def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_modulated_init(n, freq, amp, seed=0):

        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement around best_alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, it_val={M:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 4.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.003 + 0.003 * min(no_improve // 100, 3)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, lr0=0.005, mom0=0.85, top_k=30, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with adaptive multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum velocity
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient from top-k peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            # (a*a)[k] = sum_j a[j]*a[k-j], d/da[m] = a[k-m] + a[m-k] (valid indices)
            # For convolution: conv[k] = sum_j a[j]*a[k-j]
            # d conv[k] / d a[m] = a[k-m] (if 0<=k-m<n) + a[m-k+n-1] ... 
            # Actually: conv[k] = sum_j a[j] * a[k-j], so d/da[m] = a[k-m] if 0<=k-m<n
            # But also from the other term: if we write sum_j a[k-j]*a[j], by symmetry
            # d conv[k]/d a[m] = a[k-m] (when 0 <= k-m <= n-1)
            # Wait, let me be more careful.
            # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
            # d/d a[m]: a[m] appears as a[j] when j=m (contributing a[k-m]) 
            #            and as a[k-j] when k-j=m i.e. j=k-m (contributing a[k-m])
            # So d conv[k]/d a[m] = a[k-m] (if 0<=k-m<n) + a[k-m] (if 0<=k-m<n)
            # Wait no: when j=m, the term is a[m]*a[k-m], derivative wrt a[m] is a[k-m].
            # When k-j=m, i.e., j=k-m, the term is a[k-m]*a[m], derivative wrt a[m] is a[k-m].
            # But these are the same term! We're double counting.
            # Actually conv[k] = sum_j a[j]*a[k-j]. If m = j, we get a[m]*a[k-m].
            # If m = k-j, i.e., j = k-m, we get a[k-m]*a[m]. Same term!
            # So d conv[k]/d a[m] = a[k-m] if 0 <= k-m < n, else 0.
            # Wait, that's only if m != k-m. If m = k-m (i.e., 2m=k), then the term is a[m]^2,
            # and the derivative is 2*a[m].
            # Let me just compute it directly:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # conv[k] = sum_{j=js}^{je} a[j] * a[k-j]
            # d/d a[m] for m in [js, je]: contributes a[k-m]
            # d/d a[m] for m in [k-je, k-js]: contributes a[k-m]
            # These overlap. The total derivative at position m is:
            #   a[k-m] if m in [js,je] AND k-m in [0,n-1]  => m in [js,je] and m in [k-n+1, k]
            #   + a[k-m] if m in [k-je, k-js] AND m in [0,n-1]
            # Simplifying: d conv[k]/d a[m] = a[k-m] * [0<=k-m<n] + a[m-k+n-1]... 
            # I'm overcomplicating this. Let me just use the fact that:
            # conv = ifft(fft(a, 2n)^2) and compute gradient numerically or via FFT.
            pass  # Skip this approach, use FFT-based gradient

        # FFT-based gradient computation
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        # conv = irfft(A^2, fft_size)[:2n-1]
        # d conv / d a is a convolution-like operation
        # Actually, let's compute gradient directly in time domain for the weighted peaks
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            # d conv[k]/d a[m] for each m
            # conv[k] = sum_j a[j]*a[k-j]
            # For a given m, it contributes when j=m (if valid) giving a[k-m],
            #                        and when k-j=m i.e. j=k-m (if valid) giving a[k-m].
            # But these are the same occurrence. Let me verify with a small example.
            # a = [a0, a1, a2], conv[k] = sum_j a[j]*a[k-j]
            # conv[0] = a0*a0. d/da0 = 2*a0. 
            # conv[1] = a0*a1 + a1*a0 = 2*a0*a1. d/da0 = 2*a1, d/da1 = 2*a0.
            # conv[2] = a0*a2 + a1*a1 + a2*a0 = 2*a0*a2 + a1^2. d/da0=2*a2, d/da1=2*a1, d/da2=2*a0.
            # conv[3] = a1*a2 + a2*a1 = 2*a1*a2. d/da1=2*a2, d/da2=2*a1.
            # conv[4] = a2*a2. d/da2 = 2*a2.
            #
            # Pattern: d conv[k]/d a[m] = 2 * a[k-m] if 0<=k-m<n AND m!=k-m
            #                                  = 2 * a[m] if m==k-m (i.e., 2m=k)
            # Actually from the examples: d/da0 conv[2] = 2*a2 = 2*a[k-m] where k=2,m=0.
            # d/da1 conv[2] = 2*a1. Here k-m=1=m, so 2*a[k-m] = 2*a1. ✓
            # d/da0 conv[1] = 2*a1 = 2*a[k-m]. ✓
            # So the formula is: d conv[k]/d a[m] = 2*a[k-m] if 0<=k-m<n, else 0.
            # But wait: conv[0] = a0*a0, d/da0 = 2*a0. k-m=0, 2*a[0]. ✓
            #
            # Hmm but that gives factor of 2 always. Let me recheck.
            # conv[k] = sum_j a[j]*a[k-j]. 
            # If I differentiate wrt a[m]:
            #   Term where j=m: d/d a[m] (a[m]*a[k-m]) = a[k-m]
            #   Term where k-j=m (j=k-m): d/d a[m] (a[k-m]*a[m]) = a[k-m]
            #   But if m = k-m (2m=k), these are the same term a[m]*a[m], and d/d a[m] = 2*a[m].
            #   If m != k-m, these are two different terms in the sum (j=m and j=k-m), each giving a[k-m].
            #   So total: 2*a[k-m] when m != k-m, and 2*a[m] when m = k-m.
            #   In both cases: 2*a[k-m] (since when m=k-m, a[k-m]=a[m]).
            #
            # BUT WAIT: the sum is over specific j values. j ranges over [0,n-1].
            # The term a[j]*a[k-j] is present only when 0<=j<n AND 0<=k-j<n.
            # So for j=m to contribute: need 0<=m<n (always true) and 0<=k-m<n.
            # For j=k-m to contribute: need 0<=k-m<n and 0<=m<n (always true).
            # So both conditions reduce to 0<=k-m<n.
            # When 0<=k-m<n: derivative = 2*a[k-m] (from both terms, or 2*a[m] if m=k-m).
            # When not: derivative = 0.
            #
            # So: d conv[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
            # This is just 2 * (a reversed and shifted)!
            
            # k-m in [0, n-1] means m in [k-(n-1), k] intersected with [0, n-1]
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]

        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        y_new = project_simplex(y + v)
        # Update y (Nesterov lookahead)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * cur_mom * cur_mom * it)) / 2.0
        # Simplified: use standard momentum
        f_new = y_new
        y = f_new + cur_mom * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate and temperature
        cur_lr *= 0.9995
        cur_temp *= 1.004
        cur_mom = min(0.97, cur_mom * 1.0003)

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.6, 0.001)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def golden_section_cd(f0, time_limit, n_multipliers=5):
    """Coordinate descent with golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_pass_improve = 0

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search on x = f[j], with others scaled to maintain sum=1
            # Objective: max_k conv(f)[k] where f[j]=x, f[others] = f[others] * (1-x)/denom
            lo = 0.0
            hi = min(0.999, 2.0 * old_val)
            # First evaluate at current point
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            # Also try the current value and a few multipliers
            cur_val = fast_convolve(f).max()
            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent (quick) ===
    print("=== Phase 1: LP-FW on parent ===")
    t1 = min(120, deadline - time.monotonic() - 800)
    if t1 > 20:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Cosine-modulated inits at various lengths ===
    print("\n=== Phase 2: Cosine-modulated initializations ===")

        (1024, 1, 0.3), (1024, 2, 0.2), (1024, 3, 0.15),

    for n_init, freq, amp in init_configs:

    print(f"\n=== Phase 4: Nesterov subgradient (best={best_score:.6f}) ===")

            elif p == 3:

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine Nesterov
        rem = deadline - time.monotonic() - 50
        if rem > 15:

                                         lr0=0.002, mom0=0.92,
                                         top_k=max(10, n_best // 4),
                                         temp0=0.00002)
            try_seq(a_opt, "final_nesterov")

        # Short LP-FW
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations with quick optimization
        rem = deadline - time.monotonic() - 10
        for _ in range(4):

    # Fallback
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
