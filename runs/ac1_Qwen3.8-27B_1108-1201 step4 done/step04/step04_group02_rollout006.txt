import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fw_lp_step(f, M, time_per_lp=5):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g normalized to sum 1, or None."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f_j * g_{k-j} <= M
    # For each k from 0 to 2n-2, the constraint involves g indices where k-j is in [0,n-1]
    # i.e., j in [max(0, k-n+1), min(n-1, k)]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g_j appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f_{k-j}
        k_start = j
        k_end = j + n  # exclusive
        A_ub[k_start:k_end, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_per_lp})
        if res.success:
            g = res.x
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def logsumexp_obj(f, alpha):
    """Compute log-sum-exp of alpha * conv(f,f) as smooth max approximation."""
    conv = fast_convolve(f)
    scaled = alpha * conv
    max_s = scaled.max()
    return max_s + np.log(np.sum(np.exp(scaled - max_s)))

def logsumexp_grad(f, alpha):
    """Gradient of log-sum-exp(alpha * conv(f,f)) w.r.t. f."""
    conv = fast_convolve(f)
    scaled = alpha * conv
    max_s = scaled.max()
    w = np.exp(scaled - max_s)
    w /= w.sum()
    # Gradient of (f*f)[k] w.r.t. f_j = 2 * f_{k-j} (shifted)
    n = len(f)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        # (f*f)[k] = sum_j f_j * f_{k-j}
        # d/d f_j = 2 * f_{k-j} (if 0 <= k-j < n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je+1] += 2.0 * alpha * w[k] * f[k - np.arange(js, je+1)]
    return grad

def smooth_descent(f0, time_limit, alpha=50.0, lr=0.005, mom=0.9):
    """Optimize using log-sum-exp smooth approximation."""
    f = project_simplex(f0.copy())
    n = len(f)
    v = np.zeros(n)
    best_f = f.copy()
    best_val = logsumexp_obj(f, alpha)
    t0 = time.monotonic()
    no_improve = 0
    cur_alpha = alpha
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        grad = logsumexp_grad(f, cur_alpha)
        v = mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        val = logsumexp_obj(f_new, cur_alpha)
        
        if val < best_val - 1e-10:
            best_val = val
            best_f = f_new.copy()
            no_improve = 0

from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_step(f, M, time_limit=10):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            gs = res.x.sum()
            if gs > 1e-15:
                return res.x / gs
    except Exception:
        pass
    return None

def smax_grad(f, alpha):
    """Gradient of log-sum-exp(alpha * conv(f,f))."""
    conv = fast_convolve(f)
    scaled = alpha * conv
    mx = scaled.max()
    w = np.exp(scaled - mx)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(w)
        w[int(np.argmax(conv))] = 1.0
        ws = 1.0

    n = len(f)

    W = np.fft.rfft(w, L)
    F = np.fft.rfft(f, L)
    corr = np.fft.irfft(W * np.conj(F), L)[:n]
    return 2.0 * alpha * corr

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Frank-Wolfe with LP oracle, smooth log-sum-exp descent, and coordinate descent.
Explores multiple sequence lengths with diverse initializations.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_max(a):
    """Compute max of auto-convolution."""
    return fast_convolve(a).max()


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, lp_time=8):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g normalized to sum 1, or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': lp_time})
        if res.success:
            gs = res.x.sum()
            if gs > 1e-15:
                return res.x / gs
    except Exception:
        pass
    return None


def smooth_grad(f, alpha):
    """Gradient of log-sum-exp(alpha * conv(f,f)) w.r.t. f."""
    conv = fast_convolve(f)
    scaled = alpha * conv
    mx = scaled.max()
    w = np.exp(scaled - mx)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(w)
        w[int(np.argmax(conv))] = 1.0
        ws = 1.0
    w /= ws
    n = len(f)
    L = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, L)
    F = np.fft.rfft(f, L)
    corr = np.fft.irfft(W * np.conj(F), L)[:n]
    return 2.0 * alpha * corr


def smooth_descent(f0, tlim, alpha0=100.0, lr0=0.003, mom=0.92):
    """Optimize using smooth log-sum-exp objective with momentum."""
    f = project_simplex(f0.copy())
    n = len(f)
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = conv_max(f)
    t0 = time.monotonic()
    no_imp = 0
    cur_a = alpha0
    cur_lr = lr0

    while time.monotonic() - t0 < tlim:
        if time.monotonic() - t0 > tlim - 0.5:
            break
        g = smooth_grad(f, cur_a)
        v = mom * v - cur_lr * g
        f_new = project_simplex(f + v)
        obj = conv_max(f_new)
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        f = f_new
        cur_a *= 1.004
        cur_lr *= 0.9996
        if no_imp > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = lr0 * 0.7
            no_imp = 0

    return best_f


def fw_optimize(f0, tlim, max_lp_time=8):
    """Frank-Wolfe with LP oracle and careful line search."""
    f = project_simplex(f0.copy())
    n = len(f)
    M = conv_max(f)
    best_f = f.copy()
    best_M = M
    t0 = time.monotonic()
    no_imp = 0
    it = 0
    m_scale = 1.0

    while time.monotonic() - t0 < tlim:
        it += 1
        remain = tlim - (time.monotonic() - t0)
        if remain < max_lp_time + 3:
            break

        g = solve_lp_direction(f, M * m_scale, lp_time=min(max_lp_time, remain - 5))
        if g is None:
            m_scale *= 1.15
            if m_scale > 5.0:
                m_scale = 1.0
                no_imp += 1
            if no_imp > 20:
                break
            continue

        if np.max(np.abs(g - f)) < 1e-12:
            no_imp += 1
            m_scale = min(m_scale * 1.02, 2.0)
            if no_imp > 40:
                break
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = M
        alphas_coarse = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        for a in alphas_coarse:
            ft = project_simplex((1.0 - a) * f + a * g)
            c = conv_max(ft)
            if c < best_val:
                best_val = c
                best_alpha = a

        # Fine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                if hi - lo < 1e-8:
                    break
                mid = (lo + hi) / 2.0
                ft = project_simplex((1.0 - mid) * f + mid * g)
                c = conv_max(ft)
                if c < best_val:
                    best_val = c
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            M = conv_max(f)
            m_scale = max(1.0, m_scale * 0.9)
            if M < best_M - 1e-13:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            m_scale = min(m_scale * 1.05, 3.0)

        if no_imp > 60:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = conv_max(f)
            m_scale = 1.0
            no_imp = 0

    return best_f, best_M


def coord_descent(f0, tlim, n_sweeps=None):
    """Coordinate descent: optimize one coordinate at a time."""
    f = project_simplex(f0.copy().astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_obj = conv_max(f)
    t0 = time.monotonic()
    no_imp = 0

    while time.monotonic() - t0 < tlim:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > tlim - 2:
                break
            # Optimize f[j] while keeping others fixed
            # (f*f)[k] is quadratic in f[j] for each k
            # We want to minimize max_k of these quadratics
            # For a given j, (f*f)[k] = f[j]*f[k-j] + f[k-j]*f[j] + rest
            #   = 2*f[j]*f[k-j] + rest_k (for k where both j and k-j are valid)
            # At k=2j: (f*f)[2j] = f[j]^2 + rest_{2j}

            fj = f[j]
            # Find the range of k affected by f[j]
            # k ranges from j to j+n-1 (where f[k-j] is in [0, n-1])
            # (f*f)[k] = f[j]*f[k-j] + sum_{i!=j} f[i]*f[k-i]
            # For k in [j, j+n-1]: coefficient of f[j] is f[k-j]
            # Also f[j] appears as f[k-j] when k-j=j, i.e., k=2j: contributes f[j]^2

            # Let's compute the current max and try to reduce it
            # Simple approach: try a few candidate values for f[j]
            # The constraint is f[j] >= 0 and sum(f) = 1, so f[j] <= 1 - sum(others)
            max_fj = max(0, 1.0 - (1.0 - fj))  # = fj, so we can increase up to 1
            # Actually with projection, we just need f[j] >= 0 and sum <= 1 (approximately)
            # Simpler: try geometric search around current value
            candidates = [max(0, fj * 0.5), max(0, fj * 0.75), fj, fj * 1.25 + 1e-8, fj * 1.5 + 2e-8]
            best_cj = None
            best_cv = conv_max(f)

            for cj in candidates:
                if cj < 0:
                    continue
                f[j] = cj
                s = f.sum()
                if s < 1e-15:
                    f[j] = fj
                    continue
                # Don't normalize here, just check if this reduces max conv
                # (normalization will be done by project_simplex)
                c = fast_convolve(f).max() / (s * s)  # normalized objective contribution
                if best_cv is None or c < best_cv:
                    best_cv = c
                    best_cj = cj

            if best_cj is not None and best_cj != fj:
                f[j] = best_cj
                f = project_simplex(f)
                obj = conv_max(f)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_imp += 1
            if no_imp > 3:
                break
        else:
            no_imp = 0

    return best_f


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    r = np.arange(1, n + 1, dtype=np.float64)
    inits.append(r / r.sum())

    ir = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ir / ir.sum())

    c = np.cos(np.pi * (x / max(1, n - 1) - 0.5))**4 + 0.01
    inits.append(c / c.sum())

    e = np.exp(-2.0 * x / n)
    inits.append(e / e.sum())

    er = np.exp(-2.0 * x[::-1] / n)
    inits.append(er / er.sum())

    pw = 1.0 / (x + 1.0)**0.5
    inits.append(pw / pw.sum())

    inits.append(project_simplex(rng.standard_normal(n)))

    inits.append(project_simplex(rng.standard_normal(n) * 0.5 + np.sin(2 * np.pi * x / n * rng.integers(1, 5))))

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        s = px.sum()
        if s > 1e-10:
            px /= s
            inits.append(px)
            inits.append(project_simplex(0.7 * px + 0.3 * np.ones(n) / n))
            inits.append(project_simplex(0.5 * px + 0.5 * ir / ir.sum()))

    return inits
