"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based Frank-Wolfe as primary method, with peak-shaving subgradient
polishing and coordinate descent for final refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Solve LP with current M (or slightly relaxed)
        g = solve_lp(f, M)
        if g is None:
            # Try relaxed M
            g = solve_lp(f, M * 1.05)
            if g is None:
                g = solve_lp(f, M * 1.1)
                if g is None:
                    no_improve += 1
                    if no_improve > 20:
                        # Restart
                        f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
                        conv = fast_convolve(f)
                        M = conv.max()
                        no_improve = 0
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check if g is a fixed point
        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 30:
                # Try perturbation
                scale = 0.002 * (1 + no_improve / 30)

def peak_shaving_sg(f0, time_limit, initial_temp=0.001, final_temp=0.00001, lr=0.005):
    """Subgradient method with softmax-weighted peaks and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    rng_local = np.random.default_rng(99999)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        progress = (time.monotonic() - t0) / total_time
        
        # Adaptive temperature
        temp = initial_temp * (final_temp / initial_temp) ** progress
        # Learning rate decay
        cur_lr = lr * (1.0 - 0.5 * progress)

        # Find top peaks
        num_peaks = max(5, min(30, n // 10))
        if len(conv) > num_peaks:
            peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
        else:
            peaks = np.arange(len(conv))

        # Softmax weights
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws > 1e-15:
            w = w / ws
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

        # Compute gradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * f[k - idx]

        # Projected gradient step
        f_new = project_simplex(f - cur_lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            # Try half step
            f_half = project_simplex(f - 0.5 * cur_lr * grad)
            conv_half = fast_convolve(f_half)
            if conv_half.max() < best_obj - 1e-14:
                best_obj = conv_half.max()
                best_f = f_half.copy()
                f = f_half
                no_improve = 0

        # Restart
        if no_improve > 200:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            no_improve = 0
            cur_lr = max(lr * 0.5, 0.001)

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng_local.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            if stall > 100:
                break
            
            # Try moving mass from j to random targets
            targets = [rng_local.integers(n), (j + n // 3) % n, (j + 2 * n // 3) % n]
            for source in targets:
                source = int(source) % n
                if source == j:
                    continue
                if f[source] < 1e-10:
                    continue
                
                for frac in [0.05, 0.15, 0.3]:
                    delta = f[source] * frac
                    if delta < 1e-12:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 50:
                # Try a small random perturbation
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                stall = 0
            elif stall > 200:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(sine / sine.sum())

    # Cosine
    cos_t = np.cos(np.pi * x / n) ** 2
    cos_t = np.maximum(cos_t, 0.001)
    inits.append(cos_t / cos_t.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based Frank-Wolfe, spectral flattening (error reduction),
peak-shaving subgradient, and coordinate descent for final refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 8, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(23456 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Solve LP
        g = solve_lp(f, M)
        if g is None:
            g = solve_lp(f, M * 1.03)
        if g is None:
            g = solve_lp(f, M * 1.08)
        if g is None:
            no_improve += 2
            if no_improve > 15:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 25:
                # Spectral restart
                F = np.fft.rfft(f, 2 * n - 1)
                P = np.abs(F) ** 2
                P_flat = P ** 0.75
                P_flat /= P_flat.sum()
                F_new = np.sqrt(P_flat * P.sum()) * np.exp(1j * np.angle(F))
                f_new = np.fft.irfft(F_new, 2 * n - 1)[:n]
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn > 1e-10:
                    f = f_new / sn
                else:
                    f = project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0.05 < best_alpha < 0.95:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Restart strategies
        if no_improve > 35:
            rst = no_improve % 4
            if rst == 0:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            elif rst == 1:
                f = project_simplex(0.85 * best_f + 0.15 * best_f[::-1])
            elif rst == 2:
                # Spectral flatten
                F = np.fft.rfft(best_f, 2 * n - 1)
                P = np.abs(F) ** 2
                P_flat = P ** 0.6
                P_flat /= P_flat.sum()
                F_new = np.sqrt(P_flat * P.sum()) * np.exp(1j * np.angle(F))
                f_new = np.fft.irfft(F_new, 2 * n - 1)[:n]
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                f = f_new / sn if sn > 1e-10 else project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
            else:
                f = project_simplex(rng_local.dirichlet(np.ones(n) * 0.6))
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def spectral_flatten(f0, time_limit, alpha_start=0.85, alpha_end=0.5, verbose=False):
    """Spectral flattening via error reduction / Gerchberg-Saxton style."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(44444 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / time_limit
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, progress * 1.5)

        # FFT
        F = np.fft.rfft(f, 2 * n - 1)
        P = np.abs(F) ** 2

        # Flatten spectrum
        P_flat = np.power(P + 1e-15, alpha)
        P_flat = P_flat / P_flat.sum() * P.sum()

        # Reconstruct
        F_new = np.sqrt(P_flat) * np.exp(1j * np.angle(F))
        f_new = np.fft.irfft(F_new, 2 * n - 1)[:n]

        # Project to non-negative
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            no_improve += 5
            if no_improve > 100:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        f_new = f_new / sn

        # Line search between f and f_new
        found_improvement = False
        for step in [1.0, 0.7, 0.5, 0.3, 0.2]:
            f_trial = (1.0 - step) * f + step * f_new
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_trial.copy()
                f = f_trial
                no_improve = 0
                found_improvement = True
                if verbose and it % 50 == 0:
                    print(f"    Spectral iter {it}: obj={best_obj:.10f}")
                break

        if not found_improvement:
            no_improve += 1

        if no_improve > 300:
            break

    return best_f, best_obj


def peak_shaving_sg(f0, time_limit, temp_start=0.003, temp_end=0.000005, lr_start=0.006):
    """Subgradient with softmax-weighted peaks and adaptive scheduling."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng_local = np.random.default_rng(88888 + n)

    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / time_limit
        progress = min(1.0, progress)
        temp = temp_start * (temp_end / temp_start) ** progress
        cur_lr = lr_start * (1.0 - 0.6 * progress)
        if cur_lr < 0.0005:
            cur_lr = 0.0005

        # Top peaks with softmax weights
        num_peaks = max(8, min(40, n // 8))
        if len(conv) > num_peaks:
            peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws > 1e-15:
            w = w / ws
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * f[k - idx]

        # Projected step
        f_new = project_simplex(f - cur_lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            # Half step
            f_half = project_simplex(f - 0.5 * cur_lr * grad)
            obj_half = fast_convolve(f_half).max()
            if obj_half < best_obj - 1e-14:
                best_obj = obj_half
                best_f = f_half.copy()
                f = f_half
                no_improve = 0

        if no_improve > 250 and time.monotonic() - t0 < time_limit * 0.8:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent: move mass between coordinate pairs to reduce peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    peak_idx = int(conv.argmax())

    t0 = time.monotonic()
    rng_local = np.random.default_rng(33333 + n)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng_local.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            if stall > 80:
                break

            # Generate diverse targets
            targets = [
                int(rng_local.integers(n)),
                int((j + n // 3) % n),
                int((j + 2 * n // 3) % n),
                int((j + rng_local.integers(1, n)) % n),
            ]

            for source in targets:
                source = source % n
                if source == j or f[source] < 1e-10:
                    continue

                for frac in [0.1, 0.2, 0.4]:
                    delta = f[source] * frac
                    if delta < 1e-13:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        peak_idx = int(conv.argmax())
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 60:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                stall = 0
            elif stall > 300:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 1.5))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    peak2 = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
             np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak2 / peak2.sum())

    sine = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(sine / sine.sum())

    cos_t = np.maximum(np.cos(np.pi * x / n) ** 2, 0.001)
    inits.append(cos_t / cos_t.sum())

    # Anti-symmetric pair
    half = n // 2
    asym = np.zeros(n)
    asym[:half] = 1.0 / half
    inits.append(asym)

    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

            a_opt, _ = spectral_flatten(state['best_seq'], time_limit=t_spec,
                                        alpha_start=0.7, alpha_end=0.4)
            try_sequence(a_opt, "spec_sg", state)
            # Follow with quick SG
            rem2 = p3_end - time.monotonic() - 5
            if rem2 > 10:
                a_opt, _ = peak_shaving_sg(state['best_seq'], time_limit=min(rem2, 40),
                                           temp_start=0.001, temp_end=0.00001,
                                           lr_start=0.003)
                try_sequence(a_opt, "sg_after_spec", state)

    # === Phase 4: Final Refinement (17%) ===
    print(f"\n=== Phase 4: Final Refinement ===")
    if state['best_seq'] is not None:
        remaining = deadline - time.monotonic()
        n_best = len(state['best_seq'])

        # Coordinate descent
        t_cd = min(remaining * 0.4, 100)
        if t_cd > 10:
            print(f"  CD polish, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(state['best_seq'], time_limit=t_cd)
            try_sequence(a_opt, "cd_main", state)

        # Quick FW squeeze
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_fw = min(rem * 0.5, 50)
            a_opt, _ = fw_lp_optimize(state['best_seq'], time_limit=t_fw)
            try_sequence(a_opt, "fw_final", state)

        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent_polish(state['best_seq'], time_limit=rem)
            try_sequence(a_opt, "cd_final", state)

        # One more spectral pass
        rem = deadline - time.monotonic() - 3
        if rem > 8:
            a_opt, _ = spectral_flatten(state['best_seq'], time_limit=rem,
                                        alpha_start=0.6, alpha_end=0.4)
            try_sequence(a_opt, "spec_final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
