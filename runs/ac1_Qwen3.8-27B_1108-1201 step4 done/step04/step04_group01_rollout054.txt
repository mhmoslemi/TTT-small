"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_smoothed_grad(f, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*(f*f))) w.r.t. f.
    Returns gradient of shape (n,) and the smoothed objective value.
    """
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    obj = (bc_max + np.log(exp_bc.sum())) / beta
    
    # Gradient: 2 * sum_k w[k] * f[k-j] for each j
    # This is 2 * cross_correlation(w, f)
    # w has length 2n-1, f has length n
    # grad[j] = 2 * sum_k w[k] * f[k-j] where 0 <= k-j < n
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_wf = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * conv_wf[n - 1:n - 1 + n]
    
    return grad, obj, conv, w

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def mirror_descent(f0, time_limit, beta_start=50.0, beta_end=5000.0):

    f = np.maximum(f0.astype(np.float64), 0.0)

    else:

    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    eta = 1.0 / np.sqrt(n)
    
    while time.monotonic() - t0 < time_limit:
        grad, smooth_obj, conv, w = lse_smoothed_grad(f, beta)
        
        # Mirror descent update: f_new[j] = f[j] * exp(-eta * grad[j]) / Z

        # Check objective

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. FW-LP for global structure optimization with high-precision LP solving
2. LSE-smoothed mirror descent and projected gradient descent for polishing
3. Active-set guided refinement using LP dual information
4. Multi-start perturbation exploration to escape local minima
5. Systematic length sweep around the parent length
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(f, w):
    """Compute gradient of LSE-smoothed objective.
    w is the softmax weights over the 2n-1 autocorrelation values.
    grad[j] = 2 * sum_k w[k] * f[k-j]
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_wf = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_wf[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP direction problem.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0. Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=50):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns (g, dual) or (None, None).
    """
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            dual = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, dual
    except Exception:
        pass
    return None, None


def fw_lp_optimize(f0, time_limit, lp_timeout=50):
    """Frank-Wolfe with LP oracle, careful line search, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g, dual = solve_lp_direction(f, M * m_relax, lp_timeout)
        if g is None:
            m_relax *= 1.05
            if m_relax > 4.0:
                m_relax = 1.0
                no_improve += 3
                if no_improve > 60:
                    break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        m_relax = 1.0

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Golden section line search
        best_alpha = 0.0
        best_val = M + 1e-15

        # Coarse scan
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(1.0, (no_improve - 30) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def lse_mirror_descent(f0, time_limit, beta_start=50.0, beta_end=10000.0):
    """Mirror descent (multiplicative weights) with LSE-smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    eta = 0.8 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_gradient(f, w)

        # Mirror descent: f_new[j] = f[j] * exp(-eta * grad[j]) / Z
        log_mult = -eta * grad
        log_mult -= log_mult.max()
        mult = np.exp(log_mult)
        f_new = f * mult
        Z = f_new.sum()
        if Z < 1e-15:
            break
        f_new = f_new / Z

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-16:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            eta = min(eta * 1.0015, 1.5 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 5:
                eta *= 0.94
            if no_improve > 80:
                f = best_f.copy()
                eta = 0.8 / np.sqrt(n)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta = min(beta * 1.003, beta_end)

    return best_f, best_obj


def lse_adam_descent(f0, time_limit, beta_start=30.0, beta_end=10000.0):
    """LSE-smoothed gradient descent with Adam optimizer and simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.5 / n
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_gradient(f, w)

        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        f_raw = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_raw)

        # Ensure minimum mass to avoid degeneracy
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-16:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 2.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.92
            if no_improve > 120:
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.5 / n
                beta = min(beta * 1.5, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta = min(beta * 1.004, beta_end)

    return best_f, best_obj


def topk_equalize(f0, time_limit, k=8):
    """Subgradient method focusing on top-k peaks of the autocorrelation.
    Tries to equalize them while reducing the maximum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    temp = 0.0001
    momentum = 0.92
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        max_val = conv.max()

        # Get top-k peaks
        if k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -k)[-k:]

        # Softmax weights over the top-k
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Subgradient: sum of weighted gradients at the selected peaks
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            pk = int(peaks[i])
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * f[pk - idx]

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-16:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.93
            if no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.005
                temp *= 5.0
                no_improve = 0

        temp = min(temp * 1.003, 0.01)

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and track best sequence."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main optimization routine. Multi-phase:
    1. FW-LP from parent (global structure)
    2. LSE Adam GD (polishing)
    3. Mirror descent (interior polishing)
    4. Top-k equalization (peak balancing)
    5. FW-LP refinement (precision)
    6. Multi-start perturbation (escape local minima)
    7. Length sweep (find optimal n)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(p / p.sum(), best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if best_tracker[1] is None:
        f_init = np.ones(448) / 448
        try_sequence(f_init, best_tracker)
        parent = f_init.copy()

    f = best_tracker[1].copy() / best_tracker[1].sum()
    n = len(f)

    # === Phase 1: FW-LP ===
    print("\n=== Phase 1: FW-LP (global) ===")
    t_rem = deadline - time.monotonic() - 700
    t_alloc = max(40, min(300, t_rem))
    print(f"  Allocating {t_alloc:.0f}s")
    a_opt, obj = fw_lp_optimize(f, time_limit=t_alloc, lp_timeout=50)
    try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE Adam GD ===
    print("\n=== Phase 2: LSE Adam GD ===")
    f = best_tracker[1].copy() / best_tracker[1].sum()
    t_rem = deadline - time.monotonic() - 550
    t_alloc = max(30, min(350, t_rem))
    print(f"  Allocating {t_alloc:.0f}s")
    a_opt, obj = lse_adam_descent(f, time_limit=t_alloc, beta_start=25.0, beta_end=8000.0)
    try_sequence(a_opt, best_tracker)

    # === Phase 3: Mirror Descent ===
    print("\n=== Phase 3: Mirror Descent ===")
    f = best_tracker[1].copy() / best_tracker[1].sum()
    t_rem = deadline - time.monotonic() - 350
    t_alloc = max(20, min(250, t_rem))
    print(f"  Allocating {t_alloc:.0f}s")
    a_opt, obj = lse_mirror_descent(f, time_limit=t_alloc, beta_start=80.0, beta_end=12000.0)
    try_sequence(a_opt, best_tracker)

    # === Phase 4: Top-k Equalization ===
    print("\n=== Phase 4: Top-k Equalization ===")
    f = best_tracker[1].copy() / best_tracker[1].sum()
    t_rem = deadline - time.monotonic() - 200
    t_alloc = max(20, min(200, t_rem))
    print(f"  Allocating {t_alloc:.0f}s")
    a_opt, obj = topk_equalize(f, time_limit=t_alloc, k=max(6, n // 50))
    try_sequence(a_opt, best_tracker)

    # === Phase 5: FW-LP Final Refinement ===
    print("\n=== Phase 5: FW-LP Refinement ===")
    f = best_tracker[1].copy() / best_tracker[1].sum()
    t_rem = deadline - time.monotonic() - 150
    t_alloc = max(15, min(150, t_rem))
    print(f"  Allocating {t_alloc:.0f}s")
    a_opt, obj = fw_lp_optimize(f, time_limit=t_alloc, lp_timeout=40)
    try_sequence(a_opt, best_tracker)

    # === Phase 6: Multi-start perturbation ===
    print("\n=== Phase 6: Multi-start ===")
    f_best = best_tracker[1].copy() / best_tracker[1].sum()
    n_best = len(f_best)

    t_rem = deadline - time.monotonic() - 100
    n_starts = min(6, max(1, int(t_rem / 55)))
    print(f"  {n_starts} restarts, remaining {t_rem:.0f}s")

    for i in range(n_starts):
        t_rem = deadline - time.monotonic() - 50
        if t_rem < 40:
            break

        rng_rs = np.random.default_rng(seed + 7000 + i * 1337)
        # Different perturbation strategies
        if i % 3 == 0:
            # Small Gaussian
            f_pert = project_simplex(f_best + rng_rs.standard_normal(n_best) * 0.001)
        elif i % 3 == 1:
            # Medium Gaussian
            f_pert = project_simplex(f_best + rng_rs.standard_normal(n_best) * 0.005)
        else:
            # Sparse perturbation (change only a few coordinates)
            f_pert = f_best.copy()
            n_changes = rng_rs.integers(5, 20)
            idx = rng_rs.choice(n_best, size=n_changes, replace=False)
            f_pert[idx] += rng_rs.standard_normal(n_changes) * 0.01
            f_pert = project_simplex(f_pert)

        t_sub = (t_rem - 50) / (n_starts - i)
        t1 = min(25, t_sub * 0.5)
        t2 = min(25, t_sub * 0.3)
        t3 = min(15, t_sub * 0.2)

        if t1 < 8:
            a_opt, _ = lse_adam_descent(f_pert, time_limit=t_sub, beta_start=100.0, beta_end=5000.0)
        else:
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t1, lp_timeout=15)
            try_sequence(a_opt, best_tracker)

            f_mid = best_tracker[1].copy() / best_tracker[1].sum() if best_tracker[1] is not None else a_opt
            if len(f_mid) == n_best:
                a_opt2, _ = lse_adam_descent(f_mid, time_limit=t2, beta_start=200.0, beta_end=8000.0)
                try_sequence(a_opt2, best_tracker)

            f_end = best_tracker[1].copy() / best_tracker[1].sum() if best_tracker[1] is not None else a_opt2
            if len(f_end) == n_best and t3 > 5:
                a_opt3, _ = fw_lp_optimize(f_end, time_limit=t3, lp_timeout=10)
                try_sequence(a_opt3, best_tracker)

    # === Phase 7: Length sweep ===
    print("\n=== Phase 7: Length Sweep ===")
    f_best = best_tracker[1].copy() if best_tracker[1] is not None else f
    n_best = len(f_best)

    # Try lengths around the best
    candidates = []
    for delta in [0, 1, -1, 2, -2, 4, -4, 8, -8, 16, -16, 32, -32]:
        nc = n_best + delta
        if 128 <= nc <= 1024 and nc != n_best:
            candidates.append(nc)
    # Add some fixed good lengths
    for nc in [384, 448, 512, 256, 640]:
        if nc not in candidates and nc != n_best:
            candidates.append(nc)

    candidates = candidates[:8]  # Limit to 8
    print(f"  Testing lengths: {candidates}")

    for n_alt in candidates:
        t_rem = deadline - time.monotonic() - 40
        if t_rem < 50:
            break

        t_alloc = min(60, t_rem - 40)
        if t_alloc < 20:
            break

        # Interpolate
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, f_best)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        t1 = t_alloc * 0.5
        t2 = t_alloc * 0.4
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t1, lp_timeout=15)
        try_sequence(a_opt, best_tracker)

        f_mid = best_tracker[1].copy()
        if f_mid is not None and len(f_mid) == n_alt:
            f_mid = f_mid / f_mid.sum()
        else:
            f_mid = a_opt / a_opt.sum()
        a_opt2, _ = lse_adam_descent(f_mid, time_limit=t2, beta_start=50.0, beta_end=5000.0)
        try_sequence(a_opt2, best_tracker)

    # === Final: One more polish on the overall best ===
    print("\n=== Final Polish ===")
    f_final = best_tracker[1].copy() / best_tracker[1].sum()
    t_rem = deadline - time.monotonic() - 10
    if t_rem > 20:
        print(f"  {t_rem:.0f}s remaining")
        a_opt, _ = lse_adam_descent(f_final, time_limit=min(t_rem * 0.6, 60),
                                     beta_start=500.0, beta_end=20000.0)
        try_sequence(a_opt, best_tracker)
        f_final = best_tracker[1].copy() / best_tracker[1].sum()
        t_rem = deadline - time.monotonic() - 5
        if t_rem > 10:
            a_opt, _ = fw_lp_optimize(f_final, time_limit=min(t_rem * 0.8, 30), lp_timeout=10)
            try_sequence(a_opt, best_tracker)

    # Return best
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]
    else:
        return [1.0 / 448] * 448
