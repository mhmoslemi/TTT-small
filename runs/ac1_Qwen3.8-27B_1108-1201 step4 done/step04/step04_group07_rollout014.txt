from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, top_k=50):
    """Compute weighted subgradient of max(conv(f,f)) at f."""
    n = len(f)
    conv = fast_convolve(f)
    k = len(conv)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    # Softmax weighting
    scale = (pv.max() - pv.min()) * 10 if pv.max() > pv.min() else 1.0
    w = np.exp(10 * (pv - pv.max()) / (pv.max() - pv.min() + 1e-15))
    w /= w.sum()
    
    sg = np.zeros(n)
    for i, p in enumerate(peaks):
        if w[i] < 1e-10:
            continue
        k_idx = int(p)
        # ∂conv[k_idx]/∂f[j] = 2*f[k_idx - j] if 0 <= k_idx - j < n
        js = np.arange(n)
        diffs = k_idx - js
        valid = (diffs >= 0) & (diffs < n)
        sg[js[valid]] += 2.0 * w[i] * f[diffs[valid]]
    
    return sg, conv

def nesterov_optimize(f0, time_limit, top_k=30, lr_init=0.01, mom_init=0.9):
    """Nesterov-accelerated gradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    conv_y = fast_convolve(y)
    best_obj = conv_y.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    mom = mom_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov step
        y = f + mom * v
        
        # Compute subgradient at y
        sg, conv_y = compute_subgradient(y, top_k)
        
        # Project gradient to simplex tangent space
        sg_centered = sg - np.mean(sg)
        
        # Update
        v = mom * v - lr * sg_centered
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_f = fast_convolve(f_new)
        obj = conv_f.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        lr *= 0.9998
        mom = min(mom * 1.0001, 0.99)
        
        if no_improve > 200:
            # Random restart
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0
    
    return best_f, best_obj

from scipy.optimize import linprog
import time


def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            best_f = f_new.copy()
            no_improve = 0
            total_steps += 1
            if verbose and it % 500 == 0:
                print(f"    Nesterov it={it}, obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1

        f = f_new

        # Adaptive step size (Polyak-like)
        lr *= 0.99985
        if no_improve > 300:
            lr = max(lr * 0.5, 0.0003)
            no_improve = 0
        if no_improve > 800:
            # Larger restart
            restart_scale = 0.003 + 0.002 * (it % 5)
            f = project_simplex(best_f + rng.standard_normal(n) * restart_scale)
            v = np.zeros(n)
            lr = max(lr_init * 0.3, 0.001)
            mom = 0.85
            no_improve = 0
            it = 0

        mom = min(mom * 1.00005, 0.95)

    return best_f, best_obj


def mass_transfer_search(f0, time_limit, verbose=False):
    """
    Greedy mass-transfer local search.
    Uses subgradient to identify promising transfer directions:
    transfer mass from high-subgradient positions to low-subgradient positions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it % 200 == 1:
            sg, conv = compute_weighted_subgradient(best_f, top_k=30)
            sg_sorted = np.argsort(sg)

        # Batch of transfers
        batch_size = 200 if n < 500 else 100
        improved = False

        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source (high sg) and destination (low sg)
            source_candidates = sg_sorted[-20:]
            dest_candidates = sg_sorted[:20]
            src = int(rng.choice(source_candidates))
            dst = int(rng.choice(dest_candidates))

            if src == dst or best_f[src] < 1e-6:
                continue

            # Try multiple transfer amounts
            base_delta = min(best_f[src] * 0.1, 0.003)
            for mult in [1.0, 0.5, 2.0, 0.25]:
                delta = base_delta * mult
                if delta < 1e-6 or delta > best_f[src] * 0.5:
                    continue

                f_new = best_f.copy()
                f_new[src] -= delta
                f_new[dst] += delta

                conv_new = fast_convolve(f_new)
                obj_new = conv_new.max()

                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            # Perturb and retry
            scale = 0.002 + 0.001 * (no_improve % 10)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv_pert = fast_convolve(f_pert)
            if conv_pert.max() < best_obj:
                best_obj = conv_pert.max()
                best_f = f_pert.copy()
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MassTransfer it={it}, obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lp_refinement(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe refinement for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv_f = fast_convolve(f)
    best_obj = conv_f.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(31337)

    while time.monotonic() - t0 < time_limit and no_improve < 30:
        it += 1
        conv_f = fast_convolve(f)
        current_obj = conv_f.max()
        M = current_obj

        # Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})

        if res.success and res.x is not None and res.x.sum() > 1e-10:
            g = res.x / res.x.sum()
        else:
            # Try with slightly relaxed M
            res = linprog(c, A_ub=A_ub, b_ub=np.full(2 * n - 1, M * 1.01),
                          bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success and res.x is not None and res.x.sum() > 1e-10:
                g = res.x / res.x.sum()
            else:
                no_improve += 2
                continue

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            continue

        # Line search
        best_alpha = 0
        best_val = current_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"    LP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else np.array([0.0])

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Hann window
    w = 0.5 * (1 - np.cos(2 * np.pi * xn))
    w = np.maximum(w, 1e-10)
    inits.append(("hann", w / w.sum()))

    # Hamming window
    w = 0.54 - 0.46 * np.cos(2 * np.pi * xn)
    w = np.maximum(w, 1e-10)
    inits.append(("hamming", w / w.sum()))

    # Blackman window
    w = 0.42 - 0.5 * np.cos(2 * np.pi * xn) + 0.08 * np.cos(4 * np.pi * xn)
    w = np.maximum(w, 1e-10)
    inits.append(("blackman", w / w.sum()))

    # Flat-top
    w = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * xn)
         + 0.277263158 * np.cos(4 * np.pi * xn)
         - 0.083578947 * np.cos(6 * np.pi * xn)
         + 0.006947368 * np.cos(8 * np.pi * xn))
    w = np.maximum(w, 1e-10)
    inits.append(("flattop", w / w.sum()))

    # Gaussian (various widths)
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        w = np.exp(-0.5 * ((xn - 0.5) / sigma) ** 2)
        inits.append((f"gauss_{sigma}", w / w.sum()))

    # Super-Gaussian
    for p in [3, 4, 6]:
        w = np.exp(-((xn - 0.5) * 2) ** p)
        inits.append((f"super_g_{p}", w / w.sum()))

    # Raised cosine (full)
    w = np.maximum(0, np.cos(np.pi * xn))
    w = np.maximum(w, 1e-10)
    inits.append(("raised_cos", w / w.sum()))

    # Multi-lobe: sum of shifted Gaussians
    for num_lobe in [2, 3, 4]:
        w = np.zeros(n)
        for li in range(num_lobe):
            center = (li + 0.5) / num_lobe
            w += np.exp(-0.5 * ((xn - center) / (0.1 / num_lobe)) ** 2)
        inits.append((f"multi_lobe_{num_lobe}", w / w.sum()))

    # Comb pattern (periodic)
    for period in [3, 5, 7, 11]:
        w = np.ones(n)
        w[period - 1::period] = 0.3
        inits.append((f"comb_{period}", w / w.sum()))

    # Linear ramp + uniform mix
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(("ramp", ramp / ramp.sum()))

    # Quadratic
    w = 1.0 - (2 * xn - 1) ** 2
    w = np.maximum(w, 1e-10)
    inits.append(("quadratic", w / w.sum()))

    # Random Dirichlet with various concentrations
    for conc in [0.1, 0.5, 1.0, 2.0, 5.0]:
        w = rng.dirichlet(np.ones(n) * conc)
        inits.append((f"dirichlet_{conc}", w))

    # Biplane: two bins
    w = np.zeros(n)
    w[0] = 0.5
    w[n - 1] = 0.5
    inits.append(("biplane", w / w.sum()))

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(("parent_interp", px / ps))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Diverse initialization (window functions, random, parent)
    Phase 2: Nesterov-accelerated subgradient descent
    Phase 3: Greedy mass-transfer local search
    Phase 4: LP refinement
    Phase 5: Final Nesterov polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, prev={old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse Initialization ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t_start = time.monotonic()

    # Determine length to use
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 384

    candidate_lengths = [n_base]
    # Try a few nearby lengths
    for offset in [-32, -16, 16, 32, 64]:
        nl = n_base + offset
        if 64 <= nl <= 800:
            candidate_lengths.append(nl)

    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_diverse_inits(n_try, rng, parent)
        for label, init in inits:
            if time.monotonic() > deadline - 550:
                break
            # Quick Nesterov burst (5 seconds) to evaluate potential
            quick_f, _ = nesterov_optimize(init, time_limit=min(5, max(1, (deadline - time.monotonic() - 500) / 20)))
            score = try_seq(quick_f, f"p1_{label}_n{n_try}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.8f} (n={n_best})")

    # === Phase 2: Main Nesterov Optimization ===
    print("\n=== Phase 2: Nesterov Optimization ===")
    remaining = deadline - time.monotonic() - 400
    t_p2 = min(remaining * 0.4, 500)

    if t_p2 > 20:
        # Run Nesterov on best
        t_each = t_p2 / 3
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_each * 0.9,
                                     top_k=35, lr_init=0.008, mom_init=0.85, verbose=True)
        try_seq(a_opt, "nest1")

        # With different parameters
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            a_opt, _ = nesterov_optimize(best_seq, time_limit=min(rem * 0.4, 150),
                                         top_k=50, lr_init=0.005, mom_init=0.9, verbose=False)
            try_seq(a_opt, "nest2")

        # From a perturbed start
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)
            a_opt, _ = nesterov_optimize(f_pert, time_limit=min(rem * 0.4, 120),
                                         top_k=30, lr_init=0.006, mom_init=0.88)
            try_seq(a_opt, "nest3_pert")

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 3: Mass Transfer Local Search ===
    print("\n=== Phase 3: Mass Transfer Search ===")
    remaining = deadline - time.monotonic() - 300
    t_p3 = min(remaining * 0.4, 400)

    if t_p3 > 20:
        t_each = t_p3 / 2
        a_opt, _ = mass_transfer_search(best_seq, time_limit=t_each * 0.9, verbose=True)
        try_seq(a_opt, "mt1")

        rem = deadline - time.monotonic() - 250
        if rem > 20:
            a_opt, _ = mass_transfer_search(best_seq, time_limit=min(rem * 0.5, 200))
            try_seq(a_opt, "mt2")

    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 4: LP Refinement ===
    print("\n=== Phase 4: LP Refinement ===")
    remaining = deadline - time.monotonic() - 200
    t_p4 = min(remaining * 0.3, 200)

    if t_p4 > 15:
        n_cur = len(best_seq)
        # LP is slow for large n, so it's best for moderate n
        if n_cur <= 500:
            a_opt, _ = lp_refinement(best_seq, time_limit=t_p4 * 0.8, verbose=True)
            try_seq(a_opt, "lp1")

            rem = deadline - time.monotonic() - 150
            if rem > 15:
                # LP from perturbed
                f_pert = project_simplex(best_seq + rng.standard_normal(n_cur) * 0.005)
                a_opt, _ = lp_refinement(f_pert, time_limit=min(rem * 0.5, 100))
                try_seq(a_opt, "lp2_pert")
        else:
            print("  Skipping LP (n too large)")

    print(f"  Phase 4 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 5: Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 30
    t_p5 = min(remaining, 200)

    if t_p5 > 15:
        # Fine Nesterov
        t_fine = t_p5 * 0.5
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_fine,
                                     top_k=50, lr_init=0.002, mom_init=0.92)
        try_seq(a_opt, "final_nest")

        # Mass transfer with very small steps
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = mass_transfer_search(best_seq, time_limit=rem * 0.7)
            try_seq(a_opt, "final_mt")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
