"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed mirror descent
(multiplicative weights) with momentum, adaptive temperature, LP-FW polishing,
and coordinate descent. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, conv, T):
    """Compute gradient of the softmax-approximated max of convolution."""
    n = len(a)
    # Softmax weights over convolution entries
    shifted = T * conv - T * conv.max()
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: sg[m] = 2 * sum_j a[j] * weights[j+m]
    # = 2 * ifft(fft(weights, P) * conj(fft(a, P)))[:n]
    P = 1 << (3 * n - 1).bit_length()
    fa = np.fft.fft(a, P)
    fw = np.fft.fft(weights, P)
    sg = 2.0 * np.fft.ifft(fw * np.conj(fa)).real[:n]
    
    return sg, weights

def mirror_descent_optimize(a0, time_limit, T_start=5.0, T_end=500.0, 
                            lr_start=1.0, mom=0.9, verbose=False):
    """Smoothed mirror descent with momentum for minimizing max(conv(a,a)) on simplex."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    lw = np.log(a + 1e-300)  # log-weights
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)  # momentum in log-space
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        
        # Adaptive temperature
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient
        sg, weights = compute_smooth_gradient(a, conv, T)
        
        # Momentum update
        v = mom * v + sg
        # Step size: decrease over time
        eta = lr_start / (1.0 + 0.01 * it)
        
        # Update log-weights
        lw_new = lw - eta * v
        
        # Convert to probability vector (softmax)
        lw_max = lw_new.max()
        exp_lw = np.exp(lw_new - lw_max)
        a_new = exp_lw / exp_lw.sum()
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lw = lw_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, reduce step size
            eta *= 0.7
            lw_new = lw - eta * v
            lw_max = lw_new.max()
            exp_lw = np.exp(lw_new - lw_max)
            a_new = exp_lw / exp_lw.sum()
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                lw = lw_new
                conv = conv_new
                no_improve = 0
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            lw = np.log(a + 1e-300)
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"      MD iter {it}: obj={best_obj:.10f}, T={T:.1f}, eta={eta:.4f}")
    
    return best_a, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    # Build constraint matrix more efficiently
    c = -np.ones(n)
    # A_ub has shape (2n-1, n), where A_ub[k, j] = f[k-j] if 0<=k-j<n
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        cols = slice(0, n)  # rows j to j+n-1
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None

def lp_fw_step(f, M, n):
    """Single LP-FW step: find best direction and do line search."""
    g = solve_lp_direction(f, M, n)
    if g is None:
        return None
    
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum
    
    # Line search
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    
    for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha < 1e-10:
        return None
    
    f_new = (1 - best_alpha) * f + best_alpha * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    return f_new

def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = project_simplex(f0.astype(np.float64))
    n = len(f)
    best_obj = fast_convolve(f).max()

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp
