`= sum_k w[k] * 2 * x[k-i]` where `w[k] = exp(beta*C[k]) / sum_j exp(beta*C[j])`

This is a weighted sum of "shifted copies" of `x`. We can compute this efficiently!

`dS/dx[i] = 2 * sum_k w[k] * x[k-i]` (with valid index constraint)

This is like a "blurred" version of `x` where the blur kernel is `w` (reflected).

Using FFT: Let `W = w` (length 2n-1) and `X = x` (length n). Then:
`sum_k w[k] * x[k-i]` is the cross-correlation of `w` and `x` at position `i`.

Cross-correlation: `R[i] = sum_k w[k] * x[k-i]`

We can compute this via FFT:
- Treat it as convolution: `R[i] = sum_k w[k] * x[k-i] = sum_k w[k] * x_flip[n-1-(k-i)]` where `x_flip = x[::-1]`
- So `R = convolve(w, x_flip)` evaluated at appropriate indices.

Actually, let me think about it differently. Define the cross-correlation as:
`(w ⋆ x)[i] = sum_k w[k] * x[k-i]`

This equals `(w * x_flip)[n-1+i]` where `*` is convolution and `x_flip = x[::-1]`.

Let me verify: `conv(w, x_flip)[m] = sum_j w[j] * x_flip[m-j] = sum_j w[j] * x[n-1-(m-j)]`
For `m = n-1+i`: `= sum_j w[j] * x[n-1-(n-1+i-j)] = sum_j w[j] * x[j-i]`

Yes! So `(w ⋆ x)[i] = conv(w, x_flip)[n-1+i]`.

So the gradient is: `grad[i] = 2 * conv(w, x_flip)[n-1+i]` for `i = 0, ..., n-1`.

This can be computed with a single FFT convolution!

Now, for the "top-k" version: instead of using all `2n-1` values in the LSE, we only use the top `k` peaks. This makes the gradient computation cheaper (sparse `w`).

For sparse `w`, the gradient is:
`grad[i] = 2 * sum_{k in top_k} w[k] * x[k-i]` (with valid index constraint `0 <= k-i < n`)

For each peak `k` in the top-k, it contributes `2*w[k] * x[k-i]` to `grad[i]` for `i` in the range `[max(0, k-n+1), min(k, n-1)]`.

So the gradient is a weighted sum of "reflected segments" of `x`.

This is efficient when k is small!

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Top-k smooth max with Adam optimizer on the simplex
- Adaptive beta (temperature) scheduling
- Periodic FW-LP steps for global improvement
- Peak-aware coordinate descent for local refinement
- Multi-start with structured initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_topk(a, conv, top_k, beta):
    """Compute gradient of top-k LSE smooth max of conv."""
    n = len(a)
    conv_size = 2 * n - 1
    
    if top_k >= conv_size:
        idx = np.arange(conv_size)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    
    vals = conv[idx]
    scaled = beta * vals
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    
    # Gradient: grad[i] = 2 * sum_{k in idx} w[k] * a[k-i] (if 0 <= k-i < n)
    grad = np.zeros(n)
    a_flip = a[::-1]
    
    # For each peak k, it contributes w[k] * a[k-i] to grad[i]
    # for i in [max(0, k-n+1), min(k, n-1)]
    # This is equivalent to: grad = 2 * cross_correlation(w_sparse, a)
    
    # Efficient computation: for each peak k, the contribution to grad is
    # 2 * w[k] * a[max(0,k-n+1):min(k+1,n)] reversed... 
    # Actually: grad[i] += 2 * w[k] * a[k-i] for valid i
    # Let j = k-i, so i = k-j. As i ranges over [max(0,k-n+1), min(k,n-1)],
    # j ranges over [max(0,k-n+1), min(k,n-1)]... 
    # j = k - i, so when i = max(0,k-n+1), j = k - max(0,k-n+1) = min(k, n-1)
    # when i = min(k,n-1), j = k - min(k,n-1) = max(0, k-n+1)
    # So j ranges from max(0,k-n+1) to min(k,n-1), same range.
    # grad[i] = 2 * w[k] * a[k-i], where a[k-i] is valid.
    
    # Let's use FFT for efficiency when top_k is moderate:
    w_full = np.zeros(conv_size)
    w_full[idx] = w
    # grad = 2 * (w_full ⋆ a) = 2 * conv(w_full, a_flip)[n-1:n-1+n]
    
    total_len = conv_size + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    F = np.fft.rfft(a_flip, fft_size)
    cross = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * cross[n-1:n-1+n]
    
    return grad, w, idx

def adam_update(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = state['m']
    v = state['v']
    t = state['t']
    t += 1
    state['t'] = t
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return params - update, {'m': m, 'v': v, 't': t}

def main_optimize(a0, time_limit, top_k_init=16, beta_start=50.0, beta_end=5000.0,
                  lr_init=0.01, use_adam=True):
    """Main optimization loop with top-k smooth max + Adam."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    beta = beta_start
    top_k = top_k_init
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    
    if use_adam:
        adam_state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        
        # Adaptive top_k: increase as we converge
        top_k_cur = min(conv.size, max(top_k, int(top_k_init * (1 + 0.1 * it / 100))))
        
        grad, w, idx = compute_grad_topk(a, conv, top_k_cur, beta)
        
        if use_adam:
            a_new, adam_state = adam_update(a, grad, adam_state, lr)
            a_new = project_simplex(a_new)
        else:
            v = momentum * v - lr * grad
            a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.95
            if no_improve > 100:
                a = best_a.copy()
                v = np.zeros(n)
                if use_adam:
                    adam_state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
                beta = min(beta * 1.3, beta_end)
                no_improve = 0
        
        # Ramp up beta slowly
        if beta < beta_end:
            beta *= 1.001
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    it={it}, obj={best_obj:.10f}, beta={beta:.1f}, top_k={top_k_cur}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Top-k LSE smooth max with Adam optimizer on the probability simplex
2. Log-barrier interior point method for exploring the interior
3. Adaptive peak tracking with FFT-based gradient computation
4. Periodic smoothing-based escapes from local minima
5. Multi-length exploration (try nearby n values)
Key insight: dC[k]/da[i] = 2*a[k-i] (valid indices), enabling efficient FFT-based gradient.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_max_grad(a, conv, top_k, beta):
    """Compute gradient of top-k LSE smooth max of conv using FFT.
    Returns (gradient, objective_estimate).
    """
    n = len(a)
    conv_size = 2 * n - 1

    if top_k >= conv_size:
        idx = np.arange(conv_size)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]

    vals = conv[idx]
    scaled = beta * (vals - vals.max())
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()

    # LSE estimate of the smooth max
    smax = vals.max() + np.log(exp_vals.sum()) / beta

    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    # = 2 * cross_correlation(w_full, a) at positions 0..n-1
    w_full = np.zeros(conv_size)
    w_full[idx] = w
    a_flip = a[::-1].copy()

    total_len = conv_size + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    F = np.fft.rfft(a_flip, fft_size)
    cross = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * cross[n - 1:n - 1 + n]

    return grad, smax


def topo_k_gradient_descent(a0, time_limit, beta_start=30.0, beta_end=10000.0,
                            top_k_start=8, top_k_max=128,
                            lr_init=0.05, use_adam=True):
    """Top-k LSE smooth max gradient descent with Adam on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    beta = beta_start
    top_k = top_k_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    if use_adam:
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        top_k_cur = min(2 * n - 1, max(top_k, int(top_k_start * (1.0 + 0.05 * it / 50))))

        grad, smax = compute_smooth_max_grad(a, conv, top_k_cur, beta)

        if use_adam:
            t_adam += 1
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = project_simplex(a_new)
        else:
            a_new = project_simplex(a - lr * grad)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
            if no_improve > 80:
                a = best_a.copy()
                if use_adam:
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                lr = max(lr_init * 0.5, 0.001)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.002

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TKGD it={it}, obj={best_obj:.10f}, beta={beta:.1f}, "
                  f"top_k={top_k_cur}, lr={lr:.6f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_a, best_obj


def barrier_optimize(a0, time_limit, mu_init=0.01, mu_decay=0.9995,
                     beta=80.0, top_k=24, lr_init=0.02):
    """Log-barrier interior point method to explore the interior of the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    mu = mu_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        grad_max, _ = compute_smooth_max_grad(a, conv, top_k, beta)

        # Barrier term gradient: -mu / a[i]
        grad = grad_max - mu / a

        # Adaptive step size with backtracking
        alpha = lr
        best_trial = a
        best_trial_obj = conv.max()
        for _ in range(8):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 1e-12)
            a_trial_sum = a_trial.sum()
            if a_trial_sum < 1e-10:
                alpha *= 0.5
                continue
            a_trial = a_trial / a_trial_sum
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_trial_obj:
                best_trial = a_trial
                best_trial_obj = obj_trial
            alpha *= 0.6

        a = best_trial
        obj = best_trial_obj

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.9

        mu = max(mu * mu_decay, 1e-10)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Barrier it={it}, obj={best_obj:.10f}, mu={mu:.2e}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a, best_obj


def smoothing_escape(a0, time_limit, n_smooths=5):

        # Create Gaussian kernel
        kernel_size = int(4 * width) * 2 + 1
        if kernel_size > n:
            kernel_size = n if n % 2 == 1 else n - 1
        x = np.arange(kernel_size) - kernel_size // 2
        kernel = np.exp(-x ** 2 / (2.0 * width ** 2))
        kernel /= kernel.sum()

        # Apply smoothing

        a_smooth = np.maximum(a_smooth, 0.0)

        a_smooth = a_smooth / ss

        if score < best_obj:
            best_obj = score
            best_a = a_opt.copy()
            print(f"    Smoothing width={width}: improved to {best_obj:.10f}")

def fw_lp_step(f0, M, lp_timeout=5):
    """Single FW step with LP oracle. Returns new f or None."""
    n = len(f0)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f0[k - j] for j in js))
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': lp_timeout})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, lp_timeout=3):

    f = np.maximum(f0.astype(np.float64), 0.0)

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-6:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()

            M = conv.max()
            if M < best_obj - 1e-14:

                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_f, best_obj

def interpolate_sequence(f, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(f)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    return f_new
