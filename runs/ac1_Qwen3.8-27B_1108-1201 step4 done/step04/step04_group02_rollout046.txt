def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient using softmax over top peaks."""
    n = len(f)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg

def freq_flatten(a, strength=0.3):
    """Gerchberg-Saxton style frequency flattening step."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Target: flat magnitude (except DC)
    # DC component should remain
    dc = A[0].real  # This is sum(a) = 1 (if normalized)
    
    # For non-DC, target magnitude = mean of current non-DC magnitudes
    if len(mag) > 1:
        non_dc_mag = mag[1:]
        target_mag = np.sqrt(np.mean(non_dc_mag ** 2))  # RMS
    else:
        target_mag = 0
    
    # Blend towards flat
    new_mag = mag.copy()
    new_mag[1:] = (1 - strength) * mag[1:] + strength * target_mag
    new_mag[0] = dc  # Keep DC
    
    # Reconstruct
    new_A = new_mag * np.exp(1j * phase)
    new_a = np.fft.irfft(new_A, fft_size)[:n]
    
    # Project to non-negative
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-15:
        return a.copy()
    return new_a / s

def coordinate_descent(a, n_passes=2, n_trials=5, delta=0.01):
    """Simple coordinate descent: try small changes to each coordinate."""
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    for _pass in range(n_passes):
        improved = False
        for i in range(n):
            best_local = a[i]
            best_val = best_obj
            for trial in range(n_trials):
                d = delta * (1 - 2 * trial / max(1, n_trials - 1))  # From +delta to -delta
                a_trial = a.copy()
                a_trial[i] = max(0, a[i] + d)
                # Renormalize
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                conv_t = fast_convolve(a_trial)
                val = conv_t.max()
                if val < best_val:
                    best_val = val
                    best_local = a_trial[i]
                    best_obj = val
                    a = a_trial.copy()
                    conv = conv_t
                    improved = True
        if not improved:
            break
    
    return a, best_obj

def subgradient_descent(f0, time_limit, initial_lr=0.005, mom=0.9, top_k=64,

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp
    
    # BB state
    f_prev = None
    sg_prev = None
    s_prev = None
    y_prev = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        
        # BB step size
        if f_prev is not None and sg_prev is not None:
            s_diff = f - f_prev
            y_diff = sg - sg_prev
            num = np.dot(s_diff, s_diff)
            den = np.dot(s_diff, y_diff)
            if den > 1e-12:
                bb_lr = num / den
                if 1e-6 < bb_lr < 0.5:
                    cur_lr = 0.6 * cur_lr + 0.4 * bb_lr
        
        f_prev = f.copy()
        sg_prev = sg.copy()
        
        # Momentum step
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        
        # Adaptive parameters
        cur_temp *= 1.005
        cur_lr *= 0.9998
        mom = min(0.99, mom + 0.0001)
        
        # Frequency flattening
        if use_freq and it % freq_every == 0 and rem > 10:
            f_freq = freq_flatten(f, strength=0.2)
            conv_freq = fast_convolve(f_freq)
            obj_freq = conv_freq.max()
            if obj_freq < best_obj:
                f = f_freq
                conv = conv_freq
                best_obj = obj_freq
                best_f = f.copy()
                v = np.zeros(n)
                no_improve = 0
        
        # Coordinate descent
        if use_coord and it % coord_every == 0 and rem > 30:
            f_coord, obj_coord = coordinate_descent(f, n_passes=1, n_trials=3, 
                                                     delta=0.005)
            if obj_coord < best_obj:
                best_obj = obj_coord
                best_f = f_coord.copy()
                f = f_coord
                conv = fast_convolve(f)
                v = np.zeros(n)
                no_improve = 0
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.003 * (1 + it // 1000)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.5
            cur_temp = temp * 0.3
            no_improve = 0
    
    return best_f, best_obj

    inits.append(ramp_i / ramp_i.sum())

    inits.append(sn / sn.sum())
    
    # Random Dirichlet (various concentrations)
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            inits.append(px)
            # Blends
            inits.append(0.7 * px + 0.3 * np.ones(n) / n)

    return inits

Looking at the problem more carefully:
- The objective is `2*n*max(conv)/(sum)^2`
- For normalized (sum=1): `2*n*max(conv)`
- Previous best: 1.505622 with n=384
- This means max(conv) ≈ 1.505622 / (2*384) ≈ 0.001960

The theoretical minimum for max(conv) with sum=1 and n elements is 1/(2n-1) ≈ 1/767 ≈ 0.001304, giving objective ≈ 1.0026.

We're at 0.001960 vs theoretical 0.001304, so we're about 50% above the theoretical minimum. There's definitely room to improve.

The key challenge is that the non-negativity constraint makes it hard to achieve a flat convolution. A complex sequence (like an m-sequence) could achieve near-perfect flatness, but real non-negative sequences are more constrained.

Let me think about what the optimal structure might look like. For a real non-negative sequence on [0, n-1]:
- The convolution at lag 0 is sum(a[i]^2) (the energy)
- The convolution at lag n-1 is sum(a[i]*a[n-1-i]) (the "anti-correlation")
- For the max to be small, ALL lags should be approximately equal

If all lags are equal to 1/(2n-1), then sum(a[i]^2) = 1/(2n-1), which means the sequence is "as spread out as possible" in the L2 sense. For a uniform sequence, sum(a[i]^2) = n*(1/n)^2 = 1/n ≈ 1/(2n) for large n, which is close to 1/(2n-1). So uniform is actually not too bad in terms of lag 0, but it has the problem that the middle lags are larger.

For uniform: conv[k] = (n-|k|)/n^2 for |k| < n. Max is at k=0: 1/n. But the "average" is 1/(2n-1) ≈ 1/(2n). So the max is about 2x the average for uniform.

To improve, we need to reduce the peak at k=0 (and other peaks) by making the sequence more "oscillatory" in some sense. But we're constrained to non-negative!

This is the fundamental tension: we want oscillation (to flatten the spectrum) but need non-negativity.

One approach that might work: **amplitude modulation**. Take a window function (like a raised cosine) and modulate it with a high-frequency component that stays non-negative:
a[i] = W[i] * (1 + alpha * cos(2*pi*f*i + phi))
where W is a window, alpha < 1 to ensure non-negativity, and f, phi are chosen to optimize flatness.

This gives us a family of parameterized sequences that we can optimize over (alpha, f, phi, window shape).

    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient descent with BB step sizes, adaptive temperature, and momentum
2. Frequency-domain flattening (Gerchberg-Saxton inspired projections)
3. Coordinate descent refinement for final polish
4. Multi-length exploration with parametric family initializations
5. Simulated-annealing-style acceptance for escaping local minima
"""

import numpy as np
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient using softmax weighting over top convolution peaks."""
    n = len(f)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
        w = np.ones(m) / m
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def freq_flatten(a, strength=0.3):
    """Gerchberg-Saxton style: blend magnitude towards flat in frequency domain."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)

    dc = mag[0]
    if len(mag) > 1:
        non_dc = mag[1:]
        target = np.sqrt(np.mean(non_dc ** 2))
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - strength) * mag[1:] + strength * target
        new_mag[0] = dc
    else:
        new_mag = mag

    new_A = new_mag * np.exp(1j * phase)
    new_a = np.fft.irfft(new_A, fft_size)[:n]
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-15:
        return a.copy()
    return new_a / s


def coordinate_descent(a, n_passes=2, n_trials=5, delta=0.005, time_limit=30):
    """Coordinate descent: try small changes to individual coordinates."""
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    t0 = time.monotonic()

    for _pass in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            best_val = best_obj
            best_shift = 0.0
            for t in range(n_trials):
                frac = (2.0 * t / max(1, n_trials - 1)) - 1.0
                d = delta * frac
                a_trial = a.copy()
                a_trial[i] = max(0.0, a[i] + d)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                conv_t = fast_convolve(a_trial)
                val = conv_t.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_shift = d
            if abs(best_shift) > 1e-15:
                a[i] = max(0.0, a[i] + best_shift)
                s = a.sum()
                a = a / s
                conv = fast_convolve(a)
                best_obj = conv.max()
                improved = True
        if not improved:
            break
    return a, best_obj


def subgradient_optimize(f0, time_limit, initial_lr=0.005, mom=0.9,
                         top_k=64, temp=0.001, use_freq=True,
                         use_coord=False, freq_every=40, coord_every=200,
                         sa_prob=0.05, verbose=False):
    """
    Subgradient descent with:
    - Barzilai-Borwein adaptive step size
    - Momentum
    - Adaptive temperature (annealing)
    - Frequency-domain flattening
    - Optional coordinate descent
    - Simulated annealing acceptance
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp
    cur_mom = mom
    sa_temp = 0.002

    f_prev = None
    sg_prev = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)

        if f_prev is not None and sg_prev is not None:
            s_diff = f - f_prev
            y_diff = sg - sg_prev
            num = np.dot(s_diff, s_diff)
            den = np.dot(s_diff, y_diff)
            if den > 1e-12:
                bb_lr = num / den
                if 1e-6 < bb_lr < 0.5:
                    cur_lr = 0.5 * cur_lr + 0.5 * bb_lr

        f_prev = f.copy()
        sg_prev = sg.copy()

        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        accept = obj_new < best_obj - 1e-14
        if not accept and sa_prob > 0:
            delta_obj = obj_new - best_obj
            if delta_obj < sa_temp and np.random.random() < np.exp(-delta_obj / sa_temp):
                accept = True

        if accept:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new

        cur_temp *= 1.004
        cur_lr *= 0.99985
        cur_mom = min(0.995, cur_mom + 0.00005)
        sa_temp *= 0.999

        if use_freq and it % freq_every == 0 and rem > 10:
            strength = 0.15 + 0.1 * (it % 3) / 3.0
            f_freq = freq_flatten(f, strength=strength)
            conv_freq = fast_convolve(f_freq)
            obj_freq = conv_freq.max()
            if obj_freq < best_obj - 1e-14:
                f = f_freq
                conv = conv_freq
                best_obj = obj_freq
                best_f = f.copy()
                v = np.zeros(n)
                no_improve = 0

        if use_coord and it % coord_every == 0 and rem > 20:
            f_coord, obj_coord = coordinate_descent(f, n_passes=1,
                                                     n_trials=3, delta=0.003,
                                                     time_limit=min(10, rem - 5))
            if obj_coord < best_obj - 1e-14:
                best_obj = obj_coord
                best_f = f_coord.copy()
                f = f_coord
                conv = fast_convolve(f)
                v = np.zeros(n)
                no_improve = 0

        if no_improve > 400:
            rng = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            noise_scale = 0.002 + 0.002 * min(no_improve / 400.0, 1.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.5
            cur_temp = temp * 0.3
            sa_temp = 0.003
            cur_mom = mom
            no_improve = 0

    return best_f, best_obj


def generate_structured_inits(n, parent=None, rng=None):

    # Random Dirichlet
    for alpha_d in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha_d)
        inits.append(r)

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            inits.append(px)
            inits.append((0.6 * px + 0.4 * np.ones(n) / n) / (0.6 * px + 0.4 * np.ones(n) / n).sum())
            inits.append((0.7 * px + 0.3 * iq / iq.sum()) / (0.7 * px + 0.3 * iq / iq.sum()).sum())

    return inits


def evaluate_and_update(a, best_seq, best_score, label=""):
    """Evaluate sequence and update best if improved. Returns (best_seq, best_score)."""
    a = np.maximum(a, 0.0).astype(np.float64)
    a = np.minimum(a, 1000.0)
    if a.sum() < 0.01:
        return best_seq, best_score
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score):
        if score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        else:
            print(f"  [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining subgradient descent,
    frequency-domain flattening, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(1.0, total_budget - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            best_seq, best_score = evaluate_and_update(parent, best_seq, best_score, "parent_init")
    except NameError:
        print("No parent available.")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # === Phase 1: Aggressive optimization of parent ===
    print("\n=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t_avail = deadline - time.monotonic() - 400
        t1 = min(250, max(50, t_avail * 0.35))
        print(f"  Subgrad+Freq on parent (n={n_p}), t={t1:.0f}s")

        a_opt, _ = subgradient_optimize(
            parent, time_limit=t1, initial_lr=0.004, mom=0.88,
            top_k=max(32, n_p // 6), temp=0.0004,
            use_freq=True, freq_every=25, use_coord=True, coord_every=120,
            sa_prob=0.03)
        best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "parent_sg")

        # Try frequency perturbations
        for st in [0.05, 0.1, 0.2, 0.3]:
            a_f = freq_flatten(best_seq, strength=st)
            best_seq, best_score = evaluate_and_update(a_f, best_seq, best_score, f"freq_{st}")

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length Exploration ===")

    lengths_to_try = [256, 320, 384, 448, 512, 640]
    if parent is not None:
        plen = len(parent)
        if plen in lengths_to_try:
            lengths_to_try.remove(plen)
        lengths_to_try.insert(0, plen)  # Already done in phase 1, but include for reference
        lengths_to_try = lengths_to_try[1:]  # Skip since done

    n_rem = deadline - time.monotonic() - 300
    n_lengths = len(lengths_to_try)
    t_per_len = max(30, int(n_rem / max(1, n_lengths)))
    t_per_len = min(t_per_len, 180)

    n_inits_to_try = 4

    for n in lengths_to_try:
        if time.monotonic() > deadline - 200:
            break

        inits = generate_structured_inits(n, parent, rng)
        selected = inits[:n_inits_to_try]

        rem = min(t_per_len, deadline - time.monotonic() - 150)
        t_per_init = max(10, rem // max(1, len(selected)))

        for i, f_init in enumerate(selected):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 8:
                break

            a_opt, _ = subgradient_optimize(
                f_init, time_limit=t_alloc, initial_lr=0.008, mom=0.9,
                top_k=max(16, n // 10), temp=0.002,
                use_freq=True, freq_every=35, use_coord=False,
                sa_prob=0.05)
            label = f"n{n}_i{i}"
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, label)

    print(f"  Phase 2 complete. Best: {best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print("\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        n_deep = 4
        for p in range(n_deep):
            rem = deadline - time.monotonic() - 80
            if rem < 25:
                break
            t_alloc = min(int(rem / (n_deep - p)), 160)

            if p == 0:
                f_init = best_seq.copy()
                kwargs_opt = {"initial_lr": 0.002, "top_k": n_best // 5,
                              "temp": 0.0001, "freq_every": 20, "coord_every": 80,
                              "mom": 0.92, "sa_prob": 0.01}
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                kwargs_opt = {"initial_lr": 0.003, "top_k": n_best // 4,
                              "temp": 0.0003, "freq_every": 25, "coord_every": 100,
                              "mom": 0.9, "sa_prob": 0.02}
            elif p == 2:
                f_freq = freq_flatten(best_seq, strength=0.4)
                f_init = project_simplex(0.6 * best_seq + 0.4 * f_freq)
                kwargs_opt = {"initial_lr": 0.002, "top_k": n_best // 3,
                              "temp": 0.0002, "freq_every": 30, "coord_every": 120,
                              "mom": 0.91, "sa_prob": 0.02}
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
                kwargs_opt = {"initial_lr": 0.005, "top_k": n_best // 3,
                              "temp": 0.0008, "freq_every": 40, "coord_every": 150,
                              "mom": 0.88, "sa_prob": 0.04}

            a_opt, _ = subgradient_optimize(f_init, time_limit=t_alloc,
                                            use_freq=True, use_coord=True, **kwargs_opt)
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, f"deep_{p}")

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent (multiple passes)
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_c, _ = coordinate_descent(best_seq, n_passes=4, n_trials=7,
                                        delta=0.002, time_limit=min(30, rem - 5))
            best_seq, best_score = evaluate_and_update(a_c, best_seq, best_score, "coord_final")

        # Fine subgradient
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_fine, _ = subgradient_optimize(
                best_seq, time_limit=min(25, rem - 5),
                initial_lr=0.0008, mom=0.95, top_k=n_best // 2,
                temp=0.00005, use_freq=False, use_coord=False, sa_prob=0.0)
            best_seq, best_score = evaluate_and_update(a_fine, best_seq, best_score, "fine_sg")

        # Micro frequency tweaks
        for st in [0.03, 0.06, 0.09, 0.12]:
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            a_f = freq_flatten(best_seq, strength=st)
            best_seq, best_score = evaluate_and_update(a_f, best_seq, best_score, f"micro_f{st}")
            # Quick subgrad from there
            if a_f.sum() > 0.01:
                a_f_norm = a_f / a_f.sum()
                a_opt, _ = subgradient_optimize(
                    a_f_norm, time_limit=min(8, rem - 3),
                    initial_lr=0.0005, top_k=n_best // 2, temp=0.00003,
                    use_freq=False, use_coord=False, sa_prob=0.0)
                best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, f"micro_f{st}_sg")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 12:
            for _ in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 6:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0003)
                a_opt, _ = subgradient_optimize(
                    f_pert, time_limit=min(8, rem - 3),
                    initial_lr=0.0003, top_k=n_best // 2, temp=0.00002,
                    use_freq=False, use_coord=True, coord_every=30, sa_prob=0.0)
                best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "tiny_pert")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback used: {best_score:.8f}")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = np.ones(256) / 256
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
