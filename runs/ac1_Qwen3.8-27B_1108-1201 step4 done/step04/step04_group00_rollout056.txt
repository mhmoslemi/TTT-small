import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate to new length and normalize."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

                improved = True

    conv = fast_convolve(a)
    return a, conv, best_a, best_obj, improved


def nesterov_gradient_phase(a, time_limit, lr_init=0.05, beta_init=0.85,
                            top_k_init=40, temp_init=0.005, verbose=False):
    """Nesterov accelerated subgradient descent."""
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    y = a.copy()
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    top_k = top_k_init
    temp = temp_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k, temp)

        v = beta * v - lr * sg
        a_new = project_simplex(a + v)

        # Nesterov momentum update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a_new + ((beta - 1.0) / beta_new) * (a_new - a)
        beta = beta_new
        a = a_new

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter adjustment
        if no_improve > 100:
            lr *= 0.85
            temp = min(temp * 1.1, 0.1)
            if no_improve > 400:
                # Restart
                rng = np.random.default_rng(it * 7 + 3)
                noise_scale = 0.002 + 0.003 * min(no_improve // 500, 3)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                y = a.copy()
                v = np.zeros(n)
                beta = beta_init
                lr = lr_init * 0.5
                top_k = top_k_init
                temp = temp_init
                no_improve = 0

        # Gradually increase top_k for finer resolution
        if it % 200 == 0:
            top_k = min(int(top_k_init + it // 50), 200)

        # Decay learning rate over time
        progress = (time.monotonic() - t0) / time_limit
        lr = max(lr_init * 0.01, lr_init * (1.0 - 0.8 * progress))

    return best_a, best_obj


def multi_phase_optimize(a0, time_limit, verbose=False):
    """Run multi-phase optimization on a single sequence."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    t0 = time.monotonic()
    deadline = t0 + time_limit

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    # Phase 0: Spectral shaping (global) - first 15%
    t_phase = time_limit * 0.15
    alpha = 0.2
    while time.monotonic() < t0 + t_phase:
        a_new = spectral_shaping_step(a, alpha)
        conv = fast_convolve(a_new)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_a = a_new.copy()
            a = a_new
        alpha = min(0.6, alpha * 1.03)
        # Small gradient step between spectral steps
        sg = compute_subgradient(a, fast_convolve(a), 30, 0.01)
        a = project_simplex(a - 0.02 * sg)
        conv = fast_convolve(a)
        if conv.max() < best_obj:
            best_obj = conv.max()
            best_a = a.copy()

    # Phase 1: Nesterov gradient - next 35%
    t_phase1 = time_limit * 0.35
    a_best, obj_best = nesterov_gradient_phase(
        best_a, time_limit=t_phase1,
        lr_init=0.05, beta_init=0.85,
        top_k_init=40, temp_init=0.005
    )
    if obj_best < best_obj:
        best_obj = obj_best
        best_a = a_best
    a = best_a

    # Phase 2: Multiplicative updates - next 20%
    t_phase2_start = time.monotonic()
    t_phase2 = time_limit * 0.20
    mult_temp = 0.02
    while time.monotonic() - t_phase2_start < t_phase2:
        conv = fast_convolve(a)
        a_new = multiplicative_pressure_update(a, conv, top_k=40, temperature=mult_temp)
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_obj - 1e-14:
            best_obj = conv_new.max()
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend: take a small step toward the multiplicative update
            a = project_simplex(0.9 * a + 0.1 * a_new)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()
        mult_temp = max(0.001, mult_temp * 0.99)

    # Phase 3: Fine Nesterov gradient - next 20%
    t_phase3 = time_limit * 0.20
    a_best, obj_best = nesterov_gradient_phase(
        best_a, time_limit=t_phase3,
        lr_init=0.01, beta_init=0.93,
        top_k_init=80, temp_init=0.001
    )
    if obj_best < best_obj:
        best_obj = obj_best
        best_a = a_best
    a = best_a

    # Phase 4: Coordinate descent polish - remaining time
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_curr = a.copy()
        conv_curr = fast_convolve(a_curr)
        passes = 0
        while time.monotonic() < deadline - 2:
            passes += 1
            a_curr, conv_curr, a_bd, obj_bd, improved = coordinate_descent_pass(
                a_curr, conv_curr, max_coords=min(n, 100)
            )
            if obj_bd < best_obj - 1e-14:
                best_obj = obj_bd
                best_a = a_bd.copy()
                a = a_bd
            if not improved and passes > 5:
                # Try a small perturbation and restart CD
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 99999)
                a_curr = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                conv_curr = fast_convolve(a_curr)
                passes = 0

    return best_a, best_obj


def make_initialization(n, parent, init_type=0, seed=42):
    """Create various initializations for a sequence of length n."""
    rng = np.random.default_rng(seed)

    if init_type == 0:
        # Parent interpolated
        if parent is not None:
            return interpolate_sequence(parent, n)
        return np.ones(n) / n

    elif init_type == 1:
        # Cosine modulation
        if parent is not None:
            base = interpolate_sequence(parent, n)
        else:
            base = np.ones(n) / n
        x = np.arange(n)
        mod = 1.0 + 0.2 * np.cos(2.0 * np.pi * x / n)
        a = base * mod
        return project_simplex(a)

    elif init_type == 2:
        # Raised cosine window
        x = np.arange(n) / max(n - 1, 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        return project_simplex(a)

    elif init_type == 3:
        # Gaussian-like window
        x = (np.arange(n) - n / 2) / (n / 4)
        a = np.exp(-0.5 * x * x)
        return project_simplex(a)

    elif init_type == 4:
        # Random phase flat spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        phase = rng.uniform(0, 2 * np.pi, size=fft_size // 2 + 1)
        A = np.exp(1j * phase)
        a = np.fft.irfft(A, fft_size)[:n]
        return project_simplex(a)

    elif init_type == 5:
        # Parent + noise
        if parent is not None:
            base = interpolate_sequence(parent, n)
            a = base + rng.standard_normal(n) * 0.05
            return project_simplex(a)
        return np.ones(n) / n

    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: spectral shaping, Nesterov gradient, multiplicative
    updates, coordinate descent, with multi-length and multi-initialization search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}", flush=True)
            if np.isfinite(parent_score) and parent_score < best_score:
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        pn = len(parent)
        # Try parent length and some alternatives
        candidate_lengths = [pn]
        # Try larger
        for factor in [1.25, 1.5, 2.0]:
            cl = int(pn * factor)
            cl = (cl // 2) * 2  # Make even
            if cl != pn and cl <= 4096:
                candidate_lengths.append(cl)
        # Try smaller
        if pn > 256:
            candidate_lengths.append(max(128, pn // 2))
    else:
        candidate_lengths = [512, 768, 1024]

    # Remove duplicates, sort
    candidate_lengths = sorted(set(candidate_lengths))
    print(f" lengths to try: {candidate_lengths}", flush=True)

    # Allocate time: give more time to promising lengths
    # We'll do a quick pass first, then deep dive on the best
    n_lengths = len(candidate_lengths)
    quick_budget = 60  # seconds per length for quick exploration
    total_quick = quick_budget * n_lengths

    # Phase A: Quick exploration across lengths and initializations
    print(f"\n=== Phase A: Quick Exploration ===", flush=True)
    length_results = {}

    for ln in candidate_lengths:
        if time.monotonic() > deadline - 60:
            break
        t_budget = min(quick_budget, deadline - time.monotonic() - 30)
        if t_budget < 10:
            break

        print(f"  Exploring n={ln} (budget={t_budget:.0f}s)", flush=True)

        # Try a few initializations
        for init_type in [0, 1, 4]:
            if time.monotonic() > deadline - 60:
                break
            a0 = make_initialization(ln, parent, init_type, seed=seed + init_type)
            t_alloc = max(5, t_budget / 3)
            a_opt, obj = multi_phase_optimize(a0, time_limit=t_alloc)
            try_seq(a_opt, f"n{ln}_init{init_type}")

        length_results[ln] = best_score  # Track which length gives best

    # Phase B: Deep optimization on the best length(s)
    print(f"\n=== Phase B: Deep Optimization (best={best_score:.8f}) ===", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20

        # Try deep optimization with different strategies
        strategies = [
            ("same_init", 0),
            ("perturb", 5),
            ("cosine", 1),
            ("rand_phase", 4),
        ]

        for strat_name, strat_type in strategies:
            rem = deadline - time.monotonic() - 15
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 300)
            a0 = make_initialization(n_best, best_seq, strat_type, seed=seed + 100)
            print(f"  Deep [{strat_name}] n={n_best}, t={t_alloc:.0f}s", flush=True)
            a_opt, obj = multi_phase_optimize(a0, time_limit=t_alloc)
            try_seq(a_opt, f"deep_{strat_name}")

    # Phase C: Try a different (longer) length with deep optimization
    print(f"\n=== Phase C: Longer Length Exploration ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        # Try a significantly longer length
        longer_n = min(4096, n_best * 2)
        if longer_n != n_best:
            rem = deadline - time.monotonic() - 20
            if rem > 60:
                a0 = interpolate_sequence(best_seq, longer_n)
                t_alloc = min(rem * 0.5, 400)
                print(f"  Longer n={longer_n}, t={t_alloc:.0f}s", flush=True)
                a_opt, obj = multi_phase_optimize(a0, time_limit=t_alloc)
                try_seq(a_opt, f"longer_{longer_n}")

                # Also try a medium length if not already tried
                med_n = int(n_best * 1.5)
                med_n = (med_n // 2) * 2
                if med_n != n_best and med_n != longer_n:
                    rem = deadline - time.monotonic() - 20
                    if rem > 40:
                        a0 = interpolate_sequence(best_seq, med_n)
                        t_alloc = min(rem * 0.5, 200)
                        print(f"  Medium n={med_n}, t={t_alloc:.0f}s", flush=True)
                        a_opt, obj = multi_phase_optimize(a0, time_limit=t_alloc)
                        try_seq(a_opt, f"med_{med_n}")

    # Phase D: Final intensive polishing
    print(f"\n=== Phase D: Final Polish (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            # Run coordinate descent intensively
            a_curr = best_seq.copy()
            conv_curr = fast_convolve(a_curr)
            cd_start = time.monotonic()
            cd_limit = min(rem, 120)
            passes = 0
            while time.monotonic() - cd_start < cd_limit:
                passes += 1
                a_curr, conv_curr, a_bd, obj_bd, improved = coordinate_descent_pass(
                    a_curr, conv_curr, max_coords=min(n_best, 150)
                )
                if obj_bd < best_score / (2 * n_best) - 1e-14:
                    # Convert: best_score = 2*n*max_conv, so max_conv = best_score/(2*n)
                    pass  # Already handled by try_seq
                if obj_bd < best_score / (2 * n_best):
                    try_seq(a_bd, f"cd_pass{passes}")
                if not improved and passes > 10:
                    break

            # Very fine gradient step
            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10:
                a_opt, obj = nesterov_gradient_phase(
                    best_seq, time_limit=min(rem2, 60),
                    lr_init=0.003, beta_init=0.95,
                    top_k_init=150, temp_init=0.0001
                )
                try_seq(a_opt, "fine_grad")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Shouldn't happen, but just in case
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
