    return 2.0 * n * conv.max()

    n = len(v)

    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

def spectral_shaping_iteration(f, n_iterations=100, beta=0.5, alpha_decay=0.99):
    """

    n = len(f)

min t s.t. t >= conv_f[k] + 2*(B*g)[k] for all k, g >= -f, sum(g)=0

This is a linearization of the objective around the current point. It finds the best first-order descent direction. This is good but might be overkill for final polishing.

For final polishing, a simpler approach might work:
- At each step, identify the single worst lag k*
- Find the pair (j, j+k*) with the largest contribution f(j)*f(j+k*)
- Reduce both by a small amount, redistribute

Let me think about the redistribution. If we reduce f(j) by δ and f(j+k*) by δ, we need to add 2δ somewhere. The best place to add it is where the "cost" (increase in max b) is lowest.

The cost of adding δ at position p is: for each lag k, the increase in b(k) is approximately δ * f(p+k) + δ * f(p-k) (if those indices exist). We want to minimize the maximum of these increases.

This is a water-filling problem! We can solve it exactly.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def harmonic_optimize(f0, n_harmonics=15, time_limit=60):
    """
    Parameterize f(j) = f0(j) * w(j) where log(w(j)) = sum c_m * phi_m(j)
    and optimize c using scipy.
    """
    f = f0.astype(np.float64).copy()
    n = len(f)
    x = np.arange(n, dtype=np.float64)
    
    # Basis functions: cos and sin harmonics + polynomial corrections
    basis = [np.ones(n)]
    for m in range(1, n_harmonics + 1):
        basis.append(np.cos(2 * np.pi * m * x / n))
        basis.append(np.sin(2 * np.pi * m * x / n))
    # Add some polynomial corrections
    for p in [2, 4]:
        basis.append(((x - n/2) / (n/2))**p)
    
    basis = np.array(basis).T  # shape (n, n_basis)
    n_basis = basis.shape[1]
    
    def cost(c):
        log_w = basis @ c
        # Clip to avoid overflow
        log_w = np.clip(log_w, -5, 5)
        w = np.exp(log_w)
        f_new = f * w
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            return 1e10
        f_new = f_new / s
        return fast_convolve(f_new).max()
    
    c0 = np.zeros(n_basis)
    t0 = time.monotonic()
    best_c = c0.copy()
    best_cost = cost(c0)
    
    # Use Nelder-Mead or Powell
    result = minimize(cost, c0, method='Nelder-Mead',
                      options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10,
                               'adaptive': True})
    if result.fun < best_cost:
        best_cost = result.fun
        best_c = result.x
    
    # Also try Powell
    result2 = minimize(cost, c0, method='Powell',
                       options={'maxiter': 1000, 'xtol': 1e-10, 'ftol': 1e-12})
    if result2.fun < best_cost:
        best_cost = result2.fun
        best_c = result2.x
    
    log_w = basis @ best_c
    log_w = np.clip(log_w, -5, 5)
    f_new = f * np.exp(log_w)
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    
    return f_new, best_cost

def directed_peak_reduction(f0, time_limit, n_peaks=5, step_size=0.01):
    """
    Directly target peak lags and reduce their contributions.
    """
    f = f0.astype(np.float64).copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        # Find peak lags
        if n_peaks >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        improved = False
        
        for k in sorted(peaks, key=lambda x: -conv[x]):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find the pair (j, j+k) with largest contribution
            if k == 0:
                # b(0) = sum f(j)^2, reduce by making f more uniform
                # Find j with largest f(j) and redistribute
                j = np.argmax(f)
                # Find position to move mass to (smallest f)
                l = np.argmin(f)
                if j == l:
                    continue
                delta = step_size * f[j]
                f_new = f.copy()
                f_new[j] -= delta
                f_new[l] += delta
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
                new_obj = fast_convolve(f_new).max()
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_f = f_new.copy()
                    f = f_new
                    improved = True
            else:
                # For lag k, find j that maximizes f(j)*f(j+k)
                valid_j = np.arange(0, n - k)
                contributions = f[valid_j] * f[valid_j + k]
                j_max = valid_j[np.argmax(contributions)]
                
                # Try reducing f(j_max) and/or f(j_max + k)
                for target in [j_max, j_max + k]:
                    # Find where to add the mass
                    # Prefer positions where f(p+k) and f(p-k) are small
                    scores = np.zeros(n)
                    for p in range(n):
                        s = 0
                        if p + k < n:
                            s += f[p + k]
                        if p - k >= 0:
                            s += f[p - k]
                        scores[p] = s
                    
                    # Among positions with small scores, pick one with small f
                    # Use a combined score
                    combined = scores + 0.5 * f
                    l_min = np.argmin(combined)
                    
                    if l_min == target:
                        continue
                    
                    delta = step_size * f[target]
                    f_new = f.copy()
                    f_new[target] -= delta
                    f_new[l_min] += delta
                    f_new = np.maximum(f_new, 0)
                    fs = f_new.sum()
                    if fs < 1e-15:
                        continue
                    f_new = f_new / fs
                    new_obj = fast_convolve(f_new).max()
                    if new_obj < best_obj - 1e-15:
                        best_obj = new_obj
                        best_f = f_new.copy()
                        f = f_new
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            # Try a random perturbation
            perturbation = rng.standard_normal(n) * 0.005
            f = project_simplex(f + perturbation)
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_f = f.copy()
            
            if no_improve > 50:
                # Larger jump
                perturbation = rng.standard_normal(n) * 0.02
                f = project_simplex(best_f + perturbation)
                no_improve = 0
        else:
            no_improve = 0
    
    return best_f, best_obj

def basin_hopping(f0, time_limit, n_harmonics=10):
    """
    Basin hopping: alternate between local optimization (harmonic) and 
    global jumps (random frequency perturbations).
    """
    f = f0.astype(np.float64).copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temperature = 0.01
    
    while time.monotonic() - t0 < time_limit:
        # Local optimization via harmonic parameterization
        if time.monotonic() - t0 > time_limit - 30:
            break
        f_opt, obj_opt = harmonic_optimize(f, n_harmonics=n_harmonics, 
                                           time_limit=min(30, time_limit - (time.monotonic() - t0)))
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_f = f_opt.copy()
        
        # Global jump
        # Perturb in frequency domain
        N = 1 << (2 * n - 1).bit_length()
        F = np.fft.rfft(best_f, N)
        # Add random phase perturbation
        phase_noise = rng.standard_normal(len(F)) * temperature
        F_new = F * np.exp(1j * phase_noise)
        f_new = np.fft.irfft(F_new, N)[:n]
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        
        # Accept with Metropolis criterion
        delta = obj_new - best_obj
        if delta < 0 or rng.random() < np.exp(-delta / (temperature * best_obj)):
            f = f_new
        else:
            f = best_f.copy()
        
        # Cool down
        temperature *= 0.97
        if temperature < 0.001:
            temperature = 0.01  # Reheat
    
    return best_f, best_obj
