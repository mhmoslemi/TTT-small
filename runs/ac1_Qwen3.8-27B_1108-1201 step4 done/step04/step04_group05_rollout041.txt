def fast_convolve(a):

"""
Multi-method optimization for minimum peak autocorrelation sequences.
Combines Frank-Wolfe LP, Alternating QP (Hankel matrix), LSE gradient descent,
and coordinate descent for fine polishing. Multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fw_lp_step(f, M):
    """Single FW step: solve LP and return new point g."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f)) / len(f)

        f = f / s

    return best_f, best_obj

def solve_hankel_qp(w, n):
    """Solve min f^T H_w f s.t. 1^T f = 1, f >= 0 where H_w[j,m] = w[j+m]."""
    # Construct the Hankel matrix
    H = np.zeros((n, n))
    for j in range(n):
        for m in range(n):
            H[j, m] = w[j + m]
    
    # Try unconstrained solution first
    try:
        H_reg = H + 1e-12 * np.eye(n)
        inv_H_1 = np.linalg.solve(H_reg, np.ones(n))
        denom = np.sum(inv_H_1)
        if abs(denom) < 1e-15:
            raise ValueError
        f_unc = inv_H_1 / denom
        if np.all(f_unc >= -1e-10):
            f_unc = np.maximum(f_unc, 0)
            f_unc /= f_unc.sum()
            return f_unc
    except Exception:
        pass
    
    # Active set: iteratively zero out negative entries
    active = np.ones(n, dtype=bool)
    for _ in range(n):
        n_active = active.sum()
        if n_active == 0:
            break
        if n_active == 1:
            f = np.zeros(n)
            f[np.argmax(active)] = 1.0
            return f
        idx = np.where(active)[0]
        H_aa = H[np.ix_(idx, idx)] + 1e-12 * np.eye(n_active)
        try:
            inv_Ha_1 = np.linalg.solve(H_aa, np.ones(n_active))
            denom = np.sum(inv_Ha_1)
            if abs(denom) < 1e-15:
                break
            f_a = inv_Ha_1 / denom
        except Exception:
            break
        if np.all(f_a >= -1e-10):
            f = np.zeros(n)
            f[idx] = np.maximum(f_a, 0)
            f /= f.sum()
            return f
        # Find the most negative entry and deactivate it
        worst = idx[np.argmin(f_a)]
        active[worst] = False
    
    # Fallback: uniform
    return np.ones(n) / n

def alternating_qp(f0, time_limit, w_smooth=0.1, verbose=False):

    it = 0
    tau = w_smooth
    
    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        # Step 1: Update w based on current convolution
        c = fast_convolve(f)
        c_scaled = c / tau
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w /= w.sum()
        
        # Step 2: Solve QP for f given w
        f_new = solve_hankel_qp(w, n)
        
        # Step 3: Check improvement
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:

            no_improve = 0
        else:
            # Try a blend
            for alpha in [0.1, 0.3, 0.5, 0.7]:
                f_blend = (1 - alpha) * f + alpha * f_new

                f_blend /= sb
                cb = fast_convolve(f_blend).max()

def lse_gradient_descent(f0, time_limit, tau=0.01, lr=0.01, verbose=False):
    """Gradient descent on the log-sum-exp smooth approximation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    it = 0
    cur_lr = lr
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        c = fast_convolve(f)
        # LSE gradient
        c_scaled = c / tau
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w /= w.sum()
        
        # Gradient: grad_j = (1/tau) * sum_m w[j+m] * f[m]
        # This is cross-correlation of w and f
        # Compute via FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f, fft_size)
        # cross-correlation: corr[j] = sum_m w[j+m] * f[m]
        # = sum_m w[j+m] * f[m] ... 
        # Actually this is (w * f_flip) convolved...
        # Let me think: grad_j = sum_{m=0}^{n-1} w_{j+m} * f_m
        # This is the cross-correlation of w and f at lag j
        # = (f * w_flip)[j + n - 1] ... 
        # Actually, let's just use the direct formula via FFT:
        # grad_j = sum_m f[m] * w[j+m]
        # This is like (f reversed) convolved with w, evaluated at position...
        # Let's use: if we define g[m] = f[m], h[m] = w[m]
        # Then grad_j = sum_m g[m] * h[j+m] = (g_flipped * h)[j + n - 1]
        # where g_flipped[m] = g[n-1-m]
        
        # Simpler: use direct FFT approach
        # grad_j = sum_{m=0}^{n-1} f[m] * w[j+m]
        # Pad f to length 2n-1 (zero-padded), pad w to length 2n-1
        # Then grad is the cross-correlation
        
        # Actually, the simplest way: 
        # Let's compute via the relationship: (f * f)[k] = sum_j f[j] f[k-j]
        # d/dc_k (f*f)[k] / df[j] = f[k-j] (if 0 <= k-j <= n-1)
        # So grad_j = (1/tau) * sum_k w[k] * f[k-j] * 1{0 <= k-j <= n-1}
        # = (1/tau) * sum_{m=0}^{n-1} w[j+m] * f[m]
        # where m = k-j

        # = (w_conv_f)[j] where w_conv_f is the cross-correlation
        
        # Using FFT: cross_correlation(w, f)[j] = IDFT(conj(DFT(f)) * DFT(w))[j]
        # or equivalently: IDFT(DFT(f[::-1]) * DFT(w))[j + n - 1]
        
        f_pad = np.zeros(fft_size)
        f_pad[:n] = f
        w_pad = np.zeros(fft_size)
        w_pad[:min(len(w), fft_size)] = w
        
        F_fft = np.fft.rfft(f_pad)
        W_fft = np.fft.rfft(w_pad)
        
        # Cross-correlation
        corr = np.fft.irfft(np.conj(F_fft) * W_fft, fft_size)
        grad = corr[:n] / tau
        
        # Gradient step on simplex
        f_new = f - cur_lr * grad
        f_new = project_simplex(f_new)
        
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            f = f_new
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0

            no_improve = 0
        
        if verbose and it % 50 == 0:

def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

    n = len(f)

    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit - 2:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 3:
                break
            for source in [(j+1) % n, (j-1) % n, rng.integers(n)]:
                source = int(source) % n
                if source == j:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = f[source] * frac
                    if delta < 1e-15:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new /= sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-14:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    
    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits

"""
Multi-method optimization for minimum peak autocorrelation sequences.
Combines Frank-Wolfe LP oracle, Alternating Hankel-QP, LSE smooth gradient descent,
and coordinate descent polishing. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import hankel


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fw_lp_step(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        g = fw_lp_step(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        best_alpha, best_val = 0.0, M
        for alpha in [0.5, 0.25, 0.75, 0.125, 0.375, 0.625, 0.1875, 0.3125]:
            ft = np.maximum((1 - alpha) * f + alpha * g, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0)
            fs = f.sum()
            if fs > 1e-15:
                f /= fs
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def solve_hankel_qp(w, n):
    """Solve min f^T H_w f s.t. 1^T f = 1, f >= 0 where H_w[j,m] = w[j+m]."""
    c_col = w[:n]
    r_row = w[n - 1:]
    H = hankel(c_col, r_row)
    H_reg = H + 1e-10 * np.eye(n)

    try:
        inv_H_1 = np.linalg.solve(H_reg, np.ones(n))
        denom = np.sum(inv_H_1)
        if abs(denom) < 1e-15:
            raise ValueError
        f_unc = inv_H_1 / denom
        if np.all(f_unc >= -1e-8):
            f_unc = np.maximum(f_unc, 0)
            f_unc /= f_unc.sum()
            return f_unc
    except Exception:
        pass

    active = np.ones(n, dtype=bool)
    for _ in range(min(n, 50)):
        n_active = active.sum()
        if n_active == 0:
            break
        if n_active == 1:
            f = np.zeros(n)
            f[np.argmax(active)] = 1.0
            return f
        idx = np.where(active)[0]
        H_aa = H[np.ix_(idx, idx)] + 1e-10 * np.eye(n_active)
        try:
            inv_Ha_1 = np.linalg.solve(H_aa, np.ones(n_active))
            denom = np.sum(inv_Ha_1)
            if abs(denom) < 1e-15:
                break
            f_a = inv_Ha_1 / denom
        except Exception:
            break
        if np.all(f_a >= -1e-8):
            f = np.zeros(n)
            f[idx] = np.maximum(f_a, 0)
            sn = f.sum()
            if sn > 1e-15:
                f /= sn
            return f
        worst = idx[np.argmin(f_a)]
        active[worst] = False

    return np.ones(n) / n


def alternating_qp(f0, time_limit, w_smooth=0.05, verbose=False):
    """Alternating minimization between peak weights and Hankel-QP for f."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    tau = w_smooth
    rng = np.random.default_rng(555 + n)

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        c = fast_convolve(f)
        c_scaled = c / tau
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w /= w.sum()

        f_new = solve_hankel_qp(w, n)
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        improved_this_iter = False
        if obj_new < best_obj - 1e-12:
            f = f_new.copy()
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            improved_this_iter = True
        else:
            for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
                f_blend = (1 - alpha) * f + alpha * f_new
                f_blend = np.maximum(f_blend, 0)
                sb = f_blend.sum()
                if sb < 1e-15:
                    continue
                f_blend /= sb
                cb = fast_convolve(f_blend).max()
                if cb < best_obj - 1e-12:
                    f = f_blend.copy()
                    best_obj = cb
                    best_f = f.copy()
                    no_improve = 0
                    improved_this_iter = True
                    break
            if not improved_this_iter:
                no_improve += 1

        if verbose and it % 30 == 0:
            print(f"    AltQP iter {it}: obj={best_obj:.10f}")

        if no_improve > 25:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0
            tau = max(tau * 0.8, 0.01)

    return best_f, best_obj


def lse_gradient_descent(f0, time_limit, tau=0.01, lr=0.005, verbose=False):
    """Gradient descent on log-sum-exp smooth approximation of max-conv."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    rng = np.random.default_rng(888 + n)

    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        c = fast_convolve(f)
        c_scaled = c / tau
        c_scaled -= c_scaled.max()
        p = np.exp(c_scaled)
        p /= p.sum()

        pf = np.zeros(fft_size)
        pf[:n] = f
        pp = np.zeros(fft_size)
        pp[:len(p)] = p

        F_fft = np.fft.rfft(pf)
        P_fft = np.fft.rfft(pp)
        cross_corr = np.fft.irfft(np.conj(P_fft) * F_fft, fft_size)
        grad = cross_corr[:n] / tau

        f_new = project_simplex(f - cur_lr * grad)
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            f = f_new.copy()
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        tau *= 1.001

        if no_improve > 150:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            cur_lr = max(cur_lr * 2.0, 0.002)
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    LSE-GD iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, tau={tau:.6f}")

    return best_f, best_obj


def pnorm_gradient_descent(f0, time_limit, p=30):
    """Gradient descent on the p-norm of the convolution (smooth for finite p)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = 0.003
    rng = np.random.default_rng(333 + n)

    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        c = fast_convolve(f)
        c_pos = np.maximum(c, 1e-15)
        c_pow = c_pos ** (p - 1)
        norm_factor = (np.sum(c_pos ** p)) ** (1.0 / p - 1.0)

        cpf = np.zeros(fft_size)
        cpf[:len(c_pow)] = c_pow
        pf = np.zeros(fft_size)
        pf[:n] = f

        C_fft = np.fft.rfft(cpf)
        F_fft = np.fft.rfft(pf)
        cross_corr = np.fft.irfft(np.conj(C_fft) * F_fft, fft_size)
        grad = norm_factor * cross_corr[:n] / p

        f_new = project_simplex(f - cur_lr * grad)
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            f = f_new.copy()
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998

        if no_improve > 100:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            cur_lr = max(cur_lr * 1.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent for fine polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(444 + n)

    while time.monotonic() - t0 < time_limit - 3:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 5:
                break
            for source in [(j + 1) % n, (j - 1) % n, (j + n // 3) % n, rng.integers(n)]:
                source = int(source) % n
                if source == j:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = f[source] * frac
                    if delta < 1e-15:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new /= sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-14:
                        f = f_new.copy()
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits


def try_seq(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method cascade: FW-LP, Alternating Hankel-QP, LSE/p-norm gradient descent,
    coordinate descent. Multi-length exploration around 384.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    state = {'best_seq': None, 'best_score': float('inf')}

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            s_val = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s_val):
                print(f"  Parent score: {s_val:.6f}")
                state['best_score'] = s_val
                state['best_seq'] = parent.copy()
    except Exception as e:
        print(f"  Parent not available: {e}")

    # === Phase 1: Exploration (20% time) ===
    print("=== Phase 1: Exploration ===")
    phase1_frac = 0.20
    phase1_end = time.monotonic() + (deadline - time.monotonic()) * phase1_frac

    lengths = [350, 370, 384, 392, 400, 420]
    if parent is not None:
        lengths = sorted(lengths, key=lambda x: abs(x - len(parent)))

    for n in lengths[:5]:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_inits(n, parent, rng)
        t_avail = max(0, phase1_end - time.monotonic() - 15)
        n_try = min(3, len(inits))
        t_per = max(10, t_avail / max(1, n_try))
        for i in range(n_try):
            if time.monotonic() > phase1_end - 10:
                break
            f_init = inits[i]
            t_alloc = min(t_per, phase1_end - time.monotonic() - 8)
            if t_alloc < 8:
                break
            a_opt, _ = fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"P1_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep Optimization (50% time) ===
    print("\n=== Phase 2: Deep Optimization ===")
    phase2_frac = 0.50
    phase2_end = time.monotonic() + (deadline - time.monotonic()) * phase2_frac

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        candidate_lengths = [n_best]
        for delta in [-16, -8, 8, 16, 24, -24]:
            cand = n_best + delta
            if 100 <= cand <= 600 and cand not in candidate_lengths:
                candidate_lengths.append(cand)

        for n_try in candidate_lengths:
            if time.monotonic() > phase2_end - 50:
                break

            if n_try == len(state['best_seq']):
                f_init = state['best_seq'].copy()
            elif parent is not None:
                f_init = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, len(parent)), parent)
                f_init = np.maximum(f_init, 0)
                f_init /= f_init.sum()
            else:
                f_init = np.ones(n_try) / n_try

            rem = phase2_end - time.monotonic() - 30
            if rem < 30:
                break

            # FW-LP (primary)
            t_fw = min(rem * 0.30, 100)
            if t_fw > 15:
                a_opt, obj = fw_optimize(f_init, time_limit=t_fw)
                try_seq(a_opt, f"P2_FW_n{n_try}", state)

            # Alternating QP (new method)
            rem = phase2_end - time.monotonic() - 25
            if rem > 15 and state['best_seq'] is not None:
                f_qp_start = state['best_seq'] if len(state['best_seq']) == n_try else f_init
                t_qp = min(rem * 0.25, 70)
                if t_qp > 12:
                    a_opt, obj = alternating_qp(f_qp_start, time_limit=t_qp, w_smooth=0.04)
                    try_seq(a_opt, f"P2_QP_n{n_try}", state)

            # LSE gradient descent (smooth)
            rem = phase2_end - time.monotonic() - 25
            if rem > 15 and state['best_seq'] is not None:
                f_lse_start = state['best_seq'] if len(state['best_seq']) == n_try else f_init
                t_lse = min(rem * 0.25, 70)
                if t_lse > 12:
                    a_opt, obj = lse_gradient_descent(f_lse_start, time_limit=t_lse,
                                                      tau=0.008, lr=0.004)
                    try_seq(a_obj if False else a_opt, f"P2_LSE_n{n_try}", state)

            # P-norm gradient descent
            rem = phase2_end - time.monotonic() - 20
            if rem > 15 and state['best_seq'] is not None:
                f_pn_start = state['best_seq'] if len(state['best_seq']) == n_try else f_init
                t_pn = min(rem * 0.2, 50)
                if t_pn > 12:
                    a_opt, obj = pnorm_gradient_descent(f_pn_start, time_limit=t_pn, p=40)
                    try_seq(a_opt, f"P2_PN_n{n_try}", state)

        # Perturbation runs from best
        if state['best_seq'] is not None:
            n_b = len(state['best_seq'])
            for ps in [0.004, 0.01, 0.025, 0.05]:
                rem = phase2_end - time.monotonic() - 20
                if rem < 15:
                    break
                noise = rng.standard_normal(n_b) * ps
                f_new = project_simplex(state['best_seq'] + noise)
                t_alloc = min(rem * 0.2, 50)
                if t_alloc > 12:
                    a_opt, _ = fw_optimize(f_new, time_limit=t_alloc)
                    try_seq(a_opt, f"P2_pert_{ps}", state)

            # Mirror blend
            rem = phase2_end - time.monotonic() - 15
            if rem > 15:
                f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
                t_alloc = min(rem * 0.4, 50)
                a_opt, _ = fw_optimize(f_mirror, time_limit=t_alloc)
                try_seq(a_opt, "P2_mirror", state)

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Polishing (remaining time) ===
    print("\n=== Phase 3: Polishing ===")
    if state['best_seq'] is not None:
        n_b = len(state['best_seq'])

        rem = deadline - time.monotonic() - 15
        if rem > 25:
            t1 = min(rem * 0.25, 90)
            a_opt, _ = lse_gradient_descent(state['best_seq'], time_limit=t1,
                                            tau=0.004, lr=0.002)
            try_seq(a_opt, "P3_LSE_fine", state)

        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t2 = min(rem * 0.2, 70)
            a_opt, _ = pnorm_gradient_descent(state['best_seq'], time_limit=t2, p=60)
            try_seq(a_opt, "P3_PN_fine", state)

        rem = deadline - time.monotonic() - 12
        if rem > 15:
            t3 = min(rem * 0.2, 60)
            a_opt, _ = alternating_qp(state['best_seq'], time_limit=t3, w_smooth=0.02)
            try_seq(a_opt, "P3_QP_fine", state)

        rem = deadline - time.monotonic() - 10
        if rem > 12:
            t4 = min(rem * 0.25, 60)
            a_opt, _ = fw_optimize(state['best_seq'], time_limit=t4)
            try_seq(a_opt, "P3_FW_final", state)

        rem = deadline - time.monotonic() - 10
        if rem > 12:
            t5 = min(rem * 0.25, 50)
            a_opt, _ = coordinate_descent(state['best_seq'], time_limit=t5)
            try_seq(a_opt, "P3_CD", state)

        rem = deadline - time.monotonic() - 8
        if rem > 10:
            t6 = min(rem * 0.4, 40)
            a_opt, _ = lse_gradient_descent(state['best_seq'], time_limit=t6,
                                            tau=0.002, lr=0.001)
            try_seq(a_opt, "P3_LSE_ultra", state)

        rem = deadline - time.monotonic() - 8
        if rem > 10:
            t7 = min(rem * 0.4, 30)
            a_opt, _ = coordinate_descent(state['best_seq'], time_limit=t7)
            try_seq(a_opt, "P3_CD_final", state)

    # Fallback
    if state['best_seq'] is None:
        print("  Using fallback uniform(384)")
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    else:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
