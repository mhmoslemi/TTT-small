import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0

def palindromic_fconv(f, g):
    """Compute conv(f, g) assuming both are palindromic.
    Returns only the first n elements (the rest are determined by symmetry)."""
    n = len(f)
    full_conv = np.convolve(f, g)
    return full_conv[:n]  # First n elements

def build_palindromic_lp_matrix(f, n):
    """Build the LP matrix for the palindromic FW-LP.
    Variables: g[0], ..., g[n/2-1] (for even n)
    Constraints: conv(f, g_pal)[k] <= M for k = 0, ..., n-1
    
    Where g_pal is the palindromic extension of the variables.
    """
    # n is even, so n/2 free variables
    half = n // 2
    # For constraint k, the coefficient of g[i] (i < half) is:
    #   f[k-i] (if 0 <= k-i < n) + f[k-(n-1-i)] (if 0 <= k-(n-1-i) < n)
    #   = f[k-i] + f[n-1-k+i] (if both valid)
    #   (these are the two contributions from g[i] and g[n-1-i])
    
    rows = []
    cols = []
    vals = []
    for k in range(n):  # Only first n constraints (due to symmetry)
        for i in range(half):
            coeff = 0.0
            # Contribution from g[i] (at position i in the full sequence)
            if 0 <= k - i < n:
                coeff += f[k - i]
            # Contribution from g[n-1-i] (at position n-1-i in the full sequence)
            j = n - 1 - i
            if 0 <= k - j < n:
                coeff += f[k - j]
            if coeff != 0.0:
                rows.append(k)
                cols.append(i)
                vals.append(coeff)
    
    return csr_matrix((vals, (rows, cols)), shape=(n, half))

def fw_lp_palindromic(f, time_limit, lp_timeout=30):
    """FW-LP on the palindromic subsimplex."""
    n = len(f)
    assert n % 2 == 0, "n must be even for palindromic method"
    half = n // 2
    
    # Normalize
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    # Ensure palindromic
    f = make_palindromic(f)
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Build LP matrix
        A = build_palindromic_lp_matrix(f, n)
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(half)
        b_ub = np.full(n, M)
        bounds = [(0.0, None)] * half
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_timeout})
            if not res.success:
                no_improve += 5
                if no_improve > 50:
                    break
                continue
        except Exception:
            no_improve += 5
            if no_improve > 50:
                break
            continue
        
        g_half = np.maximum(res.x, 0.0)
        g_sum_half = g_half.sum()
        if g_sum_half < 1e-15:
            break
        
        # Extend to full palindromic sequence
        g = np.zeros(n)
        g[:half] = g_half
        g[half:] = g_half[::-1]
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = make_palindromic(ft)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine-tune with bisection
        if best_alpha > 0:
            lo = max(0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):

                ft = (1 - mid) * f + mid * g
                ft = make_palindromic(ft)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = make_palindromic(f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Random restart
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(half) * 0.003
            f_half = np.maximum(best_f[:half] + noise, 0)
            f = np.zeros(n)
            f[:half] = f_half
            f[half:] = f_half[::-1]
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindromic FW-LP (reduced LP with symmetry)
2. Mirror descent with exact subgradient
3. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def make_palindromic(a):
    return (a + a[::-1]) / 2.0

def normalize_palindromic(a):
    a = make_palindromic(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    return a

def build_pal_lp_matrix(f, n):
    half = n // 2
    rows = []
    cols = []
    vals = []
    for k in range(n):
        for i in range(half):
            coeff = 0.0
            idx1 = k - i
            if 0 <= idx1 < n:
                coeff += f[idx1]
            idx2 = k - n + 1 + i
            if 0 <= idx2 < n:
                coeff += f[idx2]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindromic FW-LP with reduced LP (exploits symmetry to halve variables/constraints)
2. Mirror descent with exact subgradient and adaptive step size
3. Nesterov-accelerated subgradient in the reduced palindromic space
4. Multi-length exploration with interpolation
Key insight: there always exists an optimal palindromic solution, so we restrict
to the palindromic subsimplex, reducing the LP from (n vars, 2n-1 cons) to (n/2 vars, n cons).
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_palindromic(a):
    """Average with reverse to enforce palindromic symmetry."""
    return (a + a[::-1]) / 2.0


def normalize_palindromic(a):
    """Enforce palindromic + non-negative + sum to 1."""
    a = make_palindromic(a.astype(np.float64))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    return a


def build_pal_lp_matrix(f, n):
    """Build sparse LP matrix for palindromic FW-LP.
    Variables: x[0..half-1] where g[i]=x[i] for i<half, g[n-1-i]=x[i] for i<half.
    Constraints: conv(f,g)[k] <= M for k=0..n-1 (first half by symmetry).
    Coefficient of x_i in constraint k: f[k-i]*1(0<=k-i<n) + f[k-n+1+i]*1(0<=k-n+1+i<n)
    """
    half = n // 2
    rows = []
    cols = []
    vals = []
    for k in range(n):
        for i in range(half):
            coeff = 0.0
            idx1 = k - i
            if 0 <= idx1 < n:
                coeff += f[idx1]
            idx2 = k - n + 1 + i
            if 0 <= idx2 < n:
                coeff += f[idx2]
            if abs(coeff) > 1e-15:
                rows.append(k)
                cols.append(i)
                vals.append(coeff)
    return csr_matrix((vals, (rows, cols)), shape=(n, half))

def nesterov_sgd_pal(f0, time_limit, lr_init=0.01):
    """Nesterov-accelerated subgradient descent with simplex projection on palindromic space."""
    n = len(f0)
    half = n // 2

    def project_simplex_pal(v):
        """Project onto palindromic simplex: sum=1, a>=0, a palindromic."""
        v_half = (v[:half] + v[half:][::-1]) / 2.0
        # Project v_half onto {x >= 0, sum(x) = 0.5}
        # Equivalent to projecting 2*v_half onto standard simplex scaled
        w = 2.0 * v_half
        m = len(w)
        u = np.sort(w)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, m + 1) > (cssv - 1)
        if not cond.any():
            w_new = np.ones(m) / m
        else:
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            w_new = np.maximum(w - theta, 0.0)
        x_half = w_new / 2.0
        result = np.zeros(n)
        result[:half] = x_half
        result[half:] = x_half[::-1]
        return result

    f = normalize_palindromic(f0)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))

        s = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1)
        s[idx] = 2.0 * f[k_star - idx]

        v = momentum * v - lr * s
        f_new = project_simplex_pal(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 0.1)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.95
                momentum = max(0.8, momentum - 0.003)
            if no_improve > 300:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001)
                momentum = 0.88
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
