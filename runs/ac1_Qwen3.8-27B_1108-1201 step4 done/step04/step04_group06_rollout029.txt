"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated gradient methods
with L2 smoothing, peak targeting, and multi-phase search strategy.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_l2(a, conv):
    """Gradient of sum(conv^2) w.r.t. a, computed via cross-correlation."""
    n = len(a)
    size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    C = np.fft.rfft(conv, size)
    xc = np.fft.irfft(np.conj(A) * C, size)[:n]
    return 4.0 * xc


def compute_grad_peak(a, conv, top_k=30, temp=0.005):
    """Soft peak-targeting gradient."""
    n = len(a)
    conv_len = 2 * n - 1
    # Get top_k peaks
    if top_k >= conv_len:
        peaks = np.arange(conv_len)
        peak_vals = conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    # Softmax weights
    scaled = peak_vals / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Build weight vector (full length)
    w_full = np.zeros(conv_len)
    w_full[peaks] = w
    
    # Gradient: 2 * (a cross-corr w_full)[j]
    size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_full, size)
    grad = 2.0 * np.fft.irfft(np.conj(A) * W, size)[:n]
    return grad


def objective(a):
    """Compute the objective value."""
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (S * S)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:size]


def grad_l2_fft(a, conv):
    """O(n log n) gradient of sum(conv^2)."""
    n = len(a)
    size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, size)
    C = np.fft.rfft(conv, size)
    return 4.0 * np.fft.irfft(np.conj(A) * C, size)[:n]


def grad_peak_fft(a, conv, top_k=50, temp=0.01):
    """Gradient of soft-peak objective."""
    n = len(a)
    conv_len = 2 * n - 1
    if top_k >= conv_len:
        idx = np.arange(conv_len)
        vals = conv
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
    s = vals / max(temp, 1e-12)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(idx)); w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    w_full = np.zeros(conv_len)
    w_full[idx] = w
    size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_full, size)
    return 2.0 * np.fft.irfft(np.conj(A) * W, size)[:n]
