def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE objective."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=10000.0, lr_mult=1.0):
    """Log-sum-exp smoothed gradient descent with adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = (2.0 / n) * lr_mult
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(bc)] = 1.0
        else:
            w = exp_bc / w_sum

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.90
                momentum = max(0.80, momentum - 0.005)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001) * lr_mult
                beta = min(beta * 2.0, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj


def pair_cd_optimize(f0, time_limit, grid_size=40, max_sweeps=100):

    no_improve_sweeps = 0

        sweep += 1

        if len(active) < 2:
            break
        n_active = len(active)
        pair_list = []
        for ii in range(n_active):

                pair_list.append((active[ii], active[jj]))
        rng.shuffle(pair_list)

        improved = False
        total_delta = 0.0

        for (i, j) in pair_list:
            ai = f[i]
            aj = f[j]
            c = ai + aj
            if c < 1e-12:
                continue

            # Compute diff array: diff[k] = f[k-i] - f[k-j]
            diff = np.zeros(2 * n - 1)
            diff[i:i + n] = f
            diff[j:j + n] -= f

            # Grid search for optimal delta in [-ai, aj]
            lo = -ai
            hi = aj
            if hi - lo < 1e-14:
                continue

            # Coarse grid
            deltas = np.linspace(lo, hi, grid_size)
            best_delta = 0.0
            best_val = conv.max()

                # new_conv = conv + 2*delta*diff

                candidate[0] += 2.0 * delta * delta
                k_ij = j - i
                if 0 <= k_ij < 2 * n - 1:
                    candidate[k_ij] -= delta * delta
                val = candidate.max()
                if val < best_val:
                    best_val = val
                    best_delta = delta

                # Apply update
                f[i] += best_delta
                f[j] -= best_delta
                # Ensure non-negativity (safety)
                if f[i] < 0:
                    f[i] = 0.0
                    f[j] += best_delta
                if f[j] < 0:
                    f[j] = 0.0
                    f[i] -= best_delta

                # Update conv
                conv += 2.0 * best_delta * diff
                conv[0] += 2.0 * best_delta * best_delta
                k_ij = j - i

                improved = True

        if current_max < best_obj - 1e-13:
            best_obj = current_max
            best_f = f.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

            no_improve_sweeps += 2

def subgrad_polish(f0, time_limit, num_peaks=5):
    """Subgradient descent focusing on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Get top peaks
        if num_peaks >= len(conv):
            peak_idx = np.arange(len(conv))
            peak_vals = conv
        else:
            peak_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]
            peak_vals = conv[peak_idx]

        # Weighted subgradient
        weights = peak_vals / peak_vals.sum()
        grad = np.zeros(n)
        for ww, k in zip(weights, peak_idx):
            if ww < 1e-15:
                continue
            k = int(k)
            # grad[i] += 2 * ww * (f[k-i] + f[i-k])
            # f[k-i] is non-zero for i in [k-(n-1), k]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx] += 2.0 * ww * f[k - idx]
            # f[i-k] is non-zero for i in [k, k+n-1]
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            if i_lo2 <= i_hi2:
                idx2 = np.arange(i_lo2, i_hi2 + 1)

        f = project_simplex(f - lr * grad)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr *= 1.001
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95

        if it % 200 == 0:
            print(f"    SubGrad it={it}, obj={best_obj:.12f}, lr={lr:.8f}")

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    alt_lengths = [432, 464, 480]
    if n_best in alt_lengths:
        alt_lengths.remove(n_best)

    for n_alt in alt_lengths[:2]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 60:
            break
        t_alloc = min(80, t_remaining - 30)
        if t_alloc < 30:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick LSE + Pair CD
        t1 = t_alloc * 0.4
        t2 = t_alloc * 0.6
        a_opt, _ = lse_optimize(f_interp, time_limit=t1,
                                beta_start=20.0, beta_end=5000.0)
        try_sequence(a_opt, best_tracker)
        if t2 > 20:
            cur = best_tracker[1] if (best_tracker[1] is not None and len(best_tracker[1]) == n_alt) else a_opt
            a_opt2, _ = pair_cd_optimize(cur, time_limit=t2, grid_size=40, max_sweeps=50)
            try_sequence(a_opt2, best_tracker)

    # === Final: one more brief pair CD on global best ===
    if best_tracker[1] is not None:
        current_best = best_tracker[1]
        t_remaining = deadline - time.monotonic() - 10
        if t_remaining > 15:
            print(f"\n=== Final Polish: {t_remaining:.0f}s ===")
            a_opt, _ = pair_cd_optimize(current_best, time_limit=t_remaining,
                                        grid_size=60, max_sweeps=50)
            try_sequence(a_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_tracker[0] = final_score
    else:
        final_seq = np.ones(448) / 448
        best_tracker[0] = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_tracker[0]:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
