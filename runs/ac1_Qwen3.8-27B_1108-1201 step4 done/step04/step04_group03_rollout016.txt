def smooth_objective_and_grad(z, t, n):
    """Compute smoothed objective and gradient w.r.t. z."""
    # Softmax parameterization
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp
    t_conv = t * conv
    t_conv_shift = t_conv - t_conv.max()
    lse = t_conv.max() + np.log(np.sum(np.exp(t_conv_shift)))
    
    obj = 2 * n * lse / t
    
    # Gradient of log-sum-exp w.r.t. conv
    softmax_weights = np.exp(t_conv_shift) / np.sum(np.exp(t_conv_shift))
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(2*n-1):
        if softmax_weights[k] < 1e-15:
            continue
        wm = k - n + 1
        we = min(n-1, k)
        for m in range(max(0, wm), we+1):
            idx = k - m
            if 0 <= idx < n:
                grad_a[m] += 2 * t * softmax_weights[k] * a[idx]
    
    # Chain rule through softmax
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_z

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weight optimization with adaptive parameters
2. L-BFGS on log-sum-exp smoothed objective (continuation method)
3. Multiple diverse restarts with structured perturbations
4. Final fine-tuning with very high smoothing

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_from_a(a):
    """Compute the true objective for a normalized non-negative sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_obj_and_grad(z, t, n):
    """Smoothed objective (log-sum-exp) and gradient w.r.t. z (softmax parameterization)."""
    # Softmax
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a_norm = a.sum()
    a = a / a_norm
    
    # Convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp (stable)
    t_conv = t * conv
    t_max = t_conv.max()
    exp_vals = np.exp(t_conv - t_max)
    lse = t_max + np.log(exp_vals.sum())
    obj = 2.0 * n * lse / t
    
    # Softmax weights for peaks
    w = exp_vals / exp_vals.sum()
    
    # Gradient w.r.t. a: df/da[m] = 4n * sum_k w[k] * a[k-m] * 1{0<=k-m<n}
    # This is 4n * cross-correlation of w and a (at appropriate lags)
    corr_full = correlate(w, a, mode='full')
    # corr_full[k + len(a) - 1] = sum_j w[j+k] * a[j] (scipy convention)
    # We want: sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j] = corr_full[m + len(a) - 1]
    cross = corr_full[n - 1:n - 1 + n]  # cross[m] = sum_j w[j+m] * a[j]
    
    grad_a = 4.0 * n * cross
    
    # Chain rule through softmax
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_z


def lbfgs_smoothed(a0, time_limit, t_start=10, t_end=500, t_steps=8, max_iter_per_step=500):
    """
    L-BFGS optimization on smoothed objective with continuation.
    Starts with small t (smooth) and increases to large t (close to true max).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    
    # Convert from a to z (inverse softmax)
    # a[i] = exp(z[i]) / sum exp(z[j])
    # z[i] = log(a[i]) works if a[i] > 0
    # For a[i] = 0, use a small value
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe / a_safe.sum())
    
    t0 = time.monotonic()
    
    # Continuation schedule
    t_schedule = np.geomspace(t_start, t_end, t_steps)
    
    for t_level in t_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Optimize at this smoothing level
        result = minimize(
            smooth_obj_and_grad,
            z,
            args=(t_level, n),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_step, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )
        
        if result.success or result.get('nit', 0) > 0:
            z = result.x
        
        # Convert back to a
        z_shift = z - z.max()
        a = np.exp(z_shift)
        a = a / a.sum()
        
        remaining = time_limit - (time.monotonic() - t0)
        print(f"    L-BFGS t={t_level:.1f}: obj={2.0*n*fast_convolve(a).max():.8f}, "
              f"smooth_obj={result.fun:.8f}, time_left={remaining:.1f}s")
    
    return a


def mult_weight_optimize(f0, time_limit, lr_init=0.05, top_k=20, temp_init=0.001,

    Multiplicative weight (mirror descent) optimization with peak focusing.
    """

    else:
        f = f / s

    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0

    cur_temp = temp_init
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

def accelerated_sg(f0, time_limit, lr=0.005, mom=0.9, top_k=30):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0

        # Nesterov lookahead
        t_new = (1 + np.sqrt(1 + 4 * t_val**2)) / 2
        beta = (t_val - 1) / t_new
        t_val = t_new
        
        conv = fast_convolve(y)
        M = conv.max()
        
        # Subgradient from top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))

            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Weight by how much each peak exceeds the mean
            excess = np.maximum(pv - pv.mean(), 0)
            total = excess.sum()
            if total > 1e-15:
                weights = excess / total
            else:
                weights = np.ones(len(peaks)) / len(peaks)
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * y[k - idx]
        
        # Accelerated update
        lr_t = 2.0 / (1 + t_val) * lr
        y_new = project_simplex(y - lr_t * sg)
        f_new = (1 - beta) * y_new + beta * f
        
        f = f_new
        y = y_new
        
        conv = fast_convolve(f)
        obj = conv.max()

            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

            t_val = 1.0
            no_improve = 0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast multiplicative weight (mirror descent) optimization with adaptive temperature
2. Nesterov-accelerated subgradient method on the simplex
3. L-BFGS on log-sum-exp smoothed objective with continuation (novel)
4. Final fine-tuning with high-resolution L-BFGS and perturbation restarts
Key innovation: L-BFGS on the smoothed objective provides much faster local
convergence than subgradient methods, especially when combined with continuation
(increasing smoothing parameter) to navigate from smooth to sharp landscape.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def true_objective(a):
    """Compute 2n * max(conv(a,a)) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_obj_and_grad(z, t, n):
    """
    Smoothed objective using log-sum-exp with softmax parameterization.
    a = softmax(z), objective = (2n/t) * log(sum_k exp(t * (a*a)[k]))
    Returns (objective, gradient w.r.t. z).
    """
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a_norm = a.sum()
    a = a / a_norm

    conv = fast_convolve(a)
    t_conv = t * conv

    # Stable log-sum-exp
    t_max = t_conv.max()
    exp_vals = np.exp(t_conv - t_max)
    exp_sum = exp_vals.sum()
    lse = t_max + np.log(exp_sum)
    obj = 2.0 * n * lse / t

    # Softmax weights over peaks
    w = exp_vals / exp_sum

    # Gradient w.r.t. a: df/da[m] = 4n * sum_k w[k] * a[k-m] * 1{0<=k-m<n}
    # Computed as cross-correlation: correlate(w, a)[m + n - 1]
    corr_full = correlate(w, a, mode='full')
    cross = corr_full[n - 1:n - 1 + n]
    grad_a = 4.0 * n * cross

    # Chain rule through softmax: grad_z[i] = a[i] * (grad_a[i] - a . grad_a)
    grad_z = a * (grad_a - np.dot(a, grad_a))

    return obj, grad_z


def lbfgs_continuation(a0, time_limit, t_levels=None, max_iter_per_level=400, verbose=True):
    """
    L-BFGS optimization with continuation on the log-sum-exp smoothed objective.
    Uses softmax parameterization a = softmax(z) to maintain positivity and sum=1.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()

    # Invert softmax: z = log(a) (softmax is scale-invariant)
    a_safe = np.maximum(a, 1e-25)
    z = np.log(a_safe)

    if t_levels is None:
        t_levels = [5, 15, 40, 100, 250, 500, 800]

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = true_objective(a)

    for t_level in t_levels:
        if time.monotonic() - t0 > time_limit:
            break

        def fun_and_grad(z_curr):
            return smooth_obj_and_grad(z_curr, t_level, n)

        try:
            result = minimize(
                fun_and_grad,
                z,
                method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': max_iter_per_level,
                    'ftol': 1e-14,
                    'gtol': 1e-10,
                    'maxls': 30
                }
            )
            if result.x is not None and result.get('nit', 0) > 0:
                z = result.x
        except Exception:
            pass

        # Convert back to a
        z_shift = z - z.max()
        a = np.exp(z_shift)
        a = a / a.sum()

        obj = true_objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        remaining = time_limit - (time.monotonic() - t0)
        if verbose:
            print(f"    L-BFGS t={t_level:.0f}: true_obj={obj:.8f}, "
                  f"smooth_obj={result.fun if hasattr(result, 'fun') else 'N/A':.6f}, "
                  f"nit={result.nit if hasattr(result, 'nit') else 0}, "
                  f"time_left={remaining:.1f}s")

    return best_a


def mult_weight_optimize(f0, time_limit, lr_init=0.05, top_k=20, temp_init=0.001,
                         decay_lr=0.9995, decay_temp=1.003):
    """
    Multiplicative weight (mirror descent) optimization with adaptive
    peak-weighting via softmax temperature annealing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= decay_lr
        cur_temp *= decay_temp

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.6, 0.005)
            cur_temp = temp_init * 2.5
            no_improve = 0

    return best_f

        # Rank-based weighting with sharpness
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** sharpness
        wsum = weights.sum()
        if wsum > 0:
            weights /= wsum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * y[k - idx]

        # Nesterov step
        lr_t = lr_scale * 2.0 / (1.0 + t_val)
        y_new = project_simplex(y - lr_t * sg)
        f_new = (1.0 - beta) * y_new + beta * f

        f = f_new
        y = y_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            t_val = 1.0
            no_improve = 0

    return best_f


def peak_surgery(f0, time_limit, num_ops_target=50000):
    """
    Direct peak reduction via targeted mass transfers.
    Identifies the peak and transfers mass from high-contribution to
    low-contribution positions to directly reduce the peak value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find the source: position j that contributes most to this peak
        # Contribution of a[j] to peak at k is a[j] * a[k-j]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = f[idx] * f[peak_k - idx]

        # Source: maximize contribution (and a[j] itself for impact)
        src_candidates = idx[np.argsort(contributions)[::-1][:5]]

        # Target: minimize a[k-j'] (least harmful to increase)
        js2 = max(0, peak_k - n + 1)
        je2 = min(n - 1, peak_k)
        idx2 = np.arange(js2, je2 + 1)
        target_scores = f[peak_k - idx2] - f[idx2] * 0.1
        tgt_candidates = idx2[np.argsort(target_scores)[:5]]

        # Try transfers and keep the best
        best_delta = 0.0
        best_transfer = None
        for src in src_candidates:
            for tgt in tgt_candidates:
                if src == tgt:
                    continue
                if f[src] < 1e-10:
                    continue
                # Try small transfer
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = f[src] * frac
                    if delta < 1e-12:
                        continue
                    f_trial = f.copy()
                    f_trial[src] -= delta
                    f_trial[tgt] += delta
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial[peak_k] < peak_val - 1e-12:
                        # Check overall max didn't increase too much
                        if conv_trial.max() < best_obj * 1.001:
                            if best_transfer is None or conv_trial.max() < conv_trial.max():
                                best_transfer = (src, tgt, delta)
                                best_delta = delta
                                break

        if best_transfer is not None:
            src, tgt, delta = best_transfer
            f[src] -= delta
            f[tgt] += delta
            f = np.maximum(f, 0)
            f = f / f.sum()

            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
        else:
            # Random perturbation to escape
            if rng.random() < 0.3:
                j1 = rng.integers(0, n)
                j2 = rng.integers(0, n)
                if j1 != j2 and f[j1] > 1e-8:
                    delta = f[j1] * 0.02
                    f[j1] -= delta
                    f[j2] += delta
                    f = np.maximum(f, 0)
                    f = f / f.sum()
            # Occasionally restart from best
            if rng.random() < 0.1:
                f = best_f * np.exp(rng.standard_normal(n) * 0.002)
                f = f / f.sum()

        # Periodic L-BFGS polish
        if int(time.monotonic() - t0) % 10 < 1 and time.monotonic() - t0 < time_limit - 20:
            t_levels_short = [50, 150, 400]
            a_polish = lbfgs_continuation(best_f, time_limit=8, t_levels=t_levels_short,
                                           max_iter_per_level=100, verbose=False)
            obj_polish = true_objective(a_polish)
            if obj_polish < best_obj:
                best_obj = obj_polish
                best_f = a_polish.copy()
                f = best_f.copy()

    return best_f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining:
    - Multiplicative weight (mirror descent) for fast global progress
    - Nesterov-accelerated subgradient for steady improvement
    - L-BFGS on smoothed objective for fast local convergence (continuation)
    - Peak surgery for targeted peak reduction
    - Perturbation restarts to escape local minima
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n = 384
        parent = np.ones(n) / n
    else:
        n = len(parent)

    # === Phase 1: Quick MW exploration (~250s) ===
    print("\n=== Phase 1: Quick MW Exploration ===")

    mw_configs = [
        {'lr_init': 0.05, 'top_k': 20, 'temp_init': 0.001, 'decay_lr': 0.9995, 'decay_temp': 1.003},
        {'lr_init': 0.03, 'top_k': 35, 'temp_init': 0.002, 'decay_lr': 0.9997, 'decay_temp': 1.002},
        {'lr_init': 0.08, 'top_k': 12, 'temp_init': 0.0005, 'decay_lr': 0.9993, 'decay_temp': 1.005},
        {'lr_init': 0.02, 'top_k': 50, 'temp_init': 0.003, 'decay_lr': 0.9998, 'decay_temp': 1.001},
    ]

    for ci, cfg in enumerate(mw_configs):
        rem = deadline - time.monotonic() - 1100
        if rem < 10:
            break
        t_alloc = min(60, rem)
        a_opt = mult_weight_optimize(parent, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"mw1_{ci}")

    # Perturbed starts
    rem = deadline - time.monotonic() - 1100
    if rem > 30:
        for pi in range(3):
            rem = deadline - time.monotonic() - 1080
            if rem < 15:
                break
            noise = rng.standard_normal(n) * 0.015
            f_pert = project_simplex(best_seq + noise)
            a_opt = mult_weight_optimize(f_pert, time_limit=min(35, rem),
                                          lr_init=0.04, top_k=25, temp_init=0.0015)
            try_seq(a_opt, f"mw_pert_{pi}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: L-BFGS Continuation (~500s) ===
    print("\n=== Phase 2: L-BFGS Continuation ===")

    rem = deadline - time.monotonic() - 700
    if rem > 30:
        t_alloc = min(400, rem)
        print(f"  L-BFGS continuation, t={t_alloc:.0f}s")
        a_opt = lbfgs_continuation(best_seq, time_limit=t_alloc,
                                    t_levels=[5, 15, 40, 100, 250, 500],
                                    max_iter_per_level=300)
        try_seq(a_opt, "lbfgs_cont")

    # Second L-BFGS pass with higher range
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(250, rem)
        print(f"  L-BFGS high-t, t={t_alloc:.0f}s")
        a_opt = lbfgs_continuation(best_seq, time_limit=t_alloc,
                                    t_levels=[100, 250, 500, 800, 1200],
                                    max_iter_per_level=250)
        try_seq(a_opt, "lbfgs_high")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Accelerated SG + MW Deep (~400s) ===
    print("\n=== Phase 3: Deep SG + MW ===")

    # Accelerated subgradient
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        t_alloc = min(150, rem)
        a_opt = accelerated_sg(best_seq, time_limit=t_alloc, lr_scale=1.0, top_k=40, sharpness=2.0)
        try_seq(a_opt, "accsg_1")

    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_alloc = min(120, rem)
        a_opt = accelerated_sg(best_seq, time_limit=t_alloc, lr_scale=0.6, top_k=60, sharpness=1.5)
        try_seq(a_opt, "accsg_2")

    # Deep MW
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        t_alloc = min(150, rem)
        a_opt = mult_weight_optimize(best_seq, time_limit=t_alloc,
                                      lr_init=0.02, top_k=40, temp_init=0.0008,
                                      decay_lr=0.9998, decay_temp=1.0015)
        try_seq(a_opt, "mw_deep")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Peak Surgery + L-BFGS Polish (~300s) ===
    print("\n=== Phase 4: Peak Surgery + Polish ===")

    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_alloc = min(150, rem)
        print(f"  Peak surgery, t={t_alloc:.0f}s")
        a_opt = peak_surgery(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "surgery")

    # Final L-BFGS at very high t
    rem = deadline - time.monotonic() - 80
    if rem > 15:
        t_alloc = min(100, rem)
        print(f"  Final L-BFGS, t={t_alloc:.0f}s")
        a_opt = lbfgs_continuation(best_seq, time_limit=t_alloc,
                                    t_levels=[300, 600, 1000, 1500],
                                    max_iter_per_level=200)
        try_seq(a_opt, "lbfgs_final")

    # Fine MW
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_alloc = min(40, rem)
        a_opt = mult_weight_optimize(best_seq, time_limit=t_alloc,
                                      lr_init=0.008, top_k=15, temp_init=0.0004,
                                      decay_lr=0.9999, decay_temp=1.001)
        try_seq(a_opt, "mw_fine")

    # === Phase 5: Perturbation + Quick Re-optimize (~80s) ===
    print("\n=== Phase 5: Perturbation Restart ===")

    rem = deadline - time.monotonic() - 15
    if rem > 20:
        num_perts = min(8, int(rem / 12))
        for pi in range(num_perts):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            # Different noise scales
            noise_scale = 0.001 * (1 + pi * 0.5)
            noise = rng.standard_normal(n) * noise_scale
            f_pert = project_simplex(best_seq + noise)

            # Quick L-BFGS polish
            t_alloc = min(10, rem)
            a_opt = lbfgs_continuation(f_pert, time_limit=t_alloc,
                                        t_levels=[100, 400, 800],
                                        max_iter_per_level=100, verbose=False)
            try_seq(a_opt, f"pert_{pi}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
