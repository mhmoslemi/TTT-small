def solve_lp(f, M):

        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coordinate_descent(f0, time_limit, seed=999):
    """Coordinate descent polishing on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            candidates = [(j + 1) % n, (j - 1) % n, (j + n // 3) % n,
                          (j + 2 * n // 3) % n, rng.integers(n)]
            for source in candidates:
                source = int(source) % n
                if source == j:
                    continue
                for frac in [0.03, 0.07, 0.12]:
                    delta = f[source] * frac
                    if delta < 1e-15:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try random restart from best
            if time.monotonic() - t0 < time_limit - 10:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                best_obj = fast_convolve(f).max()
                if best_obj > best_f.max() if False else True:
                    best_obj = fast_convolve(best_f).max()
            else:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    ramp = (x + 1)
    inits.append(ramp / ramp.sum())

    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 7.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 7.0) ** 2)))
    inits.append(peak / peak.sum())

    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    pl = (x + 1.0) ** (-0.3)
    inits.append(pl / pl.sum())

    # Spectral initialization: flat-ish spectrum
    nfft = 1 << (2 * n - 2).bit_length()
    for _ in range(3):
        phase = rng.uniform(0, 2 * np.pi, nfft // 2 + 1)
        mag = 1.0 + 0.2 * rng.standard_normal(nfft // 2 + 1)
        mag = np.maximum(mag, 0.1)
        spec = mag * np.exp(1j * phase)
        spec[0] = spec[0].real
        spec_full = np.zeros(nfft, dtype=complex)
        spec_full[:nfft // 2 + 1] = spec
        spec_full[nfft // 2 + 1:] = np.conj(spec[1:nfft // 2][::-1])
        sig = np.fft.irfft(spec_full, nfft)[:n]
        sig = np.maximum(sig, 0)
        ss = sig.sum()
        if ss > 1e-10:
            inits.append(sig / ss)

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            inits.append(project_simplex(0.8 * (px / ps) + 0.2 * (px[::-1] / ps)))

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid FISTA warm-up + LP-based FW + polishing.
    Multi-length exploration with spectral initializations and Nesterov momentum.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}", flush=True)
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}", flush=True)
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    # === Phase 1: Fast FISTA exploration (20% time) ===
    print("=== Phase 1: FISTA Warm-up Exploration ===", flush=True)
    phase1_end = time.monotonic() + (float(budget_s) * 0.20)

    lengths = [256, 320, 384, 448, 512]
    if parent is not None:
        pn = len(parent)
        if pn in lengths:
            lengths.remove(pn)
        lengths.insert(0, pn)
    lengths.sort(key=lambda x: abs(x - (len(parent) if parent is not None else 384)))

    # Quick FISTA on parent if available
    if parent is not None:
        t_rem = phase1_end - time.monotonic()
        if t_rem > 30:
            t_alloc = min(60, t_rem * 0.4)
            print(f"  FISTA on parent (n={len(parent)}), t={t_alloc:.0f}s", flush=True)
            a_opt, obj = fista_warmup(parent, time_limit=t_alloc, T_init=0.004,
                                      top_k=25, seed=111)
            try_sequence(a_opt, "fista_parent", state)

    # Multi-length FISTA
    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_inits(n, parent, rng)
        n_try = min(4, len(inits))
        t_per = max(10, (phase1_end - time.monotonic() - 15) / (n_try * max(1, len(lengths) - lengths.index(n))))
        t_per = min(t_per, 40)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break
            f_init = inits[i]
            a_opt, obj = fista_warmup(f_init, time_limit=t_alloc,
                                      T_init=0.006, top_k=20,
                                      restart_period=500, noise_scale=0.01,
                                      seed=seed + n * 10 + i)
            try_sequence(a_opt, f"fista_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}", flush=True)

    # === Phase 2: Deep LP-based FW optimization (55% time) ===
    print(f"\n=== Phase 2: Deep LP-based FW Optimization ===", flush=True)
    phase2_end = time.monotonic() + (float(budget_s) * 0.55)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}, time available={(phase2_end - time.monotonic()):.0f}s", flush=True)

        # Main deep FW run
        t_main = min(300, (phase2_end - time.monotonic()) * 0.45)
        if t_main > 30:
            print(f"  Main FW run: t={t_main:.0f}s", flush=True)
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_main,
                                        verbose=True, use_momentum=True, seed=555)
            try_sequence(a_opt, "fw_main", state)

        # Perturbation runs with different scales
        perturb_scales = [0.003, 0.008, 0.02, 0.05]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 20
            if rem < 20:
                break
            t_alloc = min(rem / (len(perturb_scales) + 1), 100)
            if t_alloc < 15:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc,
                                        use_momentum=True, seed=seed + int(ps * 1000))
            try_sequence(a_opt, f"fw_pert_{ps}", state)

        # Mirror blend
        rem = phase2_end - time.monotonic() - 15
        if rem > 25:
            f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
            a_opt, obj = fw_lp_optimize(f_mirror, time_limit=min(rem, 80),
                                        use_momentum=True, seed=888)
            try_sequence(a_opt, "fw_mirror", state)

        # Also try FISTA from current best (different optimizer might escape)
        rem = phase2_end - time.monotonic() - 10
        if rem > 20:
            a_opt, obj = fista_warmup(state['best_seq'], time_limit=min(rem, 80),
                                      T_init=0.003, top_k=15,
                                      restart_period=1000, noise_scale=0.005,
                                      seed=444)
            try_sequence(a_opt, "fista_phase2", state)

    # === Phase 3: Polishing (25% time) ===
    print(f"\n=== Phase 3: Polishing ===", flush=True)
    phase3_end = deadline - 5

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}", flush=True)

        # Coordinate descent (long)
        rem = phase3_end - time.monotonic() - 50
        if rem > 20:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 120),
                                            seed=333)
            try_sequence(a_opt, "cd_long", state)

        # Short FW squeeze
        rem = phase3_end - time.monotonic() - 30
        if rem > 15:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 60),
                                        use_momentum=True, seed=666)
            try_sequence(a_opt, "fw_squeeze1", state)

        # FISTA fine-tune (low temperature, small noise)
        rem = phase3_end - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = fista_warmup(state['best_seq'], time_limit=min(rem, 60),
                                      T_init=0.002, cool_rate=1.0003,
                                      top_k=10, restart_period=2000,
                                      noise_scale=0.002, seed=222)
            try_sequence(a_opt, "fista_fine", state)

        # Final coordinate descent
        rem = phase3_end - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 40),
                                            seed=111)
            try_sequence(a_opt, "cd_final", state)

        # One more FW squeeze if time allows
        rem = phase3_end - time.monotonic() - 1
        if rem > 10:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 30),
                                        use_momentum=True, seed=777)
            try_sequence(a_opt, "fw_squeeze2", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}", flush=True)

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})", flush=True)
    return [float(x) for x in state['best_seq']]
