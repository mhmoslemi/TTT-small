import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective_normalized(f):
    """Compute 2*n*max(conv) for normalized f (sum=1)."""
    n = len(f)

def subgradient_max_conv(f, conv, top_k=64, temp=0.001):
    """Subgradient of max(conv(a,a)) with softmax over top peaks."""
    n = len(f)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0

        w /= wsum

            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)

def fw_direction(f, M, n_constraints=None):
    """Solve LP to find Frank-Wolfe direction."""
    n = len(f)
    c = -np.ones(n)
    
    if n_constraints is None:
        # Use all constraints
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        # Use subset of constraints (the active ones)
        conv = fast_convolve(f)
        active = np.argpartition(conv, -n_constraints)[-n_constraints:]
        A_ub = np.zeros((n_constraints, n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            A_ub[i, idx] = f[k - idx]
        b_ub = np.full(n_constraints, M)
    
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sg(f, conv, top_k=64, temp=0.001):
    """Subgradient of max(conv(a,a)) with softmax weighting over top peaks."""
    n = len(f)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def solve_fw_lp(f, M, top_k_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k peaks, g >= 0."""
    n = len(f)
    c = -np.ones(n)

    if top_k_constraints is not None and top_k_constraints < (2 * n - 1):
        conv = fast_convolve(f)
        active = np.argpartition(conv, -top_k_constraints)[-top_k_constraints:]
        n_c = len(active)
        A_ub = np.zeros((n_c, n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            A_ub[i, idx] = f[k - idx]
        b_ub = np.full(n_c, M)
    else:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

                conv_test = fast_convolve(f_test)
                if conv_test.max() < M - 1e-15:
                    f = f_test
                    M = conv_test.max()
                    break

        if fast_convolve(f).max() < best_obj - 1e-14:
            best_obj = fast_convolve(f).max()
            best_f = f.copy()

    return best_f


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x / max(n - 1, 1) - 0.5  # centered in [-0.5, 0.5]

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine^p windows for different p
    for p in [2, 4, 6, 8]:
        c = np.cos(np.pi * xm) ** p + 0.005
        inits.append(c / c.sum())

    # Gaussian-like
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * (xm / sigma) ** 2)
        inits.append(g / g.sum())

    # Linear ramps
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    # Quadratic
    q = 1.0 + 2.0 * xm ** 2
    inits.append(q / q.sum())

    # Sine
    sn = np.sin(np.pi * (x / n + 0.5)) + 0.02
    inits.append(sn / sn.sum())

    # Random Dirichlet
    for alpha_val in [0.1, 0.3, 1.0]:
        r = rng.dirichlet(np.ones(n) * alpha_val)
        inits.append(r)

    # Parent-based
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            p_resized = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            p_resized = np.maximum(p_resized, 0)
            if p_resized.sum() > 1e-10:
                p_resized = p_resized / p_resized.sum()
                inits.append(p_resized)
                # Blends
                inits.append((0.7 * p_resized + 0.3 * np.ones(n) / n))
                inits.append((0.5 * p_resized + 0.5 * (np.cos(np.pi * xm) ** 4 + 0.005)))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: hybrid optimization combining FW-LP (active set),
    subgradient descent, peak shaving, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_p = len(parent)
            obj_p = 2.0 * n_p * fast_convolve(parent).max()
            print(f"Parent loaded: n={n_p}, objective={obj_p:.8f}")
    except Exception:
        pass

    # === Phase 1: Parent optimization with FW-LP active set ===
    print("\n=== Phase 1: Parent FW-LP (Active Set) ===")
    if parent is not None:
        n_p = len(parent)
        # Use top K constraints for speed
        for k_constraints in [min(n_p, 50), min(n_p, 100)]:
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            t_alloc = min(rem // 2, 150)
            print(f"  FW active-set K={k_constraints}, t={t_alloc:.0f}s")
            a_opt, obj = fw_optimize(parent, time_limit=t_alloc, top_k_constraints=k_constraints)
            try_seq(a_opt, f"parent_fw_k{k_constraints}")

            # Peak shave the result
            a_shaved = peak_shave(a_opt, n_rounds=3)
            try_seq(a_shaved, f"parent_fw_k{k_constraints}_shaved")

    # === Phase 2: SG on parent ===
    print("\n=== Phase 2: Parent SG ===")
    if parent is not None:
        n_p = len(parent)
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  SG on parent, t={t_alloc:.0f}s")
            a_opt, obj = sg_optimize(parent, time_limit=t_alloc,
                                     initial_lr=0.004, top_k=n_p // 5, temp=0.0003,
                                     lp_escape=True, lp_every=100,
                                     n_constraints=min(n_p, 80))
            try_seq(a_opt, "parent_sg")
            a_shaved = peak_shave(a_opt, n_rounds=3)
            try_seq(a_shaved, "parent_sg_shaved")

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length Exploration ===")

    lengths = []
    if parent is not None:
        lp = len(parent)
        for dl in [0, -16, 16, -32, 32, -64, 64, -96, 96, 128, 160]:
            L = lp + dl
            if 128 <= L <= 1024 and L not in lengths:
                lengths.append(L)
    else:
        lengths = [256, 384, 512, 640]

    if parent is not None:
        lengths.sort(key=lambda L: abs(L - len(parent)))

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        if parent is not None and n == len(parent):
            continue

        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break

        inits = generate_inits(n, parent, rng)
        n_inits = len(inits)
        t_per = max(12, rem / max(1, n_inits * 2))
        t_per = min(t_per, 60)

        for i, f_init in enumerate(inits[:min(n_inits, 8)]):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)
            if t_alloc < 10:
                break

            # SG first (faster)
            a_opt, obj = sg_optimize(f_init, time_limit=min(t_alloc * 0.5, t_alloc - 10),
                                     initial_lr=0.01, top_k=max(32, n // 8), temp=0.001,
                                     lp_escape=True, lp_every=80,
                                     n_constraints=max(32, n // 4))
            try_seq(a_opt, f"n{n}_i{i}")

            # FW refine
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 12:
                a_opt2, obj2 = fw_optimize(a_opt, time_limit=min(rem2, 30),
                                           top_k_constraints=max(32, n // 4))
                try_seq(a_opt2, f"n{n}_i{i}_fw")

    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Deep optimization ===
    print("\n=== Phase 4: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple FW runs with different starting points
        starts = [
            best_seq.copy(),
            project_simplex(best_seq + rng.standard_normal(n_best) * 0.003),
            project_simplex(best_seq + rng.standard_normal(n_best) * 0.008),
            project_simplex(0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)),
            project_simplex(0.7 * best_seq + 0.3 * (best_seq[::-1])),
        ]

        for p, f_init in enumerate(starts):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem // (len(starts) - p), 120)

            # FW with active set
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc,
                                     top_k_constraints=min(n_best, 80))
            try_seq(a_opt, f"deep_fw_{p}")

            # Peak shave
            a_shaved = peak_shave(a_opt, n_rounds=5)
            try_seq(a_shaved, f"deep_fw_{p}_shaved")

        # SG deep runs
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem // (4 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
                cfg = dict(initial_lr=0.002, top_k=n_best // 4, temp=0.0002,
                           lp_escape=True, lp_every=80, n_constraints=min(n_best, 60))
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
                cfg = dict(initial_lr=0.004, top_k=n_best // 3, temp=0.0004,
                           lp_escape=True, lp_every=100, n_constraints=min(n_best, 60))
            else:
                f_init = project_simplex(0.95 * best_seq + 0.05 * (np.ones(n_best) / n_best))
                cfg = dict(initial_lr=0.001, top_k=n_best // 2, temp=0.0001,
                           lp_escape=True, lp_every=100, n_constraints=min(n_best, 40))

            a_opt, obj = sg_optimize(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{p}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG with very small steps
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = sg_optimize(best_seq, time_limit=min(rem, 35),
                                     initial_lr=0.0008, mom=0.96, top_k=n_best // 2,
                                     temp=0.00003, lp_escape=True, lp_every=80,
                                     n_constraints=min(n_best, 50))
            try_seq(a_opt, "polish_sg")

        # FW polish (full constraints for precision)
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = fw_optimize(best_seq, time_limit=min(rem, 30),
                                     top_k_constraints=None)
            try_seq(a_opt, "polish_fw")

        # Final peak shave
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_shaved = peak_shave(best_seq, n_rounds=10)
            try_seq(a_shaved, "polish_shave")

        # Micro perturbations + quick polish
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, obj = sg_optimize(f_pert, time_limit=min(rem, 8),
                                         initial_lr=0.0005, mom=0.97,
                                         top_k=n_best // 2, temp=0.00001)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
