    # f[k-j] is non-zero only when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    
    # Efficient computation using FFT:
    # This is the cross-correlation of w and f.
    # corr[j] = sum_k w[k] f[k-j]
    # = ifft(FFT(w) * conj(FFT(f)))[j]
    
    fft_size = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    corr = np.fft.irfft(W * np.conj(F), fft_size)[:n]
    
    return 2.0 * corr

def lse_mirror_descent(f0, time_limit, beta_start=10, beta_end=1000, lr_start=0.01, lr_end=0.0001):
    """Mirror descent with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive parameters (exponential schedule)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad = compute_lse_gradient(f, conv, beta)
        
        # Mirror descent update
        log_f = np.log(np.maximum(f, 1e-30))
        log_f_new = log_f - lr * grad
        f_new = np.exp(log_f_new - log_f_new.max())
        f_new /= f_new.sum()
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()
        
        f = f_new
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        total_iters += 1
        
        if obj < best_obj - 1e-12:

def sg_descent_v2(f0, time_limit, initial_lr=0.005, mom=0.9, top_k=32, temp=0.0005):

def swap_local_search(f0, time_limit, max_swaps_per_sweep=200):

    f = np.maximum(f0.astype(np.float64), 0.0)

            if i == j:
                continue

                continue

                if delta > f[i]:
                    delta = f[i]
                f_try = f.copy()
                f_try[i] -= delta
                f_try[j] += delta
                f_try /= f_try.sum()

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # This is a convolution, so the matrix is banded (Toeplitz-like)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)[k] includes term f[j]*g[k-j] for valid k-j
        # i.e., g[m] appears in C[k] = sum_j f[j] g[k-j] with coefficient f[j] when m = k-j, k = m+j
        # So A_ub[k, m] += f[k-m] for valid k-m
        for m in range(n):
            k = m + j
            if 0 <= k < 2*n-1:
                A_ub[k, m] += f[j]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def fw_lp(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.02, 3.0)
            continue
        
        # Bisect line search
        best_alpha = 0.0
        best_val = M
        lo, hi = 0.0, 1.0
        for _ in range(10):
            mid = (lo + hi) / 2.0
            ft = (1.0 - mid) * f + mid * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                lo = mid
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj / (2.0 * n) - 1e-14:

                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            no_improve += 1

def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)

    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine
    x = np.arange(n, dtype=np.float64)
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())

            px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(px / ps)
    
    return inits

import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(f, conv, beta):
    """Compute gradient of logsumexp(beta * conv) / beta w.r.t. f using FFT correlation."""
    n = len(f)
    m = conv.max()
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -300, 300)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    fft_size = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    corr = np.fft.irfft(W * np.conj(F), fft_size)[:n]
    return 2.0 * beta * corr


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def lse_mirror_descent(f0, time_limit, beta_start=5.0, beta_end=1000.0,
                       lr_start=0.015, lr_end=0.0001):
    """Mirror descent with log-sum-exp smoothing and adaptive beta/lr."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** 0.5 * ((lr_end / lr_start) ** 0.5 * progress)
        lr = max(lr, 1e-6)
        grad = compute_lse_gradient(f, conv, beta)
        log_f = np.log(np.maximum(f, 1e-30))
        update = -lr * grad
        update = np.clip(update, -5.0, 5.0)
        log_f_new = log_f + update
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        fsum = f_new.sum()
        if fsum < 1e-30:
            f_new = np.ones(n) / n
        else:
            f_new /= fsum
        f = f_new
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f = np.maximum(f, 1e-15)
            f /= f.sum()
            conv = fast_convolve(f)
            no_improve = 0
    return best_f, best_obj


def sg_descent(f0, time_limit, initial_lr=0.005, mom=0.92, top_k=48, temp=0.0005):
    """Subgradient descent with momentum, adaptive temp, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.9996
        cur_temp *= 1.004
        if no_improve > 600:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = initial_lr * 0.4
            cur_temp = temp * 0.3
            no_improve = 0
    return best_f, best_obj


def solve_lp(f, M, time_limit=25):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(n):
            k = m + j
            if 0 <= k < 2 * n - 1:
                A_ub[k, m] += f[j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp(f0, time_limit, lp_time=20):
    """Frank-Wolfe with LP oracle and bisection line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        remaining_lp = min(lp_time, max(5, rem - 5) // 3)
        g = solve_lp(f, M * m_exp, time_limit=remaining_lp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue
        best_alpha = 0.0
        best_val = M
        lo, hi = 0.0, 1.0
        for _ in range(12):
            mid = (lo + hi) / 2.0
            ft = (1.0 - mid) * f + mid * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                lo = mid
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj / (2.0 * n) - 1e-14:

                best_f = f.copy()

            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
    return best_f, best_obj


def swap_local_search(f0, time_limit, max_frac=0.3):
    """Local search by randomly swapping mass between coordinate pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.3:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j or f[i] < 1e-12:
            continue
        deltas = rng.uniform(0.005, max_frac, size=3)
        improved = False
        for delta_frac in sorted(deltas):
            delta = delta_frac * f[i]
            f_try = f.copy()
            f_try[i] -= delta
            f_try[j] += delta
            if f_try[i] < 0:
                continue
            f_try /= f_try.sum()
            conv_try = fast_convolve(f_try)
            M_try = conv_try.max()
            if M_try < M - 1e-14:
                f = f_try
                conv = conv_try
                M = M_try
                obj = 2.0 * n * M
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_f = f.copy()
                    stall = 0
                improved = True
                break
        if not improved:
            stall += 1
        if stall > 5000:
            break
    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):

    lengths = [384, 256, 512]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        if parent is not None and n == len(parent):
            continue

        inits = generate_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 120
        t_per = max(25, rem // max(1, len(inits)))
        t_per = min(t_per, 80)

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)
            if t_alloc < 20:
                break
            a_opt, _ = sg_descent(f_init, time_limit=t_alloc,
                                   initial_lr=0.008, top_k=max(24, n // 10), temp=0.001)
            try_seq(a_opt, f"n{n}_init{i}")

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print("\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Long LSE mirror descent
        rem = deadline - time.monotonic() - 120
        if rem > 40:
            t_alloc = min(200, int(rem * 0.4))
            print(f"  Deep LSE (n={n_best}), t={t_alloc}s")
            a_opt, _ = lse_mirror_descent(best_seq, time_limit=t_alloc,
                                           beta_start=10.0, beta_end=5000.0,
                                           lr_start=0.006, lr_end=0.00003)
            try_seq(a_opt, "deep_lse")

        # SG descent with fine parameters
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(150, int(rem * 0.4))
            print(f"  Deep SG (n={n_best}), t={t_alloc}s")
            a_opt, _ = sg_descent(best_seq, time_limit=t_alloc,
                                   initial_lr=0.002, top_k=max(48, n_best // 4), temp=0.0001)
            try_seq(a_opt, "deep_sg")

        # FW-LP for global moves
        rem = deadline - time.monotonic() - 80
        if rem > 40:
            t_alloc = min(120, int(rem * 0.5))
            print(f"  FW-LP (n={n_best}), t={t_alloc}s")
            a_opt, _ = fw_lp(best_seq, time_limit=t_alloc, lp_time=15)
            try_seq(a_opt, "deep_fw")

        # Perturbed LSE
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(80, int(rem * 0.5))
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            print(f"  Perturbed LSE, t={t_alloc}s")
            a_opt, _ = lse_mirror_descent(f_pert, time_limit=t_alloc,
                                           beta_start=5.0, beta_end=2000.0,
                                           lr_start=0.01, lr_end=0.0001)
            try_seq(a_opt, "perturbed_lse")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine LSE
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(50, int(rem * 0.5))
            a_opt, _ = lse_mirror_descent(best_seq, time_limit=t_alloc,
                                           beta_start=1000.0, beta_end=20000.0,
                                           lr_start=0.002, lr_end=0.00001)
            try_seq(a_opt, "polish_lse")

        # Very fine SG
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(35, int(rem * 0.5))
            a_opt, _ = sg_descent(best_seq, time_limit=t_alloc,
                                   initial_lr=0.0008, top_k=n_best // 2, temp=0.00001)
            try_seq(a_opt, "polish_sg")

        # Swap local search
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            t_alloc = min(30, int(rem * 0.6))
            a_opt, _ = swap_local_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish_swap")

        # Another LSE after swap
        rem = deadline - time.monotonic() - 5
        if rem > 8 and best_seq is not None:
            a_opt, _ = lse_mirror_descent(best_seq, time_limit=min(20, rem - 3),
                                           beta_start=2000.0, beta_end=50000.0,
                                           lr_start=0.001, lr_end=0.000005)
            try_seq(a_opt, "final_lse")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
