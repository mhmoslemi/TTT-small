import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_from_conv(conv, n):
    """Compute 2*n*max(conv) for a normalized (sum=1) sequence."""
    return 2.0 * n * conv.max()


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smooth_optimize(f0, time_limit):
    """
    L-BFGS-B optimization of log-sum-exp smoothed objective.
    Uses parameterization a[-1] = 1 - sum(a[:-1]) to handle the simplex constraint.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    # Start with sharp tau, decrease over time
    taus = [200, 100, 50, 30, 20]
    tau_idx = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and tau_idx < len(taus):
        if time.monotonic() - t0 > time_limit:
            break
        tau = taus[tau_idx]
        tau_idx += 1
        tau_time = min(300, time_limit * (2 * tau_idx - 1) / (2 * len(taus)))

        # Work with first n-1 free variables
        current = f.copy()

        def smooth_obj(x):
            # x has n-1 elements, a[-1] = 1 - sum(x)
            a = np.empty(n)
            a[:-1] = x
            a[-1] = 1.0 - x.sum()
            if np.any(a < -1e-10):
                return 1e10
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            # Log-sum-exp for numerical stability
            max_conv = conv.max()
            shifted = tau * (conv - max_conv)
            log_sum = max_conv + np.log(np.sum(np.exp(shifted))) / tau
            return log_sum

        def smooth_grad(x):
            a = np.empty(n)
            a[:-1] = x
            a[-1] = 1.0 - x.sum()
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            max_conv = conv.max()
            shifted = tau * (conv - max_conv)
            weights = np.exp(shifted)
            ws = weights.sum()
            if ws < 1e-30:
                weights = np.zeros_like(conv)
                weights[np.argmax(conv)] = 1.0
            else:
                weights /= ws
            # Gradient of max_k sum_i+j=k a_i a_j wrt a
            grad_a = np.zeros(n)
            for k in range(len(conv)):
                if weights[k] < 1e-15:
                    continue
                for j in range(max(0, k - n + 1), min(n, k + 1)):
                    idx = k - j
                    if 0 <= idx < n:
                        grad_a[j] += 2.0 * weights[k] * a[idx]
            # Chain rule for parameterization: d a_i / d x_j
            # a_i = x_i for i < n-1, a_{n-1} = 1 - sum(x)
            grad_x = grad_a[:-1].copy()
            for j in range(n - 1):
                grad_x[j] -= grad_a[-1]
            # Also account for normalization (a = a_raw / sum(a_raw))
            # For simplicity, skip this (it's a small effect when sum is close to 1)
            return grad_x

        # L-BFGS optimization
        x0 = current[:-1].copy()
        bounds = [(0.0, 1.0)] * (n - 1)

        result = minimize(smooth_obj, x0, method='L-BFGS-B',
                          jac=smooth_grad, bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8,
                                   'maxls': 50})

        if result.success or result.fun < smooth_obj(x0):
            a_new = np.empty(n)
            a_new[:-1] = result.x
            a_new[-1] = 1.0 - result.x.sum()
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3:
            # Perturb and continue
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            f_pert = best_f * np.exp(rng.standard_normal(n) * 0.01)
            f_pert = f_pert / f_pert.sum()
            best_f = f_pert

    return best_f, best_obj


def nesterov_subgradient(f0, time_limit):
    """
    Nesterov-accelerated subgradient method for minimizing max(conv(a,a)) on the simplex.
    Key: uses Nesterov's momentum for O(1/T^2) convergence rate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    # Nesterov state
    x = f.copy()  # current point
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    # Step size: eta_t = diameter / (L * sqrt(2*t))
    # Diameter of simplex ~ sqrt(2), Lipschitz constant L ~ 4
    D = np.sqrt(2.0)
    L = 4.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(y)
        M = conv.max()

        # Subgradient at y: gradient of the active max constraint
        k_star = int(np.argmax(conv))
        # grad_j = 2 * a_{k_star - j}
        grad = np.zeros(n)
        for j in range(max(0, k_star - n + 1), min(n, k_star + 1)):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * y[idx]

        # Step size with Nesterov decay
        eta = D / (L * np.sqrt(2.0 * it))
        # Cap step size for stability
        eta = min(eta, 0.05)

        # Nesterov update
        beta_t = 2.0 / (it + 3.0)
        z = y + ((beta_t - 1.0) / (2.0 / (it + 2.0) if it > 1 else 1.0)) * (y - x) if it > 1 else y
        # Simpler momentum form
        v = 0.9 * v - eta * grad
        x_new = x + v
        x_new = project_simplex(x_new)

        # Nesterov lookahead
        gamma = 2.0 / (it + 3.0)
        y_new = (1.0 - gamma) * x_new + gamma * y
        y_new = project_simplex(y_new)

        x = x_new
        y = y_new

        conv = fast_convolve(y)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Also check x (sometimes it's better)
        conv_x = fast_convolve(x)
        if conv_x.max() < best_obj:
            best_obj = conv_x.max()
            best_f = x.copy()
            no_improve = 0

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            y = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            x = y.copy()
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_sweeps=50):
    """
    Coordinate descent with binary search for optimal value of each coordinate.
    For each coordinate j, find the best value t for a_j that minimizes max(conv).
    Uses binary search over t ∈ [0, 1] (since objective is scale-invariant,
    we work with the ratio max_conv / sum^2).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max() / (f.sum() ** 2)
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    def eval_coord(j, t):
        """Evaluate objective when a[j] = t (no renormalization, scale-invariant)."""
        a = f.copy()
        a[j] = t
        s = a.sum()
        if s < 1e-15:
            return 1e10
        conv = fast_convolve(a)
        return 2.0 * n * conv.max() / (s * s)

    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < n_sweeps:
        if time.monotonic() - t0 > time_limit:
            break
        sweep += 1

        # Use random order to avoid bias
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Binary search for optimal a[j]
            # The function is unimodal in a_j (for fixed other coordinates)
            lo = max(0.0, f[j] * 0.1)
            hi = min(1.0, f[j] * 10.0 + 0.01)

            # Try a few initial points to bracket the minimum
            candidates = [f[j] * 0.5, f[j] * 0.75, f[j], f[j] * 1.25, f[j] * 1.5,
                          max(0, f[j] - 0.01), min(1.0, f[j] + 0.01)]
            best_t = f[j]
            best_val = eval_coord(j, f[j])
            for ct in candidates:
                ct = max(0.0, min(1.0, ct))
                if ct == f[j]:
                    continue
                val = eval_coord(j, ct)
                if val < best_val:
                    best_val = val
                    best_t = ct

            # Refine with binary search around best_t
            if best_t > 0 and best_t < 1:
                span = max(0.005, abs(best_t - f[j]) * 0.5)
                lo = max(0.0, best_t - span)
                hi = min(1.0, best_t + span)
                for _ in range(12):
                    mid1 = lo + (hi - lo) / 3.0
                    mid2 = hi - (hi - lo) / 3.0
                    v1 = eval_coord(j, mid1)
                    v2 = eval_coord(j, mid2)
                    if v1 < v2:
                        hi = mid2
                    else:
                        lo = mid1
                    mid = (lo + hi) / 2.0
                    vm = eval_coord(j, mid)
                    if vm < best_val:
                        best_val = vm
                        best_t = mid

            if best_val < best_obj - 1e-15:
                f[j] = best_t
                # Renormalize
                s = f.sum()
                if s > 1e-15:
                    f = f / s
                    best_obj = 2.0 * n * fast_convolve(f).max()
                    best_f = f.copy()

    return best_f, best_obj / (f.sum() ** 2) if f.sum() > 0 else f.copy(), best_obj


def pairwise_swap_optimize(f0, time_limit):
    """
    Local search via pairwise mass transfers.
    For each pair (i,j), find the optimal amount of mass to transfer.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        # Pick random pairs
        pairs_per_iter = min(n * 2, 200)
        improved = False

        for _ in range(pairs_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try transferring mass from i to j (and vice versa)
            # Parameterize: a_i' = a_i - δ, a_j' = a_j + δ
            # Find δ that minimizes max(conv)
            # Use ternary search on δ ∈ [-a_i, a_j]
            lo = -f[i] * 0.8
            hi = f[j] * 0.8

            # Quick scan
            deltas = [0.0, lo * 0.25, lo * 0.5, hi * 0.25, hi * 0.5,
                      f[i] * 0.1, -f[i] * 0.1, f[j] * 0.1, -f[j] * 0.1]
            best_d = 0.0
            best_val = best_obj

            for d in deltas:
                d = max(-f[i] * 0.9, min(f[j] * 0.9, d))
                a_try = f.copy()
                a_try[i] -= d
                a_try[j] += d
                if a_try[i] < -1e-12 or a_try[j] < -1e-12:
                    continue
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                conv = fast_convolve(a_try / st)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_d = d

            if best_d != 0.0 and best_val < best_obj - 1e-15:
                f[i] -= best_d
                f[j] += best_d
                f = np.maximum(f, 0.0)
                st = f.sum()
                if st > 1e-15:
                    f = f / st
                    best_obj = 2.0 * n * fast_convolve(f).max()
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Restart with perturbation
                rng2 = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
                f = best_f * np.exp(rng2.standard_normal(n) * 0.008)
                f = f / f.sum()
                no_improve = 0
        else:
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted (battledore-like shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine taper
    c = np.cos(np.pi * (x + 0.5) / (2 * n)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Sinusoidal modulation
    s = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits.append(s / s.sum())

    # Power law
    p = (x + 1) ** (-0.5) * ((n - x) + 1) ** (-0.5)
    inits.append(p / p.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining:
    Phase 1: FW-LP quick refinement
    Phase 2: Smoothed L-BFGS optimization
    Phase 3: Nesterov subgradient
    Phase 4: Coordinate descent
    Phase 5: Pairwise swap local search
    Phase 6: Final polish
    Explores multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_p = 384
        parent = np.ones(n_p) / n_p
        print(f"Using uniform parent, n={n_p}")

    n_p = len(parent)

    # === Phase 1: FW-LP Quick (~150s) ===
    print("\n=== Phase 1: FW-LP Quick ===")
    rem = deadline - time.monotonic()
    t1 = min(120, rem * 0.1)
    if t1 > 20:
        print(f"  FW-LP on parent (n={n_p}), t={t1:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
        try_seq(a_opt, "fw_parent")

    # Try alternative init briefly
    rem = deadline - time.monotonic()
    if rem > 50:
        inits = generate_inits(n_p, parent, rng)
        for fi in [1, 2, 3]:
            if time.monotonic() > deadline - 1300:
                break
            if fi < len(inits):
                t_alloc = min(30, deadline - time.monotonic() - 1250)
                if t_alloc < 10:
                    break
                a_opt, _ = fw_lp_optimize(inits[fi], time_limit=t_alloc)
                try_seq(a_opt, f"fw_init_{fi}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Smoothed L-BFGS (~350s) ===
    print("\n=== Phase 2: Smoothed L-BFGS ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t2 = min(350, rem * 0.25)
        if t2 > 50:
            print(f"  L-BFGS smooth, n={n_best}, t={t2:.0f}s")
            a_opt, _ = smooth_optimize(best_seq, time_limit=t2)
            try_seq(a_opt, "lbfgs_smooth")

            # Also try from a perturbed start
            rem = deadline - time.monotonic()
            if rem > 100:
                pert = best_seq * np.exp(rng.standard_normal(n_best) * 0.02)
                pert = pert / pert.sum()
                t2b = min(150, rem * 0.4)
                if t2b > 30:
                    a_opt2, _ = smooth_optimize(pert, time_limit=t2b)
                    try_seq(a_opt2, "lbfgs_pert")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Nesterov Subgradient (~400s) ===
    print("\n=== Phase 3: Nesterov Subgradient ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t3 = min(400, rem * 0.25)
        if t3 > 50:
            print(f"  Nesterov SG, n={n_best}, t={t3:.0f}s")
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=t3)
            try_seq(a_opt, "nesterov_sg")

            # Second run with perturbation
            rem = deadline - time.monotonic()
            if rem > 100:
                pert = best_seq * np.exp(rng.standard_normal(n_best) * 0.01)
                pert = pert / pert.sum()
                t3b = min(200, rem * 0.4)
                if t3b > 30:
                    a_opt2, _ = nesterov_subgradient(pert, time_limit=t3b)
                    try_seq(a_opt2, "nesterov_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Coordinate Descent (~300s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(300, rem * 0.2)
        if t4 > 30:
            print(f"  Coord descent, n={n_best}, t={t4:.0f}s")
            a_opt, _, _ = coordinate_descent(best_seq, time_limit=t4, n_sweeps=100)
            try_seq(a_opt, "coord_desc")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Pairwise Swap (~250s) ===
    print("\n=== Phase 5: Pairwise Swap ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t5 = min(250, rem * 0.2)
        if t5 > 30:
            print(f"  Pairwise swap, n={n_best}, t={t5:.0f}s")
            a_opt, _ = pairwise_swap_optimize(best_seq, time_limit=t5)
            try_seq(a_opt, "pairwise")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Try different lengths ===
    print("\n=== Phase 6: Multi-length exploration ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        target_scores = {384: best_score, 512: 1.6, 448: 1.6}
        for n_try in [512, 448, 256]:
            if time.monotonic() > deadline - 100:
                break
            # Interpolate parent to new length
            if parent is not None:
                px = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, len(parent)), parent)
                px = np.maximum(px, 0)
                ps = px.sum()
                if ps > 1e-10:
                    px = px / ps
                    if evaluate_sequence(px.tolist()) < target_scores.get(n_try, 1.6):
                        t_alloc = min(100, deadline - time.monotonic() - 80)
                        if t_alloc > 20:
                            print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
                            a_opt, _ = fw_lp_optimize(px, time_limit=t_alloc)
                            try_seq(a_opt, f"fw_n{n_try}")
            # Also try from current best
            if best_seq is not None:
                bx = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, len(best_seq)), best_seq)
                bx = np.maximum(bx, 0)
                bs = bx.sum()
                if bs > 1e-10:
                    bx = bx / bs
                    t_alloc = min(80, deadline - time.monotonic() - 60)
                    if t_alloc > 15:
                        a_opt, _ = fw_lp_optimize(bx, time_limit=t_alloc)
                        try_seq(a_opt, f"fw_best_n{n_try}")

    print(f"  Phase 6 best: {best_score:.8f}")

    # === Phase 7: Final Polish ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 30:
            # Quick coordinate descent
            t_alloc = min(rem * 0.5, 120)
            if t_alloc > 15:
                a_opt, _, _ = coordinate_descent(best_seq, time_limit=t_alloc, n_sweeps=50)
                try_seq(a_opt, "polish_cd")

            # Quick pairwise
            rem = deadline - time.monotonic()
            if rem > 20:
                t_alloc = min(rem * 0.5, 80)
                if t_alloc > 10:
                    a_opt, _ = pairwise_swap_optimize(best_seq, time_limit=t_alloc)
                    try_seq(a_opt, "polish_ps")

            # Tiny perturbations
            rem = deadline - time.monotonic()
            if rem > 20:
                for pi in range(3):
                    if time.monotonic() > deadline - 10:
                        break
                    noise = rng.standard_normal(n_best) * 0.002
                    f_pert = project_simplex(best_seq + noise)
                    t_alloc = min(20, (deadline - time.monotonic() - 10) / (3 - pi))
                    if t_alloc < 5:
                        break
                    a_opt, _ = nesterov_subgradient(f_pert, time_limit=t_alloc)
                    try_seq(a_opt, f"polish_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
