def optimize_softmax(a_init, tau_schedule, max_iters=200, rng=None):
    n = len(a_init)
    # Initialize x such that softmax(x) ≈ a_init
    a_norm = a_init / np.sum(a_init)
    x = np.log(a_norm + 1e-30)  # inverse softmax
    
    best_score = local_score(a_init)
    best_a = a_init.copy()
    
    for tau in tau_schedule:
        def objective_and_grad(x):
            a = softmax(x)
            obj, grad_a = compute_conv_and_grad(a, tau)
            grad_x = a * (grad_a - np.dot(a, grad_a))
            return obj, grad_x
        
        result = minimize(objective_and_grad, x, method='L-BFGS-B', 
                         jac=True, options={'maxiter': max_iters, 'gtol': 1e-8})
        a = softmax(result.x)
        a = np.clip(a, 1e-10, None)
        a = a / np.sum(a)  # renormalize
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

"""
Optimization of non-negative step functions to minimize peak autocorrelation ratio.
Uses softmax parameterization with L-BFGS-B, LP-based refinement, and local search.
Multi-scale approach: optimize at increasing sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

# ============ Helper Functions ============

def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_stable(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

def compute_softmax_obj_grad(a, tau):
    """Compute softened max objective and gradient w.r.t. a.
    
    Returns (objective, gradient_wrt_a)
    """
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp  # softmax weights over conv positions
    
    obj = c_max + np.log(sum_exp) / tau
    
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # Using FFT: g[i] = (a * w_rev)[2n-2-i]
    v = w[::-1]  # reversed, length 2n-1
    conv_aw = fft_convolve(a, v)  # length n + 2n-2 = 3n-2
    # g[i] = conv_aw[2n-2-i] for i=0..n-1
    g = conv_aw[n-1 : 2*n-1][::-1]
    
    grad_a = 2.0 * g
    return obj, grad_a

def softmax_lbfgs(a_init, tau_schedule, max_iters_per_tau=100, time_limit=60.0):
    """Optimize using softmax parameterization + L-BFGS-B with tau annealing."""
    t0 = time.monotonic()
    n = len(a_init)
    a_norm = a_init.astype(np.float64).copy()
    a_norm = np.clip(a_norm, 1e-10, None)
    a_norm = a_norm / np.sum(a_norm)
    x = np.log(a_norm)  # inverse softmax
    
    best_score = local_score(a_init)
    best_a = a_init.copy()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_and_grad(x):
            a = softmax_stable(x)
            obj, grad_a = compute_softmax_obj_grad(a, tau)
            # Chain rule: dF/dx = a * (grad_a - dot(a, grad_a))
            grad_x = a * (grad_a - np.dot(a, grad_a))
            return obj, grad_x
        
        result = minimize(obj_and_grad, x, method='L-BFGS-B', jac=True,
                         options={'maxiter': max_iters_per_tau, 'gtol': 1e-10})
        
        a_new = softmax_stable(result.x)
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / np.sum(a_new)
        
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Update x for next tau
        x = np.log(np.clip(a_new, 1e-30, None))
    
    return best_a, best_score

def build_constraint_matrix(f):
    """Build (2n-1) x n constraint matrix for LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_direction(f, M):
    """Solve LP to find best descent direction."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except:
        pass
    return np.ones(n) / n

"""
Multi-phase optimization for minimizing peak autocorrelation ratio of non-negative step functions.
Combines: (1) softmax-parameterized L-BFGS-B with soft-max smoothing and tau annealing,
(2) LP-based directional refinement, (3) simulated annealing local search,
(4) multi-scale approach (optimize at n=512, 768, 1024).
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

# ============ Helper Functions ============

def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score (fast local version, no clamping to 1000)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_stable(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    return e / s

def compute_softmax_obj_grad(a, tau):
    """Compute softened-max objective and its gradient w.r.t. a.
    
    The softened max is: c_max + (1/tau)*log(sum_k exp(tau*(c[k]-c_max)))
    Gradient: grad_a[i] = 2 * sum_j a[j] * w[i+j] where w = softmax(tau*(c-c_max))
    """
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    
    obj = c_max + np.log(sum_exp) / tau
    
    # Gradient via FFT cross-correlation trick
    # g[i] = sum_j a[j] * w[i+j] = (a * w_rev)[2n-2-i]
    v = w[::-1]  # reversed w, length 2n-1
    conv_aw = fft_convolve(a, v)  # length 3n-2
    g = conv_aw[n - 1 : 2 * n - 1][::-1]
    
    grad_a = 2.0 * g
    return obj, grad_a

def softmax_lbfgs(a_init, tau_schedule, max_iters_per_tau=80, time_limit=120.0):
    """Optimize using softmax parameterization + L-BFGS-B with tau annealing.
    
    Returns (best_a, best_score).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a_norm = np.clip(a_init.astype(np.float64), 1e-10, None)
    a_norm = a_norm / np.sum(a_norm)
    x = np.log(a_norm)
    
    best_score = local_score(a_init)
    best_a = a_init.copy()
    
    for ti, tau in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_and_grad(xv):
            a = softmax_stable(xv)
            obj, grad_a = compute_softmax_obj_grad(a, tau)
            # Chain rule through softmax: dF/dx = a * (grad_a - <a, grad_a>)
            dot_ak = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot_ak)
            return obj, grad_x
        
        try:
            result = minimize(obj_and_grad, x, method='L-BFGS-B', jac=True,
                             options={'maxiter': max_iters_per_tau, 'gtol': 1e-12,
                                      'ftol': 1e-12, 'maxls': 50})
            a_new = softmax_stable(result.x)
        except Exception:
            a_new = softmax_stable(x)
        
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / np.sum(a_new)
        score = local_score(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Update x for next tau level
        x = np.log(np.clip(a_new, 1e-30, None))
        
        if ti % 3 == 0:
            print(f"    tau={tau:.0f}, score={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score

def build_constraint_matrix(f):
    """Build (2n-1) x n matrix A where row k, col j = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n

def lp_refine(f0, time_limit, rng, verbose=False):
    """LP-based iterative refinement with adaptive step size."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = np.clip(f0.astype(np.float64), 0, None)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        f = np.ones(n) * (S / n)
    else:
        f = f * (S / sum_f)
    
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.1
    stall_count = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        b_0 = solve_lp_direction(f, M)
        iter_count += 1
        
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = (S / sum_b) * b_0
        
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1, t * 0.05]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            new_score = local_score(f_new)
            if new_score < current_score:
                f = f_new
                current_score = new_score
                t = min(trial_t * 1.1, 0.3)
                improved = True
                stall_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break
        
        if not improved:
            stall_count += 1
            t *= 0.85
            if stall_count > 40:
                # Escape local minimum with perturbation
                sigma = 0.02 + 0.01 * min(stall_count / 100.0, 2.0)
                f = f * np.exp(rng.normal(0, sigma, n))
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * (S / sf)
                    current_score = local_score(f)
                stall_count = 0
                t = 0.08
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score

def local_search_polish(a_init, time_limit, rng):
    """Simulated annealing local search with targeted peak reduction."""
    current = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99985
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.30:
            # Targeted peak reduction: reduce contributors to peak, boost others
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 3:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.012, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.60:
            # Multi-block small
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.75:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                total = current[i] + current[j]
                frac = rng.beta(2, 2)
                candidate[i] = total * frac
                candidate[j] = total * (1 - frac)
        elif r < 0.85:
            # Triple redistribution
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            total = current[i] + current[j] + current[k]
            fracs = rng.dirichlet(np.ones(3) * 2.0)
            candidate[idx] = total * fracs
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.007, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score

def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining softmax+L-BFGS-B (with tau annealing),
    LP-based directional refinement, and simulated annealing local search.
    Uses multi-scale approach: optimize at n=512, then upsample to n=768, n=1024.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # --- Get initial sequence ---
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    
    if best_seq is None:
        n0 = 512
        a = make_chirp_init(n0, 0.5, rng)
        s = local_score(a)
        print(f"Generated chirp init: n={n0}, score={s:.6f}")
        best_score = s
        best_seq = a.copy()
    
    # ---- Phase 1: Softmax + L-BFGS-B at original length ----
    n_curr = len(best_seq)
    remaining = deadline - time.monotonic()
    p1_time = min(150.0, remaining * 0.12)
    print(f"\n=== Phase 1: Softmax+LBFGS (n={n_curr}, budget={p1_time:.0f}s) ===")
    if p1_time > 20:
        tau_schedule = [1.0, 5.0, 15.0, 50.0, 150.0, 500.0, 1500.0, 5000.0]
        a_opt, s_opt = softmax_lbfgs(best_seq, tau_schedule, max_iters_per_tau=60, time_limit=p1_time)
        print(f"  Result: score={s_opt:.6f} (was {best_score:.6f})")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ---- Phase 2: LP refinement ----
    remaining = deadline - time.monotonic()
    p2_time = min(300.0, remaining * 0.22)
    print(f"\n=== Phase 2: LP refinement (n={len(best_seq)}, budget={p2_time:.0f}s) ===")
    if p2_time > 30:
        f_opt, s_opt = lp_refine(best_seq, p2_time, rng, verbose=True)
        print(f"  Result: score={s_opt:.6f} (was {best_score:.6f})")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = f_opt.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ---- Phase 3: Local search polish ----
    remaining = deadline - time.monotonic()
    p3_time = min(200.0, remaining * 0.15)
    print(f"\n=== Phase 3: Local search (n={len(best_seq)}, budget={p3_time:.0f}s) ===")
    if p3_time > 30:
        a_opt, s_opt = local_search_polish(best_seq, p3_time, rng)
        print(f"  Result: score={s_opt:.6f} (was {best_score:.6f})")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ---- Phase 4: Upsample and optimize at larger lengths ----
    for target_n in [768, 1024]:
        remaining = deadline - time.monotonic()
        if remaining < 400:
            print(f"\nSkipping n={target_n} (insufficient time: {remaining:.0f}s)")
            break
        
        print(f"\n=== Phase 4: Upsample to n={target_n} ===")
        a_large = resample_sequence(best_seq, target_n)
        
        # Softmax + LBFGS
        t4a = min(100.0, remaining * 0.25)
        if t4a > 20:
            print(f"  Softmax+LBFGS (budget={t4a:.0f}s)")
            tau_sched = [1.0, 10.0, 50.0, 200.0, 800.0]
            a_opt, s_opt = softmax_lbfgs(a_large, tau_sched, max_iters_per_tau=50, time_limit=t4a)
            print(f"    score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
        
        # LP refinement
        remaining = deadline - time.monotonic()
        t4b = min(180.0, remaining * 0.30)
        if t4b > 30:
            print(f"  LP refinement (budget={t4b:.0f}s)")
            f_opt, s_opt = lp_refine(best_seq, t4b, rng, verbose=False)
            print(f"    score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = f_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # ---- Phase 5: Final refinement on the best sequence ----
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 5: Final refinement (n={len(best_seq)}, budget={remaining-20:.0f}s) ===")
        
        # LP
        remaining = deadline - time.monotonic()
        lp_time = min(remaining * 0.45, 250.0)
        if lp_time > 30:
            f_opt, s_opt = lp_refine(best_seq, lp_time, rng, verbose=True)
            print(f"  After LP: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = f_opt.copy()
        
        # Local search
        remaining = deadline - time.monotonic()
        ls_time = min(remaining * 0.6, 250.0)
        if ls_time > 30:
            a_opt, s_opt = local_search_polish(best_seq, ls_time, rng)
            print(f"  After LS: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
        
        # One more short LP pass
        remaining = deadline - time.monotonic()
        if remaining > 60:
            f_opt, s_opt = lp_refine(best_seq, remaining * 0.5, rng, verbose=False)
            print(f"  After final LP: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = f_opt.copy()
    
    # ---- Final verification and return ----
    print(f"\n{'='*50}")
    print(f"Best score (local): {best_score:.8f}, n={len(best_seq)}")
    
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"FINAL: evaluate_sequence score={final_score:.8f}, n={len(final_list)}")
        return final_list
    
    # Fallback
    return [1.0] * 512
