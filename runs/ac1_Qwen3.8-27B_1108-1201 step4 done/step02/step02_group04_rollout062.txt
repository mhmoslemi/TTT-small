    size = 2 * n - 1

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Mirror descent (multiplicative weights) with subgradient of max-conv
2. LP-based min-max direction finding (direct peak minimization)
3. Frequency-domain equalization (Gerchberg-Saxton style)
4. Support refinement via simulated annealing
5. Adaptive hybrid strategy with warm-start from previous solution
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_conv(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_subgradient(a, active):
    """Compute the (averaged) subgradient of max_k (a*a)[k].
    active: boolean array of length 2n-1 indicating which conv entries are active.
    Returns vector of length n."""
    n = len(a)
    n_active = active.sum()
    if n_active == 0:
        return np.zeros(n)
    a_rev = a[::-1]
    size = n + (2 * n - 1) - 1
    fft_size = 1 << (size - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Af = np.fft.rfft(active.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Ar * Af, fft_size)
    d = conv_full[n - 1: 2 * n - 1] / n_active
    return 2.0 * d


def mirror_descent_opt(f0, time_limit, eta0=0.5, active_frac=0.95, patience_max=80):
    """Mirror descent (multiplicative weights) with subgradient of max-conv."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        threshold = M * active_frac
        active = conv >= threshold
        if not active.any():
            active[np.argmax(conv)] = True

        d = compute_subgradient(a, active)
        d_centered = d - d.mean()

        a_new = a * np.exp(-eta * d_centered)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s

        conv_new = fast_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            patience = 0
        else:
            patience += 1
            eta *= 0.97
            if patience > patience_max:
                break

    return best_a, best_obj


def build_lp_matrix(n):
    """Build sparse structure for cross-convolution constraints."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    return rows, cols


def solve_lp_minmax(f, rows, cols, n, timeout=10):
    """Solve min t s.t. (f*g)[k] <= t for all k, g>=0, sum(g)=1.
    Returns (g, t) or (None, None) on failure."""
    vals = np.tile(f, n)
    n_constraints = 2 * n - 1
    A_cc = sparse.csc_matrix((vals, (rows, cols)), shape=(n_constraints, n))

    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: (f*g)[k] - t <= 0 for all k
    # A_ub = [A_cc, -I]
    I_neg = sparse.csc_matrix(-np.eye(n_constraints))
    A_ub = sparse.bmat([[A_cc, I_neg]], format='csc')
    b_ub = np.zeros(n_constraints)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        if g.sum() > 1e-12:
            g /= g.sum()
        return g, float(res.x[-1])
    return None, None


def lp_alt_optimize(f0, time_limit, patience_max=80):
    """LP-guided alternating optimization with min-max LP formulation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_conv(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = build_lp_matrix(n)

    t0 = time.monotonic()
    t_blend = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        lp_timeout = min(15.0, remaining * 0.1)
        g, t_lp = solve_lp_minmax(f, rows, cols, n, timeout=lp_timeout)
        if g is None:
            g = f.copy()
            if iter_count > 50:
                break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t_blend, 0.8 * t_blend, 0.5 * t_blend, 0.3 * t_blend, 0.1 * t_blend]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_conv(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t_blend = min(1.0, t_blend * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t_blend *= 0.85
            if patience > patience_max:
                break

        if iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={t_blend:.3f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def freq_equalize(f0, time_limit, n_iter_max=5000):
    """Frequency-domain equalization (Gerchberg-Saxton style).
    Iterates between flattening |A(f)|^2 in frequency and projecting to non-negative in time."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    patience = 0
    alpha = 0.5  # Blending factor for frequency modification

    for _ in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute DTFT
        A = np.fft.rfft(a)
        P = np.abs(A)**2
        P_mean = P.mean()

        if P.min() < 1e-30:
            A_modified = A
        else:
            # Flatten: A_new = A / sqrt(P/P_mean) blended with A
            scale = np.sqrt(P_mean / np.maximum(P, 1e-30))
            A_modified = A * (1 - alpha) + A * scale * alpha

        # Back to time domain
        a_new = np.fft.irfft(A_modified, n)
        a_new = np.maximum(a_new, 0.0)
        s2 = a_new.sum()
        if s2 < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s2

        conv_new = fast_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            alpha *= 0.99
            if patience > 200:
                break

    return best_a, best_obj


def support_refinement(f0, time_limit):
    """Try zeroing small entries to see if objective improves."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return a, float('inf')
    a /= s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    # Sort indices by value
    sorted_idx = np.argsort(a)

    # Try removing entries from smallest to largest
    for remove_frac in [0.1, 0.2, 0.3]:
        if time.monotonic() - t0 > time_limit:
            break
        n_remove = int(n * remove_frac)
        a_trial = best_a.copy()
        a_trial[sorted_idx[:n_remove]] = 0
        s_t = a_trial.sum()
        if s_t < 1e-12:
            continue
        a_trial /= s_t
        conv_t = fast_conv(a_trial)
        obj_t = 2.0 * n * conv_t.max()
        if obj_t < best_obj - 1e-10:
            best_obj = obj_t
            best_a = a_trial.copy()
            print(f"    Support refinement: removed {n_remove} entries, obj={obj_t:.6f}")

    # Random support perturbations
    for _ in range(100):
        if time.monotonic() - t0 > time_limit - 5:
            break
        a_trial = best_a.copy()
        # Randomly zero some entries
        n_zeros = rng.integers(1, n // 10 + 1)
        zero_idx = rng.choice(n, size=n_zeros, replace=False)
        a_trial[zero_idx] = 0
        # Randomly restore some zeroed entries
        if (a_trial == 0).any():
            zero_positions = np.where(a_trial == 0)[0]
            if len(zero_positions) > 0:
                restore_count = min(len(zero_positions), rng.integers(0, n_zeros + 1))
                if restore_count > 0:
                    restore_idx = rng.choice(zero_positions, size=restore_count, replace=False)
                    a_trial[restore_idx] = best_a[restore_idx] * 0.5

        s_t = a_trial.sum()
        if s_t < 1e-12:
            continue
        a_trial /= s_t
        conv_t = fast_conv(a_trial)
        obj_t = 2.0 * n * conv_t.max()
        if obj_t < best_obj - 1e-10:
            best_obj = obj_t
            best_a = a_trial.copy()

    return best_a, best_obj


def combined_optimize(f0, time_limit):
    """Combined optimization: alternate between mirror descent, LP, and freq equalization."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    n = len(a)

    t0 = time.monotonic()
    phase_duration = max(30, time_limit / 5.0)

    phases = [
        ('mirror', lambda fa, tl: mirror_descent_opt(fa, tl, eta0=0.5, active_frac=0.95)),
        ('lp', lambda fa, tl: lp_alt_optimize(fa, tl, patience_max=60)),

        ('freq', lambda fa, tl: freq_equalize(fa, tl, n_iter_max=3000)),
    ]

    for phase_name, phase_func in phases:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed - 20
        if remaining < 20:
            break
        alloc = min(phase_duration, remaining)
        if alloc < 10:
            break

        print(f"  Phase '{phase_name}', allocating {alloc:.0f}s")
        a_new, obj_new = phase_func(a, alloc)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"    Improved: {best_obj:.8f}")
        else:
            a = best_a.copy()

    # Final support refinement
    remaining = time_limit - (time.monotonic() - t0) - 10
    if remaining > 15:
        print(f"  Phase 'support', allocating {remaining:.0f}s")
        a_new, obj_new = support_refinement(a, remaining * 0.5)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new

    # Final polish

    if remaining > 20:
        a_new, obj_new = mirror_descent_opt(best_a, remaining, eta0=0.1, active_frac=0.92, patience_max=50)
        if obj_new < best_obj - 1e-12:

            best_a = a_new.copy()

    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-6)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # Periodic patterns
    for period in [2, 3, 4, 5, 8]:
        p = np.ones(n)
        p[period - 1::period] = 0.1
        p /= p.sum()
        inits[f'per{period}'] = p

    # Two-peak
    bi = np.zeros(n)
    w = max(1, n // 8)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    if (n // 2 - w) > 0:
        bi[n // 2 - w:n // 2 + w] = 0.5
    bi /= np.maximum(bi.sum(), 1e-12)
    inits['twopeak'] = bi

    # Cosine window
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb = np.maximum(cb, 0.01)
    cb /= cb.sum()
    inits['cos'] = cb

    # Random Dirichlet
    rr = rng.dirichlet(np.ones(n) * 0.2)
    inits['rnd1'] = rr
    rr2 = rng.dirichlet(np.ones(n) * 2.0)
    inits['rnd2'] = rr2

    # Sawtooth
    saw = (x % (n // 4)) / (n // 4)
    saw = np.maximum(saw, 0.05)
    saw /= saw.sum()
    inits['saw'] = saw

    return inits

    print(f"Total remaining: {remaining:.0f}s")

    for n in lengths:
        if time.monotonic() > deadline - 30:
            break

        n_time = time_allocs[n]
        if n_time < 60:
            continue

        print(f"\n{'='*50}")
        print(f"Optimizing length n={n}, time budget={n_time:.0f}s")
        print(f"{'='*50}")

        # Generate initializations
        inits = gen_inits(n, rng)

        # Add parent as initialization
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
            pp_sum = pp.sum()
            if pp_sum > 1e-10:
                inits['parent'] = pp / pp_sum

        # Determine how many inits to try based on time
        n_init_limit = max(3, int(n_time / 60))
        init_items = list(inits.items())

        # Prioritize: parent first, then diverse inits
        if 'parent' in inits:
            init_items = [('parent', inits['parent'])] + [x for x in init_items if x[0] != 'parent']

        init_items = init_items[:min(n_init_limit, len(init_items))]

        for name, a0 in init_items:
            if time.monotonic() > deadline - 20:
                break

            alloc = max(30, (deadline - time.monotonic() - 20) / max(1, len(init_items)))
            if alloc < 20:
                break

            print(f"\n  Init '{name}', time={alloc:.0f}s")
            a_opt, obj_opt = combined_optimize(a0, alloc)

            a_list = a_opt.tolist()
            score = float(evaluate_sequence(a_list))

            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  ** NEW BEST: n={n}, init={name}, score={score:.6f}")
                else:
                    print(f"  n={n}, init={name}, score={score:.6f} (best={best_score:.6f})")

    # Final deep polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        print(f"\n{'='*50}")
        print(f"Final deep polish: {remaining:.0f}s remaining")
        print(f"{'='*50}")

        n_best = len(best_seq)
        a_best = best_seq / best_seq.sum()

        # Multiple polishing rounds
        for round_idx in range(3):
            if time.monotonic() > deadline - 15:
                break
            alloc = remaining / (3 - round_idx)

            # Mirror descent with different params
            a_new, obj_new = mirror_descent_opt(a_best, alloc * 0.4,
                                                 eta0=0.2 * (0.7 ** round_idx),
                                                 active_frac=0.93 - 0.02 * round_idx,
                                                 patience_max=60)
            sc = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
                a_best = a_new / a_new.sum()
                print(f"  Polish round {round_idx} (mirror): score={sc:.6f}")

            if time.monotonic() > deadline - 15:
                break

            # LP optimization
            alloc2 = min(alloc * 0.4, deadline - time.monotonic() - 15)
            if alloc2 > 10:
                a_new2, obj_new2 = lp_alt_optimize(a_best, alloc2, patience_max=80)
                sc2 = float(evaluate_sequence(a_new2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_new2.copy()
                    a_best = a_new2 / a_new2.sum()
                    print(f"  Polish round {round_idx} (LP): score={sc2:.6f}")

            if time.monotonic() > deadline - 15:
                break

            # Frequency equalization
            alloc3 = min(alloc * 0.2, deadline - time.monotonic() - 15)
            if alloc3 > 5:
                a_new3, obj_new3 = freq_equalize(a_best, alloc3, n_iter_max=2000)
                sc3 = float(evaluate_sequence(a_new3.tolist()))
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = a_new3.copy()
                    a_best = a_new3 / a_new3.sum()
                    print(f"  Polish round {round_idx} (freq): score={sc3:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL RESULT: n={len(best_seq)}, score={best_score:.6f}")
    print(f"{'='*50}")

    return [float(max(0.0, x)) for x in best_seq]
