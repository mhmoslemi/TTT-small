Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-based subgradient optimization,
L-BFGS refinement, and coordinate descent polishing. Explores multiple sequence
lengths and diverse initializations to escape local minima.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Objective value assuming sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * float(fast_convolve(a).max())


def compute_subgrad(a):
    """
    Subgradient of max_k c[k] at point a.
    Returns the subgradient vector (before multiplying by 2n).
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    # Get positions achieving (near) max
    tol = 1e-8 * max_c if max_c > 0 else 1e-12
    max_pos = np.where(c >= max_c - tol)[0]
    
    grad = np.zeros(n)
    for k in max_pos:
        # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        j = np.arange(n)
        ki = k - j
        valid = (ki >= 0) & (ki < n)
        grad[valid] += 2.0 * a[ki[valid]]
    
    grad /= len(max_pos)
    return grad


def adam_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer with subgradient for the normalized objective.
    Works on the probability simplex (sum=1, a>=0).
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad = compute_subgrad(a)
        grad = 2.0 * n * grad  # Full subgradient of objective
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate schedule
        lr_t = lr * (1.0 / (1.0 + 0.01 * t))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-12:
            break
        a_new /= ns
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Random restart if stuck
        if no_improve > 2000:
            noise = np.random.default_rng(t).dirichlet(np.ones(n) * 0.5)
            a = 0.7 * a + 0.3 * noise
            a /= a.sum()
            no_improve = 0
    
    return best_a, best_obj


def lbfgs_optimize(a0, time_limit):

    a = np.maximum(a0.astype(np.float64), 0.0).copy()

            return 1e10, np.zeros_like(x)
        x = x / xs
        f = obj_value(x)
        g = compute_subgrad(x) * 2.0 * n
        # Adjust gradient for the normalization constraint
        # d/dx [obj(x/sum(x))] = (I - x*1^T/sum(x)) @ grad_obj
        # Simplified: just use the raw subgradient
        return f, g
    
    t0 = time.monotonic()

    try:
        result = scipy.optimize.minimize(
            fun_and_grad, a, method='L-BFGS-B', jac=True,
            bounds=[(0, None)] * n,
            options={'maxiter': max_iter, 'maxfun': 100000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        if result.success or result.fun < best_obj:
            a_new = np.maximum(result.x, 0.0)
            a_new /= a_new.sum()
            o = obj_value(a_new)
            if o < best_obj:
                best_obj = o
                best_a = a_new.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: for each coordinate, find the best value
    that minimizes the objective (given others fixed).
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = n * 3
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        improved = False
        for j in range(n):
            # Try different values for a[j]
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try redistributing a[j] to other coordinates
            # and also just scaling
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5, 0.0]
            
            for new_val in candidates:
                if new_val < 0:
                    continue
                a_trial = a.copy()
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue
                # Distribute delta to other coordinates proportionally
                other_sum = a.sum() - old_val
                if other_sum < 1e-12:
                    continue
                a_trial[j] = new_val
                for i in range(n):
                    if i != j:
                        a_trial[i] -= delta * a[i] / other_sum
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = obj_value(a_trial)
                if o < best_obj - 1e-10:
                    best_obj = o
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += n
        else:
            no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam subgradient optimization,
L-BFGS-B refinement, and coordinate descent polishing. Explores multiple
sequence lengths (up to 2048) and diverse initializations. Uses subgradient
of the max auto-correlation peak with momentum-based Adam updates, projected
onto the probability simplex.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(fast_convolve(a).max())


def compute_subgrad(a):
    """
    Subgradient of max_k c[k] at point a.
    dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    Returns averaged subgradient over max-achieving positions.
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    if max_c < 1e-15:
        return np.zeros(n)
    tol = max(1e-10, 1e-8 * max_c)
    max_pos = np.where(c >= max_c - tol)[0]

    grad = np.zeros(n)
    for k in max_pos:
        ki = k - np.arange(n)
        valid = (ki >= 0) & (ki < n)
        grad[valid] += 2.0 * a[ki[valid]]

    grad /= float(len(max_pos))
    return grad


def adam_optimize(a0, time_limit, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, restart_every=3000):
    """
    Adam optimizer with subgradient on the probability simplex.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    rng = np.random.default_rng(12345)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad = compute_subgrad(a)
        grad = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr * (0.5 ** (t / 5000.0))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-12:
            break
        a_new /= ns

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if t % restart_every == 0:
            # Check if stuck: compare current to best
            cur_obj = obj_value(a)
            if cur_obj > best_obj * 1.01:
                # Perturb
                noise = rng.dirichlet(np.ones(n) * 0.5)
                a = 0.6 * a + 0.4 * noise
                a /= a.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

    return best_a, best_obj


def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def projected_subgrad_optimize(a0, time_limit, lr=0.01, momentum=0.9):
    """Heavy-ball momentum with projection onto simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    vel = np.zeros(n)
    t = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad = compute_subgrad(a) * 2.0 * n
        lr_t = lr * (1.0 / (1.0 + 0.005 * t))

        vel = momentum * vel - lr_t * grad
        a_new = proj_simplex(a + vel, 1.0)
        a_new /= a_new.sum() if a_new.sum() > 1e-12 else 1.0

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def lbfgs_optimize(a0, time_limit):
    """L-BFGS-B with subgradient (relaxation: use mean of top few peaks)."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    def fun_and_grad(x):
        x = np.maximum(x, 0.0)
        xs = x.sum()
        if xs < 1e-12:
            return 1e10, np.zeros_like(x)
        x = x / xs
        c = fast_convolve(x)
        max_c = c.max()
        # Use top 3 peaks for smoother gradient
        tol = 0.05 * max_c
        top_pos = np.where(c >= max_c - tol)[0]
        if len(top_pos) == 0:
            top_pos = [int(np.argmax(c))]

        grad = np.zeros(n)
        for k in top_pos:
            ki = k - np.arange(n)
            valid = (ki >= 0) & (ki < n)
            grad[valid] += 2.0 * x[ki[valid]]
        grad /= float(len(top_pos))
        grad = 2.0 * n * grad
        # Center the gradient (account for normalization)
        grad = grad - grad.mean()
        return 2.0 * n * max_c, grad

    try:
        result = scipy.optimize.minimize(
            fun_and_grad, a, method='L-BFGS-B', jac=True,
            bounds=[(0, None)] * n,
            options={'maxiter': 2000, 'maxfun': 50000, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        a_new = np.maximum(result.x, 0.0)
        ns = a_new.sum()
        if ns > 1e-12:
            a_new /= ns
            o = obj_value(a_new)
            if o < best_obj:
                best_obj = o
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_obj


def coordinate_descent(a0, time_limit, frac_list=None):
    """
    Coordinate descent: try changing one element at a time,
    redistributing mass proportionally to others.
    """
    if frac_list is None:
        frac_list = [0.3, 0.5, 0.7, 1.3, 1.7, 2.5, 0.0]

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 3 * n

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        improved = False
        # Process coordinates in order of largest value first (more impactful)
        order = np.argsort(a)[::-1]
        for j in order:
            old_val = a[j]
            if old_val < 1e-14:
                continue
            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue

            for frac in frac_list:
                new_val = old_val * frac
                if new_val < 0:
                    continue
                if abs(new_val - old_val) < 1e-14:
                    continue
                a_trial = a.copy()
                delta = new_val - old_val
                a_trial[j] = new_val
                a_trial[np.arange(n) != j] -= delta * a[np.arange(n) != j] / other_sum
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = obj_value(a_trial)
                if o < best_obj - 1e-10:
                    best_obj = o
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    old_val = new_val
                    other_sum = 1.0 - old_val
                    improved = True
                    break

        if not improved:
            stuck += n
        else:
            stuck = 0

    return best_a, best_obj


def get_parent(n):

def make_initializations(n, rng):
    """Generate diverse initializations for sequence length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        g /= g.sum()
        inits['gauss_%s' % sf] = g

    # Edge-concentrated
    for sf in [0.2, 0.35]:
        d = np.minimum(x, n - 1.0 - x) / (sf * n)
        g = np.exp(-d * d)
        g = np.maximum(g, 0)
        g /= g.sum()
        inits['edge_%s' % sf] = g

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.2)

    # Sparse random
    num_nz = max(2, n // 4)
    idx = rng.choice(n, size=num_nz, replace=False)
    sparse = np.zeros(n)
    sparse[idx] = rng.dirichlet(np.ones(num_nz) * 0.5)
    inits['sparse'] = sparse

    # Sinusoidal
    for freq in [1.0, 2.0, 3.0, 5.0]:
        sig = np.abs(np.sin(np.pi * freq * x / n)) + 0.05
        inits['sin_%s' % freq] = sig / sig.sum()

    # Arithmetic progression (ramp)
    ramp = np.linspace(0.1, 1.0, n)
    inits['ramp'] = ramp / ramp.sum()

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        # Noisy parent
        noisy = parent + 0.05 * rng.standard_normal(n)
        noisy = np.maximum(noisy, 0)
        noisy /= noisy.sum()
        inits['parent_noisy'] = noisy

    return inits
