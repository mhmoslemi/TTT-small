Uses L-BFGS-B with adaptive soft-max smoothing, diverse parametric initializations,

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def smooth_obj(a, tau):
    """Smooth approximation using log-sum-exp."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf'), np.zeros_like(a)
    
    # Soft max: (1/tau) * log(sum(exp(tau * c)))
    c_max = c.max()
    soft_max = c_max + np.log(np.sum(np.exp(tau * (c - c_max)))) / tau
    
    obj = 2.0 * n * soft_max / (S * S)
    
    # Gradient
    # d(soft_max)/da[i] = sum_m (exp(tau*c[m]) / sum(exp(tau*c[m]))) * dc[m]/da[i]
    w = np.exp(tau * (c - c_max))
    w /= w.sum()
    
    # dc[m]/da[i] = a[m-i] if 0 <= m-i < n (for the "other" index)
    # Actually: c[m] = sum_j a[j]*a[m-j], dc[m]/da[i] = a[m-i] + a[m-i] = 2*a[m-i] (if valid and m-i != i)
    # More carefully: dc[m]/da[i] = sum_{j: m-j=i or j=i} a[other]
    # = a[m-i] (if 0<=m-i<n) counted once for each occurrence
    # c[m] = sum_{j=max(0,m-n+1)}^{min(n-1,m)} a[j]*a[m-j]
    # dc[m]/da[i] = a[m-i] if 0<=m-i<n AND i is in valid range, + a[m-i] if m-i is in valid range and i is the "other"
    # Actually: dc[m]/da[i] = a[m-i] * (number of times i appears in the sum for c[m])
    # i appears as "j" when j=i, giving a[i]*a[m-i]
    # i appears as "m-j" when m-j=i, i.e., j=m-i, giving a[m-i]*a[i]
    # So dc[m]/da[i] = a[m-i] + a[m-i] = 2*a[m-i] if 0<=m-i<n and i != m-i
    #               = a[i] if i = m-i (i.e., 2i=m) and valid
    #               = 0 otherwise
    
    # Simpler: dc[m]/da[i] = 2*a[m-i] if 0<=m-i<n and 2*i != m, else a[m-i] if 2*i==m and 0<=m-i<n, else 0
    # = a[m-i] * (2 - [2i==m]) if 0<=m-i<n, else 0
    
    # For the gradient, we need: grad[i] = (1/S²) * 2n * sum_m w[m] * dc[m]/da[i] - (2n * soft_max / S³) * 2
    # (the second term comes from d(1/S²)/da[i] = -2/S³)
    
    # Let's compute sum_m w[m] * dc[m]/da[i] for each i
    # = sum_m w[m] * a[m-i] * (2 - [2i==m]) (for 0<=m-i<n)
    # = 2 * sum_m w[m] * a[m-i] - sum_m w[m] * a[m-i] * [2i==m]
    # The second term: [2i==m] means m=2i, so it's w[2i]*a[0] if 2i < 2n-1 (which it always is for i < n)
    # Wait, a[m-i] when m=2i is a[i]. So the correction is w[2i]*a[i].
    
    # Let's use the FFT-based approach for efficiency
    # cross[i] = sum_m w[m]*a[m-i] for 0<=m-i<n
    # This is a cross-correlation. Let's compute it.
    
    # For each i, cross[i] = sum_{m: 0<=m-i<n} w[m]*a[m-i] = sum_{j=0}^{n-1} w[i+j]*a[j]
    # where j = m-i, so m = i+j, and we need 0 <= i+j <= 2n-2.

    # So for most i, cross[i] = sum_{j=0}^{n-1} w[i+j]*a[j]
    # For i = n-1, j = n-1 gives m = 2n-2 which is valid. So actually all are valid.

    # This is the "valid" cross-correlation of w (length 2n-1) and a (length n).

    # Actually, let's just use a direct approach with FFT.

    # We want sum_j a[j]*w[i+j] = sum_j a[j]*w[(i+j)] 
    # Let k = i+j, j = k-i: sum_k a[k-i]*w[k] = conv(a, w)[k] evaluated...

    # This is the "cross-correlation" <a, v[i:i+n]>

    # Hmm that's not quite right either.
    
    # Let me just do it directly with a loop for n <= 1024 (should be fast enough):
    grad_a = np.zeros(n)
    for i in range(n):
        # cross[i] = sum_{j=0}^{n-1} w[i+j]*a[j]
        # w[i:i+n] has length n (since i < n and w has length 2n-1, i+n <= 2n-1)
        cross_i = np.dot(w[i:i+n], a)
        # Correction for the [2i==m] term: when m=2i, contribution is a[i] (not 2*a[i])
        # So the actual dc[2i]/da[i] = a[i] (not 2*a[i])
        # The formula sum_m w[m]*dc[m]/da[i] = 2*cross_i - w[2i]*a[i] (if 2i < 2n-1)
        correction = w[2*i] * a[i] if 2*i < len(w) else 0
        grad_a[i] = 2*cross_i - correction

    # obj = 2n * soft_max / S²
    # d(obj)/da[i] = 2n * (grad_a[i] * S² - soft_max * 2*S) / S⁴
    #              = 2n * grad_a[i] / S² - 4n * soft_max / S³
    #              = (2n/S²) * grad_a[i] - (4n*soft_max/S³)

Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))².
Uses L-BFGS-B with soft-max smoothing (adaptive temperature), diverse parametric
initializations, 1D active-set coordinate descent, and randomized restarts.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))²."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def make_objective_fn(n, tau):
    """Create the objective and gradient function for L-BFGS-B in softmax parameterization."""
    S_t = np.sqrt(2.0 * n)
    
    def fun_and_grad(xf):
        x = np.zeros(n)
        x[1:] = xf
        x_max = x.max()
        exp_x = np.exp(x - x_max)
        a = S_t * exp_x / exp_x.sum()
        
        c = np.convolve(a, a)
        c_max = c.max()
        e = np.exp(tau * (c - c_max))
        e_sum = e.sum()
        soft_max = c_max + np.log(e_sum) / tau
        obj = 2.0 * n * soft_max / (S_t * S_t)
        
        if not np.isfinite(obj):
            return obj, np.zeros(n - 1)
        
        w = e / e_sum
        
        # Gradient of soft_max w.r.t. a
        # grad_sm[i] = sum_m w[m] * dc[m]/da[i]
        # dc[m]/da[i] = 2*a[m-i] if 0<=m-i<n and i != m-i, else a[m-i] if 2i==m, else 0
        # = a[m-i]*(2 - [2i==m]) if 0<=m-i<n, else 0
        
        # cross[i] = sum_{j=0}^{n-1} w[i+j]*a[j] (using convolution)
        a_flipped = a[::-1]
        conv_w_a = np.convolve(w, a_flipped)
        cross = conv_w_a[n-1:n-1+n]
        
        grad_sm = 2.0 * cross.copy()
        # Correction: for each i, subtract w[2i]*a[i] (since dc[2i]/da[i] = a[i], not 2*a[i])
        for i in range(n):
            if 2 * i < len(w):
                grad_sm[i] -= w[2 * i] * a[i]
        
        # d(obj)/da[i] = (2n/S_t²) * grad_sm[i] - 4n*soft_max/S_t³
        d_obj_da = (2.0 * n / (S_t * S_t)) * grad_sm - (4.0 * n * soft_max / (S_t ** 3))
        
        # Chain rule: d(obj)/dx[j] for j >= 1
        # da[i]/dx[j] = a[i]*delta[ij] - a[i]*a[j]/S_t
        # d(obj)/dx[j] = d_obj_da[j]*a[j] - (a[j]/S_t)*sum_i d_obj_da[i]*a[i]
        trace = np.dot(d_obj_da, a)
        grad = np.empty(n - 1)
        for j in range(1, n):
            grad[j - 1] = d_obj_da[j] * a[j] - (a[j] / S_t) * trace
        
        return obj, grad
    
    return fun_and_grad


def lbfgs_optimize(a_init, n, tau_seq, deadline, maxiter_per_tau=1000):
    """Run L-BFGS-B with a sequence of tau values (annealing schedule)."""
    S_t = np.sqrt(2.0 * n)
    a = np.maximum(a_init, 1e-15)
    a = S_t * a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Initial parameterization
    x0 = np.zeros(n)
    x0[1:] = np.log(a[1:] / a[0])
    xf = x0[1:].copy()
    
    for tau in tau_seq:
        if time.monotonic() > deadline:
            break
        
        fn = make_objective_fn(n, tau)
        
        try:
            result = minimize(
                fn, xf, method='L-BFGS-B', jac=True,
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
            )
            xf = result.x
        except Exception:
            pass
        
        # Recover a
        x = np.zeros(n)
