    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_A(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        lo = max(0, j-n+1)  # Actually: A[k,j] = f[k-j] if 0<=k-j<n
        # For fixed j, k ranges from j to j+n-1, and A[k,j] = f[k-j]
        A[j:j+n, j] = f
    return A

def solve_lp(f, M, x0=None):
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_iterate(f_init, time_limit, verbose=False):
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = 0.3  # step size
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        M = np.max(c)
        
        g = solve_lp(f, M)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1-t)*f + t*g_rescaled
        f_new = np.clip(f_new, 0, None)
        s = np.sum(f_new)
        if s < 1e-10:
            t *= 0.5
            continue
        f_new *= (sum_f / s)
        
        score = local_score(f_new)
        f = f_new
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def subgradient_polish(f_init, time_limit, n_iter=5000, verbose=False):
    """Subgradient descent with averaged iterates."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    f *= target_sum / np.sum(f)
    
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()

    f_avg = f.copy()
    count = 0

    decay = 0.9999
    
    for i in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(f, f)
        k_star = int(np.argmax(c))
        M = c[k_star]
        
        # Subgradient of max(c) w.r.t. f
        # d/d f[i] of c[k_star] = f[k_star - i] + f[i - k_star]
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += f[j1]
            if 0 <= j2 < n and j2 != j1:  # avoid double counting when j1==j2
                grad[i] += f[j2]
            elif 0 <= j2 < n and j1 == j2:
                # Actually when k_star = 0 and i = 0, j1 = j2 = 0
                # But c[0] = f[0]^2, so d/d f[0] = 2*f[0]
                # Let me handle this more carefully
                pass
        
        # Actually, let me just compute it properly:
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                val += f[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                val += f[j2]
            grad[i] = val
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad /= grad_norm
        
        # Update
        f = f - lr * grad
        f = np.clip(f, 0, None)
        
        # Renormalize
        s = np.sum(f)
        if s < 1e-10:
            f = f_init.copy()
            continue
        f *= target_sum / s
        
        # Running average
        f_avg += f
        count += 1
        
        # Check score of averaged iterate
        if i % 100 == 0:
            score_avg = local_score(f_avg / count)
            if score_avg < best_score:
                best_score = score_avg
                best_f = (f_avg / count).copy()
        
        lr *= decay
    
    # Final check
    score = local_score(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

Wait, the subgradient approach with a loop over i for computing the gradient is O(n) per step, which for n=256 is fine. But we can do it faster using the convolution structure.

Actually, the subgradient of c[k*] with respect to f[i] is:
∂c[k*]/∂f[i] = f[k*-i] (if 0 ≤ k*-i < n) + f[i-k*] (if 0 ≤ i-k* < n)

But note that when i = k*, both terms give f[0], so the total is 2*f[0].
When k* = 0 and i = 0: c[0] = f[0]², ∂c[0]/∂f[0] = 2*f[0]. With our formula: f[0] + f[0] = 2*f[0]. ✓

For efficiency, we can compute this as:
grad[i] = f[k*-i] (if valid) + f[i-k*] (if valid)

This is just a "shifted" version of f, which can be computed with array indexing.
