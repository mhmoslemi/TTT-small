    n = len(f)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid convex optimization approach:
1. Water-filling / flattening iterations (exploiting KKT structure)
2. LP-guided steps for escaping flat regions
3. Coordinate descent for final polishing
Tries multiple lengths and diverse initializations within the time budget.

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def flattening_iteration(f, delta=0.01, alpha=0.5, n_active_ratio=0.2):
    """One water-filling iteration to flatten the convolution.
    
    At the KKT optimum, the cross-correlation of f with the active lag weights
    should be constant. This iteration moves f toward that condition.
    
    Parameters:
    - f: probability vector (sum=1, non-negative)
    - delta: threshold for "active" lags (within delta of max)
    - alpha: blending parameter (0 < alpha <= 1)
    - n_active_ratio: maximum fraction of lags to consider as active
    
    Returns: updated f, improvement info
    """
    n = len(f)
    conv = fast_convolve(f)
    h = conv.max()
    
    if h < 1e-15:
        return f, 0.0
    
    # Find active lags (those close to the maximum)
    active = np.where(conv >= h - delta)[0]
    
    # Limit active set size
    max_active = max(1, int(n_active_ratio * (2 * n - 1)))
    if len(active) > max_active:
        # Keep the top max_active
        active_convs = conv[active]
        top_idx = np.argsort(active_convs)[-max_active:]
        active = active[top_idx]
    
    # Compute pressure: p[i] = sum_{k in active} f[k-i]
    p = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        p[j_start:j_end+1] += f[k - j_end : k - j_start + 1][::-1]
    
    # Water-filling update: f_new[i] = f[i] / p[i]^alpha
    # Use safe division to handle zeros
    p_safe = np.maximum(p, 1e-15)
    f_new = f / np.power(p_safe, alpha)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f, 0.0
    f_new /= s
    
    new_h = fast_convolve(f_new).max()
    return f_new, new_h


def lp_guided_step(f, M, n, timeout=10):
    """One LP-guided step: find g that maximizes sum(g) s.t. g>=0, (f*g)[k]<=M."""
    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (np.array(rows_list), np.array(cols_list))), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-12:
            g /= s
            return g
    return None


def coordinate_descent(f, time_limit, n_steps_per_coord=3):

    given all others fixed."""
    f = f.copy()
    n = len(f)
    t0 = time.monotonic()

    # Let d = f[i]_new - f[i]_old

    # Actually, if we only change f[i] and renormalize, it's more complex.
    # Let's use a simpler approach: for each i, try a few candidate values

    best_score = fast_convolve(f).max()

            old_val = f[i]
            if old_val < 1e-12:

def gen_initializations(n, rng, parent=None):

            pp = pp / pp.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive water-filling iterations
with LP-guided escape steps and multi-scale active set management.
The key insight is that at the KKT optimum, the weighted cross-correlation of f
with itself must be constant on the support. We iteratively enforce this condition.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_gradient_smoothed(f, conv, temperature=0.01):
    """Compute smoothed subgradient of max(conv) at f.
    
    Uses softmax weighting over lags to get a smooth gradient.
    Returns (gradient vector, max value).
    """
    n = len(f)
    h = conv.max()
    
    # Softmax weights
    scaled = (conv - h) / temperature
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Gradient: grad[i] = sum_k w[k] * 2*f[k-i] (for valid k-i)
    grad = np.zeros(n)
    # Use thresholding for efficiency
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        grad[j_start:j_end+1] += 2.0 * w[k] * f[k - j_end : k - j_start + 1][::-1]
    
    return grad, h


def water_filling_step(f, conv, delta, alpha):
    """One water-filling step: adjust f to make the cross-correlation flatter.
    
    Parameters:
    - f: probability vector
    - conv: auto-convolution of f
    - delta: threshold for active lags
    - alpha: blending strength (0 to 1)
    
    Returns: new f
    """
    n = len(f)
    h = conv.max()
    
    if h < 1e-15:
        return f
    
    # Active lags
    active_threshold = h - delta
    active = np.where(conv >= active_threshold)[0]
    
    # Limit active set
    max_count = max(1, int(0.25 * (2*n-1)))
    if len(active) > max_count:
        active_convs = conv[active]
        idx = np.argsort(active_convs)[-max_count:]
        active = active[idx]
    
    # Compute pressure
    p = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        p[j_start:j_end+1] += f[k - j_end : k - j_start + 1][::-1]
    
    # Normalize pressure
    p_mean = p.mean()
    if p_mean < 1e-15:
        return f
    
    # Update: f_new[i] ∝ f[i] / (p[i]/p_mean)^alpha
    ratio = p / p_mean
    f_new = f * np.power(np.maximum(ratio, 1e-10), -alpha)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    f_new /= s
    
    return f_new


def lp_step(f, M, n, timeout=8):
    """LP-guided step."""
    rows_list = []
    cols_list = []
    for i in range(n):
        base = i
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    
    vals = np.tile(f, n).tolist()
    rows_arr = np.array(rows_list, dtype=np.int32)
    cols_arr = np.array(cols_list, dtype=np.int32)
    A = sparse.csc_matrix((vals, (rows_arr, cols_arr)), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-12:
            g /= s
            return g
    return None


def full_optimize(f0, time_limit, use_lp=True):
    """Full optimization pipeline for a single sequence."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    best_h = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Parameters
    delta = 0.01  # Active set width
    alpha = 0.5   # Water-filling strength
    temp = 0.005  # Smoothing temperature
    lp_interval = 30  # Do LP every 30 iterations
    
    iter_count = 0
    patience = 0
    last_lp_time = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        
        # --- Flattening step ---
        f_new = water_filling_step(f, conv, delta, alpha)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        h_new = conv_new.max()
        
        if h_new < best_h - 1e-14:
            f = f_new
            conv = conv_new
            best_h = h_new
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
        
        # --- Adaptive parameter adjustment ---
        if patience % 10 == 0 and patience > 0:
            # Narrow the active set and increase strength
            delta = max(delta * 0.8, 0.0005)
            alpha = min(alpha * 1.05, 0.95)
            temp = max(temp * 0.9, 0.0005)
            
            if patience > 50:
                # Reset with wider parameters
                delta = 0.02
                alpha = 0.4
                temp = 0.01
                patience = 0
        
        # --- LP step (periodic) ---
        if use_lp and iter_count % lp_interval == 0:
            lp_time = time.monotonic()
            if lp_time - last_lp_time > 5 and elapsed < time_limit * 0.7:
                M_current = conv.max()
                g = lp_step(f, M_current, n, timeout=min(5, time_limit - elapsed))
                if g is not None:
                    # Try blending
                    for t in [0.3, 0.5, 0.7]:
                        f_try = (1-t)*f + t*g
                        f_try = np.maximum(f_try, 0)
                        s_try = f_try.sum()
                        if s_try < 1e-12:
                            continue
                        f_try /= s_try
                        h_try = fast_convolve(f_try).max()
                        if h_try < best_h - 1e-14:
                            f = f_try
                            conv = fast_convolve(f)
                            best_h = h_try
                            best_f = f.copy()
                            patience = 0
                            break
                last_lp_time = time.monotonic()
    
    return best_f, best_h

I need a `project_simplex` function. The projection onto the probability simplex is:
Given z, find p such that p_i >= 0, sum(p) = 1, and ||p - z||_2 is minimized.

The solution is: p_i = max(z_i - τ, 0) where τ is chosen so that sum_i max(z_i - τ, 0) = 1.

τ can be found by sorting z and trying O(n) candidate values.

```python
def project_simplex(v):
    """Project v onto the probability simplex {p: p>=0, sum(p)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is the standard algorithm from Duchi et al. (2008).

    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def lp_step(f, M, n, rows, cols, timeout=6):
    """LP-guided step: max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-12:
            return g / s
    return None


def optimize_one(f0, time_limit, use_lp=True):
    """Full optimization for a single sequence using hybrid method."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    best_h = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Adaptive parameters
    temp = 0.008
    delta_wf = 0.015
    alpha_wf = 0.5
    eta_scale = 1.0
    beta = 0.0  # Start without momentum, add later
    y = f.copy()  # Nesterov ghost point
    rows, cols = None, None
    if use_lp:
        rows, cols = build_lp_matrix(n)

    iter_count = 0
    patience = 0
    last_lp_iter = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0

        # --- Subgradient step ---
        grad, h_cur = smoothed_subgradient(y, conv, temp)
        eta = eta_scale / (np.sqrt(iter_count + 1) * max(h_cur, 0.001))
        z = y - eta * grad
        f_new = project_simplex(z)

        # Nesterov update
        beta_cur = min(beta, 2.0 / (np.sqrt(iter_count) + 1.0))
        y_new = (1.0 + beta_cur) * f_new - beta_cur * f
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-12:
            y_new = f_new
        else:
            y_new /= sy

        f = f_new
        y = y_new
        conv = fast_convolve(y)
        h_new = conv.max()

        if h_new < best_h - 1e-14:
            best_h = h_new
            best_f = y.copy()
            patience = 0
        else:
            patience += 1

        # --- Water-filling when stuck ---
        if patience > 15 and patience % 15 == 0:
            f_wf = water_filling_update(y, conv, delta_wf, alpha_wf)
            conv_wf = fast_convolve(f_wf)
            h_wf = conv_wf.max()
            if h_wf < best_h - 1e-14:
                f = f_wf
                y = f_wf.copy()
                conv = conv_wf
                best_h = h_wf
                best_f = f.copy()
                patience = 0
            # Adjust parameters
            delta_wf = max(delta_wf * 0.85, 0.002)
            alpha_wf = min(alpha_wf * 1.05, 0.9)
            temp = max(temp * 0.95, 0.001)

        if patience > 0 and patience % 8 == 0:
            temp = max(temp * 0.97, 0.001)
            eta_scale = min(eta_scale * 1.02, 3.0)

        # --- LP step (periodic) ---
        if use_lp and iter_count - last_lp_iter >= 25 and patience > 10:
            if elapsed < time_limit * 0.75:
                remaining_for_lp = min(5.0, time_limit - elapsed - 1.0)
                if remaining_for_lp > 1.0:
                    M_cur = conv.max()
                    g = lp_step(y, M_cur, n, rows, cols, timeout=remaining_for_lp)
                    last_lp_iter = iter_count
                    if g is not None:
                        improved = False
                        for t_blend in [0.4, 0.6, 0.3, 0.7, 0.5]:
                            f_try = (1 - t_blend) * y + t_blend * g
                            f_try = np.maximum(f_try, 0)
                            st = f_try.sum()
                            if st < 1e-12:
                                continue
                            f_try /= st
                            h_try = fast_convolve(f_try).max()
                            if h_try < best_h - 1e-14:
                                f = f_try
                                y = f_try.copy()
                                conv = fast_convolve(f_try)
                                best_h = h_try
                                best_f = f.copy()
                                patience = 0
                                improved = True
                                break
                        if not improved:
                            # Small perturbation to escape
                            y = y + 0.001 * np.random.randn(n)
                            y = np.maximum(y, 0)
                            y /= y.sum()

    return best_f, best_h


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(('uni', np.ones(n) / n))

    for sf in [0.08, 0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'g{int(sf * 100)}', g))

    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(('cos1', c / c.sum()))

    c2 = 0.5 * (1 + np.cos(4 * np.pi * x / n))
    inits.append(('cos2', c2 / c.sum()))

    bi = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    inits.append(('bim', bi / bi.sum()))

    r = rng.dirichlet(np.ones(n) * 0.05)
    inits.append(('rnd', r))

    rnp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('rmp', rnp / rnp.sum()))

    tri = np.maximum(1.0 - 2.0 * np.abs(x / n - 0.5), 0.01)
    inits.append(('tri', tri / tri.sum()))

    # Multi-frequency
    mf = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x / n) + 0.2 * np.cos(2 * np.pi * 7 * x / n)
    inits.append(('mf', mf / mf.sum()))

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0)
            sp = pp.sum()
            if sp > 1e-10:
                inits.append(('par', pp / sp))
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            sp = pp.sum()
            if sp > 1e-10:
                inits.append(('par', pp / sp))

    # Fourier random phase
    for tr in range(2):
        fft_size = 1 << (2 * n - 1).bit_length()
        phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        F = np.zeros(fft_size // 2 + 1, dtype=complex)
        F[0] = float(n)
        F[1:] = np.exp(1j * phase[1:])
        f_fft = np.fft.irfft(F, fft_size)[:n]
        f_fft = np.maximum(f_fft, 0)
        sf_ = f_fft.sum()
        if sf_ > 1e-10:
            inits.append((f'fft{tr}', f_fft / sf_))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: hybrid optimization with multi-start."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            # Evaluate parent
            sc_par = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(sc_par):
                best_score = sc_par
                best_seq = parent.copy()
                print(f"Parent score: {sc_par:.6f}, n={len(parent)}")
    except Exception:
        pass

    lengths = [512, 768, 1024]
    all_results = []

    # Phase 1: Exploration
    print("Phase 1: Exploration")
    for n in lengths:
        if time.monotonic() > deadline - 150:
            break

        inits = gen_inits(n, rng, parent)
        n_cfgs = len(inits)
        remaining = deadline - time.monotonic() - 150
        time_per = max(8.0, remaining / max(1, n_cfgs * len(lengths)))
        print(f"  n={n}: {n_cfgs} inits, ~{time_per:.0f}s each")

        for name, a0 in inits:
            if time.monotonic() > deadline - 80:
                break
            ta = min(time_per, deadline - time.monotonic() - 80)
            if ta < 8:
                break

            t_st = time.monotonic()
            f_opt, h_opt = optimize_one(a0, time_limit=ta, use_lp=True)
            elapsed = time.monotonic() - t_st
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"    n={n} {name:5s} score={score:.6f} ({elapsed:.0f}s) *NEW BEST*")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Exploitation on top results
    all_results.sort(key=lambda x: x[0])
    top = all_results[:6]
    remaining = deadline - time.monotonic() - 40
    if remaining > 30 and top:
        print(f"Phase 2: Exploitation ({remaining:.0f}s)")
        t_each = remaining / len(top)
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 25:
                break
            ta = min(t_each, deadline - time.monotonic() - 25)
            if ta < 10:
                break
            f2, h2 = optimize_one(f, time_limit=ta, use_lp=True)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"  Deep [{idx}] n={n} {name} score={sc2:.6f}")

    # Phase 3: Final intensive polish on the best
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            # Try with and without LP
            for use_lp in [True, False]:
                if time.monotonic() > deadline - 8:
                    break
                tr = min(remaining / (2 if use_lp else 1), deadline - time.monotonic() - 8)
                if tr < 10:
                    break
                f3, h3 = optimize_one(best_seq, time_limit=tr, use_lp=use_lp)
                sc3 = float(evaluate_sequence(f3.tolist()))
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = f3.copy()
                    print(f"  Polish (lp={use_lp}) score={sc3:.6f}")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Ensure valid output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(512)

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
