```python
import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)
```

```python
import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_soft_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation peak."""
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    # Soft-max weights
    c_shifted = (c - np.max(c)) / tau
    w = np.exp(c_shifted)
    w /= np.sum(w)
    
    # Compute gradient using FFT
    N = 1

    W = np.fft.rfft(w, N)

    conv_aw = np.fft.irfft(A * W, N)
    q = conv_aw[0 : n]
    
    return p + q

    start = time.monotonic()
    n = len(a)
    N = 1
    while N < 2 * n - 1:

        P = np.abs(F) ** 2
        
        # Flatten: P_new = P^alpha normalized
        P_flat = P ** alpha
        P_flat = P_flat * (np.sum(P) / np.sum(P_flat))
        
        # Reconstruct with original phase

        # Inverse FFT
        a_new = np.fft.irfft(F_new, N)[:n].copy()
        
        # Project to non-negative

        if s < 1e-10:

        a_new = a_new * np.sum(a) / s
```

conv_flip[k] = sum_{j=0}^{n-1} a_flip[j] * w[k-j] where a_flip[j] = a[n-1-j]
= sum_{j=0}^{n-1} a[n-1-j] * w[k-j]
Let m = n-1-j:
= sum_{m=0}^{n-1} a[m] * w[k-(n-1-m)]
= sum_{m=0}^{n-1} a[m] * w[k-n+1+m]

We want p[i] = sum_{m=0}^{n-1} a[m] * w[i+m]
So k-n+1+m = i+m → k = i+n-1
Therefore p[i] = conv_flip[i+n-1]

conv_flip has length 2n-1 (indices 0 to 2n-2).
p[i] for i=0: conv_flip[n-1]
p[i] for i=n-1: conv_flip[2n-2]
So p = conv_flip[n-1 : 2n-1] ✓

For q:
conv_aw[k] = sum_{j=0}^{n-1} a[j] * w[k-j]
q[i] = sum_{j=0}^{n-1} a[j] * w[i-j] = conv_aw[i]
So q = conv_aw[0 : n] ✓

```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Combines spectral flattening (fast global optimization), adaptive gradient
descent with soft-max peak (refinement), and simulated annealing (fine-tuning).
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_soft_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation peak w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    
    # Soft-max weights
    c_max = np.max(c)
    w = np.exp((c - c_max) / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient using FFT
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    
    # p[i] = sum_j a[j] * w[i+j] = conv(flip(a), w)[i+n-1]
    a_flip = a[::-1]
    AF = np.fft.rfft(a_flip, N)
    conv_flip = np.fft.irfft(AF * W, N)
    p = conv_flip[n - 1:2 * n - 1]
    
    # q[i] = sum_j a[j] * w[i-j] = conv(a, w)[i]
    conv_aw = np.fft.irfft(A * W, N)
    q = conv_aw[0:n]
    
    return p + q


def spectral_flatten(a_init, time_limit, rng, verbose=False):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    best_a = a.copy()
    best_score = local_score(a)
    iteration = 0
    stagnant = 0
    
    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / time_limit)
        
        # Alpha schedule: start aggressive, end gentle
        alpha = 0.25 + 0.70 * progress ** 0.5
        
        F = np.fft.rfft(a, N)
        P = np.abs(F) ** 2
        
        # Flatten power spectrum
        P_flat = P ** alpha
        P_sum = np.sum(P)
        P_flat_sum = np.sum(P_flat)
        if P_flat_sum < 1e-300:
            break
        P_flat = P_flat * (P_sum / P_flat_sum)
        
        # Reconstruct with modified magnitude, same phase
        F_new = np.sqrt(P_flat) * np.exp(1j * np.angle(F))
        a_new = np.fft.irfft(F_new, N)[:n].copy()
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            # Failed, try random restart
            a_new = np.abs(a + 0.1 * rng.standard_normal(n))
            s = np.sum(a_new)
            a_new = a_new * np.sum(a) / s
        
        a_new = a_new * np.sum(a) / s
        
        score = local_score(a_new)
        iteration += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            # Accept with some probability to explore
            if rng.random() < 0.3:
                a = a_new
            stagnant += 1
            if stagnant > 100:
                # Add perturbation
                a = a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a * np.sum(a) / np.sum(a)
                stagnant = 0
        
        if verbose and iteration % 200 == 0:
            print(f"    SF iter={iteration}, alpha={alpha:.3f}, score={score:.6f}, best={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_a, best_score


def gradient_optimize(a_init, time_limit, rng, verbose=False):
    """Adaptive gradient descent with soft-max peak and momentum."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    sum_a = np.sum(a)
    
    tau_init = 8.0 / n
    tau_min = 0.3 / n
    lr = 0.3
    momentum = np.zeros(n)
    beta = 0.9
    
    iteration = 0
    stagnant = 0
    
    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / time_limit)
        tau = tau_init * (tau_min / tau_init) ** progress
        
        grad = compute_soft_gradient(a, tau)
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum
        
        # Project and normalize
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new * sum_a / s
        
        score = local_score(a_new)
        iteration += 1
        
        if score < best_score - 1e-13:
            a = a_new
            best_score = score
            best_a = a.copy()
            stagnant = 0
            lr = min(3.0, lr * 1.01)
            beta = min(0.98, beta + 0.001)
        else:
            stagnant += 1
            if stagnant > 30:
                lr *= 0.7
                stagnant = 0
                # Perturbation
                pert_size = min(0.1, 0.02 * stagnant)
                a = a + pert_size * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a * sum_a / np.sum(a)
            if stagnant > 200:
                # Big restart
                a = a * (1 + 0.2 * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = a * sum_a / np.sum(a)
                lr = 0.2
                momentum = np.zeros(n)
                stagnant = 0
        
        if verbose and iteration % 500 == 0:
            print(f"    GD iter={iteration}, tau={tau:.4f}, lr={lr:.3f}, score={score:.6f}, best={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_a, best_score


def sa_optimize(a_init, time_limit, rng, verbose=False):
    """Simulated annealing with targeted peak-reduction moves."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score
    sum_a = np.sum(a)
    
    temperature = 0.004
    cooling = 0.99985
    iteration = 0
    
    while time.monotonic() - start < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            # Compute contributions
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            # Reduce top contributors
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            # Boost non-peak elements
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 10:
                n_boost = min(10, len(other_idx) // 3)
                boost_idx = rng.choice(other_idx, size=n_boost, replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_boost))
        elif r < 0.55:
            # Smooth block modification
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            pos = int(rng.integers(0, n - width))
            candidate[pos:pos + width] *= np.exp(rng.normal(0, 0.025, size=width))
        elif r < 0.70:
            # Paired adjustment (sum-preserving)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.normal(0, 0.03 * sum_a / n)
                candidate[i] = max(0.0, a[i] + delta)
                candidate[j] = max(0.0, a[j] - delta)
        elif r < 0.85:
            # Global smooth perturbation
            candidate *= np.exp(rng.normal(0, 0.008, size=n))
        else:
            # Larger exploration
            width = max(2, int(rng.integers(2, max(3, n // 4))))
            pos = int(rng.integers(0, n - width))
            candidate[pos:pos + width] += rng.normal(0, 0.05 * sum_a / n, size=width)
            candidate = np.maximum(candidate, 0.0)
        
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate * sum_a / s
        
        new_score = local_score(candidate)
        iteration += 1
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                # Slight temperature bump on improvement
                temperature = min(0.01, temperature * 1.01)
        else:
            delta = new_score - current_score
            if delta < temperature:
                if rng.random() < np.exp(-delta / max(temperature, 1e-15)):
                    a = candidate
                    current_score = new_score
        
        temperature *= cooling
        if temperature < 0.0005:
            temperature = 0.0005 + 0.001 * rng.random()
    
    return best_a, best_score
```

Looking at this problem, I need to minimize `2n * max(conv(a,a)) / sum(a)^2` for a non-negative sequence. The previous approach used LP-based iteration which is slow. I'll use a different strategy combining:

1. **Spectral flattening** - fast O(n log n) iterations that reshape the power spectrum
2. **Adaptive gradient descent** with soft-max peak (FFT-accelerated gradient)
3. **Targeted simulated annealing** with peak-aware moves
4. **Multiple initializations** and phase cycling

Key insight: the gradient of the autocorrelation peak can be computed in O(n log n) using FFT cross-correlations, making gradient-based refinement much faster than LP.

```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Combines spectral flattening, FFT-accelerated gradient descent with soft-max
peak approximation, and targeted simulated annealing.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_gradient(a, tau, top_k=None):
    """Compute gradient of soft-max autocorrelation peak w.r.t. a.
    If top_k is set, only weight the top-k peaks."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    
    if top_k is not None and top_k < len(c):
        top_indices = np.argsort(c)[-top_k:]
        mask = np.zeros(len(c))
        mask[top_indices] = 1.0
        w = mask * np.exp((c - c_max) / tau)
    else:
        w = np.exp((c - c_max) / tau)
    
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    
    # p[i] = sum_j a[j]*w[i+j] via conv(flip(a), w)
    a_flip = a[::-1]
    AF = np.fft.rfft(a_flip, N)
    conv_flip = np.fft.irfft(AF * W, N)
    p = conv_flip[n - 1:2 * n - 1]
    
    # q[i] = sum_j a[j]*w[i-j] via conv(a, w)
    conv_aw = np.fft.irfft(A * W, N)
    q = conv_aw[0:n]
    
    return p + q


def spectral_flatten(a_init, time_limit, rng, verbose=False):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    best_a = a.copy()
    best_score = local_score(a)
    iteration = 0
    stagnant = 0
    
    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / max(time_limit, 1e-6))
        
        # Alpha schedule with some randomization
        base_alpha = 0.2 + 0.75 * progress ** 0.4
        alpha = base_alpha + 0.1 * rng.standard_normal()
        alpha = np.clip(alpha, 0.1, 0.98)
        
        F = np.fft.rfft(a, N)
        P = np.abs(F) ** 2
        
        # Flattening: compress dynamic range
        P_flat = P ** alpha
        P_sum = np.sum(P)
        P_flat_sum = np.sum(P_flat)
        if P_flat_sum < 1e-300:
            break
        P_flat = P_flat * (P_sum / P_flat_sum)
        
        # Reconstruct
        F_new = np.sqrt(P_flat) * np.exp(1j * np.angle(F))
        a_new = np.fft.irfft(F_new, N)[:n].copy()
        
        # Project and normalize
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.abs(a + 0.1 * rng.standard_normal(n))
            s = np.sum(a_new)
            a_new = a_new * np.sum(a) / s
        else:
            a_new = a_new * np.sum(a) / s
        
        score = local_score(a_new)
        iteration += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            if rng.random() < 0.4:
                a = a_new
            stagnant += 1
            if stagnant > 150:
                a = a + 0.08 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a * np.sum(a_init) / max(np.sum(a), 1e-10)
                stagnant = 0
        
        if verbose and iteration % 500 == 0:
            print(f"    SF iter={iteration}, alpha={alpha:.3f}, score={score:.6f}, best={best_score:.6f}, t={time.monotonic()-start:.0f}s")
    
    return best_a, best_score


def spectral_flatten_v2(a_init, time_limit, rng, verbose=False):
    """Alternative spectral flattening using soft-clip: P/(P+c) compression."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    best_a = a.copy()
    best_score = local_score(a)
    iteration = 0
    
    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / max(time_limit, 1e-6))
        
        F = np.fft.rfft(a, N)
        P = np.abs(F) ** 2
        P_mean = np.mean(P)
        
        # Soft-clip: compress high values, leave low values more intact
        c_param = P_mean * (0.1 + 2.0 * (1.0 - progress))
        P_new = P / (1.0 + P / (c_param * 5.0))
        P_new = P_new * (np.sum(P) / np.sum(P_new))
        
        F_new = np.sqrt(P_new) * np.exp(1j * np.angle(F))
        a_new = np.fft.irfft(F_new, N)[:n].copy()
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new * np.sum(a) / s
        
        score = local_score(a_new)
        iteration += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        elif rng.random() < 0.5:
            a = a_new
    
    return best_a, best_score


def gradient_optimize(a_init, time_limit, rng, verbose=False):
    """Adaptive gradient descent with soft-max peak, momentum, and top-k."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    sum_a = np.sum(a)
    
    tau_init = 10.0 / n
    tau_final = 0.5 / n
    lr = 0.4
    momentum = np.zeros(n)
    beta = 0.85
    use_topk = False
    top_k_val = 5
    
    iteration = 0
    stagnant = 0
    
    while time.monotonic() - start < time_limit:
        progress = min(1.0, (time.monotonic() - start) / max(time_limit, 1e-6))
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Switch to top-k when tau is small
        if tau < 2.0 / n and not use_topk:
            use_topk = True
        
        tk = top_k_val if use_topk else None
        grad = compute_gradient(a, tau, tk)
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum
        
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new * sum_a / s
        
        score = local_score(a_new)
        iteration += 1
        
        if score < best_score - 1e-13:
            a = a_new
            best_score = score
            best_a = a.copy()
            stagnant = 0
            lr = min(4.0, lr * 1.005)
            beta = min(0.97, beta + 0.0005)
        else:
            stagnant += 1
            if stagnant > 50:
                lr *= 0.6
                stagnant = 0
                a = a + 0.03 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a * sum_a / max(np.sum(a), 1e-10)
            if stagnant > 300:
                a = np.abs(a + 0.15 * rng.standard_normal(n))
                a = a * sum_a / max(np.sum(a), 1e-10)
                lr = 0.3
                momentum = np.zeros(n)
                beta = 0.85
                stagnant = 0
        
        if verbose and iteration % 1000 == 0:
            print(f"    GD iter={iteration}, tau={tau:.4f}, lr={lr:.3f}, topk={use_topk}, score={score:.6f}, best={best_score:.6f}, t={time.monotonic()-start:.0f}s")
    
    return best_a, best_score


def sa_optimize(a_init, time_limit, rng, verbose=False):
    """Simulated annealing with targeted peak-aware moves."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score
    sum_a = np.sum(a)
    
    temperature = 0.003
    cooling = 0.9999
    min_temp = 0.0003
    iteration = 0
    best_iter = 0
    since_best = 0
    
    while time.monotonic() - start < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 10:
                n_boost = min(8, len(other_idx) // 4)
                boost_idx = rng.choice(other_idx, size=n_boost, replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_boost))
        elif r < 0.55:
            # Smooth local block
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            pos = int(rng.integers(0, n - width))
            candidate[pos:pos + width] *= np.exp(rng.normal(0, 0.02, size=width))
        elif r < 0.70:
            # Two-point swap (sum-preserving)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.normal(0, 0.025 * sum_a / n)
                candidate[i] = max(0.0, a[i] + delta)
                candidate[j] = max(0.0, a[j] - delta)
        elif r < 0.82:
            # Global gentle perturbation
            candidate *= np.exp(rng.normal(0, 0.006, size=n))
        elif r < 0.92:
            # Multi-block
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                pos = int(rng.integers(0, n - width))
                candidate[pos:pos + width] *= np.exp(rng.normal(0, 0.018, size=width))
        else:
            # Larger exploration
            width = max(2, int(rng.integers(2, max(3, n // 5))))
            pos = int(rng.integers(0, n - width))
            candidate[pos:pos + width] += rng.normal(0, 0.04 * sum_a / n, size=width)
            candidate = np.maximum(candidate, 0.0)
        
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate * sum_a / s
        
        new_score = local_score(candidate)
        iteration += 1
        since_best += 1
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                best_iter = iteration
                since_best = 0
        else:
            delta_e = new_score - current_score
            if delta_e < 2.0 * temperature:
                if rng.random() < np.exp(-delta_e / max(temperature, 1e-15)):
                    a = candidate
                    current_score = new_score
        
        temperature = max(min_temp, temperature * cooling)
        
        # Periodic temperature reset if stuck too long
        if since_best > 5000:
            temperature = min(0.01, temperature * 3.0)
            since_best = 0
        
        if verbose and iteration % 5000 == 0:
            print(f"    SA iter={iteration}, temp={temperature:.6f}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-start:.0f}s")
    
    return best_a, best_score


def hill_climb(a_init, time_limit, rng):
    """Simple coordinate-descent style hill climbing."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    sum_a = np.sum(a)
    
    while time.monotonic() - start < time_limit:
        # Try modifying a random subset
        k = int(rng.integers(1, max(2, n // 16)))
        idx = rng.choice(n, size=k, replace=False)
        
        for i in idx:
            # Try scaling a[i] by various factors
            for factor in [0.7, 0.85, 1.0, 1.15, 1.3]:
                trial = a.copy()
                trial[i] = a[i] * factor
                s = np.sum(trial)
                if s < 1e-10:
                    continue
                trial = trial * sum_a / s
                sc = local_score(trial)
                if sc < best_score:
                    a = trial
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score
```
