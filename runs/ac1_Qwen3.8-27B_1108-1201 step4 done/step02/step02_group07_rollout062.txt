Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import coo_matrix


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A (dense) where A@b = conv(f,b). Shape (2n-1, n)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[:hi - lo]
    return A


def solve_lp(f, M, timeout=3.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

                      bounds=[(0, None)] * n,
                      method='highs',

    return None

    center = (n - 1) / 2.0
    width = n * (0.15 + 0.1 * rng.random())
    a = 1.0 + 0.8 * np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_chirp(n, rng):
    t = np.arange(n, dtype=np.float64)
    window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = 0.5 + 0.5 * window + 0.2 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_random_windowed(n, rng):
    base = 0.5 + 0.5 * rng.random(n)
    t = np.arange(n, dtype=np.float64)
    w = 0.3 + 0.7 * (0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1))))
    a = base * w
    return np.clip(a, 0.01, None)


def optimize_lp_steps(f_init, time_limit, t_step=0.05, M_factor=1.0, verbose=False):
    """LP-iteration optimizer with adaptive M factor."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    current_t = t_step

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = M_factor * float(np.max(c))

        b = solve_lp(f, M, timeout=2.0)
        if b is None:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        # Line search
        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 2.0 * t_step, 0.5 * t_step]:
            f_cand = np.clip((1.0 - t_try) * f + t_try * b, 0.01, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = np.clip((1.0 - best_t) * f + best_t * b, 0.01, 1000.0)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
            current_t = min(0.15, current_t * 1.1)
        else:
            stall += 1
            if stall % 20 == 0:
                current_t = max(0.02, current_t * 0.7)

        if verbose and it % 50 == 0:
            print(f"    [lp {time.monotonic()-t0:6.1f}s] it={it} score={best_s:.6f} best={best_score:.6f} stall={stall}")

        if stall >= 100:
            rng_esc = np.random.default_rng(42 + it * 7919)
            f = f + 0.02 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
            current_t = t_step

    return best_f, best_score


def subgradient_polish(f_init, time_limit, verbose=False):
    """Subgradient descent with heavy-ball momentum."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (n / S)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.005
    it = 0
    lr_decay = 0.9999

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        top_k = min(7, len(c))
        top_idx = np.argsort(c)[-top_k:]
        max_c = c[top_idx[0]]

        # Compute subgradient: average over active peaks
        g = np.zeros(n)
        weights = np.zeros(top_k)
        for ii, k in enumerate(top_idx):
            k = int(k)
            w = (c[k] / max_c) ** 4
            weights[ii] = w
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[k - j] * w
        g /= max(np.sum(weights), 1e-15)

        # Normalize gradient to unit norm for stable step size
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            g = g / g_norm

        # Center the gradient (maintain sum constraint)
        g = g - np.mean(g)

        momentum = beta * momentum + (1.0 - beta) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (n / float(np.sum(f)))

        it += 1
        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            else:
                lr *= lr_decay
            if verbose:
                print(f"    [subgrad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def peak_target_search(f_init, time_limit, verbose=False):
    """Local search that specifically targets reducing the peak."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    temp = 0.003
    cooling = 0.99985
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        candidate = f.copy()
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Compute contribution of each element to the peak
        contribs = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                contribs[j] = f[j] * f[idx]

        r = rng.random()

        if r < 0.5:
            # Reduce high-contribution elements, increase low-contribution
            sorted_idx = np.argsort(contribs)
            n_reduce = max(3, n // 10)
            n_increase = max(3, n // 15)
            reduce_set = sorted_idx[-n_reduce:]
            increase_set = sorted_idx[:n_increase]
            candidate[reduce_set] *= np.exp(rng.normal(-0.04, 0.02, size=n_reduce))
            candidate[increase_set] *= np.exp(rng.normal(0.03, 0.02, size=n_increase))
        elif r < 0.75:
            # Smooth perturbation
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.01, 1000.0)
        new_score = score_fast(candidate)

        current_score = score_fast(f)
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
            temp *= 0.99
        elif new_score < current_score:
            f = candidate
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            f = candidate

        temp *= cooling
        it += 1

        if verbose and it % 200 == 0:
            print(f"    [peak {time.monotonic()-t0:6.1f}s] it={it} score={score_fast(f):.6f} best={best_score:.6f} temp={temp:.5f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def refine_lp(f_init, time_limit, verbose=False):
    """Final LP refinement with very small steps and lower M factor."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))
        S = float(np.sum(f))

        # Try slightly reduced M to push peak down
        for M_frac in [1.0, 0.95, 0.9]:
            b = solve_lp(f, M_frac * M, timeout=1.5)
            if b is not None:
                Sb = float(np.sum(b))
                if Sb > 1e-12:
                    b = b * (S / Sb)
                    best_t = 0.03
                    best_s = np.inf
                    for t_try in [0.03, 0.06, 0.015]:
                        f_cand = np.clip((1.0 - t_try) * f + t_try * b, 0.01, 1000.0)
                        s_cand = score_fast(f_cand)
                        if s_cand < best_s:
                            best_s = s_cand
                            best_t = t_try

                    f = np.clip((1.0 - best_t) * f + best_t * b, 0.01, 1000.0)

                    if best_s < best_score - 1e-10:
                        best_score = best_s
                        best_f = f.copy()
                        stall = 0
                    else:
                        stall += 1
                    break

        if stall >= 50:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search: LP optimization + subgradient + peak-target + final polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = score_fast(provided)
            print(f"Provided: n={len(provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = provided.copy()
        except Exception:
            pass

    print(f"\n=== Phase 1: Multi-start LP optimization ===")

    # Define candidate (n, init_type) pairs
    tasks = []
    n_values = [192, 256, 320, 384, 512]
    init_types = ['hann', 'flat_noise', 'cosine_mod', 'bartlett', 'random_windowed', 'chirp']

    if provided is not None:
        pn = len(provided)
        tasks.append((pn, 'provided'))
        # Also try interpolating to nearby lengths
        for dn in [-64, -32, 32, 64]:
            tasks.append((pn + dn, 'provided_interp'))

    for n in n_values:
        for init_name in init_types[:4]:
            tasks.append((n, init_name))

    # Time allocation: weight by 1/n (smaller n = faster LP = more iterations)
    time_weights = []
    for n, _ in tasks:
        time_weights.append(128.0 / n)
    total_w = sum(time_weights)

    it_global = 0
    for idx, (n, init_name) in enumerate(tasks):
        if time.monotonic() > deadline - 400:
            break

        n = max(32, n)

        if init_name == 'provided' and provided is not None and len(provided) == n:
            f0 = provided.copy()
        elif init_name == 'provided_interp' and provided is not None:
            x_old = np.linspace(0, 1, len(provided))
            x_new = np.linspace(0, 1, n)
            f0 = np.interp(x_new, x_old, provided)
        elif init_name == 'hann':
            f0 = make_init_hann(n, rng)
        elif init_name == 'bartlett':
            f0 = make_init_bartlett(n, rng)
        elif init_name == 'flat_noise':
            f0 = make_init_flat_noise(n, rng)
        elif init_name == 'cosine_mod':
            f0 = make_init_cosine_mod(n, rng, m=1 + rng.integers(0, 3))
        elif init_name == 'chirp':
            f0 = make_init_chirp(n, rng)
        elif init_name == 'geometric':
            f0 = make_init_geometric(n, rng)
        elif init_name == 'two_hump':
            f0 = make_init_two_hump(n, rng)
        else:
            f0 = make_init_random_windowed(n, rng)

        # Allocate time: more for smaller n, less for later tasks
        remaining = deadline - time.monotonic() - 400
        frac = time_weights[idx] / sum(time_weights[idx:])
        t_alloc = max(5.0, remaining * frac * 0.8)
        t_alloc = min(t_alloc, 150.0)

        # Skip if very little time or if this n already converged well
        if t_alloc < 5:
            continue

        it_global += 1
        print(f"  [{it_global:2d}] n={n:4d} init={init_name:16s} alloc={t_alloc:.0f}s", end="")

        f_best, s_best = optimize_lp_steps(f0, t_alloc, t_step=0.05, M_factor=1.0)
        print(f" => {s_best:.6f} (best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}) ***")

    print(f"\n  Phase 1 complete. Best: {best_score:.6f}")

    # Phase 2: Subgradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 200:
        pt = min(250.0, deadline - time.monotonic() - 200)
        if pt > 20:
            print(f"\n=== Phase 2: Subgradient polish ({pt:.0f}s) ===")
            f_pol, s_pol = subgradient_polish(best_seq, pt, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = f_pol.copy()
                print(f"  Polished: {best_score:.6f}")

    # Phase 3: Peak-target local search
    if best_seq is not None and time.monotonic() < deadline - 100:
        ls_t = min(150.0, deadline - time.monotonic() - 100)
        if ls_t > 15:
            print(f"\n=== Phase 3: Peak-target search ({ls_t:.0f}s) ===")
            f_ls, s_ls = peak_target_search(best_seq, ls_t, verbose=True)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = f_ls.copy()
                print(f"  After peak search: {best_score:.6f}")

    # Phase 4: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 10:
        fl_t = min(60.0, deadline - time.monotonic() - 8)
        if fl_t > 10:
            print(f"\n=== Phase 4: Final LP refinement ({fl_t:.0f}s) ===")
            f_fin, s_fin = refine_lp(best_seq, fl_t, verbose=True)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Phase 5: Quick subgradient + peak search on result
    if best_seq is not None and time.monotonic() < deadline - 5:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            print(f"\n=== Phase 5: Final polish ({rem:.0f}s) ===")
            f_p5, s_p5 = subgradient_polish(best_seq, rem * 0.5)
            if s_p5 < best_score:
                best_score = s_p5
                best_seq = f_p5.copy()
            if time.monotonic() < deadline - 3:
                rem2 = deadline - time.monotonic() - 3
                if rem2 > 5:
                    f_p5b, s_p5b = peak_target_search(best_seq, rem2)
                    if s_p5b < best_score:
                        best_score = s_p5b
                        best_seq = f_p5b.copy()
            print(f"  Final: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")

        if final_score < best_score:
            best_score = final_score
            best_seq = np.array(final_list)

        return final_list

    # Fallback
    return [1.0] * 256
