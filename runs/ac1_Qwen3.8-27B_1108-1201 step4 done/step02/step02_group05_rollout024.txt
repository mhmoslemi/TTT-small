import numpy as np
import scipy.optimize
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def peak_gradient(a, top_k=1):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    # Find indices where c is near the peak
    threshold = M * (1 - 1e-6)
    peak_indices = np.where(c >= threshold)[0]
    
    # Compute gradient: dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n
    grad = np.zeros(n)
    for k in peak_indices:
        # For each peak index k, add contribution to gradient
        # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # i ranges from lo to hi, and k-i ranges from k-hi to k-lo
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    
    grad /= len(peak_indices)
    return grad, M

def multiplicative_update(a, n_iter=100, top_k=3, decay=0.5):
    """Multiplicative weight update to reduce peak."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    for it in range(n_iter):
        c = fast_convolve(a)
        M = c.max()
        
        # Find top-k peak indices
        peak_idx = np.argsort(c)[-top_k:]
        
        # Compute blame for each element
        blame = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            blame[idx] += a[k - idx]
        
        blame /= len(peak_idx)
        
        # Multiplicative update: reduce high-blame elements
        # a_new[i] = a[i] / (1 + alpha * blame[i])
        alpha = 0.1 * (1 - it / n_iter * decay)
        a = a / (1.0 + alpha * blame)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    
    return a

def projected_gradient_descent(a, n_iter=500, step=None, top_k=5):
    """Projected gradient descent on the peak objective."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = obj_norm(a)
    
    if step is None:
        step = 0.01
    
    for it in range(n_iter):
        grad, M = peak_gradient(a, top_k=top_k)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        s = step * M / (g_norm + 1e-15)
        s = min(s, 0.1)  # Cap step size
        
        # Gradient step
        a_new = a - s * grad
        
        # Project: clip to [0, inf] and normalize
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_sum
        
        obj = obj_norm(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        
        # Adaptive step: reduce if no improvement
        if obj >= best_obj:
            step *= 0.9
    
    return best_a, best_obj

    # Build constraint matrix (sparse would be ideal but let's use dense for now)

    A = np.zeros((len(c), n))

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    """2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def peak_subgrad(a, top_k=1):
    """Subgradient of max_k c[k] where c=conv(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    if M < 1e-18:
        return np.zeros(n), M
    pk = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in pk:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= float(len(pk))
    return grad, M

def exp_peak_shave(a, n_iter=200, top_k=3, alpha_init=0.5, alpha_decay=0.98):
    """Exponential multiplicative peak shaving."""
    n = len(a)
    a = np.maximum(a, 1e-18).copy()
    a /= a.sum()
    alpha = alpha_init
    best_a = a.copy()
    best_obj = obj_norm(a)
    for it in range(n_iter):
        c = fast_convolve(a)
        M = c.max()
        if M < 1e-18:
            break
        pk = np.argsort(c)[-top_k:]
        blame = np.zeros(n)
        for k in pk:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            blame[idx] += a[k - idx]
        blame /= float(len(pk))
        # Exponential scaling
        a = a * np.exp(-alpha * blame / M)
        a = np.maximum(a, 1e-18)
        a /= a.sum()
        obj = obj_norm(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        alpha *= alpha_decay

def subgrad_descent(a, n_iter=500, step_init=0.05, top_k=5, step_decay=0.95):
    """Subgradient descent with projection."""
    n = len(a)

    a /= a.sum()
    best_a = a.copy()
    best_obj = obj_norm(a)
    step = step_init

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        s = min(step * M / (gnorm + 1e-15), 0.2)
        a_new = a - s * grad
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-12:

        else:

        if obj < best_obj - 1e-12:

            best_a = a_new.copy()
            a = a_new.copy()
        else:

def lp_direction(f, M, time_limit=30):
    """LP: find g>=0 max sum(g) s.t. conv(f,g) <= M."""
    n = len(f)
    n_c = 2 * n - 1
    A = np.zeros((n_c, n))
    for j in range(n):
        end = min(n_c, j + n)
        A[j:end, j] = f[:end - j]
    c_obj = -np.ones(n)
    b = np.full(n_c, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                                     method='highs', options={'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()
