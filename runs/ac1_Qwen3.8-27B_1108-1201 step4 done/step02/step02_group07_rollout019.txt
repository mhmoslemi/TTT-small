"""
Optimized peak-convolution minimizer for non-negative sequences.
Uses iterative soft-gradient descent, LP-based direction finding,
and peak-targeted local search to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n+m-1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def normalize_sum(a, target=1.0):
    S = np.sum(a)
    if S < 1e-15:
        return a
    return a * (target / S)


def simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Use the algorithm from Duchi et al. (2011)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def soft_gradient_step(a, t, lr):
    """One step of soft-gradient descent toward flattening the convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    # Compute softmax weights
    c_max = np.max(c)
    w = np.exp((c - c_max) / t)
    w /= np.sum(w)
    # Gradient: d(sum_k w[k]*c[k])/da[i] = 2 * sum_{j} w[i+j]*a[j] (valid j)
    grad = np.zeros(n)
    for i in range(n):
        # j ranges over [0, n-1] such that 0 <= i+j <= 2n-2
        j_min = 0
        j_max = min(n-1, 2*n-2-i)
        grad[i] = 2.0 * np.dot(w[i+j_min:i+j_max+1], a[j_min:j_max+1])
    # Remove mean component (to maintain sum = 1)
    grad -= np.mean(grad)
    return a, grad, c, w


def soft_gradient_descent(a_init, time_limit):
    """Iterative soft-gradient descent with adaptive temperature."""
    a = normalize_sum(np.asarray(a_init, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    
    t_temp = 0.5  # temperature
    lr = 0.01
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1:
            break
        
        a, grad, c, w = soft_gradient_step(a, t_temp, lr)
        a = a - lr * grad
        a = np.maximum(a, 0)
        a = simplex_project(a)
        
        it += 1
        if it % 50 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            # Adapt parameters
            t_temp *= 0.95
            if stall > 10:
                lr *= 0.7
                t_temp = max(0.01, t_temp * 2)
                stall = 0
        
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    return best_a, best_score
