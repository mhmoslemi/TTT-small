    # Maximize sum(g) subject to g >= 0, (f*g)_k <= M for all k
    # (f*g)_k = sum_j f_j * g_{k-j} where g_{k-j} = 0 if k-j out of range

        A_ub = np.zeros((2 * n - 1, n))
        for m in range(n):
            A_ub[m:m + n, m] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        # Use top-k constraints based on where f is large
        # The constraint (f*g)_k <= M is most binding where f has mass
        # For simplicity, just use all
        A_ub = np.zeros((2 * n - 1, n))
        for m in range(n):
            A_ub[m:m + n, m] = f
        b_ub = np.full(2 * n - 1, M)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        return g
    return f.copy()


def optimize_lp(f0, time_limit, t_init=0.5, t_decay=0.999, min_t=0.05,
                max_iter=None):
    """
    LP-based best response optimization.
    Iteratively: compute M=max(f*f), solve LP for best response g,
    update f <- (1-t)*f + t*g_normalized.
    """
    f = f0.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Also track averaged iterate
    avg_f = f.copy()
    avg_count = 0

    t0 = time.monotonic()
    t = t_init
    iterate = 0

    while time.monotonic() - t0 < time_limit:
        if max_iter is not None and iterate >= max_iter:
            break

        M = conv.max()
        if M < 1e-14:
            break

        g = lp_best_response(f, M)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g_norm = g / g_sum

        # Update
        f_new = (1 - t) * f + t * g_norm
        f_new = project_simplex(f_new)

        # Track average
        avg_f = (avg_f * avg_count + f_new) / (avg_count + 1)
        avg_count += 1

        f = f_new
        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
        # Also check averaged iterate
        avg_conv = fast_convolve(avg_f)
        avg_obj = avg_conv.max()
        if avg_obj < best_obj - 1e-14:
            best_obj = avg_obj
            best_f = avg_f.copy()

        t = max(min_t, t * t_decay)
        iterate += 1

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr_init=0.01, momentum=0.9,
                       top_k=30, temp_init=0.0001):
    """Fine subgradient polishing on the simplex."""
    f = f0.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 5000:
            break

        # Compute subgradient at top peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        else:
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()

        # Softmax weights
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * f[k - idx]

        # Update
        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        f = f_new
        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        temp *= 1.001
        momentum = min(0.98, momentum + 0.0001)

    return best_f, best_obj


def mirror_polish(f0, time_limit, eta=0.3, t_param=50.0):
    """Multiplicative weight (mirror descent) polishing."""
    f = f0.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    t_cur = t_param
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break

        max_conv = conv.max()
        if max_conv < 1e-14:
            break

        # Softmax over convolution peaks
        scaled = t_cur * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.01
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[k_int] * 2.0 * f[k - idx]

        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_update = -eta * sg_centered / max_conv
        log_update -= log_update.max()
        f = f * np.exp(log_update)
        f = np.maximum(f, 0)
        s = f.sum()
        f = f / s if s > 1e-15 else np.ones(n) / n

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            eta *= 0.7
            t_cur *= 2.0
            if eta < 0.05:
                eta = 0.3
                t_cur = t_param
                no_improve = 0

    return best_f, best_obj


def generate_starts(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng()
    starts = {}
    x = np.arange(n, dtype=np.float64)

    starts['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    starts['ramp'] = ramp
    starts['ramp_rev'] = ramp[::-1].copy()

    tri = np.minimum(x + 1, n - x)
    tri /= tri.sum()
    starts['triangle'] = tri

    for sf in [0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        starts[f'gauss_{sf}'] = g

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    starts['bimodal'] = bi

    starts['random'] = rng.dirichlet(np.ones(n) * 0.5)

    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.sum() > 1e-10:
            if len(p) >= n:
                pp = p[:n]
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                starts['parent'] = pp / ps

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: multi-phase LP + subgradient + mirror descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Quick LP scan over lengths and starts
    print("=== Phase 1: LP Quick Scan ===")
    lengths = [128, 256, 384, 512]
    phase1_end = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.3)
    results = []

    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break
        starts = generate_starts(n, parent, rng)
        names = list(starts.keys())
        # Limit starts per length
        if len(names) > 5:
            priority = ['parent', 'uniform', 'ramp', 'bimodal', 'random', 'triangle', 'gauss_0.2']
            names = [nm for nm in priority if nm in starts]
            names = names[:5]

        time_per = max(8.0, (phase1_end - time.monotonic() - 10) / max(1, len(names)))

        for name in names:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(time_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 4:
                break

            f0 = starts[name]
            f_opt, obj = optimize_lp(f0, time_limit=t_alloc, t_init=0.6,
                                     t_decay=0.9995, min_t=0.2)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, f_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = f_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on top configs with varying step sizes
    print("=== Phase 2: Deep LP Optimization ===")
    top_configs = []
    seen = set()
    for score, n, name, f_opt in results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, f_opt))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    # Add some fresh starts at best length
    best_n = len(best_sequence) if best_sequence is not None else 256
    fresh_starts = generate_starts(best_n, parent, rng)
    for nm in ['random', 'gauss_0.2', 'bimodal']:
        if nm in fresh_starts:
            top_configs.append((best_n, fresh_starts[nm]))

    remaining = deadline - time.monotonic() - 60.0
    n_configs = len(top_configs)
    time_per_deep = max(10.0, remaining / max(1, n_configs * 2))

    for i, (n, f0) in enumerate(top_configs):
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 5:
            break

        # Try multiple step size schedules
        for t_init in [0.5, 0.3, 0.8]:
            if time.monotonic() > deadline - 30:
                break
            t_single = t_alloc / 3.0
            if t_single < 3:
                break

            f_opt, obj = optimize_lp(f0, time_limit=t_single,
                                     t_init=t_init, t_decay=0.999, min_t=0.1)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = f_opt.copy()
                print(f"  Deep LP: n={n}, t0={t_init}, score={score:.6f}")

    print(f"Phase 2 best: {best_score:.6f}")

    # Phase 3: Polish with subgradient and mirror descent
    print("=== Phase 3: Polishing ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # Subgradient polish with different params
        for cfg in [
            {'lr_init': 0.005, 'momentum': 0.92, 'top_k': 20, 'temp_init': 0.00005},
            {'lr_init': 0.002, 'momentum': 0.95, 'top_k': 40, 'temp_init': 0.0001},
            {'lr_init': 0.01, 'momentum': 0.88, 'top_k': 15, 'temp_init': 0.00003},
        ]:
            if time.monotonic() > deadline - 15:
                break
            f_opt, obj = subgradient_polish(best_sequence,
                                            time_limit=min(60, remaining / 4), **cfg)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = f_opt.copy()
                print(f"  SG polish: score={score:.6f}")

        # Mirror polish
        if time.monotonic() < deadline - 15:
            f_opt, obj = mirror_polish(best_sequence,
                                       time_limit=min(60, remaining / 3),
                                       eta=0.2, t_param=100.0)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = f_opt.copy()
                print(f"  Mirror polish: score={score:.6f}")

        # LP polish with very small step size
        if time.monotonic() < deadline - 10:
            f_opt, obj = optimize_lp(best_sequence,
                                     time_limit=min(90, remaining / 2),
                                     t_init=0.15, t_decay=0.9995, min_t=0.05)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = f_opt.copy()
                print(f"  LP polish: score={score:.6f}")

        # Perturbation + re-optimize
        if time.monotonic() < deadline - 10:
            for _ in range(3):
                if time.monotonic() > deadline - 10:
                    break
                noise_scale = 0.003
                noise = rng.standard_normal(n_best) * noise_scale
                f_pert = project_simplex(best_sequence + noise)

                # Short LP + SG
                f_opt, obj = optimize_lp(f_pert, time_limit=min(30, remaining / 4),
                                         t_init=0.3, t_decay=0.999, min_t=0.1)
                if time.monotonic() < deadline - 15:
                    f_opt, obj = subgradient_polish(f_opt, time_limit=min(20, remaining / 4),
                                                    lr_init=0.003, momentum=0.92,
                                                    top_k=20, temp_init=0.00005)

                score = float(evaluate_sequence(f_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = f_opt.copy()
                    print(f"  Perturb+opt: score={score:.6f}")

    # Phase 4: Try a different length if time remains
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 4: Alternative Length ===")
        best_n = len(best_sequence)
        alt_n = int(best_n * 1.5) if best_n <= 300 else int(best_n * 0.75)
        alt_n = max(64, min(1024, alt_n))

        if alt_n != best_n:
            # Interpolate parent to new length
            p = np.interp(np.linspace(0, 1, alt_n),
                          np.linspace(0, 1, best_n), best_sequence)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                p = p / ps

            remaining4 = deadline - time.monotonic() - 30
            f_opt, obj = optimize_lp(p, time_limit=min(120, remaining4 * 0.6),
                                     t_init=0.5, t_decay=0.999, min_t=0.15)
            if time.monotonic() < deadline - 20:
                f_opt, obj = subgradient_polish(f_opt, time_limit=min(40, remaining4 * 0.3),
                                                lr_init=0.005, momentum=0.92,
                                                top_k=20, temp_init=0.00005)

            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = f_opt.copy()
                print(f"  Alt len {alt_n}: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
