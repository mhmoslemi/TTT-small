Optimization of non-negative step sequences to minimize peak autocorrelation score.
Uses tilted-window initializations, smooth-max L-BFGS-B, LP-based refinement,

the symmetric bound of 2.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_max_conv(a, T):
    """Compute smooth max of conv(a,a) and its gradient.
    Returns (value, gradient)."""
    c = fft_convolve(a, a)
    # Shift for numerical stability
    c_shifted = c - np.max(c)
    exp_c = np.exp(T * c_shifted)
    log_sum_exp = np.log(np.sum(exp_c)) + np.max(c)
    w = exp_c / np.sum(exp_c)  # softmax weights
    value = log_sum_exp / T
    
    # Gradient computation
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = sum_k w[k] * d conv[a,a][k] / da[i]
        # d conv[a,a][k] / da[i] = 2*a[k-i] if 0<=k-i<n and k!=2i, a[i] if k==2i
        k_start = max(0, i)  # k-i >= 0 => k >= i
        k_end = min(2*n-2, i+n-1)  # k-i <= n-1 => k <= i+n-1
        # Sum over k in [k_start, k_end]
        if k_end >= k_start:
            indices = np.arange(k_start, k_end+1)
            m_vals = indices - i  # m = k-i, in [0, n-1]
            # For k = 2i: correction
            if 0 <= 2*i <= 2*n-2 and 2*i >= k_start and 2*i <= k_end:
                # w[2i] * a[i] instead of w[2i] * 2*a[i]
                grad[i] = 2.0 * np.sum(w[indices] * a[m_vals]) - w[2*i] * a[i]
            else:
                grad[i] = 2.0 * np.sum(w[indices] * a[m_vals])
    return value, grad

"""
Optimizes non-negative step sequences to minimize peak autocorrelation score.
Uses tilted-window initializations exploiting asymmetry, smooth-max L-BFGS-B,
LP-based refinement, and local search. Key insight: asymmetric sequences beat
the symmetric bound of 2 by making a[i]*a[n-1-i] products small.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_max_and_grad(a, T):
    """Returns (smooth_max_value, gradient) of smooth max of conv(a,a)."""
    c = fft_convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    value = c_max + np.log(sum_exp) / T

    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)

        two_i = 2 * i

        grad[i] = s
    return value, grad

def make_tilted_hann(n, tilt, rng=None):
    """Asymmetric Hann window with linear tilt."""
    t = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    linear = 1.0 + tilt * (2.0 * t / max(1, n - 1) - 1.0)
    a = hann * linear
    if rng is not None:
        a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    t = np.arange(n, dtype=np.float64)

def build_conv_matrix(f, n_lags=None):
    """Build matrix A where A @ b = conv(f, b)."""
    n = len(f)
    if n_lags is None:
        n_lags = 2 * n - 1
    A = np.zeros((n_lags, n), dtype=np.float64)
    for j in range(n):
        lo = max(0, j - (n_lags - 1))
        hi = min(n - 1, n_lags - 1 - j)
        if hi >= lo:
            A[lo:hi + 1 + j - lo, j] = f[j - lo:j - lo + (hi - lo + 1)]
    # Actually let's be more careful
    A = np.zeros((n_lags, n), dtype=np.float64)
    for j in range(n):
        # conv(f,b)[k] gets contribution f[j]*b[k-j] when 0<=k-j<n, i.e., j<=k<j+n
        k_lo = j
        k_hi = min(n_lags - 1, j + n - 1)
        if k_hi >= k_lo:

    return A

            a_cand = (1.0 - t_try) * a + t_try * b
            a_cand = np.clip(a_cand, 0.001, 1000.0)
            s_cand = score_fast(a_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        a = (1.0 - best_t) * a + best_t * b
        a = np.clip(a, 0.001, 1000.0)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 20 == 0:
            print(f"    LP it={it:3d} elapsed={time.monotonic()-t0:.0f}s score={best_s:.6f} best={best_score:.6f} stall={stall}")

        if stall >= 50:
            # Perturb to escape
            rng_esc = np.random.default_rng(42 + it * 7919)
            a += 0.05 * rng_esc.standard_normal(n)
            a = np.clip(a, 0.01, 1000.0)
            stall = 0
            it = 0

    return best_a, best_score


def local_search(a0, time_limit, verbose=False):
    """Perturbation-based local search focusing on peak-contributing elements."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    temp = 0.001
    cooling = 0.99995
    it = 0

    while time.monotonic() - t0 < time_limit - 5:
        candidate = a.copy()
        r = rng.random()

        if r < 0.5:
            # Target peak-contributing elements
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = a[j] * a[idx]
            total = np.sum(contribs)
            if total > 0:
                probs = contribs / total
                # Perturb top contributors
                top_k = max(5, n // 10)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.015, size=top_k))
                # Boost others
                others = np.setdiff1d(np.arange(n), top_idx)
                if len(others) > 0:
                    n_boost = min(8, len(others))
                    boost_idx = rng.choice(others, size=n_boost, replace=False)
                    candidate[boost_idx] *= np.exp(rng.normal(0.02, 0.015, size=n_boost))
        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.9:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))
        else:
            # Asymmetry-adjusting perturbation
            # Shift mass from one side to the other
            side = 1 if rng.random() < 0.5 else -1
            mask_left = np.arange(n) < n // 2
            mask_right = ~mask_left
            factor = np.exp(side * rng.uniform(0.01, 0.04))
            candidate[mask_left] *= factor
            candidate[mask_right] /= factor

        candidate = np.clip(candidate, 0.001, 1000.0)
        new_score = score_fast(candidate)
        it += 1

        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            temp *= 0.99  # reward: cool faster
        elif rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-15)):
            a = candidate

        temp *= cooling

        if verbose and it % 200 == 0:
            print(f"    LS it={it:5d} elapsed={time.monotonic()-t0:.0f}s best={best_score:.6f} temp={temp:.6f}")

    return best_a, best_score


def upsample_sequence(a, new_n):
    """Upsample sequence to new length via interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.001, None)


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization: multi-phase search for best non-negative sequence."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def remaining():
        return max(0.0, deadline - time.monotonic())

    def update_best(a, tag=""):
        nonlocal best_score, best_seq
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_seq = np.clip(a.copy(), 0.001, 1000.0)
            print(f"  *** NEW BEST [{tag}]: score={s:.6f}, n={len(a)} ***")
            return True
        return False

    # --- Get initial candidates ---
    candidates = []
    lengths_to_try = [256, 384, 512, 768]

    # Use provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        candidates.append(('provided', provided))
        print(f"Provided: n={len(provided)}, score={score_fast(provided):.6f}")

    # Generate diverse initializations
    for n in lengths_to_try:
        candidates.append((f'hann_t03_n{n}', make_tilted_hann(n, 0.3, rng)))
        candidates.append((f'hann_t05_n{n}', make_tilted_hann(n, 0.5, rng)))
        candidates.append((f'gauss_n{n}', make_tilted_gaussian(n, 0.4, 0.35, 0.25)))
        candidates.append((f'power_n{n}', make_power_law(n, 1.5, direction=1, rng=rng)))
        candidates.append((f'stair_n{n}', make_staircase(n, 8, rng)))

    print(f"\n=== Phase 1: L-BFGS-B optimization ({len(candidates)} candidates) ===")
    p1_time = min(400.0, remaining() * 0.28)
    print(f"  Time budget: {p1_time:.0f}s")

    T_schedule = [1.0, 3.0, 10.0, 20.0, 40.0]
    per_candidate_time = max(5.0, p1_time / len(candidates))

    for name, a0 in candidates:
        if remaining() < per_candidate_time + 10:
            break
        t_alloc = min(per_candidate_time, remaining() - 10)
        if t_alloc < 3:
            break
        a_opt, s_opt = optimize_lbfbs(a0, T_schedule, t_alloc, verbose=False)
        update_best(a_opt, f'LBFBS {name}')

    # --- Phase 2: LP-based refinement ---
    print(f"\n=== Phase 2: LP-based refinement ===")
    if best_seq is not None and remaining() > 60:
        p2_time = min(500.0, remaining() * 0.35)
        print(f"  Time budget: {p2_time:.0f}s, n={len(best_seq)}")
        a_lp, s_lp = optimize_lp(best_seq, p2_time, t_step=0.05, verbose=True)
        update_best(a_lp, 'LP')

    # --- Phase 3: Local search ---
    print(f"\n=== Phase 3: Local search ===")
    if best_seq is not None and remaining() > 50:
        p3_time = min(300.0, remaining() * 0.25)
        print(f"  Time budget: {p3_time:.0f}s")
        a_ls, s_ls = local_search(best_seq, p3_time, verbose=True)
        update_best(a_ls, 'LS')

    # --- Phase 4: L-BFGS-B refinement on best ---
    print(f"\n=== Phase 4: Final L-BFGS-B polish ===")
    if best_seq is not None and remaining() > 40:
        p4_time = min(150.0, remaining() * 0.2)
        print(f"  Time budget: {p4_time:.0f}s")
        T_fine = [10.0, 20.0, 50.0, 100.0]
        a_fin, s_fin = optimize_lbfbs(best_seq, T_fine, p4_time, verbose=True)
        update_best(a_fin, 'FINAL_LBFBS')

    # --- Phase 5: Try upsampled version ---
    if best_seq is not None and remaining() > 80:
        n_orig = len(best_seq)
        n_new = min(2 * n_orig, 1024)
        if n_new > n_orig:
            print(f"\n=== Phase 5: Upsampled to n={n_new} ===")
            a_up = upsample_sequence(best_seq, n_new)
            # Quick L-BFGS-B
            T_up = [5.0, 15.0, 30.0]
            t_up = min(120.0, remaining() - 20)
            a_up_opt, s_up = optimize_lbfbs(a_up, T_up, t_up, verbose=True)
            update_best(a_up_opt, f'UP_n{n_new}')

    # --- Phase 6: Final LP on best ---
    if best_seq is not None and remaining() > 30:
        print(f"\n=== Phase 6: Final LP ===")
        t_final = min(80.0, remaining() - 10)
        a_final_lp, s_final_lp = optimize_lp(best_seq, t_final, t_step=0.03, verbose=False)
        update_best(a_final_lp, 'FINAL_LP')

    # --- Final local search ---
    if best_seq is not None and remaining() > 20:
        t_lst = remaining() - 5
        print(f"\n=== Final local search ({t_lst:.0f}s) ===")
        a_last, s_last = local_search(best_seq, t_lst, verbose=False)
        update_best(a_last, 'LAST_LS')

    # --- Verify with evaluate_sequence ---
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    return [1.0] * 256
