"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent 
with peak-focused subgradients, LP refinement, and multi-scale initialization.
Key improvement over LP-only approach: much faster per-iteration (O(n log n) 
vs O(n^2) or worse), allowing more iterations in the time budget.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgrad_peak(a, k_star, n):
    """Compute subgradient of b[k_star] w.r.t. a.
    b[k] = sum_i a[i]*a[k-i] (valid range)
    db[k]/da[j] = 2*a[k-j] if 0<=k-j<n and j != k-j
                  2*a[j] if 2j=k (i.e., j=k/2)
                  a[k-j] if only one appearance valid (shouldn't happen for interior)
    """
    g = np.zeros(n)
    for j in range(n):
        k_minus_j = k_star - j
        if 0 <= k_minus_j < n:
            if k_minus_j == j:
                g[j] = 2.0 * a[j]
            else:
                g[j] = 2.0 * a[k_minus_j]
    return g


def compute_subgrad_smooth(a, b, n, top_k=5, beta=10.0):
    """Compute a smoothed subgradient using top-k peaks with softmax weighting."""
    # Get top-k peaks
    k_max = np.argsort(b)[-top_k:]
    # Softmax weights
    vals = b[k_max]
    vals_shifted = vals - vals.max()
    w = np.exp(beta * vals_shifted)
    w /= w.sum()
    
    g = np.zeros(n)
    for idx, k in enumerate(k_max):
        k = int(k)
        for j in range(n):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                if k_minus_j == j:
                    g[j] += w[idx] * 2.0 * a[j]
                else:
                    g[j] += w[idx] * 2.0 * a[k_minus_j]
    return g

def make_lp_structure(n):
    """Precompute indices for cross-conv LP constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

            f_new /= s

            obj_new = 2.0 * n * b_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = b_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.1)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.8
            if patience > 40:

def coordinate_descent(a0, time_limit, max_coord_steps=50):
    """Coordinate descent: for each coordinate, find optimal value that reduces max-conv."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()
    
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    # Precompute for each k, which (i, k-i) pairs contribute
    # For efficiency, just do direct updates
    
    coord_order = np.arange(n)
    for step in range(max_coord_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        rng.shuffle(coord_order)
        for j in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try perturbing a[j]
            # The effect on b[k] for k in [j, j+n-1]:
            # b[k] changes by: (a[j]+delta)*a[k-j] - a[j]*a[k-j] = delta*a[k-j] (for first factor)
            # plus for the second factor: a[k-(j+delta)]... no, this is discrete
            # Actually b[k] = sum_i a[i]*a[k-i], so changing a[j] by delta:
            # affects b[k] for k = j + i (where a[i] is the other factor)
            # delta_b[k] = delta * a[k-j] (for k-j in [0,n-1])
            #           + delta * a[j] (for k-j = j, i.e., k=2j) -- already included above
            # Wait, more carefully:
            # b[k] = ... + a[j]*a[k-j] + ... + a[k-j]*a[j] + ...
            # If we change a[j] to a[j]+delta:
            # New b[k] = ... + (a[j]+delta)*a[k-j] + ... + a[k-j]*(a[j]+delta) + ...
            # = b[k] + delta*a[k-j] + delta*a[k-j] = b[k] + 2*delta*a[k-j]
            # (for k-j valid and k-j != j)
            # If k-j = j (k=2j): b[k] = ... + a[j]^2 + ...
            # New: (a[j]+delta)^2 = a[j]^2 + 2*delta*a[j] + delta^2
            # So effect is 2*delta*a[j] + delta^2
            
            # Try a few delta values
            a_save = a[j]
            for delta_frac in [-0.3, -0.1, 0.1, 0.3, -0.5, 0.5]:
                a[j] = max(0.0, a_save * (1 + delta_frac))
                # Quick check: recompute only affected b values
                # For speed, just do full conv every few steps
                if j % 32 == 0 or delta_frac in [-0.3, 0.3]:
                    b_new = fast_convolve(a)
                    obj_new = 2.0 * n * b_new.max()
                    if obj_new < best_obj - 1e-10:
                        best_obj = obj_new
                        best_a = a.copy()
                        b = b_new
                        break
            a[j] = a_save  # revert if not improved
    
    return best_a, best_obj


def gen_initializations(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    # Gaussian bumps
    for sf in [0.08, 0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    # Two bumps
    for pos in [0.2, 0.3]:
        two_bump = np.exp(-0.5 * ((x - pos * n) / (0.08 * n)) ** 2) + \
                   np.exp(-0.5 * ((x - (1-pos) * n) / (0.08 * n)) ** 2)
        two_bump /= two_bump.sum()
        inits[f'tb{int(pos*100)}'] = two_bump
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb /= cb.sum()
    inits['cos'] = cb
    
    # Random (Dirichlet with low concentration)
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Edge-heavy
    edge = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.4 * n), 0.0) ** 0.5
    edge /= edge.sum()
    inits['edge'] = edge
    
    return inits

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgrad_peak(a, k_star, n):
    """Subgradient of b[k_star] w.r.t. a."""
    g = np.zeros(n)
    for j in range(n):
        kmj = k_star - j
        if 0 <= kmj < n:
            if kmj == j:
                g[j] = 2.0 * a[j]
            else:
                g[j] = 2.0 * a[kmj]
    return g


def mirror_descent_peak(a0, time_limit, eta0=0.4, top_k=1, beta=10.0,

def make_lp_structure(n):
    """Precompute indices for LP constraint matrix (2n-1 x n)."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=8):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    b = fast_convolve(f)
    M = b.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        # Slightly tighten M each iteration for more aggressive moves
        M_eff = M * (1.0 - 0.001 * min(iter_count, 50))

        g = solve_lp(f, M_eff, rows, cols, n, timeout=6)
        if g is None:
            g = solve_lp(f, M, rows, cols, n, timeout=6)
            if g is None:
                break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            break

        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.15 * t, 0.05 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            b_new = fast_convolve(f_new)
            obj_new = 2.0 * n * b_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = b_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.12)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.75
            if patience > 40:
                # Perturb to escape
                rng_esc = np.random.default_rng(iter_count)
                f = f + rng_esc.normal(0, 0.01 * t, n)
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f /= s
                    b = fast_convolve(f)
                    M = b.max()
                    if 2.0 * n * M < best_obj:
                        best_obj = 2.0 * n * M
                        best_f = f.copy()
                patience = max(0, patience - 20)
                t = max(t, 0.3)
            if patience > 80:
                break

    return best_f, best_obj


def parametric_objective(params, n, basis):
    """Evaluate objective for parametric sequence a[j] = max(0, basis @ params)."""
    a = basis @ params
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return 1e6
    a /= a.sum()
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def parametric_search(n, time_limit, seed=42):
    """Optimize a spline-parametrized sequence."""
    rng = np.random.default_rng(seed)
    # Use RBF basis with diverse centers
    n_basis = min(20, n // 8)
    centers = np.sort(rng.random(n_basis))
    widths = 0.05 + 0.2 * rng.random(n_basis)

    x = np.linspace(0, 1, n)
    basis = np.zeros((n, n_basis))
    for i, (c, w) in enumerate(zip(centers, widths)):
        basis[:, i] = np.exp(-0.5 * ((x - c) / w) ** 2)
    # Add a constant basis function
    basis = np.hstack([basis, np.ones((n, 1))])
    n_params = n_basis + 1

    def obj(params):
        return parametric_objective(params, n, basis)

    # Multistart Nelder-Mead
    best_x = None
    best_f = np.inf
    n_starts = max(3, int(time_limit // 15))

    t0 = time.monotonic()
    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        x0 = rng.random(n_params) * 0.5 + 0.25
        res = optimize.minimize(obj, x0, method='Nelder-Mead',
                                options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.fun < best_f:
            best_f = res.fun
            best_x = res.x.copy()

    if best_x is None:
        return np.ones(n) / n, 2.0

    a = basis @ best_x
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a, best_f


def gen_initializations(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # Two bumps
    for pos in [0.2, 0.3]:
        tb = np.exp(-0.5 * ((x - pos * n) / (0.08 * n)) ** 2) + \
             np.exp(-0.5 * ((x - (1 - pos) * n) / (0.08 * n)) ** 2)
        tb /= tb.sum()
        inits[f'tb{int(pos * 100)}'] = tb

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb /= cb.sum()
    inits['cos'] = cb

    # Random
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.3)

    # Edge-heavy
    edge = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.45 * n), 0.0) ** 0.7
    edge /= edge.sum()
    inits['edge'] = edge

    # Quadratic residue inspired (for prime-like structure)
    if n > 10:
        qr = np.zeros(n)
        for i in range(int(np.sqrt(n))):
            qr[(i * i) % n] += 1.0
        qr = qr + 0.1
        qr /= qr.sum()
        inits['qr'] = qr

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: multi-strategy optimization combining mirror descent,
    LP alternating, and parametric search across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(score):
                best_score = score
                best_seq = p.copy()
                print(f"Parent: n={len(p)}, score={score:.6f}")
    except Exception:
        pass

    all_results = []

    # === Phase 1: Quick mirror descent screening ===
    print("Phase 1: Mirror descent screening")
    lengths = [256, 512, 1024]
    screened = []

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        inits = gen_initializations(n, rng)

        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()

        remaining = deadline - time.monotonic() - 300
        time_per = max(4.0, remaining / max(1, len(inits) * len(lengths)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(time_per * 1.2, deadline - time.monotonic() - 150)
            if t_alloc < 3:
                break

            f_opt, obj_opt = mirror_descent_peak(
                a0, time_limit=t_alloc, eta0=0.3, top_k=1,
                patience_limit=60, noise_prob=0.02, seed=hash((n, name)) % 100000
            )
            score = float(evaluate_sequence(f_opt.tolist()))

            if np.isfinite(score):
                screened.append((score, n, name, f_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:6s} score={score:.6f}")

    screened.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # === Phase 2: LP refinement on top candidates ===
    top_k_results = min(8, len(screened))
    if top_k_results > 0:
        remaining = deadline - time.monotonic() - 120
        if remaining > 60:
            print(f"Phase 2: LP refinement on top {top_k_results}, {remaining:.0f}s left")
            t_each = remaining / top_k_results
            for idx in range(top_k_results):
                if time.monotonic() > deadline - 30:
                    break
                sc, n, name, f = screened[idx]
                t_alloc = min(t_each, deadline - time.monotonic() - 30)
                if t_alloc < 8:
                    break

                # LP phase
                f2, obj2 = lp_optimize(f, time_limit=t_alloc * 0.65)
                sc2 = float(evaluate_sequence(f2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = f2.copy()
                    print(f"  LP[{idx}] n={n} {name} score={sc2:.6f}")

                # Mirror polish
                if time.monotonic() < deadline - 10:
                    tr = min(t_alloc * 0.35, deadline - time.monotonic() - 10)
                    f3, _ = mirror_descent_peak(f2, time_limit=tr,
                                                eta0=0.1, top_k=3, beta=20.0,
                                                patience_limit=60, seed=idx * 7 + 1)
                    sc3 = float(evaluate_sequence(f3.tolist()))
                    if np.isfinite(sc3) and sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"  Pol[{idx}] n={n} {name} score={sc3:.6f}")

    # === Phase 3: Parametric search for diverse exploration ===
    if time.monotonic() < deadline - 60:
        print("Phase 3: Parametric search")
        for n in [256, 512]:
            if time.monotonic() > deadline - 30:
                break
            t_param = min(40.0, deadline - time.monotonic() - 30)
            if t_param < 10:
                break
            a_pars, obj_pars = parametric_search(n, time_limit=t_param, seed=seed + n)
            # Polish the parametric result
            a_pol, _ = mirror_descent_peak(a_pars, time_limit=min(15.0,
                                deadline - time.monotonic() - 10),
                                eta0=0.2, top_k=1, patience_limit=50)
            sc = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Param n={n} score={sc:.6f}")

    # === Phase 4: Intensive final optimization on best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final intensive optimization")
        f_best = best_seq.copy()
        f_best = np.maximum(f_best, 0.0)
        f_best /= f_best.sum()
        n_best = len(f_best)

        tr = deadline - time.monotonic() - 10

        # Extended LP
        t_lp = min(tr * 0.5, 120.0)
        if t_lp > 10:
            f_final, _ = lp_optimize(f_best, time_limit=t_lp)
            sc = float(evaluate_sequence(f_final.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f_final.copy()
                f_best = f_final.copy()
                print(f"  Final LP score={sc:.6f}")

        tr = deadline - time.monotonic() - 5

        # Multiple mirror descent rounds with different params
        for params in [(0.1, 1, 5.0), (0.05, 3, 15.0), (0.03, 5, 30.0),
                       (0.08, 2, 10.0), (0.02, 1, 8.0)]:
            if tr < 3:
                break
            eta_p, tk, beta_p = params
            t_round = min(tr / 2.5, 20.0)
            if t_round < 2:
                break
            f_r, _ = mirror_descent_peak(f_best, time_limit=t_round,
                                         eta0=eta_p, top_k=tk, beta=beta_p,
                                         patience_limit=80, seed=42 + tk * 100)
            sc_r = float(evaluate_sequence(f_r.tolist()))
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = f_r.copy()
                f_best = f_r.copy()
                print(f"  Final MD eta={eta_p} tk={tk} score={sc_r:.6f}")
            tr = deadline - time.monotonic() - 5

    # === Phase 5: Length resampling ===
    if best_seq is not None and time.monotonic() < deadline - 8:
        print("Phase 5: Length resampling")
        n_cur = len(best_seq)
        for scale in [2, 0.5]:
            n_try = int(n_cur * scale)
            if n_try < 64 or n_try > 2048:
                continue
            if time.monotonic() > deadline - 5:
                break
            f_res = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_cur), best_seq)
            f_res = np.maximum(f_res, 0.0)
            f_res /= f_res.sum()
            t_res = min(20.0, deadline - time.monotonic() - 5)
            f_new, _ = mirror_descent_peak(f_res, time_limit=t_res,
                                           eta0=0.2, top_k=1, patience_limit=50)
            sc_new = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_seq = f_new.copy()
                print(f"  Resample n={n_try} score={sc_new:.6f}")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
