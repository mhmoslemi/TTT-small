"""
Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines LP-based refinement, anti-palindrome
gradient steps, and targeted local search. Exploits the reversal structure
of the center peak for direct gradient computation.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_sparse_conv_matrix(a):
    """Build sparse convolution matrix where A @ g = conv(a, g)."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For each k (row), the non-zero entries are at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve_fast(a, M, active_lags=None):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for active lags, g >= 0.
    If active_lags is None, use all lags.
    """
    n = len(a)
    A = build_sparse_conv_matrix(a)
    
    if active_lags is not None:
        A = A[active_lags, :]
        b = np.full(len(active_lags), M)
    else:
        b = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=True, use_active=True):
    """
    LP-based iterative refinement with active set and line search.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = fft_convolve(a, a)
    best_peak = float(np.max(best_conv))
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        conv_a = fft_convolve(a, a)
        M = float(np.max(conv_a))
        
        # Determine active lags (those within 5% of max)
        if use_active:
            threshold = M * 0.95
            active_mask = conv_a >= threshold
            active_lags = np.where(active_mask)[0].tolist()
            if len(active_lags) > 50:
                # Use top 50
                top_idx = np.argsort(conv_a)[-50:]
                active_lags = top_idx.tolist()
        else:
            active_lags = None
        
        g = lp_solve_fast(a, M, active_lags)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g = normalize_seq(g, S)
        
        # Line search for best blend
        best_t = 0.0
        best_new_peak = M
        ts = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]
        for t_try in ts:
            b = (1 - t_try) * a + t_try * g
            b = np.maximum(b, 0.0)
            b = normalize_seq(b, S)
            new_peak = float(np.max(fft_convolve(b, b)))
            if new_peak < best_new_peak:
                best_new_peak = new_peak
                best_t = t_try
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = normalize_seq(a, S)
        
        new_conv = fft_convolve(a, a)
        new_peak = float(np.max(new_conv))
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a.copy()
        
        if verbose and iteration % 50 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    LP iter {iteration}: peak={best_peak:.6f}, score={score:.6f}, t={best_t:.3f}, active={len(active_lags)}")
        
        if best_t < 0.01:
            break
    
    return best_a


def anti_palindrome_step(a, eta, time_check=None):
    """
    Multiplicative anti-palindrome step: pushes a toward being anti-symmetric
    about the center, reducing the center peak a^T J a.
    """
    n = len(a)
    S = np.sum(a)
    
    # Compute the "anti-palindrome gradient": for each pair (i, n-1-i),
    # reduce the larger one and increase the smaller one.
    a_new = a.copy()
    for i in range(n // 2):
        j = n - 1 - i
        diff = a[i] - a[j]
        # Multiplicative update
        a_new[i] = a[i] * np.exp(-eta * diff)
        a_new[j] = a[j] * np.exp(eta * diff)
    
    # If n is odd, the center element stays
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, S)
    return a_new


def vectorized_anti_palindrome(a, eta):
    """Vectorized anti-palindrome step."""
    n = len(a)
    S = np.sum(a)
    
    # Create the reversal
    a_rev = a[::-1]
    
    # Compute log-ratio: for pair (i, n-1-i), the correction is proportional to a[i] - a[n-1-i]
    # a_new[i] = a[i] * exp(-eta * (a[i] - a[n-1-i]))
    # a_new[n-1-i] = a[n-1-i] * exp(eta * (a[i] - a[n-1-i]))
    
    diff = a - a[::-1]  # diff[i] = a[i] - a[n-1-i]
    
    a_new = a * np.exp(-eta * diff)
    
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, S)
    return a_new


def anti_palindrome_optimize(a, time_limit, verbose=True):
    """
    Iterative anti-palindrome optimization with adaptive eta.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = float(np.max(fft_convolve(a, a)))
    
    eta = 0.5
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = vectorized_anti_palindrome(a, eta)
        new_peak = float(np.max(fft_convolve(candidate, candidate)))
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = candidate.copy()
            a = candidate
            stall = 0
            eta = min(eta * 1.1, 5.0)
        else:
            a = candidate  # Still move, to explore
            stall += 1
            eta *= 0.95
        
        if stall > 50:
            # Revert to best and reduce eta
            a = best_a.copy()
            eta *= 0.5
            stall = 0
            if eta < 0.01:
                break
        
        step += 1
        if verbose and step % 200 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Anti-pal step {step}: score={score:.6f}, eta={eta:.3f}")
    
    return best_a


def coordinate_descent(a, time_limit, verbose=True, rng=None):
    """
    Coordinate descent: try small perturbations to individual elements
    that contribute to the top peaks.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = float(np.max(fft_convolve(a, a)))
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        top_k = 5
        peak_lags = np.argsort(conv)[-top_k:]
        
        # For each top peak, find contributing elements and try to reduce them
        improved = False
        for k in peak_lags:
            # Elements contributing to conv[k]: a[i] * a[k-i] for valid i
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            
            # Find the element with maximum contribution
            max_contrib = 0
            max_idx = -1
            for i in range(j_min, j_max + 1):
                contrib = a[i] * a[k - i]
                if contrib > max_contrib:
                    max_contrib = contrib
                    max_idx = i
            
            if max_idx >= 0:
                # Try reducing this element
                for factor in [0.8, 0.7, 0.5, 0.9]:
                    candidate = a.copy()
                    candidate[max_idx] *= factor
                    # Compensate by increasing a random other element
                    other = rng.integers(0, n)
                    while other == max_idx:
                        other = rng.integers(0, n)
                    candidate[other] *= (1.0 / factor) ** 0.5
                    
                    candidate = np.maximum(candidate, 0.0)
                    candidate = normalize_seq(candidate, S)
                    new_peak = float(np.max(fft_convolve(candidate, candidate)))
                    
                    if new_peak < best_peak:
                        best_peak = new_peak
                        best_a = candidate.copy()
                        a = candidate
                        improved = True
                        break
                    elif new_peak < conv[k] and rng.random() < 0.3:
                        a = candidate
        
        if not improved:
            # Try random perturbations
            for _ in range(10):
                candidate = a.copy()
                idx = rng.integers(0, n)
                candidate[idx] *= np.exp(rng.normal(0, 0.1))
                candidate = np.maximum(candidate, 0.0)
                candidate = normalize_seq(candidate, S)
                new_peak = float(np.max(fft_convolve(candidate, candidate)))
                if new_peak < best_peak:
                    best_peak = new_peak
                    best_a = candidate.copy()
                    a = candidate
        
        if verbose and int(time.monotonic() - t0) % 30 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Coord t={time.monotonic()-t0:.0f}s: score={score:.6f}")
    
    return best_a


def make_chirp(n, rng):
    """Create a chirp signal (linear frequency sweep)."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.5 + 1.0 * rng.random()
    a = 1.0 + 0.5 * np.cos(np.pi * alpha * t * t / n)
    a = np.clip(a, 0.01, None)
    return a


def make_sine(n, f, amp, phase, rng):
    """Create a sinusoidal modulation."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def smooth_anti_palindrome(n, rng):
    """
    Create a smooth anti-palindromic sequence:
    a[i] = 1 + c * sin(pi * i / n) which is anti-symmetric about n/2.
    """
    t = np.arange(n, dtype=np.float64)
    c = 0.3 + 0.4 * rng.random()
    a = 1.0 + c * np.sin(np.pi * t / n)
    a = np.clip(a, 0.01, None)
    return a
