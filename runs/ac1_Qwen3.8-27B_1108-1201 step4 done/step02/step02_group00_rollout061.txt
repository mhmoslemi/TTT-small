"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-domain (softmax) gradient
descent with Adam optimizer, log-sum-exp temperature annealing, multi-length
exploration, and periodic perturbation restarts. Key innovations over prior
approach: natural gradient on the simplex via softmax parameterization,
per-coordinate adaptive learning rates, and efficient gradient computation
via cross-correlation.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_logsumexp(a, conv, beta):
    """
    Compute the exact gradient of (1/beta)*log(sum_k exp(beta*conv[k]))
    with respect to a. Returns grad of the unscaled log-sum-exp (times beta
    is absorbed into the softmax weights).
    
    grad_a[i] = 2 * sum_k w[k] * a[k-i]
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    return grad_a, w


def optimize_logdomain(a_init, time_limit, beta_start=5.0, beta_end=5000.0,
                       lr_start=0.05, lr_end=0.002,
                       adam_b1=0.9, adam_b2=0.999, adam_eps=1e-8,
                       warmup_frac=0.05, perturb_interval=2000,
                       perturb_scale=0.02):

    Log-domain (softmax) gradient descent with Adam.
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    theta = np.log(a)
    theta -= theta.max()  # center for numerical stability
    
    m = np.zeros(n)
    v_arr = np.zeros(n)
    step = 0
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    stir = 0
    
    while time.monotonic() - t0 < time_limit:

        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)
        
        # Schedule learning rate with warmup
        if frac < warmup_frac:
            lr = lr_start * min(1.0, frac / warmup_frac)
        else:
            lr_progress = (frac - warmup_frac) / (1.0 - warmup_frac)
            lr = lr_start * (lr_end / lr_start) ** lr_progress
        
        # Compute gradient
        conv = fast_convolve(a)
        grad_a, w = compute_grad_logsumexp(a, conv, beta)
        
        # Convert to theta-space (natural gradient on simplex)
        dot_ag = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot_ag)
        
        # Adam update
        m = adam_b1 * m + (1.0 - adam_b1) * grad_theta
        v_arr = adam_b2 * v_arr + (1.0 - adam_b2) * grad_theta * grad_theta
        m_hat = m / (1.0 - adam_b1 ** step)

            a = a_new / s

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-12:
                continue
            k = int(peaks[ii])
            wi = w[ii] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 2000:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            lr = max(lr_init * 0.3, lr * 0.5)
        
        lr *= 0.9998
        temp *= 1.001
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # U-shaped (edges higher)
    u = 1.0 + 0.5 * (2.0 * x / (n - 1) - 1.0) ** 2
    u /= u.sum()
    inits['u_shaped'] = u
    
    # Inverted U (center higher)
    iu = 1.0 + 0.5 * (1.0 - (2.0 * x / (n - 1) - 1.0) ** 2)
    iu /= iu.sum()
    inits['inv_u'] = iu
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = cos_w / cos_w.sum()
    inits['cosine'] = cos_w
    
    # Raised cosine
    rc = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    rc = rc / rc.sum()
    inits['raised_cos'] = rc
    
    # Bimodal (two peaks at 1/4 and 3/4)
    bi = np.exp(-((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.75) / (n * 0.1)) ** 2)
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Edge concentration
    edge = np.zeros(n)
    edge[:max(1, n // 8)] = 1.0
    edge[-max(1, n // 8):] = 1.0
    edge /= edge.sum()
    inits['edge'] = edge
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using log-domain gradient descent
    with Adam, subgradient refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    # Phase 1: Quick exploration
    print("=== Phase 1: Exploration ===")
    lengths = [128, 256, 512, 1024]
    
    phase1_deadline = time.monotonic() + min(350.0, (deadline - time.monotonic()) * 0.2)
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        
        # Prioritize
        priority = ['parent', 'parent_up', 'uniform', 'u_shaped', 'cosine', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                    [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]
        
        time_per = max(15.0, (phase1_deadline - time.monotonic() - 20) / (len(init_names) * len(lengths)))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(time_per, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = optimize_logdomain(
                a0, time_limit=t_alloc,
                beta_start=5.0, beta_end=500.0,
                lr_start=0.05, lr_end=0.005,
                perturb_interval=max(500, n), perturb_scale=0.02)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  n={n}, {name}: obj={obj_opt:.8f}, score={score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 6:
            top_configs.append((n, name, a_opt))
            seen.add(key)
    
    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best_so_far', best_sequence.copy()))
    
    # Add fresh starts at best length
    best_n = len(best_sequence) if best_sequence is not None else 256
    fresh_names = ['u_shaped', 'cosine', 'bimodal']
    inits_b = generate_initializations(best_n, parent, rng)
    for fn in fresh_names:
        if fn in inits_b:
            top_configs.append((best_n, f'fresh_{fn}', inits_b[fn]))
    
    remaining = deadline - time.monotonic() - 80.0
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))
    
    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 50)
        if t_alloc < 10:
            break
        
        # Try two-phase approach
        a_opt, obj_opt = optimize_two_phase(
            a0, time_limit=t_alloc * 0.7)
        
        score = float(evaluate_sequence(a_opt.tolist()))
        method_tag = "two_phase"
        
        # If time remaining, also try subgradient
        t_remaining = deadline - time.monotonic() - 50
        if t_remaining > 15 and t_alloc * 0.3 > 10:
            a_sg, obj_sg = optimize_subgrad_refined(
                a_opt, time_limit=t_alloc * 0.3,
                lr_init=0.015, momentum=0.92,
                top_k=max(20, n // 20), temp_init=0.0002)
            
            score_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score_sg) and score_sg < score:
                score = score_sg
                a_opt = a_sg
                method_tag = "two_phase+sg"
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep {i}: n={n}, {name}, {method_tag}, score={score:.6f}")
    
    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        
        # Polish with very low learning rate and high beta
        polish_configs = [
            {'beta_start': 200.0, 'beta_end': 10000.0, 'lr_start': 0.01, 'lr_end': 0.0005},
            {'beta_start': 50.0, 'beta_end': 5000.0, 'lr_start': 0.02, 'lr_end': 0.001},
            {'beta_start': 1000.0, 'beta_end': 50000.0, 'lr_start': 0.005, 'lr_end': 0.0002},
        ]
        
        remaining = deadline - time.monotonic() - 20.0
        t_per = remaining / (len(polish_configs) + 3)
        
        for cfg in polish_configs:
            if time.monotonic() > deadline - 25:
                break
            a_opt, obj_opt = optimize_logdomain(
                best_sequence, time_limit=min(t_per, deadline - time.monotonic() - 20),
                perturb_interval=10000, perturb_scale=0.001,
                **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish: score={score:.6f}")
        
        # Small perturbation + re-optimize
        if time.monotonic() < deadline - 30:
            for _ in range(3):
                if time.monotonic() > deadline - 25:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = np.maximum(best_sequence + noise, 1e-10)
                a_pert = a_pert / a_pert.sum()
                t_p = min(30.0, deadline - time.monotonic() - 20)
                if t_p < 5:
                    break
                a_opt, obj_opt = optimize_logdomain(
                    a_pert, time_limit=t_p,
                    beta_start=500.0, beta_end=10000.0,
                    lr_start=0.01, lr_end=0.0005,
                    perturb_interval=10000, perturb_scale=0.001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb: score={score:.6f}")
        
        # Subgradient polish
        if time.monotonic() < deadline - 30:
            t_p = min(30.0, deadline - time.monotonic() - 20)
            if t_p > 5:
                a_opt, obj_opt = optimize_subgrad_refined(
                    best_sequence, time_limit=t_p,
                    lr_init=0.005, momentum=0.95,
                    top_k=n_best // 10, temp_init=0.0001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG polish: score={score:.6f}")
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
