```python
def fast_convolve(a):

def obj_and_grad(a, beta=50.0):
    """Soft-max objective and gradient on the simplex (sum=1, a>=0)."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    
    # Soft max
    scaled = beta * conv
    scaled_max = scaled.max()
    log_sum_exp = scaled_max / beta + np.log(np.exp(scaled - scaled_max).sum()) / beta
    # Actually: logsumexp(β*conv) = β * (max(conv) + log(sum(exp(β*conv - β*max(conv)))/len))
    # Let me just compute it directly
    exp_conv = np.exp(beta * (conv - conv.max()))
    log_sum = conv.max() + np.log(exp_conv.sum()) / beta
    obj = 2.0 * n * log_sum  # since sum(a)=1, (sum a)^2 = 1
    
    # Gradient
    w = exp_conv / exp_conv.sum()  # softmax weights, sum to 1
    # d log_sum / da[i] = sum_k w[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n else 0
    grad = np.zeros(n)
    for k_idx in np.where(w > 1e-10)[0]:
        k = int(k_idx)
        wi = w[k_idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # grad[j] += 2 * w[k] * a[k-j] for j in [j_start, j_end]
        # a[k-j] for j in [j_start, j_end] is a[k-j_end : k-j_start+1][::-1]
        grad[j_start:j_end+1] += 2.0 * wi * a[k - j_end:k - j_start + 1][::-1]
    
    grad *= 2.0 * n  # from the 2n factor
    
    return obj, grad
```

Hmm, but the issue with L-BFGS-B and the simplex constraint is that L-BFGS-B only handles box constraints. The simplex constraint `sum(a)=1, a>=0` is an equality + inequality constraint.

I could:
1. Parameterize `a` in terms of `n-1` free variables (e.g., `a[n-1] = 1 - sum(a[0:n-1])`) - but this is awkward with L-BFGS-B.
2. Use the log-sum-exp parameterization: `a[i] = exp(z[i]) / sum(exp(z))` where `z` is free. This automatically satisfies the simplex constraint.
3. Use projected gradient methods (as before but with better convergence).

Option 2 is cleanest for L-BFGS-B:

Let me also think about what initializations might be good:

For the autocorrelation to be flat, we want the signal to have a "random" spectral character. But it must be non-negative. So we want a non-negative signal whose Fourier magnitude spectrum is as flat as possible.

Ideas:
- `a[i] = 1 + α*cos(2πf₁i + φ₁) + β*cos(2πf₂i + φ₂)` (sum of cosines, clipped to be non-negative)
- `a[i] = 1 + α*Walsh_Hadamard[i]` (clipped)
- Random non-negative with specific spectral shaping

Actually, a key insight: if `a[i] = 1/n + f[i]` where `f` has mean 0, then the DC component of the FFT of `a` is 1, and other components are the FFT of `f`. The autocorrelation will have a DC peak of `1/(2n-1)` plus the autocorrelation of `f`.

To minimize the peak, we want `f` to have a flat power spectrum, i.e., `f` should look like white noise (but with the constraint that `a >= 0`, i.e., `f[i] >= -1/n`).

So a good initialization might be: `a[i] = 1/n + c * w[i]` where `w` is white noise with `min(w) >= -1/(nc)` (ensuring non-negativity).

Or: `a[i] = (1/n) * (1 + t * (2*U[i] - 1))` where `U` is uniform [0,1] and `t` is a parameter controlling the "noisiness" (t=0 gives uniform, t=1 gives [0, 2/n]).

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with softmax parameterization, Adam optimizer, simulated annealing,
and multi-start with diverse initializations including spectral shaping.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0]*a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def softmax_param_obj_grad(z, beta):
    """Objective and gradient in softmax parameterization."""
    n = len(z)
    # Compute a from z via softmax
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = fast_convolve(a)
    conv_max = conv.max()
    scaled = beta * (conv - conv_max)
    exp_s = np.exp(scaled)
    exp_s_sum = exp_s.sum()
    w = exp_s / exp_s_sum
    
    log_sum = conv_max + np.log(exp_s_sum) / beta
    obj = 2.0 * n * log_sum
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k_idx in active:
        k = int(k_idx)
        wi = w[k_idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        grad_a[j_start:j_end+1] += 2.0 * wi * a[k - j_end:k - j_start + 1][::-1]
    grad_a *= 2.0 * n
    
    # Chain rule for softmax
    a_dot_grad = a @ grad_a
    grad_z = a * (grad_a - a_dot_grad)
    
    return obj, grad_z

def optimize_lbgfsb(a0, time_limit, betas=None):
    """L-BFGS-B optimization using softmax parameterization."""
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if betas is None:
        betas = [50, 100, 200, 500, 1000]
    
    t0 = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        z = np.log(a + 1e-30)  # Initialize z from current a
        
        def fun(z_input):
            return softmax_param_obj_grad(z_input, beta)[0]
        
        def fun_and_grad(z_input):
            return softmax_param_obj_grad(z_input, beta)
        
        t_remaining = time_limit - (time.monotonic() - t0)
        maxiter = max(10, int(200 * t_remaining / time_limit))
        
        try:
            result = minimize(fun_and_grad, z, method='L-BFGS-B',
                            options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10,
                                    'maxls': 50})
            
            z_opt = result.x
            a_opt = np.exp(z_opt - z_opt.max())
            a_opt = a_opt / a_opt.sum()
            obj_opt = compute_obj(a_opt)
            
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                a = a_opt  # Update starting point for next beta
        except Exception:
            pass
    
    return best_a, best_obj

def adam_optimize(a0, time_limit, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                  multi_peak=5, temperature=0.01):
    """Adam optimizer with projected gradient on simplex."""
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    no_improve = 0
    lr_schedule = [lr]
    current_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        conv = fast_convolve(a)
        
        if multi_peak > 1:
            # Weighted subgradient over top peaks
            top_k = np.argpartition(conv, -multi_peak)[-multi_peak:]
            peak_vals = conv[top_k]
            scaled = peak_vals / temperature
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
            
            grad = np.zeros(n)
            for ii in range(multi_peak):
                k = int(top_k[ii])
                wi = w[ii]
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                grad[j_start:j_end+1] += 2.0 * wi * a[k - j_end:k - j_start + 1][::-1]
        else:
            k_max = int(np.argmax(conv))
            grad = np.zeros(n)
            j_start = max(0, k_max - n + 1)
            j_end = min(n - 1, k_max)
            grad[j_start:j_end+1] = 2.0 * a[k_max - j_end:k_max - j_start + 1][::-1]
        
        # Project to tangent space
        grad_tangent = grad - np.mean(grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_tangent
        v = beta2 * v + (1 - beta2) * grad_tangent ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive LR
        if no_improve % 500 == 0 and no_improve > 0:
            current_lr *= 0.8
            if current_lr < 0.001:
                # Restart with perturbation
                rng = np.random.default_rng(t)
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                current_lr = lr
                no_improve = 0
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, t_init=1.0, t_min=0.001, cooling=0.9999):
    """Simulated annealing on the simplex."""
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    current_obj = compute_obj(a)
    best_obj = current_obj
    best_a = a.copy()
    
    temp = t_init
    
    while time.monotonic() - t0 < time_limit and temp > t_min:
        # Propose: random perturbation
        delta = rng.standard_normal(n) * 0.01
        a_new = project_simplex(a + delta)
        
        obj_new = compute_obj(a_new)
        delta_obj = obj_new - current_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / temp):
            a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
        
        temp *= cooling
    
    return best_a, best_obj

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Noisy uniform (different noise levels)
    for noise_level in [0.3, 0.5, 0.7, 0.9]:
        noise = rng.standard_normal(n) * noise_level
        a = 1.0 + noise
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits[f'noisy_{noise_level}'] = a / s
    
    # Cosine-weighted
    x = np.arange(n, dtype=np.float64)
    for f in [1, 2, 3, 5, 8]:
        for alpha in [0.3, 0.5, 0.7]:
            a = 1.0 + alpha * np.cos(2 * np.pi * f * x / n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                inits[f'cos_f{f}_a{alpha}'] = a / s
    
    # Multi-frequency
    for trial in range(3):
        freqs = rng.integers(1, n//4+1, size=3)
        phases = rng.uniform(0, 2*np.pi, size=3)
        amps = rng.uniform(0.2, 0.6, size=3)
        a = np.ones(n)
        for fr, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(2*np.pi*fr*x/n + ph)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits[f'multi_{trial}'] = a / s
    
    # Ramps and triangles
    ramp = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = ramp[::-1] / ramp.sum()
    tri = np.minimum(x, n-1-x) + 1
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = n // 3
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.power(x + 1, p)
        inits[f'power_{p}'] = a / a.sum()
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.power(n - x + 1, p)
        inits[f'power_rev_{p}'] = a / a.sum()
    
    # From parent
    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent'] = p / s
        else:
            # Resample
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent_interp'] = p / s
    
    return inits
```

```python
    """Flatten the power spectrum of a using Gerchberg-Saxton style iteration."""
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        if n_iter is not None and iter_count >= n_iter:
            break
        iter_count += 1
        
        # Forward FFT
        A = np.fft.rfft(a)  # length n//2+1
        
        if target == 'geomean':
            # Geometric mean of magnitudes (excluding DC)
            mags = np.abs(A)
            # Use all magnitudes for the target
            target_mag = np.full(len(A), mags.mean())
            # Or use a smoothed version
        elif target == 'flat':
            target_mag = np.full(len(A), mags[0] / np.sqrt(2*n - 1))
            target_mag[0] = mags[0]  # Keep DC
        
        # Replace magnitudes, keep phases
        phases = np.angle(A)
        A_new = target_mag * np.exp(1j * phases)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, n)
        
        # Clip to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj
```

```python
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    L = 2 * n - 1

    phase_iter = 2000
    alpha_cur = alpha

        t_batch_start = time.monotonic()
        while time.monotonic() - t_batch_start < 1.0:  # ~1 second per batch

            A = np.fft.fft(a, L)

            # Target: compress the magnitude range

            target = mags.copy()

            # Compress: target[k] = mags[k]^alpha for k > 0
            # But we need to maintain the overall scale
```

```python
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines L-BFGS-B (softmax parameterization), spectral domain flattening,
Adam optimizer, and multi-start diverse initializations.
Key innovations over prior work: proper gradient via softmax reparameterization,
Gerchberg-Saxton spectral flattening, and adaptive multi-peak weighting.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return 2.0 * n * conv.max() / (s * s)


def conv_gradient(a, k):

    j_end = min(n - 1, k)
    grad[j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
    return grad
```
