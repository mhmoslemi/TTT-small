def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    # Cosine windows with different parameters
    for sf in [0.3, 0.5, 0.7, 1.0]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf * n * 0.5))**2)
        g = np.maximum(g, 1e-6)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    # Cosine shaped
    c = 0.5 * (1 + np.cos(2*np.pi*(x+0.5)/n))
    c = np.maximum(c, 1e-6)
    inits['cos'] = c / c.sum()
    
    # Triangle
    tri = 1.0 - 2.0*np.abs(x/n - 0.5)
    tri = np.maximum(tri, 0.01)
    inits['tri'] = tri / tri.sum()
    
    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'dir{int(alpha*10)}'] = r

            p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['par'] = p / ps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Smoothed objective (softmax approximation of max) with L-BFGS-B and t-annealing
2. Nesterov/momentum subgradient descent with periodic LP guidance
3. Targeted coordinate descent polishing (pairwise mass transfers)
4. Diverse initializations including parent sequence and multiple lengths
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_loss_and_grad(a, t):
    """Compute smoothed 2n*max(b)/sum(a)^2 and gradient using softmax approximation."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return 2.0 * n, np.zeros(n)
    b = fast_convolve(a)
    b_max = b.max()
    shifted = (b - b_max) / t
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_b / exp_sum
    M_smooth = b_max + t * np.log(exp_sum)
    loss = 2.0 * n * M_smooth / (S * S)
    # Gradient of M_smooth w.r.t. a_i = 2 * sum_k p[k] * a[k-i]
    grad_M = np.zeros(n)
    # Only iterate over significant p entries
    thresh = p.max() * 1e-8
    active = np.where(p > thresh)[0]
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_M[lo:hi + 1] += p[k] * 2.0 * a[k - lo:k - hi + 1][::-1]
    # d(loss)/d(a_i) = 2n * (grad_M[i] * S - 2*M_smooth) / S^3
    grad_loss = 2.0 * n * (grad_M * S - 2.0 * M_smooth) / (S * S * S)
    return loss, grad_loss


def lbfgsb_optimize(a0, time_limit):
    """Optimize using L-BFGS-B with t-annealing on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    n = len(a)

    def loss_fn(a_flat):
        a_pos = np.maximum(a_flat, 0.0)
        return smooth_loss_and_grad(a_pos, t_current[0])[0]

    def grad_fn(a_flat):
        a_pos = np.maximum(a_flat, 0.0)
        return smooth_loss_and_grad(a_pos, t_current[0])[1]

    t_current = [1.0]
    best_a = a.copy()
    best_score = float('inf')
    t0 = time.monotonic()

    # Anneal t from 1.0 down to 0.005
    t_schedule = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    remaining_per = time_limit / len(t_schedule)

    for t_val in t_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        t_current[0] = t_val
        phase_time = min(remaining_per, time_limit - (time.monotonic() - t0))
        if phase_time < 1:
            break

        def obj(x):
            return smooth_loss_and_grad(np.maximum(x, 0.0), t_val)[0]

        def obj_grad(x):
            return smooth_loss_and_grad(np.maximum(x, 0.0), t_val)[1]

        try:
            res = optimize.minimize(obj, a, jac=obj_grad, method='L-BFGS-B',
                                    bounds=[(0, None)] * n,
                                    options={'maxiter': 200, 'maxfun': 5000,
                                             'ftol': 1e-12, 'gtol': 1e-8,
                                             'maxls': 50})
            if res.fun < best_score:
                best_score = res.fun
                best_a = np.maximum(res.x, 0.0)
                a = best_a.copy()
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        print(f"    L-BFGS t={t_val:.4f}, loss={best_score:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_score

    best_a = a.copy()

    v = np.zeros(n)

    eta = 0.03
    t0 = time.monotonic()
    patience = 0
    it = 0

    if use_lp and n <= 600:
        r, c = [], []
        for i in range(n):
            for j in range(n):
                r.append(i + j)
                c.append(i)
        lp_rows = np.array(r, dtype=np.int32)
        lp_cols = np.array(c, dtype=np.int32)

    while time.monotonic() - t0 < time_limit:

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= 0.999 * max_b)[0]

        sg = np.zeros(n)

        for k in active:

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)

        v = mu * v - eta * sg
        a_new = project_simplex(a + v)

        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.8
                mu = max(0.5, mu * 0.98)
                if eta < 1e-6:
                    break
            if patience > 500:
                break

        # Periodic LP
        if use_lp and lp_rows is not None and it % 150 == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 20:
                try:
                    M = b.max()
                    vals = np.tile(a, n)
                    A = sparse.csc_matrix((vals, (lp_rows, lp_cols)), shape=(2 * n - 1, n))
                    res = optimize.linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                                           bounds=[(0, None)] * n, method='highs',
                                           options={'disp': False, 'time_limit': 3})
                    if res.success:
                        g = np.maximum(res.x, 0.0)
                        gs = g.sum()
                        if gs > 1e-12:
                            g /= gs
                            a_lp = project_simplex(0.4 * a + 0.6 * g)
                            b_lp = fast_convolve(a_lp)
                            obj_lp = 2.0 * n * b_lp.max()
                            if obj_lp < best_obj:
                                best_obj = obj_lp
                                best_a = a_lp.copy()
                                a = a_lp
                                v = np.zeros(n)
                except Exception:
                    pass

    return best_a, best_obj


def coord_descent_polish(a0, time_limit, seed=42):
    """Targeted coordinate descent: transfer mass between pairs to reduce peak autocorrelation."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(a)
        peak_lag = int(np.argmax(b))

        # Focus on pairs related to the peak lag
        n_trials = min(500, max(100, int((time_limit - (time.monotonic() - t0)) * 50)))
        for _ in range(n_trials):
            # Biased selection: prefer coordinates near peak
            r = rng.random()
            if r < 0.6:
                i = rng.integers(max(0, peak_lag - n + 1), min(n, peak_lag + 1))
            else:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            ai, aj = a[i], a[j]
            if ai < 1e-12 and aj < 1e-12:
                continue

            # Try transferring mass
            max_transfer = min(ai, aj) * 0.5
            best_t = 0.0
            current_obj = 2.0 * n * b.max() if _ == 0 else None
            if current_obj is None:
                current_obj = 2.0 * n * b.max()

            for frac in [0.2, 0.5, 0.8]:
                t_val = frac * max_transfer
                # i -> j
                a[i] -= t_val
                a[j] += t_val
                b_t = fast_convolve(a)
                obj_t = 2.0 * n * b_t.max()
                if obj_t < current_obj - 1e-14:
                    best_t = t_val
                else:
                    a[i] += t_val
                    a[j] -= t_val
                # j -> i
                a[i] += t_val
                a[j] -= t_val
                b_t2 = fast_convolve(a)
                obj_t2 = 2.0 * n * b_t2.max()
                if obj_t2 < current_obj - 1e-14:
                    best_t = -t_val
                else:
                    a[i] -= t_val
                    a[j] += t_val

            if abs(best_t) > 1e-12:
                a[i] += best_t
                a[j] -= best_t
                a = np.maximum(a, 0.0)
                a = project_simplex(a)
                b = fast_convolve(a)
                obj = 2.0 * n * b.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                peak_lag = int(np.argmax(b))

        if not improved:
            stagnation += 1
            if stagnation > 30:
                break
        else:
            stagnation = 0

    return best_a, best_obj


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.25, 0.4, 0.6, 0.8]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n * 0.5)) ** 2)
        g = np.maximum(g, 1e-8)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    c = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n))
    c = np.maximum(c, 1e-8)
    inits['cos'] = c / c.sum()

    # Double-humped
    d = 0.5 * (1.0 + np.cos(4.0 * np.pi * (x + 0.5) / n))
    d = np.maximum(d, 1e-8)
    inits['dble'] = d / d.sum()

    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 0.02)
    inits['tri'] = tri / tri.sum()

    for alpha in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'dir{int(alpha * 10)}'] = r

    # Edge-concentrated
    edge = np.maximum(np.abs(x / n - 0.5) * 2.0, 0.05)
    inits['edge'] = edge / edge.sum()

    if parent is not None:
        if parent.size >= n:
            p = np.maximum(parent[:n], 0.0)
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            inits['par'] = p / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: multi-phase optimization for minimum peak autocorrelation."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    lengths = [256, 384, 512, 768, 1024]
    all_results = []

    # Phase 1: L-BFGS-B with t-annealing (fast, smooth)
    print("Phase 1: L-BFGS-B optimization with t-annealing")
    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = gen_inits(n, rng, parent)
        remaining = deadline - time.monotonic() - 600
        time_per = max(3.0, remaining / max(1, len(inits) * len(lengths)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 300:
                break
            ta = min(time_per, deadline - time.monotonic() - 300)
            if ta < 2:
                break
            f_opt, loss_opt = lbfgsb_optimize(a0, time_limit=ta)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:10s} score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Subgradient + LP
    all_results.sort(key=lambda x: x[0])
    remaining = deadline - time.monotonic() - 120
    if remaining > 60 and all_results:
        print(f"Phase 2: Subgradient + LP on top results, {remaining:.0f}s left")
        top = all_results[:8]
        t_each = remaining / len(top)
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 60:
                break
            ta = min(t_each, deadline - time.monotonic() - 60)
            if ta < 5:
                break
            use_lp = (n <= 512) and (ta > 30)
            f2, obj2 = subgrad_optimize(f, time_limit=ta, use_lp=use_lp)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                all_results[idx] = (sc2, n, name, f2)
                print(f"  Subgrad [{idx}] n={n} {name} score={sc2:.6f}")

    # Phase 3: Coordinate descent polish
    all_results.sort(key=lambda x: x[0])
    remaining = deadline - time.monotonic() - 60
    if remaining > 30 and all_results:
        print(f"Phase 3: Coordinate descent polish, {remaining:.0f}s left")
        top = all_results[:5]
        t_each = remaining / len(top)
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 20:
                break
            ta = min(t_each, deadline - time.monotonic() - 20)
            if ta < 3:
                break
            f3, obj3 = coord_descent_polish(f, time_limit=ta, seed=100 + idx)
            sc3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(sc3) and sc3 < best_score:
                best_score = sc3
                best_seq = f3.copy()
                print(f"  CD [{idx}] n={n} {name} score={sc3:.6f}")

    # Phase 4: Final combined polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 5:
        tr = deadline - time.monotonic() - 5
        print(f"Phase 4: Final polish, {tr:.0f}s")
        # Subgradient round
        if tr > 15:
            f4, _ = subgrad_optimize(best_seq, time_limit=tr * 0.4, use_lp=(len(best_seq) <= 512))
            sc4 = float(evaluate_sequence(f4.tolist()))
            if np.isfinite(sc4) and sc4 < best_score:
                best_score = sc4
                best_seq = f4.copy()
                print(f"  Final subgrad score={sc4:.6f}")
        # CD round
        if time.monotonic() < deadline - 3:
            tr2 = deadline - time.monotonic() - 3
            f5, _ = coord_descent_polish(best_seq, time_limit=tr2, seed=999)
            sc5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_seq = f5.copy()
                print(f"  Final CD score={sc5:.6f}")

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
