The idea: We want to minimize `max_k conv[k]`. This is equivalent to minimizing `M` subject to `conv[k] <= M`. Using IRLS:

5. This is a QP (quadratic objective, linear constraints)
6. Update `a`
7. Repeat

The QP in step 4 can be solved with standard QP solvers. Let me check if this would be effective.

```python
def sqp_step(a, M):
    """One SQP iteration: solve QP to reduce max convolution."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Linearize: conv[k] ≈ c[k] + J[k,:] @ da where da = a_new - a
    # c[k] = conv[k], J[k,i] = ∂conv[k]/∂a[i]
    
    # For the QP, we want to minimize M_new
    # s.t. c[k] + J[k,:] @ da <= M_new for all k
    #      a + da >= 0
    #      sum(da) = 0 (maintain sum=1)
    
    # This is: minimize M_new
    # s.t. J[k,:] @ da - M_new <= -c[k] for all k
    #      da >= -a
    #      sum(da) = 0
    
    # Variables: da[0..n-1], M_new
    # This is an LP!
    
    # Let's set it up:
    # minimize M_new
    # s.t. for each k: sum_i J[k,i]*da[i] - M_new <= -c[k]
    #      for each i: da[i] >= -a[i]
    #      sum_i da[i] = 0
    
    # Build the constraint matrix
    # Number of variables: n + 1 (da[0..n-1] and M_new)
    # Number of inequality constraints: (2n-1) [conv constraints] + n [non-negativity]
    # Equality constraints: 1 (sum = 0)
    
    # For the conv constraints:
    # sum_i J[k,i]*da[i] - M_new <= -c[k]
    # J[k,0..n-1] * da[0..n-1] + [-1]*M_new <= -c[k]
    
    # Compute J (Jacobian of conv with respect to a)
    J = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        idx1 = k - np.arange(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        J[k, valid1] += a[idx1[valid1]]
        idx2 = np.arange(n) - k
        valid2 = (idx2 >= 0) & (valid2 < n)
        J[k, valid2] += a[idx2[valid2]]
    
    # Set up the LP
    # c_obj = [0, 0, ..., 0, 1] (minimize M_new)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Conv constraints: J[k,:] @ da - M_new <= -c[k]
    A_conv = np.hstack([J, -np.ones((2*n-1, 1))])
    b_conv = -conv
    
    # Non-negativity: da[i] >= -a[i] → -da[i] <= a[i]
    A_nn = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_nn = a.copy()
    
    A_ub = np.vstack([A_conv, A_nn])
    b_ub = np.concatenate([b_conv, b_nn])
    
    # Equality: sum(da) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: da[i] is free (bounded implicitly by non-negativity constraint)
    # M_new >= 0
    bounds = [(None, None)] * n + [(0, None)]
    
    res = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs', options={'disp': False})
    
    if res.success:
        da = res.x[:n]
        M_new = res.x[n]
        return a + da, M_new
    
    return a, M
```

```python
    iter_count = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        a_sq, _ = sqp_step(a)
        if a_sq is None:
            break

        # Check if SQP solution is essentially the same
        if np.max(np.abs(a_sq - a)) < 1e-14:
            patience += 1
            if patience > 5:
                break
            continue

        # Backtracking line search
        improved = False
        for t_try in [t, 0.8 * t, 0.6 * t, 0.4 * t, 0.2 * t, 0.1 * t]:
            a_try = (1.0 - t_try) * a + t_try * a_sq
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-12:
                continue
            a_try /= s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-14:
                a = a_try
                best_obj = obj_try
                best_a = a.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                if verbose and iter_count % 20 == 0:
                    print(f"      SQP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}")
                break

        if not improved:
            patience += 1
            t *= 0.7
            if patience > 40:
                break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, K=10, eta0=0.03):
    """Top-K subgradient optimization with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    momentum = np.zeros(n)
    mu = 0.8
    patience = 0
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            patience = 0
            eta = min(eta * 1.02, 0.3)
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.85
                mu = max(0.5, mu * 0.99)
            if patience > 200:
                break

        # Compute top-K subgradient
        if K < len(conv):
            top_k = np.argpartition(conv, -K)[-K:]
        else:
            top_k = np.arange(len(conv))
        sg = np.zeros(n)
        for k in top_k:
            k = int(k)
            idx1 = k - np.arange(n)
            valid1 = (idx1 >= 0) & (idx1 < n)
            sg[valid1] += a[idx1[valid1]]
            idx2 = np.arange(n) - k
            valid2 = (idx2 >= 0) & (idx2 < n)
            sg[valid2] += a[idx2[valid2]]
        sg /= len(top_k.astype(float)).item() if isinstance(len(top_k), float) else float(len(top_k))

        # Momentum update
        momentum = mu * momentum - eta * sg
        a_new = project_simplex(a + momentum)
        a = a_new

    return best_a, best_obj


def lp_direction_optimize(f0, time_limit, verbose=False):
    """LP-based direction finding (maximize sum(g) s.t. (f*g)[k] <= M)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    # Precompute LP structure
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.tile(f, n)
    A_mat = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    t = 1.0
    patience = 0
    iter_count = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = optimize.linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds,
                                   method='highs', options={'disp': False, 'time_limit': 10})
        except Exception:
            break

        if not res.success:
            break
        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            break

        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.08)
                improved = True
                # Update LP matrix for next iteration
                vals_new = np.tile(f, n)
                A_mat = sparse.csc_matrix((vals_new, (rows, cols)), shape=(2 * n - 1, n))
                break

        if not improved:
            patience += 1
            t *= 0.75
            if patience > 35:
                break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits['cos1'] = cb / cb.sum()

    cb2 = 0.5 + 0.3 * np.cos(4 * np.pi * x / n)
    cb2 = np.maximum(cb2, 0.01)
    inits['cos2'] = cb2 / cb2.sum()

    cb3 = 0.5 + 0.3 * np.cos(2 * np.pi * x / (n / 3.0))
    cb3 = np.maximum(cb3, 0.01)
    inits['cos3'] = cb3 / cb3.sum()

    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x / n - 0.5)))
    inits['rc'] = rc / rc.sum()

    # Sine-based
    sn = np.abs(np.sin(np.pi * (x + 0.5) / n))
    inits['sin'] = sn / sn.sum()

    return inits


def evaluate_normalized(a):
    """Compute objective for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: SQP + LP hybrid optimization with diverse initializations.
    Combines SQP (direct peak minimization via linearized LP) with LP direction
    finding and subgradient polishing."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    lengths = [128, 256, 512, 1024]
    all_results = []

    # Phase 1: Quick scan with SQP (30% of time)
    print("Phase 1: SQP scan across lengths and initializations")
    phase1_budget = max(60.0, (budget_s - 15.0) * 0.30)
    phase1_deadline = time.monotonic() + phase1_budget

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = gen_inits(n, rng)

        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()

        n_cfgs = len(inits)
        remaining = max(10.0, phase1_deadline - time.monotonic() - 10)
        time_per = max(5.0, remaining / max(1, n_cfgs * 4))

        for name, a0 in inits.items():
            if time.monotonic() > phase1_deadline - 10:
                break
            ta = min(time_per, phase1_deadline - time.monotonic() - 5)
            if ta < 3:
                break

            # Use SQP
            f_opt, obj_opt = sqp_optimize(a0, time_limit=ta)
            score = float(evaluate_sequence(f_opt.tolist()))

            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:5s} score={score:.6f}")

    print(f"Phase 1 done. Best score: {best_score:.6f}")

    # Phase 2: Deep optimization on top results (45% of time)
    all_results.sort(key=lambda x: x[0])
    top = all_results[:6]

    phase2_budget = max(60.0, (budget_s - 15.0) * 0.45)
    phase2_deadline = time.monotonic() + phase2_budget

    if top:
        print(f"Phase 2: Deep optimization on top {len(top)}, budget={phase2_budget:.0f}s")
        t_each = phase2_budget / len(top)

        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > phase2_deadline - 15:
                break
            ta = min(t_each, phase2_deadline - time.monotonic() - 10)
            if ta < 5:
                break

            # SQP deep run
            f2, obj2 = sqp_optimize(f, time_limit=ta * 0.5)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"  SQP [{idx}] n={n} {name} score={sc2:.6f}")

            # LP direction optimization
            if time.monotonic() < phase2_deadline - 10:
                ta_lp = min(ta * 0.4, phase2_deadline - time.monotonic() - 10)
                if ta_lp > 3:
                    f3, obj3 = lp_direction_optimize(f2, time_limit=ta_lp)
                    sc3 = float(evaluate_sequence(f3.tolist()))
                    if np.isfinite(sc3) and sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"  LP [{idx}] n={n} {name} score={sc3:.6f}")

            # SQP again from LP result
            if time.monotonic() < phase2_deadline - 5:
                ta_sq = min(ta * 0.2, phase2_deadline - time.monotonic() - 5)
                if ta_sq > 2:
                    f4, _ = sqp_optimize(f3, time_limit=ta_sq)
                    sc4 = float(evaluate_sequence(f4.tolist()))
                    if np.isfinite(sc4) and sc4 < best_score:
                        best_score = sc4
                        best_seq = f4.copy()
                        print(f"  SQP2 [{idx}] n={n} {name} score={sc4:.6f}")

    # Phase 3: Subgradient polishing (15% of time)
    if best_seq is not None and time.monotonic() < deadline - 10:
        phase3_deadline = time.monotonic() + max(30.0, (budget_s - 15.0) * 0.15)
        print(f"Phase 3: Subgradient polishing, best={best_score:.6f}")

        for K in [5, 10, 20, 50]:
            if time.monotonic() > phase3_deadline - 5:
                break
            ta = min(30.0, phase3_deadline - time.monotonic() - 5)
            if ta < 3:
                break
            f5, _ = subgradient_optimize(best_seq, time_limit=ta, K=K, eta0=0.02)
            sc5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_seq = f5.copy()
                print(f"  Subgrad K={K} score={sc5:.6f}")

    # Phase 4: Final alternating SQP + LP (10% of time)
    if best_seq is not None and time.monotonic() < deadline - 5:
        phase4_deadline = time.monotonic() + max(20.0, (budget_s - 15.0) * 0.10)
        print(f"Phase 4: Final alternating, best={best_score:.6f}")

        n_best = len(best_seq)
        f_cur = best_seq.copy()
        stall_count = 0

        while time.monotonic() < phase4_deadline - 3:
            # SQP step
            t_sq = min(8.0, phase4_deadline - time.monotonic() - 3)
            if t_sq > 2:
                f_new, _ = sqp_optimize(f_cur, time_limit=t_sq)
                sc_new = float(evaluate_sequence(f_new.tolist()))
                if np.isfinite(sc_new) and sc_new < best_score - 1e-10:
                    best_score = sc_new
                    best_seq = f_new.copy()
                    f_cur = f_new.copy()
                    stall_count = 0
                    print(f"  Final SQP score={sc_new:.6f}")
                else:
                    stall_count += 1

            # LP step
            if time.monotonic() < phase4_deadline - 3:
                t_lp = min(8.0, phase4_deadline - time.monotonic() - 3)
                if t_lp > 2:
                    f_new, _ = lp_direction_optimize(f_cur, time_limit=t_lp)
                    sc_new = float(evaluate_sequence(f_new.tolist()))
                    if np.isfinite(sc_new) and sc_new < best_score - 1e-10:
                        best_score = sc_new
                        best_seq = f_new.copy()
                        f_cur = f_new.copy()
                        stall_count = 0
                        print(f"  Final LP score={sc_new:.6f}")
                    else:
                        stall_count += 1

            if stall_count > 4:
                break

    # Phase 5: Last resort - try a few random perturbations of the best
    if best_seq is not None and time.monotonic() < deadline - 3:
        n_best = len(best_seq)
        for trial in range(10):
            if time.monotonic() > deadline - 3:
                break
            perturbation = rng.normal(0, 0.01, n_best)
            f_pert = best_seq + perturbation
            f_pert = np.maximum(f_pert, 0)
            f_pert /= f_pert.sum()
            sc_pert = float(evaluate_sequence(f_pert.tolist()))
            if np.isfinite(sc_pert) and sc_pert < best_score:
                best_score = sc_pert
                best_seq = f_pert.copy()
                print(f"  Perturb [{trial}] score={sc_pert:.6f}")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
```
