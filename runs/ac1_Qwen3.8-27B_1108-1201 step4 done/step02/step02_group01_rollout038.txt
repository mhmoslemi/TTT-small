def w_seg(w, lo, hi):
    """Get segment of w from lo to hi (inclusive), padding with 0."""
    n = len(w)
    if lo >= n:
        return np.zeros(hi - lo + 1)
    hi = min(hi, n)
    return w[lo:hi]

def a_seg(a, lo, hi):
    """Get segment of a from lo to hi (inclusive), padding with 0."""
    n = len(a)
    if lo >= n or hi <= 0:
        length = hi - lo + 1
        if length <= 0:
            return np.zeros(0)
        return np.zeros(max(0, length))
    lo = max(lo, 0)
    hi = min(hi, n)
    if hi <= lo:
        return np.zeros(0)
    return a[lo:hi]

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s ** 2))


def normalize(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def build_lp_matrix_subset(a, k_indices):
    """Build convolution constraint matrix for a subset of lags."""
    n = len(a)
    A = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[row, j] = a[k - j]
    return A


def lp_solve_subset(a, M, k_indices):
    """Solve LP: maximize sum(g) s.t. conv(a,g)[k] <= M for k in k_indices, g >= 0."""
    n = len(a)
    A = build_lp_matrix_subset(a, k_indices)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(len(k_indices), M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)

def smooth_adam_optimize(a, time_limit, verbose=True, T_start=0.3, T_end=0.005):
    """
    Optimize using smooth log-sum-exp objective with Adam optimizer.
    Uses simulated annealing acceptance for escaping local minima.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S)
    t0 = time.monotonic()

    best_a = a.copy()
    best_score_val = np.max(fft_convolve(a, a))

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.95, 1e-8
    lr = 0.05
    t_step = 0

    # Schedule
    total_planned = max(1000, int(time_limit * 200))
    no_improve = 0
    stall_threshold = 500

    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        frac = min(1.0, t_step / total_planned)
        T = T_start * (T_end / T_start) ** frac

        # Compute convolution and weights
        conv = fft_convolve(a, a)
        conv_shifted = conv - np.max(conv)
        weights = np.exp(conv_shifted / T)
        weights /= np.sum(weights)

        # Compute gradient
        grad = compute_smooth_gradient(a, weights, n)

        # Normalization constraint: project gradient to be orthogonal to ones vector
        # (to maintain sum after update)
        grad_mean = np.mean(grad)
        grad = grad - grad_mean

        # Adam update
        t_step += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_step)
        v_hat = v_state / (1 - beta2 ** t_step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)

        # Evaluate
        new_conv_max = np.max(fft_convolve(a_new, a_new))
        current_conv_max = np.max(conv)

        # Acceptance criterion (simulated annealing)
        if new_conv_max < best_score_val:
            best_score_val = new_conv_max
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        elif new_conv_max <= current_conv_max:
            a = a_new
            no_improve += 1
        else:
            # Annealing: accept with probability exp(-delta / temp)
            delta = new_conv_max - current_conv_max
            accept_temp = 0.001 * (1.0 - frac) + 0.0001
            if np.random.random() < np.exp(-delta / (accept_temp * current_conv_max)):
                a = a_new
            no_improve += 1

        # Restart mechanism
        if no_improve > stall_threshold:
            no_improve = 0
            lr *= 0.5
            if lr < 0.001:
                # Perturb from best
                perturb = np.abs(np.random.default_rng(t_step).normal(0, 0.02, n))
                a = best_a * (1 + perturb)
                a = np.maximum(a, 0.0)
                a = normalize(a, S)
                lr = 0.05

        if verbose and t_step % 500 == 0:
            sc = 2.0 * n * best_score_val / (S * S)
            print(f"    SmoothAdam step {t_step}: T={T:.4f}, score={sc:.6f}, lr={lr:.4f}")

    return best_a


def gradient_refine(a, time_limit, verbose=False):
    """Subgradient optimization targeting the peak lag(s)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    lr = 1.0 / n
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_val = np.max(conv)
        # Get all near-peak lags
        threshold = peak_val * 0.999
        peak_lags = np.where(conv >= threshold)[0]

        # Subgradient from all near-peak lags
        grad = np.zeros(n)
        for pk in peak_lags:
            for i in range(n):
                if 0 <= pk - i < n:
                    grad[i] += a[pk - i]
                if 0 <= i - pk < n:
                    grad[i] += a[i - pk]
        grad /= len(peak_lags)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)

        new_conv_max = np.max(fft_convolve(a_new, a_new))
        if new_conv_max < best_conv_max - 1e-12:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        elif new_conv_max < peak_val:
            a = a_new
            no_improve += 1
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.7
                no_improve = 0

        # Adaptive lr
        if step % 200 == 0 and step > 0:
            if no_improve == 0:
                lr = min(lr * 1.05, 3.0 / n)

        step += 1
        if verbose and step % 500 == 0:
            sc = 2.0 * n * best_conv_max / (S * S)
            print(f"    GradRefine step {step}: score={sc:.6f}, lr={lr:.4f}")

    return best_a


def mirror_descent(a, time_limit, verbose=False):

    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S)

    t0 = time.monotonic()

    step = 0
    T = 0.1

    while time.monotonic() - t0 < time_limit:

        frac = min(1.0, step / 3000)

        conv = fft_convolve(a, a)

        weights = np.exp(conv_shifted / T)
        weights /= np.sum(weights)

        grad = compute_smooth_gradient(a, weights, n)

        # Multiplicative update (mirror descent with entropy)
        log_a = np.log(a)
        log_a = log_a - eta * grad
        a_new = np.exp(log_a - np.max(log_a))
        a_new = a_new / np.sum(a_new) * S
        a_new = np.maximum(a_new, 1e-15)

def make_initialization(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        freq = 0.3 + 0.5 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n)
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.2 + 0.4 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3 + int(rng.integers(0, 4))):
            f = int(rng.integers(1, n // 3))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.1 + 0.15 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'gap':
        a = np.ones(n)
        gap_start = n // 4
        gap_end = 3 * n // 4
        a[gap_start:gap_end] = 0.3 + 0.3 * rng.random()
    elif init_type == 'triangle_mod':
        a = 1.0 + 0.5 * np.abs(np.sin(np.pi * t / n))
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

    def update_best(a):
        nonlocal best_score_val, best_seq
        a_clip = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        s = score(a_clip)
        if s < best_score_val:
            best_score_val = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        sc = score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={sc:.6f}")
        update_best(provided_seq)

    # Phase 1: Smooth Adam optimization from provided sequence
    print("\n=== Phase 1: Smooth Adam on provided sequence ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(240.0, (deadline - time.monotonic()) * 0.3)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = smooth_adam_optimize(provided_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After SmoothAdam: score={best_score_val:.6f}")

    # Phase 2: LP refinement on best
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(180.0, (deadline - time.monotonic()) * 0.25)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = lp_refine_fast(best_seq, t_limit, verbose=True, subset_size=60)
        update_best(result)
        print(f"  After LP: score={best_score_val:.6f}")

    # Phase 3: Try other sequence lengths
    print("\n=== Phase 3: Other lengths ===")
    if provided_seq is not None:
        for n_try in [384, 448, 512, 576, 704, 768, 896]:
            if time.monotonic() > deadline - 300:
                break
            t_per = min(40.0, (deadline - time.monotonic()) * 0.1)
            if t_per < 10:
                break
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            result = smooth_adam_optimize(a0, t_per * 0.5, verbose=False)
            result = lp_refine_fast(result, t_per * 0.5, verbose=False, subset_size=40)
            improved = update_best(result)
            if improved:
                print(f"  *** n={n_try}: new best score={best_score_val:.6f} ***")

    # Phase 4: Gradient refinement
    print("\n=== Phase 4: Gradient refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.2)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = gradient_refine(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After Gradient: score={best_score_val:.6f}")

    # Phase 5: More LP on best
    print("\n=== Phase 5: LP polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 30:
            t_limit = min(remaining * 0.5, 120.0)
            print(f"  LP polish, time_limit={t_limit:.0f}s")
            result = lp_refine_fast(best_seq, t_limit, verbose=True, subset_size=50)
            update_best(result)
            print(f"  After LP polish: score={best_score_val:.6f}")

    # Phase 6: Mirror descent
    print("\n=== Phase 6: Mirror descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            t_limit = min(remaining * 0.5, 60.0)
            print(f"  Mirror descent, time_limit={t_limit:.0f}s")
            result = mirror_descent(best_seq, t_limit, verbose=True)
            update_best(result)
            print(f"  After Mirror: score={best_score_val:.6f}")

    # Phase 7: Final smooth Adam on best
    print("\n=== Phase 7: Final SmoothAdam ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            t_limit = min(remaining, 60.0)
            print(f"  Final SmoothAdam, time_limit={t_limit:.0f}s")
            result = smooth_adam_optimize(best_seq, t_limit, verbose=False,
                                          T_start=0.1, T_end=0.001)
            update_best(result)
            print(f"  After final SmoothAdam: score={best_score_val:.6f}")

    # Phase 8: Final targeted local search
    print("\n=== Phase 8: Targeted local search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            n = len(best_seq)
            a = normalize(best_seq.copy(), np.sqrt(2.0 * n))
            best_local = np.max(fft_convolve(a, a))
            S = np.sqrt(2.0 * n)

            for iteration in range(100000):
                if time.monotonic() > deadline:
                    break
                conv = fft_convolve(a, a)
                peak_idx = int(np.argmax(conv))

                # Try reducing elements that contribute to the peak
                n_perturb = max(3, n // 16)
                # Find top contributors to peak
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]

                top_idx = np.argsort(contribs)[-n_perturb:]
                candidate = a.copy()
                for idx in top_idx:
                    candidate[idx] *= np.exp(rng.normal(-0.03, 0.015))

                # Increase some other elements
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = max(2, n // 25)
                    inc_idx = rng.choice(other, size=min(n_inc, len(other)), replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=len(inc_idx)))

                candidate = np.clip(candidate, 0.0, None)
                candidate = normalize(candidate, S)
                new_peak = np.max(fft_convolve(candidate, candidate))

                if new_peak < best_local:
                    best_local = new_peak
                    a = candidate

            updated = update_best(a)
            if updated:
                print(f"  *** Targeted search improved: score={best_score_val:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score_val:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
