"""
Soft-max L-BFGS optimization with temperature annealing for minimizing
normalized peak autocorrelation of non-negative step functions.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import next_fast_len
import time


def fft_autocorr(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    """Compute the actual score 2*n*max(conv)/sum^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_autocorr(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def soft_max_conv_value(a, T):
    """Compute soft-max of conv(a,a) with temperature T."""
    conv = fft_autocorr(a)
    c_max = np.max(conv)
    if T < 1e-10:
        return c_max
    exps = np.exp((conv - c_max) / T)
    return float(c_max + T * np.log(np.sum(exps)))


def soft_max_conv_grad(a, T):
    """Compute value and gradient of soft-max of conv(a,a)."""
    n = len(a)
    L = 2 * n - 1
    conv = fft_autocorr(a)
    c_max = np.max(conv)
    
    if T < 1e-10:
        # Return subgradient at the peak
        peak_idx = int(np.argmax(conv))
        grad = np.zeros(n)
        for j in range(n):
            k_minus_j = peak_idx - j
            if 0 <= k_minus_j < n:
                grad[j] = 2.0 * a[k_minus_j]
        return c_max, grad
    
    exps = np.exp((conv - c_max) / T)
    p = exps / np.sum(exps)
    
    # grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # = 2 * (p_rev * a)[L-1-j] where p_rev = reverse(p)
    p_rev = p[::-1]
    size = next_fast_len(L + n - 1)
    conv_full = np.fft.irfft(np.fft.rfft(p_rev, size) * np.fft.rfft(a, size), size)[:L + n - 1]
    
    # Need conv_full[L-1-j] for j=0,...,n-1
    # L-1-j ranges from L-1=2n-2 down to L-n=n-1
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    
    obj = c_max + T * np.log(np.sum(exps))
    return obj, grad


def lbfgs_optimize(a0, T, maxiter=200, verbose=False):
    """Run L-BFGS-B to minimize soft-max of conv(a,a)."""
    def fun_and_grad(x):
        x = np.maximum(x, 1e-8)  # Ensure positivity for log
        val, grad = soft_max_conv_grad(x, T)
        return val, grad
    
    res = minimize(fun_and_grad, a0.copy(), method='L-BFGS-B',
                   jac=True, options={'maxiter': maxiter, 'gtol': 1e-6,
                                       'iprint': -1})
    result = np.clip(res.x, 0.0, None)
    if verbose:
        score = compute_score(result)
        print(f"      L-BFGS (T={T:.4f}): iter={res.nit}, f={res.fun:.6f}, score={score:.6f}")
    return result


def anneal_optimize(a0, n_temps=8, verbose=True):
    """Temperature annealing with L-BFGS at each step."""
    a = a0.copy()
    # Determine temperature range based on current convolution spread
    conv = fft_autocorr(a)
    spread = np.max(conv) - np.mean(conv)
    if spread < 1e-10:
        spread = np.max(conv) * 0.1
    
    T_start = spread * 0.5
    T_end = spread * 0.01
    
    # Geometric temperature schedule
    temps = np.geomspace(T_start, T_end, n_temps)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for i, T in enumerate(temps):
        a = lbfgs_optimize(a, T, maxiter=100, verbose=False)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose:
            print(f"    T={T:.4f}: score={best_score:.6f}")
    
    return best_a


def local_search(a0, time_limit=60.0, verbose=False):
    """Local search with targeted moves to reduce the peak."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_autocorr(a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Compute contributions to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j] (valid indices)
        contribs = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contribs[j] = a[j] * a[k]
        
        # Strategy: reduce top contributors, increase non-contributors
        top_k = max(3, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)
        
        for attempt in range(10):
            candidate = a.copy()
            
            # Reduce top contributors
            n_reduce = max(2, top_k // 2)
            reduce_idx = top_idx[rng.choice(len(top_idx), size=n_reduce, replace=False)]
            candidate[reduce_idx] *= np.exp(rng.normal(-0.03, 0.01, size=n_reduce))
            
            # Increase non-contributors
            if len(other_idx) > 0:
                n_inc = max(2, len(other_idx) // 10)
                inc_idx = other_idx[rng.choice(len(other_idx), size=n_inc, replace=False)]
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
            
            candidate = np.clip(candidate, 0.01, None)
            
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
            elif new_score < current_peak * 1.02 and rng.random() < 0.2:
                a = candidate  # Accept with some probability (simulated annealing)
        
        # Block perturbation
        if rng.random() < 0.3:
            width = max(1, n // 20)
            start = int(rng.integers(0, n - width))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
            candidate = np.clip(candidate, 0.01, None)
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
    
    if verbose:
        print(f"    Local search: final score={best_score:.6f}")
    return best_a


def make_initialization(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'cosine':
        alpha = 0.2 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(2 * np.pi * t / n)
    elif init_type == 'bicost':
        a = 1.0 + 0.2 * np.cos(4 * np.pi * t / n) - 0.1 * np.cos(2 * np.pi * t / n)
    elif init_type == 'ramp_up':
        a = 0.5 + 0.5 * t / (n - 1)
    elif init_type == 'u_shaped':
        center = (n - 1) / 2.0
        a = 1.0 + 0.3 * ((t - center) / center) ** 2
    elif init_type == 'sine_wave':
        f = 1 + int(rng.integers(0, 3))
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 'random_smooth':
        a = 1.0 + 0.2 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    
    a = np.clip(a, 0.01, None)
    return a


def normalize_to_sum(a, target_sum):
    """Normalize array to have a specific sum."""
    s = np.sum(a)
    if s < 1e-12:
        return a * target_sum / len(a)
    return a * target_sum / s


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

        fun_and_grad, a.copy(), method='L-BFGS-B', jac=True,
        options={'maxiter': maxiter, 'gtol': 1e-4, 'ftol': 1e-8, 'iprint': -1}
    )
    result = np.clip(res.x, 0.01, None)
    return result


def anneal_from(a, n_temps=8, t0=None, verbose=False):
    """Temperature annealing schedule from a starting sequence."""
    if t0 is None:
        t0 = time.monotonic()
    a = a.copy()
    conv = fft_autocorr(a)
    spread = float(np.max(conv) - np.mean(conv))
    if spread < 1e-10:
        spread = float(np.max(conv)) * 0.1 + 1e-6

    T_start = spread * 0.5
    T_end = spread * 0.005
    temps = np.geomspace(T_start, T_end, n_temps)

    best_a = a.copy()
    best_score = compute_score(a)

    for i, T in enumerate(temps):
        if time.monotonic() - t0 > 600:
            break
        a = lbfgs_step(a, float(T), maxiter=120)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose:
            print(f"      T={T:.5f} (iter {i+1}/{n_temps}): score={best_score:.6f}")

    return best_a


def local_search_refine(a0, time_limit=60.0, seed=42):
    """Targeted local search to reduce the convolution peak."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(seed)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_autocorr(a)
        peak_idx = int(np.argmax(conv))
        current_peak = float(conv[peak_idx])

        # Compute contributions to peak: conv[peak_idx][j] contribution from a[j]
        contribs = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contribs[j] = 2.0 * a[j] * a[k]

        top_k = max(3, n // 12)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)

        for attempt in range(8):
            candidate = a.copy()

            # Reduce top contributors
            n_reduce = max(2, top_k // 3)
            reduce_idx = top_idx[rng.choice(len(top_idx), size=n_reduce, replace=False)]
            candidate[reduce_idx] *= np.exp(rng.normal(-0.02, 0.008, size=n_reduce))

            # Increase non-contributors to maintain sum
            if len(other_idx) > 0:
                n_inc = max(2, len(other_idx) // 15)
                inc_idx = other_idx[rng.choice(len(other_idx), size=n_inc, replace=False)]
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))

            candidate = np.clip(candidate, 0.01, None)
            new_score = compute_score(candidate)

            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
            elif new_score < best_score * 1.01 and rng.random() < 0.15:
                a = candidate

        # Block perturbation
        if rng.random() < 0.4:
            width = max(1, n // 25)
            start = int(rng.integers(0, n - width + 1))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(
                rng.normal(0.0, 0.015, size=width)
            )
            candidate = np.clip(candidate, 0.01, None)
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate

        iteration += 1

    return best_a


def make_init(n, init_type, rng):
    """Create diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'cosine1':
        a = 1.0 + 0.25 * rng.uniform(0.5, 1.0) * np.cos(2 * np.pi * t / n)
    elif init_type == 'cosine2':
        a = 1.0 + 0.2 * np.cos(4 * np.pi * t / n) + 0.1 * np.cos(2 * np.pi * t / n)
    elif init_type == 'u_shape':
        center = (n - 1) / 2.0
        a = 1.0 + 0.4 * ((t - center) / center) ** 2
    elif init_type == 'u_shape_strong':
        center = (n - 1) / 2.0
        a = 0.8 + 0.6 * ((t - center) / center) ** 2
    elif init_type == 'sine_multi':
        a = 1.0
        for _ in range(3):
            f = 1 + int(rng.integers(0, 4))
            amp = rng.uniform(0.05, 0.15)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + ph)
    elif init_type == 'random_smooth':
        a = 1.0 + 0.15 * rng.standard_normal(n)
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
    elif init_type == 'parabola_inv':
        center = (n - 1) / 2.0
        a = 1.2 - 0.5 * ((t - center) / center) ** 2
    else:
        a = np.ones(n)

    return np.clip(a, 0.01, None)


def normalize_sum(a, target):
    """Normalize to a target sum."""
    s = float(np.sum(a))
    if s < 1e-12:
        return a * target / len(a)
    return a * target / s


def interpolate_to(a, n_new):
    """Interpolate sequence to new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: soft-max L-BFGS with temperature annealing,
    diverse initializations at multiple lengths, and local search refinement.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(a):
        nonlocal best_score, best_seq
        if a is None:
            return False
        a_c = np.clip(a, 0.0, 1000.0)
        if np.sum(a_c) < 0.01:
            return False
        s = compute_score(a_c)
        if s < best_score:
            best_score = s
            best_seq = a_c.copy()
            return True
        return False

    # Load provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided: n={len(provided)}, score={compute_score(provided):.6f}")
        try_update(provided)

    # --- Phase 1: Annealing from provided sequence ---
    print("\n=== Phase 1: Annealing from provided ===")
    if provided is not None:
        n = len(provided)
        a = normalize_sum(provided, np.sqrt(2.0 * n))
        t0 = time.monotonic()
        result = anneal_from(a, n_temps=10, t0=t0, verbose=True)
        if try_update(result):
            print(f"  Phase 1 improved: {best_score:.6f}")
        else:
            print(f"  Phase 1: no improvement (score={compute_score(result):.6f})")

    # --- Phase 2: Interpolated lengths ---
    print("\n=== Phase 2: Interpolated lengths ===")
    if provided is not None:
        for n_try in [384, 448, 512, 640, 768, 896, 1024, 1280]:
            if time.monotonic() > deadline - 180:
                break
            if n_try == len(provided):
                continue
            a0 = interpolate_to(provided, n_try)
            a0 = normalize_sum(a0, np.sqrt(2.0 * n_try))
            a0 = np.clip(a0, 0.01, None)
            result = anneal_from(a0, n_temps=5)
            if try_update(result):
                print(f"  *** n={n_try}: score={best_score:.6f} ***")

    # --- Phase 3: Diverse initializations ---
    print("\n=== Phase 3: Diverse initializations ===")
    inits = [
        (512, 'u_shape'), (512, 'cosine1'), (512, 'sine_multi'),
        (640, 'u_shape'), (640, 'cosine1'), (640, 'u_shape_strong'),
        (768, 'u_shape'), (768, 'cosine2'),
        (448, 'u_shape_strong'), (448, 'parabola_inv'),
        (896, 'u_shape'), (896, 'cosine1'),
        (384, 'u_shape'), (384, 'sine_multi'),
    ]
    for n, init_type in inits:
        if time.monotonic() > deadline - 120:
            break
        a0 = make_init(n, init_type, rng)
        a0 = normalize_sum(a0, np.sqrt(2.0 * n))
        result = anneal_from(a0, n_temps=5)
        if try_update(result):
            print(f"  *** n={n}, {init_type}: score={best_score:.6f} ***")

    # --- Phase 4: Local search on best ---
    print("\n=== Phase 4: Local search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_limit = min(remaining * 0.6, 100.0)
            print(f"  Local search (t={t_limit:.0f}s):")
            result = local_search_refine(best_seq, time_limit=t_limit, seed=seed)
            if try_update(result):
                print(f"  Local search improved: {best_score:.6f}")
            else:
                print(f"  Local search: no improvement")

    # --- Phase 5: Final annealing on best ---
    print("\n=== Phase 5: Final annealing ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            n = len(best_seq)
            a = normalize_sum(best_seq, np.sqrt(2.0 * n))
            conv = fft_autocorr(a)
            spread = float(np.max(conv) - np.mean(conv)) + 1e-8
            T_start = spread * 0.3
            T_end = spread * 0.003
            n_steps = max(3, int(remaining / 35))
            temps = np.geomspace(T_start, T_end, n_steps)

            for T in temps:
                if time.monotonic() > deadline - 15:
                    break
                a = lbfgs_step(a, float(T), maxiter=80)
                try_update(a)

            print(f"  After final annealing: {best_score:.6f}")

    # --- Phase 6: Final local search polish ---
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            result = local_search_refine(best_seq, time_limit=remaining, seed=seed + 1)
            try_update(result)

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0} ===")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        verified = evaluate_sequence(final_list)
        print(f"Verified score: {verified:.6f}")
        return final_list

    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]

    return [1.0] * 256
