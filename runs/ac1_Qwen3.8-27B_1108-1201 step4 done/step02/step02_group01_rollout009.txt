    b = np.array([conv[2*n-2-i] for i in range(n)])

"""
Smoothed gradient descent with FFT-accelerated gradients, adaptive log-sum-exp
smoothing, multiplicative updates, Nesterov momentum, and active-set LP refinement
for minimizing the normalized peak autocorrelation of non-negative step functions.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def compute_weighted_grad_fast(a, weights):
    """
    Fast computation of grad_conv[i] = 2 * sum_k weights[k] * a[k-i] * 1[0<=k-i<n]
    using FFT cross-correlation trick.
    """
    n = len(a)
    w_rev = weights[::-1]
    size = 1
    while size < n + (2 * n - 1):
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:n + 2 * n - 1]
    # b[i] = conv[2n-2-i]
    b = conv[2 * n - 2 - np.arange(n)]
    return 2.0 * b


def compute_weighted_grad_direct(a, weights):
    """Direct O(n^2) computation for verification or small n."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        max_m = min(n - 1, 2 * n - 2 - i)
        if max_m >= 0:
            grad[i] = 2.0 * np.dot(weights[i:i + max_m + 1], a[:max_m + 1])
    return grad


def smoothed_grad_step(a, mu, lr, velocity=None, momentum=0.0):
    """
    One step of smoothed gradient descent with multiplicative updates.
    Returns (a_new, conv_max, lse_value).
    """
    n = len(a)
    S = np.sum(a)
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    
    # Softmax weights (numerically stable)
    shifted = (conv - conv_max) / mu
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient via FFT
    grad = compute_weighted_grad_fast(a, weights)
    
    # Momentum
    if velocity is not None and momentum > 0:
        v = momentum * velocity + grad
    else:
        v = grad
        velocity = v.copy()
    
    # Multiplicative update
    # The actual gradient of the objective includes factor 2n*mu/S^2,
    # but since it's constant across coordinates, it just scales lr.
    a_new = a * np.exp(-lr * v)
    a_new = np.maximum(a_new, 1e-12)
    a_new = normalize_seq(a_new)
    
    lse = mu * (conv_max / mu + np.log(np.sum(exp_shifted)))
    return a_new, conv_max, velocity


def active_set_lp(a, M, fraction=0.3, max_constraints=100):
    """
    LP with active set: only include constraints where conv >= fraction * max.
    Returns optimal g.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    threshold = M * fraction
    active = np.where(conv >= threshold)[0]
    
    # Limit number of constraints
    if len(active) > max_constraints:
        # Keep the top max_constraints
        vals = conv[active]
        top_idx = np.argsort(vals)[-max_constraints:]
        active = active[top_idx]
    
    if len(active) == 0:
        return a.copy()
    
    # Build constraint matrix
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[idx, j] = a[k - j]
    
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(len(active), M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return np.zeros(n)


def lp_refine_fast(a, time_limit, verbose=False):
    """
    LP-based iterative refinement with active set and line search.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        # Solve active-set LP
        g = active_set_lp(a, M, fraction=0.85, max_constraints=80)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c = normalize_seq(g)
        
        # Line search for best blend
        best_t = 0.0
        best_conv_t = M
        for t_try in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max < best_conv_t:
                best_conv_t = b_conv_max
                best_t = t_try
            if b_conv_max > M + 1e-8:
                break
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * c
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)
        
        new_conv_max = np.max(fft_convolve(a, a))
        if new_conv_max < best_conv_max - 1e-10:
            best_conv_max = new_conv_max
            best_a = a.copy()
        
        if verbose and iteration % 50 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}")
    
    return best_a


def smoothed_gradient_optimize(a, time_limit, verbose=True):
    """
    Smoothed gradient descent with adaptive mu, backtracking, and momentum.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    mu = 0.5  # Smoothing parameter
    lr = 0.3
    momentum = 0.7
    velocity = None
    stall_count = 0
    step = 0
    mu_decay = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        a_new, conv_max, velocity = smoothed_grad_step(a, mu, lr, velocity, momentum)

        if new_conv_max < best_conv_max - 1e-10:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new

        elif new_conv_max < conv_max * 1.001:
            a = a_new
            stall_count += 1
        else:
            stall_count += 1
            lr *= 0.9

        if verbose and step % 200 == 0:
            score = 2.0 * n * best_conv_max / (S * S)

                a = candidate

        contribs = np.zeros(n)

            block_start = max(0, min_pos - n // 20)

def make_spectral_init(n, rng):

    Aim for a sequence whose autocorrelation has low sidelobes.

    num_freq = n // 2 + 1
    magnitudes = np.ones(num_freq) * (1.0 + 0.1 * rng.standard_normal(num_freq))
    magnitudes = np.abs(magnitudes)
    phases = rng.uniform(0, 2 * np.pi, size=num_freq)
    spectrum = magnitudes * np.exp(1j * phases)
    spectrum[0] = np.abs(spectrum[0])  # Make DC positive
    a = np.fft.irfft(spectrum, n)
    a = np.abs(a) + 0.1  # Make non-negative
    a = normalize_seq(a)
    return a


def make_smooth_init(n, rng):
    """Create a smooth initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(rng.integers(2, 5)):
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.cos(2 * np.pi * f * t / n + phase)
    a = np.clip(a, 0.01, None)
    return normalize_seq(a)


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining smoothed gradient descent,
    active-set LP refinement, and targeted local search.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"  *** New best: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    # Phase 1: Smoothed gradient descent on provided sequence
    print("\n=== Phase 1: Smoothed gradient descent ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(300.0, (deadline - time.monotonic()) * 0.25)
        print(f"  SGD on provided seq (n={n}), time_limit={t_limit:.0f}s")
        result = smoothed_gradient_optimize(provided_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After SGD: score={best_score:.6f}")
    
    # Phase 2: Active-set LP refinement
    print("\n=== Phase 2: Active-set LP refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(200.0, remaining * 0.2)
        print(f"  LP refine (n={len(best_seq)}), time_limit={t_limit:.0f}s")
        result = lp_refine_fast(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After LP: score={best_score:.6f}")
    
    # Phase 3: More SGD with smaller mu (sharper focus)
    print("\n=== Phase 3: Fine SGD ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(200.0, remaining * 0.2)
        print(f"  Fine SGD (n={len(best_seq)}), time_limit={t_limit:.0f}s")
        result = smoothed_gradient_optimize(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After fine SGD: score={best_score:.6f}")
    
    # Phase 4: Try different lengths
    print("\n=== Phase 4: Different lengths ===")
    lengths_to_try = [512, 640, 768, 896, 1024]
    if provided_seq is not None:
        lengths_to_try.append(len(provided_seq))
    
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 200:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        
        remaining = deadline - time.monotonic() - 200
        t_per = min(80.0, remaining / max(1, len(lengths_to_try)))
        
        # Interpolate from best or provided
        if best_seq is not None:
            a0 = interpolate_sequence(best_seq, n_try)
        elif provided_seq is not None:
            a0 = interpolate_sequence(provided_seq, n_try)
        else:
            a0 = make_smooth_init(n_try, rng)
        
        a0 = np.clip(a0, 0.01, None)
        
        # Quick SGD
        result = smoothed_gradient_optimize(a0, t_per, verbose=False)
        update_best(result)
    
    # Phase 5: LP refinement on best
    print("\n=== Phase 5: LP refinement on best ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_limit = min(remaining, 150.0)
            print(f"  LP refine best (n={len(best_seq)}), time_limit={t_limit:.0f}s")
            result = lp_refine_fast(best_seq, t_limit, verbose=True)
            update_best(result)
            print(f"  After LP: score={best_score:.6f}")
    
    # Phase 6: Final polish - coordinate descent + block perturbation
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            half_time = remaining / 2.0
            print(f"  CD polish, time_limit={half_time:.0f}s")
            result = coordinate_descent_polish(best_seq, half_time, verbose=True)
            update_best(result)
            
            remaining = deadline - time.monotonic() - 15
            if remaining > 10:
                print(f"  Block perturbation, time_limit={remaining:.0f}s")
                result = block_perturbation_search(best_seq, remaining, verbose=True)
                update_best(result)
    
    # Phase 7: One final LP if time permits
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print(f"\n=== Phase 7: Final LP ===")
            result = lp_refine_fast(best_seq, remaining, verbose=True)
            update_best(result)
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score:.6f} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 640
