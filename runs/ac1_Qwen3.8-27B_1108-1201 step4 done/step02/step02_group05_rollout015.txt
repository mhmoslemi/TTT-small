"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Phase 1: Diverse initialization including spectral flat-PSD design
Phase 2: Iterative spectral shaping (Gerchberg-Saxton style)
Phase 3: Smooth gradient descent with log-sum-exp and annealing
Phase 4: Local search with diverse move types

import numpy as np
import time
import math

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    return 2.0 * len(a) * fast_convolve(a).max()

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s

def spectral_flatten(a, max_iter=200, beta=0.3):
    """Iteratively flatten the power spectral density."""
    n = len(a)
    a = normalize(a.copy())
    fft_size = 1 << (2*n-2).bit_length()
    
    for it in range(max_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: reduce dynamic range of magnitude
        mean_mag = mag.mean()
        ratio = mag / (mean_mag + 1e-15)
        new_mag = mean_mag * ratio ** (1.0 - beta)
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new
    
    return a

def smooth_gradient_step(a, temperature, lr):
    """One step of gradient descent using log-sum-exp smooth approximation."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    c_scaled = (c - c_max) / temperature
    c_scaled = np.clip(c_scaled, -500, 0)
    exp_c = np.exp(c_scaled)
    p = exp_c / exp_c.sum()
    
    # Gradient: d(logsumexp(c/T))/da[i] = (1/T) * sum_k p[k] * a[k-i]
    # where the sum is over k such that 0 <= k-i < n
    # grad[i] = (1/T) * sum_m p[m+i] * a[m] for m=0..n-1
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = np.dot(p[i:i+n], a)
    
    grad = 2.0 * n * grad / temperature
    
    # Gradient step with projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    a_new = normalize(a_new)
    
    return a_new

def gradient_descent_anneal(a0, max_iter=200, t_start=0.1, t_end=0.001, lr_start=0.01, lr_end=0.001):
    """Gradient descent with temperature annealing."""
    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = obj_norm(a)
    
    for it in range(max_iter):
        frac = it / max(1, max_iter - 1)
        # Exponential annealing
        temperature = t_start * (t_end / t_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a_new = smooth_gradient_step(a, temperature, lr)
        
        obj_new = obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def make_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

        inits[f'gauss_{sf}'] = g / g.sum()
    
    if n > 1:

        inits['hann'] = np.maximum(hann, 0)

    # V-shaped (high at edges, low in middle)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral shaping, smooth gradient descent with annealing and momentum,
and diverse local search.
"""

import numpy as np
import time
import math

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def linear_conv(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]

def obj_norm(a):
    """Objective assuming sum(a) is already 1."""
    return 2.0 * len(a) * fast_convolve(a).max()

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s

def spectral_flatten(a, max_iter=100, beta=0.3):
    """Iteratively flatten the power spectral density (Gerchberg-Saxton style)."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for it in range(max_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        mean_mag = mag.mean()
        ratio = mag / (mean_mag + 1e-15)
        new_mag = mean_mag * ratio ** (1.0 - beta)
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new
    
    return a

def compute_smooth_gradient(a, temperature):
    """Compute gradient of 2n * logsumexp(conv(a,a)/T) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    c_scaled = (c - c_max) / temperature
    c_scaled = np.clip(c_scaled, -500, 0)
    exp_c = np.exp(c_scaled)
    p = exp_c / exp_c.sum()
    
    # grad[i] = (2n/T) * sum_m p[m+i] * a[m]
    # = (2n/T) * (p * a_rev)[n-1+i]
    q = a[::-1]
    conv_pq = linear_conv(p, q)
    raw_grad = conv_pq[n - 1:2 * n - 1]
    
    return 2.0 * n / temperature * raw_grad

def gd_step_with_line_search(a, temperature, lr_init, max_backtrack=10):
    """Gradient descent step with backtracking line search."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    c_scaled = (c - c_max) / temperature
    c_scaled = np.clip(c_scaled, -500, 0)
    exp_c = np.exp(c_scaled)
    p = exp_c / exp_c.sum()
    lse_val = c_max + temperature * math.log(exp_c.sum())
    current_obj = 2.0 * n * lse_val
    
    grad = compute_smooth_gradient(a, temperature)
    
    lr = lr_init
    for _ in range(max_backtrack):
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            lr *= 0.5
            continue
        a_new /= s
        
        # Evaluate smooth objective at new point
        c_new = fast_convolve(a_new)
        c_new_max = c_new.max()
        c_new_scaled = (c_new - c_new_max) / temperature
        c_new_scaled = np.clip(c_new_scaled, -500, 0)
        exp_new = np.exp(c_new_scaled)
        lse_new = c_new_max + temperature * math.log(exp_new.sum())
        new_obj = 2.0 * n * lse_new
        
        if new_obj < current_obj - 1e-12:
            return a_new
        lr *= 0.5
    
    return a  # No improvement, return original

def gradient_descent_anneal(a0, max_iter=200, t_start=0.1, t_end=0.002, lr_start=0.01, lr_end=0.001, momentum=0.9):
    """Gradient descent with temperature annealing and momentum."""
    a = normalize(a0.astype(np.float64).copy())
    vel = np.zeros_like(a)
    best_a = a.copy()
    best_real_obj = obj_norm(a)
    
    for it in range(max_iter):
        frac = it / max(1, max_iter - 1)
        temperature = t_start * (t_end / t_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a_new = gd_step_with_line_search(a, temperature, lr, max_backtrack=5)
        
        # Update momentum
        vel = momentum * vel + a_new  # Hmm, this isn't quite right for constrained optimization
        
        # Actually, let's skip momentum for now and just do simple GD
        a = a_new
        
        # Check real objective
        real_obj = obj_norm(a)
        if real_obj < best_real_obj:
            best_real_obj = real_obj
            best_a = a.copy()
    
    return best_a, best_real_obj

def local_search_diverse(a, time_limit, seed=42):
    """Local search with diverse move types."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 300
    
    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        move_type = rng.integers(0, 5)
        found = False
        
        if move_type == 0:
            # Pairwise transfer
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8]:
                amt = a[i] * frac
                if amt < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize(a_trial)
                o = obj_norm(a_trial)
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_trial
                    found = True
                    break
        
        elif move_type == 1:
            # Block shift
            block_size = rng.integers(2, max(3, n // 8))
            if block_size >= n:
                continue
            start = rng.integers(0, n - block_size)
            shift = rng.integers(1, max(2, n // 6)) * (1 if rng.random() < 0.5 else -1)
            new_start = start + shift
            if 0 <= new_start and new_start + block_size <= n:
                a_trial = a.copy()
                block = a_trial[start:start + block_size].copy()
                a_trial[start:start + block_size] = 0
                a_trial[new_start:new_start + block_size] += block
                a_trial = normalize(a_trial)
                o = obj_norm(a_trial)
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_trial
                    found = True
        
        elif move_type == 2:
            # Gaussian perturbation
            center = rng.integers(0, n)
            width = rng.uniform(2, n / 6)
            sign = 1.0 if rng.random() < 0.5 else -1.0
            height = sign * rng.uniform(0.01, 0.05)
            x = np.arange(n, dtype=np.float64)
            bump = height * np.exp(-0.5 * ((x - center) / width) ** 2)
            a_trial = a + bump
            a_trial = normalize(a_trial)
            o = obj_norm(a_trial)
            if o < best_obj - 1e-12:
                best_obj = o
                a = a_trial
                found = True
        
        elif move_type == 3:
            # Sinusoidal perturbation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.03)
            x = np.arange(n, dtype=np.float64)
            pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
            a_trial = a + pert
            a_trial = normalize(a_trial)
            o = obj_norm(a_trial)
            if o < best_obj - 1e-12:
                best_obj = o
                a = a_trial
                found = True
        
        elif move_type == 4:
            # Mirror/flip perturbation
            alpha = rng.uniform(0.1, 0.5)
            a_trial = (1 - alpha) * a + alpha * a[::-1]
            a_trial = normalize(a_trial)
            o = obj_norm(a_trial)
            if o < best_obj - 1e-12:
                best_obj = o
                a = a_trial
                found = True
        
        if found:
            stuck = 0
        else:
            stuck += 1
    
    return a, best_obj
