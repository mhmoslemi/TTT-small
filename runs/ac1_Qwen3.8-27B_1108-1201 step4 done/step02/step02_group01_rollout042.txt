```python
Multi-strategy optimization for minimizing peak autocorrelation of non-negative sequences.
Combines LP-based directional search, coordinate descent with efficient delta updates,

"""

import numpy as np

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):

    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)

        j_max = min(k, n - 1)

            A[k, j] = a[k - j]
    return A
```

```python
"""
Multi-strategy optimization for minimizing peak autocorrelation of non-negative sequences.
Uses LP-based directional search with exact line search, subgradient methods with momentum,
and coordinate descent for final polish. Employs FFT-based convolution and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * (S / len(a))
    return a * S / s

def build_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def lp_solve(a, M):
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.status == 0:
        return res.x
    return np.zeros(n)

def lp_refine(a, time_limit, verbose=False):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a))
    rng = np.random.default_rng(777)
    stagnant = 0
    
    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a)
        M = np.max(conv_a)
        
        g = lp_solve(a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g_norm = normalize_seq(g, S)
        
        # Line search for best blend
        best_t = 0.0
        best_peak = M
        best_b = None
        
        # Try a range of t values (geometric spacing for better coverage)
        t_values = np.concatenate([
            np.linspace(0.01, 0.1, 5),
            np.linspace(0.1, 0.3, 5),
            np.linspace(0.3, 0.6, 5),
            np.linspace(0.6, 1.0, 3)
        ])
        
        for t_try in t_values:
            b = (1 - t_try) * a + t_try * g_norm
            b = np.clip(b, 0.0, None)
            if np.sum(b) < 1e-10:
                continue
            b = normalize_seq(b, S)
            peak = np.max(fft_convolve(b))
            if peak < best_peak:
                best_peak = peak
                best_t = t_try
                best_b = b.copy()
        
        if best_t == 0.0:
            # Stagnant, try perturbation
            stagnant += 1
            if stagnant > 50:
                # Add noise
                a_pert = a * np.exp(rng.normal(0, 0.02, n))
                a = normalize_seq(np.clip(a_pert, 0, None), S)
                stagnant = 0
            continue
        
        a = best_b
        new_peak = np.max(fft_convolve(a))
        
        if new_peak < best_conv_max - 1e-12:
            best_conv_max = new_peak
            best_a = a.copy()
            stagnant = 0
            if verbose and iteration % 20 == 0:
                score = 2.0 * n * best_conv_max / (S * S)
                print(f"    LP iter {iteration}: score={score:.6f}")
        else:
            stagnant += 1
            # Try smaller step
            if stagnant > 20:
                a_pert = a * np.exp(rng.normal(0, 0.01, n))
                a = normalize_seq(np.clip(a_pert, 0, None), S)
                stagnant = 0
    
    return best_a

def subgradient_optimize(a, time_limit, verbose=False):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a))
    
    # Polyak step size parameters
    peak_lower = 1.0 + 1.0 / (2 * n - 1)  # Theoretical lower bound for normalized peak
    k_counter = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak = np.max(conv)
        peak_idx = int(np.argmax(conv))
        
        # Subgradient: partner vector for the peak lag
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            i1 = peak_idx - j
            if 0 <= i1 < n:
                val += a[i1]
            i2 = j - peak_idx
            if 0 <= i2 < n:
                val += a[i2]
            grad[j] = val
        # Note: if peak_idx == 2*j for some j, we double-count. Fix:
        # Actually, the subgradient of conv(a,a)[k] w.r.t. a[j] is:
        # a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # When k-j == j (k=2j), this gives 2*a[j], which is correct
        # because d/dx (x*x) = 2x
        # So the formula is correct.
        
        # Polyak step size
        gap = peak - peak_lower
        if gap < 1e-6:
            gap = 1e-6
        grad_norm_sq = np.sum(grad ** 2)
        if grad_norm_sq < 1e-12:
            break
        lr = gap / grad_norm_sq
        lr = min(lr, 1.0 / np.sqrt(k_counter + 1))
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        new_peak = np.max(fft_convolve(a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
        
        k_counter += 1
        if verbose and k_counter % 1000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Subgrad iter {k_counter}: score={score:.6f}, gap={gap:.6f}")
    
    return best_a

def coordinate_descent(a, time_limit, verbose=False):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a))
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        a = normalize_seq(a, S)
        conv = fft_convolve(a)
        current_peak = np.max(conv)
        order = rng.permutation(n)
        improved_this_sweep = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-12:
                candidates = [S * 0.001, S * 0.005, S * 0.01]
            else:
                candidates = [old_val * f for f in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]]
            
            best_candidate_val = old_val
            best_candidate_peak = current_peak
            best_candidate_conv = None
            
            for new_val in candidates:
                if new_val < 0 or abs(new_val - old_val) < 1e-14:
                    continue
                a[i] = new_val
                a_norm = normalize_seq(a, S)
                new_conv = fft_convolve(a_norm)
                peak = np.max(new_conv)
                if peak < best_candidate_peak:
                    best_candidate_peak = peak
                    best_candidate_val = new_val
                    best_candidate_conv = new_conv.copy()
                    best_candidate_a = a_norm.copy()
                a[i] = old_val
            
            if best_candidate_val != old_val:
                a = best_candidate_a
                conv = best_candidate_conv
                current_peak = best_candidate_peak
                improved_this_sweep = True
                if current_peak < best_peak:
                    best_peak = current_peak
                    best_a = a.copy()
        
        if not improved_this_sweep:
            break
        
        if verbose:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    CD: score={score:.6f}")
    
    return best_a

def make_initialization(n, init_type, rng):

    if init_type == 'constant':

    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        freq = 0.1 + 0.5 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n)
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for non-negative sequences. Key insights:

1. **The linear convolution (not autocorrelation)** is what's being maximized, and its peak is at the center for symmetric sequences.
2. **Edge-weighted sequences** (higher at edges, lower in middle) can reduce the center peak because `conv(n-1) = sum_j a[j]*a[n-1-j]` benefits from anti-correlation between `a[j]` and `a[n-1-j]`.
3. **A linearly varying sequence** `a[i] = 1 + alpha*(2i/(n-1)-1)` naturally gives score ≈ `2*(1-alpha^2)` by this analysis, already approaching our target.

I'll combine: LP-based directional search with exact line search, subgradient with momentum, coordinate descent, and edge-weighted initializations.

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based directional search with adaptive line search, subgradient methods
with momentum, coordinate descent, and edge-weighted initializations that exploit
the structure of linear convolution (center peak reduction via anti-symmetric mass).
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * (S / len(a))
    return a * S / s


def build_lp_matrix(a):
    """Build convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M, time_limit=5.0):
    """Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement with adaptive line search and perturbation escape."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a))
    rng = np.random.default_rng(12345)
    stagnant = 0
    no_improve_count = 0

    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a)
        M = np.max(conv_a)

        # Solve LP
        lp_time = min(3.0, max(0.5, (time_limit - (time.monotonic() - t0)) * 0.01))
        g = lp_solve(a, M, time_limit=lp_time)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g_norm = normalize_seq(g, S)

        # Line search: find best blend parameter
        best_t = 0.0
        best_peak = M
        best_b = None

        # Geometric spacing for t values (better coverage)
        t_values = np.concatenate([
            np.linspace(0.02, 0.08, 4),
            np.linspace(0.08, 0.2, 4),
            np.linspace(0.2, 0.4, 4),
            np.linspace(0.4, 0.7, 3),
            [0.9]
        ])

        for t_try in t_values:
            b = (1.0 - t_try) * a + t_try * g_norm
            b = np.clip(b, 0.0, None)
            if np.sum(b) < 1e-10:
                continue
            b = normalize_seq(b, S)
            peak = np.max(fft_convolve(b))
            if peak < best_peak:
                best_peak = peak
                best_t = t_try
                best_b = b.copy()

        # Also try geometric blend
        a_safe = np.maximum(a, 1e-10)
        ratio = np.clip(g_norm / a_safe, 1e-10, 100.0)
        for t_try in [0.05, 0.1, 0.15, 0.25]:
            b = a_safe * (ratio ** t_try)
            b = normalize_seq(b, S)
            peak = np.max(fft_convolve(b))
            if peak < best_peak:
                best_peak = peak
                best_t = t_try
                best_b = b.copy()

        if best_t == 0.0:
            stagnant += 1
            no_improve_count += 1
            if stagnant % 30 == 0:
                # Perturb to escape
                noise_scale = 0.01 + 0.01 * min(stagnant / 100.0, 1.0)
                a_pert = a * np.exp(rng.normal(0, noise_scale, n))
                a = normalize_seq(np.clip(a_pert, 0, None), S)
            if stagnant > 500:
                break
            continue

        a = best_b
        new_peak = np.max(fft_convolve(a))
        stagnant = 0

        if new_peak < best_conv_max - 1e-12:
            best_conv_max = new_peak
            best_a = a.copy()
            no_improve_count = 0
            if verbose and iteration % 20 == 0:
                score = 2.0 * n * best_conv_max / (S * S)
                print(f"    LP iter {iteration}: score={score:.6f}, t={best_t:.3f}")
        else:
            no_improve_count += 1

        if no_improve_count > 300:
            break

    return best_a


def subgradient_optimize(a, time_limit, verbose=False):
    """Subgradient method with momentum for minimizing the peak of conv(a,a)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a))

    peak_lower = 1.0 + 1.0 / n  # Rough lower bound for min peak / (S^2/n)
    k_counter = 0
    momentum = np.zeros(n)
    beta = 0.85
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak = np.max(conv)
        peak_idx = int(np.argmax(conv))

        # Subgradient of peak w.r.t. a
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            i1 = peak_idx - j
            if 0 <= i1 < n:
                val += a[i1]
            i2 = j - peak_idx
            if 0 <= i2 < n:
                val += a[i2]
            grad[j] = val

        # Polyak step size
        gap = peak - peak_lower
        if gap < 1e-4:
            gap = 1e-4
        grad_norm_sq = np.sum(grad ** 2)
        if grad_norm_sq < 1e-12:
            break
        lr = min(gap / grad_norm_sq, 2.0 / np.sqrt(k_counter + 1.0))

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_peak = np.max(fft_convolve(a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reduce momentum and try again
                beta = max(0.5, beta * 0.95)
                momentum *= 0.5
                stall = 0

        k_counter += 1
        if verbose and k_counter % 2000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Subgrad iter {k_counter}: score={score:.6f}, gap={gap:.6f}")

    return best_a


def coordinate_descent(a, time_limit, verbose=False):
    """Coordinate descent with multiplicative perturbations for final polish."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a))
    rng = np.random.default_rng(777)

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        a = normalize_seq(a, S)
        conv = fft_convolve(a)
        current_peak = np.max(conv)
        order = rng.permutation(n)
        improved_this_sweep = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-12:
                candidates = [S / n * f for f in [0.5, 1.0, 2.0, 3.0]]
            else:
                candidates = [old_val * f for f in [0.4, 0.6, 0.8, 1.25, 1.5, 2.0]]

            best_candidate_val = old_val
            best_candidate_peak = current_peak
            best_candidate_a = None

            for new_val in candidates:
                if new_val < 0 or abs(new_val - old_val) < 1e-14:
                    continue
                a[i] = new_val
                a_norm = normalize_seq(a, S)
                new_conv = fft_convolve(a_norm)
                peak = np.max(new_conv)
                if peak < best_candidate_peak:
                    best_candidate_peak = peak
                    best_candidate_val = new_val
                    best_candidate_a = a_norm.copy()

            if best_candidate_a is not None:
                a = best_candidate_a
                current_peak = best_candidate_peak
                improved_this_sweep = True
                if current_peak < best_peak:
                    best_peak = current_peak
                    best_a = a.copy()

        sweep += 1
        if not improved_this_sweep:
            break
        if verbose and sweep % 5 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    CD sweep {sweep}: score={score:.6f}")

    return best_a


def make_initialization(n, init_type, rng):
    """Create various initial sequences with emphasis on edge-weighted structures."""
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0  # Normalized to [-1, 1]

    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'linear':
        alpha = 0.3 + 0.5 * rng.random()
        a = 1.0 + alpha * x
    elif init_type == 'quadratic_edge':
        alpha = 0.3 + 0.5 * rng.random()
        a = 1.0 + alpha * x * x
    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        freq = 0.2 + 0.6 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n)
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.2 + 0.4 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(4):
            f = int(rng.integers(1, n // 3))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * f * t / n + phase)
    elif init_type == 'edge_boost':
        # Higher at edges, lower in middle
        alpha = 0.3 + 0.5 * rng.random()
        a = 1.0 - alpha * np.exp(-0.5 * (x / 0.3) ** 2)
    elif init_type == 'random_smooth':
        raw = rng.standard_normal(n)
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        a = 1.0 + 0.3 * np.convolve(raw, kernel, mode='same')
    elif init_type == 'spectral_flat':
        A = np.ones(n // 2 + 1)
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        A = A * np.exp(1j * phases)
        a = np.fft.irfft(A, n)
        a = 1.0 + 0.3 * a / max(np.max(np.abs(a)), 1e-10)
    else:
        a = np.ones(n)

    return np.clip(a, 0.01, None)


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. LP-based refinement from provided sequence and diverse inits
    2. Subgradient optimization with momentum
    3. Coordinate descent polish
    4. Final LP + CD on best
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        prov_score = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={prov_score:.6f}")
        best_score = prov_score
        best_seq = provided_seq.copy()

    # Phase 1: LP-based refinement (main workhorse)
    print("\n=== Phase 1: LP-based refinement ===")

    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(350.0, (deadline - time.monotonic()) * 0.3)
        print(f"  LP refine provided seq (n={n}), time_limit={t_limit:.0f}s")
        result = lp_refine(provided_seq, t_limit, verbose=True)
        r_score = local_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={len(best_seq)} ***")

    # Try edge-weighted initializations at good lengths
    init_configs = [
        (640, 'linear', 80),
        (640, 'edge_boost', 60),
        (512, 'linear', 50),
        (768, 'linear', 50),
        (512, 'edge_boost', 40),
        (640, 'quadratic_edge', 40),
    ]

    for n_init, init_type, t_per in init_configs:
        remaining = deadline - time.monotonic()
        if remaining < t_per + 100:
            break
        a0 = make_initialization(n_init, init_type, rng)
        t_actual = min(t_per, remaining * 0.1)
        if t_actual < 15:
            break
        result = lp_refine(a0, t_actual, verbose=False)
        r_score = local_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** NEW BEST (n={n_init}, {init_type}): score={best_score:.6f} ***")
        else:
            print(f"  n={n_init}, {init_type}: score={r_score:.6f}")

    # Try interpolation of best to other lengths
    if best_seq is not None:
        for n_try in [512, 768, 896]:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            t_per = min(50.0, remaining * 0.08)
            if t_per < 15:
                break
            a0 = interpolate_sequence(best_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            result = lp_refine(a0, t_per, verbose=False)
            r_score = local_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** NEW BEST (interp n={n_try}): score={best_score:.6f} ***")

    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: Subgradient optimization
    print("\n=== Phase 2: Subgradient optimization ===")
    if best_seq is not None:
        t_limit = min(150.0, (deadline - time.monotonic()) * 0.2)
        if t_limit > 30:
            print(f"  Subgrad (n={len(best_seq)}), time_limit={t_limit:.0f}s")
            result = subgradient_optimize(best_seq, t_limit, verbose=True)
            r_score = local_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** NEW BEST after subgrad: score={best_score:.6f} ***")

    # Phase 3: Coordinate descent polish
    print("\n=== Phase 3: Coordinate descent ===")
    if best_seq is not None:
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.15)
        if t_limit > 20:
            print(f"  CD (n={len(best_seq)}), time_limit={t_limit:.0f}s")
            result = coordinate_descent(best_seq, t_limit, verbose=True)
            r_score = local_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** NEW BEST after CD: score={best_score:.6f} ***")

    # Phase 4: Final LP refinement on best
    print("\n=== Phase 4: Final LP refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 40:
            t_limit = min(remaining, 150.0)
            print(f"  Final LP (n={len(best_seq)}), time_limit={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            r_score = local_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** NEW BEST after final LP: score={best_score:.6f} ***")

    # Phase 5: Final CD polish
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            result = coordinate_descent(best_seq, remaining, verbose=False)
            r_score = local_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** NEW BEST after final CD: score={best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
```
