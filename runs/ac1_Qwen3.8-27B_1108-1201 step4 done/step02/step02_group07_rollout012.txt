"""
Hybrid optimization for minimum peak-to-mean autocorrelation of non-negative sequences.
Combines subgradient descent, LP-based peak reduction, spectral initialization,
and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = rfft(a, size)
    B = rfft(b, size)
    return irfft(A * B, size)[:n + m - 1]

def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def normalize(a):
    """Normalize so sum = 1."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def subgradient_step(a, lr, top_k=3):
    """One subgradient step to reduce max convolution value."""
    n = len(a)
    c = fft_convolve(a, a)
    # Find top-k peaks
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        # dc[ki]/da[i] = 2*a[ki-i] if 0<=ki-i<n, with correction if ki=2i
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for i in range(lo, hi + 1):
            diff = ki - i
            if 0 <= diff < n:
                if ki == 2 * i:
                    g[i] += a[diff] / top_k
                else:
                    g[i] += 2.0 * a[diff] / top_k
    # Update
    a_new = a - lr * g
    a_new = np.clip(a_new, 0.0, None)
    return a_new, float(np.max(c))

def spectral_init(n, rng, shape='flat'):
    """Initialize using spectral method (Gerchberg-Saxton-like)."""
    a = np.ones(n) / n
    for it in range(50):
        # FFT with zero-padding
        size = 2 * n - 1
        A = fft(a, size)
        # Target: flat magnitude (with DC being largest)
        mag = np.abs(A)
        # Compute target magnitude: sqrt of flattened power
        power = mag ** 2
        # Make power more uniform: blend with its mean
        target_power = 0.7 * power + 0.3 * np.ones_like(power) * np.mean(power)
        target_mag = np.sqrt(target_power)
        # Preserve DC
        target_mag[0] = mag[0]
        # Reconstruct
        A_new = target_mag * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))
        a_new = np.clip(a_new, 0.0, None)
        if np.sum(a_new) < 1e-10:

        a = a_new / np.sum(a_new)
