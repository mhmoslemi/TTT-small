"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected gradient descent with soft-max annealing, backtracking line search,
LP-based refinement, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=1.0):
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    arange = np.arange(1, n+1, dtype=np.float64)
    # Find largest k where u[k-1] > (cumsum[k-1] - target_sum) / k
    rho_arr = np.where(u > (cumsum - target_sum) / arange)[0]
    if len(rho_arr) == 0:
        return np.full(n, target_sum / n)
    k = rho_arr[-1] + 1
    theta = (cumsum[k-1] - target_sum) / k
    result = np.maximum(v - theta, 0)
    # Fix floating point
    result_sum = result.sum()
    if abs(result_sum - target_sum) > 1e-10:
        result += (target_sum - result_sum) / n
        result = np.maximum(result, 0)
    return result

def soft_gradient(a, tau):
    """Compute gradient of log-sum-exp(tau * autocorrelation) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * correlate(w, a, 'valid')[i]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

def hard_gradient(a):
    """Gradient using only the peak lag."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def backtracking_line_search(a, grad, lr_init, alpha=0.01, beta=0.5, max_iter=20):
    """
    Find step size using Armijo backtracking.
    a is on the simplex (sum=1, a>=0).
    Returns new a and the step size used.
    """
    n = len(a)
    f0 = compute_obj(a)
    lr = lr_init
    for _ in range(max_iter):
        # Gradient step
        a_try = a - lr * grad
        # Project onto simplex
        a_try = project_simplex(a_try, target_sum=1.0)
        f_try = compute_obj(a_try)
        if f_try <= f0 - alpha * lr * np.dot(grad, a - a_try):
            return a_try, lr
        lr *= beta
    return a, 0.0

def lp_refine(a, deadline, max_iters=50):

    n = len(a)
    if n > 768:
        return a, compute_obj(a)
    
    a = normalize(a, target_sum=1.0)
    c = np.convolve(a, a)
    M = float(c.max())
    best = a.copy()
    best_score = compute_obj(a)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]
    
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})

                break
            g = res.x
        except Exception:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-8:
            break
        g_norm = g * (1.0 / g_sum)
        
        # Try blending
        improved = False
        for t_blend in [0.5, 0.3, 0.2, 0.1, 0.7, 0.4]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, target_sum=1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True

                c = np.convolve(a, a)
                M = float(c.max())
                for j in range(n):
                    for m in range(j, j + n):
                        A[m, j] = a[m - j]

def coordinate_descent(a, deadline, max_rounds=5):
    """Coordinate descent polishing."""
    n = len(a)
    a = normalize(a.copy(), target_sum=1.0)
    best_score = compute_obj(a)
    patience = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            candidates = [0.0, orig * 0.5, orig * 1.5, orig * 0.8, orig * 1.2]
            
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a = normalize(a, target_sum=1.0)
                s = compute_obj(a)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = val
                a[i] = orig
            
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, target_sum=1.0)
                best_score = best_s
                improved = True
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))
        
        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0
    
    return a, best_score

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=1.0):
    """Scale a to have the given sum."""
    S = float(a.sum())
    if S < 1e-12:
        return np.full_like(a, target / len(a))
    return a * (target / S)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    # Find largest k such that u[k-1] > (cumsum[k-1] - target_sum) / k
    mask = u > (cumsum - target_sum) / arange
    if not mask.any():
        return np.full(n, target_sum / n)
    k = int(np.max(np.where(mask)[0])) + 1
    theta = (cumsum[k - 1] - target_sum) / k
    result = np.maximum(v - theta, 0.0)
    # Correct for floating point
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        # Distribute correction to positive entries
        pos_mask = result > 0
        n_pos = pos_mask.sum()
        if n_pos > 0:
            result[pos_mask] += diff / n_pos
    return result


def soft_gradient(a, tau):
    """Gradient of log(sum_k exp(tau * c_k)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * correlate(w, a, 'valid')[i]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def pgd_optimize(a0, deadline, tau_start=1.0, tau_end=100.0, lr_init=0.05):
    """Projected gradient descent with soft-max annealing and backtracking."""
    n = len(a0)
    a = normalize(np.clip(a0, 0, 1000), target_sum=1.0)
    best = a.copy()
    best_score = compute_obj(a)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        # Anneal temperature sinusoidally (increase overall, with small oscillations)
        progress = min(1.0, it / 30000.0)
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.5)
        
        # Compute gradient
        grad = soft_gradient(a, tau)
        
        # Backtracking line search
        lr = lr_init
        f0 = compute_obj(a)
        step_taken = False
        for _ in range(12):
            a_try = a - lr * grad
            a_try = project_simplex(a_try, target_sum=1.0)
            f_try = compute_obj(a_try)
            if f_try < f0 - 1e-12:
                a = a_try
                step_taken = True
                break
            lr *= 0.5
        
        if step_taken:
            if it % 50 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-10:
                    best_score = s
                    best = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            no_improve += 1
        
        # Periodic perturbation to escape local minima
        if it % 1000 == 999:
            noise = np.random.randn(n) * (0.02 + 0.01 * progress)
            a_pert = a + noise
            a_pert = project_simplex(a_pert, target_sum=1.0)
            s_pert = compute_obj(a_pert)
            # Accept with probability based on improvement (or always for exploration)
            if s_pert < best_score or np.random.rand() < 0.3:
                a = a_pert
            if s_pert < best_score:
                best = a_pert.copy()
                best_score = s_pert
        
        # Early stop if stuck for too long
        if no_improve > 20:
            # Try a bigger perturbation
            noise = np.random.randn(n) * 0.1
            a_pert = a + noise
            a_pert = project_simplex(a_pert, target_sum=1.0)
            a = a_pert
            no_improve = 0
    
    return best, best_score


def lp_refine(a, deadline, max_iters=30, m_factor=1.0):
    """LP-based refinement."""
    n = len(a)
    if n > 768:
        return a, compute_obj(a)
    
    a = normalize(a.copy(), target_sum=1.0)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = float(c.max()) * m_factor
        
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for m in range(j, j + n):
                A[m, j] = a[m - j]
        
        b = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
        except Exception:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-8:
            break
        g_norm = g * (1.0 / g_sum)
        
        # Try blending
        improved = False
        for t_blend in [0.5, 0.3, 0.7, 0.2, 0.4, 0.6]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, target_sum=1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            # Small gradient nudge to escape

            a_new = project_simplex(a_new, target_sum=1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
            else:
                # Try with m_factor slightly larger
                pass
    
    return best, best_score


def coordinate_descent(a, deadline, max_rounds=5):
    """Coordinate descent polishing."""
    n = len(a)
    a = normalize(a.copy(), target_sum=1.0)
    best_score = compute_obj(a)
    patience = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-10:
                continue
            candidates = [0.0, orig * 0.3, orig * 0.7, orig * 1.3, orig * 1.7]
            
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a = normalize(a, target_sum=1.0)
                s = compute_obj(a)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = val
                a[i] = orig
            
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, target_sum=1.0)
                best_score = best_s
                improved = True
                # Update peak lag
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))
        
        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0
    
    return a, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected gradient descent with soft-max annealing, backtracking line search,
periodic LP refinement, and coordinate descent polishing.
Focuses on n=384 with multi-start and structured randomization.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=1.0):
    """Scale a to have the given sum."""
    S = float(a.sum())
    if S < 1e-12:
        return np.full_like(a, target / max(1, len(a)))
    return a * (target / S)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using sorted algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    mask = u > (cumsum - target_sum) / arange
    if not mask.any():
        return np.full(n, target_sum / n)
    k = int(np.max(np.where(mask)[0])) + 1
    theta = (cumsum[k - 1] - target_sum) / k
    result = np.maximum(v - theta, 0.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        pos_mask = result > 0
        n_pos = int(pos_mask.sum())
        if n_pos > 0:
            result[pos_mask] += diff / n_pos
    return result


def soft_gradient(a, tau):
    """Gradient of log(sum_k exp(tau * c_k)) w.r.t. a. Uses FFT-free correlate."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * np.correlate(w, a, 'valid')[i]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

        hi = min(n - 1, k_idx)

    no_improve = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / 20000.0)

        # Choose gradient type: mostly soft-max, occasionally top-k
        if use_mixed_grad and it % 500 in (400, 450):
            grad = topk_gradient(a, k=3)
        else:
            tau = tau_start + (tau_end - tau_start) * (progress ** 0.6)
            grad = soft_gradient(a, tau)

        # Backtracking line search
        lr = lr_init * (1.0 - 0.5 * progress)
        lr = max(lr, 0.005)
        f0 = compute_obj(a)
        step_taken = False
        for _ in range(10):
            a_try = a - lr * grad
            a_try = project_simplex(a_try, target_sum=1.0)
            f_try = compute_obj(a_try)
            if f_try < f0 - 1e-14:
                a = a_try
                step_taken = True
                break
            lr *= 0.4

        if step_taken:
            if it % 30 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-12:
                    best_score = s
                    best = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            no_improve += 1

        # Periodic perturbation
        if it % 800 == 799:
            amplitude = 0.01 + 0.03 * (1.0 - progress)
            noise = rng.standard_normal(n) * amplitude
            a_pert = a + noise
            a_pert = project_simplex(a_pert, target_sum=1.0)
            s_pert = compute_obj(a_pert)
            if s_pert < best_score:
                best = a_pert.copy()
                best_score = s_pert
            # Accept perturbation with some probability (exploration)
            if s_pert < best_score + 0.01 or rng.random() < 0.2:
                a = a_pert

        # Stuck detection: big restart
        if no_improve > 30:
            amplitude = 0.05 + 0.1 * (1.0 - progress)
            a_pert = a + rng.standard_normal(n) * amplitude
            a_pert = project_simplex(a_pert, target_sum=1.0)
            a = a_pert
            no_improve = 0

    return best, best_score


def lp_refine(a, deadline, max_iters=25, m_margin=1.0):
    """
    LP-based refinement: given current a, solve LP to find g maximizing sum(g)
    subject to conv(a,g) <= M, g >= 0. Then blend.
    """
    n = len(a)
    if n > 768:
        return a, compute_obj(a)

    a = normalize(a.copy(), target_sum=1.0)
    best = a.copy()
    best_score = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(c.max()) * m_margin

        # Build constraint matrix A[m,j] = a[m-j] if 0 <= m-j < n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for m in range(j, j + n):
                A[m, j] = a[m - j]

        b = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 45})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-8:
            break
        g_norm = g * (1.0 / g_sum)

        improved = False
        for t_blend in [0.4, 0.25, 0.6, 0.15, 0.35, 0.5]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, target_sum=1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-12:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Gradient nudge to escape
            grad = soft_gradient(a, 15.0)
            a_new = a - 0.003 * grad
            a_new = project_simplex(a_new, target_sum=1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
            else:
                # Try slightly relaxed M
                m_margin *= 1.02

    return best, best_score


def coordinate_descent(a, deadline, max_rounds=4):
    """Coordinate descent polishing with focus on peak-reducing moves."""
    n = len(a)
    a = normalize(a.copy(), target_sum=1.0)
    best_score = compute_obj(a)
    patience = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-8:
                continue

            # Generate candidates: focus on reducing contribution to peak
            j = k_star - i
            if 0 <= j < n and a[j] > 0.1 * orig:
                # This pair contributes to the peak; try reducing more
                candidates = [0.0, orig * 0.2, orig * 0.5, orig * 0.8, orig * 1.2]
            else:
                candidates = [0.0, orig * 0.5, orig * 1.5, orig * 0.8, orig * 1.2]

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a = normalize(a, target_sum=1.0)
                s = compute_obj(a)
                if s < best_s - 1e-13:
                    best_s = s
                    best_val = val
                a[i] = orig

            if best_val != orig:
                a[i] = best_val
                a = normalize(a, target_sum=1.0)
                best_score = best_s
                improved = True
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))

        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0

    return a, best_score


def generate_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # From previous best (most important start)
    have_prev = False
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= n and np.all(np.isfinite(gs)) and gs.sum() > 0.01:
            starts.append(gs[:n].copy())
            have_prev = True
            # Small perturbations
            for _ in range(4):
                pert = gs[:n] + rng.standard_normal(n) * 0.02
                starts.append(np.maximum(pert, 0.0))
            # Circular shifts
            for shift in [1, 2, 3, 5]:
                starts.append(np.roll(gs[:n], shift).copy())
            # Mirror
            starts.append(gs[:n][::-1].copy())
    except NameError:
        pass

    # Random uniform-like
    for _ in range(6):
        starts.append(rng.random(n) * 0.4 + 0.1)

    # Cosine modulations (various frequencies)
    for freq in [1, 2, 3, 5, 7, 11, 13, 17]:
        phase = rng.uniform(0, 2 * np.pi)
        starts.append(0.2 + 0.25 * np.cos(2 * np.pi * i * freq / n + phase))

    # Anti-symmetric bias (reduce center autocorrelation)
    a = rng.random(n)
    a[:n // 2] *= 1.8
    a[n // 2:] *= 0.5
    starts.append(np.maximum(a, 0.01))
    a2 = rng.random(n)
    a2[:n // 2] *= 0.5
    a2[n // 2:] *= 1.8
    starts.append(np.maximum(a2, 0.01))

    # Sparse patterns
    for period in [2, 3, 4]:
        a = np.zeros(n)
        a[::period] = rng.random(n // period + 1) * 0.4 + 0.1
        starts.append(a)

    # Gaussian bumps
    for center_frac, width in [(0.25, 0.08), (0.75, 0.08), (0.5, 0.15), (0.3, 0.12), (0.7, 0.12)]:
        starts.append(np.exp(-0.5 * ((i - n * center_frac) / (n * width)) ** 2))

    # Two bumps (anti-symmetric placement)
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.07)) ** 2) +
                  0.1 * np.exp(-0.5 * ((i - n * 0.75) / (n * 0.15)) ** 2))
    starts.append(np.exp(-0.5 * ((i - n * 0.75) / (n * 0.07)) ** 2) +
                  0.1 * np.exp(-0.5 * ((i - n * 0.25) / (n * 0.15)) ** 2))

    # Linear ramp
    starts.append(0.1 + 0.4 * i / n)
    starts.append(0.1 + 0.4 * (1 - i / n))

    # Comb-like (alternating)
    a = np.zeros(n)
    a[::2] = 0.4
    a[1::2] = 0.1
    starts.append(a + rng.random(n) * 0.05)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses projected gradient descent with soft-max annealing,
    backtracking line search, LP refinement, and coordinate descent polishing.
    Focuses on n=384 (previous best length) with multi-start strategy.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Try to get initial score from height_sequence_1
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size > 0 and np.all(np.isfinite(gs)) and gs.sum() > 0.01:
            initial_score = compute_obj(gs)
            best_seq = gs.copy()
            best_score = initial_score
            print(f"Initial from height_sequence_1: n={len(gs)}, score={initial_score:.6f}")
    except NameError:
        print("No height_sequence_1 available, starting fresh")

    target_n = 384

    # Primary phase: intensive optimization at n=384
    print(f"\n=== Phase 1: Intensive optimization at n={target_n} ===")
    t_p1_end = t_start + (deadline - t_start) * 0.65
    starts = generate_starts(rng, target_n)
    print(f"  {len(starts)} starting sequences")

    n_starts = len(starts)
    for idx, a0 in enumerate(starts):
        if time.monotonic() > t_p1_end:
            break
        remaining = t_p1_end - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: first start (prev best) gets most time
        if idx == 0:
            t_single = min(remaining * 0.30, 300)
        elif idx < 4:
            t_single = min(remaining * 0.15, 150)
        elif idx < 10:
            t_single = min(remaining * 0.08, 100)
        else:
            t_single = min(remaining / max(1, n_starts - idx) * 0.6, 60)
        t_single = max(15, t_single)
        if t_single > remaining - 10:
            t_single = remaining - 10
        if t_single < 12:
            break

        print(f"  [{idx:2d}/{n_starts}] t={t_single:.0f}s", end="")

        a_work = np.clip(a0, 0, 1000).copy()
        a_local_best = a_work.copy()
        s_local_best = compute_obj(a_local_best)

        # Phase A: PGD
        t_pgd_end = time.monotonic() + t_single * 0.65
        t_pgd_end = min(t_pgd_end, t_p1_end - 30)
        a_pgd, s_pgd = pgd_optimize(a_work, t_pgd_end)
        if s_pgd < s_local_best:
            a_local_best = a_pgd
            s_local_best = s_pgd

        # Phase B: LP refinement
        if time.monotonic() < t_p1_end - 20:
            t_lp_end = min(t_p1_end - 10, time.monotonic() + t_single * 0.2)
            a_lp, s_lp = lp_refine(a_local_best, t_lp_end, max_iters=15)
            if s_lp < s_local_best:
                a_local_best = a_lp
                s_local_best = s_lp

        # Phase C: Coordinate descent
        if time.monotonic() < t_p1_end - 5:
            t_cd_end = min(t_p1_end - 3, time.monotonic() + t_single * 0.1)
            a_cd, s_cd = coordinate_descent(a_local_best, t_cd_end, max_rounds=3)
            if s_cd < s_local_best:
                a_local_best = a_cd
                s_local_best = s_cd

        if s_local_best < best_score:
            best_score = s_local_best
            best_seq = a_local_best.copy()
            print(f" -> NEW BEST: {best_score:.6f}")
        else:
            print(f" -> {s_local_best:.6f}")

    # Secondary phase: try a few other lengths
    print(f"\n=== Phase 2: Other lengths ===")
    for alt_n in [256, 512, 320, 448]:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(120, (deadline - time.monotonic()) - 40)
        if t_alloc < 30:
            break
        t_alt_end = time.monotonic() + t_alloc
        print(f"  n={alt_n}, t={t_alloc:.0f}s")

        a0 = rng.random(alt_n) * 0.4 + 0.1
        # Also try shifted version of best if lengths are compatible
        if best_seq is not None and len(best_seq) >= alt_n:
            a0 = best_seq[:alt_n].copy()

        a_pgd, s_pgd = pgd_optimize(a0, min(t_alt_end, time.monotonic() + t_alloc * 0.7),
                                     tau_start=5.0, tau_end=80.0, lr_init=0.06)
        a_final, s_final = a_pgd, s_pgd

        if alt_n <= 512 and time.monotonic() < t_alt_end - 10:
            a_lp, s_lp = lp_refine(a_final, t_alt_end - 5, max_iters=10)
            if s_lp < s_final:
                a_final, s_final = a_lp, s_lp

        if alt_n <= 384 and time.monotonic() < t_alt_end - 5:
            a_cd, s_cd = coordinate_descent(a_final, t_alt_end - 2, max_rounds=2)
            if s_cd < s_final:
                a_final, s_final = a_cd, s_cd

        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"    NEW BEST at n={alt_n}: {best_score:.6f}")
        else:
            print(f"    n={alt_n}: {s_final:.6f}")

    # Final phase: intensive polish on the best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 3: Final polish (score={best_score:.6f}) ===")
        n_best = len(best_seq)
        t_final_end = deadline - 10
        remaining = t_final_end - time.monotonic()

        # More PGD with different hyperparams
        a_pol, s_pol = pgd_optimize(best_seq, time.monotonic() + remaining * 0.4,
                                     tau_start=20.0, tau_end=200.0, lr_init=0.03)
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol
            print(f"  After PGD2: {best_score:.6f}")

        # LP with more iterations
        if n_best <= 512 and time.monotonic() < t_final_end - 15:
            a_pol, s_pol = lp_refine(best_seq, time.monotonic() + remaining * 0.3,
                                      max_iters=30, m_margin=1.0)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol
                print(f"  After LP: {best_score:.6f}")

        # Coordinate descent
        if n_best <= 512 and time.monotonic() < t_final_end - 5:
            a_pol, s_pol = coordinate_descent(best_seq, t_final_end - 2, max_rounds=8)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol
                print(f"  After CD: {best_score:.6f}")

        # One more LP pass
        if n_best <= 512 and time.monotonic() < t_final_end - 3:
            a_pol, s_pol = lp_refine(best_seq, t_final_end - 1, max_iters=10)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol
                print(f"  After LP2: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: n={len(best_seq)}, score={final_score:.6f}")
        print(f"{'='*50}")
        try:
            ev = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {ev:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 256
