            k = int(k)
            if k < 0 or k >= len(conv):

            j_end = min(n - 1, k)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth-max (log-sum-exp) L-BFGS-B optimization with adaptive beta continuation
- Multiple length exploration (128-1024) with diverse initializations
- Hard-max subgradient descent polish with momentum
- Random restarts to escape local minima
Key innovation: continuation method on beta smooths the non-convex problem,
allowing L-BFGS-B to efficiently navigate the landscape before switching to
sharp (high-beta) optimization and subgradient polish.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_obj(x, n, beta):
    """Smooth objective: 2n * LSE(beta * conv(softmax(x))) / beta."""
    a = softmax(x)
    conv = fast_convolve(a)
    return 2.0 * n * logsumexp(beta * conv) / beta

    """Gradient of smooth objective w.r.t. log-space parameters."""
    a = softmax(x)
    size = 2 * n - 1
    fft_size1 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size1)
    conv = np.fft.irfft(A * A, fft_size1)[:size]

    # Softmax weights over convolution peaks
    conv_shifted = beta * (conv - conv.max())
    w = np.exp(conv_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Gradient w.r.t. a: g_i = 4n * sum_{k} w[k] * a[k-i] for valid k-i
    # This is 4n * cross_correlation(w, a)[i]
    # Using FFT: ifft(fft(w, N) * conj(fft(a, N)))[:n] gives cross-corr at lags 0..n-1
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    B = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(B), N)
    g = 4.0 * n * corr[:n]

    # Gradient w.r.t. x via softmax Jacobian:
    # grad_x_j = a_j * (g_j - sum_i a_i * g_i)
    inner = np.dot(a, g)
    grad_x = a * (g - inner)

    return grad_x


def optimize_smooth(x0, n, beta, time_limit, maxiter=300):
    """
    Optimize smooth objective using L-BFGS-B with time limit.
    Returns (x_opt, a_opt, obj_opt_hard).
    """
    t0 = time.monotonic()

    def callback(xk):
        return (time.monotonic() - t0) > time_limit

    try:
        result = minimize(
            _smooth_obj, x0, args=(n, beta),
            jac=_smooth_grad, args=(n, beta),
            method='L-BFGS-B',
            callback=callback,
            options={'maxiter': maxiter, 'maxfun': 5000, 'gtol': 1e-10,
                     'ftol': 1e-12}
        )
        x_opt = result.x
    except Exception:
        x_opt = x0

    a_opt = softmax(x_opt)
    conv = fast_convolve(a_opt)
    obj_opt = 2.0 * n * conv.max()
    return x_opt, a_opt, obj_opt


def subgradient_polish(a0, time_limit, lr_init=0.02, momentum=0.85, top_k=30,
                       temp_init=0.001, perturb_every=2000, perturb_scale=0.005):
    """
    Hard-max projected subgradient descent with momentum on the simplex.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    momentum_cur = momentum
    batch = 50

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute subgradient over top-k peaks
            if top_k >= len(conv):
                peaks = np.arange(len(conv))
                peak_vals = conv.copy()
            else:
                peaks = np.argpartition(conv, -top_k)[-top_k:]
                peak_vals = conv[peaks]

            # Soft weights over peaks
            if temp > 1e-15:
                scaled = peak_vals / temp
                scaled -= scaled.max()
                w = np.exp(scaled)
                w_sum = w.sum()
                if w_sum < 1e-15:
                    w = np.zeros(len(peaks))
                    w[np.argmax(peak_vals)] = 1.0
                else:
                    w /= w_sum
            else:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0

            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                wi = w[i] * 2.0
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += wi * a[k - idx]

            v = momentum_cur * v - lr * sg
            a_new = project_simplex(a + v)
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Periodic perturbation
        if no_improve >= perturb_every:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            lr = max(lr_init * 0.3, lr * 0.5)

        # Schedule
        lr *= 0.9995
        temp *= 1.003
        momentum_cur = min(0.95, momentum_cur + 0.0003)

    return best_a, best_obj


def generate_x0_list(n, parent, rng):
    """Generate diverse initializations in log-space for length n."""
    x0_list = []
    x = np.linspace(0, 1, n)

    # Uniform
    x0_list.append(np.zeros(n))

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 1e-12)
            x0_list.append(np.log(pp / pp.sum()))
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 1e-12)
            x0_list.append(np.log(pp / pp.sum()))

    # Triangular (peak in middle)
    tri = -4.0 * (x - 0.5) ** 2
    x0_list.append(tri)

    # Inverted triangle (peaks at edges)
    inv_tri = -4.0 * x * (1 - x)
    x0_list.append(inv_tri)

    # Gaussian (narrow peak)
    g = -3.0 * ((x - 0.5) / 0.15) ** 2
    x0_list.append(g)

    # Bimodal
    bi = -2.0 * (np.minimum(x, 1 - x) / 0.2) ** 2
    x0_list.append(bi)

    # Ramp
    x0_list.append(2.0 * x)

    # Random
    x0_list.append(rng.standard_normal(n) * 0.5)

    # Cosine
    x0_list.append(np.cos(2 * np.pi * x))

    return x0_list


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Smooth L-BFGS-B exploration over lengths and initializations
    Phase 2: High-beta L-BFGS-B refinement on top solutions
    Phase 3: Hard-max subgradient polish
    Phase 4: Final random restarts and extended polish
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # Phase 1: Exploration with moderate beta
    print("=== Phase 1: Smooth Exploration ===")
    lengths = [128, 192, 256, 320, 384, 512, 768, 1024]
    candidates = []  # (score, n, a_opt)

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        x0_list = generate_x0_list(n, parent, rng)
        # Limit initializations based on time
        time_per_n = max(20.0, (deadline - time.monotonic() - 300) / (len(lengths) + 1 - lengths.index(n)))
        n_inits = min(len(x0_list), max(2, int(time_per_n / 15)))

        for idx in range(n_inits):
            if time.monotonic() > deadline - 250:
                break

            x0 = x0_list[idx]
            t_alloc = min(25.0, deadline - time.monotonic() - 200)
            if t_alloc < 3:
                break

            # Beta continuation: start moderate, increase
            beta1 = 30
            x_opt, a_opt, obj_opt = optimize_smooth(x0, n, beta1, time_limit=t_alloc, maxiter=200)

            # Quick evaluation with hard max
            score = 2.0 * n * fast_convolve(a_opt).max()

            candidates.append((score, n, a_opt, x_opt))

            # Also evaluate via the official function
            real_score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(real_score):
                score = real_score
                if score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  P1: n={n}, init={idx}, score={score:.6f}")

        print(f"  Length {n} done. Best so far: {best_score:.6f}")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 complete. Top score: {candidates[0][0]:.6f}")

    # Phase 2: High-beta refinement on top candidates
    print("=== Phase 2: High-Beta Refinement ===")
    top_k_refine = min(10, len(candidates))
    refined = []

    for i in range(top_k_refine):
        if time.monotonic() > deadline - 200:
            break

        score_i, n_i, a_i, x_i = candidates[i]
        t_alloc = min(40.0, deadline - time.monotonic() - 150)
        if t_alloc < 5:
            break

        # Increase beta for sharper optimization
        beta2 = 150
        x_opt, a_opt, obj_opt = optimize_smooth(x_i, n_i, beta2, time_limit=t_alloc, maxiter=300)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            refined.append((score, n_i, a_opt, x_opt))
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  P2: n={n_i}, from cand {i}, score={score:.6f}")

        # Even higher beta
        if time.monotonic() < deadline - 100:
            beta3 = 500
            t_alloc2 = min(20.0, deadline - time.monotonic() - 80)
            if t_alloc2 > 3:
                x_opt2, a_opt2, obj_opt2 = optimize_smooth(x_opt, n_i, beta3,
                                                            time_limit=t_alloc2, maxiter=200)
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(score2):
                    refined.append((score2, n_i, a_opt2, x_opt2))
                    if score2 < best_score:
                        best_score = score2
                        best_a = a_opt2.copy()
                        print(f"  P2b: n={n_i}, beta=500, score={score2:.6f}")

    refined.sort(key=lambda x: x[0])
    print(f"Phase 2 complete. Best: {best_score:.6f}")

    # Phase 3: Subgradient polish
    print("=== Phase 3: Subgradient Polish ===")
    top_polish = min(5, len(refined)) if refined else min(3, len(candidates))
    source = refined if refined else [(c[0], c[1], c[2], c[3]) for c in candidates[:3]]

    for i in range(top_polish):
        if time.monotonic() > deadline - 100:
            break

        score_i, n_i, a_i, _ = source[i]
        t_alloc = min(60.0, deadline - time.monotonic() - 60)
        if t_alloc < 5:
            break

        a_polish, obj_polish = subgradient_polish(
            a_i, time_limit=t_alloc, lr_init=0.015,
            momentum=0.88, top_k=max(15, n_i // 20), temp_init=0.0005)

        score = float(evaluate_sequence(a_polish.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_polish.copy()
            print(f"  P3: n={n_i}, from src {i}, score={score:.6f}")

    print(f"Phase 3 complete. Best: {best_score:.6f}")

    # Phase 4: Final refinement on the best solution
    print("=== Phase 4: Final Refinement ===")
    if best_a is not None:
        n_best = len(best_a)
        remaining = deadline - time.monotonic() - 15.0

        if remaining > 30:
            # Multiple subgradient passes with different params
            configs = [
                {'lr_init': 0.01, 'momentum': 0.92, 'top_k': n_best // 15, 'temp_init': 0.0003},
                {'lr_init': 0.005, 'momentum': 0.95, 'top_k': n_best // 10, 'temp_init': 0.0001},
                {'lr_init': 0.02, 'momentum': 0.85, 'top_k': max(20, n_best // 25), 'temp_init': 0.001},
            ]
            t_per_pass = remaining / (len(configs) + 3)

            for cfg in configs:
                if remaining < 5:
                    break
                a_p, obj_p = subgradient_polish(best_a, time_limit=t_per_pass, **cfg)
                score = float(evaluate_sequence(a_p.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_p.copy()
                    print(f"  P4 subgrad: score={score:.6f}")
                remaining = deadline - time.monotonic() - 15.0
                t_per_pass = remaining / max(1, len(configs) + 2 - configs.index(cfg))

        # Smooth re-optimization from the best point
        if remaining > 15:
            a_norm = best_a / best_a.sum()
            x_start = np.log(a_norm + 1e-12)
            for beta_test in [300, 800]:
                if remaining < 5:
                    break
                t_alloc = min(20.0, remaining / 2)
                x_opt, a_opt, obj_opt = optimize_smooth(x_start, n_best, beta_test,
                                                         time_limit=t_alloc, maxiter=200)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  P4 smooth beta={beta_test}: score={score:.6f}")
                remaining = deadline - time.monotonic() - 15.0

        # Random perturbations + re-optimize
        if remaining > 15:
            for trial in range(5):
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_a + noise)
                t_alloc = min(15.0, remaining / 2)
                a_opt, obj_opt = subgradient_polish(a_pert, time_limit=t_alloc,
                                                     lr_init=0.008, momentum=0.93,
                                                     top_k=n_best // 15, temp_init=0.0002)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  P4 perturb {trial}: score={score:.6f}")
                remaining = deadline - time.monotonic() - 15.0

    # Fallback
    if best_a is None:
        best_a = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}")
    return [float(x) for x in best_a]
