def fast_convolve(a):

    return result[:size]

def gen_initializations(n, rng, parent=None):

    inits = {}

    for sf in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf*n))**2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2*np.pi*x/n))
    cb /= cb.sum()
    inits['cos'] = cb
    
    # Bimodal
    bm = np.exp(-0.5*((x-0.25*n)/(0.1*n))**2) + np.exp(-0.5*((x-0.75*n)/(0.1*n))**2)
    bm /= bm.sum()
    inits['bi'] = bm
    
    # U-shaped
    u = 0.5 * (1 - np.cos(2*np.pi*x/n)) + 0.01
    u /= u.sum()
    inits['u'] = u
    
    # Power law
    r = np.arange(1, n+1, dtype=np.float64)
    for p in [0.5, 2.0]:
        pl = r**p
        pl /= pl.sum()
        inits[f'pl{int(p*10)}'] = pl
    
    # Random
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5) / 1.0  # dirichlet already sums to 1
    
    # Parent (if available)
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_gradient(a, w):
    """
    Compute the convolution gradient: dg[j] = sum_m a[m] * w[m+j]

    Tconv = T * conv

    if s < 1e-300:

        w[np.argmax(conv)] = 1.0
    else:

def lp_step(f, time_limit=20.0):
    """
    One LP-guided step: find g maximizing sum(g) s.t. f*g <= M, g >= 0.
    Returns blended result or None if LP fails.
    """
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    # Build the constraint: (f*g)[k] <= M for k=0..2n-2
    # (f*g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i]
    # So for variable g[j], it appears in constraints k=j, j+1, ..., j+n-1
    # with coefficients f[k-j]

    rows_list = []
    cols_list = []
    vals_list = []

    for j in range(n):
        for k in range(j, j + n):
            if k >= 2 * n - 1:
                break
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(f[k - j])

    try:
        import scipy.sparse as sparse
        import scipy.optimize as optimize

        A = sparse.csc_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                               options={'disp': False, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-12:
                return None
            g /= gs
            # Blend
            for t_try in [0.5, 0.3, 0.2, 0.1]:
                f_new = (1.0 - t_try) * f + t_try * g
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s < 1e-12:
                    continue
                f_new /= s
                conv_new = fast_convolve(f_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < 2.0 * n * M:
                    return f_new
            return None
    except Exception:
        pass
    return None


def gen_inits(n, rng, parent=None):

    inits = {}

        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    ar /= ar.sum()
    inits['anti'] = ar

    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random with spikes
    rs = np.ones(n) * 0.01
    n_spikes = max(3, n // 32)
    spike_pos = rng.choice(n, size=n_spikes, replace=False)
    rs[spike_pos] = 1.0
    rs /= rs.sum()
    inits['spike'] = rs

    # Sine wave
    sw = np.abs(np.sin(np.pi * x / n)) + 0.05
    sw /= sw.sum()
    inits['sine'] = sw

    # Triangular
    tri = 1.0 - np.abs(x - 0.5 * n) / (0.5 * n)
    tri = np.maximum(tri, 0.02)
    tri /= tri.sum()
    inits['tri'] = tri

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: multi-stage optimization combining adaptive mirror descent,
    active-set refinement, LP-guided steps, and controlled restarts.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(0.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 20.0  # 20s safety margin

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: len={p.size}, score={parent_score:.6f}")
    except Exception as e:
        print(f"No parent: {e}")

    # Determine lengths to try
    lengths_to_try = [512]
    if parent is not None and parent.size not in lengths_to_try:
        lengths_to_try.append(parent.size)
    for ln in [256, 384, 768, 1024]:
        if ln not in lengths_to_try and ln <= 2048:
            lengths_to_try.append(ln)
    lengths_to_try.sort()

    all_results = []

    # === Phase 1: Diverse quick scan ===
    print("=" * 60)
    print("Phase 1: Diverse initialization + quick optimization")
    print("=" * 60)

    # Prioritize parent's length and 512
    lengths_priority = sorted(lengths_to_try, key=lambda x: 0 if x == (parent.size if parent is not None else 512) else (1 if x == 512 else 2))

    for n_try in lengths_priority:
        if time.monotonic() > deadline - 180:
            break

        inits = gen_inits(n_try, rng, parent)
        n_cfgs = len(inits)
        remaining = deadline - time.monotonic() - 180
        time_per_cfg = max(4.0, remaining / max(1, n_cfgs * max(1, len(lengths_priority) - lengths_priority.index(n_try))))
        # Cap per-cfg time for phase 1
        time_per_cfg = min(time_per_cfg, 25.0)

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 150:
                break
            ta = time_per_cfg
            if ta < 3:
                break

            a_opt, _ = mirror_descend(a0, time_limit=ta, T_start=5.0, T_end=80.0,
                                       eta_start=0.4, eta_min=0.01)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                all_results.append((score, n_try, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  n={n_try:4d} {name:7s} score={score:.6f}")

    print(f"Phase 1 complete. Best so far: {best_score:.6f}")

    # Sort results
    all_results.sort(key=lambda x: x[0])
    top_n = min(8, len(all_results))
    top_results = all_results[:top_n]

    # === Phase 2: Deep optimization on top candidates ===
    print("=" * 60)
    print(f"Phase 2: Deep optimization on top {top_n} candidates")
    print("=" * 60)

    remaining = deadline - time.monotonic() - 120
    if remaining > 60 and top_results:
        t_each = remaining / top_n

        for idx, (sc, n, name, a) in enumerate(top_results):
            if time.monotonic() > deadline - 80:
                break
            ta = min(t_each, deadline - time.monotonic() - 80)
            if ta < 10:
                break

            print(f"  [{idx}] n={n} {name} current_score={sc:.6f}, budget={ta:.0f}s")

            # Step 2a: Extended mirror descent with high T
            t_md = ta * 0.45
            a1, _ = mirror_descend(a, time_limit=t_md, T_start=10.0, T_end=500.0,
                                   eta_start=0.25, eta_min=0.003)
            sc1 = float(evaluate_sequence(a1.tolist()))
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_seq = a1.copy()
                print(f"    MD: {sc1:.6f}")

            # Step 2b: Active-set refinement
            if time.monotonic() < deadline - 60:
                t_as = min(ta * 0.25, deadline - time.monotonic() - 60)
                a2, _ = active_set_refine(a1, time_limit=t_as, top_k=3, eta=0.2)
                sc2 = float(evaluate_sequence(a2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a2.copy()
                    print(f"    AS: {sc2:.6f}")
                a_cur = a2
            else:
                a_cur = a1

            # Step 2c: LP-guided step (if time permits)
            if time.monotonic() < deadline - 50 and n <= 768:
                a_lp = lp_step(a_cur, time_limit=15.0)
                if a_lp is not None:
                    sc_lp = float(evaluate_sequence(a_lp.tolist()))
                    if np.isfinite(sc_lp) and sc_lp < best_score:
                        best_score = sc_lp
                        best_seq = a_lp.copy()
                        print(f"    LP: {sc_lp:.6f}")
                    a_cur = a_lp if (np.isfinite(sc_lp) and sc_lp < 2.0 * n * fast_convolve(a_cur).max()) else a_cur

            # Step 2d: Final mirror descent polish
            if time.monotonic() < deadline - 40:
                t_polish = min(ta * 0.25, deadline - time.monotonic() - 40)
                a3, _ = mirror_descend(a_cur, time_limit=t_polish, T_start=50.0, T_end=1000.0,
                                       eta_start=0.1, eta_min=0.001)
                sc3 = float(evaluate_sequence(a3.tolist()))
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = a3.copy()
                    print(f"    Polish: {sc3:.6f}")

    # === Phase 3: Restarts from best with perturbations ===
    print("=" * 60)
    print("Phase 3: Controlled restarts")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 30
        n_restarts = max(1, int(remaining / 22))
        n_restarts = min(n_restarts, 40)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            # Different perturbation strategies
            if r % 3 == 0:
                # Gaussian noise proportional to value
                noise = rng.normal(0, 0.15, n_best)
                a_pert = np.maximum(best_seq * (1.0 + noise), 0.0)
            elif r % 3 == 1:
                # Random spike perturbation
                a_pert = best_seq.copy()
                n_spikes = max(5, n_best // 20)
                pos = rng.choice(n_best, size=n_spikes, replace=False)
                a_pert[pos] += rng.uniform(0, 0.5, n_spikes) * best_seq.mean()
            else:
                # Eta-perturbation: scale a subset
                a_pert = best_seq.copy()
                mask = rng.random(n_best) < 0.3
                a_pert[mask] *= rng.uniform(0.5, 2.0, mask.sum())

            s = a_pert.sum()
            if s < 1e-10:
                a_pert = best_seq.copy()
            else:
                a_pert /= s

            tr = min(18.0, deadline - time.monotonic() - 20)
            if tr < 3:
                break

            a_r, _ = mirror_descend(a_pert, time_limit=tr * 0.6, T_start=10.0, T_end=200.0)
            # Quick active set
            if time.monotonic() < deadline - 15:
                a_r, _ = active_set_refine(a_r, time_limit=tr * 0.3, top_k=3, eta=0.15)

            sc_r = float(evaluate_sequence(a_r.tolist()))
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {r:2d}: {sc_r:.6f} (NEW BEST)")

    # === Phase 4: Final intensive polish ===
    print("=" * 60)
    print("Phase 4: Final intensive polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 5:
        tr = deadline - time.monotonic() - 5
        n_final = len(best_seq)

        # Multiple polish passes with different T settings
        for tp, eta in [(100.0, 0.08), (300.0, 0.05), (1000.0, 0.02)]:
            if tr < 5:
                break
            t_pass = min(tr * 0.35, 60.0)
            a_f, _ = mirror_descend(best_seq, time_limit=t_pass, T_start=tp, T_end=tp * 5,
                                    eta_start=eta, eta_min=0.001)
            sc_f = float(evaluate_sequence(a_f.tolist()))
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  Final T={tp:.0f}: {sc_f:.6f}")
            tr = deadline - time.monotonic() - 5

        # Active set final
        if time.monotonic() < deadline - 5:
            t_as = deadline - time.monotonic() - 5
            a_af, _ = active_set_refine(best_seq, time_limit=t_as, top_k=2, eta=0.1)
            sc_af = float(evaluate_sequence(a_af.tolist()))
            if np.isfinite(sc_af) and sc_af < best_score:
                best_score = sc_af
                best_seq = a_af.copy()
                print(f"  Final AS: {sc_af:.6f}")

    # Safety: ensure valid output
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        best_seq = np.ones(512) / 512.0
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL RESULT: n={len(best_seq)}, score={final_score:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
