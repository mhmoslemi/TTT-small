"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradient descent with Nesterov acceleration,
LP-based refinement, asymmetric initializations, and multi-start strategy.
Key insight: asymmetric sequences can achieve objective < 2 (unlike symmetric/uniform).
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to given target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def smooth_max_grad(c, tau):
    """Compute weights w_k = softmax(tau * c_k) for the log-sum-exp gradient."""
    c = np.asarray(c, dtype=np.float64)
    c_max = np.max(c)
    logits = tau * (c - c_max)
    logits = np.clip(logits, -500.0, 500.0)
    w = np.exp(logits)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    return w


def soft_grad(a, tau):
    """
    Gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])) w.r.t. a.
    = 2 * cross-correlation of w with a (valid part).
    """
    n = len(a)
    c = np.convolve(a, a)
    w = smooth_max_grad(c, tau)
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def hard_grad(a):
    """Gradient using hard max (single peak)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def project_simplex(a, target_sum):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / S)


def nesterov_step(a, prev_a, t_prev, tau, S_t, lr):
    """
    One Nesterov-accelerated step.
    Returns (new_a, new_t).
    """
    n = len(a)
    # Nesterov extrapolation
    t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
    beta = (t_prev - 1.0) / t_curr
    y = a + beta * (a - prev_a)
    y = np.maximum(y, 0.0)

    # Gradient at y
    grad = soft_grad(y, tau)

    # Project gradient (remove mean to stay on sum constraint)
    grad -= np.mean(grad)

    # Step
    x_new = y - lr * grad
    x_new = project_simplex(x_new, S_t)

    return x_new, t_curr


def backtrack_line_search(a, S_t, tau, lr, max_iters=30):
    """
    Backtracking line search for the smoothed objective.
    Returns step and new point.
    """
    n = len(a)
    c0 = np.convolve(a, a)
    f0 = np.max(c0)  # We minimize max(c) subject to sum constraint

    grad = soft_grad(a, tau)
    grad -= np.mean(grad)

    c = lr
    for _ in range(max_iters):
        a_new = project_simplex(a - c * grad, S_t)
        c_new_conv = np.convolve(a_new, a_new)
        f_new = np.max(c_new_conv)
        if f_new <= f0 - 1e-4 * c * np.sum(grad * grad):
            return a_new, c
        c *= 0.5
    return a, c


def lp_refine(a, M, verbose=False):
    """
    LP: max sum(g) s.t. conv(a,g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    if n > 1024:
        return None

    c_obj = -np.ones(n)
    # Build constraint matrix
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]

    b = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

                i = int(rng.integers(0, n))
                orig = a[i]
                a[i] = orig * (1.0 + 0.3 * rng.standard_normal())
                a = np.maximum(a, 0.0)
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_score - 1e-10:
                    best_score = s
                    improved = True
                else:
                    a[i] = orig
                    a = normalize(a, S_t)

        if not improved:
            break

    return a, best_score


def optimize_sequence(a0, deadline, max_iters_gd=20000):
    """
    Full optimization pipeline for a single starting sequence.
    Returns (best_seq, best_score).
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)
    t_start = time.monotonic()

    # Phase 1: Nesterov-accelerated gradient descent with soft max
    t_curr = 1.0
    prev_a = a.copy()
    tau = 2.0
    lr = 0.5
    no_improve = 0

    phase1_end = t_start + (deadline - t_start) * 0.5
    it = 0
    while time.monotonic() < phase1_end and it < max_iters_gd:
        it += 1

        # Anneal tau (increase for sharper gradients)
        progress = it / max_iters_gd
        tau = 2.0 + 48.0 * progress
        # Anneal learning rate
        lr = 0.5 * (0.9998 ** it)
        lr = max(lr, 0.01)

        a_new, t_curr = nesterov_step(a, prev_a, t_curr, tau, S_t, lr)
        prev_a = a
        a = a_new

        if it % 500 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-8:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 6:
                    # Restart Nesterov
                    t_curr = 1.0
                    prev_a = a.copy()
                    no_improve = 0
                    lr = 0.3

    print(f"    Phase 1 (Nesterov GD): score={best_score:.6f}, iters={it}")

    # Phase 2: LP-based refinement
    a = best.copy()
    phase2_end = t_start + (deadline - t_start) * 0.8
    a = normalize(a, S_t)
    no_improve_lp = 0
    it_lp = 0

    while time.monotonic() < phase2_end and it_lp < 50:
        it_lp += 1
        c = np.convolve(a, a)
        M = float(np.max(c))

        g = lp_refine(a, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g_norm = g * (S_t / g_sum)

        improved = False
        for t_blend in [0.8, 0.6, 0.4, 0.25, 0.1]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            no_improve_lp += 1
            if no_improve_lp >= 4:
                break
        else:
            no_improve_lp = 0

    print(f"    Phase 2 (LP): score={best_score:.6f}, iters={it_lp}")

    # Phase 3: Short Nesterov GD to escape LP local minima
    if time.monotonic() < phase2_end + 60:
        a = best.copy()
        t_curr = 1.0
        prev_a = a.copy()
        tau = 20.0
        lr = 0.2
        phase3_end = min(time.monotonic() + 60.0, deadline - 30)

        for it2 in range(5000):
            if time.monotonic() > phase3_end:
                break
            tau = 20.0 + 20.0 * (it2 / 1000.0)
            tau = min(tau, 60.0)
            lr = 0.2 * (0.999 ** it2)
            lr = max(lr, 0.02)

            a_new, t_curr = nesterov_step(a, prev_a, t_curr, tau, S_t, lr)
            prev_a = a
            a = a_new

            if it2 % 500 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-8:
                    best_score = s
                    best = a.copy()

        a = best.copy()
        print(f"    Phase 3 (post-LP GD): score={best_score:.6f}")

    # Phase 4: Coordinate descent polish
    if n <= 512 and time.monotonic() < deadline - 10:
        a_pol, s_pol = coordinate_descent(best, deadline - 5, max_rounds=3)
        if s_pol < best_score:
            best = a_pol
            best_score = s_pol
        print(f"    Phase 4 (coord desc): score={best_score:.6f}")

    return best, best_score


def generate_initializations(rng, n):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    i = np.arange(n, dtype=np.float64)
    rn = i / n

    # Asymmetric ramps (inspired by n=3 optimal structure)
    inits.append(1.0 + 2.0 * rn)              # Linear ramp
    inits.append(1.0 + 2.0 * rn ** 2)         # Quadratic ramp
    inits.append(1.0 + 1.0 * rn)              # Gentle linear
    inits.append(np.exp(1.5 * rn))            # Exponential ramp
    inits.append(1.0 + 3.0 * rn ** 0.5)       # Square root ramp
    inits.append(1.0 / (1.0 + rn))            # Decaying (reverse)
    inits.append(1.0 + 2.0 * (1.0 - rn))      # Reverse linear ramp

    # Twin structure: rise then fall, but asymmetric
    peak_pos = 0.7 * n
    inits.append(np.exp(-((i - peak_pos) / (n * 0.3)) ** 2) + 0.2)

    # Cosine-modulated
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * rn * 1.0))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * rn * 3.0) + 0.5 * rn)

    # Random
    inits.append(rng.random(n) * 0.5 + 0.25)
    inits.append(0.5 + 0.5 * rng.random(n))

    # Power law
    inits.append(0.3 + 0.7 * rn ** 1.5)
    inits.append(0.3 + 0.7 * rn ** 0.5)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-start optimization with Nesterov GD,
    LP refinement, and coordinate descent. Returns best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try multiple lengths
    lengths = [256, 384, 512]

    print(f"Starting optimization, budget={budget_s}s")
    print(f"Timeline: start={t_start:.1f}, deadline={deadline:.1f}")

    # --- Phase A: Quick scan across lengths and initializations ---
    # Give each (length, init) combo a quick optimization
    scan_results = []
    for n in lengths:
        inits = generate_initializations(rng, n)
        # Pick top 4 most diverse initializations
        # (we'll just take the first 4 + 1 random)
        chosen = inits[:4] + [inits[4]]

        for idx, a0 in enumerate(chosen):
            if time.monotonic() > deadline - 600:
                break

            # Quick optimization (limited time)
            quick_deadline = time.monotonic() + 60.0
            seq, score = optimize_sequence(a0, quick_deadline, max_iters_gd=5000)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [scan n={n}, init={idx}] score={score:.6f}")
            else:
                print(f"  [scan n={n}, init={idx}] score={score:.6f}")

            scan_results.append((n, score, idx))

    print(f"\nAfter scan: best_score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase B: Deep optimization on best length ---
    if best_seq is not None:
        best_n = len(best_seq)
        print(f"\nDeep optimization at n={best_n}, time remaining={deadline - time.monotonic():.0f}s")

        # Try several diverse starts at the best length
        inits_deep = generate_initializations(rng, best_n)
        # Add perturbed version of current best
        for trial in range(3):
            pert = rng.normal(0, 0.15, best_n)
            a0 = np.maximum(best_seq / best_seq.sum() * np.sqrt(2 * best_n) + pert, 0)
            inits_deep.append(a0)

        # Also try the best initialization type from scan
        # Sort detailed inits by how close they are to best_seq
        inits_deep = inits_deep[:8]  # Take at most 8

        for idx, a0 in enumerate(inits_deep):
            if time.monotonic() > deadline - 120:
                break

            remaining = deadline - time.monotonic() - 90
            t_alloc = min(remaining / max(1, len(inits_deep) - idx), 300.0)
            if t_alloc < 30:
                t_alloc = max(30.0, remaining * 0.3)

            seq, score = optimize_sequence(a0, time.monotonic() + t_alloc, max_iters_gd=50000)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [deep {idx}] New best: score={best_score:.6f}")
            else:
                print(f"  [deep {idx}] score={score:.6f}")

    # --- Phase C: Final intensive polish ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\nFinal polish, time remaining={deadline - time.monotonic():.0f}s")
        remaining = deadline - time.monotonic() - 20
        seq, score = optimize_sequence(best_seq, time.monotonic() + remaining, max_iters_gd=100000)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Final polish improved: score={best_score:.6f}")
        else:
            print(f"  Final polish: no improvement (score={score:.6f})")

    # --- Validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal validation: n={len(best_seq)}, computed_score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence score: {eval_score:.6f}")
            if eval_score < best_score:
                best_score = eval_score
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        # Ensure all values are valid
        final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        return final

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
