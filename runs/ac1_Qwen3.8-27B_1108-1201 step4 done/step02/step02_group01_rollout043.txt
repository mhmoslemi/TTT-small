"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines improved LP-based refinement with
multi-peak equalization, adaptive line search, stochastic restarts, and
multiple sequence lengths. Uses FFT-based convolution for speed.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def self_convolve(a):
    """Compute autocorrelation (linear convolution with itself)."""
    return fft_convolve(a, a)


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = self_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have sum S."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones_like(a) * S / len(a)
    return a * S / s


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M, constraint_indices=None):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0.
    If constraint_indices given, only enforce those constraints.
    """
    n = len(a)
    A = build_conv_matrix(a)
    if constraint_indices is not None:
        A_sub = A[constraint_indices, :]
    else:
        A_sub = A
    c = -np.ones(n)
    b_ub = np.full(A_sub.shape[0], M)
    res = linprog(c, A_ub=A_sub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status == 0:
        return res.x
    return None


def line_search(a, g, max_evals=30):

    vals = []
    for t in t_grid:
        b = (1 - t) * a + t * g
        b = np.clip(b, 0.0, None)
        b = normalize_seq(b, S)
        v = np.max(self_convolve(b))
        vals.append(v)
        if v < best_val:
            best_val = v
            best_t = t

    # Refine around best

    if 0 < idx < len(t_grid) - 1:
        t_lo = t_grid[idx - 1]
        t_hi = t_grid[idx + 1]

            b = (1 - t_mid) * a + t_mid * g

    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, None)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        conv = self_convolve(a)
        M = np.max(conv)

        # Get top K lag indices
        top_k = np.argsort(conv)[::-1][:K]

        # Try progressively tighter M values
        success = False
        for frac in [1.0, 0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
            M_try = M * frac
            g = solve_lp(a, M_try, constraint_indices=top_k)
            if g is not None:
                g_sum = np.sum(g)
                if g_sum > 1e-10:
                    g = normalize_seq(g, S)
                    t_best, val_best = line_search(a, g)
                    if t_best > 0.0:
                        a_new = (1 - t_best) * a + t_best * g
                        a_new = np.clip(a_new, 0.0, None)
                        a_new = normalize_seq(a_new, S)
                        new_score = local_score(a_new)
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a_new.copy()
                            a = a_new
                        success = True
                        break

        if not success:
            break

        if verbose and it % 20 == 0:
            print(f"    MultiPeak iter {it}: score={best_score:.6f}")

    return best_a


def subgradient_optimize(a, time_limit=60.0, verbose=False):
    """
    Subgradient method with mirror descent (exponential update)
    to reduce the peak of the autocorrelation.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, None)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(self_convolve(a))

    # Use log-parameterization: a[i] = exp(w[i]) * S/n initially
    w = np.log(np.maximum(a, 1e-10) * n / S)
    lr = 0.1
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Reconstruct a from w
        a = np.exp(w)
        a = a / np.sum(a) * S

        conv = self_convolve(a)
        peak_idx = int(np.argmax(conv))

        # Compute subgradient at peak
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i - peak_idx
            if 0 <= j2 < n:
                grad[i] += a[j2]

        # Mirror descent step
        w -= lr * grad / S
        # Clamp to prevent overflow
        w = np.clip(w, -20, 5)

        # Renormalize
        a_new = np.exp(w)
        a_new = a_new / np.sum(a_new) * S
        a_new = np.clip(a_new, 0.0, None)

        new_max = np.max(self_convolve(a_new))
        if new_max < best_conv_max:
            best_conv_max = new_max
            best_a = a_new.copy()
            w = np.log(np.maximum(a_new, 1e-10) * n / S)
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.9
            if stall > 500:
                lr *= 0.5
                stall = 0

        if verbose:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_conv_max / (S * S)
            if int(elapsed * 2) % 10 == 0:
                print(f"    SubGrad t={elapsed:.0f}s: score={score:.6f}")

    return best_a


def stochastic_local_search(a, time_limit=60.0, verbose=False):
    """
    Stochastic local search: random perturbations targeting peak lags.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, None)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = self_convolve(a)
        peak_val = np.max(conv)
        peak_idx = int(np.argmax(conv))

        # Compute contributions to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Strategy: reduce top contributors, boost others
        top_k = max(5, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]

        candidate = a.copy()
        # Reduce top contributors
        for idx in top_idx:
            factor = np.exp(-rng.uniform(0.02, 0.15))
            candidate[idx] *= factor

        # Boost random other elements
        other = np.setdiff1d(np.arange(n), top_idx)
        n_boost = max(5, n // 16)
        boost_idx = rng.choice(other, size=min(n_boost, len(other)), replace=False)
        candidate[boost_idx] *= np.exp(rng.uniform(0.02, 0.15, size=len(boost_idx)))

        candidate = np.clip(candidate, 0.0, None)
        candidate = normalize_seq(candidate, S)

        new_score = local_score(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
        elif new_score < best_score * 1.01 and rng.random() < 0.2:
            a = candidate

        # Occasionally do a larger perturbation
        if rng.random() < 0.1:
            candidate2 = a * np.exp(rng.normal(0, 0.05, size=n))
            candidate2 = np.clip(candidate2, 0.0, None)
            candidate2 = normalize_seq(candidate2, S)
            s2 = local_score(candidate2)
            if s2 < best_score:
                best_score = s2
                best_a = candidate2.copy()
                a = candidate2

    return best_a


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def make_structured_init(n, rng, init_type='chirp'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.2 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.3))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.1 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_sine':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 8)))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'bump':
        center = n * rng.uniform(0.2, 0.8)
        sigma = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'alternating':
        a = 1.0 + 0.3 * np.cos(np.pi * t / n * 2)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def optimize_sequence(a, remaining_time, verbose=False):
    """
    Apply multiple optimization phases to a sequence.
    Returns the best sequence found.
    """
    n = len(a)
    a = np.clip(a, 0.0, None)
    best = a.copy()
    best_score = local_score(best)

    # Phase 1: LP refinement
    t_lp = min(remaining_time * 0.35, 120.0)
    t_lp = max(t_lp, 10.0)
    if verbose:
        print(f"  LP refine (n={n}, t={t_lp:.0f}s)")
    result = lp_refine_improved(best, t_lp, verbose=verbose)
    s = local_score(result)
    if s < best_score:
        best = result
        best_score = s

    # Phase 2: Multi-peak equalization
    t_mp = min(remaining_time * 0.2, 60.0)
    t_mp = max(t_mp, 5.0)
    elapsed = time.time()
    if time.time() - elapsed < t_mp:
        if verbose:
            print(f"  Multi-peak LP (t={t_mp:.0f}s)")
        result = multi_peak_lp(best, K=5, time_limit=t_mp, verbose=verbose)
        s = local_score(result)
        if s < best_score:
            best = result
            best_score = s

    # Phase 3: Subgradient
    t_sg = min(remaining_time * 0.15, 45.0)
    t_sg = max(t_sg, 5.0)
    if time.time() - elapsed < t_sg + t_mp:
        if verbose:
            print(f"  Subgradient (t={t_sg:.0f}s)")
        result = subgradient_optimize(best, time_limit=t_sg, verbose=verbose)
        s = local_score(result)
        if s < best_score:
            best = result
            best_score = s

    # Phase 4: Stochastic local search
    t_sl = min(remaining_time * 0.15, 45.0)
    t_sl = max(t_sl, 5.0)
    if time.time() - elapsed < t_sl + t_sg + t_mp:
        if verbose:
            print(f"  Stochastic (t={t_sl:.0f}s)")
        result = stochastic_local_search(best, time_limit=t_sl, verbose=verbose)
        s = local_score(result)
        if s < best_score:
            best = result
            best_score = s

    # Phase 5: Final LP polish
    t_final = min(remaining_time * 0.1, 30.0)
    if t_final > 3.0:
        if verbose:
            print(f"  Final LP (t={t_final:.0f}s)")
        result = lp_refine_improved(best, t_final, verbose=False)
        s = local_score(result)
        if s < best_score:
            best = result
            best_score = s

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization combining LP refinement,
    multi-peak equalization, subgradient methods, and stochastic search.
    Tries multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def update_best(a):
        nonlocal best_seq, best_score
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"  *** NEW BEST: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # === Phase 1: Optimize provided sequence at native length ===
    if provided_seq is not None:
        print("\n=== Phase 1: Optimize provided sequence ===")
        remaining = deadline - time.monotonic()
        if remaining > 60:
            result, score = optimize_sequence(provided_seq, remaining * 0.3, verbose=True)
            update_best(result)
            print(f"  After Phase 1: score={best_score:.6f}")

    # === Phase 2: Try interpolated lengths ===
    if provided_seq is not None:
        print("\n=== Phase 2: Interpolated lengths ===")
        target_lengths = [576, 640, 704, 768, 832, 896]
        # Filter out lengths too close to provided
        n_prov = len(provided_seq)
        target_lengths = [ln for ln in target_lengths if abs(ln - n_prov) > 32]

        for n_try in target_lengths:
            if time.monotonic() > deadline - 120:
                break
            remaining = min((deadline - time.monotonic()) - 120, 90.0)
            if remaining < 20:
                break
            a0 = interpolate_seq(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            print(f"  Trying n={n_try}, time_budget={remaining:.0f}s")
            result, score = optimize_sequence(a0, remaining, verbose=False)
            if score < best_score:
                update_best(result)

    # === Phase 3: Structured initializations ===
    print("\n=== Phase 3: Structured initializations ===")
    init_configs = [
        (384, 'chirp'), (512, 'sine'), (640, 'multi_sine'),
        (448, 'bump'), (512, 'alternating'), (384, 'multi_sine'),
    ]
    for n_init, init_type in init_configs:
        if time.monotonic() > deadline - 90:
            break
        remaining = min((deadline - time.monotonic()) - 90, 45.0)
        if remaining < 15:
            break
        a0 = make_structured_init(n_init, rng, init_type)
        print(f"  Init n={n_init} type={init_type}, budget={remaining:.0f}s")
        result, score = optimize_sequence(a0, remaining, verbose=False)
        if score < best_score:
            update_best(result)

    # === Phase 4: Stochastic restarts from best ===
    print("\n=== Phase 4: Stochastic restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(10):
            if time.monotonic() > deadline - 60:
                break
            remaining = min((deadline - time.monotonic()) - 60, 35.0)
            if remaining < 10:
                break

            # Perturb best sequence
            sigma = 0.03 + 0.02 * rng.random()
            a_perturbed = best_seq * np.exp(rng.normal(0, sigma, size=n_best))
            a_perturbed = np.clip(a_perturbed, 0.01, None)
            a_perturbed = normalize_seq(a_perturbed, np.sqrt(2.0 * n_best))

            result, score = optimize_sequence(a_perturbed, remaining, verbose=False)
            if score < best_score:
                update_best(result)
            else:
                print(f"  Restart {restart}: score={score:.6f} (no improvement)")

    # === Phase 5: Final intensive LP on best ===
    print("\n=== Phase 5: Final intensive optimization ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 30:
            print(f"  Final LP, budget={remaining:.0f}s")
            result = lp_refine_improved(best_seq, remaining * 0.6, verbose=True)
            s = local_score(result)
            if s < best_score:
                best_score = s
                best_seq = result
                print(f"  After final LP: score={best_score:.6f}")

            # Multi-peak on the result
            remaining2 = deadline - time.monotonic() - 10
            if remaining2 > 15:
                result = multi_peak_lp(best_seq, K=7, time_limit=remaining2 * 0.5, verbose=False)
                s = local_score(result)
                if s < best_score:
                    best_score = s
                    best_seq = result
                    print(f"  After final multi-peak: score={best_score:.6f}")

            # Stochastic polish
            remaining3 = deadline - time.monotonic() - 5
            if remaining3 > 10:
                result = stochastic_local_search(best_seq, time_limit=remaining3 * 0.7, verbose=False)
                s = local_score(result)
                if s < best_score:
                    best_score = s
                    best_seq = result
                    print(f"  After stochastic polish: score={best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate={final_score:.6f}, local={best_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
