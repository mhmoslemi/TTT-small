def fast_convolve(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def objective(a):
    """Compute 2*n*max_conv for a probability vector (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def multiplicative_update(f, tau, eta, n_iter=1000):
    """Multiplicative weight updates to minimize max convolution."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj, _ = objective(f)
    
    for it in range(n_iter):
        conv = fast_convolve(f)
        cmax = conv.max()
        if cmax < 1e-15:
            break
        # Softmax weights
        scaled = conv / max(tau, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute pressure: p[i] = sum_k w[k] * a[k-i]
        # This is a cross-correlation of w with f
        # p = ifft(fft(w) * conj(fft(f))) ... but w has length 2n-1 and f has length n

        p = np.zeros(n)
        for i in range(n):

            # dc[k]/df[i] = f[k-i] for k from i to i+n-1 (clipped to 0..2n-2)

            idx_start = k_start - i
            idx_end = k_end - i
            p[i] = np.dot(w[k_start:k_end+1], f[idx_start:idx_end+1])
        
        # Update

        lf -= lf.max()
        f_new = f * np.exp(lf)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-12:

        else:

        obj_new, _ = objective(f_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_obj

def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

def make_lp_matrix(n):
    """Precompute sparse structure for (f*g) constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    return rows, cols


def solve_lp(f, M, rows, cols, n, timeout=10.0):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

            f_new = (1.0 - t_try) * f + t_try * g
            f_new = normalize(f_new)
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.8
            if patience > 40:
                break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {it}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def greedy_peak_shave(f, time_limit):
    """Greedy local search: reduce peak contributors, redistribute to low-pressure positions."""
    n = len(f)
    f = normalize(f.copy())
    best_f = f.copy()
    best_obj, _ = compute_obj(f)

    t0 = time.monotonic()
    it = 0
    patience = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        c_max = conv[k_star]

        # Find main contributor to the peak
        # c[k_star] = sum_i f[i] * f[k_star - i]
        contributors = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors[i] = f[i] * f[j]

        i_max = int(np.argmax(contributors))
        if contributors[i_max] < 1e-10:
            break

        # Compute pressure for redistribution
        p = compute_pressure(conv, f, power=3.0)
        p_min_idx = int(np.argmin(p))

        # Try reducing f[i_max] by delta and adding to f[p_min_idx]
        delta = f[i_max] * 0.1
        f_new = f.copy()
        f_new[i_max] -= delta
        f_new[p_min_idx] += delta
        f_new = normalize(f_new)

        obj_new, _ = compute_obj(f_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            patience = 0
        else:
            patience += 1
            # Try smaller delta
            delta = f[i_max] * 0.03
            f_new2 = f.copy()
            f_new2[i_max] -= delta
            f_new2[p_min_idx] += delta
            f_new2 = normalize(f_new2)
            obj_new2, _ = compute_obj(f_new2)
            if obj_new2 < best_obj - 1e-12:
                best_obj = obj_new2
                best_f = f_new2.copy()
                f = f_new2
                patience = 0
            else:
                if patience > 100:
                    break

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Greedy iter {it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def spectral_init(n, rng):
    """Create an initial distribution using spectral shaping."""
    f = np.ones(n) / n
    fft_size = 2 * (2 * n - 1)
    for _ in range(50):
        A = np.fft.fft(f, fft_size)
        mags = np.abs(A)
        # Flatten non-DC magnitudes toward mean
        target = mags[1:].mean()
        blend = 0.3
        new_mags = mags.copy()
        new_mags[1:] = (1 - blend) * mags[1:] + blend * target
        phases = np.angle(A)
        new_A = new_mags * np.exp(1j * phases)
        f = np.real(np.fft.ifft(new_A, fft_size))[:n]
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s
    return f


def gen_inits(n, rng):
    """Generate diverse initializations including asymmetric patterns."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uni'] = np.ones(n) / n

    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # Asymmetric bump (left-heavy)
    for frac in [0.3, 0.4, 0.6]:
        center = frac * n
        g = np.exp(-0.5 * ((x - center) / (0.2 * n)) ** 2)
        g /= g.sum()
        inits[f'asym{int(frac * 100)}'] = g

    # Two bumps (asymmetric positions)
    for c1, c2 in [(0.2, 0.7), (0.3, 0.8), (0.15, 0.6)]:
        g1 = np.exp(-0.5 * ((x - c1 * n) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - c2 * n) / (0.1 * n)) ** 2)
        g = g1 + g2
        g /= g.sum()
        inits[f'dual{int(c1 * 100)}{int(c2 * 100)}'] = g

    # Cosine pattern
    for k in [1, 2, 3, 5]:
        cb = 0.5 + 0.4 * np.cos(2 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
        cb = np.maximum(cb, 0.01)
        cb /= cb.sum()
        inits[f'cos{k}'] = cb

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Uniform ramp down
    r2 = np.arange(n, 0, -1, dtype=np.float64)
    inits['ramp2'] = r2 / r2.sum()

    # Spectral
    inits['spec'] = spectral_init(n, rng)

    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr

    # Block pattern
    blk = np.zeros(n)
    w = max(1, n // 8)
    for start in range(0, n, 2 * w):
        blk[start:start + w] = 1.0
    blk /= blk.sum()
    inits['block'] = blk

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: multiplicative updates + LP + greedy polishing."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    print(f"Parent available: {parent is not None}, deadline in {deadline - time.monotonic():.0f}s")

    n = 512
    rows, cols = make_lp_matrix(n)
    all_results = []

    # ===== Phase 1: Multiplicative updates from diverse inits =====
    print("Phase 1: Multiplicative updates")
    p1_deadline = deadline - 900.0  # Reserve ~900s for Phase 2+3

    inits = gen_inits(n, rng)

    if parent is not None:
        if parent.size >= n:
            inits['parent'] = normalize(parent[:n].copy())
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            inits['parent'] = normalize(pp)

    # Sort inits: parent first, then others
    init_names = list(inits.keys())
    if 'parent' in init_names:
        init_names.remove('parent')
        init_names = ['parent'] + init_names

    remaining_p1 = max(30.0, p1_deadline - time.monotonic())
    time_per_init = max(5.0, remaining_p1 / len(init_names))

    for name in init_names:
        if time.monotonic() > p1_deadline:
            break
        ta = min(time_per_init, p1_deadline - time.monotonic())
        if ta < 3:
            break

        a0 = inits[name]
        print(f"  Init '{name}' ({ta:.0f}s budget)")

        # Try multiple parameter settings
        results_for_init = []
        params = [
            (2.0, 0.5, 0.0),
            (3.0, 0.4, 0.0),
            (1.5, 0.3, 0.0),
            (4.0, 0.6, 0.001),
        ]
        time_per_param = ta / len(params)

        for power_p, eta_p, shrink_p in params:
            if time.monotonic() > p1_deadline:
                break
            tr = min(time_per_param, p1_deadline - time.monotonic())
            if tr < 2:
                break

            f_opt, obj_opt = mult_update(a0, time_limit=tr, tau_power=power_p, eta0=eta_p, shrink_thresh=shrink_p)
            score = float(evaluate_sequence(f_opt.tolist()))
            results_for_init.append((score, f_opt))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f_opt.copy()
                print(f"    New best: {score:.6f} (init={name}, power={power_p})")

        all_results.extend(results_for_init)

    print(f"Phase 1 done. Best: {best_score:.6f}, {len(all_results)} candidates")

    # ===== Phase 2: LP refinement on top candidates =====
    print("Phase 2: LP refinement")
    all_results.sort(key=lambda x: x[0])
    top_k = min(6, len(all_results))
    top_candidates = all_results[:top_k]

    p2_deadline = deadline - 300.0
    remaining_p2 = max(30.0, p2_deadline - time.monotonic())
    time_per_lp = remaining_p2 / max(1, top_k)

    lp_results = []
    for idx, (sc, f) in enumerate(top_candidates):
        if time.monotonic() > p2_deadline:
            break
        ta = min(time_per_lp, p2_deadline - time.monotonic())
        if ta < 8:
            break

        print(f"  LP [{idx}] score={sc:.6f} ({ta:.0f}s)")
        f_lp, obj_lp = lp_optimize(f, time_limit=ta * 0.65, rows=rows, cols=cols)
        sc_lp = float(evaluate_sequence(f_lp.tolist()))

        if np.isfinite(sc_lp):
            lp_results.append((sc_lp, f_lp))
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = f_lp.copy()
                print(f"    LP improved: {sc_lp:.6f}")

        # Follow up with multiplicative update on LP result
        if time.monotonic() < p2_deadline - 5:
            tr2 = min(30.0, p2_deadline - time.monotonic() - 5)
            f_mu, obj_mu = mult_update(f_lp, time_limit=tr2, tau_power=3.0, eta0=0.3)
            sc_mu = float(evaluate_sequence(f_mu.tolist()))
            if np.isfinite(sc_mu):
                lp_results.append((sc_mu, f_mu))
                if sc_mu < best_score:
                    best_score = sc_mu
                    best_seq = f_mu.copy()
                    print(f"    MU after LP: {sc_mu:.6f}")

    all_results.extend(lp_results)
    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ===== Phase 3: Greedy peak shaving + final polishing =====
    print("Phase 3: Greedy + final polish")
    if best_seq is not None:
        # Greedy peak shaving
        tr_greedy = min(120.0, deadline - time.monotonic() - 60)
        if tr_greedy > 10:
            print(f"  Greedy peak shaving ({tr_greedy:.0f}s)")
            f_g, obj_g = greedy_peak_shave(best_seq, time_limit=tr_greedy)
            sc_g = float(evaluate_sequence(f_g.tolist()))
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = f_g.copy()
                print(f"    Greedy improved: {sc_g:.6f}")

        # Final multiplicative polishing with different parameters
        remaining = deadline - time.monotonic() - 10
        polish_params = [
            (2.0, 0.2, 0.0),
            (5.0, 0.15, 0.0),
            (3.0, 0.1, 0.005),
            (1.0, 0.25, 0.0),
        ]
        time_per_polish = max(5.0, remaining / len(polish_params))
        for power_p, eta_p, shrink_p in polish_params:
            if time.monotonic() > deadline - 5:
                break
            tr_p = min(time_per_polish, deadline - time.monotonic() - 5)
            if tr_p < 3:
                break
            f_p, _ = mult_update(best_seq, time_limit=tr_p, tau_power=power_p, eta0=eta_p, shrink_thresh=shrink_p)
            sc_p = float(evaluate_sequence(f_p.tolist()))
            if np.isfinite(sc_p) and sc_p < best_score:
                best_score = sc_p
                best_seq = f_p.copy()
                print(f"    Polish (power={power_p}, eta={eta_p}): {sc_p:.6f}")

        # One more LP on the final best
        if time.monotonic() < deadline - 15:
            tr_final_lp = min(60.0, deadline - time.monotonic() - 10)
            f_fl, _ = lp_optimize(best_seq, time_limit=tr_final_lp, rows=rows, cols=cols)
            sc_fl = float(evaluate_sequence(f_fl.tolist()))
            if np.isfinite(sc_fl) and sc_fl < best_score:
                best_score = sc_fl
                best_seq = f_fl.copy()
                print(f"    Final LP: {sc_fl:.6f}")

    # ===== Phase 4: Try n=1024 if time permits =====
    if time.monotonic() < deadline - 200:
        print("Phase 4: Trying n=1024")
        n2 = 1024
        rows2, cols2 = make_lp_matrix(n2)
        inits2 = gen_inits(n2, rng)
        if parent is not None:
            if parent.size >= n2:
                inits2['parent'] = normalize(parent[:n2].copy())
            else:
                pp = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, parent.size), parent)
                inits2['parent'] = normalize(pp)

        tr_each = min(60.0, (deadline - time.monotonic() - 30) / 4)
        for name in ['parent', 'uni', 'g20', 'spec']:
            if name not in inits2 or time.monotonic() > deadline - 30:
                continue
            f_o, _ = mult_update(inits2[name], time_limit=tr_each, tau_power=3.0, eta0=0.4)
            sc = float(evaluate_sequence(f_o.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f_o.copy()
                print(f"  n=1024 {name}: {sc:.6f}")

    if best_seq is None:
        best_seq = np.ones(n)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}")
    return [float(x) for x in best_seq]
