Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Strategy: Multi-start optimization combining soft-max gradient descent, 
direct peak reduction, LP-based refinement, and coordinate descent.
Uses diverse initializations including geometric sequences and perturbed 
previous best solutions.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(c.max())
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=1.0):
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def gradient_max(a):
    """Gradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def gradient_softmax(a, tau):
    """Gradient of soft-max tau * log(sum exp(tau * conv)) w.r.t. a, simplified."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad


def gradient_topk(a, K=3):
    """Gradient using top-K peaks with equal weighting."""
    n = len(a)
    c = np.convolve(a, a)
    top_indices = np.argsort(c)[-K:]
    grad = np.zeros(n)
    for k in top_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += (2.0 / K) * a[j]
    return grad


def lp_direction(a, M, n_max=512):
    """Find LP direction: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    # A[m, j] = a[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            if m < 2 * n - 1:
                A[m, j] = a[m - j]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    n = len(a)

    decay = 0.9
    lr = lr0
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < 50000:
        it += 1
        
        if tau_schedule == 'increasing':

            tau = 10.0

        a = a_new
        # Anneal lr

    return best, best_score
