def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    #---Security Checks---
    if not isinstance(sequence, list):
        return np.inf  # Reject non-list
    
    if not sequence:  # Reject empty list
        return np.inf
        
    # Check each element for validity
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)) or not isinstance(x, np.floating):
            return np.inf  # Reject bool and non-float
        
        if np.isnan(x) or np.isinf(x):
            return np.inf  # Reject NaN and infinity
            
    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]
    
    # Apply defensive clip (just to mimic the evaluator's clipping)
    sequence = [max(0.0, min(1000.0, x)) for x in sequence]
    
    n = len(sequence)
    if n < 1:
        return np.inf
        
    try:
        S = sum(sequence)
        # Protect against sum too close to zero
        if S < 0.01:
            return np.inf
            
        # Compute full linear convolution
        convolution = np.convolve(sequence, sequence, mode='full')
        max_conv = float(convolution.max())
        
        result = float(2.0 * n * max_conv / (S ** 2))
        return result
        
    except Exception as e:
        return np.inf

    if deadline - time.monotonic() > 60:
        print("Further refinement would benefit, but time is limited")
        return best_candidate
    else:
        print("Time budget exhausted for the search")
        return best_candidate
