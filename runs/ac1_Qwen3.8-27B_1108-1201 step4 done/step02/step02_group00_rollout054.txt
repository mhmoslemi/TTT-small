"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with smooth max (log-sum-exp) approximation and softmax parameterization
2. Multi-scale (multi-T) optimization
3. Multi-length search
4. Projected gradient polishing
5. Multi-start with diverse initializations

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(conv, T):
    cmax = conv.max()
    if T < 1e-12:
        return cmax, np.zeros_like(conv)
    shifted = np.clip((conv - cmax) / T, -50.0, 0.0)
    e = np.exp(shifted)
    weights = e / e.sum()
    return cmax + T * np.log(e.sum()), weights

def lbfgs_smooth(z_init, T, maxiter=5000, tol=1e-10):
    """Optimize using L-BFGS with softmax parameterization."""
    n = len(z_init)
    
    def fun_and_grad(z):
        a = np.exp(z - z.max())
        a /= a.sum()
        conv = fast_convolve(a)
        cmax = conv.max()
        shifted = np.clip((conv - cmax) / T, -50.0, 0.0)
        e = np.exp(shifted)
        e_sum = e.sum()
        w = e / e_sum
        obj = 2.0 * n * (cmax + T * np.log(e_sum))
        
        # Gradient w.r.t. a: d(obj)/da[m] = 2n * sum_k w[k] * d(conv[k])/da[m]
        # d(conv[k])/da[m] = 2*a[k-m] if 0<=k-m<n
        # So d(obj)/da[m] = 4n * sum_k w[k]*a[k-m] = 4n * (w cross-correlate a)[m]
        
        # Cross-correlation via FFT
        a_flipped = a[::-1]
        fft_size = 1 << (2 * n + n - 2).bit_length()
        W = np.fft.rfft(w, fft_size)
        Af = np.fft.rfft(a_flipped, fft_size)
        cross = np.fft.irfft(W * Af, fft_size)
        grad_a = 4.0 * n * cross[n - 1: 2 * n - 1]
        
        # Chain rule through softmax: d(a[m])/d(z[p]) = a[m]*(delta[m,p] - a[p])
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        return obj, grad_z
    
    result = minimize(fun_and_grad, z_init, method='L-BFGS-B',
                      jac=True, options={'maxiter': maxiter, 'gtol': tol,
                                         'maxfun': maxiter * 2})
    z_opt = result.x
    a_opt = np.exp(z_opt - z_opt.max())
    a_opt /= a_opt.sum()
    return a_opt, result.fun, result

def projected_gradient(a_init, T, time_limit, lr_init=0.01, momentum=0.9):
    """Projected gradient descent on the simplex."""
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a /= s
    n = len(a)
    v = np.zeros(n)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth max gradient
        cmax = conv.max()
        shifted = np.clip((conv - cmax) / T, -50.0, 0.0)
        e = np.exp(shifted)
        w = e / e.sum()
        
        # Gradient of smooth_max w.r.t. a
        a_flipped = a[::-1]
        fft_size = 1 << (2 * n + n - 2).bit_length()
        W = np.fft.rfft(w, fft_size)
        Af = np.fft.rfft(a_flipped, fft_size)
        cross = np.fft.irfft(W * Af, fft_size)
        grad = 4.0 * n * cross[n - 1: 2 * n - 1]
        
        # Center the gradient (project onto tangent space of simplex)
        grad -= grad.mean()
        
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur = 2.0 * n * conv.max()
        
        if cur < best_obj - 1e-12:
            best_obj = cur
            best_a = a.copy()
        
        it += 1
        lr *= 0.9995
        
        if it % 1000 == 0:
            print(f"    PG iter {it}: obj={best_obj:.6f}, lr={lr:.6f}")
    
    return best_a, best_obj

def make_init(n, parent, mode, rng):

    elif mode == 'ramp':

    elif mode == 'ramp_dec':

    elif mode == 'triangle':

    elif mode == 'gauss_center':

    elif mode == 'gauss_edge':

    elif mode == 'bimodal':

    elif mode == 'random':

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation method with L-BFGS on softmax parameterization (smooth max via log-sum-exp)
2. Multi-length search (128-512)

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smgrad_z(z, T):
    """
    Compute objective and gradient for softmax parameterization.
    a = softmax(z), minimize 2n * smooth_max(conv(a,a), T).
    """
    n = len(z)
    a = np.exp(z - z.max())
    a_sum = a.sum()
    a = a / a_sum
    
    conv = fast_convolve(a)
    
    cmax = conv.max()
    shifted = np.clip((conv - cmax) / T, -50.0, 0.0)
    e = np.exp(shifted)
    e_sum = e.sum()
    w = e / e_sum
    
    obj = 2.0 * n * (cmax + T * np.log(e_sum + 1e-30))
    
    # Cross-correlation of w (len 2n-1) and a (len n)
    # (w cross a)[m] = sum_k w[k] * a[k-m] for m = 0,...,n-1
    a_flipped = a[::-1]
    fft_size = 1 << (2 * n + n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a_flipped, fft_size)
    cross = np.fft.irfft(W * Af, fft_size)
    grad_a = 4.0 * n * cross[n - 1: 2 * n - 1]
    
    # Chain rule through softmax
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lbfgs_optimize(a_init, T, maxiter=5000, gtol=1e-8):
    """Run L-BFGS-B on softmax parameterization."""
    z0 = np.log(a_init + 1e-18)
    
    def fun_and_grad(z):
        return compute_smgrad_z(z, T)
    
    result = minimize(fun_and_grad, z0, method='L-BFGS-B',
                      jac=True, options={'maxiter': maxiter, 'gtol': gtol,
                                         'maxfun': maxiter * 3, 'ftol': 1e-12})
    z_opt = result.x
    a_opt = np.exp(z_opt - z_opt.max())
    a_opt /= a_opt.sum()
    return a_opt, result.fun, result.nit


def continuation_lbfgs(a_init, T_start=1.0, T_end=0.001, T_factor=0.5,
                       maxiter_per=2000, time_limit=60.0):
    """
    Continuation method: start with large T, gradually decrease.
    """
    a = a_init.copy()
    t0 = time.monotonic()
    T = T_start
    
    while T > T_end and time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        maxit = max(500, int(maxiter_per * remaining / time_limit))
        a, val, nit = lbfgs_optimize(a, T, maxiter=maxit, gtol=1e-7)
        T *= T_factor
        if nit == 0:
            break
    
    return a


def projected_gradient_refine(a_init, T, time_limit, lr_init=0.003, momentum=0.9):

    a = project_simplex(a_init.copy())
    n = len(a)
    v = np.zeros(n)

def make_init(n, parent, mode, rng):
    """Create initial sequence of length n."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif mode == 'ramp_dec':
        a = np.arange(n, 0, -1, dtype=np.float64)
        return a / a.sum()
    elif mode == 'triangle':
        x = np.arange(n, dtype=np.float64)
        a = np.minimum(x, n - 1 - x) + 1.0
        return a / a.sum()
    elif mode == 'gauss_center':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - (n - 1) / 2) / (n / 4)) ** 2)
        return a / a.sum()
    elif mode == 'gauss_left':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-x / (n / 3))
        return a / a.sum()
    elif mode == 'bimodal':
        a = np.zeros(n)
        edge = max(1, n // 4)
        a[:edge] = 1.0
        a[-edge:] = 1.0
        return a / a.sum()
    elif mode == 'bimodal_soft':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - n * 0.15) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.85) / (n * 0.1)) ** 2)
        return a / a.sum()
    elif mode == 'random':
        return rng.dirichlet(np.ones(n) * 0.3)
    elif mode == 'parent':
        if parent is not None:
            if len(parent) >= n:
                p = parent[:n].copy()
            else:
                p = np.interp(np.linspace(0, 1, n),
                             np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 1e-15)
            return p / p.sum()
        return np.ones(n) / n
    elif mode == 'sine':
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x / n)
        return a / a.sum()
    elif mode == 'sine2':
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.7 * np.sin(4 * np.pi * x / n + np.pi / 4)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n

        for T_final in [0.001, 0.0005, 0.0003]:
            if remaining < 3:
                break
            a_opt, val, nit = lbfgs_optimize(a_cur, T_final, maxiter=5000, gtol=1e-10)
            score = update_best(a_opt, f"polish_T={T_final},nit={nit}")
            if a_opt is not None and score < best_score:
                a_cur = a_opt
            remaining = deadline - time.monotonic() - 10.0
        
        # Projected gradient fine polish
        if remaining > 5:
            a_pg = projected_gradient_refine(a_cur, 0.001, time_limit=min(30.0, remaining * 0.5),
                                             lr_init=0.001, momentum=0.93)
            score = update_best(a_pg, "final_pg")
        
        # Try small perturbations + re-optimize
        if remaining > 10:
            n_best = len(best_seq)
            for _ in range(3):
                if remaining < 4:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_seq + noise)
                a_opt, val, nit = lbfgs_optimize(a_pert, 0.001, maxiter=2000, gtol=1e-9)
                score = update_best(a_opt, "perturb")
                remaining = deadline - time.monotonic() - 10.0
    
    # ===== Phase 4: Try a couple more lengths if time permits =====
    print("=== Phase 4: Additional Lengths ===")
    if best_seq is not None:
        best_n = len(best_seq)
        extra_lengths = [l for l in [64, 96, 160, 224, 320, 448, 640, 768]
                        if abs(l - best_n) > best_n * 0.2 and time.monotonic() < deadline - 60]
        
        if extra_lengths:
            for n_try in extra_lengths[:3]:
                if time.monotonic() > deadline - 30:
                    break
                a0 = make_init(n_try, parent, 'parent', rng)
                a_opt = continuation_lbfgs(a0, T_start=0.5, T_end=0.002,
                                           T_factor=0.5, maxiter_per=3000,
                                           time_limit=40.0)
                score = update_best(a_opt, f"extra_n={n_try}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final validation
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but just in case
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_seq)}")
    print(f"Total time: {elapsed:.1f}s")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
