Mixed-method optimization for minimizing peak autocorrelation of non-negative
step functions. Combines soft-max gradient descent (fast, O(n log n) per iter),
active-set LP direction finding (principled but slower), and coordinate descent
polish. Uses multiple starting points and adaptive restarts.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def project_simplex(x, S):
    """Project x onto {a >= 0, sum a = S}."""
    n = len(x)
    lo = -np.max(x) - 1.0
    hi = 0.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        a = np.maximum(0.0, x - mid)
        s = np.sum(a)
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    
    return np.maximum(0.0, x - (lo + hi) / 2.0)


def softmax_gradient(c, f, beta):
    """Compute gradient of soft-max autocorrelation."""
    n = len(f)
    mc = np.max(c)
    scaled = beta * (c - mc)
    w = np.exp(scaled - np.max(scaled))
    w /= np.sum(w)
    # grad[i] = 2 * sum_l w[i+l] * f[l]
    grad = 2.0 * np.correlate(w, f, mode='valid')
    return grad


def softmax_optimize(f0, time_limit, rng, verbose=False):
    """Optimize using soft-max gradient descent."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0.0, None)
    sf = np.sum(f)
    if sf < 0.01:
        f = np.ones(n) * (S / n)
    else:
        f = f * (S / sf)
    
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    beta = 5.0  # Initial sharpness
    lr = 0.01
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        M = np.max(c)
        
        grad = softmax_gradient(c, f, beta)
        # Normalize gradient by M for scale invariance
        grad = grad / (M + 1e-12)
        
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            f_new = f - trial_lr * grad
            f_new = project_simplex(f_new, S)
            new_score = local_score(f_new)

def lp_direction(f, M, top_k=50):
    """Solve LP for descent direction using top-k active constraints."""
    n = len(f)
    c = fft_convolve(f, f)
    if top_k < 2 * n - 1:
        top_pos = np.argsort(c)[-top_k:]
    else:
        top_pos = np.arange(2 * n - 1)
    
    # Build constraint matrix
    A = np.zeros((len(top_pos), n))
    for idx, k in enumerate(top_pos):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_pos), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def make_chirp(n, alpha, rng):

def make_multi_tone(n, rng):

        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)

def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)

def make_u_shape(n, depth, rng):
    """Create a U-shaped sequence (large at edges, small in middle)."""
    t = np.linspace(0, 1, n)
    a = 1.0 - depth * np.sin(np.pi * t)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bimodal(n, gap, rng):
    """Create a bimodal sequence."""
    a = np.ones(n)
    center = n // 2
    half_gap = max(1, int(gap * n))
    # Suppress the middle
    mask = np.abs(np.arange(n) - center) < half_gap
    a[mask] *= 0.3
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

Mixed-method optimization for minimizing peak autocorrelation of non-negative
step functions. Combines soft-max gradient descent (fast per-iteration),

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score (fast local version)."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def project_simplex(x, S):
    """Project x onto {a >= 0, sum a = S} via bisection."""
    n = len(x)
    lo = float(-np.max(x)) - 1.0
    hi = 0.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        a = np.maximum(0.0, x - mid)
        s = np.sum(a)
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14 * max(1.0, abs(lo)):
            break
    return np.maximum(0.0, x - (lo + hi) / 2.0)


def softmax_gradient(c, f, beta):
    """Compute gradient of soft-max of autocorrelation w.r.t. f."""
    n = len(f)
    mc = np.max(c)

    w = np.exp(scaled - np.max(scaled))
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    grad = 2.0 * np.correlate(w, f, mode='valid')
    return grad

            f_new = project_simplex(f_new, S)
            new_score = local_score(f_new)
            if new_score < current_score * (1.0 - 1e-9):
                f = f_new
                current_score = new_score
                lr = min(trial_lr * 1.12, 0.3)
                improved = True
                stall = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break

        if not improved:
            stall += 1
            lr *= 0.85
            if stall > 40:
                f = f * np.exp(rng.normal(0, 0.025, n))
                f = project_simplex(f, S)
                current_score = local_score(f)
                lr = 0.005
                beta = 8.0
                stall = 0
            elif stall > 15:
                beta = min(beta * 1.04, 300.0)

        iter_count += 1
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    softmax: iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, beta={beta:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_f, best_score


def lp_optimize(f0, time_limit, rng, verbose=False):
    """Optimize using LP-based direction finding with full constraint set."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0.0, None)
    sf = np.sum(f)
    if sf < 0.01:
        f = np.ones(n) * (S / n)
    else:
        f = f * (S / sf)

    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.1
    stall = 0
    iter_count = 0

    # Pre-build the full constraint matrix structure
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # We'll rebuild when f changes significantly (but for now, rebuild each iter)

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        M = np.max(c)

        # Use top-K active constraints for speed (K adaptive)
        top_k = min(2 * n - 1, max(50, int(50 + 2 * stall)))
        if top_k < 2 * n - 1:
            top_pos = np.argsort(c)[-top_k:]
        else:
            top_pos = np.arange(2 * n - 1)

        # Build constraint matrix
        A = np.zeros((len(top_pos), n))
        for idx, k in enumerate(top_pos):
            lo_j = max(0, k - n + 1)
            hi_j = min(k, n - 1)
            if hi_j >= lo_j:
                A[idx, lo_j:hi_j + 1] = f[k - hi_j:k - lo_j + 1][::-1]

        c_obj = -np.ones(n)
        b_ub = np.full(len(top_pos), M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': time_limit})
        except Exception:
            break

        if not res.success:
            stall += 20
            if stall > 80:
                break
            continue

        g = np.maximum(res.x, 0.0)
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g * (S / sum_g)

        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0.0, None)
            sf_new = np.sum(f_new)
            if sf_new < 0.01:
                continue
            f_new = f_new * (S / sf_new)
            new_score = local_score(f_new)
            if new_score < current_score * (1.0 - 1e-9):
                f = f_new
                current_score = new_score
                t = min(trial_t * 1.08, 0.35)
                improved = True
                stall = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break

        if not improved:
            stall += 1
            t *= 0.8
            if stall > 50:
                f = f * np.exp(rng.normal(0, 0.035, n))
                f = project_simplex(f, S)
                current_score = local_score(f)
                t = 0.08
                stall = 0
            elif stall > 20 and stall % 15 == 0:
                f = f * np.exp(rng.normal(0, 0.06, n))
                f = project_simplex(f, S)
                current_score = local_score(f)
                t = 0.05

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={t:.4f}, top_k={top_k}, elapsed={elapsed:.1f}s")

    return best_f, best_score
