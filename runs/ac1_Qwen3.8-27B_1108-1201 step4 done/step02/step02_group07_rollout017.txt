from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):

    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))

    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

        A[j:j+n, j] = f
    return A

def solve_lp_direction(f, M, timeout=5.0):
    n = len(f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def optimize_lp(f_init, time_limit, step_init=0.1):
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    step = step_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        b = solve_lp_direction(f, M, timeout=4.0)
        if b is None:
            break
        
        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)
        
        # Adaptive line search
        best_t = step
        best_s = score_fast(f)
        for t_try in [step, 2*step, 0.5*step, 0.25*step, 1.5*step]:
            cand = (1-t_try)*f + t_try*b
            cand = np.clip(cand, 0.0, 1000.0)
            s_cand = score_fast(cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1-best_t)*f + best_t*b
        f = np.clip(f, 0.0, 1000.0)
        it += 1
        
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
            # Increase step if improving
            step = min(step * 1.05, 0.3)
        else:
            stall += 1
            step = max(step * 0.95, 0.01)
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{elapsed:6.1f}s] n={n} it={it:4d} score={best_s:.6f} best={best_score:.6f} step={step:.3f} stall={stall}")
        
        if stall >= 50:

            f += 0.05 * rng.standard_normal(n) * np.mean(f)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
            step = 0.1
    
    return best_f, best_score

def peak_shaving(f, time_limit):
    """Directly reduce elements that contribute to the peak."""
    n = len(f)
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        
        # Compute contributions to the peak
        contribs = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                contribs[j] = f[j] * f[idx]
        
        # Reduce top contributors, increase others
        top_k = max(3, n // 16)
        top_indices = np.argsort(contribs)[-top_k:]
        other_indices = np.argsort(contribs)[:-top_k]
        
        reduce_amt = rng.uniform(0.01, 0.05) * np.mean(f)
        increase_amt = reduce_amt * top_k / max(1, len(other_indices))
        
        f[top_indices] -= reduce_amt * rng.uniform(0.5, 1.0, size=top_k)
        chosen = rng.choice(other_indices, size=min(8, len(other_indices)), replace=False)
        f[chosen] += increase_amt * rng.uniform(0.5, 1.5, size=len(chosen))
        f = np.clip(f, 0.01, 1000.0)
        
        s = score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
    
    return best_f, best_score

def gradient_descent(f_init, time_limit):
    """Subgradient descent on the max convolution."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f / S * np.sqrt(2*n)  # normalize
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f, f)
        # Use top-k peaks for smooth subgradient
        top_k = min(10, 2*n-1)
        top_idx = np.argsort(c)[-top_k:]
        
        # Subgradient of max(c): average of gradients at top peaks
        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - (n-1))
            hi = min(n-1, ki)
            for j in range(lo, hi+1):
                g[j] += 2.0 * f[ki - j]
        g /= float(top_k)
        
        # Center the gradient (since we're on the constraint sum(f) = const)
        g -= np.mean(g)
        
        momentum = beta * momentum + (1-beta) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, 1000.0)
        f = f / float(np.sum(f)) * np.sqrt(2*n)
        
        it += 1
        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            print(f"    [grad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score

"""
Optimized LP-based + spectral approach for minimizing peak convolution score.
Uses reduced-constraint LP for speed, adaptive line search, multi-start across lengths,
spectral flattening heuristic, and peak-directed local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def solve_lp_reduced(f, c_conv, M, n_active=40, timeout=4.0):
    """Solve LP with only the top n_active constraints for speed.
    Returns direction b or None."""
    n = len(f)
    # Select the most active constraints
    if len(c_conv) > n_active:
        top_idx = np.argsort(c_conv)[-n_active:]
    else:
        top_idx = np.arange(len(c_conv))
    
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build reduced constraint matrix
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(top_idx):
        ki = int(k)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            A[row, j] = f[ki - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_full(f, M, timeout=5.0):
    """Solve the full LP with all constraints."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search(f, direction, base_score, step_init=0.08):
    """Golden-section style line search for best step size."""
    S = float(np.sum(f))
    Sb = float(np.sum(direction))
    if Sb < 1e-12:
        return 0.0, base_score, f.copy()
    
    direction = direction * (S / Sb)
    best_t = step_init
    best_s = base_score
    best_f = f.copy()
    
    # Try a range of step sizes
    candidates = [step_init, 0.5 * step_init, 0.25 * step_init,
                  2.0 * step_init, 0.1 * step_init, 1.5 * step_init]
    for t_try in candidates:
        if t_try <= 0 or t_try > 1.0:
            continue
        cand = (1.0 - t_try) * f + t_try * direction
        cand = np.clip(cand, 0.0, 1000.0)
        s_cand = score_fast(cand)
        if s_cand < best_s:
            best_s = s_cand
            best_t = t_try
            best_f = cand
    
    return best_t, best_s, best_f


def optimize_lp(f_init, time_limit, use_reduced=True, n_active=40, verbose=True):
    """Main LP-iteration optimizer with adaptive steps and reduced constraints."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    step = 0.08
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 5.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        if use_reduced:
            b = solve_lp_reduced(f, c, M, n_active=n_active, timeout=3.0)
        else:
            b = solve_lp_full(f, M, timeout=4.0)
        
        if b is None:
            # Fallback: try with fewer constraints
            b = solve_lp_reduced(f, c, M, n_active=20, timeout=2.0)
            if b is None:
                break
        
        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        
        # Line search
        t_best, s_best, f_new = line_search(f, b, best_score, step)
        
        if t_best == 0.0:
            stall += 5
            step = max(0.01, step * 0.8)
            if stall > 60:
                break
            continue
        
        f = f_new
        it += 1
        
        if s_best < best_score - 1e-12:
            best_score = s_best
            best_f = f.copy()
            stall = 0
            # Gradually increase step if consistently improving
            step = min(step * 1.03, 0.2)
        else:
            stall += 1
            step = max(step * 0.97, 0.01)
        
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{elapsed:6.1f}s] n={n} it={it:4d} score={best_score:.6f} step={step:.4f} stall={stall}")
        
        if stall >= 60:
            # Escape local optimum
            rng_esc = np.random.default_rng(1000 + it * 31 + n)
            noise_scale = 0.03 * np.mean(f)
            f = f + noise_scale * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            # Restore normalization
            S = float(np.sum(f))
            S_target = float(np.sum(best_f))
            f = f * (S_target / S)
            step = 0.08
            stall = 0
            best_score = score_fast(f)
            if best_score > score_fast(best_f):
                f = best_f.copy()
                best_score = score_fast(f)
    
    return best_f, best_score


def spectral_flatten(f_init, time_limit, verbose=True):
    """Try to flatten the convolution via spectral manipulation."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        # Compute spectrum
        F = rfft(f, fft_size)
        # Compute squared spectrum (gives convolution in frequency domain)
        F_sq = F * F
        # IFFT to get circular convolution
        c_circ = irfft(F_sq, fft_size)
        c_lin = np.concatenate([c_circ[fft_size - n + 1:], c_circ[:n - 1]])
        
        # Find the target level (mean of positive part)
        positive = c_lin[c_lin > 0]
        if len(positive) == 0:
            break
        target = np.max(c_lin) * 0.7
        
        # Modify spectrum: reduce components that contribute to peaks
        # This is a heuristic: reduce the magnitude of F where it creates peaks
        c_peaks = np.where(c_lin > target)[0]
        
        # Adjust F magnitude: reduce where autocorrelation of F is high
        F_mag = np.abs(F)
        F_phase = np.angle(F)
        
        # Reduce peaks in |F|^2
        F_mag_sq = F_mag ** 2
        mean_mag_sq = np.mean(F_mag_sq)
        # Compress the dynamic range
        F_mag_sq_new = np.sqrt(F_mag_sq * mean_mag_sq)  # geometric mean compression
        F_mag_new = np.sqrt(F_mag_sq_new)
        
        # Keep DC component
        F_mag_new[0] = F_mag[0]
        
        F_new = F_mag_new * np.exp(1j * F_phase)
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.clip(f_new, 0.01, 1000.0)
        
        # Blend with original
        alpha = 0.1
        f_cand = (1 - alpha) * f + alpha * f_new
        f_cand = np.clip(f_cand, 0.01, 1000.0)
        
        s_cand = score_fast(f_cand)
        if s_cand < best_score:
            best_score = s_cand
            best_f = f_cand.copy()
            f = f_cand
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"    [spectral {time.monotonic()-t0:6.1f}s] it={it} score={best_score:.6f}")
    
    return best_f, best_score


def peak_directed_search(f_init, time_limit, verbose=True):
    """Directly target the convolution peak with coordinated moves."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        
        # Compute contributions to the peak
        contribs = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                contribs[j] = f[j] * f[idx]
        
        # Strategy: reduce top contributors, redistribute mass
        total_contrib = np.sum(contribs)
        if total_contrib < 1e-15:
            break
        
        # Find elements with high contribution
        threshold = np.percentile(contribs[contribs > 0], 80)
        high_mask = contribs > threshold
        low_mask = (contribs <= threshold) & (f > 0.01)
        
        if not np.any(high_mask) or not np.any(low_mask):
            # Try random perturbation
            f += 0.02 * np.mean(f) * rng.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            it += 1
            continue
        
        # Reduce high contributors proportionally to their contribution
        reduce_frac = np.array([min(0.15, 0.05 + 0.02 * (contribs[i] / max(total_contrib, 1e-15) * 10)) 
                                for i in np.where(high_mask)[0]])
        high_indices = np.where(high_mask)[0]
        
        mass_removed = 0.0
        for idx, frac in zip(high_indices, reduce_frac):
            removed = f[idx] * frac
            f[idx] -= removed
            mass_removed += removed
        
        # Distribute mass to low contributors
        low_indices = np.where(low_mask)[0]
        n_low = len(low_indices)
        if n_low > 0:
            # Distribute more to elements that would reduce the peak
            # (elements far from the peak alignment)
            dist_scores = []
            for i in low_indices:
                dist_scores.append(abs(i - (peak_idx - i)))  # distance from self-alignment
            dist_scores = np.array(dist_scores)
            probs = dist_scores / max(1.0, np.sum(dist_scores))
            probs = np.clip(probs, 0.01, None)
            probs = probs / np.sum(probs)
            
            # Give mass to a fraction of low contributors
            n_add = max(3, n_low // 4)
            add_indices = low_indices[rng.choice(n_low, size=n_add, replace=False, p=probs)]
            
            per_add = mass_removed / n_add
            for i in add_indices:
                f[i] += per_add
        
        f = np.clip(f, 0.01, 1000.0)
        
        # Also try increasing some low elements and decreasing high ones
        s = score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    [peak {time.monotonic()-t0:6.1f}s] it={it} score={best_score:.6f}")
    
    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=True):
    """Subgradient descent with momentum on the max convolution."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (np.sqrt(2.0 * n) / S)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.005
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f, f)
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(c)[-top_k:]
        
        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(top_k)
        g -= np.mean(g)
        
        momentum = beta * momentum + (1.0 - beta) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, 1000.0)
        S = float(np.sum(f))
        f = f * (np.sqrt(2.0 * n) / S)
        
        it += 1
        if it % 500 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [grad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
            # Adaptive learning rate
            if s < best_score + 0.001:
                lr = min(lr * 1.1, 0.02)
            else:
                lr = max(lr * 0.95, 0.001)
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def make_init(n, style, rng):
    """Generate initial sequence of given style."""
    t = np.arange(n, dtype=np.float64)
    if style == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.03 * rng.standard_normal(n) * np.mean(a)
    elif style == 'raised':
        a = np.cos(np.pi * t / max(1, n - 1)) ** 2
        a += 0.02 * rng.standard_normal(n) * np.mean(a)
    elif style == 'gaussian':
        sigma = n / 4.0
        a = np.exp(-0.5 * ((t - n / 2.0) / sigma) ** 2)
        a += 0.02 * rng.standard_normal(n) * np.mean(a)
    elif style == 'cosh':
        a = 1.0 / np.cosh((t - n / 2.0) / (n / 6.0))
        a += 0.02 * rng.standard_normal(n) * np.mean(a)
    elif style == 'sqrt':
        x = t / max(1, n - 1)
        a = np.sqrt(x * (1 - x))
        a += 0.02 * rng.standard_normal(n) * np.mean(a)
    else:
        a = 0.5 + 0.5 * np.sin(np.pi * t / max(1, n - 1))
        a += 0.03 * rng.standard_normal(n) * np.mean(a)
    
    return np.clip(a, 0.01, None)
