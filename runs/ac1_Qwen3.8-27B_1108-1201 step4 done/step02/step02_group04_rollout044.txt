"""
Minimizes 2*n*max(auto_conv(a))/(sum(a))^2 using:
1. LP-based alternating optimization with peak targeting
2. Simulated annealing for exploration
3. Multi-start with diverse perturbations
4. Final coordinate descent polishing
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def norm_obj(a):
    """Compute normalized objective 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_lp_matrix(f, n):
    """Build the constraint matrix B where B[k,j] = f[k-j] for valid indices."""
    B_rows = []
    B_cols = []
    B_vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            B_vals.append(f[k - j])
            B_rows.append(k)
            B_cols.append(j)
    B = sparse.csc_matrix((B_vals, (B_rows, B_cols)), shape=(2 * n - 1, n))
    return B

def solve_lp_direction(f, M, n, peak_frac=0.0, reduction=0.005, timeout=10):
    """Solve LP to find direction g.
    If peak_frac > 0, target peak reduction."""
    B = build_lp_matrix(f, n)
    
    conv = fast_convolve(f)
    
    if peak_frac > 0:
        # Identify peak positions (top peak_frac fraction of max)
        threshold = M * (1.0 - peak_frac)
        peak_mask = conv >= threshold
        b = M * (1.0 + reduction)
        b[peak_mask] = M * (1.0 - reduction)
    else:
        b = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=B, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-12:
            return g / g.sum()
    return None

def lp_optimize(f0, time_limit, n, peak_frac=0.05, reduction=0.005, verbose=True):
    """LP-based alternating optimization with peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        g = solve_lp_direction(f, M, n, peak_frac=peak_frac, reduction=reduction)
        if g is None:
            # Try without peak targeting
            g = solve_lp_direction(f, M, n, peak_frac=0.0, reduction=0.0)
            if g is None:
                break
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t, 0.05*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                conv = conv_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.02)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.9
            if patience > 40:
                break
        
        if verbose and iter_count % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

def sa_optimize(f0, time_limit, n, rng, verbose=True):

    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    
    conv = fast_convolve(f)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_f = f.copy()
    
    T = 0.005
    T_min = 1e-7
    cooling = 0.9995
    
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        r = rng.random()
        
        if r < 0.3:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * (T / 0.005)
            noise = rng.normal(0, scale, n)
            f_new = f + noise
        elif r < 0.5:
            # Targeted peak reduction
            conv_cur = fast_convolve(f)
            peak_k = int(np.argmax(conv_cur))
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            # Contributions to peak
            idx = np.arange(j_start, j_end + 1)
            contrib = f[idx] * f[peak_k - idx]
            # Reduce the largest contributor
            max_local = int(np.argmax(contrib))
            max_idx = idx[max_local]
            f_new = f.copy()
            move = 0.005 * f[max_idx] * rng.random()
            f_new[max_idx] -= move
            # Add to a random position far from peak
            candidates = np.setdiff1d(np.arange(n), idx)
            if len(candidates) > 0:
                add_idx = candidates[rng.integers(len(candidates))]
                f_new[add_idx] += move
        elif r < 0.7:
            # Swap
            i, j = rng.choice(n, 2, replace=False)
            f_new = f.copy()
            delta = 0.01 * min(f[i], f[j]) * rng.random()
            f_new[i] -= delta
            f_new[j] += delta
        elif r < 0.85:
            # Block perturbation: modify a random block
            block_size = rng.integers(1, max(2, n // 32))
            start = rng.integers(0, n - block_size)
            f_new = f.copy()
            f_new[start:start+block_size] *= (1.0 + rng.normal(0, 0.1, block_size))

            f_new = f.copy()
            mid = n // 2

            edge_region_candidates = np.concatenate([np.arange(n//8), np.arange(n - n//8, n)])

                target = edge_region_candidates[rng.integers(len(edge_region_candidates))]
                f_new[target] += move
        
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s
        
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        delta_obj = obj_new - current_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            f = f_new
            current_obj = obj_new
            if current_obj < best_obj:
                best_obj = current_obj
                best_f = f.copy()
        
        T = max(T * cooling, T_min)
        
        if T == T_min and step % 2000 == 0:
            T = 0.003  # Reheat
            if verbose:
                print(f"    SA reheat at step {step}, best={best_obj:.8f}")

            print(f"    SA step {step}, current={current_obj:.8f}, best={best_obj:.8f}, T={T:.6f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

        # Find the best coordinate to adjust
        best_i = -1

        # Also add some random candidates

                f_trial = f.copy()
                f_trial[i] -= amt

                min_k_candidates = np.argsort(conv)[:10]
                for add_k in min_k_candidates:
