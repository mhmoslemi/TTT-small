"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed FISTA (accelerated
projected gradient) with adaptive temperature, multi-restart exploration,
and coordinate-descent polishing. Key innovations over prior approach:
(1) log-sum-exp smoothing provides dense gradient from all peaks simultaneously,
(2) FISTA acceleration for O(1/sqrt(T)) convergence,
(3) adaptive temperature schedule that transitions from exploration to exploitation,
(4) two-coordinate deskew as a polishing step.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Scale to target_sum / 1 = target_sum
    # Project onto {x >= 0, sum(x) = S} where S = target_sum
    # This is equivalent to projecting (v) onto the S-simplex
    # Standard trick: project onto probability simplex, then scale by S
    # But simpler: use the standard algorithm with S instead of 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_val = u * np.arange(1, n + 1) - (cssv - target_sum)
    cond = rho_val > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_logsumexp(C, t):
    """Compute logsumexp(C/t)*t and the softmax weights."""
    C_max = C.max()
    if C_max < 1e-30:
        return C_max, np.full(len(C), 1.0 / len(C))
    shifted = (C - C_max) / t
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    val = C_max + t * np.log(w_sum)
    return val, w


def compute_smooth_gradient(w, a):
    """
    Compute gradient of logsumexp(conv(a,a)/t) w.r.t. a.
    grad[m] = sum_k w[k] * dc_k/da_m = sum_k w[k] * 2*a_{k-m} (for valid indices)
    = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    """
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = 2.0 * np.dot(w[m:m+n], a)
    return grad


def fista_smoothed(a0, time_limit, t_init=None, t_final=1e-6, 
                    momentum_init=0.5, momentum_growth=1.001,
                    n_restarts=5, restart_frac=0.2, perturb_scale=0.01):

    FISTA with log-sum-exp smoothing for minimizing 2n*max(conv)/(sum)^2
    on the simplex {a >= 0, sum(a) = 1}.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / a_sum
    n = len(a)
    
    # Compute initial values
    C = fast_autocorr(a)
    C_max = C.max()
    if t_init is None:
        t_init = C_max / 20.0
    t = t_init
    
    best_obj = 2.0 * n * C_max  # Using exact max for tracking
    best_a = a.copy()
    
    t0 = time.monotonic()
    total_steps = 0
    no_improve = 0
    max_no_improve = 500
    
    # FISTA variables
    y = a.copy()
    v = a.copy()
    k_idx = 1.0  # step counter for momentum
    
    # Step size initialization
    step = 1.0 / np.linalg.norm(compute_smooth_gradient(*smooth_logsumexp(C, t), a))**2 + 1e-8
    # Actually, let's use a fixed step size based on problem scale
    # L (Lipschitz constant) ~ 2 * 2n * max(a) / t (rough)
    # For now, use backtracking
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth objective and gradient at y
        smooth_val, w = smooth_logsumexp(C, t)
        grad = compute_smooth_gradient(w, a)  # gradient w.r.t. a (will recompute)
        
        # Actually, we need gradient at y, not a
        C_y = fast_autocorr(y)
        smooth_val_y, w_y = smooth_logsumexp(C_y, t)
        grad_y = compute_smooth_gradient(w_y, y)
        
        # Objective value (using exact max for tracking)
        obj_y = 2.0 * n * C_y.max()
        
        # FISTA update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k_idx**2)) / 2.0
        momentum = (k_idx - 1.0) / t_k
        momentum = min(momentum, momentum_growth ** (k_idx % 1000))
        
        v_new = project_simplex(y - step * grad_y)
        y = v_new + momentum * (v_new - a)
        y = project_simplex(y)  # ensure y stays on simplex
        a = v_new
        
        k_idx += 1.0
        
        # Update C and check improvement
        C = fast_autocorr(a)
        cur_obj = 2.0 * n * C.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        # Cool down over time
        progress = (time.monotonic() - t0) / time_limit
        t = t_init * (t_final / t_init) ** progress
        
        # Adaptive step size (simple geometric decay)
        step *= 0.9999
        
        # Restart if stuck
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 10000)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(best_a + noise)
            y = a.copy()
            v = a.copy()
            C = fast_autocorr(a)
            k_idx = 1.0
            step = 1.0 / (2.0 * n * a.max() / t + 1e-8)
            no_improve = 0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale smoothed projected
gradient (log-sum-exp approximation of max), FISTA acceleration, L-BFGS-B
polishing, flat-autocorrelation initialization, and multi-restart exploration.
Key ideas: (1) dense smooth gradient from all convolution peaks simultaneously
via softmax weighting, (2) adaptive temperature cooling schedule, (3) Nesterov
momentum for faster convergence, (4) structured initialization from theoretical
flat-autocorrelation recursion, (5) L-BFGS-B for final polishing.

import numpy as np
import time
from scipy.optimize import minimize


def fast_autocorr(a):
    """Linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_logsumexp(C, t):
    """Compute t*log(sum(exp(C/t))) and softmax weights. Returns (value, weights)."""
    C_max = C.max()
    if C_max < 1e-30:
        return 0.0, np.full(len(C), 1.0 / len(C))
    shifted = (C - C_max) / t
    shifted = np.clip(shifted, -200, 200)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(C))
        w[np.argmax(C)] = 1.0
        return C_max, w
    w = w / w_sum
    val = C_max + t * np.log(w_sum)
    return val, w


def compute_grad(w, a):
    """
    Gradient of logsumexp(conv(a,a)/t) w.r.t. a.
    grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    """
    n = len(a)
    grad = np.empty(n)
    for m in range(n):
        grad[m] = 2.0 * np.dot(w[m:m + n], a)
    return grad


def project_prob_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_autocorr_init(n):
    """
    Generate initialization by recursively solving for flat autocorrelation.
    c_k = 1/(2n-1) for all k. Gives a good starting point.
    """
    target = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(target)
    for k in range(1, n):
        # c_k = 2*a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j]
        determined = 0.0
        for j in range(1, k):
            determined += a[j] * a[k - j]
        a_k = (target - determined) / (2.0 * a[0])
        a[k] = max(a_k, 0.0)
    s = a.sum()
    if s > 1e-15:

    return a

    n = len(a)

    C = fast_autocorr(a)

    k_counter = 1.0

    if step_init is None:
        # Estimate Lipschitz constant: L ~ 4*n*max(a)/t + 4*n*max(a)
        step = min(1.0, 0.5 / (n * (a.max() / t + a.max()) + 1e-10))
        step = max(step, 1e-4)

    no_improve = 0
    patience = 2000
    rng = np.random.default_rng(12345)

    iter_count = 0
    while time.monotonic() - t0 < time_limit and iter_count < max_iters:
        # Compute objective and gradient at y
        C_y = fast_autocorr(y)
        smooth_val_y, w_y = smooth_logsumexp(C_y, t)
        grad_y = compute_grad(w_y, y)

        # FISTA momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k_counter * k_counter)) / 2.0
        momentum = (k_counter - 1.0) / t_k
        momentum = min(momentum, 0.98)

        # Gradient step + projection
        v = project_prob_simplex(y - step * grad_y)

        # Update y
        y = v + momentum * (v - a)
        y = project_prob_simplex(y)

        a = v
        k_counter += 1.0
        iter_count += 1

        # Track best (using exact max)
        C = fast_autocorr(a)
        cur_obj = 2.0 * n * C.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature cooling
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        t = t_init * (t_min / t_init) ** (progress ** 0.7)

        # Adaptive step size
        step *= 0.99995
        step = max(step, 1e-6)

        # Restart if stuck
        if no_improve >= patience:
            noise = rng.standard_normal(n) * 0.005
            a = project_prob_simplex(best_a + noise)
            y = a.copy()
            k_counter = 1.0
            C = fast_autocorr(a)
            step = min(1.0, 0.5 / (n * (a.max() / t + a.max()) + 1e-10))
            no_improve = 0
            patience = int(patience * 1.5)

    return best_a, best_obj


def optimize_pg_simple(a0, time_limit):
    """
    Simple multi-scale projected gradient without momentum.
    More robust but slower convergence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)

    C = fast_autocorr(a)
    C_max = C.max()
    best_obj = 2.0 * n * C.max()
    best_a = a.copy()

    t0 = time.monotonic()
    temps = [C_max / 4, C_max / 8, C_max / 16, C_max / 32,
             C_max / 64, C_max / 128, C_max / 256, C_max / 512,
             C_max / 1024, C_max / 2048, 1e-7]

    for t in temps:
        if time.monotonic() - t0 > time_limit * 0.95:
            break
        time_per_t = (time_limit * 0.95 - (time.monotonic() - t0)) / (len(temps) - temps.index(t))
        step = min(1.0, 0.3 / (n * a.max() / t + 1e-10))

        for _ in range(100000):
            if time.monotonic() - t0 > time_limit * 0.95:
                break
            C = fast_autocorr(a)
            _, w = smooth_logsumexp(C, t)
            grad = compute_grad(w, a)
            a = project_prob_simplex(a - step * grad)
            cur_obj = 2.0 * n * C.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            step *= 0.9999

    return best_a, best_obj


def optimize_lbfgs(a0, time_limit, t_smooth=0.01):
    """
    L-BFGS-B polishing using log-sum-exp smoothed objective.
    Works on the simplex via (n-1) free variables.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)

    # Use n-1 free variables (drop last)
    x0 = a[:-1].copy()

    def objective(x):
        a_full = np.empty(n)
        a_full[:-1] = x
        a_full[-1] = max(1.0 - x.sum(), 0.0)
        a_full = np.maximum(a_full, 0.0)
        C = fast_autocorr(a_full)
        val, _ = smooth_logsumexp(C, t_smooth)
        return 2.0 * n * val

    def gradient(x):
        a_full = np.empty(n)
        a_full[:-1] = x
        a_full[-1] = max(1.0 - x.sum(), 0.0)
        a_full = np.maximum(a_full, 0.0)
        C = fast_autocorr(a_full)
        _, w = smooth_logsumexp(C, t_smooth)
        grad_full = compute_grad(w, a_full)
        return grad_full[:-1].copy()

    deadline = time.monotonic() + time_limit
    stop_flag = [False]

    def callback(xk):
        if time.monotonic() > deadline:
            stop_flag[0] = True
            return True
        return None

    bounds = [(0.0, 1.0)] * (n - 1)

    try:
        result = minimize(objective, x0, method='L-BFGS-B', jac=gradient,
                          bounds=bounds, callback=callback,
                          options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxls': 50})
    except Exception:
        return a0, 2.0 * n * fast_autocorr(a0).max()

    x_opt = np.maximum(result.x, 0.0)
    a_opt = np.empty(n)
    a_opt[:-1] = x_opt
    a_opt[-1] = max(1.0 - x_opt.sum(), 0.0)
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    a_opt = a_opt / s if s > 1e-15 else a_opt

    C = fast_autocorr(a_opt)
    obj = 2.0 * n * C.max()
    return a_opt, obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n on the probability simplex."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Flat autocorrelation
    inits['flat_ac'] = flat_autocorr_init(n)

    # Cosine (raised cosine)
    x = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
    cos_init = 0.5 * (1 + np.cos(2 * np.pi * x))
    cos_init = np.maximum(cos_init, 0)
    s = cos_init.sum()
    inits['cosine'] = cos_init / s if s > 0 else np.ones(n) / n

    # Sinc-like (windowed)
    t = np.arange(n) - (n - 1) / 2.0
    sigma = n / 4.0
    gauss = np.exp(-t**2 / (2 * sigma**2))
    s = gauss.sum()
    inits['gaussian'] = gauss / s if s > 0 else np.ones(n) / n

    # Bimodal (edges)
    bm = np.zeros(n)
    edge = max(1, n // 5)
    bm[:edge] = 1.0
    bm[-edge:] = 1.0
    s = bm.sum()
    inits['bimodal'] = bm / s if s > 0 else np.ones(n) / n

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.2)

    # Concentrated (center peak)
    center = np.zeros(n)
    center[n // 2] = 1.0
    # Smooth it
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    center = np.convolve(center, kernel, mode='same')
    center = np.maximum(center, 0)
    s = center.sum()
    inits['center_peak'] = center / s if s > 0 else np.ones(n) / n

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent'] = p / s
        elif len(parent) > n:
            # Downsample
            idx = np.linspace(0, len(parent) - 1, n)
            p = np.interp(idx, np.arange(len(parent)), np.maximum(parent, 0))
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent_ds'] = p / s
        else:
            # Upsample
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent_up'] = p / s

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        a_list = [float(max(0.0, x)) for x in a]
        score = evaluate_sequence(a_list)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {label} n={len(a)}, score={score:.6f}")
        elif np.isfinite(score):
            print(f"  {label} n={len(a)}, score={score:.6f}")

    # === Phase 1: Quick scan (25% of time) ===
    print("=== Phase 1: Quick Scan ===")
    phase1_end = time.monotonic() + (deadline - time.monotonic()) * 0.25

    lengths = [128, 256, 512]
    if parent is not None and 64 <= len(parent) <= 1024:
        # Add parent length if reasonable
        parent_n = len(parent)
        if parent_n not in lengths:
            lengths.append(parent_n)
            lengths.sort()

    scan_results = []
    for n in lengths:
        if time.monotonic() > phase1_end - 10:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize good initializations
        priority = ['parent', 'parent_ds', 'parent_up', 'flat_ac', 'cosine',
                    'gaussian', 'uniform', 'bimodal', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]  # Limit per length

        t_per_init = max(5.0, min(20.0, (phase1_end - time.monotonic()) / max(1, len(init_names) * len(lengths))))

        for name in init_names:
            if time.monotonic() > phase1_end - 5:
                break
            t_alloc = min(t_per_init, phase1_end - time.monotonic() - 3)
            if t_alloc < 3:
                break

            a0 = inits[name]
            # Quick FISTA run
            a_opt, obj_opt = optimize_fista(a0, time_limit=t_alloc,
                                            t_init=max(fast_autocorr(a0).max() / 10, 1e-5))
            update_best(a_opt, f"scan n={n}/{name}")
            scan_results.append((2.0 * n * fast_autocorr(a_opt).max(), n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best internal: {scan_results[0][0]:.6f} if scan_results else 'none'")

    # === Phase 2: Deep optimization (50% of time) ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.55

    # Select top configurations
    top_configs = []
    seen_keys = set()
    for obj, n, name, a in scan_results:
        key = (n, name)
        if key not in seen_keys:
            top_configs.append((n, name, a))
            seen_keys.add(key)
        if len(top_configs) >= 5:
            break

    # Always include parent at its own length
    if parent is not None:
        pn = len(parent)
        if pn not in [c[0] for c in top_configs] and 64 <= pn <= 2048:
            p_norm = np.maximum(parent, 0)
            s = p_norm.sum()
            if s > 1e-10:
                top_configs.append((pn, 'parent_deep', p_norm / s))

    # Add flat_ac at best length
    if best_sequence is not None:
        best_n = len(best_sequence)
        flat_n = best_n if best_n in [128, 256, 512, 1024] else 256
        top_configs.append((flat_n, 'flat_ac_fresh', flat_autocorr_init(flat_n)))

    remaining = phase2_end - time.monotonic()
    n_configs = len(top_configs)
    t_per_config = max(10.0, remaining / max(1, n_configs * 2))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_end - 10:
            break
        t_alloc = min(t_per_config, phase2_end - time.monotonic() - 5)
        if t_alloc < 5:
            break

        print(f"  Deep opt {i+1}/{n_configs}: n={n}, {name}, t={t_alloc:.1f}s")

        # FISTA with longer run
        a_opt, obj_opt = optimize_fista(
            a0, time_limit=t_alloc * 0.6,
            t_init=max(fast_autocorr(a0).max() / 8, 1e-5),
            t_min=1e-8)
        update_best(a_opt, f"deep_fista n={n}/{name}")

        # L-BFGS polish on the result
        t_lbfgs = t_alloc * 0.4
        if time.monotonic() < phase2_end - 5 and t_lbfgs > 3:
            a_lbg, obj_lbg = optimize_lbfgs(a_opt, time_limit=t_lbfgs,
                                            t_smooth=fast_autocorr(a_opt).max() / 50)
            update_best(a_lbg, f"deep_lbfgs n={n}/{name}")

    # === Phase 3: Final polish (remaining time) ===
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        C_best = fast_autocorr(best_sequence)
        C_max_best = C_best.max()

        # Multiple FISTA passes with decreasing temperature
        polish_temps = [C_max_best / 20, C_max_best / 50, C_max_best / 100,
                        C_max_best / 200, C_max_best / 500]

        for i, t_pol in enumerate(polish_temps):
            if time.monotonic() > deadline - 10:
                break
            t_polish = min(60.0, (deadline - time.monotonic() - 5) / max(1, len(polish_temps) - i))
            a_pol, obj_pol = optimize_fista(
                best_sequence, time_limit=t_polish,
                t_init=t_pol, t_min=t_pol * 0.1)
            update_best(a_pol, f"polish_fista t={t_pol:.2e}")

        # L-BFGS with very small smoothing
        if time.monotonic() < deadline - 15:
            C_new = fast_autocorr(best_sequence)
            t_lbg = max(C_new.max() / 200, 1e-8)
            a_lbg, obj_lbg = optimize_lbfgs(best_sequence,
                                            time_limit=min(60, (deadline - time.monotonic() - 5)),
                                            t_smooth=t_lbg)
            update_best(a_lbg, "polish_lbfgs_tiny")

        # Random perturbation restarts
        if time.monotonic() < deadline - 20:
            for attempt in range(5):
                if time.monotonic() > deadline - 10:
                    break
                noise_scale = 0.01 / (attempt + 1)
                noise = rng.standard_normal(n_best) * noise_scale
                a_pert = project_prob_simplex(best_sequence + noise)
                t_remaining = deadline - time.monotonic() - 5
                t_this = min(30.0, t_remaining / max(1, 5 - attempt))
                a_rec, obj_rec = optimize_fista(
                    a_pert, time_limit=t_this,
                    t_init=max(fast_autocorr(a_pert).max() / 30, 1e-7),
                    t_min=1e-8)
                update_best(a_rec, f"perturb_{attempt}")

    # === Final validation ===
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence([float(max(0, x)) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.8f}")
    return [float(max(0.0, x)) for x in best_sequence]
