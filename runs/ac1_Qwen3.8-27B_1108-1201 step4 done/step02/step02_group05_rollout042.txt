"""
Searches for non-negative step-function heights minimizing
    2*n*max(conv(a,a))/(sum(a))^2

The search combines:
  * exact block coordinate descent on the scale-invariant objective,
  * linearized trust-region LP steps to escape coordinate-local minima,
  * basin hopping via multiplicative/sparse/mixture mutations,
  * resampling to nearby lengths and edge trimming.
"""

import numpy as np
import scipy.optimize
import time
import math


def normalize_sequence(a):
    x = np.asarray(a, dtype=float)
    if x.ndim != 1:
        x = x.ravel()
    if x.size == 0:
        return np.ones(1, dtype=float)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-14:
        x = np.ones(x.size, dtype=float)
        s = float(x.size)
    x = x / s
    return x


def trim_sequence(x):
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    if x.size <= 1:
        return x
    m = float(x.max())
    tol = 1e-12 * max(1.0, m)
    nz = np.where(x > tol)[0]
    if nz.size == 0:
        return x
    return x[nz[0]:nz[-1] + 1]


def make_state(a):
    x = normalize_sequence(a)
    x = trim_sequence(x)
    x = normalize_sequence(x)
    if x.size == 0:
        x = np.ones(1, dtype=float)
    c = np.convolve(x, x)
    return x, c


def internal_objective(a):
    x = normalize_sequence(a)
    x = trim_sequence(x)
    x = normalize_sequence(x)
    if x.size == 0:
        return float("inf")
    c = np.convolve(x, x)
    return 2.0 * float(x.size) * float(c.max())


def safe_evaluate(x):
    y = normalize_sequence(x)
    y = trim_sequence(y)
    y = normalize_sequence(y)
    if y.size == 0:
        y = np.ones(1, dtype=float)
    try:
        lst = [float(v) for v in y]
        sc = float(evaluate_sequence(lst))
        if np.isfinite(sc):
            return sc
    except Exception:
        pass
    return float("inf")


def coord_obj_value(c, i, v, old, n, two_x, base_max):
    d = float(v) - float(old)
    S = 1.0 + d
    if S <= 1e-12:
        return float("inf")
    if abs(d) < 1e-15:
        return 2.0 * float(n) * float(base_max)
    ct = c.copy()
    ct[i:i + n] += d * two_x
    ct[2 * i] += d * d
    return 2.0 * float(n) * float(ct.max()) / (S * S)


def coordinate_candidates(old, n, v_hi):
    vals = [0.0, float(old), float(v_hi)]

    for fac in (0.01, 0.03, 0.10, 0.30, 0.50, 0.70, 1.30, 2.00, 3.00, 5.00, 8.00, 15.00):
        vals.append(float(old) * float(fac))

    inv = 1.0 / float(n)
    for fac in (0.05, 0.10, 0.30, 1.00, 3.00, 10.00, 30.00, 100.00):
        vals.append(inv * float(fac))

    for base in (1e-8, 1e-7, 1e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2,
                 1e-1, 2e-1, 5e-1, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0):
        vals.append(float(base))

    out = []
    for val in vals:
        if not np.isfinite(val):
            continue
        v = float(val)
        if v < 0.0:
            v = 0.0
        if v > float(v_hi):
            v = float(v_hi)
        dup = False
        for u in out:
            if abs(v - u) <= 1e-12 * max(1.0, abs(v), abs(u)):
                dup = True
                break
        if not dup:
            out.append(v)
    out.sort()
    return out


def pair_first_key(p):
    return p[0]


def golden_coord_value(c, i, old, lo, hi, n, two_x, base_max):
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = coord_obj_value(c, i, a1, old, n, two_x, base_max)
    f2 = coord_obj_value(c, i, a2, old, n, two_x, base_max)

    best_f = f1 if f1 < f2 else f2
    best_v = a1 if f1 < f2 else a2

    for _ in range(25):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = coord_obj_value(c, i, a1, old, n, two_x, base_max)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = coord_obj_value(c, i, a2, old, n, two_x, base_max)

        if f1 < best_f:
            best_f = f1
            best_v = a1
        if f2 < best_f:
            best_f = f2
            best_v = a2

    v = 0.5 * (lo + hi)
    f = coord_obj_value(c, i, v, old, n, two_x, base_max)
    if f < best_f:
        best_f = f
        best_v = v
    return best_f, best_v


def optimize_coordinate(x, c, two_x, i, n):
    old = float(x[i])
    base_max = float(c.max())
    base_obj = 2.0 * float(n) * base_max

    v_hi = 100.0 * (1.0 - old) + 10.0
    if v_hi < 30.0:
        v_hi = 30.0
    if v_hi > 1000.0:
        v_hi = 1000.0

    unique = coordinate_candidates(old, n, v_hi)
    pairs = []
    for v in unique:
        f = coord_obj_value(c, i, v, old, n, two_x, base_max)
        pairs.append((f, v))

    pairs.sort(key=pair_first_key)
    best_f = pairs[0][0]
    best_v = pairs[0][1]

    top_count = min(3, len(pairs))
    for r in range(top_count):
        f, v = pairs[r]
        if f > best_f + 1e-12:
            break

        idx = 0
        best_diff = abs(unique[0] - v)
        for k in range(1, len(unique)):
            diff = abs(unique[k] - v)
            if diff < best_diff:
                best_diff = diff
                idx = k

        if idx == 0:
            lo = 0.0
            hi = unique[1] if len(unique) > 1 else min(2.0, v_hi)
        elif idx == len(unique) - 1:
            lo = unique[-2]
            hi = min(1000.0, unique[-1] * 3.0 + 1.0)
            if hi <= lo:
                hi = lo + max(1e-6, 0.1 * lo)
        else:
            lo = unique[idx - 1]
            hi = unique[idx + 1]
            if hi <= lo:
                hi = lo + 1e-6

        f_g, v_g = golden_coord_value(c, i, old, lo, hi, n, two_x, base_max)
        if f_g < best_f:
            best_f = f_g
            best_v = v_g

    if best_f < base_obj - 1e-11:
        return best_v, best_f
    return None, base_obj


def update_coordinate(x, c, two_x, i, v, n):
    old = float(x[i])
    d = float(v) - old
    if abs(d) <= 1e-15:
        return False
    S = 1.0 + d
    if S <= 1e-12:
        return False

    c[i:i + n] += d * two_x
    c[2 * i] += d * d
    x[i] = float(v)

    invS = 1.0 / S
    x *= invS
    c *= invS * invS

    two_x *= invS
    two_x[i] = 2.0 * x[i]
    return True


def coordinate_refine(a, time_limit, seed):
    x, c = make_state(a)
    n = x.size
    if n == 0 or time_limit <= 0:
        return x, internal_objective(x)

    obj = 2.0 * float(n) * float(c.max())
    x_best = x.copy()
    obj_best = obj
    two_x = 2.0 * x

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    sweep = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        if obj > obj_best + 1e-7:
            x = x_best.copy()
            c = np.convolve(x, x)
            obj = obj_best
            two_x = 2.0 * x

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            v, f_new = optimize_coordinate(x, c, two_x, i, n)
            if v is not None:
                if update_coordinate(x, c, two_x, i, v, n):
                    obj = 2.0 * float(n) * float(c.max())
                    if obj < obj_best - 1e-11:
                        obj_best = obj
                        x_best = x.copy()
                        improved = True

        sweep += 1
        if sweep % 3 == 0:
            c = np.convolve(x, x)
            two_x = 2.0 * x
            obj = 2.0 * float(n) * float(c.max())
            if obj < obj_best - 1e-12:
                obj_best = obj
                x_best = x.copy()
                improved = True

        if not improved:
            stagnant += 1
            if stagnant >= 4:
                break
        else:
            stagnant = 0

    x = x_best.copy()
    x = normalize_sequence(x)
    c = np.convolve(x, x)
    obj = 2.0 * float(x.size) * float(c.max()) if x.size else float("inf")
    return x, obj


def evaluate_alpha(x, delta, alpha, n):
    y = x + alpha * delta
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if s <= 1e-12:
        return float("inf"), None, None
    y = y / s
    c = np.convolve(y, y)
    o = 2.0 * float(n) * float(c.max())
    return o, y, c


def golden_alpha(x, delta, lo, hi, n):
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1, y1, c1 = evaluate_alpha(x, delta, a1, n)
    f2, y2, c2 = evaluate_alpha(x, delta, a2, n)

    if f1 < f2:
        best_f, best_y, best_c = f1, y1, c1
    else:
        best_f, best_y, best_c = f2, y2, c2

    for _ in range(25):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            y2 = y1
            c2 = c1
            a1 = hi - gr * (hi - lo)
            f1, y1, c1 = evaluate_alpha(x, delta, a1, n)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            y1 = y2
            c1 = c2
            a2 = lo + gr * (hi - lo)
            f2, y2, c2 = evaluate_alpha(x, delta, a2, n)

        if f1 < best_f:
            best_f, best_y, best_c = f1, y1, c1
        if f2 < best_f:

    a = 0.5 * (lo + hi)
    f, y, c = evaluate_alpha(x, delta, a, n)

def try_lp_step(x, c, n):
    if n < 2:
        return None

    L = 2 * n - 1
    U = max(0.02, 10.0 / float(n))
    beta = 0.05

    A_ub = np.zeros((L + 2 * n, n + 2), dtype=float)
    b_ub = np.empty(L + 2 * n, dtype=float)

    for s in range(L):
        lo = s - (n - 1)
        if lo < 0:
            lo = 0
        hi = s
        if hi >= n:
            hi = n - 1

        if lo <= hi:
            j_start = s - hi
            j_end = s - lo
            A_ub[s, lo:hi + 1] = 2.0 * x[j_start:j_end + 1][::-1]

        A_ub[s, n] = -1.0
        b_ub[s] = -float(c[s])

    row = L
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n + 1] = -1.0
        b_ub[row] = 0.0
        row += 1

        A_ub[row, i] = -1.0
        A_ub[row, n + 1] = -1.0
        b_ub[row] = 0.0
        row += 1

    A_eq = np.zeros((1, n + 2), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = []
    for i in range(n):
        bounds.append((-float(x[i]), float(U)))
    bounds.append((-10.0, 10.0))
    bounds.append((0.0, float(U)))

    c_obj = np.zeros(n + 2, dtype=float)
    c_obj[n] = 1.0
    c_obj[n + 1] = beta

    try:
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"disp": False}
        )
    except Exception:
        return None

    if not res.success:
        return None

    delta = np.asarray(res.x[:n], dtype=float)
    if not np.all(np.isfinite(delta)):
        return None

    base_obj = 2.0 * float(n) * float(c.max())
    alphas = [0.0, 0.01, 0.03, 0.10, 0.20, 0.35, 0.50, 0.70, 1.00]
    grid_objs = []
    best_obj = base_obj
    best_y = None
    best_c = None
    best_idx = 0
    min_val = float("inf")

    for k, a in enumerate(alphas):
        f, y, cc = evaluate_alpha(x, delta, float(a), n)
        grid_objs.append(f)
        if f < min_val:
            min_val = f
            best_idx = k
        if f < best_obj - 1e-12:
            best_obj = f
            best_y = y
            best_c = cc

    if best_y is None:
        return None

    if best_idx == 0:
        lo = 0.0
        hi = alphas[1] if len(alphas) > 1 else 1.0
    elif best_idx == len(alphas) - 1:
        lo = alphas[-2]
        hi = 1.0
    else:
        lo = alphas[best_idx - 1]
        hi = alphas[best_idx + 1]

    f_g, y_g, c_g = golden_alpha(x, delta, lo, hi, n)
    if f_g < best_obj:
        best_obj = f_g
        best_y = y_g
        best_c = c_g

    if best_obj < base_obj - 1e-10:
        x_best = normalize_sequence(best_y)
        x_best = trim_sequence(x_best)
        x_best = normalize_sequence(x_best)
        c_best = np.convolve(x_best, x_best)
        o = 2.0 * float(x_best.size) * float(c_best.max())
        return x_best, c_best, o

    return None


def pertur0b_sequence(x, rng, strength):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        x = np.ones(1, dtype=float)
    noise = rng.standard_normal(x.size) * float(strength)
    y = x * np.exp(noise)
    y = trim_sequence(y)
    y = normalize_sequence(y)
    return y


def mix_sequence(x, rng, mix, alpha):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        x = np.ones(1, dtype=float)
    z = rng.dirichlet(np.full(x.size, alpha))
    y = (1.0 - float(mix)) * x + float(mix) * z
    y = trim_sequence(y)
    y = normalize_sequence(y)
    return y


def blank_sequence(x, rng, fraction):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return np.ones(1, dtype=float)
    k = max(1, int(float(fraction) * x.size))
    if k >= x.size:
        k = max(1, x.size - 1)
    y = x.copy()
    idx = rng.choice(x.size, size=k, replace=False)
    y[idx] = 0.0
    y = trim_sequence(y)
    y = normalize_sequence(y)
    return y


def random_perturb_sequence(x, rng, strength):
    typ = int(rng.integers(0, 4))
    if typ == 0:
        return pertur0b_sequence(x, rng, max(0.01, strength))
    if typ == 1:
        alpha = max(0.05, 50.0 / max(1, x.size))
        return mix_sequence(x, rng, min(0.5, 0.1 + strength), alpha)
    if typ == 2:
        frac = 0.02 + 0.10 * float(rng.random())
        return blank_sequence(x, rng, frac)
    y = pertur0b_sequence(x, rng, max(0.01, 0.5 * strength))
    alpha = max(0.1, 100.0 / max(1, y.size))
    return mix_sequence(y, rng, 0.10, alpha)


def resample_sequence(a, m):
    x = normalize_sequence(a)
    n = x.size
    if n == 0:
        x = np.ones(1, dtype=float)
        n = 1
    if n == m:
        return x
    if n == 1:
        y = np.full(m, float(x[0]), dtype=float)
    else:
        old_x = np.linspace(0.0, 1.0, n)
        new_x = np.linspace(0.0, 1.0, m)
        y = np.interp(new_x, old_x, x)
    y = trim_sequence(y)
    y = normalize_sequence(y)
    return y

    if len(out) > 26:
        out = out[:26]
    return out


def track_score_key(t):
    return t[1]


def register_sequence(tracks, x, score, obj):
    if not np.isfinite(score):
        return
    x = normalize_sequence(trim_sequence(x))
    n = x.size
    if n == 0:
        return

    best_idx = -1
    best_score = float("inf")
    for i in range(len(tracks)):
        tr = tracks[i]
        if tr[0].size == n:
            if tr[1] < best_score:
                best_score = tr[1]
                best_idx = i

    if best_idx != -1 and score < best_score - 1e-12:
        tracks[best_idx][0] = x.copy()
        tracks[best_idx][1] = float(score)
        tracks[best_idx][2] = float(obj)
        tracks[best_idx][3] = 0
    elif len(tracks) < 16:
        tracks.append([x.copy(), float(score), float(obj), 0, len(tracks)])


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start_time = time.monotonic()
    total_time = max(60.0, float(budget_s) - 30.0)
    deadline = start_time + total_time
    rng = np.random.default_rng(seed)

    pool = build_pool(seed)
    tracks = []
    for a in pool:
        x, c = make_state(a)
        obj = 2.0 * float(x.size) * float(c.max()) if x.size else float("inf")
        score = safe_evaluate(x)
        if np.isfinite(score):
            tracks.append([x, score, obj, 0, len(tracks)])

    if not tracks:
        x = np.ones(128, dtype=float) / 128.0
        return [float(v) for v in x]

    tracks.sort(key=track_score_key)
    global_x = tracks[0][0].copy()
    global_score = tracks[0][1]
    print(f"Initial best: {global_score:.6f} length={global_x.size}", flush=True)

    quick_limit = min(6, len(tracks))
    quick_deadline = start_time + 0.10 * total_time
    qi = 0
    while qi < quick_limit and time.monotonic() < quick_deadline:
        now = time.monotonic()
        t_alloc = min(30.0, quick_deadline - now, 45.0)
        if t_alloc < 5.0:
            break
        track = tracks[qi]
        opt_x, opt_obj = coordinate_refine(track[0], t_alloc, seed + 1000 + qi * 7919)
        opt_x = normalize_sequence(trim_sequence(opt_x))
        score = safe_evaluate(opt_x)
        if np.isfinite(score):
            if score < track[1] - 1e-12:
                track[0] = opt_x
                track[1] = score
                track[2] = opt_obj
                track[3] = 0
            else:
                track[3] += 1
            if score < global_score:
                global_score = score
                global_x = opt_x.copy()
        print(f"Quick {qi + 1}: best={global_score:.6f}", flush=True)
        qi += 1

    tracks.sort(key=track_score_key)
    iter_count = 0

    while time.monotonic() < deadline - 20.0:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 20.0:
            break
        iter_count += 1

        best_idx = 0
        for i in range(1, len(tracks)):
            if tracks[i][1] < tracks[best_idx][1]:
                best_idx = i

        if rng.random() < 0.75:
            idx = best_idx
        else:
            others = []
            for i in range(len(tracks)):
                if i != best_idx:
                    others.append(i)
            if not others:
                idx = best_idx
            else:
                idx = int(others[int(rng.integers(len(others)))])

        track = tracks[idx]

        if track[3] >= 3:
            strength = 0.05 + 0.25 * float(rng.random())
            if global_score < 1.51:
                strength *= 0.6
            trial_x = random_perturb_sequence(track[0], rng, strength)
        else:
            trial_x = track[0].copy()

        t_alloc = min(75.0, remaining * 0.25, remaining - 15.0)
        if t_alloc < 5.0:
            break

        opt_x, opt_obj = coordinate_refine(trial_x, t_alloc, seed + iter_count * 104729)
        opt_x = normalize_sequence(trim_sequence(opt_x))
        score = safe_evaluate(opt_x)

        if np.isfinite(score):
            if score < track[1] - 1e-12:
                track[0] = opt_x
                track[1] = score
                track[2] = opt_obj
                track[3] = 0
            else:
                track[3] += 1

            if score < global_score:
                global_score = score
                global_x = opt_x.copy()

        if iter_count % 4 == 0 and global_x.size <= 1024:
            try:
                clp = np.convolve(global_x, global_x)
                lpres = try_lp_step(global_x, clp, global_x.size)
                if lpres is not None:
                    lx, lc, lo = lpres
                    ls = safe_evaluate(lx)
                    if np.isfinite(ls) and ls < global_score:
                        global_score = ls
                        global_x = lx.copy()
                        register_sequence(tracks, lx, ls, lo)
            except Exception:
                pass

        if iter_count % 5 == 0:
            print(f"Iter {iter_count}: best={global_score:.6f} len={global_x.size}", flush=True)

    now = time.monotonic()
    final_deadline = deadline - 5.0
    if now < final_deadline - 10.0:
        t_alloc = final_deadline - now
        if t_alloc > 10.0:
            opt_x, opt_obj = coordinate_refine(global_x, t_alloc, seed + 987654321)
            opt_x = normalize_sequence(trim_sequence(opt_x))
            score = safe_evaluate(opt_x)
            if np.isfinite(score) and score < global_score:
                global_score = score
                global_x = opt_x.copy()
            register_sequence(tracks, global_x, global_score, opt_obj)

    if time.monotonic() < deadline - 5.0 and global_x.size <= 1024:
        try:
            clp = np.convolve(global_x, global_x)
            lpres = try_lp_step(global_x, clp, global_x.size)
            if lpres is not None:
                lx, lc, lo = lpres
                score = safe_evaluate(lx)
                if np.isfinite(score) and score < global_score:
                    global_score = score
                    global_x = lx.copy()
        except Exception:
            pass

    now = time.monotonic()
    if now < deadline - 5.0:
        t_alloc = deadline - 5.0 - now
        if t_alloc > 5.0:
            opt_x, opt_obj = coordinate_refine(global_x, t_alloc, seed + 555555)
            opt_x = normalize_sequence(trim_sequence(opt_x))
            score = safe_evaluate(opt_x)
            if np.isfinite(score) and score < global_score:
                global_score = score
                global_x = opt_x.copy()

    global_x = normalize_sequence(trim_sequence(global_x))
    final_score = safe_evaluate(global_x)
    if not np.isfinite(final_score):
        global_x = np.ones(128, dtype=float) / 128.0
        final_score = safe_evaluate(global_x)

    print(f"FINAL SCORE: {final_score:.6f} length={global_x.size}", flush=True)
    return [float(v) for v in global_x]
