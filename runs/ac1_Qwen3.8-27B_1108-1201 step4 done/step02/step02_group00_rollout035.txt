import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_fft(a, w):
    """Compute sg[i] = 2 * sum_k a[k] * w[i+k] using FFT."""
    n = len(a)
    total_len = n + len(w) - 1
    N = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(np.conj(A) * W, N)
    return 2.0 * R[:n]

Actually, one more important consideration: the LP direction finding. Let me think about this more carefully.

The LP is:
```
maximize sum(b)
subject to conv(a_eval, b) <= M (element-wise)
           b >= 0
```

where `a_eval` is the current sequence and `M = max(conv(a_eval, a_eval))`.

The idea is: if we move from `a_eval` toward `b` (i.e., `a_new = (1-t)*a_eval + t*b`), then:
`conv(a_new, a_new) = (1-t)^2 conv(a_eval, a_eval) + 2t(1-t) conv(a_eval, b) + t^2 conv(b, b)`

We want this to have max ≤ M. A sufficient condition is:
- `conv(a_eval, b) <= M` (ensured by LP)
- `conv(b, b) <= M / t^2` (not ensured by LP, but if `sum(b)` is large and `b` is spread out, `conv(b,b)` is small)

Actually, the condition `conv(a_eval, b) <= M` ensures that the cross-term doesn't push us over M. The `conv(b,b)` term is `t^2 * conv(b,b)`, which is small for small t. So for sufficiently small t, the peak won't increase.

But we want the peak to DECREASE. For that, we need `conv(b, b)` to be smaller than `conv(a_eval, a_eval)` at the peak locations. The LP maximizes `sum(b)`, which tends to spread `b` out, reducing its peak auto-convolution.

This is a good heuristic, but it's not guaranteed to always decrease the peak. That's why we need multiple iterations and occasional restarts.

Let me also think about a variant: instead of `conv(a, b) <= M`, use `conv(a, b) <= M * alpha` for some `alpha < 1`. This is a stricter constraint that more aggressively pushes `b` away from the peak locations of `a`, potentially giving a better direction.

Another variant: use weighted constraints `conv(a, b)[j] <= M * (1 + epsilon * (1 - w_j))` where `w_j` is the weight from the log-sum-exp. This focuses the LP on the most important constraints.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Log-sum-exp smoothed projected gradient descent with FFT-based subgradients
2. LP-based direction finding (complementary to gradient)
3. Convex optimization via CVXPY for small n
4. Multi-phase search with adaptive parameters and restarts
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_fft(a, w):
    """Compute subgradient of LSE-smoothed max convolution."""
    n = len(a)
    N = 1 << (n + len(w) - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(np.conj(A) * W, N)
    return 2.0 * R[:n]

def get_conv_max(a):
    return fast_convolve(a).max()

def score_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * get_conv_max(a) / (s * s)

def lp_direction(a, M, n, alpha=1.0):
    """Solve LP for direction: max sum(b) s.t. conv(a,b) <= M*alpha, b >= 0."""
    # Build sparse constraint matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        for i in range(i_start, i_end + 1):
            rows.append(j)
            cols.append(i)
            vals.append(a[j - i])
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * alpha)
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-10:
                return b / bs
    except Exception:
        pass
    return None

def solve_cvxpy_small(n, time_limit=60):
    """Solve the convex problem for small n using CVXPY."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == 1]
    # Build convolution constraints
    for j in range(2 * n - 1):
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        term = cp.sum([a[i] * a[j - i] for i in range(i_start, i_end + 1)])
        constraints.append(term <= t)
    problem = cp.Problem(cp.Minimize(t), constraints)
    t0 = time.monotonic()
    try:
        problem.solve(solver=cp.SCS, max_iters=100000, eps=1e-7, verbose=False)
        if a.value is not None and np.isfinite(problem.objective.value):
            a_val = np.maximum(a.value, 0)
            a_val = a_val / a_val.sum()
            obj = 2.0 * n * get_conv_max(a_val)
            return a_val, obj
    except Exception as e:
        print(f"CVXPY error: {e}")
    return None, None

def optimize_hybrid(a0, time_limit, mu_init=50.0, mu_max=500.0,
                    eta_init=0.05, eta_decay=0.9995,
                    lp_interval=150, perturb_interval=400,
                    perturb_scale=0.005):
    """
    Hybrid optimizer: LSE smoothed gradient + periodic LP direction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    mu = mu_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # LSE weights
        scaled = mu * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = subgrad_fft(a, w)
        sg = sg - sg.mean()  # center to preserve sum
        
        # Gradient step + projection
        a = project_simplex(a - eta * sg)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        it += 1
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        eta = max(eta * eta_decay, 1e-6)
        mu = min(mu * 1.003, mu_max)
        
        # LP direction
        if lp_interval > 0 and it % lp_interval == 0:
            M_cur = conv.max()
            b = lp_direction(a, M_cur, n, alpha=1.0)
            if b is not None:
                for t_mix in [0.05, 0.1, 0.2]:
                    a_mix = (1.0 - t_mix) * a + t_mix * b
                    a_mix = project_simplex(a_mix)
                    conv_mix = fast_convolve(a_mix)
                    obj_mix = 2.0 * n * conv_mix.max()
                    if obj_mix < best_obj:
                        a = a_mix
                        best_obj = obj_mix
                        best_a = a_mix.copy()
                        no_improve = 0
                        break
        
        # Perturbation restart
        if no_improve >= perturb_interval:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            eta = eta_init * 0.3
            mu = mu_init
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),

                           np.maximum(parent, 0))
            ps = pp.sum()

def subgrad_direct(a, w):
    n = len(a)
    # Build matrix W where W[i,k] = w[i+k]
    # Then sg = 2 * W @ a
    indices = np.arange(n)
    W = w[None, :] + indices[:, None]  # This doesn't work directly
    # Actually: W[i, k] = w[i + k] for i = 0..n-1, k = 0..n-1
    # w has length 2n-1, so i+k ranges from 0 to 2n-2. OK.
    idx = (np.arange(n)[:, None] + np.arange(n)[None, :])  # shape (n, n)
    W = w[idx]  # W[i, k] = w[i+k]
    sg = 2.0 * (W @ a)
    return sg
