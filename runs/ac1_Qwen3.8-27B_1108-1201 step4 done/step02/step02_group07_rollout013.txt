from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M):
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': 5.0, 'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, verbose=True):
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        b = solve_lp(f, M)
        if b is None:
            break
        
        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)
        
        # Adaptive step size: try multiple values
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.15, 0.1, 0.05, 0.02, 0.01]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1
        
        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{elapsed:6.1f}s] it={it:5d} score={best_s:.8f} best={best_score:.8f} stall={stall}")

            # Escape: perturb
            rng = np.random.default_rng(42 + it)
            f += 0.02 * rng.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
    
    # Final check
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    
    return best_f, best_score

LP-based iterative optimization for minimizing peak autocorrelation of non-negative sequences.
Uses fast HiGHS LP subproblems with adaptive step sizes, subgradient polish, and multi-phase
time allocation to achieve the best score within the time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, A=None):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)

            best_seq = provided.copy()

    n = len(provided) if provided is not None else 256

    print(f"\n=== Phase 1: LP optimization (n={n}) ===")
    t1 = min(900.0, deadline - time.monotonic() - 200.0)
    if t1 > 20 and best_seq is not None:
        f1, s1 = lp_optimize(best_seq, t1, verbose=True)
        print(f"  Phase 1 result: {s1:.8f} (best={best_score:.8f})")
        if s1 < best_score:
            best_score = s1
            best_seq = f1.copy()
            print(f"  *** NEW BEST: {best_score:.8f} ***")

    print(f"\n=== Phase 2: Gradient polish ===")
    t2 = min(120.0, deadline - time.monotonic() - 100.0)
    if t2 > 15 and best_seq is not None:
        f2, s2 = gradient_polish(best_seq, t2, verbose=True)
        print(f"  Phase 2 result: {s2:.8f} (best={best_score:.8f})")
        if s2 < best_score:
            best_score = s2
            best_seq = f2.copy()
            print(f"  *** NEW BEST: {best_score:.8f} ***")

    print(f"\n=== Phase 3: Local refinement ===")
    t3 = min(80.0, deadline - time.monotonic() - 60.0)
    if t3 > 10 and best_seq is not None:
        f3, s3 = local_refine(best_seq, t3, verbose=True)
        print(f"  Phase 3 result: {s3:.8f} (best={best_score:.8f})")
        if s3 < best_score:
            best_score = s3
            best_seq = f3.copy()
            print(f"  *** NEW BEST: {best_score:.8f} ***")

    print(f"\n=== Phase 4: Final LP refinement ===")
    t4 = min(60.0, deadline - time.monotonic() - 5.0)
    if t4 > 5 and best_seq is not None:
        f4, s4 = lp_optimize(best_seq, t4, verbose=False)
        print(f"  Phase 4 result: {s4:.8f} (best={best_score:.8f})")
        if s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  *** NEW BEST: {best_score:.8f} ***")

    # Final validation
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.8f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    return [1.0] * 256
