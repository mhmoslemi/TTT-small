import numpy as np
from scipy.optimize import linprog

import sys

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def smooth_peak(a, T):
    """Compute soft approximation to max of autocorrelation."""
    c = fft_convolve(a, a)
    # For numerical stability, subtract max
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / T)
    return c_max + T * np.log(np.sum(exp_c))

def smooth_peak_gradient(a, T):
    """Compute gradient of soft peak approximation."""
    c = fft_convolve(a, a)
    c_max = np.max(c)

    p = exp_c / np.sum(exp_c)

    grad = np.zeros(n)
    for m in range(n):
        grad[m] = 2.0 * np.dot(a, p[m:m+n])
    return grad
