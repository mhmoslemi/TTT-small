def project_simplex(v):
    n = len(v)
    if n == 1:

    cssv = np.cumsum(u)

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_subgrad(a0, time_limit, tol=0.005, lr_init=0.01, 
                      perturb_every=5000, perturb_scale=0.003,
                      max_peak_tol=0.05):

    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    x_prev = a.copy()
    t_val = 1.0
    lr = lr_init
    no_improve = 0
    t0 = time.monotonic()
    batch = 30
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead point
        gamma = (t_val - 1.0) / (t_val + 1.0)

def polish_sequence(a0, time_limit, lr_init=0.001, tol=0.001):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() - t0 < time_limit:
        max_val = conv.max()
        # Use all peaks within 2% of max
        active = np.where(conv >= max_val * (1.0 - 0.02))[0]
        
        # Uniform weights over active peaks
        sg = compute_subgradient(a, active, np.ones(len(active)) / len(active))

        momentum = mu * momentum - lr * sg
        a_new = project_simplex(a + momentum)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Decay lr slowly
        lr *= 0.99995
    
    return best_a, best_obj

Hmm, wait. Let me reconsider. The Nesterov method I wrote has an issue: the standard Nesterov FISTA is designed for differentiable objectives. For a max of smooth functions (which is what we have), the subgradient is only a subgradient, not a true gradient. The Nesterov acceleration may not work as well.

Let me think about this differently. The objective `f(a) = max_k b_k(a)` where `b_k(a) = sum_j a[j]*a[k-j]` is convex (as a max of convex functions - each `b_k` is a quadratic form which is convex on the simplex... wait, is `b_k` convex?

`b_k(a) = a^T Q_k a` where `Q_k` has specific structure. For the diagonal entries: `Q_k[i, k-i] = 1` (if both indices are in range). This matrix is generally NOT positive semi-definite, so `b_k` is not necessarily convex.

For example, for k=0: `b_0(a) = a[0]^2` which is convex.
For k=1: `b_1(a) = 2*a[0]*a[1]` which is NOT convex (the Hessian is [[0,1],[1,0]] which has eigenvalues ±1).

So the objective is NOT convex! This means we can't use standard convex optimization tools, and the landscape may have many local minima.

Given this non-convexity, the approach of subgradient + momentum + perturbation is reasonable. The key is to be aggressive about escaping local minima.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using momentum subgradient descent
with Nesterov lookahead, adaptive active set, and aggressive exploitation
of the parent construction.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sg(a, active, weights=None):
    """Compute subgradient at a for weighted sum of active peaks."""
    n = len(a)
    if weights is None:
        weights = np.ones(len(active)) / len(active)
    sg = np.zeros(n)
    for idx in range(len(active)):
        k = int(active[idx])
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        i_s = max(0, k - n + 1)
        i_e = min(n - 1, k)
        if i_s <= i_e:
            idxs = np.arange(i_s, i_e + 1)
            sg[idxs] += 2.0 * w * a[k - idxs]
    return sg


def optimize(a0, time_limit, lr=0.02, mu=0.90, peak_tol=0.02,
             nesterov=True, perturb_scale=0.003, perturb_patience=3000):
    """
    Momentum subgradient with optional Nesterov lookahead.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    a_prev = a.copy()  # For Nesterov
    t0 = time.monotonic()
    no_improve = 0
    lr_cur = lr
    patience = perturb_patience
    
    while time.monotonic() - t0 < time_limit:
        batch = 20
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            if nesterov:
                # Nesterov lookahead
                gamma = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * (no_improve + 1.0) / (no_improve + 1.0)))
                # Simplified: use a fixed extrapolation coefficient
                y = a + mu * (a - a_prev)
                y = project_simplex(y)
            else:
                y = a.copy()
            
            conv_y = fast_convolve(y)
            max_y = conv_y.max()
            
            # Adaptive active set
            active = np.where(conv_y >= max_y * (1.0 - peak_tol))[0]
            
            # Softmax weights
            diff = (max_y - conv_y[active]) / (max_y * 0.005 + 1e-15)
            w = np.exp(-diff)
            w_sum = w.sum()
            if w_sum > 1e-15:
                w = w / w_sum
            else:
                w = np.ones(len(active)) / len(active)
            
            sg = compute_sg(y, active, w)
            
            # Update
            v = mu * v - lr_cur * sg
            a_prev = a.copy()
            a = project_simplex(a + v)
            conv = fast_convolve(a)
            cur_obj = conv.max()
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
