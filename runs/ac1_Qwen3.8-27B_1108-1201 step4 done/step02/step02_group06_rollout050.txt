"""
Multi-phase optimization for minimum normalized peak autocorrelation.
Uses iterative QP for peak reduction, smooth-max gradient descent,
LP-based direction finding, and simulated annealing polish.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_lag_matrix(k, n):
    """Build symmetric matrix Q_k such that a^T Q_k a = conv(a,a)[n-1+k]."""
    Q = np.zeros((n, n))
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            if k == 0:
                Q[i, j] = 1.0
            else:
                Q[i, j] = 0.5
                Q[j, i] = 0.5
    return Q

def solve_qp_peak(a, k, S, method='SLSQP'):
    """Solve QP: minimize a^T Q_k a subject to sum(a) = S, a >= 0."""
    n = len(a)
    Q = build_lag_matrix(k, n)
    
    x0 = a.copy()
    bounds = [(0, None)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S, 
                    'jac': lambda x: np.ones(n)}]
    
    def objective(x):
        return x @ Q @ x
    
    def gradient(x):
        return 2 * Q @ x
    
    result = minimize(objective, x0, jac=gradient, method=method,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 1000, 'ftol': 1e-12})
    if result.success:
        return np.clip(result.x, 0, None)
    return None

def qp_iterative_optimize(f_init, time_limit, top_k=5, verbose=False):
    """Iteratively minimize the top lags using QP."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        peak_val = np.max(conv)
        peak_idx = int(np.argmax(conv))
        k = peak_idx - (n - 1)  # lag
        
        # Find top_k lags
        top_indices = np.argsort(conv)[-top_k:][::-1]
        
        # Build combined Q matrix for top lags
        Q_combined = np.zeros((n, n))
        weights = np.zeros(len(top_indices))
        for idx, ti in enumerate(top_indices):
            lag = ti - (n - 1)
            Q_lag = build_lag_matrix(lag, n)
            # Weight by relative value
            weights[idx] = conv[ti] / peak_val
            Q_combined += weights[idx] * Q_lag
        
        # Solve QP
        x0 = f.copy()
        bounds = [(0, None)] * n
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S,
                        'jac': lambda x: np.ones(n)}]
        
        def objective(x):
            return x @ Q_combined @ x
        
        def gradient(x):
            return 2 * Q_combined @ x
        
        result = minimize(objective, x0, jac=gradient, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-10})
        
        it += 1
        if result.success:
            f_new = np.clip(result.x, 0, None)
            s_new = np.sum(f_new)
            if s_new < 1e-10:
                continue
            f_new = f_new * (S / s_new)
            score_new = local_score(f_new)
            
            if score_new < best_score:
                best_score = score_new
                best_f = f_new.copy()
                f = f_new
            elif it % 10 == 0:
                # Accept anyway to avoid cyclic behavior
                f = (0.7 * f + 0.3 * f_new)
                f = np.clip(f, 0, None)
                f = f * (S / np.sum(f))
        
        if verbose and it % 20 == 0:
            print(f"  QP iter={it}, peak_lag={k}, score={local_score(f):.6f}, best={best_score:.6f}")
    
    return best_f, best_score

    size = 1
    while size < n + m - 1:

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

    n = len(a)
    if n == 0:

    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    if top_k is not None and top_k < len(conv):

        idx = np.argsort(conv)[-top_k:]
        w = np.zeros_like(conv)
        top_vals = conv[idx]
        shifted = (top_vals - top_vals.max()) / tau
        exp_vals = np.exp(shifted)
        w[idx] = exp_vals / np.sum(exp_vals)
    else:
        shifted = (conv - conv.max()) / tau
        exp_conv = np.exp(shifted)
        w = exp_conv / np.sum(exp_conv)

    # = 2 * conv(w, a[::-1])[n-1 : 2n-1]
    corr = fft_convolve(w, a[::-1])
    g = 2.0 * corr[n-1 : 2*n-1]
    return g, conv

            phase_frac = frac / phase1_frac
            tau = tau_start * (tau_mid / tau_start) ** phase_frac
        elif frac < phase1_frac + phase2_frac:
            # Phase 2: tau from tau_mid to tau_end
            phase_frac = (frac - phase1_frac) / phase2_frac
            tau = tau_mid * (tau_end / tau_mid) ** phase_frac
        else:
            tau = tau_end
        
        # Compute a from theta
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        p = e / np.sum(e)
        a = p * S
        
        # Compute gradient
        g_a, conv = smooth_max_grad(a, tau, top_k=min(50, 2*n-1))
        
        # Gradient w.r.t. theta
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)
        
        # Adam update
        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g_theta
        m2 = beta2 * m2 + (1 - beta2) * g_theta ** 2
        m1_hat = m1 / (1 - beta1 ** step)
        m2_hat = m2 / (1 - beta2 ** step)
        
        # Adaptive learning rate
        lr = 1.0 / np.sqrt(1 + 0.01 * step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Track best
        score = local_score(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        if verbose and step % 500 == 0:
            print(f"    Grad step={step}, tau={tau:.5f}, score={score:.6f}, best={best_score:.6f}")
    
    # Convert back
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def build_constraint_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp(f, M, time_limit=20):
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success:
        return result.x
    return None

def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = solve_lp(f, M, time_limit=min(15, time_limit - (time.monotonic() - t0)))
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)
        
        score_new = local_score(f_new)
        iteration += 1
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            if t < 0.8:
                t = min(0.8, t * 1.05)
        else:
            f = f_new
            stagnant += 1
            if stagnant > 20:
                t *= 0.7
                stagnant = 0
            if stagnant > 80:
                f = f + 0.02 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.2
                stagnant = 0
        
        if verbose and iteration % 20 == 0:
            print(f"    LP iter={iteration}, t={t:.3f}, score={local_score(f):.6f}, best={best_score:.6f}")
    
    return best_f, best_score

def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0a = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    iter_count = 0
    
    while time.monotonic() - t0a < time_limit:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.3:
            # Target-based: reduce peak contributors
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.75:
            # Two block perturbations
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Swap-based
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            tmp = candidate[i1]
            candidate[i1] = candidate[i2]
            candidate[i2] = tmp
        
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)
        
        new_score = local_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score

def qp_peak_reduce(f_init, time_limit, verbose=False):
    """Iteratively solve QPs to reduce peak autocorrelation lags."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Get top lags
        top_k = min(5, 2*n-1)
        top_indices = np.argsort(conv)[-top_k:][::-1]
        
        # Build combined Q
        Q = np.zeros((n, n))
        for ti in top_indices:
            lag = ti - (n - 1)
            w = conv[ti] / peak_val
            for i in range(n):
                j = i + lag
                if 0 <= j < n:
                    if lag == 0:
                        Q[i, j] += w
                    else:
                        Q[i, j] += 0.5 * w
                        Q[j, i] += 0.5 * w
        
        # Solve QP: minimize x^T Q x s.t. sum(x) = S, x >= 0
        x0 = f.copy()
        bounds = [(0, None)] * n
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S,
                        'jac': lambda x: np.ones(n)}]
        
        def obj(x):
            return x @ Q @ x
        
        def grad(x):
            return 2 * Q @ x
        
        result = minimize(obj, x0, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-10})
        
        it += 1
        if result.success:
            f_new = np.clip(result.x, 0, None)
            s_new = np.sum(f_new)
            if s_new < 1e-10:
                continue
            f_new = f_new * (S / s_new)
            score_new = local_score(f_new)
            
            if score_new < best_score:
                best_score = score_new
                best_f = f_new.copy()
                f = f_new
            elif it % 5 == 0:
                # Blend to avoid stalling
                alpha = 0.3
                f = (1 - alpha) * f + alpha * f_new
                f = np.clip(f, 0, None)
                f = f * (S / np.sum(f))
        
        # Occasionally add perturbation
        if it % 20 == 0:
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0, None)
            f = f * (S / np.sum(f))
        
        if verbose and it % 10 == 0:
            print(f"  QP iter={it}, peak={peak_val:.6f}, score={local_score(f):.6f}, best={best_score:.6f}")
    
    return best_f, best_score

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 3)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_spectral_init(n, rng):
    """Initialize using flat-spectrum approach."""
    # Random phases with constant magnitude
    freqs = np.fft.rfftfreq(n)
    phases = rng.uniform(0, 2*np.pi, size=len(freqs))
    # Make real: conjugate symmetry
    if len(freqs) > 1:
        phases[0] = 0
        if n % 2 == 0 and len(freqs) > n//2:
            phases[-1] = 0
    A = np.ones(len(freqs)) * np.exp(1j * phases)
    a = np.fft.irfft(A, n)
    a = np.abs(a)  # Take absolute value for non-negativity
    if np.sum(a) < 1e-10:
        a = np.ones(n)
    a = a / np.sum(a)
    return a
