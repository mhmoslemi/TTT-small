Multi-phase optimization for minimizing peak linear convolution of non-negative sequences.
Uses gradient-based methods with smooth max approximation, parametric initialization,
LP refinement, and multi-start strategy.
"""

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def conv_and_grad(a, method='active', K=10):
    """
    Compute score and gradient.
    method: 'exact' for true max, 'active' for top-K average, 'smooth' for log-sum-exp
    """
    n = len(a)
    c = fft_convolve(a, a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    if method == 'exact':
        idx = int(np.argmax(c))
        score = 2.0 * n * c[idx] / (S * S)
        # Gradient: dc_idx/da = 2 * R_idx * a
        g = np.zeros(n)
        for j in range(n):
            k = idx - j
            if 0 <= k < 2 * n - 1:
                g[j] = 2.0 * a[k] if 0 <= k < n else 0.0
        # Wait, (R_idx a)[j] = a[k] where k = idx - j, but only if 0 <= k < n
        # Actually, (R_k a)[j] = a[k-j] if 0 <= k-j < n
        # So dc_k/da[j] = 2 * a[k-j] if 0 <= k-j < n
        g = np.zeros(n)
        for j in range(n):
            k = idx - j
            if 0 <= k < n:
                g[j] = 2.0 * a[k]
        score_grad = g / (S * S) - 2.0 * c[idx] / (S ** 3) * np.ones(n)
        return 2.0 * n * c[idx] / (S * S), 2.0 * n * score_grad
    
    elif method == 'active':
        top_idx = np.argsort(c)[-K:]
        c_top = c[top_idx]
        avg_c = float(np.mean(c_top))
        score = 2.0 * n * avg_c / (S * S)
        g = np.zeros(n)
        for idx in top_idx:
            idx = int(idx)
            for j in range(n):
                k = idx - j
                if 0 <= k < n:
                    g[j] += 2.0 * a[k]
        g /= K
        score_grad = g / (S * S) - 2.0 * avg_c / (S ** 3) * np.ones(n)
        return 2.0 * n * avg_c / (S * S), 2.0 * n * score_grad
    
    elif method == 'smooth':
        tau = 0.01 * S * S / n  # scale tau appropriately
        c_scaled = c / tau
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        w = exp_c / np.sum(exp_c)
        avg_c = float(np.dot(w, c))
        score = 2.0 * n * avg_c / (S * S)
        
        # g[j] = sum_k w[k] * 2*a[k-j] (where valid)

            j_min = max(0, k - (n - 1))

        return score, 2.0 * n * score_grad

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based gradient computation, L-BFGS-B with smooth max approximation,
parametric initialization families, LP refinement, and multi-start.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def obj_and_grad(a, tau_factor=0.05):
    """
    Objective and gradient using smooth max (log-sum-exp) approximation.
    tau is scaled relative to the max convolution value.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)
    c_max = float(np.max(c))
    tau = tau_factor * c_max if c_max > 0 else 1.0
    
    # Softmax weights
    c_scaled = c / tau
    c_scaled_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_scaled_max)
    w = exp_c / np.sum(exp_c)
    
    avg_c = float(np.dot(w, c))
    score = 2.0 * n * avg_c / (S * S)
    
    # Compute s[j] = sum_m w[j+m] * a[m] via FFT
    a_rev = a[::-1]
    conv_w_arev = fft_convolve(w, a_rev)
    s = conv_w_arev[n - 1:2 * n - 1]
    
    # Gradient
    grad = 4.0 * n * s / (S * S) - 4.0 * n * avg_c / (S ** 3) * np.ones(n)
    
    return score, grad


def obj_and_grad_active(a, K=5):
    """
    Objective and gradient using top-K active set.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)
    top_idx = np.argsort(c)[-K:]
    c_top = c[top_idx]
    avg_c = float(np.mean(c_top))
    score = 2.0 * n * avg_c / (S * S)
    
    # Build weight vector (1/K for top K, 0 otherwise)
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / K
    
    # Compute s[j] via FFT
    a_rev = a[::-1]
    conv_w_arev = fft_convolve(w, a_rev)
    s = conv_w_arev[n - 1:2 * n - 1]
    
    grad = 4.0 * n * s / (S * S) - 4.0 * n * avg_c / (S ** 3) * np.ones(n)
    
    return score, grad


def lbfgs_optimize(a_init, time_limit, tau_factor=0.05, verbose=True):
    """
    Optimize using L-BFGS-B with smooth max objective.
    Returns (best_a, best_score).
    """
    a_init = np.clip(np.asarray(a_init, dtype=np.float64), 0.01, 1000.0)
    n = len(a_init)
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = score_fast(a_init)
    
    def wrapped_obj_grad(x):
        x = np.clip(x, 0.01, 1000.0)
        s, g = obj_and_grad(x, tau_factor)
        if np.isinf(s):
            return 1e10, np.zeros_like(x)
        return s, g
    
    try:
        res = minimize(
            wrapped_obj_grad,
            a_init,
            method='L-BFGS-B',
            jac=True,
            bounds=[(0.01, 1000.0)] * n,
            options={
                'maxiter': 500,
                'maxfun': 2000,
                'ftol': 1e-12,
                'gtol': 1e-8,
            }
        )
        
        if res.fun < best_score and np.isfinite(res.fun):
            a_res = np.clip(res.x, 0.01, 1000.0)
            true_score = score_fast(a_res)
            if true_score < best_score:
                best_score = true_score
                best_a = a_res.copy()
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    L-BFGS done in {elapsed:.1f}s, score={best_score:.6f}")
    
    return best_a, best_score

def lp_refine(f, M, timeout=5.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
