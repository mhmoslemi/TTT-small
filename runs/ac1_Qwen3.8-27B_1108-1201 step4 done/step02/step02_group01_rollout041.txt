def lp_with_constraint_generation(a, M, max_constraints=50):
    """LP with constraint generation: start with few constraints, add as needed."""
    n = len(a)
    c = -np.ones(n)
    constraints_a = []
    constraints_b = []
    
    # Start with top lags
    conv_a = fft_convolve(a, a)
    top_lags = np.argsort(conv_a)[::-1][:max_constraints]
    
    for lag in top_lags:
        row = np.zeros(n)
        for j in range(n):
            i1 = lag - j
            if 0 <= i1 < n:
                row[j] = a[i1]
            i2 = j - lag
            if 0 <= i2 < n:
                row[j] = a[i2]
        # Actually for conv(a,g)[lag] = sum_j a[lag-j] * g[j] (with bounds)
        # So row[j] = a[lag-j] if 0 <= lag-j < n, else 0
        # Wait, conv(a,g)[k] = sum_i a[i] * g[k-i]
        # So for fixed k and varying g[j]: conv(a,g)[k] = sum_j a[k-j] * g[j]

        row = np.zeros(n)

            if 0 <= idx < n:
                row[j] = a[idx]
        constraints_a.append(row)
        constraints_b.append(M)
    
    res = linprog(c, A_ub=np.array(constraints_a), b_ub=np.array(constraints_b),
                  bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    
    g = res.x
    
    # Verify all constraints
    conv_ag = fft_convolve(a, g)
    if np.max(conv_ag) <= M * (1 + 1e-6):
        return g
    
    # Add violated constraints
    violated = np.where(conv_ag > M * (1 + 1e-6))[0]

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))

def normalize(a, S=1.0):
    """Normalize to have sum S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * S / s

def build_row(a, lag):
    """Build the row of the LP matrix for a given lag constraint."""
    n = len(a)
    row = np.zeros(n)
    for j in range(n):
        idx = lag - j
        if 0 <= idx < n:
            row[j] = a[idx]
    return row

def solve_lp(a, M, top_k=None):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for top_k lags, g >= 0."""
    n = len(a)
    if top_k is None:
        top_k = 2 * n - 1  # all lags
    
    conv_a = fft_convolve(a, a)
    sorted_lags = np.argsort(conv_a)[::-1][:top_k]
    
    A = np.array([build_row(a, lag) for lag in sorted_lags])
    c = -np.ones(n)
    b = np.full(top_k, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None

def line_search_blend(a, g, M, S=1.0):
    """Find best blend parameter t: minimize max(conv((1-t)a + t*g, ...)) subject to <= M."""
    n = len(a)
    g_norm = normalize(g, S)
    
    best_t = 0.0
    best_val = np.inf
    
    # Try a range of t values
    for t in np.linspace(0.0, 1.0, 21):
        b = (1 - t) * a + t * g_norm
        b = np.maximum(b, 0.0)
        s = np.sum(b)
        if s < 0.01:
            continue
        b = normalize(b, S)
        val = np.max(fft_convolve(b, b))
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, g_norm

def lp_iterate(a, time_limit, S=1.0, verbose=False):
    """LP-based iteration with adaptive M and line search."""
    n = len(a)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_a = fft_convolve(a, a)
        current_peak = np.max(conv_a)
        
        if current_peak >= best_peak:
            # Reset to best
            a = best_a.copy()
            conv_a = fft_convolve(a, a)
            current_peak = np.max(conv_a)
        
        # Try different M values
        for M_factor in [1.0, 0.999, 0.998, 0.995]:
            M = M_factor * current_peak
            if time.monotonic() - t0 > time_limit:
                break

            if g is None or np.sum(g) < 1e-10:
                continue
            
            g_norm = normalize(g, S)
            
            # Line search for best blend
            best_t = 0.0
            best_blended_peak = current_peak
            
            for t in np.linspace(0.01, 1.0, 30):
                b = (1 - t) * a + t * g_norm
                b = np.maximum(b, 0.0)
                b = normalize(b, S)
                peak_b = np.max(fft_convolve(b, b))
                if peak_b < best_blended_peak:
                    best_blended_peak = peak_b
                    best_t = t
            
            if best_t > 0 and best_blended_peak < best_peak - 1e-10:
                a = (1 - best_t) * a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = normalize(a, S)
                best_peak = best_blended_peak
                best_a = a.copy()
                if verbose:

                    print(f"    LP iter {iteration}: M={M:.4f}, t={best_t:.3f}, peak={best_peak:.6f}, score={score:.6f}, time={elapsed:.0f}s")

def momentum_gradient(a, time_limit, S=1.0, verbose=False):
    """Momentum-based gradient descent on max(conv(a,a))."""
    n = len(a)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    v = np.zeros(n)  # momentum
    lr = 0.1 / n
    beta = 0.9
    stall = 0
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Subgradient
        grad = np.zeros(n)
        k = peak_idx
        for j in range(n):
            if j + k < n:
                grad[j] += a[j + k]
            if j - k >= 0:
                grad[j] += a[j - k]
        # Note: this is actually the gradient of the quadratic form
        # For j where j+k == j-k (i.e., k=0), we get 2*a[j]
        # For k != 0, we get a[j+k] + a[j-k] (with bounds)
        
        # Momentum step
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)
        
        new_peak = np.max(fft_convolve(a_new, a_new))
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 100:
            lr *= 0.9
            v *= 0.5
            stall = 0
        
        a = a_new
        
        if verbose and step % 1000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    MomGrad step {step}: peak={best_peak:.6f}, score={score:.6f}, lr={lr:.6f}")
    
    return best_a

Optimized LP-Gradient hybrid for minimizing normalized peak autocorrelation
of non-negative step functions. Uses restricted-constraint LP for speed,
adaptive M reduction, proper line search, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a, S=1.0):
    """Normalize sequence to have sum S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * S / s


def build_conv_matrix(a):
    """Build full convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_restricted(a, lags):
    """Build restricted convolution matrix for given lags."""
    n = len(a)
    A = np.zeros((len(lags), n))
    for ii, k in enumerate(lags):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[ii, j] = a[k - j]
    return A


def solve_lp_full(a, M):
    """Solve full LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None


def solve_lp_restricted(a, M, top_k=50):
    """Solve LP with only top-k lag constraints."""
    n = len(a)
    conv_a = fft_convolve(a, a)
    top_lags = np.argsort(conv_a)[::-1][:top_k]
    A = build_conv_matrix_restricted(a, top_lags)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(top_k, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None


def line_search(a, g, S=1.0, n_trials=40):
    """Find best blend parameter t to minimize max(conv)."""
    g_norm = normalize(g, S) if np.sum(g) > 0 else None
    if g_norm is None:
        return 0.0, a.copy()
    
    best_t = 0.0
    best_val = np.max(fft_convolve(a, a))
    best_b = a.copy()
    
    # Coarse search
    for t in np.linspace(0.0, 1.0, n_trials):
        b = (1.0 - t) * a + t * g_norm
        b = np.maximum(b, 0.0)
        b = normalize(b, S)
        val = np.max(fft_convolve(b, b))
        if val < best_val:
            best_val = val
            best_t = t
            best_b = b
    
    # Fine search around best
    if best_t > 0:
        t_lo = max(0.0, best_t - 1.0 / n_trials)
        t_hi = min(1.0, best_t + 1.0 / n_trials)
        for t in np.linspace(t_lo, t_hi, 10):
            b = (1.0 - t) * a + t * g_norm
            b = np.maximum(b, 0.0)
            b = normalize(b, S)
            val = np.max(fft_convolve(b, b))
            if val < best_val:
                best_val = val
                best_t = t
                best_b = b
    
    return best_t, best_b

def gradient_optimize(a, time_limit, S=1.0, verbose=False):
    """Momentum-based gradient descent."""
    n = len(a)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    v = np.zeros(n)
    lr = 0.5 / n
    beta = 0.85
    stall = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # Compute subgradient
        grad = np.zeros(n)
        k = peak_idx
        if k == 0:
            grad = 2.0 * a
        else:
            for j in range(n):
                g = 0.0
                if j + k < n:
                    g += a[j + k]
                if j - k >= 0:
                    g += a[j - k]
                grad[j] = g

        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)

            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still update a for momentum
            stall += 1

            lr *= 0.9
            v *= 0.5
            stall = 0
            # Reset to best if we've been stuck

                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

            score = 2.0 * n * best_peak / (S * S)

    return best_a
