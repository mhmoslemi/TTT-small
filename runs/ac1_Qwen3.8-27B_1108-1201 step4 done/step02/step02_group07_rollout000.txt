"""

score of non-negative sequences. Uses adaptive-constraint LP iterations,
Nesterov-accelerated subgradient descent, and targeted local search.
Key improvements: golden-section line search, lazy constraint LP,
spectral-aware initialization, and Nesterov momentum with Polyak steps.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def conv_matrix(f):
    """Build matrix A where A@b = conv(f, b) (length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, n_constraints=None, timeout=10.0):
    """Solve max sum(b) s.t. conv(f,b) <= M (for selected constraints), b >= 0.
    If n_constraints is set, only use the top n_constraints constraint rows.
    Returns b or None."""
    from scipy.optimize import linprog
    n = len(f)
    c_full = fft_convolve(f, f)
    if n_constraints is not None and n_constraints < len(c_full):
        # Use only top constraints (highest f*f values correlate with tightest)
        # Actually, use all constraints but let solver handle it
        # For speed, use constraints where c_full is in top n_constraints
        idx = np.argsort(c_full)[::-1][:n_constraints]
        A = np.zeros((len(idx), n), dtype=np.float64)
        for j in range(n):
            A[:len(idx), j] = f[idx - j:idx - j + n] if idx[0] - j >= 0 else 0
        # Rebuild properly
        A = np.zeros((len(idx), n), dtype=np.float64)
        for j in range(n):
            start = 0
            end = min(len(idx), n + j)  # this is getting complex, let's just use full
        # Simpler: just use full matrix
        A = conv_matrix(f)
        idx_sorted = np.argsort(c_full)[::-1][:n_constraints]
        A = A[idx_sorted, :]
        b_ub = np.full(len(idx_sorted), M)
    else:
        A = conv_matrix(f)
        b_ub = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def golden_section_search(f, b_dir, S_target, iters=20):
    """Find best t in [0,1] to minimize score of (1-t)*f + t*b_dir_scaled."""
    b_scaled = b_dir * (S_target / max(np.sum(b_dir), 1e-15))
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = score_fast(np.clip((1 - c) * f + c * b_scaled, 0.0, 1000.0))
    fd = score_fast(np.clip((1 - d) * f + d * b_scaled, 0.0, 1000.0))
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = score_fast(np.clip((1 - c) * f + c * b_scaled, 0.0, 1000.0))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = score_fast(np.clip((1 - d) * f + d * b_scaled, 0.0, 1000.0))
    t_best = (a + b) / 2.0
    return t_best


def subgradient_score(f, top_k=3):
    """Compute subgradient of score(a) = 2n*max(conv(a,a))/S^2 with respect to a.
    Returns gradient vector (before projection/normalization)."""
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return np.zeros(n)
    c = fft_convolve(f, f)
    # Find top-k peak positions
    top_idx = np.argsort(c)[-top_k:]
    max_b = float(c[top_idx[0]])
    # Score = 2n*max_b/S^2
    # d(score)/d(a_j) = 2n/S^2 * d(max_b)/d(a_j) - 4n*max_b/S^3
    # d(max_b)/d(a_j) averaged over top peaks:
    g_b = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            g_b[j] += 2.0 * f[ki - j]
    g_b /= float(len(top_idx))
    g = (2.0 * n / (S * S)) * g_b - (4.0 * n * max_b / (S * S * S))
    return g


def nesterov_subgrad(f_init, time_limit, verbose=True):
    """Nesterov-accelerated subgradient descent with Polyak step size."""
    n = len(f_init)
    # Normalize to sum = sqrt(2n) for numerical stability
    S0 = float(np.sum(f_init))
    f = f_init.copy() * (np.sqrt(2.0 * n) / max(S0, 1e-10))
    y = f.copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    # Lower bound on score (theoretical): 2n/(2n-1)
    LB = 2.0 * n / (2.0 * n - 1.0)
    momentum = 0.0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        g = subgradient_score(y, top_k=3)
        # Polyak step: t_k = (score(y) - LB) / ||g||^2
        score_y = score_fast(y)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-20:
            break
        t_k = min((score_y - LB) / g_norm_sq, 0.01)
        t_k = max(t_k, 1e-6)

        # Gradient step from y
        y_new = y - t_k * g
        y_new = np.clip(y_new, 0.0, 1000.0)

        # Nesterov update
        beta_new = 2.0 / (it + 3.0)
        f_new = y_new + (beta_new - 1.0) * (y_new - f)
        f_new = np.clip(f_new, 0.0, 1000.0)

        # Renormalize to maintain sum
        S_new = float(np.sum(f_new))
        if S_new > 1e-10:
            f_new = f_new * (float(np.sqrt(2.0 * n)) / S_new)

        f = f_new
        y = f_new
        it += 1

        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [nesterov {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f} t_k={t_k:.6f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def coordinate_local_search(f_init, time_limit, verbose=True):
    """Targeted coordinate-wise local search focusing on peak contributors."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S0 = float(np.sum(f))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temp = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        c = fft_convolve(f, f)
        peak_pos = int(np.argmax(c))

        # Find top contributors to the peak
        contribs = np.zeros(n)
        for j in range(n):
            idx = peak_pos - j
            if 0 <= idx < n:
                contribs[j] = f[j] * f[idx]

        # Choose a coordinate to perturb: weighted by contribution
        top_frac = max(10, n // 8)
        top_indices = np.argsort(contribs)[-top_frac:]
        j = int(rng.choice(top_indices))

        # Perturbation: reduce or increase
        delta_scale = rng.uniform(0.01, 0.08)
        if rng.random() < 0.7:
            # Reduce (to lower the peak)
            f[j] *= (1.0 - delta_scale)
        else:
            # Increase (might help redistribute)
            f[j] *= (1.0 + delta_scale)

        f[j] = max(0.01, f[j])

        # Renormalize
        S = float(np.sum(f))
        f = f * (S0 / S)

        s = score_fast(f)
        it += 1

        if s < best_score:
            best_score = s
            best_f = f.copy()
            temp = max(temp * 0.95, 1e-6)
        elif rng.random() < np.exp(-(s - best_score) / max(temp, 1e-15)):
            temp = max(temp * 0.99, 1e-6)
        else:
            # Revert
            S = float(np.sum(f))
            # Recompute from best
            f = best_f.copy()

        if it % 500 == 0 and verbose:
            print(f"    [local {time.monotonic()-t0:6.1f}s] it={it} best={best_score:.6f} temp={temp:.6f}")

    return best_f, best_score


def spectral_init(n, rng):
    """Initialize using spectral method: random phase FFT -> take magnitude -> IFFT."""
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    # Random positive spectrum
    A = rng.chisquare(df=2, size=N) * 0.5 + 1.0
    # Make it conjugate symmetric for real output
    a = irfft(A * 0.5 + A[:1], N)
    a = a[:n]
    a = np.abs(a) + 0.1
    return np.clip(a, 0.01, None)


def windowed_init(n, rng, window_type='hann'):
    """Generate windowed initialization."""
    t = np.arange(n, dtype=np.float64)
    if window_type == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    elif window_type == 'blackman':
        w = 0.42 - 0.5 * np.cos(2.0 * np.pi * t / max(1, n - 1)) + 0.08 * np.cos(4.0 * np.pi * t / max(1, n - 1))
    elif window_type == 'flat_top':
        a1, a2, a3, a4, a5 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
        w = a1 - a2 * np.cos(2 * np.pi * t / (n - 1)) + a3 * np.cos(4 * np.pi * t / (n - 1)) - a4 * np.cos(6 * np.pi * t / (n - 1)) + a5 * np.cos(8 * np.pi * t / (n - 1))
    else:
        w = np.ones(n)
    a = 0.5 + w + 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def lp_optimize(f_init, time_limit, K_start=5, K_end=None, verbose=True):

    n = len(f_init)
    f = f_init.astype(np.float64).copy()

    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()

        M = float(np.max(c))
        S_cur = float(np.sum(f))

        progress = min(1.0, it / 50.0)
        K = int(K_start + (K_end - K_start) * progress)
        K = min(K, 2 * n - 1)

        # Solve LP with top-K constraints
        A_full = conv_matrix(f)
        if K < 2 * n - 1:
            # Select constraints: use all (safer) but timeout handles speed

            A_sub = A_full[idx, :]
            b_ub_sub = np.full(K, M)
        else:
            A_sub = A_full
            b_ub_sub = np.full(2 * n - 1, M)

        c_obj = -np.ones(n)
        timeout_lp = min(8.0, max(2.0, (time_limit - (time.monotonic() - t0)) / 3.0))
        try:
            res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub_sub,

            if not res.success:

                continue
            b_dir = res.x

            stall += 1
            if stall > 30:
                break
            continue

                break
            continue

def multi_peak_lp(f_init, time_limit, verbose=True):

            print(f"  *** Improved: {best_score:.6f} ***")

    # Phase 5: Try a different length with fresh start
    if time.monotonic() < deadline - 80:
        print("\n=== Phase 5: Alternative lengths ===")
        for n_try in [384, 512]:
            if time.monotonic() > deadline - 60:
                break
            for wtype in ['blackman', 'flat_top']:
                if time.monotonic() > deadline - 60:
                    break
                f0 = windowed_init(n_try, rng, wtype)
                t_alloc = min(120.0, deadline - time.monotonic() - 40.0)
                if t_alloc < 30:
                    break
                print(f"  Trying n={n_try}, window={wtype}, time={t_alloc:.0f}s")
                f_opt, s_opt = lp_optimize(f0, t_alloc, K_start=5, K_end=2 * n_try - 1, verbose=False)
                if s_opt < best_score:
                    best_score = s_opt
                    best_seq = f_opt.copy()
                    print(f"  *** NEW BEST: n={n_try}, score={best_score:.6f} ***")

    # Phase 6: Final polish with local search
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_alloc = min(60.0, deadline - time.monotonic() - 10.0)
        if t_alloc > 15:
            print(f"\n=== Phase 6: Final polish ({t_alloc:.0f}s) ===")
            f_opt, s_opt = coordinate_local_search(best_seq, t_alloc, verbose=False)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = f_opt.copy()
                print(f"  Final: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    return [1.0] * 256
