"""
Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses LP-based direction finding with line search,
diverse non-symmetric initializations, gradient polishing, and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix A for LP. A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success:
        return result.x
    return None


def lp_optimize_with_linesearch(f_init, time_limit, verbose=False):
    """LP-based iterative optimization with line search for step size."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    iteration = 0
    rng_local = np.random.default_rng(42)
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if verbose and iteration % 20 == 0:
            print(f"    LP iter={iteration}, score={local_score(f):.6f}, "
                  f"best={best_score:.6f}, elapsed={elapsed:.1f}s")
        
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = solve_lp(f, M)
        if g is None or np.sum(g) < 1e-10:
            break
        
        # Rescale g to have same L1 norm as f
        sum_g = np.sum(g)
        g_rescaled = g * (sum_f / sum_g)
        
        # Line search
        best_t = 0.0
        best_score_candidate = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t) * f + t * g_rescaled
            f_trial = np.clip(f_trial, 0.0, None)
            s = np.sum(f_trial)
            if s < 1e-10:
                continue
            f_trial = f_trial * (sum_f / s)
            score_trial = local_score(f_trial)
            if score_trial < best_score_candidate:
                best_score_candidate = score_trial
                best_t = t
        
        if best_t == 0.0:
            # No improvement found, try a small random perturbation
            stagnant += 1
            if stagnant > 30:
                f = f + 0.05 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                stagnant = 0
            if stagnant > 100:
                break
            continue
        
        f = (1.0 - best_t) * f + best_t * g_rescaled
        f = np.clip(f, 0.0, None)
        s = np.sum(f)
        f = f * (sum_f / s)
        iteration += 1
        stagnant = 0
        
        if local_score(f) < best_score:
            best_score = local_score(f)
            best_f = f.copy()
    
    return best_f, best_score

def sa_polish(f_init, time_limit, rng):

        candidate = f.copy()

            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort
            top_idx = top_idx(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.50:
            # Local perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.70:
            # Pair swap (preserves sum)
            i, j = rng.choice(n, 2, replace=False)
            swap = candidate[i].copy()
            candidate[i] = candidate[j]
            candidate[j] = swap
            # Small perturbation after swap
            candidate[i] *= np.exp(rng.normal(0, 0.01))
            candidate[j] *= np.exp(rng.normal(0, 0.01))
        elif r < 0.85:
            # Asymmetric shift
            shift = int(rng.integers(1, max(2, n // 30)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= cooling

    return best_f, best_score


def make_exp_decay_init(n, rng, alpha_range=(0.5, 3.0)):
    """Exponential decay initialization (strongly non-symmetric)."""
    alpha = rng.uniform(*alpha_range)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_ramp_init(n, rng, slope_range=(0.5, 2.0)):
    """Linear ramp initialization."""
    slope = rng.uniform(*slope_range)
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * t / n
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng, alpha_range=(0.2, 0.8)):
    """Chirp-like initialization."""
    alpha = rng.uniform(*alpha_range)
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng, n_tones=5):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_asymmetric_multitone(n, rng):
    """Asymmetric multi-tone: mix of low and high frequencies with amplitude ramp."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    # Add asymmetric components
    a += 0.3 * np.exp(-t / (n * 0.3))  # Positive DC offset decaying
    for _ in range(4):
        k = int(rng.integers(1, max(2, n // 5)))
        amp = rng.uniform(0.05, 0.12)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP with line search, diverse initializations,
    gradient polish, and simulated annealing.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # === Phase 1: LP-based optimization with diverse initializations ===
    print("\n" + "=" * 60)
    print("Phase 1: LP-based iterative optimization")
    print("=" * 60)

    n_prov = len(provided_seq) if provided_seq is not None else 256
    
    # Define search configurations: (n, init_func, init_args)
    def _exp_decay(n_, rng_):
        return make_exp_decay_init(n_, rng_)
    def _ramp(n_, rng_):
        return make_ramp_init(n_, rng_)
    def _chirp(n_, rng_):
        return make_chirp_init(n_, rng_)
    def _multitone(n_, rng_):
        return make_multitone_init(n_, rng_)
    def _random(n_, rng_):
        return make_random_init(n_, rng_, 0.2 + 0.3 * rng_.random())
    def _asym(n_, rng_):
        return make_asymmetric_multitone(n_, rng_)
    def _provided(n_, rng_):
        if provided_seq is not None:
            if len(provided_seq) == n_:
                return provided_seq.copy()
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n_)
            return np.clip(np.interp(x_new, x_old, provided_seq), 0.01, None)
        return np.ones(n_)

    search_configs = [
        (n_prov, _provided, 0.25),
        (n_prov, _exp_decay, 0.15),
        (n_prov, _ramp, 0.10),
        (n_prov, _asym, 0.10),
        (n_prov, _chirp, 0.10),
        (512, _exp_decay, 0.10),
        (512, _ramp, 0.08),
        (n_prov, _multitone, 0.07),
        (384, _exp_decay, 0.05),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = deadline - time.monotonic()

    for n, init_func, frac in search_configs:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(20.0, frac / total_frac * available_time * 0.85)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 20)
        if time_alloc < 15:
            break

        a0 = init_func(n, rng)
        print(f"\n  n={n}, init, alloc={time_alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = lp_optimize_with_linesearch(a0, time_alloc, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  LP done: score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")

    # === Phase 2: Gradient polish ===
    print("\n" + "=" * 60)
    print("Phase 2: Gradient-based polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(200.0, remaining * 0.4)
