def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_subgrad(f, conv, n, top_k=10):
    """Compute a smooth subgradient using top-k peak positions."""
    sg = np.zeros(n)
    thresh = conv.max() * 0.9
    active = np.where(conv > thresh)[0]
    weights = conv[active] - thresh
    weights = np.maximum(weights, 0)
    if weights.sum() < 1e-15:
        active = np.array([np.argmax(conv)])
        weights = np.array([1.0])
    else:
        weights /= weights.sum()
    
    for idx, k in enumerate(active):
        k = int(k)
        w = weights[idx]
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                sg[j] += w * f[ki]
    return sg

def precompute_lp_matrix(n):
    """Precompute sparse matrix for cross-conv constraints."""
    rows_list = []
    cols_list = []
    for j in range(n):
        for k in range(2 * n - 1):
            ki = k - j
            if 0 <= ki < n:
                rows_list.append(k)
                cols_list.append(j)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    return rows, cols

def solve_lp_warm(f, M, n, rows, cols, x0=None, timeout=10):
    """Solve max sum(g) s.t. (f*g)[k] <= M, g >= 0, with warm start."""
    # Build sparse matrix
    vals = np.empty(len(rows), dtype=np.float64)
    # For each (row=k, col=j), value is f[k-j]
    # rows[i] = k, cols[i] = j, so values[i] = f[rows[i] - cols[i]]
    vals = f[rows - cols]
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    options = {'disp': False, 'time_limit': timeout}
    if x0 is not None:
        # HiGHS doesn't directly support warm start via linprog,
        # but we can try passing it
        pass
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=options)
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = precompute_lp_matrix(n)
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    momentum = np.zeros(n)
    g_prev = None
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        g = solve_lp_warm(f, M, n, rows, cols, x0=g_prev, timeout=8)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        g_prev = g.copy()
        
        # Momentum
        direction = g - f
        momentum = momentum_coef * momentum + (1 - momentum_coef) * direction
        momentum /= max(np.max(np.abs(momentum)), 1e-15)
        
        if np.max(np.abs(g - f)) < 1e-13 and np.max(np.abs(momentum)) < 1e-12:
            break
        
        # Backtracking line search with momentum direction
        improved = False
        for t_try in [t, 0.8*t, 0.6*t, 0.4*t, 0.2*t, 0.1*t]:
            f_new = f + t_try * momentum
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:

            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

                noise = rng_dirichlet(n)

                f = np.maximum(f, 0)
                f /= f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                patience = 0
                t = 0.5
    
    return best_f, best_obj

The spectral method is interesting. If I want the auto-conv to be flat, I want the "power spectrum" to be as flat as possible. For a sequence a, the power spectrum is |A|^2 where A = FFT(a). The auto-conv (circular) is IFFT(|A|^2). For linear auto-conv, it's approximately the circular one if the sequence is "band-limited" enough.

In the frequency domain, this is: minimize max_k (circular auto-conv)[k] subject to a >= 0, sum(a) = 1.

The "flat" power spectrum corresponds to a being "white noise." But we need a >= 0, which constrains the feasible set.

1. Start with random a
2. Compute A = FFT(a)
3. Modify A to make |A| more flat (e.g., replace |A| with a smoothed version)
4. a_new = IFFT(A_modified)
5. Project a_new to be non-negative (set negative values to 0)
6. Normalize a_new to sum 1
7. Repeat

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def objective_val(a):

    rows_list = []
    cols_list = []

            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, n, rows, cols, timeout=10):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    vals = f[rows - cols]
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method='highs', options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def ift_optimize(f0, time_limit, alpha=0.5, iter_per_check=50):
    """IFT-based spectral balancing optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    patience = 0

    while time.monotonic() - t0 < time_limit:
        # Compute FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2

        # Compress the power spectrum
        p_max = power.max()
        if p_max < 1e-15:
            break
        p_norm = power / p_max

        # Apply alpha compression: new_power = p_norm ^ alpha
        # This makes the spectrum flatter (alpha < 1 compresses dynamic range)
        new_power = p_norm ** alpha

        # Preserve the DC component (sum)
        new_power[0] = power[0]

        # Reconstruct
        # We need to find a phase such that IFFT(sqrt(new_power) * phase) is non-negative
        # Simple approach: use the original phase
        new_A = np.sqrt(new_power) * (A / np.abs(A + 1e-30))
        f_new = np.fft.irfft(new_A, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-12:
            break
        f_new /= sn

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj * (1 - 1e-10):
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            patience = 0
        else:
            patience += 1
            if patience > 100:
                # Try different alpha
                alpha = alpha * 0.9 if alpha > 0.1 else alpha * 1.1
                alpha = np.clip(alpha, 0.1, 2.0)
                patience = 0
            if patience > 500:
                break

    return best_f, best_obj


def gen_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-6)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal (U-shape)
    u = (x / (n - 1) - 0.5) ** 2
    u = u / u.sum()
    inits['u_shape'] = u

    # Cosine
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    c = np.maximum(c, 1e-6)
    c /= c.sum()
    inits['cosine'] = c

    # Quadratic residues pattern (approximate)

    # Create a pattern based on bit-reversal
    br = np.array([int(format(i, '09b')[::-1], 2) for i in range(n)])
    # Use this to create a structured pattern
    pattern = 0.5 + 0.5 * np.cos(2 * np.pi * br / n * 1.0)

    inits['rand'] = rng.dirichlet(np.ones(n) * 0.5)

    saw = (x % max(1, n // 8)) / max(1, n // 8)
    saw = np.maximum(saw, 1e-6)

    inits['sawtooth'] = saw

3. Check if sum(a) >= target (e.g., 1). If not, increase M and repeat.
4. Iterate until convergence.

- Binary search on M
- For each M, run coordinate descent to find the max sum achievable
