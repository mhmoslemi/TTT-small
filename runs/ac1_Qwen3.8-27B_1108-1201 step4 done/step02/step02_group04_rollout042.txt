from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * target_sum / s

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def compute_lse_gradient(a, w, n):
    """Compute sum_k w[k] * 2*a[k-j] for each j using FFT cross-correlation."""
    # w has length 2n-1, a has length n

    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    c = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * c

    exp_vals = np.exp(t_conv_shifted)
    w = exp_vals / exp_vals.sum()
    
    # LSE value (log of sum of exps, shifted for numerical stability)
    LSE = (t_conv.max() + np.log(exp_vals.sum())) / t
    
    # Objective: 2n * LSE / S^2
    obj = 2.0 * n * LSE / (S * S)
    
    # Gradient of LSE w.r.t. a[j]:
    # d LSE / d a[j] = (1/t) * sum_k exp(t*conv[k]) / sum exp(t*conv[m]) * 2*a[k-j]
    # = (1/t) * sum_k w[k] * 2*a[k-j]
    grad_lse = compute_lse_gradient(a, w, n) / t
    
    # d obj / d a[j] = 2n/S^2 * d LSE / da[j] - 2n*LSE/S^3 * 1
    # (since d/d a[j] of 1/S^2 = -2/S^3)
    grad_obj = (2.0 * n / (S * S)) * grad_lse - (2.0 * n * LSE / (S ** 3))
    
    return obj, grad_obj

def adam_update(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with simplex projection."""
    m, v, t = state
    t = t + 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = simplex_project(a_new)
    new_state = (m, v, t)
    return a_new, new_state

"""
Minimizes 2*n*max(auto_conv(a))/(sum(a))^2 using:
1. Parametric family search for diverse starting points
2. Adam-optimized log-sum-exp smoothed descent (main optimization)
3. Subgradient refinement with Nesterov momentum
4. LP-based alternating polish
Key insight: for non-negative a, the auto-conv peak can be reduced below 1/n
by using non-symmetric (tilted) distributions, exploiting the fact that
(a*a)[n-1] = sum a[i]*a[n-1-i] < sum a[i]^2 for skewed a.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)


def next_pow2(n):
    return 1 << (max(n, 1) - 1).bit_length()


def compute_grad_lse(a, w, n):
    """Compute (1/t)*sum_k w[k]*2*a[k-j] for each j via FFT cross-correlation.
    c[j] = sum_i a[i] * w[i+j], using conj(FFT(a))*FFT(w)."""
    fft_size = next_pow2(3 * n)
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    c = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * c


def lse_obj_grad(a, t):
    """Compute log-sum-exp smoothed objective and gradient.
    Objective: 2n * (1/t)*log(sum exp(t*(a*a)[k])) / (sum a)^2"""
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    S2 = S * S
    tc = t * conv
    tc_shift = tc - tc.max()
    exp_vals = np.exp(tc_shift)
    W_sum = exp_vals.sum()
    w = exp_vals / W_sum
    LSE = (tc.max() + np.log(W_sum)) / t
    obj = 2.0 * n * LSE / S2
    grad_lse = compute_grad_lse(a, w, n) / t
    grad = (2.0 * n / S2) * grad_lse - (4.0 * n * LSE / (S2 * S))
    return obj, grad, conv, w


def adam_step(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update with simplex projection."""
    m, v, t = state
    t_new = t + 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** t_new)
    v_hat = v / (1.0 - beta2 ** t_new)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = simplex_project(a_new)
    return a_new, (m, v, t_new)


def nesterov_step(a, vel, grad, lr, mu=0.9):
    """Nesterov accelerated gradient step with simplex projection."""
    a_y = a + mu * vel
    vel_new = mu * vel - lr * grad
    a_new = simplex_project(a_y + vel_new)
    return a_new, vel_new


def subgradient_at_kstar(a, k_star, n):
    """Compute subgradient of max auto-conv at lag k_star.
    grad_j (a*a)[k_star] = 2*a[k_star - j] if 0 <= k_star - j <= n-1, else 0."""
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        # grad[j] = 2*a[k_star - j] for j in [j_start, j_end]
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = 2.0 * a[k_star - indices]
    return grad


def make_lp_structure(n):
    """Precompute sparse structure for cross-conv LP: (2n-1) x n matrix A where A[k,i] encodes (f*g)[k]."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_cross_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_refine(f0, time_limit):
    """LP-guided alternating refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    S = f.sum()
    if S < 1e-12:
        f = np.ones(n) / n
        S = 1.0
    f = f / S

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    step = 1.0
    patience = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_cross_lp(f, M, rows, cols, n, timeout=8)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [step, 0.6 * step, 0.4 * step, 0.2 * step, 0.1 * step]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new = f_new / s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                step = min(1.0, step * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            step *= 0.85
            if patience > 40:
                break
        if it % 100 == 0:
            print(f"      LP iter {it}, obj={best_obj:.8f}, remaining={time_limit-(time.monotonic()-t0):.0f}s")

    return best_f, best_obj


def gen_parametric(n, rng):
    """Generate diverse parametric initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Linear ramps with various slopes
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
        a = 1.0 + alpha * (2.0 * x / (n - 1) - 1.0)
        a = np.maximum(a, 0.01)
        inits[f'lin{int(alpha*10)}'] = a / a.sum()

    # Exponential decay
    for rate in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-rate * x / n)
        inits[f'exp{int(rate*10)}'] = a / a.sum()

    # Cosine window
    for freq in [1, 2, 3]:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        inits[f'cos{freq}'] = a / a.sum()

    # Quadratic
    for c in [0.3, 0.5, 0.7]:
        a = 1.0 - c * (x / (n - 1) - 0.5) ** 2 * 4.0
        a = np.maximum(a, 0.01)
        inits[f'quad{int(c*10)}'] = a / a.sum()

    # Sigmoid tilt
    for k in [5, 10, 20]:
        a = 1.0 / (1.0 + np.exp(-k * (x / n - 0.5)))
        inits[f'sig{k}'] = a / a.sum()

    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 0.1)

    return inits

    top_k = min(5, len(results))
    remaining = deadline - time.monotonic() - 200.0
    time_per_cand = max(30.0, remaining / top_k)

    for idx in range(top_k):
        if time.monotonic() > deadline - 200.0:
            break
        obj_init, name, a0 = results[idx]
        ta = min(time_per_cand, deadline - time.monotonic() - 200.0)
        if ta < 15:
            break

        print(f"  Optimizing [{name}] (init obj={obj_init:.6f}, time={ta:.0f}s)")
        a_opt, obj_opt = main_optimize(a0, time_limit=ta, t_start=15.0, t_end=400.0,
                                       lr_start=0.04, lr_end=0.0005)
        update_best(a_opt, f"adam:{name}")

    print(f"  Phase 2 done, best so far: {best_score:.6f}")

    # ========== Phase 3: Subgradient refinement ==========
    print("Phase 3: Subgradient refinement")
    if best_seq is not None and time.monotonic() < deadline - 150.0:
        ta = min(400.0, deadline - time.monotonic() - 150.0)
        a_sub, obj_sub = subgrad_refine(best_seq, time_limit=ta, lr_start=0.001, mu=0.9)
        update_best(a_sub, "subgrad")

    # Additional subgrad from top candidates
    for idx in range(min(3, top_k)):
        if time.monotonic() > deadline - 100.0:
            break
        _, name, a0 = results[idx]
        ta = min(100.0, deadline - time.monotonic() - 100.0)
        if ta < 10:
            break
        a_sub, _ = subgrad_refine(a0, time_limit=ta, lr_start=0.001, mu=0.9)
        update_best(a_sub, f"subgrad2:{name}")

    print(f"  Phase 3 done, best so far: {best_score:.6f}")

    # ========== Phase 4: LP polish ==========
    print("Phase 4: LP polish")
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        ta = min(250.0, deadline - time.monotonic() - 30.0)
        a_lp, obj_lp = lp_refine(best_seq, time_limit=ta)
        update_best(a_lp, "LP")

        # Second LP from a slightly perturbed version
        if time.monotonic() < deadline - 20.0:
            a_pert = simplex_project(best_seq + rng_global.standard_normal(n) * 0.01)
            ta2 = min(80.0, deadline - time.monotonic() - 20.0)
            if ta2 > 10:
                a_lp2, _ = lp_refine(a_pert, time_limit=ta2)
                update_best(a_lp2, "LP2")

    # ========== Phase 5: Final coordinate descent polish ==========
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        a = best_seq.copy().astype(np.float64)
        a = a / a.sum()
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        t0 = time.monotonic()
        step = 0.01

        while time.monotonic() - t0 < 60:
            conv = fast_convolve(a)
            k_star = int(np.argmax(conv))
            grad = subgradient_at_kstar(a, k_star, n)
            grad = grad - grad.mean()  # keep sum ~ 1

            improved = False
            for s in [step, step * 0.5, step * 0.25]:
                a_try = simplex_project(a - s * grad)
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < current_obj - 1e-14:
                    a = a_try
                    current_obj = obj_try
                    step = min(0.05, step * 1.1)
                    improved = True
                    break

            if not improved:
                step *= 0.8
                if step < 1e-5:
                    break

        update_best(a, "coord")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Ensure valid output
    final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = float(evaluate_sequence(final))
    if not np.isfinite(final_score):
        final = [1.0] * n
        final_score = float(evaluate_sequence(final))

    print(f"\nFINAL: n={len(final)}, score={final_score:.6f}")
    return final
