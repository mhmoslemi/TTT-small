"""
Optimized LP-based refinement with sparse convolution matrix, exact line search,
perturbation restarts, and coordinate descent polish for minimizing the
normalized peak autocorrelation of non-negative step functions.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the objective score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize to target sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def precompute_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_fast(a, M, rows, cols, warm_start=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Uses precomputed structure."""
    n = len(a)
    vals = np.empty(len(rows), dtype=np.float64)
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        vals[idx] = a[k - j]
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).to_csr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    if warm_start is not None:
        x0 = warm_start
    else:
        x0 = np.zeros(n)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, 1000.0)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return np.zeros(n)


def exact_line_search(a, g, M, n):
    """Find the largest t in [0,1] such that the blended sequence has peak <= M * (1+eps)."""
    S = np.sum(a)
    g_normalized = g * S / max(np.sum(g), 1e-12)
    
    best_t = 0.0
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Try to find the largest t where peak doesn't increase
    lo, hi = 0.0, 1.0
    for _ in range(25):
        mid = (lo + hi) / 2.0
        b = (1 - mid) * a + mid * g_normalized
        b = np.clip(b, 0.0, None)
        b = normalize_seq(b, S)
        conv_max = np.max(fft_convolve(b, b))
        if conv_max <= best_conv_max + 1e-10:
            lo = mid
        else:
            hi = mid
    
    if lo > 1e-6:
        b = (1 - lo) * a + lo * g_normalized
        b = np.clip(b, 0.0, None)
        b = normalize_seq(b, S)
        return b, lo
    return a, 0.0

def subgradient_refine(a, time_limit, verbose=False):
    """
    Subgradient-based refinement: directly target the peak lag(s).
    Uses weighted subgradient from all near-peak lags.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    lr = 0.5 / n
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a, a)
        current_peak = np.max(conv)
        
        # Find near-peak lags (within 5% of max)
        threshold = current_peak * 0.95
        active_lags = np.where(conv >= threshold)[0]
        
        # Compute subgradient: weighted average of gradients at active lags
        weights = np.zeros(len(active_lags))
        for i, lag in enumerate(active_lags):
            weights[i] = conv[lag] ** 2
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for w, k in zip(weights, active_lags):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * w * a[k - j]
        
        # Step: decrease in gradient direction, maintain non-negativity and sum
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * S / s
        
        new_peak = np.max(fft_convolve(a_new, a_new))
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 5.0 / n)
        else:
            stall += 1
            lr *= 0.95
            if stall > 100:
                lr = 0.5 / n
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a *= np.exp(rng.normal(0, 0.02, n))
                a = np.clip(a, 0, None)
                a = normalize_seq(a, S)
                best_a = a.copy()
                best_peak = np.max(fft_convolve(a, a))
                stall = 0
        
        if verbose and stall % 500 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Subgrad: score={score:.6f}, lr={lr:.5f}, stall={stall}")
    
    return best_a

def interpolate_sequence(a, n_new):
    """Interpolate a sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def make_asymmetric_init(n, rng, init_type='ramp'):
    """Create asymmetric initial sequences."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if init_type == 'ramp':
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * (2 * t - 1)
    elif init_type == 'asym_sine':
        f = 1 + rng.integers(0, 3)
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.sin(np.pi * t) * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi))
    elif init_type == 'step':
        x0 = 0.3 + 0.4 * rng.random()
        a = np.where(t < x0, 1.5, 0.5)
    elif init_type == 'poly':
        c = rng.normal(0, 0.3, 3)
        a = 1.0 + c[0] * t + c[1] * t**2 + c[2] * t**3
    else:
        a = 1.0 + 0.3 * (2 * t - 1)
    
    return np.clip(a, 0.01, None)

            t_limit = min(remaining, 200.0)
            print(f"  Final LP, time={t_limit:.0f}s")
            result = lp_refine_improved(best_seq, t_limit, verbose=True, allow_increase=False)
            if update_best(result):
                print(f"  *** Final LP: score={best_score:.6f} ***")
    
    # Phase 7: Coordinate descent polish
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            result = coordinate_descent(best_seq, min(remaining, 100.0), verbose=False)
            if update_best(result):
                print(f"  *** Final coord: score={best_score:.6f} ***")
    
    # Final answer
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq)} ===")
    
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"  evaluate_sequence confirms: {final_score:.6f}")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
