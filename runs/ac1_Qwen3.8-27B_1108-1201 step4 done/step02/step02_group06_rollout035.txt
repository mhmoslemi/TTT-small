```python
        a[j] = a_j_new
        a[i] = a_i_new

            best_a = a.copy()

        # Actually, let's undo and try a line search
        a[i] = a_i_new + move  # restore
        a[j] = a_j_new - move  # restore

        best_local_score = np.inf
        best_local_t = 0

            a_temp[j] += move

                best_local_score = s
                best_local_t = t_try
                best_local_a = a_temp.copy()

        if best_local_score < best_score:
            best_score = best_local_score
            best_a = a.copy()
    
    return best_a, best_score
```

Actually, let me reconsider the problem. The score is:
`score = 2*n*max(conv(a,a))/(sum(a))^2`

For `n=256`, if we achieve `score = 1.5`, then `max(conv)/(sum)^2 = 1.5/(2*256) = 0.00293`.

If `sum(a) = 1`, then `max(conv) = 0.00293`.
The "average" value of the convolution (sum of all conv values)/(2n-1) = `(sum(a))^2/(2n-1) = 1/511 ≈ 0.00196`.

So the peak is only about 1.5x the average. That's a very flat autocorrelation!

For comparison, a white noise sequence would have `max(conv) ≈ (sum(a))^2 / n * (some factor)`... actually for random `a_i ~ Uniform(0,1)` with `sum = 1`, `a_i ≈ 1/n`, and `conv[n-1] ≈ sum a_i^2 ≈ n*(1/n)^2 = 1/n`. So `max(conv) ≈ c/n` for some constant c, giving score ≈ `2n*c/n = 2c`. For truly random, c is around 1-2, giving score around 2-4.

To get score 1.5, we need the peak to be much below what random gives. This requires a very specific structure.

Let me think about what structures achieve low peak:
- **Barker sequences** (for complex): have ideal autocorrelation properties, but they're complex-valued
- **Golay complementary sequences**: their sum of autocorrelations is zero
- **MAM sequences (Minimum Amplitude Modulation)**: designed to have low peak sidelobe

For real non-negative sequences, the problem is harder because we can't use negative values to cancel.

Let me think about the "greedy" approach:
1. Start with a uniform sequence
2. The peak is at the center: `conv[n-1] = sum a_i * a_{n-1-i}`
3. To reduce this, we want to make `a_i` and `a_{n-1-i}` "anti-correlated" (one high, one low)
4. But this might increase other lags

A natural structure: make the sequence "skewed" - high on one side, low on the other. For example:
`a_i = exp(alpha * i / n)` for some alpha > 0.

Let's check: `a_i = e^(alpha*i/n)`, `sum = (e^alpha - 1)/alpha * 1/n * ... `

Actually, let me just try different parametric families and see which gives the best score:

Family 1: `a_i = 1 + alpha * (2*i/n - 1)` (linear sweep)

For the local optimization, let me think about what's most effective:

**Option A: Frank-Wolfe with away steps** (as described above)
- Pros: Simple, efficient, guaranteed to stay in simplex
- Cons: Slow convergence (O(1/t))

**Option B: Projected subgradient**
- Compute subgradient of `max conv`
- Take a step in the negative subgradient direction
- Project back onto simplex
- Pros: Can use adaptive step sizes, potentially faster
- Cons: Oscillations, need careful tuning

**Option C: Multiplicative weights / mirror descent**
- Update: `a_i <- a_i * exp(-eta * g_i)` then renormalize
- This is mirror descent with entropy regularizer
- Pros: Naturally handles non-negativity, can be fast
- Cons: Same subgradient issues

**Option D: Coordinate descent on individual elements**
- For each coordinate, find the best value that minimizes the objective
- Pros: Simple, guaranteed improvement at each step
- Cons: Very slow for high dimensions

**Option E: The LP approach from the previous program**
- Pros: Very effective (achieved 1.511)
- Cons: Slow (each LP takes seconds)

I think the best strategy is:
1. Parametric search to get a good initial point (fast)
2. Frank-Wolfe / mirror descent to refine (medium speed, many iterations)
3. A few LP iterations to polish (slow but very effective per iteration)
4. Maybe a final round of simulated annealing

```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Uses parametric initialization, mirror descent (multiplicative weights),
Frank-Wolfe with away steps, LP polishing, and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_subgradient(a, active_lags):
    """Compute subgradient of max_k conv(a,a)_k at point a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # g[lo:hi+1] += a[k-hi : k-lo+1][::-1]
            seg = a[k-hi:k-lo+1][::-1].copy()
            g[lo:hi+1] += seg
    return g * (2.0 / len(active_lags))

def get_active_lags(conv, tol=0.01, max_lags=30):
    """Get the set of active lags (where conv is close to its max)."""
    M = np.max(conv)
    active = np.where(conv >= M * (1 - tol))[0]
    if len(active) > max_lags:
        # Keep only the top max_lags
        idx = np.argsort(conv)[-max_lags:]
        active = idx
    return active

def mirror_descent(a_init, time_limit, verbose=False):
    """Mirror descent with multiplicative weight updates."""
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    best_score = local_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    eta = 0.5  # step size
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        M = np.max(conv)
        active = get_active_lags(conv, tol=0.02, max_lags=20)
        g = compute_subgradient(a, active)
        
        # Mirror descent update: a_i <- a_i * exp(-eta * g_i)
        # Use log for numerical stability
        log_a = np.log(a + 1e-30)
        log_a = log_a - eta * g
        # Normalize
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / np.sum(a_new)
        
        # Check improvement
        score = local_score(a_new)
        iteration += 1
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            eta *= 0.95
            a = a_new  # still move forward to explore
        
        if eta < 0.001:
            eta = 0.1  # reset if too small
        
    return best_a, best_score

def fw_away_steps(a_init, time_limit, verbose=False):
    """Frank-Wolfe with away steps."""
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    best_score = local_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        M = np.max(conv)
        active = get_active_lags(conv, tol=0.01, max_lags=15)
        g = compute_subgradient(a, active)
        
        # Find the best coordinate to add mass
        j = np.argmin(g)
        # Find the worst coordinate to remove mass
        i = np.argmax(g)
        
        if i == j:
            break
            
        # Try several step sizes with away step
        best_local = best_score
        best_local_a = a.copy()
        for frac in [0.5, 0.3, 0.1, 0.05]:
            move = frac * a[i]
            if move < 1e-10:
                continue
            a_temp = a.copy()
            a_temp[i] -= move
            a_temp[j] += move
            s = local_score(a_temp)
            if s < best_local:
                best_local = s
                best_local_a = a_temp
        
        if best_local < best_score:
            best_score = best_local
            best_a = best_local_a.copy()
            a = best_local_a
            stagnant = 0
        else:
            # Try without away step (just add to j, distribute from all others)
            a_temp = a.copy()
            # Add small mass to j, take from all others proportionally
            delta = 0.01 / n
            a_temp[j] += delta
            others = np.delete(np.arange(n), j)
            a_temp[others] -= delta * a_temp[others] / a_temp[others].sum()
            a_temp = np.clip(a_temp, 0, None)
            a_temp = a_temp / a_temp.sum()
            s = local_score(a_temp)
            if s < best_score:
                best_score = s
                best_a = a_temp.copy()
                a = a_temp
                stagnant = 0
            else:
                a = best_local_a  # take the best we found
                stagnant += 1
                if stagnant > 100:
                    # Restart from best
                    a = best_a.copy()
                    stagnant = 0
        
        iteration += 1
        if verbose and iteration % 500 == 0:
            print(f"    FW iter={iteration}, score={local_score(a):.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def lp_polish(f_init, time_limit):
    """LP-based polishing (same idea as previous program but focused)."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        # Build constraint matrix for LP
        # We want: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0
        # conv(f,g)_k = sum_j f_j * g_{k-j}
        # This is a linear constraint in g
        
        A = np.zeros((2*n-1, n))
        for j in range(n):
            # g_j contributes to conv[k] for k = j, j+1, ..., j+n-1
            # conv[k] gets f[k-j] * g_j
            lo = j
            hi = min(2*n-2, j+n-1)
            A[lo:hi+1, j] = f[0:hi-lo+1]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True, 'time_limit': 20})
        
        if not result.success:
            break
        
        g = result.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale to match sum
        g_rescaled = g * (sum_f / sum_g)
        
        # Blend
        t_blend = 0.3
        f_new = (1 - t_blend) * f + t_blend * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        f_new = f_new * (sum_f / np.sum(f_new))
        
        score_new = local_score(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score

def parametric_init(n, rng):
    """Try various parametric families and return the best."""
    t = np.arange(n) / n
    best_score = np.inf
    best_a = np.ones(n) / n
    
    # Linear
    for alpha in np.linspace(-3, 3, 21):
        a = np.clip(1 + alpha * (2*t - 1), 0.001, None)
        s = local_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    # Exponential  
    for alpha in np.linspace(-4, 4, 33):
        a = np.exp(alpha * t)
        s = local_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    # Cosine with various frequencies
    for k in range(1, min(20, n//8+1)):
        for alpha in np.linspace(-0.8, 0.8, 17):
            a = np.clip(1 + alpha * np.cos(2*np.pi*k*t + rng.uniform(0, 2*np.pi)), 0.001, None)
            s = local_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
    
    # Quadratic
    for alpha in np.linspace(-2, 2, 17):
        for beta in np.linspace(-2, 2, 17):
            a = np.clip(1 + alpha*(2*t-1) + beta*(2*t-1)**2, 0.001, None)
            s = local_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
    
    return best_a, best_score

def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    
    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-directed perturbation
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            # Find contributing elements
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            # Reduce top contributors, increase bottom
            top_k = max(3, n // 8)
            sorted_idx = np.argsort(contribs)
            reduce_idx = sorted_idx[-top_k:]
            increase_idx = sorted_idx[:top_k]
            candidate[reduce_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            candidate[increase_idx] *= np.exp(rng.normal(0.04, 0.015, size=top_k))
        elif r < 0.7:
            # Block perturbation
            width = int(rng.integers(2, max(3, n // 8)))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.85:
            # Single element
            idx = int(rng.integers(0, n))
            candidate[idx] *= np.exp(rng.normal(0, 0.05))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            temperature *= 0.999
            continue
        candidate = candidate * (sum_f / cand_sum)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score
        
        temperature *= 0.9998
    
    return best_f, best_score
```

"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def annealed_grad_descent(a_init, time_limit, tau_start=None, tau_end=None, verbose=False):
    """
    Gradient descent on the smoothed objective f_tau(a) = tau * log(sum_k exp(conv_k/tau)).
    Uses Adam optimizer with adaptive learning rate.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()

    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.2 / n

    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    lr = 0.5

    best_score = local_score(a)
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        conv = fft_convolve(a, a)
        max_conv = np.max(conv)
        shifted = (conv - max_conv) / tau
        exp_shifted = np.exp(shifted)
        p = exp_shifted / np.sum(exp_shifted)

        grad = np.zeros(n)

        if grad_scale > 1e-10:
            grad = grad / grad_scale

        step += 1

        v = beta2 * v + (1 - beta2) * (grad ** 2)

        v_hat = v / (1 - beta2 ** step)
        lr_eff = lr / (1 + 0.005 * step ** 0.5)

def lp_step(f, M):
    """Single LP step: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        A[lo:hi + 1, j] = f[0:hi - lo + 1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 15})
    if result.success:
        return result.x
    return None

        # Decide: LP or gradient based on time
        # Use LP for longer stretches, gradient for shorter bursts
        if lp_count < 15 and remaining > 60:
            # Do an LP step
            conv = fft_convolve(f, f)
            M = np.max(conv)
            g = lp_step(f, M)
            if g is None:
                break
            sum_g = np.sum(g)
            if sum_g < 1e-10:
                break
            g_rescaled = g * (sum_f / sum_g)

            # Adaptive blend: stronger blend when LP solution is very different
            diff = np.sum(np.abs(g_rescaled - f)) / (sum_f + 1e-10)
            t_blend = min(0.5, 0.2 + 0.3 * diff)

            f_new = (1 - t_blend) * f + t_blend * g_rescaled
            f_new = np.clip(f_new, 1e-10, None)
            f_new = f_new * (sum_f / np.sum(f_new))

            score_new = local_score(f_new)
            lp_count += 1
            f = f_new

            if score_new < best_score:
                best_score = score_new
                best_f = f_new.copy()

            if verbose:
                print(f"    LP {lp_count}: score={score_new:.6f}, best={best_score:.6f}, "
                      f"blend={t_blend:.3f}, elapsed={time.monotonic()-t0:.0f}s")
        else:
            # Do a gradient burst (60 seconds worth, or less if less time)
            grad_time = min(60.0, remaining - 20)
            if grad_time < 10:
                break
            f_grad, s_grad = annealed_grad_descent(
                f, grad_time,
                tau_start=3.0 / n, tau_end=0.3 / n,
                verbose=verbose
            )
            f = f_grad
            grad_count += 1
            if s_grad < best_score:
                best_score = s_grad
                best_f = f_grad.copy()

    return best_f, best_score


def sa_polish(f_init, time_limit, rng):
    """Simulated annealing with peak-directed perturbations."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.003
    iters = 0

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.45:
            # Peak-directed: reduce contributors to peak, boost non-contributors
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(4, n // 10)
            sorted_idx = np.argsort(contribs)
            reduce_set = sorted_idx[-top_k:]
            increase_set = sorted_idx[:top_k]
            candidate[reduce_set] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            candidate[increase_set] *= np.exp(rng.normal(0.04, 0.015, size=top_k))
        elif r < 0.7:
            # Block perturbation
            width = int(rng.integers(2, max(3, n // 6)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.85:
            # Dual block (two blocks)
            for _ in range(2):
                width = int(rng.integers(1, max(2, n // 10)))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        else:
            # Global small
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))

        candidate = np.clip(candidate, 0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            temperature *= 0.999
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iters += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= 0.9997

    return best_f, best_score
