Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterated local search with
LP-based directional optimization, simulated annealing for escaping local
optima, and subgradient refinement. Employs multi-length search and diverse
initializations to explore the non-convex landscape.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                  'dual_feasibility_tolerance': 1e-10,
                                                  'primal_feasibility_tolerance': 1e-10})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(a_init, time_limit, blend_init=0.5, blend_min=0.005):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    max_blend_halves = 20

    halve_count = 0
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check if g is very close to a (no useful direction)
        if np.abs(g - a).max() < 1e-10:
            break

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 8:

            stall = 0
            halve_count += 1
            if halve_count > max_blend_halves:

    return best_a, best_obj

    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterated local search (ILS)
with LP-based directional optimization, simulated annealing perturbations,
subgradient refinement, and multi-length exploration with diverse initializations.
The ILS framework repeatedly applies LP local search from perturbed starting
points to escape local minima in the non-convex landscape.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """2*n*max(conv(a,a)) assuming sum(a)=1."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] for valid indices."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[k - je:k - js + 1][::-1]
    return A


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def perturb_block_swap(a, rng, n_blocks=5):
    """Swap mass between random blocks."""
    n = len(a)
    result = a.copy()
    block_size = max(1, n // n_blocks)
    for _ in range(n_blocks):
        i = int(rng.integers(n - block_size))
        j = int(rng.integers(n - block_size))
        if abs(i - j) < block_size:
            continue
        bi = slice(i, i + block_size)
        bj = slice(j, j + block_size)
        mass_i = result[bi].sum()
        mass_j = result[bj].sum()
        total = mass_i + mass_j
        if total < 1e-15:
            continue
        split = 0.5 + rng.standard_normal() * 0.3
        split = max(0.05, min(0.95, split))
        result[bi] = split * total / block_size
        result[bj] = (1.0 - split) * total / block_size
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return project_simplex(result)
