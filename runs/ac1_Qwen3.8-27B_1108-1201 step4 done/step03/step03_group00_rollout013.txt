import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_optimize(f0, time_limit, top_k=None, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    """Adam-based optimization of max(conv(f,f)) subject to sum(f)=1, f>=0."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(4, n // 16)
    
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    t = 0
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Softmax weighting
        temp = 0.01 * best_obj + 1e-10
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Adam update
        m = b1 * m + (1 - b1) * sg
        v = b2 * v + (1 - b2) * sg ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            # Don't update f if no improvement (or use a smaller step)
            # Actually for Adam we should still update f
            f = f_new
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr * 0.5
            if lr < 0.0001:
                break
    
    return best_f, best_obj

import numpy as np

def fast_convolve(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, top_k, temp):
    """Compute weighted subgradient of max(conv(f,f)) using top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    if temp <= 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        contrib = f[k - je:k - js + 1][::-1]
        sg[js:je + 1] += 2.0 * w[i] * contrib
    return sg, conv


def subgradient_optimize(f0, time_limit, top_k=16, base_lr=0.008, base_temp=0.03,
                         anneal_rate=1.002, noise_scale=0.003):
    """
    Subgradient descent with adaptive multi-peak weighting and annealing.
    Minimizes max(conv(f,f)) subject to sum(f)=1, f>=0.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv0 = fast_convolve(f)
    best_obj = conv0.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = base_lr
    temp = base_temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, conv = compute_subgradient(f, top_k, temp)

        # Normalize subgradient for stable stepping
        sg_norm = np.sqrt(np.sum(sg * sg))
        if sg_norm > 1e-15:
            step = lr * sg / sg_norm
            f_new = f - step
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = f
            else:
                f_new = f_new / sn

            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1

        # Anneal temperature
        temp *= anneal_rate

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.9
            if no_improve % 400 == 0:
                # Perturb to escape local minimum
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * noise_scale
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                no_improve = 0
                lr = base_lr * 0.7
                temp = base_temp

        if lr < 1e-10:
            break

    return best_f, best_obj


def adam_subgrad_optimize(f0, time_limit, top_k=20, base_lr=0.003,
                          b1=0.9, b2=0.999, eps=1e-8, anneal_rate=1.004):
    """Adam-based subgradient optimization with adaptive peak weighting."""
    f = f0.astype(np.float64).copy()

        f = f / s

    t = 0

    conv0 = fast_convolve(f)
    best_obj = conv0.max()

    temp = 0.02

    while time.monotonic() - t0 < time_limit:
        t += 1

        m = b1 * m + (1.0 - b1) * sg

        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0.0)

            f_new = f_new / sn

def coord_descent_polish(f0, time_limit, n_cands=5):

    passed = 0

    while time.monotonic() - t0 < time_limit:
        passed += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Generate candidates: fractions of old value
            fracs = [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5]
            candidates = [old_val * fr for fr in fracs]

            best_c = old_val
            best_v = conv.max()

            for c in candidates:
                if c < 0:
                    continue
                f[j] = c
                # Quick eval: only check if this could improve

                conv_new = fast_convolve(f)
                v_new = conv_new.max()
                if v_new < best_v:
                    best_v = v_new
                    best_c = c

            f[j] = best_c
            if best_c != old_val:
                improved = True
                conv = fast_convolve(f)

        # Renormalize
        sn = f.sum()
        if sn > 1e-15:
            f = f / sn
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()

        if not improved:
            break

    return best_f, best_obj

    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Two peaks
    two_peak = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
               np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits.append(two_peak / two_peak.sum())

    # Sine wave
    sine = np.abs(np.sin(2 * np.pi * x / n)) + 0.05
    inits.append(sine / sine.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Normalize all
    inits = [a / a.sum() for a in inits if a.sum() > 1e-10]
    return inits


def multi_peak_flatten(f0, time_limit, n_peaks=5, step_frac=0.005):
    """
    Targeted peak flattening: identify dominant convolution peaks and
    reduce them by moving mass from contributing positions.
    """
    f = f0.astype(np.float64).copy()
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_max = conv.max()

        if current_max < best_obj - 1e-15:
            best_obj = current_max
            best_f = f.copy()

        # Find top peaks
        peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        # For the highest peak, find dominant contributing pair
        k = int(peak_indices[0])
        js = max(0, k - n + 1)
        je = min(n - 1, k)

        # Find pair with max product
        max_prod = -1
        best_i = -1
        best_j = -1
        for i in range(js, je + 1):
            j = k - i
            prod = f[i] * f[j]
            if prod > max_prod:
                max_prod = prod
                best_i = i
                best_j = j

        if best_i < 0 or max_prod < 1e-15:
            break

        # Determine source (larger of the pair) and destination (smallest)
        if best_i == best_j:
            src = best_i
        else:
            src = best_i if f[best_i] > f[best_j] else best_j

        # Destination: position with minimum contribution to all peaks
        # Simple heuristic: position with smallest value that doesn't contribute to top peaks
        peak_set = set(int(p) for p in peak_indices)
        low_candidates = []
        for pos in range(n):
            # Check if this position contributes significantly to any peak
            max_contrib = 0
            for pk in peak_indices[:3]:
                pk = int(pk)
                if 0 <= pk - pos < n:
                    max_contrib = max(max_contrib, f[pos] * f[pk - pos])
            if max_contrib < 0.3 * max_prod:
                low_candidates.append((f[pos], pos))

        if low_candidates:
            low_candidates.sort()
            dst = low_candidates[0][1]
        else:
            dst = int(np.argmin(f))

        if src == dst:
            continue

        # Move mass
        eps = step_frac * f[src]
        f[src] -= eps
        f[dst] += eps
        f = np.maximum(f, 0)
        f = f / f.sum()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining subgradient methods,
    coordinate descent, and targeted peak flattening.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick exploration (300s) ===
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = min(300, deadline - time.monotonic() - 1200)
    if t_phase1 < 30:
        t_phase1 = 30

    if parent is not None:
        # Optimize parent quickly first
        t_alloc = min(80, t_phase1 * 0.5)
        print(f"  Quick subgrad on parent (n={len(parent)}), t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(parent, time_limit=t_alloc, top_k=16,
                                        base_lr=0.006, base_temp=0.02)
        try_seq(a_opt, "p1_parent")

        # Also try Adam on parent
        t_alloc = min(80, t_phase1 * 0.3)
        rem = deadline - time.monotonic() - 1100
        if rem > 30:
            print(f"  Quick Adam on parent, t={t_alloc:.0f}s")
            a_opt, _ = adam_subgrad_optimize(parent, time_limit=min(t_alloc, rem),
                                             top_k=20, base_lr=0.003)
            try_seq(a_opt, "p1_adam_parent")

    # Try other lengths
    other_lengths = [256, 320, 448, 512, 200, 160]
    rem = deadline - time.monotonic() - 1100
    for n_try in other_lengths:
        if rem < 40:
            break
        inits = generate_inits(n_try, parent, rng)
        t_per = max(15, min(30, rem / max(1, len(other_lengths))))
        rem = deadline - time.monotonic() - 1000

        if rem < 20:
            break

        best_for_n = None
        best_for_n_obj = float('inf')

        for i, f_init in enumerate(inits[:4]):
            if rem < 15:
                break
            t_alloc = min(t_per, rem - 5)
            if t_alloc < 10:
                break
            a_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc,
                                              top_k=max(8, n_try // 24), base_lr=0.007)
            if obj < best_for_n_obj:
                best_for_n_obj = obj
                best_for_n = a_opt
            rem = deadline - time.monotonic() - 900

        if best_for_n is not None:
            try_seq(best_for_n, f"p1_n{n_try}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep optimization (900s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300

        # Long subgradient run
        t_alloc = min(300, rem * 0.5)
        print(f"  Deep subgrad (n={n_best}), t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, top_k=24,
                                        base_lr=0.004, base_temp=0.015,
                                        anneal_rate=1.003, noise_scale=0.002)

        # Adam run

        if rem > 50:
            t_alloc = min(250, rem * 0.6)
            print(f"  Deep Adam (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = adam_subgrad_optimize(best_seq, time_limit=t_alloc,
                                             top_k=32, base_lr=0.002,
                                             b1=0.92, b2=0.995, anneal_rate=1.005)
            try_seq(a_opt, "p2_adam")

        # Peak flattening
        rem = deadline - time.monotonic() - 100
        if rem > 50:
            t_alloc = min(200, rem)
            print(f"  Peak flatten (n={n_best}), t={t_alloc:.0f}s")

    # === Phase 3: Fine polish (400s) ===
    print(f"\n=== Phase 3: Fine Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100

        # Local search with very small steps
        if rem > 50:
            t_alloc = min(150, rem * 0.5)
            print(f"  Local search (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = local_search_polish(best_seq, time_limit=t_alloc,
                                           step_size=0.001, n_trials=50)
            try_seq(a_opt, "p3_local")

        # Coordinate descent (if fast enough)
        rem = deadline - time.monotonic() - 50
        if rem > 60 and n_best <= 500:
            t_alloc = min(120, rem)
            print(f"  Coord descent (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p3_coord")

        # Another subgrad round on the polished result
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_alloc = min(80, rem)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, top_k=32,
                                            base_lr=0.002, base_temp=0.008)
            try_seq(a_opt, "p3_subgrad2")

    print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Final micro-optimization ===
    print(f"\n=== Phase 4: Final Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10

        if rem > 20:
            # Very fine subgradient
            t_alloc = min(40, rem * 0.6)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, top_k=40,
                                            base_lr=0.001, base_temp=0.005,
                                            anneal_rate=1.001, noise_scale=0.001)
            try_seq(a_opt, "p4_fine")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            for _ in range(5):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                rng2 = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                f_pert = project_simplex(best_seq + rng2.standard_normal(n_best) * 0.0005)
                a_opt, _ = subgradient_optimize(f_pert, time_limit=min(rem, 10),
                                                top_k=40, base_lr=0.0005, base_temp=0.003)
                try_seq(a_opt, "p4_perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
