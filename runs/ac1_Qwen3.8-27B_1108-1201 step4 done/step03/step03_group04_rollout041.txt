"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines subgradient descent with adaptive temperature, Polyak step size,
momentum, and coordinated local search. Explores multiple sequence lengths
and uses the parent sequence for warm-starting.
"""

import numpy as np
import time

def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def subgradient_descent(a0, time_limit, verbose=True, verbose_every=1000, 
                        lr_init=0.01, momentum_beta=0.9, top_k_frac=0.1):
    """
    Subgradient descent on the simplex with:
    - Multi-peak Softmax weighting
    - Adaptive temperature (decreasing)
    - Momentum
    - Backtracking line search
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        conv_max = conv.max()
        cur_obj = 2.0 * n * conv_max
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Multi-peak subgradient
        top_k = max(3, int(n * top_k_frac))

def local_peak_reduction(a0, time_limit, verbose=True):
    """
    Local search that explicitly targets the top peaks in the convolution.
    For the peak at lag k, try to reduce a_j * a_{k-j} for dominant pairs.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        conv_max = conv.max()
        cur_obj = 2.0 * n * conv_max
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= max_stall:
            # Restart with perturbation
            a = project_simplex(a + 0.002 * np.random.default_rng(it).standard_normal(n))
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        # Get top peaks
        top_k = 5
        peaks = np.argsort(conv)[::-1][:top_k]
        
        # For each peak, try to reduce it
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            
            # Compute contributions: a[j] * a[k-j] for j in [j_start, j_end]
            idx = np.arange(j_start, j_end + 1)
            contributions = a[idx] * a[k - idx]
            
            # Find the dominant pair
            dom_local = np.argmax(contributions)
            j_dom = idx[dom_local]
            other_dom = k - j_dom
            
            if j_dom == other_dom:
                continue  # can't split
                    
            # Try to transfer mass from the larger element to a "safe" position
            if a[j_dom] > a[other_dom]:
                src = j_dom
                dst_candidates = np.where((np.arange(n) != src) & (np.arange(n) != k - src))[0]
            else:
                src = other_dom
                dst_candidates = np.where((np.arange(n) != src) & (np.arange(n) != k - src))[0]
            
            if len(dst_candidates) == 0:
                continue
            
            # Try transferring a small amount to a random safe position
            transfer_amt = a[src] * 0.05
            if transfer_amt < 1e-10:
                continue
            
            for trial in range(3):
                dst = dst_candidates[np.random.randint(len(dst_candidates))]
                a_trial = a.copy()
                a_trial[src] -= transfer_amt
                a_trial[dst] += transfer_amt
                if a_trial[src] < 0:
                    continue
                a_trial = project_simplex(a_trial)
                
                conv_trial = fast_convolve(a_trial)
                obj_trial = 2.0 * n * conv_trial.max()
                
                if obj_trial < cur_obj - 1e-14:
                    a = a_trial
                    break
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPR it={it}, best={best_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def harmonic_init(n, rng):
    """Initialize with a harmonic-like (smooth oscillation) pattern."""
    x = np.arange(n, dtype=np.float64)
    # Sum of a few sine waves with positive values
    a = 1.0
    for k in range(1, 4):
        a += 0.5 * np.sin(2 * np.pi * k * x / n + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp_up'] = r.copy()
    inits['ramp_down'] = r[::-1].copy()
    
    # Two-sided ramp (V-shape)
    v = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0
    v /= v.sum()
    inits['v_shape'] = v.copy()
    
    # Inverted V (peak in center)
    iv = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0
    iv = iv.max() - iv + 1
    iv /= iv.sum()
    inits['inv_v'] = iv.copy()
    
    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # Edge-concentrated
    edge = max(1, n // 6)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Harmonic
    inits['harmonic'] = harmonic_init(n, rng)
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random smooth (low-pass filtered random)
    r_smooth = rng.standard_normal(n)
    # Simple moving average
    kernel = np.ones(7) / 7
    r_smooth = np.convolve(r_smooth, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0)
    if r_smooth.sum() > 1e-10:
        r_smooth /= r_smooth.sum()
    else:
        r_smooth = np.ones(n) / n
    inits['rand_smooth'] = r_smooth.copy()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits

"""
Multi-phase optimizer combining LP-based Frank-Wolfe, Toeplitz-based Newton steps,
and subgradient polishing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: warm-started LP oracle, exact 1D quadratic line search,
Toeplitz mean-field update as cheap direction, and targeted peak reduction.
"""

import numpy as np
import time

def fast_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_obj(a):
    """Compute 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_conv_matrix_sparse(a, n):
    """Build (2n-1) x n matrix A where (A @ b)[k] = conv(a,b)[k]."""
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_oracle(a, M, n, x0=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b."""
    from scipy.optimize import linprog
    A = build_conv_matrix_sparse(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    if x0 is not None and np.all(np.isfinite(x0)):
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 3.0, 'primal_start': x0})
    else:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 3.0})
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb > 1e-15:
            return b / sb
    return None

def line_search_exact(a, b, n, max_iter=50):
    """Exact 1D line search for t in [0,1] minimizing max_k conv((1-t)a+tb, (1-t)a+tb).
    The objective is a convex piecewise-quadratic in t."""
    n = len(a)
    conv_aa = fast_convolve(a)
    conv_ab = np.convolve(a, b)
    conv_bb = fast_convolve(b)
    
    best_t = 0.0
    best_val = (2.0 * n) * conv_aa.max()
    
    # The objective f(t) = 2n * max_k [(1-t)^2 conv_aa_k + 2t(1-t) conv_ab_k + t^2 conv_bb_k]
    # is convex in t. Use ternary search.
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = (1-m1)**2 * conv_aa + 2*m1*(1-m1) * conv_ab + m1**2 * conv_bb
        v2 = (1-m2)**2 * conv_aa + 2*m2*(1-m2) * conv_ab + m2**2 * conv_bb
        obj1 = 2.0 * n * v1.max()
        obj2 = 2.0 * n * v2.max()
        if obj1 < obj2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-10:
            break
    
    t_star = (lo + hi) / 2.0
    v_star = (1-t_star)**2 * conv_aa + 2*t_star*(1-t_star) * conv_ab + t_star**2 * conv_bb
    obj_star = 2.0 * n * v_star.max()
    
    if obj_star < best_val:
        return t_star, obj_star
    return 0.0, best_val

def toeplitz_direction(a, conv, n, temperature=0.01):
    """Compute a direction using the Toeplitz mean-field approximation.
    M = sum_k w_k * T_k where M_{j,p} = w_{j+p}.
    Solve M @ x = 1, project to simplex."""
    w = np.exp(conv / (temperature * conv.max() + 1e-15))
    w /= w.sum()
    
    # Build Toeplitz matrix M where M_{j,p} = w_{j+p}
    M = np.zeros((n, n))
    for j in range(n):
        for p in range(n):
            M[j, p] = w[j + p]
    
    # Solve M @ x = 1
    try:
        x = np.linalg.solve(M, np.ones(n))
    except np.linalg.LinAlgError:
        try:
            x = np.linalg.lstsq(M, np.ones(n), rcond=None)[0]
        except:
            return None
    
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-15:
        return None
    return x / s

def fpw_optimize(a0, time_limit, verbose=False, verbose_every=500, use_lp=True, use_toeplitz=False):
    """Frank-Wolfe with LP oracle and optional Toeplitz direction."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    x0_lp = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        got_direction = False
        
        # LP oracle
        if use_lp:
            b = lp_oracle(a, M, n, x0=x0_lp)
            if b is not None:
                t_star, obj_new = line_search_exact(a, b, n)
                if t_star > 1e-6:
                    a = (1 - t_star) * a + t_star * b
                    x0_lp = b.copy()
                    conv = fast_convolve(a)
                    M = conv.max()
                    cur_obj = 2.0 * n * M
                    if cur_obj < best_obj - 1e-14:
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    got_direction = True
        
        # Toeplitz direction as fallback or supplement
        if not got_direction and use_toeplitz:
            d = toeplitz_direction(a, conv, n)
            if d is not None:
                t_star, obj_new = line_search_exact(a, d, n)
                if t_star > 1e-6:
                    a = (1 - t_star) * a + t_star * d
                    conv = fast_convolve(a)
                    M = conv.max()
                    cur_obj = 2.0 * n * M
                    if cur_obj < best_obj - 1e-14:
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
        
        # Stall detection and restart
        if no_improve >= max_stall:
            rng_local = np.random.default_rng(it + 12345)
            noise_scale = 0.002 * (1.0 + no_improve / max_stall)
            a = project_simplex(a + rng_local.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            M = conv.max()
            x0_lp = a.copy()
            no_improve = 0
            max_stall = int(max(50, max_stall * 1.4))
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.91, top_k=10, 
                       temp_init=0.05, temp_decay=0.999):
    """Subgradient descent with multi-peak Softmax on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        conv_max = conv.max()
        cur_obj = 2.0 * n * conv_max
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Select top-k peaks
        peaks = np.argsort(conv)[::-1][:top_k]
        
        # Softmax weights
        w = np.exp((conv[peaks] - conv_max) / (temp * conv_max + 1e-15))
        w /= w.sum()
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            partner = k - idx
            g[idx] += 2.0 * w[i] * a[partner]
        
        # Update
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        v = momentum * v - lr * g / g_norm
        a_new = project_simplex(a + v)
        
        # Check and adapt
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new > best_obj * 1.005:
            lr = max(lr * 0.5, 1e-10)
            a_new = project_simplex(a - v * 0.5)
        else:
            a = a_new
            temp *= temp_decay
        
        it += 1
    
    return best_a, best_obj

def peak_pair_swap(a, n_peaks=8, n_trials=20, rng=None):
    """Try to reduce top peaks by swapping mass between coefficient pairs."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    improved = False
    
    peaks = np.argsort(conv)[::-1][:n_peaks]
    
    for _ in range(n_trials):
        k = int(peaks[np.random.randint(len(peaks))])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        contributions = a[idx] * a[k - idx]
        
        if contributions.max() < 1e-10:
            continue
        
        dom = idx[np.argmax(contributions)]
        other = k - dom
        if dom == other:
            continue
        
        # Transfer from larger to a random "safe" position
        src = dom if a[dom] > a[other] else other
        amt = a[src] * np.random.uniform(0.01, 0.08)
        if amt < 1e-10:
            continue
        
        # Choose destination carefully: avoid creating new peaks
        valid_dst = np.where(a < a.mean() * 1.5)[0]
        valid_dst = valid_dst[valid_dst != src]
        if len(valid_dst) == 0:
            continue
        dst = valid_dst[np.random.randint(len(valid_dst))]
        
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = project_simplex(a_trial)
        
        conv_trial = fast_convolve(a_trial)
        obj_trial = 2.0 * n * conv_trial.max()
        
        if obj_trial < cur_obj - 1e-14:
            a = a_trial
            cur_obj = obj_trial
            improved = True
    
    return a, improved

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp_up'] = r.copy()
    inits['ramp_down'] = r[::-1].copy()
    
    # V-shape (minimum in center)
    v = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0
    v /= v.sum()
    inits['v_shape'] = v.copy()
    
    # Edge-concentrated
    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.3*n))**2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # Asymmetric Gauss (shifted center)
    for frac in [0.3, 0.7]:
        g_asym = np.exp(-0.5 * ((x - frac*(n-1)) / (0.25*n))**2)
        g_asym /= g_asym.sum()
        inits[f'gauss_{int(frac*10)}'] = g_asym.copy()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)
    
    # Random smooth
    rs = rng.standard_normal(n)
    ks = np.ones(9) / 9
    rs = np.convolve(rs, ks, mode='same')
    rs = np.maximum(rs, 0)
    if rs.sum() > 1e-10:
        rs /= rs.sum()
    else:
        rs = np.ones(n) / n
    inits['rand_smooth'] = rs.copy()
    
    # Quasi-periodic
    freqs = rng.choice([3, 5, 7, 11, 13], size=2, replace=False)
    qp = 1.0
    for f in freqs:
        qp += 0.4 * np.sin(2*np.pi*f*x/n + rng.uniform(0, 2*np.pi))
    qp = np.maximum(qp, 0)
    if qp.sum() > 1e-10:
        qp /= qp.sum()
    else:
        qp = np.ones(n) / n
    inits['quasi_periodic'] = qp.copy()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits
