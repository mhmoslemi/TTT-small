LP-based iterative method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the bilinear structure: at each step, solve an LP to find a direction
that keeps conv(a, dir) bounded, then take a convex combination step.
Followed by gradient-based polishing with exact max.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft


def fast_convolve(a, b=None):
    """Linear convolution of a with b (or with itself if b is None) via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    B = fft(b, fft_size)
    return np.real(ifft(A * B, fft_size))[:size]


def obj_value(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp_direction(a, M):
    """
    Solve the LP: maximize sum(b) subject to b >= 0, conv(a,b)[k] <= M for all k.
    Returns the optimal b (normalized to sum=1).
    """
    n = len(a)
    # Build constraint matrix A_lp of shape (2n-1, n)
    # A_lp[k, j] = a[k-j] if 0 <= k-j < n else 0
    A_lp = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_lp[k, j] = a[idx]
    
    # Objective: minimize -sum(b) => c = -1 for all
    c = -np.ones(n)
    
    # Constraints: A_lp @ b <= M (scalar)
    # b >= 0 (handled by bounds)
    res = linprog(c, A_ub=A_lp, b_ub=np.ones(2*n-1)*M, bounds=(0, None), method='highs')
    
    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-10:
            return b / s
        else:
            return a.copy()
    else:
        return a.copy()

    if s < 1e-10:
        return np.inf
    return 2.0 * n * conv_max(a) / (s * s)


def build_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a, M):
    n = len(a)
    A_lp = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_lp, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-10:
            return b / s
    return a.copy()


def line_search_combo(a, b, n):
    best_t = 0.0
    best_obj = 2.0 * n * conv_max(a)
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]:
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = 2.0 * n * conv_max(a_new)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def lp_iterate(a, n_iters=500, deadline=None):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * conv_max(a)
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        conv = fast_convolve(a)
        M = conv.max()
        b = solve_lp_direction(a, M)
        if np.allclose(b, a, atol=1e-9):
            no_improve += 1
            if no_improve > 30:
                break
            continue
        t, obj_new = line_search_combo(a, b, n)
        if t > 0:
            a_new = (1 - t) * a + t * b
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
            else:
                a_new = a
            full_obj = 2.0 * n * conv_max(a_new)
            if full_obj < best_obj * (1 - 1e-10):
                best_obj = full_obj
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 60:
            noise = 0.005 * rng.standard_normal(n)
            a_p = a + noise
            a_p = np.maximum(a_p, 0)
            s = a_p.sum()
            if s > 1e-10:
                a = a_p / s
            no_improve = 0
    return best_a


def softmax_obj(x, n):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    return 2.0 * n * conv_max(a)


def softmax_jac(x, n):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    M = conv.max()
    tol = 1e-8 * M
    active = np.where(conv >= M - tol)[0]
    grad_a = np.zeros(n)
    for k in active:
        valid = np.arange(max(0, k - n + 1), min(n, k + 1))
        idx = k - valid
        grad_a[valid] += 2.0 * a[idx]
    grad_a /= len(active)
    c = np.dot(a, grad_a)
    return a * (grad_a - c)


def polish_lbfgsb(a, maxiter=300):
    n = len(a)
    a_pos = np.maximum(a, 1e-15)
    a_pos = a_pos / a_pos.sum()
    x0 = np.log(a_pos)
    res = minimize(softmax_obj, x0, jac=softmax_jac, args=(n,),
                   method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = res.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def smooth_max_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    shifted = beta * (conv - conv.max())
    shifted = np.clip(shifted, -500, 500)
    sm = conv.max() + np.log(np.exp(shifted).sum()) / beta
    return 2.0 * n * sm


def smooth_max_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    shifted = beta * (conv - conv.max())
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w = w / w.sum()
    grad_a = np.zeros(n)
    for k in range(len(w)):
        if w[k] > 1e-15:
            valid = np.arange(max(0, k - n + 1), min(n, k + 1))
            idx = k - valid
            grad_a[valid] += 2.0 * w[k] * a[idx]
    c = np.dot(a, grad_a)
    return a * (grad_a - c)


def polish_smooth(a, beta, maxiter=300):
    n = len(a)
    a_pos = np.maximum(a, 1e-15)
    a_pos = a_pos / a_pos.sum()
    x0 = np.log(a_pos)
    res = minimize(smooth_max_obj, x0, jac=smooth_max_jac, args=(n, beta),
                   method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = res.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def flat_spectrum_init(n, rng):
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def diverse_init(n, parent, rng):
    inits = {}
    inits['uniform'] = np.ones(n) / n
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    for sf in [0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        inits['gauss_%.2f' % sf] = g / g.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['flat_spec2'] = flat_spectrum_init(n, rng)
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: length=%d, score=%.6f" % (len(p), objective(p)))
    except Exception:
        pass

    def score_and_update(a, label=""):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0).astype(np.float64)
        s = a.sum()
        if s < 1e-8:
            return float('inf')
        a = a / s
        score = float(objective(a))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST: %.6f n=%d %s (t=%.0fs)" % (score, len(a), label, time.monotonic() - t_start))
        return score

    # Phase 1: LP iteration on parent length
    print("=== Phase 1: LP Iteration (n=256) ===")
    n_primary = 256
    if parent is not None:
        a_init = np.interp(np.linspace(0, 1, n_primary), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()
    else:
        a_init = flat_spectrum_init(n_primary, rng)

    score_and_update(a_init, "init")
    lp_deadline = min(t_start + 250.0, deadline - 800.0)
    a_lp = lp_iterate(a_init, n_iters=2000, deadline=lp_deadline)
    score_and_update(a_lp, "LP")

    # Phase 2: L-BFGS-B polish on primary length
    print("=== Phase 2: L-BFGS-B Polish (n=256) ===")
    p2_deadline = min(t_start + 500.0, deadline - 600.0)
    a_cur = a_lp.copy()

    # Smooth polish first (easier to converge)
    for beta in [40.0, 20.0, 10.0, 5.0]:
        if time.monotonic() > p2_deadline - 20:
            break
        a_cur = polish_smooth(a_cur, beta=beta, maxiter=200)
        score_and_update(a_cur, "smooth beta=%.0f" % beta)

    # Exact max polish
    for _ in range(3):
        if time.monotonic() > p2_deadline - 10:
            break
        a_cur = polish_lbfgsb(a_cur, maxiter=300)
        score_and_update(a_cur, "exact")

    # Phase 3: LP from polished solution, then re-polish
    print("=== Phase 3: LP Round 2 + Polish ===")
    p3_deadline = min(t_start + 700.0, deadline - 500.0)
    a_lp2 = lp_iterate(a_cur, n_iters=500, deadline=p3_deadline - 100)
    score_and_update(a_lp2, "LP2")
    a_lp2 = polish_smooth(a_lp2, beta=15.0, maxiter=200)
    a_lp2 = polish_lbfgsb(a_lp2, maxiter=300)
    score_and_update(a_lp2, "polish2")

    # Phase 4: Multi-length exploration
    print("=== Phase 4: Multi-Length Exploration ===")
    other_lengths = [192, 320, 384, 448, 512]

    for n_alt in other_lengths:
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(180.0, deadline - time.monotonic() - 100)
        if t_alloc < 60:
            break
        print("  Trying n=%d (budget %.0fs)" % (n_alt, t_alloc))

        inits = diverse_init(n_alt, best_sequence if best_sequence is not None else parent, rng)
        a_best_alt = None
        best_alt_score = float('inf')

        # Try top initializations
        priority = ['parent', 'flat_spec', 'flat_spec2', 'uniform', 'gauss_0.20', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]

        for name in init_names:
            if time.monotonic() > deadline - 120:
                break
            t_single = min(50.0, (deadline - time.monotonic() - 100) / max(1, len(init_names)))
            if t_single < 20:
                break
            a0 = inits[name]
            # Quick LP
            lp_dl = time.monotonic() + min(20.0, t_single * 0.5)
            a_opt = lp_iterate(a0, n_iters=200, deadline=lp_dl)
            # Quick polish
            if time.monotonic() < deadline - 80:
                a_opt = polish_smooth(a_opt, beta=20.0, maxiter=100)
            sc = score_and_update(a_opt, "n=%d %s" % (n_alt, name))
            if sc < best_alt_score:
                best_alt_score = sc
                a_best_alt = a_opt.copy()

        # If this length beat our best, do deeper optimization
        if a_best_alt is not None and best_alt_score < best_score * 1.01:
            if time.monotonic() < deadline - 60:
                a_best_alt = polish_lbfgsb(a_best_alt, maxiter=300)
                score_and_update(a_best_alt, "n=%d deep" % n_alt)

    # Phase 5: Final refinement on best
    print("=== Phase 5: Final Refinement ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20
        n_best = len(best_sequence)

        # Alternating LP + smooth polish
        a_fin = best_sequence.copy()
        for round in range(5):
            if remaining < 15:
                break
            # LP step
            lp_dl = time.monotonic() + max(10, remaining / 8)
            a_fin = lp_iterate(a_fin, n_iters=100, deadline=lp_dl)
            score_and_update(a_fin, "fin LP r%d" % round)
            remaining = deadline - time.monotonic() - 20
            if remaining < 10:
                break
            # Smooth polish
            a_fin = polish_smooth(a_fin, beta=8.0, maxiter=150)
            score_and_update(a_fin, "fin smooth r%d" % round)
            remaining = deadline - time.monotonic() - 20
            if remaining < 10:
                break

        # Final exact polish
        if remaining > 15:
            a_fin = polish_lbfgsb(a_fin, maxiter=200)
            score_and_update(a_fin, "fin exact")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = objective(best_sequence)

    # Final verification via evaluate_sequence
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("FINAL: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
