"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced subgradient descent
with adaptive peak weighting, LP-based peak shaving, coordinate descent,
and multi-length exploration with simulated annealing restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """Compute weighted subgradient based on top-K peaks."""
    n = len(a)
    # Select top-K peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weighting
    peak_vals = conv[peaks]
    if temperature > 0:
        scaled = (peak_vals - peak_vals.max()) / temperature
        w = np.exp(scaled)
        w /= w.sum()
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    
    # Compute weighted gradient
    sg = np.zeros(n)
    for idx, k in enumerate(peaks):
        if w[idx] < 1e-12:
            continue
        k = int(k)
        wi = 2.0 * w[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] += wi * a[k - j]
    
    return sg


def subgradient_descent(a_init, time_limit, lr_init=0.02, momentum=0.9,
                        top_k=30, temp=0.0005, decay=0.9999,
                        restart_every=200, restart_noise=0.005):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(a, conv, top_k, temp)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        lr *= decay
        
        if no_improve >= restart_every:
            rng = np.random.default_rng(iter_count)
            a = project_simplex(best_a + rng.standard_normal(n) * restart_noise)
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5
    
    return best_a, best_obj


def peak_shave_lp(a, conv, M, n, max_active=50):
    """
    LP to find direction that reduces peaks most uniformly.
    Minimize z s.t. grad_k^T d <= z for k in active set,
    sum(d) = 0, d >= -a.
    """
    # Select active lags (those near the peak)
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) > max_active:
        # Keep only the top max_active
        active_vals = conv[active]
        top_idx = np.argsort(active_vals)[::-1][:max_active]
        active = active[top_idx]
    
    if len(active) == 0:
        return None
    
    # Build the LP
    # Variables: d[0..n-1], z
    # Minimize z
    # Constraints:
    #   sum(d) = 0 (equality)
    #   grad_k^T d - z <= 0 for k in active (i.e., grad_k^T d <= z)
    #   d[i] >= -a[i] (i.e., -d[i] <= a[i])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize z
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_ub_rows = []
    b_ub_vals = []
    
    for k in active:
        k = int(k)
        row = np.zeros(n + 1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row[j] = 2.0 * a[k - j]
        row[n] = -1.0  # -z
        A_ub_rows.append(row)
        b_ub_vals.append(0.0)
    
    # d[i] >= -a[i] => -d[i] <= a[i]
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(a[i])
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * (n + 1)
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
        if result.success:
            return result.x[:n]
    except Exception:
        pass
    return None


def lp_direction_optimize(a_init, time_limit, n_peaks=30):
    """LP-based peak shaving optimization."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step_size = 0.1
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        d = peak_shave_lp(a, conv, M, n, max_active=n_peaks)
        if d is None:
            break
        
        # Take a step in direction d
        # d should decrease the peaks, so a + t*d should be better
        a_new = a + step_size * d
        a_new = project_simplex(a_new)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a.copy()
            stall = 0
            step_size = min(0.5, step_size * 1.1)
        else:
            stall += 1
            step_size *= 0.7
            if step_size < 0.001:
                # Reset with noise
                rng = np.random.default_rng(hash(tuple(a[:5])) % 2**31)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(a)
                step_size = 0.1
                stall = 0
                if stall > 20:
                    break
    
    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=50, step_frac=0.1):
    """Coordinate descent: move mass between pairs to reduce peak."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        M = conv.max()
        # Focus on coordinates that contribute to the peak
        peak_lags = np.where(conv >= M * 0.9)[0]
        
        for k in peak_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            
            for j in range(j_start, j_end + 1):
                if a[j] < 1e-8:
                    continue
                # Try moving some mass from j to j+1 or j-1
                for dj in [-1, 1]:
                    j2 = j + dj
                    if j2 < 0 or j2 >= n:
                        continue
                    delta = a[j] * step_frac
                    a_test = a.copy()
                    a_test[j] -= delta
                    a_test[j2] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        conv = conv_test
                        best_obj = obj_test
                        improved = True
        
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Ramp (monotonically increasing)
    ramp = (x + 1.0) ** 2
    inits['ramp'] = ramp / ramp.sum()
    
    # Inverted ramp
    inv_ramp = (n - x) ** 2
    inits['inv_ramp'] = inv_ramp / inv_ramp.sum()
    
    # Edge concentration
    edge = np.zeros(n)
    e = max(1, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sine-based
    sine = np.abs(np.sin(np.pi * x / n)) + 0.01
    inits['sine'] = sine / sine.sum()
    
    # Cosine-tapered
    cos_t = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits['cos_taper'] = cos_t / cos_t.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            if len(p) == n:
                inits['parent'] = p / p.sum()
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(p)), p)
                p_interp = np.maximum(p_interp, 0)
                ps = p_interp.sum()
                if ps > 1e-10:
                    inits['parent_interp'] = p_interp / ps
    
    return inits

1. Identify the peak lag k*.
2. The gradient at k* is `2 * a[k*-j]` for `j` in the valid range.
3. To reduce `(a*a)_{k*}`, we need to decrease the "alignment" of `a` with its flip at lag k*.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced subgradient descent
with backtracking line search, LP-based peak shaving, direct peak reduction,
and multi-length exploration with adaptive restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient over top-K peak lags."""
    n = len(a)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    pmax = peak_vals.max()
    if temperature > 0:
        scaled = (peak_vals - pmax) / temperature
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum > 1e-30:
            w /= wsum
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        wi = 2.0 * w[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] += wi * a[k - j]
    return sg


def subgradient_optimize(a_init, time_limit, lr_init=0.02, momentum=0.92,
                         top_k=40, temp=0.0003, decay=0.9998,

    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        sg = compute_weighted_subgradient(a, conv, top_k, temp)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        v = momentum * v - lr * sg
        if backtracking:
            accepted = False
            trial_lr = lr
            for bt in range(max_bt):
                a_candidate = project_simplex(a + trial_lr * v / max(lr, 1e-15))
                conv_cand = fast_convolve(a_candidate)
                obj_cand = 2.0 * n * conv_cand.max()

def peak_shave_lp(a, conv, max_active=25, time_limit_lp=3.0):
    """LP to find direction d minimizing max peak growth, sum(d)=0, d>=-a."""
    n = len(a)
    M = conv.max()
    threshold = M * 0.97
    active = np.where(conv >= threshold)[0]
    if len(active) > max_active:
        active_vals = conv[active]
        top_idx = np.argsort(active_vals)[::-1][:max_active]
        active = active[top_idx]
    if len(active) == 0:
        return None
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in active:
        k = int(k)
        row = np.zeros(n + 1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row[j] = 2.0 * a[k - j]
        row[n] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(a[i] + 1e-12)
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(None, None)] * (n + 1)
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
        if result.success:
            d = result.x[:n]
            d_norm = np.linalg.norm(d)
            if d_norm > 1e-15:

    except Exception:
        pass
    return None

                    best_move = (a_test.copy(), conv_test, obj_test)
        if best_move is not None:
            a = best_move[0]
            conv = best_move[1]
            best_obj = best_move[2]
            best_a = a.copy()
        else:
            break
    return best_a, best_obj


def coordinate_descent_fast(a_in, max_rounds=30, n_trials_per_coord=3):
    """Fast coordinate descent: try shifting mass to neighbors."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    for rnd in range(max_rounds):
        improved = False
        M = conv.max()
        peak_lags = np.where(conv >= M * 0.85)[0]
        for k in peak_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end - j_start + 1 > 50:
                idxs = np.random.default_rng(rnd * 31 + k).choice(
                    range(j_start, j_end + 1), size=50, replace=False)
            else:
                idxs = range(j_start, j_end + 1)
            for j in idxs:
                j = int(j)
                if a[j] < 1e-8:
                    continue
                for dj in [-1, 1, -2, 2]:
                    j2 = j + dj
                    if j2 < 0 or j2 >= n or j2 == j:
                        continue
                    for frac in [0.03, 0.08, 0.15]:
                        delta = a[j] * frac
                        if delta < 1e-10:
                            continue
                        a_test = a.copy()
                        a_test[j] -= delta
                        a_test[j2] += delta
                        a_test = np.maximum(a_test, 0.0)
                        conv_test = fast_convolve(a_test)
                        obj_test = 2.0 * n * conv_test.max()
                        if obj_test < best_obj - 1e-13:
                            a = a_test
                            conv = conv_test
                            best_obj = obj_test
                            improved = True
        if not improved:
            break
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    ramp = (x + 1.0) ** 2
    inits['ramp'] = ramp / ramp.sum()
    inv_ramp = (n - x + 1.0) ** 2
    inits['inv_ramp'] = inv_ramp / inv_ramp.sum()
    edge = np.zeros(n)
    e = max(1, n // 6)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    sine = np.abs(np.sin(np.pi * (x + 0.5) / n)) + 0.05
    inits['sine'] = sine / sine.sum()
    cos_t = 0.5 * (1 + np.cos(2 * np.pi * (x + 0.5) / n)) + 0.05
    inits['cos_taper'] = cos_t / cos_t.sum()
    # Power-law ramp
    pw = (x + 1.0) ** 4
    inits['power_ramp'] = pw / pw.sum()
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            if len(p) == n:
                inits['parent'] = p / p.sum()
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(p)), p)
                p_interp = np.maximum(p_interp, 0)
                ps = p_interp.sum()
                if ps > 1e-10:
                    inits['parent_interp'] = p_interp / ps
    return inits


def optimize_pipeline(a_init, time_limit):
    """Run full optimization pipeline: subgrad -> LP -> direct reduction -> CD."""
    a = a_init.copy()
    n = len(a)
    t0 = time.monotonic()
    # Phase 1: Subgradient
    t1 = (time_limit - t0) * 0.35
    if t1 > 3:
        a, _ = subgradient_optimize(a, t1, lr_init=0.02, momentum=0.92,
                                     top_k=50, temp=0.0003, decay=0.9998,
                                     restart_every=250, restart_noise=0.004)
    # Phase 2: LP peak shaving
    t2 = (time_limit - time.monotonic()) * 0.3
    if t2 > 5:
        a_lp, obj_lp = lp_peak_shave(a, t2, n_peaks=15, init_step=0.06)
        conv_a = fast_convolve(a)
        if obj_lp < 2.0 * n * conv_a.max():
            a = a_lp
    # Phase 3: Direct peak reduction
    t3 = (time_limit - time.monotonic()) * 0.2
    if t3 > 3:
        a_dr, obj_dr = direct_peak_reduction(a, t3, max_rounds=150)
        conv_a = fast_convolve(a)
        if obj_dr < 2.0 * n * conv_a.max():
            a = a_dr
    # Phase 4: Final subgradient with small steps
    t4 = (time_limit - time.monotonic()) * 0.3
    if t4 > 3:
        a, _ = subgradient_optimize(a, t4, lr_init=0.005, momentum=0.95,
                                     top_k=80, temp=0.0001, decay=0.9999,
                                     restart_every=200, restart_noise=0.002)
    # Phase 5: Coordinate descent
    t5 = time_limit - (time.monotonic() - t0)
    if t5 > 2:
        rounds = max(5, min(40, int(t5)))
        a = coordinate_descent_fast(a, max_rounds=rounds)
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: subgradient + LP peak shaving + direct reduction + CD.
    Explores multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    best_sequence = None
    best_score = float('inf')
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_score = par_score
            best_sequence = p.copy()
    except Exception:
        pass
    # Phase 1: Length and init exploration
    print("=== Phase 1: Exploration ===")
    lengths = [384, 448, 480, 512, 544, 576, 640, 768]
    if parent is not None:
        pl = len(parent)
        if pl not in lengths:
            lengths.append(pl)
        for dl in [-16, -8, 8, 16, 24, 32]:
            candidate = pl + dl
            if candidate >= 128 and candidate not in lengths:
                lengths.append(candidate)
    lengths = [l for l in lengths if 128 <= l <= 1024]
    lengths.sort()
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.25)
    phase1_end = time.monotonic() + phase1_budget
    results = []
    t_per = phase1_budget / max(1, len(lengths) * 2)
    for n in lengths:
        if time.monotonic() > phase1_end - 15:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_interp', 'ramp', 'inv_ramp', 'triangle',
                'uniform', 'power_ramp', 'bimodal', 'gauss', 'random']
        names = [nm for nm in prio if nm in inits][:2]
        for nm in names:
            if time.monotonic() > phase1_end - 8:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 6:
                break
            a0 = inits[nm]
            a_opt = optimize_pipeline(a0, t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, nm, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {nm}: {score:.6f}")
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    top_cands = []
    seen_lens = set()
    for sc, n, nm, a in results[:8]:
        if n not in seen_lens:
            top_cands.append((n, a.copy()))
            seen_lens.add(n)
        if len(top_cands) >= 5:
            break
    if best_sequence is not None:
        top_cands.append((len(best_sequence), best_sequence.copy()))
    remaining = deadline - time.monotonic() - 250.0
    t_per_cand = max(40.0, remaining / max(1, len(top_cands)))
    for n, a0 in top_cands:
        if time.monotonic() > deadline - 230:
            break
        t_alloc = min(t_per_cand, deadline - time.monotonic() - 200)
        if t_alloc < 20:
            break
        print(f"  Deep n={n}, budget={t_alloc:.0f}s")
        # Multiple subgradient passes with different params
        a_deep = a0.copy()
        for pass_i in range(3):
            if time.monotonic() > deadline - 180:
                break
            t_pass = (t_alloc / 3.0) * (1.0 - 0.25 * pass_i)
            if t_pass < 8:
                break
            a_p, obj_p = subgradient_optimize(
                a_deep, t_pass,
                lr_init=0.015 * (1 - 0.2 * pass_i),
                momentum=0.92 + 0.02 * pass_i,
                top_k=60 + 20 * pass_i,
                temp=0.0002 * (1 - 0.2 * pass_i),
                decay=0.9999,
                restart_every=300,
                restart_noise=0.003 * (1 - 0.2 * pass_i))
            conv_cur = fast_convolve(a_deep)
            if obj_p < 2.0 * n * conv_cur.max() - 1e-10:
                a_deep = a_p
        # LP peak shaving
        if time.monotonic() < deadline - 160:
            t_lp = min(40.0, (deadline - time.monotonic() - 140) * 0.3)
            if t_lp > 5:
                a_lp, obj_lp = lp_peak_shave(a_deep, t_lp, n_peaks=12, init_step=0.05)
                conv_cur = fast_convolve(a_deep)
                if obj_lp < 2.0 * n * conv_cur.max() - 1e-10:
                    a_deep = a_lp
        # Direct peak reduction
        if time.monotonic() < deadline - 130:
            t_dr = min(30.0, (deadline - time.monotonic() - 110) * 0.3)
            if t_dr > 5:
                a_dr, obj_dr = direct_peak_reduction(a_deep, t_dr, max_rounds=200)
                conv_cur = fast_convolve(a_deep)
                if obj_dr < 2.0 * n * conv_cur.max() - 1e-10:
                    a_deep = a_dr
        # Coordinate descent
        if time.monotonic() < deadline - 100:
            a_deep = coordinate_descent_fast(a_deep, max_rounds=50)
        score = float(evaluate_sequence(a_deep.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_deep.copy()
            print(f"    Deep n={n}: {score:.6f}")
    # Phase 3: Final polishing
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        n_b = len(best_sequence)
        remaining = deadline - time.monotonic() - 50.0
        for cycle in range(5):
            if time.monotonic() > deadline - 40:
                break
            t_cyc = min(remaining / max(1, 5 - cycle), 40.0)
            if t_cyc < 6:
                break
            # Subgradient
            a_s, obj_s = subgradient_optimize(best_sequence, t_cyc * 0.4,
                                               lr_init=0.003, momentum=0.96,
                                               top_k=100, temp=0.00005,
                                               decay=0.99995,
                                               restart_every=300,
                                               restart_noise=0.001)
            conv_cur = fast_convolve(best_sequence)
            if obj_s < 2.0 * n_b * conv_cur.max() - 1e-11:
                best_sequence = a_s
            # Direct reduction
            a_dr, obj_dr = direct_peak_reduction(best_sequence, t_cyc * 0.3,
                                                 max_rounds=100)
            conv_cur = fast_convolve(best_sequence)
            if obj_dr < 2.0 * n_b * conv_cur.max() - 1e-11:
                best_sequence = a_dr
            # CD
            best_sequence = coordinate_descent_fast(best_sequence, max_rounds=30)
            score = float(evaluate_sequence(best_sequence.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                print(f"  Polish {cycle}: {score:.6f}")
            # Small perturbations
            if time.monotonic() < deadline - 30:
                for ns in [0.002, 0.005]:
                    a_p = project_simplex(best_sequence + rng.standard_normal(n_b) * ns)
                    a_p = optimize_pipeline(a_p, min(4.0, (deadline - time.monotonic() - 20) * 0.1))
                    sc = float(evaluate_sequence(a_p.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_sequence = a_p.copy()
                        print(f"  Perturb({ns}) cycle {cycle}: {sc:.6f}")
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(512) / 512
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
