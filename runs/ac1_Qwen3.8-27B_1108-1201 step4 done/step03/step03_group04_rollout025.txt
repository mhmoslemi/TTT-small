```python
Hybrid optimization combining fast coordinate descent, LP-based Frank-Wolfe,
and Nesterov-accelerated subgradient methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Incremental convolution updates in coordinate descent (O(n) per step vs O(n log n))
2. Top-K peak constraint LP (faster oracle, similar quality)
3. Nesterov acceleration for subgradient phase
4. Smart coordinate ordering based on peak contribution

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix_topk(a, M, top_k=50):
    """Build sparse matrix for top-K convolution constraints."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top K peak positions
    k = min(top_k, len(conv))
    peak_positions = np.argpartition(conv, -k)[-k:]
    
    rows = []
    cols = []
    vals = []
    for pos in peak_positions:
        pos = int(pos)
        j_start = max(0, pos - n + 1)
        j_end = min(n - 1, pos)
        for j in range(j_start, j_end + 1):
            rows.append(len(rows))  # row index
            cols.append(j)
            vals.append(a[pos - j])
    
    # Actually, let me just build the full matrix but with only top-K rows
    # The matrix has shape (top_k, n)
    n_rows = len(peak_positions)
    rows = []
    cols = []
    vals = []
    for idx, pos in enumerate(peak_positions):
        pos = int(pos)
        j_start = max(0, pos - n + 1)
        j_end = min(n - 1, pos)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[pos - j])
    
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n)), peak_positions


def lp_oracle_topk(a, M, top_k=50):
    """Solve LP: max sum(b) s.t. top-K conv constraints, b >= 0."""
    n = len(a)
    A, peak_pos = build_conv_matrix_topk(a, M, top_k)
    c_obj = -np.ones(n)
    b_ub = np.full(len(peak_pos), M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_oracle_full(a, M):
    """Solve LP: max sum(b) s.t. all conv constraints, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
```

```python
"""
Hybrid optimizer combining fast coordinate descent (with incremental convolution
updates), LP-based Frank-Wolfe (top-K peak constraints), and Nesterov-accelerated
subgradient methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: vectorized CD with O(n) per-coordinate updates via incremental
convolution, adaptive LP oracle using only binding constraints, and multi-phase
time allocation focusing compute on the best candidate.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_topk_conv_matrix(a, top_k=60):
    """Build sparse matrix for top-K peak convolution constraints."""
    n = len(a)
    conv = fast_convolve(a)
    k = min(top_k, len(conv))
    peak_positions = np.argpartition(conv, -k)[-k:]
    peak_positions = np.sort(peak_positions)

    rows = []
    cols = []
    vals = []
    for idx, pos in enumerate(peak_positions):
        pos = int(pos)
        j_start = max(0, pos - n + 1)
        j_end = min(n - 1, pos)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[pos - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_positions), n))
    return A, peak_positions, conv[peak_positions]


def lp_oracle_topk(a, M, top_k=60):
    """LP oracle: max sum(b) s.t. top-K conv constraints, b >= 0."""
    n = len(a)
    A, peak_pos, peak_vals = build_topk_conv_matrix(a, top_k)
    c_obj = -np.ones(n)
    b_ub = np.full(len(peak_pos), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_oracle_full(a, M):
    """LP oracle with all constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_iteration(a, M, use_topk=True, top_k=60):
    """One Frank-Wolfe iteration: solve LP oracle, line search."""
    n = len(a)
    if use_topk:
        b = lp_oracle_topk(a, M, top_k)
    else:
        b = lp_oracle_full(a, M)
    if b is None:
        return a, M, 0.0
    sb = b.sum()
    if sb < 1e-15:
        return a, M, 0.0
    b = b / sb

    # Line search
    best_t = 0.0
    best_obj = objective_value(a)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        candidate = (1 - t) * a + t * b
        obj = objective_value(candidate)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if best_t > 0.01:
        new_a = (1 - best_t) * a + best_t * b
        new_M = fast_convolve(new_a).max()
        return new_a, new_M, best_t
    return a, M, 0.0


def fw_optimize(a0, time_limit, verbose=False, verbose_every=100, use_topk=True, top_k=60):
    """Frank-Wolfe optimization with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit:
        new_a, new_M, t = fw_iteration(a, M, use_topk, top_k)
        if t > 0:
            a = new_a
            M = new_M
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= max_stall:
            rng = np.random.default_rng(it)
            scale = 0.002 * (1 + no_improve // max_stall)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            if max_stall > 1000:
                break

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, verbose_every=100):
    """
    Fast coordinate descent with incremental convolution updates.
    Uses vectorized operations for speed.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()

    t0 = time.monotonic()
    p = 0
    stagnation = 0
    max_stag = 50

    # Precompute trial multipliers
    trial_mults = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0])
    trial_abs = np.array([1e-5, 1e-4, 1e-3, 5e-3, 0.01, 0.05])

    while time.monotonic() - t0 < time_limit:
        # Smart coordinate ordering based on peak contribution
        peak_k = int(conv.argmax())
        indices = np.arange(n)
        j_peak = peak_k - indices
        valid = (j_peak >= 0) & (j_peak < n)
        contrib = np.zeros(n)
        contrib[valid] = a[indices[valid]] * a[j_peak[valid]]
        order = np.argsort(-contrib)

        for i in order:
            old_val = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Extract affected region
            k_range = np.arange(k_lo, k_hi + 1)
            j_range = k_range - i
            is_self = (j_range == i)

            # Max over unchanged lags
            max_unchanged = 0.0
            if k_lo > 0:
                v = conv[:k_lo].max()
                if v > max_unchanged:
                    max_unchanged = v
            if k_hi + 1 < len(conv):
                v = conv[k_hi + 1:].max()
                if v > max_unchanged:
                    max_unchanged = v

            base_affected = conv[k_range]
            # Paired values (for cross terms)
            a_j = np.zeros(len(k_range))
            if not is_self.all():
                a_j[~is_self] = a[j_range[~is_self]]

            s_others = s - old_val

            # Build trial array
            trials = np.concatenate([trial_abs, old_val * trial_mults[1:]])
            trials = trials[trials >= 0]

            best_x = old_val
            cur_max_conv = max(base_affected.max(), max_unchanged)
            best_f = 2.0 * n * cur_max_conv / (s * s)

            for x in trials:
                delta = x - old_val
                new_affected = base_affected + 2.0 * delta * a_j
                if is_self.any():
                    new_affected[is_self] = base_affected[is_self] - old_val * old_val + x * x
                mx = new_affected.max()
                if max_unchanged > mx:
                    mx = max_unchanged
                s_new = s_others + x
                if s_new < 1e-15:
                    continue
                f = 2.0 * n * mx / (s_new * s_new)
                if f < best_f:
                    best_f = f
                    best_x = x

            # Apply update
            if best_x != old_val:
                delta = best_x - old_val
                a[i] = best_x
                s = s_others + best_x
                conv[k_range] = base_affected + 2.0 * delta * a_j
                if is_self.any():
                    conv[k_range[is_self]] = base_affected[is_self] - old_val * old_val + best_x * best_x

                # Check global improvement (expensive, do less frequently)
                # Just track locally
                pass

        # Periodic global check
        if p % 10 == 0:
            cur_obj = 2.0 * n * conv.max() / (s * s)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation > max_stag:
            # Perturb to escape
            rng = np.random.default_rng(p)
            scale = 0.003 * (1 + stagnation // max_stag)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
                s = 1.0
            conv = fast_convolve(a)
            stagnation = 0
            max_stag = int(max_stag * 1.4)
            if max_stag > 300:
                break

        p += 1
        if verbose and p % verbose_every == 0:
            elapsed = time.monotonic() - t0
            cur_obj = 2.0 * n * conv.max() / (s * s)
            print(f"    CD pass {p}: obj={best_obj:.8f}, cur={cur_obj:.8f}, elapsed={elapsed:.1f}s")

    # Final check
    final_obj = 2.0 * n * conv.max() / (s * s)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    return best_a, best_obj


def subgradient_nesterov(a0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    y = a.copy()

    conv_y = fast_convolve(y)
    best_obj = 2.0 * n * conv_y.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = 1
    t_prev = 1

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        conv = fast_convolve(y)
        peak_k = int(conv.argmax())

        # Subgradient of max(conv) w.r.t. a: 2 * reflected_shift(a, peak_k)
        g = np.zeros(n)
        indices = np.arange(n)
        j_peak = peak_k - indices
        valid = (j_peak >= 0) & (j_peak < n)
        g[valid] = 2.0 * y[j_peak[valid]]

        # Nesterov momentum
        mu = 2.0 / (t + 3.0)
        a_new = y - (mu / n) * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

        # Check objective
        conv_a = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_a.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()

        # Update y (Nesterov extrapolation)
        y = a_new + ((t - 1.0) / (t + 2.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = a_new
        t_prev = t
        t += 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={t}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    inits['ramp_rev'] = r[::-1].copy()

    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()

    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Asymmetric concentrated
    asym = np.zeros(n)
    asym[:n//3] = 1.0
    asym /= asym.sum()
    inits['left_conc'] = asym.copy()

    asym2 = np.zeros(n)
    asym2[-n//3:] = 1.0
    asym2 /= asym2.sum()
    inits['right_conc'] = asym2.copy()

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                interp = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(parent)), pp / ps)
                inits['parent_interp'] = interp / interp.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimization:
    Phase 1: Quick length/init scan
    Phase 2: LP-FW optimization
    Phase 3: Coordinate descent (main workhorse)
    Phase 4: Nesterov subgradient polish
    Phase 5: Final LP-FW pass
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Determine candidate lengths
    if parent is not None:
        parent_n = len(parent)
        candidate_lengths = sorted(set([
            parent_n,
            max(64, parent_n - 64),
            parent_n + 64,
            max(64, parent_n - 128),
            parent_n + 128,
            256, 384, 512, 768
        ]))
    else:
        candidate_lengths = [256, 384, 512, 768]

    # Filter to reasonable range
    candidate_lengths = [n for n in candidate_lengths if 64 <= n <= 2048]

    print(f"=== Phase 1: Quick Scan ===")
    print(f"  Candidate lengths: {candidate_lengths}")

    # Quick scan: for each length, try a few inits with brief CD
    for n in candidate_lengths[:8]:
        if time.monotonic() > deadline - 120:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'uniform', 'edge', 'ramp',
                    'triangle', 'gauss', 'left_conc', 'right_conc', 'random', 'ramp_rev']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        for name in init_names[:5]:
            if time.monotonic() > deadline - 60:
                break
            a0 = inits[name]
            # Brief CD (5 seconds)
            a_opt, obj_opt = coordinate_descent(a0, time_limit=5.0, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n}, init={name}: score={score:.6f}")

    print(f"  After Phase 1: best={best_score:.6f}")

    if best_sequence is None:
        # Fallback
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback: score={best_score:.6f}")

    # === Phase 2: LP-FW Optimization ===
    print(f"\n=== Phase 2: LP-FW Optimization ===")
    remaining = deadline - time.monotonic() - 30
    t_fw = min(300.0, remaining * 0.25)
    if t_fw > 10:
        a_opt, obj_opt = fw_optimize(best_sequence, time_limit=t_fw,
                                     verbose=True, verbose_every=200,
                                     use_topk=True, top_k=80)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 2 score={score:.6f} (local={obj_opt:.6f})")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 3: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 60
    t_cd = remaining * 0.50
    if t_cd > 10:
        a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=t_cd,
                                            verbose=True, verbose_every=200)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 3 score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 4: Nesterov Subgradient Polish ===
    print(f"\n=== Phase 4: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic() - 40
    t_sg = min(120.0, remaining * 0.3)
    if t_sg > 10:
        a_opt, obj_opt = subgradient_nesterov(best_sequence, time_limit=t_sg,
                                              verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 4 score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 5: Final LP-FW ===
    print(f"\n=== Phase 5: Final LP-FW ===")
    remaining = deadline - time.monotonic() - 20
    t_final = min(120.0, remaining * 0.6)
    if t_final > 5:
        a_opt, obj_opt = fw_optimize(best_sequence, time_limit=t_final,
                                     verbose=True, verbose_every=200,
                                     use_topk=True, top_k=100)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 5 score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 6: Final CD polish ===
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 6: Final CD Polish ===")
        remaining = deadline - time.monotonic() - 5
        t_cd2 = max(5.0, remaining * 0.7)
        a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=t_cd2,
                                            verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 6 score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
