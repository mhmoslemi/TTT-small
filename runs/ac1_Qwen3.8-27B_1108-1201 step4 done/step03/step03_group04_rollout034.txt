```python
def flat_spectrum_iteration(a, n_grid, n_iter, S):
    """Iteratively flatten the spectrum while maintaining a >= 0 and sum(a) = S."""
    n = len(a)
    a = a.copy()
    
    for it in range(n_iter):
        # Compute spectrum on fine grid
        A = np.fft.fft(a, n_grid)
        
        # Get |A|^2
        power = np.abs(A)**2
        
        # Target: flatten, keeping DC (index 0) fixed at S^2
        # Simple approach: replace peaks with their median
        median_power = np.median(power[1:])  # exclude DC
        peak_mask = power > median_power * 1.5
        peak_mask[0] = False  # don't touch DC
        
        # Reduce peaks
        A_new = A.copy()
        # Scale down peaks
        for freq in np.where(peak_mask)[0]:
            target = median_power
            if power[freq] > 0:
                scale = np.sqrt(target / power[freq])
                A_new[freq] *= scale
        
        # Transform back
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        
        # Renormalize
        if a_new.sum() > 0:
            a_new *= S / a_new.sum()
        else:
            a_new = np.ones(n) / n * S
        
        a = a_new
    
    return a
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent on the simplex with log-sum-exp approximation,
frequency-domain inspired perturbations, and multi-length scanning.
Key innovation: efficient FFT-based gradient computation and spectrum flattening steps.

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def soft_max_gradient(a, tau=0.1):
    """
    Compute gradient of soft_max(conv(a,a), tau) = tau * log(sum(exp(conv/tau))).
    Returns (gradient, soft_max_value).
    """
    n = len(a)
    conv = fast_convolve(a)  # length 2n-1
    # Numerically stable softmax
    shifted = conv / tau - conv.max() / tau
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    w = exp_vals / Z  # weights for each lag
    soft_max_val = tau * (np.log(Z) + conv.max() / tau)
    # Gradient: grad[j] = sum_k w[k] * a[k-j] for valid k-j in [0, n-1]
    # This is np.correlate(w, a, 'valid')
    grad = np.correlate(w, a, 'valid')
    return grad, soft_max_val


def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gradient_descent_simplex(a0, time_limit, tau_init=0.05, lr_init=0.01,
                             momentum=0.9, verbose=False, verbose_every=500,
                             tau_decay=0.999, kick_every=300, kick_scale=0.01):
    """
    Gradient descent on simplex with soft-max objective.
    a0: initial point in simplex (sum=1, >=0)
    Returns best (a, objective) found.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    tau = tau_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    it = 0
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit:
        grad, sm_val = soft_max_gradient(a, tau)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = proj_simplex(a_new)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        tau *= tau_decay
        lr *= 0.9998

        # Kick to escape local minima
        if stall >= max_stall:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * kick_scale
            a = proj_simplex(a + noise)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            stall = 0
            max_stall = int(max_stall * 1.5)
            kick_scale *= 0.7
            lr = lr_init * 0.5

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD it={it}, obj={best_obj:.6f}, tau={tau:.4f}, lr={lr:.6f}, stall={stall}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def spectral_flatten(a0, n_grid=None, n_iter=200, verbose=False):
    """
    Frequency-domain inspired iteration to flatten |A(w)|^2.
    Maintains a >= 0 and sum(a) = constant.
    """
    n = len(a0)
    if n_grid is None:
        n_grid = 1 << (4 * n - 1).bit_length()
    a = np.maximum(a0.copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S  # normalize to sum=1

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_iter):
        A = np.fft.fft(a, n_grid)
        power = np.abs(A) ** 2

        # Target: minimize peaks while keeping A[0] = 1 (DC)
        # Excluding DC and its mirror
        mask = np.ones(n_grid, dtype=bool)
        mask[0] = False
        mask[n_grid // 2] = False  # Nyquist

        peak_thresh = np.percentile(power[mask], 90)
        peak_mask = (power > peak_thresh) & mask

        # Reduce peaks: scale down amplitudes at peak frequencies
        A_new = A.copy()
        target_power = np.median(power[mask])
        for idx in np.where(peak_mask)[0]:
            if power[idx] > target_power * 1.01:
                scale = np.sqrt(target_power / power[idx])
                A_new[idx] *= (1.0 + 0.3 * (scale - 1.0))  # partial correction

        # Transform back
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / sn  # normalize back to sum=1

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()

        a = a_new
        if verbose and it % 50 == 0:
            print(f"    SpecFlatten it={it}, obj={best_obj:.6f}")

    return best_a, best_obj


def multi_peak_subgradient_step(a, top_k=20, temperature=0.01):
    """
    Compute a weighted subgradient using top-k peaks of conv(a,a).
    Returns the gradient vector.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Get top-k peak indices
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    # Softmax weights over peak values
    peak_vals = conv[peaks]
    shifted = peak_vals / temperature - peak_vals.max() / temperature
    w = np.exp(shifted)
    w /= w.sum()

    # Accumulate gradient
    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        if w[i] < 1e-10:
            continue
        # grad contribution from conv[k]: a[k-j] for j where 0<=k-j<n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        grad[j_start:j_end + 1] += w[i] * a[k - j_end:k - j_start + 1][::-1]

    return grad

    conv = fast_convolve(a)

    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit:
        grad = multi_peak_subgradient_step(a, top_k=top_k, temperature=temperature)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        v = momentum * v - lr * grad
        a_new = proj_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:

            stall = 0

        lr *= 0.9997

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n, all normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Sine taper (smooth edges)
    sine = np.sin(np.pi * (x + 1) / (n + 1))
    inits['sine'] = sine / sine.sum()

    # Cosine taper
    cos_t = np.cos(np.pi * x / (2 * n)) ** 2
    inits['cosine'] = cos_t / cos_t.sum()

    # Triangle (peak in middle)
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2.0) / n
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge concentrated
    edge_w = 1.0 / (1.0 + np.exp(-((x - n / 2) / (n / 6))))
    inits['edge'] = edge_w / edge_w.sum()

    # Opposite edge (ramp up)
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    # Gaussian
    gauss = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    inits['gauss'] = gauss / gauss.sum()

    # Random smooth (low-pass filtered random)
    r = rng.random(n)
    # Simple low-pass: moving average
    kernel_size = max(3, n // 50)
    if kernel_size % 2 == 0:
        kernel_size += 1
    k = np.ones(kernel_size) / kernel_size
    r_smooth = np.convolve(r, k, mode='same')
    r_smooth = np.maximum(r_smooth, 0.01)
    inits['smooth_rand'] = r_smooth / r_smooth.sum()

    # Frequency-domain flat: random phase, inverse FFT magnitude
    n_grid = 1 << (4 * n - 1).bit_length()
    phase = rng.uniform(0, 2 * np.pi, n_grid)
    A = np.exp(1j * phase)
    A[0] = abs(A[0])  # keep DC positive
    a_freq = np.fft.ifft(A).real[:n]
    a_freq = np.maximum(a_freq, 0.0)
    if a_freq.sum() > 1e-10:
        inits['freq_flat'] = a_freq / a_freq.sum()
    else:
        inits['freq_flat'] = np.ones(n) / n

    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_interp / ps

    return inits

        # Priority order
        priority = ['parent', 'parent_interp', 'freq_flat', 'sine', 'cosine',
                    'smooth_rand', 'uniform', 'gauss', 'triangle', 'ramp', 'edge']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        n_inits = len(init_names)
        time_per_init = max(5.0, 100.0 / n_inits)

        for name in init_names[:8]:
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 30)
            if t_alloc < 4:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t_alloc={t_alloc:.1f}s")
            a_opt, obj_opt = combined_optimizer(a0, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    score={score:.6f}, local_obj={obj_opt:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # === Phase 2: Deep optimization on best ===
    if time.monotonic() < deadline - 90:
        print("=== Phase 2: Deep Optimization ===")
        remaining = deadline - time.monotonic() - 60.0
        t_deep = remaining * 0.45
        if t_deep > 10:
            a_opt, obj_opt = combined_optimizer(best_sequence, time_limit=t_deep,
                                                 verbose=True, verbose_every=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 3: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0
        n_restarts = max(1, int(remaining // 25))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            scale = 0.002 * (1 + i * 0.3)
            noise = rng.standard_normal(n_best) * scale
            a_pert = proj_simplex(best_sequence + noise)
            t_alloc = min(20.0, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break
            a_opt, obj_opt = combined_optimizer(a_pert, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")
                else:
                    print(f"  Restart {i}: score={score:.6f}")

    # === Phase 4: Final spectral polish ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 4: Final Spectral Polish ===")
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5:
            n_best = len(best_sequence)
            n_grid = 1 << (8 * n_best - 1).bit_length()
            a_norm = best_sequence / best_sequence.sum()
            a_opt, obj_opt = spectral_flatten(a_norm, n_grid=n_grid,
                                               n_iter=int(remaining * 30))
            # Scale back
            a_opt_scaled = a_opt * best_sequence.sum()
            score = float(evaluate_sequence(a_opt_scaled.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt_scaled.copy()

    # === Phase 5: Try a few different lengths near the best ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Neighbor Lengths ===")
        n_best = len(best_sequence)
        neighbor_lengths = [n // 2, n * 3 // 2, n * 2]
        neighbor_lengths = [nl for nl in neighbor_lengths if 128 <= nl <= 2048]
        neighbor_lengths = list(set(neighbor_lengths))

        for nl in neighbor_lengths[:2]:
            if time.monotonic() > deadline - 15:
                break
            inits = generate_inits(nl, best_sequence, rng)
            # Use parent interpolation and a couple others
            for name in ['parent_interp', 'uniform', 'sine']:
                if name not in inits:
                    continue
                if time.monotonic() > deadline - 12:
                    break
                t_alloc = min(15.0, deadline - time.monotonic() - 12)
                if t_alloc < 5:
                    break
                a0 = inits[name]
                a_opt, obj_opt = combined_optimizer(a0, time_limit=t_alloc, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    print(f"  n={nl}, init={name}: score={score:.6f}")
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
