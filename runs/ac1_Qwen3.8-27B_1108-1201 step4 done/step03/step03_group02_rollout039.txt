"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with step-function initializations and projected subgradient polishing.
Key insight: step functions (asymmetric) achieve lower peak autocorrelation
than uniform sequences for non-negative inputs.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_value(a):
    """Compute the objective value for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    return A


def lp_maximize_sum(a, M, conv_matrix=None):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    if conv_matrix is None:
        conv_matrix = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=conv_matrix, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a, conv_matrix=None):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    if conv_matrix is None:
        conv_matrix = build_conv_matrix_sparse(a)
    num_rows = 2 * n - 1
    # Variables: b[0..n-1], M'
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = conv_matrix.toarray()
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

            if b is None:

            b_sum = b.sum()
            if b_sum < 1e-15:

        else:

            if b_sum < 1e-15:

            no_improve = 0
            step = min(0.9, step * 1.03)
        else:
            no_improve += 1
            step = max(step_min, step * 0.95)

        a = a_new
        conv = conv_new

            step = step_init
            no_improve = 0
            conv_matrix = build_conv_matrix_sparse(a)

        if verbose and it % 50 == 0:

def subgradient_step(a, conv, top_k, temperature):
    """Weighted subgradient over top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def optimize_subgrad(a0, time_limit, lr_init=0.05, momentum=0.90, top_k=50,
                     temp_init=0.0005, perturb_every=2000, perturb_scale=0.003):
    """Projected subgradient with momentum and periodic perturbation."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_step(a, conv, top_k, temp)
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.2, lr * 0.6)

    return best_a, best_obj


def make_step_init(n, step_pos, ratio, smooth=0):
    """Create a step function initialization.
    
    Values are 1 for j < step_pos and ratio for j >= step_pos,
    optionally smoothed over 'smooth' elements around the step.
    """
    a = np.ones(n)
    if smooth > 0:
        # Smooth transition
        for j in range(n):
            if j < step_pos - smooth // 2:
                a[j] = 1.0
            elif j >= step_pos + smooth // 2:
                a[j] = ratio
            else:
                t = (j - (step_pos - smooth // 2)) / smooth
                t = np.clip(t, 0, 1)
                a[j] = 1.0 + (ratio - 1.0) * t
    else:
        a[step_pos:] = ratio
    a = np.maximum(a, 0)
    return a / a.sum()


def make_ramp_init(n, ratio):
    """Linear ramp from 1 to ratio."""
    a = np.linspace(1.0, ratio, n)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_smooth_step_init(n, center, width, ratio):

    a = 1.0 + (ratio - 1.0) / (1.0 + np.exp(-(x - center) / width))
    a = np.maximum(a, 0)
    return a / a.sum()

def optimize_with_truncation(a_full, n_use):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using step-function initializations,
LP-based iterative refinement, and subgradient polishing.
Key insight: asymmetric step functions achieve lower peak autocorrelation
than uniform sequences for non-negative inputs, as they shift the convolution
peak away from the symmetric center.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_value(a):
    """Compute the objective value for a non-negative sequence (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    rows = np.zeros(num_rows * n, dtype=np.int32)
    cols = np.zeros(num_rows * n, dtype=np.int32)
    vals = np.zeros(num_rows * n, dtype=np.float64)
    idx = 0
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = range(j_start, j_end + 1)
        vals[idx:idx + count] = a[k - np.arange(j_start, j_end + 1)]
        idx += count
    # Trim to actual size
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(num_rows, n))
    return A


def lp_maximize_sum(a, M, conv_matrix=None):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    if conv_matrix is None:
        conv_matrix = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=conv_matrix, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a, conv_matrix=None):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    if conv_matrix is None:
        conv_matrix = build_conv_matrix_sparse(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Build A_ub as sparse
    import scipy.sparse as sp
    A_ub_sparse = sp.bmat([[conv_matrix, -np.ones((num_rows, 1))]]).tocsr()
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub_sparse, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def lp_restricted(a, M, top_k=200, conv_matrix=None):

    conv = fast_convolve(a)

    c = -np.ones(n)

    res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:

    return None

def subgradient_step(a, conv, top_k, temperature):
    """Weighted subgradient over top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def optimize_subgrad(a0, time_limit, lr_init=0.05, momentum=0.92, top_k=50,
                     temp_init=0.0005, perturb_every=1500, perturb_scale=0.002):
    """Projected subgradient with momentum and periodic perturbation."""

        a = np.ones(len(a)) / len(a)
    else:

    n = len(a)

    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(99)

        sg = subgradient_step(a, conv, top_k, temp)
        v = momentum * v - lr * sg

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            perturb_every = int(perturb_every * 1.5)
            lr = max(lr_init * 0.15, lr * 0.5)

    return best_a, best_obj


def make_step_init(n, step_pos, ratio, smooth=0):
    """Create a step function initialization."""
    a = np.ones(n)
    if smooth > 0:
        for j in range(n):
            if j < step_pos - smooth // 2:
                a[j] = 1.0
            elif j >= step_pos + smooth // 2:
                a[j] = ratio
            else:
                t = (j - (step_pos - smooth // 2)) / max(1, smooth)
                t = np.clip(t, 0, 1)
                a[j] = 1.0 + (ratio - 1.0) * t
    else:
        a[step_pos:] = ratio
    a = np.maximum(a, 0)
    return a / a.sum()


def make_ramp_init(n, ratio):
    """Linear ramp from 1 to ratio."""
    a = np.linspace(1.0, ratio, n)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_smooth_step_init(n, center, width, ratio):
    """Smooth step using sigmoid."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + (ratio - 1.0) / (1.0 + np.exp(-(x - center) / max(width, 0.1)))
    a = np.maximum(a, 0)
    return a / a.sum()


def make_two_level_init(n, frac, ratio):
    """Two-level: first frac*n elements = 1, rest = ratio."""
    a = np.ones(n)
    split = int(n * frac)
    a[split:] = ratio
    a = np.maximum(a, 0)
    return a / a.sum()


def optimize_analytical(n, time_budget=20.0):
    """Sweep over analytical initializations to find the best starting point."""
    best_a = None
    best_obj = float('inf')
    t0 = time.monotonic()

    # Two-level (step) functions with various parameters
    fracs = [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6]
    ratios = [1.3, 1.5, 1.8, 2.0, 2.5, 3.0, 4.0, 5.0, 7.0, 10.0]

    for frac in fracs:
        if time.monotonic() - t0 > time_budget:
            break
        for r in ratios:
            if time.monotonic() - t0 > time_budget:
                break
            a = make_two_level_init(n, frac, r)
            obj = objective_value(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    # Smooth steps
    for center_frac in [0.4, 0.5, 0.6]:
        if time.monotonic() - t0 > time_budget:
            break
        center = int(n * center_frac)
        for width in [2, 5, 10, 20]:
            for r in [1.5, 2.0, 3.0, 5.0]:
                if time.monotonic() - t0 > time_budget:
                    break
                a = make_smooth_step_init(n, center, width, r)
                obj = objective_value(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    # Ramps
    for r in [1.5, 2.0, 3.0, 5.0, 10.0]:
        if time.monotonic() - t0 > time_budget:
            break
        a = make_ramp_init(n, r)
        obj = objective_value(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def try_trim(a):
    """Try to improve objective by trimming near-zero ends."""
    best_a = a.copy()
    best_obj = objective_value(a)
    n = len(a)
    
    for trim in range(0, min(50, n // 4)):
        if trim == 0:
            continue
        a_t = a[trim:n - trim] if n - 2 * trim > 10 else a
        if len(a_t) <= 10:
            continue
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t = a_t / s
        obj = objective_value(a_t)
        if obj < best_obj:
            best_obj = obj
            best_a = a_t.copy()
    
    return best_a, best_obj
