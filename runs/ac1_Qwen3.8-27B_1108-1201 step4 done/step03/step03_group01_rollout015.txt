def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_from_conv(conv, n):
    """Compute objective from precomputed convolution."""
    max_b = conv.max()
    return 2.0 * n * max_b  # assuming sum(a) = 1

def compute_subgradient(a, conv, n, top_k=5, temp=0.001):
    """
    Compute a weighted subgradient of max(conv(a,a)).
    Uses softmax weighting over near-peak positions.
    """
    max_val = conv.max()
    # Find positions where conv is near the max
    threshold = max_val * (1 - 0.01)  # within 1% of max
    peak_indices = np.where(conv >= threshold)[0]
    
    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv)])
    
    # Softmax weighting
    peak_vals = conv[peak_indices]
    scaled = (peak_vals - max_val) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peak_indices))
        w[0] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        if w[idx] < 1e-12:
            continue
        k = int(peak_indices[idx])
        # grad_i = w[idx] * (a[k-i] + a[i-k]) with bounds

            ki = k - i
            if 0 <= ki < n:
                val += a[ki]
            ik = i - k
            if 0 <= ik < n:
                val += a[ik]
            grad[i] += w[idx] * val
    
    return grad

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute the objective assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def subgradient_step(a, conv, n, temp=0.0005, top_k=10):
    """Compute a weighted subgradient of max(conv)."""
    max_val = conv.max()
    # Softmax weights over near-peaks
    threshold = max_val - 0.05 * max_val  # within 5% of max
    peak_idx = np.where(conv >= threshold)[0]
    
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(conv)])
    
    # Limit to top_k peaks
    if len(peak_idx) > top_k:
        peak_vals = conv[peak_idx]
        top_positions = np.argsort(peak_vals)[-top_k:]
        peak_idx = peak_idx[top_positions]
    
    peak_vals = conv[peak_idx]
    scaled = (peak_vals - max_val) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peak_idx))
        w[0] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient
    grad = np.zeros(n)
    for idx in range(len(peak_idx)):
        if w[idx] < 1e-12:
            continue
        k = int(peak_idx[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx_range = np.arange(i_start, i_end + 1)
            grad[idx_range] += w[idx] * 2.0 * a[k - idx_range]
    
    return grad

def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum_init=0.9, 
                          temp=0.0005, top_k=10, batch=1):
    """
    Accelerated subgradient descent on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        grad = subgradient_step(a, conv, n, temp=temp, top_k=top_k)
        
        # Heavy-ball momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        a = a_new
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9995
        temp = max(0.0001, temp * 0.999)
        momentum = min(0.97, momentum + 0.0002)
        
        # Restart if stuck
        if no_improve >= 500:
            rng = np.random.default_rng(iteration)
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = project_simplex(a)
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0
        
        iteration += 1
    
    return best_a, best_obj

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lse_objective_and_grad(z, beta, n):
    """
    Compute the log-sum-exp objective and gradient with respect to z,
    where a = softmax(z).
    """
    # Softmax
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    
    # Convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp
    beta_conv = beta * conv
    beta_conv_max = beta_conv.max()
    lse = (beta_conv_max + np.log(np.sum(np.exp(beta_conv - beta_conv_max)))) / beta
    
    # Weights for the softmax over conv values
    w = np.exp(beta_conv - beta_conv_max)
    w_sum = w.sum()
    w = w / w_sum  # This gives exp(β*c_k) / sum exp(β*c_j) = w_k
    
    # Gradient of lse w.r.t. a: sum_k w_k * 2 * a_{k-i} (with bounds)
    grad_a = np.zeros(n)
    # For efficiency, only compute for significant w_k
    threshold = 1e-10
    active = w > threshold
    active_idx = np.where(active)[0]
    
    for idx in active_idx:
        k = int(idx)
        if w[idx] < threshold:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            grad_a[i_start:i_end+1] += w[idx] * 2.0 * a[k - (np.arange(i_start, i_end+1))]
    
    # Chain rule: grad_z = a * (grad_a - a @ grad_a)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return lse, grad_z

def fista_optimize(a0, time_limit, beta_init=10, beta_final=500, lr=None):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    
    # Previous iterate for momentum
    b = a.copy()
    t = 1.0
    
    # Lipschitz constant estimate
    # ∇f_β(a) is bounded; we need L such that ||∇f(a) - ∇f(b)|| <= L||a-b||
    # For now, use an adaptive L
    L = 1.0
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Extrapolation
        a_new = a + ((t - 1) / (t + 1)) * (a - b)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        # Gradient step with LSE smoothing
        conv = fast_convolve(a_new)
        beta_conv = beta * conv
        beta_max = beta_conv.max()
        w = np.exp(beta_conv - beta_max)
        w = w / w.sum()
        
        # Gradient of LSE w.r.t. a
        grad = np.zeros(n)
        active = w > 1e-10
        for idx in np.where(active)[0]:
            k = int(idx)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                grad[i_start:i_end+1] += w[idx] * 2.0 * a_new[k - np.arange(i_start, i_end+1)]
        
        # Backtracking line search for L
        step = 1.0 / L
        a_trial = a_new - step * grad
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial / s_trial
        
        # Check descent (simplified: just accept)
        b = a.copy()
        a = a_trial
        t = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
        
        # Evaluate
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Gradually increase beta
        beta *= 1.001
        if beta > beta_final:
            beta = beta_final
        
        # Restart if stuck
        if no_improve >= 200:
            a = best_a.copy()
            b = a.copy()
            t = 1.0
            L = 1.0
            no_improve = 0
        
        # Update L
        L *= 1.001
    
    return best_a, best_obj
