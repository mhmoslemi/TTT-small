def solve_inner_qp(q, a0, n, max_iters=1000, lr=0.01):
    """Solve min_{a in simplex} sum_k q_k conv(a,a)[k] using projected gradient."""
    a = a0.copy()
    for _ in range(max_iters):
        # Gradient: 2 * sum_k q_k * a[k-i] for each i
        # This is 2 * (q correlated with a)
        # cross_corr[i] = sum_k q_k * a[k-i] (with appropriate bounds)
        # = sum_m q_{m+i} * a[m]
        
        # Using FFT: if we define the full sum, it's a convolution
        # Let's compute it directly for clarity, or via FFT
        
        # Actually, let's think about it as:
        # grad[i] = 2 * sum_{j: 0<=j<n, 0<=i+j<2n-1} q_{i+j} * a[j]
        
        # This is like a "Hankel" operation. We can compute it as:
        # For each i, grad[i]/2 = <q_{i:i+2n-2}, a> (dot product of slice of q with a)
        
        # Using the fact that this is a matrix-vector product with a Hankel matrix,
        # we can use FFT-based tricks, or just do it directly for moderate n.
        
        # For efficiency, note that:
        # sum_j q_{i+j} a[j] = cross-correlation of (q, a) at lag i
        # = (q * flip(a))[i + n - 1] (using standard convolution)
        
        # Let's just compute it using the relationship:
        # H @ a where H[i,j] = q_{i+j}
        # H is a Hankel matrix, and H @ a can be computed via FFT in O(n log n)
        
        # Method: H[i,j] = q[i+j]. Let b = flip(a) (b[j] = a[n-1-j]).
        # Then (H @ a)[i] = sum_j q[i+j] a[j] = sum_j q[i+j] b[n-1-j]
        # Let k = n-1-j, so j = n-1-k:
        # = sum_k q[i+n-1-k] b[k]
        # = (convolve(q, b))[i+n-1] (linear convolution)
        
        b = a[::-1].copy()
        # Full linear convolution of q (length 2n-1) and b (length n)
        # has length 2n-1+n-1 = 3n-2
        # We want indices i+n-1 for i=0,...,n-1
        fft_size = 1 << (3*n).bit_length()
        Q = np.fft.rfft(q, fft_size)
        B = np.fft.rfft(b, fft_size)
        conv_qb = np.fft.irfft(Q * B, fft_size)
        # conv_qb[m] = sum_k q[k] * b[m-k] for valid k, m-k
        # We want sum_k q[i+n-1-k] * b[k] = conv_qb[i+n-1]
        grad = 2.0 * conv_qb[n-1:n-1+n]
        
        # Projected gradient step
        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.ones(n) / n
    
    return a

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a minimax (duel) algorithm
in the primal-dual space, combined with weighted log-sum-exp smoothing,
Nesterov-accelerated gradient descent, and peak-shaving coordinate descent.

Key innovation: Uses the dual variable q (distribution over lags) to
guide the primal optimization, concentrating effort on "active" peaks.
This provides a principled way to flatten the autocorrelation spectrum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(q, v, n):
    """Compute H @ v where H[i,j] = q[i+j] (Hankel matrix).
    Uses FFT: (H @ v)[i] = (q * reverse(v))[i + n - 1].
    """
    b = v[::-1]
    fft_size = 1 << (len(q) + n - 1).bit_length()
    Q = np.fft.rfft(q, fft_size)
    B = np.fft.rfft(b, fft_size)
    conv = np.fft.irfft(Q * B, fft_size)
    return conv[n - 1:n - 1 + n]


def hankel_obj(q, a, n):
    """Compute a^T H a = sum_k q_k conv(a,a)[k]."""
    conv = fast_convolve(a)
    return np.dot(q, conv[:len(q)])


def solve_inner_qp(q, a_init, n, time_limit=5.0, tol=1e-10):
    """Solve min_{a in simplex} sum_k q_k conv(a,a)[k] using FISTA with backtracking."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = a.copy()
    t = 1.0
    lr = 0.5  # initial step size
    
    t0 = time.monotonic()
    prev_obj = hankel_obj(q, a, n)
    
    while time.monotonic() - t0 < time_limit:
        # Gradient at z: 2 * H @ z
        grad_z = 2.0 * hankel_matvec(q, z, n)
        
        # FISTA step
        a_new = project_simplex(z - lr * grad_z)
        
        # Backtracking if needed (check objective)
        obj_new = hankel_obj(q, a_new, n)
        obj_z = hankel_obj(q, z, n)
        if obj_new > obj_z + 1e-12:
            # Reduce step size
            lr = lr * 0.5
            a_new = project_simplex(z - lr * grad_z)
            obj_new = hankel_obj(q, a_new, n)
            if obj_new > obj_z + 1e-12:
                # Just keep z
                a_new = z.copy()
                obj_new = obj_z
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z_new = a_new + (t / t_new - 1.0) * (a_new - a)
        z = np.maximum(z_new, 0.0)
        zs = z.sum()
        if zs > 0:
            z = z / zs
        a = a_new
        t = t_new
        
        # Convergence check
        if abs(obj_new - prev_obj) < tol * max(1.0, prev_obj):
            break
        prev_obj = obj_new
        
        # Increase lr if we didn't need backtracking
        if obj_new <= obj_z:
            lr = min(lr * 1.05, 2.0)
    
    return a, prev_obj


def duel_algorithm(n, a_init, time_limit, eta_init=0.5, seed=42):
    """
    Minimax algorithm:
    Outer: update q (dual weights on lags)
    Inner: solve QP for a (primal)
    
    Returns (a, q, best_obj).
    """
    rng = np.random.default_rng(seed)
    
    # Initialize q uniformly
    q = np.ones(2 * n - 1) / (2 * n - 1)
    
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    epoch = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        epoch += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        # Adaptive eta
        eta = eta_init * (1.0 - 0.5 * frac)
        
        # Inner solve
        inner_time = max(0.5, min(10.0, (time_limit - (time.monotonic() - t0)) / max(1, 50 - epoch)))
        a_inner, obj_inner = solve_inner_qp(q, a, n, time_limit=inner_time)
        
        # Compute autocorrelation
        conv = fast_convolve(a_inner)
        cur_max = conv.max()
        
        # Update best
        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_a = a_inner.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_inner
        
        # Update q (dual): concentrate on peaks
        # Multiplicative weights update
        r = conv / max(conv.max(), 1e-30)
        q = q * np.exp(eta * r)
        q = q / q.sum()
        
        # Occasional sharpening
        if epoch % 20 == 0:
            q = q ** 1.5
            q = q / q.sum()
        
        if no_improve > 50:
            # Reset q to uniform and try a different initial point
            q = np.ones(2 * n - 1) / (2 * n - 1)
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            no_improve = 0
        
        if epoch % 100 == 0:
            print(f"    Duel epoch {epoch}: obj={best_obj:.8f}, cur={cur_max:.8f}", flush=True)
    
    return best_a, q, best_obj


def weighted_smoothed_gradient(a, q_weights, beta):
    """
    Gradient of log(sum_k exp(beta * q_k * conv(a,a)[k])) / beta.
    This is a weighted version of the log-sum-exp smoothing.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Weighted log-sum-exp
    weighted = beta * q_weights[:len(conv)] * conv
    weighted -= weighted.max()
    p = np.exp(weighted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # Gradient: 2 * sum_k p_k * q_k * d(conv_k)/d(a[i])
    # d(conv_k)/d(a[i]) = 2 * a[k-i] (for valid k-i)
    # So grad[i] = 4 * sum_k p_k * q_k * a[k-i]
    #             = 4 * (softmax_weighted * a)[i] (Hankel-like)
    
    w = p * q_weights[:len(conv)]
    # grad[i] = 4 * sum_j w[i+j] * a[j] (Hankel matvec with w instead of q)
    # Actually: grad[i] = 2 * sum_k (p_k * q_k) * 2 * a[k-i] = 4 * sum_k w_k * a[k-i]
    # = 4 * (w * reverse(a))[i + n - 1]
    
    b = a[::-1]
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    B = np.fft.rfft(b, fft_size)
    conv_wa = np.fft.irfft(W * B, fft_size)
    
    return 4.0 * conv_wa[n - 1:n - 1 + n]


def optimize_weighted_smoothed(a0, q_weights, time_limit, beta_init=20.0, beta_end=500.0,
                                 lr_init=0.01, momentum=0.92, seed=0):
    """Nesterov-accelerated projected gradient on weighted smooth objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    total_iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac * beta
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 0:
            y = y / ys
        
        grad = weighted_smoothed_gradient(y, q_weights, beta)
        
        # Momentum update
        v = momentum * v - lr_init * grad * (1.0 - 0.5 * frac)
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Perturbation
        if no_improve >= 300:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
        
        # Update lr
        lr_init = max(0.0005, lr_init * (1.0 - 0.3 * frac))
    
    return best_a, best_obj


def peak_shaving_improved(a0, time_limit, seed=0):

                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                        elif obj_try < max_val:
                            # Accept if it reduces the current peak even if not the global best
                            a = a_try.copy()
                            improved = True
                            break
                        else:
                            break  # Larger frac won't help if smaller didn't
                
                if improved:
                    break
            
            if improved:
                break
        
        iters += 1
        
        # Periodic: try random moves
        if not improved and iters % 200 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * 0.05
                a_try = a.copy()
                a_try[i] -= amt
