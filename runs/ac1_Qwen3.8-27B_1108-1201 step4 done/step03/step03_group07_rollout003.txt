def smooth_max_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_vals / total
    sm = x_max + np.log(total) / beta
    return sm, w

def grad_smooth_conv(a, w):
    # Gradient of sum_k w[k] * conv[k] w.r.t. a
    # conv[k] = sum_i a[i]*a[k-i]
    # d(conv[k])/da[j] = 2*a[k-j] (with appropriate boundary handling)
    # sum_k w[k] * d(conv[k])/da[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * (w * a)[j] ... wait, this is a cross-correlation
    # Actually: sum_k w[k] * a[k-j] for k where k-j is in [0,n-1]
    # Let m = k-j, then k = m+j, and we sum over m where 0 <= m < n and 0 <= m+j < 2n-1
    # = sum_m w[m+j] * a[m] (for valid m)
    # This is a cross-correlation of w with a, evaluated at lag j
    n = len(a)
    # Use FFT for efficiency
    N = 1 << (n + len(w) - 1 + 63).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    # cross-correlation: sum_k w[k] * a[k-j]
    # This is the convolution of w with reverse(a), evaluated at appropriate index
    # Or: ifft(fft(w) * conj(fft(a)))
    result = np.real(np.fft.ifft(W * np.conj(A), N))
    grad = 2.0 * result[:n]
    return grad

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n+1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(cond.nonzero()[0]) + 1
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def objective_smooth(a, beta):
    conv = fast_convolve(a)
    sm, _ = smooth_max_weights(conv, beta)
    return 2.0 * len(a) * sm

def gerchberg_saxton_step(a, n_iters=100):
    # Iterate between time domain (non-negative) and frequency domain (flat magnitude)
    n = len(a)
    fft_size = 1 << (2*n - 1 + 63).bit_length()
    a = a.copy()
    a = a / a.sum()
    
    for _ in range(n_iters):
        # Go to frequency domain
        A = np.fft.fft(a, fft_size)
        # Set magnitude to flat (except DC)
        target_mag = np.ones(fft_size) / np.sqrt(fft_size)
        target_mag[0] = np.abs(A[0])  # Keep DC
        # Enforce Hermitian symmetry for real signal
        phase = A / np.abs(A + 1e-300)
        A_new = target_mag * phase
        # Ensure Hermitian symmetry
        A_new[0] = np.abs(A_new[0])
        for k in range(1, fft_size // 2 + 1):
            A_new[fft_size - k] = np.conj(A_new[k])
        # Go back to time domain
        a = np.real(np.fft.ifft(A_new, fft_size))[:n]
        a = np.maximum(a, 0)  # Project to non-negative
        s = a.sum()
        if s > 1e-15:
            a /= s
        else:
            a = np.ones(n) / n
            break
    return a

def subgradient_step(a, step_size):
    # Take a subgradient step on the actual max
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    max_indices = np.where(conv >= max_val - 1e-10)[0]
    # Subgradient direction
    sg = np.zeros(n)
    for k in max_indices:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                if k == 2*j:
                    sg[j] += a[idx] / len(max_indices)
                else:
                    sg[j] += 2*a[idx] / len(max_indices)
    # Projected gradient step
    a_new = project_simplex(a - step_size * sg)
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
1. Gerchberg-Saxton (intensity holography) initialization for flat autocorrelation
2. Smooth L-BFGS-B optimization with adaptive temperature
3. Subgradient polishing of the actual max
4. Multi-start over diverse initializations and sequence lengths
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_vals / total
    sm = x_max + np.log(total) / beta
    return sm, w


def grad_smooth_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    N = 1 << (n + len(w) - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    result = np.real(ifft(W * np.conj(A), N))
    return 2.0 * result[:n]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(cond.nonzero()[0]) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj(x, n, beta):
    """Objective for scipy minimize with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_weights(conv, beta)
    return 2.0 * n * sm


def softmax_jac(x, n, beta):
    """Jacobian for scipy minimize with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_weights(conv, beta)
    grad_a = grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(softmax_obj, x0, jac=softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 30})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def optimize_pg(a0, beta, lr=0.05, n_iter=300):
    """Projected gradient descent on simplex with smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj * (1 - 1e-8):
            best_obj = 2.0 * n * fast_convolve(a).max()
            best_a = a.copy()

        grad_a = grad_smooth_conv(a, w)
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break

        t = lr
        for _ in range(8):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        a = a_new
        lr *= 0.999

    # Final check
    final_obj = 2.0 * n * fast_convolve(a).max()
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    return best_a, best_obj


def gerchberg_saxton(a_init, n_iters=200):
    """GS/intensity holography for flat autocorrelation."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a
    a /= s
    fft_size = 1 << (2 * n - 1 + 63).bit_length()

    for _ in range(n_iters):
        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-300)
        # Target: flat magnitude except DC
        target_mag = np.zeros(fft_size)
        target_mag[0] = mag[0]  # preserve DC
        target_mag[1:] = mag[1:].mean()
        # Actually, we want total power to match
        # sum |A|^2 = fft_size * sum(a^2) by Parseval
        # For flat: |A_k|^2 = const for k>=1, |A_0|^2 = (sum a)^2 = 1
        # By Parseval: 1 + (fft_size-1)*c = fft_size * sum(a^2)
        # We want to minimize sum(a^2) which means minimize c
        # But c is determined by the constraint... Let's just use flat.
        A_new = target_mag * phase
        # Enforce Hermitian symmetry
        A_new[0] = np.abs(A_new[0])
        half = fft_size // 2
        A_new[1:half + 1] = (A_new[1:half + 1] + np.conj(A_new[fft_size - half:fft_size - 1:-1])) / 2
        if fft_size % 2 == 0:
            A_new[half] = np.abs(A_new[half])
        a = np.real(ifft(A_new, fft_size))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        else:
            a = np.ones(n) / n
            break
    return a


def subgradient_polish(a, n_iter=100, lr=0.01):
    """Polish using subgradient of the actual (non-smooth) max."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        max_val = conv.max()
        # Find all near-max indices
        threshold = max_val - 1e-8
        max_mask = conv >= threshold
        w = max_mask.astype(float)
        w /= w.sum()

        grad = grad_smooth_conv(a, w)
        # Adaptive step size
        t = lr / (1.0 + it * 0.01)
        for _ in range(6):
            a_new = project_simplex(a - t * grad)
            obj_new = 2.0 * n * fast_convolve(a_new).max()
            if obj_new <= best_obj * (1 + 1e-10):
                break
            t *= 0.5
        a = a_new
        obj_cur = 2.0 * n * fast_convolve(a).max()
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a.copy()
        # Decay learning rate
        lr *= 0.99

    return best_a, best_obj


def coordinate_descent(a, n_sweeps=5):
    """Simple coordinate descent: try scaling each element."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for sweep in range(n_sweeps):
        for i in range(n):
            old_val = a[i]
            # Try reducing
            for scale in [0.95, 0.9, 0.8, 0.7]:
                a[i] = old_val * scale
                obj = 2.0 * n * fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a[i] = old_val
                    break
            # Try increasing (if current is small)
            if old_val < 1.0 / n:
                for scale in [1.1, 1.2, 1.5]:
                    a[i] = old_val * scale
                    obj = 2.0 * n * fast_convolve(a).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    else:
                        a[i] = old_val
                        break
        a = best_a.copy()
        a = a / a.sum()
        best_obj = 2.0 * n * fast_convolve(a).max()
        best_a = a.copy()

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated
    edge = np.zeros(n)
    falloff = np.minimum(x, n - 1 - x).astype(float)
    edge = np.exp(-falloff / (n * 0.1))
    edge /= edge.sum()
    inits['edge_conc'] = edge

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Two Gaussians
    g2 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.15 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_two'] = g2

    # Sine
    s = np.abs(np.sin(np.pi * (x + 0.5) / n)) + 0.1
    s /= s.sum()
    inits['sine'] = s

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_resized'] = pp / ps
                inits['parent_noise'] = project_simplex(pp / ps + rng.standard_normal(n) * 0.01)

    return inits

                break
            maxiter = int(min(300, max(50, t_alloc * 25)))
            a_opt, _ = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
            score = score_and_update(a_opt)
            # Also try PG for robustness
            if time.monotonic() < phase2_end - 20:
                a_pg, _ = optimize_pg(a_opt, beta=beta * 0.5, lr=0.03, n_iter=100)
                score = score_and_update(a_pg)
                if score < best_score:
                    a_opt = a_pg
            a_cur = a_opt

    # Try alternative lengths
    for n_alt in alt_lengths:
        if time.monotonic() > phase2_end - 30:
            break
        if n_alt == n_target and best_sequence is not None and len(best_sequence) == n_target:
            continue
        print("  Trying n=%d" % n_alt)

        # Initialize from best
        if best_sequence is not None:
            a_init = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_sequence)), best_sequence)
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s > 1e-10:
                a_init /= s
            else:
                a_init = np.ones(n_alt) / n_alt
        else:
            a_init = np.ones(n_alt) / n_alt

        # GS
        a_gs = gerchberg_saxton(a_init, n_iters=100)
        score = score_and_update(a_gs)

        # L-BFGS-B
        if time.monotonic() < phase2_end - 20:
            t_alloc = min(60.0, phase2_end - time.monotonic() - 15)
            maxiter = int(min(300, max(50, t_alloc * 20)))
            for beta in [30.0, 10.0, 5.0]:
                if time.monotonic() > phase2_end - 10:
                    break
                a_opt, _ = optimize_lbfgsb(a_gs, beta=beta, maxiter=maxiter // 2)
                score = score_and_update(a_opt)
                a_gs = a_opt

    print("Phase 2 done. Best so far: %.8f" % best_score)

    # Phase 3: Fine-tuning (last 350s)
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)

        # Subgradient polishing
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            print("  Subgradient polishing...")
            t_sg = min(100.0, remaining * 0.3)
            n_sg_iter = int(min(200, t_sg * 3))
            a_sg, _ = subgradient_polish(best_sequence, n_iter=n_sg_iter, lr=0.005)
            score = score_and_update(a_sg)
            if score < best_score:
                pass  # already updated

        remaining = deadline - time.monotonic() - 30

        # Coordinate descent
        if remaining > 30:
            print("  Coordinate descent...")
            a_cd, _ = coordinate_descent(best_sequence, n_sweeps=3)
            score = score_and_update(a_cd)
            if score < best_score:
                pass

        remaining = deadline - time.monotonic() - 30

        # Low-beta L-BFGS-B polish
        if remaining > 30:
            print("  Low-beta L-BFGS-B polish...")
            for beta in [2.0, 1.5, 1.0]:
                if remaining < 15:
                    break
                t_alloc = min(50.0, remaining / 3)
                if t_alloc < 8:
                    break
                maxiter = int(min(300, max(50, t_alloc * 25)))
                a_opt, _ = optimize_lbfgsb(best_sequence, beta=beta, maxiter=maxiter)
                score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 30

        # Random perturbations
        if remaining > 40:
            print("  Random perturbation search...")
            n_pert = min(8, int(remaining / 20))
            for i in range(n_pert):
                if remaining < 15:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
                # Quick optimize
                t_alloc = min(40.0, remaining / 2)
                if t_alloc < 8:
                    break
                maxiter = int(min(200, max(30, t_alloc * 15)))
                a_opt, _ = optimize_lbfgsb(a_pert, beta=10.0, maxiter=maxiter)
                score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 30

        # Final subgradient + PG combo
        if remaining > 20:
            print("  Final polish...")
            a_final, _ = subgradient_polish(best_sequence, n_iter=50, lr=0.003)
            score = score_and_update(a_final)
            if score < best_score:
                pass
            if remaining > 15:
                a_final2, _ = optimize_pg(best_sequence, beta=5.0, lr=0.01, n_iter=100)
                score = score_and_update(a_final2)

    # Final verification
    if best_sequence is None:
        # Fallback: uniform sequence
        best_sequence = np.ones(256) / 256

    final_score = evaluate_sequence(best_sequence.tolist())
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but just in case
        best_sequence = np.ones(256) / 256
        best_score = evaluate_sequence(best_sequence.tolist())

    print("\nFINAL SCORE: %.8f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
