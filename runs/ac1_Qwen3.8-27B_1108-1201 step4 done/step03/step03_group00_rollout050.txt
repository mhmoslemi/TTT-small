    # Wait, that's the same: 2*a[k-i] when k=2i gives 2*a[i]. OK.
    # But we need to be careful: if k-i is not a valid index (outside [0,n-1]), then that term doesn't exist.
    
    # So d(conv[k])/da[i] = a[k-i] * (1 if 0<=k-i<n else 0) + a[k-i] * (1 if 0<=k-i<n and k-i!=i else 0)
    # = a[k-i] * (1 + 1) if 0<=k-i<n and k!=i  ... hmm this is getting confusing
    # Let me just say: d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n and k != 2*i... no
    # Ugh, let me just be very explicit:
    # conv[k] = sum_{j in S} a[j] * a[k-j] where S = {j : 0<=j<n, 0<=k-j<n}
    # d(conv[k])/da[i] = sum_{j in S} [delta(j,i)*a[k-j] + delta(k-j,i)*a[j]]
    # = (if i in S: a[k-i]) + (if k-i in S: a[k-i])
    # Note: k-i in S means 0<=k-i<n and 0<=i<n (the second is always true)
    # So: d(conv[k])/da[i] = (a[k-i] if 0<=k-i<n) + (a[k-i] if 0<=k-i<n)
    # Wait that's wrong. Let me redo.
    # d(conv[k])/da[i] = sum_{j in S} [delta_{j,i} * a[k-j] + delta_{k-j,i} * a[j]]
    # First term: sum_{j in S} delta_{j,i} * a[k-j] = a[k-i] if i in S (i.e., 0<=i<n and 0<=k-i<n)
    # Second term: sum_{j in S} delta_{k-j,i} * a[j] = sum_{j in S} delta_{j,k-i} * a[j] = a[k-i] if k-i in S (i.e., 0<=k-i<n and 0<=i<n)
    # Since 0<=i<n always (i is our index), both conditions reduce to 0<=k-i<n.
    # So d(conv[k])/da[i] = 2 * a[k-i] if 0<=k-i<n, else 0.
    # EXCEPT when k-i = i (i.e., k=2i), in which case we're double-counting the same term a[i]*a[i].
    # When k=2i: the first term gives a[2i-i] = a[i], and the second term also gives a[2i-i] = a[i].
    # But in the sum conv[2i] = sum_j a[j]*a[2i-j], the term j=i appears only ONCE as a[i]*a[i].
    # So d(a[i]*a[i])/da[i] = 2*a[i], which is what we get (2*a[i]). So it's correct!
    
    # Therefore: d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    # And: grad[i] = sum_k p[k] * 2*a[k-i] * indicator(0<=k-i<n)
    # = 2 * sum_{k: 0<=k-i<n} p[k] * a[k-i]
    # = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]  (substituting j=k-i, so k=i+j)
    #   where we need i+j <= 2n-2 (since k goes up to 2n-2)
    
    # So grad[i] = 2 * sum_{j=0}^{min(n-1, 2n-2-i)} a[j] * p[i+j]
    # This is 2 * cross_correlation(a, p)[i]
    
    # Compute using FFT
    conv_len = 2 * n - 1
    fft_size = 1 << (n + conv_len - 2).bit_length()
    A = np.fft.rfft(a[::-1], fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * P, fft_size)
    grad = 2.0 * cross[n-1:n-1+n]
    
    # Handle the ReLU: if x[i] < 0, gradient should be 0
    grad[x < 0] = 0.0
    
    return val, grad

def coordinate_descent_step(a, max_iter=100):
    """One pass of cyclic coordinate descent."""
    n = len(a)
    conv = fast_autoconv(a)
    current_max = conv.max()
    
    for i in range(n):
        # For coordinate i, the terms in conv that involve a[i] are:
        # conv[k] for k in [i, i+n-1] (where a[k-i] is the paired element)
        # Specifically, conv[k] includes the term a[i]*a[k-i] (and a[k-i]*a[i] if k-i != i)
        # So conv[k] = base[k] + a[i]*a[k-i] + (a[k-i]*a[i] if k-i != i and k-i valid)
        # = base[k] + 2*a[i]*a[k-i] (if k != 2i)
        # = base[k] + a[i]^2 (if k = 2i)
        
        # Find the max of conv as a function of a[i] = t:
        # For each k in [i, min(i+n-1, 2n-2)]:
        #   if k == 2i: conv[k] = base[k] + t^2
        #   else: conv[k] = base[k] + 2*a[k-i]*t
        # For k outside this range: conv[k] = base[k] (unchanged)
        
        # We want to minimize max(conv) over t >= 0.
        
        # The max is the max of:
        # - Fixed values: max of conv[k] for k not in [i, i+n-1]
        # - Variable: max over k in [i, min(i+n-1, 2n-2)] of (base[k] + 2*a[k-i]*t) or (base[2i] + t^2)
        
        # This is a 1D convex minimization problem.
        # We can solve it by bisection on the optimal value, or by finding where the derivative changes sign.
        
        # Let's use a simple approach: evaluate at a few points and do bisection.
        
        fixed_max = 0.0
        # Max of conv[k] for k NOT involving a[i]
        # k involves a[i] if i <= k <= i+n-1 (and k <= 2n-2)
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        
        # Get fixed max
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_start:k_end+1] = False
        if mask.any():
            fixed_max = conv[mask].max()
        else:
            fixed_max = 0.0
        
        # Variable part: for k in [k_start, k_end]:
        # coeff[k-i] * t + base[k] where coeff depends on whether k==2i
        # Let's compute the "slope" and "intercept" for each relevant k
        
        # For efficiency, let's just do a golden section or bisection on t.
        # The function f(t) = max(fixed_max, max_{k in range} (b_k + c_k * t + q_k * t^2))
        # where q_k = 1 if k==2i else 0, c_k = 2*a[k-i] if k!=2i else 0, b_k = conv[k] - a[i]*a[k-i] - a[i]*a[k-i] (or - a[i]^2 if k==2i)
        
        # Actually, let's just compute base values:
        old_ai = a[i]
        # base[k] = conv[k] - contribution of a[i]
        # contribution of a[i] to conv[k] = 2*a[i]*a[k-i] if k!=2i, or a[i]^2 if k==2i
        
        # For the 1D problem, let's parameterize t = a[i] and find min_t f(t)
        # f(t) = max(fixed_max, max_{k in [k_start,k_end]} g_k(t))
        # where g_k(t) = base_k + 2*a[k-i]*t (for k != 2i)
        #              = base_k + t^2 (for k = 2i)
        
        # base_k = conv[k] - 2*a[i]*a[k-i] (for k != 2i)
        #         = conv[k] - a[i]^2 (for k = 2i)
        
        # Since f(t) is convex in t, we can use ternary search or bisection on derivative.
        
        # Simple approach: try a few values and pick the best
        candidates = [0.0, old_ai * 0.5, old_ai * 0.8, old_ai, old_ai * 1.2, old_ai * 1.5]
        # Also add the value where the max transitions
        best_t = old_ai
        best_val = current_max
        
        for t in candidates:
            if t < 0:
                t = 0.0
            # Compute new conv for affected k values
            new_conv_k = np.zeros(k_end - k_start + 1)
            for idx, k in enumerate(range(k_start, k_end + 1)):
                if k == 2 * i:
                    new_conv_k[idx] = conv[k] - old_ai * old_ai + t * t
                else:
                    new_conv_k[idx] = conv[k] - 2 * old_ai * a[k - i] + 2 * t * a[k - i]
            new_max = max(fixed_max, new_conv_k.max())
            if new_max < best_val:
                best_val = new_max
                best_t = t
        
        # Refine around best_t
        lo = max(0.0, best_t * 0.7)
        hi = best_t * 1.3
        for _ in range(10):
            mid = (lo + hi) / 2
            new_conv_k = np.zeros(k_end - k_start + 1)
            for idx, k in enumerate(range(k_start, k_end + 1)):
                if k == 2 * i:
                    new_conv_k[idx] = conv[k] - old_ai * old_ai + mid * mid
                else:
                    new_conv_k[idx] = conv[k] - 2 * old_ai * a[k - i] + 2 * mid * a[k - i]
            new_max = max(fixed_max, new_conv_k.max())
            if new_max < best_val:
                best_val = new_max
                best_t = mid
                hi = mid
            else:
                lo = mid
        
        a[i] = best_t
        # Update conv
        for k in range(k_start, k_end + 1):
            if k == 2 * i:
                conv[k] += best_t * best_t - old_ai * old_ai
            else:
                conv[k] += 2 * (best_t - old_ai) * a[k - i]
    
    return a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth max (log-sum-exp) approximation
with L-BFGS-B optimization, temperature continuation, coordinate descent polishing,
and multiplicative weight updates. Explores multiple sequence lengths.

import numpy as np

def fast_autoconv(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_logsumexp(conv, t):
    """Compute t*logsumexp(conv/t) and softmax weights."""
    m = conv.max()
    scaled = (conv - m) / t
    exp_scaled = np.exp(scaled)
    log_s = np.log(exp_scaled.sum())
    val = m + t * log_s
    p = exp_scaled / exp_scaled.sum()
    return val, p

def compute_gradient_fft(a, p):
    """Compute gradient of smooth max w.r.t. a.
    grad[i] = 2 * sum_j a[j] * p[i+j] (for valid indices).
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (n + conv_len - 2).bit_length()
    A = np.fft.rfft(a[::-1], fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * P, fft_size)
    return 2.0 * cross[n - 1:n - 1 + n]

def smooth_obj_and_grad(x, t):

def cd_polish(a, n_passes=100, time_limit=30.0, verbose=False):

    a = np.maximum(a.astype(np.float64), 0.0)

    n = len(a)

    t0 = time.monotonic()
    improvements = 0

        if time.monotonic() - t0 > time_limit:
            break
        improved_this_pass = False
        for i in range(n):
            old_ai = a[i]
            k_start = i

            # Fixed max
            if k_start > 0 or k_end < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_start:k_end + 1] = False
                fixed_max = conv[mask].max()
            else:
                fixed_max = 0.0

            # Find best t for coordinate i
            def eval_t(t):
                mv = fixed_max
                for k in range(k_start, k_end + 1):
                    ji = k - i
                    if k == 2 * i:
                        v = conv[k] - old_ai * old_ai + t * t
                    else:
                        v = conv[k] + 2.0 * (t - old_ai) * a[ji]
                    if v > mv:
                        mv = v
                return mv

            # Golden section search on [0, 2*old_ai + 0.1]
            hi = max(2.0 * old_ai + 0.05, 0.1)
            lo = 0.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d2 = lo + gr * (hi - lo)
            fc = eval_t(c)
            fd = eval_t(d2)
            for _ in range(20):
                if fc < fd:
                    hi = d2
                    d2 = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_t(c)
                else:
                    lo = c
                    c = d2
                    fc = fd
                    d2 = lo + gr * (hi - lo)
                    fd = eval_t(d2)

            best_t = (lo + hi) / 2.0
            best_f = eval_t(best_t)

            if best_f < best_max - 1e-14:

                diff = best_t - old_ai
                for k in range(k_start, k_end + 1):

                if new_max < best_max:

        max_c = conv.max()
        # Soft weights
        t_smooth = max_c * 0.01 + 1e-8
        p = np.exp((conv - max_c) / t_smooth)
        p /= p.sum()
        grad = compute_gradient_fft(a, p)
        # Multiplicative update
        a = a * np.exp(-eta * grad / grad.max() if grad.max() > 0 else 0.0)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

        val = fast_autoconv(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        eta *= 0.999

    return best_a

def subgradient_polish(a, time_limit=30.0, lr_init=0.002, mom=0.9):
    """Subgradient descent with momentum on the hard max."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    best_a = a.copy()
    best_val = fast_autoconv(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autoconv(a)
        max_c = conv.max()

        # Find all active (or near-active) peaks
        threshold = max_c - 1e-6
        active = np.where(conv >= threshold)[0]

        # Subgradient: average of subgradients at active peaks
        # d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += a[k - idx]
        sg *= 2.0 / len(active)

        v = mom * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a
        else:
            a_new = a_new / sn

        val = fast_autoconv(a_new).max()
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Reset with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

        lr *= 0.9997

    return best_a

def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(rc / rc.sum())

    # Dirichlet random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    bim = g1 + g2
    inits.append(bim / bim.sum())

    # If parent available, resample
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using smooth max L-BFGS-B,
    coordinate descent, multiplicative updates, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, current score={evaluate_sequence(parent.tolist()):.8f}")
            best_seq = parent.copy()
            best_score = evaluate_sequence(parent.tolist())
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Smooth L-BFGS-B with temperature continuation ===
    print("\n=== Phase 1: Smooth L-BFGS-B Optimization ===")

    n_parent = len(parent) if parent is not None else 384

    # Primary: optimize parent with smooth method
    if parent is not None:
        t_avail = min(deadline - time.monotonic() - 300, 500)
        if t_avail > 30:
            print(f"  Smooth L-BFGS-B on parent (n={n_parent}), budget={t_avail:.0f}s")
            a_opt, _ = optimize_smooth(parent, time_limit=t_avail, t_init=None,
                                       n_stages=10, max_iter_per_stage=40)
            try_seq(a_opt, "smooth_parent")

    # Try other initializations for parent length
    if parent is not None:
        inits = generate_structured_inits(n_parent, rng, parent)
        t_per = min(deadline - time.monotonic() - 200, 100) / max(1, len(inits) - 1)
        for i, f_init in enumerate(inits[1:], 1):  # Skip uniform (already covered)
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(t_per, rem - 10)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_smooth(f_init, time_limit=t_alloc, n_stages=6,
                                       max_iter_per_stage=25)
            try_seq(a_opt, f"smooth_i{i}")

    # Try different lengths
    other_lengths = [512, 256, 384, 640, 448, 320]
    if parent is not None:
        other_lengths = [l for l in other_lengths if l != n_parent]
    other_lengths = other_lengths[:4]  # Limit to 4 other lengths

    for n_try in other_lengths:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        t_alloc = min(60, rem - 20)
        if t_alloc < 15:
            break
        f_init = np.ones(n_try) / n_try
        # Add some structure
        x = np.arange(n_try)
        f_init = f_init * (1.0 + 0.3 * np.cos(2 * np.pi * x / n_try))
        f_init = np.maximum(f_init, 0)
        f_init = f_init / f_init.sum()

        a_opt, _ = optimize_smooth(f_init, time_limit=t_alloc, n_stages=6,
                                   max_iter_per_stage=30)
        try_seq(a_opt, f"smooth_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Coordinate Descent Polishing ===
    print(f"\n=== Phase 2: Coordinate Descent ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80

        # CD on current best
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD on best (n={n_best}), budget={t_cd:.0f}s")
            a_cd = cd_polish(best_seq, n_passes=200, time_limit=t_cd)
            try_seq(a_cd, "cd_best")

        # CD with perturbation
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            noise = rng.standard_normal(n_best) * 0.005
            a_pert = np.maximum(best_seq + noise, 0)
            a_pert = a_pert / a_pert.sum()
            t_cd = min(rem - 10, 80)
            a_cd = cd_polish(a_pert, n_passes=100, time_limit=t_cd)
            try_seq(a_cd, "cd_perturb")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Subgradient + Multiplicative Polish ===
    print(f"\n=== Phase 3: Subgradient & Multiplicative Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 40

        # Subgradient polish
        if rem > 20:
            t_sg = min(rem * 0.4, 80)
            a_sg = subgradient_polish(best_seq, time_limit=t_sg, lr_init=0.003, mom=0.92)
            try_seq(a_sg, "sg_polish")

        # Multiplicative updates
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_mu = min(rem - 10, 40)
            a_mu = multiplicative_updates(best_seq, n_iters=3000, eta_init=0.05,
                                          time_limit=t_mu)
            try_seq(a_mu, "mult_update")

        # Combined: CD after subgradient
        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_seq is not None:
            a_final = cd_polish(best_seq, n_passes=50, time_limit=min(rem - 5, 30))
            try_seq(a_final, "cd_final")

    # === Phase 4: Final Fine-tuning ===
    print(f"\n=== Phase 4: Final Fine-tuning ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10

        # Very fine CD
        if rem > 8:
            a_fine = cd_polish(best_seq, n_passes=500, time_limit=min(rem - 5, 60))
            try_seq(a_fine, "cd_fine")

        # Tiny perturbations + quick CD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for trial in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                a_p = np.maximum(best_seq + noise, 0)
                a_p = a_p / a_p.sum()
                a_r = cd_polish(a_p, n_passes=30, time_limit=max(2, rem - 2))
                try_seq(a_r, f"tiny_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
