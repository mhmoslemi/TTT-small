For the "linear" cross-correlation, we want:
g[j] = sum_{k=0}^{2n-2} w[k] * a[k-j] (where a[k-j] = 0 if k-j < 0 or k-j >= n)

The circular version will wrap around, giving extra terms. But if L is large enough (L >= 3n-2), there's no wrap-around for the indices we care about:
- k ranges from 0 to 2n-2
- k-j ranges from -j to 2n-2-j
- For j >= 0: the negative part (k-j < 0) corresponds to k < j, and in the circular version, a[(k-j) mod L] = a[L + k - j] which is 0 since L + k - j >= L > n (as long as L >= n + 2n-2 = 3n-2)

So with L >= 3n-2, the circular cross-correlation equals the linear cross-correlation for all j. 

So: g = ifft(fft(w, L) * conj(fft(a, L)))[:n] where L >= 3n-2.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def full_convolve_fft(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)[:size]
    return C

def compute_gradient_fft(a, w, L):
    """Compute g[j] = sum_k w[k] * a[k-j] using FFT (cross-correlation)."""
    n = len(a)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(W * np.conj(A), L)[:n]
    return C

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_optimize(a0, time_limit, verbose=False):
    """Optimize using projected subgradient with soft-max peaking."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    L = 1 << (3 * n - 2 - 1).bit_length()  # FFT size for cross-correlation
    
    conv = full_convolve_fft(a)
    peak = conv.max()
    best_obj = 2.0 * n * peak
    best_a = a.copy()
    
    # Optimizer state
    a_prev = a.copy()
    momentum = 0.9
    lr = 0.05
    tau = 0.05  # Soft-max temperature (fraction of peak)
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        conv = full_convolve_fft(a)
        peak = conv.max()
        obj = 2.0 * n * peak
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute soft-max weights
        tau_val = max(tau * peak, 1e-10)
        log_w = (conv - peak) / tau_val
        log_w -= log_w.max()  # numerical stability
        w = np.exp(log_w)
        w /= w.sum()
        
        # Compute gradient via FFT
        g = compute_gradient_fft(a, w, L)
        
        # Update with momentum
        velocity = a - a_prev
        a_prev = a.copy()
        a_new = a - lr * g + momentum * velocity
        
        # Project to simplex
        a = project_simplex(a_new)
        
        # Adaptive parameters
        if it % 100 == 0:
            lr *= 0.98
            tau = max(0.02, tau * 0.99)

            rng = np.random.default_rng(it)
            scale = 0.01 * (1 + it / 10000)
            a = np.maximum(a + rng.standard_normal(n) * scale, 0.0)

            a_prev = a.copy()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            lr = max(0.001, lr * 1.5)  # increase lr on restart

            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.6f}, peak={peak:.6e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

**Approach: Water-filling / Balanced convolution**

The idea is to iteratively adjust the sequence so that the convolution becomes more balanced (flat). This is similar to the "iterative proportional fitting" (IPF) or "Sinkhorn" algorithm.

Algorithm:
1. Start with a in the simplex
2. Compute conv = a * a
3. Compute target = mean(conv) = 1/(2n-1)
4. For each lag k where conv[k] > target:
   - Reduce the contributing a[j] values (j = k-n+1 to k, intersected with [0,n-1])
   - Distribute the reduction proportionally to a[j]
5. For each lag k where conv[k] < target:
   - Increase the contributing a[j] values
   - Distribute the increase proportionally
6. Project to simplex
7. Repeat

This is like a multiplicative weight update. It directly targets the flattening of the convolution.

"""
Hybrid subgradient + LP optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated soft-max subgradient descent with momentum for fast iteration,
combined with LP-based Frank-Wolfe for final polishing. Multi-length, multi-start
search with adaptive restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def full_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:size]


def gradient_from_weights(a, w, L):
    """Compute g[j] = sum_k w[k]*a[k-j] via FFT cross-correlation."""
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(W * np.conj(A), L)[:len(a)]


def project_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    nz = np.nonzero(cond)[0]
    if len(nz) == 0:
        return np.ones(n) / n
    rho = nz[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def sg_optimize(a0, time_limit, lr_init=0.02, momentum=0.92, tau_init=0.05,
                verbose=False, seed=0):
    """Projected subgradient with soft-max peaking and momentum."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0.0))
    L = 1 << (3 * n - 1).bit_length()

    conv = full_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    a_prev = a.copy()
    lr = lr_init
    tau = tau_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stall_limit = 400
    rng = np.random.default_rng(seed)

        conv = full_convolve(a)
        peak = conv.max()

            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Soft-max weights on convolution peaks
        tv = max(tau * peak, 1e-12)
        lw = (conv - peak) / tv
        lw -= lw.max()
        w = np.exp(lw)
        w /= w.sum()

        # Gradient via FFT
        g = gradient_from_weights(a, w, L)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Momentum update
        vel = a - a_prev
        a_prev = a.copy()
        a_new = a - lr * g + momentum * vel
        a = project_simplex(a_new)

        # Adaptive parameters
        if it % 500 == 0 and it > 0:
            lr = max(0.001, lr * 0.997)
            tau = max(0.01, tau * 0.999)

            print(f"    SG it={it}, best={best_obj:.6f}, el={elapsed:.1f}s")

    return best_a, best_obj

    """Build sparse matrix for conv(a, b) as a linear function of b."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
