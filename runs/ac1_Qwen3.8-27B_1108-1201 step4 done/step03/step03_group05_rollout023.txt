"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def correct_gradient(a, p, n):
    """
    Correct gradient of (1/beta)*logsumexp(beta*c) w.r.t. a.
    grad[i] = 2 * sum_m a[m] * p[i+m] (cross-correlation of a with p)
    Using FFT: IFFT(conj(FFT(a)) * FFT(p))
    """
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def smoothed_obj_value(a, beta):
    """Compute the smoothed log-sum-exp objective value."""
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    val = scaled.max() + np.log(np.exp(scaled).sum()) / beta
    return val


def optimize_smoothed_correct(a0, time_limit, beta_init=50.0, beta_end=500.0,
                              lr_init=0.01, momentum=0.9,
                              perturb_every=300, perturb_scale=0.003, seed=0):
    """
    Nesterov accelerated projected gradient descent with CORRECT gradient.
    Input: a0 on simplex (sum=1, non-negative).
    Returns (best_a, best_max_conv).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * (frac ** 0.7)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        conv_y = fast_convolve(y)

        # Compute softmax probabilities
        scaled = beta * conv_y
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p_sum = exp_vals.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv_y)
            p[np.argmax(conv_y)] = 1.0
        else:
            p = exp_vals / p_sum

        # Correct gradient
        grad = correct_gradient(y, p, n)

        # Momentum update
        v = momentum * v - lr_init * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation

            ns = perturb_scale * (1.0 + 0.3 * np.sin(total_iters * 0.007))
            noise = rng.standard_normal(n) * ns
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)

    return best_a, best_obj


def peak_equalization(a0, time_limit, seed=0):
    """
    Targeted peak reduction: iteratively identify top peaks and redistribute
    mass from high-contribution pairs to low-contribution positions.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        current_max = conv.max()

        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 200:
            # Global perturbation
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            v_reset = True
            stagnation = 0
            continue

        # Find top peaks
        n_peaks = min(8, 2 * n - 1)
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_idx = peak_idx[conv[peak_idx] > current_max * 0.9]
        peak_idx = sorted(peak_idx, key=lambda k: -conv[k])

        improved = False
        for k in peak_idx[:4]:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            if i_start > i_end:
                continue

            # Find pairs contributing to this peak
            contributions = []
            for i in range(i_start, i_end + 1):
                contributions.append((a[i] * a[k - i], i, k - i))

            if not contributions:
                continue

            contributions.sort(key=lambda x: -x[0])

            # Try reducing the top-contributing pair
            for contrib, i, j in contributions[:3]:
                if contrib < 1e-12:
                    continue

                # Try to move mass from the larger element to a "safe" position
                src = max(i, j, key=lambda x: a[x])
                amt = a[src] * 0.05

                # Try several target positions
                for _ in range(5):
                    dst = rng.integers(0, n)
                    if abs(dst - src) < 1:
                        continue

                    if a[dst] > a[src] * 1.5:
                        continue

                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try

                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-12:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

                if improved:
                    break

            if improved:
                break

        # Also try small random adjustments if stuck
        if not improved and stagnation % 50 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = min(a[i], a[j]) * 0.03
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-12:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()

    return best_a, best_obj


def coordinate_refine(a0, time_limit, seed=0):
    """
    Fine-grained coordinate descent: try small adjustments to individual
    or paired coordinates to reduce the max convolution.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        current_max = conv.max()
        max_idx = np.argmax(conv)

        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_a = a.copy()

        # For the peak at max_idx, try reducing contributions
        k = int(max_idx)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)

        # Find the element with highest "coupling" at the peak
        if i_start <= i_end:
            # a[i] * a[k-i] is the contribution; gradient of c_k w.r.t. a[i] is 2*a[k-i]
            # To reduce c_k, decrease a[i] where a[k-i] is large
            coupling = np.array([a[k - i] for i in range(i_start, i_end + 1)])
            order = np.argsort(-coupling)

            for idx in order[:3]:
                i = i_start + int(idx)
                j = k - i

                # Try moving a small amount from i to a random other position
                for _ in range(4):
                    dst = rng.integers(0, n)
                    if dst == i or dst == j:
                        continue

                    for frac in [0.02, 0.05, 0.1]:
                        amt = a[i] * frac
                        if amt < 1e-10:
                            continue

                        a_try = a.copy()
                        a_try[i] -= amt
                        a_try[dst] += amt
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try

                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj - 1e-13:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Bell / raised cosine
    tri = 1.0 / (1.0 + 0.3 * ((x - n / 2) / (n / 4)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    # Various Gaussians
    for sf in [0.15, 0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # sinc-like (based on the theoretical optimal shape)
    sinc = np.sinc(x / n - 0.5)
    sinc = np.maximum(sinc, 0.01)
    sinc /= sinc.sum()
    inits['sinc'] = sinc

    # Random Dirichlet
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add random restarts at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.5)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 2.0)))

    remaining = deadline - time.monotonic() - 100.0
    n_configs = len(top_configs)
    per_config = max(20.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(per_config, 250.0)
        if t_alloc < 15:
            break

        print(f"  Config {i}: n={n}, time={t_alloc:.0f}s", flush=True)

        # Sub-phase A: Smoothed gradient (55%)
        t_a = t_alloc * 0.55
        a_a, obj_a = optimize_smoothed_correct(
            a0, t_a, beta_init=50, beta_end=400,
            lr_init=0.008, momentum=0.92,
            perturb_every=250, perturb_scale=0.003,
            seed=int(rng.integers(0, 100000)))

        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    Grad: {score_a:.6f}", flush=True)

        # Sub-phase B: Peak equalization (25%)
        t_b = t_alloc * 0.25
        if t_b > 8:
            a_b, obj_b = peak_equalization(a_a, t_b, seed=int(rng.integers(0, 100000)))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    PeakEq: {score_b:.6f}", flush=True)
            a_a = a_b

        # Sub-phase C: Fine gradient (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_c, obj_c = optimize_smoothed_correct(
                a_a, t_c, beta_init=200, beta_end=800,
                lr_init=0.003, momentum=0.95,
                perturb_every=200, perturb_scale=0.001,
                seed=int(rng.integers(0, 100000)))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Refine: {score_c:.6f}", flush=True)

    # Phase 3: Final polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0

        # Coordinate refinement
        t_coord = remaining * 0.30
        if t_coord > 10:
            a_cr, obj_cr = coordinate_refine(best_sequence, t_coord, seed=int(rng.integers(0, 100000)))
            score_cr = float(evaluate_sequence(a_cr.tolist()))
            if np.isfinite(score_cr) and score_cr < best_score:
                best_score = score_cr
                best_sequence = a_cr.copy()
                print(f"  Coord: {best_score:.6f}", flush=True)

        # High-beta gradient
        remaining = deadline - time.monotonic() - 10.0
        t_hb = remaining * 0.35
        if t_hb > 5:
            a_hb, obj_hb = optimize_smoothed_correct(
                best_sequence, t_hb, beta_init=500, beta_end=2000,
                lr_init=0.001, momentum=0.97,
                perturb_every=100, perturb_scale=0.001,
                seed=int(rng.integers(0, 100000)))
            score_hb = float(evaluate_sequence(a_hb.tolist()))
            if np.isfinite(score_hb) and score_hb < best_score:
                best_score = score_hb
                best_sequence = a_hb.copy()
                print(f"  HighBeta: {best_score:.6f}", flush=True)

        # Final peak equalization
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 8:
            a_pe, obj_pe = peak_equalization(best_sequence, remaining, seed=int(rng.integers(0, 100000)))
            score_pe = float(evaluate_sequence(a_pe.tolist()))
            if np.isfinite(score_pe) and score_pe < best_score:
                best_score = score_pe
                best_sequence = a_pe.copy()
                print(f"  FinalPeak: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
