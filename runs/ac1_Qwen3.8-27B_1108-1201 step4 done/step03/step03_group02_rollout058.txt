"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Smoothed log-sum-exp gradient descent with Nesterov momentum on the simplex
2. LP-based iterative refinement (max-sum and min-peak variants)
3. Exact subgradient polishing on the non-smooth max objective
Multiple lengths and initializations are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def raw_objective(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_grad(a, tau):
    """
    Smoothed objective and gradient on the simplex.
    F(a) = (2n/tau) * log(sum_k exp(tau * conv(a,a)[k]))
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    shifted = tau * (conv - m)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    p = exp_vals / Z
    F = (2.0 * n / tau) * np.log(Z) + 2.0 * n * m
    
    # Gradient: 4n * sum_k p_k * a[k-i] for valid k-i
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        length = k_hi - k_lo + 1
        if length > 0:
            grad[i] = 4.0 * n * np.dot(p[k_lo:k_hi + 1], a[:length])
    return F, grad

def nesterov_smoothed_descent(a0, time_limit, tau_init=50.0, tau_decay=0.9995,
                                lr_init=0.01, momentum_init=0.9):
    """
    Nesterov momentum + projected gradient on the simplex
    with log-sum-exp smoothing and annealing.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    momentum = momentum_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)
        
        F, grad = smooth_obj_grad(y, tau)
        
        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_proj
        a_new = a + v
        a_new = project_simplex(a_new)
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Annealing
        if it % 100 == 0:
            tau = max(tau * 0.995, 2.0)
        if no_improve > 500:
            lr *= 0.7
            no_improve = 0
        if no_improve > 3000:
            break
        if lr < 1e-8:
            break
    
    return best_a, best_obj

def build_conv_matrix(a):
    """Build the convolution constraint matrix for the LP."""
    n = len(a)
    num_rows = 2 * n - 1
    # For efficiency, use a sparse-like construction
    # (A @ b)[k] = sum_j a[k-j] * b[j] for valid j
    # This is a banded Toeplitz-like matrix
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def lp_refine_max_sum(a, M, max_iters=200, step=0.3, time_limit=60):
    """
    LP-based refinement: iteratively solve max sum(b) s.t. b>=0, conv(a,b)<=M,
    then blend. Returns improved sequence.
    """
    n = len(a)
    a_cur = np.maximum(a, 0.0).astype(np.float64)
    s = a_cur.sum()
    if s < 1e-15:
        return a
    a_cur = a_cur / s  # Normalize to simplex
    
    conv = fast_convolve(a_cur)
    best_obj = 2.0 * n * conv.max()
    best_a = a_cur.copy()
    
    A = build_conv_matrix(a_cur)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_cur = conv.max()
        
        # Solve: max sum(b) s.t. A_ub @ b <= M_cur, b >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_cur)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Check convergence
        if np.max(np.abs(b - a_cur)) < 1e-12:
            break
        
        # Blend
        a_new = (1.0 - step) * a_cur + step * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a_cur = a_new
        conv = conv_new
        
        # Update the LP matrix since a changed
        if it % 5 == 0:
            A = build_conv_matrix(a_cur)
    
    return best_a

def lp_refine_min_peak(a, max_iters=200, time_limit=60):
    """
    LP-based refinement: iteratively minimize peak of conv(a,b) with sum(b)=1.
    """
    n = len(a)
    a_cur = np.maximum(a, 0.0).astype(np.float64)
    s = a_cur.sum()
    if s < 1e-15:
        return a
    a_cur = a_cur / s
    
    conv = fast_convolve(a_cur)
    best_obj = 2.0 * n * conv.max()
    best_a = a_cur.copy()
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = build_conv_matrix(a_cur)
        num_rows = 2 * n - 1
        
        # Solve: min M s.t. sum(b)=1, b>=0, A@b <= M
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((num_rows, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        b = np.maximum(res.x[:n], 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        if np.max(np.abs(b - a_cur)) < 1e-12:
            break
        
        a_new = 0.5 * a_cur + 0.5 * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a_cur = a_new
        conv = conv_new
    
    return best_a

def subgradient_polish(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=20,
                       temp_init=0.001):
    """
    Subgradient method on the exact (non-smooth) max objective.
    Focuses on the top-k peaks with temperature-weighted subgradient.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Compute weighted subgradient over top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        
        if temp > 1e-15:
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = 2.0 * w[i]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]
        
        # Project gradient
        sg_proj = sg - np.mean(sg)
        
        v = momentum * v - lr * sg_proj
        a_new = project_simplex(a + v)
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.9995
        temp *= 0.999
        
        if no_improve > 2000:
            # Perturbation
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(lr_init * 0.2, lr * 0.8)
    
    return best_a

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0)
        else:
            p = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps
    
    for freq in [1, 2, 3]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            inits[f'cos_{freq}'] = a / a_sum
    
    sigma = 0.25
    g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Bimodal
    b = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
        np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = b / b.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Nesterov accelerated projected gradient with log-sum-exp smoothing and annealing
2. LP-based iterative refinement (max-sum and min-peak variants)
3. Exact subgradient polishing focused on active peaks
4. Multi-length exploration with diverse initializations
All methods work on the probability simplex for scale invariance.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def raw_objective(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, tau):
    """
    Smoothed objective and gradient on the simplex (sum(a)=1 assumed).
    F(a) = (2n/tau) * log(sum_k exp(tau * conv(a,a)[k]))
    Gradient: 4n * sum_k p_k * a[k-i] for valid k-i.
    """
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    shifted = tau * (conv - m)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    p = exp_vals / Z
    F = (2.0 * n / tau) * np.log(Z) + 2.0 * n * m

    grad = np.zeros(n)
    for i in range(n):
        k_hi = min(2 * n - 2, i + n - 1)
        length = k_hi - i + 1
        if length > 0:
            grad[i] = 4.0 * n * np.dot(p[i:k_hi + 1], a[:length])
    return F, grad


def nesterov_smoothed_descent(a0, time_limit, tau_init=50.0,
                              lr_init=0.005, momentum_init=0.9):
    """
    Nesterov momentum + projected gradient on the simplex
    with log-sum-exp smoothing and adaptive annealing.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    momentum = momentum_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr_min = lr_init * 1e-3

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)

        F, grad = smooth_obj_grad(y, tau)

        # Project gradient onto tangent space of simplex
        grad_proj = grad - np.mean(grad)

        v = momentum * v - lr * grad_proj
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive annealing
        if it % 50 == 0 and tau > 3.0:
            tau *= 0.997
        if no_improve > 300:
            lr *= 0.75
            no_improve = 0
        if no_improve > 2000:
            break
        if lr < lr_min:
            break

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        k_start = j
        k_end = min(num_rows, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine_max_sum(a0, time_limit, step=0.3, max_iters=500):
    """
    LP-based refinement: iteratively solve max sum(b) s.t. b>=0, conv(a,b)<=M.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    A = build_conv_matrix(a)
    t0 = time.monotonic()
    no_lp_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M_cur = conv.max()
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_cur)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if res.status != 0:
            break
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        diff = np.max(np.abs(b - a))
        if diff < 1e-11:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_lp_improve = 0
        else:
            no_lp_improve += 1

        a = a_new
        conv = conv_new

        if it % 3 == 0:
            A = build_conv_matrix(a)

        if no_lp_improve > 100:
            step *= 0.8
            if step < 0.05:
                break

    return best_a


def lp_refine_min_peak(a0, time_limit, max_iters=500):
    """
    LP-based refinement: iteratively minimize max_k conv(a,b)[k] s.t. sum(b)=1.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = build_conv_matrix(a)
        num_rows = 2 * n - 1

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((num_rows, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if res.status != 0:
            break
        b = np.maximum(res.x[:n], 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.max(np.abs(b - a)) < 1e-11:
            break

        a_new = 0.5 * a + 0.5 * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        if no_improve > 80:
            break

    return best_a


def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.92,
                       top_k=30, temp_init=0.0005):
    """
    Subgradient method on the exact max objective with temperature-weighted
    peaks and periodic perturbation to escape local minima.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(777)
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]

        if temp > 1e-18:
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = 2.0 * w[i]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        sg_proj = sg - np.mean(sg)

        v = momentum * v - lr * sg_proj
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9997
        temp *= 0.9995

        if no_improve > 3000 + perturb_count * 2000:
            scale = 0.002 * (1.0 + 0.5 * perturb_count)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_count += 1
            lr = max(lr_init * 0.3, lr)

        if lr < 1e-9:
            break

    return best_a


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0)
        else:
            from numpy import interp as np_interp
            p = np_interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps

    for freq in [1, 2, 3, 5]:
        a_c = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a_c = np.maximum(a_c, 0.0)
        sc = a_c.sum()
        if sc > 0:
            inits[f'cos_{freq}'] = a_c / sc

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    b = (np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2) +
         np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2))
    inits['bimodal'] = b / b.sum()

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    inits['tri'] = tri / tri.sum()

    return inits

                a_p, time_limit=t_alloc, tau_init=tau_val,
                lr_init=0.004, momentum_init=0.9)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: LP-based refinement ===
    print("=== Phase 2: LP Refinement ===")
    t2 = time.monotonic()

    if best_sequence is not None:
        configs = [
            ('max_sum', 0.4), ('max_sum', 0.25), ('max_sum', 0.6),
            ('min_peak', 0.5), ('min_peak', 0.5),
        ]
        for method, step_val in configs:
            t_alloc = min(80.0, deadline - time.monotonic() - 500)
            if t_alloc < 10:
                break
            if method == 'max_sum':
                a_opt = lp_refine_max_sum(
                    best_sequence, time_limit=t_alloc, step=step_val,
                    max_iters=400)
            else:
                a_opt = lp_refine_min_peak(
                    best_sequence, time_limit=t_alloc, max_iters=400)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Subgradient polish ===
    print("=== Phase 3: Subgradient Polish ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        for lr_val in [0.004, 0.002, 0.008]:
            t_alloc = min(60.0, deadline - time.monotonic() - 300)
            if t_alloc < 10:
                break
            a_opt = subgradient_polish(
                best_sequence, time_limit=t_alloc, lr_init=lr_val,
                momentum=0.93, top_k=40, temp_init=0.0003)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Explore other lengths ===
    print("=== Phase 4: Other Lengths ===")
    t4 = time.monotonic()

    for n_try in [256, 1024]:
        t_remain = deadline - time.monotonic() - 200
        if t_remain < 50:
            break
        t_alloc = min(80.0, t_remain)
        if t_alloc < 20:
            break

        inits = generate_inits(n_try, parent, rng)
        init_names = [nm for nm in ['uniform', 'parent', 'cos_1', 'random']
                      if nm in inits]
        t_per = t_alloc / max(1, len(init_names))

        for name in init_names:
            if t_per < 8:
                break
            a_opt, _ = nesterov_smoothed_descent(
                inits[name], time_limit=t_per, tau_init=60.0,
                lr_init=0.006, momentum_init=0.9)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Deep LP polish on best ===
    print("=== Phase 5: Deep LP Polish ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        # Alternating LP with very small steps
        for _ in range(3):
            t_alloc = min(60.0, deadline - time.monotonic() - 50)
            if t_alloc < 8:
                break
            a_opt = lp_refine_max_sum(
                best_sequence, time_limit=t_alloc * 0.5, step=0.15,
                max_iters=800)
            try_update(a_opt)
            t_alloc = min(50.0, deadline - time.monotonic() - 40)
            if t_alloc < 5:
                break
            a_opt = lp_refine_min_peak(
                best_sequence, time_limit=t_alloc, max_iters=800)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Final micro-polish ===
    print("=== Phase 6: Micro-Polish ===")
    t6 = time.monotonic()

    if best_sequence is not None:
        # Very low temperature gradient
        t_alloc = min(40.0, deadline - time.monotonic() - 15)
        if t_alloc > 5:
            a_opt, _ = nesterov_smoothed_descent(
                best_sequence, time_limit=t_alloc, tau_init=500.0,
                lr_init=0.001, momentum_init=0.95)
            try_update(a_opt)

        # Tiny subgradient steps
        t_alloc = min(30.0, deadline - time.monotonic() - 10)
        if t_alloc > 3:
            a_opt = subgradient_polish(
                best_sequence, time_limit=t_alloc, lr_init=0.001,
                momentum=0.95, top_k=60, temp_init=0.0001)
            try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, t={time.monotonic()-t6:.0f}s")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        print("WARNING: Using fallback uniform sequence")

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}, "
          f"total_time={time.monotonic()-(deadline-budget_s+30):.0f}s")
    return [float(x) for x in best_sequence]
