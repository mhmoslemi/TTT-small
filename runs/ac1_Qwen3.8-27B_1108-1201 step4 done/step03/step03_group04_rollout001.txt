"""
Improved LP-based Frank-Wolfe with top-K constraint selection, finer line search,
softmax gradient polishing, and strategic restarts to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: (1) top-K LP oracle for speed, (2) 30-point line search,
(3) entropic/softmax gradient polish phase, (4) better time management and restarts.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear (full) convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    theta = (cv[i] - 1.0) / (i + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def build_conv_matrix_rows(a, row_indices):
    """Build sparse matrix A for selected rows: (A @ b)[i] = conv(a,b)[row_indices[i]]."""
    n = len(a)
    K = len(row_indices)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(row_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def lp_oracle_topk(a, M, K=80, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for top-K peaks, b >= 0."""
    n = len(a)
    conv_a = fast_convolve(a)
    # Select top K peaks
    if K >= len(conv_a):
        top_k_idx = np.arange(len(conv_a))
    else:
        top_k_idx = np.argpartition(conv_a, -K)[-K:]
    top_k_idx = np.sort(top_k_idx)
    
    A_mat = build_conv_matrix_rows(a, top_k_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_k_idx), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def line_search(a, b, n_points=30):
    """Find best t in [0,1] for a_t = (1-t)*a + t*b minimizing the objective."""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Include t=0 and t=1
    ts = np.linspace(0.0, 1.0, n_points)
    for t in ts:
        candidate = (1.0 - t) * a + t * b
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def fpw_optimize(a0, time_limit, verbose=True, verbose_every=100, K=80, max_stall=150):
    """Frank-Wolfe with top-K LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()

        a = a / s

    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP oracle
        b = lp_oracle_topk(a, M, K=K, time_limit=2.0)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        # Check if stuck (direction is same as current)
        if np.allclose(a, b, atol=1e-12):
            no_improve += 1
        else:
            t, obj_new = line_search(a, b, n_points=30)
            if t < 0.005:
                no_improve += 1
            else:
                a_new = (1.0 - t) * a + t * b
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                a = a_new
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-15:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            conv = fast_convolve(a)

def softmax_polish(a0, time_limit, verbose=True, verbose_every=500):
    """Polish using entropic/softmax gradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    
    # Precompute FFT size for cross-correlation
    fft_len = 1 << (3 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Anneal: lambda increases over time
        progress = (time.monotonic() - t0) / time_limit
        lam = 50.0 + 500.0 * progress ** 1.5
        
        # Softmax weights over peaks
        shifted = conv - M
        exp_vals = np.exp(lam * shifted)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / w_sum
        
        # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
        # Use FFT for efficiency
        W = np.fft.rfft(w, fft_len)
        A = np.fft.rfft(a, fft_len)
        corr = np.fft.irfft(W * np.conj(A), fft_len)
        # corr[k] = sum_n w[n+k] * a[n] (with zero-padding interpretation)
        # We want g[j] = 2 * sum_k w[k] * a[k-j] = 2 * corr[j] (for j in [0, n-1])
        # Actually: corr[j] = sum_m w[m+j]*a[m] = sum_k w[k]*a[k-j] (with k=m+j)
        # Yes, that's what we want
        g = 2.0 * corr[:n]
        
        # Normalize gradient to have unit norm for stable updates
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Check objective
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
        
        # Decay learning rate
        lr *= 0.9997
        it += 1
        
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    Softmax it={it}, best_obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def topk_subgradient_polish(a0, time_limit, K=10, verbose=True, verbose_every=200):
    """Polish using subgradient of top-K peaks with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    y = a.copy()  # Nesterov momentum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        t_k = 2.0 / (it + 2.0)
        y = (1.0 - t_k) * a + t_k * best_a  # simplified
        
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # Top-K peaks
        top_k_idx = np.argpartition(conv_y, -K)[-K:]
        peak_vals = conv_y[top_k_idx]
        
        # Weights: exp((peak - M_y) / T)
        T = max(0.01 * M_y, 1e-10)
        weights = np.exp((peak_vals - M_y) / T)
        weights /= weights.sum()
        
        # Subgradient: g[j] = 2 * sum_{k in topK} w_k * a_y[k-j]
        g = np.zeros(n)
        for i, k in enumerate(top_k_idx):
            wid = weights[i]
            if wid < 1e-15:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                g[j] += 2.0 * wid * y[k - j]
        
        # Step size (Polyak-like)
        lr = 2.0 / (it + 10.0)
        a_new = y - lr * g
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it}, best_obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP-FW optimization with top-K constraints, followed by
    softmax gradient polish and subgradient refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    # Determine starting point
    if parent is not None:
        n_start = len(parent)
        a_start = parent / parent.sum()
        print(f"Starting from parent: n={n_start}")
    else:
        n_start = 384
        a_start = np.ones(n_start) / n_start
        print(f"No parent, using uniform init: n={n_start}")
    
    # Evaluate starting point
    score = float(evaluate_sequence(a_start.tolist()))
    print(f"Initial score: {score:.8f}")
    if np.isfinite(score):
        best_score = score
        best_sequence = a_start.copy()
    
    n = len(a_start)
    
    # === Phase 1: LP-FW main optimization ===
    print("\n=== Phase 1: LP-FW Optimization ===")
    t_phase1 = min(1000.0, (deadline - time.monotonic()) * 0.55)
    if t_phase1 > 10:
        a_opt, obj_opt = fpw_optimize(a_start, time_limit=t_phase1,
                                      verbose=True, verbose_every=100, K=80)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Phase 1 score={score:.8f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
    
    # === Phase 2: Softmax gradient polish ===
    print("\n=== Phase 2: Softmax Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0
        t_phase2 = min(400.0, remaining * 0.4)
        if t_phase2 > 10:
            a_opt, obj_opt = softmax_polish(best_sequence, time_limit=t_phase2,
                                            verbose=True, verbose_every=500)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 3: Top-K Subgradient Polish ===
    print("\n=== Phase 3: Subgradient Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0
        t_phase3 = min(300.0, remaining * 0.4)
        if t_phase3 > 10:
            a_opt, obj_opt = topk_subgradient_polish(best_sequence, time_limit=t_phase3,
                                                     K=8, verbose=True, verbose_every=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 4: More LP-FW on best ===
    print("\n=== Phase 4: LP-FW Refinement ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0
        t_phase4 = min(200.0, remaining * 0.5)
        if t_phase4 > 10:
            a_opt, obj_opt = fpw_optimize(best_sequence, time_limit=t_phase4,
                                          verbose=True, verbose_every=200, K=100)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 5: Perturbation restarts ===
    print("\n=== Phase 5: Perturbation Restarts ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_restarts = max(1, int(remaining // 25))
        n_restarts = min(n_restarts, 10)
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            # Different perturbation scales
            scales = [0.001, 0.003, 0.005, 0.01, 0.02, 0.002, 0.004, 0.008, 0.015, 0.001]
            scale = scales[i % len(scales)]
            noise = rng.standard_normal(n) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            
            t_alloc = min(20.0, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            
            a_opt, obj_opt = fpw_optimize(a_pert, time_limit=t_alloc,
                                          verbose=False, K=60, max_stall=50)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.8f} (improved!)")
                else:
                    print(f"  Restart {i}: score={score:.8f} (no improvement)")
    
    # === Phase 6: Final softmax polish on best ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 5:
            a_opt, obj_opt = softmax_polish(best_sequence, time_limit=remaining,
                                            verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final polish: score={score:.8f} (improved!)")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
