def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Symmetry enforcement (p_j = p_{n-1-j})
3. LP-based directional optimization with symmetry-reduced constraints
4. Coordinate descent polishing
5. Multi-length search (n = 512 to 2048)

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def symmetrize(p):
    """Enforce p_j = p_{n-1-j}."""
    return (p + p[::-1]) / 2.0


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_sym(v, n):
    """Project onto symmetric simplex: {p >= 0, sum(p)=1, p_j = p_{n-1-j}}."""
    # Project onto full simplex, then symmetrize, then renormalize
    p = project_simplex(v)
    p = symmetrize(p)
    s = p.sum()
    if s > 1e-15:
        p /= s
    else:
        p = np.ones(n) / n
    return p


def smooth_objective(p, eps):
    """Compute eps * log(sum exp(c_k/eps)) where c = p*p."""
    c = fast_convolve(p)
    c_max = c.max()
    # Numerical stability
    return c_max + eps * np.log(np.sum(np.exp((c - c_max) / eps)))


def smooth_gradient(p, w):
    """
    Compute gradient of eps*log(sum exp(c_k/eps)) w.r.t. p.
    w = softmax(c/eps), i.e., w_k = exp(c_k/eps) / sum exp(c_k/eps)
    grad_j = sum_k w_k * (p_{k-j} + p_{j-k})
    """
    n = len(p)
    # w has length 2n-1
    # term1[j] = sum_k w[k] * p[k-j] (cross-correlation)
    # term2[j] = sum_k w[k] * p[j-k] (convolution)
    
    N = 1
    while N < 4 * n:
        N *= 2
    
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p, N)
    
    # term1: cross-correlation = IFFT(W * conj(P))
    term1 = np.fft.irfft(W * np.conj(P), N)[:n]
    
    # term2: convolution = IFFT(W * P)
    term2 = np.fft.irfft(W * P, N)[:n]
    
    return term1 + term2


def gradient_descent_smooth(p_init, time_limit, sym=True):
    """
    Gradient descent on the simplex with log-sum-exp smoothing.
    Uses adaptive epsilon (starts large, decreases).
    """
    n = len(p_init)
    p = np.maximum(p_init.astype(np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p /= s
    if sym:
        p = symmetrize(p)
        p = p / p.sum()
    
    t0 = time.monotonic()
    eps = n * 0.1  # Initial temperature
    eps_min = 0.01
    eps_decay = 0.995
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(p)
        c_max = c.max()
        
        # Compute softmax weights
        c_shifted = (c - c_max) / eps
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        
        # Compute gradient
        grad = smooth_gradient(p, w)
        
        # Momentum update
        v = momentum * v - lr * grad
        p_new = p + v
        
        if sym:
            p_new = symmetrize(p_new)
        
        p_new = project_simplex(p_new)
        
        # Check objective
        c_new = fast_convolve(p_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
            p = p_new
        else:
            no_improve += 1
            #Revert
            p = p  # keep old p
            v *= 0.5
        
        # Adaptive epsilon
        if no_improve > 20:
            eps = max(eps_min, eps * eps_decay)
            no_improve = 0
        
        # Decay learning rate
        lr *= 0.999
        
        if lr < 0.0001:
            break
    
    return best_p, 2.0 * n * best_obj

"""

2. Symmetry enforcement (p_j = p_{n-1-j}) to halve search space

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def symmetrize(p):
    """Enforce p_j = p_{n-1-j}."""
    return (p + p[::-1]) * 0.5


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_sym(v, n):
    """Project onto symmetric probability simplex."""
    p = project_simplex(v)
    p = symmetrize(p)
    s = p.sum()
    if s > 1e-15:
        p /= s
    else:
        p = np.ones(n) / n
    return p


def smooth_gradient(p, w):
    """
    Compute gradient of eps*log(sum exp(c_k/eps)) w.r.t. p.
    w = softmax(c/eps) with length 2n-1.
    Result: grad_j = 2 * sum_m w_{j+m} * p_m  (cross-correlation)
    """
    n = len(p)
    N = 1
    while N < 4 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p, N)
    cross = np.fft.irfft(W * np.conj(P), N)[:n]
    return 2.0 * cross


def gradient_descent_smooth(p_init, time_limit, sym=True, eps_init=None, eps_min=0.005):
    """
    Projected gradient descent with log-sum-exp smoothing and adaptive temperature.
    Returns (best_p, best_conv_max).
    """
    n = len(p_init)
    p = np.maximum(p_init.astype(np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p /= s
    if sym:
        p = symmetrize(p)
        p = p / p.sum()

    if eps_init is None:
        eps_init = 0.1 * n

    t0 = time.monotonic()
    eps = eps_init
    lr = 0.005
    momentum = 0.85
    v = np.zeros(n)
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(p)
        c_max = c.max()
        if c_max < 1e-15:
            break

        # Softmax weights (numerically stable)
        c_shifted = (c - c_max) / eps
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        grad = smooth_gradient(p, w)

        # Momentum update
        v = momentum * v - lr * grad
        p_new = p + v

        if sym:
            p_new = symmetrize(p_new)
        p_new = project_simplex(p_new)

        c_new = fast_convolve(p_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_p = p_new.copy()
            p = p_new
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.5

        # Adaptive temperature: decrease when stuck
        if no_improve > 30:
            eps = max(eps_min, eps * 0.9)
            no_improve = 0
            lr = max(0.0005, lr * 0.7)

        # Decay learning rate
        lr *= 0.9995

        if lr < 0.0003:
            break

        iteration += 1

    return best_p, best_obj


def build_sym_lp_matrix(f, n):
    """
    Build constraint matrix A for symmetric LP.
    f is symmetric of length n (even).
    Returns A of shape (n, n//2) such that (f*g_sym)[k] = (A @ q)[k] for k=0..n-1.
    """
    half = n // 2
    A = np.zeros((n, half))
    for k in range(n):
        for m in range(half):
            # q_g[m] contributes to g[m] and g[n-1-m]
            # (f*g)[k] = sum_i f[i]*g[k-i]
            # g[k-i] = q[k-i] if 0<=k-i<half, else q[n-1-(k-i)] if half<=k-i<n
            coeff = 0.0
            if 0 <= k - m < n:
                ko = k - m  # index in g from first copy
                # g[ko] = q[ko] if ko < half else q[n-1-ko]
                if ko < half:
                    pass  # already counted
                else:
                    pass
            # More direct: coefficient of q[m] in (f*g)[k]
            # q[m] appears in g[m] and g[n-1-m]
            # Contribution from g[m]: f[k-m] (if 0<=k-m<n)
            # Contribution from g[n-1-m]: f[k-(n-1-m)] = f[k-n+1+m] (if 0<=k-n+1+m<n)
            if 0 <= k - m < n:
               coeff += f[k - m]
            if 0 <= k - n + 1 + m < n:
                coeff += f[k - n + 1 + m]
            A[k, m] = coeff
    return A


def solve_sym_lp(f, M, n):
    """
    Solve symmetric LP: max sum(g) s.t. (f*g)[k] <= M for k=0..n-1, g>=0, g symmetric.
    Returns g (full length n) or None.
    """
    half = n // 2
    A = build_sym_lp_matrix(f, n)
    b_ub = np.full(n, M)
    c = -np.ones(half)
    bounds = [(0, None)] * half
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            q = np.maximum(result.x, 0.0)
            # Reconstruct full g
            g = np.zeros(n)
            g[:half] = q
            g[half:] = q[::-1]
            return g
    except Exception:
        pass
    return None


def optimize_lp_sym(f_init, time_limit, blend_init=0.4, blend_min=0.005,
                    blend_halve_every=10):
    """LP-based directional optimization with symmetry."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    a = symmetrize(a)
    a = project_simplex(a)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_sym_lp(a, M, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex_sym(a_new, n)
        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min * 1.1:
                break

        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng(int(best_obj * 1e10) % (2**31))
            a = project_simplex_sym(a + rng_local.standard_normal(n) * 0.02, n)
            conv = fast_convolve(a)
            stall = 0

    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30):
    """Subgradient descent focusing on top-k peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    a = symmetrize(a)
    a = project_simplex(a)
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Weighted subgradient: emphasize highest peaks
        scaled = peak_vals / (0.01 * pmax + 1e-10)
        w = scaled / scaled.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a_new = symmetrize(a + v)
        a_new = project_simplex(a_new)
        conv = fast_convolve(a_new)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.5

        if no_improve > 100:
            a = project_simplex_sym(a + np.random.default_rng().standard_normal(n) * 0.005, n)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.7)

        lr *= 0.999

    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=50):
    """Coordinate descent: try moving mass between pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    a = symmetrize(a)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()

    for rnd in range(max_rounds):
        improved = False
        # Sample random pairs for efficiency with large n
        rng_cd = np.random.default_rng(rnd * 7 + 13)
        if n <= 200:
            pairs = [(i, j) for i in range(n) for j in range(n) if i != j]
        else:
            idx = rng_cd.integers(0, n, size=min(5000, n * 20))
            pairs = []
            for ii in range(0, len(idx) - 1, 2):
                i, j = int(idx[ii]), int(idx[ii + 1])
                if i != j:
                    pairs.append((i, j))

        for i, j in pairs:
            if a[i] < 1e-8:
                continue
            for frac in [0.1, 0.05, 0.02]:
                delta = a[i] * frac
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                a_test = symmetrize(a_test)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            break

    return a

    inits['uniform'] = np.ones(n) / n

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = symmetrize(tri / tri.sum())

    # Edge
    e = max(1, n // 6)
    edge = np.zeros(n)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - mid) / (0.25 * n)) ** 2)
    inits['gauss'] = symmetrize(g / g.sum())

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = symmetrize(g2 / g2.sum())

    # Cosine window
    cos_w = np.cos(np.pi * (x - mid) / n) ** 2
    cos_w = np.maximum(cos_w, 1e-6)
    inits['cosine'] = symmetrize(cos_w / cos_w.sum())

    # Random symmetric
    q = rng.dirichlet(np.ones(n // 2) * 0.5)
    p_rand = np.zeros(n)
    p_rand[:n // 2] = q
    p_rand[n // 2:] = q[::-1]
    inits['random'] = p_rand / p_rand.sum()

    # Plateau (flat top with cosine edges)
    w = max(1, n // 8)
    plat = np.zeros(n)
    plat[w:n - w] = 1.0
    for i in range(w):
        plat[i] = 0.5 * (1 + np.cos(np.pi * i / w))
        plat[n - 1 - i] = plat[i]
    inits['plateau'] = symmetrize(plat / plat.sum())

    if parent is not None and len(parent) > 5:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
                if len(parent) == n:
                inits['parent'] = symmetrize(p)
            else:
                p_resized = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)), p)
                p_resized = np.maximum(p_resized, 0)
                p_resized /= p_resized.sum()
                inits['parent'] = symmetrize(p_resized)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimum of 2n*max(conv(a,a))/(sum(a))^2.
    Uses smooth gradient descent, symmetric LP, subgradient, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass

    # Phase 1: Gradient descent scan
    print("=== Phase 1: Gradient Descent Scan ===")
    t0 = time.monotonic()
    lengths = [512, 768, 1024, 1536]
    results = []

    # Time allocation for phase 1
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.25)
    phase1_end = time.monotonic() + phase1_time

    all_names = ['parent', 'cosine', 'bimodal', 'plateau', 'uniform',
                 'triangle', 'gauss', 'edge', 'random']

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        names = [nm for nm in all_names if nm in inits][:4]

        t_per = max(15.0, (phase1_end - time.monotonic() - 20) / (len(lengths) * 4))

        for name in names:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 8)
            if t_alloc < 5:
                break

            a0 = inits[name].copy()
            a_opt, conv_max = gradient_descent_smooth(
                a0, t_alloc, sym=True,
                eps_init=0.05 * n, eps_min=0.01)

            score = 2.0 * n * conv_max
            results.append((score, n, name, a_opt.copy(), conv_max))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n} {name}: score={score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization with LP + gradient
    print("=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    top_list = []
    seen = set()
    for score, n, name, a_opt, conv_max in results[:6]:
        key = (n, name)
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None:
        if (len(best_sequence),) not in [(t[0],) for t in top_list]:
            top_list.append((len(best_sequence), best_sequence.copy()))

    phase2_end = time.monotonic() + min(900.0, (deadline - time.monotonic()) * 0.55)
    t_per_candidate = max(30.0, (phase2_end - t2_start) / max(1, len(top_list) * 2))

    for n, a0 in top_list:
        if time.monotonic() > phase2_end - 60:
            break
        t_remaining = phase2_end - time.monotonic() - 40
        if t_remaining < 20:
            break
        t_alloc = min(t_per_candidate, t_remaining)

        # LP optimization
        t_lp = t_alloc * 0.5
        if t_lp > 10:
            a_lp, obj_lp = optimize_lp_sym(a0, t_lp, blend_init=0.3,
                                           blend_min=0.003, blend_halve_every=12)
            score_lp = 2.0 * n * obj_lp
            if score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  LP n={n}: {score_lp:.6f}")
                a0 = a_lp
            else:
                a0 = a_lp  # still use it as starting point

        # Gradient descent refinement
        t_gd = min(60.0, t_alloc * 0.3)
        if t_gd > 5 and time.monotonic() < phase2_end - 30:
            a_gd, obj_gd = gradient_descent_smooth(
                a0, t_gd, sym=True,
                eps_init=0.02 * n, eps_min=0.005)
            score_gd = 2.0 * n * obj_gd
            if score_gd < best_score:
                best_score = score_gd
                best_sequence = a_gd.copy()
                print(f"  GD n={n}: {score_gd:.6f}")

        # Subgradient polish
        t_sg = min(30.0, t_alloc * 0.2)
        if t_sg > 5 and time.monotonic() < phase2_end - 20:
            a_sg, obj_sg = subgradient_polish(
                a0, t_sg, lr_init=0.005, momentum=0.92, top_k=40)
            score_sg = 2.0 * n * obj_sg
            if score_sg < best_score:
                best_score = score_sg
                best_sequence = a_sg.copy()
                print(f"  SG n={n}: {score_sg:.6f}")

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        phase3_end = deadline - 15
        a_cur = best_sequence.copy()
        n_cur = len(a_cur)

        for cycle in range(5):
            if time.monotonic() > phase3_end - 10:
                break
            t_cyc = min(60.0, (phase3_end - time.monotonic() - 10) / (5 - cycle))
            if t_cyc < 5:
                break

            # Coordinate descent
            t_cd = min(20.0, t_cyc * 0.4)
            if t_cd > 3:
                a_cd = coordinate_descent(a_cur, max_rounds=30)
                score_cd = float(evaluate_sequence(a_cd.tolist()))
                if np.isfinite(score_cd) and score_cd < best_score:
                    best_score = score_cd
                    best_sequence = a_cd.copy()
                    a_cur = a_cd.copy()
                    print(f"  CD cycle {cycle}: {score_cd:.6f}")

            # LP with small blend
            t_lp = min(20.0, t_cyc * 0.3)
            if t_lp > 3 and time.monotonic() < phase3_end - 8:
                a_lp, obj_lp = optimize_lp_sym(a_cur, t_lp, blend_init=0.15,
                                               blend_min=0.002, blend_halve_every=8)
                score_lp = 2.0 * n_cur * obj_lp
                if score_lp < best_score:
                    best_score = score_lp
                    best_sequence = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  LP polish {cycle}: {score_lp:.6f}")

            # Gradient descent with small eps
            t_gd = min(15.0, t_cyc * 0.25)
            if t_gd > 3 and time.monotonic() < phase3_end - 5:
                a_gd, obj_gd = gradient_descent_smooth(
                    a_cur, t_gd, sym=True, eps_init=0.01 * n_cur, eps_min=0.003)
                score_gd = 2.0 * n_cur * obj_gd
                if score_gd < best_score:
                    best_score = score_gd
                    best_sequence = a_gd.copy()
                    a_cur = a_gd.copy()
                    print(f"  GD polish {cycle}: {score_gd:.6f}")

        # Random restart perturbations
        if time.monotonic() < phase3_end - 15:
            for ps in [0.005, 0.01, 0.02]:
                if time.monotonic() > phase3_end - 10:
                    break
                for _ in range(3):
                    if time.monotonic() > phase3_end - 8:
                        break
                    noise = rng.standard_normal(n_cur) * ps
                    a_p = project_simplex_sym(best_sequence + noise, n_cur)
                    a_opt, obj_opt = gradient_descent_smooth(
                        a_p, min(15.0, phase3_end - time.monotonic() - 5),
                        sym=True, eps_init=0.02 * n_cur, eps_min=0.005)
                    score = 2.0 * n_cur * obj_opt
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        n_fb = 512
        best_sequence = np.ones(n_fb) / n_fb
        best_sequence = symmetrize(best_sequence)

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(512) / 512
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
