def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, T, top_k=None):

def solve_lp_oracle(f, M, time_limit=None):
    """Solve LP: max sum(g) s.t. conv(f, g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_subgradient(a, T, top_k):
    """Compute subgradient of LSE-smoothed max auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    size = len(conv)
    
    if top_k >= size:
        idx = np.arange(size)
        vals = conv
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
    
    scaled = vals / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(w))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-15:
            continue
        k = int(idx[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        j_arr = np.arange(js, je + 1)
        sg[j_arr] += 2.0 * w[i] * a[k - j_arr]
    
    return sg, conv.max()

def lp_fw_step(f, M, m_exp=1.0):
    """Single FW step with LP oracle."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M * m_exp)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None
    g = res.x
    gs = g.sum()
    if gs < 1e-15:
        return None, None
    g = g / gs
    return g, res
