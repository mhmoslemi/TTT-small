    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    # Maintain list of cuts: (gradient_vector, rhs)
    # Cut: grad @ a <= t + rhs
    cuts = []
    
    t_star = float('inf')
    
    for iter in range(max_iters):
        conv = fast_convolve(a)
        t_current = conv.max()
        
        # Add top-k cuts based on current a
        # Use cuts where conv_k is near the max
        threshold = t_current * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Add cuts for active indices (avoid duplicates)
        for k in active:
            if k not in [c[2] for c in cuts]:
                grad = compute_subgrad_conv(a, k)
                rhs = -conv[k] + np.dot(grad, a)
                cuts.append((grad, rhs, k))
        
        # Also add the current argmax
        k_max = int(np.argmax(conv))
        if k_max not in [c[2] for c in cuts]:
            grad = compute_subgrad_conv(a, k_max)
            rhs = -conv[k_max] + np.dot(grad, a)
            cuts.append((grad, rhs, k_max))
        
        # Solve LP: min t
        # subject to: grad_i @ a <= t + rhs_i for all cuts i
        #              sum(a) = 1
        #              a >= 0
        #
        # Rewrite: grad_i @ a - t <= rhs_i  (linear in (a, t))
        
        from scipy.optimize import linprog
        # Variables: [a_0, ..., a_{n-1}, t]
        # Minimize: [0, ..., 0, 1] @ [a, t]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each cut: grad @ a - t <= rhs
        # => [grad, -1] @ [a, t] <= rhs
        if len(cuts) > 0:
            A_ub = np.zeros((len(cuts), n + 1))
            b_ub = np.zeros(len(cuts))
            for i, (grad, rhs, k) in enumerate(cuts):
                A_ub[i, :n] = grad
                A_ub[i, n] = -1.0
                b_ub[i] = rhs
        else:
            A_ub = None
            b_ub = None
        
        # Equality: sum(a) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Bounds: a_i >= 0, t unbounded below (but will be >= 0 anyway)
        bounds = [(0, None)] * n + [(-np.inf, None)]
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')

        conv_new = fast_convolve(a_new)

        a = a_new

            break
        
        # Limit total cuts for efficiency

            cuts = cuts[-max_cuts:]
    
    return a, t_star

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-start projected subgradient with momentum and adaptive step size
2. Cutting-plane LP heuristic for direct peak reduction
3. L-BFGS-B smooth refinement with beta annealing
4. 1D coordinate descent on the exact non-smooth objective
5. Random perturbation restarts and final polish
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta, top_k=None):
    """Compute smooth max and weights, optionally restricting to top_k values."""
    if top_k is not None and top_k < len(x):
        idx = np.argpartition(x, -top_k)[-top_k:]
        x_used = x[idx]
        x_max = x_used.max()
        shifted = beta * (x_used - x_max)
        shifted = np.clip(shifted, -100, 100)
        exp_shifted = np.exp(shifted)
        total = exp_shifted.sum()
        if total < 1e-30:
            w = np.zeros_like(x)
            w[np.argmax(x)] = 1.0
            return x_max, w
        w_used = exp_shifted / total
        sm = x_used.max() + np.log(total) / beta
        w = np.zeros_like(x)
        w[idx] = w_used
        return sm, w
    else:
        x_max = x.max()
        shifted = beta * (x - x_max)
        shifted = np.clip(shifted, -100, 100)
        exp_shifted = np.exp(shifted)
        total = exp_shifted.sum()
        if total < 1e-30:
            w = np.zeros_like(x)
            w[np.argmax(x)] = 1.0
            return x_max, w
        w = exp_shifted / total
        sm = x_max + np.log(total) / beta
        return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of sum_k w[k]*conv_k w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def compute_subgrad_conv_fast(a, k):
    """Compute subgradient of conv_k(a) w.r.t. a using FFT circular correlation trick."""
    n = len(a)
    # conv_k = sum_i a_i * a_{k-i}
    # d(conv_k)/da_j = a_{k-j} (if valid) + a_{j-k} (if valid)
    # This is equivalent to: circular cross-correlation of a with a shifted by k
    fft_size = 1 << (n - 1).bit_length()
    A = fft(a, fft_size)
    # circ_corr[k] = sum_j a_j * a_{j-k} (circular)
    # For linear conv subgradient, we need the linear version
    # grad[j] = a[k-j] + a[j-k] with bounds
    grad = np.zeros(n)
    # a[k-j] for j in range(n): valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += a[k - np.arange(lo1, hi1 + 1)]
    # a[j-k] for j in range(n): valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        grad[lo2:hi2+1] += a[np.arange(lo2, hi2 + 1) - k]
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta, top_k):
    """Objective for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta, top_k)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta, top_k):
    """Jacobian for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta, top_k)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300, top_k=None):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta, top_k),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 40})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def subgradient_descent(a0, n_iter=500, momentum=0.9, time_budget=30):
    """Projected subgradient descent with momentum and adaptive step size."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    # Polyak lower bound: max(conv) >= 1/(2n-1) when sum=1
    f_star = 1.0 / (2 * n - 1) * 0.9

    t_start = time.monotonic()
    alpha = 0.05

    for it in range(n_iter):
        if time.monotonic() - t_start > time_budget:
            break

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Subgradient: gradient of the active piece
        k_max = int(np.argmax(conv))
        grad = compute_subgrad_conv_fast(a, k_max)

        # Polyak step size
        gap = conv.max() - f_star
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break
        alpha_polyak = gap / grad_norm_sq
        alpha = alpha_polyak * (1.0 + it / n_iter)  # Slightly aggressive early on
        alpha = min(alpha, 0.5)

        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm * np.linalg.norm(grad)

        # Step
        a_new = project_simplex(a - alpha * v)

        # Check improvement
        conv_new = fast_convolve(a_new)
        if 2.0 * n * conv_new.max() < best_obj * 1.01:
            a = a_new
        else:
            # Halve step and try again
            a_new = project_simplex(a - alpha * 0.25 * v)
            conv_new = fast_convolve(a_new)
            if 2.0 * n * conv_new.max() < best_obj * 1.01:
                a = a_new
            else:
                alpha *= 0.5

    return best_a


def cutting_plane_optimize(a0, max_iters=100, time_budget=60):
    """Cutting-plane LP heuristic for minimizing max(conv)."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    # Cuts: list of (gradient_array, rhs)
    # Constraint: grad @ x - t <= rhs  where x = [a, t]
    cuts = []
    cut_indices = set()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t_start > time_budget:
            break

        conv = fast_convolve(a)
        cur_max = conv.max()

        if cur_max < best_obj:
            best_obj = cur_max
            best_a = a.copy()

        # Add cuts for top violations
        sorted_idx = np.argsort(conv)[::-1]
        n_add = min(5, n)
        added = 0
        for k in sorted_idx:
            k = int(k)
            if k in cut_indices:
                continue
            if conv[k] < cur_max * 0.85:
                break
            grad = compute_subgrad_conv_fast(a, k)
            rhs = np.dot(grad, a) - conv[k]
            cuts.append((grad, rhs))
            cut_indices.add(k)
            added += 1
            if added >= n_add:
                break

        # If no new cuts added, still try to solve
        if added == 0 and len(cuts) > 10:
            # Remove some old cuts and try different linearization
            # Re-linearize the top constraint at current a
            k_max = int(np.argmax(conv))
            grad = compute_subgrad_conv_fast(a, k_max)
            rhs = np.dot(grad, a) - conv[k_max]
            # Update existing cut if present
            for i, (g, r) in enumerate(cuts):
                if np.allclose(g, compute_subgrad_conv_fast(best_a, k_max), atol=1e-6):
                    cuts[i] = (grad, rhs)
                    break
            else:
                cuts.append((grad, rhs))
                cut_indices.add(k_max)

        # Build and solve LP
        m = len(cuts)
        if m == 0:
            continue

        # Variables: [a_0, ..., a_{n-1}, t]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t

        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, (grad, rhs) in enumerate(cuts):
            A_ub[i, :n] = grad
            A_ub[i, n] = -1.0
            b_ub[i] = rhs

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(-np.inf, None)]

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': max(1, time_budget - (time.monotonic() - t_start))})
        except Exception:
            break

        if not result.success:
            break

        a_lp = np.maximum(result.x[:n], 0)
        a_lp_sum = a_lp.sum()
        if a_lp_sum < 1e-10:
            break
        a_lp = a_lp / a_lp_sum

        # Check true objective
        conv_lp = fast_convolve(a_lp)
        lp_max = conv_lp.max()

        # Accept if improved
        if lp_max < cur_max * 1.001:
            a = a_lp
        else:
            # Blend with current
            blend = 0.5
            a_new = blend * a_lp + (1 - blend) * a
            a_new = np.maximum(a_new, 0)
            a_new_sum = a_new.sum()
            if a_new_sum > 1e-10:
                a_new = a_new / a_new_sum
                conv_new = fast_convolve(a_new)
                if conv_new.max() < cur_max:
                    a = a_new

        # Limit cuts
        if len(cuts) > 500:
            cuts = cuts[-300:]
            cut_indices = set()

    return best_a


def coordinate_descent(a0, n_passes=30, time_budget=60):
    """Coordinate descent: optimize one coordinate at a time using 1D search."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t_start = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t_start > time_budget:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t_start > time_budget:
                break

            old_val = a[i]
            # Compute objective as function of a[i] = x
            # S = sum of other elements
            S = a.sum() - old_val
            if S < 1e-10:
                continue

            # conv_k(x) = base_k + x * c_k
            # base_k = conv_k with a[i]=0
            a[i] = 0.0
            conv_base = fast_convolve(a)
            # c_k = a[k-i] + a[i-k] (with bounds)
            c = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n:
                    val += a[i - k]
                c[k] = val
            a[i] = old_val

            # Find optimal x >= 0
            # f(x) = max_k (conv_base[k] + x*c[k]) / (S + x)^2
            # Try candidates: x=0, and for each k with c[k]>0: x_k = S - 2*conv_base[k]/c[k]
            candidates = [0.0, old_val, 2.0 * S / n]

            for k in range(2 * n - 1):
                if c[k] > 1e-15:
                    x_k = S - 2.0 * conv_base[k] / c[k]
                    if 0 <= x_k <= 4.0 * S / n:
                        candidates.append(x_k)

            candidates.append(S / n)  # uniform-like
            candidates = [max(0.0, min(x, 4.0 * S / n)) for x in candidates]
            candidates = list(set([round(x, 10) for x in candidates]))

            best_x = old_val
            best_f = float('inf')

            for x in candidates:
                a[i] = x
                conv_try = fast_convolve(a)
                f_try = 2.0 * n * conv_try.max() / (a.sum() ** 2) if a.sum() > 1e-10 else float('inf')
                if f_try < best_f:
                    best_f = f_try
                    best_x = x
            a[i] = best_x

            if best_x != old_val:
                improved = True
                s = a.sum()
                if s > 1e-10:
                    a = a / s

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian centered
    sigma = n * 0.25
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Edge-concentrated
    edge_sigma = n * 0.15
    e1 = np.exp(-0.5 * (x / edge_sigma) ** 2)
    e2 = np.exp(-0.5 * ((n - 1 - x) / edge_sigma) ** 2)
    bi = e1 + e2
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Random Dirichlet (concentrated)
    inits['rand_conc'] = rng.dirichlet(np.ones(n) * 2.0)
    # Random Dirichlet (sparse)
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.3)
    # Random Dirichlet (uniform)
    inits['rand_unif'] = rng.dirichlet(np.ones(n) * 1.0)

    # Binary-like: random 0/1 then add small epsilon
    binary = rng.random(n) > 0.4
    binary = binary.astype(float)
    binary += 0.01
    binary /= binary.sum()
    inits['binary'] = binary

    # Flat spectrum initialization
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    for _ in range(10):
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        for k in range(1, fft_size // 2 + 1):
            mag = 1.0
            A[k] = mag * np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = mag * np.exp(-1j * phases[k])
        a_try = np.real(ifft(A, fft_size))[:n]
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try > 1e-10:
            a_try /= s_try
            inits['flat_spec'] = a_try
            break

    # M-sequence inspired (maximal length sequence pattern)
    # Use a periodic pattern from a primitive polynomial
    m_seq = np.zeros(n)
    period = 1
    for _ in range(20):
        period = (2 ** rng.integers(2, 10)) - 1
        if period >= n:
            break
        # Generate simple pattern
        pattern = (rng.random(period) > 0.5).astype(float)
        tile = np.tile(pattern, n // period + 1)[:n]
        m_seq = tile + 0.05
    m_seq /= m_seq.sum()
    inits['mseq'] = m_seq

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    return inits


def multi_beta_lbfgsb(a0, betas=None, time_budget=60):
    """L-BFGS-B with decreasing beta schedule."""
    if betas is None:
        betas = [80.0, 40.0, 20.0, 10.0, 5.0, 3.0]
    n = len(a0)
    a = a0.copy()
    t_start = time.monotonic()

    for beta in betas:
        if time.monotonic() - t_start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - t_start) - 5
        if remaining < 3:
            break
        maxiter = int(min(400, max(30, remaining * 25)))
        a = optimize_lbfgsb(a, beta=beta, maxiter=maxiter, top_k=max(5, len(a) // 4))

    return a

        pass

                score_candidate(a_cur)

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Coordinate descent refinement
    print("=== Phase 3: Coordinate Descent ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0
        if remaining > 20:
            t_cd = min(remaining * 0.6, 120.0)
            a_cd = coordinate_descent(best_sequence, n_passes=40, time_budget=t_cd)
            score_candidate(a_cd)
            if a_cd is not None and 2.0 * len(a_cd) * fast_convolve(a_cd).max() < \
                    2.0 * len(best_sequence) * fast_convolve(best_sequence).max():
                best_sequence = a_cd

    # Phase 4: Final polish
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0

        # Low-beta L-BFGS-B
        if remaining > 15:
            t_single = min(remaining * 0.4, 60.0)
            a_pol = optimize_lbfgsb(best_sequence, beta=2.0, maxiter=int(min(500, t_single * 30)), top_k=5)
            score_candidate(a_pol)

        # Subgradient with very low momentum for fine tuning
        if remaining > 15:
            t_sg = min(remaining * 0.3, 40.0)
            a_sg2 = subgradient_descent(best_sequence, n_iter=300, momentum=0.7, time_budget=t_sg)
            score_candidate(a_sg2)

        # Random perturbations
        if remaining > 20:
            n_restarts = min(5, int(remaining / 15))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 15:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                a_opt = optimize_lbfgsb(a_pert, beta=5.0, maxiter=150, top_k=8)
                score_candidate(a_opt)

        # Coordinate descent one more time
        if time.monotonic() < deadline - 15:
            t_cd2 = min(30.0, deadline - time.monotonic() - 15)
            if t_cd2 > 5:
                a_cd2 = coordinate_descent(best_sequence, n_passes=20, time_budget=t_cd2)
                score_candidate(a_cd2)

        # Final cutting-plane
        if time.monotonic() < deadline - 10:
            t_cp2 = min(30.0, deadline - time.monotonic() - 10)
            if t_cp2 > 5:
                a_cp2 = cutting_plane_optimize(best_sequence, max_iters=50, time_budget=t_cp2)
                score_candidate(a_cp2)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
