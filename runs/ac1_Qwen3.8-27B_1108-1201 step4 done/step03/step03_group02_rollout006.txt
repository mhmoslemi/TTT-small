from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def lp_direction(a, mode='min_peak'):
    """
    Solve LP to get direction.
    mode='min_peak': min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0
    mode='max_sum': max sum(b) s.t. conv(a,b)_k <= M_current, b>=0
    Returns (b, M_bound) where b is normalized to sum=1.
    """
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix(a)
    
    if mode == 'min_peak':
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        A_ub = np.zeros((num_rows, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            b = np.maximum(res.x[:n], 0)
            M_bound = res.x[n]
            b_sum = b.sum()
            if b_sum > 1e-15:
                b = b / b_sum
            return b, M_bound
        return None, None
    else:  # max_sum
        M_current = fast_convolve(a).max()
        c = -np.ones(n)
        b_ub = np.full(num_rows, M_current)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            b = np.maximum(res.x, 0)
            b_sum = b.sum()
            if b_sum > 1e-15:
                b = b / b_sum
            return b, M_current
        return None, None

        obj = 2.0 * n * conv.max()
        if obj < best_obj:

def subgradient_step(a, convex, top_k=20, temperature=0.001):
    """Subgradient step targeting top-k peaks."""
    n = len(a)
    if top_k >= len(convex):
        peaks = np.arange(len(convex))
    else:
        peaks = np.argpartition(convex, -top_k)[-top_k:]
    
    # Weight by similarity to max
    peak_vals = convex[peaks]
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = 2.0 * w[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with golden-section line search, multi-scale exploration, subgradient polishing,
and perturbation restarts. Key improvement over previous: proper 1D line search
along LP directions instead of fixed step sizes, plus frequency-aware
initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        js = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(js) > 0:
            A[k, js] = a[k - js]
    return A


def lp_direction_min_peak(a):
    """Solve: min M s.t. conv(a,b)_k <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        M_bound = float(res.x[n])
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
        return b, M_bound
    return None, None


def lp_direction_max_sum(a, M_bound):
    """Solve: max sum(b) s.t. conv(a,b)_k <= M_bound, b>=0. Returns normalized b."""
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(num_rows, M_bound)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
        return b
    return None


def golden_section_search(a, b, n_iter=30):
    """
    Minimize f(t) = max_k conv((1-t)*a + t*b, (1-t)*a + t*b)_k
    over t in [0, 1] using golden section search.
    Returns (t_opt, a_opt).
    """
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618

    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    a_c = (1 - c) * a + c * b
    a_c = np.maximum(a_c, 0)
    s = a_c.sum()
    if s < 1e-15:
        return 0.0, a.copy()
    a_c = a_c / s
    fc = 2.0 * n * fast_convolve(a_c).max()

    a_d = (1 - d) * a + d * b
    a_d = np.maximum(a_d, 0)
    s = a_d.sum()
    if s < 1e-15:
        return 0.0, a.copy()
    a_d = a_d / s
    fd = 2.0 * n * fast_convolve(a_d).max()

    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            a_c = (1 - c) * a + c * b
            a_c = np.maximum(a_c, 0)
            s = a_c.sum()
            if s < 1e-15:
                break
            a_c = a_c / s
            fc = 2.0 * n * fast_convolve(a_c).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            a_d = (1 - d) * a + d * b
            a_d = np.maximum(a_d, 0)
            s = a_d.sum()
            if s < 1e-15:
                break
            a_d = a_d / s
            fd = 2.0 * n * fast_convolve(a_d).max()

    t_opt = (lo + hi) / 2.0
    a_opt = (1 - t_opt) * a + t_opt * b
    a_opt = np.maximum(a_opt, 0)
    s = a_opt.sum()
    if s < 1e-15:
        return 0.0, a.copy()
    a_opt = a_opt / s
    return t_opt, a_opt


def lp_line_search_optimize(a0, time_limit, max_iters=500, gss_iters=25):
    """LP direction + golden section line search."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    use_min_peak = True
    gss_count = gss_iters

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_min_peak:
            b, M_bound = lp_direction_min_peak(a)
            if b is None:
                b = lp_direction_max_sum(a, conv.max())
                if b is None:
                    break
            use_min_peak = False
        else:
            b = lp_direction_max_sum(a, conv.max())
            if b is None:
                b, M_bound = lp_direction_min_peak(a)
                if b is None:
                    break
            use_min_peak = True

        if np.linalg.norm(b - a) < 1e-12:
            break

        t_opt, a_new = golden_section_search(a, b, n_iter=gss_count)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if gss_count > 15:
                gss_count = 15
        else:
            no_improve += 1
            gss_count = max(10, gss_count - 2)

        a = a_new
        conv = conv_new

        if no_improve > 40:
            break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.015, momentum=0.9,
                         top_k=30, temp=0.0004):
    """Projected subgradient with momentum and perturbation."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        # Compute weighted subgradient over top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = 2.0 * w[i]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        lr *= 0.9997

        if no_improve > 1500:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            lr = max(lr_init * 0.2, lr * 0.5)

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Parent (resampled if needed)
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits.append(('parent', pp / ps))
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits.append(('parent_res', pp / ps))

    # Cosine (various frequencies)
    for freq in [1, 2, 3]:
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        inits.append(('cos_%d' % freq, a))

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits.append(('gauss', g))

    # Random Dirichlet
    inits.append(('random', rng.dirichlet(np.ones(n) * 0.4)))

    # Tapered (raised cosine edges)
    a = np.ones(n)
    edge = n // 8
    if edge > 2:
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge))
        a[:edge] = taper
        a[-edge:] = taper[::-1]
    a = np.maximum(a, 0.01)
    a /= a.sum()
    inits.append(('tapered', a))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: LP + golden section line search from diverse inits
    Phase 2: Deep LP refinement from best
    Phase 3: Subgradient polishing
    Phase 4: Perturbation + LP re-optimization
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_p = len(parent)
            obj_p = 2.0 * n_p * fast_convolve(parent).max()
            print(f"Parent: n={n_p}, obj={obj_p:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    best_sequence = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")

    # === Phase 1: LP + Line Search from multiple inits ===
    print("=== Phase 1: LP + Golden Section Line Search ===")
    t1_start = time.monotonic()

    # Decide which lengths to try
    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 512

    lengths_to_try = [n_main]
    if n_main != 256:
        lengths_to_try.append(256)
    if n_main != 1024:
        lengths_to_try.append(1024)

    # Budget allocation: main length gets 70%, others split the rest
    phase1_budget = (total_deadline - t1_start) * 0.55
    main_budget = phase1_budget * 0.7
    other_budget = phase1_budget * 0.3 / max(1, len(lengths_to_try) - 1)

    for L in lengths_to_try:
        if L == n_main:
            t_alloc_total = main_budget
        else:
            t_alloc_total = other_budget

        if time.monotonic() + 30 > total_deadline:
            break

        inits = generate_inits(L, parent, rng)
        # Prioritize parent and uniform
        priority = ['parent', 'parent_res', 'uniform', 'random', 'cos_1', 'tapered']
        inits.sort(key=lambda x: (priority.index(x[0]) if x[0] in priority else 99, x[0]))
        inits = inits[:6]

        t_per_init = t_alloc_total / len(inits)

        for name, a0 in inits:
            if time.monotonic() + 10 > total_deadline:
                break
            t_single = min(t_per_init, total_deadline - time.monotonic() - 60)
            if t_single < 5:
                break
            print(f"  n={L}, init={name}, time={t_single:.0f}s")
            a_opt, obj = lp_line_search_optimize(a0, time_limit=t_single, max_iters=400)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Deep LP refinement ===
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = total_deadline - (total_deadline - time.monotonic()) * 0.25
        # Run multiple rounds with alternating LP modes
        for round_num in range(4):
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(50.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a_opt, obj = lp_line_search_optimize(
                best_sequence, time_limit=t_alloc,
                max_iters=500, gss_iters=20)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Subgradient polishing ===
    print("=== Phase 3: Subgradient Polishing ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        phase3_deadline = total_deadline - (total_deadline - time.monotonic()) * 0.12
        for lr in [0.015, 0.008, 0.004]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(35.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_opt, obj = subgradient_optimize(
                best_sequence, time_limit=t_alloc, lr_init=lr,
                momentum=0.9, top_k=max(20, n_best // 25), temp=0.0003)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Perturbation + LP re-optimization ===
    print("=== Phase 4: Perturbations ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        phase4_deadline = total_deadline - 20
        scales = [0.001, 0.002, 0.004, 0.008, 0.015, 0.003, 0.006]
        for i, scale in enumerate(scales):
            if time.monotonic() > phase4_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(25.0, phase4_deadline - time.monotonic())
            if t_alloc < 3:
                break
            a_opt, obj = lp_line_search_optimize(
                a_pert, time_limit=t_alloc, max_iters=300, gss_iters=20)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Final safety
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    else:
        final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(max(0.0, x)) for x in best_sequence]
