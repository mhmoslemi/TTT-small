def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

(D_θ a)_i,j = a_i * (δ_ij - a_j)

This is the "natural gradient" or "Fisher information metric gradient." It's scale-invariant and works well for simplex-constrained problems.

Minimizes 2n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient

import numpy as np
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices):
    """Compute (2/K) * sum_{k in S} a[k-j] as a function of j."""
    n = len(a)
    K = len(peak_indices)
    if K == 0:
        return np.zeros(n)
    sg = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] += a[k - idx]
    sg *= 2.0 / K
    return sg


def find_top_k_peaks(conv, K, frac=0.92):
    """Find top-K peak indices within frac of the maximum."""
    max_val = conv.max()
    if max_val <= 0:
        return np.array([np.argmax(conv)], dtype=int)
    threshold = max_val * frac
    candidates = np.where(conv >= threshold)[0]
    if len(candidates) > K:
        vals = conv[candidates]
        order = np.argsort(vals)[::-1][:K]
        candidates = candidates[order]
    return candidates.astype(int)


def optimize_subgrad(a0, time_limit, K_start=1, K_end=48, seed=0,
                     mom_base=0.88, lr_base=0.04):
    """Nesterov-accelerated subgradient with BB step size and adaptive K."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    a_prev = a.copy()
    sg_prev = None
    s_BB = None
    y_BB = None

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        K = int(K_start + (K_end - K_start) * (frac ** 0.6))
        K = max(K_start, min(K, max(2, n // 4)))

        momentum = mom_base + 0.08 * (1.0 - frac)
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        conv_y = fast_convolve(y)
        peaks = find_top_k_peaks(conv_y, K, frac=0.90)
        sg = compute_subgradient(y, peaks)

        # BB step size
        if s_BB is not None and y_BB is not None:
            sy = np.dot(s_BB, y_BB)
            ss = np.dot(s_BB, s_BB)
            if sy > 1e-10 and ss > 1e-10:
                lr = np.clip(ss / sy, lr_base * 0.1, lr_base * 3.0)
            else:
                lr = lr_base / (1.0 + 0.08 * iter_count)
        else:
            lr = lr_base / (1.0 + 0.08 * iter_count)

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        if sg_prev is not None:
            s_BB_new = a_new - a_prev
            y_BB_new = sg - sg_prev
            if np.dot(s_BB_new, y_BB_new) > 1e-10:
                s_BB = s_BB_new
                y_BB = y_BB_new

        a_prev = a_new
        sg_prev = sg
        a = a_new
        iter_count += 1

        conv_new = fast_convolve(a)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 350:
            scale = 0.008 * (1.0 + 1.5 * frac)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(a + noise)
            v *= 0.4
            no_improve = 0
            s_BB = None
            y_BB = None
            perturb_count += 1
            if perturb_count > 8 and frac > 0.5:
                # Stronger restart
                noise = rng.standard_normal(n) * 0.02
                a = project_simplex(a + noise)
                v *= 0.2

    return best_a, best_obj


def optimize_mirror_desc(a0, time_limit, K_start=1, K_end=32, seed=0,
                         mom_base=0.85, lr_base=0.4):
    """Mirror descent (natural gradient) with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    theta = np.log(a + 1e-30)
    v = np.zeros(n)
    theta_prev = theta.copy()
    grad_prev = None
    s_BB = None
    y_BB = None

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        K = int(K_start + (K_end - K_start) * (frac ** 0.5))
        K = max(K_start, min(K, max(2, n // 4)))

        momentum = mom_base + 0.05 * (1.0 - frac)
        theta_y = theta + momentum * v
        # Softmax with numerical stability
        a_y = theta_y - theta_y.max()
        a_y = np.exp(a_y)
        a_y = a_y / a_y.sum()

        conv_y = fast_convolve(a_y)
        peaks = find_top_k_peaks(conv_y, K, frac=0.90)
        grad_a = compute_subgradient(a_y, peaks)

        # Natural gradient: grad_theta_i = a_i * (grad_a_i - <a, grad_a>)
        mean_grad = np.dot(a_y, grad_a)
        grad_theta = a_y * (grad_a - mean_grad)

        if s_BB is not None and y_BB is not None:
            sy = np.dot(s_BB, y_BB)
            ss = np.dot(s_BB, s_BB)
            if sy > 1e-10 and ss > 1e-10:
                lr = np.clip(ss / sy, lr_base * 0.1, lr_base * 3.0)
            else:
                lr = lr_base / (1.0 + 0.1 * iter_count)
        else:
            lr = lr_base / (1.0 + 0.1 * iter_count)

        v = momentum * v - lr * grad_theta
        theta_new = theta + v

        a_new = theta_new - theta_new.max()
        a_new = np.exp(a_new)
        a_new = a_new / a_new.sum()

        if grad_prev is not None:
            s_BB_new = theta_new - theta_prev
            y_BB_new = grad_theta - grad_prev
            if np.dot(s_BB_new, y_BB_new) > 1e-10:
                s_BB = s_BB_new
                y_BB = y_BB_new

        theta_prev = theta_new
        grad_prev = grad_theta
        theta = theta_new
        iter_count += 1

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 400:
            noise = rng.standard_normal(n) * (0.05 + 0.1 * frac)
            theta = theta + noise
            v *= 0.3
            no_improve = 0
            s_BB = None
            y_BB = None

    return best_a, best_obj


def coordinate_polish(a0, time_limit, seed=0):
    """Coordinate descent: move mass between pairs to reduce peak."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()
        peaks = find_top_k_peaks(conv_cur, 5, frac=0.95)

        improved = False
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            contribs = a[idx] * a[k - idx]
            if len(contribs) == 0:
                continue
            # Find the pair with max contribution
            max_i_local = np.argmax(contribs)
            i1 = int(idx[max_i_local])
            i2 = int(k - idx[max_i_local])

            # Try moving mass from the larger of (i1, i2) to a random location
            src = i1 if a[i1] >= a[i2] else i2
            dst = src if i1 == i2 else (i2 if a[i1] >= a[i2] else i1)

            for _ in range(5):
                j = int(rng.integers(0, n))
                if j == src or (abs(j - src) < 3 and n > 10):
                    continue
                for frac_move in [0.03, 0.05, 0.08, 0.12]:
                    amt = a[src] * frac_move
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-12:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random restart within neighborhood
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    amt = a[i] * 0.05
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    a = a_try
                stall = 0
        else:
            stall = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    x_norm = x / max(n - 1, 1)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()

    # Exponential (geometric)
    for alpha in [0.5, 1.0, 2.0]:
        exp_seq = np.exp(alpha * x_norm)
        exp_seq /= exp_seq.sum()
        inits[f'exp_{alpha}'] = exp_seq
        exp_seq2 = np.exp(-alpha * x_norm)
        exp_seq2 /= exp_seq2.sum()
        inits[f'exp_neg_{alpha}'] = exp_seq2

    # Bell / Gaussian
    for sigma in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x_norm - 0.5) / sigma) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sigma*100)}'] = g

    # Off-center Gaussian
    g_off = np.exp(-0.5 * ((x_norm - 0.3) / 0.2) ** 2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off

    # V-shape (valley in middle)
    v_shape = np.abs(x_norm - 0.5) + 0.1
    v_shape /= v_shape.sum()
    inits['v_shape'] = v_shape

    # U-shape (convex)
    u_shape = (x_norm - 0.5) ** 2 + 0.05
    u_shape /= u_shape.sum()
    inits['u_shape'] = u_shape

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x_norm))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Sine
    sin_w = np.sin(np.pi * x_norm) ** 2 + 1e-6
    sin_w /= sin_w.sum()
    inits['sine'] = sin_w

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet
    inits['dirichlet_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dirichlet_2'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['dirichlet_01'] = rng.dirichlet(np.ones(n) * 0.1)

    # Asymmetric: product of ramp and bell
    asym = ramp * g_off * 10 + 0.01
    asym = np.maximum(asym, 0)
    asym /= asym.sum()
    inits['asym_product'] = asym

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        # Perturbed parent
        if 'parent_crop' in inits or 'parent_up' in inits or 'parent_ds' in inits:
            key = [k for k in ['parent_crop', 'parent_up', 'parent_ds'] if k in inits][0]
            pp = inits[key] + rng.standard_normal(n) * 0.02
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_perturb'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: scan lengths/inits, deep-optimize with two methods,
    then polish with coordinate descent and fine subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # Phase 1: Scan
    print("=== Phase 1: Scan ===", flush=True)
    t_phase1 = time.monotonic()
    phase1_budget = 300.0

    lengths = [256, 320, 384, 448, 512]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        inits = generate_initializations(n, parent, rng)
        names = list(inits.keys())

        # Priority
        priority = ['parent', 'parent_crop', 'parent_up', 'parent_ds',
                    'parent_perturb', 'v_shape', 'u_shape', 'asym_product',
                    'gauss_25', 'exp_1.0', 'raised_cos', 'uniform',
                    'dirichlet_05', 'sine']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:8]

        time_rem = phase1_budget - (time.monotonic() - t_phase1)
        per_init = max(8.0, time_rem / max(1, len(names) * len(lengths)))

        for name in names:
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(per_init, 40.0)
            if t_alloc < 5:
                break

            a0 = inits[name]
            # Use subgradient for speed in scan phase
            a_opt, obj_opt = optimize_subgrad(
                a0, t_alloc, K_start=1, K_end=24,
                seed=int(rng.integers(0, 100000)))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Opt ===", flush=True)

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 4:
            break

    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))
        best_n = len(best_sequence)
    else:
        best_n = 384

    # Add diverse restarts at best length
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.3)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.5)))

    remaining = deadline - time.monotonic() - 200.0
    n_configs = len(top_configs)
    per_config = max(30.0, remaining / max(1, n_configs))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(per_config, 350.0)
        if t_alloc < 15:
            break

        print(f"  Config {i}: n={n}, alloc={t_alloc:.0f}s", flush=True)

        # Method A: Subgradient (40%)
        t_a = t_alloc * 0.40
        if t_a > 5:
            a_a, _ = optimize_subgrad(
                a0, t_a, K_start=1, K_end=48,
                mom_base=0.88, lr_base=0.04,
                seed=int(rng.integers(0, 1000000)))
            sa = float(evaluate_sequence(a_a.tolist()))
            if np.isfinite(sa) and sa < best_score:
                best_score = sa
                best_sequence = a_a.copy()
                print(f"    SubGrad: {sa:.6f}", flush=True)

        # Method B: Mirror descent (35%)
        t_b = t_alloc * 0.35
        if t_b > 5:
            a_b, _ = optimize_mirror_desc(
                a_a if t_a > 5 else a0, t_b,
                K_start=1, K_end=32,
                mom_base=0.85, lr_base=0.4,
                seed=int(rng.integers(0, 1000000)))
            sb = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(sb) and sb < best_score:
                best_score = sb
                best_sequence = a_b.copy()
                print(f"    Mirror: {sb:.6f}", flush=True)
            a_a = a_b

        # Method C: Second subgradient pass (25%)
        t_c = t_alloc * 0.25
        if t_c > 5:
            a_c, _ = optimize_subgrad(
                a_a, t_c, K_start=4, K_end=64,
                mom_base=0.92, lr_base=0.02,
                seed=int(rng.integers(0, 1000000)))
            sc = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_c.copy()
                print(f"    SubGrad2: {sc:.6f}", flush=True)

    # Phase 3: Polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0

        # Coordinate descent
        t_cd = min(remaining * 0.30, 120.0)
        if t_cd > 10:
            a_cd, _ = coordinate_polish(best_sequence, t_cd,
                                        seed=int(rng.integers(0, 1000000)))
            s_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(s_cd) and s_cd < best_score:
                best_score = s_cd
                best_sequence = a_cd.copy()
                print(f"  CoordDesc: {best_score:.6f}", flush=True)

        # Fine subgradient with high K
        remaining = deadline - time.monotonic() - 20.0
        t_fg = min(remaining * 0.35, 150.0)
        if t_fg > 10:
            a_fg, _ = optimize_subgrad(
                best_sequence, t_fg, K_start=8, K_end=80,
                mom_base=0.93, lr_base=0.01,
                seed=int(rng.integers(0, 1000000)))
            s_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(s_fg) and s_fg < best_score:
                best_score = s_fg
                best_sequence = a_fg.copy()
                print(f"  FineSubGrad: {best_score:.6f}", flush=True)

        # Mirror descent polish
        remaining = deadline - time.monotonic() - 15.0
        t_mp = min(remaining * 0.30, 100.0)
        if t_mp > 5:
            a_mp, _ = optimize_mirror_desc(
                best_sequence, t_mp, K_start=4, K_end=48,
                mom_base=0.90, lr_base=0.2,
                seed=int(rng.integers(0, 1000000)))
            s_mp = float(evaluate_sequence(a_mp.tolist()))
            if np.isfinite(s_mp) and s_mp < best_score:
                best_score = s_mp
                best_sequence = a_mp.copy()
                print(f"  MirrorPolish: {best_score:.6f}", flush=True)

        # Final coordinate descent
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            a_fc, _ = coordinate_polish(best_sequence, remaining,
                                        seed=int(rng.integers(0, 1000000)))
            s_fc = float(evaluate_sequence(a_fc.tolist()))
            if np.isfinite(s_fc) and s_fc < best_score:
                best_score = s_fc
                best_sequence = a_fc.copy()
                print(f"  FinalCoord: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}", flush=True)
    return [float(x) for x in best_sequence]
