def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def subgradient_descent(a0, time_limit, lr_init=0.01, decay=0.9995, 
                         momentum=0.0, top_k=1, seed=0):
    """
    Direct subgradient descent on max of auto-convolution.
    Uses the top-k averaged subgradient for robustness.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient using top-k peaks
        sorted_indices = np.argsort(conv)[::-1]
        active = sorted_indices[:top_k]
        
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for j in range(n):
                jk = k - j
                if 0 <= jk < n:
                    grad[j] += 2.0 * a[jk]
        grad /= top_k
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Check if update helps
        conv_new = fast_convolve(a_new)
        if conv_new.max() < cur_obj + 0.01 * lr:
            a = a_new
        else:
            # Reject, try smaller step
            v = momentum * v - lr * 0.5 * grad
            a = project_simplex(a + v)
        
        iters += 1
        lr = lr_init * (decay ** iters)
        
        # Perturbation if stuck
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            v = np.zeros(n)
            no_improve = 0
            top_k = min(top_k + 1, 5)
        
    return best_a, best_obj


def peak_equalization(a0, time_limit, seed=0):

            if total_contrib < 1e-15:

                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

            for k in top_peaks:

                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    risk[i] += a[i] * a[k - i]
                    risk[k - i] += a[i] * a[k - i]
            
            # Safe positions are those with lowest risk
            safe_positions = np.argsort(risk)[:max(3, n // 10)]
            risky_positions = np.argsort(risk)[::-1][:max(3, n // 10)]
            
            for i_src in risky_positions:
                for i_dst in safe_positions:
                    if i_src == i_dst:
                        continue

                            break

                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

                a = project_simplex(a + noise)

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
- Direct subgradient descent targeting the max convolution peak
- LP-based direction finding for non-convex escape
- Peak equalization via active set identification
- Log-sum-exp smoothed gradient descent

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_at_peak(a, k):
    """
    Gradient of conv[k] w.r.t. a.
    d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    for j in range(j_start, j_end + 1):
        grad[j] = 2.0 * a[k - j]
    return grad


def subgradient_topk(a, conv, top_k=3):
    """Averaged subgradient using top-k peaks."""
    n = len(a)
    sorted_idx = np.argsort(conv)[::-1]
    active = sorted_idx[:top_k]
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        grad += grad_at_peak(a, k)
    grad /= top_k
    return grad


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None
    if not result.success or result.x is None:
        return None
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def smooth_gradient(a, conv, beta):
    """Smoothed gradient via log-sum-exp."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]

def optimize_smoothed(a0, time_limit, beta_init=50.0, beta_end=500.0,
                      lr_init=0.012, momentum=0.9, seed=0, top_k_switch=300):
    """
    Log-sum-exp smoothed gradient descent with Nesterov momentum
    and periodic switching to subgradient mode.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)

        # Choose gradient type
        if no_improve > top_k_switch:

        else:
            grad = smooth_gradient(y, conv_y, beta)

        # Momentum update

        a = project_simplex(a + v)

        conv = fast_convolve(a)

            a = project_simplex(a + noise)
            v *= 0.2

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    # Bell / raised cosine variants
    for center_frac in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            g = np.exp(-0.5 * ((x - center_frac * n) / (width * n)) ** 2)
            g = np.maximum(g, 1e-10)
            g /= g.sum()
            inits[f'gauss_{int(center_frac*100)}_{int(width*100)}'] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Sinc-like (for low peak sidelobe properties)
    xc = x - n / 2
    sinc = np.sinc(xc / (n * 0.3))
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 0:
        sinc /= sinc.sum()
        inits['sinc'] = sinc

    # Windowed (Hann, Blackman)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    hann = np.maximum(hann, 1e-6)
    hann /= hann.sum()
    inits['hann'] = hann

    blackman = (0.42 - 0.5 * np.cos(2 * np.pi * x / max(n - 1, 1))
                + 0.08 * np.cos(4 * np.pi * x / max(n - 1, 1)))
    blackman = np.maximum(blackman, 1e-6)
    blackman /= blackman.sum()
    inits['blackman'] = blackman

    # Random
    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['rand_c'] = rng.dirichlet(np.ones(n) * 0.2)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

            if t_alloc < 5:
                break

            a0 = inits[name]

            # Quick optimization with smoothed gradient
            a_opt, obj_opt = optimize_smoothed(
                a0, t_alloc, beta_init=40, beta_end=200,
                lr_init=0.01, momentum=0.88, seed=rng.integers(0, 10000),
                top_k_switch=200)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f} (n={len(best_sequence)} if best_sequence else None)", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)

    # Select top candidates for deep optimization
    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen_keys:
            top_configs.append((n, a_opt))
            seen_keys.add(key)
        if len(top_configs) >= 4:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add diverse restarts at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.4)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.0)))

    # Remove duplicates (same n and very similar vectors)
    filtered = []
    for n, a0 in top_configs:
        is_dup = False
        for n2, a2 in filtered:
            if n == n2:
                cos_sim = np.dot(a0, a2) / (np.linalg.norm(a0) * np.linalg.norm(a2) + 1e-10)
                if cos_sim > 0.99:
                    is_dup = True
                    break
        if not is_dup:
            filtered.append((n, a0))
    top_configs = filtered[:6]

    remaining = deadline - time.monotonic() - 60.0
    per_config = max(20.0, remaining / len(top_configs))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 50:
            break
        t_alloc = min(per_config, 300.0)
        if t_alloc < 15:
            break

        print(f"  Config {i}: n={n}, t_alloc={t_alloc:.0f}s", flush=True)

        # Sub-phase A: Hybrid optimizer (60% of allocation)
        t_a = t_alloc * 0.55
        a_a, obj_a = optimize_hybrid(
            a0, t_a, seed=rng.integers(0, 100000))
        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    Hybrid: {score_a:.6f}", flush=True)

        # Sub-phase B: Smoothed gradient refinement (25%)
        t_b = t_alloc * 0.25
        if t_b > 10:
            a_b, obj_b = optimize_smoothed(
                a_a, t_b, beta_init=100, beta_end=600,
                lr_init=0.005, momentum=0.92,
                seed=rng.integers(0, 100000), top_k_switch=150)
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    Smooth: {score_b:.6f}", flush=True)
            a_a = a_b

        # Sub-phase C: LP + small steps (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_c, obj_c = optimize_hybrid(
                a_a, t_c, seed=rng.integers(0, 100000))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Refine: {score_c:.6f}", flush=True)

    # Phase 3: Intensive polish on the best
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        best_n = len(best_sequence)

        # Multiple polish attempts with different strategies
        polish_strategies = [
            ('high_beta_smooth', 0.25),
            ('hybrid_fine', 0.30),
            ('lp_polish', 0.25),
            ('subgrad_fine', 0.20),
        ]

        for pname, frac in polish_strategies:
            if time.monotonic() > deadline - 10:
                break
            t_p = remaining * frac
            if t_p < 5:
                continue

            print(f"  Polish: {pname}, t={t_p:.0f}s", flush=True)

            if pname == 'high_beta_smooth':
                a_p, _ = optimize_smoothed(
                    best_sequence, t_p, beta_init=300, beta_end=1500,
                    lr_init=0.002, momentum=0.95,
                    seed=rng.integers(0, 100000), top_k_switch=100)
            elif pname == 'hybrid_fine':
                a_p, _ = optimize_hybrid(best_sequence, t_p, seed=rng.integers(0, 100000))
            elif pname == 'lp_polish':
                # LP-based: repeatedly solve LP with tight constraint
                a_p = best_sequence.copy()
                a_p = a_p / a_p.sum()
                t_lp_start = time.monotonic()
                while time.monotonic() - t_lp_start < t_p:
                    conv_p = fast_convolve(a_p)
                    M_p = conv_p.max()
                    b_p = lp_direction(a_p, M_p * 1.0005, len(a_p))
                    if b_p is None:
                        break
                    improved = False
                    for t in [0.1, 0.05, 0.03, 0.02, 0.01]:
                        a_try = (1 - t) * a_p + t * b_p
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        score_try = float(evaluate_sequence(a_try.tolist()))
                        if np.isfinite(score_try) and score_try < best_score - 1e-8:
                            best_score = score_try
                            best_sequence = a_try.copy()
                            a_p = a_try.copy()
                            improved = True
                            break
                    if not improved:
                        break
                a_p = a_p
            elif pname == 'subgrad_fine':
                a_p, _ = optimize_smoothed(
                    best_sequence, t_p, beta_init=2000, beta_end=10000,
                    lr_init=0.001, momentum=0.97,
                    seed=rng.integers(0, 100000), top_k_switch=50)

            score_p = float(evaluate_sequence(a_p.tolist()))
            if np.isfinite(score_p) and score_p < best_score:
                best_score = score_p
                best_sequence = a_p.copy()
                print(f"    After {pname}: {score_p:.6f}", flush=True)

        # Final micro-adjustments: single-coordinate moves
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5 and best_sequence is not None:
            a_fin = best_sequence.copy()
            a_fin = np.maximum(a_fin, 0)
            a_fin = a_fin / a_fin.sum()
            n_fin = len(a_fin)
            a_fin_score = float(evaluate_sequence(a_fin.tolist()))

            t_fin_start = time.monotonic()
            improvement_threshold = 1e-8
            consecutive_fails = 0

            while time.monotonic() - t_fin_start < remaining and consecutive_fails < 500:
                conv_fin = fast_convolve(a_fin)
                max_idx = int(np.argmax(conv_fin))

                # Find contributing positions
                i_start = max(0, max_idx - n_fin + 1)
                i_end = min(n_fin - 1, max_idx)

                if i_end < i_start:
                    break

                contributions = np.array([a_fin[i] * a_fin[max_idx - i] for i in range(i_start, i_end + 1)])
                if contributions.sum() < 1e-15:
                    break

                # Try moving from the biggest contributor
                i_max = i_start + int(np.argmax(contributions))
                improved = False

                for _ in range(5):
                    j = int(rng.integers(0, n_fin))
                    if j == i_max or abs(j - i_max) < 2:
                        continue

                    for frac in [0.005, 0.01, 0.02, 0.03]:
                        amt = a_fin[i_max] * frac
                        if amt < 1e-12:
                            break
                        a_try = a_fin.copy()
                        a_try[i_max] -= amt
                        a_try[j] += amt
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < conv_fin.max() - improvement_threshold:
                            a_fin = a_try
                            new_score = float(evaluate_sequence(a_fin.tolist()))
                            if np.isfinite(new_score) and new_score < best_score:
                                best_score = new_score
                                best_sequence = a_fin.copy()
                            improved = True
                            consecutive_fails = 0
                            break
                    if improved:
                        break

                if not improved:
                    consecutive_fails += 1

            print(f"  After micro-adjust: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
