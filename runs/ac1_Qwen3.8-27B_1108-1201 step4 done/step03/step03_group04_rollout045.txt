def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:

    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) structured initializations, (2) Nesterov accelerated subgradient
with peak-focused softmax weighting, (3) LP-based Frank-Wolfe polishing,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj(a):
    """Compute 2*n*max(conv) for normalized sequence (sum~1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([1.0])
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def peak_subgradient(a, conv, top_k=40, temperature=None):
    """Compute weighted subgradient focusing on top peaks using softmax weighting."""
    n = len(a)
    k = min(top_k, len(conv))
    if k < 1:
        return np.zeros(n)
    
    peak_indices = np.argpartition(conv, -k)[-k:]
    peak_values = conv[peak_indices]
    max_pv = peak_values.max()
    
    if temperature is None:
        temperature = max(0.01 * (max_pv - peak_values.min() + 1e-15), 1e-10)
    
    weights = np.exp((peak_values - max_pv) / temperature)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    
    grad = np.zeros(n)
    for i in range(k):
        if weights[i] < 1e-12:
            continue
        ki = int(peak_indices[i])
        w = weights[i]
        valid_start = max(0, ki - n + 1)
        valid_end = min(n - 1, ki)
        if valid_start <= valid_end:
            indices = np.arange(valid_start, valid_end + 1)
            grad[indices] += 2.0 * w * a[ki - indices]
    
    return grad


def nesterov_subgradient(a0, time_limit, top_k=40, temp_base=0.005,
                         lr_scale=0.5, momentum_decay=0.9999):
    """Nesterov accelerated subgradient method on the simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    y = a.copy()
    v = np.zeros(n)
    t_iter = 1.0
    momentum = 0.9
    
    t0 = time.monotonic()
    it = 0
    patience = 0
    max_patience = 3000
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        
        # Ensure y is on simplex
        y = project_simplex(y)
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        g = peak_subgradient(y, conv_y, top_k=top_k, temperature=temp_base)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Step size
        lr = lr_scale / np.sqrt(t_iter)
        
        # Update
        a_new = project_simplex(y - lr * (g - g.mean()))
        
        # Update momentum
        v = (a_new - a) / (1.0 + momentum) if momentum > 0 else (a_new - a)
        
        # Decay momentum slightly
        momentum = min(0.99, momentum * momentum_decay + 0.99 * (1 - momentum_decay))
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        t_iter += 1
        
        # Anneal temperature
        temp_base = max(0.001, 0.005 * (0.999 ** it))
        
        # Reset if stuck
        if patience > max_patience:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            y = a.copy()
            v = np.zeros(n)
            patience = 0
            max_patience = int(max_patience * 1.4)
        
        it += 1
    
    return best_a, best_obj


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix A where (A@b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((a[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=2.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        lp_time = min(2.0, remaining * 0.3)
        b = lp_oracle(a, M, time_limit=lp_time)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-9):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_local = compute_obj(a)
            for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
                cand = (1 - t) * a + t * b
                obj_c = compute_obj(cand)
                if obj_c < best_local:
                    best_local = obj_c
                    best_t = t
            
            if best_t > 0.01:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 12345)
            a = project_simplex(a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    FW-LP it={it}, obj={best_obj:.6f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_rounds=None):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            # Try modifying coordinate j
            old_val = a[j]
            if old_val < 1e-12:
                # Try setting it to a small positive value
                for frac in [0.001, 0.002, 0.005]:
                    a[j] = frac
                    a_new = a / a.sum()
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        improved = True
                    else:
                        a[j] = old_val
                        break
            else:

                for factor in [0.5, 0.7, 1.3, 1.5, 2.0, 0.1]:
                    a[j] = old_val * factor
                    s_new = a.sum()
                    if s_new < 1e-15:

                        continue
                    a_new = a / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        improved = True
                        break
                    a[j] = old_val
        
        it += 1
        if not improved and it > 1:

            rng = np.random.default_rng(it * 7919)

            a[idx] *= rng.choice([0.5, 1.5, 2.0, 0.1])

    return best_a, best_obj


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Edge concentrated (mass at both ends)
    width = max(1, n // 4)
    edge = np.zeros(n)
    edge[:width] = 1.0
    edge[-width:] = 1.0
    edge /= edge.sum()
    inits['edge'] = edge.copy()
    
    # Gaussian (centered)
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - (n - 1) / 2.0) / (sigma_frac * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sigma_frac*100)}'] = g.copy()
    
    # Wilson-like: 1/sqrt(distance from edge)
    x = np.arange(n, dtype=np.float64)
    dist_left = x + 1
    dist_right = n - x
    wilson = 1.0 / (np.sqrt(dist_left) + np.sqrt(dist_right) + 1)
    wilson /= wilson.sum()
    inits['wilson'] = wilson.copy()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sparse random
    sparse = rng.random(n)
    threshold = 0.6
    sparse[sparse < threshold] = 0.0
    if sparse.sum() > 1e-10:
        sparse /= sparse.sum()
    else:
        sparse = rng.dirichlet(np.ones(n))
    inits['sparse'] = sparse.copy()
    
    # Alternating (checkerboard-like)
    alt = np.ones(n)
    alt[::2] = 0.5
    alt /= alt.sum()
    inits['alternating'] = alt.copy()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.clip(parent, 0, 1000.0)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_interp / ps
    
    # Difference-set inspired: use quadratic residue pattern
    # For prime n, use Legendre symbol pattern (approximated for non-prime)
    quad = np.zeros(n)
    for i in range(n):
        # Simple periodic pattern based on squares
        quad[i] = 1.0 if (i * i) % max(1, n // 3) < n // 6 else 0.3
    quad /= quad.sum()
    inits['quad'] = quad.copy()
    
    return inits

        print("=== Phase 6: Final LP ===")
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_opt, obj_opt = fw_lp_optimize(best_sequence, time_limit=remaining, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP score={score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    else:
        print("\nWARNING: Final score is not finite!")
    
    return [float(x) for x in best_sequence]
