"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M (componentwise), g >= 0.
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum, result
    except Exception:
        pass
    return None, 0.0, None


def subgradient_descent(a0, time_limit, lr_init=0.02, momentum_init=0.9, top_k=60, temp_init=0.001, batch=20):

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    gradient_sq = np.zeros(n)  # AdaGrad accumulator
    
    _, conv = _obj_and_conv(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    momentum = momentum_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find active peaks with soft weighting
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            
            # Adaptive temperature: lower when peaks are more concentrated
            top1 = peak_vals.max()
            top_mean = peak_vals.mean()
            peak_spread = max(top1 - top_mean, 1e-10)
            temp_adaptive = temp * (1.0 + peak_spread / (top1 + 1e-10))
            
            scaled = peak_vals / max(temp_adaptive, 1e-10)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum

            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            # AdaGrad adaptive LR
            gradient_sq += sg * sg
            lr_adaptive = lr / (np.sqrt(gradient_sq) + 1e-8)

            v = momentum * v - lr_adaptive * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
                v = np.zeros(n)
            
            _, conv = _obj_and_conv(a, n)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            iteration += 1
        
        # Decay learning rate
        lr *= 0.9995
        temp *= 1.003
        momentum = min(0.98, momentum + 0.0002)

            scale = 0.003 + 0.002 * min(no_improve // 200, 5)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_and_conv(a, n):
    """Return (2*n*max(conv), conv) for sequence a of length n."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M):
    """
    Solve LP: maximize sum(g) s.t. (a*g)(k) <= M for all k, g >= 0.
    Returns (g_normalized, g_sum, dual_vars) or (None, 0, None).
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = result.x
            g_sum = float(g.sum())
            dual = None
            try:
                if result.ineqlin is not None and result.ineqlin.marginals is not None:
                    dual = result.ineqlin.marginals
            except Exception:
                pass
            if g_sum > 1e-10:
                return g / g_sum, g_sum, dual
    except Exception:
        pass
    return None, 0.0, None

def coord_descent(a0, time_limit, max_sweeps=200):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_and_conv(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        any_improve = False

        for j in range(n):
            w = a[j]

            # Precompute influence: a_shift[k] = a(k-j) for k=0..2n-2
            # a_shift[k] = a[k-j] if 0<=k-j<n else 0
            # This is a placed at positions j..j+n-1 in a vector of length 2n-1
            a_shift = np.zeros(2 * n - 1)
            a_shift[j:j + n] = a

            k_quad = 2 * j  # position where d^2 term appears
            quad_valid = (0 <= k_quad < 2 * n - 1)

            # Current objective at d=0 (after renormalization, which is identity)
            # f(d) = 2n * max_k[conv(k) + 2d*a_shift(k) + d^2*[k==2j]] / (s+d)^2
            # We search for d in [-w + eps, w] (keep a[j] in [0, 2w])

            best_d = 0.0
            best_f = best_obj  # current objective (d=0 is always feasible)

            # Try a geometric set of d values
            candidates = [0.0]
            for fac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                candidates.append(fac * w)
                candidates.append(-fac * w * 0.8)  # going negative is harder
            candidates.append(0.3 * w)
            candidates.append(-0.3 * w)

            for d in candidates:
                new_w = w + d
                if new_w < 0:
                    continue
                new_s = s + d
                if new_s < 1e-15:
                    continue
                new_conv = conv + 2.0 * d * a_shift
                if quad_valid:
                    new_conv[k_quad] += d * d
                obj_try = 2.0 * n * new_conv.max() / (new_s * new_s)
                if obj_try < best_f - 1e-15:
                    best_f = obj_try
                    best_d = d

            if best_d != 0.0:
                a[j] = w + best_d
                s = s + best_d
                # Renormalize
                a = a / s
                s = 1.0
                _, conv = obj_and_conv(a, n)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-15:
                    best_obj = cur_obj
                    best_a = a.copy()
                    any_improve = True

        if not any_improve:
            break

    return best_a, best_obj
