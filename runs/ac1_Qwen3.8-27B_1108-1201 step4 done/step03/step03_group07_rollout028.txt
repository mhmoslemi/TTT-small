def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=500):
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-15})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def optimize_pg_smooth(a0, beta, lr=0.05, n_iter=500):
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

        obj = 2.0 * n * sm

            best_a = a.copy()

def active_set_refine(a, n_iter=200, tol=1e-10):
    """
    Refine using active set approach:
    - Identify the top-k lags where convolution is near maximum
    - Try to equalize them using gradient steps on the max
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_score = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        max_val = conv.max()
        score = 2.0 * n * max_val
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active lags (within 1% of max)
        threshold = max_val * 0.999
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient of the max (using uniform weights over active lags)
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        
        grad = compute_grad_smooth_conv(a, w)
        
        # Project gradient to the tangent space of the simplex
        grad_tangent = grad - np.mean(grad)
        
        # Gradient descent step
        step = 0.01 / (np.linalg.norm(grad_tangent) + 1e-15)
        a_new = project_simplex(a - step * grad_tangent)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        
        if score_new < score:
            a = a_new
        else:
            # Try smaller step
            step *= 0.1
            a_new = project_simplex(a - step * grad_tangent)
            conv_new = fast_convolve(a_new)
            score_new = 2.0 * n * conv_new.max()
            if score_new < score:
                a = a_new
            else:
                break
        
        # Decay step size
        step *= 0.99
    
    return best_a


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive multi-stage optimization:
1) Multi-beta L-BFGS-B annealing from parent sequence
2) Projected gradient with very high beta for precision
3) Active-set refinement targeting peak lags directly
4) Multi-restart with structured perturbations
5) Exploration of nearby sequence lengths
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a via FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_obj(x, n, beta):
    """Objective for L-BFGS-B with softmax parametrization."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def soft_jac(x, n, beta):
    """Jacobian for L-BFGS-B with softmax parametrization."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def run_lbfgsb(a0, beta, maxiter=500):
    """Run L-BFGS-B optimization from a0 with given beta."""
    n = len(a0)
    a_clipped = np.maximum(a0, 1e-30)
    a_clipped = a_clipped / a_clipped.sum()
    x0 = np.log(a_clipped)
    try:
        result = minimize(soft_obj, x0, jac=soft_jac, args=(n, beta),
                          method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxls': 100, 'ftol': 1e-18, 'gtol': 1e-12})
        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        return a_opt
    except Exception:
        return a0


def run_pg_high_beta(a0, beta, n_iter=300, lr=0.02):
    """Projected gradient descent with high-beta smooth max on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_true_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        true_obj = 2.0 * n * conv.max()

        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a.copy()

        grad = compute_grad_conv(a, w)
        grad_tan = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-16:
            break

        step = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(12):
            a_new = project_simplex(a - step * grad_tan)

                a = a_new
                improved = True
                break
            step *= 0.5

        lr *= 0.998

        if gnorm < 1e-15:
            break

        step = step_size / (gnorm + 1e-15)
        a_new = project_simplex(a - step * grad_tan)
        conv_new = fast_convolve(a_new)

                    break

def spectral_init(n, rng, n_trials=30):
    """Initialize via random phase spectrum, keeping best non-negative projection."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        ph = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * ph[k - 1])
            A[fft_size - k] = np.exp(-1j * ph[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * ph[-1])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a
