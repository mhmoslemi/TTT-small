**"Anti-peak" initialization:**

**KKT refinement:**
At the optimum, for all j with a[j] > 0:
sum_k w[k] * a[k-j] = const (where w are the softmax weights over the active lags)

This means the cross-correlation of w with a should be constant (on the support of a). We can use this to define a fixed-point iteration:

1. Compute conv, M, w (softmax over top lags)
2. Compute c[j] = sum_k w[k] * a[k-j] (cross-correlation)
3. target = mean(c[j] for j where a[j] > 0)
4. Update: a[j] = a[j] * (target / c[j])^t (for small t, like exponential tilting)
5. Project to non-negative, renormalize

This is similar to iterative proportional fitting / Sinkhorn's algorithm. It should converge to a local optimum where the KKT conditions are approximately satisfied.

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def smooth_max(x, beta):
    """Compute smooth max (log-sum-exp approximation) and softmax weights."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    sm = x_max + np.log(total) / beta
    return sm, w


def cross_correlation_w_a(w, a):
    """Compute cross[j] = sum_k w[k] * a[k-j] via FFT."""
    n = len(a)
    N = next_fast_len(2 * n - 1)
    a_rev_padded = np.zeros(N)
    a_rev_padded[0] = a[0]
    if n > 1:
        a_rev_padded[N - n + 1:] = a[1:][::-1]
    W = fft(w, N)
    A_rev = fft(a_rev_padded, N)
    return np.real(ifft(W * A_rev, N))[:n]


def compute_objective_and_gradient(x, n, beta):
    """Compute objective and gradient for log-softmax parametrization."""
    a = np.exp(x - x.max())
    a /= a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max(conv, beta)
    obj = 2.0 * n * sm

    cross = cross_correlation_w_a(w, a)
    grad_a = 4.0 * n * cross
    lagrange = np.dot(a, grad_a)
    grad_x = a * (grad_a - lagrange)

    return obj, grad_x

def projected_gradient_refine(a, n_iters=3000, lr=0.02, beta=50.0):
    """Projected gradient descent with backtracking on the simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iters):

            best_a = a.copy()

        cross = cross_correlation_w_a(w, a)
        grad_a = 4.0 * n * cross
        # Project gradient to tangent space of simplex
        grad_a = grad_a - np.mean(grad_a)
        norm_g = np.linalg.norm(grad_a)
        if norm_g < 1e-14:
            break

        step_lr = lr
        for _ in range(8):
            a_new = a - step_lr * grad_a
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:

            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            step_lr *= 0.5

        a = a_new
        lr *= 0.9997

def init_flat_spectrum(n, rng):

    if N % 2 == 0:

    # Add small random perturbation to break symmetry
    a = a + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def init_from_parent(n, parent, rng):
    """Resample parent sequence to length n."""
    if parent is None or len(parent) == 0:
        return None
    p = np.maximum(parent, 0.0)
    ps = p.sum()
    if ps < 1e-10:
        return None
    p = p / ps
    if len(p) == n:
        return p.copy()
    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    inits['uniform'] = init_uniform(n)
    inits['gauss_03'] = init_gaussian(n, 0.3)
    inits['gauss_05'] = init_gaussian(n, 0.5)
    inits['bimodal'] = init_bimodal(n)
    inits['barker'] = init_barker_tiled(n, rng)
    inits['flat_spec'] = init_flat_spectrum(n, rng)
    inits['zc_like'] = init_zadoff_chu_like(n, rng)
    inits['random_dir'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sp'] = rng.dirichlet(np.ones(n) * 0.05)

    parent_init = init_from_parent(n, parent, rng)
    if parent_init is not None:
        inits['parent'] = parent_init

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: initial scan, deep L-BFGS-B, KKT refinement,
    projected gradient polish, and random restarts.
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
            # Score the parent
            parent_score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
                print("Parent score: %.6f" % parent_score)
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-8:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            elapsed = time.monotonic() - t_start
            print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), elapsed))
        return score if np.isfinite(score) else float('inf')

    # Phase 1: Quick scan over lengths and initializations
    print("=== Phase 1: Quick Scan ===")
    lengths = [192, 256, 320, 384]
    phase1_deadline = t_start + 400.0

    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        inits = generate_inits(n, parent, rng)

        # Priority order
        priority = ['parent', 'flat_spec', 'zc_like', 'gauss_03', 'barker', 'random_dir']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:6]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            a0 = inits[name]
            # Quick L-BFGS-B with moderate settings
            a_opt, _ = optimize_lbfgsb_multibeta(
                a0, betas=(50.0, 30.0, 15.0), maxiter_per_phase=150)
            score_and_update(a_opt)

    print("Phase 1 done. Best: %.6f" % best_score)

    # Phase 2: Deep optimization on best length
    print("=== Phase 2: Deep Optimization ===")
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    # Focus on the best length and one alternative
    target_n = best_n
    phase2_deadline = deadline - 250.0

    # Deep multi-beta L-BFGS-B from current best
    if best_sequence is not None and time.monotonic() < phase2_deadline - 30:
        print("  Deep optimizing n=%d from best..." % target_n)
        t_avail = min(phase2_deadline - time.monotonic() - 20, 300.0)
        maxiter = int(min(600, max(100, t_avail * 3)))
        a_opt, _ = optimize_lbfgsb_multibeta(
            best_sequence, betas=(100.0, 60.0, 30.0, 15.0, 8.0, 4.0, 2.0),
            maxiter_per_phase=maxiter // 7)
        score_and_update(a_opt)

    # KKT refinement (novel step)
    if best_sequence is not None and time.monotonic() < phase2_deadline - 20:
        print("  KKT refinement...")
        a_kkt, _ = kkt_refine(best_sequence, n_steps=300, t=0.4, beta=60.0)
        score_and_update(a_kkt)
        if score_and_update(a_kkt) < best_score:
            # Re-optimize from KKT result
            if time.monotonic() < phase2_deadline - 20:
                a_opt, _ = optimize_lbfgsb_multibeta(
                    a_kkt, betas=(40.0, 20.0, 10.0, 5.0), maxiter_per_phase=200)
                score_and_update(a_opt)

    # Try alternating KKT and L-BFGS-B
    if best_sequence is not None and time.monotonic() < phase2_deadline - 30:
        print("  Alternating KKT + L-BFGS-B...")
        a_cur = best_sequence.copy()
        for round_i in range(4):
            if time.monotonic() > phase2_deadline - 20:
                break
            # KKT step
            a_kkt, _ = kkt_refine(a_cur, n_steps=100, t=0.5, beta=40.0)
            score_and_update(a_kkt)
            a_cur = a_kkt
            # L-BFGS-B step
            a_opt, _ = optimize_lbfgsb_multibeta(
                a_cur, betas=(20.0, 10.0, 5.0), maxiter_per_phase=200)
            score_and_update(a_opt)
            a_cur = a_opt

    # Phase 3: Multi-start with diverse initializations
    print("=== Phase 3: Multi-Start ===")
    phase3_deadline = deadline - 150.0

    for n in [best_n, max(128, best_n // 2), min(512, best_n * 2)]:
        if time.monotonic() > phase3_deadline - 30:
            break
        inits = generate_inits(n, parent, rng)
        if best_sequence is not None:
            pp = init_from_parent(n, best_sequence, rng)
            if pp is not None:
                inits['from_best'] = pp

        priority3 = ['from_best', 'parent', 'flat_spec', 'zc_like', 'gauss_05', 'barker']
        init_names = [nm for nm in priority3 if nm in inits] + \
                     [nm for nm in inits if nm not in priority3]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() > phase3_deadline - 20:
                break
            a0 = inits[name]
            a_opt, _ = optimize_lbfgsb_multibeta(
                a0, betas=(60.0, 30.0, 15.0, 8.0), maxiter_per_phase=250)
            score_and_update(a_opt)
            # KKT polish
            if time.monotonic() < phase3_deadline - 10:
                a_kkt, _ = kkt_refine(a_opt, n_steps=150, t=0.3, beta=50.0)
                score_and_update(a_kkt)

    print("Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Fine-tuning
    print("=== Phase 4: Fine-tuning ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0

        # Very low beta L-BFGS-B for precision
        for beta in [3.0, 2.0, 1.5, 1.0]:
            if remaining < 15:
                break
            maxiter = int(min(400, max(50, remaining // 4)))
            a_opt, _ = optimize_lbfgsb_multibeta(
                best_sequence, betas=(beta,), maxiter_per_phase=maxiter)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 15.0

        # KKT with very low step size (polish)
        if remaining > 20:
            a_kkt, _ = kkt_refine(best_sequence, n_steps=500, t=0.1, beta=30.0)
            score_and_update(a_kkt)
            remaining = deadline - time.monotonic() - 15.0

        # Projected gradient with low beta (polish)
        if remaining > 20:
            n_iters = int(min(5000, remaining * 15))
            a_pg, _ = projected_gradient_refine(best_sequence, n_iters=n_iters,
                                                lr=0.01, beta=20.0)
            score_and_update(a_pg)
            remaining = deadline - time.monotonic() - 15.0

        # Random perturbations + re-optimize
        if remaining > 40:
            for trial in range(6):
                if remaining < 15:
                    break
                    pass
                scale = 0.002 + 0.003 * rng.random()
                noise = rng.standard_normal(len(best_sequence)) * scale
                a_pert = best_sequence.copy()
                a_pert = a_pert + noise * a_pert
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_opt, _ = optimize_lbfgsb_multibeta(
                        a_pert, betas=(15.0, 8.0, 4.0), maxiter_per_phase=150)
                    score_and_update(a_opt)
                    remaining = deadline - time.monotonic() - 15.0

        # Final KKT + L-BFGS-B cycle
        if remaining > 30:
            a_final = kkt_refine(best_sequence, n_steps=200, t=0.2, beta=20.0)[0]
            score_and_update(a_final)
            if remaining > 20:
                a_opt, _ = optimize_lbfgsb_multibeta(
                    a_final, betas=(5.0, 2.0, 1.0), maxiter_per_phase=200)
                score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
