Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] for valid j
    # For efficiency, use the banded structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fp_peak_reduction(a, n_peaks=10, reduction=0.005):
    """
    QP-based peak reduction: find smallest step d that reduces the top peaks.
    Minimizes ||d||^2 subject to linearized peak constraints, a+d >= 0, sum(d) = 0.
    """
    n = len(a)
    c = fast_convolve(a)
    peak_locs = np.argsort(c)[-n_peaks:]
    
    # Build linear constraints: sum_j a[k-j]*d[j] <= -c[k]*reduction/2
    # For each peak k, the valid j range is [max(0, k-n+1), min(n-1, k)]
    
    # We'll use scipy minimize with SLSQP
    # Variables: d (length n)
    # Objective: ||d||^2
    # Constraints:
    #   For each peak k: sum_{j in valid} a[k-j] * d[j] + c[k]*reduction/2 <= 0
    #   For each i: d[i] + a[i] >= 0 (i.e., -d[i] <= a[i])
    #   sum(d) = 0
    
    def obj(d):
        return np.dot(d, d)
    
    def grad(d):
        return 2.0 * d
    
    constraints = []
    # Equality: sum(d) = 0
    constraints.append({'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)})
    
    # Inequality constraints for peaks
    for k in peak_locs:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # coeff_j = a[k-j] for j in [js, je]
        coeffs = np.zeros(n)
        coeffs[js:je+1] = a[k - np.arange(js, je+1)]
        rhs = c[k] * reduction / 2.0
        # constraint: sum_j coeffs_j * d_j <= -rhs
        # i.e., -sum_j coeffs_j * d_j + rhs >= 0
        constraints.append({'type': 'ineq', 'fun': lambda d, c=coeffs, r=rhs: np.dot(-c, d) + r,
                           'jac': lambda d, c=coeffs: -c})
    
    # Bounds: d[i] >= -a[i]
    bounds = [(-a[i], None) for i in range(n)]
    
    res = minimize(obj, np.zeros(n), jac=grad, constraints=constraints, 
                   bounds=bounds, method='SLSQP', options={'maxiter': 200, 'ftol': 1e-12})
    
    if res.success:
        return res.x
    return None

def sg_polyak(a, time_limit, top_k=20, lb_factor=1.0):
    """
    Subgradient descent with Polyak step size.
    The lower bound on max(conv) is 1/(2n-1) (with sum(a)=1).
    """
    n = len(a)
    f = np.maximum(a.astype(np.float64), 0.0)
    f = f / f.sum()
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    lower_bound = 1.0 / (2 * n - 1)
    
    t0 = time.monotonic()
    no_improve = 0
    step_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(f)
        cur_obj = c.max()
        
        # Find peaks
        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]
        
        # Compute subgradient (average over peaks with weight proportional to value)
        # Use weighted average: peaks with higher values get more weight
        pv = c[peaks]
        # Softmax weighting for smoothness
        scaled = (pv - pv.max()) / (1e-8 + pv.max() - pv.min() + 1e-8)
        w = np.exp(scaled * 0.5)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Polyak step size
        sg_norm_sq = np.dot(sg, sg)
        if sg_norm_sq < 1e-30:
            break
        gap = max(0.0, cur_obj - lower_bound)
        eta = step_scale * gap / sg_norm_sq
        
        # Cap the step
        eta = min(eta, 0.1)
        
        # Gradient step + projection
        f_new = project_simplex(f - eta * sg)
        
        if fast_convolve(f_new).max() < best_obj - 1e-12:
            best_obj = fast_convolve(f_new).max()
            best_f = f_new.copy()
            no_improve = 0
            step_scale = min(step_scale * 1.05, 5.0)
        else:
            no_improve += 1
            step_scale = max(step_scale * 0.95, 0.1)
        
        f = f_new
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
            step_scale = 1.0
    
    return best_f, best_obj

def fw_lp_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    n = len(f)

    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()

    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in np.linspace(0.1, 1.0, 10):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            c = fast_convolve(f)
            M = c.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            c = fast_convolve(f)
            M = c.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def compute_subgradient(a, conv, n_peaks=20, temperature=1.0):
    """Compute weighted subgradient of max(conv(a,a)) using exponential peak weighting."""
    n = len(a)
    if n_peaks >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

    pv = conv[peaks]
    # Exponential weighting: higher peaks get more weight
    scaled = (pv - pv.max()) / (temperature * (pv.max() + 1e-15))
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg


def sg_adaptive(a0, time_limit, top_k=30, temp=0.001, damping=0.999,
                max_step=0.05, min_step=1e-6, restart_every=500):
    """
    Subgradient descent with adaptive Polyak step size and exponential peak weighting.
    Uses the lower bound 1/(2n-1) for the excess.
    """
    n = len(a0)
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    lower_bound = 1.0 / (2 * n - 1)

    t0 = time.monotonic()
    no_improve = 0
    step_scale = 1.0
    v = np.zeros(n)  # momentum
    mom = 0.9
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()
        gap = max(1e-10, cur_obj - lower_bound)

        sg = compute_subgradient(f, conv, n_peaks=top_k, temperature=temp)
        sg_norm_sq = np.dot(sg, sg)

        if sg_norm_sq < 1e-30:
            break

        # Polyak step with damping
        eta = step_scale * gap / sg_norm_sq
        eta = np.clip(eta, min_step, max_step)

        # Momentum update
        v = mom * v - eta * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            step_scale = min(step_scale * 1.02, 10.0)
            # Decay momentum slowly
            mom = min(mom * 1.0005, 0.97)
        else:
            no_improve += 1
            step_scale = max(step_scale * 0.98, 0.05)

        f = f_new

        # Periodic restart
        if no_improve > restart_every:
            rng = np.random.default_rng((it * 31 + 17) % 1000000)
            scale = 0.001 + 0.003 * (no_improve // (restart_every * 4))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
            step_scale = 1.0
            mom = 0.9
            temp *= 1.5  # Increase temperature for more exploration

    return best_f, best_obj


def qp_peak_shave(a, n_peaks=12, reduction=0.004, max_iter_qp=50):
    """
    QP-based peak shaving: minimize ||d||^2 subject to linearized peak constraints.
    Uses scipy SLSQP.
    """
    n = len(a)
    conv = fast_convolve(a)
    peak_locs = np.argsort(conv)[-n_peaks:]

    def obj(d):
        return np.dot(d, d)

    def grad(d):
        return 2.0 * d

    constraints = []
    # Equality: sum(d) = 0
    constraints.append({
        'type': 'eq',
        'fun': lambda d: np.sum(d),
        'jac': lambda d: np.ones(n)
    })

    # Inequality constraints for peaks (linearized)
    for k in sorted(peak_locs):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # d(conv[k])/d(a[j]) = 2*a[k-j] for j in [js, je]
        coeffs = np.zeros(n)
        coeffs[js:je + 1] = 2.0 * a[k - np.arange(js, je + 1)]
        rhs = conv[k] * reduction
        # Constraint: coeffs @ d >= -rhs, i.e., -coeffs @ d + rhs >= 0 (ineq form)
        def constr_fun(d, c=coeffs, r=rhs):
            return np.dot(c, d) + r
        def constr_jac(d, c=coeffs):
            return c
        constraints.append({'type': 'ineq', 'fun': constr_fun, 'jac': constr_jac})

    bounds = [(-a[i] * 0.5, None) for i in range(n)]

    try:
        res = minimize(obj, np.zeros(n), jac=grad, constraints=constraints,
                       bounds=bounds, method='SLSQP',
                       options={'maxiter': max_iter_qp, 'ftol': 1e-14})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    it = 0
    m_exp = 1.0

        it += 1

        g = solve_lp(f, M * m_exp)

            m_exp *= 1.12
            if m_exp > 6.0:
                m_exp = 1.0

        if g_sum < 1e-15:

        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-8:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.02, 3.0)
            continue

        # Fine grid line search (40 points)

        for alpha in alphas:

            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:

                best_alpha = alpha

            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:

            else:
                no_improve += 1

            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 13 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(g / g.sum())

    # U-shape (edge concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.2
    inits.append(c1 / c1.sum())

    # Two-peak
    p1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two_peak = p1 + p2
    inits.append(two_peak / two_peak.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

                a_qp = np.maximum(a_qp, 0)
                s = a_qp.sum()
                if s < 1e-15:
                    break
                a_qp = a_qp / s
                n_qp_iters += 1

                if n_qp_iters % 5 == 0:
                    try_seq(a_qp, f"qp_{n_qp_iters}")

            if n_qp_iters > 0:
                try_seq(a_qp, f"qp_final_{n_qp_iters}")
                print(f"  QP: {n_qp_iters} iterations")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final Polish ===
    print("\n=== Phase 4: Final Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Quick FW-LP
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(50, rem)
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Fine SG
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(40, rem)
            a_opt, _ = sg_adaptive(best_seq, time_limit=t_alloc,
                                   top_k=15, temp=0.0001,
                                   damping=0.9997, max_step=0.005,
                                   restart_every=300)
            try_seq(a_opt, "final_sg")

        # Tiny perturbation + quick optimization
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            for trial in range(2):
                rem = deadline - time.monotonic() - 15
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = sg_adaptive(f_pert, time_limit=min(rem, 20),
                                       top_k=20, temp=0.0002,
                                       damping=0.999, max_step=0.01,
                                       restart_every=200)
                try_seq(a_opt, f"perturb_{trial}")

        # Another QP round on best
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_qp2 = best_seq.copy() / best_seq.sum()
            t_qp2 = time.monotonic()
            n_qp2 = 0
            while time.monotonic() - t_qp2 < min(rem, 40):
                d = qp_peak_shave(a_qp2, n_peaks=8, reduction=0.002)
                if d is None:
                    break
                a_qp2 = np.maximum(a_qp2 + d, 0)
                s = a_qp2.sum()
                if s < 1e-15:
                    break
                a_qp2 = a_qp2 / s
                n_qp2 += 1
            if n_qp2 > 0:
                try_seq(a_qp2, "final_qp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
