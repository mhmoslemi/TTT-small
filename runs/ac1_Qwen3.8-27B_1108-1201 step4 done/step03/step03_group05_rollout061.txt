from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum

    # grad_i = 2 * sum_{m} a[m] * p[m+i] (cross-correlation of a and p)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * np.conj(P), fft_size)
    return 2.0 * cross[:n]


def spectral_flatness_penalty(a, weight):
    """
    Gradient of a penalty that encourages flat power spectrum.
    This helps guide optimization toward low-peak autocorrelation.
    """
    n = len(a)
    A = np.fft.rfft(a)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    if P_mean < 1e-15:
        return np.zeros(n)
    # Gradient of sum((P - mean)^2) w.r.t. a
    # d P[k]/d a[i] involves A[k] and e^{i2pk i/n}
    # Use: grad_i = 2 * Re(IFFT(A * (P - P_mean).conj())) ... simplified
    # Actually for real a: d|A[k]|^2/da[i] = 2*Re(A[k] * conj((2π/n)*i*e^{-2πiki/n})*... )
    # Simpler: use the fact that FFT is linear
    # |A[k]|^2 = A[k]*conj(A[k]), d/da = A[k]*conj(dA[k]/da) + conj(A[k])*dA[k]/da
    # dA[k]/da[i] = e^{-2πiki/n} (for DFT), so:
    # d|A[k]|^2/da[i] = 2*Re(A[k]*e^{2πiki/n})
    # gradient_i = 2 * sum_k (P[k] - P_mean) * 2*Re(A[k]*e^{2πiki/n})
    #            = 4 * Re(IFFT(A * (P - P_mean)))
    # Note: for rfft, we need to handle the Hermitian symmetry
    grad_freq = A * (P - P_mean)
    # For real-valued IFFT of the full spectrum:
    grad = 4.0 * np.fft.irfft(grad_freq, n) * weight
    return grad


def optimize_adaptive(a0, time_limit, beta_init=50.0, beta_end=2000.0,
                      lr_init=0.02, momentum=0.9, top_k_frac=0.3,
                      spec_weight=0.0, seed=0, backtracking=True):
    """
    Adaptive smoothed gradient descent with:
    - Backtracking line search
    - Adaptive beta (increases when stalled)
    - Top-k peak concentration
    - Optional spectral regularization
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    lr = lr_init
    beta = beta_init
    top_k = max(n, int(top_k_frac * (2 * n - 1)))

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)
        grad = compute_gradient(y, conv_y, beta, top_k)

        # Add spectral regularization if enabled
        if spec_weight > 0:
            grad += spectral_flatness_penalty(y, spec_weight)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        # Backtracking line search
        if backtracking:
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            step = 1.0
            attempts = 0
            while obj_new > best_obj * 1.05 and step > 1e-4 and attempts < 10:
                step *= 0.5
                a_new = project_simplex(a + step * v)
                v = momentum * v - lr * grad
                conv_new = fast_convolve(a_new)
                obj_new = conv_new.max()
                attempts += 1

        a = a_new
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            # Slightly increase momentum when improving
            momentum = min(0.97, momentum + 0.001)
        else:
            no_improve += 1

        # Adaptive beta: increase when stalled
        if no_improve > 200:
            beta = min(5000.0, beta * 1.5)
            top_k = max(n // 2, int(top_k * 0.7))
            no_improve = 0
        elif no_improve > 500:
            # Perturb to escape
            noise_scale = 0.002 * (1.0 + 0.5 * np.sin(total_iters * 0.005))
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            v *= 0.2
            no_improve = 0
            lr = max(0.0002, lr * 0.7)
            beta = max(beta_init, beta * 0.5)

        # Adaptive learning rate
        lr = lr_init * (1.0 - 0.6 * frac)
        lr = max(0.0001, lr)

    return best_a, best_obj


def coordinate_pair_descent(a0, time_limit, seed=0):
    """
    Pair-swap coordinate descent: pick two coordinates, optimally redistribute
    their mass to reduce the max convolution.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        cur_max = conv_cur.max()

        # Find the peak position
        peak_k = int(np.argmax(conv_cur))

        # Pick pair (i, j) where i is near the peak and j is elsewhere
        # The peak at position k involves pairs (i, k-i). Pick a contributor.
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_end < i_start:
            i = rng.integers(0, n)
        else:
            # Pick the index with highest contribution to the peak
            contribs = np.array([a[i] * a[peak_k - i] for i in range(i_start, i_end + 1)])
            i = i_start + int(np.argmax(contribs))

        # Pick j: a position far from i with low value
        j_candidates = np.where((np.abs(np.arange(n) - i) > n // 4) & (a < best_a.max() * 0.5))[0]
        if len(j_candidates) == 0:
            j_candidates = np.where(np.abs(np.arange(n) - i) > 5)[0]
        if len(j_candidates) == 0:
            j_candidates = np.arange(n)
        j = j_candidates[rng.integers(len(j_candidates))]

        # Optimal t: minimize max of (conv + t*(contribution changes))
        # c_k(t) = c_k + t*(a_{k-i} - a_{k-j}) for the peak
        # But this affects many lags. Use bisection on 1D problem.
        total = a[i] + a[j]

        def obj_at_t(t):
            a_try = a.copy()
            a_try[i] = max(0, a[i] + t)
            a_try[j] = max(0, a[j] - t)
            if a_try[i] + a_try[j] != total:
                # Renormalize the pair
                s2 = a_try[i] + a_try[j]
                if s2 < 1e-15:
                    return cur_max + 1
                a_try[i] = total / 2
                a_try[j] = total / 2
            conv_try = fast_convolve(a_try)
            return conv_try.max()

        # Golden section search for optimal t in [-a[j]/2, a[i]/2]
        t_lo = -a[j] * 0.8
        t_hi = a[i] * 0.8
        if t_hi <= t_lo:
            iters += 1
            continue

        # Quick ternary search
        gr = (np.sqrt(5) - 1) / 2
        t1 = t_hi - gr * (t_hi - t_lo)
        t2 = t_lo + gr * (t_hi - t_lo)
        f1 = obj_at_t(t1)
        f2 = obj_at_t(t2)
        for _ in range(12):
            if f1 < f2:
                t_hi = t2
                t2 = t1
                f2 = f1
                t1 = t_hi - gr * (t_hi - t_lo)
                f1 = obj_at_t(t1)
            else:
                t_lo = t1
                t1 = t2
                f1 = f2
                t2 = t_lo + gr * (t_hi - t_lo)
                f2 = obj_at_t(t2)

        t_opt = (t_lo + t_hi) / 2
        a_new = a.copy()
        a_new[i] = max(0, a[i] + t_opt)
        a_new[j] = max(0, a[j] - t_opt)
        s_new = a_new.sum()
        if s_new < 1e-10:
            iters += 1
            continue
        a_new = a_new / s_new
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        iters += 1

    return best_a, best_obj


def lp_direction_fast(a, M, n, time_limit=20):
    """
    Fast LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    Uses a sparse-friendly approach.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def structured_initializations(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Chirp-like (quadratic phase) mapped to non-negative
    chirp = 0.5 + 0.5 * np.cos(2 * np.pi * x * (0.5 + 0.1 * np.random.default_rng(rng.integers(1000)).random()) / n)
    chirp += 1e-6
    chirp /= chirp.sum()
    inits['chirp'] = chirp

    # Raised cosine with various widths
    for w in [0.3, 0.5, 0.7]:
        rc = 0.5 * (1 + np.cos(2 * np.pi * x / (w * n)))
        rc = np.maximum(rc, 1e-8)
        rc /= rc.sum()
        inits[f'raised_cos_{int(w*100)}'] = rc

    # Gaussian at various positions and widths
    for pos in [0.3, 0.5, 0.7]:
        for sf in [0.15, 0.25, 0.4]:
            g = np.exp(-0.5 * ((x - pos * n) / (sf * n)) ** 2)
            g /= g.sum()
            inits[f'gauss_{int(pos*10)}_{int(sf*100)}'] = g

    # Bimodal (edges)
    edge = max(2, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Triangular (peak center)
    tri = 1.0 - 2.0 * np.abs(x / (n - 1) - 0.5)
    tri = np.maximum(tri, 1e-8)
    tri /= tri.sum()
    inits['triangle'] = tri

    # Inverse triangle (valley center)
    inv_tri = 2.0 * np.abs(x / (n - 1) - 0.5) + 0.1
    inv_tri /= inv_tri.sum()
    inits['inv_triangle'] = inv_tri

    # Random Dirichlet
    inits['dirichlet_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dirichlet_1'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['dirichlet_2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Sparse (random subset)
    for sparsity in [0.3, 0.5]:
        k = max(2, int(n * sparsity))
        idx = rng.choice(n, size=k, replace=False)
        sp = np.zeros(n)
        sp[idx] = 1.0
        sp /= sp.sum()
        inits[f'sparse_{int(sparsity*100)}'] = sp

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_down'] = pp / ps

        # Perturbed parent
        pp = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, max(1, len(parent))),
                       parent if len(parent) > 0 else np.ones(max(1, len(parent))))
        pp = np.maximum(pp, 0)
        noise = rng.standard_normal(n) * 0.05
        pp = np.maximum(pp + noise, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent_perturbed'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining adaptive gradient descent,
    pair-descent polish, LP direction finding, and spectral guidance.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # Phase 1: Multi-length scan with quick optimization
    print("=== Phase 1: Multi-length scan ===", flush=True)
    lengths = [128, 192, 256, 320, 384, 448, 512]

    scan_results = []
    t_p1_start = time.monotonic()
    p1_budget = 200.0

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        inits = structured_initializations(n, parent, rng)
        names = list(inits.keys())

        # Priority
        priority = ['parent_crop', 'parent_interp', 'parent_down', 'parent_perturbed',
                    'chirp', 'raised_cos_50', 'gauss_5_25', 'dirichlet_1',
                    'uniform', 'bimodal']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:5]

        time_rem = p1_budget - (time.monotonic() - t_p1_start)
        per_init = max(5.0, time_rem / max(1, len(names) * max(1, len(lengths) - lengths.index(n))))

        for name in names:
            if time.monotonic() > deadline - 280:
                break
            t_alloc = min(per_init, 30.0)
            if t_alloc < 4:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_adaptive(
                a0, t_alloc, beta_init=30, beta_end=200,
                lr_init=0.015, momentum=0.9, top_k_frac=0.4,
                seed=rng.integers(0, 100000))

            score = evaluate_sequence(a_opt.tolist())
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    if scan_results:
        print(f"Phase 1 best from scan: {scan_results[0][0]:.6f}, overall best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization of top configurations
    print("=== Phase 2: Deep optimization ===", flush=True)

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add a few diverse restarts at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    inits_best = structured_initializations(best_n, parent, rng)
    for name in ['chirp', 'dirichlet_05']:
        if name in inits_best:
            top_configs.append((best_n, inits_best[name].copy()))

    remaining = deadline - time.monotonic() - 120.0
    n_configs = len(top_configs)
    per_config = max(20.0, remaining / max(1, n_configs))

    for i, (n_cfg, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(per_config, 250.0)
        if t_alloc < 15:
            break

        print(f"  Config {i}: n={n_cfg}, t_alloc={t_alloc:.1f}s", flush=True)

        # Sub-phase A: Main gradient descent (50%)
        t_a = t_alloc * 0.45
        a_a, obj_a = optimize_adaptive(
            a0, t_a, beta_init=50, beta_end=1500,
            lr_init=0.012, momentum=0.92, top_k_frac=0.25,
            spec_weight=0.0,
            seed=rng.integers(0, 1000000))
        score_a = evaluate_sequence(a_a.tolist())
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    Grad: {score_a:.6f}", flush=True)

        # Sub-phase B: LP direction (25%)
        t_b = t_alloc * 0.25
        if t_b > 10:
            a_b, obj_b = _lp_refine(a_a, t_b, rng)
            score_b = evaluate_sequence(a_b.tolist())
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    LP: {score_b:.6f}", flush=True)

        # Sub-phase C: Refine with high beta (20%)
        t_c = t_alloc * 0.20
        if t_c > 8:
            a_c, obj_c = optimize_adaptive(
                a_a, t_c, beta_init=500, beta_end=5000,
                lr_init=0.005, momentum=0.95, top_k_frac=0.1,
                seed=rng.integers(0, 1000000))
            score_c = evaluate_sequence(a_c.tolist())
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Refine: {score_c:.6f}", flush=True)

        # Sub-phase D: Pair descent polish (10%)
        t_d = t_alloc * 0.10
        if t_d > 5:
            a_d, obj_d = coordinate_pair_descent(a_a, t_d, seed=rng.integers(0, 1000000))
            score_d = evaluate_sequence(a_d.tolist())
            if np.isfinite(score_d) and score_d < best_score:
                best_score = score_d
                best_sequence = a_d.copy()
                print(f"    PairDec: {score_d:.6f}", flush=True)

    # Phase 3: Final polish
    print("=== Phase 3: Final polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # High-precision gradient
        t_fg = remaining * 0.4
        if t_fg > 10:
            a_fg, obj_fg = optimize_adaptive(
                best_sequence, t_fg, beta_init=1000, beta_end=10000,
                lr_init=0.003, momentum=0.96, top_k_frac=0.05,
                seed=42)
            score_fg = evaluate_sequence(a_fg.tolist())
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {score_fg:.6f}", flux=True) if False else print(f"  FineGrad: {score_fg:.6f}", flush=True)

        # Pair descent
        remaining = deadline - time.monotonic() - 15.0
        t_pd = remaining * 0.35
        if t_pd > 8:
            a_pd, obj_pd = coordinate_pair_descent(best_sequence, t_pd, seed=999)
            score_pd = evaluate_sequence(a_pd.tolist())
            if np.isfinite(score_pd) and score_pd < best_score:
                best_score = score_pd
                best_sequence = a_pd.copy()
                print(f"  PairPolish: {score_pd:.6f}", flush=True)

        # One more gradient pass
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            a_final, obj_final = optimize_adaptive(
                best_sequence, remaining, beta_init=2000, beta_end=20000,
                lr_init=0.001, momentum=0.97, top_k_frac=0.03,
                seed=777)
            score_final = evaluate_sequence(a_final.tolist())
            if np.isfinite(score_final) and score_final < best_score:
                best_score = score_final
                best_sequence = a_final.copy()
                print(f"  Final: {score_final:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]


def _lp_refine(a0, time_limit, rng):
    """LP-based refinement: repeatedly solve LP and interpolate."""
    rng2 = np.random.default_rng(rng.integers(0, 2**31))
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, conv, beta, top_k=None):
    """
    Gradient of (1/beta)*logsumexp(beta*conv) w.r.t. a.
    Uses cross-correlation via FFT. Optionally concentrates on top-k peaks.
    """

        thresh_idx = np.argpartition(conv, -top_k)[-top_k]
        mask = np.zeros(len(conv), dtype=bool)
        mask[thresh_idx] = True
        scaled = np.zeros(len(conv))
        scaled[mask] = beta * conv[mask]
        max_s = scaled[mask].max()
        scaled[mask] -= max_s
        p = np.exp(scaled)
        p_sum = p.sum()

            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum
    else:
        scaled = beta * conv
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum

    P = np.fft.rfft(p, fft_size)

def spectral_flatness_grad(a, weight):
    """
    Gradient of weight * sum((|A[k]|^2 - mean)^2) to encourage flat spectrum.
    """
    n = len(a)
    A = np.fft.rfft(a)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    if P_mean < 1e-15:
        return np.zeros(n)
    grad_freq = A * (P - P_mean)
    grad = 4.0 * np.fft.irfft(grad_freq, n) * weight
    return grad

def structured_initializations(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Chirp
    freq = 0.5 + 0.5 * rng.random()
    chirp = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    chirp = np.maximum(chirp, 1e-8)
    chirp /= chirp.sum()
    inits['chirp'] = chirp

    # Raised cosine
    for w in [0.4, 0.6, 0.8]:
        rc = 0.5 * (1 + np.cos(2 * np.pi * x / (w * n)))
        rc = np.maximum(rc, 1e-8)
        rc /= rc.sum()
        inits[f'rc_{int(w*10)}'] = rc

    # Gaussians
    for pos in [0.35, 0.5, 0.65]:
        for sf in [0.15, 0.3]:
            g = np.exp(-0.5 * ((x - pos * n) / (sf * n)) ** 2)
            g /= g.sum()
            inits[f'g_{int(pos*100)}_{int(sf*100)}'] = g

    # Bimodal
    edge = max(2, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Dirichlet
    inits['dir_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dir_1'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['dir_2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Sparse
    for sp in [0.3, 0.5]:
        k = max(2, int(n * sp))
        idx = rng.choice(n, size=k, replace=False)
        sp_seq = np.zeros(n)
        sp_seq[idx] = 1.0
        sp_seq /= sp_seq.sum()
        inits[f'sp_{int(sp*100)}'] = sp_seq

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        else:
