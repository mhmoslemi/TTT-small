def qp_subproblem(a, conv, t_level, n_constraints=8):
    """
    Solve QP: min ||a' - a||^2 s.t. sum a' = 1, a' >= 0, 
    sum_i a'_i * a_fixed[k-i] <= t for top peaks.
    """
    n = len(a)
    # Get top peaks
    peak_idx = np.argpartition(conv, -n_constraints)[-n_constraints:]
    peak_idx = sorted(peak_idx.tolist())
    
    # Build linear constraint matrix: for each peak k, constraint is sum_i a'_i * a[k-i] <= t
    A_ub = np.zeros((len(peak_idx), n))
    for row, k in enumerate(peak_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A_ub[row, i] = a[k - i]  # coefficient of a'_i
    
    b_ub = np.full(len(peak_idx), t_level)
    
    # Objective: min (a'-a)^T (a'-a) = a'^T a' - 2 a^T a' + const
    # Gradient: 2a' - 2a
    # Hessian: 2I
    
    def obj(x):
        return np.sum((x - a)**2)
    
    def grad(x):
        return 2.0 * (x - a)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)},
    ]
    for row in range(len(peak_idx)):
        k = int(peak_idx[row])
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        coeffs = np.zeros(n)
        for i in range(i_start, i_end + 1):
            coeffs[i] = a[k - i]
        constraints.append({
            'type': 'ineq', 
            'fun': lambda x, c=coeffs, t=t_level: t_level - c @ x,
            'jac': lambda x, c=coeffs: -c
        })
    
    bounds = [(0, None)] * n
    
    try:
        result = minimize(obj, a, jac=grad, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            a_new = np.maximum(result.x, 0)
            s = a_new.sum()
            if s > 1e-10:
                return a_new / s
    except:
        pass
    return None

def optimize_peak_targeting(a0, time_limit, top_k=10, lr_init=0.05, seed=0):
    """

    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.9
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        
        # Adaptive top_k: start with more peaks, narrow down
        cur_k = max(3, int(top_k * (1.0 - 0.5 * frac)))
        
        conv_cur = fast_convolve(a)
        grad = peak_gradient(a, conv_cur, top_k=cur_k)
        
        # Nesterov
        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad_y = peak_gradient(y, conv_y, top_k=cur_k)
        
        v = momentum * v - lr * grad_y
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        total_iters += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = a_new  # still update a for momentum consistency
        
        # Adaptive learning rate
        if no_improve == 50:
            lr *= 0.7
            no_improve = 0
        elif no_improve > 200:

            noise = rng.standard_normal(n) * 0.01 * (1.0 + frac)
            a = project_simplex(a + noise)
            v *= 0.1
            lr = max(0.001, lr_init * 0.5)
            no_improve = 0

def coordinate_descent_peaks(a0, time_limit, seed=0):
    """
    Coordinate descent targeting peak-reducing pair moves.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

        a = a / s

    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        cur_max = conv_cur.max()
        
        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            
            if i_end < i_start:
                continue
            
            # Find pair with max contribution
            contribs = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            max_idx = np.argmax(contribs)
            i_star = i_start + max_idx
            
            # Try to reduce a[i_star] or a[k - i_star]
            j_star = k - i_start - max_idx  # the paired element
            
            # Try moving mass from i_star to a random other location
            for attempt in range(5):
                j = rng.integers(0, n)
                if j == i_star:
                    continue
                
                # Binary search for max transfer that reduces the peak
                lo, hi = 0.0, a[i_star] * 0.5

                    mid = (lo + hi) / 2
                    a_try = a.copy()
                    a_try[i_star] -= mid
                    a_try[j] += mid
                    c_try = fast_convolve(a_try)
                    if c_try.max() < cur_max:
                        lo = mid
                    else:
                        hi = mid
                
                if lo > 0:
                    a_try = a.copy()
                    a_try[i_star] -= lo
                    a_try[j] += lo
                    c_try = fast_convolve(a_try)
                    if c_try.max() < best_obj:
                        best_obj = c_try.max()
                        best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:

            i = rng.integers(0, n)
            j = rng.integers(0, n)

                        best_a = a_try.copy()
                    a = a_try

def generate_asymmetric_inits(n, rng):
    """Generate initializations that break left-right symmetry."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Exponential decay (mass on left)
    exp_d = np.exp(-x / (n * 0.3))
    exp_d /= exp_d.sum()
    inits['exp_left'] = exp_d
    
    # Exponential decay (mass on right)  
    exp_r = np.exp(-((n-1-x) / (n * 0.3)))
    exp_r /= exp_r.sum()
    inits['exp_right'] = exp_r
    
    # Double exponential (mass on both edges, dip in middle)
    dexp = np.exp(-((x - n*0.15)/(n*0.15))**2) + np.exp(-((x - n*0.85)/(n*0.15))**2)
    dexp /= dexp.sum()
    inits['double_exp'] = dexp
    
    # Linear ramp (mass increasing)
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Inverted ramp
    inits['inv_ramp'] = ramp[::-1].copy()
    
    # S-curve (logistic)
    logi = 1.0 / (1.0 + np.exp(-4.0 * (x / n - 0.5)))
    logi /= logi.sum()
    inits['logistic'] = logi
    
    # Power law
    power = np.power(x + 1, 0.5)
    power /= power.sum()
    inits['power'] = power
    
    # Random with asymmetry
    rand_asym = rng.dirichlet(np.concatenate([np.ones(n//2)*3.0, np.ones(n - n//2)*0.5]))
    inits['rand_asym'] = rand_asym
    
    # Random uniform
    inits['rand_unif'] = rng.dirichlet(np.ones(n))
    
    # Smooth random (low-pass filtered random)
    rand_smooth = rng.standard_normal(n)
    # Simple moving average smoothing
    kernel = np.ones(5) / 5
    rand_smooth = np.convolve(rand_smooth, kernel, mode='same')
    rand_smooth = np.abs(rand_smooth)
    rand_smooth /= rand_smooth.sum()
    inits['rand_smooth'] = rand_smooth
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_gradient(a, conv, top_k=10):
    """
    Compute gradient of the max-conv approximation focusing on top-k peaks.
    grad[j] = 2 * sum_k w_k * a[k-j] (for valid indices)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    N = len(conv)  # 2n-1

    if top_k >= N:
        idx = np.arange(N)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]

    weights = conv[idx].astype(np.float64).copy()
    w_sum = weights.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    weights /= w_sum

    # Build full weight vector
    w_full = np.zeros(N)
    for k, w in zip(idx, weights):
        w_full[int(k)] = w

    # Cross-correlation: x[j] = sum_n w_full[n+j] * a[n]
    # = ifft(fft(w_full) * conj(fft(a)))[:n] with proper zero-padding
    fft_size = 1 << (N + n - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)

    return 2.0 * cross[:n]


def full_smoothed_gradient(a, conv, beta):
    """Full log-sum-exp smoothed gradient (more expensive but global)."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:

    else:

def spectral_init(n, rng, target_flatness=0.5):
    """
    Generate a sequence with near-flat autocorrelation using spectral method.
    1. Target flat autocorrelation spectrum
    2. Compute magnitude spectrum as sqrt of target
    3. Random phase, inverse FFT, rectify
    """
    # Target autocorrelation: blend between flat and delta
    # Flat: all values = 1/(2n-1)
    # Partially flat: value at lag 0 is larger
    N = 2 * n - 1
    target = np.ones(N) / N
    # Add some concentration at lag 0
    target[n - 1] = target[n - 1] * (1.0 + target_flatness * 2.0)

    # Frequency domain: |FFT(a)|^2 should correspond to target autocorrelation
    # FFT of autocorrelation = |FFT(a)|^2
    fft_size = 1 << (N - 1).bit_length()
    target_fft = np.fft.rfft(target, fft_size)
    # Ensure non-negative (should be, but numerical issues)
    target_fft = np.maximum(target_fft, 0)
    magnitude = np.sqrt(target_fft)

    # Random phase
    phase = rng.uniform(0, 2 * np.pi, size=magnitude.shape)
    phase[0] = 0  # DC component real
    spectrum = magnitude * np.exp(1j * phase)

    # Inverse FFT
    a_complex = np.fft.irfft(spectrum, fft_size)
    a_real = a_complex[:n]

    # Rectify and normalize
    a = np.maximum(a_real, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback: try with different phase
        for _ in range(10):
            phase = rng.uniform(0, 2 * np.pi, size=magnitude.shape)
            phase[0] = 0
            spectrum = magnitude * np.exp(1j * phase)
            a_complex = np.fft.irfft(spectrum, fft_size)
            a_real = a_complex[:n]
            a = np.maximum(a_real, 0)
            s = a.sum()
            if s > 1e-5 * n:
                break
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def mseq_init(n, rng):
    """
    Initialize from a pseudo-random ±1 sequence (m-sequence inspired).
    Map +1 -> high, -1 -> low (but non-negative).
    """
    # Generate random ±1 sequence
    bits = rng.integers(0, 2, size=n) * 2 - 1  # ±1

    # Map to non-negative: +1 -> 1, -1 -> eps
    eps = 0.05
    a = np.where(bits > 0, 1.0, eps)
    a = a / a.sum()

    # Apply a mild spatial variation to avoid being too "digital"
    x = np.arange(n) / n
    modulation = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(0.1, 0.4) * x * n)
    a = a * modulation
    a = np.maximum(a, 0)
    a = a / a.sum()

    return a


def generate_inits(n, parent, rng):

    inits['mseq2'] = mseq_init(n, rng)

    exp_l /= exp_l.sum()
    inits['exp_left'] = exp_l

    exp_r = np.exp(-(n - 1 - x) / (n * 0.25))
    exp_r /= exp_r.sum()
    inits['exp_right'] = exp_r

    # Double peak (edges)
    dp = np.exp(-((x - n * 0.1) / (n * 0.12)) ** 2) + np.exp(-((x - n * 0.9) / (n * 0.12)) ** 2)
    dp /= dp.sum()
    inits['double_peak'] = dp

    # Logistic (S-curve, asymmetric)
    logi = 1.0 / (1.0 + np.exp(-5.0 * (x / n - 0.4)))
    logi = np.maximum(logi, 1e-8)
    logi /= logi.sum()
    inits['logistic'] = logi

    # Random Dirichlet (various concentrations)
    inits['rand_conc'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_flat'] = rng.dirichlet(np.ones(n) * 3.0)
    # Asymmetric Dirichlet
    inits['rand_asym'] = rng.dirichlet(np.concatenate([
        np.ones(n // 2) * 2.0, np.ones(n - n // 2) * 0.5
    ]))

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        # Adaptive top_k: start broad, narrow down
        cur_k = max(3, int(top_k * (1.0 - 0.4 * frac)))

        if mode == 'peak':
            conv_y = fast_convolve(a + momentum * v)
            grad = peak_gradient(a + momentum * v, conv_y, top_k=cur_k)
        else:
            beta = 50.0 + 200.0 * frac
            conv_y = fast_convolve(a + momentum * v)
            grad = full_smoothed_gradient(a + momentum * v, conv_y, beta)

        # Nesterov momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        total_iters += 1
        a = a_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate and perturbation
        if no_improve == 80:
            lr *= 0.7
            no_improve = 0
        elif no_improve > 300:
            # Structured perturbation: anti-peak direction
            conv_cur = fast_convolve(a)
            anti_grad = peak_gradient(a, conv_cur, top_k=3)
            a = project_simplex(a + 0.02 * anti_grad + rng.standard_normal(n) * 0.005)
            v *= 0.1
            lr = max(0.002, lr_init * 0.4)
            no_improve = 0

        # Decay learning rate
        lr = max(0.0005, lr_init * (1.0 - 0.85 * frac))

    return best_a, best_obj


def optimize_qp_phases(a0, time_limit, seed=0):
    """
    Iterative QP: linearize top peak constraints, solve QP, check improvement.
    Uses scipy minimize with SLSQP.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        cur_max = conv_cur.max()

        # Try multiple reduction targets
        found = False
        for reduction in [0.99, 0.98, 0.96, 0.94, 0.90]:
            t_target = cur_max * reduction
            n_cons = min(8, 2 * n - 1)
            peak_idx = np.argpartition(conv_cur, -n_cons)[-n_cons:]

            # Build linear constraint: for peak k, sum_i a'_i * a_fixed[k-i] <= t_target
            A_rows = []
            b_vals = []
            for k in peak_idx:
                k = int(k)
                row = np.zeros(n)
                i_start = max(0, k - n + 1)
                i_end = min(k, n - 1)
                for i in range(i_start, i_end + 1):
                    row[i] = current_a[k - i]
                A_rows.append(row)
                b_vals.append(t_target)

            A_mat = np.array(A_rows)

            def obj(x):
                return np.sum((x - current_a) ** 2)

            def grad_qp(x):
                return 2.0 * (x - current_a)

            constraints_list = [
                {'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)},
            ]
            for row in A_rows:
                constraints_list.append({
                    'type': 'ineq',
                    'fun': lambda x, r=row, t=t_target: t - r @ x,
                    'jac': lambda x, r=row: -r,
                })

            bounds = [(0, None)] * n

            try:
                result = minimize(obj, current_a, jac=grad_qp, method='SLSQP',
                                  bounds=bounds, constraints=constraints_list,
                                  options={'maxiter': 100, 'ftol': 1e-9})
                if result.success:
                    a_new = np.maximum(result.x, 0)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a_new = a_new / sn
                        conv_new = fast_convolve(a_new)
                        obj_new = conv_new.max()

                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                        if obj_new < cur_max * 0.999:
                            current_a = a_new.copy()
                            found = True
                            break
            except Exception:
                pass

        if not found:
            stuck += 1
            if stuck > 15:
                # Perturb
                noise = rng.standard_normal(n) * 0.015
                current_a = project_simplex(current_a + noise)
                stuck = 0

    return best_a, best_obj


def coordinate_polish(a0, time_limit, seed=0):
    """
    Coordinate descent: find peak-contributing pairs and reduce them.
    Uses binary search for max beneficial transfer.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        cur_max = conv_cur.max()
        iterations += 1

        # Find top peaks
        n_peaks = min(4, 2 * n - 1)
        peak_idx = np.argpartition(conv_cur, -n_peaks)[-n_peaks:]

        improved = False
        for k in sorted(peak_idx, key=lambda x: -conv_cur[x]):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)

            if i_end < i_start:
                continue

            # Find max-contributing pair
            contribs = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if contribs.max() < 1e-15:
                continue
            max_i = i_start + int(np.argmax(contribs))
            paired = k - max_i

            # Try transferring from max_i to a random other position
            for attempt in range(3):
                j = int(rng.integers(0, n))
                if j == max_i:
                    continue

                # Binary search for max transfer
                lo, hi = 0.0, a[max_i] * 0.4
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    a_try = a.copy()
                    a_try[max_i] -= mid
                    a_try[j] += mid
                    c_try = fast_convolve(a_try)
                    if c_try.max() < cur_max * 0.999:
                        lo = mid
                    else:
                        hi = mid

                if lo > 1e-7:
                    a_try = a.copy()
                    a_try[max_i] -= lo
                    a_try[j] += lo
                    c_try = fast_convolve(a_try)
                    if c_try.max() < best_obj:
                        best_obj = c_try.max()
                        best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Random pair adjustment
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-6:
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    c_try = fast_convolve(a_try)
                    if c_try.max() < best_obj:
                        best_obj = c_try.max()
                        best_a = a_try.copy()
                    a = a_try
                    break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: spectral/asymmetric init -> gradient descent -> QP -> coordinate polish.
    Key innovations over previous: peak-focused gradient (faster), spectral inits,
    asymmetry exploitation, QP for large jumps.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # === Phase 1: Exploration ===
    print("=== Phase 1: Exploration ===", flush=True)
    t_phase1_start = time.monotonic()
    phase1_budget = 500.0

    lengths = [256, 384, 512]
    results = []

    for n in lengths:
        if time.monotonic() > deadline - 700:
            break
        t_start = time.monotonic()
        t_alloc_n = min(phase1_budget / len(lengths), deadline - time.monotonic() - 680)

        inits = generate_inits(n, parent, rng)
        names = list(inits.keys())

        # Priority: parent, spectral, asymmetric random
        priority = ['parent', 'parent_ds', 'parent_up', 'spectral_1', 'spectral_2',
                    'mseq', 'rand_asym', 'exp_left', 'logistic', 'double_peak',
                    'spectral_0', 'rand_conc', 'mseq2', 'exp_right', 'rand_flat']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:6]

        per_init = t_alloc_n / max(len(names), 1)

        for name in names:
            if time.monotonic() > deadline - 680:
                break
            t_alloc = min(per_init, 50.0)
            if t_alloc < 8:
                break

            a0 = inits[name]
            mode = 'peak' if 'spectral' not in name else 'smooth'
            a_opt, obj = optimize_gradient(
                a0, t_alloc, mode=mode, top_k=8,
                lr_init=0.03, momentum=0.9,
                seed=int(rng.integers(0, 100000)))

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.6f}, time={time.monotonic()-t_phase1_start:.0f}s", flush=True)

    # === Phase 2: Deep optimization ===
    print("=== Phase 2: Deep Optimization ===", flush=True)

    # Build candidate list from top results + diverse
    candidates = []
    if best_sequence is not None:
        candidates.append(best_sequence.copy())

    seen_n = {len(candidates[0])} if candidates else set()
    for score, n, name, a_opt in results:
        if len(a_opt) not in seen_n:
            candidates.append(a_opt.copy())
            seen_n.add(len(a_opt))
        if len(candidates) >= 4:
            break

    # Add fresh random at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    candidates.append(spectral_init(best_n, rng, 0.5))
    candidates.append(mseq_init(best_n, rng))

    remaining = deadline - time.monotonic() - 400.0
    n_cand = len(candidates)
    per_cand = max(25.0, remaining / max(n_cand, 1))

    for i, a0 in enumerate(candidates):
        if time.monotonic() > deadline - 380:
            break
        t_alloc = min(per_cand, 180.0)
        if t_alloc < 15:
            break

        print(f"  Cand {i}: n={len(a0)}, t={t_alloc:.0f}s", flush=True)

        # A: Peak gradient (55%)
        t_a = t_alloc * 0.55
        a_a, _ = optimize_gradient(
            a0, t_a, mode='peak', top_k=6,
            lr_init=0.02, momentum=0.92,
            seed=int(rng.integers(0, 1000000)))
        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    PeakGrad: {score_a:.6f}", flush=True)

        # B: QP (25%)
        t_b = t_alloc * 0.25
        if t_b > 15:
            a_b, _ = optimize_qp_phases(a_a, t_b, seed=int(rng.integers(0, 1000000)))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    QP: {score_b:.6f}", flush=True)
            a_a = a_b

        # C: Smooth gradient refine (20%)
        t_c = t_alloc * 0.20
        if t_c > 8:
            a_c, _ = optimize_gradient(
                a_a, t_c, mode='smooth', top_k=4,
                lr_init=0.008, momentum=0.94,
                seed=int(rng.integers(0, 1000000)))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    SmoothRefine: {score_c:.6f}", flush=True)

    # === Phase 3: Final polish ===
    print("=== Phase 3: Final Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0

        # Coordinate descent
        t_cd = remaining * 0.40
        if t_cd > 15:
            a_cd, _ = coordinate_polish(best_sequence, t_cd, seed=777)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  Coord: {best_score:.6f}", flush=True)

        # QP polish
        remaining = deadline - time.monotonic() - 10.0
        t_qp = remaining * 0.30
        if t_qp > 10:
            a_qp, _ = optimize_qp_phases(best_sequence, t_qp, seed=555)
            score_qp = float(evaluate_sequence(a_qp.tolist()))
            if np.isfinite(score_qp) and score_qp < best_score:
                best_score = score_qp
                best_sequence = a_qp.copy()
                print(f"  QP2: {best_score:.6f}", flush=True)

        # Final gradient + coordinate
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 15:
            t_g = remaining * 0.5
            a_g, _ = optimize_gradient(
                best_sequence, t_g, mode='peak', top_k=4,
                lr_init=0.005, momentum=0.95, seed=12345)
            score_g = float(evaluate_sequence(a_g.tolist()))
            if np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()
                print(f"  FinalGrad: {best_score:.6f}", flush=True)

            remaining = deadline - time.monotonic() - 3.0
            if remaining > 10:
                a_f, _ = coordinate_polish(best_sequence, remaining, seed=42)
                score_f = float(evaluate_sequence(a_f.tolist()))
                if np.isfinite(score_f) and score_f < best_score:
                    best_score = score_f
                    best_sequence = a_f.copy()
                    print(f"  FinalCoord: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
