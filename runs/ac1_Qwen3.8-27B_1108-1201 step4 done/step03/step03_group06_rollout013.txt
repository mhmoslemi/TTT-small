"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses full LP direction finding, backtracking line search, multiplicative weight
updates, and multi-phase search with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k.
    Returns A_ub where A_ub @ b gives conv(a, b)."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def solve_lp_direction(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in active_indices (or all)."""
    n = len(a)
    num_k = 2 * n - 1
    
    if active_indices is None:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(num_k, M)
    else:
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[active_indices]
        b_ub = np.full(len(active_indices), M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x, res.fun
    except Exception:
        pass
    return None, None

def lp_iterate(a, M, use_all_constraints=True):
    """One LP iteration. Returns (b, lp_sum) where b is the LP solution on simplex."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find active (nearly tight) constraints
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    
    if use_all_constraints or len(active) < 5:
        b, _ = solve_lp_direction(a, M, active_indices=None)
    else:
        b, _ = solve_lp_direction(a, M, active_indices=active)
    
    if b is None:
        return None
    
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None
    return b / b_sum

def evaluate_normalized(a):
    """Evaluate objective for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lp_optimize(a0, time_limit, seed=0, verbose=False, full_lp=True):
    """LP-based optimization with backtracking line search."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 100
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        b = lp_iterate(a, M, use_all_constraints=full_lp)
        if b is None:
            break
        
        # Check if b is close to a (convergence)
        diff = np.abs(b - a).max()
        if diff < 1e-8:
            no_improve += 1
            if no_improve > 5:
                break
            # Perturb slightly
            noise = rng.standard_normal(n) * 0.001
            a = project_simplex(a + noise)
            continue
        
        # Backtracking line search
        best_t = 0.0
        best_new_obj = cur_obj
        for t in [0.9, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            obj_new = evaluate_normalized(a_new)
            if obj_new < best_new_obj:
                best_new_obj = obj_new
                best_t = t
        
        if best_t > 0 and best_new_obj < cur_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            cur_obj = best_new_obj
            a = a_new
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Perturb to escape
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                cur_obj = evaluate_normalized(a)
                no_improve = 0
                max_no_improve = int(max_no_improve * 1.3)
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, cur={cur_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def multiplicative_weight_optimize(a0, time_limit, seed=0, verbose=False):
    """Mirror descent with KL divergence (multiplicative weight updates)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = 1.0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Subgradient: find peaks and compute gradient
        # Use top few peaks for smoothing
        top_k = min(10, 2*n-1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Compute subgradient (average over peaks)
        grad = np.zeros(n)
        for k in peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
                j2 = i - k
                if 0 <= j2 < n:
                    grad[i] += a[j2]
        grad /= len(peaks)
        
        # Multiplicative weight update: a_new[i] ∝ a[i] * exp(-eta * grad[i])
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        obj_new = evaluate_normalized(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                eta *= 0.9
            if no_improve > 200:
                # Perturb
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                eta = 1.0
                no_improve = 0
        
        # Decrease eta slowly
        eta = max(0.01, eta * 0.999)
    
    return best_a, best_obj

def generate_init(n, parent=None, seed=0):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine
    if n > 2:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Triangular (peak at edges)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build A_ub for LP: (a*b)[k] <= M for all k. A_ub.shape = (2n-1, n)."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        target = 1.0 / (2 * n - 1)  # Ideal flat level

        # Find peaks above target
        excess = np.maximum(conv - target, 0)
        total_excess = excess.sum()
        if total_excess < 1e-12:
            break

        # Weight subgradients by excess
        # For peak at position k: grad_k[i] = a[k-i] + a[i-k] (valid indices)
        # Use FFT-based computation for efficiency
        above_target = excess > 1e-10
        peak_indices = np.where(above_target)[0]

        # Compute weighted gradient
        # grad[i] = sum_k w_k * (a[k-i] + a[i-k])
        # = cross_correlation(w, a)[i] + convolution(w, a)[i] (with appropriate indexing)
        w = excess / total_excess  # normalized weights

        # Use FFT: grad[i] = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
        fft_size = 1 << (2 * n - 1).bit_length()
        W = np.fft.fft(w, fft_size)
        A = np.fft.fft(a, fft_size)

        # First term: sum_k w[k]*a[k-i] = cross-correlation
        term1 = np.fft.ifft(A * np.conj(W)).real[:n]
        # Second term: sum_k w[k]*a[i-k] = convolution-like
        term2 = np.fft.ifft(A * W).real[:n]

        grad = term1 + term2

        # Projected gradient step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        a_new = project_simplex(a - eta * grad)
        obj_new = evaluate_norm(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.02, 1.0)  # Slowly increase step
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                eta *= 0.9
            if no_improve > 150:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                eta = 0.05
                no_improve = 0

    return best_a, best_obj


def lp_with_flattening(a0, time_limit, seed=0, verbose=False):
    """
    Combined LP + peak shaving: alternate between LP direction and
    peak-shaving subgradient steps.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    mode = 'lp'  # alternate between 'lp' and 'shave'
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if mode == 'lp':
            b, _ = solve_lp_full(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

            # Line search
            best_t = 0.0
            best_new = best_obj
            for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                a_try = (1.0 - t) * a + t * b
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = evaluate_norm(a_try)
                if obj_try < best_new:
                    best_new = obj_try
                    best_t = t

            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * b
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if evaluate_norm(a) < best_obj:
                    best_obj = evaluate_norm(a)
                    best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 30:
                mode = 'shave'
                no_improve = 0
        else:
            # Peak shaving step
            target = 1.0 / (2 * n - 1)
            excess = np.maximum(conv - target, 0)
            total_excess = excess.sum()
            if total_excess < 1e-12:
                mode = 'lp'
                continue

            w = excess / total_excess
            fft_size = 1 << (2 * n - 1).bit_length()
            W = np.fft.fft(w, fft_size)
            A = np.fft.fft(a, fft_size)
            term1 = np.fft.ifft(A * np.conj(W)).real[:n]
            term2 = np.fft.ifft(A * W).real[:n]
            grad = term1 + term2

            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
                for eta in [0.1, 0.05, 0.02, 0.01]:
                    a_try = project_simplex(a - eta * grad)
                    obj_try = evaluate_norm(a_try)
                    if obj_try < best_obj:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                        break

            no_improve += 1
            if no_improve > 20:
                mode = 'lp'
                no_improve = 0

        # Periodic perturbation
        if iter_count % 300 == 299:
            conv_check = fast_convolve(a)
            if 2 * n * conv_check.max() >= best_obj - 1e-9:
                noise = rng.standard_normal(n) * 0.005
                a_pert = project_simplex(a + noise)
                if evaluate_norm(a_pert) < best_obj:
                    a = a_pert

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid iter {iter_count}: obj={best_obj:.7f}, mode={mode}, t={elapsed:.1f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: quick scan -> deep LP -> hybrid LP+shaving -> polish.
    Uses full LP (all constraints), chirp initializations, and backtracking.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Exploration (90s) ==========
    print("=== Phase 1: Exploration ===")
    t0 = time.monotonic()
    phase1_time = 90.0

    lengths = [192, 256, 320, 384, 448]
    scan_results = []

    for n in lengths:
        remaining = t0 + phase1_time - time.monotonic()
        if remaining < 15:
            break
        inits = generate_inits(n, parent, rng)

        # Prioritize: parent, chirps, random, bimodal
        priority = ['parent', 'parent_ds', 'parent_up', 'chirp_0', 'chirp_1',
                    'rand1', 'bimodal', 'uniform', 'chirp_2']
        init_names = [nm for nm in priority if nm in inits][:4]

        time_per = max(5.0, (remaining - 5) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() + 5 > t0 + phase1_time:
                break
            t_alloc = min(time_per, t0 + phase1_time - time.monotonic() - 3)
            if t_alloc < 4:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                         seed=n + hash(name) % 1000,
                                         use_full=True, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    scan_results.sort(key=lambda x: x[0])
    for s, n, nm, _ in scan_results[:5]:
        print(f"  Scan: n={n}, {nm}, score={s:.7f}")
    print(f"Phase 1 done. Best: {best_score:.7f}")

    # ========== Phase 2: Deep LP (600s) ==========
    print("=== Phase 2: Deep LP ===")
    t1 = time.monotonic()
    phase2_time = 600.0

    if best_sequence is not None:
        n_best = len(best_sequence)
    else:
        n_best = 320
        best_sequence = np.ones(n_best) / n_best

    # Build list of starting points
    starts = [best_sequence.copy()]
    for score, n, name, a_opt in scan_results[:4]:
        if n == n_best:
            starts.append(a_opt.copy())
        else:
            pp = np.interp(np.linspace(0, 1, n_best),
                           np.linspace(0, 1, len(a_opt)), a_opt)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                starts.append(pp / ps)

    # Add fresh diverse starts at best length
    fresh = generate_inits(n_best, parent, rng)
    for key in ['chirp_0', 'chirp_1', 'rand2', 'bimodal']:
        if key in fresh:
            starts.append(fresh[key].copy())
            if len(starts) >= 8:
                break

    time_per_run = phase2_time / max(1, len(starts))
    time_per_run = max(30.0, time_per_run)

    for i, a_start in enumerate(starts):
        remaining = t1 + phase2_time - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per_run, remaining - 15)
        if t_alloc < 20:
            break

        a_opt, obj_opt = lp_optimize(a_start, time_limit=t_alloc,
                                     seed=i * 31 + 7, use_full=True, verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP [{i}]: n={n_best}, score={score:.7f}")

    print(f"Phase 2 done. Best: {best_score:.7f}")

    # ========== Phase 3: Hybrid LP + Peak Shaving (400s) ==========
    print("=== Phase 3: Hybrid ===")
    t2 = time.monotonic()
    phase3_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_opt, obj_opt = lp_with_flattening(best_sequence, time_limit=phase3_time * 0.5,
                                            seed=123, verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Hybrid: score={score:.7f}")

        remaining = t2 + phase3_time - time.monotonic()
        if remaining > 60:
            # Try from perturbed version
            noise = rng.standard_normal(n_best) * 0.003
            a_pert = project_simplex(best_sequence + noise)
            a_opt2, obj_opt2 = lp_with_flattening(a_pert, time_limit=remaining - 20,
                                                   seed=456, verbose=False)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  Hybrid pert: score={score2:.7f}")

    print(f"Phase 3 done. Best: {best_score:.7f}")

    # ========== Phase 4: Peak Shaving Focus (200s) ==========
    print("=== Phase 4: Peak Shaving ===")
    t3 = time.monotonic()
    phase4_time = 200.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_opt, obj_opt = peak_shaving(best_sequence, time_limit=phase4_time,
                                      seed=789, verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Shaving: score={score:.7f}")

    print(f"Phase 4 done. Best: {best_score:.7f}")

    # ========== Phase 5: Final Polish (remaining time - 30s) ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)

        # Final LP
        t_alloc = min(remaining * 0.4, 80.0)
        if t_alloc > 15:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                         seed=999, use_full=True, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.7f}")

        # Tiny perturbations
        remaining2 = deadline - time.monotonic() - 25
        if remaining2 > 10:
            scales = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015]
            for scale in scales:
                if time.monotonic() > deadline - 28:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 28:
                        break
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb s={scale}: score={score:.7f}")

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
