import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def subgradient(a, num_active=1):
    """Compute (sub)gradient of max_k (a*a)[k]."""
    n = len(a)
    conv = fast_convolve(a)
    # Find top num_active peaks
    if num_active == 1:
        k_star = np.argmax(conv)
        active = [k_star]
        weights = [1.0]
    else:
        # Use top peaks with softmax weighting
        peaks = np.sort(conv)[::-1][:num_active]
        # Weight by how close to max
        max_val = conv.max()
        epsilon = 1e-6
        active = [k for k in range(len(conv)) if conv[k] > max_val - (max_val - peaks[-1]) - epsilon]
        if not active:
            active = [np.argmax(conv)]
        # Weights proportional to (conv[k] - second_max)
        w = np.array([conv[k] - (max_val - (max_val - conv[active[-1]]))/2 for k in active])
        # Simpler: just use softmax on top peaks
        top_idx = np.argsort(conv)[::-1][:num_active]
        w = conv[top_idx]
        w -= w.max()
        w = np.exp(w * 10)  # sharp softmax
        w /= w.sum()
        active = top_idx.tolist()
        weights = w.tolist()
    
    grad = np.zeros(n)
    for k, w in zip(active, weights):
        for i in range(n):
            g = 0.0
            if 0 <= k - i < n:
                g += a[k - i]
            if 0 <= i - k < n:
                g += a[i - k]
            grad[i] += w * g
    return grad, conv, np.max(conv)

"""
Optimized subgradient/gradient descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed gradients, backtracking line search, Nesterov momentum,
and multi-phase search with diverse initializations.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def smoothed_obj_and_grad(a, beta, top_m=None):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    conv = fast_convolve(a)
    
    if top_m is not None and top_m < len(conv):
        # Only use top-m peaks
        sorted_conv = np.sort(conv)[::-1]
        threshold = sorted_conv[top_m - 1]
        mask = conv >= threshold
        conv_used = np.where(mask, conv, -np.inf)
    else:
        conv_used = conv
    
    scaled = beta * conv_used
    scaled = scaled - scaled[np.isfinite(scaled)].max()
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    
    if exp_sum < 1e-300:
        # Degenerate case
        k_star = np.argmax(conv)
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            ji = k_star - i
            if 0 <= ji < n:
                val += a[ji]
            ji2 = i - k_star
            if 0 <= ji2 < n:
                val += a[ji2]
            grad[i] = val
        return beta * conv[k_star] + np.log(exp_sum) / beta if exp_sum > 0 else beta * conv[k_star], grad, conv
    
    weights = exp_scaled / exp_sum
    obj = (conv_used.max() + np.log(exp_sum)) / beta  # approximation of log-sum-exp / beta
    
    # Actually: log(sum exp(beta*conv_k)) / beta = max(conv) + log(sum exp(beta*(conv_k - max))) / beta
    # Let me redo:
    max_conv = conv_used[np.isfinite(conv_used)].max()
    shifted = beta * (conv_used - max_conv)
    shifted = np.where(np.isfinite(shifted), shifted, -1000)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    weights = exp_shifted / exp_sum
    obj = max_conv + np.log(exp_sum) / beta
    
    # Gradient: sum_k weights[k] * grad_k
    grad = np.zeros(n)
    active_k = np.where(weights > 1e-10)[0]
    for k in active_k:
        w = weights[k]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)

        if i2_min <= i2_max:
            right = a[i2_min:i2_max+1]
            grad[i2_min:i2_max+1] += w * right

def gradient_descent(a0, time_limit, beta_init=200.0, beta_max=5000.0,
                     lr_init=0.01, momentum=0.9, top_m=None,
                     verbose=False, seed=0):
    """Smoothed gradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    beta = float(beta_init)
    lr = float(lr_init)
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        obj, grad, conv = smoothed_obj_and_grad(a, beta, top_m)
        
        # Normalize gradient for stable updates
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad_norm = grad / g_norm
        else:
            break
        
        # Momentum update
        v = momentum * v - lr * grad_norm
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.8
                v *= 0.5
            if no_improve > 300:
                # Perturb and restart
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = float(lr_init) * 0.5
                beta = float(beta_init) * 0.5
                no_improve = 0
        
        # Slowly increase beta
        beta = min(float(beta_max), beta * 1.002)
        
        iter_count += 1
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD iter {iter_count}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.6f}, t={elapsed:.1f}s")
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

def fw_optimize(a0, time_limit, top_m=5, verbose=False, seed=0):

    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        # Compute subgradient using top-m active lags
        top_idx = np.argsort(conv)[::-1][:top_m]
        w = conv[top_idx].copy()
        w -= w.max()
        w = np.exp(w * 20)
        w /= w.sum()

            wi = w[idx]
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            left = a[max(0, k-n+1):min(n-1, k)+1][::-1]
            grad[i_min:i_max+1] += wi * left
            i2_min = max(k, 0)
            i2_max = min(n - 1, k + n - 1)
            if i2_min <= i2_max:
                right = a[i2_min:i2_max+1]
                grad[i2_min:i2_max+1] += wi * right
        
        # Frank-Wolfe: find minimizer of linear oracle
        # min_{s in simplex} grad^T s = grad[argmin grad]
        # So s = e_{argmin grad}
        s_idx = np.argmin(grad)
        
        # Step size: use FW rule t = 2/(iter+2) or line search
        t_fw = 2.0 / (iter_count + 2)

        # Actually s = e_{s_idx}, so a_new = (1-t)*a + t*e_{s_idx}

        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < obj:
            a = a_new
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a.copy()
        else:
            # Try smaller step
            t_fw = t_fw * 0.5
            a_new = (1.0 - t_fw) * a + t_fw * np.zeros(n)

            conv_new = fast_convolve(a_new)

            if obj_new < obj:
                a = a_new
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a, best_obj
