"""
Multi-scale optimization with hybrid subgradient/mirror descent and LP refinement
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def subgrad_step(a, conv, n, lr, top_k=20):
    """
    Compute a subgradient step. Returns (new_a, gradient).
    Uses softmax-weighted peaks for smooth gradient approximation.
    """
    # Get top-k peaks
    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    
    # Compute weights (softmax over peak values)
    peak_vals = conv[peak_idx]
    max_pv = peak_vals.max()
    exp_vals = np.exp(-(peak_vals - max_pv) * 10.0 / max(max_pv - peak_vals.min(), 1e-10))
    weights = exp_vals / exp_vals.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for i in range(len(peak_idx)):
        if weights[i] < 1e-10:
            continue
        k = int(peak_idx[i])
        # b[k] = sum_j a[j] * a[k-j]
        # db[k]/da[j] = a[k-j] for valid k-j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += weights[i] * a[k - idx]
    
    return grad

def mirror_descent_optimize(a0, time_limit, lr=None, momentum=0.9, top_k=20, restart_every=500):
    """
    Mirror descent (exponential) optimization for non-negative sequences.
    Uses KL divergence as Bregman divergence.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    if lr is None:
        lr = 1.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    conv = fast_convolve(a)
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        grad = subgrad_step(a, conv, n, lr, top_k)
        
        # Mirror descent step (exponential)
        # a_new = a * exp(-lr * (grad - mean(grad)))
        # But we also want momentum
        v = momentum * v - lr * grad
        a_new = a * np.exp(-v)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        conv = fast_convolve(a)
        obj = compute_objective(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > restart_every:
            lr *= 0.5
            no_improve = 0
            # Perturb
            rng = np.random.default_rng(iteration)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            conv = fast_convolve(a)
        
        iteration += 1
        
        # Decay learning rate slowly
        lr *= (1 - 1e-6)
    
    return best_a, best_obj

    if s < 1e-15:

    n = len(a)

    t0 = time.monotonic()
    no_improve = 0

        # Batch of 10 steps

            if time.monotonic() - t0 > time_limit:

            grad = subgrad_step(a, conv, n, lr, top_k)
            
            v = momentum * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:

            else:
                a = a_new / s_new
            
            conv = fast_convolve(a)
            obj = compute_objective(a)

                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adapt parameters

        momentum = min(0.98, momentum + 0.0001)
        temp *= temp_decay
        
        if no_improve > 500:

def lp_refine(a0, time_limit, max_iters=200):
    """
    LP-based refinement. Solve LP to find best direction, then line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    # Build the convolution matrix once (it changes as a changes, so we rebuild each time)
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and iteration < max_iters:
        # Build constraint matrix: C[g] = a * g (convolution)
        C = np.zeros((2 * n - 1, n))
        for i in range(n):
            C[i:i+n, i] = a
        
        # LP: maximize sum(g) s.t. C@g <= M*1.001, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.001)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', 
                           options={'presolve': True, 'time_limit': 5})
        except Exception:
            no_improve += 1
            iteration += 1
            continue
        
        if not result.success:
            no_improve += 1
            iteration += 1
            continue
        
        g = result.x
        g_sum = g.sum()
        
        if g_sum <= 1.0 + 1e-6:
            no_improve += 1
        else:
            # Direction: g - a (both normalized to sum 1)
            g_norm = g / g_sum
            d = g_norm - a
            d_norm = np.linalg.norm(d)
            
            if d_norm < 1e-12:
                no_improve += 1
            else:
                # Line search
                best_t = 0.0
                best_obj_ls = best_obj
                for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_objective(a_try)
                    if obj_try < best_obj_ls - 1e-13:
                        best_obj_ls = obj_try
                        best_t = t
                
                if best_t > 0:
                    a_new = a + best_t * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve > 50:

            rng = np.random.default_rng(iteration * 7)

def upsample(a, factor=2):
    """Upsample a sequence by inserting zeros between elements."""
    n = len(a)
    new_n = n * factor
    new_a = np.zeros(new_n)
    new_a[::factor] = a
    return new_a

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Random Dirichlet
    concentration = 0.3
    inits['random'] = rng.dirichlet(np.ones(n) * concentration)
    
    # Gaussian
    sigma = n * 0.25
    g = np.exp(-0.5 * ((x - n/2) / sigma) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Sine
    si = np.sin(np.pi * x / n) + 0.1
    inits['sine'] = si / si.sum()
    
    # Power law
    pw = (x + 1) ** 0.3
    inits['power'] = pw / pw.sum()
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_ds'] = pp / pp.sum()
            else:
                # Upsample
                pp = upsample(p, factor=max(1, n // len(p)))
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_up'] = pp / pp.sum()
    
    return inits

    conv = fast_convolve(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s ** 2)


def soft_objective(a, k=20):
    """Soft objective: 2n * mean(top-k of conv) / (sum)^2."""
    n = len(a)
    conv = fast_convolve(a)
    if k >= len(conv):
        top_k_sum = conv.sum()
    else:
        top_k = np.partition(conv, -k)[-k:]
        top_k_sum = top_k.sum()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * top_k_sum / (k * s ** 2), conv


def compute_gradient(a, conv, n, k=20, soft=True):
    """
    Compute the (sub)gradient of the objective with respect to a.
    For soft: gradient of 2n * mean(top-k) / S^2
    For hard: subgradient of 2n * max / S^2
    """
    s = a.sum()
    if s < 1e-15:
        return np.zeros(n)
    
    if soft and k < len(conv):
        # Soft gradient
        top_k_vals = np.partition(conv, -k)[-k:]
        threshold = top_k_vals.min()
        # Gradient of sum of top-k w.r.t. a[i]
        # For each k_idx where conv[k_idx] >= threshold:
        #   d(conv[k_idx])/da[i] = a[k_idx - i] (if valid)
        active = np.where(conv >= threshold - 1e-15)[0]
        # For ties at the boundary, use fractional weight
        grad_peak = np.zeros(n)
        for k_idx in active:
            weight = 1.0
            if abs(conv[k_idx] - threshold) < 1e-12:
                # Tied at boundary - use 0.5 weight (approximate)
                weight = 0.5
            j_start = max(0, k_idx - n + 1)
            j_end = min(n - 1, k_idx)
            idx = np.arange(j_start, j_end + 1)
            grad_peak[idx] += weight * a[k_idx - idx]
        
        # Objective: 2n * sum_topk / (k * S^2)
        # dObj/da[i] = 2n/(k*S^2) * grad_peak[i] - 4n*sum_topk/(k*S^3)
        sum_topk = conv[active].sum()  # approximate
        coef1 = 2.0 * n / (k * s ** 2)
        coef2 = 4.0 * n * sum_topk / (k * s ** 3)
        grad = coef1 * grad_peak - coef2
    else:
        # Hard gradient (subgradient)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        grad_peak = np.zeros(n)
        for k_idx in active:
            j_start = max(0, k_idx - n + 1)
            j_end = min(n - 1, k_idx)
            idx = np.arange(j_start, j_end + 1)
            grad_peak[idx] += a[k_idx - idx]
        
        coef1 = 2.0 * n / s ** 2
        coef2 = 4.0 * n * max_val / s ** 3
        grad = coef1 * grad_peak - coef2
    
    return grad


def gradient_descent(a0, time_limit, method='projected', lr_init=0.01, momentum=0.9,
                     top_k=30, soft_k=50, soft_phase=True):
    """
    Gradient descent with momentum for the autocorrelation minimization.
    method: 'projected' (additive) or 'mirror' (multiplicative)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    batch_size = 20
    
    # Phase 1: soft objective
    # Phase 2: hard objective
    soft = soft_phase
    soft_phase_end = time_limit * 0.5 if soft_phase else 0
    
    while time.monotonic() - t0 < time_limit:
        # Phase transition
        if soft and (time.monotonic() - t0) > soft_phase_end:
            soft = False
            lr = lr_init * 0.3
            v *= 0.5  # reduce momentum velocity
        
        current_k = top_k
        current_soft_k = soft_k if soft else top_k
        
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            if soft:
                grad = compute_gradient(a, conv, n, k=current_soft_k, soft=True)
            else:
                grad = compute_gradient(a, conv, n, k=current_k, soft=False)
            
            if method == 'mirror':
                v = momentum * v - lr * grad
                log_a = np.log(np.maximum(a, 1e-30)) + v
                a_new = np.exp(log_a - log_a.max())
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                else:
                    a = best_a.copy()
            else:
                v = momentum * v - lr * grad
                a_new = a + v
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                else:
                    a = best_a.copy()
            
            conv = fast_convolve(a)
            obj = compute_objective(a)
            iter_count += 1
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adapt
        if no_improve > 0 and no_improve % 100 == 0:
            lr *= 0.95
        if no_improve > 500:
            # Restart
            rng = np.random.default_rng(iter_count)
            scale = 0.005 + 0.005 * min(no_improve / 500.0, 5.0)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            lr = lr_init * 0.5
            v = np.zeros(n)
        
        # Slow decay
        lr *= (1.0 - 1e-7)
    
    return best_a, best_obj
