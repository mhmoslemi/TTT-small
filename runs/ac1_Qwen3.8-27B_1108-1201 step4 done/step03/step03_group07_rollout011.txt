"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Power-mean smooth optimization with Adam optimizer
- Spectral domain iterative flattening (Fourier-based)
- Exact subgradient descent with momentum
- Peak-shaving heuristics
- Multi-length exploration with resampling
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def fast_convolve_grad(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    grad[m] = 2 * sum_k weights[k] * a[k-m] (if 0 <= k-m < n)
    """
    n = len(a)
    # grad[m] = 2 * sum_j a[j] * weights[m+j] (where m+j in [0, 2n-2])
    # Use FFT: this is a cross-correlation
    fft_size = 1 << (2 * n - 1 + n - 1 + 7).bit_length()
    W = fft(weights, fft_size)
    A = fft(a, fft_size)
    # cross-correlation: ifft(conj(W) * A) gives sum_k W[k+m] * a[m+... ]
    # Actually: ifft(conj(W) * A)[m] = sum_k conj(W[k]) * A[(k-m) mod fft_size]
    # For linear: = sum_k W[k] * a[k-m] (with proper bounds)
    # So ifft(conj(W) * A)[m] = sum_k W[k] * a[k-m] for m where k-m is in range
    # = sum_j W[j+m] * a[j] (renaming k = j+m)
    # This is what we want (times 2)!
    grad = np.real(ifft(np.conj(W) * A, fft_size))[:n] * 2.0
    return grad


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n * max(conv) / (sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def power_mean_optimize(a0, p, n_iters, lr, deadline):
    """Adam optimization of p-th power mean of convolution."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_convolve(a)
        # Weight for gradient: w[k] = p * conv[k]^(p-1) / (sum conv^p)^(1/p) * p^(1/p)
        # Actually for power mean: (sum c^p)^(1/p), gradient is p * c^(p-1) / (sum c^p)^(1/p)
        # = p * c^(p-1) / NM where NM = (sum c^p)^(1/p)
        conv_p = conv ** p
        sum_conv_p = conv_p.sum()
        if sum_conv_p < 1e-30:
            break
        NM = sum_conv_p ** (1.0 / p)
        weights = p * conv ** (p - 1) / (NM * NM ** (p - 1))
        # Simplify: weights = p * conv^(p-1) / NM^p = p * conv^(p-1) / sum_conv_p
        weights = p * conv ** (p - 1) / sum_conv_p
        
        grad = fast_convolve_grad(a, weights)
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Check improvement periodically
        if it % 20 == 0 or it == n_iters - 1:
            obj = compute_objective(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def subgradient_optimize(a0, n_iters, lr, deadline, momentum=0.9):
    """Subgradient descent with momentum on exact max."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s
    
    vel = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_convolve(a)
        k_max = np.argmax(conv)
        
        # Subgradient: grad[m] = 2*a[k_max - m] if 0 <= k_max - m < n
        grad = np.zeros(n)
        for m in range(n):
            idx = k_max - m
            if 0 <= idx < n:
                grad[m] = 2.0 * a[idx]
        
        vel = momentum * vel - lr * grad
        a_new = project_simplex(a + vel)
        
        if it % 10 == 0 or it == n_iters - 1:
            obj = compute_objective(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        a = a_new
        # Decay learning rate
        lr *= 0.999
    
    return best_a, best_obj


def spectral_flatten(a0, n_iters, deadline):
    """Iterative spectral method to flatten autocorrelation."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s
    
    L = 1 << (2 * n - 1 - 1).bit_length()
    target = 1.0 / (2 * n - 1)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        A = fft(a, L)
        A2 = A * A
        c = np.real(ifft(A2, L))[:2 * n - 1]
        
        # Compute adjustment factor in frequency domain
        # We want c to be flat at target level
        # c_target_fft = target * ones(L) (approximately, for the first 2n-1)
        # A2_target = FFT(c_target) approximately
        # But c_target is only defined for 2n-1 values, rest are 0
        c_target = np.zeros(L)
        c_target[:2 * n - 1] = target
        A2_target = fft(c_target, L)
        
        # Adjust A: A_new^2 should be closer to A2_target
        # A_new = A * sqrt(A2_target / A^2) = A * A2_target / A^2 * A = A2_target / A
        # But this might not be a valid FFT of a non-negative sequence
        A2_adj = A2 * 0.7 + A2_target * 0.3  # Blend
        # Take square root (in frequency domain, element-wise)
        # A_new = sqrt(A2_adj) (element-wise complex sqrt)
        # But this might not give a real time-domain signal
        
        # Alternative: adjust magnitude only
        mag_A2 = np.abs(A2)
        mag_target = np.abs(A2_target)
        # Reduce peaks in A2 magnitude
        if mag_A2.max() > 0:
            norm_factor = np.ones(L)
            peak_ratio = mag_A2 / mag_A2.max()
            # Compress the dynamic range
            norm_factor = 0.5 + 0.5 * peak_ratio
        else:
            norm_factor = np.ones(L)
        
        A_new = A * norm_factor
        a_new = np.real(ifft(A_new, L))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj = compute_objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def peak_shave(a, n_steps=20, deadline=None):
    """Reduce the maximum of the convolution by adjusting contributing elements."""
    n = len(a)
    a = a.copy()
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        
        # Find all (i, j) pairs with i+j = k_max
        # Reduce a[i] and a[j] proportionally to their product
        indices = []
        for i in range(max(0, k_max - n + 1), min(n, k_max + 1)):
            j = k_max - i
            if 0 <= j < n:
                indices.append(i)
        
        if not indices:
            break
        
        idx_arr = np.array(indices)
        contribs = a[idx_arr] * a[[k_max - i for i in indices]]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break
        
        # Reduce elements that contribute most to the peak
        reduction = 0.05
        for i in indices:
            j = k_max - i
            a[i] -= reduction * a[i] * a[j] / total_contrib * (conv[k_max] - conv.mean()) / max(conv.mean(), 1e-15)
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return a
        a = a / s
    
    return a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_resampled = np.interp(x_new, x_old, a)
    a_resampled = np.maximum(a_resampled, 0.0)
    s = a_resampled.sum()
    if s > 1e-10:
        a_resampled = a_resampled / s
    else:
        a_resampled = np.ones(new_n) / new_n
    return a_resampled


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    L = 1 << (n - 1).bit_length()
    A = np.zeros(L, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, L // 2)
    for k in range(1, L // 2):
        A[k] = np.exp(1j * phases[k - 1])
        A[L - k] = np.exp(-1j * phases[k - 1])
    if L % 2 == 0:
        A[L // 2] = 1.0
    a = np.real(ifft(A, L))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Power-mean smooth optimization with Adam and Nesterov acceleration
- Spectral domain iterative flattening
- Exact subgradient descent with momentum
- Peak-shaving and hot/cold water-filling heuristics
- Multi-length exploration with resampling
- Direct O(n^2) gradient computation for accuracy
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def fast_convolve_grad(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    grad[m] = 2 * sum_j a[j] * weights[j+m] (where j+m in [0, 2n-2])
    Uses FFT cross-correlation.
    """
    n = len(a)
    L = 1 << (3 * n + 7).bit_length()
    A = fft(a, L)
    W = fft(weights, L)
    cross = np.real(ifft(A * np.conj(W), L))
    indices = np.array([(L - m) % L for m in range(n)])
    grad = 2.0 * cross[indices]
    return grad


def compute_objective(a):
    """Compute 2n * max(conv) / (sum)^2 directly via FFT."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def power_mean_optimize(a0, p, n_iters, lr, deadline):
    """Adam optimization of p-th power mean of convolution (smooth max approx)."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_obj = compute_objective(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_convolve(a)
        conv_p = conv ** p
        sum_conv_p = conv_p.sum()
        if sum_conv_p < 1e-30:
            break
        weights = p * conv ** (p - 1) / sum_conv_p

        grad = fast_convolve_grad(a, weights)

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)

        a_new = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps_adam))

        if it % 25 == 0 or it == n_iters - 1:
            obj = compute_objective(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def nesterov_optimize(a0, p, n_iters, lr, deadline):
    """Nesterov-accelerated optimization of p-th power mean."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_obj = compute_objective(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        momentum = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        y = project_simplex(y + (it / momentum) * (y - a))

        conv = fast_convolve(y)
        conv_p = conv ** p
        sum_conv_p = conv_p.sum()
        if sum_conv_p < 1e-30:
            break
        weights = p * conv ** (p - 1) / sum_conv_p
        grad = fast_convolve_grad(y, weights)

        a_new = project_simplex(y - lr * grad)

        if it % 25 == 0 or it == n_iters - 1:
            obj = compute_objective(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def subgradient_optimize(a0, n_iters, lr, deadline, decay=0.9995):
    """Subgradient descent with momentum on exact max of convolution."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s

    vel = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_objective(a)
    current_lr = lr

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))

        grad = np.zeros(n)
        for m in range(n):
            idx = k_max - m
            if 0 <= idx < n:
                grad[m] = 2.0 * a[idx]

        vel = 0.9 * vel - current_lr * grad
        a_new = project_simplex(a + vel)

        if it % 15 == 0 or it == n_iters - 1:
            obj = compute_objective(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        a = a_new
        current_lr *= decay

    return best_a, best_obj


def spectral_flatten(a0, n_iters, deadline, blend=0.3):
    """Iterative spectral method to flatten autocorrelation magnitude."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s

    L = 1 << (2 * n - 1 - 1).bit_length()
    target = 1.0 / (2 * n - 1)

    best_a = a.copy()
    best_obj = compute_objective(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = fft(a, L)
        A2 = A * A
        mag_A2 = np.abs(A2)

        if mag_A2.max() < 1e-15:
            break

        # Compress dynamic range of |A|^2
        log_mag = np.log(mag_A2 + 1e-15)
        log_target = np.full(L, np.log(target * L))  # target for flat
        # Blend current log-magnitude toward target
        new_log_mag = (1 - blend) * log_mag + blend * log_target
        new_mag = np.exp(new_log_mag)

        # Preserve phase
        phase = np.angle(A2)
        A2_new = new_mag * np.exp(1j * phase)

        # A_new = sqrt(A2_new) (element-wise complex sqrt)
        # Use polar form: sqrt(r*exp(i*th)) = sqrt(r)*exp(i*th/2)
        A_new = np.sqrt(new_mag) * np.exp(0.5j * phase)

        a_new = np.real(ifft(A_new, L))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def hot_cold_waterfill(a0, n_iters, deadline, strength=0.05):
    """Adjust a to reduce peaks (hot) and fill troughs (cold) in convolution."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_obj = compute_objective(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_convolve(a)
        k_hot = int(np.argmax(conv))
        k_cold = int(np.argmin(conv))

        # Reduce contribution to hot spot
        adjust = np.zeros(n)
        for m in range(n):
            idx_h = k_hot - m
            idx_c = k_cold - m
            if 0 <= idx_h < n:
                adjust[m] -= strength * a[idx_h]
            if 0 <= idx_c < n:
                adjust[m] += strength * a[idx_c]

        a_new = project_simplex(a + adjust)

        obj = compute_objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def peak_shave(a, n_steps, deadline):
    """Reduce the maximum convolution value by directly shrinking contributors."""
    n = len(a)
    a = a.copy()
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        mean_conv = conv.mean()

        if conv[k_max] <= mean_conv * 1.01:
            break

        indices = []
        for i in range(max(0, k_max - n + 1), min(n, k_max + 1)):
            j = k_max - i
            if 0 <= j < n:
                indices.append(i)

        if not indices:
            break

        excess = conv[k_max] - mean_conv
        for i in indices:
            j = k_max - i
            a[i] -= 0.03 * a[i] * a[j] / max(a[i] * a[j], 1e-15) * excess / max(conv[k_max], 1e-15)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return a
        a = a / s

    return a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_res = np.interp(x_new, x_old, a)
    a_res = np.maximum(a_res, 0.0)
    s = a_res.sum()
    if s > 1e-10:
        a_res = a_res / s
    else:
        a_res = np.ones(new_n) / new_n
    return a_res


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum (random phase)."""
    L = 1 << (n - 1).bit_length()
    A = np.zeros(L, dtype=complex)
    A[0] = np.sqrt(n)
    half = L // 2
    phases = rng.uniform(0, 2 * np.pi, half)
    for k in range(1, half):
        A[k] = np.exp(1j * phases[k - 1])
        A[L - k] = np.exp(-1j * phases[k - 1])
    if L % 2 == 0:
        A[half] = 1.0
    a = np.real(ifft(A, L))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            inits['parent'] = resample_sequence(parent, n)

    inits['flat_spec'] = flat_spectrum_init(n, rng)

    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gaussian'] = g

    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    return inits


def score_sequence(a):
    """Score a sequence using the evaluator."""
    if a is None or len(a) == 0:
        return float('inf')
    a_clean = np.maximum(a, 0.0)
    s = a_clean.sum()
    if s < 1e-10:
        return float('inf')
    try:
        score = float(evaluate_sequence(a_clean.tolist()))
    except Exception:
        return float('inf')
    return score if np.isfinite(score) else float('inf')


def propose_candidate(budget_s=1800):
    """
    Main entry point. Multi-phase search combining power-mean Adam/Nesterov
    optimization, spectral flattening, subgradient descent, hot/cold water-filling,
    and peak shaving to minimize 2n*max(conv)/(sum)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    rng = np.random.default_rng(42)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print("Parent loaded: length=%d" % len(parent))
    except Exception:
        pass

    def update_best(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        sc = score_sequence(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_sequence = np.maximum(a, 0.0).copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" %
                  (sc, len(a), time.monotonic() - t_start))
        return sc

    # === Phase 1: Quick scan of initializations (first 250s) ===
    print("=== Phase 1: Quick Scan ===")
    phase1_end = t_start + 250.0

    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 256

    inits = generate_initializations(n_main, parent, rng)
    init_priority = ['parent', 'flat_spec', 'uniform', 'gaussian', 'bimodal', 'random']
    init_names = [nm for nm in init_priority if nm in inits] + \
                 [nm for nm in inits.keys() if nm not in init_priority]

    for name in init_names:
        if time.monotonic() > phase1_end - 15:
            break
        a0 = inits[name]
        t_alloc = min(40.0, phase1_end - time.monotonic() - 10)
        n_iters = int(min(300, t_alloc * 30))
        a_opt, _ = power_mean_optimize(a0, p=32, n_iters=n_iters, lr=0.04,
                                       deadline=time.monotonic() + t_alloc)
        update_best(a_opt)

    print("Phase 1 best: %.6f" % best_score)

    # === Phase 2: Deep optimization with p-schedule (next 550s) ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_end = t_start + 800.0

    if best_sequence is not None:
        a_cur = best_sequence.copy()
        n_cur = len(a_cur)
    else:
        a_cur = np.ones(n_main) / n_main
        n_cur = n_main

    # P-schedule: high to low for progressive sharpening
    p_schedule = [128, 64, 32, 16, 8, 4]
    remaining_p2 = phase2_end - time.monotonic()
    time_per_p = remaining_p2 / (len(p_schedule) + 2)

    for p in p_schedule:
        if time.monotonic() > phase2_end - 30:
            break
        t_alloc = min(time_per_p, phase2_end - time.monotonic() - 20)
        if t_alloc < 10:
            break
        n_iters = int(min(3000, t_alloc * 25))
        lr_val = max(0.005, 0.15 / np.sqrt(p))
        a_opt, _ = power_mean_optimize(a_cur, p=p, n_iters=n_iters, lr=lr_val,
                                       deadline=time.monotonic() + t_alloc)
        sc = update_best(a_opt)
        a_cur = a_opt

    # Nesterov pass
    if time.monotonic() < phase2_end - 40:
        t_alloc = min(100.0, phase2_end - time.monotonic() - 20)
        n_iters = int(min(2000, t_alloc * 20))
        a_opt, _ = nesterov_optimize(a_cur, p=8, n_iters=n_iters, lr=0.02,
                                     deadline=time.monotonic() + t_alloc)
        update_best(a_opt)
        a_cur = a_opt

    print("Phase 2 best: %.6f" % best_score)

    # === Phase 3: Subgradient + spectral + waterfill (next 350s) ===
    print("=== Phase 3: Boosting Strategies ===")
    phase3_end = t_start + 1150.0

    base = best_sequence.copy() if best_sequence is not None else a_cur

    # Subgradient
    if time.monotonic() < phase3_end - 120:
        t_alloc = min(100.0, phase3_end - time.monotonic() - 60)
        n_iters = int(min(5000, t_alloc * 30))
        a_opt, _ = subgradient_optimize(base, n_iters=n_iters, lr=0.015,
                                        deadline=time.monotonic() + t_alloc)
        sc = update_best(a_opt)
        if np.isfinite(sc) and sc < best_score:
            base = a_opt

    # Spectral flattening
    if time.monotonic() < phase3_end - 60:
        t_alloc = min(80.0, phase3_end - time.monotonic() - 30)
        n_iters = int(min(300, t_alloc * 3))
        a_opt, _ = spectral_flatten(base, n_iters=n_iters,
                                    deadline=time.monotonic() + t_alloc)
        update_best(a_opt)

    # Hot/cold water-filling
    if time.monotonic() < phase3_end - 30:
        t_alloc = min(60.0, phase3_end - time.monotonic() - 15)
        n_iters = int(min(500, t_alloc * 5))
        a_opt, _ = hot_cold_waterfill(base, n_iters=n_iters,
                                      deadline=time.monotonic() + t_alloc)
        update_best(a_opt)

    # Peak shave + re-optimize
    if time.monotonic() < phase3_end - 20:
        a_shaved = peak_shave(base, n_steps=30, deadline=time.monotonic() + 15)
        sc = update_best(a_shaved)
        if np.isfinite(sc) and sc < best_score:
            base = a_shaved
            if time.monotonic() < phase3_end - 10:
                a_opt, _ = power_mean_optimize(base, p=8, n_iters=300, lr=0.015,
                                               deadline=time.monotonic() + 20)
                update_best(a_opt)

    print("Phase 3 best: %.6f" % best_score)

    # === Phase 4: Multi-length exploration (next 250s) ===
    print("=== Phase 4: Multi-length Exploration ===")
    phase4_end = t_start + 1450.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        alt_lengths = sorted(set([
            max(64, n_best - 32), max(64, n_best - 16),
            n_best + 16, n_best + 32, n_best + 64
        ]))
    else:
        alt_lengths = [192, 256, 320, 384]

    for n_alt in alt_lengths:
        if time.monotonic() > phase4_end - 30:
            break
        ref_seq = best_sequence if best_sequence is not None else parent
        inits_alt = generate_initializations(n_alt, ref_seq, rng)

        init_p4 = ['parent', 'flat_spec', 'uniform']
        names_p4 = [nm for nm in init_p4 if nm in inits_alt]

        for name in names_p4[:2]:
            if time.monotonic() > phase4_end - 10:
                break
            t_alloc = min(50.0, phase4_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            a0 = inits_alt[name]
            n_iters = int(min(500, t_alloc * 20))
            a_opt, _ = power_mean_optimize(a0, p=16, n_iters=n_iters, lr=0.03,
                                           deadline=time.monotonic() + t_alloc)
            update_best(a_opt)

    print("Phase 4 best: %.6f" % best_score)

    # === Phase 5: Final polish (remaining time) ===
    print("=== Phase 5: Final Polish ===")

    if best_sequence is not None:
        remaining = deadline - time.monotonic()

        # Another round of subgradient with very small lr
        if remaining > 40:
            t_alloc = min(80.0, remaining * 0.3)
            n_iters = int(min(5000, t_alloc * 40))
            a_opt, _ = subgradient_optimize(best_sequence, n_iters=n_iters,
                                            lr=0.005, deadline=time.monotonic() + t_alloc)
            update_best(a_opt)
            remaining = deadline - time.monotonic()

        # Peak shave
        if remaining > 20:
            a_shaved = peak_shave(best_sequence, n_steps=40,
                                  deadline=time.monotonic() + min(15, remaining * 0.3))
            update_best(a_shaved)
            remaining = deadline - time.monotonic()

        # Random perturbations + quick optimize
        if remaining > 30:
            n_pert = min(8, int(remaining / 12))
            for _ in range(n_pert):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.001 + 0.003 * rng.uniform()
                noise = rng.standard_normal(len(best_sequence)) * scale
                a_pert = project_simplex(best_sequence + noise)
                t_alloc = max(5.0, min(20.0, (deadline - time.monotonic() - 5) / 2))
                n_iters = int(min(400, t_alloc * 25))
                a_opt, _ = power_mean_optimize(a_pert, p=16, n_iters=n_iters,
                                               lr=0.015, deadline=time.monotonic() + t_alloc)
                update_best(a_opt)

        # Final low-p optimization for precision
        if remaining > 15:
            t_alloc = min(30.0, remaining - 5)
            n_iters = int(min(1000, t_alloc * 30))
            a_opt, _ = power_mean_optimize(best_sequence, p=4, n_iters=n_iters,
                                           lr=0.005, deadline=time.monotonic() + t_alloc)
            update_best(a_opt)

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = score_sequence(best_sequence)
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
