"""
Hybrid multiplicative gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smooth-max approximation, multiplicative (mirror) descent on the simplex,
periodic LP-based corrections, and multi-phase exploration with diverse initializations.
Key improvements over previous: multiplicative updates (maintain non-negativity naturally),
adaptive beta schedule, Nesterov-like momentum, full-constraint LP, and better time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k=0..2n-2."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M, top_k=None, conv=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M."""
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[peaks]
    else:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(len(A_ub), M)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def smooth_convolve_and_grad(a, beta):
    """Compute softmax-weighted gradient of max_k q_k(a).
    Returns (grad, conv, w)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # Using FFT: grad = IFFT(FFT(a)*conj(FFT(w)))[:n] + IFFT(FFT(a)*FFT(w))[:n]
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()

    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)

    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]

    grad = cross + conv_wa
    return grad, conv, w


def multiplicative_gradient_optimize(a0, time_limit, beta_init=200.0, beta_max=8000.0,
                                     lr_init=0.08, momentum=0.7, seed=0,
                                     lp_interval=50, top_k=None, verbose=False):
    """Multiplicative (mirror) gradient descent with smooth max and LP corrections."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)  # momentum accumulator

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient
        grad, conv, w = smooth_convolve_and_grad(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Momentum
        v = momentum * v - lr * grad

        # Multiplicative update: a_new[i] = a[i] * exp(v[i]) / Z
        # This naturally maintains a >= 0 and sum(a) = 1
        exp_v = np.exp(np.clip(v, -50, 50))
        a_new = a * exp_v
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Keep the old a, adjust step
            if no_improve % 15 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 100:
                # Reset with perturbation
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.7
                no_improve = 0

        # Periodic LP correction
        if lp_interval > 0 and iter_count % lp_interval == 0 and iter_count > 0:
            conv_curr = fast_convolve(a)
            M = conv_curr.max()
            b = solve_lp_direction(a, M, top_k=top_k, conv=conv_curr)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    # Take a step toward the LP solution
                    t_step = min(1.0, 5.0 / (iter_count + 1))
                    a_lp = (1.0 - t_step) * a + t_step * b
                    a_lp = np.maximum(a_lp, 0.0)
                    s_lp = a_lp.sum()
                    if s_lp > 1e-15:
                        a_lp = a_lp / s_lp
                    conv_lp = fast_convolve(a_lp)
                    obj_lp = 2.0 * n * conv_lp.max()
                    if obj_lp < best_obj - 1e-12:
                        best_obj = obj_lp
                        best_a = a_lp.copy()
                        a = a_lp
                        v = np.zeros(n)
                        no_improve = 0

        # Adaptively increase beta
        beta = min(beta_max, beta * 1.003)
        # Slowly decay learning rate
        lr = max(0.001, lr_init * 0.9995 ** iter_count)

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MUL iter {iter_count}: obj={best_obj:.6f}, beta={beta:.0f}, "
                  f"lr={lr:.5f}, no_imp={no_improve}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def lp_optimize(a0, time_limit, step_init=1.0, top_k=None, relax=1.0,
                seed=0, verbose=False):
    """LP-based optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        b = solve_lp_direction(a, M * relax, top_k=top_k, conv=conv)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                step *= 0.8
            if no_improve > 30:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                step = step_init * 0.6
                no_improve = 0

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Cosine window (good for low sidelobes)
    if n > 1:
        cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        cos_w = np.maximum(cos_w, 0.001)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Triangular
    tri = np.minimum(x.astype(np.float64), n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Gaussians
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    # Bimodal (mass at edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # W-kaiser like
    alpha = 5.0
    if n > 1:
        kaiser = np.i0(np.pi * alpha * np.sqrt(1 - ((2 * x / (n - 1)) - 1) ** 2)) / np.i0(np.pi * alpha)
        kaiser = np.maximum(kaiser, 0.001)
        kaiser /= kaiser.sum()
        inits['kaiser'] = kaiser

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.array(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if len(p) >= n:
            pp = p[:n]
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(p) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def evaluate_obj_normalized(a):
    """Evaluate objective assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

        # Final gradient polish
        remaining2 = deadline - time.monotonic()
        t_grad = min(remaining2 * 0.3, 40.0)
        if t_grad > 10:
            a_opt, obj_opt = multiplicative_gradient_optimize(
                best_sequence, time_limit=t_grad,
                beta_init=5000, lr_init=0.005, momentum=0.9,
                lp_interval=100, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish Grad: score={score:.6f}")

        # Tiny perturbations
        remaining3 = deadline - time.monotonic() - 30
        if remaining3 > 30:
            n_tries = int(min(remaining3 / 3, 30))
            for i in range(n_tries):
                if time.monotonic() > deadline - 35:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: score={best_score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
