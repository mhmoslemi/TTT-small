    c[-1] = 1.0  # minimize t

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if total < 1e-300:

    return e / total

    grad_obj = grad / t
    return obj, grad_obj


def bfgs_smoothed(a0, time_limit, t_init=50.0, max_iters=2000):
    """BFGS on the smoothed objective using exp parameterization."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize x in the exp parameterization
    # a = softmax(x), so x = log(a) (for a > 0)
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_cur = t_init
    count = 0
    
    def fun_and_grad(x):
        nonlocal count
        count += 1
        if time.monotonic() - t0 > time_limit:
            return best_obj * (1 + 1e-10), np.zeros(len(x))
        a_curr = exp_param_to_simplex(x)
        obj, grad_a = log_sum_exp_smoothed(a_curr, t_cur)
        # Chain rule: dx/dx via softmax
        # a_j = softmax(x)_j = exp(x_j) / sum exp(x_i)
        # da_j/dx_i = a_j * (delta(i,j) - a_i)
        # dObj/dx_i = sum_j dObj/da_j * da_j/dx_i = sum_j grad_a_j * a_j * (delta(i,j) - a_i)
        #           = grad_a_i * a_i - a_i * sum_j grad_a_j * a_j
        #           = a_i * (grad_a_i - sum_j grad_a_j * a_j)
        sg = np.dot(grad_a, a_curr)
        grad_x = a_curr * (grad_a - sg)
        return obj, grad_x
    
    options = {'maxiter': max_iters, 'gtol': 1e-10, 'ftol': 1e-12}
    
    # Run BFGS in multiple passes with decreasing temperature
    for pass_i, t_val in enumerate([t_init, t_init*0.5, t_init*0.25, t_init*0.1]):
        if time.monotonic() - t0 > time_limit - 10:
            break
        t_cur = t_val
        try:
            res = minimize(fun_and_grad, x, method='L-BFGS-B', jac=True, options=options)
            if res.success or res.fun < best_obj:
                a_new = exp_param_to_simplex(res.x)
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                x = res.x
        except Exception as e:
            print(f"  BFGS pass {pass_i} error: {e}")
            break
    
    return best_a, best_obj

def generate_diverse_inits(n, parent, rng, num=8):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Parent (resized)
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent.copy(), 0)
        else:
            pp = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    # Anti-palindromic: high on one side, low on other
    # This reduces the center convolution value
    t = x / (n - 1)
    anti = 1.0 + 0.9 * (2*t - 1)  # linear ramp from 0.1 to 1.9
    anti = np.maximum(anti, 0.01)
    inits['ramp'] = anti / anti.sum()
    
    # Cosine modulations
    for freq in [1, 2, 3, 5]:
        c = 1.0 + 0.9 * np.cos(2 * np.pi * freq * x / n)
        c = np.maximum(c, 0.01)
        inits[f'cos_{freq}'] = c / c.sum()
    
    # Gaussian
    for sigma in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sigma*n))**2)
        inits[f'gauss_{sigma}'] = g / g.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Two-peak (anti-symmetric placement)
    p1 = int(n * 0.2)
    p2 = int(n * 0.8)
    two = np.zeros(n)
    two[max(0,p1-5):p1+6] = 1.0
    two[max(0,p2-5):p2+6] = 0.3  # asymmetric
    two = np.maximum(two, 0.01)
    inits['two_peak_asym'] = two / two.sum()
    
    # Cauchy-like (heavy tails)
    cauchy = 1.0 / (1.0 + ((x - 0.5*n) / (0.3*n))**2)
    inits['cauchy'] = cauchy / cauchy.sum()
    
    return inits


def subgradient_step(a, conv, num_peaks, temperature):
    """Compute weighted subgradient over top peaks."""
    n = len(a)
    if num_peaks >= 2*n-1:
        peaks = np.arange(2*n-1)
        vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
        vals = conv[peaks]
    
    if temperature > 1e-15:
        scaled = vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(vals)] = 1.0
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based iteration
with exponential-parameterization L-BFGS polishing, diverse structured
initialization (ramp/cauchy/anti-palindromic), and adaptive phase allocation.
Key improvements: backtracking line search, exp-param BFGS to avoid projection
artifacts, and adaptive temperature scheduling for the smoothed objective.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2n*max(conv)/(sum)^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def build_cross_conv_matrix(a):
    """Build matrix M where (M @ b)[k] = cross_conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    M = np.zeros((num_rows, n))
    for k in range(num_rows):
        js = np.arange(max(0, k - n + 1), min(n, k + 1))
        M[k, js] = a[k - js]
    return M


def lp_min_peak_direction(a, M=None):
    """Solve: min t s.t. sum(b)=1, b>=0, (M@b)[k]<=t for all k."""
    n = len(a)
    if M is None:
        M = build_cross_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = M
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def lp_max_sum_direction(a, M, M_peak):
    """Solve: max sum(b) s.t. b>=0, (M@b)[k]<=M_peak for all k."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_peak)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=M, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exp_param_to_simplex(x):
    """Convert unconstrained x to simplex via softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    total = e.sum()
    if total < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / total


def lse_grad(a, t):
    """Compute log-sum-exp smoothed max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    smax = scaled.max()
    log_sum = smax + np.log(np.exp(scaled - smax).sum())
    obj = log_sum / t
    p = np.exp(scaled - smax)
    p /= p.sum()
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        if k_start <= k_end:
            grad[j] = 2.0 * np.dot(p[k_start:k_end + 1], a[k_start - j:k_end - j + 1])
    grad /= t
    return obj, grad


def lse_obj_grad_x(x, t, n):
    """Objective and gradient in exp-parameterization x (where a = softmax(x))."""
    a = exp_param_to_simplex(x)
    obj, grad_a = lse_grad(a, t)
    sg = np.dot(grad_a, a)
    grad_x = a * (grad_a - sg)
    return obj, grad_x


def bfgs_smoothed_optimize(a0, time_limit, t_schedule):
    """L-BFGS-B on smoothed objective with exp parameterization."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for t_val in t_schedule:
        if time.monotonic() - t0 > time_limit - 5:
            break
        state = {'n': n, 't': t_val}

        def _fun_grad(xv, st):
            return lse_obj_grad_x(xv, st['t'], st['n'])

        options = {'maxiter': 500, 'gtol': 1e-9, 'ftol': 1e-11}
        try:
            res = minimize(_fun_grad, x, args=(state,), method='L-BFGS-B',
                           jac=True, options=options)
            a_new = exp_param_to_simplex(res.x)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
            x = res.x
        except Exception:
            pass

    bs = best_a.sum()
    if bs > 1e-15:
        best_a = best_a / bs
    return best_a, best_obj


def lp_iterate(a0, time_limit, method, step_init, max_iters=2000):
    """LP-based iterative refinement with backtracking line search."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    M = build_cross_conv_matrix(a)
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 300 and it < max_iters:
        it += 1
        M_peak = conv.max()

        if method == 'min_peak':
            b, _ = lp_min_peak_direction(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b = lp_max_sum_direction(a, M, M_peak)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            break

        # Backtracking line search
        a_new = None
        cur_obj = best_obj
        trial = step
        for _ in range(15):
            candidate = (1.0 - trial) * a + trial * b
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                trial *= 0.5
                continue
            candidate = candidate / cs
            conv_c = fast_convolve(candidate)
            obj_c = 2.0 * n * conv_c.max()
            if obj_c < best_obj:
                a_new = candidate
                cur_obj = obj_c
                break
            trial *= 0.5

        if a_new is None:
            a_new = (1.0 - step * 0.5) * a + (step * 0.5) * b
            a_new = np.maximum(a_new, 0.0)
            es = a_new.sum()
            if es < 1e-15:
                break
            a_new = a_new / es
            conv = fast_convolve(a_new)
            cur_obj = 2.0 * n * conv.max()
        else:
            conv = fast_convolve(a_new)
            cur_obj = 2.0 * n * conv.max()

        a = a_new

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            step = min(0.65, step * 1.02)
        else:
            no_improve += 1
            step = max(0.05, step * 0.92)
            if no_improve > 60:
                method = 'max_sum' if method == 'min_peak' else 'min_peak'
                step = step_init
                no_improve = 0

        if it % 3 == 0:
            M = build_cross_conv_matrix(a)

    bs = best_a.sum()
    if bs > 1e-15:
        best_a = best_a / bs
    return best_a, best_obj


def subgrad_optimize(a0, time_limit, lr_init, num_peaks, temp):
    """Projected subgradient with momentum and periodic perturbation."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        if num_peaks >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
            vals = conv[peaks]

        scaled = vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = 0.88 * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9996
        if no_improve > 800:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            lr = max(0.008, lr)
        if no_improve > 4000:
            break

    bs = best_a.sum()
    if bs > 1e-15:
        best_a = best_a / bs
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent.copy(), 0.0)
        else:
            pp = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Linear ramp (asymmetric - reduces center convolution)
    ramp = 0.1 + 1.8 * t
    inits['ramp'] = ramp / ramp.sum()

    # Reverse ramp
    rramp = 0.1 + 1.8 * (1.0 - t)
    inits['rramp'] = rramp / rramp.sum()

    # Cosine modulations
    for freq in [1, 2, 3]:
        c = 1.0 + 0.85 * np.cos(2.0 * np.pi * freq * t)
        c = np.maximum(c, 0.02)
        inits[f'cos_{freq}'] = c / c.sum()

    # Sine (asymmetric)
    s1 = 1.0 + 0.8 * np.sin(np.pi * t)
    s1 = np.maximum(s1, 0.02)
    inits['sine'] = s1 / s1.sum()

    # Gaussian
    for sig in [0.25, 0.4]:
        g = np.exp(-0.5 * ((t - 0.5) / sig) ** 2)
        inits[f'gauss_{sig}'] = g / g.sum()

    # Cauchy
    ca = 1.0 / (1.0 + ((t - 0.5) / 0.25) ** 2)
    inits['cauchy'] = ca / ca.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)

    # Piecewise: high-low-high (bimodal)
    bimodal = np.zeros(n)
    bimodal[:n // 4] = 1.0
    bimodal[3 * n // 4:] = 0.7
    bimodal = np.maximum(bimodal + 0.05, 0.01)
    inits['bimodal'] = bimodal / bimodal.sum()

    return inits


def assign_time_budget(total_budget, n_phases):
    """Allocate time to phases with decreasing weights."""
    weights = np.array([0.35, 0.25, 0.20, 0.12, 0.08])
    weights = weights[:n_phases]
    weights /= weights.sum()
    return weights * total_budget


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: diverse LP exploration -> deep refinement ->
    BFGS polish -> subgradient -> perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    all_results = []

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, compute_obj={compute_objective(p):.6f}")
    except Exception:
        pass

    def _update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  >> New best: {score:.6f} (n={len(a)})")
        all_results.append((score, a.copy(), len(a)))
        return score

    # PHASE 1: LP exploration
    print("=== Phase 1: LP Exploration ===")
    t1 = time.monotonic()

    lengths = [512]
    if parent is not None and len(parent) != 512:
        lengths.insert(0, len(parent))
    for L in [256, 384, 640]:
        if L not in lengths:
            lengths.append(L)

    for n in lengths[:3]:
        remain = deadline - time.monotonic() - 800
        if remain < 60:
            break
        inits = generate_inits(n, parent, rng)
        names = ['parent', 'ramp', 'rramp', 'cos_1', 'sine', 'uniform', 'random']
        names = [nm for nm in names if nm in inits][:5]
        t_each = max(8.0, remain / (len(names) * 2))

        for nm in names:
            remain = deadline - time.monotonic() - 700
            if remain < 8:
                break
            ta = min(t_each, remain)
            if ta < 5:
                break
            for meth in ['min_peak', 'max_sum']:
                ts = ta / 2.0
                if ts < 3:
                    continue
                a0 = inits[nm]
                a_opt, _ = lp_iterate(a0, time_limit=ts, method=meth, step_init=0.3)
                _update(a_opt)

    print(f"Phase 1: best={best_score:.6f}, t={time.monotonic()-t1:.0f}s")

    # PHASE 2: Deep refinement on top candidates
    print("=== Phase 2: Deep Refinement ===")
    t2 = time.monotonic()

    if best_sequence is not None:
        p2_deadline = deadline - 500
        # Sort and take top
        valid = [(s, a, n) for s, a, n in all_results if np.isfinite(s)]
        valid.sort(key=lambda x: x[0])

        for i in range(min(3, len(valid))):
            if time.monotonic() > p2_deadline:
                break
            _, a_i, _ = valid[i]
            ta = min(150.0, p2_deadline - time.monotonic() - 10)
            if ta < 15:
                break
            for meth in ['min_peak', 'max_sum']:
                ts = ta / 2.5
                if ts < 5:
                    continue
                for si in [0.25, 0.15, 0.4]:
                    ts2 = ts / 3.0
                    if ts2 < 3:
                        continue
                    a_opt, _ = lp_iterate(a_i, time_limit=ts2, method=meth, step_init=si)
                    _update(a_opt)

    print(f"Phase 2: best={best_score:.6f}, t={time.monotonic()-t2:.0f}s")

    # PHASE 3: BFGS polish
    print("=== Phase 3: BFGS Polish ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        p3_deadline = deadline - 250
        t_scheds = [[200.0, 100.0, 50.0, 25.0],
                    [100.0, 50.0, 20.0, 10.0],
                    [50.0, 25.0, 10.0, 5.0]]
        for sched in t_scheds:
            if time.monotonic() > p3_deadline:
                break
            ta = min(60.0, p3_deadline - time.monotonic() - 5)
            if ta < 10:
                break
            a_opt, _ = bfgs_smoothed_optimize(best_sequence, time_limit=ta, t_schedule=sched)
            _update(a_opt)
            # Also try from a slightly perturbed version
            noise = rng.standard_normal(len(best_sequence)) * 0.002
            a_pert = project_simplex(best_sequence + noise)
            ta2 = min(30.0, p3_deadline - time.monotonic() - 5)
            if ta2 > 5:
                a_opt2, _ = bfgs_smoothed_optimize(a_pert, time_limit=ta2, t_schedule=sched)
                _update(a_opt2)

    print(f"Phase 3: best={best_score:.6f}, t={time.monotonic()-t3:.0f}s")

    # PHASE 4: Subgradient
    print("=== Phase 4: Subgradient ===")
    t4 = time.monotonic()

    if best_sequence is not None:
        p4_deadline = deadline - 80
        n_best = len(best_sequence)
        for lr in [0.02, 0.01, 0.005]:
            if time.monotonic() > p4_deadline:
                break
            ta = min(35.0, p4_deadline - time.monotonic() - 3)
            if ta < 5:
                break
            a_opt, _ = subgrad_optimize(best_sequence, time_limit=ta, lr_init=lr,
                                        num_peaks=max(10, n_best // 25), temp=0.0004)
            _update(a_opt)

    print(f"Phase 4: best={best_score:.6f}, t={time.monotonic()-t4:.0f}s")

    # PHASE 5: Final perturbations + short LP
    print("=== Phase 5: Final Perturbations ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        p5_deadline = deadline - 5
        n_best = len(best_sequence)
        for scale in [0.001, 0.003, 0.0005, 0.005, 0.002]:
            if time.monotonic() > p5_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            ta = min(25.0, p5_deadline - time.monotonic())
            if ta < 3:
                break
            a_opt, _ = lp_iterate(a_pert, time_limit=ta, method='min_peak', step_init=0.2)
            _update(a_opt)

    print(f"Phase 5: best={best_score:.6f}, t={time.monotonic()-t5:.0f}s")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
