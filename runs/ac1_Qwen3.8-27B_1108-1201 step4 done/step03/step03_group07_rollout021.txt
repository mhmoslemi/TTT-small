def fast_convolve(a):

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using increasing-beta smooth optimization
with asymmetric initializations (key: symmetric sequences are bounded at 2),
multi-length search, and high-precision final refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a where c = conv(a,a)."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj(x, n, beta):
    """Objective for L-BFGS-B in log-space."""
    x_s = x - x.max()
    a = np.exp(x_s)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def softmax_jac(x, n, beta):
    """Jacobian for L-BFGS-B in log-space."""
    x_s = x - x.max()
    a = np.exp(x_s)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    ga = compute_grad_conv_weighted(a, w)
    lag = np.dot(a, ga)
    return a * (ga - lag)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Run L-BFGS-B in log-space."""
    n = len(a0)
    a0c = np.maximum(a0, 1e-30)
    a0c /= a0c.sum()
    x0 = np.log(a0c)
    result = minimize(softmax_obj, x0, jac=softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-12})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    return a_opt


def anneal_beta(a0, betas, maxiter_per_beta=200):
    """Anneal through increasing betas (smooth to sharp)."""
    a = a0.copy()
    for i, beta in enumerate(betas):
        remaining = len(betas) - i
        a = optimize_lbfgsb(a, beta=beta, maxiter=maxiter_per_beta)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
    return a

        a /= a.sum()
        inits['exp_decay_%s' % rate] = a

        a = np.exp(-rate * (n - 1 - x) / n)

        a /= a.sum()

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using increasing-beta smooth L-BFGS-B
annealing (correct direction: smooth-to-sharp), asymmetric initializations
(key insight: symmetric non-negative sequences have objective >= 2, so
asymmetry is required for improvement below 2), multi-length search,
and direct subgradient peak-refinement at the final stage.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).
    Result: 2 * cross_correlation(w, a) truncated to len(a)."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_objective(x, n, beta):
    """Objective in log-space for scipy minimize."""
    x_s = x - x.max()
    a = np.exp(x_s)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jacobian(x, n, beta):
    """Jacobian in log-space for scipy minimize."""
    x_s = x - x.max()
    a = np.exp(x_s)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    ga = compute_grad_conv_weighted(a, w)
    lag = np.dot(a, ga)
    return a * (ga - lag)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Run L-BFGS-B with softmax (log-space) parameterization."""
    n = len(a0)
    a0c = np.maximum(a0, 1e-30)
    a0c /= a0c.sum()
    x0 = np.log(a0c)
    result = minimize(_softmax_objective, x0, jac=_softmax_jacobian,
                      args=(n, beta), method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-12})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    return a_opt


def anneal_increasing_beta(a0, betas, maxiter_per_beta=200):
    """Anneal through increasing betas (smooth to sharp)."""
    a = a0.copy()
    for beta in betas:
        a = optimize_lbfgsb(a, beta=beta, maxiter=maxiter_per_beta)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
    return a


def subgradient_peak_step(a, n_steps=200, step_scale=0.01):
    """Direct subgradient descent on the true max (not smooth approx)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s

    for it in range(n_steps):
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        # subgradient: 2 * a[k_star - j] for j where 0 <= k_star - j < n
        subs = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            subs[j] = 2.0 * a[k_star - j]
        # Normalize subgradient
        sn = np.linalg.norm(subs)
        if sn < 1e-15:
            break
        subs = subs / sn
        # Adaptive step size with backtracking
        step = step_scale * (1.0 / (1.0 + 0.01 * it))
        cur_obj = 2.0 * n * conv.max()
        accepted = False
        for _ in range(12):
            a_new = project_simplex(a - step * subs)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj:
                a = a_new
                cur_obj = obj_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step_scale *= 0.7
            if step_scale < 1e-8:
                break
    return a


def gen_initializations(n, rng, parent=None):
    """Generate diverse initializations, emphasizing asymmetric shapes."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Exponential decay (asymmetric - key for beating 2.0)
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.exp(-rate * x / (n - 1))
        a = np.maximum(a, 0)
        a /= a.sum()
        inits['exp_d_%s' % rate] = a

    # Exponential rise
    for rate in [1.0, 2.0]:
        a = np.exp(-rate * (n - 1 - x) / (n - 1))
        a = np.maximum(a, 0)
        a /= a.sum()
        inits['exp_r_%s' % rate] = a

    # Power law decay
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (x + 1.0) ** (-p)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits['pow_%s' % p] = a

    # Skewed Gaussians
    for center in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for width in [0.1, 0.15, 0.25]:
            a = np.exp(-0.5 * ((x - center * (n - 1)) / (width * n)) ** 2)
            a = np.maximum(a, 0)
            a /= a.sum()
            inits['sg_%s_%s' % (center, width)] = a

    # Ramp
    a = np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()
    inits['ramp'] = a

    # Uniform (baseline = 2.0)
    inits['uniform'] = np.ones(n) / n

    # Hann (symmetric, will be ~2.0)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    a /= a.sum()
    inits['hann'] = a

    # Bimodal (two peaks)
    for pos in [0.2, 0.3]:
        a = np.exp(-0.5 * ((x - pos * (n - 1)) / (0.08 * n)) ** 2) + \
            np.exp(-0.5 * ((x - (1 - pos) * (n - 1)) / (0.08 * n)) ** 2)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits['bimod_%s' % pos] = a

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Frequency-domain: try to get flatter auto-correlation
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    for _ in range(20):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        for k in range(1, fft_size // 2 + 1):
            ph = rng.uniform(0, 2.0 * np.pi)
            A[k] = np.exp(1j * ph)
            if k < fft_size - k:
                A[fft_size - k] = np.exp(-1j * ph)
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    inits['freq_flat'] = best_a

    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            pp = np.maximum(pp, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_rs'] = pp / ps

    return inits
