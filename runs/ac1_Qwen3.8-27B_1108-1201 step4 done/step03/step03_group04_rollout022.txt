def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(n) + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)

Let me also consider: for the LP approach, the bottleneck is solving the LP. For n=384, the LP has 384 vars and 767 constraints. HiGHS should handle this in ~0.1-1s per call. Over 1800s, we might get 200-1800 LP calls. Not bad.

But with projected gradient, each iteration is O(n log n) (two FFTs), which for n=384 is maybe 0.1ms. So we can do millions of iterations!

The tradeoff is that projected gradient converges more slowly (O(1/sqrt(T)) for non-smooth, O(1/T) for smooth with exact line search). But with enough iterations, we should get close.

Let me also think about using **scipy.optimize.minimize** with the smoothed objective. For n=384, L-BFGS-B should work well since the smoothed function is C-infinity. The memory for L-BFGS is O(m*n) where m is the history (default 10), so 10*384 = 3840 floats, which is fine.

The challenge is the constraint: sum(a) = 1 and a >= 0. L-BFGS-B handles bounds (a >= 0) but not equality constraints (sum = 1). I can handle this by:
- Option A: Work in an affine subspace (fix one element, optimize the rest)
- Option B: Use a penalty method (add (sum(a) - 1)^2 to the objective)
- Option C: Normalize after each iteration (project onto simplex)

Option C is simplest and what the projected gradient does. For scipy, I'll use Option B (penalty) or just do my own projected gradient.

Let me go with a hybrid:
- Use my own fast projected gradient for the bulk of the optimization

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines fast FFT-based projected gradient (with log-sum-exp smoothing and Nesterov acceleration),
LP-based Frank-Wolfe polishing, parametric smooth initializations, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1)) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - tau, 0.0)
    return result


def smooth_obj_grad(a, t):
    """Compute log-sum-exp smoothed objective and gradient for 2n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Numerically stable log-sum-exp
    max_c = conv.max()
    exp_arg = t * (conv - max_c)
    # Clip for numerical safety
    exp_arg = np.clip(exp_arg, -50, 50)
    lse = max_c + np.log(np.sum(np.exp(exp_arg))) / t
    
    # Softmax weights
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: 2 * conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_aw = fast_convolve(a_rev, w)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]
    
    return lse, grad, conv, w


def phpw_optimize(a0, time_limit, t_init=10.0, t_growth=1.001, verbose=False):
    """
    Projected gradient with Nesterov acceleration and adaptive log-sum-exp temperature.
    Minimizes 2n * max(conv(a,a)) over the simplex.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    t = t_init
    lr = 1.0 / (n ** 0.5)
    beta = 0.9  # momentum
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        obj_val, grad, conv, w = smooth_obj_grad(a, t)
        
        # Gradient step with momentum
        v = beta * v - lr * grad
        a_new = simplex_project(a + v)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        cur_obj = 2.0 * n * M_new
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            beta = 0.95
            no_improve = 0
        
        # Increase temperature (sharpen the max approximation)
        t *= t_growth
        
        # Stuck handling
        if no_improve > max_stall:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.01
            a = simplex_project(a + noise)
            v = np.zeros(n)
            lr = 1.0 / (n ** 0.5)
            beta = 0.9
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            t = max(t_init, t * 0.5)  # reset temperature partially
        
        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG it={it}, obj={best_obj:.6f}, M={M_new:.4e}, t={t:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return mat


def lp_oracle(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def fpw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M)
        if b is None:
            # Try with slightly relaxed M
            b = lp_oracle(a, M * 1.001)
            if b is None:
                break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_local = 2.0 * n * M
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                candidate = (1 - t) * a + t * b
                conv_c = fast_convolve(candidate)
                obj_c = 2.0 * n * conv_c.max()
                if obj_c < best_local:
                    best_local = obj_c
                    best_t = t
            
            if best_t > 0.01:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > 100:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.005
            a = simplex_project(a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def make_smooth_inits(n, parent, rng):
    """Generate diverse smooth initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Linear ramp
    ramp = x + 0.1
    inits['ramp_up'] = ramp / ramp.sum()
    inits['ramp_down'] = (1 - x + 0.1) / (1 - x + 0.1).sum()
    
    # Quadratic
    quad = (x - 0.5)**2 + 0.1
    inits['quad'] = quad / quad.sum()
    inits['quad_inv'] = (1 - (x - 0.5)**2 * 0.5 + 0.01) / (1 - (x - 0.5)**2 * 0.5 + 0.01).sum()
    
    # Exponential
    exp_ramp = np.exp(x * 2)
    inits['exp_up'] = exp_ramp / exp_ramp.sum()
    inits['exp_down'] = np.exp(-x * 2)
    inits['exp_down'] = inits['exp_down'] / inits['exp_down'].sum()
    
    # Cosine
    cos = np.cos(x * np.pi * 0.5) + 0.1
    inits['cosine'] = cos / cos.sum()
    
    # Sine
    sine = np.sin(x * np.pi) + 0.05
    inits['sine'] = sine / sine.sum()
    
    # Gaussian
    gauss = np.exp(-2 * (x - 0.3)**2 / 0.2)
    inits['gauss_left'] = gauss / gauss.sum()
    gauss2 = np.exp(-2 * (x - 0.7)**2 / 0.2)
    inits['gauss_right'] = gauss2 / gauss2.sum()
    
    # Edge concentrated
    edge = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) * 0.3 + np.abs(x - 0.5) * 1.4 + 0.01
    inits['edge'] = edge / edge.sum()
    
    # V-shape (high at edges, low in middle)
    vshape = np.abs(x - 0.5) * 2 + 0.05
    inits['vshape'] = vshape / vshape.sum()
    
    # Random smooth
    for i in range(3):
        raw = rng.dirichlet(np.ones(n) * 0.3)
        # Smooth it
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        smoothed = np.convolve(raw, kernel, mode='same')
        smoothed = np.maximum(smoothed, 0)
        if smoothed.sum() > 1e-10:
            inits[f'random_smooth_{i}'] = smoothed / smoothed.sum()
    
    # Random (not smooth)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits
