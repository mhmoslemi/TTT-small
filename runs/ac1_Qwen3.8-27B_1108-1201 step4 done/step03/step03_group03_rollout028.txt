"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive softmax-subgradient
descent with momentum, structured sinusoidal perturbations, and multi-phase
time management. Exploits the structure of the auto-convolution gradient
(reversed slices of a) for efficient computation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_gradient(a, peaks, weights):
    """Compute weighted sum of gradients of conv at peak positions.
    
    Gradient of conv_k with respect to a[i] is 2*a[k-i] (0 if k-i out of range).
    """
    n = len(a)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # sg[j] += 2*w*a[k-j] for j in [j_start, j_end]
        src = a[k - j_end : k - j_start + 1][::-1]
        sg[j_start:j_end + 1] += 2.0 * w * src
    return sg


def subgradient_optimize(a_init, time_limit, top_k=50, momentum=0.9,
                         lr_init=0.02, temp_init=0.1, temp_decay=0.999,
                         perturb_interval=200, seed=42):
    """Subgradient descent with softmax-weighted peaks and momentum.
    
    Returns (best_a, best_obj) where obj = 2*n*max(conv) for normalized a.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find top-K peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        
        c_max = peak_vals.max()
        if c_max < 1e-15:
            break
        
        # Softmax weights
        scaled = (peak_vals - c_max) / temp
        weights = np.exp(scaled)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros(len(peaks))
            weights[np.argmax(peak_vals)] = 1.0
        else:
            weights /= w_sum
        
        # Compute weighted gradient
        sg = compute_weighted_gradient(a, peaks, weights)
        # Scale by 2n (the objective factor)
        sg *= 2.0 * n
        
        # Momentum update
        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        iters += 1
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8
            v *= 0.5
            no_improve = 0
        
        # Periodic perturbation
        if iters % perturb_interval == 0 and best_obj > 1.5:
            # Sinusoidal perturbation
            freq = rng.choice(np.arange(1, n // 4 + 1))
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.01 * a.max()
            perturbation = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a = project_simplex(a + perturbation)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
        
        # Decay temperature (make optimization more focused)
        temp = max(0.001, temp * temp_decay)
    
    return best_a, best_obj


def sinusoidal_perturb(a, n_frequencies=5, amplitude=0.02, seed=0):
    """Add sinusoidal perturbations at random frequencies."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a_pert = a.copy()
    for _ in range(n_frequencies):
        freq = rng.integers(1, n // 3 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = amplitude * a.max() * rng.uniform(0.5, 1.5)
        a_pert += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_simplex(a_pert)


def block_perturb(a, n_blocks=3, amplitude=0.03, seed=0):
    """Add block perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a_pert = a.copy()
    for _ in range(n_blocks):
        start = rng.integers(0, n)
        length = rng.integers(max(1, n // 20), n // 4)
        length = min(length, n - start)
        amp = amplitude * a.max() * rng.choice([-1, 1]) * rng.uniform(0.5, 1.5)
        a_pert[start:start + length] += amp
    return project_simplex(np.maximum(a_pert, 0))


def coordinate_polish(a_in, max_rounds=40, n_trials=8):
    """Coordinate descent: try moving small amounts of mass between coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    rng = np.random.default_rng(123)
    
    for rnd in range(max_rounds):
        improved = False
        # Try random pairs of coordinates
        for _ in range(n_trials):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = a[i] * frac
                if delta < 1e-10:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                if a_test[i] < 0:
                    continue
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                    break
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = {}

    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    inits['triangle'] = np.maximum(tri, 0.01)
    inits['triangle'] /= inits['triangle'].sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Cosine-bell
    cos_bell = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2
    inits['cos_bell'] = cos_bell / cos_bell.sum()
    
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            if len(p) == n:

            else:
                p_resized = np.interp(np.linspace(0, 1, n),

                p_resized = np.maximum(p_resized, 0)
                if p_resized.sum() > 1e-10:

    return inits
