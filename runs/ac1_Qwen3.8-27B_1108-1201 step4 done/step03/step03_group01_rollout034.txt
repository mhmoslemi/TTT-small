"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
1. LP-based direction finding targeting top peaks
2. Smoothed gradient-based optimization (L-BFGS-B with log-sum-exp softening)
3. Final refinement with adaptive restarts

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute the true objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_obj_with_conv(a):
    """Compute objective and return both the value and the convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, None
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s), conv


def lp_direction_full(a, conv_a=None):
    """
    Solve LP: minimize t
    subject to: conv_a[k] + 2*conv(a,d)[k] <= t for all k
                sum(d) = 0
                d >= -a
    Returns direction d or None.
    """
    n = len(a)
    if conv_a is None:
        conv_a = fast_convolve(a)
    
    num_constr = 2 * n - 1
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((num_constr, n + 1))
    b_ub = np.zeros(num_constr)
    
    for k in range(num_constr):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A_ub[k, j_start:j_end+1] = 2.0 * a[k - j_start:k - j_end][::-1]
            # a[k-j] for j = j_start, ..., j_end
            # a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
            # = a[k-j_start : k-j_end-1 : -1]
            # Actually: a[k-j] for j in range(j_start, j_end+1)
            # = a[[k-j for j in range(j_start, j_end+1)]]
            # = a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
            # This is a slice a[k-j_start : k-j_end-1 : -1] if k-j_start > k-j_end
            # k-j_start >= k-j_end always (since j_start <= j_end)
            # So: a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
            # = reversed a[k-j_end : k-j_start+1]
        
        A_ub[k, n] = -1.0
        b_ub[k] = -conv_a[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n].copy()
        if np.linalg.norm(d) > 1e-15:
            return d
    return None
