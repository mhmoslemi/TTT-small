def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def solve_lp_banded(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses banded structure for efficiency."""

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Hybrid optimizer combining smooth gradient descent (log-sum-exp approximation),
coordinate descent polish, and Frank-Wolfe with LP to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key innovations: adaptive kappa schedule, cyclic coordinate descent for final polish,
and multi-phase time budgeting.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_descent(a0, time_limit, kappa_init=1.0, kappa_end=0.0005,
                            lr_init=0.008, momentum_init=0.9, beta1=0.9, beta2=0.999,
                            restart_patience=400, restart_scale=0.004):
    """Smooth gradient descent using log-sum-exp approximation with Adam-like updates."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    kappa = kappa_init
    lr = lr_init
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    no_improve = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_convolve(a)
        c_scaled = c / kappa
        c_max = c_scaled.max()
        w = np.exp(c_scaled - c_max)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # Compute gradient of logsumexp(c/kappa) w.r.t. a
        grad = np.zeros(n)
        active_threshold = 1e-10
        active = np.where(w > active_threshold)[0]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[k] * a[k - idx]
        grad *= 2.0 / kappa

        # Adam update
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / bc1
        v_hat = v / bc2
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + 1e-8))

        # Schedule
        kappa = max(kappa * 0.9997, kappa_end)
        lr = max(lr * 0.99995, lr_init * 0.3)

        # Track best
        c = fast_convolve(a)
        val = c.max()
        if val < best_val - 1e-14:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart
        if no_improve > restart_patience:
            rng = np.random.default_rng((int(time.monotonic() * 10000) + t) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * restart_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            kappa = max(kappa * 3, kappa_init * 0.3)
            no_improve = 0
            restart_scale = max(restart_scale * 0.7, 0.001)

    return best_a, best_val


def coordinate_descent_polish(a0, time_limit, max_passes=50):

                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old
                continue

            # Build the linear coefficient for each convolution position
            # delta_c_k = 2*(t - old) * a[k-i] for k != 2i, a[k-i] valid

            base[2 * i] = c[2 * i] + old * old  # so that c[2i] + t^2 - old^2 = base[2i] + t^2 - 2*old^2...

            def obj_1d(t):

                return result.max()

                else:
                    lo = m1

                c = fast_convolve(a)
                val = c.max()

                    best_a = a.copy()
                improved = True

                a[i] = max(t_opt, 0.0)

        if not improved:
            break

    return best_a, best_val


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = [np.ones(n) / n]
    x = np.arange(n, dtype=np.float64)

    # Sinusoidal modulations
    for freq in [1, 2, 3, 5]:
        for phase in [0.0, np.pi / 3.0, np.pi / 2.0]:

            a = np.maximum(a, 0.0)
            if a.sum() > 1e-10:
                inits.append(a / a.sum())

    # Edge-weighted
    d = np.minimum(x, n - 1 - x).astype(np.float64)

    # Gaussian center
    a = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)

    a = (d + 1.0) ** 0.7
    inits.append(a / a.sum())

    a = np.minimum(x, n - 1 - x).astype(np.float64) + 0.1
    inits.append(a / a.sum())

    # Random Dirichlet
    for _ in range(3):
        inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Parent-based
    if parent is not None:

    print(f"\n=== Phase 5: Final Polish (deadline in {deadline - time.monotonic():.0f}s) ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine smooth GD
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, obj = smooth_gradient_descent(best_seq, time_limit=min(rem, 60),
                                                  kappa_init=0.05, kappa_end=0.00001,
                                                  lr_init=0.001, momentum_init=0.95)
            try_seq(a_opt, "fine_sgd")

        # CD polish again
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=min(rem, 50), max_passes=30)
            try_seq(a_opt, "fine_cd")

        # Tiny perturbations + quick polish
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            for p in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                t_alloc = min(rem - 2, 20)
                if t_alloc > 3:
                    a_opt, obj = smooth_gradient_descent(f_pert, time_limit=t_alloc,
                                                          kappa_init=0.02, kappa_end=0.00001,
                                                          lr_init=0.0005, momentum_init=0.96)
                    try_seq(a_opt, f"micro_{p}")

        # One more CD on the absolute best
        rem = deadline - time.monotonic() - 3
        if rem > 3 and best_seq is not None:
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=min(rem, 15), max_passes=10)
            try_seq(a_opt, "last_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
