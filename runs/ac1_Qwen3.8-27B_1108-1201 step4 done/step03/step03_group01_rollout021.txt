    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2*n - 1, n))
    for i in range(n):
        C[i:i+n, i] = a
    return C

def lp_direction(a, M):
    """Solve LP to find direction."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def nesterov_subgradient(a, n, max_iter, lr, momentum):
    """Nesterov-accelerated subgradient descent."""
    v = np.zeros(n)
    y = a.copy()
    best_a = a.copy()
    best_obj = np.inf
    
    for it in range(max_iter):
        conv = fast_convolve(y)
        cmax = conv.max()
        obj = 2.0 * n * cmax / (y.sum()**2)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
        
        # Subgradient of max(c) = max(conv)
        # ∂c[k]/∂a[j] = 2*a[k-j] (when valid)
        # For the max, take uniform weight over peaks
        peak_thresh = cmax - 1e-10
        peaks = np.where(conv >= peak_thresh)[0]
        
        sg = np.zeros(n)
        for k in peaks:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += y[k - idx]
        
        sg = sg / len(peaks)  # average over peaks
        
        # Nesterov update
        t_k = (1 + np.sqrt(1 + 4*(it+1)**2)) / 2
        v_new = momentum * v - lr * sg
        y_new = y + (1 + momentum) * v_new - momentum * v
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s > 1e-15:
            y_new = y_new / s
        y = y_new
    
    return best_a, best_obj
