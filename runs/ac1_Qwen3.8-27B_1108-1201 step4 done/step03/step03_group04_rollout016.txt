def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_subgradient(a, conv, top_k=8, temperature=0.05):
    """Compute softmax-weighted subgradient of max(conv(a,a)) using top peaks."""
    n = len(a)
    # Find top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        vals = conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[peaks]
    
    # Softmax weights
    v_max = vals.max()
    exps = np.exp((vals - v_max) / (temperature * max(v_max, 1e-15)))
    w = exps / exps.sum()
    
    # Weighted subgradient
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # d(conv[a,a][k])/d(a[j]) = 2*a[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += 2.0 * w[i] * a[k - idx]
    return g

    y = a.copy()

    beta = 0.9

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0

    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        g = softmax_subgradient(y, conv_y, top_k=top_k, temperature=temperature)
        
        # Momentum update
        v = beta * v + (1.0 - beta) * g
        
        # Mirror descent step: a_new = a * exp(-eta * v), then project to simplex

def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit_lp=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit_lp})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def simulated_annealing_peak(a0, time_limit, verbose=False):
    """Simulated annealing with peak-targeting structured moves."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    it = 0
    T = 0.01  # Initial temperature
    cool = 0.999
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        # Generate a structured move targeting the peak
        k_peak = int(np.argmax(conv))
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        idx = np.arange(j_start, j_end + 1)
        
        if len(idx) > 0:
            # Pick a random contributing pair
            pi = int(rng.integers(0, len(idx)))
            i1 = int(idx[pi])
            i2 = k_peak - i1
        else:
            i1 = int(rng.integers(0, n))
            i2 = i1
        
        # Move: transfer mass from (i1, i2) to (j1, j2)
        move_type = int(rng.integers(0, 3))
        a_new = a.copy()
        
        if move_type == 0:
            # Reduce one hot element, add to a cold one
            j_cold = int(rng.integers(0, n))
            delta = a[i1] * 0.05 * rng.random()
            a_new[i1] -= delta
            a_new[j_cold] += delta
        elif move_type == 1:
            # Swap mass between two elements
            j_swap = int(rng.integers(0, n))
            if j_swap != i1 and j_swap != i2:
                delta = min(a[i1], a[j_swap]) * 0.1 * rng.random()
                a_new[i1] += delta
                a_new[j_swap] -= delta
        else:
            # Reduce both elements of the peak pair proportionally
            delta1 = a[i1] * 0.03 * rng.random()
            delta2 = a[i2] * 0.03 * rng.random() if i2 != i1 else 0.0
            j_cold1 = int(rng.integers(0, n))
            j_cold2 = int(rng.integers(0, n)) if i2 != i1 else j_cold1
            a_new[i1] -= delta1
            a_new[j_cold1] += delta1
            if i2 != i1:
                a_new[i2] -= delta2
                a_new[j_cold2] += delta2
        
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        # Acceptance criterion
        if obj_new < cur_obj:
            a = a_new
            conv = conv_new
            cur_obj = obj_new
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(obj_new - cur_obj) / T):
                a = a_new
                conv = conv_new
                cur_obj = obj_new
        
        T *= cool
        if T < 1e-6:
            T = 0.001  # Reheat occasionally
        
        it += 1
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA it={it}, T={T:.4f}, best={best_obj:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Linear ramp
    r = np.linspace(0.1, 1.0, n)
    inits['ramp'] = r / r.sum()
    
    # Reverse ramp
    r2 = np.linspace(1.0, 0.1, n)
    inits['ramp_rev'] = r2 / r2.sum()
    
    # Triangle
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # Edge-concentrated
    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    inits['edge'] = ec / ec.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal (two Gaussians)
    bm = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.1 * n)) ** 2)
    inits['bimodal'] = bm / bm.sum()
    
    # Random Dirichlet (concentrated)
    inits['rand_conc'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random Dirichlet (uniform-ish)
    inits['rand_uni'] = rng.dirichlet(np.ones(n) * 1.0)
    
    # Sparse random
    sparse = rng.random(n)
    mask = rng.random(n) > 0.5
    sparse = sparse * mask
    if sparse.sum() > 1e-10:
        inits['sparse'] = sparse / sparse.sum()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy, multi-length search to minimize the evaluation objective.
    Combines mirror descent, coordinate descent, LP-Frank-Wolfe, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    def update_best(a_opt, label=""):
        nonlocal best_sequence, best_score
        if a_opt is not None and len(a_opt) > 0:
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  NEW BEST: score={score:.6f}, n={len(a_opt)} [{label}]")
            return score
        return float('inf')
    
    # === Phase 1: Quick mirror descent scan ===
    print("=== Phase 1: Mirror Descent Scan ===")
    lengths = [256, 384, 512]
    
    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        
        # Priority order
        priority = ['parent', 'parent_interp', 'uniform', 'bimodal', 'edge',
                    'triangle', 'gauss', 'ramp', 'rand_conc', 'sparse', 'rand_uni', 'ramp_rev']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)
        
        # Limit to top 5 inits per length
        for name in init_names[:5]:
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(30.0, deadline - time.monotonic() - 60)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                     top_k=10, temperature=0.03,
                                                     eta0=0.03, decay=0.9997)
            update_best(a_opt, f"MD n={n} {name}")
    
    print(f"After Phase 1: best={best_score:.6f}")
    
    # === Phase 2: Coordinate descent on best ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 2: Coordinate Descent ===")
        remaining = deadline - time.monotonic() - 60
        t_cd = min(120.0, remaining * 0.4)
        if t_cd > 10:
            a_opt, obj_opt = coordinate_descent_peak(best_sequence, time_limit=t_cd)
            update_best(a_opt, "CD")
    
    # === Phase 3: LP Frank-Wolfe on best ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 3: LP Frank-Wolfe ===")
        n_lp = len(best_sequence)
        remaining = deadline - time.monotonic() - 30
        # Only do LP if n is manageable (sparse matrix build is O(n^2))
        if n_lp <= 512:
            t_lp = min(300.0, remaining * 0.5)
            if t_lp > 10:
                a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_lp,
                                                 verbose=True, verbose_every=50)
                update_best(a_opt, "LP-FW")
    
    # === Phase 4: Simulated Annealing ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 4: Simulated Annealing ===")
        remaining = deadline - time.monotonic() - 30
        t_sa = min(150.0, remaining * 0.4)
        if t_sa > 10:
            a_opt, obj_opt = simulated_annealing_peak(best_sequence, time_limit=t_sa)
            update_best(a_opt, "SA")
    
    # === Phase 5: Second round of optimization on best ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Polish ===")
        remaining = deadline - time.monotonic() - 20
        t_polish = min(200.0, remaining * 0.5)
        if t_polish > 10:
            # Mirror descent polish with smaller steps
            a_opt, obj_opt = mirror_descent_optimize(best_sequence, time_limit=t_polish,
                                                     top_k=15, temperature=0.01,
                                                     eta0=0.005, decay=0.9999)
            update_best(a_opt, "MD Polish")
    
    # === Phase 6: Try different lengths from best ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 6: Length Transfer ===")
        remaining = deadline - time.monotonic() - 20
        n_best = len(best_sequence)
        
        for n_new in [max(128, n_best // 2), min(1024, n_best * 2)]:
            if time.monotonic() > deadline - 30:
                break
            if n_new == n_best or n_new < 64 or n_new > 1024:
                continue
            
            # Interpolate to new length
            a_interp = np.interp(np.linspace(0, 1, n_new),
                                 np.linspace(0, 1, n_best),
                                 np.maximum(best_sequence, 0.0))
            a_interp = a_interp / a_interp.sum()
            
            t_alloc = min(60.0, max(10, 0.5 * (deadline - time.monotonic() - 20)))
            if t_alloc < 8:
                break
            
            a_opt, obj_opt = mirror_descent_optimize(a_interp, time_limit=t_alloc,
                                                     top_k=10, temperature=0.03,
                                                     eta0=0.03, decay=0.9997)
            update_best(a_opt, f"LenTransfer n={n_new}")
    
    # === Phase 7: Final LP pass if time remains ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        n_final = len(best_sequence)
        remaining = deadline - time.monotonic() - 15
        if n_final <= 512 and remaining > 20:
            print("=== Phase 7: Final LP ===")
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=remaining - 5,
                                             verbose=False)
            update_best(a_opt, "Final LP")
    
    # === Phase 8: Final coordinate descent polish ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            print("=== Phase 8: Final CD ===")
            a_opt, obj_opt = coordinate_descent_peak(best_sequence, time_limit=remaining - 3)
            update_best(a_opt, "Final CD")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
