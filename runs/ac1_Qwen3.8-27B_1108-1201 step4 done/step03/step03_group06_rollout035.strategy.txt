        def project_simplex(v):
            n = len(v)
            if n == 1:
                return np.array([max(v[0], 0.0)])

            u = np.sort(v)[::-1]
            cssv = np.cumsum(u)
            rho = np.arange(1, n + 2, dtype=bool)  # workaround for indexing

            cond = u * rho > (cssv - 1)

            if notcond.any():
                rho_val = np.max(rho[cond])
                theta = (cssv[rho_val - 1] - 1.0) / rho_val
                return np.maximum(v - theta, 0.0)

            else:
                return np.ones(n) * (1.0 / float(n))
