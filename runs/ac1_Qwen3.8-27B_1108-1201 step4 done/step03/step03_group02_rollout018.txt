def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    # Project onto {x >= 0, sum(x) = 1}
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):

    return A

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with backtracking line search, spectral-damping heuristics, and multi-length
exploration. Leverages asymmetry to beat the symmetric lower bound of 2.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_max_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        if b.sum() > 1e-15:
            return b / b.sum()
    return None


def lp_min_peak(a):
    """Solve: min t s.t. sum(b)=1, b>=0, conv(a,b)[k] <= t for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        if b.sum() > 1e-15:
            return b / b.sum()
    return None


def peak_damping_direction(a, conv, top_k=10):
    """Compute a direction that reduces peak autocorrelation lags via damping."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    total_peak = peak_vals.sum()
    if total_peak < 1e-15:
        return None
    
    w = peak_vals / total_peak
    penalty = np.zeros(n)
    
    for i in range(top_k):
        k = int(peaks[i])
        wi = w[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            partner = k - j
            if a[j] > a[partner]:
                penalty[j] += wi * a[partner]
            elif a[partner] > a[j]:
                penalty[j] += wi * a[j]
            else:
                penalty[j] += wi * a[j] * 0.5
    
    direction = a - 0.5 * penalty
    direction = np.maximum(direction, 0)
    s = direction.sum()
    if s < 1e-15:
        return None
    return direction / s

                no_improve = 0

                found = True

    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 20
        n_best = len(best_sequence)

        for scale in [0.003, 0.008, 0.001, 0.012, 0.005, 0.015]:
            if time.monotonic() > phase4_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(25.0, phase4_deadline - time.monotonic())
            if t_alloc < 5:
                break
            a_opt, _ = lp_refine(a_pert, time_limit=t_alloc, method='min_peak',
                                 max_iters=400, use_damping=True)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic() - t4_start:.0f}s")

    # Phase 5: Final squeeze - alternate LP methods rapidly
    print("=== Phase 5: Final Squeeze ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        t_alloc = deadline - time.monotonic() - 10
        if t_alloc > 20:
            a_opt, _ = lp_refine(best_sequence, time_limit=t_alloc * 0.5,
                                 method='max_sum', max_iters=2000, use_damping=True)
            try_update(a_opt)
            t_alloc2 = deadline - time.monotonic() - 5
            if t_alloc2 > 10:
                a_opt, _ = lp_refine(best_sequence, time_limit=t_alloc2 * 0.8,
                                     method='min_peak', max_iters=2000,
                                     use_damping=False)
                try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic() - t5_start:.0f}s")

    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
