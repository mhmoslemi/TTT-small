"""
Optimized step function search using first-order linearized LP direction finding.
Key innovation: uses the full gradient of the auto-convolution quadratic form
(a[k-i] + a[i-k]) for LP constraints, giving more accurate descent directions
than the previous (a*b)[k] <= M approach. Combined with backtracking line search,
subgradient fallback, and symmetry exploitation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_gradient_matrix(a):
    """Build the gradient matrix G where G[k,i] = d(a*a)[k]/da[i]."""
    n = len(a)
    num_k = 2 * n - 1
    G = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            G[k, i] += a[k - i]
        i_min2 = max(0, k - (n - 1))
        i_max2 = min(n - 1, k + n - 1)
        for i in range(i_min2, i_max2 + 1):
            ik = i - k
            if 0 <= ik < n:
                G[k, i] += a[ik]
    return G


def build_gradient_matrix_v2(a):
    """More efficient gradient matrix using the structure G = C + C^T."""
    n = len(a)
    num_k = 2 * n - 1
    # C[k,i] = a[k-i] if 0 <= k-i < n, else 0
    # G = C + C^T
    # G[k,i] = C[k,i] + C[i,k] = a[k-i] + a[i-k] (where valid)
    G = np.zeros((num_k, n))
    # For each i, G[k,i] gets a[k-i] when 0<=k-i<n, i.e., i<=k<n+i
    # And G[k,i] gets a[i-k] when 0<=i-k<n, i.e., k<=i<k+n
    for i in range(n):
        # First term: a[k-i] valid when 0 <= k-i < n => i <= k < n+i
        k_start = max(0, i)
        k_end = min(num_k - 1, n - 1 + i)
        if k_start <= k_end:
            G[k_start:k_end+1, i] += a[k_end - i : k_start - i : -1] if False else \
                np.array([a[k - i] for k in range(k_start, k_end + 1)])
        # Second term: a[i-k] valid when 0 <= i-k < n => i-n < k <= i
        k_start2 = max(0, i - n + 1)
        k_end2 = min(num_k - 1, i)
        if k_start2 <= k_end2:
            for k in range(k_start2, k_end2 + 1):
                G[k, i] += a[i - k]
    return G


def build_gradient_matrix_fast(a):
    """Build gradient matrix G[k,i] = a[k-i] (if valid) + a[i-k] (if valid)."""
    n = len(a)
    num_k = 2 * n - 1
    G = np.zeros((num_k, n))
    # For each k, the valid i range for a[k-i] is: max(0, k-n+1) <= i <= min(n-1, k)
    # For a[i-k], valid when 0 <= i-k < n, i.e., k <= i < k+n, so i in [max(k,0), min(n-1, k+n-1)] = [k, min(n-1,k+n-1)] if k>=0
    # Simpler: just iterate
    for k in range(num_k):
        # a[k-i] valid: 0 <= k-i < n => k-n < i <= k => max(0, k-n+1) <= i <= min(n-1, k)
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            G[k, idx] += a[k - idx]
        # a[i-k] valid: 0 <= i-k < n => k < i-k+n... => k <= i < k+n => max(0,k) <= i <= min(n-1, k+n-1)
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            G[k, idx2] += a[idx2 - k]
    return G


def solve_linearized_lp(a, conv_val):
    """
    Solve LP: min t s.t. conv_val[k] + G[k,:]·d <= t for all k,
    sum(d) = 0, d >= -a.
    Returns (d, t_opt) or (None, None).
    """
    n = len(a)
    num_k = 2 * n - 1
    G = build_gradient_matrix_fast(a)

    # Variables: [d (n vars), t (1 var)]
    n_vars = n + 1

    c = np.zeros(n_vars)
    c[n] = 1.0  # minimize t

    # Inequality: G @ d - t <= -conv_val  =>  [G | -1] @ [d;t] <= -conv_val
    A_ub = np.zeros((num_k, n_vars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -conv_val.copy()

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Bounds: d_i >= -a_i (no upper bound), t free
    bounds = [(max(-a[i] - 1e-10, -a[i]), None) for i in range(n)] + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None


def solve_original_lp(a, M):
    """
    Original LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns b or None.
    """
    n = len(a)
    num_k = 2 * n - 1

    # Build (a*b) matrix: row k has a[k-i] at position i
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A_ub[k, idx] = a[k - idx]

    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def subgradient_step(a, T=None):
    """Compute subgradient of max(conv(a,a)) using active set."""
    n = len(a)
    if T is None:
        T = max(5, n // 16)
    conv = fast_convolve(a)
    # Get top T lags
    if T >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -T)[-T:]

    # Weight: uniform on active set
    w = np.zeros(len(conv))
    w[top_idx] = 1.0 / T

    # grad_i = sum_k w[k] * (a[k-i] + a[i-k])
    # First term: sum_k w[k] * a[k-i] (cross-correlation of a and w)
    # Second term: sum_k w[k] * a[i-k] (convolution of a and w)

    fft_size = 1 << (2 * n).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w

    A_fft = np.fft.rfft(a_pad)
    W_fft = np.fft.rfft(w_pad)

    # Cross-correlation: sum_j a[j] * w[j+i]
    # = irfft(conj(A) * W)
    cross = np.fft.irfft(np.conj(A_fft) * W_fft)[:n]
    # Convolution: sum_j a[j] * w[i-j]
    # = irfft(A * W)
    conv_term = np.fft.irfft(A_fft * W_fft)[:n]

    grad = cross + conv_term
    return grad, conv


def evaluate_normalized(a):
    """Compute 2n * max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

        alpha = step
        improved = False
        for _range_check in range(8):
            a_try = a + alpha * direction
            a_try = normalize(a_try)
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                improved = True
                break
            alpha *= 0.5
            if alpha < 0.001:
                break

        if not improved:
            no_improve += 1
            if no_improve == 1:
                step *= 0.6
            elif no_improve > 20:
                # Try a perturbation
                noise_scale = 0.01 * (1 + no_improve // 20)
                noise = rng.standard_normal(n) * noise_scale
                a = normalize(a + noise)
                step = 0.3
                no_improve = 0
            elif no_improve > 50:
                # Bigger perturbation
                noise = rng.standard_normal(n) * 0.05
                a = normalize(a + noise)
                step = 0.5
                no_improve = 0
                if verbose:
                    print(f"    Big perturbation at iter {iter_count}")

        # Every 50 iterations without improvement, try symmetry
        if no_improve > 50 and iter_count % 50 == 0:
            a_sym = (a + a[::-1]) / 2.0
            a_sym = normalize(a_sym)
            conv_sym = fast_convolve(a_sym)
            obj_sym = 2.0 * n * conv_sym.max()
            if obj_sym < best_obj:
                best_obj = obj_sym
                best_a = a_sym.copy()
                a = a_sym
                no_improve = 0
                if verbose:
                    print(f"    Symmetry improvement: {best_obj:.6f}")

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Iter {iter_count}: obj={best_obj:.6f}, step={step:.3f}, "
                  f"no_improve={no_improve}, method={method_used}, t={elapsed:.0f}s/{time_limit:.0f}s")

        # Gradually reduce step
        if improved:
            step = min(1.0, step * 1.1)
        else:
            step = max(0.01, step * 0.95)

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_structured_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Tapered (higher in middle)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['tapered'] = tri

    # Edge-heavy
    edge = 1.0 / (1.0 + np.exp(-((x - n/2) / (n * 0.2))))
    edge = edge - edge.min() + 0.1
    edge /= edge.sum()
    inits['edge_heavy'] = edge

    # Gaussian
    sigma = n * 0.25
    g = np.exp(-0.5 * ((x - n/2) / sigma) ** 2)
    g /= g.sum()
    inits['gaussian'] = g

    # Bimodal
    bi = np.zeros(n)
    bi[n//4] = 1.0
    bi[3*n//4] = 1.0
    for i in range(n):
        bi[i] = np.exp(-0.5*((x[i]-n//4)/(n*0.15))**2) + np.exp(-0.5*((x[i]-3*n//4)/(n*0.15))**2)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization using linearized LP direction finding.
    Phase 1: Quick exploration of different lengths and initializations.
    Phase 2: Deep optimization of best candidates.
    Phase 3: Final polishing with very small steps.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    def try_update(a, label=""):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a_clean = np.clip(a, 0, 1000)
        score = evaluate_sequence(a_clean.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_clean.copy()
            print(f"  {label}: score={score:.6f} (n={len(a_clean)})")

    # ========== Phase 1: Quick Exploration (3 minutes) ==========
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = 180.0

    lengths = [192, 256, 320, 384, 512]
    scan_results = []

    for n in lengths:
        if time.monotonic() - t_phase1 > phase1_time - 10:
            break
        inits = generate_structured_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize
        priority = ['parent', 'parent_ds', 'parent_up', 'tapered', 'gaussian',
                    'bimodal', 'uniform', 'random', 'random2', 'edge_heavy']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        per_init_time = max(8.0, (phase1_time - (time.monotonic() - t_phase1) - 5) /
                            max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() - t_phase1 > phase1_time - 5:
                break
            t_alloc = min(per_init_time, phase1_time - (time.monotonic() - t_phase1) - 3)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_from_start(a0, time_limit=t_alloc,
                                                  seed=n + hash(name) % 10000, verbose=False)
            try_update(a_opt, f"Scan n={n},{name}")
            scan_results.append((obj_opt, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ========== Phase 2: Deep Optimization (most of remaining time) ==========
    print("=== Phase 2: Deep Optimization ===")
    phase2_start = time.monotonic()
    phase2_end = deadline - 120.0  # Leave 120s for phase 3
    phase2_time = max(60.0, phase2_end - phase2_start)

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 320

    # Collect starting points
    deep_starts = []
    seen = set()
    for obj, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)

    # Always include current best
    if best_sequence is not None and (best_n, 'best') not in seen:
        deep_starts.append((best_n, best_sequence.copy()))

    # Add fresh starts at best length
    inits_best = generate_structured_initializations(best_n, parent, rng)
    for extra in ['tapered', 'random', 'gaussian']:
        if extra in inits_best and (best_n, extra) not in seen:
            deep_starts.append((best_n, inits_best[extra].copy()))
            seen.add((best_n, extra))

    # Allocate time to each start
    time_per_start = phase2_time / max(1, len(deep_starts))
    time_per_start = max(30.0, time_per_start)

    for i, (n_start, a_start) in enumerate(deep_starts):
        remaining_phase2 = (phase2_end - 10) - time.monotonic()
        if remaining_phase2 < 30:
            break
        t_alloc = min(time_per_start, remaining_phase2)

        print(f"  Deep opt [{i}]: n={n_start}, time={t_alloc:.0f}s")
        a_opt, obj_opt = optimize_from_start(a_start, time_limit=t_alloc,
                                              seed=1000 + i * 7, verbose=True)
        try_update(a_opt, f"Deep[{i}]")

        # Try symmetry
        a_sym = normalize((a_opt + a_opt[::-1]) / 2.0)
        try_update(a_sym, f"Deep[{i}]_sym")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Final Polish ==========
    print("=== Phase 3: Final Polish ===")
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = evaluate_sequence(best_sequence.tolist())

    n_best = len(best_sequence)
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Very careful optimization with small steps
        t_polish = min(remaining - 20, 100.0)
        print(f"  Polishing for {t_polish:.0f}s at n={n_best}")
        a_opt, obj_opt = optimize_from_start(best_sequence, time_limit=t_polish,
                                              seed=9999, verbose=True)
        try_update(a_opt, "Polish")

    # Tiny perturbations
    remaining = deadline - time.monotonic() - 10
    if remaining > 10:
        n_tries = min(int(remaining / 2), 30)
        for i in range(n_tries):
            if time.monotonic() > deadline - 10:
                break
            for scale in [0.0005, 0.001, 0.002, 0.005]:
                noise = rng.standard_normal(n_best) * scale
                a_pert = normalize(best_sequence + noise)
                try_update(a_pert, f"Micro[{i}]s={scale}")
                # Also try symmetry
                a_pert_sym = normalize((a_pert + a_pert[::-1]) / 2.0)
                try_update(a_pert_sym, f"Micro[{i}]s={scale}_sym")

    # Final verification
    final_score = evaluate_sequence(best_sequence.tolist())
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = evaluate_sequence(best_sequence.tolist())

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
