"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated subgradient for exploration, LP with adaptive active set and
line search for refinement, coordinate descent for fine-tuning, and length
exploration. Exploits scale invariance (normalizes sum to 1).
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def subgradient_of_conv(a, conv, beta=100.0):
    """Compute smoothed subgradient of max(conv(a,a))."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # Use FFT for efficiency
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    # First term: sum_k w[k] * a[k-i] = cross-correlation
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    # Second term: sum_k w[k] * a[i-k] = convolution
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return grad


def accelerated_subgradient(a0, time_limit, beta_init=100.0, lr_init=0.1,
                            nesterov=True, seed=0, verbose=False):
    """Accelerated subgradient method with Nesterov averaging on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = objective_normalized(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    y = a.copy()
    v = np.zeros(n)
    t_val = 1.0
    beta = beta_init
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        if nesterov:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            gamma = (t_val - 1.0) / t_new
            y = a + gamma * (a - y)
        else:
            y = a.copy()

        # Compute subgradient at y
        _, conv_y = objective_normalized(y)
        grad = subgradient_of_conv(y, conv_y, beta)

        # Step
        v = v - lr * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1.0:
            v = v / v_norm
        a_new = project_simplex(y + v)

        # Evaluate
        _, conv_new = objective_normalized(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9
            if no_improve > 300:
                # Perturb
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                t_val = 1.0
                y = a.copy()
                lr = lr_init * 0.8
                no_improve = 0

        a = a_new
        t_val = t_new
        beta = min(2000.0, beta * 1.002)
        iter_count += 1

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AS iter {iter_count}: obj={best_obj:.6f}, lr={lr:.4f}, t={elapsed:.1f}s")

    _, conv_final = objective_normalized(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def build_active_set_matrix(a, active_idx):
    """Build constraint matrix for active set only. Returns (A_ub, b_idx)."""
    n = len(a)
    num_k = len(active_idx)
    A_rows = []
    for k in active_idx:
        row = np.zeros(n)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            row[i] = a[k - i]
        A_rows.append(row)
    return np.array(A_rows), active_idx


def solve_lp_direction_fast(a, M, active_idx):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in active_idx."""
    n = len(a)
    A_ub, _ = build_active_set_matrix(a, active_idx)
    b_ub = np.full(len(active_idx), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search(a, direction, best_obj, n, max_iters=20):
    """Golden section line search along direction from a."""
    direction = np.maximum(direction, 0.0)
    d_sum = direction.sum()
    if d_sum < 1e-15:
        return a.copy(), 0.0
    direction = direction / d_sum

    # Golden section search in [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def eval_at(t):
        a_t = (1.0 - t) * a + t * direction
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            return 1e10
        a_t = a_t / s
        obj, _ = objective_normalized(a_t)
        return obj

    c_val = eval_at(c)
    d_val = eval_at(d)

    best_t = 0.0
    best_f = best_obj

    for _ in range(max_iters):
        if c_val < d_val:
            hi = d
            d = c
            d_val = c_val
            c = hi - gr * (hi - lo)
            c_val = eval_at(c)
        else:
            lo = c
            c = d
            c_val = d_val
            d = lo + gr * (hi - lo)
            d_val = eval_at(d)

        if abs(hi - lo) < 1e-6:
            break

    t_opt = (lo + hi) / 2.0
    f_opt = eval_at(t_opt)
    if f_opt < best_f:
        best_t = t_opt
        best_f = f_opt

    a_opt = (1.0 - best_t) * a + best_t * direction
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt / s
    return a_opt, best_f


def lp_optimize_adaptive(a0, time_limit, step_init=0.8, max_active=40,

    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break

        _, conv = objective_normalized(a)
        M = conv.max()

        # Adaptive active set: top peaks + near-peaks
        num_constr = min(2 * n - 1, max(20, max_active))
        sorted_idx = np.argsort(conv)[::-1]
        active_idx = sorted_idx[:num_constr].tolist()

        # Also include any that are > 0.95 * M (near-active)
        near_active = np.where(conv > 0.95 * M)[0]
        for k in near_active:
            if k not in active_idx:
                active_idx.append(int(k))
        active_idx = active_idx[:max_active + 10]

        b = solve_lp_direction_fast(a, M * relax, active_idx)
        if b is None:
            # Fallback: use gradient direction
            grad = subgradient_of_conv(a, conv, beta=200.0)
            b = np.maximum(-grad, 0.0)
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

            break
        b = b / b_sum

        if use_line_search:

        else:
            a_new = (1.0 - step) * a + step * b
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 > 1e-15:
                a_new = a_new / s2
            obj_new, _ = objective_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

        else:

            if not use_line_search:

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    _, conv_final = objective_normalized(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def coordinate_descent(a0, time_limit, seed=0, verbose=False):
    """Pairwise coordinate descent on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = objective_normalized(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    # Focus on indices with significant weight
    significant = np.where(a > 1.0 / n * 0.5)[0]
    if len(significant) < 10:
        significant = np.where(a > 0)[0]

    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Pick a random pair from significant indices
        if len(significant) < 2:
            break
        i, j = rng.choice(significant, size=2, replace=False)
        if i == j:
            continue

        c = a[i] + a[j]
        if c < 1e-12:
            continue

        # Binary search for best alpha in [0, c]
        lo, hi = 0.0, c
        best_alpha = a[i]
        best_f = best_obj

        for _ in range(12):
            mid = (lo + hi) / 2.0
            a[i] = mid
            a[j] = c - mid
            obj, _ = objective_normalized(a)
            if obj < best_f:
                best_f = obj
                best_alpha = mid
                if obj < 1.0 * (2.0 * n / (2 * n - 1)):
                    break
            a[i] = best_alpha
            a[j] = c - best_alpha

            if obj > best_f:
                lo = mid
            else:
                hi = mid

        a[i] = best_alpha
        a[j] = c - best_alpha

        if best_f < best_obj - 1e-12:
            best_obj = best_f
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                no_improve = 0

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD iter {iter_count}: obj={best_obj:.6f}, t={elapsed:.1f}s")

    _, conv_final = objective_normalized(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def symmetric_lp_optimize(a0, time_limit, seed=0, verbose=False):
    """LP optimization with symmetry constraint a[i] = a[n-1-i]."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    # Enforce symmetry
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        a_sym = np.ones(n) / n
    else:
        a_sym = a_sym / s

    half = (n + 1) // 2
    # Work with half the vector
    # a[0..half-1] determines a[half..n-1] by symmetry

    # For odd n: a[0..(n-1)/2-1], a[(n+1)/2..n-1] = a[(n-1)/2-1..0], middle is free

    if n % 2 == 0:

        h = a_sym[:half_n].copy()

    def expand(h):

            return np.concatenate([h, h[-2::-1]])

    best_obj = 2.0 * n * conv.max()

            row = np.zeros(half_n)

                # Contribution of h[i] to conv[k]
                # In full vector, h[i] appears at position i and n-1-i
                full_i1 = i
                full_i2 = n - 1 - i
                val = 0.0

                j1 = k - full_i1
                if 0 <= j1 < n:
                    # a_full[j1] = h[j1] or h[n-1-j1]
                    if j1 < half_n:

                    else:
                        idx2 = n - 1 - j1
                        if idx2 < half_n:
                            row[i] += 1.0  # h[idx2] contribution... this is getting complex

        b = solve_lp_direction_fast(a_full, M, active_idx)

            bs = b.sum()
            if bs < 1e-15:
                break
            b = b / bs

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Enforce symmetry on direction
        b_sym = (b + b[::-1]) / 2.0
        b_sym = np.maximum(b_sym, 0.0)
        bs = b_sym.sum()
        if bs < 1e-15:
            break
        b_sym = b_sym / bs

        a_new = (1.0 - step) * a_full + step * b_sym
        a_new = np.maximum(a_new, 0.0)
        # Enforce symmetry
        a_new = (a_new + a_new[::-1]) / 2.0
        s2 = a_new.sum()
        if s2 > 1e-15:

        _, conv_new = objective_normalized(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:

            a_full = a_new
            no_improve = 0
        else:
            no_improve += 1
            step *= 0.9
            if no_improve > 30:
                noise = rng.standard_normal(n) * 0.005
                noise = (noise + noise[::-1]) / 2.0
                a_full = project_simplex(a_full + noise)
                a_full = (a_full + a_full[::-1]) / 2.0
                a_full = a_full / a_full.sum()
                h = a_full[:half_n]
                step = 0.7
                no_improve = 0

    a_result = a_full if n % 2 == 0 else a_full
    # Make sure it's properly symmetric
    a_result = (a_result + a_result[::-1]) / 2.0
    a_result = np.maximum(a_result, 0.0)
    s = a_result.sum()
    if s > 1e-15:
        a_result = a_result / s

    _, conv_final = objective_normalized(a_result)
    final_obj = 2.0 * n * conv_final.max()

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Symmetric structures
    inits['sym_tri'] = (np.minimum(x, n - 1 - x) + 1.0)
    inits['sym_tri'] /= inits['sym_tri'].sum()

    # Bimodal (edges) - often good for convolution problems
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Single edge bump
    bump = np.exp(-x / (0.15 * n))
    inits['bump_left'] = bump / bump.sum()

    # Center bump
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gauss_center'] = g / g.sum()

    # Wide Gaussian
    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.4 * n)) ** 2)
    inits['gauss_wide'] = g2 / g2.sum()

    # Random sparse
    sparse = np.zeros(n)
    num_nonzero = max(5, n // 10)
    idx = rng.choice(n, size=num_nonzero, replace=False)
    sparse[idx] = rng.random(num_nonzero) + 0.1
    inits['sparse'] = sparse / sparse.sum()

    # Random dense
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Cosine taper (good for spectral properties)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    # Symmetrized random
    r = rng.random(n)
    r_sym = (r + r[::-1]) / 2.0
    r_sym = np.maximum(r_sym, 0.01)
    inits['sym_random'] = r_sym / r_sym.sum()

    return inits


def resample_sequence(a, new_n):
    """Resample sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Fast exploration with accelerated subgradient across lengths/initializations
    2. LP refinement with adaptive active set and line search
    3. Coordinate descent fine-tuning
    4. Symmetric optimization
    5. Length exploration and final polish
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    def time_left():
        return max(0, deadline - time.monotonic())

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Fast Exploration (250s) ==========
    print("=== Phase 1: Fast Exploration ===")
    phase1_end = start_time + 250.0

    lengths = [192, 256, 320, 384, 448, 512, 640]
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize
        priority = ['parent', 'parent_interp', 'bimodal', 'sym_tri', 'gauss_center',
                    'cosine', 'sym_random', 'random', 'uniform', 'sparse']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:6]

        time_per = max(5.0, min(30.0, (phase1_end - time.monotonic() - 10) /
                                max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names))))

        for name in init_names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(time_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, obj_opt = accelerated_subgradient(a0, time_limit=t_alloc,
                                                     beta_init=150, lr_init=0.08,
                                                     seed=n * 31 + hash(name) % 1000)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1: n={n}, {name}, obj={obj_opt:.6f}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ========== Phase 2: LP Refinement (700s) ==========
    print("=== Phase 2: LP Refinement ===")
    phase2_end = start_time + 950.0

    # Collect starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:8]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(deep_starts) >= 5:
            break

    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    # LP configurations
    lp_configs = [
        {'step_init': 0.8, 'max_active': 40, 'relax': 1.0, 'use_ls': True},
        {'step_init': 0.6, 'max_active': 60, 'relax': 1.0, 'use_ls': True},
        {'step_init': 1.0, 'max_active': 30, 'relax': 1.01, 'use_ls': True},
        {'step_init': 0.5, 'max_active': 80, 'relax': 0.99, 'use_ls': False},
    ]

    num_runs = len(deep_starts) * len(lp_configs)
    time_per_run = max(8.0, (phase2_end - time.monotonic() - 10) / max(1, num_runs))

    run_count = 0
    for i, (n_start, a_start) in enumerate(deep_starts):
        for j, cfg in enumerate(lp_configs):
            if time.monotonic() > phase2_end - 15:
                break
            t_alloc = min(time_per_run, phase2_end - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a_opt, obj_opt = lp_optimize_adaptive(
                a_start, time_limit=t_alloc,
                step_init=cfg['step_init'], max_active=cfg['max_active'],
                relax=cfg['relax'], use_line_search=cfg['use_ls'],
                seed=run_count * 13 + 7)

            score = float(evaluate_sequence(a_opt.tolist()))
            run_count += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2 LP [{run_count}]: n={n_start}, cfg{j}, score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Coordinate Descent (400s) ==========
    print("=== Phase 3: Coordinate Descent ===")
    phase3_end = start_time + 1350.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        cd_time = max(10.0, min(100.0, (phase3_end - time.monotonic() - 10) / 4))

        for i in range(4):
            if time.monotonic() > phase3_end - 10:
                break
            t_alloc = min(cd_time, phase3_end - time.monotonic() - 10)
            if t_alloc < 5:
                break

            if i > 0:
                # Slightly perturb
                noise = rng.standard_normal(n_best) * 0.002
                a_start = project_simplex(best_sequence + noise)
            else:
                a_start = best_sequence.copy()

            a_opt, obj_opt = coordinate_descent(a_start, time_limit=t_alloc,
                                                 seed=100 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 CD [{i}]: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Symmetric + Additional LP (250s) ==========
    print("=== Phase 4: Symmetric + LP ===")
    phase4_end = start_time + 1600.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Symmetric optimization
        t_alloc = max(10.0, min(80.0, (phase4_end - time.monotonic() - 20) / 3))
        if t_alloc > 8:
            a_opt, obj_opt = symmetric_lp_optimize(best_sequence, time_limit=t_alloc,
                                                    seed=555)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 Sym: score={score:.6f}")

        # More LP with different configs
        for cfg_i, (ma, rl) in enumerate([(50, 1.0), (35, 1.02), (65, 0.98)]):
            if time.monotonic() > phase4_end - 10:
                break
            t_alloc = max(8.0, min(50.0, phase4_end - time.monotonic() - 10))
            if t_alloc < 5:
                break

            a_opt, obj_opt = lp_optimize_adaptive(
                best_sequence, time_limit=t_alloc,
                step_init=0.5, max_active=ma, relax=rl,
                use_line_search=True, seed=200 + cfg_i)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 LP [{cfg_i}]: score={score:.6f}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Length Exploration + Final Polish (remaining) ==========
    print("=== Phase 5: Final Polish ===")

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Try nearby lengths
        for delta in [-32, -16, 16, 32, 48, 64]:
            if time.monotonic() > deadline - 30:
                break
            new_n = n_best + delta
            if new_n < 64 or new_n > 2048:
                continue

            a_resampled = resample_sequence(best_sequence, new_n)
            t_alloc = max(5.0, min(30.0, deadline - time.monotonic() - 20))
            if t_alloc < 5:
                break

            a_opt, obj_opt = lp_optimize_adaptive(
                a_resampled, time_limit=t_alloc,
                step_init=0.7, max_active=40, relax=1.0,
                use_line_search=True, seed=new_n)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P5 n={new_n}: score={score:.6f}")

        # Final fine LP
        if time.monotonic() < deadline - 20:
            t_alloc = max(5.0, min(40.0, deadline - time.monotonic() - 15))
            a_opt, obj_opt = lp_optimize_adaptive(
                best_sequence, time_limit=t_alloc,
                step_init=0.3, max_active=50, relax=1.0,
                use_line_search=True, seed=999)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P5 Final LP: score={score:.6f}")

        # Tiny perturbations
        if time.monotonic() < deadline - 10:
            n_best = len(best_sequence)
            for i in range(8):
                if time.monotonic() > deadline - 8:
                    break
                for scale in [0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  P5 pert s={scale}: score={score_pert:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
