"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Log-sum-exp smoothed gradient descent with momentum for fast convergence
2. LP-based direction finding for escape from local minima
3. Direct-convolution polishing for final precision
4. Multi-phase annealing with adaptive parameters
Key innovation: efficient FFT-based gradient computation for the smoothed objective,
combined with careful temperature annealing and structural diversity in initialization.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def direct_convolve(a):
    """Direct O(n^2) auto-convolution for precision."""
    n = len(a)
    size = 2 * n - 1
    result = np.zeros(size, dtype=np.float64)
    for i in range(n):
        result[i:i + n] += a[i] * a
    return result


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, mu):
    """Compute softmax weights for log-sum-exp approximation of max."""
    scaled = conv / mu
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_gradient(a, w):
    """
    Compute gradient of sum_k w[k]*conv_k(w.r.t. a_i) efficiently via FFT.
    grad[i] = 2 * sum_k w[k] * a[k-i] for valid k.
    This is a cross-correlation computed via FFT.
    """
    n = len(a)
    a_rev = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    corr_full = np.fft.irfft(W * A_rev, fft_size)
    corr = corr_full[n - 1:n - 1 + n]
    return 2.0 * corr


def smoothed_descent(a0, time_limit, mu_start=None, mu_end=1e-10,
                     lr_init=0.008, momentum=0.92, verbose=False):
    """
    Smoothed gradient descent with momentum for log-sum-exp objective.
    Uses adaptive temperature annealing and step size decay.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if mu_start is None:
        conv_init = fast_convolve(a)
        mu_start = conv_init.max() * 0.03

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr_init
    cur_mu = mu_start
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)

        # Anneal temperature: start high, end very low
        cur_mu = mu_start * (mu_end / mu_start) ** (progress ** 0.7)
        cur_mu = max(cur_mu, 1e-12)

        # Compute gradient
        w = softmax_weights(conv, cur_mu)
        grad = compute_gradient(a, w)

        # Momentum update
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)

        # Check if we moved
        move = np.max(np.abs(a_new - a))
        if move < 1e-15:
            no_improve += 1
            if no_improve > 200:
                # Reset with smaller step
                v = np.zeros(n)
                cur_lr *= 0.5
                no_improve = 0
                continue
        else:
            a = a_new
            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive learning rate
        if no_improve > 50:
            cur_lr *= 0.95
        elif no_improve > 100:
            cur_lr *= 0.9

        if verbose and it % 1000 == 0:
            print(f"      Smoothed it={it}: obj={best_obj:.10f}, mu={cur_mu:.2e}, lr={cur_lr:.2e}, noimp={no_improve}")

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-15:
            return g / g_sum
    return None


def lp_escape(a, time_limit, verbose=False):
    """LP-based Frank-Wolfe iteration for escape from local minima."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(a, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
                continue
            continue

        # Fixed point check
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search: try different alphas
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      LP it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 50:
            break

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent polishing on the exact max-convolution objective.
    For each coordinate, do a small 1D optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

        full_pass += 1
        pass_improve = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Try perturbing coordinate i
            a_old = a[i]
            if a_old < 1e-12:
                # Try to make it positive

                if delta > 1e-14:
                    a[i] = a_old + delta
                    a[(i + 1) % n] -= delta * 0.5
                    a[(i - 1) % n] -= delta * 0.5
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = direct_convolve(a)
                    if conv.max() < best_obj - 1e-13:
                        best_obj = conv.max()
                        best_a = a.copy()
                        pass_improve = True
                    else:
                        a[i] = a_old
                        a[(i + 1) % n] += delta * 0.5
                        a[(i - 1) % n] += delta * 0.5
                        a = a / a.sum()
                continue

            # Try small decreases
            for frac in [0.001, 0.005, 0.01]:
                delta = a_old * frac
                a[i] = a_old - delta
                # Redistribute to neighbors
                a[(i + 1) % n] += delta / 2

                a = np.maximum(a, 0)

                conv = direct_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-13:

                    best_a = a.copy()

                    a_old = a[i]
                    break
                else:
                    a = best_a.copy()

        if pass_improve:
            no_improve = 0
            a = best_a.copy()
        else:
            no_improve += 1
            if no_improve > 3:
                break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Gaussian centered
    sigma = 0.25 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits.append(g / g.sum())

    # Edge-concentrated (U-shaped)
    edge_w = np.minimum(x, n - 1 - x)
    edge_w = 1.0 / (edge_w + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Bimodal
    bim = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) +
           np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits.append(bim / bim.sum())

    # Periodic structure
    period = max(2, n // 10)
    per = 1.0 + 0.3 * np.cos(2 * np.pi * x / period)
    inits.append(np.maximum(per, 0.01) / np.maximum(per, 0.01).sum())

    # Parent interpolated
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Parent with small noise
        pn = px + rng.standard_normal(n) * px.mean() * 0.1
        pn = np.maximum(pn, 0)
        if pn.sum() > 1e-10:
            inits.append(pn / pn.sum())

    return inits

    score_direct = 2.0 * n_final * conv_final.max() / (sum_final ** 2)
    score_evaluate = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFinal (direct): {score_direct:.8f}")
    print(f"Final (evaluate): {score_evaluate:.8f}")
    print(f"n={len(best_seq)}, max_conv={conv_final.max():.10f}, sum={sum_final:.6f}")

    if np.isfinite(score_evaluate):
        best_score = score_evaluate

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
