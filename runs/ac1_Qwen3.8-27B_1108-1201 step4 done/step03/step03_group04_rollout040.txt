"""
Multi-method optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Combines LP-based Frank-Wolfe, adaptive subgradient, multiplicative updates,
and peak-targeted coordinate descent with multi-length scanning.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def local_obj(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def multiplicative_update(a, conv, lr=0.01, n_peaks=10):
    """Multiplicative (Winnow-style) update targeting peaks."""
    n = len(a)
    # Find top peaks
    top_k = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    # Softmax weighting
    temp = max(peak_vals.max() * 0.1, 1e-15)
    w = np.exp((peak_vals - peak_vals.max()) / temp)
    w /= w.sum()
    
    # Compute gradient
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += w[i] * a[k - idx]
    
    # Multiplicative update
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new


def subgradient_update(a, conv, lr=0.01, momentum=None, n_peaks=20):
    """Subgradient update with multi-peak targeting."""
    n = len(a)
    top_k = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    # Softmax weighting
    temp = max(peak_vals.max() * 0.05, 1e-15)
    w = np.exp((peak_vals - peak_vals.max()) / temp)
    w /= w.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += 2.0 * w[i] * a[k - idx]
    
    new_a = a - lr * g
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-15:

def subgradient_polish(a0, time_limit, lr_init=0.003, momentum=0.9, verbose=True, verbose_every=500):
    """Polish with projected subgradient on simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a /= a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.03, 1e-15)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new /= s
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= 0.9997
        it += 1
        if verbose and it % verbose_every == 0:
            print(f"    SubGrad it={it}, obj={best_obj:.6f}")

    score = float(evaluate_sequence(best_a.tolist()))
    return best_a, score


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    inits['ramp_rev'] = r[::-1].copy()

    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()

    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Broader Gaussian
    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_narrow'] = g2.copy()

    # Cosine bump
    cos_bump = np.cos(np.pi * x / (n - 1)) ** 2 + 0.1
    cos_bump /= cos_bump.sum()
    inits['cosine'] = cos_bump.copy()

    # Two-sided exponential
    center = (n - 1) / 2
    exp2 = np.exp(-np.abs(x - center) / (n / 6))
    exp2 /= exp2.sum()
    inits['exp2'] = exp2.copy()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sparse
    sparse = rng.random(n)
    sparse[sparse < 0.5] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits

"""
Multi-method optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Combines LP-based Frank-Wolfe, multiplicative (Winnow) updates, adaptive
subgradient, and peak-equalization coordinate descent with multi-length
scanning and smart restarts. Key innovations over previous approach:
  - Multiplicative updates preserve non-negativity naturally and explore
    different regions of the optimization landscape.
  - Peak-equalization moves directly address the imbalance between peaks.
  - Adaptive LP M-tightening for more aggressive exploration.
  - Multi-length scanning (256, 384, 512) to find globally better n.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    out = np.fft.irfft(A * B, fft_size)[:size]
    return out


def local_obj(a):
    """Compute 2*n*max(conv) assuming sum(a)=1. Returns (obj, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 120
    m_tighten = 1.0

    while time.monotonic() - t0 < time_limit:
        # Slightly tighten M over time to force more exploration
        M_eff = M * m_tighten
        b = lp_oracle(a, M_eff, time_limit=lp_time)
        if b is None:
            b = lp_oracle(a, M * 1.02, time_limit=lp_time)
            if b is None:
                break
        sb = b.sum()
        if sb < 1e-15:
            break
        b /= sb

        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search with more points
            best_t = 0.0
            best_local = 2.0 * n * fast_convolve(a).max()
            for t in np.linspace(0.02, 0.95, 10):
                candidate = (1 - t) * a + t * b
                cand_obj, _ = local_obj(candidate)
                if cand_obj < best_local:
                    best_local = cand_obj
                    best_t = t

            if best_t < 0.01:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                    m_tighten = max(0.95, m_tighten * 0.999)
                else:
                    no_improve += 1

        if no_improve >= max_stall:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise = rng.standard_normal(n) * 0.02
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.4)
            m_tighten = 1.0

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.6f}, M={M:.4e}, t={elapsed:.1f}s")

    return best_a, best_obj


def multiplicative_optimize(a0, time_limit, verbose=True, verbose_every=2000):
    """Iterative multiplicative (Winnow) updates targeting top peaks."""
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-15)
    a /= a.sum()
    best_a = a.copy()
    best_obj, _ = local_obj(a)

    lr = 0.08
    t0 = time.monotonic()
    it = 0
    n_peaks = min(25, n)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

        # Softmax weighting with adaptive temperature
        temp = max(peak_vals.max() * 0.08, 1e-15)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        # Compute gradient (subgradient of max conv)
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += w[i] * a[k - idx]

        # Multiplicative update
        a = a * np.exp(-lr * g)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        cur_obj, _ = local_obj(a)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        # Decay learning rate
        lr *= 0.9992
        if lr < 0.001:
            lr = 0.001

        it += 1
        if verbose and it % verbose_every == 0:
            print(f"    Mult it={it}, obj={best_obj:.6f}, lr={lr:.4f}")

    score = float(evaluate_sequence(best_a.tolist()))
    return best_a, score


def subgradient_polish(a0, time_limit, lr_init=0.003, momentum=0.91,
                       verbose=True, verbose_every=2000):
    """Polish with projected subgradient with multi-peak softmax weighting."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a /= a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    n_peaks = min(25, n)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

        # Adaptive temperature softmax
        spread = peak_vals.max() - peak_vals.min()
        temp = max(spread * 0.3, peak_vals.max() * 0.03, 1e-15)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new /= s
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= 0.9998
        momentum = min(0.95, momentum + 0.00001)
        it += 1
        if verbose and it % verbose_every == 0:
            print(f"    SubGrad it={it}, obj={best_obj:.6f}")

    score = float(evaluate_sequence(best_a.tolist()))
    return best_a, score


def peak_equalize(a, n_steps=500, verbose=False):
    """
    Directly equalize top peaks by transferring mass from
    high-contributing positions to low-contributing ones.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_obj, _ = local_obj(a)

    for step in range(n_steps):
        conv = fast_convolve(a)
        # Find top peak
        k_max = int(conv.argmax())
        peak_val = conv[k_max]

        # Find contributors to the peak
        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        idx = np.arange(j_start, j_end + 1)
        contributions = a[idx] * a[k_max - idx]

        # Find the pair with maximum contribution
        pair_idx = np.argmax(contributions)
        j_hi = idx[pair_idx]
        j_lo = k_max - j_hi  # the other member of the pair

        # Find a position to transfer mass to (away from peak)
        # Choose position that minimizes its contribution to the peak
        # and is far from the peak's support
        all_pos = np.arange(n)
        peak_support = set(idx.tolist())
        candidates = [p for p in all_pos if p not in peak_support and a[p] < 0.3 * a.mean() * n]
        if not candidates:
            candidates = [p for p in all_pos if p not in peak_support]
        if not candidates:
            break

        # Pick the candidate that would most reduce the peak if we transfer to it
        # (i.e., it doesn't pair with the peak)
        j_target = candidates[0]

        # Transfer a small amount of mass
        delta = 0.001 * max(a[j_hi], a[j_lo])
        a[j_hi] = max(0, a[j_hi] - delta)
        a[j_target] = a[j_target] + delta

        # Renormalize
        a /= a.sum()

        cur_obj, _ = local_obj(a)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        elif verbose and step % 100 == 0:
            print(f"      PE step={step}, obj={best_obj:.6f}")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    inits['uniform'] = np.ones(n) / n

    # Cosine bump (smooth, symmetric)
    cos_b = np.cos(np.pi * (x - center) / (n - 1)) ** 2 + 0.05
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b.copy()

    # Various Gaussians
    for sigma_frac, name in [(0.25, 'gauss_wide'), (0.15, 'gauss_mid'),
                             (0.08, 'gauss_narrow')]:
        g = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
        g /= g.sum()
        inits[name] = g.copy()

    # Edge-concentrated
    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()

    # Ram
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()

    # Two-sided exponential
    exp2 = np.exp(-np.abs(x - center) / (n / 8))
    exp2 /= exp2.sum()
    inits['exp2'] = exp2.copy()

    # Plateau (flat center, tapered edges)
    plateau_width = n // 3
    plateau = np.ones(n)
    taper = plateau_width // 2
    if taper > 1:
        for i in range(taper):
            f = 1.0 - i / taper
            plateau[i] = f
            plateau[n - 1 - i] = f
    plateau /= plateau.sum()
    inits['plateau'] = plateau.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse
    sparse = rng.random(n)
    sparse[sparse < 0.6] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()

    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits


def hybrid_optimize(a0, time_limit, verbose=False):
    """
    Three-stage hybrid: LP-FW (coarse) -> Multiplicative (mid) -> Subgradient (fine).
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a /= a.sum()

    t0 = time.monotonic()

    # Stage 1: LP-FW (40% of time)
    t_fw = time_limit * 0.40
    if t_fw > 10:
        a, obj_fw = fw_optimize(a, t_fw, verbose=verbose, verbose_every=500)
    else:
        obj_fw, _ = local_obj(a)

    elapsed = time.monotonic() - t0
    remaining = time_limit - elapsed - 5
    if remaining < 5:
        score = float(evaluate_sequence(a.tolist()))
        return a, score

    # Stage 2: Multiplicative (35% of time)
    t_mult = remaining * 0.40
    if t_mult > 5:
        a, _ = multiplicative_optimize(a, t_mult, verbose=False)

    elapsed = time.monotonic() - t0
    remaining = time_limit - elapsed - 3
    if remaining < 3:
        score = float(evaluate_sequence(a.tolist()))
        return a, score

    # Stage 3: Subgradient polish (remaining)
    a, score = subgradient_polish(a, remaining, lr_init=0.002, momentum=0.92,
                                  verbose=verbose, verbose_every=5000)
    return a, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase, multi-method search to minimize the evaluation objective.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # === Phase 1: Multi-length, multi-init scan ===
    print("=== Phase 1: Multi-length Hybrid Scan ===")

    # Prioritize parent length, then try others
    if parent is not None:
        lengths = [len(parent), 512, 256]
    else:
        lengths = [384, 512, 256]
    # Remove duplicates preserving order
    seen = set()
    lengths_unique = []
    for ln in lengths:
        if ln not in seen and ln >= 64:
            lengths_unique.append(ln)
            seen.add(ln)
    lengths = lengths_unique

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        # Time allocation: parent length gets 50%, others 25% each
        if parent is not None and n == len(parent):
            t_total_n = min(600, (deadline - time.monotonic() - 200) * 0.55)
        else:
            t_total_n = min(300, (deadline - time.monotonic() - 200) * 0.25)

        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())

        priority = ['parent', 'parent_interp', 'cosine', 'uniform', 'gauss_mid',
                    'plateau', 'edge', 'gauss_wide', 'triangle', 'exp2',
                    'ramp', 'gauss_narrow', 'random', 'sparse']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        # Pick top inits based on available time
        n_inits = min(len(init_names), max(3, int(t_total_n // 60)))
        t_per_init = t_total_n / n_inits

        for name in init_names[:n_inits]:
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 200)
            if t_alloc < 15:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.0f}s")

            a_opt, score = hybrid_optimize(a0, time_limit=t_alloc, verbose=False)
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"    ** NEW BEST: score={score:.6f} **")
                else:
                    print(f"    score={score:.6f}")

    print(f"\nAfter Phase 1: best={best_score:.6f}, "
          f"n={len(best_sequence) if best_sequence is not None else 0}")

    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback: score={best_score:.6f}")

    # === Phase 2: Intensive LP-FW on best ===
    if time.monotonic() < deadline - 120:
        print("=== Phase 2: Intensive LP-FW ===")
        remaining = deadline - time.monotonic() - 100
        t_deep = remaining * 0.40
        if t_deep > 30:
            a_opt, obj_opt = fw_optimize(best_sequence, time_limit=t_deep,
                                         verbose=True, verbose_every=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 LP: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Multiplicative polish ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 3: Multiplicative Polish ===")
        remaining = deadline - time.monotonic() - 50
        t_mult = remaining * 0.35
        if t_mult > 10:
            a_opt, score = multiplicative_optimize(best_sequence, time_limit=t_mult,
                                                   verbose=True, verbose_every=3000)
            if np.isfinite(score):
                print(f"  Phase 3 Mult: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Subgradient deep polish ===
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print("=== Phase 4: Subgradient Deep Polish ===")
        remaining = deadline - time.monotonic() - 30
        t_sub = remaining * 0.45
        if t_sub > 10:
            a_opt, score = subgradient_polish(best_sequence, time_limit=t_sub,
                                              lr_init=0.0015, momentum=0.93,
                                              verbose=True, verbose_every=5000)
            if np.isfinite(score):
                print(f"  Phase 4 SubGrad: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Final LP burst ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            print("=== Phase 5: Final LP Burst ===")
            a_opt, obj_opt = fw_optimize(best_sequence, time_limit=remaining,
                                         verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 5 LP: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 6: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 6: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 15
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 12:
                break
            # Vary perturbation scale
            scales = [0.002, 0.005, 0.01, 0.003, 0.008, 0.015]
            scale = scales[i % len(scales)]
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert /= a_pert.sum()
            t_alloc = min(15.0, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            a_opt, obj_opt = fw_optimize(a_pert, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: ** NEW BEST score={score:.6f} **")
                else:
                    print(f"  Restart {i}: score={score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    print(f"{'='*50}")
    return [float(x) for x in best_sequence]
