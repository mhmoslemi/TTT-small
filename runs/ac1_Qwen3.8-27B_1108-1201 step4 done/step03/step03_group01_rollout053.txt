import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def solve_lp_direction(a, conv_a):
    """
    Solve LP: minimize t
    subject to: 2*(a*d)[k] + conv_a[k] <= t for k = 0, ..., 2n-2
                 d[i] >= -a[i] for i = 0, ..., n-1
                 sum(d) = 0
    
    Variables: [d_0, ..., d_{n-1}, t] (n+1 variables)
    """
    n = len(a)
    # Build the constraint matrix for 2*(a*d)[k]
    # (a*d)[k] = sum_i a[i] * d[k-i] where 0<=i<n, 0<=k-i<n
    # So (a*d)[k] = sum_i a[k-i] * d[i] where max(0,k-n+1) <= i <= min(n-1,k)
    
    # A_ub rows: for each k, 2*(a*d)[k] - t <= conv_a[k] - 0 ... 
    # Actually: 2*(a*d)[k] + conv_a[k] <= t
    # => 2*(a*d)[k] - t <= -conv_a[k]
    
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n + 1))
    b_ub = np.zeros(num_constraints)
    
    for k in range(num_constraints):
        # (a*d)[k] = sum_i a[k-i-0] * d[i] where i ranges over valid indices
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = 2.0 * a[k - i]
        A_ub[k, n] = -1.0  # coefficient of t
        b_ub[k] = -conv_a[k]
    
    # Non-negativity: d[i] >= -a[i] => -d[i] <= a[i]
    A_nn = np.zeros((n, n + 1))
    for i in range(n):
        A_nn[i, i] = -1.0
    b_nn = a.copy()
    
    # Combine: A_ub_total = [A_ub; A_nn], b_ub_total = [b_ub; b_nn]
    A_ub_total = np.vstack([A_ub, A_nn])
    b_ub_total = np.concatenate([b_ub, b_nn])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = [0.0]
    
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(None, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub_total, b_ub=b_ub_total, 
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        t_val = result.x[n]
        return d, t_val
    return None, None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def solve_lp_direction(a, conv_a):
    """
    Solve LP: minimize t
    subject to: conv_a[k] + 2*(a*d)[k] <= t, k=0,...,2n-2
                d[i] >= -a[i], i=0,...,n-1
                sum(d) = 0
    Returns (d, t_val) or (None, None) on failure.
    """
    n = len(a)
    num_conv = 2 * n - 1
    
    # Build A_ub: (num_conv + n) x (n+1)
    A_ub = np.zeros((num_conv + n, n + 1))
    b_ub = np.zeros(num_conv + n)
    
    # Convolution linearization constraints
    for k in range(num_conv):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # coefficient of d[j] is 2*a[k-j] for j in [i_start, i_end]
        # k-j ranges from k-i_end to k-i_start, i.e., from max(0,k-n+1) to min(n-1,k)
        coeffs = 2.0 * a[k - i_end : k - i_start + 1][::-1]
        A_ub[k, i_start:i_end+1] = coeffs
        A_ub[k, n] = -1.0
        b_ub[k] = -conv_a[k]
    
    # Non-negativity: d[i] >= -a[i]  =>  -d[i] <= a[i]
    A_ub[num_conv:, np.arange(n)] = -1.0
    b_ub[num_conv:] = a
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     method='highs')
    
    if result.success:
        return result.x[:n], float(result.x[n])
    return None, None


def line_search(a, d):
    """Find best alpha in [0, alpha_max] for a + alpha*d."""
    # Max alpha for non-negativity
    neg_mask = d < -1e-15
    if neg_mask.any():
        alpha_max = (a[neg_mask] / (-d[neg_mask])).min()
        alpha_max = min(alpha_max, 2.0)
    else:
        alpha_max = 2.0
    
    if alpha_max <= 1e-12:
        return a.copy()
    
    base_obj = compute_objective(a)
    best_a = a.copy()
    best_obj = base_obj
    
    # Try candidate alphas
    candidates = np.array([1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.025, 0.0125])
    for frac in candidates:
        alpha = alpha_max * frac
        if alpha <= 0:
            break
        a_try = a + alpha * d
        if (a_try < -1e-10).any():
            continue
        a_try = np.maximum(a_try, 0.0)
        obj = compute_objective(a_try)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_try.copy()
    
    # Refine around best
    # Find which fraction worked best
    if best_obj < base_obj - 1e-12:
        for frac in [0.9, 1.1, 0.8, 1.2, 0.7, 1.3]:
            # We need to track the actual alpha, let's just try around the best
            pass
    
    return best_a


def line_search_refined(a, d, n_evals=15):
    """More careful line search with bisection."""
    neg_mask = d < -1e-15
    if neg_mask.any():
        alpha_max = (a[neg_mask] / (-d[neg_mask])).min() * 0.999
        alpha_max = min(alpha_max, 3.0)
    else:
        alpha_max = 3.0
    
    if alpha_max <= 1e-12:
        return a.copy()
    
    base_obj = compute_objective(a)
    best_a = a.copy()
    best_obj = base_obj
    
    # Coarse search
    for alpha in [alpha_max * f for f in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]]:
        if alpha <= 0:
            continue
        a_try = a + alpha * d
        if (a_try < -1e-10).any():
            continue
        a_try = np.maximum(a_try, 0.0)
        obj = compute_objective(a_try)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_try.copy()
            best_alpha = alpha
    
    if best_obj >= base_obj - 1e-12:
        return a.copy()
    
    # Fine search around best_alpha
    lo = best_alpha * 0.7
    hi = min(best_alpha * 1.3, alpha_max)
    for _ in range(4):
        mid = (lo + hi) / 2
        a_mid = a + mid * d
        if (a_mid < -1e-10).any():
            hi = mid
            continue
        a_mid = np.maximum(a_mid, 0.0)
        obj_mid = compute_objective(a_mid)
        if obj_mid < best_obj - 1e-14:
            best_obj = obj_mid
            best_a = a_mid.copy()
            lo = mid
        else:
            hi = mid
    
    return best_a


def spectral_perturb(a, rng, strength=0.02):
    """Apply a random Fourier-mode perturbation."""
    n = len(a)
    # Random frequency
    freq = rng.integers(1, n // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    perturbation = strength * np.cos(2 * np.pi * freq * x / n + phase)
    a_new = a + perturbation
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def random_perturb(a, rng, strength=0.01):
    """Apply random perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def lp_optimize(a0, time_limit, verbose=False, max_perturbs=50):
    """
    LP-based iterative optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv_a = fast_convolve(a)
    best_obj = 2.0 * n * conv_a.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    rng = np.random.default_rng(12345 + n)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_convolve(a)
        d, t_val = solve_lp_direction(a, conv_a)
        
        if d is None:
            no_improve += 5
        else:
            # Check if this is a descent direction
            current_max = conv_a.max()
            if t_val < current_max - 1e-10:
                a_new = line_search_refined(a, d)
                new_obj = compute_objective(a_new)
                if new_obj < best_obj - 1e-13:
                    a = a_new
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and iteration % 50 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
                else:
                    no_improve += 1
                    # Still update a slightly for progress
                    neg_mask = d < -1e-15
                    if neg_mask.any():
                        alpha_max = (a[neg_mask] / (-d[neg_mask])).min() * 0.1
                    else:
                        alpha_max = 0.1
                    a_try = a + alpha_max * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a = a_try / s_try
            else:
                no_improve += 1
        
        iteration += 1
        
        # Handle stagnation
        if no_improve >= 30 + 10 * perturb_count:
            if perturb_count < max_perturbs:
                # Choose perturbation type
                if perturb_count % 3 == 0:
                    strength = 0.01 + 0.005 * perturb_count
                    a = spectral_perturb(best_a, rng, strength)
                elif perturb_count % 3 == 1:
                    strength = 0.008 + 0.004 * perturb_count
                    a = random_perturb(best_a, rng, strength)
                else:
                    # Structured: reduce peaks, boost troughs
                    conv_a = fast_convolve(a)
                    peaks = np.where(conv_a > 0.9 * conv_a.max())[0]
                    # Move mass away from positions that contribute to peaks
                    grad = np.zeros(n)
                    for k in peaks[:5]:
                        i_start = max(0, k - n + 1)
                        i_end = min(n - 1, k)
                        grad[i_start:i_end+1] += a[i_start:k+1][::-1]
                    a_new = a - 0.01 * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                
                no_improve = 0
                perturb_count += 1
            else:
                break
        
        if no_improve > 200:
            break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Cosine bump
    cos_bump = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits['cosine'] = cos_bump / cos_bump.sum()
    
    # Sine
    inits['sine'] = (np.sin(np.pi * x / n) + 0.05) / (np.sin(np.pi * x / n) + 0.05).sum()
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits['ramp'] = ramp / ramp.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase LP-based search with spectral perturbations
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}, score={compute_objective(p):.6f}")
    except Exception:
        pass
    
    # Phase 1: Quick scan across lengths and initializations
    print("=== Phase 1: Quick LP Scan ===")
    t_start = time.monotonic()
    phase1_time = 400.0
    phase1_deadline = t_start + phase1_time
    
    lengths = [512, 256]
    if deadline - time.monotonic() > 800:
        lengths.append(1024)
    
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        
        inits = generate_initializations(n, parent, rng)
        # Priority order
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'cosine', 'sine', 'random', 'gaussian']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]
        
        t_per_init = max(20.0, (phase1_time - 20.0) / (len(lengths) * 4))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, verbose=False)
            score = compute_objective(a_opt)
            
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")
    
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on best candidates
    print("=== Phase 2: Deep LP Optimization ===")
    t_start2 = time.monotonic()
    
    # Sort and pick top candidates
    scan_results.sort(key=lambda x: x[0])
    top_candidates = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths or score < best_score:
            top_candidates.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_candidates) >= 3:
            break
    
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 60.0
    t_per_deep = max(30.0, remaining * 0.55 / max(1, len(top_candidates)))
    
    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 50:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 40)
        if t_alloc < 15:
            break
        
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, verbose=False)
        score = compute_objective(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # Phase 3: Intensive polish on the global best
    print("=== Phase 3: Intensive Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 20:
            # Multiple restarts from the best with different perturbations
            n_best = len(best_sequence)
            n_restarts = max(1, int(remaining / 40))
            t_per_restart = remaining / (n_restarts + 1)
            
            for ri in range(n_restarts):
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = min(t_per_restart, deadline - time.monotonic() - 10)
                if t_alloc < 8:
                    break
                
                if ri == 0:
                    a_start = best_sequence.copy()
                else:
                    # Vary perturbation strength
                    strength = 0.005 + 0.01 * ri
                    if ri % 2 == 0:
                        a_start = spectral_perturb(best_sequence, rng, strength)
                    else:
                        a_start = random_perturb(best_sequence, rng, strength)
                
                a_opt, obj_opt = lp_optimize(a_start, time_limit=t_alloc, verbose=False)
                score = compute_objective(a_opt)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {ri}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # Phase 4: Final attempt with different length
    print("=== Phase 4: Alternative Length ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 30:
            n_alt = len(best_sequence) * 2 if len(best_sequence) <= 512 else len(best_sequence) // 2
            if n_alt not in [x[1] for x in scan_results]:
                # Upsample or downsample the best sequence
                if n_alt > len(best_sequence):
                    a_alt = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_sequence)), best_sequence)
                else:
                    a_alt = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_sequence)), best_sequence)
                a_alt = np.maximum(a_alt, 0.0)
                s_alt = a_alt.sum()
                if s_alt > 1e-15:
                    a_alt = a_alt / s_alt
                
                t_alloc = min(remaining - 10, 80.0)
                if t_alloc > 10:
                    a_opt, obj_opt = lp_optimize(a_alt, time_limit=t_alloc, verbose=False)
                    score = compute_objective(a_opt)
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Alt length n={n_alt}: {score:.6f}")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final verification with evaluate_sequence
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    else:
        # Fallback to uniform
        best_sequence = np.ones(256) / 256
        print(f"FINAL (fallback): n=256")
    
    return [float(x) for x in best_sequence]
