"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-based iterative
refinement with backtracking line search, multiple LP formulations,
multi-length exploration, and Adam-polished final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_maximize_min(a):
    """Solve: max t s.t. conv(a,b)[k] >= t for all k, sum(b)=1, b>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    # Variables: b (n vars) + t (1 var) = n+1 vars
    c = np.zeros(n + 1)
    c[-1] = -1.0  # maximize t
    # conv(a,b)[k] >= t  =>  -conv(a,b)[k] + t <= 0
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = -A
    A_ub[:, n] = 1.0
    b_ub = np.zeros(num_rows)
    # sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def line_search(a, b, n):
    """Find best step t in [0,1] minimizing max(conv((1-t)a + t*b, ...))."""
    a = a / a.sum()
    b = b / b.sum()
    best_t = 0.0
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for t in np.linspace(0.05, 1.0, 20):
        a_t = (1 - t) * a + t * b
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-15:
            continue
        a_t = a_t / s
        obj = 2.0 * n * fast_convolve(a_t).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def optimize_lp_iterative(a0, time_limit, method_schedule=None, max_iters=1000,
                          line_search=True, ls_interval=3):
    """LP-based iterative refinement with adaptive methods and line search."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.3
    no_improve = 0
    it = 0
    method_idx = 0
    methods = method_schedule or ['max_sum', 'max_min', 'max_sum', 'min_peak']
    
    # Pre-build matrix (rebuild when a changes significantly)
    A = build_conv_matrix(a)
    A_rebuild_counter = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        M = conv.max()

        method = methods[method_idx % len(methods)]
        
        if method == 'max_sum':
            b = lp_maximize_sum(a, M)
        elif method == 'max_min':
            b, _ = lp_maximize_min(a)
        elif method == 'min_peak':
            b, _ = lp_minimize_peak(a)
        else:
            b = lp_maximize_sum(a, M)

        if b is None or b.sum() < 1e-15:
            method_idx += 1
            continue

        b = b / b.sum()

        if np.linalg.norm(b - a) < 1e-10:
            # Convergence - try different method
            method_idx += 1
            step = min(0.5, step * 1.1)
            continue

        # Line search every ls_interval iterations
        if line_search and it % ls_interval == 0:
            t_opt, _ = line_search(a, b, n)
            step = max(0.05, min(0.8, t_opt * 1.2))
        else:
            step = min(0.6, step * 1.02)

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            step *= 0.5
            continue
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            step *= 0.85

        a = a_new
        conv = conv_new

        # Periodically switch methods
        if no_improve > 30:
            method_idx += 1
            no_improve = 0
            step = 0.3

        if no_improve > 200:
            break

        # Rebuild matrix periodically (since a has changed)
        A_rebuild_counter += 1
        if A_rebuild_counter >= 10:
            A = build_conv_matrix(a)
            A_rebuild_counter = 0

    return best_a, best_obj


def optimize_adam_smoothed(a0, time_limit, lr_init=0.01, t_init=20.0,
                           max_iters=10000, beta1=0.9, beta2=0.999):
    """Adam optimization on log-sum-exp smoothed objective."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = t_init
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute smoothed objective gradient
        conv = fast_convolve(a)
        scaled = t * conv
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp  # softmax weights over conv positions

        # Gradient of 2n * (1/t) * log(sum(exp(t*conv[k])) - t*max(conv))
        # = 2n * sum_k p[k] * d(conv[k])/d(a)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-12:
                continue
            wk = p[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # d conv[k] / d a[j] = a[k-j] for j in [j_start, j_end]
            for j in range(j_start, j_end + 1):
                grad[j] += wk * a[k - j]
        grad *= 2.0 * n  # multiply by 2n (the 1/t and log terms cancel in gradient)

        # Wait, the gradient of (1/t) * log(sum exp(t*conv)) w.r.t. a is:
        # (1/t) * (sum exp(t*conv[k]) * t * dconv[k]/da) / sum exp(t*conv[k])
        # = sum p[k] * dconv[k]/da
        # So the gradient of the smoothed objective = 2n * sum p[k] * dconv[k]/da
        # which is what I have. Good.

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a = a - lr_init * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Project to simplex
        a = project_simplex(a)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature
        if it % 500 == 0 and t < 200:
            t *= 1.5

        if no_improve > 3000:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Cosine modulations
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                inits[f'cos_{freq}_{amp}'] = a / s

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.2)

    # Piecewise constant
    for n_blocks in [4, 8]:
        block_size = n // n_blocks
        vals = rng.uniform(0.5, 1.5, n_blocks)
        a = np.repeat(vals, block_size)
        if n % n_blocks != 0:
            a = np.concatenate([a, rng.uniform(0.5, 1.5, n % n_blocks)])
        a = np.maximum(a, 0)
        inits[f'blocks_{n_blocks}'] = a / a.sum()

    return inits
