Minimizes 2n*max(a*a)/(sum(a))^2 using L-BFGS-B on a log-sum-exp

def _compute_score_fast(a):
    """Fast objective computation (no evaluator overhead)."""
    n = len(a)
    c = np.convolve(a, a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def _run_lbfgsb(x0, beta, lam, maxiter, deadline):
    """Run L-BFGS-B minimizing logsumexp(beta*c) + lam*(sum-1)^2."""
    n = len(x0)

    def func(x):
        c = np.convolve(x, x)
        s = np.sum(x)
        if s < 1e-10:
            return 1e10
        mc = beta * c
        mc_max = np.max(mc)
        exp_vals = np.exp(mc - mc_max)
        obj = mc_max + np.log(np.sum(exp_vals))
        obj += lam * (s - 1.0) ** 2
        return obj

    def grad_func(x):
        c = np.convolve(x, x)
        s = np.sum(x)
        if s < 1e-10:
            return np.zeros(n)
        mc = beta * c
        mc_max = np.max(mc)
        exp_vals = np.exp(mc - mc_max)
        p = exp_vals / np.sum(exp_vals)
        gx = 2.0 * beta * np.correlate(p, x, mode='valid')
        gx += 2.0 * lam * (s - 1.0)
        return gx

    try:
        result = minimize(func, x0, jac=grad_func, method='L-BFGS-B',
                          bounds=[(0, None)] * n,

                                   'ftol': 1e-14, 'gtol': 1e-9})
        return result.x if result.x is not None else x0
