    """Project v onto {x >= 0, sum(x) = target}."""

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 via smooth gradient descent on the simplex.

Uses a combination of:
- Log-sum-exp smoothing (top-k) to approximate the max of the self-convolution
- Variance reduction of the convolution profile
- Adam optimizer with simplex projection
- Running average (Polyak-Ruppert) for convergence
- Multiple restarts with different initializations
"""

import numpy as np
import time

def _project_simplex(v, target):
    """Project v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64).copy()
    m = len(v)
    if target <= 0:
        return np.zeros(m)
    u = -np.sort(-v)
    cssv = np.cumsum(u)
    rho = np.sum(u * (np.arange(m) + 1) > (cssv - target)) - 1
    if rho < 0:
        return np.full(m, target / m)
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0)


def _smooth_logsumexp_grad(p, tau, top_k):
    """Gradient of (1/tau)*log(sum_k exp(tau*c_k)) where c=conv(p,p).
    Uses only top-k terms for efficiency."""
    n = len(p)
    c = np.convolve(p, p)
    part = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[part]
    max_v = top_vals.max()
    exps = np.exp(tau * (top_vals - max_v))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(top_k):
        k = part[idx]
        wt = w[idx]
        if wt < 1e-15:
            continue
        shift = k - np.arange(n)
        valid = (shift >= 0) & (shift < n)
        grad[valid] += 2.0 * wt * p[shift[valid]]
    return grad


def _variance_grad(p):
    """Gradient of Var(conv(p,p)) = (1/(2n-1))*sum_k(c_k - mu)^2.
    For sum(p)=1, mu = 1/(2n-1)."""
    n = len(p)
    c = np.convolve(p, p)
    mu = 1.0 / (2 * n - 1)
    d = c - mu
    g = np.correlate(d, p, mode='valid')
    return (4.0 / (2 * n - 1)) * g

def _run_gd(p_init, n_iters, tau, top_k, var_wt, alpha, name, deadline, global_best, global_best_score):
    """Run Adam-based gradient descent with simplex projection and averaging."""
    n = len(p_init)
    p = p_init.copy()
    p_avg = p.copy()
    best_local_score = global_best_score
    best_local_p = global_best.copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8

    for t in range(1, n_iters + 1):
        if time.monotonic() > deadline:
            break

        g1 = _smooth_logsumexp_grad(p, tau=tau, top_k=top_k)
        g2 = _variance_grad(p)
        grad = g1 + var_wt * g2

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * grad * grad
        m_hat = m_buf / (1.0 - b1 ** t)
        v_hat = v_buf / (1.0 - b2 ** t)
        lr = alpha / np.sqrt(t + 1.0)
        p_new = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        p = _project_simplex(p_new, 1.0)
