def fft_convolve(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    return c

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    max_c = c.max()
    return 2.0 * n * max_c / (s * s)

import numpy as np
import time

"""
Subgradient and soft-max optimization to minimize 2n*max(self-conv)/sum^2
for non-negative sequences. Uses FFT-based convolution, adaptive step sizes,
momentum, and multiple restarts with structured initializations.
"""

def fft_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    return c

def compute_score(a):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    s = a.sum()
    if n == 0 or s < 0.01:
        return np.inf
    c = fft_convolve(a)
    max_c = c.max()
    return float(2.0 * n * max_c / (s * s))

def get_subgradient(a, peak_tol=0.01, max_peaks=50):
    """Compute subgradient of max_k c[k] w.r.t. a."""
    n = len(a)
    c = fft_convolve(a)
    max_c = c.max()
    
    tol = peak_tol * max_c
    active = np.where(c >= max_c - tol)[0]
    if len(active) > max_peaks:
        idx = np.argsort(c[active])[::-1][:max_peaks]
        active = active[idx]
    
    h_grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            h_grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    
    if len(active) > 0:
        h_grad /= len(active)
    
    return h_grad, max_c, c

        # Momentum
        v = beta_mom * v + (1.0 - beta_mom) * direction
        a_new = a - lr * v
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Check score
        score = compute_score(a_new)
