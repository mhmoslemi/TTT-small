def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0: return v

    cond = (x - (w - 1.0)/rho) > 0

import numpy as np
import time


def _score_key(item):
    return item[0]


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize_sequence(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 1e-18:
        return None
    return f / s


def _resample_log(f, n_out):
    f = _normalize_sequence(f)
    if f is None:
        return np.ones(int(n_out), dtype=np.float64) / float(int(n_out))
    if f.size == int(n_out):
        return f.copy()
    x_old = np.linspace(0.0, 1.0, f.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, int(n_out), dtype=np.float64)
    log_old = np.log(f + 1e-18)
    log_new = np.interp(x_new, x_old, log_old)
    out = np.exp(log_new - float(np.max(log_new)))
    s = float(np.sum(out))
    if s <= 0 or not np.isfinite(s):
        return np.ones(int(n_out), dtype=np.float64) / float(int(n_out))
    return out / s


def _conv_fast(f):
    f = _normalize_sequence(f)
    if f is None:
        return np.zeros(0, dtype=np.float64)
    n = f.size
    m = 2 * n - 1
    L = _next_pow2(m)
    F = np.fft.rfft(f, L)
    q = np.fft.irfft(F * F, L)[:m]
    q = np.asarray(q, dtype=np.float64)
    return np.maximum(q, 0.0)


def _score_fast(f):
    f = _normalize_sequence(f)
    if f is None:
        return np.inf
    q = _conv_fast(f)
    if q.size == 0:
        return np.inf
    qmax = float(np.max(q))
    if not np.isfinite(qmax) or qmax <= 0.0:
        return np.inf
    return float(2.0 * f.size * qmax)


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    x = np.sort(v)[::-1]
    w = np.cumsum(x)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (x - (w - 1.0) / rho) > 0.0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    last = int(np.nonzero(cond)[0][-1])
    rho_val = float(last + 1)
    theta = (w[last] - 1.0) / rho_val
    out = np.maximum(v - theta, 0.0)
    s = float(np.sum(out))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def _hankel_mul(v, W, L2, M, idx):
    F = np.fft.rfft(v, L2)
    c = np.fft.irfft(F * W, L2)
    return c[idx]


def _balance_once(f0, W, L2, M, idx, b, iters, eta, deadline):
    p = np.asarray(f0, dtype=np.float64).copy()
    for _ in range(int(iters)):
        if time.monotonic() > float(deadline):
            break
        u = _hankel_mul(p, W, L2, M, idx)
        if not np.all(np.isfinite(u)):
            break
        c = float(np.mean(u))
        v = _hankel_mul(u, W, L2, M, idx)
        if not np.all(np.isfinite(v)):
            break
        grad = v - c * b
        gmax = float(np.max(np.abs(grad)))
        if not np.isfinite(gmax) or gmax <= 1e-14:
            break
        p = p - (float(eta) / gmax) * grad
        p = _project_simplex(p)
    return p


def _perturb_log(f, rng, amp):
    f = _normalize_sequence(f)
    if f is None:
        return None
    n = f.size
    L = _next_pow2(n)
    half = L // 2
    freq = np.fft.rfftfreq(L) * float(L)
    F = np.zeros(half + 1, dtype=np.complex128)
    if half > 0:
        F[1:] = rng.normal(size=half) + 1j * rng.normal(size=half)
    F[0] = 0.0
    F *= float(amp) * np.exp(-0.5 * (freq / 8.0) ** 2)
    noise = np.fft.irfft(F, L, n)[:n]
    std = float(np.std(noise))
    if std > 1e-12:
        noise = noise / std
    else:
        noise = np.zeros(n, dtype=np.float64)
    logf = np.log(f + 1e-18)
    logf = logf + 0.5 * float(amp) * noise
    out = np.exp(logf - float(np.max(logf)))
    s = float(np.sum(out))
    if s <= 0 or not np.isfinite(s):
        return f.copy()
    return out / s


def _exponential_seq(n, beta):
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    f = np.exp(float(beta) * x)
    return _normalize_sequence(f)


def _active_optimize(f0, time_limit, seed):
    f0 = _normalize_sequence(f0)
    if f0 is None:
        return np.ones(256, dtype=np.float64) / 256.0, 2.0
    rng = np.random.default_rng(int(seed))
    start = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    deadline = start + time_limit
    if time_limit < 1.0:
        return f0.copy(), _score_fast(f0)

    n = f0.size
    M = 2 * n - 1
    L2 = _next_pow2(3 * n - 2)
    idx = M - 1 - np.arange(n, dtype=np.int64)
    ones = np.ones(n, dtype=np.float64)

    best = f0.copy()
    best_sc = _score_fast(best)
    curr = f0.copy()
    curr_sc = best_sc
    stuck = 0
    outer = 0

    while time.monotonic() < deadline - 3.0:
        outer += 1
        q = _conv_fast(curr)
        if q.size == 0:
            break
        qmax = float(np.max(q))
        if not np.isfinite(qmax) or qmax <= 0.0:
            break

        if outer <= 3:
            lam = 4.0
        elif outer <= 8:
            lam = 8.0
        elif outer <= 20:
            lam = 16.0
        elif outer <= 45:
            lam = 32.0
        else:
            lam = float(rng.uniform(8.0, 140.0))

        tol = qmax / lam
        z = (q - qmax) / tol
        z = np.where(z > -60.0, z, -60.0)
        w = np.exp(z)
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.full(M, 1.0 / float(M), dtype=np.float64)
        else:
            w = w / ws

        W = np.fft.rfft(w[::-1], L2)
        b = _hankel_mul(ones, W, L2, M, idx)
        if not np.all(np.isfinite(b)):
            b = ones / float(n)

        improved = False
        for eta in (0.35, 0.18):
            if time.monotonic() > deadline - 5.0:
                break
            pbal = _balance_once(curr, W, L2, M, idx, b, 180, eta, deadline)
            sc_bal = _score_fast(pbal)
            if not np.isfinite(sc_bal):
                continue
            for alpha in (1.0, 0.75, 0.5, 0.25, 0.1):
                cand = alpha * pbal + (1.0 - alpha) * curr
                sc = _score_fast(cand)
                if np.isfinite(sc):
                    if sc < best_sc:
                        best = cand.copy()
                        best_sc = sc
                    if sc < curr_sc * 1.001:
                        curr = cand.copy()
                        curr_sc = sc
                        improved = True

        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck % 5 == 0:
                if rng.random() < 0.25:
                    cand = 0.5 * curr + 0.5 * best
                    cand = _normalize_sequence(cand)
                    if cand is not None:
                        curr = cand
                        curr_sc = _score_fast(curr)
                else:
                    cand = _perturb_log(curr, rng, float(rng.uniform(0.05, 0.30)))
                    if cand is not None:
                        curr = cand
                        curr_sc = _score_fast(curr)
                if not np.isfinite(curr_sc):
                    curr = best.copy()
                    curr_sc = best_sc

        if np.isfinite(best_sc) and np.isfinite(curr_sc) and best_sc < curr_sc * 0.99 and rng.random() < 0.20:
            cand = _resample_log(best, n)
            cand_sc = _score_fast(cand)
            if np.isfinite(cand_sc):
                curr = cand.copy()
                curr_sc = cand_sc

    return best, best_sc


def _gen_initial_candidates(n, seed):
    rng = np.random.default_rng(int(seed))
    cands = []
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    x2 = x * x
    x3 = x2 * x
    x4 = x2 * x2

    try:
        if "height_sequence_1" in globals():
            cands.append(_resample_log(height_sequence_1, n))
    except Exception:
        pass

    for beta in (0.0, 0.5, 0.8, 1.0, 1.2, 1.4, 1.8, 2.5):
        cands.append(_exponential_seq(n, beta))

    for A in np.linspace(2.0, 7.0, 10):
        A = float(A)
        for B in np.linspace(-6.0, 1.0, 13):
            B = float(B)
            cands.append(1.0 + A * x + B * x2)
            for C in (-2.0, -1.0, 0.0, 1.0, 2.0, 3.0):
                cands.append(1.0 + A * x + B * x2 + float(C) * x3)

    for _ in range(420):
        K = int(rng.integers(3, 8))
        amps = rng.normal(0.0, 1.0, K + 1)
        phases = rng.uniform(0.0, 2.0 * np.pi, K + 1)
        th = amps[0] * np.ones(n, dtype=np.float64)
        for k in range(1, K + 1):
            th = th + amps[k] * np.sin(2.0 * np.pi * float(k) * x + phases[k])
        trend = float(rng.normal(0.0, 1.3))
        quad = float(rng.normal(0.0, 1.0))
        th = th + trend * x + quad * x2
        th -= float(np.mean(th))
        th = np.clip(th, -4.0, 4.0)
        cands.append(np.exp(th))

    for _ in range(120):
        m = int(rng.integers(2, 5))
        f = np.zeros(n, dtype=np.float64)
        for _j in range(m):
            c = float(rng.uniform(0.08, 0.92))
            w = float(rng.uniform(0.05, 0.32))
            a = float(rng.uniform(0.15, 1.0))
            f += a * np.exp(-0.5 * ((x - c) / w) ** 2)
        f += 0.03
        cands.append(f)

    return cands


def _score_candidates(cands):
    out = []
    for c in cands:
        f = _normalize_sequence(c)
        if f is None:
            continue
        sc = _score_fast(f)
        if np.isfinite(sc):
            out.append((sc, f))
    out.sort(key=_score_key)
    return out
