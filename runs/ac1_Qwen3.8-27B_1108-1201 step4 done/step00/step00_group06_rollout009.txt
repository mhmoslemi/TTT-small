import numpy as np

def compute_convolution(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    # Need FFT size >= 2n-1
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fa2 = fa * fa
    conv = np.fft.irfft(fa2, N)
    return conv[:2*n-1]

def score(a):

    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    conv = compute_convolution(a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)
