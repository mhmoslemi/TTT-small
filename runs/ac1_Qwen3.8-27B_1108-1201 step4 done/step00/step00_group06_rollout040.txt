```python
    """Solve LP with lazy constraints."""
    # Active constraints (indices k)
    active = [n-1]  # Start with center
    
    g = None

        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0

            j_start = max(0, k - n + 1)
```

```python
def convolve_fast(a, b):

    # Round up to next power of 2
    while size < n + m - 1:

    fa = fft(a, size)
    fb = fft(b, size)
    return np.real(ifft(fa * fb, size))[:n + m - 1]
```

```python
"""
Minimizes 2*n*max(np.convolve(a,a))/(sum(a))^2 for non-negative sequence a.
Uses LP-based alternating method (solving max sum(g) s.t. (a*g)_inf <= M)
combined with multiplicative weights subgradient descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _convolve_full(a, b):
    """Compute full linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:out_len]
    return out

def _normalize(a):
    """Normalize to unit sum, handling edge cases."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _lp_step(f, M, n, active_set, max_constraints=500):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for k in active_set.
    Returns (g, new_active_set).
    """
    if not active_set:
        active_set = [n - 1]
    
    # Cap the active set
    if len(active_set) > max_constraints:
        active_set = sorted(active_set)[:max_constraints]
    
    k_list = sorted(active_set)
    n_active = len(k_list)
    
    # Build constraint matrix A (n_active x n)
    # A[row, j] = f[k - j] if 0 <= k-j < n, else 0
    A = np.zeros((n_active, n), dtype=np.float64)
    for i, k in enumerate(k_list):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # A[i, j_lo..j_hi] = f[k-j_hi .. k-j_lo] (reversed)
            A[i, j_lo:j_hi+1] = f[k - j_hi : k - j_lo + 1][::-1]
    
    b = np.full(n_active, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, 
                  bounds=[(0, None)] * n, method='highs')
    
    if not res.success:
        return None, active_set
    
    g = np.maximum(res.x, 0.0)
    
    # Check all constraints and find new violations
    new_active = set(k_list)
    for k in range(2 * n - 1):
        if k in new_active:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        val = np.dot(f[k - j_hi : k - j_lo + 1][::-1], g[j_lo:j_hi+1])
        if val > M * 1.001:
            new_active.add(k)
    
    return g, list(new_active)

def _lp_optimize(a, n, max_iters=500, deadline=float('inf')):
    """Run LP-based alternating optimization."""
    a = _normalize(a.copy())
    active_set = [n - 1]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_full(a, a)
        M = float(np.max(c))
        
        if M < 1e-15:
            break
        
        g, active_set = _lp_step(a, M, n, active_set)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        g_norm = g / g_sum
        
        # Adaptive step size
        t = 0.15
        a_new = (1.0 - t) * a + t * g_norm
        
        # Verify improvement
        c_new = _convolve_full(a_new, a_new)
        M_new = float(np.max(c_new))
        
        if M_new < M:
            a = a_new
        else:
            # Try smaller step
            t = 0.05
            a_new = (1.0 - t) * a + t * g_norm
            c_new = _convolve_full(a_new, a_new)
            M_new = float(np.max(c_new))
            if M_new < M:
                a = a_new
            else:
                # Try even smaller
                t = 0.02
                a_new = (1.0 - t) * a + t * g_norm
                c_new = _convolve_full(a_new, a_new)
                M_new = float(np.max(c_new))
                if M_new < M:
                    a = a_new
                else:
                    break  # No improvement possible with this LP solution
    
    return a

def _mw_optimize(a, n, max_iters=50000, deadline=float('inf'), eta=0.3):
    """Multiplicative weights subgradient descent."""
    a = _normalize(a.copy())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_full(a, a)
        M = float(np.max(c))
        
        # Find argmax (or top-k for robustness)
        k_star = int(np.argmax(c))
        
        # Gradient of (a*a)[k_star] w.r.t. a
        # grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        grad[j_lo:j_hi+1] = 2.0 * a[k_star - j_hi : k_star - j_lo + 1][::-1]
```

```python
"""
Minimizes 2*n*max(np.convolve(a,a))/(sum(a))^2 for non-negative sequence a.
Uses LP-based alternating method (solving max sum(g) s.t. ||a*g||_inf <= M)
combined with multiplicative weights subgradient descent for refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _convolve_full(a, b):
    """Compute full linear convolution of a and b using FFT (real input/output)."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    size = 1
    while size < out_len:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:out_len]
    return out


def _normalize(a):
    """Normalize to unit sum, handling edge cases."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _lp_solve(f, M, n, active_set):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for k in active_set.
    Returns (g, updated_active_set).
    """
    if not active_set:
        active_set = [n - 1]
    
    # Limit active set size for performance
    max_c = 400
    if len(active_set) > max_c:
        active_set = active_set[:max_c]
    
    k_list = list(active_set)
    n_act = len(k_list)
    
    # Build constraint matrix A (n_act x n)
    A = np.zeros((n_act, n), dtype=np.float64)
    for i, k in enumerate(k_list):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[i, j_lo:j_hi+1] = f[k - j_hi:k - j_lo + 1][::-1]
    
    b = np.full(n_act, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0.0, None)] * n, method='highs')
    
    if not res.success:
        return None, active_set
    
    g = np.maximum(res.x, 0.0)
    
    # Find violated constraints (not in active set)
    conv_val = _convolve_full(f, g)
    violated = np.where((conv_val > M * 1.0001) & (~np.isin(np.arange(2*n-1), k_list)))[0]
    
    new_active = list(active_set) + [int(k) for k in violated[:50]]
    
    return g, new_active


def _lp_optimize(a, n, max_iters=400, deadline=float('inf')):
    """Run LP-based alternating optimization to reduce max convolution peak."""
    a = _normalize(np.maximum(a, 1e-10).astype(np.float64))
    active_set = [n - 1]
    prev_M = float('inf')
    stuck_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_full(a, a)
        M = float(np.max(c))
        
        if M >= prev_M - 1e-12:
            stuck_count += 1
        else:
            stuck_count = 0
        
        if stuck_count > 20:
            break
        prev_M = M
        
        if M < 1e-10:
            break
        
        g, active_set = _lp_solve(a, M, n, active_set)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        g_norm = g / g_sum
        
        # Try multiple step sizes
        improved = False
        for t in [0.2, 0.1, 0.05, 0.02]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 1e-10)
            c_new = _convolve_full(a_new, a_new)
            M_new = float(np.max(c_new))
            if M_new < M * 0.999:
                a = a_new
                improved = True
                break
        if not improved:
            break
    
    return a


def _mw_optimize(a, n, max_iters=50000, deadline=float('inf'), eta_scale=1.0):
    """Multiplicative weights subgradient descent for fine-tuning."""
    a = _normalize(np.maximum(a, 1e-10).astype(np.float64))
    best_a = a.copy()
    best_M = float(np.max(_convolve_full(a, a)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_full(a, a)
        M = float(np.max(c))
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # Use top-3 lags for gradient (avoids oscillation from hard argmax)
        top_indices = np.argsort(c)[-3:]
        grad = np.zeros(n, dtype=np.float64)
        for k in top_indices:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]
        grad /= 3.0
        
        # Adaptive step size
        max_grad = float(np.max(grad))
        if max_grad < 1e-15:
            break
        eta = eta_scale / max_grad
        
        # Multiplicative update
        a *= np.exp(-eta * grad)
        a = _normalize(a)
    
    return best_a


def _random_local_search(a, n, max_iters=5000, deadline=float('inf'), rng=None):
    """Simple randomized local search as a polishing step."""
    if rng is None:
        rng = np.random.default_rng()
    
    a = _normalize(np.maximum(a, 0.0).astype(np.float64))
    best_a = a.copy()
    best_M = float(np.max(_convolve_full(a, a)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Block perturbation

        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.1, width))
        candidate = _normalize(np.maximum(candidate, 0.0))
        
        c = _convolve_full(candidate, candidate)
        M = float(np.max(c))
        
        if M < best_M:
            best_M = M
            best_a = candidate
    
    return best_a
```
