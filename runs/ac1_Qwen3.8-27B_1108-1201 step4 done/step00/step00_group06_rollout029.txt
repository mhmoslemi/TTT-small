        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            delta = frac * c[s_max]
            if delta >= c[s_max]:
                continue
            c_new = c.copy()
            c_new[s_max] -= delta
            c_new[s_min] += delta
            b_new = compute_conv_sparse(positions, c_new)
            max_new = b_new.max()
            if max_new < best_score:
                best_score = max_new
                best_delta = delta
                c_trial = c_new.copy()
        
        if best_delta > 0:
            c = c_trial
        else:
            # Try smaller fractions
            for frac in [0.001, 0.005, 0.01]:
                delta = frac * c[s_max]
                c_new = c.copy()
                c_new[s_max] -= delta
                c_new[s_min] += delta
                b_new = compute_conv_sparse(positions, c_new)
                max_new = b_new.max()
                if max_new < max_b:
                    c = c_new
                    break
            else:
                # No improvement possible with this pair; break
                if best_max_b - max_b < tol:
                    break
    
    return best_c, best_max_b

A Golomb ruler of order k is a set of k marks such that all pairwise differences are distinct. A Sidon set (B_2 set) is a set where all pairwise sums are distinct (or at most 2 representations: one from the pair itself and one from the reversed pair).
