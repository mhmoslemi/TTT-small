def solve_lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    g = cp.Variable(n)
    
    # Build constraint matrix (sparse Toeplitz)
    # (f*g)_k = sum_j f_j * g_{k-j} for valid indices
    # We need: (f*g)_k <= M for all k = 0, ..., 2n-2
    
    # Build the constraint: T @ g <= M * ones
    # T is (2n-1) x n, T[k, j] = f_j if 0 <= k-j < n, else 0
    # i.e., T[k, j] = f[j] if k-n+1 <= j <= k, else 0
    
    # Build as sparse matrix
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [T @ g <= M]
    constraints.append(g >= 0)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None
    
    if g.value is None:
        return None
    
    return np.clip(g.value, 0, 1000)

def build_partial_matrix(f, positions):
    """Build sparse constraint matrix for selected convolution positions."""
    n = len(f)
    k = len(positions)
    if k == 0:
        return sp.csr_matrix((0, n))
    
    pos_arr = np.array(positions, dtype=np.int64)
    j_starts = np.maximum(0, pos_arr - n + 1).astype(np.int32)
    j_ends = (np.minimum(pos_arr, n - 1) + 1).astype(np.int32)
    
    row_counts = (j_ends - j_starts).astype(np.int32)
    total = int(row_counts.sum())
    
    rows = np.repeat(np.arange(k, dtype=np.int32), row_counts)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    
    idx = 0
    for i in range(k):
        js = j_starts[i]
        je = j_ends[i]
        cnt = je - js
        cols[idx:idx+cnt] = np.arange(js, je)
        vals[idx:idx+cnt] = f[js:je]
        idx += cnt
    
    return sp.csr_matrix((vals, (rows, cols)), shape=(k, n))
