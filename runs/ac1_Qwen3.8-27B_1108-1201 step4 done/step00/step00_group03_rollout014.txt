def _fft_convolve_self(a, N):
    """Compute full self-convolution of a using FFT. N >= 2*len(a)-1."""
    n = len(a)
    A = np.fft.rfft(a, N)
    B = A * A
    return np.fft.irfft(B, N)[:2*n-1]

def _fft_cross_correlate(x, y, N):
    """Compute cross-correlation c_j = sum_k x[k]*y[k-j] for j=0..len(y)-1.
    N >= len(x) + len(y) - 1."""
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)

def _compute_b(a, N):
    """Compute self-convolution using FFT."""
    n = len(a)
    A = np.fft.rfft(a, N)
    B = A * A
    return np.fft.irfft(B, N)[:2*n-1]

def _cross_corr(x, y, N):
    """Cross-correlation: c[j] = sum_k x[k]*y[k-j] for j=0..len(y)-1."""
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    return np.fft.irfft(X * np.conj(Y), N)[:len(y)]

def _pnorm_grad(a, p, N, N2):

    n = len(a)
