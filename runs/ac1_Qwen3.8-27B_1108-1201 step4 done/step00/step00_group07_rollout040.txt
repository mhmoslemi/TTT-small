```python
def solve_lp_first_order(f, M, n, n_iters=100, lr=0.1, rng=None):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0 using multiplicative weights."""
    g = np.ones(n) / n
    for _ in range(n_iters):
        # Compute F @ g = convolve(f, g)
        r = fftconvolve(f, g, mode='full')
        # Find violations
        violations = np.maximum(r - M, 0)
        if violations.max() < 1e-10:
            break
        # Update g
        # For each j, the "score" is the sum of violations of constraints involving g[j]
        # Constraint k involves g[j] if 0 <= k-j < n, i.e., j <= k < j+n
        # score[j] = sum_{k: j <= k < j+n} violations[k]
        # This is a "sliding window sum" of violations
        # violations has length 2n-1, and we need window sums of length n
        # score[j] = sum_{k=j}^{j+n-1} violations[k] (with appropriate bounds)
        # This is: cumsum approach
        cumsum = np.zeros(2*n)
        cumsum[1:] = np.cumsum(violations)
        # score[j] = cumsum[j+n] - cumsum[j] for j = 0, ..., n-1
        # But violations only has 2n-1 elements (indices 0 to 2n-2)
        # So cumsum has 2n elements (indices 0 to 2n-1)
        # score[j] = cumsum[j+n] - cumsum[j] for j = 0, ..., n-1
        # j+n ranges from n to 2n-1, which is valid for cumsum
        scores = cumsum[n:2*n] - cumsum[0:n]
        # Update
        g = g * np.exp(-lr * scores / max(M, 1e-10))
        g = np.clip(g, 0, None)
        s = g.sum()
        if s < 1e-10:
            g = np.ones(n) / n
        else:
            g = g / s
    return g
```
