"""
Low-peak auto-convolution search.

Objective: minimize  2*n*max(conv(a,a)) / (sum(a))^2  over non-negative a.
(= score returned by evaluate_sequence).

Approach
--------
1) Build a set of cheap parametric "seed" sequences (uniform, two-block,
   monotone ramps / power ramps and their reverses) at a fixed working length
   and keep the best evaluated one.
2) Run an iterative refinement based on the linear-programming off/on idea:
   given current f with M = max(conv(f,f)), solve the LP
       maximize   sum(g)
       subject to g >= 0,  (f * g)[l] <= M   for every lag l
   (a sparse convolution/Hankel constraint matrix), rescale g to match the
   mass of f, and blend  f <- (1-t) f + t g .  Re-accept only when the exact
   evaluator improves the incumbent.  A few cheap block perturbations are
   interleaved to polish.  Everything is time-guarded and falls back to the
   best seed if anything is too slow.
"""

import time
import math

import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# Helpers (all top level, no closures / lambdas)

def _clean_seq(x):
    """Return a finite, non-negative float array (clip to [0, 1000])."""
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.nan_to_num(x, nan=0.0, posinf=1000.0, neginf=0.0)
    x = np.clip(x, 0.0, 1000.0)
    return x


def _safe_eval(seq):
    """Call evaluate_sequence, returning +inf on any error."""
    try:
        s = float(evaluate_sequence([float(v) for v in seq]))
        return s
    except Exception:
        return float("inf")


def _pick_length(base):
    """Choose a working length that keeps the LP sparse matrix bounded."""
    try:
        n = int(np.asarray(base).size)
    except Exception:
        n = 2048
    if n < 512:
        n = 1024
    if n > 2048:
        n = 2048
    return n


def _seed_uniform(n):
    return np.ones(n, dtype=np.float64)


def _seed_twoblock(n, r=2.0, frac=0.5):
    m = max(1, int(round(n * frac)))
    m = min(m, n - 1)
    a = np.empty(n, dtype=np.float64)
    a[:m] = r
    a[m:] = 1.0
    return a


def _seed_ramp(n, p=1.0):
    """Monotone decreasing power ramp: a_i = (n-1-i)**p, last entry -> 0."""
    idx = np.arange(n, dtype=np.float64)
    v = (n - 1 - idx)
    v[-1] = 0.0
    a = v ** p
    return a


def _seed_families(n):
    """Yield a list of candidate seed arrays of length n."""
    fam = []
    fam.append(_seed_uniform(n))
    fam.append(_seed_twoblock(n, r=2.0, frac=0.5))
    fam.append(_seed_twoblock(n, r=3.0, frac=0.5))
    for p in (1.0, 1.5, 2.0, 3.0, 4.0):
        a = _seed_ramp(n, p)
        fam.append(a)
        fam.append(a[::-1].copy())
    # a ramp with a small floor to test robustness
    a = _seed_ramp(n, 2.0)
    a = a + 0.1
    fam.append(a)
    return fam


def _conv_max(a):
    """max of full linear convolution a*a via FFT (fast, n up to ~4096)."""
    n = a.size
    size = 1
    m = 2 * n - 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return float(np.max(c[:m]))


def _normalize_mass(a, target):
    s = float(np.sum(a))
    if s <= 0.0:
        return None
    return a * (target / s)


def _lp_refine_step(f, deadline):
    """
    One LP refinement on f (length n). Returns a blended candidate or None.
    Maximizes sum(g) s.t. g>=0 and (f*g)[l] <= M for all lags.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n < 2:
        return None
    M = _conv_max(f)
    if not np.isfinite(M) or M <= 0.0:
        return None
    target = float(np.sum(f))

    # Build sparse Hankel/convolution matrix A with A[l, j] = f[l-j]
    # when 0 <= l-j < n else 0.  Each super-diagonal (offset k) is constant f[k].
    diag_len = 2 * n - 1
    diagonals = []
    offsets = []
    for k in range(n):
        # offset k -> entry (row, col) with row - col = k
        length = min(diag_len - k, n)
        if length <= 0:
            continue
        d = np.full(length, float(f[k]), dtype=np.float64)
        diagonals.append(d)
        offsets.append(k)
    A = sparse.diags(diagonals, offsets, shape=(diag_len, n), format="csr")
    b_ub = np.full(diag_len, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if res is None or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    g = _normalize_mass(g, target)
    if g is None:
        return None

    # Try a couple of blend factors, accept the best that improves f.
    base_score = float(evaluate_sequence([float(v) for v in f]))
    best = None
    best_s = base_score
    for t in (1.0, 0.7, 0.4):
        cand = (1.0 - t) * f + t * g
        cand = np.clip(cand, 0.0, 1000.0)
        s = _safe_eval(cand)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best = cand
    # respect deadline on caller side; just return the improved one (or None)
    if best is not None:
        return best
    # fall back: even if evaluation of blend is unavailable, return g-blend for caller to score
    cand = 0.5 * (f + g)
    if time.monotonic() >= deadline:
        return None
    return None


def _block_perturb(base, rng, n):
    """Cheap non-negative local perturbation of `base` (length n)."""
    cand = base.copy()
    width = max(4, int(n // 40))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.15)))
    if rng.random() < 0.5:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
