def compute_conv(a):

    n = len(a)

        a, score = gradient_descent_smooth(
            init, alpha, lr, max_iters=2000,
            time_limit=remaining, normalize_to=target_sum)
        candidates.append((score, n, a.copy()))
        update_best(a, score, f"GD {name} n={n}")

    # ===== PHASE 2: LP Refinement =====
    print(f"=== Phase 2: LP Refinement (budget {phase2_time:.0f}s) ===")
    candidates.sort(key=lambda x: x[0])
    top_k = min(4, len(candidates))
    phase2_deadline = max(start_time + phase1_time, time.monotonic()) + phase2_time

    for idx in range(top_k):
        if time.monotonic() > phase2_deadline:
            break
        score, n, a = candidates[idx]
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"  Refining candidate {idx} (n={n}, init_score={score:.6f})")
        a_ref, score_ref = lp_refine(a, max_iters=40, time_limit=remaining * 0.9)
        if score_ref < best_overall_score:
            update_best(a_ref, score_ref, f"LP cand{idx} n={n}")
        # Also update in candidates list
        candidates[idx] = (score_ref, n, a_ref.copy())

    # ===== PHASE 3: Alternating GD + LP with perturbation =====
    print(f"=== Phase 3: Alternating Polish (budget {phase3_time:.0f}s) ===")
    if best_overall is not None and time.monotonic() < _GLOBAL_DEADLINE:
        phase3_deadline = min(time.monotonic() + phase3_time, _GLOBAL_DEADLINE)
        a = best_overall.copy()
        n = len(a)
        target_sum = np.sqrt(2.0 * n)

        for round_idx in range(6):
            if time.monotonic() > phase3_deadline:
                break
            remaining = phase3_deadline - time.monotonic()
            if remaining < 20:
                break

            # Small random perturbation to escape local minima
            if round_idx > 0 and round_idx % 2 == 0:
                pert = 0.05 * rng.standard_normal(n)
                a_pert = np.maximum(0, a + pert)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert * (target_sum / s_pert)
                    score_pert = objective(a_pert)
                    if score_pert < best_overall_score * 1.01:
                        a = a_pert

            # GD step
            gd_time = min(remaining * 0.4, 60)
            a_gd, score_gd = gradient_descent_smooth(
                a, alpha, lr * 2, max_iters=500,
                time_limit=gd_time, normalize_to=target_sum)
            if score_gd < best_overall_score:
                update_best(a_gd, score_gd, f"GD-P3 r{round_idx}")
                a = a_gd.copy()

            # LP step
            if time.monotonic() < phase3_deadline:
                lp_time = min(phase3_deadline - time.monotonic(), 60)
                if lp_time > 20:
                    a_lp, score_lp = lp_refine(a, max_iters=15, time_limit=lp_time)
                    if score_lp < best_overall_score:
                        update_best(a_lp, score_lp, f"LP-P3 r{round_idx}")
                        a = a_lp.copy()

    # ===== Final validation =====
    print(f"=== Final Validation ===")
    elapsed = time.monotonic() - start_time
    if best_overall is not None:
        final_score = try_exact(best_overall)
        print(f"  Exact score: {final_score:.6f} (elapsed {elapsed:.1f}s)")
        if not np.isfinite(final_score) or final_score > best_overall_score * 1.01:
            # Fallback
            if given_seq is not None:
                fb_score = try_exact(given_seq)
                if np.isfinite(fb_score):
                    best_overall = given_seq.copy()
                    best_overall_score = fb_score
                    print(f"  Fallback to given: score={fb_score:.6f}")
                else:
                    best_overall = np.ones(1000)
                    best_overall_score = try_exact(best_overall)
            else:
                best_overall = np.ones(1000)
                best_overall_score = try_exact(best_overall)
        else:
            best_overall_score = final_score
    else:
        if given_seq is not None:
            best_overall = given_seq.copy()
        else:
            best_overall = np.ones(2000)
        best_overall_score = try_exact(best_overall)

    # Final safety check
    best_overall = np.maximum(0, best_overall)
    best_overall = np.minimum(1000.0, best_overall)
    final_check = try_exact(best_overall)
    if not np.isfinite(final_check):
        best_overall = np.ones(1000)
        final_check = try_exact(best_overall)

    print(f"  FINAL: n={len(best_overall)}, score={final_check:.6f}, elapsed={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in best_overall]
