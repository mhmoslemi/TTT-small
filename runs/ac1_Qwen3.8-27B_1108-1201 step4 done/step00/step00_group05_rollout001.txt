import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except:
        base = np.ones(4498, dtype=np.float64)
    
    n = len(base)
    base = np.clip(base, 0, 1000)
    S = base.sum()
    if S < 1e-10:
        base = np.ones(n)
        S = n
    base = base / S  # normalize to sum = 1
    
    best = base.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial score: {best_score}")
    
    deadline = time.monotonic() + budget_s - 10
    
    f = best.copy()
    
    while time.monotonic() < deadline:
        # Compute M
        g = np.convolve(f, f, mode='full')
        M = max(g)
        
        # Solve LP: max sum(b) s.t. A @ b <= M, b >= 0
        # A is (2n-1) x n matrix where A[k, j] = f(k-j) if 0 <= k-j < n, else 0
        # For efficiency, build A as a sparse matrix
        
        from scipy import sparse
        
        # Build A
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            # (f*b)(k) = sum_j f(k-j) * b(j) for j where 0 <= k-j < n
            # i.e., j in [max(0, k-n+1), min(n-1, k)]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
        
        A = sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
        
        # LP: max c^T b s.t. A b <= M, b >= 0
        # linprog minimizes, so we negate: min -c^T b s.t. A b <= M, b >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            break
        
        b = res.x
        sum_b = b.sum()
        
        if sum_b < 1e-10:
            break
        
        # Scale b to have sum = 1 (same as f)
        b = b / sum_b
        
        # Update
        t = 0.5  # step size
        f_new = (1 - t) * f + t * b
        f_new = np.clip(f_new, 0, None)

            best_score = score

        f = f_new
