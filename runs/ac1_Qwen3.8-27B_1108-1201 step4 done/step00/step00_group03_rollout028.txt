    Multi-start coordinate descent to minimize 2n*max(conv(a,a))/sum(a)^2.
    Tries the given starting sequence and several alternatives.
    """
    total_start = time.monotonic()
    total_deadline = total_start + max(0, budget_s - 5.0)
    
    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    
    parent = np.clip(parent, 0.0, 1000.0)
    
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(parent.tolist()))
    print(f"Initial score: {best_score:.6f}, length={len(best_seq)}")
    
    # Strategy 1: Coordinate descent on the original sequence
    time_per_attempt = min(600.0, (total_deadline - time.monotonic()) * 0.5)
    if time.monotonic() < total_deadline:
        print(f"Running coordinate descent on original (length {len(parent)})...")
        t0 = time.monotonic()
        seq, score = run_coordinate_descent(parent, time_per_attempt, seed=seed)
        elapsed = time.monotonic() - t0
        # Verify with the actual evaluator
        real_score = float(evaluate_sequence(seq.tolist()))
        if real_score < best_score:
            best_score = real_score
            best_seq = seq.copy()
        print(f"  CD on original: score={real_score:.6f} (time={elapsed:.1f}s)")
    
    # Strategy 2: Try a symmetric version
    if time.monotonic() < total_deadline:
        n = len(parent)
        sym = 0.5 * (parent + parent[::-1])
        sym = np.clip(sym, 0, 1000)
        t0 = time.monotonic()
        seq, score = run_coordinate_descent(sym, min(300.0, (total_deadline - time.monotonic()) * 0.3), seed=seed+1)
        real_score = float(evaluate_sequence(seq.tolist()))
        if real_score < best_score:
            best_score = real_score
            best_seq = seq.copy()
        print(f"  CD on symmetric: score={real_score:.6f} (time={time.monotonic()-t0:.1f}s)")
    
    # Strategy 3: Try a random non-negative sequence of same length
    if time.monotonic() < total_deadline:
        rng = np.random.default_rng(seed + 100)
        rand_seq = rng.exponential(scale=1.0, size=n)
        t0 = time.monotonic()
        seq, score = run_coordinate_descent(rand_seq, min(300.0, (total_deadline - time.monotonic()) * 0.3), seed=seed+2)
        real_score = float(evaluate_sequence(seq.tolist()))
        if real_score < best_score:
            best_score = real_score
            best_seq = seq.copy()
        print(f"  CD on random exp: score={real_score:.6f} (time={time.monotonic()-t0:.1f}s)")
    
    # Strategy 4: Try a different length (shorter might be easier to optimize)
    if time.monotonic() < total_deadline:
        for try_n in [512, 1024]:
            if time.monotonic() >= total_deadline:
                break
            rng = np.random.default_rng(seed + try_n)
            # Start with a slightly structured random sequence
            init = rng.exponential(scale=1.0, size=try_n)
            # Make it slightly symmetric
            init = 0.7 * init + 0.3 * init[::-1]
            t0 = time.monotonic()
            remaining = (total_deadline - time.monotonic()) * 0.25
            seq, score = run_coordinate_descent(init, remaining, seed=seed + try_n)
            real_score = float(evaluate_sequence(seq.tolist()))
            if real_score < best_score:
                best_score = real_score
                best_seq = seq.copy()
            print(f"  CD on random n={try_n}: score={real_score:.6f} (time={time.monotonic()-t0:.1f}s)")
    
    # Strategy 5: Continue coordinate descent on the best found so far
    if time.monotonic() < total_deadline:
        remaining = total_deadline - time.monotonic()
        if remaining > 30:
            t0 = time.monotonic()
            seq, score = run_coordinate_descent(best_seq, remaining, seed=seed + 999)
            real_score = float(evaluate_sequence(seq.tolist()))
            if real_score < best_score:
                best_score = real_score
                best_seq = seq.copy()
            print(f"  Final CD refinement: score={real_score:.6f} (time={time.monotonic()-t0:.1f}s)")
    
    print(f"\nFinal best score: {best_score:.6f} (length={len(best_seq)})")
    return [float(x) for x in best_seq]
