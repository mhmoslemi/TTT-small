import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} (water-filling)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    idx = np.where(cond)[0]
    if idx.size == 0:
        rho = n - 1
    else:
        rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    w = w / s
    return w


def fast_obj(a, n):
    """Objective 2*n*max(conv(a,a)) assuming sum(a)=1 (scale-invariant form)."""
    b = np.convolve(a, a)
    return 2.0 * n * float(b.max())


def topk_grad(b, a, n, k):
    """Smoothed subgradient: mean over top-k peak positions of 2*a[k-i] contribution."""
    k = int(min(len(b), max(1, k)))
    order = np.argsort(b)[::-1][:k]
    grad = np.zeros(n, dtype=np.float64)
    for pos in order:
        pos = int(pos)
        i_min = pos - (n - 1)
        if i_min < 0:
            i_min = 0
        i_max = pos
        if i_max > n - 1:
            i_max = n - 1
        if i_max >= i_min:
            lo = pos - i_max
            hi = pos - i_min
            grad[i_min:i_max + 1] += 2.0 * a[lo:hi + 1][::-1]
    grad /= float(k)
    return grad


def make_init(n, kind, p, rng):
    """Build a raw (unnormalized) non-negative increasing-ish sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    if kind == "pow":
        a = np.power(x, float(p))
    elif kind == "sin":
        if n > 1:
            t = x / float(n - 1)
        else:
            t = x
        a = np.power(np.sin(np.pi * t / 2.0), float(p))
    elif kind == "hybrid":
        if n > 1:
            t = x / float(n - 1)
        else:
            t = x
        a = np.power(x, float(p)) * (0.6 + 0.4 * np.cos(np.pi * t))
        a = a / (0.6 + 0.0)
    else:
        a = x.copy()
    a = np.maximum(a, 0.0)
    if not np.all(np.isfinite(a)) or a.sum() <= 1e-300:
        a = x.copy()
    return a


def block_perturb(a, rng):
    """Random positive multiplicative block scaling (helps escape local minima)."""
    c = a.copy()
    n = c.size
    lo = max(1, n // 300)
    hi = max(lo + 1, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, 0.18)))
    if rng.random() < 0.3:
        c *= float(np.exp(rng.normal(0.0, 0.08)))
    return c

    a = project_simplex(a0.copy())

    step_base = 1.0
    no_improve = 0
    k = int(min(256, max(16, n // 24)))
    max_step = 5.0

        score = cur

        # Try a few step sizes (backtracking).

                  0.15 * step_base / gnorm,
                  0.05 * step_base / gnorm):
            if s <= 0.0:
                continue
            a_new = project_simplex(a - s * grad)
            sc = fast_obj(a_new, n)
            if sc < score - 1e-10:
                a = a_new
                score = sc
                improved = True
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
                break

        if improved:
            no_improve = 0
            step_base = min(step_base * 1.08, max_step)
        else:
            no_improve += 1
            step_base = step_base * 0.7
            if no_improve > 35:
                # Escalate: random-ish structured perturbation to escape.
                a = project_simplex(block_perturb(a, rng))
                score = fast_obj(a, n)
                if score < best_score:
                    best_score = score
                    best = a.copy()
                no_improve = 0
                step_base = 1.0

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Minimise 2n*max(conv(a,a))/sum(a)^2 via projected gradient minimax search.

    Starts from a family of asymmetric increasing sequences (power/sin ramps) whose
    self-convolution is far flatter than a box (objective ~4/3 for large n), then
    runs projected gradient descent on the top-peak proxy with line search and
    structured perturbations to escape local minima.  Returns the best list found.
    """
    budget_s = float(budget_s) if budget_s is not None else 1800.0
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Choose a working length: prefer the provided construction length if sane.
    n = 2535
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if h.ndim == 1 and 256 <= h.size <= 6000 and np.all(np.isfinite(h)) and h.sum() > 0:
            n = int(h.size)
    except Exception:
        n = 2535
    n = int(min(max(n, 256), 6000))

    # (kind, p) initialisations to explore.
    inits = [
        ("pow", 1.0),
        ("pow", 0.85),
        ("pow", 1.15),
        ("pow", 0.70),
        ("sin", 1.5),
        ("hybrid", 1.0),
    ]

    best_a = None
    best_score = np.inf

    n_init = len(inits)
    per_init_budget = max(20.0, (budget_s - 30.0) / max(1, n_init))

    for (kind, p) in inits:
        if time.monotonic() >= global_deadline:
            break
        raw = make_init(n, kind, p, rng)
        a0 = project_simplex(raw)
        stop = min(global_deadline, time.monotonic() + per_init_budget)
        label = f"{kind}/{p}"
        a_opt, sc = optimize_local(a0, n, stop, global_deadline, rng, label, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"[*] new global best ({label}) obj={best_score:.6f}")

    # Seed-refinement using the provided construction as an extra start point.
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if (h.ndim == 1 and h.size == n and np.all(np.isfinite(h)) and h.sum() > 0
                and time.monotonic() < global_deadline):
            a_h = project_simplex(np.maximum(h, 0.0))
            stop = min(global_deadline, time.monotonic() + max(20.0, per_init_budget))
            a_opt, sc = optimize_local(a_h, n, stop, global_deadline, rng, "h1", verbose=True)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"[*] new global best (h1) obj={best_score:.6f}")
    except Exception:
        pass

    # Final polish with whatever time remains, starting from the global best.
    if best_a is not None and time.monotonic() < global_deadline - 5.0:
        stop = global_deadline - 2.0
        a_opt, sc = optimize_local(best_a, n, stop, global_deadline, rng, "polish", verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"[*] polish improved obj={best_score:.6f}")

    if best_a is None:
        best_a = np.ones(n, dtype=np.float64) / n

    result = np.clip(best_a, 0.0, 1000.0)
    result = project_simplex(result)  # keep sum=1 and non-negative exactly

    try:
        true_score = float(evaluate_sequence([float(x) for x in result]))  # noqa: F821
        print(f"[+] final evaluate_sequence = {true_score:.6f}  (n={n})")
    except Exception:
        print(f"[+] final proxy obj = {best_score:.6f}  (n={n})")

    return [float(x) for x in result]
