Minimize  2 n max_k (s*s)_k / (sum s)^2  by pure projected gradient flow in log-space.
Periodic coordinate-descent polishes break out of saddle points of the nonconvex max.
"""
import numpy as np
import time


def _gstep(s, k0, K, lr):
    """One projected gradient step in log-space for nonnegative s (sum fixed)."""
    n = len(s)
    conv = np.convolve(s, s, mode='full')
    S = float(np.sum(s))
    if S <= 0.0:
        return s, np.inf, 0
    m2 = conv.size
    kk = np.arange(m2)
    top = np.argpartition(conv, -(K - 1))[-K:]
    M = float(np.max(conv[top]))
    if M <= 0.0:
        return s, np.inf, 0
    T = np.zeros(n, dtype=np.float64)

def _score(s):
    n = len(s)
    conv = np.convolve(s, s, mode='full')
    S = float(np.sum(s))
    if S <= 0.0:
        return np.inf
    return float(2.0 * n * float(np.max(conv)) / (S * S))


def _coord_polish(s, deadline, rng, K, iters):
    """Local coordinate descent in log-space: burn the top active peaks one lag at a time."""
    n = len(s)
    best = s.copy()
    best_sc = _score(best)
    for _ in range(iters):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(best, best, mode='full')
        S = float(np.sum(best))
        M = float(np.max(conv))
        order = np.argpartition(-conv, topk if (topk := K) < conv.size else conv.size - 1)
        for k in order[:K]:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi - lo < 2:
                continue
            idx = np.arange(lo, hi)
            a = best[idx].copy()
            # 1-D problem: pick a'>=0, sum=a.sum(), minimize (a*b_k)^2 where b_k = s[k-i]
            b = best[k - hi + 1:k - lo + 1][::-1]
            # optimal a_i proportional to b_i but with sum fixed and clipped at 0
            ratio = a.sum() / max(1e-300, float(np.sum(b)))
            newa = b * ratio
            # keep the same support as before to avoid churn
            newa[a <= 0] = 0.0
            ns = float(np.sum(newa))
            if ns <= 0.0:
                continue
            newa = newa / ns
            cand = best.copy()
            cand[idx] = newa
            sc = _score(cand)
            if sc < best_sc - 1e-12:
                best = cand
                best_sc = sc
    return best, best_sc
