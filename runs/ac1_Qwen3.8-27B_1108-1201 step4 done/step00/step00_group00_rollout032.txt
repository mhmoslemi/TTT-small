from scipy.fft import fft, ifft
import time

def fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n = len(a) + len(b) - 1
    fa = fft(a, n)
    fb = fft(b, n)
    return np.real(ifft(fa * fb))

def softmax(x, T):
    """Stable softmax with temperature."""
    x = x / T
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def compute_score(a):
    """Compute the exact score."""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return 2 * n * max_b / (np.sum(a)**2)

Gradient descent with log-sum-exp smoothing to minimize the peak-to-sum ratio
of the autocorrelation of a non-negative sequence. Uses FFT for efficient
convolution and geometric temperature annealing for convergence.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    # Round up to next power of 2 for efficiency
    n_fft = 1
    while n_fft < n:
        n_fft *= 2
    fa = fft(a, n_fft)
    fb = fft(b, n_fft)
    result = np.real(ifft(fa * fb))
    return result[:n]


def stable_logsumexp(x, T):
    """Compute T * log(sum(exp(x/T))) stably, and the softmax weights."""
    x_scaled = x / T
    x_max = np.max(x_scaled)
    x_shifted = x_scaled - x_max
    e = np.exp(x_shifted)
    s = np.sum(e)
    p = e / s  # softmax weights
    log_sum_exp = T * (x_max + np.log(s))
    return log_sum_exp, p


def compute_exact_score(a):
    """Compute the exact score (matching evaluate_sequence)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return float(2 * n * max_b / (s * s))
