import numpy as np
import time

def compute_obj(a):
    """Fast computation of 2*n*max(conv(a,a))/(sum(a))^2 using FFT."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # FFT-based full convolution
    N = 2 * n - 1
    Nfft = 1 << (N - 1).bit_length()  # next power of 2
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A * A, Nfft)[:N]
    return 2.0 * n * conv.max() / (s * s)

def get_gradient(a, k_positions, weights):
    """Compute the subgradient of max conv at the given positions.
    
    grad[j] = 2 * sum_m weights[m] * a[k_m - j] (if 0 <= k_m - j < n)
    """
    n = a.shape[0]
    grad = np.zeros(n)
    for k, w in zip(k_positions, weights):
        # We need a[k-j] for j = 0, ..., n-1
        # k-j is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # grad[j] += 2*w*a[k-j] for j in [j_min, j_max]
            # k-j ranges from k-j_min to k-j_max
            indices = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max+1] += 2.0 * w * a[indices]
    return grad

def simplex_project_simple(v):
    """Project v onto {x >= 0, sum(x) = 1} using clip + normalize."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s == 0:
        return np.ones(len(v)) / len(v)
    return v / s

def simplex_project_exact(v):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
on the simplex, with multiple initializations and adaptive step sizes.
Exploits the structure of the convolution gradient (reversed/shifted a)
for O(n) gradient computation and FFT-based O(n log n) objective evaluation.
"""

import numpy as np
import time

def compute_obj(a):
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    N = 2 * n - 1
    Nfft = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A * A, Nfft)[:N]
    return 2.0 * n * conv.max() / (s * s)

def get_subgradient(a, k_positions, weights):
    n = a.shape[0]
    grad = np.zeros(n)
    for k, w in zip(k_positions, weights):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            indices = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max+1] += 2.0 * w * a[indices]
    return grad

def simplex_project(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def run_pgd(a, max_iterations, deadline, rng, verbose=False):
    """Run projected gradient descent from initial point a (on the simplex)."""
    n = a.shape[0]
    a = a.copy()
    a = simplex_project(a)
    
    best_a = a.copy()
    best_score = compute_obj(a)
    
    lr = 0.5
    no_improve = 0
    top_k = 5
    
    for it in range(max_iterations):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution and find peak positions
        N = 2 * n - 1
        Nfft = 1 << (N - 1).bit_length()
        A = np.fft.rfft(a, Nfft)
        conv = np.fft.irfft(A * A, Nfft)[:N]
        
        s = a.sum()
        score = 2.0 * n * conv.max() / (s * s)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500 and it > 100:
            # Restart with perturbation
            a = simplex_project(a + rng.normal(0, 0.01, n))
            lr = 0.5
            no_improve = 0
            continue
        
        # Find top positions
        if top_k < N:
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            top_vals = conv[top_idx]
            # Weights: proportional to how close to max
            max_val = conv.max()
            weights = top_vals / (max_val * top_k)
            weights = weights / weights.sum()
        else:
            top_idx = [N // 2]
            weights = [1.0]
        
        # Compute subgradient
        grad = get_subgradient(a, top_idx, weights)
        
        # Gradient step
        a = a - lr * grad
        a = simplex_project(a)
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
        elif no_improve == 0:
            lr = min(lr * 1.05, 2.0)
    
    return best_a, best_score
