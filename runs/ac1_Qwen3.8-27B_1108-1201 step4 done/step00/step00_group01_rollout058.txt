"""
Multi-start projected gradient optimization for minimizing the step function
evaluation score. Uses log-space parameterization, FFT-based convolution,
soft-max smoothing for the peak, and Adam optimizer with adaptive step size.
"""

import numpy as np
import time

def fft_autocorr(a):
    """Compute a * a (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:size]
    # Clip tiny negative values from FFT roundoff
    c = np.maximum(c, 0)
    return c

def loss_and_grad_logspace(x, beta=50.0):
    """
    Compute loss and gradient w.r.t. x where a = exp(x).
    Loss = 2n * soft_max(conv(a)) / (sum(a))^2
    """
    a = np.exp(x)
    n = len(a)
    s = np.sum(a)
    
    # Convolution
    c = fft_autocorr(a)
    
    # Soft-max of c
    scaled = beta * c
    cm = np.max(scaled)
    exps = np.exp(scaled - cm)
    sum_exps = np.sum(exps)
    M = (cm + np.log(sum_exps)) / beta
    w = exps / sum_exps  # weights
    
    # dM/da_j = 2 * sum_i a_i * w_{i+j} for j=0..n-1
    # Compute g[j] = sum_i a_i * w_{i+j}
    # Using: g = convolve(a, w[::-1])[n-1:2n-1][::-1]
    w_rev = w[::-1]
    N = 1
    while N < n + len(w_rev) - 1:
        N <<= 1
    conv_full = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(w_rev, N), N)[:n + len(w_rev) - 1]
    g = conv_full[n-1:2*n-1][::-1].copy()
    
    # dM/da = 2*g
    # dLoss/da = 2n * (2g/s - 2M/s^2) = 4n*(g/s - M/s^2)
    # dLoss/dx_i = a_i * dLoss/da_i
    grad_a = 4.0 * n * (g / s - M / (s * s))
    grad_x = a * grad_a
    
    loss = 2.0 * n * M / (s * s)
    return loss, grad_x

def adam_step(params, grads, t, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update step. params and grads are dicts."""
    m = {}
    v = {}
    for key in params:
        grad = grads[key]
        if key not in m:
            params[key]['m'] = np.zeros_like(params[key]['x'])
            params[key]['v'] = np.zeros_like(params[key]['x'])
        m_t = b1 * params[key]['m'] + (1 - b1) * grad
        v_t = b2 * params[key]['v'] + (1 - b2) * grad**2
        m_hat = m_t / (1 - b1**(t+1))
        v_hat = v_t / (1 - b2**(t+1))
        params[key]['m'] = m_t
        params[key]['v'] = v_t
        params[key]['x'] -= lr * m_hat / (np.sqrt(v_hat) + eps)

from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def solve_lp_for_b(a, n):
    """Given a (length n), solve LP to find b minimizing max_k (b*a)_k subject to b>=0, sum(b)=1."""
    # Variables: b_0, ..., b_{n-1}, t (total n+1)
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    # Constraints: (b*a)_k - t <= 0 for k=0..2n-2
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Build constraint matrix A_ub (sparse)
    # Row k: [a_0, a_1, ..., a_k, 0, ..., 0] - [0, 0, 0, ..., 1] (for k < n)
    # More precisely:
    # For k = 0..n-1: row k has a_i in column i for i=0..k, and -1 in column n (the t column)
    # For k = n..2n-2: row k has a_{k-n+1}, ..., a_{n-1} in columns 0..(2n-2-k), and -1 in column n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        # (b*a)_k = sum_i b_i * a_{k-i} for valid i
        # Valid i: max(0, k-n+1) <= i <= min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
        # -t term
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    b_ub = np.zeros(2 * n - 1)  # all constraints are <= 0
    
    A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    
    # b >= 0: handled by bounds
    bounds = [(0, None)] * n + [(0, None)]  # b_i >= 0, t >= 0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs', options={'disp': False})
    
    return result

def fft_convolve(a):
    """Fast linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:size]
