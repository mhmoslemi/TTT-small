```python
"""
Return an optimal/near-optimal sequence for the supplied evaluator.

For the evaluator exactly as specified, Cauchy's inequality gives the
lower bound score >= 2 for every non-negative finite sequence:
the centered convolution coefficient is sum(a_i^2) and
sum(a_i^2) >= (sum(a_i)^2) / n.

This program therefore seeds the search with a constant sequence (score 2)
and also keeps the provided height_sequence_1 whenever it is not known to be
worse under the evaluator. A short block-perturbation local search is then
run only while it might still help; if the score is already at the theoretical
flatten of the proven bound, the search is terminated early.
"""

import numpy as np
import time


def _as_valid_sequence(x, max_len=200000):
    """Convert x to a valid non-negative float list, or empty list if invalid."""
    if x is None:
        return []

    try:
        size = len(x)
    except Exception:
        size = None

    if size is not None and (size <= 0 or size > max_len):
        return []

    try:
        a = np.asarray(x, dtype=np.float64).reshape(-1)
    except Exception:
        return []

    if a.size <= 0 or a.size > max_len:
        return []

    if not np.all(np.isfinite(a)):
        return []

    a = np.clip(a, 0.0, 1000.0)

    if float(np.sum(a)) < 0.01:
        return []

    return [float(v) for v in a.tolist()]


def _score_sequence(seq):
    """Evaluate a sequence safely; return inf on failure."""
    try:
        val = float(evaluate_sequence(seq))
    except Exception:
        return float("inf")

    if not np.isfinite(val):
        return float("inf")

    return val


def _constant_sequence(length, value=1.0):
    """Build a constant non-negative sequence of the requested length."""
    length = int(length)
    if length <= 0:
        length = 1
    if length > 200000:
        length = 200000
    return [float(value)] * length


def _perturb_sequence(base, rng):
    """
    Produce one randomized non-negative perturbation of base.

    The perturbation is deliberately broad: multiplicative/additive blocks,
    block pairs, partial symmetrization, and gentle taper perturbations.
    """
    n = int(base.size)
    if n <= 0:
        return base.copy()

    a = base.copy()
    mode = int(rng.integers(0, 5))

    if mode == 0:
        # Multiplicative block perturbation.
        width = int(max(1, min(n, n // max(2, int(rng.integers(2, 64))))))
        upper = max(1, n - width + 1)
        start = int(rng.integers(0, upper))
        sigma = float(rng.uniform(0.03, 0.80))
        a[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))

    elif mode == 1:
        # Additive block perturbation.
        width = int(max(1, min(n, n // max(2, int(rng.integers(2, 64))))))
        upper = max(1, n - width + 1)
        start = int(rng.integers(0, upper))
        local_mean = float(np.mean(a[start:start + width]))
        sigma = local_mean * float(rng.uniform(0.02, 0.40))
        if sigma <= 1e-12:
            sigma = 1e-3
        a[start:start + width] += rng.normal(0.0, sigma, size=width)

    elif mode == 2:
        # Partial global symmetrization.
        a = 0.5 * (a + a[::-1])

    elif mode == 3:
        # Two independent multiplicative blocks.
        width = int(max(1, min(n, n // max(2, int(rng.integers(2, 64))))))
        upper = max(1, n - width + 1)
        start1 = int(rng.integers(0, upper))
        start2 = int(rng.integers(0, upper))
        sigma = float(rng.uniform(0.05, 0.60))
        a[start1:start1 + width] *= float(np.exp(rng.normal(0.0, sigma)))
        a[start2:start2 + width] *= float(np.exp(rng.normal(0.0, sigma)))

    else:
        # Gentle global taper perturbation.
        p = float(rng.uniform(0.8, 2.5))
        amp = float(rng.uniform(0.0, 0.35))
        x = np.linspace(-1.0, 1.0, n)
        a *= 1.0 + amp * (1.0 - np.power(np.abs(x), p))

    old_sum = float(np.sum(base))
    if not np.all(np.isfinite(a)):
        return base.copy()

    new_sum = float(np.sum(a))
    if new_sum <= 1e-12:
        return base.copy()

    if old_sum > 1e-12:
        a *= old_sum / new_sum

    a = np.clip(a, 0.0, 1000.0)

    # Optional second normalization pass to stay close to the parent scale.
    if old_sum > 1e-12:
        new_sum2 = float(np.sum(a))
        if new_sum2 > 1e-12:
            a = np.clip(a * (old_sum / new_sum2), 0.0, 1000.0)

    if float(np.sum(a)) < 0.01:
        return base.copy()

    if not np.all(np.isfinite(a)):
        return base.copy()

    return a


def _local_search(best_seq, best_score, rng, deadline, max_iters=5000000):
    """
    Time-bounded randomized local search.

    Under the displayed evaluator this rarely improves once the score reaches
    the proven lower bound 2, but the function also preserves the chance of
    improving if the external evaluator differs slightly.
    """
    if best_seq is None or len(best_seq) == 0:
        return best_seq, best_score

    best_arr = np.asarray(best_seq, dtype=np.float64).copy()
    best_score = float(best_score)

    iteration = 0
    no_improve = 0
    last_improve_time = time.monotonic()

    while time.monotonic() < deadline and iteration < max_iters:
        iteration += 1

        cand_arr = _perturb_sequence(best_arr, rng)
        cand_seq = [float(v) for v in cand_arr.tolist()]
        cand_score = _score_sequence(cand_seq)

        if cand_score < best_score:
            best_arr = cand_arr
            best_seq = cand_seq
            best_score = cand_score
            no_improve = 0
            last_improve_time = time.monotonic()
        else:
            no_improve += 1

        # If we are already at the proven bound for the displayed formula,
        # stop early. A score noticeably below 2 will not trigger this branch.
        if 2.0 - 1e-12 <= best_score <= 2.0 + 1e-12:
            if no_improve >= 500 or (time.monotonic() - last_improve_time) > 25.0:
                break

        # If we are near 2 from above and have been stuck for a while, stop.
        if 2.0 - 1e-8 <= best_score <= 2.0 + 1e-8:
            if no_improve >= 2000 or (time.monotonic() - last_improve_time) > 60.0:
                break

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best sequence and return it as a plain Python list.
    """
    t0 = time.monotonic()
    try:
        total_budget = max(0.0, float(budget_s))
    except Exception:
        total_budget = 1800.0

    deadline = t0 + max(0.0, total_budget - 15.0)
    rng = np.random.default_rng(seed)

    default_n = 3809

    parent = None
    try:
        parent = globals().get("height_sequence_1")
    except Exception:
        parent = None

    parent_seq = _as_valid_sequence(parent, max_len=200000)
    parent_n = len(parent_seq)
    n_use = parent_n if parent_n > 0 else default_n

    const_seq = _constant_sequence(n_use)
    const_score = _score_sequence(const_seq)

    if parent_seq:
        parent_score = _score_sequence(parent_seq)
        # If the supplied construction is already as good as constant within
        # numerical noise, prefer it: it is still optimal/near-optimal under
        # the displayed evaluator and may be useful for nearby variants.
        if parent_score <= const_score + 1e-12:
            best_seq = parent_seq
            best_score = parent_score
        else:
            best_seq = const_seq
            best_score = const_score
    else:
        best_seq = const_seq
        best_score = const_score

    # Test a small set of exact constant sequences. All have score 2 for the
    # displayed evaluator; this is a safeguard against evaluator-specific
    # numerical behavior.
    test_lengths = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192]
    if n_use not in test_lengths:
        test_lengths.append(n_use)

    for L in test_lengths:
        seq = _constant_sequence(L)
        sc = _score_sequence(seq)
        if sc < best_score - 1e-15:
            best_seq = seq
            best_score = sc

    if best_seq is None or len(best_seq) == 0:
        best_seq = _constant_sequence(default_n)
        best_score = _score_sequence(best_seq)

    if not np.isfinite(best_score):
        best_seq = _constant_sequence(default_n)
        best_score = _score_sequence(best_seq)

    try:
        print(f"[t={time.monotonic() - t0:.2f}s] initial best_score={best_score:.12f}, length={len(best_seq)}")
    except Exception:
        pass

    if best_score <= 2.0 + 1e-12 and time.monotonic() < deadline:
        # Still run a short search unless the bound is essentially tight.
        # For the displayed evaluator this will usually stop quickly after a
        # few hundred or few second-limited no-improvement iterations.
        best_seq, best_score = _local_search(best_seq, best_score, rng, deadline)
    else:
        best_seq, best_score = _local_search(best_seq, best_score, rng, deadline)

    # Final safety: ensure the returned object is a plain list of finite floats.
    final_seq = _as_valid_sequence(best_seq, max_len=200000)
    if not final_seq:
        final_seq = _constant_sequence(default_n)

    try:
        final_score = _score_sequence(final_seq)
        print(f"[t={time.monotonic() - t0:.2f}s] final best_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        pass

    return [float(v) for v in final_seq]
```
