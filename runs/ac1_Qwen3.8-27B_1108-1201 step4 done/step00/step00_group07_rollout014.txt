            a = np.exp(alpha * np.arange(n) / (n-1))

    for n in [256, 512, 1024, 2048]:

                a = np.maximum(a, 0)

    for n in [256, 512, 1024, 2048]:

            a = np.exp(-x**2 / (2 * (sigma_frac * n)**2))
