```
def make_weights(conv, tau, n):
    m = conv.size
    peak = float(np.max(conv))
    if peak <= 0 or not np.isfinite(peak):
        w=np.full(m, 1.0/m); return w
    frac = tau/peak
    z=(conv-peak)/tau
    z=np.clip(z, -50.0, 0.0)
    w=np.exp(z)
    sw=w.sum()
    if sw <= 1e-300:
       w=np.zeros(m); w[int(np.argmax(conv))]=1; return w
    w /= sw
    if frac < 0.08:
       blend = min(0.4, 0.4 * (0.08 - frac) / 0.08)
       if blend > 0:
          tc = min(max(16, n//32), m)
          if tc < m:
             idx=np.argpartition(-conv, tc)[:tc]
          else: idx=np.arange(m)
          tp=np.zeros(m); tp[idx]=1.0/idx.size
          w = (1.0-blend)*w + blend*tp
    return w
```

```python
"""
FFT-based mirror-descent optimizer for minimizing
    J(a) = 2*n*max(conv(a,a)) / sum(a)^2
for a nonnegative real sequence a of moderate length.

The main loop maintains a fixed positive sum S = min(n, 1000), uses an
FFT approximation to log-sum-exp subgradients of conv(a,a), applies
multiplicative/log-domain updates, and periodically perturbs/restarts
to escape local minima.  A short final verification uses evaluate_sequence.
"""

import numpy as np
import time


def normalize_to_sum(a, target):
    """Clamp to feasible range and scale to exactly target sum."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a

    max_val = 1000.0
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, max_val)
    s = float(a.sum())

    if s <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)

    a = a * (float(target) / s)
    a = np.minimum(a, max_val)
    s2 = float(a.sum())

    if s2 <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)

    if s2 < float(target):
        slack = max_val - a
        tot = float(slack.sum())
        if tot > 0:
            a = a + slack * ((float(target) - s2) / tot)
        else:
            a = np.full(n, target / n, dtype=np.float64)
    elif s2 > float(target) * (1.0 + 1e-12):
        a = a * (float(target) / s2)
        a = np.minimum(a, max_val)
        s3 = float(a.sum())
        if s3 > 0:
            a = a * (float(target) / s3)

    return a


def from_logs_to_array(x, target):
    """Convert relative log-values to a nonnegative array with sum target."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return x

    if not np.all(np.isfinite(x)):
        return np.full(n, target / n, dtype=np.float64)

    xmax = float(np.max(x))
    xr = np.clip(x - xmax, -50.0, 0.0)
    y = np.exp(xr)
    s = float(y.sum())

    if s <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)

    a = (float(target) / s) * y
    a = np.minimum(a, 1000.0)
    s2 = float(a.sum())

    if s2 <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)

    if s2 < float(target):
        slack = 1000.0 - a
        tot = float(slack.sum())
        if tot > 0:
            a = a + slack * ((float(target) - s2) / tot)
        else:
            a = np.full(n, target / n, dtype=np.float64)
    elif s2 > float(target) * (1.0 + 1e-12):
        a = a * (float(target) / s2)
        a = np.minimum(a, 1000.0)

    return a


def downsample_to_length(x, n_new):
    """Resize a 1-D sequence to n_new by block-averaging or nearest sampling."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n_old = x.size
    if n_old == n_new:
        return x

    if n_old == 0:
        return np.full(n_new, 1.0)

    if n_old > n_new:
        out = np.empty(n_new, dtype=np.float64)
        for i in range(n_new):
            lo = int((i * n_old) / n_new)
            hi = int(((i + 1) * n_old) / n_new)
            if hi <= lo:
                hi = lo + 1
            out[i] = float(np.mean(x[lo:hi]))
        return out

    out = np.empty(n_new, dtype=np.float64)
    for i in range(n_new):
        j = int(i * n_old / n_new)
        out[i] = x[j]
    return out


def score_from_conv(conv, n, s):
    """Compute J from an already-computed full convolution."""
    c = np.asarray(conv, dtype=np.float64)
    if c.size == 0 or s <= 0.01:
        return float("inf")
    peak = float(np.max(c))
    if peak < 0.0 or not np.isfinite(peak):
        peak = 0.0
    return float(2.0 * n * peak / (s * s))


def autocorr_rfft(a, L):
    """Full self-convolution via real FFT.  Returns (conv, FFT-of-a)."""
    A = np.fft.rfft(a, L)
    two = 2 * a.size - 1
    c = np.fft.irfft(A * A, n=L)[:two]
    if c.size:
        c[c < 0.0] = 0.0
    return c, A


def gradient_rfft(weights, A, n, L):
    """
    For weights w over conv positions k, compute
    g_i = sum_k w_k a_{k-i} over valid k, then the full subgradient is 2*g.
    Here we return g without the final factor 2; caller scales if needed.
    """
    P = np.fft.rfft(weights, L)
    g = np.fft.irfft(P * np.conj(A), n=L)[:n]
    if g.size:
        g[g < 0.0] = 0.0
    return g


def make_weights(conv, tau, n):
    """Softmax weights over convolution peaks, with a small top-blend sweep."""
    m = conv.size
    if m == 0:
        return np.zeros(0)

    peak = float(np.max(conv))
    if peak <= 0.0 or not np.isfinite(peak):
        return np.full(m, 1.0 / m, dtype=np.float64)

    tau = float(max(tau, peak * 1e-6, 1e-300))
    z = (conv - peak) / tau
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    sw = float(w.sum())

    if sw <= 1e-300:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(conv))] = 1.0
        return w

    w = w / sw
    frac = tau / peak

    # When the soft max is very sharp, blend in a flat top-K mask.
    if frac < 0.08:
        blend = min(0.45, 0.45 * (0.08 - frac) / 0.08)
        if blend > 0.0:
            tc = int(min(max(16, n // 32), m))
            if tc < m:
                idx = np.argpartition(-conv, tc)[:tc]
            else:
                idx = np.arange(m)
            if idx.size > 0:
                tp = np.zeros(m, dtype=np.float64)
                tp[idx] = 1.0 / idx.size
                w = (1.0 - blend) * w + blend * tp

    return w


def perturb_multiplicative(a, rng, target, strength):
    """Multiplicative local/random block perturbation, then renormalize."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    target = float(target)

    if n <= 1:
        return normalize_to_sum(a, target)

    if rng.random() < 0.72:
        low = max(2, n // 120)
        high = max(low + 1, min(n // 6, n - 1))
        if high >= n:
            high = n - 1
        if low >= high:
            low = 1
            high = 2
        width = int(rng.integers(low + 1, high + 1))
        width = min(width, n - 1)
        if width >= 1:
            start = int(rng.integers(0, n - width))
            a[start:start + width] *= np.exp(
                rng.normal(0.0, strength, size=width)
            )
    else:
        a *= np.exp(rng.normal(0.0, 0.30 * strength, size=n))

    return normalize_to_sum(a, target)


def make_restart_sequence(n, kind, rng, target):
    """Deterministic/random structured restart initializer."""
    idx = np.arange(n, dtype=np.float64)
    if n <= 1:
        a = np.ones(n, dtype=np.float64)
    else:
        t = idx / float(n - 1)
        if kind == 0:
            a = np.ones(n, dtype=np.float64)
        elif kind == 1:
            a = 2.0 - t
        elif kind == 2:
            a = 1.0 + t
        elif kind == 3:
            fmax = max(2, n // 100)
            f = int(rng.integers(1, fmax + 1))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            a = 1.0 + 0.40 * np.cos(2.0 * np.pi * f * t + phase)
        elif kind == 4:
            a = np.exp(rng.normal(0.0, 0.30, size=n))
        elif kind == 5:
            a = np.ones(n, dtype=np.float64)
            cut = n // 2
            if cut > 0:
                a[cut:] = 0.5
        elif kind == 6:
            # Low-center, high-edge shape.
            a = np.sqrt(np.maximum(1.0 - t * (1.0 - t), 1e-3))
        elif kind == 7:
            a = np.exp(np.linspace(0.60, -0.30, n))
        else:
            a = 1.0 + 0.35 * np.sin(2.0 * np.pi * idx / n) ** 2

    a = np.maximum(a, 0.0)
    return normalize_to_sum(a, target)


def run_optimization(n, a_init, time_budget, rng):
    """
    Run the adaptive mirror-descent loop for time_budget seconds.
    Returns (best_sequence_list, best_logged_score).
    """
    n = int(n)
    if n <= 0:
        return [], float("inf")

    target = min(float(n), 1000.0)
    a = normalize_to_sum(np.asarray(a_init, dtype=np.float64), target)
    best = a.copy()

    L = 1 << (max(1, 3 * n - 2)).bit_length()

    conv0, _ = autocorr_rfft(a, L)
    best_score = score_from_conv(conv0, n, target)

    # One expensive ground-truth check at the start for moderate sizes.
    if n <= 8000:
        try:
            ver = float(evaluate_sequence(best.tolist()))
            if np.isfinite(ver):
                best_score = ver
        except Exception:
            pass

    if not np.isfinite(best_score):
        best_score = 2.0

    current_score = best_score
    starts = time.monotonic()
    deadline = starts + max(1.0, float(time_budget))

    lr = 0.32
    lr_min = 0.02
    lr_max = 0.95
    tau_frac = 0.25
    tau_min = 0.03
    tau_max = 0.50

    last_best = starts
    last_perturb = starts
    last_restart = starts
    last_print = starts

    iter_count = 0

    while time.monotonic() < deadline:
        conv, A = autocorr_rfft(a, L)
        now = time.monotonic()

        peak = float(np.max(conv)) if conv.size else 0.0
        if peak < 0.0 or not np.isfinite(peak):
            a = best.copy()
            continue

        cur = float(2.0 * n * peak / (target * target))

        if np.isfinite(cur) and cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
            last_best = now
            if now - last_print > 10.0:
                print(f"  score={best_score:.8f} n={n} iter={iter_count}")
                last_print = now
        elif not np.isfinite(cur):
            a = best.copy()
            continue

        # If the current iterate is far worse than the best, nudge it back.
        if cur > best_score * 1.25 and now - last_perturb > 5.0:
            a = perturb_multiplicative(best, rng, target, 0.25)
            last_perturb = now
            iter_count += 1
            continue

        tau = max(peak * tau_frac, peak * 1e-6, 1e-300)
        w = make_weights(conv, tau, n)
        g = gradient_rfft(w, A, n, L)
        gmean = float(np.mean(g))

        if gmean <= 1e-300 or not np.isfinite(gmean):
            a = perturb_multiplicative(a, rng, target, 0.25)
            last_perturb = now
            iter_count += 1
            continue

        grad = g / gmean
        x = np.log(np.maximum(a, 1e-300))

        accepted = False
        trial = lr
        for _ in range(3):
            step = trial * grad
            step = np.clip(step, 0.0, 1.0)
            cand = from_logs_to_array(x - step, target)

            conv_c, _ = autocorr_rfft(cand, L)
            cand_score = score_from_conv(conv_c, n, target)

            if np.isfinite(cand_score) and cand_score < cur - 1e-12:
                a = cand
                cur = cand_score
                accepted = True

                if cur < best_score - 1e-12:
                    best_score = cur
                    best = cand.copy()
                    last_best = time.monotonic()
                    if time.monotonic() - last_print > 10.0:
                        print(f"  score={best_score:.8f} n={n} iter={iter_count}")
                        last_print = time.monotonic()

                lr = min(lr_max, trial * 1.08)
                tau_frac = max(tau_min, tau_frac * 0.96)
                break

            trial *= 0.62

        if not accepted:
            lr = max(lr_min, trial * 0.70)
            tau_frac = min(tau_max, tau_frac * 1.05)

        now = time.monotonic()

        # Stagnation handling: local perturb, then best-noise restart,
        # then broad structured restart.
        if now - last_best > 25.0 and now - last_perturb > 8.0:
            strength = 0.18 + 0.18 * min(1.0, (now - last_best) / 150.0)
            a = perturb_multiplicative(a, rng, target, strength)
            lr = 0.30
            tau_frac = 0.25
            last_perturb = now

        if now - last_best > 120.0 and now - last_restart > 30.0:
            xbest = np.log(np.maximum(best, 1e-300))
            noise = rng.normal(0.0, 0.30, size=n)
            a = from_logs_to_array(xbest + noise, target)
            lr = 0.35
            tau_frac = 0.25
            last_restart = now
            last_perturb = now

        if now - last_best > 360.0 and now - last_restart > 60.0:
            kind = int(rng.integers(0, 8))
            a = make_restart_sequence(n, kind, rng, target)
            lr = 0.40
            tau_frac = 0.30
            last_restart = now
            last_perturb = now

        iter_count += 1

    seq = [float(v) for v in best]

    try:
        fin = float(evaluate_sequence(seq))
    except Exception:
        fin = best_score

    score_out = fin if np.isfinite(fin) else best_score
    if not np.isfinite(score_out):
        score_out = 2.0

    return seq, float(score_out)


def propose_candidate(seed=42, budget_s=1800):
    """
    Public entry point.  Chooses a robust length, evaluates several structured
    initializers, then spends most of the budget in two adaptive phases and
    returns the best feasible sequence found.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(30.0, float(budget_s))
    effective = max(20.0, total_budget - 15.0)
    t0 = time.monotonic()

    # Obtain a base sequence from the supplied global if possible.
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        raw = np.where(np.isfinite(raw), raw, 0.0)
    except Exception:
        raw = np.ones(4096, dtype=np.float64)

    if raw.size == 0:
        raw = np.ones(4096, dtype=np.float64)

    n = int(raw.size)
    if n < 2048:
        if n > 0:
            raw = downsample_to_length(raw, 4096)
        else:
            raw = np.ones(4096, dtype=np.float64)
        n = int(raw.size)
    elif n > 8192:
        raw = downsample_to_length(raw, 8192)
        n = int(raw.size)

    target = min(float(n), 1000.0)
    L = 1 << (max(1, 3 * n - 2)).bit_length()

    # Choose the best among the supplied sequence and several structured
    # initializers using fast FFT scores.
    raw_norm = normalize_to_sum(raw, target)
    conv_raw, _ = autocorr_rfft(raw_norm, L)
    best_init_score = score_from_conv(conv_raw, n, target)
    best_init = raw_norm

    kinds = range(8)
    for kind in kinds:
        cand = make_restart_sequence(n, int(kind), rng, target)
        convc, _ = autocorr_rfft(cand, L)
        sc = score_from_conv(convc, n, target)
        if np.isfinite(sc) and sc < best_init_score:
            best_init_score = sc
            best_init = cand

    # Phase 1: longest continuous run from the best initializer.
    phase1_time = 0.65 * effective
    seq, score = run_optimization(n, best_init, phase1_time, rng)

    elapsed = time.monotonic() - t0
    remaining = effective - max(0.0, elapsed)

    # Phase 2: continue from current best, or attempt a fresh structured
    # restart if phase 1 is still clearly below the target region.
    if remaining > 25.0:
        if not np.isfinite(score) or score > 1.85:
            kind = int(rng.integers(0, 8))
            init2 = make_restart_sequence(n, kind, rng, target)
            seq2, score2 = run_optimization(n, init2, remaining, rng)
            if np.isfinite(score2) and score2 < score:
                seq, score = seq2, score2
        else:
            seq2, score2 = run_optimization(n, seq, remaining, rng)
            if np.isfinite(score2) and score2 < score:
                seq, score = seq2, score2

    # Final feasibility cleanup.
    if not isinstance(seq, list):
        seq = [float(x) for x in seq]
    else:
        seq = [float(x) for x in seq]

    seq = normalize_to_sum(np.asarray(seq, dtype=np.float64), target)
    out = [float(x) for x in seq]

    try:
        final_score = float(evaluate_sequence(out))
    except Exception:
        final_score = score

    if not np.isfinite(final_score) or final_score > 2.0:
        # Conservative fallback: a uniform sequence always evaluates to 2.
        out = [float(target) / n for _ in range(n)]

    return out
```
