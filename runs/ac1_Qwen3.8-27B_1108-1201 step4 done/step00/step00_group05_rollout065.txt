"""
Multi-scale search for a non-negative step sequence minimizing
    J = 2*n*max(convolve(a,a)) / (sum a)^2
Strategy:
  1. Exhaustive-ish Nelder-Mead restarts in log-space for small n (2..12),
     where unbalanced (non-uniform) sequences can clearly beat the uniform
     value of 2.
  2. For medium/large n (20..~1200) run a memory-bounded, exact-evaluation
     multiplicative local search with adaptive step size, several structured
     start points, and occasional mirroring, all sharing a single deadline
     and a single global best.
The exact evaluate_sequence() is always authoritative.
"""

import time
import numpy as np


# ----------------------------------------------------------------------------
# generic helpers (top level, no closures / lambdas)
# ----------------------------------------------------------------------------

def _valid(x):
    return (np.isfinite(x) and x > 0.0)


def _objective_from_log(loga):
    a = np.exp(np.clip(loga, -30.0, 7.0))
    a = np.clip(a, 0.0, 1000.0)
    return float(evaluate_sequence(a.tolist()))


def _objective_from_a(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    return float(evaluate_sequence(a.tolist()))


def _structured_start(n, kind, rng):
    """A few deterministic-but-randomized initial shapes for local search."""
    if kind == "uniform":
        return np.ones(n, dtype=np.float64)
    if kind == "ramp":
        t = np.linspace(0.0, 1.0, n)
        return np.exp(rng.normal(0.0, 1.0) + 1.5 * t)
    if kind == "ramp_rev":
        t = np.linspace(0.0, 1.0, n)
        return np.exp(rng.normal(0.0, 1.0) + 1.5 * (1.0 - t))
    if kind == "two_block":
        a = np.ones(n, dtype=np.float64)
        cut = int(rng.integers(1, n))
        a[:cut] = float(rng.uniform(0.3, 3.0))
        a[cut:] = float(rng.uniform(0.3, 3.0))
        return a
    if kind == "asym_gauss":
        i = np.arange(n, dtype=np.float64)
        mu = float(rng.uniform(0.2, 0.8) * n)
        sig = float(rng.uniform(0.08, 0.45) * n)
        return np.exp(-0.5 * ((i - mu) / sig) ** 2)
    # fallback
    return np.exp(rng.normal(0.0, 0.5, n))


def _nelder_mead_n(n, deadline, initial_lost, rng):
    """Small-n global-ish search in log-space via many Nelder-Mead restarts."""
    from scipy.optimize import minimize

    def f(t):
        val = _objective_from_log(t)
        if not np.isfinite(val):
            return 1e9
        return val

    best_a = None
    best_s = float(initial_lost)
    kinds = ["uniform", "ramp", "ramp_rev", "two_block", "asym_gauss"]
    restarts = 0
    while time.monotonic() < deadline:
        kind = kinds[restarts % len(kinds)]
        if kind == "uniform":
            t0 = np.zeros(n, dtype=np.float64)
        elif kind in ("ramp", "ramp_rev"):
            t = np.linspace(0.0, 1.0, n)
            t0 = 1.5 * t if kind == "ramp" else 1.5 * (1.0 - t)
            t0 = t0 + rng.normal(0.0, 0.2, n)
        elif kind == "two_block":
            t0 = rng.normal(0.0, 0.6, n)
            cut = int(rng.integers(1, n))
            t0[:cut] += float(rng.uniform(-1.0, 1.0))
            t0[cut:] += float(rng.uniform(-1.0, 1.0))
        else:
            i = np.arange(n, dtype=np.float64)
            mu = float(rng.uniform(0.2, 0.8) * n)
            sig = float(rng.uniform(0.08, 0.45) * n)
            t0 = -0.5 * ((i - mu) / sig) ** 2 + rng.normal(0.0, 0.1, n)
        t0 = t0 + rng.normal(0.0, 0.15, n)  # jitter every start

        res = minimize(
            f, t0, method="Nelder-Mead",
            options={"maxiter": 120, "xatol": 1e-5, "fatol": 1e-6,
                     "adaptive": True},
        )
        if time.monotonic() > deadline:
            break
        val = f(res.x)
        if np.isfinite(val) and val < best_s:
            best_s = val
            best_a = np.exp(np.clip(res.x, -30.0, 7.0))
            best_a = np.clip(best_a, 0.0, 1000.0)
            print("  [NM n=%d] improved -> %.6f (restart %d, %s)"
                  % (n, best_s, restarts, kind))
        restarts += 1
        if restarts >= 40:  # cap small-n effort per n
            break
    return best_a, best_s


def _local_search_n(n, deadline, initial_lost, rng):
    """Medium/large-n exact-evaluation multiplicative local search."""
    best_a = None
    best_s = float(initial_lost)

    kinds = ["ramp", "ramp_rev", "two_block", "asym_gauss", "uniform"]
    for kind in kinds:
        if time.monotonic() >= deadline:
            break
        a = _structured_start(n, kind, rng).astype(np.float64)
        s = _objective_from_a(a)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_a = a.copy()
            print("  [LS n=%d] better start %s -> %.6f" % (n, kind, best_s))
        if not np.isfinite(s):
            s = best_s

        cur = a.copy()
        cur_s = s
        sigma = 0.15
        stall = 0
        it = 0
        while time.monotonic() < deadline:
            width = max(2, int(n * rng.uniform(0.02, 0.25)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand = cur.copy()
            cand[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))
            if rng.random() < 0.25:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
            ns = _objective_from_a(cand)
            it += 1
            if np.isfinite(ns) and ns < cur_s:
                cur = cand
                cur_s = ns
                sigma = min(0.8, sigma * 1.05)
                stall = 0
                if ns < best_s:
                    best_s = ns
                    best_a = cur.copy()
                    if it % 200 == 0:
                        print("  [LS n=%d] it %d -> %.6f" % (n, it, best_s))
            else:
                cur = cand  # anneal-style move to escape basin
                cur_s = ns if np.isfinite(ns) else cur_s
                sigma = max(0.02, sigma * 0.85)
                stall += 1
            if stall > 600:
                break
    return best_a, best_s

    return out
