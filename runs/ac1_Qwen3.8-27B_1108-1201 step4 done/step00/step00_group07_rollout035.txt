import numpy as np
import time

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Find lambda such that sum(max(v - lambda, 0)) = S
    # This is equivalent to finding the S-th "level" of v
    # Sort v in decreasing order
    w = np.sort(v)[::-1]
    # Find the largest k such that sum(w[:k]) - k*(w[k] + 1/k) > S
    # (standard simplex projection algorithm)
    cumsum = np.cumsum(w)
    rho = np.where(w * np.arange(1, n+1) > (cumsum - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1] + 1  # 1-indexed
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0)

def grad_at_k(a, k, n):
    """Compute gradient of c[k] = sum_j a[j]a[k-j] w.r.t. a."""
    # dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n, else 0

    return grad

**Alternative: Use the Frank-Wolfe (conditional gradient) method**

The Frank-Wolfe method is designed for constrained optimization. At each step:
1. Compute the gradient g of the objective at the current point.
2. Solve the linear minimization: min_{s in S} <g, s> where S is the constraint set.
3. Update: a_new = (1-γ)a + γ*s

For our constraint set S = {x >= 0, sum(x) = S_0}, the linear minimization is:

The solution is: put all mass on the component with the smallest g[m]. So s = S_0 * e_{m*} where m* = argmin g[m].

This gives: a_new = (1-γ)a + γ * S_0 * e_{m*}

This is very simple! At each step, we just increase the component with the smallest gradient and decrease all others proportionally.

"""
Gradient-based optimization for minimizing the autocorrelation peak of non-negative sequences.
Uses FFT-accelerated convolution, Adam optimizer with simplex projection, and multiple restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    # Find largest rho such that w[rho-1] >= (cumsum[rho-1] - S) / rho
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0] + 1))
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, n, top_indices, c):
    """Compute subgradient of max(c) using top-k indices."""
    grad = np.zeros(n)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 0:
        return grad
    for w, k in zip(top_vals / total, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def mirror_descent_step(a, grad, S, eta):
    """Multiplicative weights update with normalization."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new * S / s
    else:
        a_new = np.full_like(a, S / len(a))
    return a_new


def run_adam_optimization(a, n, S, max_time, rng, lr_init, top_k=5):
    """Run Adam optimization with simplex projection."""
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    best_local = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_score = c.max()
        
        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        
        # Scale gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)
        a = a_new
        
        # Learning rate schedule
        if t % 100 == 0:
            lr = lr_init * (0.5 ** (t / 500))
        
        # Periodic evaluation with exact function
        if t % 500 == 0:
            exact = evaluate_sequence(a.tolist())
            if np.isfinite(exact) and exact < best_score + 0.01:
                pass  # consistent
        
    return best_local, best_score


def run_mirror_optimization(a, n, S, max_time, rng, eta_init, top_k=5):
    """Run mirror descent (multiplicative weights) optimization."""
    best_local = a.copy()
    best_score = fft_convolve_self(a).max()
    eta = eta_init
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_score = c.max()
        
        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        
        # Scale gradient for multiplicative update
        gmax = np.abs(grad).max()
        if gmax > 1e-12:
            grad = grad / gmax
        
        a_new = mirror_descent_step(a, grad, S, eta)
        a = a_new
        
        # Decay eta
        if t % 200 == 0:
            eta *= 0.98
        
    return best_local, best_score

"""
Gradient-based optimization to minimize 2n*max_conv/sum^2 for non-negative sequences.
Uses FFT-accelerated convolution, Adam optimizer with simplex projection,
mirror descent (multiplicative weights), and multiple restarts with adaptive time allocation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, n, top_indices, c):
    """Compute weighted subgradient of max(c) at top-k indices."""
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_adam(a_init, n, S, max_time, rng, lr_init, top_k):

    v = np.zeros(n, dtype=np.float64)

    lr = lr_init
    best_local = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    start = time.monotonic()

        t += 1
        c = fft_convolve_self(a)
        cur_score = c.max()

            best_local = a.copy()

        grad = compute_subgradient(a, n, top_indices, c)
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm
        else:
            break

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        v_hat = v / (1.0 - beta2 ** t)
