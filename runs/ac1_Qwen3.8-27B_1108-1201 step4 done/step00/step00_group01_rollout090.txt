def next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p *= 2
    return p


def fft_len_conv(n):
    return next_pow2(2 * int(n) - 1)


def fft_len_grad(n):
    return next_pow2(3 * int(n) - 2)


def normalize_pos(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    x = np.clip(x, 0.0, 1000.0)
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(x, dtype=np.float64) / float(x.size)
    return x / s


def self_conv(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    out = np.fft.irfft(fa * fa, fft_len)
    return out[:2 * len(a) - 1]


def fast_score(a, conv_len=None):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return float("inf")
    a = a / s
    if conv_len is None:
        conv_len = fft_len_conv(a.size)
    conv = self_conv(a, conv_len)
    conv = np.maximum(conv, 0.0)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * float(a.size) * m)


def weights_from_conv(conv, beta_rel=20.0):
    cmax = float(np.max(conv))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros_like(conv, dtype=np.float64), 0.0
    s = beta_rel * (conv - cmax) / cmax
    s = np.minimum(s, 0.0)
    w = np.exp(s)
    maxw = float(np.max(w))
    if not np.isfinite(maxw) or maxw <= 0.0:
        w = np.zeros_like(conv, dtype=np.float64)
        idx = np.flatnonzero(conv >= cmax - 1e-12)
        if idx.size == 0:
            idx = np.array([0], dtype=int)
        w[idx] = 1.0
        return w / float(w.sum()), cmax
    w[w < maxw * 1e-7] = 0.0
    ws = float(np.sum(w))
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros_like(conv, dtype=np.float64)
        idx = np.flatnonzero(conv >= cmax - 1e-12)
        if idx.size == 0:
            idx = np.array([0], dtype=int)
        w[idx] = 1.0
        ws = float(w.sum())
    return w / ws, cmax


def grad_from_weights(a, w, grad_fft_len):
    n = len(a)
    m = len(w)
    b = np.zeros(grad_fft_len, dtype=np.float64)
    b[:m] = w[::-1]
    fa = np.fft.rfft(a, grad_fft_len)
    fb = np.fft.rfft(b, grad_fft_len)
    conv = np.fft.irfft(fa * fb, grad_fft_len)
    start = m - 1
    g = conv[n - 1:start + 1][::-1]
    g = np.maximum(g, 0.0)
    return g


def make_starts(K, random_count=6):
    K = int(K)
    if K <= 1:
        return [np.array([1.0], dtype=np.float64)]

    starts = []
    starts.append(np.ones(K, dtype=np.float64) / float(K))

    half = K // 2
    if half > 0 and K - half > 0:
        m = np.zeros(K, dtype=np.float64)
        m[:half] = (2.0 / 3.0) / float(half)
        m[half:] = (1.0 / 3.0) / float(K - half)
        starts.append(normalize_pos(m))
        starts.append(normalize_pos(m[::-1]))

    t = np.linspace(0.0, 1.0, K, dtype=np.float64)
    profiles = [
        (0.586, 0.828),
        (0.586, -0.828),
        (0.5, 1.0),
        (1.0, 0.5),
        (1.0, -0.5),
        (0.2, 1.5),
        (1.8, -0.75),
        (0.75, 1.25),
    ]
    for a0, b0 in profiles:
        h = a0 + b0 * t
        h = np.maximum(h, 0.0)
        if np.sum(h) > 0.0:
            starts.append(normalize_pos(h))

    idx = np.arange(1, K + 1, dtype=np.float64)
    for p in (0.35, 0.6, 1.0, 1.8):
        v = np.power(idx, p)
        starts.append(normalize_pos(v))
        starts.append(normalize_pos(v[::-1]))

    for _ in range(int(random_count)):
        noise = _GLOBAL_RNG.standard_normal(K)
        win = max(2, K // 8)
        kernel = np.ones(win, dtype=np.float64) / float(win)
        sm = np.convolve(noise, kernel, mode="same")
        v = np.abs(sm) + 1e-3
        starts.append(normalize_pos(v))

    clean = []
    for x in starts:
        x = normalize_pos(x)
        if x.shape == (K,) and np.all(np.isfinite(x)):
            clean.append(x)
    return clean


def m_to_short(m, K):
    K = int(K)
    m = np.asarray(m, dtype=np.float64).copy()
    if m.size != K:
        m = normalize_pos(m)
    m = np.maximum(m, 1e-300)
    s = float(np.sum(m))
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(K - 1, dtype=np.float64)
    m = m / s
    return np.log(m[:-1]) - np.log(m[-1])


def short_to_m(p, K):
    K = int(K)
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size != K - 1:
        p = p[:K - 1] if p.size > K - 1 else np.pad(p, (0, K - 1 - p.size))
    if not np.all(np.isfinite(p)):
        return np.ones(K, dtype=np.float64) / float(K)
    full = np.zeros(K, dtype=np.float64)
    full[:-1] = p
    full -= full.max()
    e = np.exp(full)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(K, dtype=np.float64) / float(K)
    return e / s


def _lbfgs_fun(p):
    K = int(_G.get("K", 0))
    beta = float(_G.get("beta", 1.0))
    conv_len = int(_G.get("conv_len", 16))
    if K <= 1:
        return 1e100
    m = short_to_m(p, K)
    conv = self_conv(m, conv_len)
    cmax = float(np.max(conv))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 1e100
    s = beta * (conv - cmax)
    s = np.minimum(s, 0.0)
    e = np.exp(s)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        return 1e100
    return float(beta * cmax + np.log(se))


def _lbfgs_jac(p):
    K = int(_G.get("K", 0))
    beta = float(_G.get("beta", 1.0))
    conv_len = int(_G.get("conv_len", 16))
    grad_len = int(_G.get("grad_len", 32))
    if K <= 1:
        return np.zeros(0, dtype=np.float64)
    m = short_to_m(p, K)
    conv = self_conv(m, conv_len)
    cmax = float(np.max(conv))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(K - 1, dtype=np.float64)
    s = beta * (conv - cmax)
    s = np.minimum(s, 0.0)
    e = np.exp(s)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        return np.zeros(K - 1, dtype=np.float64)
    w = e / se
    g = grad_from_weights(m, w, grad_len)
    grad_m = 2.0 * beta * g
    avg = float(np.dot(m, grad_m))
    grad_full = m * (grad_m - avg)
    out = grad_full[:-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def base_random_search(m0, deadline, conv_len):
    m = normalize_pos(m0)
    best = m.copy()
    best_s = fast_score(best, conv_len)
    n = m.size
    while time.monotonic() < float(deadline):
        cand = best.copy()
        w = max(1, n // 20)
        st = int(_GLOBAL_RNG.integers(0, max(1, n - w + 1)))
        cand[st:st + w] *= float(np.exp(_GLOBAL_RNG.normal(0.0, 0.35)))
        for _ in range(2):
            idx = int(_GLOBAL_RNG.integers(0, n))
            cand[idx] *= float(np.exp(_GLOBAL_RNG.normal(0.0, 0.35)))
        cand = normalize_pos(cand)
        s = fast_score(cand, conv_len)
        if np.isfinite(s) and s < best_s:
            best = cand
            best_s = s
    return best, float(best_s)


def search_base(K, deadline, extra_starts=None):

    deadline = float(deadline)
    conv_len = fft_len_conv(K)

    if best is None:

def upsample_base(base, target_K):
    base = np.asarray(base, dtype=np.float64)
    K = max(1, int(base.size))
    target_K = max(1, int(target_K))
    out = np.zeros(target_K, dtype=np.float64)
    for i in range(K):
        a = int(round(i * target_K / float(K)))
        b = int(round((i + 1) * target_K / float(K)))
        if b > a:
            out[a:b] = float(base[i]) / float(b - a)
    return normalize_pos(out)


def expand_base(base, N):
    base = np.asarray(base, dtype=np.float64)
    K = int(base.size)
    N = int(N)
    if K <= 1:
        return np.ones(N, dtype=np.float64) / float(N)
    lens = np.full(K, N // K, dtype=int)
    rem = N % K
    if rem > 0:
        lens[:rem] += 1
    lens = np.maximum(lens, 1)
    a = np.repeat(base / lens, lens)
    return normalize_pos(a)


def two_block_sequence(N):
    N = int(N)
    a = np.zeros(N, dtype=np.float64)
    half = max(1, N // 2)
    a[:half] = (2.0 / 3.0) / float(half)
    a[half:] = (1.0 / 3.0) / float(max(1, N - half))
    return normalize_pos(a)


def ramp_sequence(N, increasing=True):
    N = int(max(2, N))
    t = np.linspace(0.0, 1.0, N, dtype=np.float64)
    if increasing:
        h = 0.586 + 0.828 * t
    else:
        h = 0.586 + 0.828 * (1.0 - t)
    return normalize_pos(np.maximum(h, 0.0))


def random_direct_sequence(N):
    N = int(max(2, N))
    t = np.linspace(0.0, 1.0, N, dtype=np.float64)
    coeffs = _GLOBAL_RNG.normal(0.0, 0.28, 6)
    phases = _GLOBAL_RNG.uniform(0.0, 2.0 * np.pi, 6)
    freqs = (0.7, 1.3, 2.1, 3.0, 4.0, 5.5)
    v = np.ones(N, dtype=np.float64)
    for j, f in enumerate(freqs):
        v += coeffs[j] * np.sin(2.0 * np.pi * f * t + phases[j])
    return normalize_pos(np.maximum(v, 0.0))


def smooth_perturb(arr):
    N = int(arr.size)
    a = np.asarray(arr, dtype=np.float64).copy()
    if N < 2:
        return normalize_pos(a)
    t = np.linspace(0.0, 1.0, N, dtype=np.float64)
    amp = float(_GLOBAL_RNG.uniform(0.08, 0.45))
    f1 = float(_GLOBAL_RNG.uniform(0.5, 3.0))
    f2 = float(_GLOBAL_RNG.uniform(3.0, 8.0))
    p1 = float(_GLOBAL_RNG.uniform(0.0, 2.0 * np.pi))
    p2 = float(_GLOBAL_RNG.uniform(0.0, 2.0 * np.pi))
    mod = 1.0 + amp * np.sin(2.0 * np.pi * f1 * t + p1) + 0.55 * amp * np.sin(2.0 * np.pi * f2 * t + p2)
    a *= np.maximum(mod, 0.05)

    width = max(2, N // 80)
    st = int(_GLOBAL_RNG.integers(0, max(1, N - width + 1)))
    a[st:st + width] *= float(np.exp(_GLOBAL_RNG.normal(0.0, 0.30)))

    return normalize_pos(a)


def full_refine(a0, budget_s):
    a = normalize_pos(np.asarray(a0, dtype=np.float64))
    N = int(a.size)
    if N < 2:
        return a, fast_score(a)

    deadline = time.monotonic() + max(0.0, float(budget_s))
    conv_len = fft_len_conv(N)
    grad_len = fft_len_grad(N)

    best = a.copy()
    best_s = fast_score(a, conv_len)
    eta = 1.0
    stag = 0

    while time.monotonic() < deadline:
        conv = self_conv(a, conv_len)
        cmax = float(np.max(conv))
        s = 2.0 * float(N) * cmax
        if not np.isfinite(s):
            break
        if s < best_s:
            best = a.copy()
            best_s = float(s)
            stag = 0

        if cmax <= 1e-300:
            break

        w, _ = weights_from_conv(conv, 20.0)
        g = grad_from_weights(a, w, grad_len)
        if not np.all(np.isfinite(g)):
            g = np.where(np.isfinite(g), g, 0.0)
        gm = g - float(np.mean(g))
        gnorm = float(np.max(np.abs(gm)))
        if not np.isfinite(gnorm) or gnorm <= 1e-18:
            a = smooth_perturb(best)
            eta = 1.0
            stag += 1
            if stag > 3:
                a = best.copy()
                stag = 0
            continue

        base_step = eta / gnorm
        base_step = min(float(base_step), 20.0)

        current_s = float(s)
        best_cand = None
        best_cand_s = current_s
        best_factor = 1.0

        for factor in (1.0, 0.5, 2.0, 4.0):
            if time.monotonic() > deadline + 0.2:
                break
            step = base_step * float(factor)
            z = np.log(a + 1e-300) - step * gm
            if not np.all(np.isfinite(z)):
                continue
            z -= float(np.max(z))
            cand = np.exp(z)
            cs = float(np.sum(cand))
            if not np.isfinite(cs) or cs <= 1e-300:
                continue
            cand = cand / cs
            cand_s = fast_score(cand, conv_len)
            if np.isfinite(cand_s) and cand_s < best_cand_s:
                best_cand = cand
                best_cand_s = float(cand_s)
                best_factor = float(factor)

        if best_cand is not None and best_cand_s < current_s * (1.0 - 1e-12):
            a = best_cand
            stag = 0
            if best_factor >= 1.0:
                eta = min(eta * 1.15, 8.0)
            else:
                eta = max(eta * 0.95, 0.1)
        else:
            eta = max(eta * 0.5, 0.05)
            stag += 1
            if stag > 8 or eta < 0.07:
                a = smooth_perturb(best)
                eta = 1.0
                stag = 0

    return best, float(best_s)

            pool[i], pool[i - 1] = pool[i - 1], pool[i]
    if len(pool) > 12:
        pool.pop()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    global _GLOBAL_RNG
    _GLOBAL_RNG = np.random.default_rng(int(seed) if seed is not None else 20240527)

    t0 = time.monotonic()
    overall_deadline = t0 + max(1.0, float(budget_s) - 20.0)

    N = choose_length()
    pool = []

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size >= 2 and np.sum(np.isfinite(parent)) >= 2:
            parent = np.clip(parent, 0.0, 1000.0)
            s = fast_score(parent)
            update_pool(pool, parent, s)
            print("parent internal score", s)
    except Exception:
        pass

    for init in (np.ones(N, dtype=np.float64), two_block_sequence(N), ramp_sequence(N, True), ramp_sequence(N, False)):
        s = fast_score(init)
        update_pool(pool, init, s)

    print("base search N", N)
    last_base = None
    configs = [(16, 10.0), (32, 25.0), (64, 80.0), (128, 130.0), (256, 120.0)]
    for K, want in configs:
        if time.monotonic() > overall_deadline:
            break
        want = min(want, overall_deadline - time.monotonic())
        if want < 1.0:
            break
        extra = []
        if last_base is not None:
            extra.append(upsample_base(last_base, K))
        m, ms = search_base(K, time.monotonic() + want, extra)
        if m is not None:
            last_base = m
            arr = expand_base(m, N)
            s = fast_score(arr)
            update_pool(pool, arr, s)
            print("base K", K, "base score", ms, "expanded score", s)

    print("internal top before refine", pool[0][0] if pool else None)

    # Refine the best expanded/base candidate.
    if pool and time.monotonic() < overall_deadline - 60.0:
        arr = pool[0][1]
        want = min(900.0, overall_deadline - time.monotonic() - 30.0)
        if want > 60.0:
            refined, rs = full_refine(arr, want)
            update_pool(pool, refined, rs)
            print("refined best expansion", rs)

    # Refine one asymmetric direct candidate.
    if time.monotonic() < overall_deadline - 40.0:
        arr = two_block_sequence(N)
        want = min(240.0, overall_deadline - time.monotonic() - 20.0)
        if want > 30.0:
            refined, rs = full_refine(arr, want)
            update_pool(pool, refined, rs)
            print("refined two-block", rs)

    # Refine a random smooth candidate if time remains.
    if time.monotonic() < overall_deadline - 40.0:
        arr = random_direct_sequence(N)
        want = min(160.0, overall_deadline - time.monotonic() - 20.0)
        if want > 20.0:
            refined, rs = full_refine(arr, want)
            update_pool(pool, refined, rs)
            print("refined random smooth", rs)

    if pool and time.monotonic() < overall_deadline - 5.0:
        arr = pool[0][1]
        want = min(120.0, overall_deadline - time.monotonic() - 10.0)
        if want > 10.0:
            refined, rs = full_refine(arr, want)
            update_pool(pool, refined, rs)
            print("final polish", rs)

    print("pool top internal", pool[0][0] if pool else None)

    best_list = None
    best_exact = float("inf")
    for _, arr in pool:
        try:
            lst = [float(x) for x in arr]
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < best_exact:
                best_exact = sc
                best_list = lst
        except Exception:
            pass

    if best_list is None:
        if pool:
            best_list = [float(x) for x in pool[0][1]]
        else:
            best_list = [1.0 / float(N)] * N

    try:
        print("chosen internal score", fast_score(best_list))
    except Exception:
        pass
    try:
        print("chosen official score", evaluate_sequence(best_list))
    except Exception:
        pass

    return best_list
