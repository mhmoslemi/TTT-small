import numpy as np
import time

def gs_flatten(a, n_iter=100, target=None):
    """Gerchberg-Saxton style iteration to flatten autocorrelation."""
    n = len(a)
    # Zero-pad to next power of 2 >= 2n-1
    L = 1
    while L < 2*n - 1:
        L *= 2
    a_padded = np.zeros(L)
    a_padded[:n] = a
    for _ in range(n_iter):
        A = np.fft.rfft(a_padded)
        mag = np.abs(A)
        if target is None:
            # Flat target: set all magnitudes to the mean
            target_mag = np.mean(mag[1:])  # exclude DC
            new_mag = np.ones_like(mag)
            new_mag[0] = mag[0]  # preserve DC (sum)
            new_mag[1:] = target_mag
        else:
            new_mag = target_mag
        # Preserve phase
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, L)
        a_padded = np.maximum(0, a_new[:n])

def gs_flatten(a_init, n_iter=200, L=None):
    """Gerchberg-Saxton iteration to flatten the autocorrelation spectrum."""
    n = len(a_init)
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L *= 2
    
    a_pad = np.zeros(L)
    a_pad[:n] = a_init
    
    for _ in range(n_iter):
        A = np.fft.rfft(a_pad)
        mag = np.abs(A)
        # Target: flat magnitude (except DC)
        target_mag = np.full_like(mag, np.mean(mag[1:]))
        target_mag[0] = mag[0]  # Preserve DC component (proportional to sum)
        
        phase = np.angle(A)
        new_A = target_mag * np.cos(phase) + 1j * target_mag * np.sin(phase)
        a_new = np.fft.irfft(new_A, L)
        a_pad = np.maximum(0.0, a_new[:n])
    
    return a_pad

∇f_smooth = (1/sum_k exp(T*g_k)) * sum_k exp(T*g_k) * ∇g_k

This is a weighted average of the individual gradients, with weights proportional to exp(T*g_k). So the lags with larger values get more weight.

    """Compute full linear convolution using FFT."""
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L *= 2
    A = np.fft.rfft(a, L)

def gs_flatten(a_init, n_iter=500, L=None):
    """Gerchberg-Saxton iteration for flat autocorrelation."""
