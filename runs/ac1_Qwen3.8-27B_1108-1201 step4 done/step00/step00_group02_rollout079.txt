"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The main idea is scale-invariant normalization (sum(a)=1), very fast FFT-based
convolutions, and several complementary local/global operators:

1. Spectral shaping:
   For current a, form a target convolution c_target by capping the current
   convolution at a decreasing peak level and re-normalizing to sum 1.
   Then take a Newton-like spectral step in the DFT domain:
       X^2 = FFT(a lifted)
       X_new ~= X * sqrt(FFT(c_target) / FFT(a*a))
   followed by projection back to non-negative length-n vectors.

2. Stochastic annealing around the spectral step (small blends, rejection of
   very bad steps, occasional Fourier/log-spectrum perturbations).

3. Active linear-programming directions for medium and large lengths:
   solve (or approximate)
       maximize sum(g)
       subject to (a*g)_k <= M for selected important k, g >= 0
   and blend the current sequence with g.

4. A few diverse deterministic/random initial seeds and length sweep.

The exact evaluator is assumed to be already available in the environment and
is called only once at the end for validation.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    from scipy import sparse
    HAVE_SCIPY_LP = True
except Exception:
    HAVE_SCIPY_LP = False


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def safe_normalize(x, target=1.0):
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0:
        x = np.ones(256, dtype=float)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-12:
        return np.ones(x.size, dtype=float) * (target / x.size)
    return x * (target / s)


def convolve_full(x):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return np.array([], dtype=float)
    m = 2 * n - 1
    N = next_pow2(m)
    F = np.fft.rfft(x, N)
    out = np.fft.irfft(F * F, N)[:m]
    out[~np.isfinite(out)] = 0.0
    out[out < 0.0] = 0.0
    return out


def fast_score(x):
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0:
        return float("inf")
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    x = np.minimum(x, 1000.0)
    s = float(x.sum())
    if s <= 1e-12:
        return float("inf")
    x = x / s
    c = convolve_full(x)
    if c.size == 0:
        return 0.0
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * x.size * mx)


def logsumexp(a):
    a = np.asarray(a, dtype=float)
    m = float(np.max(a))
    if not np.isfinite(m):
        return m
    return m + float(np.log(np.sum(np.exp(a - m))))


def random_fourier_init(n, rng):
    i = np.arange(n, dtype=float)
    x = np.ones(n, dtype=float)
    kmax = max(2, min(8, n // 4))
    for _ in range(kmax):
        f = int(rng.integers(1, kmax + 1))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.05, 1.0))
        x += amp * np.cos((2.0 * np.pi * f * i) / max(1, n) + phase)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-12:
        x = np.ones(n, dtype=float)
    else:
        x = x / s
    return x


def make_inits(n, rng, parent=None, num=4):
    inits = []

    if parent is not None:
        p = np.asarray(parent, dtype=float).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size >= 2 and float(p.sum()) > 0.0:
            xp = np.linspace(0.0, 1.0, n)
            fp = np.interp(xp, np.linspace(0.0, 1.0, p.size), p)
            fp = np.maximum(fp, 0.0)
            s = float(fp.sum())
            if s > 0.0:
                inits.append(fp / s)

    i = np.arange(n, dtype=float)
    for a, sign in ((0.6, 1.0), (0.8, -1.0)):
        x = 1.0 + sign * a * np.cos((2.0 * np.pi * i) / max(1, n))
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s > 0.0:
            inits.append(x / s)

    x = rng.exponential(1.0, n) + 1e-8
    inits.append(x / x.sum())

    inits.append(random_fourier_init(n, rng))

    d = float(rng.choice([0.35, 0.5, 0.65]))
    x = (rng.random(n) < d).astype(float)
    x += 0.05 * rng.normal(0.0, 1.0, n)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s > 0.0:
        inits.append(x / s)

    while len(inits) < num:
        x = rng.random(n) + 0.2
        inits.append(x / x.sum())

    return inits[:num]


def cap_normalize_c(c, cap):
    c = np.asarray(c, dtype=float).ravel()
    L = c.size
    if L == 0:
        return np.array([], dtype=float)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 0.0:
        return np.full(L, 1.0 / L, dtype=float)
    c = c / s

    cap = float(cap)
    if not np.isfinite(cap) or cap <= 0.0:
        cap = float(c.max())
    cap = max(cap, 1.0 / L + 1e-15)
    cap = min(cap, 1.0)

    t = np.minimum(c, cap)
    for _ in range(8):
        res = 1.0 - float(t.sum())
        if abs(res) <= 1e-13:
            break
        if res > 0.0:
            room = cap - t
            room[room < 0.0] = 0.0
            tot = float(room.sum())
            if tot <= 0.0:
                break
            t = t + res * room / tot
        else:
            amt = t.copy()
            tot = float(amt.sum())
            if tot <= 0.0:
                break
            t = np.maximum(t + res * amt / tot, 0.0)  # res is negative

    t = np.where(np.isfinite(t), t, 0.0)
    t = np.maximum(t, 0.0)
    t = np.minimum(t, cap + 1e-12)
    s = float(t.sum())
    if s <= 0.0:
        return np.full(L, 1.0 / L, dtype=float)
    return t / s


def enforce_hermitian(X):
    X = np.asarray(X, dtype=complex)
    N = X.size
    if N == 0:
        return X
    Xh = X.copy()
    Xh[0] = Xh[0].real
    half = N // 2
    if N % 2 == 0:
        Xh[half] = Xh[half].real
        idx = np.arange(1, half)
        if idx.size:
            Xh[N - idx] = np.conj(Xh[idx])
    else:
        idx = np.arange(1, half + 1)
        if idx.size:
            Xh[N - idx] = np.conj(Xh[idx])
    return Xh


def spectral_step(x, target, rho=0.8, clip_m=10.0, eps=1e-10):
    x = np.asarray(x, dtype=float).ravel()
    target = np.asarray(target, dtype=float).ravel()
    n = x.size
    m = 2 * n - 1
    N = next_pow2(m)

    xpad = np.zeros(N, dtype=float)
    xpad[:n] = x
    X = np.fft.fft(xpad)

    tpad = np.zeros(N, dtype=float)
    if target.size >= m:
        tpad[:m] = target[:m]
    else:
        tpad[:target.size] = target
    T = np.fft.fft(tpad)

    C = X * X
    scale = max(1.0, float(np.max(np.abs(T))))
    Csafe = C + eps * scale

    ratio = T / Csafe
    with np.errstate(all="ignore"):
        sq = np.sqrt(ratio)
    mag = np.abs(sq)
    clipped = np.clip(mag, 1.0 / clip_m, clip_m)
    scaled = np.zeros_like(sq)
    mask = mag > 0.0
    if mask.any():
        scaled[mask] = sq[mask] * (clipped[mask] / mag[mask])

    Xp = X * scaled
    if rho < 1.0:
        Xp = (1.0 - rho) * X + rho * Xp
    elif rho > 1.0:
        Xp = X + (rho - 1.0) * Xp

    Xh = enforce_hermitian(Xp)
    xraw = np.fft.ifft(Xh).real
    x2 = xraw[:n]
    x2 = np.where(np.isfinite(x2), x2, 0.0)
    x2 = np.maximum(x2, 0.0)
    s = float(x2.sum())
    if not np.isfinite(s) or s <= 1e-8:
        return safe_normalize(x)
    return x2 / s


def fourier_perturb(x, rng, amp=0.2):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    N = next_pow2(n)
    X = np.fft.rfft(x, N)
    phase = rng.uniform(-amp, amp, X.shape)
    Xp = X * np.exp(1j * phase)
    x2 = np.fft.irfft(Xp, N)[:n]
    x2 = np.where(np.isfinite(x2), x2, 0.0)
    x2 = np.maximum(x2, 0.0)
    s = float(x2.sum())
    if s <= 1e-8:
        return safe_normalize(x)
    return x2 / s


def make_weights(c, mode, focus, tau):
    c = np.asarray(c, dtype=float).ravel()
    m = c.size
    if m == 0:
        return np.array([], dtype=float)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.ones(m, dtype=float) / m
    q = c / cmax
    if mode == 0:
        z = focus * q
    elif mode == 1:
        z = np.maximum(q - tau, 0.0)
        z = z * z
    else:
        z = np.maximum(q - tau, 0.0)
    s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        z = focus * q
        s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(m, dtype=float) / m
    return z / s


def correlate_grad(w, x):
    w = np.asarray(w, dtype=float).ravel()
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    L = next_pow2(w.size + n - 1)
    W = np.fft.rfft(w, L)
    X = np.fft.rfft(x[::-1], L)
    z = np.fft.irfft(W * X, L)[n - 1:n - 1 + n]
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.maximum(z, 0.0)
    return 2.0 * z


def gradient_candidate(x, c, rng, eta=0.25, noise=0.08, focus=20.0, tau=0.8):
    x = np.asarray(x, dtype=float).ravel()
    w = make_weights(c, 1, focus, tau)
    g = correlate_grad(w, x)
    gm = float(np.max(g))
    if gm <= 0.0 or not np.isfinite(gm):
        return safe_normalize(x)
    logx = np.log(np.maximum(x, 1e-300))
    step = -eta * (g / gm) + noise * rng.standard_normal(x.size)
    step = np.clip(step, -3.0, 3.0)
    logx = logx + step
    logx = logx - logsumexp(logx)
    x2 = np.exp(logx)
    return safe_normalize(x2)


def spectral_optimize(x, time_limit, rng, target_score=1.45):
    x = safe_normalize(x)
    n = x.size
    if n == 0:
        return x, float("inf")

    c = convolve_full(x)
    score = float(2.0 * n * float(np.max(c))) if c.size else float("inf")
    best = x.copy()
    best_score = score

    if not np.isfinite(time_limit) or time_limit <= 0.0:
        return best, best_score

    desired_cap = max(target_score / (2.0 * n), 1.0 / (2.0 * n - 1.0) + 1e-15)
    cap = score / (2.0 * n) if np.isfinite(score) and score > 0.0 else 1.0 / (2.0 * n - 1.0)
    rho = 0.8
    no_improve = 0
    it = 0
    start = time.monotonic()
    last_print = 0.0

    while time.monotonic() - start < time_limit:
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not np.isfinite(score):
            break

        mx = score / (2.0 * n)
        if no_improve > 250:
            cap = mx * 0.99
        else:
            cap = max(desired_cap, min(cap, mx * 0.99))
        cap = max(cap, 1.0 / (2.0 * n - 1.0) + 1e-15)

        target = cap_normalize_c(c, cap)
        x2 = spectral_step(x, target, rho)
        c2 = convolve_full(x2)
        sc2 = float(2.0 * n * float(np.max(c2))) if c2.size else float("inf")

        if not np.isfinite(sc2) or sc2 >= score:
            for b in (0.5, 0.25, 0.1):
                xb = (1.0 - b) * x + b * x2
                xb = safe_normalize(xb)
                cb = convolve_full(xb)
                scb = float(2.0 * n * float(np.max(cb))) if cb.size else float("inf")
                if np.isfinite(scb) and (not np.isfinite(sc2) or scb < sc2):
                    x2, c2, sc2 = xb, cb, scb

        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best = x2.copy()
            no_improve = 0

        ok = False
        if np.isfinite(sc2):
            if sc2 < score:
                ok = True
            elif sc2 < score + max(0.01, 0.02 * max(1.0, score)) and rng.random() < 0.08:
                ok = True

        if ok:
            x, c, score = x2, c2, sc2
            rho = min(0.95, rho * 1.01)
        else:
            rho = max(0.35, rho * 0.95)

        if no_improve > 900:
            if rng.random() < 0.7:
                x = fourier_perturb(x, rng, 0.22)
            else:
                x = gradient_candidate(x, c, rng, eta=0.35, noise=0.12, focus=25.0, tau=0.75)
            c = convolve_full(x)
            score = float(2.0 * n * float(np.max(c))) if c.size else float("inf")
            cap = max(desired_cap, score / (2.0 * n) if np.isfinite(score) else 1.0 / (2.0 * n - 1.0))
            no_improve = 0
            rho = 0.8

        it += 1
        now = time.monotonic()
        if (it % 800 == 0 and now - last_print > 25.0) or (no_improve % 2000 == 0 and now - last_print > 25.0):
            print("spectral n=%d score=%.6f best=%.6f cap=%.6g it=%d t=%.1f" %
                  (n, score if np.isfinite(score) else float("nan"), best_score, cap, it, now - start))
            last_print = now

    return best, best_score


def solve_active_lp(f, M, rows):
    if not HAVE_SCIPY_LP:
        return None
    f = np.asarray(f, dtype=float).ravel()
    n = f.size
    K = rows.size
    if K <= 0 or n <= 0:
        return None

    row_lists = []
    col_lists = []
    data_lists = []
    for r in range(K):
        k = int(rows[r])
        if k < 0 or k > 2 * n - 2:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo > hi:
            continue
        cols = np.arange(lo, hi + 1, dtype=np.int32)
        coeffs = f[k - cols]
        row_idx = np.full(cols.shape, r, dtype=np.int32)
        row_lists.append(row_idx)
        col_lists.append(cols)
        data_lists.append(coeffs)

    if len(row_lists) == 0:
        A = sparse.csr_matrix((K, n))
    else:
        rr = np.concatenate(row_lists)
        cc = np.concatenate(col_lists)
        dd = np.concatenate(data_lists)
        mask = np.isfinite(dd)
        A = sparse.csr_matrix((dd[mask], (rr[mask], cc[mask])), shape=(K, n))

    obj = -np.ones(n, dtype=float)
    b = float(M) * np.ones(K, dtype=float)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:
        return None

    if res is not None and getattr(res, "success", False) and getattr(res, "x", None) is not None:
        g = np.asarray(res.x, dtype=float)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s > 1e-12:
            return g
    return None


def lp_active_optimize(x, time_limit, rng, target_score=1.45, K=250):
    if not HAVE_SCIPY_LP:
        return safe_normalize(x), fast_score(x)

    x = safe_normalize(x)
    n = x.size
    c = convolve_full(x)
    score = float(2.0 * n * float(np.max(c))) if c.size else float("inf")
    best = x.copy()
    best_score = score

    if not np.isfinite(time_limit) or time_limit <= 0.0:
        return best, best_score

    start = time.monotonic()
    calls = 0
    no_improve = 0
    Kmax = int(K)

    while time.monotonic() - start < max(10.0, time_limit - 3.0):
        c = convolve_full(x)
        score = float(2.0 * n * float(np.max(c))) if c.size else float("inf")
        if not np.isfinite(score):
            break
        if score < best_score - 1e-12:
            best_score = score
            best = x.copy()
            no_improve = 0

        M = float(np.max(c)) * (1.0 + 1e-9)
        if M <= 0.0 or not np.isfinite(M):
            break

        Kuse = min(Kmax, c.size)
        rows = np.argsort(c)[-Kuse:]
        g = solve_active_lp(x, M, rows)
        if g is None:
            break
        g = safe_normalize(g)

        improved = False
        for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            x2 = (1.0 - t) * x + t * g
            x2 = safe_normalize(x2)
            c2 = convolve_full(x2)
            sc2 = float(2.0 * n * float(np.max(c2))) if c2.size else float("inf")
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best = x2.copy()
                x = x2
                c = c2
                score = sc2
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 4:
                break

        calls += 1
        if calls >= 30:
            break

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    max_time = max(20.0, budget - 12.0)
    deadline = start + max_time
    rng = np.random.default_rng(seed)

    try:
        parent_raw = height_sequence_1
        if parent_raw is None:
            parent = None
        else:
            parent = np.asarray(parent_raw, dtype=float).ravel()
            parent = np.where(np.isfinite(parent), parent, 0.0)
            parent = np.maximum(parent, 0.0)
            if parent.size == 0:
                parent = None
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(6962, dtype=float)

    pnorm = safe_normalize(parent)
    best_seq = pnorm.copy()
    best_score = fast_score(pnorm)
    print("initial parent len=%d fast_score=%.6f" % (best_seq.size, best_score))

    # First: spend meaningful time improving the provided/parent length.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        parent_time = min(420.0, rem * 0.35)
        bseq, bscore = spectral_optimize(pnorm, parent_time, rng, target_score=1.45)
        if np.isfinite(bscore) and bscore < best_score:
            best_score = bscore
            best_seq = bseq.copy()
            print("parent improved fast_score=%.6f" % best_score)

        if best_score > 1.55 and time.monotonic() < deadline - 60.0:
            lp_time = min(300.0, max(30.0, deadline - time.monotonic() - 60.0))
            K = 180 if best_seq.size > 3000 else 300
            lseq, lscore = lp_active_optimize(best_seq, lp_time, rng, target_score=1.45, K=K)
            if np.isfinite(lscore) and lscore < best_score:
                best_score = lscore
                best_seq = lseq.copy()
                print("parent LP improved fast_score=%.6f" % best_score)

    # Sweep useful lengths.
    lengths = [256, 512, 1024, 2048, 4096]
    for n in lengths:
        if time.monotonic() >= deadline - 30.0:
            break

        if n <= 512:
            alloc = 90.0
        elif n <= 1024:
            alloc = 120.0
        elif n <= 2048:
            alloc = 150.0
        else:
            alloc = 180.0
        alloc = min(alloc, max(30.0, deadline - time.monotonic() - 30.0))
        if alloc <= 0.0:
            break

        inits = make_inits(n, rng, parent, num=3)
        per = alloc / float(len(inits))

        for x0 in inits:
            if time.monotonic() >= deadline - 30.0:
                break
            bseq, bscore = spectral_optimize(x0, per, rng, target_score=1.45)
            if np.isfinite(bscore) and bscore < best_score:
                best_score = bscore
                best_seq = bseq.copy()
                print("length %d improved fast_score=%.6f" % (n, best_score))

            if n <= 4096 and np.isfinite(bscore) and bscore < best_score + 0.35:
                rem2 = min(60.0, max(20.0, deadline - time.monotonic() - 15.0))
                if rem2 > 20.0:
                    K = min(300, max(50, n))
                    lseq, lscore = lp_active_optimize(bseq, rem2, rng, target_score=1.45, K=K)
                    if np.isfinite(lscore) and lscore < best_score:
                        best_score = lscore
                        best_seq = lseq.copy()
                        print("length %d LP improved fast_score=%.6f" % (n, best_score))

    # Final polishing on the best sequence found.
    rem = deadline - time.monotonic()
    if rem > 40.0:
        if best_seq.size <= 4096 and rem > 200.0:
            spec_time = rem * 0.65
            lp_time = rem * 0.35
        else:
            spec_time = rem * 0.88
            lp_time = rem * 0.12

        bseq, bscore = spectral_optimize(best_seq, spec_time, rng, target_score=1.40)
        if np.isfinite(bscore) and bscore < best_score:
            best_score = bscore
            best_seq = bseq.copy()
            print("focus spectral fast_score=%.6f" % best_score)

        if lp_time > 20.0 and best_seq.size <= 4096:
            lseq, lscore = lp_active_optimize(best_seq, lp_time, rng, target_score=1.40, K=250)
            if np.isfinite(lscore) and lscore < best_score:
                best_score = lscore
                best_seq = lseq.copy()
                print("focus LP fast_score=%.6f" % best_score)

    # Finalize: cleanup, normalize, return a plain Python list.
    seq = np.asarray(best_seq, dtype=float).ravel()
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.maximum(seq, 0.0)
    if seq.size == 0:
        seq = np.ones(256, dtype=float)
    s = float(seq.sum())
    if s <= 1e-8:
        seq = np.ones(seq.size if seq.size > 0 else 256, dtype=float)
    else:
        seq = seq / s
    seq = np.minimum(seq, 1000.0)

    candidate = [float(v) for v in seq]

    # One exact validation call. The evaluator is assumed to be in scope.
    try:
        exact = evaluate_sequence(candidate)
        print("final exact_score=%s len=%d" % (exact, len(candidate)))
        if not np.isfinite(exact):
            fb = safe_normalize(parent)
            candidate = [float(v) for v in fb]
    except Exception as e:
        print("exact evaluation failed: %s" % str(e))

    return candidate
