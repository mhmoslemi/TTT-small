"""
Minimize  2*n*max(conv(a,a)) / (sum(a))^2  over non-negative a.

Key idea: for *symmetric* non-negative sequences the centre autocorrelation
equals sum(a_i^2), which is minimised by the uniform vector (score = 2).
Therefore any improvement below 2 requires an *asymmetric* sequence.

Algorithm
  1. Generate a family of asymmetric starting vectors (exponential decays,
     ramps, half-windows, gamma / random-walk shapes, and the supplied parent).
  2. Minimise a smooth-max surrogate  tau*log(sum exp(C_k/tau))  of
     max_k C_k  with a projected Adam step onto the probability simplex,
     annealing tau down over the run.  Gradients are computed with O(n log n)
     FFT cross-correlations.
  3. Occasionally polish a candidate with the LP direction from the prompt
     (max sum(b) s.t. conv(a,b)_k <= M, b>=0) and a small convex blend,
     which is beneficial whenever the LP's total mass S_b > 1.
  4. Track the best (lowest evaluated) sequence and do a final long refinement
     on it.  The parent is used as a warm start (after a symmetry-breaking
     tilt) so the search is not trapped in the symmetric basin.
"""

import numpy as np
import time

# --------------------------------------------------------------------------- #
#  FFT convolution / cross-correlation helpers                                #
# --------------------------------------------------------------------------- #

def fft_convolve(a, b):
    """Linear (full) convolution of two 1-D real arrays via rFFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def perplex_conv(a):
    """convolve(a, a) via FFT."""
    return fft_convolve(a, a)


def simplex_project(v):
    """Euclidean projection of v onto the probability simplex; returns sum==1."""
    n = len(v)
    if n == 0:
        return np.ones(1)
    if not np.any(v > 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 0:
        return np.ones(n) / n
    return out / s


def smooth_max_grad(a, tau):
    """
    Gradient of  tau*log(sum_k exp(C_k/tau))  w.r.t. a,  C = a*a (linear conv).

    Returns (grad, C).  grad has the same length as a.
    """
    C = perplex_conv(a)
    n = len(a)
    cmax = C.max()
    w = np.exp((C - cmax) / tau)
    w /= w.sum()
    # grad_i = 2 * sum_j w[i+j] * a[j]  = 2 * (w_rev * a)[i + 2n - 2]
    w_rev = w[::-1]
    conv = fft_convolve(w_rev, a)          # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * conv[(2 * n - 2):(3 * n - 2)]
    return grad, C

#  Core optimiser: projected smooth-max Adam                                   #

def optimize_smooth(a0, time_budget, rng, do_lp=False):
    """
    Minimise max(conv(a,a)) starting from a0 (normalised to sum 1, a0>=0).
    Uses an Adam step onto the simplex + temperature annealing, with optional
    LP polish.  Returns the best a found (sum 1).
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n)
    a /= a.sum()

    C0 = perplex_conv(a)
    c0 = float(C0.max())
    best = a.copy()
    best_c = c0

    tau0 = max(c0 * 0.1, 1e-12)
    tau_min = max(c0 * 1e-4, 1e-14)

    m = np.zeros(n)
    v = np.zeros(n)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    lr = 0.1
    n_iter_max = 200000

    t0 = time.monotonic()
    it = 0
    fp_iters = 0

    while (time.monotonic() - t0) < time_budget and it < n_iter_max:
        it += 1
        # temperature anneal over the bounds of time_budget
        prog = min(1.0, (time.monotonic() - t0) / max(1e-9, time_budget))
        tau = tau0 * (tau_min / tau0) ** prog

        grad, C = smooth_max_grad(a, tau)
        c_now = float(C.max())

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = simplex_project(a)

        # track best (cheap-ish every few iters)
        if it % 20 == 0:
            cc = perplex_conv(a).max()
            if cc < best_c:
                best_c = cc
                best = a.copy()

        # occasional LP polish on a copy of the current best
        if do_lp and n <= 1200 and fp_iters < 30 and it % 800 == 0:
            fp_iters += 1
            cand = lp_direction_polish(best)
            cc = float(perplex_conv(cand).max())
            if cc < best_c:
                best_c = cc
                best = cand
                a = cand.copy()
                m[:] = 0.0
                v[:] = 0.0

    # final exact tracking
    cc = float(perplex_conv(best).max())
    if cc < best_c:
        best_c = cc
    return best

#  LP direction polish (from the prompt): only used for moderate n             #

def lp_direction_polish(a):
    """
    a -> (1-t)a + t*(b/sum(b))  where b solves  max sum(b) s.t. conv(a,b)<=M,
    b>=0.  Blending is beneficial iff sum(b) > 1.  Returns a (sum 1) sequence.
    """
    import scipy.sparse as sp
    from scipy.optimize import linprog

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    C = perplex_conv(a)
    M = float(C.max())

    # A[k, j] = a[k-j] if 0 <= k-j < n else 0 ;  k = 0..2n-2, j = 0..n-1
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    res = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(2 * n - 1, M),
        bounds=[(0.0, None)] * n,
        method="highs",
    )
    if res.status != 0:
        return a.copy()

    b = np.clip(res.x, 0.0, None)
    sb = b.sum()
    if not (np.isfinite(sb)) or sb <= 1.001:
        return a.copy()

    bnorm = b / sb
    for t in (0.1, 0.25, 0.5, 0.75):
        cand = simplex_project((1.0 - t) * a + t * bnorm)
        cc = float(perplex_conv(cand).max())
        if cc < best_peak(a):
            a = cand
    return a


def best_peak(a):
    return float(perplex_conv(a).max())

def make_starts(n, rng, parent):
    starts = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # exponential decays (forward & backward) -- the main asymmetric family
    for rate in (0.5, 1.0, 1.6, 2.4, 3.5, 5.0, 8.0):
        starts.append(np.exp(-rate * x / n))
        starts.append(np.exp(-rate * (n - 1 - x) / n))

    # power-law / polynomial ramps
    for p in (0.5, 1.0, 1.5, 2.0):
        starts.append((x + 1.0) ** p)
        starts.append((n - x) ** p)

    # half-windows (mass on one side only)
    half = max(2, n // 2)
    hh = np.arange(half, dtype=np.float64)
    w_rect = np.zeros(n); w_rect[:half] = 1.0
    starts.append(w_rect)
    w_hann = np.zeros(n)
    w_hann[:half] = 0.5 * (1.0 + np.cos(2 * np.pi * hh / max(1, half - 1)))
    starts.append(w_hann)
    w_tri = np.zeros(n)
    w_tri[:half] = 1.0 - hh / max(1.0, half)
    starts.append(w_tri)
    # mirror to the other side
    starts.append(w_rect[::-1].copy())
    starts.append(w_hann[::-1].copy())
    starts.append(w_tri[::-1].copy())

    # random gamma / exponential shapes (asymmetric, smooth)
    for _ in range(3):
        shape = rng.gamma(shape=1.0 + 4.0 * rng.random(), scale=1.0, size=n)
        starts.append(shape)
        starts.append(shape[::-1].copy())

    # random asymmetric log-linear tilt on top of a window
    win = 0.5 * (1.0 + np.cos(2 * np.pi * x / max(1, n - 1)))
    slope = rng.uniform(1.5, 4.0)
    starts.append(win * np.exp(-slope * x / n))
    starts.append(win * np.exp(-slope * (n - 1 - x) / n))

    # parent, warm-started with a symmetry-breaking tilt if it fits
    if parent is not None:
        if parent.size == n:
            p = parent.copy()
        elif parent.size > n:
            p = parent[:n].copy()
        else:
            p = np.pad(parent, (0, n - parent.size), mode="edge")
        p = np.clip(p, 0.0, None)
        if p.sum() > 0:
            starts.append(p * np.exp(-1.3 * x / n))
            starts.append(p * np.exp(-1.3 * (n - 1 - x) / n))
            starts.append(p * np.exp(-0.6 * x / n))

    # normalise each
    out = []
    for stt in starts:
        stt = np.asarray(stt, dtype=np.float64)
        stt = np.clip(stt, 0.0, None)
        s = stt.sum()
        if s < 1e-10 or not np.all(np.isfinite(stt)):
            continue
        out.append(stt / s)
    return out

                break
            do_lp = (n <= 1000)
            aopt = optimize_smooth(stt, slice_t, rng, do_lp=do_lp)
            consider(aopt)

    # final long refinement on the best sequence found so far
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        ref = optimize_smooth(best_seq, rem, rng, do_lp=(len(best_seq) <= 1200))
        consider(ref)

    consider(best_seq)  # ensure a finite, valid return
    return [float(v) for v in best_seq]
