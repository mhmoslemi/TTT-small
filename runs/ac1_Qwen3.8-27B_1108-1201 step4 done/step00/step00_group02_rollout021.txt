import numpy as np
from scipy.optimize import linprog
import time

def solve_lp_direction(f, M):
    """Solve the LP to find direction g that maximizes sum(g) subject to ||f*g||_inf <= M, g >= 0."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f * g)[k] <= M for k = 0, ..., 2n-2
    # (f * g)[k] = sum_i f[i] * g[k-i] where valid
    # For a given k, the coefficient of g[j] is f[k-j] if valid.
    # g[j] appears in constraint k when 0 <= k-j < n, i.e., j <= k < j+n, i.e., k-j is in [0, n-1].
    # So the coefficient of g[j] in constraint k is f[k-j] if 0 <= k-j < n, else 0.
    
    # Build the constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # This is a banded Toeplitz-like matrix.
    # For efficiency, let's build it as a sparse matrix.
    from scipy.sparse import lil_matrix
    
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        # g[j] appears with coefficient f[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A[k, j] = f[k-j]
    
    A = A.tocsr()
    b_constraint = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_constraint, bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    return result.x
