    """Solve max sum(x) s.t. (a*x)[k] <= M for all k, x >= 0."""
    active = [np.argmax(np.convolve(a, a))]  # Start with the peak
    
    for _ in range(max_iterations):
        # Build LP with current active constraints
        x = solve_lp(a, active, M, n)
        if x is None:
            return None
        
        # Check feasibility against ALL constraints
        ax = np.convolve(a, x)
        violations = np.where(ax > M * (1 + 1e-9))[0]

            return x  # Feasible!
        
        # Add violating constraints (limit to top 20)
        if len(violations) > 20:
            violations = violations[np.argsort(ax[violations])[::-1][:20]]
        active = list(set(active + list(violations)))

            cols.extend(js.tolist())
            vals.extend(a[js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

def solve_lp(a, active_indices, M, n):

    # Actually A_ub is just the matrix, M goes into b_ub

    return res.x if res.success else None

"""
LP-based cutting plane optimization for minimizing 2n*max(convolve(a,a))/sum(a)^2.
Uses iterative LP relaxations with constraint generation (cutting planes) and
blending to find sequences with flat autocorrelation profiles.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    """FFT-based linear convolution of two 1D arrays."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_score_fast(a):
    """Compute score = 2n*max(conv(a,a))/sum(a)^2 without calling evaluate_sequence."""
    n = len(a)
    b = fft_convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def build_constraint_matrix(a, active_indices, n):
    """Build sparse matrix A where A[i,j] = a[k-j] if valid, for constraint row i corresponding to position k=active_indices[i]."""
    rows = []
    cols = []
    vals = []
    n_active = len(active_indices)
    for i in range(n_active):
        k = active_indices[i]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            for jj in js:
                rows.append(i)
                cols.append(int(jj))
                vals.append(float(a[jj]))
    if not vals:
        return csr_matrix((n_active, n))
    return csr_matrix((vals, (rows, cols)), shape=(n_active, n))


def solve_lp_single(a, active_indices, M, n):
    """Solve max sum(x) s.t. (a*x)[k] <= M for k in active_indices, x >= 0."""
    A_ub = build_constraint_matrix(a, active_indices, n)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except:
        pass
    return None


def lp_with_cutting_planes(a, M, n, max_cuts=40):
    """
    Solve: max sum(x) s.t. (a*x)[k] <= M for ALL k in [0, 2n-2], x >= 0.
    Uses cutting plane method: start with peak constraint, add violated ones.
    """
    b = fft_convolve(a, a)
    peak_pos = int(np.argmax(b))
    active = [peak_pos]

    for _ in range(max_cuts):
        x = solve_lp_single(a, active, M, n)
        if x is None:
            return None

        # Check all constraints
        ax = fft_convolve(a, x)
        tol = M * 1e-7 + 1e-15
        viol = np.where(ax > M + tol)[0]

        if len(viol) == 0:
            return x

        # Add most violated constraints (up to 30 per round)
        if len(viol) > 30:
            top_idx = viol[np.argsort(ax[viol])[::-1][:30]]
        else:
            top_idx = viol
        new_active = list(set(active) | set(int(v) for v in top_idx))
        active = new_active

    # Return last LP solution as approximate
    x = solve_lp_single(a, active, M, n)
    return x


def optimize_sequence(a, deadline):
    """
    Iterative LP-based optimization.
    At each step: solve LP to find direction x, blend toward x.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s  # Normalize to sum 1

    best = a.copy()
    best_score = compute_score_fast(a)

    t = 0.3
    stall = 0
    iter_count = 0

    while time.monotonic() < deadline - 5:
        iter_count += 1
        b = fft_convolve(a, a)
        M = float(np.max(b))

        # Solve LP with cutting planes
        x = lp_with_cutting_planes(a, M, n, max_cuts=25)
        if x is None:
            stall += 3
            t = max(0.01, t * 0.7)
            if stall > 50:
                break
            continue

        sum_x = float(np.sum(x))
        if sum_x < 1e-12:
            stall += 3
            break
        x = x / sum_x  # Normalize x to sum 1

        # Try blending
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1, 0.05]:
            a_new = (1.0 - trial_t) * a + trial_t * x
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-6:
                continue
            a_new = a_new / s_new
            sc = compute_score_fast(a_new)
            if sc < best_score - 1e-12:
                a = a_new
                best = a_new.copy()
                best_score = sc
                improved = True

                break

        if iter_count % 50 == 0:
            print(f"    iter {iter_count}: score={best_score:.6f}, M={M:.6f}, t={t:.3f}")

    return best, best_score


def local_search_refinement(a, deadline, rng):
