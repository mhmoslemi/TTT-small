def _get_active(c, max_k=80, frac=0.90):
    """Get active lag indices: top max_k plus all lags above frac*max."""
    M = float(c.max())
    top = np.argsort(c)[::-1][:max_k]
    thr = frac * M
    extra = np.where(c >= thr)[0]
    active = np.unique(np.concatenate([top, extra]))
    if len(active) > max_k + 100:
        active = active[np.argsort(c[active])[::-1]][:max_k + 100]
    return active, M


def _lp_dir(f, active, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active, g >= 0."""
    n = len(f)
    K = len(active)
    A = np.zeros((K, n))
    for ii, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[ii, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    b = np.full(K, M)
    co = -np.ones(n)
    try:
        r = linprog(co, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                    options={"presolve": True, "time_limit": 5.0})
        if r.success or (r.status == 1 and r.x is not None):
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def _perturb_smooth(f, rng, scale=0.1):

    n = len(f)
    freq = int(rng.integers(1, 80))
    phase = rng.uniform(0, 2.0 * np.pi)
    mod = 1.0 + scale * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    f2 = f * mod
    f2 = np.maximum(f2, 1e-10)
    f2 = f2 / np.sum(f2)
    return f2


def _perturb_random(f, rng, scale=0.05):

    f2 = f * np.exp(rng.normal(0, scale, len(f)))
