"""
Minimize  f(a) = 2*n*max(convolve(a,a)) / sum(a)^2   for a_i >= 0.

Key facts used:
  * f is scale invariant, so only the relative shape of a matters.
  * The uniform (constant) sequence always gives f = 2.0.
  * Information-theoretic lower bound is 2n/(2n-1)  (flat autocorrelation),
    which is NOT attainable with non-negative entries, hence target < 2 is
    reachable but > 1.  The optimum is a compromise shape (edge-weighted,
    flatter aperiodic autocorrelation than the triangle of the constant case).

Algorithm:
  * objective is smoothed with a temperature-scaled log-sum-exp (LSE) of the
    full convolution, giving a differentiable surrogate of the hard max;
  * analytic gradient with respect to a (convolution / cross-correlation),
    optimized with L-BFGS-B over the simplex-ish box 0 <= a_i <= 1000;
  * multiple starts over several sequence lengths and an edge-peaked family
    a(i) ~ 1 + beta*cos(2*pi*i/(n-1)), plus an ILS (perturb + re-optimize)
    loop to escape local minima;
  * the authoritative, unsmoothed evaluate_sequence is always the final
    scorer; the best finite sequence is returned.
"""

import math
import time

import numpy as np


# ----------------------------------------------------------------------
# differentiable surrogate objective and exact gradient
# ----------------------------------------------------------------------
def _lse_and_p(R, T):
    """Return (L, p) where L = LSE(R)/T (smooth max) and p = softmax of T*R."""
    m = float(np.max(R))
    e = np.exp(T * (R - m))
    s = float(np.sum(e))
    L = (m + math.log(s) / T)
    p = e / s
    return L, p


def _obj_grad(a, T):
    """Smooth objective 2*n*L/S^2 and its gradient w.r.t. a (a >= 0)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0 or np.sum(a) < 1e-9:
        return 1e30, np.zeros_like(a)
    R = np.convolve(a, a)                 # length 2n-1, non-negative
    L, p = _lse_and_p(R, T)
    S = float(np.sum(a))
    # Q[i] = sum_{j} a[j] * p[i+j]   (cross-correlation of a and p)
    C1 = np.convolve(p, a[::-1])          # length 3n-2
    Q = C1[0:n][::-1]
    val = 2.0 * n * L / (S * S)
    grad = (4.0 * n / (S * S)) * Q - (4.0 * n * L / (S * S * S)) * 1.0
    return val, grad


# ----------------------------------------------------------------------
# starting shapes
# ----------------------------------------------------------------------
def edge_peaked(n, beta, rng):
    """a(i) ~ 1 + beta*cos(2*pi*i/(n-1)) : high at both ends, low in middle."""
    if n < 2:
        return np.ones(2)
    t = np.arange(n, dtype=np.float64)
    base = 1.0 + beta * np.cos(2.0 * math.pi * t / (n - 1))
    # add gentle randomisation so ILS / restarts explore different basins
    base = base + rng.normal(0.0, max(1e-3, 0.02 * beta), n)
    base = np.clip(base, 1e-3, None)
    return base / base.sum()


def random_start(n, rng):
    return (0.5 + 0.5 * rng.random(n))

# one optimisation run from a given initial shape
# ----------------------------------------------------------------------
def optimise(a0, T, maxiter, deadline):
    """L-BFGS-B minimisation of the smooth surrogate; returns best a (clipped)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return np.clip(a0, 0.0, 1000.0)

    a0 = np.clip(np.asarray(a0, dtype=np.float64), 1e-6, 1000.0)
    n = a0.size

    def fun(x):
        v, g = _obj_grad(x, T)
        return v, g

    res = minimize(
        fun, a0, jac=True, method="L-BFGS-B",
        bounds=[(1e-6, 1000.0)] * n,
        options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-7,
                 "maxls": 40},
        callback=None,
    )
    a = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
    if np.isfinite(a).all():
        return a
    return np.clip(a0, 0.0, 1000.0)


# ----------------------------------------------------------------------
# main search
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Time-boxed multi-start ILS with a differentiable flat-autocorrelation
    surrogate; returns the best non-negative sequence found (list of floats)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    # ---- valid fallbacks (always scoreable) ----
    fallback = np.ones(1024, dtype=np.float64)            # uniform -> f = 2
    best = fallback.copy()
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = 2.0
    if not np.isfinite(best_score):
        best_score = 2.0

    # seed from previously found construction if present and valid
    try:
        p0 = np.asarray(height_sequence_1, dtype=np.float64)
        if (p0.ndim == 1 and p0.size >= 8
                and np.isfinite(p0).all() and np.sum(p0) > 1e-6):
            p0 = np.clip(p0, 0.0, 1000.0)
            s0 = float(evaluate_sequence(p0.tolist()))
            if np.isfinite(s0) and s0 < best_score:
                best = p0.copy()
                best_score = s0
                print("seed from height_sequence_1: score=%.6f n=%d"
                      % (best_score, p0.size))
    except Exception:
        pass

    # keep a small pool of the best shapes found so far
    length_pool = [int(best.size)]

    def try_candidate(a):
        """Score with authoritative evaluator; update globals; return score."""
        nonlocal best, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or float(np.sum(a)) < 1e-3:
            return np.inf
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return np.inf
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            if a.size not in length_pool:
                length_pool.append(a.size)
            print("  new best score=%.6f  n=%d" % (sc, a.size))
        return sc if np.isfinite(sc) else np.inf

    # ---- parameter grid for restarts ----
    n_list = sorted(set([192, 256, 384, 512, 768, 1024, 1536, 2048]))
    n_list = [n for n in n_list]
    betas = [-0.9, -0.7, -0.5, -0.3, 0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 0.97]
    Ts = [80.0, 200.0, 500.0, 1200.0]          # higher T -> closer to hard max

    init_count = 0
    # ---- phase 1: diverse multi-start ----
    for n in n_list:
        if time.monotonic() > deadline - 60.0:
            break
        for beta in betas:
            if time.monotonic() > deadline - 30.0:
                break
            for T in (200.0, 600.0):
                if time.monotonic() > deadline - 20.0:
                    break
                a0 = edge_peaked(n, beta, rng)
                a0 = optimise(a0, T, maxiter=120, deadline=deadline)
                try_candidate(a0)
                init_count += 1
                if time.monotonic() > deadline:
                    break
        # a pure-uniform reference for this length (guarantees a 2.0 anchor)
        try_candidate(np.ones(n, dtype=np.float64))

    # also refine the inherited / random starts
    for _ in range(len(n_list)):
        if time.monotonic() > deadline - 30.0:
            break
        n = n_list[int(rng.integers(0, len(n_list)))]
        for T in Ts:
            if time.monotonic() > deadline - 10.0:
                break
            a0 = random_start(n, rng)
            a0 = optimise(a0, T, maxiter=120, deadline=deadline)
            try_candidate(a0)

    # ---- phase 2: ILS on the current best to escape local minima ----
    base = best.copy()
    T_best = 600.0
    step_scale = 0.15
    ills = 0
    while time.monotonic() < deadline - 15.0:
        # mix best shape with a fresh edge-peaked shape for diversity
        m = base.size
        for _ in range(3):
            if time.monotonic() > deadline - 15.0:
                break
            b = 0.0
            a0 = edge_peaked(m, b, rng)
            mix = 0.3 * base + 0.7 * a0
            mix = mix + rng.normal(0.0, step_scale, m)
            mix = np.clip(mix, 1e-6, None)
            # try a couple temperatures
            for T in (T_best, 1200.0):
                if time.monotonic() > deadline - 15.0:
                    break
                a1 = optimise(mix, T, maxiter=100, deadline=deadline)
                try_candidate(a1)
            ills += 1
        # slowly grow the perturbation (ILS schedule)
        step_scale = min(0.6, step_scale + 0.02)

    # ---- final: one very-high-temperature polish of the global best ----
    if time.monotonic() < deadline:
        a_pol = optimise(best.copy(), 2000.0, maxiter=200, deadline=deadline)
        try_candidate(a_pol)

    # ---- hard safety: make sure what we return is valid & finite ----
    result = best.copy()
    result = np.clip(result, 0.0, 1000.0)
    try:
        fin = float(evaluate_sequence(result.tolist()))
    except Exception:
        fin = np.inf
    if (not np.isfinite(fin)) or fin >= best_score:
        result = np.ones(1024, dtype=np.float64)

    print("FINAL n=%d  score=%.6f  (initial %d + %d ILS)"
          % (result.size, best_score, init_count, ills))
    return [float(x) for x in result]
