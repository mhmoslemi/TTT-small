def _add_grad_single(g, a, k, w):
    """Add w * grad_b_k to g in-place."""
    n = a.size
    # g[i] += w * a[k-i] for 0 <= k-i < n, i.e., k-n+1 <= i <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        g[lo1:hi1+1] += w * a[k-hi1 : k-lo1+1][::-1]
    # g[i] += w * a[i-k] for 0 <= i-k < n, i.e., k <= i <= k+n-1
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2+1] += w * a[lo2-k : hi2-k+1]
