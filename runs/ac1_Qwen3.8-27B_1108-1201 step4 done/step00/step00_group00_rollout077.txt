```python
"""
FFT-accelerated search for nonnegative sequences minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

Key ideas used:
- The evaluator's objective is scale invariant as long as values stay in [0,1000].
- Fast exact evaluation is done by FFT-based linear convolution.
- Strong deterministic starting profiles are used, especially low-start ramps and
  power-law shapes, which already avoid the constant-sequence local minimum.
- A mixed local search is run:
    * log-softmax smoothed-objective gradient steps,
    * multiplicative Gaussian noise,
    * block scaling perturbations,
    * peak-aware mass redistribution.
- Multiple lengths are explored, and the final best candidate is verified with
  the authoritative evaluate_sequence() if available.
"""

import numpy as np
import time


def _next_pow2(m: int) -> int:
    """Return the smallest power of two >= m."""
    x = 1
    m = max(1, int(m))
    while x < m:
        x <<= 1
    return x


def _norm_mean1(a: np.ndarray) -> np.ndarray:
    """
    Clip to [0,1000] and scale to mean 1.

    The objective is scale invariant for admissible values, so this keeps the
    search numerically stable without changing the score.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(a.size, dtype=np.float64)

    b = a * (float(a.size) / s)

    # If normalization somehow made values too large, re-clip and renormalize.
    if b.size and b.max() > 1000.0:
        b = np.clip(b, 0.0, 1000.0)
        s2 = float(b.sum())
        if np.isfinite(s2) and s2 > 1e-12:
            b = b * (float(b.size) / s2)

    return b


def fast_eval(a: np.ndarray) -> float:
    """
    Fast evaluation using FFT convolution.

    This matches evaluate_sequence() up to roundoff when the input is valid and
    not clipped by the 0/1000 sanitizers.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf")

    # Normalize to mean 1. The final formula then uses sum = n.
    a = a * (float(n) / s)

    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)

    if b.size < 2 * n - 1:
        return float("inf")

    # Roundoff can create tiny negative values; the max is what matters.
    bmax = float(np.max(b[:2 * n - 1]))
    if not np.isfinite(bmax):
        return float("inf")

    return float(2.0 * float(n) * bmax / (float(n) * float(n)))


def smooth_grad(a: np.ndarray, T: float):
    """
    Return a smoothed surrogate objective and a usable gradient/subgradient.

    The surrogate replaces max(conv(a,a)) by a log-sum-exp smoothing.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf"), np.zeros(0, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf"), np.zeros(n, dtype=np.float64)

    a = a * (float(n) / s)
    S = float(n)

    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    bmax = float(np.max(b))
    if not np.isfinite(bmax):
        return float("inf"), np.zeros(n, dtype=np.float64)

    T = max(float(T), 1e-8)
    shifted = (b - bmax) / T
    m = float(np.max(shifted))
    expb = np.exp(shifted - m)
    psum = float(expb.sum())
    if not np.isfinite(psum) or psum <= 1e-300:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = expb / psum

    smooth_max = bmax + T * (np.log(psum) + m)
    fval = float(2.0 * n * smooth_max / (S * S))

    if n == 1:
        c = np.array([a[0]], dtype=np.float64)
    else:
        p_rev = p[::-1]
        conv_len = 3 * n - 2
        N2 = _next_pow2(conv_len)
        fa2 = np.fft.rfft(a, N2)
        fp = np.fft.rfft(p_rev, N2)
        conv = np.fft.irfft(fa2 * fp, N2)[:conv_len]
        # c[i] = conv[2n-2-i]
        c = conv[2 * n - 2 : n - 2 : -1]
        if c.size != n:
            c = np.zeros(n, dtype=np.float64)

    grad = (2.0 * n / (S * S)) * (2.0 * c) - (4.0 * n * smooth_max / (S ** 3)) * a
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return fval, grad


def peak_shave(a: np.ndarray, rng: np.random.Generator, strength: float) -> np.ndarray:
    """
    Heuristic peak reduction.

    Find the largest convolution bin, reduce elements that contribute strongly
    to it, and redistribute their mass to elements with low derivative for that
    same bin.
    """
    a = _norm_mean1(a)
    n = a.size
    if n <= 2:
        return a.copy()

    strength = float(min(max(strength, 0.0), 0.35))
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    k = int(np.argmax(b))

    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    valid = np.arange(lo, hi, dtype=np.int64)

    contrib = np.zeros(n, dtype=np.float64)
    deriv = np.zeros(n, dtype=np.float64)

    if valid.size:
        contrib[valid] = a[valid] * a[k - valid]
        deriv[valid] = 2.0 * a[k - valid]

    top_k = max(1, min(n // 50, n))
    if top_k >= n:
        top_idx = np.arange(n)
    else:
        top_idx = np.argpartition(-contrib, top_k)[:top_k]

    cand = a.copy()
    old = cand[top_idx].copy()
    reduce_amt = strength * old
    cand[top_idx] -= reduce_amt
    saved = float(reduce_amt.sum())

    if saved > 0.0:
        scores = deriv + 1e-12
        scores[top_idx] = np.inf

        max_add = n - top_idx.size
        m = min(max_add, max(1, min(3 * top_idx.size, n // 10)))
        if m > 0:
            if m >= scores.size:
                add_idx = np.nonzero(np.isfinite(scores))[0]
            else:
                add_idx = np.argpartition(scores, m)[:m]
            if add_idx.size:
                cand[add_idx] += saved / float(add_idx.size)

    return _norm_mean1(np.clip(cand, 0.0, 1000.0))


def make_power_profile(n: int, alpha: float) -> np.ndarray:
    """Return normalized (i+1)^alpha style profile, with i from 0 to n-1."""
    x = np.arange(n, dtype=np.float64)
    if alpha == 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = (x + 1.0) ** float(alpha)
    return _norm_mean1(a)


def make_exp_profile(n: int, beta: float) -> np.ndarray:
    """Return normalized exp(beta*x) profile."""
    x = (np.arange(n, dtype=np.float64) / max(1, n - 1))
    a = np.exp(float(beta) * x)
    return _norm_mean1(a)


def random_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Generate a diverse random nonnegative starting point.

    Many good sequences are smooth monotone-ish profiles, so this favors
    power/exponential/polynomial shapes.
    """
    n = max(64, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    r = float(rng.random())

    if r < 0.38:
        # Power law with occasional very low exponent.
        alpha = float(rng.uniform(0.15, 2.5))
        a = (x + 1e-4) ** alpha
    elif r < 0.62:
        # Exponential skew.
        beta = float(rng.uniform(0.3, 3.0))
        a = np.exp(beta * x)
    elif r < 0.82:
        # Low-degree polynomial profile, floored to stay nonnegative.
        c0 = abs(float(rng.normal(0.6, 0.4))) + 0.2
        c1 = float(rng.normal(1.0, 0.8))
        c2 = float(rng.normal(0.7, 0.8))
        c3 = float(rng.normal(0.2, 0.6))
        a = c0 + c1 * x + c2 * x * x + c3 * x * x * x
    else:
        # Noisy baseline.
        a = 1.0 + 0.9 * np.abs(rng.normal(0.0, 1.0, n))

    a = np.clip(a, 1e-8, 1000.0)
    return _norm_mean1(a)


def profile_candidates(n: int, rng: np.random.Generator) -> list:
    """A diversified set of strong deterministic and random initial points."""
    n = max(64, int(n))
    out = []

    # Very important: a simple ramp already avoids the constant local minimum.
    out.append(_norm_mean1(np.arange(n, dtype=np.float64)))

    # Additional power/exponential profiles.
    for alpha in (0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5):
        out.append(make_power_profile(n, alpha))

    for beta in (0.5, 1.0, 1.5, 2.0, 3.0):
        out.append(make_exp_profile(n, beta))

    # A few offset linear/quadratic shapes.
    for off in (0.15, 0.35, 0.6):
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        out.append(_norm_mean1(off + x))
        out.append(_norm_mean1(off + x * x))

    # Random diverse starts.
    for _ in range(5):
        out.append(random_sequence(rng, n))

    # Remove exact duplicates cheaply by length? Not necessary.
    return out


def propose_candidate(seed: int = 42, budget_s: int = 1800, **kwargs):
    """
    Search for the best nonnegative sequence of step-function heights.
    """
    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    budget = max(120.0, float(budget_s))
    deadline = t0 + (budget - 15.0)

    # ------------------------------------------------------------------
    # Choose reasonable length(s) to explore.
    # ------------------------------------------------------------------
    n_base = 4096
    global_seq = globals().get("height_sequence_1", None)
    if global_seq is not None:
        try:
            g = np.asarray(global_seq, dtype=np.float64).ravel()
            if g.size >= 64 and np.all(np.isfinite(g)) and float(np.sum(np.clip(g, 0, 1000))) > 1e-6:
                n_base = int(g.size)
        except Exception:
            pass

    # Keep final exact convolve reasonably safe while still using longer n.
    n_base = int(min(max(n_base, 512), 8192))

    lengths = []
    for m in (1.0, 0.75, 1.15, 1.35, 1.5):
        nn = int(round(n_base * m))
        nn = int(min(max(nn, 512), 8192))
        if nn not in lengths:
            lengths.append(nn)

    # Include a few solid standard sizes.
    for nn in (2048, 4096, 6144, 8192):
        if nn not in lengths:
            lengths.append(nn)

    lengths = lengths[:6]

    # ------------------------------------------------------------------
    # Initial multistart over profiles and lengths.
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    start_budget = min(60.0, 0.08 * budget)

    for n in lengths:
        if time.monotonic() - t0 > start_budget and best is not None:
            break

        for cand in profile_candidates(n, rng):
            if time.monotonic() - t0 > start_budget and best is not None:
                break
            try:
                sc = fast_eval(cand)
            except Exception:
                sc = float("inf")
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = float(sc)

    if best is None or not np.isfinite(best_score):
        # Guaranteed decent fallback: simple ramp.
        best = _norm_mean1(np.arange(n_base, dtype=np.float64))
        best_score = fast_eval(best)

    print(f"initial best: n={best.size} score={best_score:.9f} t={time.monotonic()-t0:8.2f}s")

    # ------------------------------------------------------------------
    # Main adaptive local search.
    # ------------------------------------------------------------------
    current = best.copy()
    curr_score = best_score
    no_improve = 0
    last_print = time.monotonic()

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        elapsed = now - t0
        prog = min(1.0, elapsed / max(1.0, budget - 15.0))
        n = current.size

        # Temperature for log-sum-exp smoothing, scaled to ~ a few percent of b.
        T = max(1.0, 0.025 * float(n) * (1.0 - 0.65 * prog))
        if no_improve > 120:
            T *= 1.8

        grad_lr = 0.05 + 0.16 * (1.0 - prog)
        if no_improve > 180:
            grad_lr += 0.08

        noise_amp = 0.018 + 0.10 * (1.0 / (1.0 + no_improve / 250.0))

        mode = float(rng.random())

        if mode < 0.33:
            # Smoothed gradient step in log-space.
            try:
                _, grad = smooth_grad(current, T)
            except Exception:
                grad = None

            if grad is not None:
                gscale = float(np.percentile(np.abs(grad), 90))
                if np.isfinite(gscale) and gscale > 1e-12:
                    base = np.maximum(current, 1e-8)
                    exponent = -float(grad_lr) * grad / gscale
                    exponent = np.clip(exponent, -3.0, 3.0)
                    cand = base * np.exp(exponent)
                else:
                    cand = current.copy()
            else:
                cand = current.copy()

        elif mode < 0.64:
            # Multiplicative noise.
            cand = current * np.exp(noise_amp * rng.normal(0.0, 1.0, n))

        elif mode < 0.84:
            # Block scaling.
            wmin = max(2, n // 500)
            wmax = max(wmin + 1, min(n, n // 5))
            if wmax <= wmin:
                width = min(n, wmin)
                start = 0
            else:
                width = int(rng.integers(wmin, wmax))
                start = int(rng.integers(0, max(1, n - width + 1)))
            width = max(1, min(width, n))
            cand = current.copy()
            block_scale = (0.08 + 0.40 * (1.0 / (1.0 + no_improve / 200.0)))
            cand[start:start + width] *= np.exp(block_scale * float(rng.normal(0.0, 1.0)))

        else:
            # Peak-aware or restart-like move.
            if no_improve > 60 and float(rng.random()) < 0.75:
                sh = 0.03 + 0.12 * float(rng.random())
                cand = peak_shave(current, rng, sh)
            elif no_improve > 500:
                # Profile restart, sometimes with a nearby length.
                nn = n
                if float(rng.random()) < 0.18:
                    choices = (
                        max(512, int(n * 4 // 5)),
                        n,
                        min(8192, int(n * 5 // 4)),
                    )
                    nn = int(rng.choice(np.asarray(choices, dtype=np.int64)))
                nn = int(min(max(nn, 512), 8192))
                cand = random_sequence(rng, nn)
            else:
                cand = current * np.exp(0.02 * rng.normal(0.0, 1.0, n))

        cand = _norm_mean1(np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0))

        try:
            sc = fast_eval(cand)
        except Exception:
            sc = float("inf")

        if np.isfinite(sc) and sc < curr_score:
            current = cand
            curr_score = float(sc)
            no_improve = 0

            if sc < best_score:
                best = current.copy()
                best_score = float(sc)
                if time.monotonic() - last_print > 8.0:
                    print(
                        f"improved: n={best.size} score={best_score:.9f} "
                        f"t={time.monotonic()-t0:8.2f}s"
                    )
                    last_print = time.monotonic()
        else:
            no_improve += 1
            # If completely stuck, escape with a good profile restart.
            if no_improve > 1200:
                restart = random_sequence(rng, current.size)
                try:
                    rs = fast_eval(restart)
                except Exception:
                    rs = float("inf")
                current = restart
                curr_score = rs
                no_improve = 0
                if np.isfinite(rs) and rs < best_score:
                    best = current.copy()
                    best_score = float(rs)

        # Short safety break if progress is very slow but still searching.
        if int(no_improve) % 10000 == 0:
            pass

    # ------------------------------------------------------------------
    # Final exact verification with a few strong fallbacks.
    # ------------------------------------------------------------------
    candidates = []
    candidates.append(best.copy())

    # Simple ramp with current best length.
    lin_cur = _norm_mean1(np.arange(max(64, best.size), dtype=np.float64))
    candidates.append(lin_cur)

    # Simple ramp at a solid large length, if different.
    lin_large = _norm_mean1(np.arange(8192, dtype=np.float64))
    if abs(int(lin_large.size) - int(best.size)) > 0:
        candidates.append(lin_large)

    # If the original height_sequence_1 exists, include its ramp length too.
    if global_seq is not None:
        try:
            orig_len = max(64, int(np.asarray(global_seq, dtype=np.float64).ravel().size))
            if 512 <= orig_len <= 8192:
                lin_orig = _norm_mean1(np.arange(orig_len, dtype=np.float64))
                if lin_orig.size not in (lin_cur.size, lin_large.size):
                    candidates.append(lin_orig)
        except Exception:
            pass

    final_seq = candidates[0]
    final_score = float("inf")

    # Limit exact calls to avoid pathological runtime.
    check_count = 0
    for seq in candidates:
        check_count += 1
        if check_count > 4:
            break
        try:
            s = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            s = float("inf")
        if np.isfinite(s) and s < final_score:
            final_score = s
            final_seq = seq

    # If exact verification failed entirely, fall back to the ramp.
    if not np.isfinite(final_score):
        final_seq = lin_cur
        print("exact verification failed; returning ramp fallback")
    else:
        print(f"final exact: n={final_seq.size} score={final_score:.9f}")

    return [float(x) for x in final_seq]
```
