"""
Alternating mirror descent (exponential weights) to minimize max self-convolution.

The objective J(a) = 2n*max(conv(a,a))/(sum(a))^2 is scale-invariant, so we fix
sum(a)=1 and minimize max(conv(a,a)). We alternate between fixing one factor and
optimizing the other via mirror descent with FFT-based gradient computation.
"""


def compute_gradient(a, b, tau):
    """Compute gradient of soft_max(conv(a,b)) with respect to b.
    
    Uses softmax weighting and np.correlate for O(n log n) computation.
    """
    conv_ab = np.convolve(a, b)
    max_val = conv_ab.max()
    weights = np.exp((conv_ab - max_val) / tau)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones_like(conv_ab) / len(conv_ab)
    else:
        weights /= wsum
    grad = np.correlate(weights, a, 'valid')
    return grad


def mirror_descent_step(a, b, n_inner, tau, lr):
    """Optimize b (fixed a) via mirror descent to minimize soft_max(conv(a,b)).
    
    Maintains b >= 0, sum(b) = 1 throughout.
    """
    n = len(a)
    b = b.copy()
    b = np.maximum(b, 1e-15)
    b /= b.sum()

        grad = compute_gradient(a, b, tau)
        # Normalize gradient for stable updates
        gmax = np.abs(grad).max()
        if gmax < 1e-30:

        grad = grad / gmax
        # Exponential weights update (multiplicative)
        b *= np.exp(-lr * grad)
        # Project to simplex {b >= 0, sum(b) = 1}
        b = np.maximum(b, 0.0)
        bsum = b.sum()
        if bsum < 1e-15:

    return b
