Minimize the peak-to-average ratio of the self-convolution of a non-negative
sequence using a cutting-plane (outer approximation) method with LP subproblems.
Starting from a ramp sequence (which already achieves ~1.33), iteratively
improves using linearized constraints from the active and near-active
convolution lags.
"""

import numpy as np
import time
from scipy.optimize import linprog

N = 6121

def compute_gradient_k(a, k):
    """Compute gradient of c[k] w.r.t. a, where c = full convolve(a,a)."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    # g[j] = 2*a[k-j] for j in [lo, hi]
    indices = np.arange(lo, hi + 1)
    g[indices] = 2.0 * a[k - indices]
    return g

def objective_value(a):
    """Compute the objective value (without the 2n factor, just max_c / (sum_a)^2)."""
    c = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2 * len(a) * c.max() / (sum_a ** 2)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using water-filling."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def solve_lp(n, cuts):
    """
    Solve: min t
           s.t. g_k @ a - t <= rhs_k for all cuts
                sum(a) = 1
                a >= 0
    """
    if len(cuts) == 0:
        return np.ones(n) / n, 1.0 / n
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    m = len(cuts)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, (g_k, rhs) in enumerate(cuts):
        A_ub[i, :n] = g_k
        A_ub[i, n] = -1.0
        b_ub[i] = rhs
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

**Approach: Frank-Wolfe (conditional gradient) method**

The Frank-Wolfe method works for minimizing a (possibly non-convex but smooth) function over a convex set. At each step:
1. Compute the gradient (or subgradient) of the objective.
2. Solve a linear minimization problem over the feasible set to find a "vertex" direction.
3. Take a step towards that vertex.

For our problem: minimize `f(a) = max_k c[k]` over the simplex `{a >= 0, sum(a) = 1}`.

Step 1: Subgradient is `g = grad c[k*](a)`.
Step 2: Minimize `<g, s>` over the simplex. The minimum is achieved at a vertex of the simplex, i.e., `s = e_j` (the j-th standard basis vector) where `j = argmin_j g[j]`.
Step 3: `a_new = (1 - gamma) * a + gamma * e_j` where `gamma` is the step size.

This is very simple! And each iteration is O(n) (just finding the argmin of `g`).

Let me analyze: `g[j] = 2*a[k*-j]` for valid `k*-j`. So `g[j]` is a (shifted) version of `2a`. The `argmin_j g[j]` is the `j` where `a[k*-j]` is minimum, i.e., where `a` has its minimum value at position `k*-j`.

So the Frank-Wolfe step would: put all the "new mass" at the position where `a` is currently smallest (in the relevant range). This makes sense: we want to "spread" mass to reduce the peak.

Let me code this up:

```python
def frank_wolfe_step(a, k_star, gamma):
    n = len(a)
    # g[j] = 2*a[k_star - j] for valid j
    # Find j* = argmin_j g[j] = argmin_j a[k_star - j]
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    # We want to minimize a[k_star - j] for j in [lo, hi]
    # Let i = k_star - j, so i ranges from k_star - hi to k_star - lo
    i_lo = k_star - hi
    i_hi = k_star - lo
    # i ranges from i_lo to i_hi
    # j* = k_star - i* where i* = argmin_{i in [i_lo, i_hi]} a[i]
    i_star = np.argmin(a[i_lo:i_hi+1]) + i_lo
    j_star = k_star - i_star
    
    # Step: a_new = (1-gamma)*a + gamma*e_{j_star}
    a_new = (1 - gamma) * a
    a_new[j_star] += gamma
    return a_new
```

"""
Algorithm: Multi-phase optimization to minimize 2n*max(convolve(a,a))/(sum(a))^2
over non-negative sequences of fixed length.

Phase 1: Grid search over parameterized families (power laws, polynomials).
Phase 2: Frank-Wolfe conditional gradient method for local optimization.
Phase 3: Random perturbation + re-optimization to escape local minima.
"""

import numpy as np
import time

N = 6121

def compute_obj(a):
    """Compute the objective value."""
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * len(a) * c.max() / (s * s)

def fw_step(a, k_star, gamma):
    """One Frank-Wolfe step."""
    n = len(a)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    i_lo = k_star - hi
    i_hi = k_star - lo
    i_star = i_lo + np.argmin(a[i_lo:i_hi + 1])
    j_star = k_star - i_star
    a_new = (1.0 - gamma) * a
    a_new[j_star] += gamma
    return a_new

def fw_optimize(a, deadline, gamma0=0.05, gamma_min=0.001):
    """Run Frank-Wolfe until deadline."""
    n = len(a)
    gamma = gamma0
    best = a.copy()
    best_obj = compute_obj(a)
    count = 0
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        k_star = c.argmax()
        a = fw_step(a, k_star, gamma)
        count += 1
        if count % 100000 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best = a.copy()
            if gamma > gamma_min:
                gamma *= 0.95
    return best, best_obj
