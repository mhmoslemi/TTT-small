import time

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2
using gradient-based methods with adaptive step sizes, soft-max gradient
approximation, simulated annealing, and periodic restarts.
"""

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)

def compute_grad_smooth(a, top_m=10, temperature=0.0):
    """Compute a smoothed subgradient using top-m convolution peaks."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.zeros(n)
    conv = np.convolve(a, a)
    
    # Get top-m positions (use soft weighting if temperature > 0)
    m = min(top_m, len(conv))
    top_idx = np.argsort(conv)[-m:]
    top_vals = conv[top_idx]
    
    if temperature > 0:
        weights = np.exp((top_vals - top_vals.max()) / temperature)
        weights /= weights.sum()
    else:
        weights = np.ones(m) / m
    
    S3 = S**3
    grad = np.zeros(n)
    
    for w, k in zip(weights, top_idx):
        Ck = conv[k]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        
        # For out-of-range j: contribution is -2*Ck/S3
        # For in-range j: contribution is (2*a[k-j]*S - 2*Ck)/S3
        g_k = np.full(n, -2.0 * Ck / S3)
        if j_min <= j_max:
            idx = k - np.arange(j_min, j_max + 1)
            g_k[j_min:j_max+1] = (2.0 * a[idx] * S - 2.0 * Ck) / S3
        
        grad += w * g_k
    
    # Multiply by 2n (the constant factor in the objective)
    grad *= 2.0 * n
    return grad

                     normalize=True, target_sum=None):

    n = len(a)
    a = a.copy()
    if normalize:
        if target_sum is None:
            target_sum = np.sum(a)
        S = np.sum(a)
        if S > 0:
            a = a * target_sum / S
    
    best_local = a.copy()
    best_score = compute_obj(a)

def block_perturb(a, rng, block_frac=(0.01, 0.1), noise_scale=0.1):
    """Perturb a random block of the sequence."""
    n = len(a)
    candidate = a.copy()
    lo = int(n * block_frac[0])
    hi = int(n * block_frac[1])
    width = int(rng.integers(lo, max(lo+1, hi)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    end = min(n, start + width)
    
    # Multiplicative perturbation
    candidate[start:end] *= np.exp(rng.normal(0, noise_scale, size=end-start))
    candidate = np.clip(candidate, 0.0, 1000.0)
    
    # Occasionally symmetrize
    if rng.random() < 0.2:
        candidate = 0.5 * (candidate + candidate[::-1])
    
    return candidate

                        block_frac=(0.01, 0.1), noise_scale=0.1):

    n = len(a)

    current_score = compute_obj(a)
    best = a.copy()
    best_score = current_score
    
    target_sum = np.sum(a)
    
    for it in range(n_iter):
        t = it / max(1, n_iter - 1)
        T = T_end + (T_start - T_end) * (1.0 - t)
        
        # Generate candidate
        candidate = block_perturb(a, rng, block_frac, noise_scale)

        if S > 0:

            a = candidate
            current_score = candidate_score
            if current_score < best_score:
                best = a.copy()
                best_score = current_score
    
    return best, best_score

import numpy as np
import time

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Adam-style adaptive gradient descent with the subgradient of the (soft) max,
simulated annealing with multiplicative block perturbations, and periodic restarts.
The subgradient is averaged over the top-k convolution peaks to handle non-smoothness.
"""


def _fast_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (S * S)


def _subgrad(a, k_top=12):
    """
    Compute averaged subgradient over top-k convolution peaks.
    Returns gradient of (2n * max_k conv[k] / S^2) approximated by averaging
    over the k_top largest convolution values.
    """
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n)
    conv = np.convolve(a, a)
    m = min(k_top, conv.size)
    # Get indices of top-m values
    idx = np.argpartition(conv, -m)[-m:]

    S2 = S * S
    S3 = S2 * S
    g = np.zeros(n)

    for k in idx:
        Ck = conv[k]
        # j range where 0 <= k-j < n: k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)

        # Base contribution for all j: -2*Ck/S^3
        g[:] += -2.0 * Ck / S3
        # Additional contribution for j in [j_lo, j_hi]: +2*a[k-j]/S^2
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            g[js] += 2.0 * a[k - js] / S2

    g *= 2.0 * n / m
    return g


def _adam_phase(a, n_steps, lr, beta1, beta2, eps, noise_scale, k_top, rng, deadline):
    """Adam optimization with noise injection and normalization."""
    n = a.size
    S_target = a.sum()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    best = a.copy()
    best_score = _fast_obj(a)
    t = 0

    while t < n_steps and time.monotonic() < deadline:
        grad = _subgrad(a, k_top)

        # Add noise to break symmetry
        if noise_scale > 0:
            grad = grad + noise_scale * rng.standard_normal(n)

        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (t + 1))
        v_hat = v_buf / (1.0 - beta2 ** (t + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.clip(a, 0.0, 1000.0)

        # Normalize to maintain sum
        S = a.sum()
        if S > 1e-10:
            a = a * S_target / S

        t += 1

        # Check improvement periodically (every 50 steps or at the end)
        if t % 50 == 0 or t == n_steps:
            score = _fast_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()

    return best, best_score


def _sa_phase(a, n_steps, T0, T1, rng, deadline, block_lo=0.005, block_hi=0.05, noise=0.15):
    """Simulated annealing with multiplicative block perturbations."""
    n = a.size
    S_target = a.sum()
    cur = a.copy()
    cur_score = _fast_obj(cur)
    best = a.copy()
    best_score = cur_score

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        T = T1 + (T0 - T1) * (1.0 - t / max(1, n_steps))

        # Generate perturbation
        cand = cur.copy()
        width = int(rng.integers(max(2, n * block_lo), max(3, n * block_hi)))
        start = int(rng.integers(0, n - width + 1))
        end = min(n, start + width)
        cand[start:end] *= np.exp(rng.normal(0.0, noise, size=end - start))

        # Occasionally symmetrize
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        S = cand.sum()
        if S > 1e-10:
            cand = cand * S_target / S

        cand_score = _fast_obj(cand)
        delta = cand_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()

    return best, best_score


def _coordinate_descent(a, n_steps, lr, rng, deadline, k_top=12):
    """
    Coordinate-descent-style: pick a random position, compute its gradient component,
    and update just that position (or a small neighborhood).
    """
    n = a.size
    S_target = a.sum()
    best = a.copy()
    best_score = _fast_obj(a)

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Compute full gradient (we need conv anyway)
        grad = _subgrad(a, k_top)
        # Pick a random coordinate to update
        j = int(rng.integers(0, n))
        a[j] -= lr * grad[j]
        a[j] = max(0.0, min(1000.0, a[j]))

        # Occasionally update a small neighborhood
        if rng.random() < 0.3:
            w = int(rng.integers(1, max(2, n // 200) + 1))
            j0 = int(rng.integers(0, n - w + 1))
            a[j0:j0+w] -= lr * grad[j0:j0+w]
            a[j0:j0+w] = np.clip(a[j0:j0+w], 0.0, 1000.0)

        # Normalize
        S = a.sum()
        if S > 1e-10:
            a = a * S_target / S

        if t % 100 == 0:
            score = _fast_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()

    return best, best_score
