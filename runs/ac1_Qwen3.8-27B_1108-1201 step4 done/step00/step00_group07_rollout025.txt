```python
def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]
```

```python
    return np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]

def _compute_subgradient(a, b, K=10, temp=0.005):
    """Compute subgradient of max(b) where b = convolve(a, a)."""
    n = len(a)
    # Find top K peaks
    if b.size <= K:
        top_k = np.argsort(b)[-K:][::-1]
    else:

    weights = np.exp((b[top_k] - max_val) / temp)
    weights = weights / weights.sum()

    grad = np.zeros(n)
```

```python
import numpy as np
import time


def _fft_self_convolve(a, fft_size):
    """Compute full self-convolution via FFT."""
    n = len(a)
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]


def _subgradient_max_conv(a, b, K=15, temp=0.01):
    """Compute subgradient of max(b) where b = convolve(a,a)."""
    n = len(a)
    if b.size <= K:
        top_k = np.argsort(b)[-K:][::-1]
    else:
        top_k = np.argpartition(b, -K)[-K:]
        top_k = top_k[np.argsort(b[top_k])[::-1]]
    max_val = b[top_k[0]]
    weights = np.exp((b[top_k] - max_val) / temp)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for idx in range(len(top_k)):
        k = int(top_k[idx])
        w = weights[idx]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        mult = np.where(k == 2 * i_vals, 1.0, 2.0)
        grad[i_vals] += w * mult * a[j_vals]
    return grad, b[top_k[0]]


def _try_optimize(a, n, fft_size, rng, deadline, best, best_score, label=""):
    """Run multiplicative gradient descent from initial a. Returns best, best_score."""
    start = time.monotonic()
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    lr = 0.8
    noise_scale = 0.05
    K = 15
    validate_every = 20
    iter_count = 0
    no_improve = 0

    while time.monotonic() < deadline:
        b = _fft_self_convolve(a, fft_size)
        grad, _ = _subgradient_max_conv(a, b, K=K, temp=0.01)

        # Multiplicative update (preserves non-negativity)
        a *= np.exp(-lr * grad)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        # Multiplicative noise
        if noise_scale > 0.001:
            a *= np.exp(rng.normal(0.0, noise_scale, size=n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        iter_count += 1

        if iter_count % validate_every == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = score
                no_improve = 0
                print(f"  [{label}] iter {iter_count}: {best_score:.6f} (t={time.monotonic()-start:.0f}s)")
            else:
                no_improve += 1

            if no_improve == 8:
                lr *= 0.75
                noise_scale *= 0.7
            elif no_improve == 16:
                a = best * np.exp(rng.normal(0.0, 0.3, size=n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                lr = 0.8
                noise_scale = 0.05
                no_improve = 0
            elif no_improve == 30:
                # Break out and let outer loop try a new init
                break

    return best, best_score

    Minimize 2n*max(convolve(a,a))/sum(a)^2 using multiplicative gradient
    descent on the peak of the self-convolution, with FFT acceleration,

    """

    deadline = t0 + budget_s - 15.0
    rng = np.random.default_rng(seed)

    except Exception:
        parent = None

    if parent is not None:
        n = parent.size
    else:
        n = 2048

    # Generate multiple initializations
    inits = []
```
