```python
"""
Search method:
- Primary: constrained smooth optimization in a reverse-complementary subspace.
  For length n, enforce a[i] + a[n-1-i] = 2/n.  This reduces the central
  convolution peak and gives a compact half-size optimization problem.
- Secondary: free projected multiplicative/additive subgradient search using
  top convolution lags as active constraints.
- Multiple lengths, random/structured restarts, and short L-BFGS-B polishes.
"""

import numpy as np
import time
import scipy.optimize as scipy_opt


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if n <= 64:
        c = np.convolve(a, a)
    else:
        m = 2 * n - 1
        size = 1
        while size < m:
            size <<= 1
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size)[:m]
    c = np.asarray(c, dtype=np.float64)
    if c.size:
        c[c < 0.0] = 0.0
    return c


def score_from_conv(c, n):
    if c.size == 0:
        return float("inf")
    return float(2.0 * int(n) * float(np.max(c)))


def score_sequence(a):
    a = normalize_nonnegative(a)
    if a.size == 0:
        return float("inf")
    c = convolve_self(a)
    return score_from_conv(c, a.size)


def cross_corr_fft(w, a):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    m = 3 * n - 2
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)
    return np.asarray(corr[:n], dtype=np.float64)


def direction_topk_direct(a, idx, weights):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    d = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(idx, weights):
        k = int(kk)
        w = float(ww)
        if w == 0.0:
            continue
        if k < 0 or k >= 2 * n - 1:
            continue
        j_min = 0 if k < n else k - n + 1
        j_max = n - 1 if k >= n else k
        if j_min < 0:
            j_min = 0
        if j_max >= n:
            j_max = n - 1
        if j_min > j_max:
            continue
        m_start = k - j_min
        m_end = k - j_max
        d[j_min:j_max + 1] += w * a[m_end:m_start + 1][::-1]
    return d


def local_search_topk(a, budget, rng, topk=None, step0=0.2, mode="mul", max_kicks=60):
    a = normalize_nonnegative(a)
    n = int(a.size)
    if n == 0 or budget <= 0.0:
        return a, score_sequence(a)

    cur = a.copy()
    c = convolve_self(cur)
    cur_score = score_from_conv(c, n)
    best = cur.copy()
    best_score = cur_score

    deadline = time.monotonic() + float(budget)
    L = 2 * n - 1
    if topk is None:
        topk = int(2.0 * np.floor(np.sqrt(n)))
        if topk < 8:
            topk = 8
    topk = int(min(L, max(1, topk)))

    step = float(step0)
    base_step = step
    no_imp = 0
    kicks = 0

    while time.monotonic() < deadline:
        if no_imp > 3000 or (no_imp > 500 and step < 1e-4):
            noise = 0.25 / np.sqrt(float(n))
            kick = rng.normal(0.0, noise, n)
            cur = normalize_nonnegative(best + kick)
            c = convolve_self(cur)
            cur_score = score_from_conv(c, n)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
            step = base_step
            no_imp = 0
            kicks += 1
            if kicks > max_kicks:
                break
            continue

        if topk < L:
            idx = np.argpartition(c, topk)[-topk:]
        else:
            idx = np.arange(L)

        weights = np.asarray(c[idx], dtype=np.float64)
        wsum = float(np.sum(weights))
        if wsum <= 0.0 or not np.isfinite(wsum):
            break
        weights = weights / wsum
        weights = np.power(weights, 1.5)
        wsum2 = float(np.sum(weights))
        if wsum2 <= 0.0:
            break
        weights = weights / wsum2

        d = direction_topk_direct(cur, idx, weights)
        dmax = float(np.max(d))
        if dmax <= 0.0 or not np.isfinite(dmax):
            break
        d = d / dmax

        old_score = cur_score
        accepted = False
        b = None
        cb = None
        sb = cur_score

        ls = step
        for _ in range(4):
            if mode == "add":
                cand = cur - ls * d
                cand = np.maximum(0.0, cand)
            else:
                exponent = np.clip(-ls * d, -30.0, 30.0)
                cand = cur * np.exp(exponent)
            cand = normalize_nonnegative(cand)
            cb2 = convolve_self(cand)
            sb2 = score_from_conv(cb2, n)
            if sb2 < cur_score - 1e-12:
                accepted = True
                b = cand
                cb = cb2
                sb = sb2
                break
            ls *= 0.5

        if accepted:
            cur = b
            c = cb
            cur_score = sb
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
            no_imp = 0
            if cur_score < old_score * 0.995:
                step = min(1.2, step * 1.10)
            elif cur_score < old_score * 0.999:
                step = min(1.2, step * 1.02)
        else:
            no_imp += 1
            step = max(1e-5, step * 0.75)

    return best, best_score


def antisym_sequence(t, n):
    n = int(n)
    m = n // 2
    a = np.full(n, 1.0 / float(n), dtype=np.float64)
    if m > 0:
        t = np.asarray(t, dtype=np.float64).ravel()[:m]
        t = np.clip(t, -1.0, 1.0)
        d = (2.0 / float(n)) * t
        h = 0.5 * d
        idx = np.arange(m)
        a[idx] += h
        a[n - 1 - idx] -= h
        a = np.maximum(a, 0.0)
        s = float(np.sum(a))
        if s > 0.0:
            a = a / s
    return a


def antisym_smooth_objective(t, n, tau):
    n = int(n)
    t = np.asarray(t, dtype=np.float64).ravel()
    t = np.clip(t, -1.0, 1.0)
    a = antisym_sequence(t, n)
    c = convolve_self(a)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return 1e100
    shift = tau * (c - maxc)
    shift = np.clip(shift, -80.0, 0.0)
    lse_shift = float(np.log(np.sum(np.exp(shift))))
    return float(maxc + lse_shift / float(tau))


def antisym_smooth_gradient(t, n, tau):
    n = int(n)
    m = n // 2
    t = np.asarray(t, dtype=np.float64).ravel()[:m]
    t = np.clip(t, -1.0, 1.0)
    a = antisym_sequence(t, n)
    c = convolve_self(a)
    maxc = float(np.max(c))
    shift = tau * (c - maxc)
    shift = np.clip(shift, -80.0, 0.0)
    w = np.exp(shift)
    ws = float(np.sum(w))
    if ws <= 0.0:
        return np.zeros(m, dtype=np.float64)
    w = w / ws
    corr = cross_corr_fft(w, a)
    grad_a = 2.0 * corr
    if m == 0:
        return np.zeros(0, dtype=np.float64)
    idx = np.arange(m)
    grad_t = (grad_a[idx] - grad_a[n - 1 - idx]) / float(n)
    return np.asarray(grad_t, dtype=np.float64)


def antisym_exact_objective(t, n):
    n = int(n)
    t = np.asarray(t, dtype=np.float64).ravel()
    tc = np.clip(t, -1.0, 1.0)
    a = antisym_sequence(tc, n)
    c = convolve_self(a)
    s = score_from_conv(c, n)
    out = t - tc
    return float(s + 10.0 * float(np.dot(out, out)))


def make_antisym_starts(n, rng, parent):
    n = int(n)
    m = n // 2
    starts = []
    if m == 0:
        starts.append(np.array([0.0], dtype=np.float64))
        return starts

    for amp in (0.25, 0.50, 0.75, 1.00):
        starts.append(np.clip(rng.uniform(-amp, amp, m), -0.999, 0.999))

    if m > 1:
        r = np.linspace(-1.0, 1.0, m)
        for amp in (0.30, 0.60, 1.00):
            starts.append(amp * r)

        ph = np.linspace(0.0, 2.0 * np.pi, m, endpoint=False)
        starts.append(0.75 * np.sin(ph))
        starts.append(0.75 * np.cos(ph))
        starts.append(0.60 * np.sin(2.0 * ph))

    ones = np.full(m, 0.70)
    starts.append(ones)

    alt = np.ones(m)
    alt[1::2] = -1.0
    starts.append(0.70 * alt)

    if parent is not None:
        p = normalize_nonnegative(parent)
        if p.size == n:
            t = np.empty(m, dtype=np.float64)
            for i in range(m):
                t[i] = (p[i] - p[n - 1 - i]) * (float(n) / 2.0)
            starts.append(np.clip(t, -0.999, 0.999))

    return starts


def try_antisym(n, budget, rng, parent=None):
    n = int(n)
    if n < 2 or budget <= 0.0:
        return None, float("inf")
    m = n // 2
    if m == 0:
        a = np.ones(1, dtype=np.float64)
        return a, 2.0

    deadline = time.monotonic() + float(budget)
    best_a = None
    best_score = float("inf")

    starts = make_antisym_starts(n, rng, parent)
    if n <= 16:
        use_starts = starts
    elif n <= 64:
        use_starts = starts[:8]
    elif n <= 512:
        use_starts = starts[:5]
    else:
        use_starts = starts[:3]

    tauses = [0.5 * float(n), 2.0 * float(n), 8.0 * float(n), 30.0 * float(n)]
    bounds = [(-1.0, 1.0)] * m

    if n <= 8:
        maxiter = 900
    elif n <= 32:
        maxiter = 450
    elif n <= 256:
        maxiter = 220
    else:
        maxiter = 90

    for t0 in use_starts:
        if time.monotonic() >= deadline:
            break

        t = np.asarray(t0, dtype=np.float64).copy()
        t = np.clip(t, -0.999, 0.999)

        for tau in tauses:
            if time.monotonic() >= deadline:
                break

            if n > 2000 and (deadline - time.monotonic()) < 0.8:
                a = antisym_sequence(t, n)
                sc = score_sequence(a)
                if np.isfinite(sc) and (best_score > sc or best_a is None):
                    best_score = sc
                    best_a = a.copy()
                break

            try:
                res = scipy_opt.minimize(
                    antisym_smooth_objective,
                    t,
                    args=(n, tau),
                    jac=antisym_smooth_gradient,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": maxiter, "gtol": 1e-5, "ftol": 1e-10, "maxls": 20},
                )
                if res is not None and hasattr(res, "x"):
                    t = np.asarray(res.x, dtype=np.float64)
                    t = np.clip(t, -0.999, 0.999)
            except Exception:
                pass

            a = antisym_sequence(t, n)
            sc = score_sequence(a)

            if n <= 12:
                rem = deadline - time.monotonic()
                if rem > 0.5:
                    try:
                        res2 = scipy_opt.minimize(
                            antisym_exact_objective,
                            t,
                            args=(n,),
                            method="Nelder-Mead",
                            options={"maxiter": 500, "fatol": 1e-9, "xatol": 1e-9},
                        )
                        if res2 is not None and hasattr(res2, "x") and res2.x is not None:
                            t2 = np.asarray(res2.x, dtype=np.float64)
                            t2 = np.clip(t2, -0.999, 0.999)
                            a2 = antisym_sequence(t2, n)
                            sc2 = score_sequence(a2)
                            if sc2 < sc:
                                a = a2
                                t = t2
                                sc = sc2
                    except Exception:
                        pass

            rem = deadline - time.monotonic()
            if rem > 0.2:
                if n <= 32:
                    pol = min(2.0, 0.25 * rem)
                elif n <= 256:
                    pol = min(1.0, 0.20 * rem)
                else:
                    pol = min(0.5, 0.20 * rem)
                if pol > 0.05:
                    a2, sc2 = local_search_topk(
                        a,
                        pol,
                        rng,
                        topk=max(4, int(2.0 * np.sqrt(n))),
                        step0=0.20,
                        mode="mul"
                    )
                    if sc2 < sc:
                        a = a2
                        sc = sc2

            if np.isfinite(sc) and (best_score > sc or best_a is None):
                best_score = sc
                best_a = a.copy()

    if best_a is not None:
        return best_a, float(best_score)

    a = np.ones(n, dtype=np.float64) / float(n)
    return a, 2.0


def random_sequence(n, rng):
    alphas = (0.5, 1.0, 2.0, 4.0)
    alpha = float(alphas[int(rng.integers(0, len(alphas)))])
    return rng.dirichlet(np.full(n, alpha, dtype=np.float64))


def free_search_length(n, budget, rng, parent=None):
    n = int(n)
    if n < 1 or budget <= 0.0:
        return None, float("inf")

    deadline = time.monotonic() + float(budget)
    best_a = None
    best_score = float("inf")

    starts = []
    for _ in range(3):
        starts.append(random_sequence(n, rng))

    m = n // 2
    if m > 0:
        t = rng.uniform(-0.85, 0.85, m)
        starts.append(antisym_sequence(t, n))
    else:
        starts.append(np.array([1.0]))

    ramp = np.linspace(0.5, 1.5, n)
    starts.append(normalize_nonnegative(ramp))
    starts.append(normalize_nonnegative(1.5 - ramp))

    if parent is not None:
        p = normalize_nonnegative(parent)
        if p.size == n:
            starts.append(p)

    n_starts = max(1, len(starts))
    for i, st in enumerate(starts):
        rem = deadline - time.monotonic()
        if rem <= 0.1:
            break
        b = min(rem, rem / float(n_starts - i))
        if b < 0.1:
            continue

        a, sc = local_search_topk(
            st,
            b,
            rng,
            topk=max(8, int(3.0 * np.sqrt(n))),
            step0=0.25,
            mode="mul"
        )
        if np.isfinite(sc) and (best_score > sc or best_a is None):
            best_score = sc
            best_a = a.copy()

        rem = deadline - time.monotonic()
        if rem > 0.3 and best_a is not None:
            b2 = min(0.25 * rem, 30.0)
            if b2 > 0.1:
                a2, sc2 = local_search_topk(
                    best_a,
                    b2,
                    rng,
                    topk=max(8, int(3.0 * np.sqrt(n))),
                    step0=0.25,
                    mode="add"
                )
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = a2

    if best_a is not None:
        return best_a, float(best_score)
    a = np.ones(n, dtype=np.float64) / float(n)
    return a, 2.0


def add_candidate(candidates, a):
    a = normalize_nonnegative(a)
    if a.size == 0:
        return
    sc = score_sequence(a)
    if not np.isfinite(sc):
        return

    dup = False
    if a.size <= 4096:
        for i in range(len(candidates)):
            ex = candidates[i][1]
            if ex.size == a.size:
                if float(np.max(np.abs(ex - a))) < 1e-5:
                    if sc < candidates[i][0]:
                        candidates[i] = (sc, a.copy())
                    dup = True
                    break

    if not dup:
        candidates.append((sc, a.copy()))

    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            if candidates[j][0] < candidates[i][0]:
                candidates[i], candidates[j] = candidates[j], candidates[i]

    if len(candidates) > 12:
        candidates[:] = candidates[:12]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(int(seed))
    start_time = time.monotonic()
    total_budget = max(1.0, float(budget_s))
    hard_deadline = start_time + total_budget
    final_reserve = 25.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.all(np.isfinite(p)) and float(np.sum(p)) > 1e-12:
            parent = p
    except Exception:
        parent = None

    candidates = []

    add_candidate(candidates, np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64))

    x = 1.0 / (4.0 * np.sqrt(3.0))
    add_candidate(candidates, np.array([0.25, 0.25 - x, 0.25 + x, 0.25], dtype=np.float64))

    if parent is not None:
        add_candidate(candidates, parent)

    base_schedule = []
    for n in range(2, 17):
        base_schedule.append((n, 4.0))
    for n in range(18, 49, 2):
        base_schedule.append((n, 4.0))
    for n in range(50, 96, 4):
        base_schedule.append((n, 6.0))
    for n in range(98, 192, 8):
        base_schedule.append((n, 8.0))
    for n in range(194, 384, 16):
        base_schedule.append((n, 12.0))

    large_schedule = [
        (384, 20.0),
        (512, 25.0),
        (768, 30.0),
        (1024, 35.0),
        (1536, 40.0),
        (2048, 45.0),
        (3072, 50.0),
        (4096, 55.0),
        (6144, 60.0),
        (8192, 60.0),
    ]
    base_schedule.extend(large_schedule)

    if parent is not None and 2 <= parent.size <= 8192:
        base_schedule.append((int(parent.size), 35.0))

    schedule = []
    seen = set()
    total_base = 0.0
    for n, b in base_schedule:
        n = int(n)
        if n in seen:
            continue
        seen.add(n)
        schedule.append((n, float(b)))
        total_base += float(b)

    phase1_budget = min(1000.0, max(120.0, total_budget * 0.55))
    if total_base > 0.0:
        scale = min(1.5, max(0.1, phase1_budget / total_base))
    else:
        scale = 1.0
    phase1_deadline = min(hard_deadline - final_reserve, start_time + phase1_budget)

    for n, b in schedule:
        now = time.monotonic()
        if now >= phase1_deadline:
            break
        rem = phase1_deadline - now
        if rem <= 0.05:
            break
        b2 = float(b) * scale
        if b2 > rem:
            b2 = rem
        if b2 < 0.05:
            continue

        a, sc = try_antisym(n, b2, rng, parent)
        if a is not None and np.isfinite(sc):
            add_candidate(candidates, a)

    phase2_deadline = hard_deadline - final_reserve
    now = time.monotonic()
    phase2_budget = max(0.0, phase2_deadline - now)

    if phase2_budget > 30.0 and candidates:
        best_a = candidates[0][1]
        n_best = int(best_a.size)

        b = min(0.25 * phase2_budget, 300.0)
        if b > 5.0:
            a2, sc2 = local_search_topk(
                best_a,
                b,
                rng,
                topk=max(16, int(4.0 * np.sqrt(n_best))),
                step0=0.25,
                mode="mul"
            )
            if a2 is not None and np.isfinite(sc2):
                add_candidate(candidates, a2)

        rem = phase2_deadline - time.monotonic()
        if rem > 30.0:
            b2 = min(0.35 * rem, 300.0)
            a3, sc3 = try_antisym(n_best, b2, rng, parent)
            if a3 is not None and np.isfinite(sc3):
                add_candidate(candidates, a3)

        rem = phase2_deadline - time.monotonic()
        if rem > 30.0:
            test_lengths = []
            if candidates:
                test_lengths.append(int(candidates[0][1].size))
            for m in (16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192):
                test_lengths.append(m)
            if parent is not None and 2 <= parent.size <= 8192:
                test_lengths.append(int(parent.size))

            unique_lengths = []
            seen_l = set()
            for m in test_lengths:
                m = int(m)
                if m not in seen_l:
                    seen_l.add(m)
                    unique_lengths.append(m)

            if unique_lengths:
                per = rem / float(len(unique_lengths))
                for m in unique_lengths:
                    rem = phase2_deadline - time.monotonic()
                    if rem <= 3.0:
                        break
                    b3 = min(120.0, max(4.0, 0.9 * per))
                    if b3 < 1.0:
                        continue
                    a4, sc4 = free_search_length(m, b3, rng, parent)
                    if a4 is not None and np.isfinite(sc4):
                        add_candidate(candidates, a4)

    if candidates:
        # Final sort, just in case.
        for i in range(len(candidates)):
            for j in range(i + 1, len(candidates)):
                if candidates[j][0] < candidates[i][0]:
                    candidates[i], candidates[j] = candidates[j], candidates[i]

        final_best_a = candidates[0][1].copy()
        final_best_score = float("inf")
        evaluated = 0
        max_evals = 4

        for sc, a in candidates:
            n = int(a.size)
            if evaluated >= max_evals:
                break
            if n > 2000 and evaluated >= 1:
                break
            if time.monotonic() >= hard_deadline - 8.0:
                break
            try:
                ev = float(evaluate_sequence([float(z) for z in a]))
            except Exception:
                ev = float("inf")
            evaluated += 1
            if np.isfinite(ev) and ev < final_best_score:
                final_best_score = ev
                final_best_a = a.copy()

        if not np.isfinite(final_best_score):
            final_best_a = candidates[0][1].copy()

        out = [float(z) for z in final_best_a]
        for i in range(len(out)):
            if out[i] < 0.0:
                out[i] = 0.0
            elif out[i] > 1000.0:
                out[i] = 1000.0
        return out

    # Fallback valid sequence.
    return [1.0 / 3.0, 2.0 / 3.0]
```
