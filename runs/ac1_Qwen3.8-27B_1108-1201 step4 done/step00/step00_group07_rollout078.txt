"""
Minimize  2*n*max(convolve(a,a)) / sum(a)^2  over non-negative sequences a.

The objective is homogeneous, so we work on the probability simplex
{a >= 0, sum(a) = 1} and minimize max_k c_k where c = a*a (full convolution),
which makes the score exactly 2*n*max_k c_k.  Because each c_k is a
(generally non-convex) quadratic, the problem is non-convex; we use a projected
aggregated-subgradient method with momentum, simplex projection, adaptive step
sizes, diverse initializations and time-bounded multi-restart search.  All
auxiliary helpers are top level (no closures / lambdas).
"""

import time
import numpy as np


# --------------------------------------------------------------------------- #
# Fast full autocorrelation via FFT  (length 2n-1)
# --------------------------------------------------------------------------- #
def _fft_convolve(a):
    """Return np.convolve(a, a) computed through an FFT (real, len 2n-1)."""
    n = a.size
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:size]


# --------------------------------------------------------------------------- #
# Projection onto the probability simplex  {x >= 0, sum(x) = 1}
# --------------------------------------------------------------------------- #
def _project_simplex(v):
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------------------- #
# One aggregated-subgradient descent step (opposite the gradient of the active
# peaks of the convolution), with momentum and simplex projection.
# State (a, mom) are mutated and returned.
# --------------------------------------------------------------------------- #
def _sgd_step(a, c, top_m, lr, beta, rng, jitter_scale):
    n = a.size

    # ---- pick the top peaks (largest c_k) and weight them toward the max ----
    if c.size <= top_m:
        peaks = np.arange(c.size)
    else:
        peaks = np.argpartition(c, -top_m)[-top_m:]
    peaks = peaks[np.argsort(c[peaks])[::-1]]          # largest first
    w = np.arange(1.0, len(peaks) + 1.0)[::-1]
    w = w / w.sum()

    # ---- aggregated gradient of max_k c_k  ----
    grad = np.zeros(n)
    for wk, k in zip(w, peaks):
        k = int(k)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += wk * 2.0 * a[idx[valid]]

    # ---- synchronous gradient with momentum (descending the active max) ----
    mom = beta * _MOM + (1.0 - beta) * grad
    _MOM[:] = mom

    a = a - lr * mom

    # ---- occasional random kick to escape flat / bad basins ----------------
    if rng.random() < 0.02:
        a = a + jitter_scale * _JITTER_SCALE_PER_N * n * rng.standard_normal(n)

    a = _project_simplex(a)
    return a


# module-level state reused by _sgd_step (kept global; no closures)
_MOM = None
_JITTER_SCALE_PER_N = 1e-4


# --------------------------------------------------------------------------- #
# Diverse starting points on the simplex of length n
# --------------------------------------------------------------------------- #
def _make_inits(n, rng, base):
    inits = []

    def add(x):
        x = np.asarray(x, dtype=np.float64)
        x = np.clip(x, 0.0, None)
        s = x.sum()
        if s > 0:
            inits.append(x / s)

    # 0) the supplied construction (usually the strongest warm start)
    if base is not None and base.size == n and np.all(np.isfinite(base)):
        add(base)

    # 1) alternating  (breaks the left-right symmetry that traps the constant map)
    x = np.ones(n); x[1::2] = 0.35
    add(x)

    # 2) pure random
    add(rng.random(n) + 0.05)

    # 3) low-frequency sine modulation
    t = np.arange(n)
    for f in (3, 5, 8):
        x = 1.0 + 0.6 * np.sin(2.0 * np.pi * t / max(1, n // f))
        add(np.maximum(x, 0.02))

    # 4) edge-weighted (mass pushed to the borders)
    tt = np.linspace(-1.0, 1.0, n)
    add(1.0 + 0.9 * tt * tt)

    # 5) two-peak block
    x = np.full(n, 0.5)
    x[: n // 4] = 1.6
    x[3 * n // 4 :] = 1.6
    add(x)

    # 6) single interior block
    x = np.zeros(n)
    lo, hi = n // 3, 2 * n // 3
    x[lo:hi] = 1.0
    add(x + 1e-3)

    # 7) smooth raised-cosine window
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * tt))
    add(0.2 + w)

    # 8) anti-symmetric-ish ramp
    add(np.linspace(1.0, 0.0, n) ** 2 + 1e-3)

    if not inits:
        inits.append(np.ones(n) / n)
    return inits


# --------------------------------------------------------------------------- #
# Bounded optimizer from one start point
# --------------------------------------------------------------------------- #
def _optimize(a_init, time_limit, rng):
    global _MOM
    a = a_init.copy()
    a = _project_simplex(a)
    n = a.size
    _MOM = np.zeros(n)

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(c.max())          # score on the simplex

    mom = 0.9
    lr = 0.6
    top_m = 4
    stall = 0
    start = time.monotonic()
    it = 0
    best_pt = best_score

    while time.monotonic() - start < time_limit:
        a = _sgd_step(a, c, top_m, lr, mom, rng, jitter_scale=jitter_scale or 1.0)
        it += 1

        # recompute correlation every few iters (cheap, keeps maxc accurate)
        if it % 2 == 0:
            c = _fft_convolve(a)
            sc = 2.0 * n * float(c.max())
            if sc < best_score - 1e-12:
                # accept (also keep a tiny plate in case eval disagrees)
                accepted = sc
                if accepted < best_score:
                    best_score = accepted
                    best_a = a.copy()
                    best_pt = best_score
                    stall = 0
                    top_m = min(8, top_m + 1)        # widen the active set a touch
                else:
                    stalled = True
                    stall += 1
                if stalled:
                    stalled = False
                    lr *= (0.9 ** min(stall, 6))
                    if stall > 8:
                        # hard restart of momentum + a random nudge
                        mom = 0.85
                        lr = max(0.05, lr)
                        a = _project_simplex(a + 0.02 * n * rng.standard_normal(n) * (1.0 / n))
                        stall = 0
                # gradual learning-rate decay
                lr = max(0.02, lr * 0.99985)
    return best_a, best_score
