def mirror_descent(a, time_limit):
    """Exponentiated gradient / mirror descent for the peak minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    eta = 1.0  # learning rate
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Compute weighted subgradient
        # Weight peaks by their closeness to max
        threshold = M * 0.9  # consider peaks above 90% of max
        peaks = np.where(conv > threshold)[0]
        
        # Softmax weights over peaks
        w = np.exp(conv[peaks] * 100)  # sharp weighting on max
        w = w / w.sum()
        
        # Gradient
        grad = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-10:
                continue
            # d conv[k] / d a[m] = 2*a[k-m] if 0<=k-m<n
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            grad[m_lo:m_hi+1] += 2 * w[i] * a[k - np.arange(m_lo, m_hi+1)]
        
        # Mirror descent update (exponentiated gradient)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Adaptive learning rate
        eta *= 0.999
        
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

- This gives: min_g 2 * sum_k w_k * (a*a)[k] (where w_k is a subgradient weight).

1. Compute conv = f*f, find k* = argmax.

3. Solve: min <s, g> subject to g in simplex.

        if max_g > 0:

            a_new[m_idx] *= (1 - reduction)

    a_new = a_new / a_new.sum()
