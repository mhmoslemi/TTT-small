"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe (full or reduced constraints)
2. Adam with log-sum-exp smoothed subgradient
3. Peak-focused local search (coordinate descent on convolution peaks)
4. Multi-n exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def conv_max(f):
    return fast_convolve(f).max()

def build_lp_matrix(f, positions):
    """Build constraint matrix for LP. positions is array of k values."""
    n = len(f)
    A = np.zeros((len(positions), n))
    for idx, k in enumerate(positions):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # A[idx, j] = f[k-j] for j in [js, je]
            # k-j ranges from k-je to k-js
            A[idx, js:je+1] = f[k-je:k-js+1][::-1]
    return A

def lp_fw_step(f, M, n, use_reduced=False, top_k=100):
    """One LP-FW step: find direction g, return g (normalized)."""
    if use_reduced and n > 256:
        conv_ff = fast_convolve(f)
        positions = np.argsort(conv_ff)[::-1][:top_k]
    else:
        positions = np.arange(2 * n - 1)
    
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, positions)
    b_ub = np.full(len(positions), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.fun > -1e10:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None

def lp_fw_optimize(f0, time_limit, use_reduced=False, top_k=80, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = normalize(f0.astype(np.float64))
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Solve LP
        if use_reduced and n > 256:
            g = lp_fw_step(f, M, n, use_reduced=True, top_k=top_k)
        else:
            g = lp_fw_step(f, M, n, use_reduced=False)
        
        if g is None:
            # Try with slightly relaxed M
            g = lp_fw_step(f, M * 1.01, n, use_reduced=use_reduced, top_k=top_k)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    # Random restart
                    rng = np.random.default_rng(it * 7919 + 42)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    M = conv_max(f)
                    no_improve = 0
                continue
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 7919 + 42)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                M = conv_max(f)
                no_improve = 0
            continue
        
        # Line search: try a few alphas
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = normalize(ft)
            val = conv_max(ft)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f = normalize(f_new)
            M = conv_max(f)
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 * min(1 + no_improve // 50, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = conv_max(f)
            no_improve = 0
    
    return best_f, best_obj

def adam_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on smoothed max objective."""
    f = normalize(f0.astype(np.float64))
    n = len(f)
    
    best_obj = conv_max(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    T_start = 10.0 / n
    T_end = 0.3 / n
    base_lr = 0.003
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max(1, int(time_limit / 0.01)))
        T = T_start * (T_end / T_start) ** progress
        lr = base_lr * (1.0 - 0.3 * progress)
        
        conv = fast_convolve(f)
        
        # Top-k weighted subgradient
        k_actual = min(max(20, n // 4), 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Adam update
        it1 = it
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 500:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                continue
            continue
        
        f = f_new
        new_M = conv_max(f)
        
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      Adam iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
    
    return best_f, best_obj

def peak_local_search(f0, time_limit, verbose=False):
    """Local search focused on reducing the convolution peak."""
    f = normalize(f0.astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_obj = conv_max(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break
        
        conv = fast_convolve(f)
        # Find the top few peaks
        top_k = min(5, 2*n-1)
        peak_positions = np.argsort(conv)[::-1][:top_k]
        
        improved = False
        for k in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            # The convolution at position k: sum_i f[i]*f[k-i]
            # Find the most contributing index pair
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contributions = f[js:je+1] * f[k-je:k-js+1][::-1]
            
            # Find the index with the largest contribution
            if len(contributions) == 0:
                continue
            max_idx = js + np.argmax(contributions)
            
            # Try moving mass from max_idx to a "safe" position
            # A safe position is one that doesn't contribute to any top peak
            # Simple heuristic: move to a random position far from the peaks
            for _ in range(3):
                target = rng.integers(0, n)
                if target == max_idx:
                    continue
                
                c = f[max_idx] + f[target]
                if c < 1e-12:
                    continue
                
                # Try different splits
                for t_frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    f_trial = f.copy()
                    f_trial[max_idx] = (1-t_frac) * c
                    f_trial[target] = t_frac * c
                    f_trial = normalize(f_trial)
                    val = conv_max(f_trial)
                    if val < best_obj - 1e-13:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        
        # Occasionally add noise to escape
        if no_improve % 50 == 49 and no_improve > 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
    
    return best_f, best_obj

def pairwise_optimize(f0, time_limit, verbose=False):
    """Pairwise coordinate optimization."""
    f = normalize(f0.astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_obj = conv_max(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 3:
            break
        
        pass_improved = False
        order = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            # Pick a few random partners
            partners = rng.choice(n, size=min(5, n-1), replace=False)
            
            for j in partners:
                if i == j:
                    continue
                c = f[i] + f[j]
                if c < 1e-12:
                    continue
                
                best_t = f[i] / c
                best_val = best_obj
                
                for t in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0, best_t]:
                    t = np.clip(t, 0, 1)
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1-t) * c
                    f_trial = normalize(f_trial)
                    val = conv_max(f_trial)
                    if val < best_val - 1e-13:
                        best_val = val
                        best_t = t
                
                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = normalize(f)
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
