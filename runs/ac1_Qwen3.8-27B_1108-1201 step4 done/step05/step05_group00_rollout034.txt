```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Mirror descent in softmax parameterization for smooth optimization
- 2-coordinate swap with golden-section search for exact local improvements
- Simulated annealing for global exploration
- Multi-n exploration and diverse restarts

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(conv, n):
    """Objective = 2*n*max(conv) (assuming sum(a)=1)."""
    return 2.0 * n * conv.max()


def softmax(v, T=1.0):
    """Stable softmax."""
    v_scaled = v / T
    v_max = v_scaled.max()
    e = np.exp(v_scaled - v_max)
    return e / e.sum()


def mirror_descent(a0, time_limit, T_init=1.0, eta_init=0.1, verbose=False):
    """
    Mirror descent in softmax parameterization.
    a = softmax(theta). Minimize max(conv(a,a)) by gradient descent on theta.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        theta = np.zeros(len(a))
    else:
        a = a / s
        # Recover theta from a: theta[i] = log(a[i]) (up to a constant)
        theta = np.log(np.maximum(a, 1e-300))
    
    n = len(a)
    T = T_init
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        a = softmax(theta, T)
        conv = fast_convolve(a)
        
        # Subgradient of max(conv): sum over max positions of dc/da
        # dc[k]/da[j] = 2*a[k-j] (with a[m]=0 for m outside [0,n-1])
        peak_idx = np.argmax(conv)
        # Subgradient at the peak
        k = peak_idx
        grad = 2.0 * a[(k - np.arange(n)) % n]  # This isn't right for linear conv
        
        # Better: compute subgradient properly
        # dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n, else 0
        js = np.arange(n)
        valid = (k - js >= 0) & (k - js < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[k - js[valid]]
        
        # Gradient in theta space: d max(conv)/dtheta[j] = sum_i (d max(conv)/da[i]) * (a[j]*delta_ij - a[j]*a[i])
        # = a[j] * (d max(conv)/da[j] - sum_i a[i] * d max(conv)/da[i])
        grad_a = grad  # subgradient w.r.t. a
        grad_theta = a * (grad_a - (a * grad_a).sum())
        
        # Update theta
        theta = theta - eta * grad_theta
        a_new = softmax(theta, T)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Roll back if not improving
            a_new = a
        
        # Adapt
        if no_improve > 20:
            eta *= 0.9
            T *= 1.02
        if no_improve > 100:
            eta *= 0.5
            T = T_init
            no_improve = 50
        
        if verbose and it % 200 == 0:
            print(f"    MD it={it}, obj={best_obj:.8f}, eta={eta:.6f}, T={T:.4f}")
    
    return best_a, best_obj

    # After change: a'[j] = a[j]-delta, a'[k] = a[k]+delta, a'[i] = a[i] for i not in {j,k}
    #
    # c'[m] - c[m] = sum_i a'[i]*a'[m-i] - sum_i a[i]*a[m-i]
    # = (terms with i=j or m-i=j or i=k or m-i=k)
    #
    # Let me just compute it directly for the code. For a given delta, the new conv is:
    # c'[m] = c[m] + delta_effect(m)
    # where delta_effect can be computed in O(1) per m.
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Pick two random coordinates to swap
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        if j == k:
            k = (j + 1) % n
        
        # We want to find delta that minimizes max(c + delta_effect)
        # delta_effect[m] is the change in c[m] when we transfer delta from j to k
        # 
        # c'[m] - c[m] = (a[j]-delta)*a[m-j] + a[m-j]*(a[j]-delta) - a[j]*a[m-j] - a[m-j]*a[j]
        #                 + (a[k]+delta)*a[m-k] + a[m-k]*(a[k]+delta) - a[k]*a[m-k] - a[m-k]*a[k]
        #                 - (cross terms if m-j==k or m-k==j)
        #
        # This is getting complicated. Let me just compute the new conv directly for candidate deltas.
        
        # Better approach: compute the effect analytically
        # c[m] involves a[j] in positions where i=j or m-i=j
        # and a[k] in positions where i=k or m-i=k
        #
        # Effect of changing a[j] by -delta:
        #   For m where 0<=m-j<n: c[m] changes by -delta*a[m-j] (from i=j term)
        #   For m where 0<=m+j<n... wait, m-i=j means i=m-j, so we need 0<=m-j<n
        #   Actually: c[m] = sum_i a[i]*a[m-i]
        #   Terms with a[j]: i=j gives a[j]*a[m-j]; i=m-j gives a[m-j]*a[j]
        #   So a[j] appears in c[m] as: a[j]*a[m-j] (if 0<=m-j<n) + a[m-j]*a[j] (if 0<=m-j<n)
        #   = 2*a[j]*a[m-j] (if 0<=m-j<n and m-j != j)
        #   = a[j]^2 (if m=2j)
        #
        # When a[j] -> a[j]-delta:
        #   Change in c[m] from j's terms:
        #   If m=2j: (a[j]-delta)^2 - a[j]^2 = -2*delta*a[j] + delta^2
        #   Else (0<=m-j<n): 2*(a[j]-delta)*a[m-j] - 2*a[j]*a[m-j] = -2*delta*a[m-j]
        #
        # Similarly for a[k] -> a[k]+delta:
        #   If m=2k: 2*delta*a[k] + delta^2
        #   Else (0<=m-k<n): 2*delta*a[m-k]
        #
        # But we also need to handle the cross-term: if m-j=k (i.e., m=j+k), then the term
        # a[j]*a[k] in c[j+k] changes to (a[j]-delta)*(a[k]+delta)
        # Change: a[j]*delta - delta*a[k] - delta^2
        # This appears once in c[j+k] from i=j (giving a[j]*a[k]) and once from i=k (giving a[k]*a[j])
        # So total change at m=j+k: 2*(a[j]*delta - delta*a[k] - delta^2)
        # But wait, this cross-term is already included in the individual effects above!
        # Because when m=j+k, the j-effect gives -2*delta*a[m-j] = -2*delta*a[k]
        # and the k-effect gives +2*delta*a[m-k] = +2*delta*a[j]
        # Sum: 2*delta*(a[j]-a[k])
        # But the actual change should be: 2*(a[j]-delta)*(a[k]+delta) - 2*a[j]*a[k]
        # = 2*(a[j]*a[k] + a[j]*delta - delta*a[k] - delta^2) - 2*a[j]*a[k]
        # = 2*(a[j]*delta - delta*a[k] - delta^2)
        # = 2*delta*(a[j]-a[k]) - 2*delta^2
        #
        # The individual effects give: 2*delta*(a[j]-a[k])
        # Missing: -2*delta^2
        #
        # So there's a second-order correction at m=j+k: -2*delta^2
        #
        # Similarly, if j+k >= n, then m=j+k might be out of range for the convolution
        # (conv has length 2n-1, indices 0 to 2n-2, so j+k is always valid since max j+k = 2n-2)
        
        # OK so the total effect at position m:
        # linear_coef[m] = coefficient of delta
        # quad_coef[m] = coefficient of delta^2
        # 
        # c'[m] = c[m] + linear_coef[m]*delta + quad_coef[m]*delta^2
        
        linear = np.zeros(2*n-1)
        quad = np.zeros(2*n-1)
        
        # j's contribution (a[j] -> a[j]-delta)
        for m in range(2*n-1):
            mj = m - j
            if 0 <= mj < n:
                if m == 2*j:
                    linear[m] += -2*a[j]  # -2*delta*a[j]
                    quad[m] += 1  # +delta^2
                else:
                    linear[m] += -2*a[mj]
        
        # k's contribution (a[k] -> a[k]+delta)
        for m in range(2*n-1):
            mk = m - k
            if 0 <= mk < n:
                if m == 2*k:
                    linear[m] += 2*a[k]
                    quad[m] += 1
                else:
                    linear[m] += 2*a[mk]
        
        # Cross-term correction at m = j+k
        m_cross = j + k
        quad[m_cross] += -2  # -2*delta^2
        
        # Also need: delta must keep a[j]-delta >= 0 and a[k]+delta >= 0
        # So: -a[k] <= delta <= a[j]
        delta_lo = -a[k] + 1e-15
        delta_hi = a[j] - 1e-15
        
        if delta_hi <= delta_lo:
            continue
        
        # Find delta in [delta_lo, delta_hi] that minimizes max_m (c[m] + linear[m]*delta + quad[m]*delta^2)
        # This is a 1D optimization problem. Use golden section or ternary search.
        # Since the function is max of convex functions (each c[m] + linear[m]*delta + quad[m]*delta^2 is convex in delta)
        # Wait, quad[m] can be negative (at m=j+k), so it's not necessarily convex.
        # Let's just use a grid + refinement approach.
        
        conv = fast_convolve(a) if it == 1 else None
        if conv is None:
            # We need conv; let's compute it
            conv = fast_convolve(a)
        
        # Quick grid search
        best_delta = 0.0
        best_max = conv.max()
        
        # Try a grid of deltas
        n_grid = 30
        deltas = np.linspace(delta_lo, delta_hi, n_grid)
        for d in deltas:
            new_conv = conv + linear*d + quad*d*d
            mx = new_conv.max()
            if mx < best_max:
                best_max = mx
                best_delta = d
        
        if best_delta == 0.0:
            no_improve += 1
        else:
            # Refine around best_delta
            d_lo = max(delta_lo, best_delta - (delta_hi - delta_lo) / n_grid)
            d_hi = min(delta_hi, best_delta + (delta_hi - delta_lo) / n_grid)
            for _ in range(15):
                d_mid = (d_lo + d_hi) / 2
                new_conv = conv + linear*d_mid + quad*d_mid*d_mid
                mx = new_conv.max()
                if mx < best_max:
                    best_max = mx
                    best_delta = d_mid
                    d_lo = d_mid if best_max < (conv + linear*d_lo + quad*d_lo*d_lo).max() else d_hi
                    # Actually let's do proper bisection
                    # If mid is better, the minimum is in [mid, hi] or [lo, mid]
                    # This is tricky for max-of-convex. Just do a fine grid.
                    d_lo = max(delta_lo, d_mid - (d_hi - d_lo) / 2)
                    d_hi = min(delta_hi, d_mid + (d_hi - d_lo) / 2)
                else:
                    d_lo = min(delta_hi, d_mid + (d_hi - d_lo) / 2)
                    d_hi = max(delta_lo, d_mid - (d_hi - d_lo) / 2)
            
            # Apply
            a[j] -= best_delta
            a[k] += best_delta
            a = np.maximum(a, 0)
            a = a / a.sum()  # Renormalize (should already be sum=1)
            
            conv = fast_convolve(a)
            mx = conv.max()
            if mx < best_obj - 1e-15:
                best_obj = mx
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            rng2 = np.random.default_rng((it * 31 + 7) % (2**31))
            a = best_a + rng2.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Softmax-parameterized gradient descent (mirror descent)
- 2-coordinate swap with analytical 1D optimization
- Simulated annealing for global exploration
- Multi-phase with diverse restarts
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_params(a):
    """Convert probability vector to log-parameters."""
    return np.log(np.maximum(a, 1e-300))


def softmax_from_params(theta):
    """Convert log-parameters to probability vector."""
    t = theta - theta.max()
    e = np.exp(t)
    return e / e.sum()


def subgradient_of_max_conv(a, conv, top_k=1):
    """
    Compute subgradient of max(conv) w.r.t. a.
    For the single peak: dc[k*]/da[j] = 2*a[k*-j] (if 0<=k*-j<n)
    For top-k peaks: sum of subgradients weighted equally.
    """
    n = len(a)
    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
        weights = np.ones(len(peak_idx)) / len(peak_idx)
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        weights = np.ones(len(peak_idx)) / len(peak_idx)
    
    grad = np.zeros(n)
    for k, w in zip(peak_idx, weights):
        k = int(k)
        js = np.arange(n)
        valid = (k - js >= 0) & (k - js < n)
        grad[valid] += w * 2.0 * a[k - js[valid]]
    
    return grad


def softmax_gradient_descent(a0, time_limit, top_k=5, eta_init=0.1, verbose=False):
    """
    Gradient descent in softmax (log) parameterization.
    a = softmax(theta). Minimizes max(conv(a,a)).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    theta = softmax_params(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        a = softmax_from_params(theta)
        conv = fast_convolve(a)
        
        # Subgradient w.r.t. a
        grad_a = subgradient_of_max_conv(a, conv, top_k)
        
        # Gradient w.r.t. theta: chain rule
        # da[i]/dtheta[j] = a[i]*(delta_ij - a[j])
        # d max(conv)/dtheta[j] = sum_i grad_a[i] * a[i]*(delta_ij - a[j])
        # = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
        mean_grad = (a * grad_a).sum()
        grad_theta = a * (grad_a - mean_grad)
        
        # Update with momentum-like decay
        theta = theta - eta * grad_theta
        
        a_new = softmax_from_params(theta)
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Don't update theta if not improving (or use a smaller step)
            theta = softmax_params(a)  # Revert
        
        # Adapt learning rate
        if no_improve > 10:
            eta *= 0.95
        elif no_improve > 50:
            eta *= 0.8
        if no_improve > 200:
            # Random restart
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            theta = softmax_params(best_a) + rng.standard_normal(n) * 0.01
            eta = eta_init * 0.5
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    SGD it={it}, obj={best_obj:.8f}, eta={eta:.6f}")
    
    return best_a, best_obj


def two_swap_sweep(a0, time_limit, n_sweeps_per_iter=100, verbose=False):
    """
    Coordinate descent: at each step, pick a random pair (j,k) and
    find the optimal delta to transfer mass from j to k.
    Uses O(n) analytical computation of the effect.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    # Precompute: for each j, which m values does a[j] affect?
    # a[j] affects c[m] for m in [j, j+n-1] (since m-j ranges from 0 to n-1)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        current_max = conv.max()
        
        # Try n_sweeps random pairs
        improved_this_iter = False
        for _ in range(n_sweeps_per_iter):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                k = (j + 1) % n
            
            # Compute linear and quadratic coefficients for the change
            # c'[m] = c[m] + lin[m]*delta + quad[m]*delta^2
            # where delta = amount transferred from j to k
            
            # a[j] -> a[j] - delta, a[k] -> a[k] + delta
            
            # Effect on c[m] from changing a[j]:
            # For m where 0 <= m-j < n: contribution is a[j]*a[m-j] (from i=j) + a[m-j]*a[j] (from i=m-j)
            #   = 2*a[j]*a[m-j] (if m != 2j) or a[j]^2 (if m = 2j)
            # Change: -2*delta*a[m-j] (if m != 2j) or -2*delta*a[j] + delta^2 (if m = 2j)
            
            lin = np.zeros(max(2*j, 2*k, j+k) + 1)  # Not efficient, let me use full size
            # Actually, let me just use the full convolution length
            lin = np.zeros(2*n - 1)
            quad = np.zeros(2*n - 1)
            
            # j's effect (subtracting delta)
            m_start_j = j  # m-j >= 0 -> m >= j
            m_end_j = j + n - 1  # m-j < n -> m < j+n
            for m in range(m_start_j, min(m_end_j + 1, 2*n-1)):
                mj = m - j
                if m == 2*j:
                    lin[m] += -2.0 * a[j]
                    quad[m] += 1.0
                else:
                    lin[m] += -2.0 * a[mj]
            
            # k's effect (adding delta)
            m_start_k = k
            m_end_k = k + n - 1
            for m in range(m_start_k, min(m_end_k + 1, 2*n-1)):
                mk = m - k
                if m == 2*k:
                    lin[m] += 2.0 * a[k]
                    quad[m] += 1.0
                else:
                    lin[m] += 2.0 * a[mk]
            
            # Cross-term correction at m = j+k
            m_cross = j + k
            if m_cross < 2*n - 1:
                quad[m_cross] += -2.0
            
            # Valid range for delta: a[j]-delta >= 0 and a[k]+delta >= 0
            delta_lo = max(0.0, -a[k]) + 1e-15  # delta >= -a[k]
            delta_hi = a[j] - 1e-15  # delta <= a[j]
            # Also delta can be negative (transfer from k to j)
            delta_lo = -a[k] + 1e-15
            delta_hi = a[j] - 1e-15
            
            if delta_hi <= delta_lo:
                continue
            
            # Minimize max_m (c[m] + lin[m]*delta + quad[m]*delta^2)
            # Use golden section search
            # The function f(delta) = max_m (c[m] + lin[m]*delta + quad[m]*delta^2)
            # is the max of quadratics, which is convex in delta.
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            a_d = delta_lo
            b_d = delta_hi
            c_g = b_d - gr * (b_d - a_d)
            d_g = a_d + gr * (b_d - a_d)
            
            f_c = conv + lin * c_g + quad * c_g * c_g
            f_d = conv + lin * d_g + quad * d_g * d_g
            fc_max = f_c.max()
            fd_max = f_d.max()
            
            for _ in range(20):
                if fc_max < fd_max:
                    b_d = d_g
                    d_g = c_g
                    fd_max = fc_max
                    c_g = b_d - gr * (b_d - a_d)
                    f_c = conv + lin * c_g + quad * c_g * c_g
                    fc_max = f_c.max()
                else:
                    a_d = c_g
                    c_g = d_g
                    fc_max = fd_max
                    d_g = a_d + gr * (b_d - a_d)
                    f_d = conv + lin * d_g + quad * d_g * d_g
                    fd_max = f_d.max()
            
            best_delta = (a_d + b_d) / 2
            new_conv = conv + lin * best_delta + quad * best_delta * best_delta
            new_max = new_conv.max()
            
            if new_max < current_max - 1e-12:
                a[j] -= best_delta
                a[k] += best_delta
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum < 1e-15:
                    a = best_a.copy()
                    break
                # Don't renormalize - we're maintaining sum=1 through the swap
                current_max = new_max
                improved_this_iter = True
        
        if improved_this_iter:
            a_sum = a.sum()
            if abs(a_sum - 1.0) > 1e-10:
                a = a / a_sum
            conv = fast_convolve(a)
            mx = conv.max()
            if mx < best_obj - 1e-14:
                best_obj = mx
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Perturb and restart
            rng2 = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = best_a + rng2.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, n_samples_per_step=5, verbose=False):
    """
    Simulated annealing with structured moves.
    Moves: pick 2-4 coordinates and redistribute their mass.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj
    
    T = 0.01  # Initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Generate a random move
        n_coords = int(rng.integers(2, 5))
        idx = rng.integers(0, n, size=n_coords)
        # Remove duplicates
        idx = np.unique(idx)
        if len(idx) < 2:
            idx = np.array([0, 1])
        
        # Save old values
        old_vals = a[idx].copy()
        old_sum = old_vals.sum()
        
        # Apply random redistribution
        new_vals = rng.dirichlet(np.ones(len(idx)) * 0.5) * old_sum
        a[idx] = new_vals
        
        new_conv = fast_convolve(a)
        new_obj = new_conv.max()
        
        # Accept/reject
        delta_obj = new_obj - current_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            current_obj = new_obj
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # Reject
            a[idx] = old_vals
        
        T *= cooling
        if T < T_min:
            T = 0.005  # Reheat
            # Perturb
            perturb_idx = rng.integers(0, n, size=n // 10)
            perturb_idx = np.unique(perturb_idx)
            if len(perturb_idx) >= 2:
                old = a[perturb_idx].copy()
                a[perturb_idx] = rng.dirichlet(np.ones(len(perturb_idx))) * old.sum()
                new_conv = fast_convolve(a)
                current_obj = new_conv.max()
    
    return best_a, best_obj


def subgradient_momentum(a0, time_limit, top_k=10, eta_init=0.005, momentum=0.9, verbose=False):
    """
    Subgradient method with momentum and adaptive step size.
    Projects to simplex after each step.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- 2-coordinate swap with vectorized analytical 1D optimization (primary)
- Softmax-parameterized subgradient descent (different geometry)
- Simulated annealing for global exploration
- Multi-phase with diverse restarts and n-values
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a, conv, top_k=5):
    """Subgradient of max(conv) w.r.t. a, averaged over top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
        weights = np.ones(len(peak_idx)) / len(peak_idx)
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        weights = np.ones(len(peak_idx)) / len(peak_idx)

    grad = np.zeros(n)
    for k, w in zip(peak_idx, weights):
        k = int(k)
        js = np.arange(n)
        valid = (k - js >= 0) & (k - js < n)
        grad[valid] += w * 2.0 * a[k - js[valid]]
    return grad

    2-coordinate swap coordinate descent.
    For each pair (j,k), analytically compute the effect of transferring mass,
    then use golden section search to find the optimal transfer.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    conv_len = 2 * n - 1

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    gr_val = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        improved_this_sweep = False
        for _ in range(pairs_per_sweep):
            if time.monotonic() - t0 > time_limit - 0.05:
                break

            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                k = (j + 1) % n

            # Compute linear and quadratic coefficients (vectorized)
            lin = np.zeros(conv_len)
            quad = np.zeros(conv_len)

            # j's effect: a[j] -> a[j] - delta
            m_start_j = j
            m_end_j = min(j + n, conv_len)
            if m_end_j > m_start_j:
                lin[m_start_j:m_end_j] -= 2.0 * a[m_start_j - j:m_end_j - j]
            if 2 * j < conv_len:
                quad[2 * j] += 1.0

            # k's effect: a[k] -> a[k] + delta
            m_start_k = k
            m_end_k = min(k + n, conv_len)
            if m_end_k > m_start_k:
                lin[m_start_k:m_end_k] += 2.0 * a[m_start_k - k:m_end_k - k]
            if 2 * k < conv_len:
                quad[2 * k] += 1.0

            # Cross-term at m = j+k
            m_cross = j + k
            if m_cross < conv_len:
                quad[m_cross] -= 2.0

            # Valid delta range: a[j]-delta >= 0, a[k]+delta >= 0
            delta_lo = -a[k] + 1e-14
            delta_hi = a[j] - 1e-14
            if delta_hi <= delta_lo:
                continue

            # Golden section search for min max(conv + lin*delta + quad*delta^2)
            a_d = delta_lo
            b_d = delta_hi
            c_g = b_d - gr_val * (b_d - a_d)
            d_g = a_d + gr_val * (b_d - a_d)

            fc_val = (conv + lin * c_g + quad * c_g * c_g).max()
            fd_val = (conv + lin * d_g + quad * d_g * d_g).max()

            for _ in range(gold_iters):
                if fc_val < fd_val:
                    b_d = d_g
                    d_g = c_g
                    fd_val = fc_val
                    c_g = b_d - gr_val * (b_d - a_d)
                    fc_val = (conv + lin * c_g + quad * c_g * c_g).max()
                else:
                    a_d = c_g
                    c_g = d_g
                    fc_val = fd_val
                    d_g = a_d + gr_val * (b_d - a_d)
                    fd_val = (conv + lin * d_g + quad * d_g * d_g).max()

            best_delta = (a_d + b_d) / 2.0
            new_max = (conv + lin * best_delta + quad * best_delta * best_delta).max()

            if new_max < current_max - 1e-13:
                a[j] -= best_delta
                a[k] += best_delta
                a = np.maximum(a, 0.0)
                current_max = new_max
                improved_this_sweep = True

        a_sum = a.sum()
        if a_sum < 1e-15:
            a = best_a.copy()
        elif abs(a_sum - 1.0) > 1e-10:
            a = a / a_sum

        if improved_this_sweep:
            new_conv = fast_convolve(a)
            mx = new_conv.max()
            if mx < best_obj - 1e-14:
                best_obj = mx
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 300:
            rng2 = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 300) / 500.0)
            a = project_simplex(best_a + rng2.standard_normal(n) * noise)
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    2-swap it={it}, obj={best_obj:.8f}, no_improve={no_improve}")

    return best_a, best_obj


def softmax_subgrad(a0, time_limit, top_k=5, eta_init=0.05, verbose=False):
    """
    Subgradient descent in softmax (log) parameterization.
    a = softmax(theta). Different geometry from Euclidean projection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    theta = np.log(np.maximum(a, 1e-300))
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(77)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        t_theta = theta - theta.max()
        e = np.exp(t_theta)
        a = e / e.sum()
        conv = fast_convolve(a)

        grad_a = subgradient_of_max_conv(a, conv, top_k)

        # Gradient in theta space
        mean_g = np.dot(a, grad_a)
        grad_theta = a * (grad_a - mean_g)

        theta_new = theta - eta * grad_theta
        t_new = theta_new - theta_new.max()
        e_new = np.exp(t_new)
        a_new = e_new / e_new.sum()

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            theta = theta_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            eta *= 0.93
        if no_improve > 80:
            eta *= 0.7
        if no_improve > 300:
            theta = np.log(np.maximum(best_a, 1e-300))
            theta = theta + rng.standard_normal(n) * 0.005
            eta = eta_init * 0.5
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    softmax it={it}, obj={best_obj:.8f}, eta={eta:.6f}")

    return best_a, best_obj


def block_swap_anneal(a0, time_limit, block_size=5, verbose=False):
    """
    Simulated annealing with block moves.
    Pick a random block of coordinates and redistribute their mass.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    T = 0.005
    T_min = 1e-7
    cooling = 0.9995

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Pick random block
        start = int(rng.integers(0, n - block_size))
        idx = np.arange(start, start + block_size)
        old_vals = a[idx].copy()
        old_sum = old_vals.sum()

        if old_sum < 1e-15:
            # Pick a different block with non-zero mass
            idx = rng.integers(0, n, size=block_size)
            idx = np.unique(idx)
            if len(idx) < 2:
                continue
            old_vals = a[idx].copy()
            old_sum = old_vals.sum()
            if old_sum < 1e-15:
                continue

        # Random Dirichlet redistribution
        alpha_param = 0.3 + 2.0 * T  # More random at high T
        new_vals = rng.dirichlet(np.ones(len(idx)) * alpha_param) * old_sum
        a[idx] = new_vals

        new_conv = fast_convolve(a)
        new_obj = new_conv.max()

        # Metropolis acceptance
        delta_obj = new_obj - current_obj
        if delta_obj < 0 or (T > T_min and rng.random() < np.exp(-delta_obj / T)):
            current_obj = new_obj
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a.copy()
        else:
            a[idx] = old_vals

        T = max(T * cooling, T_min)
        if T <= T_min * 1.01:
            # Reheat
            T = 0.003
            # Perturb
            pidx = rng.integers(0, n, size=max(3, n // 20))
            pidx = np.unique(pidx)
            if len(pidx) >= 2:
                pv = a[pidx].copy()
                ps = pv.sum()
                if ps > 1e-15:
                    a[pidx] = rng.dirichlet(np.ones(len(pidx)) * 0.5) * ps
                    current_obj = fast_convolve(a).max()

    return best_a, best_obj


def subgrad_momentum(a0, time_limit, top_k=10, eta_init=0.003, momentum=0.92, verbose=False):
    """Subgradient with Heavy Ball momentum and simplex projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(55)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        grad_a = subgradient_of_max_conv(a, conv, top_k)

        v = momentum * v - eta * grad_a
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        eta *= 0.9998
        momentum = min(0.97, momentum + 0.00005)

        if no_improve > 100:
            eta *= 0.9
        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            eta = eta_init * 0.6
            no_improve = 0

    return best_a, best_obj
```
