import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_qp_subproblem(p, n, max_iter=500):
    """Solve min a^T H(p) a s.t. a >= 0, sum(a) = 1 using active set."""
    active = set(range(n))
    
    for it in range(max_iter):
        S = sorted(active)
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            a = np.zeros(n)
            a[S[0]] = 1.0
            return a

        # Build sub-matrix H[S,S]
        H_SS = np.array([[p[S[i] + S[j]] for j in range(k)] for i in range(k)])
        
        # Solve [2*H_SS, 1; 1^T, 0] [x; lambda] = [0; 1]
        KKT = np.zeros((k + 1, k + 1))
        KKT[:k, :k] = 2.0 * H_SS
        KKT[:k, k] = 1.0
        KKT[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0
        
        try:
            sol = np.linalg.solve(KKT, rhs)
            x = sol[:k]
            lam = sol[k]
        except np.linalg.LinAlgError:
            sol, _, _, _ = np.linalg.lstsq(KKT, rhs, rcond=None)
            x = sol[:k]
            lam = sol[k]

        # Check for negative entries
        removed = False
        new_active = set()
        for i, idx in enumerate(S):
            if x[i] < -1e-14:
                removed = True
            else:
                new_active.add(idx)
        
        if removed:
            active = new_active
            continue

        # Build full solution
        a = np.zeros(n)
        for i, idx in enumerate(S):
            a[idx] = max(0.0, x[i])

        # Compute lambda from KKT (recompute for stability)
        Ma = hankel_mv(p, a)
        lam = -np.mean([2.0 * Ma[idx] for idx in S])

        # Check KKT for inactive variables
        added = False
        for j in range(n):
            if j not in active:
                if 2.0 * Ma[j] + lam < -1e-12:
                    active.add(j)
                    added = True

        if not added:
            return a

    # Fallback
    a = np.zeros(n)
    a[list(active)] = 1.0 / len(active)
    return a


def fw_lp_optimize(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_obj = M
    best_f = f.copy()

    no_improve = 0

    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:

                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.06, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent using log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 30:
            T *= 1.08
            lr *= 0.95
        if no_improve > 80:
            T *= 0.7
        if no_improve > 250:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.005 * min(1.0, (no_improve - 250) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def game_iterate(a0, time_limit, T_init=0.001):

def population_search(a0, time_limit, pop_size=80, n_gen_max=5000):
    """Population-based search with structured mutations."""
    n = len(a0)
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()

    # Initialize population
    rng = np.random.default_rng(42)
    pop = [a0.copy()]
    for _ in range(pop_size - 1):
        noise_scale = rng.uniform(0.003, 0.02)
        candidate = project_simplex(a0 + rng.standard_normal(n) * noise_scale)
        pop.append(candidate)

    # Evaluate initial population
    objs = []
    for i, a in enumerate(pop):
        objs.append(fast_convolve(a).max())

    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit and gen < n_gen_max:
        gen += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        # Sort by objective
        order = np.argsort(objs)
        pop_arr = np.array(pop)
        pop_sorted = pop_arr[order]
        objs_sorted = np.array(objs)[order]

        # Update best
        if objs_sorted[0] < best_obj - 1e-15:
            best_obj = objs_sorted[0]
            best_a = pop_sorted[0].copy()

        # Early termination if converged
        if gen > 50 and objs_sorted[0] * 1.0001 < objs_sorted[-1]:
            # Check if we're making progress
            if gen > 100 and objs_sorted[0] < best_obj + 1e-12:
                break

        # Create new generation
        new_pop = []
        
        # Elitism: keep top half
        n_elite = pop_size // 2
        new_pop.extend(list(pop_sorted[:n_elite]))

        # Create offspring
        n_offspring = pop_size - n_elite
        for _ in range(n_offspring):
            mutation_type = rng.integers(0, 5)
            
            if mutation_type == 0:
                # Gaussian mutation from a random elite
                parent_idx = rng.integers(0, n_elite)
                scale = rng.uniform(0.001, 0.01)
                child = project_simplex(pop_sorted[parent_idx] + rng.standard_normal(n) * scale)
                
            elif mutation_type == 1:
                # Crossover
                p1 = rng.integers(0, n_elite)
                p2 = rng.integers(0, n_elite)
                alpha = rng.uniform(0.2, 0.8)
                child = project_simplex((1 - alpha) * pop_sorted[p1] + alpha * pop_sorted[p2])
                
            elif mutation_type == 2:
                # Local swap: pick two random positions, swap values
                parent_idx = rng.integers(0, n_elite)
                parent = pop_sorted[parent_idx].copy()
                i, j = rng.integers(0, n, size=2)
                new_val_i = parent[j]
                new_val_j = parent[i]
                parent[i] = new_val_i
                parent[j] = new_val_j
                child = np.maximum(parent, 0)
                s = child.sum()
                if s > 1e-10:
                    child = child / s
                else:
                    child = pop_sorted[parent_idx]
                    
            elif mutation_type == 3:
                # Block shift: shift a random block of values
                parent_idx = rng.integers(0, n_elite)
                parent = pop_sorted[parent_idx].copy()
                block_len = rng.integers(2, min(20, n // 4))
                start = rng.integers(0, max(1, n - block_len))
                shift = rng.integers(-3, 4)
                if shift != 0:
                    new_parent = parent.copy()
                    new_parent[start + shift:start + shift + block_len] = parent[start:start + block_len]
                    new_parent[start:start + block_len] = 0
                    child = np.maximum(new_parent, 0)
                    s = child.sum()
                    if s > 1e-10:
                        child = child / s
                    else:
                        child = parent
                else:
                    child = parent
            else:
                # Perturb top contributors to peak
                parent_idx = rng.integers(0, n_elite)
                parent = pop_sorted[parent_idx].copy()
                conv = fast_convolve(parent)
                k_peak = int(np.argmax(conv))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                affected = np.arange(js, je + 1)
                if len(affected) > 0:
                    noise = rng.standard_normal(len(affected)) * 0.005
                    perturbed = parent.copy()
                    perturbed[affected] += noise
                    child = project_simplex(perturbed)
                else:
                    child = project_simplex(parent + rng.standard_normal(n) * 0.002)

            new_pop.append(child)

        # Evaluate new population
        pop = new_pop
        new_objs = []
        for a in pop:
            objs.append(fast_convolve(a).max())
        # Take last pop_size objects
        objs = objs[-pop_size:]

        # Periodically re-center around best
        if gen % 100 == 0:
            for i in range(pop_size):
                if rng.random() < 0.3:
                    pop[i] = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                    pop[i] = np.maximum(pop[i], 0)
                    s = pop[i].sum()
                    if s > 1e-10:
                        pop[i] = pop[i] / s

    return best_a, best_obj


def coordinate_descent_peak(a0, time_limit):
    """Coordinate descent focusing on reducing the peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Find affected indices
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)

        if len(affected) < 2:
            break

        # Try to reduce the peak by adjusting affected coordinates
        peak_val = conv[k_peak]
        
        # Strategy: move mass from high-contributing to low-contributing
        # Contribution of a[j] to peak: a[j] * a[k_peak - j]
        contributions = np.array([a[j] * a[k_peak - j] for j in affected])
        
        # Find the highest and lowest contributing pairs
        max_idx_local = np.argmax(contributions)
        min_idx_local = np.argmin(contributions)
        max_j = affected[max_idx_local]
        min_j = affected[min_idx_local]

        # Try several epsilon values
        improved = False
        for eps in [0.0001, 0.0003, 0.001, 0.003, 0.01]:
            if a[max_j] < eps:
                continue
            candidate = a.copy()
            candidate[max_j] -= eps
            candidate[min_j] += eps
            candidate = np.maximum(candidate, 0)
            s = candidate.sum()
            if s < 1e-10:
                continue
            candidate = candidate / s
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try random perturbation of affected coords
            rng = np.random.default_rng((it * 999983) % (2**31))
            for _ in range(3):
                noise = np.zeros(n)
                noise[affected] = rng.standard_normal(len(affected)) * 0.001
                candidate = project_simplex(a + noise)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    break

        # Periodic restart
        if it % 500 == 0:
            rng = np.random.default_rng(it * 42)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Anti-symmetric (high at edges, low at center)
    anti = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    anti = 1.0 - anti
    anti = np.maximum(anti, 0.01)
    inits.append(anti / anti.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / max(n - 1, 1) - 0.5)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search with population-based refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent]))
    for n_try in [n_parent - 64, n_parent + 64, n_parent // 2, n_parent * 2, 512, 768, 1024]:
        if 128 <= n_try <= 2048 and n_try not in candidate_ns:
            candidate_ns.append(n_try)

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - (budget_s * 0.55)
        if remaining < 15:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 25)
        print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"fw_n{n_try}")

    # Quick gradient pass on best so far
    if best_seq is not None:
        remaining = deadline - time.monotonic() - (budget_s * 0.5)
        if remaining > 10:
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=min(remaining, 20),
                                             T_init=0.001, lr_init=0.03)
            try_seq(a_opt, "quick_grad")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP refinement ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - (budget_s * 0.4)
        if remaining > 20:
            t_alloc = min(remaining, 250)
            print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

            # Perturbed restarts
            for scale in [0.005, 0.01, 0.02, 0.03]:
                remaining = deadline - time.monotonic() - (budget_s * 0.3)
                if remaining < 15:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(remaining, 60))
                try_seq(a_opt2, f"fw_pert_{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Game iterate + Smoothed gradient ===
    print("\n=== Phase 3: Game Iterate & Smoothed Gradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Game iterate
        remaining = deadline - time.monotonic() - (budget_s * 0.25)
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Game iterate, t={t_alloc:.0f}s")
            a_opt, _ = game_iterate(best_seq, time_limit=t_alloc, T_init=0.001)
            try_seq(a_opt, "game")

            # Second game iterate with different T
            remaining = deadline - time.monotonic() - (budget_s * 0.2)
            if remaining > 15:
                t_alloc = min(remaining, 80)
                a_opt, _ = game_iterate(best_seq, time_limit=t_alloc, T_init=0.005)
                try_seq(a_opt, "game_T0.005")

        # Smoothed gradient with different settings
        for T_init, lr_init, label in [(0.0005, 0.02, "sg_T5e-4"),
                                        (0.002, 0.03, "sg_T2e-3"),
                                        (0.0001, 0.01, "sg_T1e-4")]:
            remaining = deadline - time.monotonic() - (budget_s * 0.15)
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Smoothed grad {label}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_seq(a_opt, label)

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Population-based search ===
    print("\n=== Phase 4: Population Search ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - (budget_s * 0.12)
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  Population search, t={t_alloc:.0f}s")
            a_opt, _ = population_search(best_seq, time_limit=t_alloc, pop_size=60)
            try_seq(a_opt, "population")

            # Second round from best
            remaining = deadline - time.monotonic() - (budget_s * 0.08)
            if remaining > 15:
                t_alloc = min(remaining, 100)
                a_opt, _ = population_search(best_seq, time_limit=t_alloc, pop_size=40)
                try_seq(a_opt, "population2")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate descent + fine polish ===
    print("\n=== Phase 5: Fine Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent on peaks
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_peak(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

        # Very fine smoothed gradient
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Fine smooth grad, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                             T_init=0.00005, lr_init=0.005)
            try_seq(a_opt, "fine_smooth")

        # Final FW-LP
        remaining = deadline - time.monotonic() - 25
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            for attempt in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining, 8),
                                                 T_init=0.0001, lr_init=0.003)
                try_seq(a_opt, f"fine_pert_{attempt}")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
