                        ft = (1.0 - alpha) * a + alpha * g
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha
                    if best_alpha > 1e-10:
                        a = (1.0 - best_alpha) * a + best_alpha * g
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        obj = fast_convolve(a).max()
                        if obj < best_obj - 1e-14:
                            best_obj = obj
                            best_a = a.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                        m_slack = 1.0
                    else:
                        no_improve += 1
                        m_slack = min(m_slack * 1.1, 5.0)
                else:
                    no_improve += 1
            else:
                no_improve += 1
                m_slack = min(m_slack * 1.2, 5.0)
        except Exception:
            no_improve += 1
            m_slack = min(m_slack * 1.2, 5.0)

        if no_improve > 50:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            no_improve = 0
            m_slack = 1.0

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent targeting the highest peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = list(range(js, je + 1))

        improved = False
        for attempt in range(3):
            # Random direction in affected subspace
            direction = np.zeros(n)
            for idx in affected:
                direction[idx] = rng.standard_normal()
            # Ensure zero mean on affected set
            direction[affected] -= direction[affected].mean()

            for scale in [0.001, 0.0005, 0.0002, 0.0001]:
                candidate = project_simplex(a + scale * direction)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved and it % 100 == 0:
            # Broader perturbation
            candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
            elif it > 500:
                break

    return best_a, best_obj


def fourier_init(n, rng):
    """Generate initialization with flat magnitude spectrum (in Fourier domain)."""
    # Random phase Fourier transform -> take real part, make positive
    freqs = np.fft.rfftfreq(n)
    # Skip DC component
    phase = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    A = np.zeros(n // 2 + 1, dtype=complex)
    A[0] = 1.0  # DC
    for k in range(1, n // 2 + 1):
        A[k] = np.exp(1j * phase[k])
    a = np.fft.irfft(A * n, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Fourier random phase
    inits.append(fourier_init(n, rng))

    # Concentrated at edges
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Two Gaussian bumps
    g = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine modulated
    c = np.abs(np.cos(2 * np.pi * x / 7.0)) + 0.5
    inits.append(c / c.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Low-frequency dominant
    A = np.zeros(n // 2 + 1, dtype=complex)
    A[0] = 1.0
    for k in range(1, min(5, n // 2)):
        A[k] = np.exp(1j * rng.uniform(0, 2 * np.pi)) / (k + 1)
    a = np.fft.irfft(A * n, n)
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        inits.append(a / a.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    - Log-parameterized L-BFGS-B with peak-focused smoothing
    - Game iterate with QP subproblem
    - LP-based Frank-Wolfe
    - Coordinate polishing
    - Multi-start with Fourier-domain initializations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, initial score={best_score:.8f}")

    # === Phase 1: Diverse initialization exploration (200s) ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t_start = time.monotonic()
    candidates = []

    # Use parent as primary
    candidates.append(("parent", parent.copy()))

    # Generate structured inits at parent length
    for i, init in enumerate(structured_init(n_parent, rng)):
        candidates.append((f"init_{i}", init))

    # Try a few different lengths
    for n_try in [n_parent - 64, n_parent + 64, 512, 1024]:
        if n_try < 64 or n_try > 2048:
            continue
        init = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_parent), parent)
        init = np.maximum(init, 0)
        init = init / init.sum()
        candidates.append((f"n{n_try}", init))

    # Quick optimization of each candidate
    time_per_cand = 35
    for label, init in candidates:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 10:
            break
        t_alloc = min(time_per_cand, remaining)
        print(f"  Quick opt: {label} (n={len(init)}, t={t_alloc:.0f}s)")
        a_opt, _ = peak_focused_sg(init, time_limit=t_alloc, T_init=0.003, lr_init=0.03, top_k=5)
        try_seq(a_opt, f"p1_{label}")

    print(f"  Phase 1 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Deep L-BFGS-B optimization (500s) ===
    print("\n=== Phase 2: L-BFGS-B Deep Optimization ===")
    t_start = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  L-BFGS-B (top_k=10, T: 0.005->0.0001), t={t_alloc:.0f}s")
            a_opt, _ = lbgfs_logparam(best_seq, time_limit=t_alloc,
                                      T_start=0.005, T_end=0.0001, top_k=10, max_restarts=5)
            try_seq(a_opt, "p2_lbgfs_k10")

            remaining = deadline - time.monotonic() - 500
            if remaining > 20:
                t_alloc = min(remaining, 200)
                print(f"  L-BFGS-B (no focus, T: 0.002->0.00005), t={t_alloc:.0f}s")
                a_opt, _ = lbgfs_logparam(best_seq, time_limit=t_alloc,
                                          T_start=0.002, T_end=0.00005, top_k=0, max_restarts=5)
                try_seq(a_opt, "p2_lbgfs_full")

    print(f"  Phase 2 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Game iterate + LP-FW (500s) ===
    print("\n=== Phase 3: Game Iterate + LP-FW ===")
    t_start = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  Game iterate (T=0.001, top_k=8), t={t_alloc:.0f}s")
            a_opt, _ = game_iterate_cvxpy(best_seq, time_limit=t_alloc, T_init=0.001, top_k=8)
            try_seq(a_opt, "p3_game_k8")

            remaining = deadline - time.monotonic() - 300
            if remaining > 20:
                t_alloc = min(remaining, 150)
                print(f"  Game iterate (T=0.003, full), t={t_alloc:.0f}s")
                a_opt, _ = game_iterate_cvxpy(best_seq, time_limit=t_alloc, T_init=0.003, top_k=0)
                try_seq(a_opt, "p3_game_full")

            remaining = deadline - time.monotonic() - 200
            if remaining > 20:
                t_alloc = min(remaining, 150)
                print(f"  LP-FW, t={t_alloc:.0f}s")
                a_opt, _ = fw_lp_fast(best_seq, time_limit=t_alloc, m_slack=1.05)
                try_seq(a_opt, "p3_lpfw")

    print(f"  Phase 3 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Fine polishing (300s) ===
    print("\n=== Phase 4: Fine Polishing ===")
    t_start = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)

        # Peak-focused SG with very low T
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            t_alloc = min(remaining, 120)
            print(f"  Peak SG (T=0.0005, k=3), t={t_alloc:.0f}s")
            a_opt, _ = peak_focused_sg(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.01, top_k=3)
            try_seq(a_opt, "p4_sg_k3")

            remaining = deadline - time.monotonic() - 100
            if remaining > 15:
                t_alloc = min(remaining, 80)
                print(f"  Game iterate (T=0.0003, k=5), t={t_alloc:.0f}s")
                a_opt, _ = game_iterate_cvxpy(best_seq, time_limit=t_alloc, T_init=0.0003, top_k=5)
                try_seq(a_opt, "p4_game_fine")

            remaining = deadline - time.monotonic() - 50
            if remaining > 10:
                t_alloc = min(remaining, 50)
                print(f"  Coordinate polish, t={t_alloc:.0f}s")
                a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "p4_coord")

    print(f"  Phase 4 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final squeeze (remaining time) ===
    print("\n=== Phase 5: Final Squeeze ===")
    t_start = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple tiny perturbations with quick re-optimization
        remaining = deadline - time.monotonic() - 20
        attempts = 0
        while remaining > 20 and attempts < 20:
            attempts += 1
            scale = 0.0005 + 0.001 * min(1.0, attempts / 10.0)
            noise = rng.standard_normal(n_best) * scale
            perturbed = project_simplex(best_seq + noise)
            t_alloc = min(15, remaining - 5)
            if t_alloc < 5:
                break
            a_opt, _ = peak_focused_sg(perturbed, time_limit=t_alloc, T_init=0.0003, lr_init=0.005, top_k=3)
            try_seq(a_opt, f"p5_pert{attempts}")
            remaining = deadline - time.monotonic() - 20

        # One final L-BFGS-B with very tight tolerance
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final L-BFGS-B, t={t_alloc:.0f}s")
            a_opt, _ = lbgfs_logparam(best_seq, time_limit=t_alloc,
                                      T_start=0.0003, T_end=0.00005, top_k=3, max_restarts=3)
            try_seq(a_opt, "p5_final_lbgfs")

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
