Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using randomized pairwise 
coordinate descent with exact 1D convex subproblem solves.

Key idea: For each pair of coordinates (i,j), the problem of optimizing 
those two (while fixing the others) reduces to a 1D convex piecewise-linear 
minimization, which can be solved exactly by bisection on the subgradient.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_1d_convex(conv, d, lo, hi, max_iter=80):
    """
    Minimize f(u) = max_k(conv[k] + u*d[k]) over u in [lo, hi].
    Returns optimal u.
    """
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        vals = conv + mid * d
        f_mid = vals.max()
        # Active set: where vals is close to max
        # Use a tight tolerance to get the correct subgradient
        tol = 1e-14 * max(1.0, f_mid)
        active = np.where(vals >= f_mid - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        slopes = d[active]
        smin = slopes.min()
        smax = slopes.max()
        if smin > 1e-15:
            # All active slopes positive -> f increasing -> go left
            hi = mid
        elif smax < -1e-15:
            # All active slopes negative -> f decreasing -> go right
            lo = mid
        else:
            # 0 in subgradient -> optimal
            break
    return (lo + hi) * 0.5

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using randomized pairwise 
coordinate descent with exact 1D convex subproblem solves.

Key innovation: For each pair of coordinates (i,j), fixing all others and 
constraining a[i]+a[j]=const reduces the problem to minimizing a convex 
piecewise-linear function of one variable, solvable exactly by bisection 
on the subgradient. This gives fast, exact local improvements that 
complement the global structure found by the parent FW-LP solution.

Additional strategies:
- Peak-aware pair selection (bias toward indices affecting current peaks)
- Periodic random restarts with controlled perturbation
- Multi-n exploration (n=768, 1024, 1536)
- Occasional full FW-LP steps for global improvements
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_1d_convex(conv, d, lo, hi, max_iter=60):
    """
    Minimize f(u) = max_k(conv[k] + u*d[k]) over u in [lo, hi].
    f is convex piecewise-linear. Uses bisection on subgradient sign.
    Returns optimal u.
    """
    if lo >= hi:
        return lo
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        vals = conv + mid * d
        f_mid = vals.max()
        # Determine subgradient: slopes of active constraints
        # Use tolerance relative to scale
        scale = max(1.0, abs(f_mid))
        tol = 1e-13 * scale
        active = np.where(vals >= f_mid - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        slopes = d[active]
        smin = slopes.min()
        smax = slopes.max()
        if smin > 1e-14:
            # All active slopes positive: f increasing here, go left
            hi = mid
        elif smax < -1e-14:
            # All active slopes negative: f decreasing here, go right
            lo = mid
        else:
            # 0 is in the subgradient: optimal
            break
    return (lo + hi) * 0.5


def pairwise_cd(a_init, time_limit, seed=42, verbose=False):
    """
    Randomized pairwise coordinate descent with exact 1D convex solves.
    Maintains a on the simplex (sum=1, a>=0).
    Uses peak-aware pair selection for efficiency.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    batch_size = 1  # will increase for speed
    
    # Preallocate d vector
    d = np.zeros(2 * n - 1)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.005:
            break
        
        # Peak-aware pair selection (every 1000 iterations do a "smart" selection)
        if it % 1000 == 0 and it > 0:
            # Find top peaks
            top_k = min(20, 2 * n - 1)
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            # Collect affected indices
            affected_set = set()
            for k in peaks:
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for idx in range(js, je + 1):
                    if a[idx] > 1e-10:
                        affected_set.add(idx)
            if len(affected_set) >= 2:
                affected_arr = np.array(list(affected_set))
            else:
                affected_arr = np.arange(n)
        
        # Select pair
        if rng.random() < 0.7:
            # Peak-aware: sample from affected indices
            if 'affected_arr' in dir() and len(affected_arr) >= 2:
                idx = rng.integers(0, len(affected_arr))
                i = int(affected_arr[idx])
                idx2 = rng.integers(0, len(affected_arr))
                j = int(affected_arr[idx2])
                if i == j:
                    j = (j + 1) % n
            else:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    j = (j + 1) % n
        else:

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
        
        p = a[i]
        q = a[j]
        c = p + q
        
        if c < 1e-15:
            it += 1
            continue
        
        # Build d[k] = a[k-i] - a[k-j]
        d[:] = 0.0
        # a[k-i] nonzero for k in [i, i+n-1]
        d[i:i + n] += a
        # a[k-j] nonzero for k in [j, j+n-1]
        d[j:j + n] -= a
        
        # Solve 1D convex problem
        u_opt = solve_1d_convex(conv, d, -p, q)
        
        if abs(u_opt) > 1e-15:
            a[i] = p + u_opt
            a[j] = q - u_opt
            # Update conv incrementally
            conv += u_opt * d
            # Clamp tiny negatives
            if conv.min() < 0:
                conv[conv < 0] = 0.0
        
        it += 1
        
        # Check objective periodically
        if it % 100 == 0:
            cur_obj = float(conv.max())
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Periodic full recompute to correct drift
        if it % 50000 == 0:
            conv = fast_convolve(a)
            cur_obj = float(conv.max())
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
        
        # Random restart if stuck
        if no_improve > 200:
            scale = 0.002 * min(3.0, no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            no_improve = 0
        
        if verbose and it % 200000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PD it={it}, obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
    
    # Final check
    conv = fast_convolve(a)
    if conv.max() < best_obj:
        best_obj = float(conv.max())
        best_a = a.copy()
    
    return best_a, best_obj


def fw_lp_step(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_short(a_init, time_limit, seed=0):
    """Short FW-LP run for global improvements (fewer iterations than full FW)."""
    n = len(a_init)

        if g is None:
            g = fw_lp_step(f, M * 1.5, n)
            if g is None:
                break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = float(conv.max())
            
            if M < best_obj - 1e-15:

                best_f = f.copy()

            no_improve += 1

def smoothed_refine(a_init, time_limit, T=0.001, lr=0.03, seed=0):
    """Smoothed gradient descent for local refinement."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        it += 1
        
        conv = fast_convolve(a)
        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: 2 * hankel_mv(p, a) = 2 * correlate(p, a, 'valid')
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        # Projected gradient step with backtracking
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = float(c.max())
            if val < best_obj - 1e-14:
                a = candidate
                best_obj = val
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
        
        if no_improve > 50:
            break
    
    return best_a, best_obj
