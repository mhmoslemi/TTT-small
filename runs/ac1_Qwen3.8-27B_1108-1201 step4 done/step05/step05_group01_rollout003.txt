Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            st = ft.sum()
            if st < 1e-15:

            val = fast_convolve(ft).max()

            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

                best_obj = M
                best_f = f.copy()

                no_improve += 1

            no_improve += 1

            conv = fast_convolve(f)

            no_improve = 0

Now for the peak-targeting coordinate descent, the idea is:
1. Find the peak lag k*
2. The peak value is conv[k*] = sum_j a[j]*a[k*-j]
3. To reduce this, we want to decrease the "aligned" pairs
4. For each coordinate m, compute how much it contributes to conv[k*]
5. Move coordinates that contribute most to the peak in the reducing direction

```python
def peak_targeting_cd(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_lag = np.argmax(conv)
        peak_val = conv[peak_lag]
        
        # Top few peaks
        top_k = np.argsort(conv)[::-1][:10]
        
        for k in top_k:
            # Contribution of each coordinate to conv[k]
            # conv[k] = sum_j a[j]*a[k-j]
            # d conv[k] / d a[m] = 2*a[k-m] if 0<=k-m<n
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            
            # Try adjusting each coordinate
            for m in range(m_lo, m_hi + 1):
                if f[m] < 1e-10:
                    continue
                
                # Try reducing f[m] and compensating by increasing others
                # Simple approach: try f[m] * factor for factor in [0.5, 0.7, 0.9, 1.1, 1.3]
                old_f_m = f[m]
                for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
                    new_f_m = old_f_m * factor
                    if new_f_m < 0:
                        continue
                    ft = f.copy()
                    ft[m] = new_f_m
                    # Renormalize
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < peak_val - 1e-10:
                        f = ft
                        peak_val = val
                        if val < best_obj:
                            best_obj = val
                            best_f = f.copy()
                        break
        
        # Check for convergence
        if best_obj - conv.max() < 1e-12:

            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
    
    return best_f, best_obj

Now, let me think about the CMA approach more carefully. The CMA algorithm for PAPR reduction works as follows:
1. Compute the time-domain signal
2. Find the peak
3. Apply a nonlinearity that "compresses" the peak (like a tanh or clipping)
4. Go back to frequency domain
5. Repeat

In our case, the "time domain" is the autocorrelation, and the "frequency domain" is the sequence itself. So:
1. Compute conv(a,a)
2. Find the peak
3. Apply a "peak reducing" operation to conv
4. Map back to the sequence domain

But step 4 is the hard part - how to map from the autocorrelation domain back to the sequence? This is essentially a "phase retrieval" problem.

A simpler approach: use the subgradient of the peak.
- The subgradient of max_k conv[k] with respect to a[m] is:
  - If k* = argmax conv: 2*a[k*-m] (if 0<=k*-m<n)

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f0)) / len(f0)

        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    step = init_step
    decay = 0.9999
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_idx = np.argmax(conv)
        
        # Subgradient at peak
        m_lo = max(0, peak_idx - n + 1)
        m_hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        grad[m_lo:m_hi+1] = 2.0 * f[peak_idx - m_lo:peak_idx - m_hi+1][::-1]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Step
        f_new = f - step * grad
        f_new = np.maximum(f_new, 0.0)
        st = f_new.sum()
        if st < 1e-15:
            step *= 0.5
            continue
        f_new = f_new / st
        
        val = fast_convolve(f_new).max()
        
        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                step *= 0.9
                if step < 0.001:
                    # Restart
                    rng = np.random.default_rng(no_improve * 11 + 3)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                    step = init_step * 0.5
                    no_improve = 0
        
        step *= decay
    
    return best_f, best_obj

    rng = np.random.default_rng(42)

    gen = 0
    mutation_rate = 0.03
    crossover_rate = 0.7
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        
        # Sort by objective
        order = np.argsort(objectives)
        population = [population[i] for i in order]
        objectives = [objectives[i] for i in order]
        
        # Elitism: keep top 5
        elite = [population[i].copy() for i in range(min(5, pop_size))]

        while len(new_pop) < pop_size:
            # Tournament selection
            parents = []
            for _ in range(2):

                parent_idx = tournament[np.argmin([objectives[i] for i in tournament])]

                child = child + noise

        population = new_pop
        objectives = [fast_convolve(ind).max() for ind in population]

        if min_obj < best_obj:
            best_obj = min_obj
            best_f = population[np.argmin(objectives)].copy()
        
        # Adaptive mutation
        if gen % 100 == 0:
            mutation_rate = max(0.005, mutation_rate * 0.95)

Looking at this problem, I need to minimize `2*n*max(convolve(a,a))/(sum(a))^2` for non-negative sequences. The key insight is this is a PAPR (peak-to-average power ratio) minimization problem on the autocorrelation.

My strategy differs from the previous approach by:

2. **CMA-style subgradient** - using the subgradient of the peak specifically
3. **Aggressive multi-length exploration** - trying n=1024, 1536 with proper interpolation

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with dual-guided
restarts, peak-targeting coordinate descent, CMA subgradient refinement,
and multi-length interpolation exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns None if infeasible."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        A_ub[k, m_lo:m_hi + 1] = f[k - m_lo:k - m_hi + 1][::-1]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and dual-guided restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.08
            if m_relax > 5.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 31 + 7)
                scale = min(0.05, 0.01 + 0.001 * (no_improve - 40))
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            continue

        # Multi-step-size line search
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        candidates = sorted(set(np.clip([fw_alpha, 0.5 * fw_alpha, 1.0,
                                          0.7, 0.5, 0.3, 0.2, 0.1, 0.05], 0.01, 1.0)))

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = max(1.0, m_relax * 0.95)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 3.0)

    return best_f, best_obj


def peak_targeting_cd(f0, time_limit, top_coords=30):
    """Coordinate descent targeting the peak autocorrelation contributors."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    stagnation = 0

    multipliers = [0.2, 0.4, 0.6, 0.8, 1.25, 1.6, 2.1, 3.0]

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Get top 3 peaks
        peak_lags = np.argsort(conv)[::-1][:3]

        pass_improved = False
        for k in peak_lags:
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            n_range = m_hi - m_lo + 1
            if n_range == 0:
                continue

            # Contribution of each coordinate to conv[k]
            contribs = f[m_lo:m_hi + 1] * f[k - m_lo:k - m_hi + 1][::-1]
            # Get top contributors
            n_top = min(top_coords, n_range)
            top_idx = np.argsort(contribs)[::-1][:n_top]

            for i in top_idx:
                m = m_lo + i
                old_val = f[m]
                if old_val < 1e-14:
                    continue

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    ft = f.copy()
                    ft[m] = new_val
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_obj - 1e-13:
                        f = ft
                        best_obj = val
                        best_f = f.copy()
                        pass_improved = True
                        break

        if not pass_improved:
            stagnation += 1
            if stagnation > 50:
                rng = np.random.default_rng(stagnation * 17 + 3)
                scale = min(0.04, 0.005 + 0.001 * stagnation)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                cv = fast_convolve(f)
                if cv.max() < best_obj:
                    best_obj = cv.max()
                    best_f = f.copy()
                stagnation = 0
        else:
            stagnation = 0

    return best_f, best_obj


def cma_refine(f0, time_limit, init_lr=0.02, top_peaks=5):
    """CMA-style subgradient method targeting top peaks with adaptive weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    lr = init_lr
    momentum = 0.7
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Softmax-weighted subgradient over top peaks
        k_actual = min(top_peaks, len(conv))
        peak_vals = np.sort(conv)[::-1][:k_actual]
        peak_lags = np.argsort(conv)[::-1][:k_actual]

        # Temperature adapts to spread
        spread = peak_vals[0] - peak_vals[-1]
        temp = max(1e-10, spread * 0.3)
        weights = np.exp((peak_vals - peak_vals[0]) / temp)
        weights /= weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for wp, k in zip(weights, peak_lags):
            if wp < 1e-12:
                continue
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            grad[m_lo:m_hi + 1] += wp * 2.0 * f[k - m_lo:k - m_hi + 1][::-1]

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        val = fast_convolve(f_new).max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.85
                if lr < 0.001:
                    rng = np.random.default_rng(it * 7 + 3)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                    v = np.zeros(n)
                    lr = init_lr * 0.5
                    no_improve = 0

        # Adaptive learning rate decay
        lr = max(0.001, lr * 0.9998)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, dtype=np.float64)
    new_x = np.linspace(0, 1, new_n, dtype=np.float64)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def cosine_init(n, freq, amp, seed=0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_level_init(n, freq1, amp1, freq2, amp2):
    """Two-level cosine modulation for richer structure."""
    j = np.arange(n)
    a = 1.0 + amp1 * np.cos(2.0 * np.pi * freq1 * j / n)
    a += amp2 * np.cos(2.0 * np.pi * freq2 * j / n + 0.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
