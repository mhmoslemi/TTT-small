    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, norm_factor) -> np.ndarray:
    A_cache = build_lp_matrix(f)
    with np.errstate(all='ignore'):
        try:
            c = np.ones(len(f))
            b_ub = np.full(2 * len(f) - 1, norm_factor)
            bounds = [(0.0, None)] * len(f)
            res = linprog(-c, A_ub=A_cache, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
            if res.success:
                return np.array(res.x)
        except Exception:
            pass
        return None

def project_simplex(v) -> np.ndarray:
    n = len(v)
    if n == 0:
        return v
    if np.min(v) >= 0 and np.max(v) <= 1:
        return v
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    u = np.array(sorted_v, dtype=sorted_v.dtype)
    for j in range(n - 1):
        if cssv[j + 1] <= (sorted_v[j] + 1) * j + 1:
            u[j + 1:] = sorted_v[j]
            break
    return np.cumsum(u) / np.sum(u)

def interpolate_to_length(f, new_n):
    if abs(new_n - len(f)) < 1e-5:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        f_new = np.ones(new_n) / new_n
        return f_new

def norm_convolve(a):
    if len(a) == 0:
        return np.array([])
    n = len(a)
    size = 2 * n - 1
    conv = np.zeros(size)
    for k in range(size):

        j_end = min(n-1, k)
        if j_start <= j_end:

            conv[k] = np.sum(a[j_array] * a[k - j_array])
        else:
            conv[k] = float('-inf')
    return conv
