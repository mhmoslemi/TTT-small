        return np.array([max(v[0], 0.0)])

    active = np.ones(n, dtype=bool)
    a = np.ones(n) / n
    
    for iteration in range(n + 10):
        # Solve restricted QP: min a^T W a s.t. sum(a) = 1 on active set
        W_a = W[active][:, active]
        ones_a = np.ones(active.sum())

            W_inv_ones = np.linalg.solve(W_a + 1e-10 * np.eye(W_a.shape[0]), ones_a)
            denom = ones_a @ W_inv_ones
            if denom <= 1e-15:
                break
            a_active = W_inv_ones / denom

            active[min_idx] = False
            continue

        grad = 2 * W @ a

        inactive_indices = np.where(~active)[0]
        active[inactive_indices[worst_inactive]] = True
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def W_matvec(w, x):
    """Compute W(w) @ x where W(w)[i,j] = w[i+j] (sum of Q_k weighted by w).
    w has length 2n-1, x has length n."""
    n = len(x)
    # result[j] = sum_i x[i] * w[i+j]
    M = np.lib.stride_tricks.sliding_window_view(w, n)
    return M @ x

    W_ones = W @ 1 (precomputed)
    W: the matrix (or a callable)
    Returns a."""
    active = np.ones(n, dtype=bool)
    
    for it in range(50):
        S = np.where(active)[0]
        m = len(S)
        if m == 0:
            return np.ones(n) / n
        if m == 1:
            a = np.zeros(n)
            a[S[0]] = 1.0
            return a
        
        W_S = W[np.ix_(S, S)]
        ones = np.ones(m)
        try:
            sol = np.linalg.solve(W_S, ones)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(W_S, ones, rcond=None)[0]
        
        denom = ones @ sol
        if denom < 1e-15:
            a = np.zeros(n)
            a[S] = 1.0 / m
            return a
        a_S = sol / denom
        
        # Check primal feasibility
        if np.min(a_S) < -1e-10:
            # Remove the most negative variable
            idx = S[np.argmin(a_S)]
            active[idx] = False
            continue
        elif np.min(a_S) < 0:
            # Clip to zero and remove
            a_S = np.maximum(a_S, 0)
            a_S = a_S / a_S.sum()
            # Remove variables that are now zero
            for i in range(m):
                if a_S[i] < 1e-10:
                    active[S[i]] = False
            continue
        
        a = np.zeros(n)
        a[S] = a_S
        
        # Check dual feasibility
        Wa = W_matvec_from_W(W, a)  # W @ a
        lambda_ = 2.0 * np.mean(Wa[S])
        # For inactive: Wa[j] >= lambda_/2
        inactive = np.where(~active)[0]
        if len(inactive) > 0:
            min_inact = np.min(Wa[inactive])
            if min_inact < lambda_/2 - 1e-8:
                # Add the most violated
                worst = inactive[np.argmin(Wa[inactive])]
                active[worst] = True
                continue
        
        return a
    
    # Fallback
    a = np.zeros(n)
    a[np.where(active)[0]] = 1.0 / active.sum()
    return a

```python
def solve_qp_fista(w, n, max_iter=2000, tol=1e-10):
    """Solve min x^T W(x) x s.t. sum(x)=1, x>=0 using FISTA.
    W is defined by w: (Wx)[i] = sum_j w[i+j] * x[j]
    w has length 2n-1."""
    # Estimate L (Lipschitz constant of gradient 2Wx)
    # L >= 2 * lambda_max(W)
    # Upper bound: 2 * max row sum of W
    # Row i sum = sum_j w[i+j] for j=0,...,n-1
    row_sums = np.array([np.sum(w[i:i+n]) for i in range(n)])
    L = 2.0 * np.max(row_sums)
    if L < 1e-15:
        return np.ones(n) / n
    
    step = 1.0 / L
    x = np.ones(n) / n
    y = x.copy()
    t = 1.0
    
    x_old = x.copy()
    for it in range(max_iter):
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = x + (t - 1) / t_new * (x - x_old)
        y = np.maximum(y, 0)
        
        # Grad step: y - step * 2 * W(y)
        Wy = W_matvec(w, y)
        x_new = y - step * 2.0 * Wy
        
        # Project onto simplex
        x_new = project_simplex(x_new)
        
        # Check convergence
        if np.max(np.abs(x_new - x)) < tol:
            break
        x_old = x
        x = x_new
        t = t_new
    
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s
    return x
```

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Exponentiated gradient (mirror descent) with adaptive peak weighting
2. FISTA-based QP refinement on top peaks
3. Multi-length exploration with structured perturbations
4. Coordinate descent polish

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_soft(conv, a, n, top_k=10, temperature=0.001):
    """Compute soft subgradient of max_k conv[k] w.r.t. a, weighted by softmax on peaks."""
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        for j in range(js, je + 1):
            offset = k - j
            if 0 <= offset < n:
                sg[j] += 2.0 * w[idx] * a[offset]
    return sg


def compute_subgradient_hard(conv, a, n, num_peaks=5):
    """Compute hard subgradient (uniform over top peaks)."""
    k_actual = min(num_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    sg = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            offset = k - j
            if 0 <= offset < n:
                sg[j] += 2.0 * a[offset]
    sg /= num_peaks
    return sg


def W_matvec(w, x):
    """Compute W(w) @ x where (Wx)[i] = sum_j w[i+j] * x[j].
    w has length 2n-1, x has length n."""
    n = len(x)
    M = np.lib.stride_tricks.sliding_window_view(w, n)
    return M @ x


def solve_qp_fista(w, n, max_iter=2000, tol=1e-12):
    """Solve min x^T W x s.t. sum(x)=1, x>=0 using FISTA.
    W defined by w: (Wx)[i] = sum_j w[i+j] * x[j]."""
    row_sums = np.array([np.sum(w[i:i + n]) for i in range(n)])
    L = 2.0 * np.max(row_sums)
    if L < 1e-15:
        return np.ones(n) / n
    step = 1.0 / L
    x = np.ones(n) / n
    y = x.copy()
    t = 1.0
    x_old = x.copy()
    for it in range(max_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = x + momentum * (x - x_old)
        y = np.maximum(y, 0.0)
        Wy = W_matvec(w, y)
        x_new = y - step * Wy
        x_new = project_simplex(x_new)
        if np.max(np.abs(x_new - x)) < tol:
            break
        x_old = x
        x = x_new
        t = t_new
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s


def build_W_from_peaks(conv, a, n, top_k=5, sharpness=0.001):
    """Build the weight vector w for the QP from peak positions.
    w[k] is proportional to the peak weight at position k."""
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / max(sharpness, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w_full = np.zeros(2 * n - 1)
        w_full[int(peaks[0])] = 1.0
        return w_full
    w = w / ws
    w_full = np.zeros(2 * n - 1)
    for i in range(k_actual):
        w_full[int(peaks[i])] = w[i]
    return w_full


def mirror_descent_optimize(a0, time_limit, top_k=10, eta_init=0.05, verbose=False):
    """Exponentiated gradient (mirror descent) on simplex with peak subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    it = 0
    no_improve = 0
    temp = 0.0005

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        obj = conv.max()

        sg = compute_subgradient_soft(conv, a, n, top_k=top_k, temperature=temp)
        sg_norm = np.max(np.abs(sg))
        if sg_norm < 1e-15:
            no_improve += 1
        else:
            sg = sg / sg_norm

        # Mirror descent update: a_new[i] ∝ a[i] * exp(-eta * sg[i])
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * sg
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Line search
        best_alpha = 0.0
        best_val = obj
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            trial = (1 - alpha) * a + alpha * a_new
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            val = fast_convolve(trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * a_new
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s

        if obj < best_obj - 1e-14:
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.01, 0.5)
                temp = temp * 1.002
            if verbose and it % 200 == 0:
                print(f"      MD iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.95
            temp *= 1.005
            if no_improve > 200:
                # Restart from best with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise)
                conv = fast_convolve(a)
                obj = conv.max()
                eta = eta_init * 0.5
                no_improve = 0

    return best_a, best_obj


def qp_optimize(a0, time_limit, top_k=5, verbose=False):
    """Iterative QP refinement: build W from peaks, solve QP via FISTA, line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        obj = conv.max()

        # Build W from top peaks
        w = build_W_from_peaks(conv, a, n, top_k=top_k, sharpness=0.0005)
        # Normalize w so that the QP is well-scaled
        w_max = w.max()
        if w_max > 1e-15:
            w = w / w_max

        # Solve QP
        a_qp = solve_qp_fista(w, n, max_iter=1500, tol=1e-12)

        # Line search
        best_alpha = 0.0
        best_val = obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            trial = (1 - alpha) * a + alpha * a_qp
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            val = fast_convolve(trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            a = (1 - best_alpha) * a + best_alpha * a_qp
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      QP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng(it * 17 + 3)
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            if time.monotonic() - t0 > time_limit:
                break

    return best_a, best_obj


def combined_optimize(a0, time_limit, verbose=False):
    """Alternating mirror descent and QP refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    phase = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        if phase == 0:
            # Mirror descent phase
            phase_time = min(rem, 8.0)
            a, obj = mirror_descent_optimize(a, time_limit=phase_time, top_k=8, eta_init=0.03)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
            phase = 1
        else:
            # QP phase
            phase_time = min(rem, 5.0)
            a, obj = qp_optimize(a, time_limit=phase_time, top_k=5)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
            phase = 0

        if verbose and it % 10 == 0:
            print(f"    Combined iter {it}: best={best_obj:.10f}, rem={rem:.0f}s")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent on individual elements, targeting peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break

        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0

        # Find top 2 peaks
        top_k = min(2, len(conv))
        peak_pos = np.argpartition(conv, -top_k)[-top_k:]

        improved = False
        for pk in peak_pos:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)

            # Try adjusting coordinates that contribute to this peak
            # Sample a few coordinates
            if je - js + 1 > 20:
                indices = np.random.default_rng(pk + int(time.monotonic() * 1000) % 10000).choice(
                    range(js, je + 1), size=min(10, je - js + 1), replace=False)
            else:
                indices = range(js, je + 1)

            for j in indices:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[j]
                if old_val < 1e-12:
                    continue

                for mult in [0.5, 0.7, 0.85, 1.15, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-13:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                no_improve = 0
        if not improved:
            no_improve += 1

    return best_a, best_obj


def generate_hann_window(n):
    """Generate a Hann-window shaped sequence normalized to sum=1."""
    x = np.linspace(0, np.pi, n)
    w = 0.5 * (1 + np.cos(x))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def generate_smoothed_random(n, seed=42, smooth=3):
    """Generate a smoothed random sequence."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    # Smooth with a boxcar
    kernel = np.ones(smooth) / smooth
    x = np.convolve(x, kernel, mode='same')
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s
    return x

        for p in range(8):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.08, 60)
            if t_alloc < 10:
                break

            # Structured perturbation: sine wave + noise
            freq = rng.uniform(1.0, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            sine = np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            noise = rng.standard_normal(n_best) * 0.002
            perturbation = 0.005 * sine + noise
            f_pert = project_simplex(best_seq + perturbation)

            a_opt, obj = combined_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{p}")

    # === Phase 7: Final polish with coordinate descent ===
    print(f"\n=== Phase 7: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        t_cd = min(rem * 0.3, 120)
        if t_cd > 15:
            print(f"  CD polish, time={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

            # Quick MD after CD
            rem = deadline - time.monotonic() - 20
            t_md = min(rem * 0.2, 60)
            if t_md > 10:
                a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_md, top_k=5, eta_init=0.01)
                try_seq(a_opt, "md_final")

            # Quick QP after CD
            rem = deadline - time.monotonic() - 15
            t_qp = min(rem * 0.2, 50)
            if t_qp > 10:
                a_opt, obj = qp_optimize(best_seq, time_limit=t_qp, top_k=4)
                try_seq(a_opt, "qp_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
