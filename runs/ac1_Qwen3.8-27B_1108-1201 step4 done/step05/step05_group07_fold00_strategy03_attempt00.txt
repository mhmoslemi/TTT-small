        best_candidate = None

            candidate_score = evaluate_sequence(candidates[candidate_len])
            improvement = (best_score - candidate_score) > 0.0001  # Security checks already handled INF cases above
            if candidate_score < best_score and improvement:
                best_score = candidate_score
                best_candidate = candidates[candidate_len]
                best_seq = best_candidate
                # Early update of best_seq for subsequent evaluations
                if best_seq is not None:
                    # Linear clipping to [0,1000] and normalization
