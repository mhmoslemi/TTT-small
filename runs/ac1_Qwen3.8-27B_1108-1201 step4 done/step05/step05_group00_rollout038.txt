def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_grad(conv, T):
    """Compute smooth max and gradient weights p = softmax(conv/T)."""
    m = conv.max()
    shifted = (conv - m) / T
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    smooth_val = m + log_sum / T
    p = exp_shifted / exp_shifted.sum()
    return smooth_val, p

def smooth_obj_grad(a, T):
    """Objective and gradient for smooth max approximation."""
    n = len(a)
    conv = fast_convolve(a)
    val, p = smooth_max_and_grad(conv, T)
    # Gradient: grad[j] = sum_k p[k] * a[k-j] for valid k-j
    # This is cross-correlation of p and a
    grad = np.correlate(p, a, mode='valid')
    return 2.0 * n * val, 2.0 * n * grad

        if remaining < 0.05:

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j+n, j] = a
        b_ub = np.full(2*n-1, M * m_exp)
        bounds = [(0, None)] * n

            if res.success:

                g_sum = g.sum()

                else:

            m_exp = 1.0

            m_exp = 1.0

Wait, I realize there's a problem with my mirror descent. The mirror descent update is:
a_new[j] ∝ a[j] * exp(-eta * grad[j])

But this can cause numerical issues when a[j] is very small (log of near-zero). Let me add more careful handling.

Also, I should think more carefully about the step size. For mirror descent with KL divergence, the optimal step size is related to the smoothness of the objective. The smooth max with parameter T has Lipschitz constant related to 1/T.

Let me also think about whether I should try a different approach entirely.

**Alternative: Use the "auxiliary function" / MAE (Majorization-Auxiliary-Function) approach**

Given current a, construct a quadratic upper bound on the objective:
f(a) <= f(a0) + <grad f(a0), a - a0> + (L/2)||a - a0||^2

where L is the Lipschitz constant of the gradient. Then minimize the upper bound (which is a QP):
min_a (L/2)||a - a0||^2 + <grad f(a0), a> + constant

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Smooth max gradient descent with adaptive temperature and backtracking
- Mirror descent (KL divergence) for global mixing
- Frank-Wolfe with LP oracle for structural improvements
- Coordinate descent with 1D line search for precise local optimization
- Multi-n exploration and perturbation-based escape from local minima
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    m = conv.max()
    shifted = (conv - m) / T
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    smooth_val = m + log_sum / T
    p = exp_shifted / exp_shifted.sum()
    return smooth_val, p


def compute_smooth_gradient(a, T):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    _, p = smooth_max_and_weights(conv, T)
    grad = np.correlate(p, a, mode='valid')
    return 2.0 * n * grad, conv


def gradient_descent_optimize(a0, time_limit, T_start=0.005, T_end=0.0001,
                              use_momentum=False, verbose=False):
    """
    Projected gradient descent with smooth max, adaptive T, backtracking,
    and optional momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)  # momentum
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.005:
            break

        # Adaptive temperature: geometric cooling
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        # Compute gradient
        conv = fast_convolve(a)
        grad, _ = compute_smooth_gradient(a, T)

        # Backtracking line search
        base_lr = 5.0 / (n * n * T * 0.1 + 1.0)
        best_lr = 0.0
        best_val = 2.0 * n * conv.max()
        best_candidate = a

        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            lr = base_lr * scale
            if use_momentum:
                v = momentum * v - lr * grad
                candidate = project_simplex(a + v)
            else:
                candidate = project_simplex(a - lr * grad)
            val = 2.0 * n * fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_lr = lr
                best_candidate = candidate

        if best_lr > 0:
            if use_momentum:
                a = best_candidate
            else:
                a = best_candidate

            cur_obj = 2.0 * n * fast_convolve(a).max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > 60:
            noise_scale = 0.002 + 0.008 * min(1.0, (no_improve - 60) / 150.0)
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))

            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    GD it={it}, T={T:.6f}, obj={best_obj:.8f}")

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, T_start=0.01, T_end=0.00005,
                            verbose=False):
    """
    Mirror descent with KL divergence and smooth max objective.
    Update: a_new[j] = a[j] * exp(-eta * grad[j]) / Z
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.005:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        # Compute gradient
        conv = fast_convolve(a)
        grad, _ = compute_smooth_gradient(a, T)

        # Mirror descent step with backtracking
        base_eta = 1.0 / (n * T + 1.0)
        best_eta = 0.0
        best_val = 2.0 * n * conv.max()

        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            eta = base_eta * scale
            # KL mirror step: a_new ∝ a * exp(-eta * grad)
            log_weights = np.log(a + 1e-30) - eta * grad
            log_weights -= log_weights.max()
            a_new = np.exp(log_weights)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-15:
                continue
            a_new = a_new / a_new_sum
            val = 2.0 * n * fast_convolve(a_new).max()
            if val < best_val - 1e-16:
                best_val = val
                best_eta = eta

        if best_eta > 0:
            eta = best_eta
            log_weights = np.log(a + 1e-30) - eta * grad
            log_weights -= log_weights.max()
            a = np.exp(log_weights)
            a = a / a.sum()

            conv = fast_convolve(a)
            cur_max = conv.max()
            if cur_max < best_conv_max - 1e-14:
                best_conv_max = cur_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            noise_scale = 0.003 + 0.01 * min(1.0, (no_improve - 80) / 200.0)
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    MD it={it}, T={T:.6f}, obj={2*n*best_conv_max:.8f}")

    return best_a, 2.0 * n * best_conv_max


def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = 2.0 * n * M

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        # Build LP constraint matrix
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            lp_time = max(0.5, remaining / 20)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': lp_time})
            if res.success:
                g = np.maximum(res.x, 0.0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                else:
                    g = None
            else:
                g = None
        except Exception:
            g = None

        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                no_improve += 10
            continue

        # Greedy line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with binary search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * a + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0

            obj = 2.0 * n * M
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 5.0)

        if no_improve > 40:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, passes=5):
    """
    Coordinate descent: for each coordinate, try multiple values and keep the best.
    Uses a smart set of trial values around the current value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()

    for p in range(passes):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        rng = np.random.default_rng((p * 31 + 7) % (2**31))
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            aj = a[j]
            if aj < 1e-15:
                # Try setting this coordinate to a small positive value
                trial_vals = [1.0 / n, 2.0 / n, 0.5 / n]
            else:
                # Try scaling the current value
                trial_vals = [aj * s for s in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 3.0]]
                # Also try absolute values
                trial_vals.extend([aj / 2, aj * 1.5, 1.0 / n, 2.0 / n, 0.5 / n])

            best_aj = aj
            best_val = best_obj

            for aj_try in trial_vals:
                if aj_try < 0 or aj_try > 1.0:
                    continue
                a_try = a.copy()
                a_try[j] = aj_try
                s = a_try.sum()
                if s < 1e-15 or s > 2.0:
                    continue
                a_try = a_try / s
                val = 2.0 * n * fast_convolve(a_try).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_aj = aj_try

            if best_aj != aj:
                a[j] = best_aj
                a = a / a.sum()
                improved = True
                cur_obj = 2.0 * n * fast_convolve(a).max()
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()

        # Fine-tuning pass with smaller perturbations
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        for j in range(0, n, max(1, n // 100)):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            aj = a[j]
            if aj < 1e-10:
                continue
            # Fine grid around current value
            for factor in [0.95, 0.97, 1.03, 1.05, 0.9, 1.1]:
                aj_try = aj * factor
                a_try = a.copy()
                a_try[j] = aj_try
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                val = 2.0 * n * fast_convolve(a_try).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_obj


def nesterov_smooth_optimize(a0, time_limit, T_start=0.003, T_end=0.00003,
                             verbose=False):
    """
    Nesterov-accelerated projected gradient descent with smooth max.
    Uses a look-ahead point for gradient evaluation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    y = a.copy()  # look-ahead point
    t_k = 1.0  # Nesterov momentum parameter

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.005:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        # Compute gradient at look-ahead point
        conv_y = fast_convolve(y)
        grad, _ = compute_smooth_gradient(y, T)

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new

        # Backtracking on step size
        base_lr = 2.0 / (n * T + 1.0)
        best_lr = 0.0
        best_val = 2.0 * n * fast_convolve(a).max()
        best_candidate = a

        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            lr = base_lr * scale
            # Nesterov update
            a_new = project_simplex(y - lr * grad)
            y_new = a_new + beta * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            y_new_sum = y_new.sum()
            if y_new_sum < 1e-15:
                continue
            y_new = y_new / y_new_sum

            val = 2.0 * n * fast_convolve(a_new).max()
            if val < best_val - 1e-16:
                best_val = val
                best_lr = lr
                best_candidate = a_new
                best_y_new = y_new

        if best_lr > 0:
            a = best_candidate
            y = best_y_new
            t_k = t_new

            cur_obj = 2.0 * n * fast_convolve(a).max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            noise_scale = 0.003 + 0.008 * min(1.0, (no_improve - 50) / 150.0)
            rng = np.random.default_rng((it * 314159 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            t_k = 1.0
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    Nesterov it={it}, T={T:.6f}, obj={best_obj:.8f}")

    return best_a, best_obj


def scipy_slsqp_optimize(a0, T, max_iter=500):
    """Use scipy SLSQP on the smooth max objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()

    def objective(a):
        conv = fast_convolve(a)
        m = conv.max()
        shifted = T * (conv - m)
        return 2.0 * n * (m + np.log(np.sum(np.exp(shifted))) / T)

    def gradient(a):
        conv = fast_convolve(a)
        m = conv.max()
        shifted = T * (conv - m)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()
        return 2.0 * n * np.correlate(p, a, mode='valid')

    def constraint_fun(a):
        return a.sum() - 1.0

    def constraint_jac(a):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = [(0, None)] * n

    try:
        result = minimize(objective, a0, jac=gradient, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-10})
        if result.success or result.fun < objective(a0):
            x = np.maximum(result.x, 0.0)
            s = x.sum()
            if s > 1e-15:
                return x / s
    except Exception:
        pass

    return a0


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for the given length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n)) ** 2))
    inits.append(g2 / g2.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Dirichlet random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Interpolated parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase hybrid approach:
    Phase 1: Quick multi-n exploration with diverse initializations
    Phase 2: Intensive multi-method optimization on best candidate
    Phase 3: Coordinate descent polish
    Phase 4: Escape attempts and final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_fallback")

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 512, 640, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break

        # Generate inits
        inits = generate_diverse_inits(n_try, parent, rng)
        best_init = None
        best_init_obj = float('inf')

        for init in inits[:4]:  # Try first 4 inits
            obj = 2.0 * n_try * fast_convolve(init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init

        if best_init is None:
            best_init = np.ones(n_try) / n_try

        # Quick gradient descent
        t_alloc = min(remaining, 30)
        a_opt, _ = gradient_descent_optimize(best_init, time_limit=t_alloc,
                                             T_start=0.005, T_end=0.001)
        try_seq(a_opt, f"gd_n{n_try}")

        # Quick mirror descent
        remaining = deadline - time.monotonic() - 900
        if remaining > 15:
            a_opt2, _ = mirror_descent_optimize(a_opt, time_limit=min(remaining, 15),
                                                T_start=0.01, T_end=0.002)
            try_seq(a_opt2, f"md_n{n_try}")

    print(f"  Phase 1 done ({time.monotonic()-t_phase:.0f}s), best={best_score:.8f}")

    # === Phase 2: Intensive optimization ===
    print("\n=== Phase 2: Intensive Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # 2a: Long gradient descent with very low T
        remaining = deadline - time.monotonic() - 700
        if remaining > 50:
            t_alloc = min(remaining, 250)
            print(f"  GD (low T) n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = gradient_descent_optimize(best_seq, time_limit=t_alloc,
                                                 T_start=0.003, T_end=0.00005,
                                                 use_momentum=True)
            try_seq(a_opt, "gd_lowT")

        # 2b: Nesterov accelerated
        remaining = deadline - time.monotonic() - 500
        if remaining > 40:
            t_alloc = min(remaining, 200)
            print(f"  Nesterov n={n_best}, t={t_alloc:.0f}s")
            a_opt2, _ = nesterov_smooth_optimize(best_seq, time_limit=t_alloc,
                                                 T_start=0.002, T_end=0.00003)
            try_seq(a_opt2, "nesterov")

        # 2c: Mirror descent
        remaining = deadline - time.monotonic() - 350
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  MD n={n_best}, t={t_alloc:.0f}s")
            a_opt3, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc,
                                                T_start=0.008, T_end=0.0001)
            try_seq(a_opt3, "md_long")

        # 2d: FW-LP
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt4, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt4, "fw_lp")

        # 2e: scipy SLSQP (if time permits)
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            print(f"  SLSQP n={n_best}")
            a_opt5 = scipy_slsqp_optimize(best_seq, T=0.001, max_iter=200)
            try_seq(a_opt5, "slsqp")
            # Follow up with GD
            remaining = deadline - time.monotonic() - 80
            if remaining > 10:
                a_opt6, _ = gradient_descent_optimize(a_opt5, time_limit=min(remaining, 15),
                                                      T_start=0.001, T_end=0.0001)
                try_seq(a_opt6, "slsqp+gd")

    print(f"  Phase 2 done ({time.monotonic()-t_phase:.0f}s), best={best_score:.8f}")

    # === Phase 3: Coordinate descent polish ===
    print("\n=== Phase 3: Coordinate Descent Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Coord descent n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc, passes=5)
            try_seq(a_opt, "coord_cd")

        # Fine GD after CD
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 60)
            a_opt2, _ = gradient_descent_optimize(a_opt, time_limit=t_alloc,
                                                  T_start=0.001, T_end=0.00002,
                                                  use_momentum=True)
            try_seq(a_opt2, "gd_after_cd")

        # Another CD pass
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            a_opt3, _ = coordinate_descent_polish(a_opt2, time_limit=min(remaining, 30),
                                                  passes=3)
            try_seq(a_opt3, "coord_cd2")

    print(f"  Phase 3 done ({time.monotonic()-t_phase:.0f}s), best={best_score:.8f}")

    # === Phase 4: Escape and final refinement ===
    print("\n=== Phase 4: Escape & Final ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Perturbation-based escape attempts
        for scale in [0.003, 0.008, 0.015, 0.03]:
            remaining = deadline - time.monotonic() - 30
            if remaining < 12:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 30)
            a_opt, _ = gradient_descent_optimize(f_pert, time_limit=t_alloc,
                                                T_start=0.002, T_end=0.0001,
                                                use_momentum=True)
            try_seq(a_opt, f"escape_s{scale}")

            # Quick CD after escape
            remaining = deadline - time.monotonic() - 20
            if remaining > 8 and best_seq is not None:
                a_opt2, _ = coordinate_descent_polish(a_opt, time_limit=min(remaining, 15),
                                                      passes=2)
                try_seq(a_opt2, f"escape_cd_s{scale}")

        # Final fine-tuning
        remaining = deadline - time.monotonic() - 10
        if remaining > 8:
            a_opt, _ = gradient_descent_optimize(best_seq, time_limit=min(remaining, 15),
                                                T_start=0.0005, T_end=0.00001)
            try_seq(a_opt, "final_gd")
        else:
            remaining = deadline - time.monotonic() - 3
            if remaining > 3:
                a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(remaining, 6),
                                                    passes=2)
                try_seq(a_opt, "final_cd")

    print(f"  Phase 4 done ({time.monotonic()-t_phase:.0f}s), best={best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
