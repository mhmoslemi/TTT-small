```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with
warm starts, correct subgradient descent, and multi-phase optimization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks, weights):
    """Compute subgradient of max_k conv(a,a)[k] at given peaks with weights."""
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            sg[m_lo:m_hi+1] += 2.0 * w * a[k - (np.arange(m_lo, m_hi+1))]
    return sg


def solve_lp_for_direction(f, M, n, warm_start=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_vars) or (None, None) if infeasible."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] for valid j
    # For each k in [0, 2n-2], the constraint is sum_j f[j]*g[k-j] <= M
    # where j ranges over [max(0,k-n+1), min(n-1,k)]
    
    # For small n, build dense matrix
    # For large n, use sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            g_idx = k - j  # this is the index in g
            rows.append(k)
            cols.append(g_idx)
            vals.append(f[j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.ineqfun.values if res.ineqfun is not None else None
    return None, None
```

```python
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build dense matrix A where A[k, m] = f[k-m] if 0<=k-m<n and 0<=m<n."""
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        k_lo = m
        k_hi = min(2 * n - 2, m + n - 1)
        f_lo = 0
        f_hi = n - 1
        A[k_lo:k_hi + 1, m] = f[f_lo:f_hi + 1]
    return A


def build_conv_matrix_sparse(f, n):
    """Build sparse matrix A where A[k, m] = f[k-m] if valid."""
    rows = []
    cols = []
    vals = []
    for m in range(n):
        k_lo = m
        k_hi = min(2 * n - 2, m + n - 1)
        rows.extend(range(k_lo, k_hi + 1))
        cols.extend([m] * (k_hi - k_lo + 1))
        vals.extend(f[0:n])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_subgradient(a, peak_indices, weights):
    """Compute subgradient of max_k conv(a,a)[k].
    grad[m] = sum_k w_k * 2*a[k-m] for valid k-m."""
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peak_indices, weights):
        k = int(k)
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            idx = np.arange(m_lo, m_hi + 1)
            sg[m_lo:m_hi + 1] += 2.0 * w * a[k - idx]
    return sg


def solve_lp_direction(f, M, n, use_sparse=False):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    if use_sparse:
        A_ub = build_conv_matrix_sparse(f, n)
    else:
        A_ub = build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def lpfw_optimize(f0, time_limit, verbose=False, m_tighten=0.999):
    """LP-Frank-Wolfe with adaptive M-tightening and accelerated steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    use_sparse = n > 800

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_scale = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        M_eff = M * m_scale
        g = solve_lp_direction(f, M_eff, n, use_sparse=use_sparse)

        if g is None:
            m_scale = min(m_scale * 1.05, 1.5)
            if m_scale > 1.5:
                m_scale = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_scale = min(m_scale * 1.02, 1.3)
            continue

        # Accelerated FW step size
        fw_alpha = 2.0 / (it + 2.0)

        # Line search: try several candidates
        alphas = sorted(set([
            0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5,
            fw_alpha, 0.6, 0.7, 0.8, 0.9, 1.0
        ]))
        alphas = [a for a in alphas if 0 < a <= 1.0]

        best_alpha = 0.0
        best_val = M

        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement
        if best_alpha > 1e-8:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_scale = max(0.99, m_scale * 0.97)

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW it{it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_scale = min(m_scale * 1.03, 1.2)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min((no_improve - 60) // 40, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_scale = 1.0
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr0=0.003, top_k=20, temp0=0.0001):
    """Subgradient descent with correct gradient, Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        if k_actual >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            pv = conv[peaks]
            scaled = pv / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
            weights = w

        sg = compute_subgradient(f, peaks, weights)

        # Nesterov-momentum update
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * 0.01)) / 2.0
        v_new = ((it * 0.01) / (it + 1.0 + 2.0 * 0.01)) * v - cur_lr * sg
        y_new = project_simplex(y + v_new)
        f = y_new
        y = project_simplex(f + v_new)
        v = v_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.001)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Cyclic coordinate descent with golden section per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass = 0

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            lo = 0.0
            hi = min(0.99, 2.0 * old_val + 0.01)

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            # Evaluate at c1, c2
            def evaluate_at(x, f_base, j_idx, denom_val):
                ft = f_base.copy()
                ft[j_idx] = x
                rest_sum = 1.0 - x
                if rest_sum < 0:
                    return 1e10
                mask = np.ones(len(f_base), dtype=bool)
                mask[j_idx] = False
                ft[mask] = f_base[mask] * (rest_sum / denom_val)
                ss = ft.sum()
                if ss < 1e-15:
                    return 1e10
                ft = ft / ss
                return fast_convolve(ft).max()

            f1 = evaluate_at(c1, f, j, denom)
            f2 = evaluate_at(c2, f, j, denom)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = evaluate_at(c1, f, j, denom)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = evaluate_at(c2, f, j, denom)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = best_obj

            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                rest_sum = 1.0 - opt_x
                if rest_sum > 0:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    ft[mask] = ft[mask] * (rest_sum / denom)
                    ft = ft / ft.sum()
                    f = ft
                    pass_improved = True
                    new_obj = fast_convolve(f).max()
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_f = f.copy()

        if pass_improved:
            no_pass = 0
        else:
            no_pass += 1
            if no_pass >= 2:
                break

    return best_f, best_obj


def cosine_init(n, freq, amp):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def ramp_init(n, alpha=1.0):
    """Ramp-like initialization."""
    j = np.arange(n)
    a = (j + 1) ** alpha
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def bump_init(n, center_frac=0.3, width=0.2):
    """Gaussian bump initialization."""
    j = np.arange(n) / n
    a = np.exp(-((j - center_frac) ** 2) / (2 * width ** 2))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                break
            target_n = len(best_seq) + delta_n
            if target_n < 64 or target_n > 2048:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            s = try_seq(f_interp, f"interp_n{target_n}")
            if s < best_score + 0.05:
                t_alloc = min(rem * 0.4, 60)
                if t_alloc > 10:
                    a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc)
                    try_seq(a_opt, f"lpfw_interp_n{target_n}")

    # === Phase 7: Final polish cycle ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Subgrad + CD alternating
        for cycle in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t1 = min(rem * 0.4, 60)
            t2 = min(rem * 0.4, 60)
            a_opt, _ = subgradient_polish(best_seq, time_limit=t1,
                                           lr0=0.001, top_k=max(10, len(best_seq) // 5),
                                           temp0=0.00001)
            try_seq(a_opt, f"final_sg_{cycle}")
            a_opt, _ = coordinate_descent(a_opt, time_limit=t2)
            try_seq(a_opt, f"final_cd_{cycle}")

    # === Phase 8: Last LP-FW push ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
