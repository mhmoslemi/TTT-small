    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(a, T):
    """
    Compute gradient of T*logsumexp(conv(a,a)/T) using FFT.
    Returns (gradient, softmax_weights, convolution).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum

    # grad[idx] = 2 * sum_j p[idx+j] * a[j]
    # = 2 * cross_correlation(p, a)[idx]
    N = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(P * np.conj(A), N)
    grad = 2.0 * R[:n]
    return grad, p, conv


def subgradient_max_conv(a):
    """Compute a subgradient of max_k conv(a,a) at a (using argmax)."""
    n = len(a)
    conv = fast_convolve(a)
    k_max = int(np.argmax(conv))
    sg = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    sg[idx] = 2.0 * a[k_max - idx]
    return sg, conv


def multi_peak_subgradient(a, tol=0.01):
    """Compute convex combination of subgradients at peaks near the max."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    threshold = max_val * (1.0 - tol)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    # Weight by how close each peak is to the max
    weights = (conv[active] - max_val * (1.0 - 2 * tol)) 
    weights = np.maximum(weights, 0.0)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(len(active))
    else:
        weights = weights / wsum
    
    sg = np.zeros(n)
    for w, k in zip(weights, active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        sg[idx] += w * 2.0 * a[k - idx]
    return sg, conv


def adam_smooth_optimize(a0, time_limit, T_init=0.001, lr_init=0.01, T_decay=0.9995,
                         beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the logsumexp-smoothed max of convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        grad, p, conv = smooth_grad_fft(a, T)
        max_val = conv.max()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = project_simplex(a - step)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        # Backtracking if objective increases too much
        if obj_new > max_val * 1.01 and t > 10:
            for shrink in [0.5, 0.25, 0.1]:
                a_new = project_simplex(a - shrink * step)
                obj_new = fast_convolve(a_new).max()
                if obj_new <= max_val * 1.001:
                    break

        a = a_new
        T = max(T * T_decay, 1e-6)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive: if stuck, try larger T temporarily
        if no_improve > 100:
            T = max(T * 2.0, 0.005)
            lr *= 0.8
        if no_improve > 300:
            # Reset with perturbation
            rng = np.random.default_rng(t * 31337 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            T = T_init
            lr = lr_init
            no_improve = 0
            best_obj = fast_convolve(a).max()

    return best_a, best_obj


def active_set_optimize(a0, time_limit, tol=0.005):
    """Optimize using multi-peak subgradient with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.05
    v = np.zeros(n)  # momentum
    mom = 0.9

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        sg, conv = multi_peak_subgradient(a, tol=tol)
        max_val = conv.max()

        # Momentum
        v = mom * v - lr * sg
        a_new = project_simplex(a + v)
        obj_new = fast_convolve(a_new).max()

        # Backtracking
        if obj_new > max_val * 1.001:
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                a_new = project_simplex(a + shrink * v)
                obj_new = fast_convolve(a_new).max()
                if obj_new < max_val + 1e-10:
                    break
            if obj_new >= max_val:
                a_new = a
                obj_new = max_val
                lr *= 0.7
        else:
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 50:
            lr *= 0.9
            mom = min(0.95, mom + 0.002)
        if no_improve > 200:
            rng = np.random.default_rng(it * 6700417 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = 0.05
            mom = 0.9
            no_improve = 0

    return best_a, best_obj


def pairwise_swap_optimize(a0, time_limit):
    """Local search: try moving mass between pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Sample random pairs
        improved = False
        n_tries = min(50000, max(1000, (int)(2000 * (time_limit - (time.monotonic() - t0)))))
        for _ in range(n_tries):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Try moving small epsilon from i to j (or vice versa)
            eps = 0.001 * max(a[i], a[j], 1.0 / n)
            for direction in [1, -1]:
                a_try = a.copy()
                a_try[i] -= direction * eps
                a_try[j] += direction * eps
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-15:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Escalate: try larger eps or全局 perturbation
            if it > 20:
                break
            eps_scale = min(0.01, 0.001 * it)

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Optimize one coordinate at a time with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            if a[i] < 1e-15:
                continue
            # Try adjusting a[i] by moving mass to a random other coordinate
            j = (i + 1) % n
            if a[j] < 1e-15:
                j = (i + 2) % n
            if a[j] < 1e-15:
                continue

            # Binary search for best transfer amount
            best_eps = 0.0
            best_val = best_obj
            for trial_frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    eps = trial_frac * a[i]
                    if direction == 1:
                        a_try = a.copy()
                        a_try[i] -= eps
                        a_try[j] += eps
                    else:
                        a_try = a.copy()
                        a_try[i] += eps * a[j] / max(a[i], 1e-15)
                        a_try[j] -= eps * a[j] / max(a[i], 1e-15)
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    val = fast_convolve(a_try).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_eps = eps * direction
                        best_a_try = a_try.copy()

            if best_eps != 0:
                a = best_a_try
                best_obj = best_val
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def spectral_reinit(n, rng):
    """Generate a random initialization by shaping the power spectrum."""
    # Start with a random phase and flat-ish magnitude
    N = 1 << (2 * n).bit_length()
    target_power = np.ones(N // 2 + 1)
    # Add some shape to avoid purely random (which gives flat conv ~ 1/n)
    freq = np.arange(N // 2 + 1) / (N // 2)
    shape = 1.0 + 0.5 * np.cos(2 * np.pi * freq) 
    target_power *= shape
    target_power[0] = 1.0  # DC
    
    mag = np.sqrt(target_power)
    phase = rng.standard_normal(N // 2 + 1) * np.pi
    phase[0] = 0
    
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[:] = mag * np.exp(1j * phase)
    
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def geometric_init(n, rng):
    """Generate geometric-like initializations with various parameters."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Various exponential decays
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.exp(-alpha * x / n)
        a = a / a.sum()
        inits.append(a)
    
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 / (x + 1.0) ** p
        a = a / a.sum()
        inits.append(a)
    
    # Sine-based
    for p in [1, 2, 3, 4]:
        a = np.sin(np.pi * (x + 0.5) / (2 * n)) ** p
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    
    # Two-sided exponential
    for alpha in [1.0, 2.0, 3.0]:
        center = n * 0.3
        a = np.exp(-alpha * np.abs(x - center) / n)
        a = a / a.sum()
        inits.append(a)
    
    return inits


def resample_to_n(parent, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(parent)
    if n_src == n_target:
        return parent.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    a = np.interp(x_tgt, x_src, parent)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a / s

    best_seq = None

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score

        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    print("\n=== Phase 1: Structural Variants ===")
    t_start = time.monotonic()

    # Try mirror
    mirror = parent[::-1]
    try_seq(mirror, "mirror")

    # Try different n values
    n_candidates = sorted(set([
        n_parent, n_parent - 4, n_parent + 4, n_parent - 16, n_parent + 16,

                a_opt2, _ = active_set_optimize(a_opt, time_limit=min(remaining, 15), tol=0.003)
                try_seq(a_opt2, f"nvar_active_{n_try}")

    print(f"  Phase 6 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.1f}s)")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
