"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent with log-sum-exp smoothing
2. Targeted peak-shaving heuristic
3. LP-based Frank-Wolfe with adaptive M
4. Multi-length exploration with interpolation
5. Pairwise coordinate optimization with golden-section line search
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, T):
    """Compute weighted subgradient of the smoothed max objective."""
    n = len(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def nesterov_gradient_descent(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    # Nesterov momentum
    y = f.copy()  # momentum position
    m = np.zeros(n)  # velocity

    # Parameters
    beta = 0.9  # momentum
    base_lr = 0.003
    T_start = 8.0 / n
    T_end = 0.3 / n
    top_k_start = max(100, n)
    top_k_end = max(20, n // 10)

    it = 0
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / (time_limit * 50))
        
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 + 0.5 * np.cos(np.pi * progress))  # cosine schedule

        # Nesterov: y = f + beta * velocity
        y = f + beta * m
        
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k, T)
        
        # Update velocity
        m = beta * m - lr * sg
        f_new = project_simplex(y + m)
        
        # Check convergence
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > max_no_improve:
                # Restart with noise
                rng = np.random.default_rng(it * 17 + 3)
                noise_scale = 0.003 + 0.002 * min(no_improve // 100, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                y = f.copy()
                m = np.zeros(n)
                no_improve = 0
                continue
            continue
        
        f = f_new
        conv = fast_convolve(f)
        new_M = conv.max()
        
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    NGD iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def peak_shave(f0, time_limit, verbose=False):
    """Directly target and reduce the worst peaks in the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find the top few peaks
        top_peaks = np.argsort(conv)[::-1][:5]
        
        improved = False
        for k in top_peaks:
            # The value at position k is sum_{i} f[i]*f[k-i]
            # To reduce it, we can reduce f at positions that contribute to this peak
            # Find which pairs contribute most
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = f[idx] * f[k - idx]
            
            # Find the pair with maximum contribution
            if len(contributions) == 0:
                continue
            max_contrib_idx = np.argmax(contributions)
            i = idx[max_contrib_idx]
            j = k - i
            
            if i == j:
                # Can't split, try reducing this element
                # Move mass from f[i] to a random other position
                if f[i] < 1e-10:
                    continue
                # Try moving a fraction of mass
                for frac in [0.1, 0.2, 0.3]:
                    mass = frac * f[i]
                    # Move to a position that doesn't contribute to this peak
                    candidates = np.random.choice(n, size=min(20, n), replace=False)
                    for dest in candidates:
                        if dest == i or (js <= dest <= je and js <= k - dest <= je):
                            continue
                        f_trial = f.copy()
                        f_trial[i] -= mass
                        f_trial[dest] += mass
                        if f_trial[i] < 0 or f_trial[dest] < 0:
                            continue
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < M - 1e-12:
                            f = f_trial
                            M = val
                            improved = True
                            break
                    if improved:
                        break
            else:
                # Transfer mass between i and j
                total = f[i] + f[j]
                if total < 1e-10:
                    continue
                curr_ratio = f[i] / total
                # Try perturbing the ratio
                for delta in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15]:
                    new_ratio = np.clip(curr_ratio + delta, 0.0, 1.0)
                    f_trial = f.copy()
                    f_trial[i] = new_ratio * total
                    f_trial[j] = (1 - new_ratio) * total
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < M - 1e-12:
                        M = val
                        improved = True
                        break
                if improved:
                    f = f_trial
                    if M < best_obj:
                        best_obj = M
                        best_f = f.copy()
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_factor = 1.0  # Adaptive M scaling

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Solve LP with current M
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            rows = np.arange(j, j + n)
            A_ub[rows, j] = f
        b_ub = np.full(2 * n - 1, M * M_factor)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            M_factor *= 1.01
            if M_factor > 1.1:
                M_factor = 1.0
                no_improve += 10
            continue

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue

        # Line search (coarse)
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            M_factor = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 40, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj

                if c < 1e-10:
                    continue

                best_t = f[i] / c
                best_val = best_obj  # Use current best as threshold

                        best_t = frac

                    else:
                        lo_g, hi_g = min(0.9, best_t - 0.1), 1.0

                    c2 = lo_g + gr * (hi_g - lo_g)
                    
                    f1 = f.copy()
                    f1[i] = c1 * c
                    f1[j] = (1 - c1) * c
                    fc1 = fast_convolve(f1).max()
                    
                    f2 = f.copy()
                    f2[i] = c2 * c
                    f2[j] = (1 - c2) * c
                    fc2 = fast_convolve(f2).max()
                    
                    for _ in range(5):
                        if hi_g - lo_g < 1e-5:
                            break
                        if fc1 < fc2:
                            hi_g = c2
                            c2 = c1
                            fc2 = fc1
                            c1 = hi_g - gr * (hi_g - lo_g)
                            f1 = f.copy()
                            f1[i] = c1 * c
                            f1[j] = (1 - c1) * c
                            fc1 = fast_convolve(f1).max()
                        else:
                            lo_g = c1
                            c1 = c2
                            fc1 = fc2
                            c2 = lo_g + gr * (hi_g - lo_g)
                            f2 = f.copy()
                            f2[i] = c2 * c
                            f2[j] = (1 - c2) * c
                            fc2 = fast_convolve(f2).max()
                    
                    best_t = (lo_g + hi_g) / 2
                    f_trial = f.copy()
                    f_trial[i] = best_t * c
                    f_trial[j] = (1 - best_t) * c
                    best_val = fast_convolve(f_trial).max()

                        f = f_trial

                        pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent with log-sum-exp smoothing and cosine LR
2. Targeted peak-shaving heuristic
3. LP-based Frank-Wolfe with adaptive constraint scaling
4. Multi-length exploration with interpolation
5. Golden-section pairwise coordinate optimization
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, T):
    """Compute weighted subgradient of smoothed max objective using top-k peaks."""
    n = len(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def nesterov_gradient_descent(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient descent with adaptive temperature and cosine LR."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)

    beta = 0.9
    base_lr = 0.004
    T_start = 10.0 / n
    T_end = 0.2 / n
    top_k_start = max(100, n)
    top_k_end = max(10, n // 12)

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / max(1, time_limit * 60))

        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (0.5 + 0.5 * np.cos(np.pi * progress))

        y = f + beta * m
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k, T)

        m = beta * m - lr * sg
        f_new = project_simplex(y + m)

        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 400:
                rng = np.random.default_rng(it * 17 + 3)
                noise_scale = 0.004 + 0.003 * min(no_improve // 100, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                m = np.zeros(n)
                no_improve = 0
                continue
            continue

        f = f_new
        conv = fast_convolve(f)
        new_M = conv.max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    NGD iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj


def peak_shave(f0, time_limit, verbose=False):
    """Directly target and reduce the worst peaks in the auto-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        top_peaks = np.argsort(conv)[::-1][:8]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contributions = f[idx] * f[k - idx]

            if len(contributions) == 0:
                continue

            top_contrib = np.argsort(contributions)[::-1][:4]
            for ci in top_contrib:
                i = idx[ci]
                j = k - i
                if i == j or j < 0 or j >= n:
                    continue

                total = f[i] + f[j]
                if total < 1e-10:
                    continue
                curr_ratio = f[i] / total

                best_local = M
                best_r = curr_ratio
                for delta in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2]:
                    new_r = np.clip(curr_ratio + delta, 0.0, 1.0)
                    f_trial = f.copy()
                    f_trial[i] = new_r * total
                    f_trial[j] = (1.0 - new_r) * total
                    f_trial = np.maximum(f_trial, 0.0)
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_local - 1e-13:
                        best_local = val
                        best_r = new_r

                if best_local < M - 1e-12:
                    f[i] = best_r * total
                    f[j] = (1.0 - best_r) * total
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    M = best_local
                    improved = True
                    if M < best_obj - 1e-13:
                        best_obj = M
                        best_f = f.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M scaling and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_factor = 1.0

    # Pre-build constraint structure (rows for each column j)
    col_rows = [np.arange(j, j + n) for j in range(n)]

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            rows = col_rows[j]
            A_ub[rows, j] = f
        b_ub = np.full(2 * n - 1, M * M_factor)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            M_factor *= 1.02
            if M_factor > 1.05:
                M_factor = 1.0
                no_improve += 5
            continue

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            M_factor = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 30, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates with golden-section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    no_improve_passes = 0
    passes = 0
    offsets = [1, 2, 3, 5, 8, 13, 21, 34]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        order = rng.permutation(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx_i]

            for offset in offsets:
                j = order[(idx_i + offset) % n]
                if i == j:
                    continue

                c = f[i] + f[j]
                if c < 1e-10:
                    continue

                # Coarse scan
                best_val = best_obj
                best_t = f[i] / c
                for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                    f_trial = f.copy()
                    f_trial[i] = frac * c
                    f_trial[j] = (1.0 - frac) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val:
                        best_val = val
                        best_t = frac

                # Golden section refinement
                if best_val < best_obj - 1e-13:
                    if best_t < 0.5:
                        lo, hi = 0.0, min(1.0, best_t + 0.15)
                    else:
                        lo, hi = max(0.0, best_t - 0.15), 1.0
                    gr = (np.sqrt(5) - 1) / 2

                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = f.copy()
                    f1[i] = c1 * c
                    f1[j] = (1.0 - c1) * c
                    fc1 = fast_convolve(f1).max()
                    f2 = f.copy()
                    f2[i] = c2 * c
                    f2[j] = (1.0 - c2) * c
                    fc2 = fast_convolve(f2).max()

                    for _ in range(6):
                        if hi - lo < 1e-5:
                            break
                        if fc1 < fc2:
                            hi = c2
                            c2 = c1
                            fc2 = fc1
                            c1 = hi - gr * (hi - lo)
                            f1 = f.copy()
                            f1[i] = c1 * c
                            f1[j] = (1.0 - c1) * c
                            fc1 = fast_convolve(f1).max()
                        else:
                            lo = c1
                            c1 = c2
                            fc1 = fc2
                            c2 = lo + gr * (hi - lo)
                            f2 = f.copy()
                            f2[i] = c2 * c
                            f2[j] = (1.0 - c2) * c
                            fc2 = fast_convolve(f2).max()

                    best_t = (lo + hi) / 2.0
                    f_trial = f.copy()
                    f_trial[i] = best_t * c
                    f_trial[j] = (1.0 - best_t) * c
                    best_val = fast_convolve(f_trial).max()

                    if best_val < best_obj - 1e-13:
                        f = f_trial
                        best_obj = best_val
                        best_f = f.copy()
                        pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Nesterov GD, peak shaving, LP-FW,
    multi-length exploration, diverse restarts, and pairwise refinement.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Nesterov GD on parent ===
    print(f"\n=== Phase 1: Nesterov GD (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 280)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = nesterov_gradient_descent(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "ngd_1")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.12, 120)
        if t2 > 15:
            print(f"  Time: {t2:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t2)
            try_seq(a_opt, "peak_shave")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.18, 220)
        if t3 > 30:
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw_3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    base_n = len(base)

    target_lengths = []
    for d in [128, -128, 256, -256, 512, -64]:
        tn = base_n + d
        if 96 <= tn <= 2048 and tn != base_n:
            target_lengths.append(tn)
    for tn in [512, 768, 1024, 1536]:
        if tn != base_n and tn not in target_lengths:
            target_lengths.append(tn)
    target_lengths.sort(key=lambda x: abs(x - base_n))
    target_lengths = target_lengths[:4]

    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)

        t_alloc = min(rem * 0.25, 100)
        if t_alloc > 15:
            print(f"  NGD n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_gradient_descent(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"ngd_n{target_n}")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_lpfw = min(rem * 0.2, 80)
            if t_lpfw > 15:
                print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s")
                a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw, verbose=False)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, 4 - p), 140)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                rng = np.random.default_rng(200 + p * 7)
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            elif p == 2:
                f_init = best_seq[::-1].copy()
                f_init = f_init / f_init.sum()
            else:
                x = np.linspace(0, 1, n_best)
                bump = np.exp(-8.0 * (x - 0.5) ** 2)
                f_init = 0.4 * bump / bump.sum() + 0.6 * best_seq
                f_init = project_simplex(f_init)

            if p % 2 == 0:
                a_opt, _ = nesterov_gradient_descent(f_init, time_limit=min(t_alloc, 100), verbose=(p == 0))
            else:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 110), verbose=(p == 0))
            try_seq(a_opt, f"diverse_{p}")

    # === Phase 6: Pairwise optimization ===
    print(f"\n=== Phase 6: Pairwise (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_pw = min(rem * 0.35, 250)
            print(f"  Time: {t_pw:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw)
            try_seq(a_opt, "pairwise")

            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt, _ = nesterov_gradient_descent(a_opt, time_limit=min(rem, 80))
                try_seq(a_opt, "ngd_after_pw")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = nesterov_gradient_descent(best_seq, time_limit=min(rem * 0.35, 80), verbose=True)
            try_seq(a_opt, "ngd_final")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.3, 60))
            try_seq(a_opt, "lpfw_final")

        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            rng = np.random.default_rng(700 + trial * 13)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
            a_opt, _ = nesterov_gradient_descent(f_pert, time_limit=min(rem, 25))
            try_seq(a_opt, f"perturb_{trial}")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = pairwise_optimize(best_seq, time_limit=min(rem, 35))
            try_seq(a_opt, "pw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
