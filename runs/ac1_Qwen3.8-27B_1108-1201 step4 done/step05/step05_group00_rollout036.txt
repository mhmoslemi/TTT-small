```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_descent_fast(a, time_limit, verbose=False):
    """
    Coordinate descent with efficient incremental convolution updates.
    For each coordinate j, we try several candidate values and pick the best.

    best_obj = 2.0 * n * conv.max()

        sweep_improvement = 0
        conv = fast_convolve(a)

            r = 1.0 - aj  # sum of others (since sum=1)
            if r < 1e-15:

            # The key candidates:
```

```python
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    a = a / a.sum()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        sweep_improved = False
        rng = np.random.default_rng(42 + sweep * 31)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            aj = a[j]
            r = 1.0 - aj
            if r < 1e-15:
                continue
            
            # Compute A_k = conv[k] - f_k(aj) for all k
            # f_k(aj) = aj^2 if k=2j; 2*aj*a[k-j] if 0<=k-j<n and k!=2j; 0 else
            # A_k = conv[k] - f_k(aj)
            # We don't need to store all A_k, just the max over inactive k's
            # and the values at active k's.
            
            # Active k's: k in [max(0, j), min(2n-2, j+n-1)] (where k-j is in [0, n-1])
            # Inactive k's: all other k's
            
            # Max over inactive k's:
            # These are k where k-j < 0 or k-j >= n
            # i.e., k < j or k >= j+n
            # A_k for inactive = conv[k] (since f_k = 0 for inactive)
            # So M_inactive = max(conv[k] for k in inactive range)
            
            k_lo = max(0, j)  # first active k
            k_hi = min(2*n-2, j+n-1)  # last active k
            
            # Inactive: [0, k_lo-1] and [k_hi+1, 2n-2]
            if k_lo > 0:
                m_inact1 = conv[:k_lo].max() if k_lo > 0 else 0
            else:
                m_inact1 = 0
            if k_hi < 2*n-2:
                m_inact2 = conv[k_hi+1:].max() if k_hi+1 < 2*n-1 else 0
            else:
                m_inact2 = 0
            M_inactive = max(m_inact1, m_inact2)
            
            # For active k's, A_k = conv[k] - f_k(aj)
            # f_k(aj):
            #   k = 2j: aj^2
            #   else: 2*aj*a[k-j]
            
            # Precompute A_k for active k's
            active_ks = range(k_lo, k_hi+1)
            # A_k values:
            A_active = np.empty(k_hi - k_lo + 1)
            for idx, k in enumerate(active_ks):
                if k == 2*j:
                    A_active[idx] = conv[k] - aj*aj
                else:
                    kj = k - j
                    A_active[idx] = conv[k] - 2*aj*a[kj]
            
            # Now for candidate x:
            # max_k (A_k + f_k(x)) = max(M_inactive, max_active (A_k + f_k(x)))
            # f_k(x) = x^2 if k=2j; 2*x*a[k-j] if k!=2j
            # (r+x)^2 in denominator
            
            # Try candidates:
            candidates = [0.0, 1.0/n, 2.0/n, 0.5/n]
            if aj > 1e-10:
                candidates.extend([aj*0.1, aj*0.5, aj*0.75, aj*1.25, aj*1.5, aj*2.0])
            
            best_x = aj
            best_val = best_obj
            
            for x in candidates:
                if x < 0:
                    continue
                if abs(x - aj) < 1e-14:
                    continue
                
                denom = (r + x) ** 2
                if denom < 1e-30:
                    continue
                
                # Compute max over active k's of (A_k + f_k(x))
                # f_k(x) = x^2 if k=2j; 2*x*a[k-j]
                if 2*j >= k_lo and 2*j <= k_hi:
                    idx_2j = 2*j - k_lo
                    f_2j = x * x
                    active_vals = A_active.copy()
                    active_vals[idx_2j] += f_2j
                else:
                    active_vals = A_active.copy()
                
                # For k != 2j in active range: f_k(x) = 2*x*a[k-j]
                # We need to add this to A_active for all active k != 2j
                # a[k-j] for k in [k_lo, k_hi]: a[0], a[1], ..., a[n-1] (since k-j ranges from k_lo-j to k_hi-j)
                k_j_lo = k_lo - j  # = max(0, j) - j = 0 (if j>=0) or j-j=0
                k_j_hi = k_hi - j  # = min(2n-2, j+n-1) - j = min(n-2, n-1) = n-1 (for j < n-1) or n-2
                # So k-j ranges from max(0, j)-(j) = 0 to min(2n-2, j+n-1)-j
                # = n-1 (if j+n-1 <= 2n-2, i.e., j <= n-1, always true)
                # Wait: k_hi = min(2n-2, j+n-1). k_hi - j = min(2n-2-j, n-1)
                # For j >= 1: 2n-2-j >= n-1 iff j < n-1. So k_hi-j = n-1 for j < n-1, and 2n-2-j for j = n-1.
                
                # Simply: the values a[k-j] for k in [k_lo, k_hi] are a[0..min(n-1, k_hi-j)]
                # which is a[max(0, k_lo-j) .. min(n-1, k_hi-j)]
                
                kj_start = max(0, k_lo - j)
                kj_end = min(n - 1, k_hi - j)
                
                # f_k(x) = 2*x*a[k-j] for k in active_ks, k != 2j
                # Vectorized:
                f_active = 2.0 * x * a[kj_start:kj_end+1]
                # But we need to map these to the right positions in active_vals
                # k - j = kj => k = k_lo + (idx) where idx = k - k_lo
                # kj = k - j = k_lo - j + idx = kj_start + idx
                # So f_active[idx] = 2*x*a[kj_start + idx] for idx = 0, ..., kj_end-kj_start
                # And this adds to A_active[idx] for the corresponding k
                
                # The active_ks range is k_lo to k_hi, with length k_hi-k_lo+1
                # kj ranges from kj_start to kj_end, with length kj_end-kj_start+1
                # kj_start = max(0, k_lo-j) and k_lo = max(0, j), so k_lo-j = max(0,j)-j = 0 for j>=0
                # Actually k_lo = max(0, j) = j (since j >= 0 always)
                # So kj_start = max(0, j-j) = 0
                # kj_end = min(n-1, k_hi-j)
                # k_hi = min(2n-2, j+n-1)
                # k_hi - j = min(2n-2-j, n-1) = n-1 (for j <= n-1, which is always)
                # So kj_end = n-1
                
                # So f_active = 2*x*a[0:n] (full vector a!)
                # And it adds to A_active for all positions where k != 2j
                
                active_vals = A_active + f_active  # adds 2*x*a[k-j] to all active
                # Subtract the double-counted part at k=2j (we added 2*x*a[j] but should add x^2)
                if 2*j >= k_lo and 2*j <= k_hi:
                    idx_2j = 2*j - k_lo
                    # We added 2*x*a[kj_start+idx_2j] = 2*x*a[j] at position 2j
                    # But the correct f_2j(x) = x^2
                    # So we need to correct: active_vals[idx_2j] = A_2j + x^2
                    # Currently: A_2j + 2*x*a[j]
                    # Correction: subtract 2*x*a[j], add x^2
                    active_vals[idx_2j] = A_active[idx_2j] + x * x
                
                max_active = active_vals.max()
                overall_max = max(M_inactive, max_active)
                val = 2.0 * n * overall_max / denom
                
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x
            
            if best_x != aj:
                a[j] = best_x
                s = a.sum()
                a = a / s
                conv = fast_convolve(a)
                sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose and sweep % 50 == 0:
                    print(f"  CD sweep {sweep}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 30:
                rng_p = np.random.default_rng(sweep * 67)
                a = best_a + rng_p.standard_normal(n) * (0.001 + 0.002 * min(1, no_improve/100))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                no_improve = 0
    
    return best_a, best_obj
```

This should be fast. Let me estimate the cost:
- Per coordinate: O(n) for computing A_active and evaluating candidates (n * num_candidates)
- Per sweep: O(n² * num_candidates) = 768² * 10 = 5.9M operations
- Plus one FFT per sweep: O(n log n) = 22K
- Total per sweep: ~6M operations
- In numpy, this is maybe 10-50ms per sweep
- In 1800 seconds: ~36,000 to 180,000 sweeps

That's more than enough. The algorithm should converge well.

Now, will coordinate descent alone be sufficient to get below 1.5027? Let me add a CVXPY-based refinement as well.

For the CVXPY approach, I'll use the "constraint generation" method:
1. Start with the best sequence found by coordinate descent
2. Compute the convolution and find the top K peaks
3. Solve the QP: min t s.t. a >= 0, sum(a)=1, conv(a,a)[k] <= t for k in top K
4. Check if other peaks are violated
5. Add violated peaks to the constraint set
6. Repeat

CVXPY with MOSEK should handle QPs with a few hundred quadratic constraints efficiently.

Actually, let me think about whether the QP is tractable. The problem is:
min t
s.t. a^T M_k a <= t for k in K (|K| constraints)
     a >= 0
     sum(a) = 1

Each M_k is a sparse symmetric PSD matrix (it's a permutation-like matrix). The QP has n+1 variables and |K| + n + 1 constraints.

For n=768 and |K|=50, this is a QP with 769 variables and 819 constraints. MOSEK should handle this easily.

Let me now put together the complete solution. I'll combine:
1. Fast coordinate descent (main workhorse)
2. CVXPY QP refinement (for global improvements)
3. Multi-start with different initializations
4. Different n values

Let me also think about what n values to try. The previous program used n=768. Let me try n in {384, 768, 1024, 1536, 2048}.

Actually, the objective is 2*n*max(conv)/sum². For larger n, the factor 2n is larger, but we can spread the mass over more positions, potentially reducing the peak. The trade-off depends on the specific sequence.

Let me write the final solution:

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Fast coordinate descent with O(n) per-candidate evaluation
2. CVXPY QP with iterative constraint generation
3. Multi-n exploration with adaptive time allocation
"""

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def coord_descent_fast(a_init, time_limit, verbose=False):
    """
    Fast coordinate descent with O(n) per-candidate evaluation.
    Exploits the structure of auto-convolution for efficient updates.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        sweep += 1
        
        sweep_improved = False
        rng = np.random.default_rng(42 + sweep * 31)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            aj = a[j]
            r = 1.0 - aj  # sum of others (since sum=1)
            if r < 1e-15:
                continue
            
            # Active k range: k where k-j is in [0, n-1]
            k_lo = j  # k-j >= 0 => k >= j
            k_hi = min(2*n-2, j+n-1)  # k-j <= n-1 => k <= j+n-1
            
            # Inactive max: max of conv[k] for k < k_lo or k > k_hi
            M_inactive = 0.0
            if k_lo > 0:
                M_inactive = max(M_inactive, conv[:k_lo].max())
            if k_hi < 2*n-2:
                M_inactive = max(M_inactive, conv[k_hi+1:].max())
            
            # Precompute A_k for active k's: A_k = conv[k] - f_k(aj)
            # f_k(aj) = aj^2 if k=2j; 2*aj*a[k-j] otherwise
            num_active = k_hi - k_lo + 1
            A_active = np.empty(num_active)
            # k-j ranges from k_lo-j to k_hi-j, i.e., from 0 to min(n-1, k_hi-j)
            # k_lo = j, so k_lo-j = 0
            # k_hi = min(2n-2, j+n-1), so k_hi-j = min(n-2, n-1) = n-1 (since j>=0)
            # Wait, for j=0: k_hi = min(2n-2, n-1) = n-1, k_hi-j = n-1
            # For j=n-1: k_hi = min(2n-2, 2n-2) = 2n-2, k_hi-j = 2n-2-(n-1) = n-1
            # So k-j always ranges from 0 to n-1. Good.
            
            # Vectorized computation of A_active
            # For k in [k_lo, k_hi], k-j is in [0, n-1]
            # f_k(aj) = 2*aj*a[k-j] for all k, EXCEPT k=2j where it's aj^2
            A_active = conv[k_lo:k_hi+1] - 2.0 * aj * a[0:n]
            # Correct k=2j:
            if k_lo <= 2*j <= k_hi:
                idx_2j = 2*j - k_lo
                # A_active[idx_2j] should be conv[2j] - aj^2
                # Currently: conv[2j] - 2*aj*a[j] = conv[2j] - 2*aj^2
                # Need: conv[2j] - aj^2
                A_active[idx_2j] += aj * aj  # add back aj^2 (since we subtracted 2aj^2, need to add aj^2)
            
            # Try candidate values for x
            # Strategy: include 0, uniform value, and perturbations of current
            candidates = [0.0, 1.0/n]
            if aj > 1e-10:
                # Log-scale perturbations
                for mult in [0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 4.0]:
                    candidates.append(aj * mult)
            else:
                candidates.extend([0.5/n, 2.0/n, 4.0/n])
            
            best_x = aj
            best_val = 2.0 * n * conv.max()  # current value
            
            for x in candidates:
                if x < 0:
                    continue
                if abs(x - aj) < 1e-14:
                    continue
                
                denom = (r + x) ** 2
                if denom < 1e-30:
                    continue
                
                # Compute max_k (A_k + f_k(x)) over active k's
                # f_k(x) = 2*x*a[k-j] for all active k, EXCEPT k=2j where f=x^2
                f_active = 2.0 * x * a[0:n]
                if k_lo <= 2*j <= k_hi:
                    idx_2j = 2*j - k_lo
                    f_active[idx_2j] = x * x
                
                active_max = (A_active + f_active).max()
                overall_max = max(M_inactive, active_max)
                val = 2.0 * n * overall_max / denom
                
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x
            
            if best_x != aj:
                a[j] = best_x
                s_a = a.sum()
                a = a / s_a
                conv = fast_convolve(a)
                sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
            cur

_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose and sweep % 100 == 0:
                    print(f"  CD sweep {sweep}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 50:
                scale = 0.001 + 0.005 * min(1.0, no_improve / 200.0)
                rng_p = np.random.default_rng(sweep * 67 + 13)
                a = best_a + rng_p.standard_normal(n) * scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                no_improve = 0
    
    return best_a, best_obj

def cvxpy_qp_refine(a_init, time_limit, max_constraints=200, verbose=False):
    """
    CVXPY QP with iterative constraint generation.
    min t s.t. a>=0, sum(a)=1, conv(a,a)[k] <= t for k in active set.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj

 = 2.0 * n * fast_convolve(a).max()
    
    max_iters = 50
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        conv = fast_convolve(a)
        
        # Select top K peaks as active constraints
        K = min(max_constraints, 2*n-1)
        # Prefer peaks that are above the current t
        idx = np.argsort(conv)[::-1][:K]
        
        # Solve QP
        a_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            a_var >= 0,
            cp.sum(a_var) == 1
        ]
        
        for k in idx:
            # Build sparse M_k
            rows = []
            cols = []
            vals = []
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            for i in range(k_lo, k_hi + 1):
                j_k = k - i
                if i == j_k:
                    rows.append(i)
                    cols.append(j_k)
                    vals.append(1.0)
                elif i < j_k:
                    rows.append(i)
                    cols.append(j_k)
                    vals.append(1.0)
                    rows.append(j_k)
                    cols.append(i)
                    vals.append(1.0)
            M_k = sp.coo_matrix((vals, (rows, cols)), shape=(n, n)).tocsc()
            constraints.append(cp.quad_form(a_var, M_k) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
        except:
            try:
                prob.solve(solver=cp clarify.SCS, verbose=False)
            except:
                break
        
        if a_var.value is not None:
            a_new = np.maximum(a_var.value, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                obj_new = 2.0 * n * fast_convolve(a_new).max()
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"  QP iter {iteration}: t={t_var.value:.8f}, obj={best_obj:.10f}")
        
        # Check convergence
        conv = fast_convolve(a)
        if conv.max() <= t_var.value + 1e-8:
            break
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800):

            a_opt, obj_opt = coord_descent_fast(best_seq, time_limit=t_alloc, verbose=True)
```
