This means the problem is NOT convex, and we can't use standard convex optimization techniques. The LP approach works because it's a different problem (minimum t such that (f*g)[k] <= t for all k, which IS linear in g for fixed f).

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fista_optimize(f0, time_limit, lr=0.005, top_k=50, temp=0.0001, verbose=False):
    """FISTA (accelerated proximal gradient) with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # FISTA variables
    x = f.copy()  # current iterate
    y = f.copy()  # extrapolated point
    t_k = 1.0     # momentum parameter
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y using top-k peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute weighted subgradient of max_k (f*f)[k] at y
        # d/dy_i [(y*y)[k]] = sum_j (delta_{i+j,k} + delta_{j+i,k}) * y[j]
        # = 2 * y[k-i] (for valid indices)
        # So the subgradient at peak k is 2*y[k-i] (reversed and shifted)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            # (y*y)[k] = sum_j y[j]*y[k-j]
            # d/dy[i] = sum_j (delta_{i,j}*y[k-j] + delta_{i,k-j}*y[j])
            # = y[k-i] + y[k-i] (if both valid) = 2*y[k-i] (if valid)
            # More carefully: d/dy[i] of sum_j y[j]*y[k-j]
            # = y[k-i] (from the first factor when j=i) + y[k-i] (from the second factor when k-j=i, i.e., j=k-i)
            # = 2*y[k-i] if 0 <= k-i < n
            # But we need to be more careful for the full convolution
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                # y[j]*y[k-j], derivative wrt y[i]:
                # if i == j: contributes y[k-j] = y[k-i]
                # if i == k-j (i.e., j = k-i): contributes y[j] = y[k-i]
                # So total contribution to sg[i] from peak k is:
                # y[k-i] if 0<=i<=je and i>=js (i is a valid j)
                # + y[k-i] if 0<=k-i<n and k-i is a valid j (i.e., js <= k-i <= je)
                pass
        
        # Simpler approach: use the fact that the subgradient of (y*y)[k] with respect to y
        # is the vector v where v[i] = y[k-i] (if 0<=k-i<n) + y[k-i] (if 0<=k-i<n)
        # Actually for the full linear convolution:
        # (y*y)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} y[j]*y[k-j]
        # d/dy[i] = sum_j [delta_{j,i}*y[k-j] + delta_{k-j,i}*y[j]]
        # = y[k-i] (if k-i is in valid range for j, i.e., max(0,k-n+1) <= k-i <= min(n-1,k))
        #   + y[k-i] (if i is in valid range for j AND k-i is in valid range)
        # Hmm this is getting complex. Let me just use:
        # d/dy[i] (y*y)[k] = 2*y[k-i] if 0 <= k-i < n, else 0
        # (This is approximately correct for the "overlap" part)
        
        # Actually, the exact subgradient:
        # (y*y)[k] = sum_j y[j] * y[k-j] (j valid)
        # d/dy[i] = sum_j [delta(j,i)*y[k-j] + delta(k-j,i)*y[j]]
        # = (if i is valid j range) y[k-i] + (if k-i is valid j range) y[k-i]
        # Let's just say it's 2*y[k-i] if both i and k-i are in [0,n-1]
        # and y[k-i] if only one is valid.
        
        # For simplicity and speed, let's use the approximation:
        # sg[i] = 2 * sum_peaks w * y[|k-i|] (with appropriate masking)
        
        # Actually the cleanest way: for each peak k, the gradient of (y*y)[k] w.r.t. y
        # is a vector g_k where g_k[i] = y[k-i] if 0<=k-i<n else 0
        # (counting both forward and backward, so actually 2*y[k-i] when valid)
        
        # Let me just compute it directly:
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            # For each i, the contribution to the gradient is:
            # y[k-i] if 0 <= k-i < n (from the "forward" term)
            # + y[k-i] if 0 <= k-i < n (from the "backward" term, when j=k-i is valid)
            # But we need j to also be valid: max(0,k-n+1) <= j <= min(n-1,k)
            # So for the first term (j=i): need max(0,k-n+1) <= i <= min(n-1,k) AND 0<=k-i<n
            # For the second term (j=k-i): need max(0,k-n+1) <= k-i <= min(n-1,k) AND 0<=i<n
            
            # Simplified: the gradient at position i is y[k-i] * (indicator that i is valid j) 
            #         + y[k-i] * (indicator that k-i is valid j)

            sg[mask1] += 2.0 * w[idx] * y[ki[mask1]]
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        x_new = project_simplex(y - lr * sg)
        # Momentum
        y_new = x_new + ((t_k - 1) / t_new) * (x_new - x)
        y_new = project_simplex(y_new)
        
        x = x_new
        y = y_new
        t_k = t_new
        
        conv = fast_convolve(y)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = y.copy()
            no_improve = 0
            if verbose and it % 100 == 0:

        temp *= 1.003

            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            x = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = x.copy()
            t_k = 1.0
            lr = 0.003
            no_improve = 0
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def compute_subgradient(y, peaks, weights, n):
    """Compute weighted subgradient of max_k (y*y)[k] at y."""
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if weights[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idxs = np.arange(lo, hi + 1)
        sg[idxs] += 2.0 * weights[idx] * y[k - idxs]
    return sg

def fista_optimize(f0, time_limit, lr=0.005, top_k=50, temp=0.0001, verbose=False):
    """FISTA accelerated proximal gradient with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    x = f.copy()
    y = f.copy()
    t_k = 1.0

    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgradient(y, peaks, w, n)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = project_simplex(y - cur_lr * sg)
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y_new = project_simplex(y_new)

        x = x_new
        y = y_new
        t_k = t_new

        conv = fast_convolve(y)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = y.copy()
            no_improve = 0
            if verbose and it % 200 == 0:

        else:
            no_improve += 1

            y = x.copy()
            t_k = 1.0

def peak_target_cd(f0, time_limit, verbose=False):

    no_improve_pass = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1, max(1, (je - js + 1) // 30)):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-13:
                    continue

                for mult in [0.5, 0.7, 1.4, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom

                    f_trial[j] = new_val

                        break

        if not improved:
            no_improve_pass += 1
            if no_improve_pass > 5:

        else:

            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = fista_optimize(f_pert, time_limit=min(rem, 20),
                                       lr=0.0008, top_k=max(15, n_best // 15),
                                       temp=0.00001)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
