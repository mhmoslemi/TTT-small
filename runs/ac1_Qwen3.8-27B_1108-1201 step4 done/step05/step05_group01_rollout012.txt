def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def full_convolve(a, b):
    """Linear convolution of a and b via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_active_lp(f, M, active_indices, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    c = -np.ones(n)
    n_constr = len(active_indices)
    A_ub = np.zeros((n_constr, n))
    b_ub = np.full(n_constr, M)
    for i, k in enumerate(active_indices):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A_ub[i, j_lo:j_hi + 1] = f[k - np.arange(j_lo, j_hi + 1)]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_min_peak_lp(f, active_indices, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for k in active_indices, g >= 0, sum(g)=1."""
    # Variables: g[0..n-1], t
    # Minimize t
    # Constraints: (f*g)[k] - t <= 0 for each active k
    #              sum(g) = 1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    # Inequality constraints: (f*g)[k] - t <= 0
    n_constr = len(active_indices)
    A_ub = np.zeros((n_constr, n + 1))
    b_ub = np.zeros(n_constr)
    for i, k in enumerate(active_indices):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A_ub[i, j_lo:j_hi + 1] = f[k - np.arange(j_lo, j_hi + 1)]
        A_ub[i, -1] = -1.0  # -t

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

        temp = 0.0001 * (1.0 + it * 0.0005)

def coordinate_descent(f0, time_limit, max_passes=100):
    """Golden section coordinate descent for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_pass_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search on x = f[j]
            lo = 0.0
            hi = min(0.999, 3.0 * old_val)

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(14):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            if opt_val < best_obj - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = np.maximum(ft, 0.0)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def cosine_init(n, freq, amp, seed=0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            cur_lr = max(lr0 * 0.5, 0.002)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: active-set LP-FW, mirror descent,
    Nesterov subgradient, coordinate descent, and multi-restart.
    """
    dtype = np.float64
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng_master = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0).astype(dtype)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=dtype)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768, dtype=dtype) / 768
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n=768, score={p_score:.6f}")

    # === Phase 1: LP-FW on parent ===
    print("\n=== Phase 1: Active-set LP-FW on parent ===")
    t1 = min(300, deadline - time.monotonic() - 800)
    if t1 > 30:
        a_opt, _ = active_set_lp_fw(parent, time_limit=t1, top_k=15, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Mirror descent polishing ===
    print(f"\n=== Phase 2: Mirror descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_md = min(rem * 0.5, 200)
            a_opt, _ = mirror_descent(best_seq, time_limit=t_md, top_k=50, lr0=0.02, decay=0.99999)
            try_seq(a_opt, "mirror_descent")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_ns = min(rem * 0.5, 150)
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_ns,
                                         lr0=0.01, mom0=0.9, top_k=40, temp0=0.00005)
            try_seq(a_opt, "nesterov")

    # === Phase 4: Second LP-FW pass ===
    print(f"\n=== Phase 4: Second LP-FW pass (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_lp2 = min(rem * 0.6, 250)
            a_opt, _ = active_set_lp_fw(best_seq, time_limit=t_lp2, top_k=20)
            try_seq(a_opt, "lpfw_pass2")

    # === Phase 5: Multi-start exploration ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        starts = []

        # Structured perturbations
        starts.append(("noise_small", project_simplex(best_seq + rng_master.standard_normal(n_best) * 0.003)))
        starts.append(("noise_med", project_simplex(best_seq + rng_master.standard_normal(n_best) * 0.01)))
        starts.append(("reverse_mix", project_simplex(0.85 * best_seq + 0.15 * best_seq[::-1])))
        starts.append(("cosine_mix", project_simplex(0.9 * best_seq + 0.1 * cosine_init(n_best, 3, 0.3))))
        starts.append(("uniform_mix", project_simplex(0.92 * best_seq + 0.08 * np.ones(n_best) / n_best)))

        for label, f_init in starts:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 80)
            if t_alloc < 15:
                break

            # Quick mirror descent then LP-FW
            a_tmp, _ = mirror_descent(f_init, time_limit=min(t_alloc * 0.4, 30), top_k=40, lr0=0.02)
            a_opt, _ = active_set_lp_fw(a_tmp, time_limit=t_alloc * 0.6, top_k=15)
            try_seq(a_opt, f"multi_{label}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Different lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [1024, 512]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.35, 120)
            a_opt, _ = active_set_lp_fw(f_interp, time_limit=t_alloc, top_k=15)
            try_seq(a_opt, f"interp_n{target_n}")

    # === Phase 7: Coordinate descent polishing ===
    print(f"\n=== Phase 7: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd = min(rem * 0.7, 200)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_desc")

    # === Phase 8: Final fine-tuning ===
    print(f"\n=== Phase 8: Final fine-tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine mirror descent
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = mirror_descent(best_seq, time_limit=min(rem, 60), top_k=60, lr0=0.005, decay=0.999999)
            try_seq(a_opt, "final_mirror")

        # Short LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = active_set_lp_fw(best_seq, time_limit=min(rem, 40), top_k=25)
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for p in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng_master.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = mirror_descent(f_pert, time_limit=min(rem, 25), top_k=50, lr0=0.01)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
