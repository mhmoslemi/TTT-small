def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def hann_window_init(n):
    """Generate a Hann-window shaped initialization."""
    j = np.arange(n)

    if s < 1e-15:

        a = np.ones(n) / n
    else:
        a = a / s
    return a


def raise_cosine_init(n, power=2):
    """Generate a raised-cosine-power initialization."""
    j = np.arange(n)
    a = np.power(0.5 * (1 - np.cos(2 * np.pi * j / (n - 1))), power)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    a = a / s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_direction(f, M, n, warm_start=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    if warm_start is not None and len(warm_start) == n:
        # Use warm start by adding it as initial guess (HiGHS supports this via x0)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', x0=warm_start)
        except:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    else:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None


def compute_subgradient(f, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient focusing on top peaks."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * f[k - indices]
    return sg


def lpfw_optimize(f0, time_limit, verbose=False, m_expansion=1.0, use_warm_start=True):
    """Improved LP-based Frank-Wolfe with warm-starting and adaptive step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_m_exp = 1.0
    last_g = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem_time = time_limit - (time.monotonic() - t0)
        if rem_time < 2.0 and it > 10:
            break

        # Solve LP direction
        M_relaxed = M * cur_m_exp
        warm = last_g if (use_warm_start and last_g is not None) else None
        g = solve_lp_direction(f, M_relaxed, n, warm_start=warm)
        
        if g is None:
            cur_m_exp *= 1.15
            if cur_m_exp > 8.0:
                cur_m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        last_g = g

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            cur_m_exp = min(cur_m_exp * 1.05, 4.0)
            continue

        # Line search with better candidates
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        
        # Generate candidate alphas
        candidates = [0.01, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5,
                      fw_alpha, 0.6, 0.7, 0.85, 0.95, 1.0]
        candidates = sorted(set([max(0, min(1, a)) for a in candidates]))

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement
        if best_alpha > 0.005:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            cur_m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            cur_m_exp = min(cur_m_exp * 1.08, 5.0)

        # Escape from stagnation
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 50, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            cur_m_exp = 1.0
            no_improve = 0
            last_g = None

    return best_f, best_obj


def nesterov_subgrad_optimize(f0, time_limit, lr0=0.004, mom0=0.88, top_k=40, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5 and it > 5:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        
        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        y_new = project_simplex(y + v)
        f_new = y_new
        y = f_new + cur_mom * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.98, cur_mom * 1.0002)

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.0005)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """Directly identify and shave peaks in the auto-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    temp = 0.001

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        n_peaks = min(20, len(conv))
        peak_indices = np.argsort(conv)[-n_peaks:][::-1]

        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            if conv[k] < M * 0.9:
                break  # Only work on significant peaks

            # Find the j that contributes most to conv[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            contrib = f[js:je+1] * f[k-je:k-js+1]
            j_idx = js + np.argmax(contrib)
            
            # Try to reduce f[j_idx] by spreading to a less harmful position
            # Find position p that minimizes increase in max conv
            old_val = f[j_idx]
            if old_val < 1e-10:
                continue
            
            # Try moving a fraction of mass to a random position
            for attempt in range(3):
                p = rng.integers(0, n)
                if p == j_idx:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = old_val * frac
                    ft = f.copy()
                    ft[j_idx] -= delta
                    ft[p] += delta
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_obj - 1e-14:
                        f = ft
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    elif val < M + temp * 0.1 and rng.random() < 0.1:
                        # Accept worsening move (annealing)
                        f = ft
                        improved = True
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            temp *= 0.95
            if no_improve > 100:
                break

        # Occasionally do a full re-optimization of the step
        if no_improve > 30 and no_improve % 30 == 0:
            # Try symmetric mirroring
            f_sym = (f + f[::-1]) / 2.0
            f_sym = np.maximum(f_sym, 0.0)
            f_sym = f_sym / f_sym.sum()
            val_sym = fast_convolve(f_sym).max()
            if val_sym < best_obj - 1e-14:
                f = f_sym
                best_obj = val_sym
                best_f = f.copy()
                no_improve = 0

    return best_f, best_obj


def spectral_smooth(f0, time_limit, lam=0.01):
    """Apply spectral smoothing to reduce convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0 and it > 5:
            break

        # In frequency domain, |FFT(a)|^2 determines the auto-convolution.
        # Flatten the spectrum slightly to reduce time-domain peaks.
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        # Gentle flattening: interpolate between current mag and its mean
        target_mag = np.mean(mag)
        new_mag = (1 - lam) * mag + lam * target_mag
        # Maintain phase
        new_A = new_mag * (A / (np.abs(A) + 1e-30))
        f_new = np.fft.irfft(new_A, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        st = f_new.sum()
        if st < 1e-15:
            break
        f_new = f_new / st

        val = fast_convolve(f_new).max()
        if val < best_obj - 1e-15:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            lam *= 0.9  # Reduce smoothing over time
        else:
            lam *= 0.8
            if lam < 0.001:
                break
            f = best_f.copy()

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_iterations=None):
    """Cyclic coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    passes = 0
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if n_iterations and passes * n + j >= n_iterations:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.9999, 3.0 * old_val)

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            mask = np.ones(n, dtype=bool)
            mask[j] = False

            def eval_coord(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                ft[mask] = f[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = eval_coord(opt_x)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                ft[mask] = f[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))

    except Exception:

                                                       top_k=max(15, n_best // 10),
                                                       temp0=0.00002)
                try_seq(a_opt2, "nesterov_fine")

    # === Phase 7: Coordinate descent polish (150s) ===
    print(f"\n=== Phase 7: Coordinate descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_cd = min(rem * 0.7, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_descent")

    # === Phase 8: Final combined polish (80s) ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Quick LP-FW
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_lpfw")

        # Quick peak shaving
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = peak_shaving(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_peak")

        # Tiny perturbations + quick optimize
        rem = deadline - time.monotonic() - 10
        for _ in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 6:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, "perturb_final")

    # Fallback
    if best_seq is None:
        best_seq = hann_window_init(n_parent)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
