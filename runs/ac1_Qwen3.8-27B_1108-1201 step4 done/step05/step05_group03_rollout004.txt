import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_from_conv(conv, n):
    """Compute objective given precomputed convolution and n."""
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def structured_init(n, method="raised_cosine", seed=42):
    """Generate structured initializations."""
    j = np.arange(n)
    center = (n - 1) / 2.0

    if method == "uniform":
        a = np.ones(n) / n
    elif method == "raised_cosine":
        a = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * (j - center) / n), 0)
    elif method == "raised_cosine2":
        a = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * (j - center) / n), 0) ** 2
    elif method == "raised_cosine3":
        a = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * (j - center) / n), 0) ** 3
    elif method == "triangular":
        a = 1.0 - 2.0 * np.abs(j - center) / (n - 1)
        a = np.maximum(a, 0)
    elif method == "gaussian":
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    elif method == "sech":
        a = 1.0 / np.cosh((j - center) / (n / 4.0))
    elif method == "random_smooth":
        rng = np.random.default_rng(seed)
        raw = rng.standard_normal(n)
        # Smooth via convolution with a Gaussian kernel
        kernel_size = max(3, n // 16)
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4 + 1) ** 2)
        a = np.convolve(raw, kernel, mode="same")
        a = np.maximum(a, 0)
    elif method == "fourier_pickup":
        # Approximate inverse FFT of a flat spectrum
        k = np.arange(n)
        a = 1.0 - 0.3 * np.cos(2 * np.pi * k / n) - 0.1 * np.cos(4 * np.pi * k / n)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                g = lp_direction(f, M * 1.03, n)
                if g is None:
                    no_improve += 1
                    if no_improve > 60:
                        rng = np.random.default_rng(it * 137 + 42)
                        f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                        conv = fast_convolve(f)
                        M = conv.max()
                        no_improve = 0
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 31 + 7)

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            ft = f + alpha * (g - f)

                hi = 1.0

            c2 = lo + gr * (hi - lo)

            fc1 = fast_convolve(ft1).max()
            ft2 = np.maximum(f + c2 * (g - f), 0)
            ft2 = ft2 / max(ft2.sum(), 1e-15)
            fc2 = fast_convolve(ft2).max()
            for _ in range(8):
                if hi - lo < 1e-7:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    ft1 = np.maximum(f + c1 * (g - f), 0)
                    ft1 = ft1 / max(ft1.sum(), 1e-15)
                    fc1 = fast_convolve(ft1).max()
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    ft2 = np.maximum(f + c2 * (g - f), 0)
                    ft2 = ft2 / max(ft2.sum(), 1e-15)
                    fc2 = fast_convolve(ft2).max()
            cand = (lo + hi) / 2.0
            ft = np.maximum(f + cand * (g - f), 0)

            if val < best_val:

def coordinate_descent_optimize(f0, time_limit, verbose=False):
    """
    2-coordinate descent: for each pair (i,j), optimize the mass transfer t
    such that a[i]=t*(a[i]+a[j]), a[j]=(1-t)*(a[i]+a[j]).
    Uses golden section search for the optimal t.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve_passes = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False

        # Generate pairs to try: mix of local and random
        pairs = []
        # Local pairs (adjacent and near-adjacent)
        for offset in [1, 2, 3, 5]:
            for i in range(n - offset):
                pairs.append((i, i + offset))
        # Random pairs
        for _ in range(n // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((i, j))

        # Shuffle
        rng.shuffle(pairs)

        for i, j in pairs:
            if time.monotonic() - t0 > time_limit:
                break

            c = f[i] + f[j]
            if c < 1e-12:
                continue

            # Current t
            t_cur = f[i] / c

            # Golden section search over t in [0, 1]
            lo = 0.0
            hi = 1.0
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            # Initial evaluations
            f_trial1 = f.copy()
            f_trial1[i] = c1 * c
            f_trial1[j] = (1 - c1) * c
            f_trial1 = f_trial1 / f_trial1.sum()
            fc1 = fast_convolve(f_trial1).max()

            f_trial2 = f.copy()
            f_trial2[i] = c2 * c
            f_trial2[j] = (1 - c2) * c
            f_trial2 = f_trial2 / f_trial2.sum()
            fc2 = fast_convolve(f_trial2).max()

            # Also check current value
            fc_cur = fast_convolve(f).max()

            for _ in range(12):
                if hi - lo < 1e-6:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    f_trial1 = f.copy()
                    f_trial1[i] = c1 * c
                    f_trial1[j] = (1 - c1) * c
                    f_trial1 = f_trial1 / f_trial1.sum()
                    fc1 = fast_convolve(f_trial1).max()
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    f_trial2 = f.copy()
                    f_trial2[i] = c2 * c
                    f_trial2[j] = (1 - c2) * c
                    f_trial2 = f_trial2 / f_trial2.sum()
                    fc2 = fast_convolve(f_trial2).max()

            t_opt = (lo + hi) / 2.0
            f_trial = f.copy()
            f_trial[i] = t_opt * c
            f_trial[j] = (1 - t_opt) * c
            f_trial = f_trial / f_trial.sum()
            fc_opt = fast_convolve(f_trial).max()

            if fc_opt < best_obj - 1e-13:
                f = f_trial
                best_obj = fc_opt
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def anneal_support(f0, time_limit, T_initial=0.05, decay=0.9995):
    """
    Simulated annealing on the sequence: randomly perturb coordinates
    and accept/reject based on objective change.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    curr_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    T = T_initial

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        T *= decay

        # Pick a random perturbation type
        ptype = rng.random()

        if ptype < 0.5:
            # Move mass between two coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = rng.standard_normal() * 0.01
            f_trial = f.copy()
            f_trial[i] += delta
            f_trial[j] -= delta
            if f_trial[i] < 0 or f_trial[j] < 0:
                continue
        elif ptype < 0.8:
            # Reshape a small block
            start = rng.integers(0, n - 8)
            end = start + rng.integers(4, 10)
            end = min(end, n)
            block = f[start:end].copy()
            bs = block.sum()
            if bs < 1e-12:
                continue
            pert = rng.standard_normal(end - start) * 0.1 * bs
            block_new = np.maximum(block + pert, 0)
            block_ns = block_new.sum()
            if block_ns < 1e-12:
                continue
            block_new *= bs / block_ns
            f_trial = f.copy()
            f_trial[start:end] = block_new
        else:
            # Small random perturbation
            f_trial = f.copy()
            idx = rng.integers(0, n)
            f_trial[idx] *= np.exp(rng.standard_normal() * 0.05)
            if f_trial[idx] < 0:
                continue

        f_trial = np.maximum(f_trial, 0)
        f_trial = f_trial / f_trial.sum()
        new_obj = fast_convolve(f_trial).max()

        delta = new_obj - curr_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            f = f_trial
            curr_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()

    return best_f, best_obj


def multi_block_coordinate(f0, time_limit, block_size=16, iterations_per_block=5, verbose=False):
    """
    Optimize blocks of coordinates by random perturbations within each block,
    keeping the block sum constant.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break

        improved = False
        block_order = rng.permutation(num_blocks)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue

            # Try multiple perturbations
            for trial in range(iterations_per_block):
                scale = 0.02 * block_sum * (1.0 + trial * 0.5)
                pert = rng.standard_normal(end - start) * scale
                trial_block = np.maximum(f[start:end] + pert, 0)
                ts = trial_block.sum()
                if ts < 1e-12:
                    continue
                trial_block = trial_block * (block_sum / ts)

                f_trial = f.copy()
                f_trial[start:end] = trial_block
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()

                if obj < best_obj - 1e-13:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Nesterov-SG, LP-FW, coordinate descent,
    block optimization, and multi-resolution exploration.
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = structured_init(768, "raised_cosine")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Try structured initializations at parent length ===
    print(f"\n=== Phase 1: Structured inits (best={best_score:.8f}) ===")
    methods = ["raised_cosine", "raised_cosine2", "gaussian", "sech", "fourier_pickup"]
    for m in methods:
        rem = deadline - time.monotonic() - 1200
        if rem < 10:
            break
        a_init = structured_init(n_parent, m)
        # Quick Nesterov optimization
        t_alloc = min(rem * 0.3, 60)
        if t_alloc > 5:
            a_opt, _ = nesterov_sg_optimize(a_init, time_limit=t_alloc)
            try_seq(a_opt, f"nesterov_{m}")

    # === Phase 2: Nesterov-SG on parent ===
    print(f"\n=== Phase 2: Nesterov-SG on parent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 1000
    t2 = min(rem, 180)
    if t2 > 20:
        print(f"  Time: {t2:.0f}s, n={n_parent}")
        a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=t2,
                                         T_start=10.0/n_parent, T_end=0.2/n_parent,
                                         lr_start=0.008, lr_end=0.0005, verbose=True)
        try_seq(a_opt, "nesterov_p2")

    # === Phase 3: LP-FW on best ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 800
        t3 = min(rem, 250)
        if t3 > 30:
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = []
    nl = len(base)
    target_lengths.append(nl * 2)
    if nl >= 512:
        target_lengths.append(nl // 2)
    target_lengths.append(1024)
    target_lengths.append(1536)
    # Remove duplicates and sort
    target_lengths = sorted(set(l for l in target_lengths if 64 <= l <= 2048))

    for target_n in target_lengths:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 600
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)

        # Nesterov
        t_alloc = min(rem * 0.3, 100)
        if t_alloc > 10:
            print(f"  Nesterov n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_sg_optimize(f_interp, time_limit=t_alloc,
                                             T_start=10.0/target_n, T_end=0.3/target_n)
            try_seq(a_opt, f"nest_n{target_n}")

        # LP-FW
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_lpfw = min(rem * 0.5, 150)
            if t_lpfw > 20:
                print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s")
                start_point = a_opt if 'a_opt' in dir() and len(a_opt) == target_n else f_interp
                a_opt2, _ = lp_fw_optimize(start_point, time_limit=t_lpfw)
                try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_cd = min(rem * 0.6, 180)
            print(f"  Time: {t_cd:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "coord_desc")

    # === Phase 6: Block coordinate optimization ===
    print(f"\n=== Phase 6: Block Coord (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_blk = min(rem * 0.7, 120)
            bs = max(8, min(32, len(best_seq) // 32))
            print(f"  Time: {t_blk:.0f}s, block_size={bs}")
            a_opt, _ = multi_block_coordinate(best_seq, time_limit=t_blk, block_size=bs)
            try_seq(a_opt, "block_coord")

    # === Phase 7: Annealing on support ===
    print(f"\n=== Phase 7: Annealing (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_ann = min(rem * 0.5, 80)
            print(f"  Time: {t_ann:.0f}s")
            a_opt, _ = anneal_support(best_seq, time_limit=t_ann, T_initial=0.03, decay=0.9998)
            try_seq(a_opt, "anneal")

    # === Phase 8: Diverse deep optimization ===
    print(f"\n=== Phase 8: Deep diverse (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem / max(1, 6 - p), 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                rng = np.random.default_rng(200 + p)
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif p == 2:
                f_init = best_seq[::-1].copy() / best_seq.sum()
            elif p == 3:
                # Blend with structured
                f_struct = structured_init(n_best, "raised_cosine2")
                f_init = project_simplex(0.8 * best_seq + 0.2 * f_struct)
            elif p == 4:
                # Blend with uniform
                f_init = project_simplex(0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best))
            else:
                rng = np.random.default_rng(500 + p)
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)

            # Alternate methods
            if p % 3 == 0:
                a_opt, _ = nesterov_sg_optimize(f_init, time_limit=min(t_alloc, 100),
                                                 T_start=8.0/n_best, T_end=0.3/n_best,
                                                 lr_start=0.006, lr_end=0.0008)
                try_seq(a_opt, f"deep_nest_{p}")
            elif p % 3 == 1:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 120))
                try_seq(a_opt, f"deep_lpfw_{p}")
            else:
                a_opt, _ = coordinate_descent_optimize(f_init, time_limit=min(t_alloc, 100))
                try_seq(a_opt, f"deep_cd_{p}")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # LP-FW final
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 100), verbose=True)
            try_seq(a_opt, "lpfw_final")

        # Coordinate descent final
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "cd_final")

        # Nesterov final with very low temperature
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=min(rem, 40),
                                             T_start=2.0/n_best, T_end=0.1/n_best,
                                             lr_start=0.003, lr_end=0.0003)
            try_seq(a_opt, "nest_final")

        # Tiny perturbations + quick LP-FW
        for trial in range(4):
            rem = deadline - time.monotonic() - 5
            if rem < 8:

            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 20))

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
