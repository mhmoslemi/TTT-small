2. Exact subgradient descent with momentum and adaptive step size

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f):
    """Exact subgradient of max_k conv(f,f)[k]. Returns (gradient, peak, conv)."""
    n = len(f)
    conv = fast_convolve(f)
    peak_val = conv.max()
    tol = 1e-10 * max(peak_val, 1e-15)
    peak_indices = np.where(conv >= peak_val - tol)[0]

    sg = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        kg_vals = k - np.arange(js, je + 1)
        sg[js:je + 1] += f[kg_vals]
    sg *= 2.0 / len(peak_indices)
    return sg, peak_val, conv

        if sg_norm > 1e-15:
            v = momentum * v - lr * sg / sg_norm
            f_new = project_simplex(f + v)
        else:
            f_new = f.copy()

        fnorm = f_new.sum()
        if fnorm < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fnorm

        new_obj = fast_convolve(f_new).max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"      SG it={it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            f = f_new

        lr *= 0.9997
        if no_improve > 200:
            lr = max(lr * 0.5, 0.001)
            v = np.zeros(n)
            no_improve = 0

        if no_improve > 600:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.3
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, verbose=False):
    """Mirror descent (entropic) for keeping mass positive."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, peak_val, conv = compute_subgradient(f)

        # Mirror descent update: a_new[i] proportional to a[i] * exp(-lr * sg[i])
        log_f = np.log(f)
        log_f -= lr * sg
        log_f -= log_f.max()  # numerical stability
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()

        new_obj = fast_convolve(f_new).max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"      MD it={it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            f = f_new

        lr *= 0.9996
        if no_improve > 300:
            lr = max(lr * 0.5, 0.0005)
            no_improve = 0

        if no_improve > 800:
            rng = np.random.default_rng(it * 7919 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.4
            no_improve = 0

    return best_f, best_obj


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            # Fallback: use subgradient direction
            sg, peak_val, _ = compute_subgradient(f)
            g = project_simplex(-sg)
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            f = f_new / sn if sn > 1e-15 else np.ones(n) / n

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 31337 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 50, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj


def generate_ramp(n, p=1.0):
    """Generate a power-law ramp: a[i] = (i+1)^p, normalized."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_exp(n, alpha=0.5):
    """Generate exponential: a[i] = exp(alpha * i / n), normalized."""
    idx = np.arange(n, dtype=np.float64) / n
    a = np.exp(alpha * idx)
    s = a.sum()
    return a / s


def generate_bimodal(n, center_frac=0.7, width=0.15):
    """Generate a skewed Gaussian-like shape."""
    idx = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((idx - center_frac) ** 2) / (2 * width ** 2))
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization exploiting lopsided (ramp-like) structure.
    Uses exact subgradient, mirror descent, LP-FW, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Explore diverse initializations across lengths ===
    print("\n=== Phase 1: Diverse initialization exploration ===")
    lengths = [384, 512, 640, 768, 832, 1024, 1280]
    shapes = {}

    for n in lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break

        # Generate several shapes for this length
        candidates = {}
        for p in [0.4, 0.7, 1.0, 1.3, 1.6]:
            candidates[f"ramp_{p}"] = generate_ramp(n, p)
        for alpha in [0.5, 1.0, 2.0]:
            candidates[f"exp_{alpha}"] = generate_exp(n, alpha)
        candidates["bimodal"] = generate_bimodal(n, 0.7, 0.2)
        candidates["bimodal2"] = generate_bimodal(n, 0.6, 0.15)

        if parent is not None:
            candidates["parent_interp"] = interpolate_sequence(parent, n)

        # Quick evaluation of each shape
        quick_time = min(8.0, rem / max(1, len(lengths)))
        for name, a in candidates.items():
            if time.monotonic() > deadline - 100:
                break
            try_seq(a, f"init_n{n}_{name}")
            shapes[(n, name)] = a

    # Pick top initialization for deep optimization
    if best_seq is None:
        best_seq = generate_ramp(512, 1.0)
        best_score = evaluate_sequence(best_seq.tolist())

    base_n = len(best_seq)
    print(f"\n  After Phase 1: best={best_score:.8f}, n={base_n}")

    # === Phase 2: Deep subgradient optimization on best ===
    print(f"\n=== Phase 2: Deep SGD on best (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t_sgd = min(rem * 0.3, 400)
    if t_sgd > 30:
        print(f"  Time budget: {t_sgd:.0f}s")
        a_opt, _ = sgd_optimize(best_seq, time_limit=t_sgd, lr_init=0.04,
                                momentum=0.93, verbose=True)
        try_seq(a_opt, "sgd_deep")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent on best (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t_md = min(rem * 0.2, 250)
    if t_md > 20 and best_seq is not None:
        print(f"  Time budget: {t_md:.0f}s")
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md,
                                           lr_init=0.015, verbose=True)
        try_seq(a_opt, "mirror_descent")

    # === Phase 4: LP-FW polishing ===
    print(f"\n=== Phase 4: LP-FW polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.25, 300)
        if t_lp > 20:
            print(f"  n={n_best}, Time budget: {t_lp:.0f}s")

            try_seq(a_opt, "lpfw_polish")

    # === Phase 5: Try nearby lengths ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = set()
        for delta in [-128, -64, 64, 128, 256, -256]:

            if 256 <= ln <= 2048:

        candidate_lengths = sorted(candidate_lengths)

        for target_n in candidate_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.25, 150)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)

            rem = deadline - time.monotonic()
            if rem > 30:

                try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 6: SGD with different hyperparameters ===
    print(f"\n=== Phase 6: SGD variant params (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg2 = min(rem * 0.2, 200)
        if t_sg2 > 20:
            a_opt, _ = sgd_optimize(best_seq, time_limit=t_sg2, lr_init=0.02,
                                    momentum=0.95, verbose=False)
            try_seq(a_opt, "sgd_slower")

        rem = deadline - time.monotonic()
        t_sg3 = min(rem * 0.15, 120)
        if t_sg3 > 15:
            a_opt, _ = sgd_optimize(best_seq, time_limit=t_sg3, lr_init=0.08,
                                    momentum=0.90, verbose=False)
            try_seq(a_opt, "sgd_faster")

    # === Phase 7: Perturbation and final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for trial in range(8):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            noise_scale = 0.001 + 0.001 * trial
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 30)
            a_opt, _ = sgd_optimize(f_pert, time_limit=t_alloc * 0.7,
                                    lr_init=0.01, momentum=0.95)

    # Fallback
    if best_seq is None:
        best_seq = generate_ramp(512, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = generate_ramp(512, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
