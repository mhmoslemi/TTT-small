    """

    active_indices: list of peak positions (in [0, 2n-2])
    weights: list of non-negative weights summing to 1
    Returns: subgradient of max(conv) w.r.t. a

        k = active_indices[idx_k]
        w = weights[idx_k]

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, active_idx, weights):
    n = len(a)
    grad = np.zeros(n)
    for i in range(len(active_idx)):
        k = active_idx[i]
        w = weights[i]
        if w < 1e-15:

    return grad

        peak_val = conv.max()

        active = np.where(conv >= threshold)[0]
        
        # Weights: proportional to how close to max

        weights = (act_vals - threshold + peak_val * 0.005)
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-15:

        grad = compute_subgrad(a, active, weights)

        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

            best_a = a_new.copy()
            no_improve = 0

                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(c, 0)  # Ensure non-negative (numerical safety)

def project_simplex(v):
    """Project v onto {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_subgradient(a, conv, peak_frac=0.99):
    """
    Compute subgradient of max(conv) at point a.
    peak_frac: fraction of max to include as "active"
    """
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    
    threshold = peak_frac * max_val
    active = np.where(conv >= threshold)[0]
    
    # Weights: linear in (conv[k] - threshold)
    w = conv[active] - threshold
    w = np.maximum(w, 1e-10)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for i in range(len(active)):
        k = int(active[i])
        wi = w[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            grad[js:je+1] += 2.0 * wi * a[k - np.arange(js, je + 1)]
    
    return grad

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Nesterov subgradient descent with adaptive peak tracking
- Golden-section pairwise coordinate descent for local refinement
- Structured peak-directed perturbations to escape local minima
- Multi-phase cycling with different hyperparameters

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT with numerical safety."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(c, 0.0)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_subgradient(a, conv, peak_frac=0.99):
    """Compute subgradient of max(conv) at point a using active peaks."""
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    threshold = peak_frac * max_val
    active = np.where(conv >= threshold)[0]
    w = conv[active] - threshold
    w = np.maximum(w, 1e-12)
    w = w / w.sum()
    grad = np.zeros(n)
    for i in range(len(active)):
        k = int(active[i])
        wi = w[i]
        if wi < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            grad[js:je + 1] += 2.0 * wi * a[k - np.arange(js, je + 1)]
    return grad

    a = np.maximum(a0.astype(np.float64), 0.0)
