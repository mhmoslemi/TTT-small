import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_grad(a, conv, T):
    """Compute gradient of LSE-smoothed max autocorrelation w.r.t. a."""
    n = len(a)
    m = conv.max()
    p = np.exp((conv - m) / T)
    p /= p.sum()
    return 2.0 * np.correlate(p, a, 'valid')


def fista_lse(a0, time_limit, T_init=0.002, T_min=0.00005,
              lr_init=0.03, verbose=False):
    """FISTA with log-sum-exp smoothing and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init

    a_prev = a.copy()
    t_mom = 1.0

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        momentum_coeff = (t_mom - 1.0) / t_new
        y = a + momentum_coeff * (a - a_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        conv_y = fast_convolve(y)
        grad = compute_softmax_grad(y, conv_y, T)

        a_new = a
        best_trial_val = conv_y.max()
        for trial_lr in [lr, lr * 0.6, lr * 0.3, lr * 0.1]:
            candidate = project_simplex(y - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_trial_val - 1e-16:
                a_new = candidate
                best_trial_val = val
                break

        a_prev = a
        a = a_new
        t_mom = t_new

        obj = best_trial_val
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.15)
        else:
            no_improve += 1
            lr = max(lr * 0.995, 0.001)
            if no_improve > 20:
                T = min(T * 1.08, 0.01)
            if no_improve > 60:
                T = max(T * 0.6, T_min)
            if no_improve > 150:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                a_prev = a.copy()
                t_mom = 1.0
                T = T_init
                lr = lr_init
                no_improve = 0

        if no_improve == 0 and T > T_min:
            T = max(T * 0.9985, T_min)

    return best_a, best_obj


def pairwise_cd(a0, time_limit, verbose=False):
    """
    Greedy pairwise coordinate descent.
    For the current peak, find the best (i,j) pair to transfer weight,
    then apply the optimal transfer amount.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.005:
            break

        conv = fast_convolve(a)
        peak_val = conv.max()
        k_peak = int(np.argmax(conv))

        # Compute potential: pot[m] = a[k_peak - m] (contribution to peak)
        pot = np.zeros(n)
        valid_start = max(0, k_peak - (n - 1))
        valid_end = min(n - 1, k_peak)
        for m in range(valid_start, valid_end + 1):
            pot[m] = a[k_peak - m]

        # Try top few peaks for more robustness
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_sorted = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        max_improvement = 0.0
        best_pair = None
        best_epsilon = 0.0

        for k_p in peak_sorted[:3]:
            pot_p = np.zeros(n)
            vs = max(0, int(k_p) - (n - 1))
            ve = min(n - 1, int(k_p))
            for m in range(vs, ve + 1):
                pot_p[m] = a[int(k_p) - m]

            # Find positions with high and low potential
            high_mask = pot_p > np.median(pot_p[pot_p > 0]) if (pot_p > 0).any() else pot_p > 0
            low_mask = pot_p < np.median(pot_p) if pot_p.size > 0 else pot_p < 0
            if high_mask.sum() < 1 or low_mask.sum() < 1:
                continue
            i_best = int(np.where(high_mask)[0][np.argmax(pot_p[high_mask])])
            j_candidates = np.where(low_mask & (pot_p < pot_p[i_best]))[0]
            if len(j_candidates) == 0:
                continue
            j_best = int(j_candidates[np.argmin(pot_p[j_candidates])])

            if i_best == j_best:
                continue

            # Try a few epsilon values
            max_eps = a[i_best] * 0.9
            for eps_frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                eps = max_eps * eps_frac
                if eps < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[i_best] -= eps
                a_trial[j_best] += eps
                c_trial = fast_convolve(a_trial)
                if c_trial.max() < peak_val - 1e-14:
                    if peak_val - c_trial.max() > max_improvement:
                        max_improvement = peak_val - c_trial.max()
                        best_pair = (i_best, j_best)
                        best_epsilon = eps

        if best_pair is not None:
            i, j = best_pair
            a[i] -= best_epsilon
            a[j] += best_epsilon
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            # Random perturbation to escape
            rng = np.random.default_rng((it * 48271 + 7) % (2**31 - 1))
            noise_scale = 0.001 + 0.002 * min(1.0, (no_improve - 50) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0

    return best_a, best_obj


def random_pairwise_sa(a0, time_limit, T_sa_init=0.0005, verbose=False):
    """
    Simulated annealing with random pairwise transfers.
    Accept moves that decrease the peak, and occasionally accept
    worse moves to escape local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    T_sa = T_sa_init
    t0 = time.monotonic()
    it = 0
    total_accepted = 0
    total_rejected = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.005:
            break

        conv = fast_convolve(a)
        peak_val = conv.max()
        k_peak = int(np.argmax(conv))

        # Pick random pair
        rng = np.random.default_rng((it * 1103515245 + 12345) % (2**31 - 1))
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Determine transfer direction: from higher "potential" to lower
        pot_i = a[k_peak - i] if 0 <= k_peak - i < n else 0.0
        pot_j = a[k_peak - j] if 0 <= k_peak - j < n else 0.0

        if pot_i >= pot_j:
            src, dst = i, j
        else:
            src, dst = j, i

        if a[src] < 1e-12:
            continue

        # Choose epsilon
        max_eps = a[src]
        eps = max_eps * min(1.0, 0.01 + 0.05 * rng.random())

        a_trial = a.copy()
        a_trial[src] -= eps
        a_trial[dst] += eps
        c_trial = fast_convolve(a_trial)
        new_obj = c_trial.max()

        delta = new_obj - current_obj
        accept = False
        if delta < 0:
            accept = True
        elif T_sa > 1e-8 and rng.random() < np.exp(-delta / T_sa):
            accept = True

        if accept:
            a = a_trial
            current_obj = new_obj
            total_accepted += 1
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()

        total_rejected += 1

        # Cool down
        if it % 100 == 0:
            T_sa *= 0.95
            if T_sa < 1e-6:
                # Reheat and reset
                T_sa = T_sa_init * 0.5
                a = project_simplex(best_a + np.random.default_rng(it).standard_normal(n) * 0.002)
                current_obj = fast_convolve(a).max()

    return best_a, best_obj


def spectrum_init(n, rng):
    """
    Generate an initial sequence by designing a flat-ish spectrum
    and projecting to non-negative time domain.
    """
    # Target: flat magnitude spectrum (except DC)
    target_mag = np.ones(n)
    target_mag[0] = np.sqrt(n)  # DC component for sum=1

    # Random phase
    phase = rng.uniform(0, 2 * np.pi, n)
    phase[0] = 0

    # Build spectrum
    A = target_mag * np.exp(1j * phase)

    # IFFT to time domain
    a = np.fft.ifft(A).real

    # Project to non-negative simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []

    # Parent (resampled)
    if parent is not None:
        p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Spectrum-based
    for _ in range(3):
        inits.append(spectrum_init(n, rng))

    # Edge-concentrated
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    return inits


def multi_peak_fista(a0, time_limit, T_init=0.001, top_r=3):
    """
    FISTA variant that focuses on the top-r peaks simultaneously.
    Uses a weighted sum of the top peaks instead of LSE.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = 0.02
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    a_prev = a.copy()
    t_mom = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a + ((t_mom - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)

        # Weight top-r peaks with exponential decay
        sorted_idx = np.argsort(conv_y)[::-1]
        top_indices = sorted_idx[:top_r]
        weights = np.exp(-np.arange(top_r) * 0.5)
        weights /= weights.sum()

        p = np.zeros(len(conv_y))
        for wi, idx in zip(weights, top_indices):
            p[idx] += wi

        # Smooth with temperature
        p_smooth = p * np.exp((conv_y - conv_y.max()) / T)
        ps = p_smooth.sum()
        if ps > 0:
            p_smooth /= ps
        else:
            p_smooth = np.exp((conv_y - conv_y.max()) / T)
            p_smooth /= p_smooth.sum()

        grad = 2.0 * np.correlate(p_smooth, y, 'valid')

        a_new = a
        best_val = conv_y.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(y - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val:
                a_new = candidate
                best_val = val
                break

        a_prev = a
        a = a_new
        t_mom = t_new

        obj = best_val
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            lr = max(lr * 0.995, 0.002)
            if no_improve > 30:
                T *= 1.1
            if no_improve > 100:
                rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                a_prev = a.copy()
                t_mom = 1.0
                T = T_init
                lr = 0.02
                no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    - FISTA with LSE smoothing (accelerated gradient with adaptive temperature)
    - Greedy pairwise coordinate descent (targeted weight transfers)
    - Simulated annealing with random pairwise moves
    - Multi-peak FISTA variant
    - Diverse multi-start (spectrum-based, random, parent)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n = 768  # Target length

    # Resample parent to target length
    parent_n = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
    parent_n = np.maximum(parent_n, 0)
    parent_n = parent_n / parent_n.sum()
    try_seq(parent_n, "parent_resampled")

    # === Phase 1: Diverse FISTA from multiple starts (~400s) ===
    print("\n=== Phase 1: Diverse FISTA Multi-Start ===")
    t_phase = time.monotonic()

    inits = diverse_inits(n, parent_n, rng)
    n_starts = min(len(inits), 5)

    for i in range(n_starts):
        remaining = deadline - time.monotonic() - 1000
        if remaining < 30:
            break
        t_alloc = min(remaining / (n_starts - i), 120)
        label = f"fista_start{i}"
        print(f"  FISTA init {i}, T=0.002, t={t_alloc:.0f}s")
        a_opt, _ = fista_lse(inits[i], time_limit=t_alloc, T_init=0.002)
        try_seq(a_opt, label)

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FISTA on best (~300s) ===
    print("\n=== Phase 2: Deep FISTA Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 800
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  Deep FISTA T=0.001, t={t_alloc:.0f}s")
            a_opt, _ = fista_lse(best_seq, time_limit=t_alloc, T_init=0.001)
            try_seq(a_opt, "deep_fista")

        remaining = deadline - time.monotonic() - 700
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Multi-peak FISTA, t={t_alloc:.0f}s")
            a_opt, _ = multi_peak_fista(best_seq, time_limit=t_alloc, T_init=0.001)
            try_seq(a_opt, "multi_peak")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Pairwise CD (~400s) ===
    print("\n=== Phase 3: Pairwise Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print(f"  Pairwise CD, t={t_alloc:.0f}s")
            a_opt, _ = pairwise_cd(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pairwise_cd")

            # Second pass from the improved result
            remaining = deadline - time.monotonic() - 400
            if remaining > 20:
                t_alloc = min(remaining, 150)
                print(f"  Pairwise CD pass 2, t={t_alloc:.0f}s")
                a_opt2, _ = pairwise_cd(a_opt, time_limit=t_alloc)
                try_seq(a_opt2, "pairwise_cd2")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Simulated Annealing (~300s) ===
    print("\n=== Phase 4: Simulated Annealing ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 250
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  SA T_sa=0.0005, t={t_alloc:.0f}s")
            a_opt, _ = random_pairwise_sa(best_seq, time_limit=t_alloc, T_sa_init=0.0005)
            try_seq(a_opt, "sa_1")

        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            # SA from a perturbed version
            perturbed = project_simplex(best_seq + rng.standard_normal(n) * 0.005)
            t_alloc = min(remaining, 150)
            print(f"  SA from perturbed, t={t_alloc:.0f}s")
            a_opt2, _ = random_pairwise_sa(perturbed, time_limit=t_alloc, T_sa_init=0.001)
            try_seq(a_opt2, "sa_perturbed")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish cycles (~250s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Alternating FISTA + pairwise CD cycles
        for cycle in range(3):
            remaining = deadline - time.monotonic() - 50
            if remaining < 20:
                break

            t_fista = min(remaining * 0.4, 60)
            t_cd = min(remaining * 0.5, 80)

            print(f"  Cycle {cycle+1}: FISTA({t_fista:.0f}s) + CD({t_cd:.0f}s)")

            a_opt, _ = fista_lse(best_seq, time_limit=t_fista, T_init=0.0005)
            try_seq(a_opt, f"polish_fista_{cycle}")

            a_opt2, _ = pairwise_cd(a_opt, time_limit=t_cd)
            try_seq(a_opt2, f"polish_cd_{cycle}")

        # Final tiny perturbations
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 5:
                    break
                noise = rng.standard_normal(n) * 0.0005
                perturbed = project_simplex(best_seq + noise)
                a_opt, _ = pairwise_cd(perturbed, time_limit=min(remaining, 8))
                try_seq(a_opt, "final_tiny")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = parent_n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
