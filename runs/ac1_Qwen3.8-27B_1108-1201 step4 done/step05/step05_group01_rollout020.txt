import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, k_peaks, weights):
    n = len(a)
    sg = np.zeros(n)
    for i, k in enumerate(k_peaks):
        w = weights[i]
        if w < 1e-14:
            continue
        # d conv[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w * a[k - indices]
    return sg

def subgradient_optimize(a0, time_limit, lr0=0.01, mom0=0.9, top_k=30, temp0=0.01, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr0
    cur_temp = temp0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top-k peaks with softmax weights
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = (pv - pv.max()) / cur_temp
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        sg = compute_subgradient(a, conv, peaks, w)
        
        # Momentum update
        v = mom0 * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
        else:

            # Try smaller step
            if no_improve > 20:
                cur_lr *= 0.9
                v *= 0.5

        # Adaptive lr

            v = np.zeros(n)
            cur_lr = lr0
            cur_temp = temp0
            no_improve = 0
            conv = fast_convolve(a)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast projected subgradient with
momentum, adaptive multi-peak softmax weighting, periodic basin-hopping restarts,
LP-Frank-Wolfe polish, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks, weights):
    """Compute weighted subgradient of max-k conv at the given peaks."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w * a[k - indices]
    return sg


def subgradient_optimize(a0, time_limit, lr0=0.02, mom0=0.85, top_k=40,
                         temp0=0.005, verbose=False):
    """Fast projected subgradient with momentum and adaptive multi-peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr0
    cur_temp = temp0
    cur_mom = mom0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top-k peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = (pv - pv.max()) / cur_temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = compute_subgradient(a, peaks, w)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Normalize gradient for stable step size
        sg = sg / sg_norm

        # Momentum update
        v = cur_mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
            # Slowly increase lr when making progress
            cur_lr = min(cur_lr * 1.002, lr0 * 3.0)
        else:
            no_improve += 1
            if no_improve > 10:
                cur_lr *= 0.95
            if no_improve > 50:
                v *= 0.5

        # Adaptive temperature (annealing)
        cur_temp *= 1.003
        if cur_temp > temp0 * 100:
            cur_temp = temp0 * 100

        # Adaptive momentum
        if no_improve > 100:
            cur_mom = min(0.95, cur_mom + 0.001)
        else:
            cur_mom = max(mom0, cur_mom - 0.0005)

        # Basin hopping restart
        if no_improve > 800:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min((no_improve - 800) // 200, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = lr0
            cur_temp = temp0
            cur_mom = mom0
            no_improve = 0
            conv = fast_convolve(a)
            if verbose:
                print(f"    [restart] it={it}, new M={conv.max():.10f}, best={best_obj:.10f}")

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
                no_improve += 10
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        alphas = sorted(set([0.05, 0.1, 0.2, 0.3, 0.5, fw_alpha, 0.7, 0.9, 1.0]))

        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def chirp_init(n, alpha, seed=0):
    """Generate a chirp (quadratic phase) initialization."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    phase = alpha * (j / n) ** 2
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * phase + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def coordinate_descent_polish(f0, time_limit, n_coords=None):
    """Coordinate descent on a subset of most important coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0

    # Identify important coordinates (those with high values)
    if n_coords is None:
        n_coords = min(n, 200)
    importance = np.sort(f)[::-1]
    threshold = importance[min(n_coords - 1, n - 1)] * 0.5
    important = np.where(f >= threshold)[0]
    if len(important) < 10:
        important = np.arange(n)

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        pass_improved = False

        for j in important:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search
            lo = 0.0
            hi = min(0.999, 3.0 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = np.maximum(ft, 0)
                fs = ft.sum()
                if fs < 1e-15:
                    return float('inf')
                ft = ft / fs
                return fast_convolve(ft).max()

            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            cur_val = fast_convolve(f).max()
            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = np.maximum(ft, 0)
                fs = ft.sum()
                if fs > 1e-15:
                    ft = ft / fs
                    f = ft
                    pass_improved = True
                    new_obj = fast_convolve(f).max()
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
