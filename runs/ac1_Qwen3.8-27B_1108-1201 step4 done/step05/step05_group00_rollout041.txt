def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # End of pass: recompute C to prevent drift
        C = fast_convolve(a)
        obj_end = C.max()

        if obj_end < best_obj - 1e-14:
            best_obj = obj_end
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and pass_num % 200 == 0:
            print(f"    CD pass {pass_num}: obj={best_obj:.10f} (elapsed {time.monotonic()-t0:.0f}s)")

        if no_improve > 80:
            # Perturb and restart
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 80) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            C = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle for global improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 6.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 48271 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    FW-LP it={it}: obj={best_obj:.10f} (elapsed {time.monotonic()-t0:.0f}s)")

    return best_f, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        if a.sum() < 1e-15:
            a = best_a.copy()
            break
        a = a / a.sum()
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.08
            lr *= 0.95
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Coordinate descent from parent (fast convergence)
    Phase 2: FW-LP for global structure
    Phase 3: Coordinate descent again (refine after FW-LP)
    Phase 4: Multiple perturbation + CD restarts
    Phase 5: FW-LP from best
    Phase 6: Final CD polish + smoothed gradient
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_default = 768
        parent = np.ones(n_default) / n_default

    n_parent = len(parent)

    # === Phase 1: Coordinate descent from parent (~250s) ===
    print("\n=== Phase 1: CD from parent ===")
    t_start = time.monotonic()
    t_alloc = min(250.0, (deadline - time.monotonic()) * 0.15)
    a_cd, obj_cd = coord_descent(parent, time_limit=t_alloc, rng=rng, verbose=True)
    try_seq(a_cd, f"cd_parent_{n_parent}")
    print(f"  Phase 1 done: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: FW-LP (~350s) ===
    print("\n=== Phase 2: FW-LP ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(350.0, remaining * 0.25)
    if t_alloc > 30:
        a_fw, obj_fw = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_fw, "fw_lp")
    print(f"  Phase 2 done: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: CD after FW-LP (~250s) ===
    print("\n=== Phase 3: CD after FW-LP ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(250.0, remaining * 0.2)
    if t_alloc > 20:
        a_cd2, _ = coord_descent(best_seq, time_limit=t_alloc, rng=rng, verbose=True)
        try_seq(a_cd2, "cd_after_fw")
    print(f"  Phase 3 done: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Multiple perturbations + CD (~350s) ===
    print("\n=== Phase 4: Perturbation + CD ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    perturb_scales = [0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03]
    for scale in perturb_scales:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(40.0, (remaining - 100) / (len(perturb_scales) - perturb_scales.index(scale)))
        if t_alloc < 10:
            break

        # Try 2 perturbations per scale
        for trial in range(2):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_sub = min(30.0, (remaining - 80) / max(1, len(perturb_scales) - perturb_scales.index(scale)))
            if t_sub < 5:
                break
            a_p, _ = coord_descent(f_pert, time_limit=t_sub, rng=rng)
            try_seq(a_p, f"pert_{scale}_{trial}")

    print(f"  Phase 4 done: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: FW-LP from best (~250s) ===
    print("\n=== Phase 5: FW-LP from best ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(250.0, remaining * 0.25)
    if t_alloc > 30:
        a_fw2, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_fw2, "fw_lp2")
        # CD after FW-LP
        remaining = deadline - time.monotonic()
        t_alloc2 = min(80.0, remaining * 0.15)
        if t_alloc2 > 10:
            a_cd3, _ = coord_descent(best_seq, time_limit=t_alloc2, rng=rng)
            try_seq(a_cd3, "cd_after_fw2")
    print(f"  Phase 5 done: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final polish (~remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic()
    # Long CD run
    t_alloc = min(remaining * 0.5, 200.0)
    if t_alloc > 10:
        a_final, _ = coord_descent(best_seq, time_limit=t_alloc, rng=rng, verbose=True)
        try_seq(a_final, "cd_final")

    # Smoothed gradient
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.3, 60.0)
    if t_alloc > 10:
        a_sg, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.02)
        try_seq(a_sg, "smoothed_final")

    # Quick CD
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.5, 40.0)
    if t_alloc > 5:
        a_cdf, _ = coord_descent(best_seq, time_limit=t_alloc, rng=rng)
        try_seq(a_cdf, "cd_very_final")

    print(f"  Phase 6 done: score={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
