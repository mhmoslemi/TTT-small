import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build the constraint matrix for LP: (T_a g)_k = sum_i a[k-i]*g[i]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[hi - lo::-1]
    return A


def solve_lp_oracle(a, M, active_indices=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    if active_indices is None:
        active_indices = range(2 * n - 1)
    active_indices = list(active_indices)
    if len(active_indices) == 0:
        active_indices = list(range(2 * n - 1))

    num_rows = len(active_indices)
    A = np.zeros((num_rows, n))
    for row, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row, lo:hi + 1] = a[hi - lo::-1]

    c = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smooth_grad_descent(a0, time_limit, T=0.001, lr=0.03):
    """Smoothed gradient descent with softmax-surrogate for max."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of smooth max: 2 * sum_k p_k * d(b_k)/d(a_j) = 2 * correlate(p, a)
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Momentum update
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        val = fast_convolve(a_new).max()
        if val < best_obj - 1e-14:
            best_obj = val
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            v = mom * v - lr * 0.5 * grad
            a_new = project_simplex(a + v)
            val = fast_convolve(a_new).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = a_new.copy()
                no_improve = 0

        a = a_new
        lr *= 0.9995
        mom = min(0.97, mom + 0.0002)

        if no_improve > 100:
            T *= 1.2
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.003 + 0.005 * min(1.0, (no_improve - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = 0.03
            T = 0.001
            no_improve = 0

    return best_a, best_obj

            # Try shifting mass from j1 to j2 (or vice versa)
            # The peak contribution from pair (j1, j2) to b[k_peak] is a[j1]*a[k_peak-j1] + a[j2]*a[k_peak-j2]

def generate_diverse_inits(n, parent, seed=42):
    """Generate diverse initializations for exploration."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2))
    inits.append(g2 / g2.sum())

    # Random (Dirichlet with small concentration)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Cosine-based
    c = np.abs(np.cos(np.pi * x / (n - 1))) + 0.1
    inits.append(c / c.sum())

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    # === Phase 4: Coordinate Descent Polish (~250s) ===
    print("\n=== Phase 4: Coordinate Descent Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 10:
            t_alloc = min(remaining, 150)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_descent")

        # Another round of FW after coordinate descent
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Post-coord FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "post_coord_fw")

    print(f"  Phase 4 done. Best: {best_score:.10f} ({time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final Refinement (~200s) ===
    print("\n=== Phase 5: Final Refinement ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 120
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Fine smooth T=1e-4, t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad_descent(best_seq, time_limit=t_alloc, T=0.0001, lr=0.01)
            try_seq(a_opt, "fine_smooth")

        # Final FW
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 10
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 8)
                a_opt, _ = smooth_grad_descent(f_pert, time_limit=t_alloc, T=0.0002, lr=0.005)
                try_seq(a_opt, "fine_perturb")

        # Peak balance one more time
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            t_alloc = min(remaining, 15)
            a_opt, _ = multi_peak_balance(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_balance")

    print(f"  Phase 5 done. Best: {best_score:.10f} ({time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, max_conv={fast_convolve(best_seq).max():.10f}")
    return [float(x) for x in best_seq]
