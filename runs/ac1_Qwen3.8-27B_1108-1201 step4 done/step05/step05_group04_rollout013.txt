def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_gradient(f, w):
    """
    Compute gradient of sum_k w[k] * (conv(f,f))[k] w.r.t. f.
    Uses the identity: grad[m] = 2 * sum_j f[j] * w[j+m]
    which is a Hankel matrix-vector product computed via FFT in O(n log n).
    """
    n = len(f)
    # Need N >= len(f) + len(w) - 1 = n + (2n-1) - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    F = np.fft.fft(f, N)
    W = np.fft.fft(w, N)
    r = np.fft.ifft(F * np.conj(W), N).real[:n]
    return 2.0 * r


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_optimize_fft(f0, time_limit, beta_start=30.0, beta_end=10000.0, 
                     lr_init=None, momentum=0.92, max_iter_no_improve=80):
    """
    LSE-smoothed Nesterov accelerated projected gradient descent with FFT gradient.
    Minimizes max(conv(f,f)) subject to f >= 0, sum(f) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    if lr_init is None:
        lr_init = 3.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    lr_max = 8.0 / n
    lr_min = 0.0005 / n
    mom = momentum
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    y = f.copy()  # Nesterov lookahead

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov lookahead
        y = f + (mom if it > 1 else 0.0) * (f - best_f) * 0 + mom * (f - f)  # standard: y = f + beta*(f - prev_f)
        # Simpler Nesterov: y = f + momentum * (f - prev_f), but we track via v
        # Use heavy ball for stability: grad computed at f
        conv = fast_convolve(f)
        beta_val = beta
        bc = beta_val * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Full gradient via FFT
        grad = fft_gradient(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 200
            break
        grad /= grad_norm

        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Adaptive lr: increase slightly on improvement
            lr = min(lr * 1.002, lr_max)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
            if no_improve >= 15:
                mom = max(0.85, mom - 0.005)
            if no_improve > max_iter_no_improve:
                # Restart from best with perturbation
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, lr_min)
                beta = min(beta * 1.8, beta_end)
                mom = 0.90
                no_improve = 0
                # Small random perturbation

                f = project_simplex(f + rng.standard_normal(n) * 0.002)

def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: A@g <= M where A is the Toepelitz of conv with f."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def raised_cosine_init(n, beta_rc=1.0):
    """Raised cosine window initialization."""
    x = np.arange(n)
    f = 0.5 * (1 + np.cos(2 * np.pi * beta_rc * (x - n/2 + 0.5) / n))
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f

    """Dolph-Chebyshev window initialization."""
    x = np.arange(n) - (n - 1) / 2.0
    # Approximate Dolph-Chebyshev using Chebyshev polynomial
    from numpy.polynomial import chebyshev as C
    # Use a raised cosine as approximation for simplicity
    t = x / ((n - 1) / 2.0 + 1e-10)
    t = np.clip(t, -1, 1)
    # 4th order Chebyshev window approximation
    f = 1.0 - 0.1 * C.chebval(t, [0, 0, 0, 0, 1])
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def kaiser_init(n, alpha=5.0):
    """Kaiser window initialization."""
    from scipy.special import i0
    x = np.arange(n) - (n - 1) / 2.0
    t = x / ((n - 1) / 2.0 + 1e-10)
    f = i0(alpha * np.sqrt(1 - t**2)) / i0(alpha)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def peak_shave(f0, time_limit, n_shave=8):
    """
    Directly reduce the top peaks by solving small sub-problems.
    For each peak location k, find the direction in f that most reduces c[k]
    while keeping other peaks from increasing too much.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    lr = 0.002

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Get top peaks
        k_top = min(n_shave, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        
        # Weight more heavily on the topmost
        weights = np.exp(3.0 * (pv - pv.max()))
        weights /= weights.sum()
        
        w_full = np.zeros(conv_len)
        w_full[peaks] = weights
        
        grad = fft_gradient(f, w_full)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad /= grad_norm
        
        # Backtracking line search
        best_alpha = 0.0
        best_val = best_obj
        alphas = [lr, lr*2, lr*0.5, lr*0.25, lr*4, lr*0.1]
        for a in alphas:
            f_new = project_simplex(f - a * grad)
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = a
        
        if best_alpha > 0:
            f = project_simplex(f - best_alpha * grad)
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_f = f.copy()
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.8
            if lr < 1e-6:
                break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization pipeline:
    Phase 1: Multi-length, multi-init exploration with fast LSE
    Phase 2: Deep LSE + peak targeting on best candidates
    Phase 3: FW-LP for hard-to-reduce peaks
    Phase 4: Final peak shaving and micro-polish
    Phase 5: Try nearby lengths for final squeeze
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    n_parent = 768
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            print(f"Parent: n={n_parent}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        n_parent = 768
        try_sequence(parent, best_tracker)

    # === Phase 1: Multi-length, multi-init quick exploration ===
    print("\n=== Phase 1: Multi-length exploration ===")
    t1_end = deadline - 900

    # Try several lengths and initializations
    lengths_to_try = sorted(set([n_parent, n_parent - 32, n_parent + 32, 
                                  n_parent - 64, n_parent + 64,
                                  512, 640, 896]), reverse=True)
    lengths_to_try = [lc for lc in lengths_to_try if 256 <= lc <= 1200]

    # For each length, try a couple of inits with quick LSE
    inits_per_length = 2
    total_budget_p1 = t1_end - time.monotonic()
    budget_per = max(15, total_budget_p1 / (len(lengths_to_try) * inits_per_length))

    tried = 0
    for nc in lengths_to_try:
        for init_idx in range(inits_per_length):
            t_remaining = t1_end - time.monotonic()
            if t_remaining < budget_per + 5 or tried >= 12:
                break
            t_alloc = min(budget_per, t_remaining * 0.7)
            if t_alloc < 10:
                break
            tried += 1

            if init_idx == 0:
                # Use parent interpolated
                f_init = interpolate_to_length(parent, nc)
            elif init_idx == 1:
                # Raised cosine
                f_init = raised_cosine_init(nc, beta_rc=1.5)
            else:
                f_init = kaiser_init(nc, alpha=4.0)

            tag = f"parent" if init_idx == 0 else "raised_cos"
            print(f"  LSE n={nc} ({tag}), t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize_fft(f_init, time_limit=t_alloc,
                                         beta_start=20.0, beta_end=3000.0,
                                         lr_init=2.0/nc, momentum=0.91)
            try_sequence(f_opt, best_tracker)

        t_remaining = t1_end - time.monotonic()
        if t_remaining < budget_per * 2:
            break

    # === Phase 2: Deep optimization on best found ===
    print(f"\n=== Phase 2: Deep LSE optimization ===")
    t2_end = deadline - 500
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Long LSE run
    t_alloc = min(300, t2_end - time.monotonic() - 100)
    if t_alloc > 30:
        print(f"  Long LSE n={n_cur}, t={t_alloc:.0f}s, beta 20->8000")
        f_opt, _ = lse_optimize_fft(current, time_limit=t_alloc,
                                     beta_start=20.0, beta_end=8000.0,
                                     lr_init=2.5/n_cur, momentum=0.92)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Peak-targeted gradient
    t_alloc = min(200, t2_end - time.monotonic() - 30)
    if t_alloc > 20:
        print(f"  Peak-target n={len(current)}, t={t_alloc:.0f}s, n_peaks=8")
        f_opt, _ = peak_target_grad_fft(current, time_limit=t_alloc,
                                         n_peaks=8, lr_init=0.003, momentum=0.93)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # More LSE with higher beta
    t_alloc = min(200, t2_end - time.monotonic() - 20)
    if t_alloc > 20:
        print(f"  High-beta LSE n={len(current)}, t={t_alloc:.0f}s, beta 100->15000")
        f_opt, _ = lse_optimize_fft(current, time_limit=t_alloc,
                                     beta_start=100.0, beta_end=15000.0,
                                     lr_init=1.5/n_cur, momentum=0.93)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: FW-LP for structural improvements ===
    print(f"\n=== Phase 3: FW-LP ===")
    t3_end = deadline - 300
    current = best_tracker[1] if best_tracker[1] is not None else current
    n_cur = len(current)

    t_alloc = min(200, t3_end - time.monotonic() - 50)
    if t_alloc > 40:
        print(f"  FW-LP n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp(current, time_limit=t_alloc, verbose=True)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Follow-up LSE after FW
    t_alloc = min(150, t3_end - time.monotonic() - 20)
    if t_alloc > 20:
        print(f"  Post-FW LSE n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize_fft(current, time_limit=t_alloc,
                                     beta_start=50.0, beta_end=20000.0,
                                     lr_init=1.0/n_cur, momentum=0.94)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Alternating refinement cycles ===
    print(f"\n=== Phase 4: Alternating refinement ===")
    t4_end = deadline - 150
    cycles = 0
    while time.monotonic() < t4_end and cycles < 10:
        cycles += 1
        current = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t4_end - time.monotonic()
        if remaining < 30:
            break

        if cycles % 3 == 1:
            # LSE
            t_alloc = min(100, remaining * 0.4)
            if t_alloc > 15:
                beta_s = 50.0 + cycles * 30
                beta_e = 5000.0 + cycles * 3000
                print(f"  Cycle {cycles}: LSE beta {beta_s:.0f}->{beta_e:.0f}, t={t_alloc:.0f}s")
                f_opt, _ = lse_optimize_fft(current, time_limit=t_alloc,
                                             beta_start=beta_s, beta_end=beta_e,
                                             lr_init=1.0/len(current), momentum=0.94)
                try_sequence(f_opt, best_tracker)

        elif cycles % 3 == 2:
            # Peak targeting
            t_alloc = min(80, remaining * 0.4)
            if t_alloc > 15:
                npk = max(3, 10 - cycles)
                print(f"  Cycle {cycles}: Peak-target n_peaks={npk}, t={t_alloc:.0f}s")
                f_opt, _ = peak_target_grad_fft(current, time_limit=t_alloc,
                                                 n_peaks=npk, lr_init=0.002, momentum=0.94)
                try_sequence(f_opt, best_tracker)
        else:
            # Peak shaving
            t_alloc = min(80, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: Peak-shave, t={t_alloc:.0f}s")
                f_opt, _ = peak_shave(current, time_limit=t_alloc, n_shave=6)
                try_sequence(f_opt, best_tracker)

    # === Phase 5: Final micro-polish and nearby lengths ===
    print(f"\n=== Phase 5: Final polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Very high beta LSE for final squeeze
    t_alloc = min(120, deadline - time.monotonic() - 60)
    if t_alloc > 20:
        print(f"  Final high-beta LSE n={n_cur}, t={t_alloc:.0f}s, beta 500->20000")
        f_opt, _ = lse_optimize_fft(current, time_limit=t_alloc,
                                     beta_start=500.0, beta_end=20000.0,
                                     lr_init=0.5/n_cur, momentum=0.95,
                                     max_iter_no_improve=60)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Final peak shave
    t_alloc = min(80, deadline - time.monotonic() - 30)
    if t_alloc > 15:
        print(f"  Final peak-shave n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = peak_shave(current, time_limit=t_alloc, n_shave=4)
        try_sequence(f_opt, best_tracker)

    # Try nearby lengths
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [16, -16, 32, -32, 48, -48]:
            n_try = n_best + off
            if n_try < 256 or n_try > 1200:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 25:
                break
            t_alloc = min(50, t_remaining * 0.5)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Nearby n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize_fft(f_init, time_limit=t_alloc,
                                         beta_start=50.0, beta_end=10000.0,
                                         lr_init=1.5/n_try, momentum=0.93)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
