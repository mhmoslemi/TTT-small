Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive penalty descent with

Key differences from previous: no LP oracle, uses smooth penalty surrogates,
FFT-based gradient computation, Nesterov momentum, and adaptive peak focusing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_grad(a, w):
    """
    Compute gradient of sum_k w_k * (a*a)_k with respect to a.
    Uses FFT for O(n log n) efficiency.
    """
    n = len(a)
    fa = a[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len).bit_length()
    W = np.fft.rfft(w, fft_size)
    FA = np.fft.rfft(fa, fft_size)
    conv = np.fft.irfft(W * FA, fft_size)
    return 2.0 * conv[n - 1: n - 1 + n]

        a = a / s
    n = len(a)

    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)

    best_obj = fast_convolve(a).max()

    no_improve = 0

        improved_this_iter = False

        for k in top_indices:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if conv[k] < best_obj * 0.95:
                break

            # Variables that affect peak k
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            affected = np.arange(js, je + 1)

            if len(affected) == 0:
                continue

            # Try to reduce contribution of each affected variable
            # For peak k: (a*a)_k = sum_i a_i * a_{k-i}
            # Reducing a_i reduces the term a_i * a_{k-i}
            # We want to shift mass from high-contributing vars to low-contributing

            # Compute contribution of each affected var to this peak
            contributions = np.zeros(len(affected))
            for idx, j in enumerate(affected):
                other = int(k) - j
                if 0 <= other < n:
                    contributions[idx] = a[j] * a[other]

            # Sort by contribution (highest first)
            order = np.argsort(contributions)[::-1]

            # Try shifting mass from high to low contributors
            if len(order) >= 2:
                src = affected[order[0]]
                dst = affected[order[-1]]
                if src == dst:
                    continue

                # Binary search for optimal shift
                best_shift_val = 0.0
                best_shift_amt = 0.0
                src_val = a[src]
                dst_val = a[dst]

                for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    cand = a.copy()
                    cand[src] -= amt
                    cand[dst] += amt
                    if cand[src] < 0:
                        cand[src] = 0
                        cand[dst] += cand[src]
                    c = fast_convolve(cand)
                    if c.max() < best_shift_val - 1e-14:
                        best_shift_val = c.max()
                        best_shift_amt = amt

                if best_shift_amt > 0:
                    a[src] -= best_shift_amt
                    a[dst] += best_shift_amt
                    a = project_simplex(a)
                    new_obj = fast_convolve(a).max()
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved_this_iter = True

        if improved_this_iter:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng((it * 31415 + 7) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 100) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            no_improve = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    peak_target it={it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def lp_guided_descent(a0, time_limit, verbose=False):
    """
    FW-lite: uses LP oracle but with warm-starting and reduced constraint set.
    Only solves LP on the top-20% most binding constraints for speed.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Use only top constraints for speed
        conv = fast_convolve(f)
        M = conv.max()
        # Keep top 40% of constraints
        n_constraints = len(conv)
        n_keep = max(n, int(0.4 * n_constraints))
        top_thresh = np.partition(conv, n_constraints - n_keep)[n_constraints - n_keep]
        active_mask = conv >= top_thresh
        active_indices = np.where(active_mask)[0]

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active, g >= 0
        c = -np.ones(n)
        A_ub_rows = []
        b_ub_vals = []
        for k in active_indices:
            row = np.zeros(n)
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            for j in range(js, je + 1):
                row[j] = f[k - j]
            A_ub_rows.append(row)
            b_ub_vals.append(M)

        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(0.1, remaining * 0.05)})
            if res.success:
                g = np.maximum(res.x, 0.0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Line search
                    best_alpha = 0.0
                    best_val = M
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = (1 - alpha) * f + alpha * g
                        ft = np.maximum(ft, 0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha

                    if best_alpha > 0:
                        f = (1 - best_alpha) * f + best_alpha * g
                        f = np.maximum(f, 0)
                        f = f / f.sum()
                        M = fast_convolve(f).max()
                        if M < best_obj - 1e-14:
                            best_obj = M
                            best_f = f.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def coordinate_polish(a0, time_limit):
    """
    Cyclic coordinate descent: for each a_i, find the optimal value
    that minimizes the peak of the convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            # Fix all variables except a_i
            # (a*a)_k = sum_j a_j * a_{k-j}
            # Terms involving a_i: a_i * a_{k-i} for each k
            # = a_i * (sum over valid k of a_{k-i} * w_k) where w_k = 1 for the max
            # Simplification: just try a few values of a_i

            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Try shifting mass from a_i to other positions
            # Find where to send the mass: to positions that minimize peak increase
            others = np.arange(n)
            others = others[others != i]

            if len(others) == 0:
                continue

            # Try a few fractions to remove
            for frac in [0.1, 0.3, 0.5]:
                amt = old_val * frac
                if amt < 1e-12:
                    continue

                # Find best destination
                best_dest = -1
                best_new_obj = best_obj

                # Sample a few destinations
                if len(others) > 20:
                    sample = np.random.default_rng(i + epoch * n).choice(others, 20, replace=False)
                else:
                    sample = others

                for dest in sample:
                    cand = a.copy()
                    cand[i] -= amt
                    cand[dest] += amt
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    cand = cand / cs
                    c = fast_convolve(cand)
                    if c.max() < best_new_obj:
                        best_new_obj = c.max()
                        best_dest = dest
                        best_cand = cand.copy()

                if best_dest >= 0 and best_new_obj < best_obj * (1 - 1e-7):
                    a = best_cand
                    best_obj = best_new_obj
                    best_a = a.copy()
                    break  # Move to next coordinate

    return best_a, best_obj


def game_iteration(a0, time_limit, T_init=0.005, verbose=False):
    """
    Primal-dual game iteration:
    - Dual: p = softmax(conv(a)/T) concentrates on peaks
    - Primal: minimize a^T M(p) a over simplex (approximate QP)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        # Dual update: softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Primal update: gradient of a^T M(p) a = 2 * M(p) a = 2 * correlate(p, a)
        grad = convolve_grad(a, p)

        # Projected gradient step with backtracking
        lr = 1.0 / (np.max(np.abs(grad)) + 1e-10)
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            cand = project_simplex(a - trial_lr * grad)
            val = fast_convolve(cand).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = cand

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature
        if no_improve > 50:
            T *= 1.05
        if no_improve > 200:
            T *= 0.7
        if no_improve > 500:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    game_it it={it}, obj={best_obj:.8f}, T={T:.6f}, {elapsed:.1f}s")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return new_a / s


def generate_initializations(n, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    sig = 0.2 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sig) ** 2)
    inits.append(g / g.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / max(n - 1, 1) - 0.5)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())

    # Inverse cosine (Wiener filter inspired)
    ic = 1.0 / (0.5 - 0.5 * np.cos(2 * np.pi * x / n) + 0.5)
    inits.append(ic / ic.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick exploration across n values and initializations
    Phase 2: Deep penalty descent + game iteration on best
    Phase 3: LP-guided refinement
    Phase 4: Peak targeting + coordinate polish
    Phase 5: Final micro-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-start_time:.0f}s)")
        return score

    start_time = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Quick Exploration (~400s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    # Candidate n values
    candidate_ns = sorted(set([n_parent, n_parent // 2, n_parent * 2,
                               512, 640, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    # Quick penalty descent on a few configurations
    for n_try in candidate_ns[:6]:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 30:
            break

        if n_try == n_parent:
            a_start = parent.copy()
        else:
            a_start = resample_sequence(parent, n_try)

        t_alloc = min(remaining, 50)
        print(f"  Quick penalty n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = penalty_descent(a_start, time_limit=t_alloc,
                                   beta_init=10.0, lr_init=0.15)
        try_seq(a_opt, f"quick_pen_n{n_try}")

        # Quick game iteration
        remaining = deadline - time.monotonic() - 1350
        if remaining > 20:
            a_opt2, _ = game_iteration(a_opt, time_limit=min(remaining, 25), T_init=0.01)
            try_seq(a_opt2, f"quick_game_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep Penalty + Game Descent (~700s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Deep penalty descent with multiple temperature settings
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print(f"  Deep penalty T=5, t={t_alloc:.0f}s")
            a_opt, _ = penalty_descent(best_seq, time_limit=t_alloc,
                                       beta_init=5.0, lr_init=0.08)
            try_seq(a_opt, "deep_pen5")

        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Deep penalty T=20, t={t_alloc:.0f}s")
            a_opt, _ = penalty_descent(best_seq, time_limit=t_alloc,
                                       beta_init=20.0, lr_init=0.05)
            try_seq(a_opt, "deep_pen20")

        # Game iteration with different temperatures
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Game T=0.002, t={t_alloc:.0f}s")
            a_opt, _ = game_iteration(best_seq, time_limit=t_alloc, T_init=0.002)
            try_seq(a_opt, "game_T0.002")

        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Game T=0.01, t={t_alloc:.0f}s")
            a_opt, _ = game_iteration(best_seq, time_limit=t_alloc, T_init=0.01)
            try_seq(a_opt, "game_T0.01")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: LP-Guided Refinement (~400s) ===
    print("\n=== Phase 3: LP-Guided ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  LP-guided n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lp_guided_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "lp_guided")

        # Perturbed LP
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            for scale in [0.005, 0.015]:
                remaining = deadline - time.monotonic() - 150
                if remaining < 15:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 60)
                print(f"  LP-guided pert={scale}, t={t_alloc:.0f}s")
                a_opt, _ = lp_guided_descent(f_pert, time_limit=t_alloc)
                try_seq(a_opt, f"lp_pert_{scale}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak Targeting + Coord Polish (~200s) ===
    print("\n=== Phase 4: Peak Targeting & Coord ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 120
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Peak targeting t={t_alloc:.0f}s")
            a_opt, _ = peak_targeting_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_target")

        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print(f"  Coord polish t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final Micro-Optimization (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine penalty descent
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Fine penalty t={t_alloc:.0f}s")
            a_opt, _ = penalty_descent(best_seq, time_limit=t_alloc,
                                       beta_init=50.0, lr_init=0.02)
            try_seq(a_opt, "fine_pen")

        # Tiny random perturbations with quick optimization
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 20
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 8)
                a_opt, _ = penalty_descent(f_pert, time_limit=t_alloc,
                                           beta_init=20.0, lr_init=0.03)
                try_seq(a_opt, "micro_perturb")

        # Try a slightly different n
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            for n_try in [n_best + n_best // 8, n_best - n_best // 8]:
                remaining = deadline - time.monotonic() - 10
                if remaining < 5 or n_try < 128:
                    continue
                a_try = resample_sequence(best_seq, n_try)
                t_alloc = min(remaining, 5)
                a_opt, _ = penalty_descent(a_try, time_limit=t_alloc,
                                           beta_init=10.0, lr_init=0.05)
                try_seq(a_opt, f"resample_n{n_try}")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, total_time={time.monotonic()-start_time:.0f}s")
    return [float(x) for x in best_seq]
