def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=20, temperature=0.001):
    """Compute weighted subgradient of max_k conv[k] using softmax over top peaks."""
    n = len(a)
    m = len(conv)
    k_actual = min(top_k, m)
    
    # Find top-k peaks
    if k_actual == m:
        peaks = np.arange(m)
        pv = conv.copy()
    else:
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
    
    # Softmax weighting
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute weighted gradient
    sg = np.zeros(n)

            sg[indices] += 2.0 * w[i] * a[k - indices]

def project_simplex(v):

    n = len(v)

def cosine_init(n, freq, amp, seed=0):

def comb_init(n, period, ratio, seed=0):

def nesterov_subgrad(a0, time_limit, lr0=0.004, mom0=0.85, top_k=25, temp0=0.0005):
    """Nesterov-accelerated projected subgradient with adaptive multi-peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient from top-k peaks with softmax weighting
        conv_y = fast_convolve(y)
        m = len(conv_y)
        k_actual = min(top_k, m)
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]
        pv = conv_y[peaks]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * y[k - indices]

        # Nesterov momentum update
        v = cur_mom * v - cur_lr * sg
        a_new = project_simplex(y + v)

        # Update Nesterov variable
        t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * it * it))
        y = a_new + (t_k - 1.0) / t_k * (a_new - y)
        y = project_simplex(y)

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameters
        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.97, cur_mom + 0.00005)

        if no_improve > 250:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            scale = 0.002 + 0.002 * min(no_improve // 250, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.6, 0.001)
            cur_temp = temp0
            no_improve = 0
            conv = fast_convolve(a)

    return best_a, best_obj

            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * a[k - indices]

        # Pick top coordinates by gradient magnitude
        top_coords = np.argpartition(-np.abs(sg), min(20, n))[:min(20, n)]
        top_coords = top_coords[np.argsort(-np.abs(sg)[top_coords])]

        for j in top_coords:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            # Golden section search on a[j]
            lo = 0.0
            hi = min(0.999, 2.5 * old_val)
            GR = (1.0 + np.sqrt(5.0)) / 2.0
            inv_gr = 1.0 / GR

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_coord(x):
                ft = a.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            cur_val = fast_convolve(a).max()
            if opt_val < cur_val - 1e-13:
                ft = a.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                a = ft
                pass_improved = True
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()

        if pass_improved:
            no_pass = 0
        else:
            no_pass += 1
            if no_pass >= 3:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: multi-resolution LP-FW, Nesterov subgradient,
    coordinate descent, and diverse exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_clipped = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a_clipped.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = cosine_init(384, 1, 0.25)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Generated parent: n={len(parent)}, score={best_score:.6f}")

    # === Phase 0: Quick LP-FW on parent ===
    print("\n=== Phase 0: Quick LP-FW on parent ===")
    rem = deadline - time.monotonic()
    t0_alloc = min(60, rem * 0.05)
    if t0_alloc > 10:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t0_alloc, verbose=True)
        try_seq(a_opt, "lpfw_parent_quick")

    # === Phase 1: Multi-resolution coarse-to-fine ===
    print("\n=== Phase 1: Multi-resolution ===")
    base = best_seq if best_seq is not None else parent

    for target_n in [128, 256, 512]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_init = interpolate_seq(base, target_n)
        # Also try a cosine init at this size
        f_cos = cosine_init(target_n, 2, 0.2)
        # Use the better starting point
        obj_init = fast_convolve(f_init).max()
        obj_cos = fast_convolve(f_cos).max()
        if obj_cos < obj_init:
            f_init = f_cos

        t_alloc = min(rem * 0.15, 80)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
        s = try_seq(a_opt, f"res_n{target_n}")

        if np.isfinite(s) and s < best_score + 0.01:
            base = a_opt

    # === Phase 2: LP-FW at target lengths ===
    print(f"\n=== Phase 2: LP-FW at target lengths (best={best_score:.6f}) ===")
    for target_n in [768, 1024]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if best_seq is not None:
            f_init = interpolate_seq(best_seq, target_n)
        else:
            f_init = cosine_init(target_n, 2, 0.2)

        t_alloc = min(rem * 0.2, 180)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(target_n == 768))
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 3: Diverse initializations ===
    print(f"\n=== Phase 3: Diverse inits (best={best_score:.6f}) ===")
    n_target = len(best_seq) if best_seq is not None else 768

    init_configs = [
        ("cos1", cosine_init(n_target, 1, 0.3)),
        ("cos2", cosine_init(n_target, 2, 0.25)),
        ("cos3", cosine_init(n_target, 3, 0.15)),
        ("cos4", cosine_init(n_target, 4, 0.12)),
        ("comb4", comb_init(n_target, 4, 0.5)),
        ("comb6", comb_init(n_target, 6, 0.4)),
        ("msin1", multi_sin_init(n_target, seed=1)),
        ("msin2", multi_sin_init(n_target, seed=2)),
    ]

    for label, f_init in init_configs:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        s = try_seq(f_init, f"init_{label}")
        if np.isfinite(s) and s < best_score + 0.05:
            t_alloc = min(rem * 0.12, 60)
            if t_alloc > 8:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
                try_seq(a_opt, f"lpfw_{label}")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 30:
            t_ns = min(rem * 0.3, 180)
            print(f"  Nesterov t={t_ns:.0f}s, n={n_best}")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_ns,
                                         lr0=0.006, mom0=0.85,
                                         top_k=max(20, n_best // 8),
                                         temp0=0.0004)
            try_seq(a_opt, "nesterov1")

            # Second pass
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt, _ = nesterov_subgrad(a_opt, time_limit=min(rem * 0.3, 100),
                                             lr0=0.002, mom0=0.92,
                                             top_k=max(15, n_best // 10),
                                             temp0=0.0001)
                try_seq(a_opt, "nesterov2")

    # === Phase 5: Short LP-FW polish ===
    print(f"\n=== Phase 5: LP-FW polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lp = min(rem * 0.25, 120)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, verbose=True)
            try_seq(a_opt, "lpfw_polish")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coord descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.4, 120)
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_desc")

    # === Phase 7: Final multi-start ===
    print(f"\n=== Phase 7: Final multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 60)

            if p == 0:
                f_init = best_seq + rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(f_init)
            elif p == 1:
                f_init = 0.8 * best_seq + 0.2 * cosine_init(n_best, 3, 0.2)
                f_init = project_simplex(f_init)
            elif p == 2:
                f_init = 0.9 * best_seq + 0.1 * best_seq[::-1]
                f_init = project_simplex(f_init)
            else:
                f_init = best_seq + rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(f_init)

            # Quick Nesterov from perturbation
            a_opt, _ = nesterov_subgrad(f_init, time_limit=t_alloc,
                                         lr0=0.003, mom0=0.9,
                                         top_k=max(15, n_best // 10),
                                         temp0=0.0002)
            try_seq(a_opt, f"restart_{p}")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(384) / 384

    # Ensure valid output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
