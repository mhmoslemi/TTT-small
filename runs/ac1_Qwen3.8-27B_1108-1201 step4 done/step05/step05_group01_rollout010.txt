"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with log-sum-exp
smoothing of the max convolution, FFT-based gradient computation, and
multi-initialization exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradients(f, T):
    """Compute gradient of logsumexp(T * conv(f)) w.r.t. f using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    Tconv = T * conv
    Tconv -= Tconv.max()
    w = np.exp(Tconv)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # grad[m] = 2 * sum_{k=0}^{n-1} f[k] * w[m+k]
    # = 2 * (f_rev * w)[m+n-1] where f_rev = f[::-1]
    a_rev = f[::-1]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv_result = np.fft.irfft(A * W, N)
    c = conv_result[n-1:3*n-2]
    return 2.0 * c, conv


def adam_optimize(f0, time_limit, T_start=10.0, T_end=5000.0, 
                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  T_ramp=5000, verbose=False):
    """Adam optimizer for log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    check_interval = max(1, T_ramp // 10)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive temperature
        if t < T_ramp:
            T = T_start * (T_end / T_start) ** (t / T_ramp)
        else:
            T = T_end
        
        # Compute gradient
        grad, conv = compute_gradients(f, T)
        
        # Proper objective (actual max, not smoothed)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 10000 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}, T={T:.1f}")
        else:
            no_improve += 1
        
        # Check actual score periodically
        if t % check_interval == 0 and t > 1000:
            if no_improve > check_interval * 3:
                # Restart with higher T
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                T_ramp = max(500, T_ramp // 2)
                no_improve = 0
                continue
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate (decay over time)
        lr_t = lr / (1.0 + t * 1e-7)
        
        f_new = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)
    
    return best_f, best_obj


def u_shaped_init(n, alpha=1.0):
    """Generate U-shaped initialization."""
    j = np.arange(n)
    x = 2.0 * j / (n - 1) - 1.0
    a = 1.0 + alpha * x ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def v_shaped_init(n, alpha=1.0):
    """Generate V-shaped initialization."""
    j = np.arange(n)
    x = 2.0 * j / (n - 1) - 1.0
    a = 1.0 + alpha * np.abs(x)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def multi_peak_init(n, num_peaks, spread=0.1):
    """Generate multi-peak initialization."""
    j = np.arange(n)
    a = np.zeros(n)
    for k in range(num_peaks):
        center = k * n / num_peaks
        a += np.exp(-0.5 * ((j - center) / (n * spread)) ** 2)
    a += 0.01  # Small floor to avoid zeros
    a /= a.sum()
    return a


def cosine_init(n, freq, amp=0.5):
    """Generate cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lp_fw_optimize(f0, time_limit, verbose=False):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Build constraint matrix
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum
            
            if np.max(np.abs(g - f)) < 1e-11:
                no_improve += 1
                if no_improve > 60:
                    break
                m_exp = min(m_exp * 1.04, 3.0)
                continue
            
            # Line search
            best_alpha = 0.0
            best_val = M
            alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
            for alpha in alphas:
                ft = f + alpha * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            
            if best_alpha > 1e-10:
                f_new = f + best_alpha * (g - f)
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new / f_new.sum()
                conv = fast_convolve(f_new)
                M = conv.max()
                m_exp = 1.0
                f = f_new
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                    no_improve = 0
                    if verbose and it % 5 == 0:
                        print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.06, 4.0)
        else:
            m_exp *= 1.1
            if m_exp > 5.0:
                break
        
        if no_improve > 100:
            break

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with log-sum-exp
smoothing, FFT-based gradient computation, multi-initialization exploration,
adaptive temperature scheduling, and LP-Frank-Wolfe final refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradients(f, T):
    """Compute gradient of logsumexp(T * conv(f)) w.r.t. f using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    Tconv = T * conv
    Tconv -= Tconv.max()
    w = np.exp(Tconv)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = f[::-1]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv_result = np.fft.irfft(A * W, N)
    c = conv_result[n - 1:2 * n - 1]
    return 2.0 * c, conv


def adam_optimize(f0, time_limit, T_start=5.0, T_end=5000.0,
                  lr=0.05, beta1=0.8, beta2=0.999, eps=1e-8,
                  T_ramp=5000, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    check_interval = max(100, T_ramp // 20)
    last_print = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        if t <= T_ramp:
            T = T_start * (T_end / T_start) ** (float(t) / T_ramp)
        else:
            T = min(T_end * (1.0 + 0.001 * (t - T_ramp)), T_end * 100.0)

        grad, conv = compute_gradients(f, T)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t - last_print > 5000:

                last_print = t

            m = np.zeros(n)
            v = np.zeros(n)
            N_new = 1 << (3 * n - 2).bit_length()
            no_improve = 0
            T_ramp = max(200, T_ramp // 3)
            continue

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr / (1.0 + 1e-6 * t)
        f_new = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

    return best_f, best_obj


def u_shaped_init(n, alpha=1.0):
    """Generate U-shaped initialization (high at ends, low in middle)."""
    j = np.arange(n)
    x = 2.0 * j / max(n - 1, 1) - 1.0
    a = 1.0 + alpha * x ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def v_shaped_init(n, alpha=1.0):
    """Generate V-shaped initialization."""
    j = np.arange(n)
    x = 2.0 * j / max(n - 1, 1) - 1.0
    a = 1.0 + alpha * np.abs(x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_init(n, freq, amp=0.5):
    """Generate cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_peak_init(n, num_peaks, spread=0.08):
    """Generate multi-peak Gaussian initialization."""
    j = np.arange(n)
    a = np.zeros(n)
    for k in range(num_peaks):
        center = (k + 0.5) * n / num_peaks
        a += np.exp(-0.5 * ((j - center) / (n * spread)) ** 2)
    a += 0.005
    a /= a.sum()
    return a


def symmetric_init(n, alpha=1.0):
    """Generate symmetric U-shaped initialization centered at both ends and middle."""
    j = np.arange(n)
    x = 2.0 * j / max(n - 1, 1) - 1.0
    a = 1.0 + alpha * (x ** 4 - 0.5 * x ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe for final refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum

            if np.max(np.abs(g - f)) < 1e-11:
                no_improve += 1
                if no_improve > 50:
                    break
                m_exp = min(m_exp * 1.05, 3.0)
                continue

            best_alpha = 0.0
            best_val = M
            alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
            for alpha in alphas:
                ft = f + alpha * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 1e-10:
                f_new = f + best_alpha * (g - f)
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new / f_new.sum()
                conv = fast_convolve(f_new)
                M = conv.max()
                m_exp = 1.0
                f = f_new
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                    no_improve = 0
                    if verbose and it % 3 == 0:
                        print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.06, 4.0)
        else:
            m_exp *= 1.1
            if m_exp > 5.0:
                break

        if no_improve > 80:
            break

    return best_f, best_obj
