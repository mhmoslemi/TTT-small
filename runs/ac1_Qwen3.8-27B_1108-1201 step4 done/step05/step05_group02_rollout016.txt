def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    w: unconstrained parameters
    t: temperature for smooth max approximation

    n = len(a)

    shifted = conv * t

    log_sum = np.log(np.exp(shifted).sum())

def fast_convolve(a):

    n = len(a)

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, conv, t):
    """Compute gradient of (1/t)log(sum exp(t*conv_k)) w.r.t. a.
    
    Returns (objective_value, gradient).
    """

    exp_tc = np.exp(tc_shifted)
    logsum = tc_shifted.max() + np.log(exp_tc.sum())

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Spectral initialization (Gerchberg-Saxton)
2. Smoothed gradient descent with Nesterov acceleration and adaptive temperature
3. Sharp gradient descent with momentum
4. Coordinate descent polish
5. Randomized restarts

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, t=10.0):
    """Compute objective and gradient of smooth max approximation."""
    n = len(a)
    conv = fast_convolve(a)
    tc = t * conv
    tc_shifted = tc - tc.max()
    exp_tc = np.exp(tc_shifted)
    logsum = tc.max() + np.log(exp_tc.sum())
    obj = logsum / t
    weights = exp_tc / exp_tc.sum()
    
    # Gradient via FFT cross-correlation
    fft_size = 1 << (2 * n - 2).bit_length()  # >= 2n-1
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    cc = np.fft.irfft(np.conj(W) * A, fft_size)
    grad = (2.0 / t) * cc[:n]
    
    return obj, grad


def sharp_gradient(a, top_k=20):
    """Compute subgradient using top_k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    peak_vals = conv.max()
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Weight peaks by their values
    pv = conv[peaks]
    min_pv = pv.min()
    max_pv = pv.max()
    if max_pv - min_pv < 1e-18:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = np.exp((pv - max_pv) / (max_pv * 0.1 + 1e-18))
        w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        m = np.arange(js, je + 1)
        grad[js:je+1] += 2.0 * w[i] * a[k - m]
    
    return peak_vals, grad


def nesterov_optimize(a_init, time_limit, t_start=5.0, t_end=500.0,
                       lr=0.01, momentum=0.9, sharp_top_k=20,
                       verbose=False):
    """Nesterov accelerated gradient descent with adaptive temperature."""
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a_init)) / len(a_init)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    
    best_a = a.copy()
    best_conv = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Adaptive temperature
        progress = min(1.0, elapsed / time_limit)
        t = t_start * (t_end / t_start) ** progress
        t = max(t_start, t)
        
        # Nesterov lookahead
        y_new = project_simplex(y + momentum * v)
        
        # Compute gradient at y_new
        conv_y = fast_convolve(y_new)
        obj_y = conv_y.max()
        
        if t > 50:  # Use sharp gradient for high temperature
            _, grad = sharp_gradient(y_new, top_k=sharp_top_k)
        else:
            _, grad = smooth_gradient(y_new, t=t)
        
        # Gradient step
        v_new = momentum * v - lr * grad
        a_new = y_new + v_new

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using accelerated gradient descent
with FFT-based gradient computation, adaptive temperature annealing,
and multi-phase optimization (smooth -> sharp -> coordinate polish).
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, t):
    """Compute smooth approximation of peak and its gradient.
    
    Uses log-sum-exp approximation: f_t(a) = (1/t) log sum_k exp(t * conv_k(a))
    """
    n = len(a)
    conv = fast_convolve(a)
    tc = t * conv
    max_tc = tc.max()
    exp_tc = np.exp(tc - max_tc)
    logsum = max_tc + np.log(exp_tc.sum())
    obj = logsum / t
    weights = exp_tc / exp_tc.sum()
    
    # Gradient: (2/t) * [cross_correlate(weights, a)][:n]
    # cross_correlate(w, a)[j] = sum_m w[j+m] * a[m]
    fft_size = 1 << (2 * n - 2).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    cc = np.fft.irfft(np.conj(W) * A, fft_size)
    grad = (2.0 / t) * cc[:n]
    
    return obj, grad


def sharp_obj_grad(a, top_k=30):
    """Compute exact peak and subgradient from top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    peak = conv.max()
    
    peaks = np.argpartition(conv, -min(top_k, len(conv)))[-min(top_k, len(conv)):]
    pv = conv[peaks]
    
    # Exponential weighting based on proximity to peak
    thresh = peak * 0.999
    mask = pv >= thresh
    if mask.sum() == 0:
        mask[np.argmax(pv)] = True
    w = np.zeros(len(peaks))
    w[mask] = 1.0 / mask.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        m_idx = np.arange(js, je + 1)
        grad[js:je + 1] += 2.0 * w[i] * a[k - m_idx]
    
    return peak, grad

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
