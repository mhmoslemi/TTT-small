import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    no_improve = 0
    it = 0

        f_look = project_simplex(f - v)
        conv_look = fast_convolve(f_look)

        conv_new = fast_convolve(f_new)

            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slowly increase learning rate
            lr = min(lr * 1.002, 10.0 / n)
        else:

                v = np.zeros(n)

                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        # Increase beta (more aggressive focusing on peaks)
        beta = min(beta * 1.005, beta_end)
    
    return best_f, best_obj
