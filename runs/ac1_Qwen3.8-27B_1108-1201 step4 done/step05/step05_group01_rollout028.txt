import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_k(a, k):
    """Compute d conv[k]/d a at point a. Returns gradient vector of length n."""
    n = len(a)
    g = np.zeros(n)
    m_lo = max(0, k - (n - 1))
    m_hi = min(n - 1, k)
    if m_lo <= m_hi:
        indices = np.arange(m_lo, m_hi + 1)
        g[indices] = 2.0 * a[k - indices]
    return g


def solve_sqp_lp(a, critical_set, conv_values):
    """Solve the linearized SQP subproblem.
    
    LP: min t s.t. sum(a')=1, a'>=0, 2*(h_k·a') - t <= conv_k(a*) for k in CS

    c[n] = 1.0

    A_ub_list = []
    b_ub_list = []

        row[n] = -1.0

        b_ub_list.append(conv_values[k])

    b_ub = np.array(b_ub_list) if b_ub_list else np.zeros(0)

    A_eq[0, :n] = 1.0

        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.02:
            break

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * a[k - indices]

        v = cur_mom * v - cur_lr * sg
        a_new = project_simplex(y + v)
        y = project_simplex(a_new + cur_mom * (a_new - a))
        a = a_new

        conv = fast_convolve(a)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.97, cur_mom * 1.0002)

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.001)
            no_improve = 0
            conv = fast_convolve(a)

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_passes_max=20):
    """Coordinate descent with golden section search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0
    no_pass_improve = 0

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit and passes < n_passes_max:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-15:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.999, 2.0 * old_val + 0.01)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_coord(x):
                ft = a.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(a).max()

            if opt_val < cur_val - 1e-14:
                ft = a.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                a = ft
                pass_improved = True
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_a, best_obj


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.1)  # Ensure strictly positive
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def multi_frequency_init(n, freqs, amps, seed=0):
    """Multi-frequency cosine initialization."""
    j = np.arange(n)
    a = np.ones(n)
    for i, (f, amp) in enumerate(zip(freqs, amps)):
        a += amp * np.cos(2.0 * np.pi * f * j / n + seed * 0.1 * i)
    a = np.maximum(a, 0.05)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def gaussian_window_init(n, sigma_frac=0.4):
    """Gaussian window initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * center
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_and_record(a, best_score_ref, label=""):
    """Try a sequence and return (is_better, score)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return False, float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score):
        return score < best_score_ref[0], score
    return False, float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Generate diverse initializations at multiple lengths
    2. Linearized SQP optimization (main workhorse)
    3. Subgradient for global exploration
    4. Coordinate descent polishing
    5. Multi-start refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = cosine_modulated_init(768, 1, 0.3)

    n_parent = len(parent)

    # === Phase 1: Quick SQP on parent ===
    print("=== Phase 1: SQP on parent ===")
    t1 = min(90, deadline - time.monotonic() - 900)
    if t1 > 20:
        a_opt, _ = sqp_optimize(parent, time_limit=t1, max_critical=20, verbose=True)
        update_best(a_opt, "sqp_parent")

    # === Phase 2: Diverse initializations at multiple lengths ===
    print(f"\n=== Phase 2: Diverse inits (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    init_configs = [
        # (n, type, params)
        (768, 'cos', (1, 0.3)),
        (768, 'cos', (2, 0.25)),
        (768, 'cos', (3, 0.2)),
        (768, 'cos', (1, 0.5)),
        (768, 'multi', ([1, 2, 3], [0.2, 0.15, 0.1])),
        (768, 'gauss', (0.3,)),
        (768, 'gauss', (0.5,)),
        (960, 'cos', (1, 0.3)),
        (960, 'cos', (2, 0.2)),
        (960, 'multi', ([1, 3], [0.25, 0.1])),
        (1024, 'cos', (1, 0.3)),
        (1024, 'cos', (2, 0.2)),
        (1024, 'multi', ([1, 2, 4], [0.2, 0.15, 0.08])),
        (1280, 'cos', (1, 0.25)),
        (1280, 'multi', ([1, 2], [0.2, 0.1])),
    ]

    for cfg in init_configs:
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        n_init, itype, params = cfg

        if itype == 'cos':
            a_init = cosine_modulated_init(n_init, params[0], params[1])
        elif itype == 'multi':
            a_init = multi_frequency_init(n_init, params[0], params[1])
        else:
            a_init = gaussian_window_init(n_init, params[0])

        s = float(evaluate_sequence(a_init.tolist()))
        if np.isfinite(s):
            print(f"  Init n={n_init} {itype}: {s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = a_init.copy()

        # If promising, run SQP on it
        if np.isfinite(s) and s < best_score + 0.05:
            t_alloc = min(rem * 0.2, 60)
            if t_alloc > 10:
                a_opt, _ = sqp_optimize(a_init, time_limit=t_alloc, max_critical=20)
                update_best(a_opt, f"sqp_{itype}_n{n_init}")

    # === Phase 3: Interpolate best to longer lengths ===
    print(f"\n=== Phase 3: Longer lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [1024, 1280, 1536]:
            if target_n <= len(best_seq):
                continue
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.35, 90)
            print(f"  SQP n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = sqp_optimize(f_interp, time_limit=t_alloc, max_critical=25)
            update_best(a_opt, f"sqp_interp_n{target_n}")

    # === Phase 4: Subgradient on best (for global exploration) ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_ns = min(rem * 0.4, 120)
            print(f"  Subgrad t={t_ns:.0f}s, n={n_best}")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_ns,
                                             lr0=0.006, mom0=0.85,
                                             top_k=max(15, n_best // 8),
                                             temp0=0.00008)
            update_best(a_opt, "subgrad")

            # Second pass with different params
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                a_opt, _ = subgradient_optimize(a_opt, time_limit=min(rem, 80),
                                                 lr0=0.002, mom0=0.9,
                                                 top_k=max(10, n_best // 10),
                                                 temp0=0.00003)
                update_best(a_opt, "subgrad_fine")

    # === Phase 5: Extended SQP on best (main optimization) ===
    print(f"\n=== Phase 5: Extended SQP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_sq = min(rem * 0.5, 200)
            print(f"  Extended SQP t={t_sq:.0f}s, n={n_best}")
            a_opt, _ = sqp_optimize(best_seq, time_limit=t_sq, max_critical=30, verbose=True)
            update_best(a_opt, "ext_sqp")

    # === Phase 6: Coordinate descent polishing ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            # For large n, do CD on a subset or with fewer passes
            t_cd = min(rem * 0.5, 120)
            max_passes = max(3, min(10, int(t_cd / max(n_best, 1) * 10)))
            print(f"  CD t={t_cd:.0f}s, passes~{max_passes}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, n_passes_max=max_passes)
            update_best(a_opt, "cd")

    # === Phase 7: Multi-start SQP from best ===
    print(f"\n=== Phase 7: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 80)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * cosine_modulated_init(n_best, 2, 0.25)
                f_init = project_simplex(f_init)
            elif p == 3:
                f_init = 0.8 * best_seq + 0.2 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 4:
                f_init = 0.85 * best_seq + 0.15 * gaussian_window_init(n_best, 0.4)
                f_init = project_simplex(f_init)
            else:
                f_init = project_simplex(best_seq * (1 + 0.02 * rng.standard_normal(n_best)))

            a_opt, _ = sqp_optimize(f_init, time_limit=t_alloc, max_critical=25)
            update_best(a_opt, f"multi_{p}")

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine SQP
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = sqp_optimize(best_seq, time_limit=min(rem, 60), max_critical=35)
            update_best(a_opt, "final_sqp")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 15
        for _ in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = sqp_optimize(f_pert, time_limit=min(rem, 15), max_critical=20)
            update_best(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
