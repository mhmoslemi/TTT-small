def project_simplex(v):

    beta = beta_start
    lr = lr_scale / (n * 5.0)
    mu = momentum
    v_mom = np.zeros(n)
    y = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute LSE weights
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient
        grad = compute_weighted_grad(y, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        grad /= gn

        # Nesterov step
        v_mom = mu * v_mom - lr * grad
        f_new = project_simplex(y + v_mom)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update Nesterov lookahead
        y = project_simplex(f_new + mu * (f_new - f))
        f = f_new

        # Adaptive scheduling
        if no_improve > 20:
            lr = max(lr * 0.95, 1.0 / (n * 1000))
        if no_improve > 100:
            mu = max(mu - 0.002, 0.5)
        if no_improve > 300:
            f = best_f.copy()
            y = f.copy()
            v_mom = np.zeros(n)
            lr = lr_scale / (n * 30.0)
            beta = min(beta * 3.0, beta_end)
            mu = 0.8
            no_improve = 0

        beta = min(beta * 1.002, beta_end)

    return best_f, best_obj
