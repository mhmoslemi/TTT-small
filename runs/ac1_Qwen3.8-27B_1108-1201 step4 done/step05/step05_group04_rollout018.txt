import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

        w_sig = w[sig_idx]

    grad = np.zeros(n)
    for i in range(len(sig_idx)):
        k = int(sig_idx[i])
        wk = w_sig[i]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]
            ms2_lo = max(0, k - hi)
            ms2_hi = min(n - 1, k - lo)
            if ms2_lo <= ms2_hi:
                ms2 = np.arange(ms2_lo, ms2_hi + 1)
                grad[ms2] += wk * f[k - ms2]
    return grad * 2.0  # factor of 2 from dc/da


def compute_peak_gradient(f, conv, n_peaks=10):
    """Compute weighted subgradient from top peaks only."""
    n = len(f)
    conv_len = len(conv)
    k_top = min(n_peaks, conv_len)
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]

    # Weight peaks by their value (emphasize higher peaks)
    w = pv / pv.sum()

    grad = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        wk = w[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]
            ms2_lo = max(0, k - hi)
            ms2_hi = min(n - 1, k - lo)
            if ms2_lo <= ms2_hi:
                ms2 = np.arange(ms2_lo, ms2_hi + 1)
                grad[ms2] += wk * f[k - ms2]
    return grad * 2.0


def lse_gradient_descent(f0, time_limit, beta_start=10.0, beta_end=20000.0,
                         max_peaks=250, lr_init=None, verbose=False):
    """LSE smoothed gradient descent on simplex with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 3.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = 0.88
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()

        # Compute gradient
        grad = compute_lse_gradient(f, conv, beta, max_peaks=max_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            no_improve += 100
            break
        grad /= gnorm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr when making progress
            lr = min(lr * 1.004, 8.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
                momentum = max(0.82, momentum - 0.003)
            if no_improve > 50:
                # Restart from best with perturbation
                rng = np.random.default_rng((it * 7919 + 13) % 999983)
                scale = 0.001 + 0.003 * min(1.0, (no_improve - 50) / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = min(beta * 2.0, beta_end)
                momentum = 0.85
                no_improve = 0
                if verbose:
                    print(f"    LSE restart it={it}, beta={beta:.1f}, obj={fast_convolve(f).max():.10f}")

        # Adaptive beta increase
        if beta < beta_end:
            beta *= 1.006
            if no_improve == 0:
                beta = min(beta * 1.003, beta_end)  # Extra boost on improvement

    return best_f, best_obj


def peak_shaving(f0, time_limit, n_peaks=8, lr_init=0.002, momentum=0.92):
    """Direct peak-shaving: target top peaks with weighted subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()

        grad = compute_peak_gradient(f, conv, n_peaks=n_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            no_improve += 100
            break
        grad /= gnorm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 0.01)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.002)
            if no_improve > 200:
                rng = np.random.default_rng((it * 31 + 7) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
                v = np.zeros(n)
                lr = lr_init * 0.3
                momentum = 0.90
                no_improve = 0

        # Gradually increase peaks targeted
        if it % 500 == 0 and n_peaks < 30:
            n_peaks = min(n_peaks + 2, 30)

    return best_f, best_obj


def fw_lp_step(f0, time_limit=120, max_iters=50, verbose=False):
    """Frank-Wolfe with LP oracle, limited iterations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        A_cache = build_lp_matrix(f)
        g = solve_lp_direction(f, M * 1.0, A_cache)
        if g is None:
            g = solve_lp_direction(f, M * 1.2, A_cache)
        if g is None:
            no_improve += 5
            if no_improve > 20:
                break
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 5
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 25:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            if verbose:
                print(f"    FW restart it={it}, obj={M:.10f}")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def symmetric_init(n, freq=3.0, amplitude=0.15):
    """Create a symmetric sinusoidal perturbation of uniform."""
    base = np.ones(n) / n
    x = np.arange(n)
    sym = np.cos(2.0 * np.pi * freq * x / n)
    if n % 2 == 0:
        sym_full = np.concatenate([sym[:n//2], sym[:n//2][::-1]])
    else:
        sym_full = np.concatenate([sym[:n//2], [0.0], sym[:n//2][::-1]])
    f = base + amplitude * sym_full
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def raised_cosine_init(n):
    """Raised cosine (Hanning-like) window."""
    x = np.arange(n)
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def alternating_refinement(f0, time_limit, max_cycles=12):
    """Alternate between LSE, peak-shaving, and occasional FW steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    cycle = 0
    beta = 50.0

    while time.monotonic() - t0 < time_limit and cycle < max_cycles:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        cycle += 1

        if cycle % 3 == 1:
            # LSE step
            t_alloc = min(100, remaining * 0.4)
            t_sub = time.monotonic()
            f_opt, obj = lse_gradient_descent(f, time_limit=t_alloc,
                                               beta_start=beta, beta_end=beta * 10,
                                               max_peaks=200)
            if obj < best_obj:
                best_obj = obj
                best_f = f_opt.copy()
                f = f_opt
                beta = min(beta * 2.0, 50000.0)
                print(f"    Alt cycle {cycle}: LSE obj={obj:.10f}, beta={beta:.0f}")
            else:
                beta = min(beta * 1.5, 50000.0)

        elif cycle % 3 == 2:
            # Peak shaving
            t_alloc = min(80, remaining * 0.35)
            f_opt, obj = peak_shaving(f, time_limit=t_alloc, n_peaks=12,
                                      lr_init=0.001, momentum=0.93)
            if obj < best_obj:
                best_obj = obj
                best_f = f_opt.copy()
                f = f_opt
                print(f"    Alt cycle {cycle}: Peak obj={obj:.10f}")

        else:
            # FW step (expensive, use sparingly)
            t_alloc = min(60, remaining * 0.3)
            f_opt, obj = fw_lp_step(f, time_limit=t_alloc, max_iters=30)
            if obj < best_obj:
                best_obj = obj
                best_f = f_opt.copy()
                f = f_opt
                print(f"    Alt cycle {cycle}: FW obj={obj:.10f}")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Multi-initialization exploration
    Phase 2: Extended LSE gradient descent with aggressive beta
    Phase 3: Alternating refinement (LSE + peak + FW)
    Phase 4: Ultra-high-beta micro-polishing
    Phase 5: Try nearby lengths
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting best: {best_tracker[0]:.8f}")

    # === Phase 1: Multi-initialization exploration ===
    print("\n=== Phase 1: Multi-Init Exploration ===")
    t1_end = deadline - 1100

    initializations = []
    # Parent-based
    initializations.append(('parent', parent.copy()))
    # Symmetric
    initializations.append(('sym3', symmetric_init(n_parent, freq=3, amplitude=0.12)))
    initializations.append(('sym5', symmetric_init(n_parent, freq=5, amplitude=0.08)))
    # Raised cosine
    initializations.append(('cosine', raised_cosine_init(n_parent)))
    # Random perturbation of parent
    rng1 = np.random.default_rng(seed + 100)
    perturbed = project_simplex(parent + rng1.standard_normal(n_parent) * 0.02)
    initializations.append(('random', perturbed))

    for name, f_init in initializations:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 30:
            break
        t_alloc = min(120, t_remaining * 0.25)
        if t_alloc < 15:
            break
        print(f"  LSE init '{name}', t={t_alloc:.0f}s")
        f_opt, obj = lse_gradient_descent(f_init, time_limit=t_alloc,
                                          beta_start=15.0, beta_end=3000.0,
                                          max_peaks=200)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Extended LSE from best ===
    print(f"\n=== Phase 2: Extended LSE (best={best_tracker[0]:.8f}) ===")
    t2_end = deadline - 550
    current = best_tracker[1].copy()
    n_cur = len(current)

    # Progressive beta LSE
    t_remaining = t2_end - time.monotonic()

        t_alloc = min(350, t_remaining)
        print(f"  LSE beta=20->5000, t={t_alloc:.0f}s")
        f_opt, obj = lse_gradient_descent(current, time_limit=t_alloc,
                                          beta_start=20.0, beta_end=5000.0,
                                          max_peaks=250, lr_init=3.0/n_cur)
        try_sequence(f_opt, best_tracker)

    # Continue with higher beta
    current = best_tracker[1].copy()
    t_remaining = t2_end - time.monotonic()
    if t_remaining > 60:
        t_alloc = min(200, t_remaining)
        print(f"  LSE beta=5000->30000, t={t_alloc:.0f}s")
        f_opt, obj = lse_gradient_descent(current, time_limit=t_alloc,
                                          beta_start=5000.0, beta_end=30000.0,
                                          max_peaks=300, lr_init=1.5/n_cur)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Alternating refinement ===
    print(f"\n=== Phase 3: Alternating Refinement (best={best_tracker[0]:.8f}) ===")
    t3_end = deadline - 300
    current = best_tracker[1].copy()
    t_remaining = t3_end - time.monotonic()

    # Peak shaving focused
    current = best_tracker[1].copy()
    t_remaining = t3_end - time.monotonic()

    print(f"\n=== Phase 4: Micro-Polishing (best={best_tracker[0]:.8f}) ===")

        print(f"  Final peak shave (2 peaks), t={t_alloc:.0f}s")
        f_opt, obj = peak_shaving(current, time_limit=t_alloc, n_peaks=2,
                                  lr_init=0.0003, momentum=0.97)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Nearby lengths ===
    print(f"\n=== Phase 5: Nearby Lengths (best={best_tracker[0]:.8f}) ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [0, 64, -64, 128, -128, 32, -32, 192, -192]
        for off in offsets:

            if n_try < 200 or n_try > 2000:

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):

            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
