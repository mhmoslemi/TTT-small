def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, current_M * 2)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

1. Reduced-constraint LP (only top-K active constraints) for much faster iterations

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak_full(f, n, current_M):
    """Solve full LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, current_M * 2)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_min_peak_reduced(f, n, important_indices, current_M):

def solve_lp_max_sum(f, M, n):

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

def subgrad_lse(conv, f, temp):
    """Subgradient of log-sum-exp smooth peak w.r.t. f."""
    n = len(f)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    # Only process significant weights for efficiency
    thresh = w.max() * 0.01
    significant = np.where(w > thresh)[0]
    for k in significant:
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        idx = np.arange(js, je + 1)
        kg = int(k) - idx
        sg[idx] += 2.0 * w[k] * f[kg]
    return sg / temp


def lse_descent(f0, time_limit, temp_init=0.002, temp_mult=1.006,
                lr=0.004, momentum=0.92, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    temp = temp_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        sg = subgrad_lse(conv, f, temp)

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        peak = conv_new.max()

        if peak < best_obj - 1e-14:
            best_obj = peak
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 300 == 0:
                print(f"    LSE iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
        else:
            no_improve += 1
            f = f_new

                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.5, 2.0, 3.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.8:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def make_chirp_init(n):
    """Create a quadratic chirp initialization."""
    x = np.arange(n) / n
    return np.abs(x - 0.5) ** 0.5


def make_windowed_init(n, center_frac=0.3):
    """Create a windowed (concentrated center) initialization."""
    x = np.arange(n) / n
    width = center_frac
    return np.exp(-((x - 0.5) / width) ** 2)


def make_periodic_init(n, freq=3):
    """Create a periodic initialization."""
    x = np.arange(n) / n
    return (1 + np.cos(2 * np.pi * freq * x)) ** 0.5


def make_bimodal_init(n):
    """Create a bimodal initialization."""
    x = np.arange(n) / n
    return np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2)


def make_ramp_init(n):
    """Create a linear ramp initialization."""
    x = np.arange(n) / n
    return x + 0.1 * np.ones(n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: adaptive reduced-constraint LP-FW,
    LSE-smoothed descent, coordinate descent, multi-length exploration,
    diverse structured initializations, and final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_seq = None
    best_score = float('inf')
    # Use a list to avoid closure issues
    state = {'best_seq': None, 'best_score': float('inf')}

    def do_try_seq(a, label):
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < state['best_score']:
            state['best_score'] = score
            state['best_seq'] = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            state['best_seq'] = parent.copy()
            state['best_score'] = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        state['best_seq'] = parent.copy()
        state['best_score'] = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={state['best_score']:.8f}")

    base_n = len(parent)

    # === Phase 0: Diverse structured initializations ===
    print("\n=== Phase 0: Structured initializations ===")
    inits = [
        (make_chirp_init(base_n), 'chirp'),
        (make_windowed_init(base_n, 0.25), 'windowed'),
        (make_periodic_init(base_n, 3), 'periodic3'),
        (make_bimodal_init(base_n), 'bimodal'),
        (make_ramp_init(base_n), 'ramp'),
    ]
    for init_sig, init_name in inits:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        init_sig = np.maximum(init_sig, 0)
        ss = init_sig.sum()
        if ss < 1e-15:
            continue
        init_sig = init_sig / ss
        t_alloc = min(45, rem * 0.08)
        if t_alloc < 15:
            break
        print(f"  Trying {init_name} init, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(init_sig, time_limit=t_alloc, use_reduced=True,
                                   top_k_constraints=40, use_min_peak=True)
        do_try_seq(a_opt, f"init_{init_name}")

    # === Phase 1: Main LP-FW (reduced constraints) on best so far ===
    print(f"\n=== Phase 1: LP-FW reduced (best={state['best_score']:.8f}) ===")
    cur_best = state['best_seq']
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.3)
    if t1 > 40:
        print(f"  n={len(cur_best)}, t={t1:.0f}s, top_k=50")
        a_opt, _ = lp_fw_optimize(cur_best, time_limit=t1, verbose=True,
                                   use_reduced=True, top_k_constraints=50,
                                   use_min_peak=True)
        do_try_seq(a_opt, "lpfw_reduced_main")

    # === Phase 2: LP-FW (full constraints) for accuracy ===
    print(f"\n=== Phase 2: LP-FW full (best={state['best_score']:.8f}) ===")
    cur_best = state['best_seq']
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.18)
    if t2 > 30:
        print(f"  n={len(cur_best)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(cur_best, time_limit=t2, verbose=True,
                                   use_reduced=False, use_min_peak=True)
        do_try_seq(a_opt, "lpfw_full")

    # === Phase 3: LSE smooth descent ===
    print(f"\n=== Phase 3: LSE Descent (best={state['best_score']:.8f}) ===")
    cur_best = state['best_seq']
    rem = deadline - time.monotonic()
    t3 = min(180, rem * 0.18)
    if t3 > 30:
        n_cur = len(cur_best)
        print(f"  n={n_cur}, t={t3:.0f}s")
        a_opt, _ = lse_descent(cur_best, time_limit=t3,
                               temp_init=0.003, temp_mult=1.007,
                               lr=0.005, momentum=0.92, verbose=True)
        do_try_seq(a_opt, "lse_descent")

        # Second LSE with different params
        rem = deadline - time.monotonic()
        t3b = min(100, rem * 0.12)
        if t3b > 20:
            a_opt, _ = lse_descent(state['best_seq'], time_limit=t3b,
                                   temp_init=0.001, temp_mult=1.01,
                                   lr=0.002, momentum=0.94)
            do_try_seq(a_opt, "lse_fine")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={state['best_score']:.8f}) ===")
    base = state['best_seq']
    base_len = len(base)
    candidate_lengths = sorted(set([
        base_len - 64, base_len - 32, base_len + 32, base_len + 64,
        base_len + 128, base_len + 256,
        512, 640, 768, 896, 1024, 1152, 1280, 1536
    ]))
    candidate_lengths = [ln for ln in candidate_lengths if 256 <= ln <= 2048 and ln != base_len]

    explored = 0
    for target_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 50 or explored >= 5:
            break
        t_alloc = min(rem * 0.2, 120)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  Exploring n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_reduced=True,
                                   top_k_constraints=40, use_min_peak=True)
        do_try_seq(a_opt, f"lpfw_n{target_n}")
        explored += 1

        # Quick LSE polish
        rem = deadline - time.monotonic()
        if rem > 25:
            a_opt2, _ = lse_descent(a_opt, time_limit=min(25, rem * 0.15),
                                     temp_init=0.001, temp_mult=1.008,
                                     lr=0.002, momentum=0.93)
            do_try_seq(a_opt2, f"lse_n{target_n}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord CD (best={state['best_score']:.8f}) ===")
    cur_best = state['best_seq']
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 120)
    if t5 > 20:
        print(f"  n={len(cur_best)}, t={t5:.0f}s")
        a_opt, _ = coordinate_descent(cur_best, time_limit=t5)
        do_try_seq(a_opt, "coord_cd")

    # === Phase 6: Perturbation + LP-FW restart ===
    print(f"\n=== Phase 6: Perturbations (best={state['best_score']:.8f}) ===")
    cur_best = state['best_seq']
    n_best = len(cur_best)
    for p in range(10):
        rem = deadline - time.monotonic() - 15
        if rem < 20:
            break
        noise = rng.standard_normal(n_best) * (0.002 + 0.0015 * p)
        f_pert = project_simplex(cur_best + noise)
        t_p = min(35, rem * 0.25)
        if t_p < 12:
            break
        a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_p, use_reduced=True,
                                   top_k_constraints=40, use_min_peak=True)
        do_try_seq(a_opt, f"perturb_{p}")

    # === Phase 7: Final LP-FW polish ===
    print(f"\n=== Phase 7: Final Polish (best={state['best_score']:.8f}) ===")
    cur_best = state['best_seq']
    rem = deadline - time.monotonic() - 20
    t7 = min(rem * 0.5, 150)
    if t7 > 15:
        print(f"  n={len(cur_best)}, t={t7:.0f}s")
        a_opt, _ = lp_fw_optimize(cur_best, time_limit=t7, use_reduced=False,
                                   use_min_peak=True)
        do_try_seq(a_opt, "lpfw_final")

    # === Phase 8: Final LSE micro-polish ===
    rem = deadline - time.monotonic() - 15
    t8 = min(rem * 0.6, 60)
    if t8 > 10 and state['best_seq'] is not None:
        a_opt, _ = lse_descent(state['best_seq'], time_limit=t8,
                               temp_init=0.0005, temp_mult=1.005,
                               lr=0.0008, momentum=0.95)
        do_try_seq(a_opt, "lse_micro")

    # Final result
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(512) / 512
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    final_score = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score
    else:
        state['best_seq'] = np.ones(512) / 512
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
