"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    conv = np.maximum(conv, 0.0)  # numerical safety
    return conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.astype(np.float64).copy()
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t, g>=0, sum(g)=1. Returns (g, t)."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_cons)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, res.x[-1]
    except Exception:
        pass
    return None, None


def goldensection_line_search(f, g, n, time_limit=1.0):
    """Golden section search for optimal alpha in f + alpha*(g-f)."""
    t0 = time.monotonic()
    golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    # Evaluate at two interior points
    c1 = hi - golden_ratio * (hi - lo)
    c2 = lo + golden_ratio * (hi - lo)
    
    def eval_at(alpha):
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()
    
    f1 = eval_at(c1)
    f2 = eval_at(c2)
    
    while (hi - lo) > 1e-8 and time.monotonic() - t0 < time_limit:
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - golden_ratio * (hi - lo)
            f1 = eval_at(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + golden_ratio * (hi - lo)
            f2 = eval_at(c2)
    
    best_alpha = (lo + hi) / 2.0
    ft = f + best_alpha * (g - f)
    ft = np.maximum(ft, 0.0)
    s = ft.sum()
    if s < 1e-15:
        return f.copy(), f
    ft = ft / s
    return best_alpha, ft

def peak_equalization(f0, time_limit, n_peaks=10, verbose=False):
    """Directly equalize the top convolution peaks via iterative adjustment."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
        
        # Find top peaks
        k_actual = min(n_peaks, len(conv))
        peak_idx = np.argpartition(conv, -k_actual)[-k_actual:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        
        # Compute subgradient: for each peak k, the gradient of b[k] w.r.t. a[j] is 2*a[k-j]
        # Weight by how much each peak exceeds the median of top peaks
        peak_vals = conv[peak_idx]
        peak_med = np.median(peak_vals)
        # Weights: peaks above median get higher weight
        weights = np.ones(len(peak_idx))
        above = peak_vals > peak_med
        weights[above] = 2.0
        weights[~above] = 0.5
        w_sum = weights.sum()
        weights /= w_sum
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if weights[i] < 1e-10:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                # b[k] = sum_i a[i]*a[k-i], db[k]/da[j] = a[k-j] + a[j] (if k-j != j)
                # More precisely: db[k]/da[j] = a[k-j] (for the term where first index is j)
                #                              + a[j] (for the term where second index is j, i.e., k-i=j => i=k-j)
                # But these are the same term if k-j = j, i.e., 2j = k
                # Actually for convolution: b[k] = sum_i a[i]*a[k-i]
                # db[k]/da[j] = a[k-j] (from term i=j) + a[j] (from term k-i=j, i.e., i=k-j)
                # If j != k-j: contribution is a[k-j] + a[j]
                # If j == k-j (k=2j): contribution is 2*a[j]
                kj = k - j
                if 0 <= kj < n:
                    sg[j] += weights[i] * (a_val := f[kj])
                    if kj != j:
                        sg[j] += weights[i] * f[j]
                    else:
                        sg[j] += weights[i] * f[j]
        
        # Gradient step with projection
        f_new = project_simplex(f - lr * sg)
        f = f_new
        
        # Adaptive learning rate
        if it % 50 == 0:
            lr *= 0.95
        
        if it % 200 == 0 and verbose:
            print(f"    Peak-eq it {it}: best={best_obj:.12f}, lr={lr:.6f}")
    
    return best_f, best_obj


def coordinate_descent_peaks(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinate pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        
        # Find top 3 peaks
        top_k = min(3, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        improved = False
        for pk in peak_positions:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            
            # For each contributing pair (j, pk-j), try to reduce the product
            # by redistributing mass
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                kj = pk - j
                if kj == j or kj < 0 or kj >= n:
                    continue
                if f[j] < 1e-12 or f[kj] < 1e-12:
                    continue
                
                # Try to reduce the product f[j]*f[kj] by moving mass
                # while keeping sum constant
                # Option: reduce f[j] by delta, increase f[kj] by delta (or vice versa)
                # But this changes other convolutions too
                # Simpler: try to equalize f[j] and f[kj] (reduces product for fixed sum)
                avg = (f[j] + f[kj]) / 2.0
                diff = f[j] - f[kj]
                if abs(diff) < 1e-12:
                    continue
                
                # Try partial equalization
                for frac in [0.1, 0.25, 0.5, 0.75]:
                    delta = frac * diff
                    f_trial = f.copy()
                    f_trial[j] = f[j] - delta
                    f_trial[kj] = f[kj] + delta
                    if f_trial[j] < 0 or f_trial[kj] < 0:
                        continue
                    # Renormalize (sum is preserved)
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Random pair tries
                rng = np.random.default_rng(it * 31337)
                for _ in range(min(50, n // 5)):
                    j = int(rng.integers(0, n))
                    k2 = int(rng.integers(0, n))
                    if j == k2 or abs(j - k2) < 2:
                        continue
                    if f[j] < 1e-10 or f[k2] < 1e-10:
                        continue
                    for delta_frac in [0.05, -0.05, 0.1, -0.1]:
                        delta = delta_frac * min(f[j], f[k2])
                        f_trial = f.copy()
                        f_trial[j] -= delta
                        f_trial[k2] += delta
                        if f_trial[j] < 0 or f_trial[k2] < 0:
                            continue
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            best_obj = val
                            best_f = f_trial.copy()
                            f = f_trial
                            improved = True
                            break
                    if improved:
                        break
                if not improved and no_improve > 60:
                    break
    
    return best_f, best_obj


def spectral_refinement(f0, time_limit, verbose=False):
    """Try to flatten the power spectrum A(w)^2 to reduce convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(f, fft_size)
        A_sq = A * A  # This is the DFT of conv(f,f)
        
        # We want ifft(Re(A_sq)) to have minimum peak
        # Try to make |A(omega)| more uniform (which makes |A^2| more uniform in magnitude)
        # Compute target: geometric mean magnitude
        mags = np.abs(A_sq)
        mags_safe = np.maximum(mags, 1e-15)
        target_mag = np.exp(np.mean(np.log(mags_safe)))
        
        # Damp the overshoots in magnitude
        alpha = 0.05
        for i in range(len(A_sq)):
            if mags[i] > target_mag * 1.1:
                # Reduce magnitude at this frequency
                factor = 1.0 - alpha * (mags[i] / target_mag - 1.0)
                factor = max(0.5, min(1.0, factor))
                A_sq[i] *= factor
            elif mags[i] < target_mag * 0.9:
                # Slightly boost
                factor = 1.0 + alpha * 0.5 * (1.0 - mags[i] / target_mag)
                factor = min(1.2, factor)
                A_sq[i] *= factor
        
        # Reconstruct: we need A_new such that A_new^2 = A_sq
        # A_new = sqrt(A_sq) (choosing a branch)
        # But this might not give a real sequence
        # Alternative: just take the real part of ifft(A_sq) and project
        conv_new = np.fft.irfft(A_sq, fft_size)[:2*n-1]
        
        # This gives us the new convolution directly!
        # We want to find f_new such that conv(f_new, f_new) = conv_new
        # This is hard. Instead, use it as a guide.
        
        # Better approach: compute f_new = ifft(sqrt(A_sq)) and project
        # sqrt of complex: use principal branch
        with np.errstate(invalid='ignore'):
            A_new = np.sqrt(A_sq)
        
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new > 1e-15:
            f_new /= s_new
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-12:
                best_obj = val
                best_f = f_new.copy()
                f = f_new
        else:
            # Fallback: just project the real part of conv
            f = project_simplex(f + 0.01 * (conv_new[:n] / conv_new.max() - 1.0/n))
            if it % 100 == 0 and verbose:
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj - 1e-12:
                    best_obj = M
                    best_f = f.copy()
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=40, temp=0.00006):
    """Subgradient with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kj = k - j
                if 0 <= kj < n:
                    sg[j] += w[i] * f[kj]
                    if kj != j:
                        sg[j] += w[i] * f[j]
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        f = f_new
        
        cur_lr *= 0.9999
        cur_temp *= 1.003
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0005)
            no_improve = 0
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-Frank-Wolfe, peak equalization, coordinate descent, and spectral methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.array(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={best_score:.10f}")
    except (NameError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512.0
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.10f}")
    
    n_base = len(parent)
    
    # === Phase 1: Enhanced LP-FW on parent ===
    print(f"\n=== Phase 1: Enhanced LP-FW (n={n_base}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.28)
    if t1 > 60:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = lp_fw_enhanced(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_enh")
    
    # === Phase 2: Peak Equalization ===
    print(f"\n=== Phase 2: Peak Equalization (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.15)
    if t2 > 30 and best_seq is not None:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a_opt, _ = peak_equalization(best_seq, time_limit=t2, n_peaks=15, verbose=True)
        try_seq(a_opt, "peak_eq")
        
        # LP-FW after peak equalization
        rem = deadline - time.monotonic()
        t2b = min(100, rem * 0.08)
        if t2b > 20:
            a_opt, _ = lp_fw_enhanced(a_opt, time_limit=t2b)
            try_seq(a_opt, "lpfw_after_eq")
    
    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.10f}) ===")
    base = best_seq if best_seq is not not None else parent
    base = best_seq if best_seq is not None else parent
    
    candidate_lengths = set()
    for delta in [-64, -32, -16, 0, 16, 32, 64, 96, 128]:
        ln = len(base) + delta
        if 256 <= ln <= 2048:
            candidate_lengths.add(ln)
    candidate_lengths = sorted(candidate_lengths)
    
    for target_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        if target_n == len(base):
            continue
        t_alloc = min(rem * 0.25, 150)
        if t_alloc < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_enhanced(f_interp, time_limit=t_alloc * 0.6)
        try_seq(a_opt, f"lpfw_n{target_n}")
        
        # Quick peak equalization on the result
        rem = deadline - time.monotonic()
        if rem > 40:
            a_opt2, _ = peak_equalization(a_opt, time_limit=min(rem, 60), n_peaks=8)
            try_seq(a_opt2, f"peq_n{target_n}")
    
    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.10f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.2, 250)
        if t_sg > 30:
            print(f"  Time: {t_sg:.0f}s, n={n_best}")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                            lr=0.002, momentum=0.94,
                                            top_k=max(30, n_best // 6), temp=0.00005)
            try_seq(a_opt, "sg_main")
            
            # Different params
            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.15, 150)
            if t_sg2 > 20:
                a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg2,
                                                lr=0.001, momentum=0.95,
                                                top_k=max(20, n_best // 10), temp=0.00003)
                try_seq(a_opt, "sg_fine")
    
    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.10f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.25, 250)
        if t_cd > 30:
            print(f"  Time: {t_cd:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent_peaks(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "coord_cd")
    
    # === Phase 6: Spectral refinement ===
    print(f"\n=== Phase 6: Spectral (best={best_score:.10f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_spec = min(rem * 0.15, 150)
        if t_spec > 20:
            print(f"  Time: {t_spec:.0f}s")
            a_opt, _ = spectral_refinement(best_seq, time_limit=t_spec, verbose=True)
            try_seq(a_opt, "spectral")
    
    # === Phase 7: Final polish with combined methods ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.10f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Quick LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, _ = lp_fw_enhanced(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "lpfw_polish")
        
        # Quick peak eq
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = peak_equalization(best_seq, time_limit=min(rem, 50), n_peaks=5)
            try_seq(a_opt, "peq_polish")
        
        # Quick CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = coordinate_descent_peaks(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "cd_polish")
    
    # === Phase 8: Random perturbations ===
    print(f"\n=== Phase 8: Perturbations (best={best_score:.10f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(10):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise_scale = 0.0005 + 0.0003 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq / best_seq.sum() + noise)
            a_opt, _ = subgradient_optimize(f_pert, time_limit=min(rem, 30),
                                            lr=0.0008, momentum=0.95,
                                            top_k=max(10, n_best // 12), temp=0.00002)
            try_seq(a_opt, f"perturb_{p}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
