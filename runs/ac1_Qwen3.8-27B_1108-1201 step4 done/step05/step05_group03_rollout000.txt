def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)

        return np.array([max(v[0], 0.0)])

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 137 + 42)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_f, best_obj


def adam_sg_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    it = 0
    no_improve = 0
    max_iters = int(time_limit / 0.01)
    
    # Adaptive temperature
    T = 3.0 / n  # Initial temperature
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max(1, max_iters))
        lr = 0.005 * (1.0 - 0.3 * progress)
        
        conv = fast_convolve(f)
        
        # Adaptive top-k based on flatness
        # If top peak is much larger than median, use small top-k
        sorted_conv = np.sort(conv)[::-1]
        ratio = sorted_conv[0] / max(sorted_conv[len(sorted_conv)//2], 1e-15)
        if ratio > 3:
            top_k = max(5, n // 16)
        elif ratio > 1.5:
            top_k = max(20, n // 8)
        else:
            top_k = max(50, n // 4)
        
        # Temperature scheduling with reheats
        T = (3.0 - 2.5 * progress) / n
        if no_improve > 100:
            T = 5.0 / n  # Reheat
            no_improve = 0
        
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w
