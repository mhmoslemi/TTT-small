"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam subgradient with adaptive top-k peak tracking
2. LP-based Frank-Wolfe with reduced constraint set for speed
3. Hot-spot guided coordinate descent exploiting antisymmetric pairs
4. Multi-resolution coarse-to-fine optimization
5. Multi-length screening for optimal n

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_topk_subgradient(f, n, top_k, temperature=None):
    """Compute weighted subgradient from top-k convolution peaks."""
    conv = fast_convolve(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    if temperature is None:
        temperature = max(0.005, 0.2 / n)
    
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws
    
    sg = np.zeros(n)
    for i_idx in range(len(peaks)):
        wi = w[i_idx]
        if wi < 1e-12:
            continue
        k = int(peaks[i_idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * wi * f[k - idx]
    return sg, conv


def adam_optimize(f0, time_limit, lr_start=0.008, lr_end=0.001,
                  top_k_start=None, top_k_end=None, verbose=False):
    """Adam optimizer with top-k subgradient and simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if top_k_start is None:
        top_k_start = max(20, n // 2)
    if top_k_end is None:
        top_k_end = max(5, n // 16)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    max_tracked = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        progress = min(1.0, it / 8000.0)
        # Exponential decay of top_k
        top_k = max(3, int(top_k_start * (top_k_end / top_k_start) ** progress))
        # Cosine LR schedule
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        # Temperature: sharp at start, very sharp at end
        temp = max(0.001, 0.3 * (1.0 - progress) / n)
        
        sg, conv = get_topk_subgradient(f, n, top_k, temp)
        M = conv.max()
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-16:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 43 + 7)
                scale = 0.002 * (1 + 0.5 * min(no_improve // 50, 3))

                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-15:

    return best_f, best_obj


def build_lp_matrix(f, n, constraint_indices):
    """Build LP constraint matrix for given convolution positions."""
    num_cons = len(constraint_indices)
    A = np.zeros((num_cons, n))
    for ii, k in enumerate(constraint_indices):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        js = np.arange(i_start, i_end + 1)
        A[ii, js] = f[k - js]
    return A

            order = rng.permutation(n)
        elif passes % 4 == 2:
            order = np.arange(n)
        else:
            order = np.arange(n)[::-1]
        
        step = remaining / (n * 0.5)
        
        for idx_i in range(0, n, 1):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = order[idx_i]
            
            for offset in offsets:
                if idx_i + offset < n:
                    j = order[idx_i + offset]
                else:
                    j = order[n - 1 - (idx_i % n)]
                if i == j:
                    continue
                
                c = f[i] + f[j]
                if c < 1e-12:
                    continue
                
                t_cur = f[i] / c
                best_t = t_cur
                best_val = best_obj
                
                # Quick 5-point scan
                for dt in [0.0, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                    t = np.clip(t_cur + dt, 0.0, 1.0)
                    if abs(t - t_cur) < 1e-10:
                        continue
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1.0 - t) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                if best_val < best_obj - 1e-14:
                    f[i] = best_t * c
                    f[j] = (1.0 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
    
    return best_f, best_obj


def block_local_search(f0, time_limit, block_size=6, verbose=False):
    """Local search on blocks of consecutive coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    
    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        if no_improve > 150:
            break
        
        improved = False
        block_order = rng.permutation(num_blocks)
        
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            start = bi * block_size
            end = min(start + block_size, n)
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue
            
            # Try random perturbation preserving block sum
            for attempt in range(4):
                pert = rng.standard_normal(end - start) * 0.02 * block_sum
                trial_block = np.maximum(f[start:end] + pert, 0)
                ts = trial_block.sum()
                if ts < 1e-12:
                    continue
                trial_block = trial_block * (block_sum / ts)
                
                f_trial = f.copy()
                f_trial[start:end] = trial_block
                val = fast_convolve(f_trial).max()
                
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def downsample_average(f, new_n):
    """Downsample by averaging blocks."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    block_size = old_n / new_n
    result = np.zeros(new_n)
    for i in range(new_n):
        start = int(i * block_size)
        end = int((i + 1) * block_size)
        end = max(end, start + 1)
        result[i] = f[start:end].mean()
    ss = result.sum()
    if ss > 1e-15:
        result /= ss
    return result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for step function minimization.
    Phases: multi-n screening, LP-FW, Adam, hotspot CD, pairwise polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0:
            return float('inf')
        a = a / a.sum()
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: score={sc:.8f} (n={len(a)})")
        return sc
    
    # --- Get parent sequence ---
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={parent_n}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        parent_n = 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={parent_n}, score={best_score:.8f}")
    
    print(f"\n{'='*60}")
    print(f"Starting optimization. Current best: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    # === Phase 1: Multi-n screening (quick Adam on different lengths) ===
    print(f"\n=== Phase 1: Multi-n Screening ===")
    rem = deadline - time.monotonic() - 900
    t_total = min(rem, 300)
    
    if t_total > 30:
        # Determine candidate n values around parent_n
        base_n = len(best_seq)
        n_candidates = sorted(set([
            384, 512, base_n,
            base_n - 64 if base_n > 448 else 448,
            base_n + 64,
            base_n + 128,
            base_n - 128 if base_n > 512 else 512,
        ]))
        n_candidates = [n for n in n_candidates if 320 <= n <= 1280]

            t_alloc = min(t_per, rem * 0.6)

            print(f"  Screening n={tgt_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(f_init, time_limit=t_alloc,
                                      lr_start=0.01, lr_end=0.002)

    # === Phase 2: LP-FW on best sequence ===
    print(f"\n=== Phase 2: LP-FW Deep (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    t_lpfw = min(rem, 350)
