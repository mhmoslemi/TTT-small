def fast_convolve(a):
    # Provided as inspiration, but we'll use our own implementation
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    energies = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(energies):
        return np.ones(n) / n
    rho = np.nonzero(energies)[0][-1] + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for i in range(2 * n - 1):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        if j_start <= j_end:
            j_indices = np.arange(j_start, j_end + 1)
            vals.extend([float(f[i - j]) for j in j_indices])
            cols.extend(j_indices.tolist())
            rows.extend([i] * len(j_indices))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
