    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j]
    # For each k, the coefficient of g[j] is f[k-j] (if valid)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] with coefficient f[k-j]
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smoothed_optimize(a0, time_limit, T_init=0.001, lr_init=0.01, 
                      mom=0.93, lp_interval=150, verbose=False):
    """
    Smoothed gradient descent with momentum, adaptive T, and LP oracle.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        # Compute softmax weights
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = convex_grad(p, a)

        # Momentum
        v = mom * v - lr * grad
        candidate = project_simplex(a + v)

        # Evaluate
        new_conv = fast_convolve(candidate)
        new_obj = new_conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = candidate.copy()
            no_improve = 0
            a = candidate
        else:
            a = candidate
            no_improve += 1

        # Adaptive learning rate
        if no_improve % 50 == 0 and no_improve > 0:
            lr *= 0.95
        if no_improve > 200:
            # Restart from best
            rng = np.random.default_rng((it * 104729) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = max(lr_init * 0.3, lr)
            T = T_init
            no_improve = 0

        # Adaptive temperature
        if no_improve > 100:
            T = min(T * 1.01, T_init * 10)
        if no_improve > 100:
            T = max(T * 0.99, T_init * 0.1)

        # LP oracle
        if lp_interval > 0 and it % lp_interval == 0 and remaining > 60:
            M_curr = fast_convolve(a).max()
            g = solve_lp_oracle(a, M_curr)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                for alpha in [0.1, 0.3, 0.5]:
                    blend = project_simplex((1 - alpha) * best_a + alpha * g)
                    bobj = fast_convolve(blend).max()
                    if bobj < best_obj - 1e-14:
                        best_obj = bobj
                        best_a = blend.copy()
                        a = blend.copy()
                        v = np.zeros(n)
                        no_improve = 0
                        break

        if verbose and it % 1000 == 0:
            print(f"    it={it}, obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")

    return best_a, best_obj


def multi_scale_optimize(target_n, time_limit, parent=None):
    """
    Multi-scale: optimize at n/4, n/2, then target_n.
    """
    if parent is not None and len(parent) == target_n:
        start = parent.copy()
    else:
        start = np.ones(target_n) / target_n

    # Start small
    small_n = max(32, target_n // 8)
    small_a = np.interp(np.linspace(0, 1, small_n), 
                        np.linspace(0, 1, target_n), start)
    small_a = np.maximum(small_a, 0)
    small_a = small_a / small_a.sum()

    t_alloc_small = min(30, time_limit * 0.05)
    small_a, _ = smoothed_optimize(small_a, time_limit=t_alloc_small, T_init=0.01, lr_init=0.05)
    
    # Upsample
    mid_n = max(64, target_n // 2)
    mid_a = np.interp(np.linspace(0, 1, mid_n), 
                      np.linspace(0, 1, small_n), small_a)
    mid_a = mid_a / mid_a.sum()
    
    t_alloc_mid = min(60, time_limit * 0.1)
    mid_a, _ = smoothed_optimize(mid_a, time_limit=t_alloc_mid, T_init=0.005, lr_init=0.03)
    
    # Upsample to target
    target_a = np.interp(np.linspace(0, 1, target_n), 
                         np.linspace(0, 1, mid_n), mid_a)
    target_a = target_a / target_a.sum()
    
    return target_a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (increasing)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())
    
    # Linear ramp (decreasing)
    ramp_d = (n - x)
    inits.append(ramp_d / ramp_d.sum())
    
    # Exponential decay
    for alpha in [0.5, 1.0, 2.0]:
        exp_d = np.exp(-alpha * x / n)
        inits.append(exp_d / exp_d.sum())
        exp_r = np.exp(-alpha * (n - 1 - x) / n)
        inits.append(exp_r / exp_r.sum())
    
    # Gaussian (different centers and widths)
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.4]:
            g = np.exp(-0.5 * ((x - center * (n-1)) / (width * n)) ** 2)
            inits.append(g / g.sum())
    
    # Power law
    for beta in [0.5, 1.0, 2.0, 3.0]:
        pw = (x + 1.0) ** (-beta)
        inits.append(pw / pw.sum())
        pw2 = (n - x + 1.0) ** (-beta)
        inits.append(pw2 / pw2.sum())
    
    # Cosine
    cos = np.abs(np.cos(np.pi * x / (2*n))) ** 2
    inits.append(cos / cos.sum())
    
    # Sine
    sin = np.sin(np.pi * x / (2*n)) ** 2 + 0.01
    inits.append(sin / sin.sum())
    
    # Random (Dirichlet)
    for alpha_dir in [0.1, 0.5, 1.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_dir))
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

    6. Final fine-tuning

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)
        return score
