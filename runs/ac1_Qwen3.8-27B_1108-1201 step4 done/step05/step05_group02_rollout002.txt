"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smoothed (log-sum-exp) optimization
with Nesterov acceleration, LP-based Frank-Wolfe polishing, and multi-length exploration.
Key innovations: FFT-based smoothed gradient computation, adaptive temperature schedule,
deterministic annealing, and hybrid method switching.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smoothed_gradient_fast(f, t):
    """Compute gradient of 2*n*logsumexp(t*conv(f,f)) efficiently via FFT.
    Returns (gradient, current_max)."""
    n = len(f)
    conv = fast_convolve(f)
    m = conv.max()
    ex = np.exp(t * (conv - m))
    w = ex / ex.sum()
    # grad[j] = sum_l w[j+l] * f[l] = (w * f_flipped)[j+n-1]
    f_flipped = f[::-1]
    conv_len = (2 * n - 1) + n - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    result = np.fft.irfft(W * F, fft_size)
    grad = result[n - 1:2 * n - 1]
    return 2.0 * n * grad, m


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

    best_f = f.copy()

    # Nesterov state
    f_k = f.copy()
    y_k = f.copy()

    t_current = t_start
    t_total = time_limit

    momentum = 0.9

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 200

        t_current = t_start * (t_end / t_start) ** progress

        # Standard Nesterov: beta_k = (1 + sqrt(1+4*(k-1)^2)) / (2*(k-1)^2 + 2)
        k = it + 1

        y_k = y_new

    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"        LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            M = fast_convolve(f).max()
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 50, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50,
                         temp=0.00005, verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"      SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Nesterov smoothed optimization (main workhorse)
    2. LP-FW polishing
    3. Higher-temperature smoothed refinement
    4. Multi-length exploration
    5. Subgradient + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(832) / 832
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Nesterov smoothed optimization ===
    print(f"\n=== Phase 1: Nesterov Smoothed (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.3)
    if t1 > 30:
        print(f"  t={t1:.0f}s, t_start=5, t_end=1000")
        a_opt, _ = nesterov_smoothed_optimize(parent, time_limit=t1,
                                              t_start=5.0, t_end=1000.0,
                                              verbose=True)
        try_seq(a_opt, "nesterov_p1")

    # === Phase 2: LP-FW polishing ===
    print(f"\n=== Phase 2: LP-FW Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(250, rem * 0.2)
    if t2 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_p2")

    # === Phase 3: Higher-temperature Nesterov refinement ===
    print(f"\n=== Phase 3: Nesterov Refinement (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(350, rem * 0.25)
    if t3 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t3:.0f}s, t_start=50, t_end=5000")
        a_opt, _ = nesterov_smoothed_optimize(best_seq, time_limit=t3,
                                              t_start=50.0, t_end=5000.0,
                                              verbose=True)
        try_seq(a_opt, "nesterov_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = set()
    for delta in [-128, -64, 64, 128, 256, 384]:
        ln = len(base) + delta
        if 384 <= ln <= 2048:
            candidate_lengths.add(ln)
    candidate_lengths = sorted(candidate_lengths)

    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 180)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  Nesterov n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_optimize(f_interp, time_limit=t_alloc,
                                              t_start=10.0, t_end=500.0)
        try_seq(a_opt, f"nesterov_n{target_n}")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.2, 200)
        if t_sg > 20:
            print(f"  n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                            lr=0.002, momentum=0.94,
                                            top_k=max(40, n_best // 6),
                                            temp=0.00004)
            try_seq(a_opt, "sg_p5")

    # === Phase 6: Final LP-FW + perturbations ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_final = min(rem, 120)
            print(f"  LP-FW final, t={t_final:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final, verbose=True)
            try_seq(a_opt, "lpfw_final")

        # Perturbations + quick Nesterov
        for p in range(8):
            rem = deadline - time.monotonic() - 15
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * (0.0008 + 0.0004 * p)
            f_pert = project_simplex(best_seq + noise)
            t_rem = min(rem, 20)
            a_opt, _ = nesterov_smoothed_optimize(f_pert, time_limit=t_rem,
                                                  t_start=100.0, t_end=3000.0)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
