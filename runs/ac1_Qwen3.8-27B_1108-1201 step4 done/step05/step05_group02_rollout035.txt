"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase strategy:
1. Multiple structured initializations (ramp, cosine, exponential, parent)
2. LP-based Frank-Wolfe with min-peak formulation

4. Peak-equalizing coordinate descent
5. Multi-length exploration and final polish

import numpy as np
from scipy.optimize import linprog
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_from_conv(conv):
    """Compute score from convolution of normalized sequence."""
    return conv.max()


def make_ramp(n, alpha=1.0):
    """Create a ramp initialization: a[i] = (i+1)^alpha normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** alpha
    s = a.sum()
    return a / s


def make_cosine(n):
    """Create a cosine bump initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / n) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_exp(n, alpha=0.5):
    """Create an exponential ramp initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * i / n)
    s = a.sum()
    return a / s


def make_bell(n, sigma_frac=0.3):
    """Create a Gaussian bell centered at 2/3 of the sequence."""
    i = np.arange(n, dtype=np.float64)
    center = 0.67 * (n - 1)
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t) or (None, None) on failure."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Returns g or None on failure."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_min_peak=True, lp_every=1):
    """LP-based Frank-Wolfe with adaptive strategy."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Use LP direction or subgradient fallback
        use_lp = (it % lp_every == 0)

        if use_lp and use_min_peak:
            g, lp_t = solve_lp_min_peak(f, n)
            if g is None:
                g = solve_lp_max_sum(f, M * 1.05, n)
                if g is not None:
                    gs = g.sum()
                    g = g / gs if gs > 1e-15 else np.ones(n) / n
                else:
                    g = None
            else:
                gs = g.sum()
                g = g / gs if gs > 1e-15 else np.ones(n) / n
        elif use_lp:
            g = solve_lp_max_sum(f, M * 1.03, n)
            if g is None:
                g = solve_lp_max_sum(f, M * 1.08, n)
            if g is None:
                g = None
            else:
                gs = g.sum()
                g = g / gs if gs > 1e-15 else np.ones(n) / n
        else:
            g = None

        if g is None:
            # Subgradient fallback: move mass away from peak-contributing positions
            conv = fast_convolve(f)
            top_k = min(10, len(conv))
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            w = pv / (pv + 1e-15)
            w /= w.sum()
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                kg = k - idx
                mask = (kg >= 0) & (kg < n)
                sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
            # Move opposite to subgradient (to reduce peak)
            g = project_simplex(f - 0.01 * sg)

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search (golden section)
        best_alpha = 0.0
        best_val = M
        direction = g - f

        # Coarse search
        for alpha in [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.4, 0.6, 0.85]:
            ft = f + alpha * direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * direction
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * direction
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 70:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.005, momentum=0.93, top_k=50, temp=0.00005,
                verbose=False, nesterov=True):
    """Subgradient method with softmax-weighted peaks, Nesterov momentum, and adaptive annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        # Use a dynamic threshold: include peaks above mean + threshold
        mean_conv = conv.mean()
        threshold = mean_conv + 0.1 * (best_obj - mean_conv)
        peak_mask = conv > threshold
        if peak_mask.sum() < 3:
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        else:
            peaks = np.where(peak_mask)[0]
            if len(peaks) > k_actual * 2:
                peaks = peaks[np.argsort(conv[peaks])[-k_actual:]]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        # Nesterov-like momentum
        if nesterov:
            f_look = project_simplex(f + momentum * v)
            # Recompute subgradient at lookahead point (approximation: use previous sg direction scaled)
            pass
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 1000 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        # Adaptive parameter updates
        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def peak_cd_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Try modifying each coordinate in the peak's support
            for j in range(js, je + 1, max(1, n // 50)):
                if time.monotonic() - t0 > time_limit:
                    break

                old_val = f[j]
                if old_val < 1e-14:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:

                    if ss < 1e-15:

                    if val < best_j_obj - 1e-16:

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        ss = f.sum()
                        if ss > 1e-15:
                            f = f / ss

                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random perturbation search
                rng = np.random.default_rng(passes * 104729 + 7)
                found = False
                for attempt in range(30):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    mult = float(rng.uniform(0.4, 1.8))
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-15:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        found = True
                        break
                if not found:
                    if no_improve > 15:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def pairwise_swap(f, time_limit, verbose=False):
    """Try pairwise mass transfers to reduce peak."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy() if 'f0' in dir() else np.maximum(f, 0.0).copy()
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Get top peaks
        top_k = min(3, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(pk)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            support = list(range(js, je + 1))

            # Try moving mass from high-contributing to low-contributing positions
            contrib = np.zeros(n)
            for j in support:
                partner = k - j
                if 0 <= partner < n:
                    contrib[j] = f[j] * f[partner]

            # Sort by contribution
            if len(support) > 10:
                idx_sorted = np.argsort(contrib[support])[::-1]
                top_half = [support[i] for i in idx_sorted[:len(support)//4]]
                bot_half = [support[i] for i in idx_sorted[-len(support)//4:]]
            else:
                top_half = support[:2]
                bot_half = support[-2:]

            for src in top_half[:5]:
                if time.monotonic() - t0 > time_limit:
                    break
                for dst in bot_half[:5]:
                    if src == dst:
                        continue
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        amt = f[src] * frac
                        f_trial = f.copy()
                        f_trial[src] -= amt
                        f_trial[dst] += amt
                        if f_trial[src] < 0:
                            continue
                        ss = f_trial.sum()
                        if ss < 1e-15:
                            continue
                        f_trial = f_trial / ss
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-16:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: structured initialization, LP-FW, subgradient,
    peak CD, pairwise swaps, multi-length exploration, and final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0).astype(np.float64)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # === Initialization phase ===
    print("=== Phase 0: Initialization ===")
    inits = []

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate structured initializations at several lengths
    n_candidates = [512, 768, 832, 1024]

    if parent is not None:
        inits.append(("parent", parent))

    for n_val in n_candidates:
        # Ramp with various powers
        for alpha in [0.5, 1.0, 1.5, 2.0]:
            a = make_ramp(n_val, alpha)
            inits.append((f"ramp_a{alpha}_n{n_val}", a))
        # Cosine
        a = make_cosine(n_val)
        inits.append((f"cosine_n{n_val}", a))
        # Exponential
        for alpha in [0.3, 0.7, 1.0]:
            a = make_exp(n_val, alpha)
            inits.append((f"exp_a{alpha}_n{n_val}", a))
        # Bell
        a = make_bell(n_val, 0.25)
        inits.append((f"bell_n{n_val}", a))

    # Evaluate all initializations
    print(f"  Evaluating {len(inits)} initializations...")
    for label, a in inits:
        score = try_seq(a, label)
        if score < best_score:
            best_seq = a.copy()

    print(f"  Best after init: {best_score:.8f} (n={len(best_seq)})")

    # Quick optimization of best few initializations
    print("\n=== Phase 1: Quick LP-FW on top initializations ===")
    # Sort inits by score, take top 3
    init_scores = []
    for label, a in inits:
        s = a.sum()
        if s < 1e-10:
            continue
        an = a / s
        conv = fast_convolve(an)
        init_scores.append((conv.max(), label, an))
    init_scores.sort(key=lambda x: x[0])

    for conv_max, label, a in init_scores[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(120, rem * 0.15)
        if t_alloc < 30:
            break
        print(f"  LP-FW on [{label}], n={len(a)}, t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(a, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"lpfw_{label}")

    # === Phase 2: Main LP-FW on best ===
    print(f"\n=== Phase 2: Main LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.2)
        if t2 > 60:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t2, verbose=True,
                                        use_min_peak=True, lp_every=2)
            try_seq(a_opt, "lpfw_main")

    # === Phase 3: Subgradient optimization ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t3 = min(400, rem * 0.25)
        if t3 > 60:
            print(f"  n={n_best}, t={t3:.0f}s")
            a_opt, obj = sg_optimize(best_seq, time_limit=t3,
                                     lr=0.004, momentum=0.94,
                                     top_k=max(40, n_best // 6),
                                     temp=0.00004, verbose=True, nesterov=True)
            try_seq(a_opt, "sg_main")

            # Second pass with different params
            rem = deadline - time.monotonic()
            t3b = min(200, rem * 0.15)
            if t3b > 40:
                a_opt, obj = sg_optimize(best_seq, time_limit=t3b,
                                         lr=0.002, momentum=0.95,
                                         top_k=max(30, n_best // 10),
                                         temp=0.00002, verbose=False, nesterov=True)
                try_seq(a_opt, "sg_fine")

    # === Phase 4: Peak CD ===
    print(f"\n=== Phase 4: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(250, rem * 0.2)
        if t4 > 50:
            print(f"  n={len(best_seq)}, t={t4:.0f}s")
            a_opt, obj = peak_cd_optimize(best_seq, time_limit=t4)
            try_seq(a_opt, "peak_cd")

    # === Phase 5: Pairwise swaps ===
    print(f"\n=== Phase 5: Pairwise Swaps (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(150, rem * 0.1)
        if t5 > 30:
            print(f"  n={len(best_seq)}, t={t5:.0f}s")
            a_opt, obj = pairwise_swap(best_seq, time_limit=t5)
            try_seq(a_opt, "swap")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = set()
        for delta in [-256, -128, -64, 64, 128, 256, 512]:
            ln = n_best + delta
            if 256 <= ln <= 2048:
                target_lengths.add(ln)
        # Also try some "natural" lengths
        for ln in [512, 768, 1024, 1280]:
            if ln not in target_lengths and abs(ln - n_best) > 100:
                target_lengths.add(ln)
        target_lengths = sorted(target_lengths)

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 100:
                break
            t_alloc = min(200, rem * 0.25)
            if t_alloc < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            # Quick LP-FW
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc * 0.6,
                                        verbose=False, lp_every=3)
            score = try_seq(a_opt, f"multin_{target_n}_lp")
            # Then subgradient polish
            rem2 = deadline - time.monotonic()
            if rem2 > 40:
                a_opt2, obj2 = sg_optimize(a_opt, time_limit=min(rem2 * 0.5, 100),
                                           lr=0.002, momentum=0.94,
                                           top_k=max(30, target_n // 8),
                                           temp=0.00003)
                try_seq(a_opt2, f"multin_{target_n}_sg")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick SG with very small lr
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, obj = sg_optimize(best_seq, time_limit=min(rem, 100),
                                     lr=0.0008, momentum=0.96,
                                     top_k=max(20, n_best // 12),
                                     temp=0.00001)
            try_seq(a_opt, "sg_polish")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=min(rem, 80),
                                        use_min_peak=True, lp_every=1)
            try_seq(a_opt, "lpfw_polish")

        # Small perturbations + quick optimization
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise_scale = 0.0008 + 0.0004 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, obj = sg_optimize(f_pert, time_limit=min(rem, 30),
                                     lr=0.0005, momentum=0.96,
                                     top_k=max(15, n_best // 15),
                                     temp=0.00001)
            try_seq(a_opt, f"perturb_{p}")

        # Peak CD final
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_opt, obj = peak_cd_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
