"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr_weighted(w, a):
    """Compute grad[m] = sum_k w[k] * a[k-m] for m=0..n-1 using FFT."""
    n = len(a)
    S = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, S)
    A = np.fft.rfft(a, S)
    return np.fft.irfft(np.conj(W) * A, S)[:n]


def subgradient_max_conv(a, eps_frac=0.01):
    """Compute subgradient of max(conv(a,a)) w.r.t. a using active set."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Active set: indices where conv is within eps of max
    threshold = M - eps_frac * M
    active_mask = conv >= threshold
    w = np.zeros(len(conv))
    w[active_mask] = 1.0
    # Normalize weights
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    grad = 2.0 * cross_corr_weighted(w, a)
    return grad, M, conv


def lse_gradient(a, beta, max_peaks=300):
    """Compute gradient of logsumexp(beta * conv(a,a)) using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # Only use top peaks for efficiency
    if max_peaks < len(conv):
        idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        w_sparse = np.zeros_like(w)
        w_sparse[idx] = w[idx]
        w_sparse /= w_sparse.sum()
        w = w_sparse
    grad = 2.0 * cross_corr_weighted(w, a)
    M = conv.max()
    return grad, M, conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_step(f, M, time_limit_lp=5.0):
    """Single Frank-Wolfe step: solve LP to find descent direction."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            if conv_trial.max() < best_obj - 1e-12:
                best_obj = conv_trial.max()
                best_f = f_trial.copy()
                f = f_trial
                improved = True
                break

        if not improved:
            # Try a random perturbation
            mask = rng.random(n) < 0.1
            if mask.sum() > 0:
                f_trial = f.copy()
                f_trial[mask] *= (1.0 + 0.02 * rng.standard_normal(mask.sum()))
                f_trial = np.maximum(f_trial, 0.0)
                s2 = f_trial.sum()
                if s2 > 1e-15:
                    f_trial /= s2
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj - 1e-12:
                        best_obj = conv_trial.max()
                        best_f = f_trial.copy()
                        f = f_trial

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Subgradient + LSE on parent and nearby lengths
    Phase 2: LP-based FW for escaping local optima
    Phase 3: High-beta LSE polishing
    Phase 4: Coordinate descent micro-polish
    Phase 5: Final nearby-length search
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, best={best_tracker[0]:.8f}")

    # === Phase 1: Subgradient + LSE exploration ===
    print("\n=== Phase 1: Subgradient + LSE Exploration ===")
    t1_end = deadline - 900

    # Try parent length with subgradient
    t_alloc = min(120, t1_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Subgradient n={n_parent}, t={t_alloc:.0f}s")
        f_opt, obj = subgradient_optimize(parent, time_limit=t_alloc, momentum=0.88, lr_init=4.0/n_parent)
        try_sequence(f_opt, best_tracker)

    # Follow with LSE
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(100, t1_end - time.monotonic())
    if t_alloc > 20:
        print(f"  LSE n={len(current)}, t={t_alloc:.0f}s")
        f_opt, obj = lse_descent(current, time_limit=t_alloc, beta_start=10.0, beta_end=3000.0)
        try_sequence(f_opt, best_tracker)

    # Try a larger n
    n_large = max(n_parent + 128, 896)
    if time.monotonic() < t1_end - 80:
        f_large = interpolate_to_length(best_tracker[1] if best_tracker[1] is not None else parent, n_large)
        t_alloc = min(80, t1_end - time.monotonic())
        if t_alloc > 20:
            print(f"  Subgradient n={n_large}, t={t_alloc:.0f}s")
            f_opt, obj = subgradient_optimize(f_large, time_limit=t_alloc, momentum=0.9, lr_init=4.0/n_large)
            try_sequence(f_opt, best_tracker)
            t_alloc = min(60, t1_end - time.monotonic())
            if t_alloc > 15:
                f_opt2, _ = lse_descent(f_opt, time_limit=t_alloc, beta_start=15.0, beta_end=2000.0)
                try_sequence(f_opt2, best_tracker)

    # Try another length
    n_med = max(n_parent - 128, 640) if n_parent > 700 else n_parent + 64
    if time.monotonic() < t1_end - 60:
        f_med = interpolate_to_length(best_tracker[1] if best_tracker[1] is not None else parent, n_med)
        t_alloc = min(60, t1_end - time.monotonic())
        if t_alloc > 20:
            print(f"  Subgradient n={n_med}, t={t_alloc:.0f}s")
            f_opt, obj = subgradient_optimize(f_med, time_limit=t_alloc, momentum=0.9, lr_init=4.0/n_med)
            try_sequence(f_opt, best_tracker)
            t_alloc = min(40, t1_end - time.monotonic())
            if t_alloc > 15:
                f_opt2, _ = lse_descent(f_opt, time_limit=t_alloc, beta_start=15.0, beta_end=1500.0)
                try_sequence(f_opt2, best_tracker)

    # === Phase 2: LP-based Frank-Wolfe ===
    print(f"\n=== Phase 2: LP-based Frank-Wolfe (best={best_tracker[0]:.8f}) ===")
    t2_end = deadline - 500
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 60:
        print(f"  FW n={len(current)}, t={t_alloc:.0f}s")
        f_opt, obj = fw_optimize(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # Second FW run with perturbation
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 40:
        rng = np.random.default_rng(seed + 1)
        f_pert = project_simplex(current + rng.standard_normal(len(current)) * 0.005)
        print(f"  FW perturbed n={len(f_pert)}, t={t_alloc:.0f}s")
        f_opt, obj = fw_optimize(f_pert, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: High-beta LSE polishing ===
    print(f"\n=== Phase 3: High-beta LSE Polishing (best={best_tracker[0]:.8f}) ===")
    t3_end = deadline - 300
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(150, t3_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE high-beta n={len(current)}, t={t_alloc:.0f}s")
        f_opt, obj = lse_descent(current, time_limit=t_alloc, beta_start=50.0, beta_end=8000.0)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(120, t3_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE very-high-beta n={len(current)}, t={t_alloc:.0f}s")
        f_opt, obj = lse_descent(current, time_limit=t_alloc, beta_start=200.0, beta_end=15000.0)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Coordinate descent micro-polish ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_tracker[0]:.8f}) ===")
    t4_end = deadline - 120
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(100, t4_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Coord descent n={len(current)}, t={t_alloc:.0f}s")
        f_opt, obj = coordinate_descent(current, time_limit=t_alloc, n_coord=15)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Final nearby-length search ===
    print(f"\n=== Phase 5: Nearby Lengths (best={best_tracker[0]:.8f}) ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [64, -64, 128, -128, 32, -32, 192, -192]
        for off in offsets:
            n_try = n_best + off
            if n_try < 256 or n_try > 1400:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 30:
                break
            t_alloc = min(50, t_remaining * 0.4)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_descent(f_init, time_limit=t_alloc, beta_start=20.0, beta_end=3000.0)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using uniform fallback")
    fallback = np.ones(512) / 512
    return [float(x) for x in fallback]
