def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(w, a):
    """Cross-convolution of w (length 2n-1) and a (length n) via FFT."""
    size = len(w) + len(a) - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(W * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, t):
    """Compute gradient of t * log(sum_k exp(c[k]/t)) where c = conv(a,a).
    Returns (gradient, smoothed_value, peaks_info)."""
    n = len(a)
    c = fast_convolve(a)
    cmax = c.max()
    # Numerically stable exp
    scaled = (c - cmax) / t
    # Clip to avoid underflow
    scaled = np.maximum(scaled, -50)
    w = np.exp(scaled)
    Z = w.sum()
    if Z < 1e-300:
        # All mass on the peak
        peak_idx = np.argmax(c)
        w = np.zeros(len(c))

    smoothed_val = t * (cmax / t + np.log(Z))  # t * log(Z) + cmax
    return grad, smoothed_val, c, cmax

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None
