def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_nonneg_sum1(v):
    """Project onto {a >= 0, sum(a) = 1} via simplex projection."""
    return project_simplex(v)

            # Weight by exponential of scaled values
            scaled = vals - vals.max()
            w = np.exp(scaled * 3.0)
            w /= w.sum()
            grad = np.zeros(n)
            for li, k in enumerate(idx):
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    grad[i] += w[li] * 2.0 * a[k - i]
            return grad, conv, idx[0]
    else:
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        grad = np.zeros(n)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_peak - i]
        conv = fast_convolve(a)
        return grad, conv, k_peak


def lse_gradient(a, beta, max_terms=500):
    """
    Gradient of LSE-beta approximation to max(conv(a,a)).
    LSE_beta(conv) = (1/beta) * log(sum(exp(beta * conv[k])))
    Gradient[i] = (1/beta) * sum_k w[k] * d(conv[k])/d(a[i])
    where w[k] = exp(beta*conv[k]) / sum(exp(beta*conv))
    Using structure: d(conv[k])/d(a[i]) = 2*a[k-i] for valid i.
    """
    n = len(a)
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Efficiently compute gradient using only significant w entries
    threshold = w.max() * 1e-6
    sig_idx = np.where(w > threshold)[0]
    if len(sig_idx) > max_terms:
        # Keep only top terms
        sig_idx = np.array(np.argsort(w)[::-1][:max_terms])
    
    grad = np.zeros(n)
    for k in sig_idx:
        k = int(k)
        wk = w[k]
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[i] += wk * 2 * a[k-i] for i in [lo, hi]
        indices = np.arange(lo, hi + 1)
        grad[indices] += 2.0 * wk * a[k - indices]
    
    return grad / beta, conv


def spectral_perturb(a, strength=0.1):
    """
    Spectral water-filling perturbation:
    Make the power spectrum flatter, then project back to non-negative simplex.
    """
    n = len(a)
    A = np.fft.rfft(a)
    P = np.abs(A) ** 2
    phases = np.angle(A)
    
    # Water-fill: raise the bottom, compress the top
    P_sorted = np.sort(P)[::-1]
    mean_P = P.mean()
    # Blend towards flat
    P_new = P + strength * (mean_P - P)
    P_new = np.maximum(P_new, P.min() * 1e-6)
    # Renormalize so that sum of P_new * P_new / P = sum of P (keeps energy)
    # Actually just renormalize to same total power
    P_new *= (P.sum() / P_new.sum())
    
    # Reconstruct signal with modified magnitudes and original phases
    A_new = np.sqrt(P_new) * phases
    a_new = np.fft.irfft(A_new, n)
    
    # Project to non-negative, sum=1
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(n) / n
    return a_new


def greedy_peak_reduction(a, n_steps=1000, n_top=5):
    """
    Greedy coordinate descent: move mass from elements that most contribute
    to the top peaks to elements that contribute least.
    """
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    for step in range(n_steps):
        conv = fast_convolve(a)
        
        # Find top peaks
        k_top = min(n_top, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        peak_vals = conv[peak_idx]
        
        # Compute weighted contribution of each element to the peaks
        # Element i contributes to peak k as: a[i] * a[k-i] (and symmetrically)
        # Weight by peak value
        contrib = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            w_k = (peak_vals[list(peak_idx).index(k)] - peak_vals.min() + 1e-10)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                contrib[i] += w_k * a[k - i]
        
        # Find source (high contrib) and destination (low contrib)
        if contrib.max() - contrib.min() < 1e-15:
            break
        
        # Move mass from top contributors to bottom contributors
        src = np.argsort(contrib)[::-1][:n]  # top n sources
        dst = np.argsort(contrib)[:n]  # bottom n destinations
        
        # Only move from positive to destinations
        src = src[a[src] > 1e-10]
        if len(src) == 0:
            break
        
        total_src = a[src].sum()
        if total_src < 1e-15:
            break
        
        # Fraction to move (aggressive but safe)
        frac = 0.02
        move_amount = total_src * frac
        
        # Distribute to destinations proportional to their "room"
        dst_slots = np.maximum(0, a.max() * 2 - a[dst])  # room to grow
        if dst_slots.sum() < 1e-15:
            dst_slots = np.ones(len(dst))
        dst_slots /= dst_slots.sum()
        
        # Distribute from sources proportional to their contribution
        src_weights = a[src] / a[src].sum()
        
        for si, ii in enumerate(src):
            a[ii] -= move_amount * src_weights[si]
        for di, jj in enumerate(dst):
            a[jj] += move_amount * dst_slots[di]
        
        # Project
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        
        # Check improvement
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        elif obj > best_obj * 1.01:
            # Revert if too bad
            a = best_a.copy()
    
    return best_a, best_obj


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=50000.0, 
                 max_terms=400, momentum_init=0.92, lr_init=None):
    """
    LSE-smoothed gradient descent with adaptive parameters.
    Returns best sequence (normalized to sum=1) and its peak conv value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_init is None:
        lr = 4.0 / n
    else:
        lr = lr_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = float(beta_start)
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = lse_gradient(f, beta, max_terms=max_terms)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad /= grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr if improving consistently
            if no_improve == 0 and it % 50 == 0:
                lr = min(lr * 1.01, 10.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
                momentum = max(0.88, momentum - 0.002)
            if no_improve > 50:
                # Restart from best with perturbation
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.5 / n)
                beta = min(beta * 2.0, beta_end)
                momentum = 0.90
                no_improve = 0
        
        # Increase beta gradually
        if beta < beta_end:
            beta *= 1.005
        max_terms = min(500, int(100 + 400 * (beta / beta_end)))
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, top_k=1, lr_decay=0.999):
    """
    Subgradient descent on the exact max of convolution.
    Uses the structure: subgrad[i] = 2*a[peak-i] for valid i.
    Step: move in direction -subgrad, projected to simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    lr = 0.05
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)
    momentum = 0.85
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv, _ = subgrad_peak(f, top_k=top_k)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Accelerated subgradient
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= lr_decay
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.001)
                no_improve = 0
    
    return best_f, best_obj


def mixed_optimize(f0, time_limit):
    """
    Alternates between LSE gradient and exact subgradient steps.
    Periodically inserts greedy coordinate moves.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    beta = 50.0
    lr_lse = 3.0 / n
    lr_sg = 0.03
    momentum = 0.90
    v = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Every 50 iterations, do greedy coordinate moves
        if it % 50 == 0 and it > 0:
            f_greedy, obj_greedy = greedy_peak_reduction(f, n_steps=200, n_top=3)
            if obj_greedy < best_obj - 1e-14:
                best_obj = obj_greedy
                best_f = f_greedy.copy()
                f = f_greedy.copy()
                v = np.zeros(n)
                no_improve = 0
                continue
        
        # Every 100 iterations, try spectral perturbation
        if it % 100 == 0 and it > 0:
            f_spec = spectral_perturb(f, strength=0.05)
            obj_spec = fast_convolve(f_spec).max()
            if obj_spec < best_obj - 1e-10:
                best_obj = obj_spec
                best_f = f_spec.copy()
                f = f_spec.copy()
                v = np.zeros(n)
                no_improve = 0
        
        # Choose between LSE and subgradient
        if it % 3 != 0:
            # LSE step
            grad, conv = lse_gradient(f, beta, max_terms=300)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                no_improve += 10
                continue
            grad /= grad_norm
            step_lr = lr_lse
        else:
            # Subgradient step
            grad, conv, _ = subgrad_peak(f, top_k=2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                no_improve += 10
                continue
            grad /= grad_norm
            step_lr = lr_sg
        
        v = momentum * v - step_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                lr_lse *= 0.96
                lr_sg *= 0.96
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr_lse = max(lr_lse * 0.4, 0.5 / n)
                lr_sg = max(lr_sg * 0.4, 0.001)
                beta = min(beta * 1.5, 50000.0)
                no_improve = 0
        
        if beta < 50000.0:
            beta *= 1.003
    
    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
