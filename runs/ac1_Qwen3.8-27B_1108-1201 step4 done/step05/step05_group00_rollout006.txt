def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""

    rows = []
    cols = []

        je = min(n - 1, int(k))
        idx = np.arange(js, je + 1)
        grad[idx] += w_k * 2.0 * a[int(k) - idx]
    
    return grad


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with sparse LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    lp_cache_time = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP oracle (with time limit on each LP call)
        lp_start = time.monotonic()
        g = solve_lp_oracle_sparse(f, M * m_exp, n)
        lp_time = time.monotonic() - lp_start
        lp_cache_time = 0.8 * lp_cache_time + 0.2 * lp_time

        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.1, 3.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search with golden section
        best_alpha = 0.0
        best_val = M

        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                if hi - lo < 1e-4:
                    break
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        # Adaptive restart
        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.005 * min(1.0, no_improve / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def soft_peak_gradient_descent(a0, time_limit, T_init=0.001, lr_init=0.03, 
                                eps_init=0.05, verbose=False):
    """
    Gradient descent with adaptive soft-peak subgradient.
    The epsilon parameter controls how many peaks we focus on.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    eps = eps_init
    momentum = 0.9
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Compute soft subgradient
        grad = soft_peak_subgradient(conv, a, epsilon=eps)

        # Momentum update with backtracking
        best_candidate = a
        best_val = current_max

        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            v = momentum * v - trial_lr * grad
            candidate = project_simplex(a + v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            # Encourage slightly larger steps on improvement
            lr = min(lr * 1.01, best_lr * 1.5)
        else:
            no_improve += 1
            lr = max(lr * 0.97, best_lr * 0.1)

        # Adapt epsilon: start wide, narrow down
        if no_improve > 50:
            eps = max(eps * 0.9, 0.005)
        if no_improve > 150:
            eps = min(eps * 1.1, 0.3)
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.003 + 0.005 * min(1.0, (no_improve - 300) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = lr_init
            eps = eps_init
            no_improve = 0
            momentum = 0.9

        # Slowly increase momentum
        momentum = min(0.97, momentum + 0.00005)

    return best_a, best_obj


def game_iterate(a0, time_limit, T_init=0.001, verbose=False):
    """
    Game-theoretic primal-dual iteration with improved convergence.
    Uses exponential moving average of peak distributions for better dual updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    n_conv = 2 * n - 1
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Maintain EMA of peak distribution
    p_ema = np.ones(n_conv) / n_conv
    ema_alpha = 0.3

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T_history = []

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)

        # Update dual (peak distribution) with EMA
        scaled = conv / max(T, 1e-10)
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-15:
            p = np.ones(n_conv) / n_conv
        else:
            p = p / p_sum
        p_ema = (1 - ema_alpha) * p_ema + ema_alpha * p
        p_ema /= p_ema.sum()

        # Solve QP: min a'^T M(p_ema) a' s.t. a' >= 0, sum(a') = 1
        # KKT: 2 * M(p_ema) * a + lambda * 1 = 0 (on support), >= 0 (off support)
        # Use iterative method for large n
        a_new = _solve_qp_hankel(p_ema, a, n, max_iter=50)

        # Blend for stability
        blend = min(1.0, 0.3 + 0.01 * it)
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature
        if no_improve > 20:
            T *= 1.05
        if no_improve > 80:
            T *= 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            p_ema = np.ones(n_conv) / n_conv
            no_improve = 0

        T_history.append(T)

    return best_a, best_obj


def _solve_qp_hankel(p, a_init, n, max_iter=100):
    """
    Solve min a^T M(p) a s.t. a >= 0, sum(a) = 1.
    M(p)_{j,m} = p[j+m].
    Uses projected gradient with adaptive step size (faster than full active set for large n).
    """
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    lr = 1.0 / (2.0 * p.max() * n + 1e-10)
    
    for _ in range(max_iter):
        Ma = hankel_mv(p, a)
        # Gradient of a^T M(p) a is 2 * M(p) * a (since M is symmetric as p[j+m] = p[m+j])
        grad = 2.0 * Ma
        
        # Simple constraint: sum(a) = 1, a >= 0
        # Projected gradient: a_new = project_simplex(a - lr * (grad - mean(grad)))
        # Actually, for equality constraint sum(a)=1, the Lagrangian gives:
        # a_new = project_simplex(a - lr * grad)
        
        for trial in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial * grad)
            # Check if objective improved
            obj_old = np.dot(a, hankel_mv(p, a))
            obj_new = np.dot(candidate, hankel_mv(p, candidate))
            if obj_new <= obj_old + 1e-15:
                a = candidate
                break
        else:
            # Try smaller step
            a = project_simplex(a - lr * 0.01 * grad)
        
        # Adaptive lr
        Ma_new = hankel_mv(p, a)
        grad_norm = np.linalg.norm(2.0 * Ma_new - 2.0 * Ma)
        if grad_norm < 1e-8:
            break
        lr = lr * 0.999  # Very slow decay

    return a


def coord_descent_peaks(a0, time_limit, verbose=False):
    """
    Coordinate descent targeting the highest peaks.
    For each top peak, try to reduce it by adjusting contributing variables.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Find top peaks
        n_peaks = min(10, len(conv))
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]

        improved = False
        for k in top_peaks:
            k = int(k)
            # Variables affecting this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            
            # Try reducing each contributing variable slightly
            # and shifting mass to variables that don't contribute to this peak
            for j_try in range(js, je + 1, max(1, (je - js) // 5)):
                # Try reducing a[j_try] and adding to a random other position
                delta = 0.01 * a[j_try]
                if delta < 1e-10:
                    continue
                
                # Try a few target positions
                for frac in [0.25, 0.5, 0.75]:
                    target = int(frac * (n - 1))
                    if target == j_try:
                        target = (target + 1) % n
                    
                    a_try = a.copy()
                    a_try[j_try] -= delta
                    a_try[target] += delta
                    
                    if a_try[j_try] < 0:
                        continue
                    
                    c = fast_convolve(a_try)
                    if c.max() < current_max - 1e-12:
                        a = a_try
                        current_max = c.max()
                        improved = True
                        break
                if improved:
                    break
            
            if improved and current_max < best_obj - 1e-14:
                best_obj = current_max
                best_a = a.copy()
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng((it * 67 + 29) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
                continue
            if no_improve > 500:
                break

    return best_a, best_obj


def fine_perturbation_search(a0, time_limit, verbose=False):
    """
    Systematic fine perturbation search around the best solution.
    Tries many small random perturbations and keeps improvements.
    Uses adaptive perturbation scale.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    scale = 0.005
    it = 0
    no_improve = 0
    total_tried = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        # Batch perturbations
        n_batch = max(50, min(500, int((time_limit - (time.monotonic() - t0)) * 50)))
        
        for _ in range(n_batch):
            total_tried += 1
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            rng = np.random.default_rng((it * 1000 + _ + 7) % (2**31))
            # Mix of full and local perturbations
            if rng.random() < 0.3:
                # Local perturbation (only ±10 neighbors of random center)
                center = rng.integers(0, n)
                lo = max(0, center - 10)
                hi = min(n, center + 10)
                pert = np.zeros(n)
                pert[lo:hi] = rng.standard_normal(hi - lo) * scale
            else:
                # Full perturbation
                pert = rng.standard_normal(n) * scale
            
            candidate = project_simplex(best_a + pert)
            c = fast_convolve(candidate)
            val = c.max()
            
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = candidate
                no_improve = 0
                # Slightly increase scale on success
                scale = min(scale * 1.01, 0.02)
        
        if no_improve > 20:
            scale *= 0.9
            no_improve = 0
        if no_improve > 100:
            scale = 0.005  # Reset
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Anti-symmetric (high at edges, low at center - targets center peak)
    anti = np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 1)
    anti = anti ** 2 + 0.01
    inits.append(anti / anti.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search with careful time management.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Quick exploration of promising n values (~150s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()

    # Focus on parent n and a few nearby values
    candidate_ns = [n_parent]
    for n_try in [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4,
                  n_parent // 2, n_parent * 2]:
        if n_try not in candidate_ns and 64 <= n_try <= 2048:
            candidate_ns.append(n_try)

    for n_try in sorted(candidate_ns, key=lambda x: abs(x - n_parent)):
        remaining = deadline - time.monotonic() - 1400
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 30)
        print(f"  Quick FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"fw_n{n_try}")

        # Quick gradient
        remaining = deadline - time.monotonic() - 1350
        if remaining > 10:
            a_opt2, _ = soft_peak_gradient_descent(a_opt, time_limit=min(remaining, 15),
                                                    T_init=0.001, lr_init=0.03, eps_init=0.05)
            try_seq(a_opt2, f"grad_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Deep FW-LP on best (~400s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 1000
        if remaining > 30:
            t_alloc = min(remaining, 350)
            print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

            remaining = deadline - time.monotonic() - 700
            if remaining > 20:

                    remaining = deadline - time.monotonic() - 600
                    if remaining < 15:
                        break
                    f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                    a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(remaining, 60))

    print(f"  Phase 2 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    print("\n=== Phase 3: Soft-Peak Gradient ===")

                                                   T_init=0.001, lr_init=0.02, eps_init=0.05)

                                                   T_init=0.0005, lr_init=0.01, eps_init=0.02)

        if remaining > 20:

                                                   T_init=0.002, lr_init=0.03, eps_init=0.15)
            try_seq(a_opt, "softgrad_eps0.15")

    print(f"  Phase 3 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    print("\n=== Phase 4: Game Iteration ===")

            a_opt, _ = game_iterate(best_seq, time_limit=t_alloc, T_init=0.001)
            try_seq(a_opt, "game_T0.001")

        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Game iterate T=0.003, t={t_alloc:.0f}s")
            a_opt, _ = game_iterate(best_seq, time_limit=t_alloc, T_init=0.003)
            try_seq(a_opt, "game_T0.003")

    print(f"  Phase 4 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Coordinate descent on peaks (~250s) ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t_start = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 300
        if remaining > 15:
            t_alloc = min(remaining, 200)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent_peaks(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Fine perturbation search (~250s) ===
    print("\n=== Phase 6: Fine Perturbation ===")
    t_start = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 200)
            print(f"  Fine perturbation, t={t_alloc:.0f}s")
            a_opt, _ = fine_perturbation_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fine_pert")

    print(f"  Phase 6 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 7: Final alternating polish (~150s) ===
    print("\n=== Phase 7: Final Polish ===")
    t_start = time.monotonic()

    if best_seq is not None:
        # Alternate between different methods with very short time budgets
        methods = [
            ("fw", lambda a, t: fw_lp_optimize(a, t)),
            ("grad", lambda a, t: soft_peak_gradient_descent(a, t, T_init=0.0005, lr_init=0.01, eps_init=0.02)),
            ("game", lambda a, t: game_iterate(a, t, T_init=0.002)),
            ("fine", lambda a, t: fine_perturbation_search(a, t)),
        ]

        for rep in range(4):
            remaining = deadline - time.monotonic() - 30
            if remaining < 20:
                break
            t_alloc = min(remaining / 4.0, 40)
            if t_alloc < 5:
                break
            
            meth_name, meth_fn = methods[rep % len(methods)]
            print(f"  Polish round {rep+1} ({meth_name}), t={t_alloc:.0f}s")
            a_opt, _ = meth_fn(best_seq, t_alloc)
            try_seq(a_opt, f"polish_{rep}_{meth_name}")

    print(f"  Phase 7 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 8: Ultra-fine final search (remaining time) ===
    print("\n=== Phase 8: Ultra-Fine Final Search ===")
    t_start = time.monotonic()

    if best_seq is not None:
        remaining = max(1.0, deadline - time.monotonic())
        if remaining > 5:
            print(f"  Ultra-fine, t={remaining:.0f}s")
            a_opt, _ = fine_perturbation_search(best_seq, time_limit=remaining, verbose=True)
            try_seq(a_opt, "ultra_fine")

            # One more gradient pass
            remaining = max(1.0, deadline - time.monotonic())
            if remaining > 10:
                a_opt, _ = soft_peak_gradient_descent(best_seq, time_limit=remaining,
                                                       T_init=0.0003, lr_init=0.005, eps_init=0.01)
                try_seq(a_opt, "final_grad")

    print(f"  Phase 8 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
