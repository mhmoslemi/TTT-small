    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k=10, temp=0.001):
    """Compute weighted subgradient of max(conv) using softmax-weighted peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[0] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)

            continue

        m_end = min(n - 1, k)

        valid_k_minus_m = k - valid_m
        sg[valid_m] += 2.0 * w[i] * a[valid_k_minus_m]

"""
Multi-phase optimization for minimum peak auto-convolution sequences.
Uses mirror descent, reduced LP-FW, and coordinate descent with 
adaptive parameters and diverse initialization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient from softmax-weighted peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[0] = 1.0

        w /= ws

        valid_m = np.arange(m_start, m_end + 1)
        valid_km = k - valid_m
        sg[valid_m] += 2.0 * w[i] * a[valid_km]
    
    return sg

    # The most restrictive constraints are those where f has large values in the overlap.
    # For now, use all constraints but with a threshold on which to include.
    
    # Better: compute max possible (f*g)[k] for each k (when g is concentrated)
    # max_g (f*g)[k] = max_j f[j] (when g is a delta at position that makes k-j the argmax of f)
    # Actually (f*g)[k] = sum_j f[j] * g[k-j], so max over g (with sum g = 1, g >= 0) 
    # is max_j f[j] (achieved by g being a delta at position k-j* where j* = argmax f)
    
    # For the reduced LP, include constraints where the "potential" is high
    # Potential for constraint k: max_j f[j] for j in [max(0,k-n+1), min(n-1,k)]
    # This is expensive to compute for all k, so use a simpler criterion
    
    # Simple approach: use all constraints (original LP)
    # The LP solver should handle ~1663 constraints fine
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        cols = np.arange(j, j + n)
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f[cols - j]  # A_ub[k, j] = f[k-j] if valid
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g
    return None


def lp_fw_step(f, n):
    """One LP-FW iteration: solve LP and do line search."""
    g = solve_reduced_lp(f, n)
    if g is None:
        return f, fast_convolve(f).max()
    
    conv_f = fast_convolve(f)
    M = conv_f.max()
    
    # Line search
    best_alpha = 0.0
    best_val = M
    
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        return (1 - best_alpha) * f + best_alpha * g, best_val
    
    return f, M


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 10:
            break
        
        peak_k = int(np.argmax(conv))
        m_start = max(0, peak_k - n + 1)
        m_end = min(n - 1, peak_k)
        
        improved = False
        # Try modifying coordinates that contribute to the peak
        candidates = list(range(m_start, m_end + 1))
        np.random.shuffle(candidates)  # Actually use a seeded shuffle
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
        rng.shuffle(candidates)
        
        for j in candidates[:50]:  # Try up to 50 random peak-contributing coords
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-10:
                continue
            
            best_mult = 1.0
            best_val = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_mult = mult
            
            if best_mult != 1.0:
                new_val = old_val * best_mult
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - new_val) / denom
                    a[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                    best_obj = best_val
                    best_a = a.copy()
                    conv = fast_convolve(a)
                    improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Random perturbation
            rng2 = np.random.default_rng(int(time.monotonic() * 5000) % 999983)
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.001)
            conv = fast_convolve(a)
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(budget_s=1800):
    """
    Multi-phase optimization for minimum peak auto-convolution.
    Phases: diverse init + MD, intensive optimization (MD + LP-FW),
    length exploration, final polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score
    
    # --- Get parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    n_base = len(parent) if parent is not None else 832
    
    # --- Phase 1: Diverse initialization + quick optimization (300s) ---
    print("\n=== Phase 1: Diverse init + quick MD ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(300, rem * 0.25)
    
    # Generate diverse candidates
    candidates = []
    
    if parent is not None:
        candidates.append(("parent", parent.copy()))
    
    n1 = n_base
    # Cosine window
    x = np.linspace(0, np.pi, n1)
    cos_w = np.sin(x) ** 2
    candidates.append(("cosine", cos_w / cos_w.sum()))
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n1) / n1))
    candidates.append(("raised_cos", rc / rc.sum()))
    
    # Gaussian (centered)
    centers = np.arange(n1)
    gauss = np.exp(-0.5 * ((centers - n1/2) / (n1/4))**2)
    candidates.append(("gaussian", gauss / gauss.sum()))
    
    # U-shape (inverted Gaussian)
    u_shape = 1.0 / (gauss + 1e-10)
    candidates.append(("u_shape", u_shape / u_shape.sum()))
    
    # Random with smoothing
    rng_init = np.random.default_rng(42)
    rand = np.abs(rng_init.standard_normal(n1))
    # Smooth it
    kernel = np.ones(5) / 5
    rand_smooth = np.convolve(rand, kernel, mode='same')
    candidates.append(("random_smooth", rand_smooth / rand_smooth.sum()))
    
    # Super-Gaussian
    super_g = np.exp(-np.abs((centers - n1/2) / (n1/4))**4)
    candidates.append(("super_gauss", super_g / super_g.sum()))
    
    # Try different n
    for n_try in [n1 - 64, n1 + 64, n1 + 128]:
        if n_try < 128:
            continue
        if parent is not None:
            interp = interpolate_sequence(parent, n_try)
            candidates.append((f"interp_n{n_try}", interp))
    
    # Quick MD on each candidate (limited time)
    t_per_cand = t_phase1 / max(len(candidates), 1)
    
    for label, cand in candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(t_per_cand, rem - 30)
        if t_alloc < 5:
            break
        print(f"  MD on {label} (n={len(cand)}), t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(cand, time_limit=t_alloc, eta0=1.0,
                                   top_k=max(20, len(cand)//10), temp0=0.0005,
                                   temp_growth=1.003)
        try_seq(a_opt, f"md_{label}")
    
    # --- Phase 2: Intensive optimization (700s) ---
    print(f"\n=== Phase 2: Intensive optimization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(700, rem * 0.45)
    
    if best_seq is not None:
