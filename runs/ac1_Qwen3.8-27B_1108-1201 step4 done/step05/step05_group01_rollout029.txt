Let me implement the "away steps" (Lerer loss step or "Frank-Wolfe with away steps"):

The FW+away algorithm:
1. Find the LP vertex `g = argmin_{v in simplex} <g, v>` (or in our case, the LP solution)
2. Find the "away" vertex: `h = argmax_{v in V} <g, v>` where V is the set of vertices in the current "face" (or just the vertex of the simplex with the largest current coefficient)
3. Take a step: `a_new = (1 - gamma_1)*a + gamma_1*g - gamma_2*h` (with appropriate step sizes)

In our case, the "vertices" of the simplex are the standard basis vectors e_i. The "away" vertex is the e_i where a_i is largest.

Actually, the "away step" in the FW context for the simplex is:
- Forward vertex: the LP solution (or the vertex that minimizes the linear approximation)
- Away vertex: the standard basis vector e_j where j = argmax(a) (the coordinate with the largest value)

The update is:
`a_new = (1 - gamma)*a + gamma*fwd - gamma_away*away_dir`

where `away_dir = e_j` (the away vertex) and we need `a_new >= 0`.

This can significantly speed up convergence.

This is equivalent to:

where `T_k` is the matrix with `T_k[j] = f[k-j]` if valid, 0 otherwise.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with away steps,
projected gradient polish, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0. Returns (g, dual_y)."""
    # Primal: max sum(g) s.t. A_ub @ g <= b_ub, g >= 0
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.ineqlin  # res.ineqlin might not be available
    return None, None


def lp_fw_away(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with away steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        
        # Solve LP
        g = solve_lp_direction(f, M, n)
        if g is None:
            M *= 1.05
            stall_count += 1
            if stall_count > 50:
                break
            continue
        stall_count = 0
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check convergence
        if np.max(np.abs(g - f)) < 1e-12:
            stall_count += 1
            if stall_count > 100:
                break
            continue
        
        # Forward step with line search
        best_alpha_f = 0.0
        best_val_f = M
        fw_alpha = 2.0 / (it + 2.0)
        
        for alpha in [fw_alpha, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val_f:
                best_val_f = val
                best_alpha_f = alpha
        
        # Away step
        # The away vertex is e_j where j = argmax(f)
        j_away = np.argmax(f)
        away_dir = -np.eye(n)[j_away]  # -e_j
        # We want to move in direction (g - f) and also away from e_j
        # Combined direction: (1-beta)*(g-f) + beta*(-e_j) for some beta
        # But we need to keep f in the simplex
        
        # Try away step: f_new = f + alpha*(g-f) - gamma*e_j (normalized)
        # The away step is useful when f has a large component at j_away
        
        best_val = best_val_f
        best_f_new = f + best_alpha_f * (g - f)
        best_f_new = np.maximum(best_f_new, 0.0) / np.maximum(best_f_new.sum(), 1e-15)
        
        # Try combined forward + away steps
        if f[j_away] > 1.0 / n * 1.5:  # Only try away if the component is significantly above average
            for beta in [0.05, 0.1, 0.2]:
                for alpha in [best_alpha_f, 0.3, 0.5]:
                    # f_new = (1 - alpha - beta*f[j_away]) * f + alpha * g + beta * f[j_away] * (1 - e_j)
                    # Simpler: f_new = f + alpha*(g-f) - beta*f[j_away]*e_j, then renormalize
                    ft = f.copy()
                    ft = ft + alpha * (g - f)
                    ft[j_away] -= beta * f[j_away]
                    if ft[j_away] < 0:
                        ft[j_away] = 0
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-10:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_f_new = ft
        
        f = best_f_new
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            stall_count = 0
            if verbose and it % 100 == 0:
                obj = 2 * n * best_obj
                print(f"    LP-FW-away iter {it}: obj={obj:.8f}, M={best_obj:.10f}")
        else:
            stall_count += 1
            if stall_count > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                M = conv.max()
                stall_count = 0
    
    return best_f, best_obj


def projected_gradient_polish(f0, time_limit, verbose=False):
    """Projected gradient with exact line search on the max of quadratic forms."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    alpha = 1.0 / (2.0 * f.max() * n)  # Initial step size estimate
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active set (indices where conv is close to max)
        tol = max(1e-12, 1e-8 * M)
        active = np.where(conv >= M - tol)[0]
        
        # Compute subgradient (average of active gradients)
        g = np.zeros(n)
        for k in active:
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            indices = np.arange(m_lo, m_hi + 1)
            g[indices] += 2.0 * f[k - indices]
        g /= len(active)
        
        # Projected gradient step with backtracking line search
        improved = False
        for _ in range(20):
            f_new = project_simplex(f - alpha * g)
            if np.max(np.abs(f_new - f)) < 1e-14:
                break
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            if M_new < M - 1e-14:
                f = f_new
                improved = True
                if M_new < best_obj:
                    best_obj = M_new
                    best_f = f.copy()
                alpha = min(alpha * 1.2, 1.0)
                break
            else:
                alpha *= 0.5
        
        if not improved:
            stall += 1
            if stall > 50:
                # Try a different step size
                alpha = 1.0 / (n * f.max() + 1e-10)
                stall = 0
            if time.monotonic() - t0 > time_limit * 0.9:
                break
        else:
            stall = 0
        
        if verbose and it % 100 == 0:
            print(f"    PG iter {it}: M={best_obj:.10f}, alpha={alpha:.6f}")
    
    return best_f, best_obj


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with away steps,

Key innovations: away-step FW for faster convergence, active-set-aware subgradients,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP direction finding.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns (g, active_mask) where active_mask indicates tight constraints.
    """
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        slack = res.slack
        active_mask = slack < 1e-10 * M
        return res.x, active_mask
    return None, None

def compute_subgradient(f, conv, n, top_k=10, temperature=1e-6):
    """Compute weighted subgradient of max(conv) using top-k peaks with softmax weights."""
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / max(temperature, 1e-12)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        indices = np.arange(m_lo, m_hi + 1)
        g[indices] += 2.0 * w[i] * f[k - indices]

    return g


def projected_gradient_polish(f0, time_limit, verbose=False, top_k=20, temp0=1e-5):
    """Projected gradient with adaptive step size and softmax-weighted subgradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    it = 0
    stall = 0
    temp = temp0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Compute subgradient
        g = compute_subgradient(f, conv, n, top_k=top_k, temperature=temp)

        # Projected gradient step with backtracking
        improved = False
        cur_alpha = alpha
        for _ in range(25):
            f_new = project_simplex(f - cur_alpha * g)
            if np.max(np.abs(f_new - f)) < 1e-15:
                break
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            if M_new < M - 1e-14:
                f = f_new
                improved = True
                if M_new < best_obj - 1e-15:
                    best_obj = M_new
                    best_f = f.copy()

def cosine_init(n, freq, amp):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a


def sine_ramp_init(n, freq, side='both'):
    """Sine-ramped initialization (tapered edges)."""
    j = np.arange(n)
    if side == 'both':
        w = np.sin(np.pi * (j + 1) / (n + 1))
    elif side == 'left':
        w = np.linspace(0.1, 1.0, n)
    else:
        w = np.linspace(1.0, 0.1, n)
    a = w * (1.0 + 0.3 * np.cos(2.0 * np.pi * freq * j / n))
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_int = np.interp(new_x, old_x, f)
    f_int = np.maximum(f_int, 0.0)
    ss = f_int.sum()
    if ss > 1e-15:
        f_int /= ss
    return f_int

            s_tmp, sq_tmp = check_and_update(f_int, best_score, best_seq, f"intel_n{target_n}", verbose=False)
            if np.isfinite(s_tmp) and s_tmp < best_score + 0.1:
                t_alloc = min(rem * 0.35, 120)
                a_opt, _ = lp_fw_away(f_int, time_limit=t_alloc)
                best_score, best_seq = check_and_update(a_opt, best_score, best_seq, f"lpfw_n{target_n}")
            rem = deadline - time.monotonic() - 150
            if rem > 20 and best_seq is not None:
                a_opt, _ = projected_gradient_polish(best_seq, time_limit=min(rem * 0.3, 60),
                                                      top_k=max(10, len(best_seq) // 12), temp0=1e-6)
                best_score, best_seq = check_and_update(a_opt, best_score, best_seq, f"pg_n{target_n}")

    # === Phase 6: Diversified initializations ===
    print(f"\n=== Phase 6: New initializations (best={best_score:.8f}) ===")
    inits = [
        (n_parent, cosine_init(n_parent, 1, 0.5)),
        (n_parent, cosine_init(n_parent, 2, 0.4)),
        (n_parent, sine_ramp_init(n_parent, 2, 'both')),
        (n_parent, cosine_init(n_parent, 3, 0.3)),
    ]
    # Also try blending with current best
    if best_seq is not None:
        blend = 0.5 * best_seq + 0.5 * cosine_init(n_parent, 1, 0.6)
        blend = np.maximum(blend, 0) / max(blend.sum(), 1e-15)
        inits.append((n_parent, blend))

    for idx, (n_init, a_init) in enumerate(inits):
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        s_tmp, _ = check_and_update(a_init, best_score, best_seq, f"init_{idx}", verbose=False)
        if np.isfinite(s_tmp) and s_tmp < best_score + 0.05:
            t_alloc = min(rem * 0.25, 80)
            a_opt, _ = lp_fw_away(a_init, time_limit=t_alloc)
            best_score, best_seq = check_and_update(a_opt, best_score, best_seq, f"lpfw_init{idx}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Cycle: PG -> LP-FW -> PG with decreasing time
        for cycle in range(4):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 60)
            if t_alloc < 10:
                break
            # Alternate between methods
            if cycle % 2 == 0:
                a_opt, _ = projected_gradient_polish(best_seq, time_limit=t_alloc,
                                                      top_k=max(8, len(best_seq) // 20),
                                                      temp0=1e-7)
                best_score, best_seq = check_and_update(a_opt, best_score, best_seq, f"final_pg{cycle}")
            else:
                a_opt, _ = lp_fw_away(best_seq, time_limit=t_alloc)
                best_score, best_seq = check_and_update(a_opt, best_score, best_seq, f"final_lpfw{cycle}")

        # Tiny perturbations with quick optimization
        rem = deadline - time.monotonic() - 10
        for p in range(6):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = projected_gradient_polish(f_pert, time_limit=min(rem, 20),
                                                  top_k=10, temp0=1e-7)
            best_score, best_seq = check_and_update(a_opt, best_score, best_seq, f"pert{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
