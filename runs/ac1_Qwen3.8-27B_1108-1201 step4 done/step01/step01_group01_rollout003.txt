def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (csv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    theta = (csv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_score(a):
    n = len(a)

def _subgrad_topM(a, b, M, temp_factor=0.1):
    """Compute weighted subgradient using top M peaks."""
    n = len(a)

        hi = min(n - 1, k)
        if lo <= hi:

            grad[idx] += w * 2.0 * a[k - idx]

from scipy.optimize import linprog

def _solve_lp_direction(f, M):
    """Solve LP: max 1^T g s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for i in range(lo, hi+1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])

    c = -np.ones(n)  # Minimize -1^T g = maximize 1^T g
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _build_conv_matrix(n, f):
    """Build the convolution matrix for fixed f."""
    # For each k, the valid i range is [max(0,k-n+1), min(n-1,k)]
    # Total non-zeros: sum_k (min(n-1,k) - max(0,k-n+1) + 1) = n^2
    
    # Use a more efficient construction
    rows_list = []
    cols_list = []
    vals_list = []
    
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        count = hi - lo + 1
        rows_list.extend([k]*count)
        cols_list.extend(range(lo, hi+1))
        vals_list.extend(f[lo:hi+1])
    
    return csr_matrix((np.array(vals_list), (np.array(rows_list), np.array(cols_list))), shape=(2*n-1, n))
