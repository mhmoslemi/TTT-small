```python
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    """Compute the score quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def solve_lp(f, M):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    # Build sparse matrix A where A[k,j] = f[k-j] if 0 <= k-j < n
    # Column j has entries at rows j, j+1, ..., j+n-1 with values f[0], ..., f[n-1]
    rows = np.repeat(np.arange(n), n)  # for each j, rows j to j+n-1
    # Actually: for column j, row i = j + k for k = 0, ..., n-1
    # So rows = [0,1,...,n-1, 1,2,...,n, 2,3,...,n+1, ..., n-1, n, ..., 2n-2]
    j_vals = np.tile(np.arange(n), n)  # column index
    k_vals = np.tile(np.arange(n), n)  # offset within column
    row_vals = j_vals + k_vals  # actual row = j + k
    
    A = sp.coo_matrix((f[k_vals], (row_vals, j_vals)), shape=(2*n-1, n)).tocsr()
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def lp_step(f, t, M=None):
    """One LP step: find g, mix with f."""
    conv = np.convolve(f, f)
    if M is None:
        M = conv.max()
    g = solve_lp(f, M)
    if g is None:
        return f
    gs = g.sum()
    if gs < 1e-10:
        return f
    g = g / gs
    f_new = (1-t)*f + t*g
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s

def make_initial(n, rng, method='uniform'):
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'ramp':
        a = np.linspace(1, 2, n)
    elif method == 'sine':
        a = 1 + 0.3*np.sin(2*np.pi*rng.random()*n/np.arange(1,n+1))  # not right
        # Better: a = 1 + 0.3 * sin(2*pi*j/n + phase)
        phase = rng.random() * 2 * np.pi
        a = 1 + 0.3 * np.sin(2*np.pi*np.arange(n)/n + phase)
    elif method == 'two_humps':
        x = np.arange(n)
        a = np.exp(-0.5*((x-n/4)/(n/8))**2) + np.exp(-0.5*((x-3*n/4)/(n/8))**2)
    elif method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'plateau':
        a = np.ones(n)
        a[:n//4] *= 1.5
        a[3*n//4:] *= 1.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a
```

```python
"""
LP-based iterative optimization with multi-phase search to minimize
the autocorrelation peak score = 2*n*max(f*f)/(sum(f))^2.

Strategy:
  Phase 1: LP-driven optimization from existing best and multiple initializations
  Phase 2: Peak-targeted local search (directed mass redistribution)
  Phase 3: Diversification via structured randomization + re-optimization
  Phase 4: Final polish and verification
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2 quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse (2n-1)xn matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    j_vals = np.tile(np.arange(n), n)
    k_vals = np.repeat(np.arange(n), n)
    row_vals = j_vals + k_vals
    data = f[k_vals]
    A = sp.coo_matrix((data, (row_vals, j_vals)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_max_sum(f, M, time_limit=20.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_min_peak(f, target_sum, time_limit=20.0):
    """
    Alternative LP: minimize t s.t. (f*g)[k] <= t for all k, sum(g) = target_sum, g >= 0.
    This directly minimizes the peak of (f*g) for a fixed total mass.
    """
    n = len(f)
    A = build_conv_matrix(n, f)
    # Variables: g[0..n-1], t. Total: n+1 variables.
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality: A@g - t <= 0 for all k -> [A | -I] @ [g; t] <= 0
    A_ineq = sp.hstack([A, -sp.identity(2 * n - 1)], format='csr')
    b_ineq = np.zeros(2 * n - 1)
    # Equality: sum(g) = target_sum -> [1...1 | 0] @ [g; t] = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            t_val = res.x[n]
            s = g.sum()
            if s < 1e-10:
                return None
            return g / s, float(t_val)
    except Exception:
        pass
    return None


def lp_step_maxsum(f, t_mix, time_limit=20.0):
    """LP step: maximize sum(g) subject to (f*g) <= M, then mix."""
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f
    g = solve_lp_max_sum(f, M, time_limit)
    if g is None:
        return f
    gs = g.sum()
    if gs < 1e-10:
        return f
    g = g / gs
    if np.linalg.norm(g - f) < 1e-10:
        return f
    f_new = (1.0 - t_mix) * f + t_mix * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def lp_step_minpeak(f, t_mix, time_limit=20.0):
    """LP step: minimize peak of (f*g) for fixed sum, then mix."""
    result = solve_lp_min_peak(f, 1.0, time_limit)
    if result is None:
        return f
    g, t_val = result
    if np.linalg.norm(g - f) < 1e-10:
        return f
    f_new = (1.0 - t_mix) * f + t_mix * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def peak_reduction_step(f, alpha=0.1):
    """
    Targeted peak reduction: find the peak of f*f, then reduce entries
    that contribute most to the peak, redistributing mass to low-contribution entries.
    """
    n = len(f)
    conv = np.convolve(f, f)
    k_peak = int(np.argmax(conv))
    # Contribution of f[j] to the peak: f[j] * f[k_peak - j] for valid j
    j_valid = np.arange(n)
    k_valid = k_peak - j_valid
    # Only valid where 0 <= k_valid < n
    mask = (k_valid >= 0) & (k_valid < n)
    contributions = np.zeros(n)
    contributions[mask] = f[j_valid[mask]] * f[k_valid[mask]]
    
    total_contrib = contributions.sum()
    if total_contrib < 1e-10:
        return f
    
    # Reduce high-contribution entries, increase low-contribution entries
    # Target: move mass from top contributors to bottom
    sorted_idx = np.argsort(contributions)
    n_reduce = max(1, n // 5)  # reduce top 20%
    n_inject = max(1, n // 5)  # inject into bottom 20%
    
    top_idx = sorted_idx[-n_reduce:]
    bot_idx = sorted_idx[:n_inject]
    
    f_new = f.copy()
    # Reduce top contributors
    reduce_amt = alpha * f[top_idx].sum()
    f_new[top_idx] -= alpha * f[top_idx]
    # Do the same for their "partners" at the peak
    for j in top_idx:
        k = k_peak - j
        if 0 <= k < n and k != j:
            f_new[k] -= alpha * 0.5 * f[k]
    # Redistribute to low contributors
    f_new[bot_idx] += reduce_amt / n_inject
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def block_perturb(f, rng, width_min=2, width_max=None, sigma=0.3):
    """Random multiplicative block perturbation."""
    n = len(f)
    if width_max is None:
        width_max = max(width_min + 1, n // 10)
    cand = f.copy()
    w = int(rng.integers(width_min, min(width_max + 1, n)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def make_initial(n, rng, method='uniform'):
    """Generate initial non-negative sequence of length n, normalized to sum 1."""
    x = np.arange(n)
    if method == 'uniform':
        a = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    elif method == 'asymmetric_ramp':
        a = 1.0 + 0.5 * x / max(1, n - 1)
    elif method == 'sine_asym':
        phase = rng.random() * 2 * np.pi
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x / n + phase)
    elif method == 'two_bumps':
        c1 = n // 4 + int(rng.integers(-n//10, n//10 + 1))
        c2 = 3 * n // 4 + int(rng.integers(-n//10, n//10 + 1))
        s = n // 10
        a = np.exp(-0.5 * ((x - c1) / s) ** 2) + 0.7 * np.exp(-0.5 * ((x - c2) / s) ** 2) + 0.1
    elif method == 'plateau_edge':
        a = np.ones(n)
        a[:n//3] *= 1.3
        a[2*n//3:] *= 1.3
    elif method == 'random_smooth':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    elif method == 'geometric':
        ratio = 1.0 + rng.random() * 0.1
        a = 0.5 + 0.5 * (x / max(1, n - 1)) ** rng.random()
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def run_lp_optimization(f, deadline, method='maxsum', t_init=0.3, t_min=0.02,
                        t_decay=0.997, lp_time=15.0):
    """Run LP iterations until deadline. Returns best found."""
    f = f.copy()
    t = t_init
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    rng = np.random.default_rng(it + hash(f.tobytes()) % 10000)
    
    while time.monotonic() < deadline:
        it += 1
        if method == 'maxsum':
            f_new = lp_step_maxsum(f, t, lp_time)
        elif method == 'minpeak':
            f_new = lp_step_minpeak(f, t, lp_time)
        else:
            # Alternate
            f_new = lp_step_maxsum(f, t, lp_time) if it % 2 == 0 else lp_step_minpeak(f, t, lp_time)
        
        moved = (not np.allclose(f_new, f, atol=1e-12))
        if not moved:
            # Try peak reduction
            f_new = peak_reduction_step(f, alpha=np.exp(-stall * 0.1))
            moved = (not np.allclose(f_new, f, atol=1e-12))
        if not moved:
            # Random perturbation to escape
            f_new = block_perturb(f, rng, width_min=2, width_max=max(5, len(f)//15),
                                   sigma=0.2 + 0.3 * min(1.0, stall / 30.0))
        
        f = f_new
        t = max(t_min, t * t_decay)
        
        sc = fast_score(f)
        if np.isfinite(sc):
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall > 80:
                # Aggressive restart from best with perturbation
                f = block_perturb(best, rng, width_min=1, width_max=max(5, len(f)//5),
                                   sigma=0.4)
                t = t_init * 0.5
                stall = 0
    return best, best_sc
```
