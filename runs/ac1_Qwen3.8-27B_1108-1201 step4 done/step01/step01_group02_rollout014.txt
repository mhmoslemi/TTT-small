"""
Improved LP-based iterative refinement with 1D line search for minimizing
peak auto-convolution of non-negative sequences.

Key improvements over previous version:
- 1D golden-section line search after each LP step (optimal step size)
- Asymmetric initializations (geometric, tilted bumps)
- Better time management with adaptive LP solve timeout

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    """Build sparse convolution matrix: row k, col j = a[j] if k-j in [0,n-1]."""
    n = len(a)
    n_rows = 2 * n - 1
    # For each column j, it contributes to rows j, j+1, ..., j+n-1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_direction(a, M, timeout=30.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized g or None."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            b = np.maximum(0.0, res.x)
            s = b.sum()
            if s > 1.0 + 1e-10:
                return b / s
    except Exception:
        pass
    return None


def line_search_optimal_t(a, g, n):
    """
    Find optimal t in [0,1] minimizing max_k[(1-t)^2 A_k + 2(1-t)t B_k + t^2 C_k]
    where A=conv(a,a), B=conv(a,g), C=conv(g,g).
    Assumes sum(a)=sum(g)=1.
    """
    A = np.convolve(a, a)
    B = np.convolve(a, g)
    C = np.convolve(g, g)
    len_conv = len(A)

    def f_of_t(t):
        vals = (1.0 - t) ** 2 * A + 2.0 * (1.0 - t) * t * B + t ** 2 * C
        return vals.max()

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

    for _ in range(60):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)

    t_star = (lo + hi) / 2.0
    # Also check endpoints
    f0 = A.max()
    f1 = C.max()
    ft = f_of_t(t_star)
    if f0 <= ft and f0 <= f1:
        return 0.0
    if f1 <= ft and f1 <= f0:
        return 1.0
    return t_star

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):

            else:
                stagnant += 5

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP v2 done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    while time.monotonic() - start_time < time_limit_s:
        # Random block perturbation
        width = max(2, int(rng.integers(n // 100, n // 5)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.08, size=stop - start))
        # Occasional global tilt
        if rng.random() < 0.2:
            x = np.arange(n, dtype=np.float64) / max(1, n - 1)
            candidate *= (1.0 + 0.1 * rng.normal() * (2.0 * x - 1.0))
        candidate = normalize_nonneg(candidate)
        s = score_sequence(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate.copy()

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def frequency_domain_init(n, rng):
    """
    Initialize using frequency domain: choose a magnitude spectrum that's
    relatively flat, then take the inverse transform (with non-negativity projection).
    """
    # Start with a smooth random spectrum
    freq = np.fft.rfft(np.zeros(n))
    k = np.arange(len(freq))
    # Low-pass with some randomness
    bandwidth = rng.uniform(0.3, 0.7) * n
    mask = np.exp(-0.5 * (k / (bandwidth / 2)) ** 2)
    freq_mag = mask * np.exp(rng.normal(0, 0.3, size=len(freq)))
    freq_phase = np.zeros(len(freq))
    a = np.fft.irfft(freq_mag * np.exp(1j * freq_phase), n=n)
    a = np.maximum(0, a)
    a = normalize_nonneg(a)
    if a.sum() < 0.01:
        return initial_raised_cosine(n)
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP with 1D line search, multiple initializations and lengths.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 30.0)

    best_score = float('inf')
    best_seq = None

    # Try to use parent sequence
    parent_seq = None
    try:
        if 'height_sequence_1' in globals():
            parent_seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
            if parent_seq is not None and len(parent_seq) > 0:
                parent_seq = normalize_nonneg(parent_seq)
                parent_score = score_sequence(parent_seq)
                print(f"Parent sequence: n={len(parent_seq)}, score={parent_score:.6f}")
                if parent_score < best_score:
                    best_score = parent_score
                    best_seq = parent_seq.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Define configurations: (n, init_type, param, time_frac)
    # Allocate more time to promising configs
    configs = []

    # If we have a parent, try refining it at various lengths
    if parent_seq is not None:
        pn = len(parent_seq)
        configs.append((pn, 'parent', None, 0.20))

    # Standard initializations at multiple lengths
    for n in [500, 800, 1200, 2000, 3000]:
        configs.append((n, 'raised_cosine', None, 0.10))
        configs.append((n, 'geometric', None, 0.08))
        configs.append((n, 'tilted_bump', None, 0.07))
        configs.append((n, 'shifted_gaussian', None, 0.06))

    # Random smooth inits
    for n in [500, 1000, 2000]:
        configs.append((n, 'random', None, 0.05))

    # Frequency domain
    for n in [500, 1000]:
        configs.append((n, 'freq_domain', None, 0.04))

    # Shuffle configs to explore diversity early, but keep parent first
    if len(configs) > 1:
        # Keep first (parent) fixed, shuffle the rest
        rest = configs[1:]
        rng.shuffle(rest)
        configs = [configs[0]] + rest

    total_time_frac = sum(c[3] for c in configs)
    # Scale time to fit budget
    available = budget_s * 0.80  # leave 20% for final polish
    scale = available / total_time_frac if total_time_frac > 0 else 1.0

    for n, init_type, param, time_frac in configs:
        elapsed = time.monotonic() - total_start
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        time_limit = min(time_frac * scale * budget_s, remaining * 0.85)
        if time_limit < 15.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time_limit={time_limit:.0f}s, elapsed={elapsed:.0f}s ===")

        if init_type == 'parent' and parent_seq is not None:
            # Resize parent to target n via interpolation
            if n == len(parent_seq):
                a = parent_seq.copy()
            else:
                old_n = len(parent_seq)
                old_x = np.linspace(0, 1, old_n)
                new_x = np.linspace(0, 1, n)
                a = np.interp(new_x, old_x, parent_seq)
                a = np.maximum(0, a)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'geometric':
            a = initial_geometric(n)
        elif init_type == 'tilted_bump':
            a = initial_tilted_bump(n, tilt=rng.uniform(0.2, 0.6))
        elif init_type == 'shifted_gaussian':
            a = initial_shifted_gaussian(n, center_frac=rng.uniform(0.4, 0.7),
                                         sigma_frac=rng.uniform(0.2, 0.35))
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        elif init_type == 'freq_domain':
            a = frequency_domain_init(n, rng)
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        if init_score < best_score:
            best_score = init_score
            best_seq = a.copy()
            print(f"  *** NEW BEST (init): score={best_score:.6f}")

        # Run LP refinement
        a_refined, score = lp_refine(a, time_limit_s=time_limit, verbose=(n <= 1000),
                                     rng=rng)

        if score < best_score:
            best_score = score
            best_seq = a_refined.copy()
            print(f"  *** NEW BEST (LP): score={score:.6f}, n={n}")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")

        if remaining > 60.0:
            # Local search
            polish_time = min(remaining * 0.3, 180.0)
            polished, score = local_search_refine(best_seq, time_limit_s=polish_time,
                                                  rng=rng, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = polished.copy()
                print(f"  *** After LS: score={best_score:.6f}")

        # More LP refinement on the best
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30.0:
            lp_time = remaining2 * 0.7
            polished2, score2 = lp_refine(best_seq, time_limit_s=lp_time, verbose=True,
                                          rng=rng)
            if score2 < best_score:
                best_score = score2
                best_seq = polished2.copy()
                print(f"  *** After final LP: score={best_score:.6f}")

        # One more round of LS + short LP
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20.0:
            polished3, score3 = local_search_refine(best_seq, time_limit_s=min(remaining3 * 0.5, 60.0),
                                                    rng=rng, verbose=False)
            if score3 < best_score:
                best_score = score3
                best_seq = polished3.copy()
                print(f"  *** After final LS: score={best_score:.6f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    result = [float(x) for x in best_seq]
    verify_score = evaluate_sequence(result)
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    print(f"Verified score: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    if np.isinf(verify_score):
        # Shouldn't happen, but fallback
        print("  WARNING: verify returned inf, using fallback")
        result = [1.0 / 500.0] * 500
        print(f"  Fallback verified: {evaluate_sequence(result):.6f}")

    return result
