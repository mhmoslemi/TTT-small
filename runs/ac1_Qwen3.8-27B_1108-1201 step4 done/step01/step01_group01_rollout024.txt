    """Solve the dual LP to find y that minimizes M*sum(y) subject to constraints."""
    n = len(f)
    # Dual: min M * sum_{k=0}^{2n-2} y_k
    # subject to: for each j: sum_k f_{k-j} * y_k >= 1, y_k >= 0
    # Convert to: min M * sum(y) subject to -A^T y <= -1, y >= 0
    
    # A^T[j,k] = f[k-j] if 0 <= k-j < n, else 0
    # So constraint j: sum_k A^T[j,k] * y_k >= 1
    # i.e., -sum_k A^T[j,k] * y_k <= -1

    L = nfft(n)

    R = np.real(ifft(P * np.conj(A)))
    return R[:n]


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = _fft_convolve(a, a)
    return 2.0 * n * np.max(b)


def _smoothed_objective(a, tau):
    """(1/tau)*log(sum exp(tau*b_k)) - stable version."""
    b = _fft_convolve(a, a)
    b_max = np.max(b)
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -700, 700)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return b_max + log_sum / tau


def _smoothed_gradient(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a, centered for sum constraint."""
    n = len(a)
    b = _fft_convolve(a, a)
    b_max = np.max(b)
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    # grad_j = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    grad = 2.0 * _fft_cross_correlate(w, a)
    grad -= np.mean(grad)
    return grad


def _fista_optimize(a_init, deadline, tau_init=3.0, tau_max=500.0,
                    eta_init=2.0, momentum_init=0.0,
                    reheat_every=800, max_iter_per_reheat=50000):
    """FISTA with adaptive tau for smoothed objective minimization."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_normalized(best_a)

    tau = tau_init
    eta = eta_init
    momentum = momentum_init
    momentum_target = 0.9

    y = a.copy()
    t_k = 1.0
    prev_a = a.copy()
    no_improve = 0
    iter_count = 0
    smooth_obj_history = []

    while time.monotonic() < deadline:
        # Check time more frequently
        if iter_count % 100 == 0 and time.monotonic() >= deadline:
            break

        # Compute current score
        b = _fft_convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute smoothed gradient at y
        grad_y = _smoothed_gradient(y, tau)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a + ((t_k - 1.0) / t_new) * (a - prev_a)
        prev_a = a.copy()
        t_k = t_new

        # Project y_new onto simplex
        y_new = _project_simplex(y_new, 1.0)

        # Gradient step
        step = eta / max(tau, 1.0)
        # Limit step magnitude
        grad_norm = np.linalg.norm(grad_y)
        if grad_norm < 1e-15:
            y = a
            continue
        a_new = y_new - step * grad_y / grad_norm
        a_new = _project_simplex(a_new, 1.0)

        # Optional: verify improvement in smoothed objective (every 10th iter)
        if iter_count % 10 == 0:
            obj_new = _smoothed_objective(a_new, tau)
            obj_old = _smoothed_objective(a, tau)
            if obj_new > obj_old + 1e-3:
                # Step too large, try half
                a_new = y_new - 0.5 * step * grad_y / grad_norm
                a_new = _project_simplex(a_new, 1.0)

        a = a_new
        y = y_new
        iter_count += 1

        # Adaptive tau: increase when stuck
        if no_improve > reheat_every:
            tau = min(tau * 1.08, tau_max)
            eta = max(eta * 0.9, 0.1)
            no_improve = 0
            # Reset momentum
            prev_a = a.copy()
            t_k = 1.0

        # Full reheat: reset everything and perturb
        if tau >= tau_max * 0.95 and no_improve > reheat_every * 3:
            # Perturb and restart
            sigma = 0.05 / np.sqrt(n)
            a = a + sigma * np.random.randn(n)
            a = _project_simplex(a, 1.0)
            y = a.copy()
            prev_a = a.copy()
            t_k = 1.0
            tau = tau_init
            eta = eta_init
            no_improve = 0
            iter_count = 0

        # Occasionally try a larger momentum
        if iter_count % 5000 == 0:
            # Decay and reset momentum periodically
            t_k = max(t_k * 0.5, 1.0)

    return best_a, best_score


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_smooth_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_four_level(n, eps):
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    return a / s


def _make_sparse_anti(n, eps, density=0.7):
    """Anti-symmetric with some zeros for sparsity."""
    a = _make_two_level(n, eps)
    # Zero out some elements in the second half
    mask = np.random.random(n // 2) > density
    a[n // 2:][mask] = 0.0
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _perturb_and_reopt(a, rng, scale, deadline):
    """Perturb and run a short optimization."""
    a_pert = a + scale * rng.standard_normal(len(a))
    a_pert = _project_simplex(a_pert, 1.0)
    return _fista_optimize(a_pert, deadline, tau_init=50.0, tau_max=200.0, eta_init=0.5)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase FFT-accelerated FISTA with adaptive log-sum-exp smoothing.
    Phase 1: Quick multi-start exploration.
    Phase 2: Deep FISTA optimization with adaptive temperature.
    Phase 3: Fine polishing with high temperature and small steps.
    Phase 4: Diversification via perturbation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except (NameError, UnboundLocalError):
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Normalize parent
    a_parent = parent.copy()
    s_parent = np.sum(a_parent)
    if s_parent > 1e-10:
        a_parent /= s_parent
    else:
        a_parent = np.ones(n) / n

    global_best = a_parent.copy()
    global_best_score = _score_normalized(global_best)
    print(f"Initial score (parent): {global_best_score:.6f}, n={n}")

    # === Phase 1: Multi-start quick optimization (first ~200s) ===
    print(f"Phase 1: Multi-start exploration")
    phase1_deadline = time.monotonic() + 150.0

    starts = [
        ("parent", a_parent),
        ("two_level_0.8", _make_two_level(n, 0.8)),
        ("two_level_1.0", _make_two_level(n, 1.0)),
        ("smooth_0.9", _make_smooth_anti(n, 0.9)),
        ("four_level_0.7", _make_four_level(n, 0.7)),
        ("random", None),  # filled below
    ]
    # Random start
    a_rand = rng.random(n)
    a_rand /= np.sum(a_rand)
    starts[5] = ("random", a_rand)

    for name, a_start in starts:
        if time.monotonic() >= phase1_deadline:
            break
        t_sub = min(30.0, phase1_deadline - time.monotonic())
        if t_sub < 2.0:
            break
        best_a, best_s = _fista_optimize(a_start, time.monotonic() + t_sub,
                                         tau_init=3.0, tau_max=100.0, eta_init=2.0)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  {name}: NEW BEST {global_best_score:.6f}")
        else:
            print(f"  {name}: {best_s:.6f} (best={global_best_score:.6f})")

    # === Phase 2: Deep optimization on best (next ~900s) ===
    print(f"\nPhase 2: Deep FISTA optimization, current best={global_best_score:.6f}")
    phase2_deadline = time.monotonic() + 900.0

    best_a, best_s = _fista_optimize(global_best, phase2_deadline,
                                     tau_init=5.0, tau_max=500.0, eta_init=2.0,
                                     reheat_every=1000)
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"  Phase 2 complete: NEW BEST {global_best_score:.6f}")
    else:
        print(f"  Phase 2 complete: no improvement (best={global_best_score:.6f})")

    # === Phase 3: Fine polishing (next ~300s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\nPhase 3: Polishing with high tau, {remaining:.0f}s remaining")
        polish_deadline = min(time.monotonic() + remaining * 0.5, deadline)

        # Use very high tau for sharp max approximation
        best_a, best_s = _fista_optimize(global_best, polish_deadline,
                                         tau_init=100.0, tau_max=1000.0,
                                         eta_init=0.5, reheat_every=500)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  Phase 3: NEW BEST {global_best_score:.6f}")
        else:
            print(f"  Phase 3: no improvement (best={global_best_score:.6f})")

    # === Phase 4: Diversification (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nPhase 4: Diversification, {remaining:.0f}s remaining")
        n_perturb = max(1, int(remaining / 80))
        per_perturb = remaining / n_perturb

        for i in range(n_perturb):
            if time.monotonic() >= deadline:
                break
            # Vary perturbation scale
            scale = 0.02 / np.sqrt(n) * (1 + 0.5 * i)
            sub_deadline = min(time.monotonic() + per_perturb * 0.8, deadline)
            best_a, best_s = _perturb_and_reopt(global_best, rng, scale, sub_deadline)
            if best_s < global_best_score:
                global_best = best_a.copy()
                global_best_score = best_s
                print(f"  Perturb {i+1}: NEW BEST {global_best_score:.6f}")

    # === Final: verify and return ===
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f} (n={n})")
    return result
