```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
on the probability simplex. The LP finds the largest non-negative direction g
whose cross-convolution with the current point a stays below the current peak.
Combined with mirror descent polishing and multi-scale search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def cross_convolve(a, g):
    """Linear cross-convolution of a and g using FFT."""
    n = len(a)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    result = np.fft.irfft(A * G, fft_size)
    return result[:size]

def solve_direction_lp(a, M):
    """
    Solve LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(a)
    # Constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # Number of constraints: 2n-1
    n_constraints = 2 * n - 1
    
    # Build constraint matrix (sparse would be better for large n, but let's start dense)
    # A[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    A_ub = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j in j_start..j_end
            # When j = j_start, k-j = k-j_start
            # When j = j_end, k-j = k-j_end
            A_ub[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(n_constraints, M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n
```
