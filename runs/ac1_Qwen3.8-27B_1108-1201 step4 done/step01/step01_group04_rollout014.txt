```python
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses alternating LP-based optimization (block coordinate descent on the peak convolution),
with FFT-accelerated convolution for large sequences and multi-length exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _lp_step(f, S_target, solver='ECOS'):
    """
    Solve: min M s.t. convolve(f, g) <= M, g >= 0, sum(g) = S_target.
    Returns (g, M).
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # convolve(f, g)[k] = sum_i f[i] * g[k-i] for valid i
    # This is a matrix-vector product: A @ g where A[k, i] = f[k-i] if valid
    # Build A as a (2n-1) x n matrix
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    # For efficiency, use the structure: A is a "convolution matrix" (toeplitz-like)
    # A[k, i] = f[max(0, k-i) ] if k-i >= 0 and k-i < n... 
    # Actually A[k,i] = f[k-i] if 0 <= k-i <= n-1, else 0
    # This means row k has f[0], f[1], ..., f[n-1] shifted
    
    # Let's build it explicitly for clarity (n up to ~2048, so matrix is ~4095 x 2048)
    # For larger n, we'd need a more efficient approach
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, i] = f[k - i]
    
    constraints = [
        A @ g <= M,
        cp.sum(g) == S_target,
        M >= 0
    ]
    
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is None:
            return None, None
        g_val = np.asarray(g.value).flatten()
        g_val = np.maximum(g_val, 0)
        M_val = float(M.value)
        return g_val, M_val
    except Exception:
        return None, None


def _lp_step_scipy(f, S_target):
    """
    LP step using scipy. Minimize M s.t. A@g <= M, g >= 0, sum(g) = S_target.
    """
    n = len(f)
    # Variables: g (n vars) + M (1 var) = n+1 vars
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - M <= 0 for each row of A
    # A is (2n-1) x n, we extend it to (2n-1) x (n+1) with last column = -1
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k, i] = f[k - i]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(g) = S_target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            M_val = max(float(res.x[-1]), 0)
            return g_val, M_val
    except Exception:
        pass
    return None, None
```

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses alternating LP (block coordinate descent) with CVXPY, multi-length exploration,
and gradient-based polishing. Starts from provided height_sequence_1.
"""

import numpy as np
import time
import cvxpy as cp


def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) * target / len(a)
    return a * (target / S)


def _build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            A[k, i] = f[k - i]
    return A


def _lp_minimize_peak(f, S_target, solver='ECOS', verbose=False):
    """
    Solve: min M s.t. convolve(f, g)[k] <= M for all k, sum(g) = S_target, g >= 0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(f)
    
    # Build constraint matrix
    A = _build_conv_matrix(f)
    
    # CVXPY formulation
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [
        A @ g <= M,
        cp.sum(g) == S_target,
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=solver)
        gv = np.asarray(g.value).flatten()
        if gv is None or not np.all(np.isfinite(gv)):
            return None, None
        gv = np.maximum(gv, 0)
        S = gv.sum()
        if S < 1e-12:
            return None, None
        gv = gv * (S_target / S)
        Mv = float(M.value) if M.value is not None else float((A @ gv).max())
        return gv, Mv
    except Exception:
        return None, None


def _gradient_step(a, lr, peak_indices=None, n_peaks=1):
    """
    Gradient step on the peak of convolve(a,a).
    If peak_indices is None, uses the top n_peaks indices.
    """
    n = len(a)
    c = np.convolve(a, a)
    
    if peak_indices is None:
        # Get top n_peaks indices
        peak_indices = np.argsort(c)[-n_peaks:]
    
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[int(k) - i]
    
    # Project out mean (fixed sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0)
    return a


def _alternating_lp(a_init, deadline, max_iters=2000, blend_schedule=None, verbose=True):
    """
    Alternating LP optimization.
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, 1000).copy()
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)
    
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    if verbose:
        print(f"  Start: n={n}, score={best_score:.6f}")
    
    # Precompute nothing (we rebuild A each step since a changes)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        
        # LP step: find g minimizing max convolve(a, g)
        g, M = _lp_minimize_peak(a, S_target)
        if g is None:
            if verbose and it % 100 == 0:
                print(f"    LP failed at iter {it}")
            break
        
        # Check if g is close to a (convergence)
        if np.allclose(g, a, atol=1e-6):
            if verbose:
                print(f"    Converged at iter {it}")
            break
        
        # Blend: a_new = (1-t)*a + t*g
        if blend_schedule is None:
            t = max(0.1, 1.0 - 0.1 * it / 100.0)
        else:
            t = blend_schedule(it)
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = _normalize(a_new, S_target)
        
        score = _compute_obj(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    iter {it}: score={score:.6f}, M={M:.4f}, t={t:.3f}")
        else:
            # Try smaller blend
            improved = False
            for t2 in [t * 0.5, t * 0.1, 0.05, 0.01]:
                a_try = (1.0 - t2) * a + t2 * g
                a_try = np.maximum(a_try, 0)
                a_try = _normalize(a_try, S_target)
                s_try = _compute_obj(a_try)
                if s_try < best_score:
                    best_score = s_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    if verbose and (it % 50 == 0 or it < 5):
                        print(f"    iter {it}: score={s_try:.6f} (t={t2:.3f})")
                    break
            if not improved:
                # Try gradient step to escape
                a_try = _gradient_step(a, 0.05)
                a_try = _normalize(a_try, S_target)
                s_try = _compute_obj(a_try)
                if s_try < best_score:
                    best_score = s_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    # Stuck, try a different approach: use g directly
                    a = _normalize(g, S_target)
                    score = _compute_obj(a)
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
        
        if verbose and (it + 1) % 200 == 0:
            print(f"    iter {it+1}: best={best_score:.6f}")
    
    return best_a, best_score


def _gradient_polish(a_init, deadline, max_iters=5000, verbose=True):
    """Polish with gradient descent using soft peak."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, 1000).copy()
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)
    
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        
        # Adaptive learning rate
        lr = 0.1 * (1.0 + np.cos(np.pi * it / max_iters)) * 0.5
        lr = max(lr, 0.001)
        
        # Soft peak: use top 5 peaks with softmax weights
        c = np.convolve(a, a)
        tau = 5.0 + 10.0 * (it / max_iters)
        weights = np.exp(tau * (c - c.max()) / c.max())
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(len(c)):
            if weights[k] < 1e-6:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += weights[k] * 2.0 * a[k - i]
        
        grad /= len(c)  # Normalize
        grad -= grad.mean()
        
        a = a - lr * grad
        a = np.maximum(a, 0)
        a = _normalize(a, S_target)
        
        if (it + 1) % 500 == 0:
            score = _compute_obj(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    final_score = _compute_obj(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses alternating LP (block coordinate descent on peak convolution), subgradient
polishing, multi-length exploration, and adaptive restart. Starts from
height_sequence_1 when available.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) * (target / len(a))
    return a * (target / S)


def _build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = f[k - j]
    return A


def _lp_step_cvxpy(f, S_target, solver='ECOS'):
    """Solve LP: min M s.t. convolve(f,g) <= M, sum(g) = S_target, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == S_target]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
        if g.value is None or M.value is None:
            return None, None
        gv = np.asarray(g.value, dtype=np.float64).flatten()
        if not np.all(np.isfinite(gv)):
            return None, None
        gv = np.maximum(gv, 0.0)
        S = gv.sum()
        if S < 1e-12:
            return None, None
        gv = gv * (S_target / S)
        Mv = float(M.value)
        return gv, Mv
    except Exception:
        return None, None


def _lp_step_scipy(f, S_target):
    """Fallback LP using scipy linprog."""
    n = len(f)
    A = _build_conv_matrix(f)
    # Variables: [g_0,...,g_{n-1}, M], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            gv = np.maximum(res.x[:n], 0.0)
            S = gv.sum()
            if S < 1e-12:
                return None, None
            gv = gv * (S_target / S)
            Mv = float(res.x[-1])
            return gv, Mv
    except Exception:
        pass
    return None, None


def _lp_step(f, S_target):
    """Try CVXPY solvers in order, fall back to scipy."""
    for solver in ['ECOS', 'SLSQP', 'SCS']:
        if solver == 'SLSQP':
            continue
        result = _lp_step_cvxpy(f, S_target, solver=solver)
        if result[0] is not None:
            return result
    return _lp_step_scipy(f, S_target)


def _subgradient_step(a, lr, n_top=1):
    """One subgradient step on peak of convolve(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    # Get top n_top peak indices
    if n_top >= len(c):
        indices = np.arange(len(c))
    else:
        indices = np.argpartition(c, -n_top)[-n_top:]
    indices = indices[np.argsort(c[indices])[::-1]]
    # Weight by softmax
    vals = c[indices]
    tau = 2.0
    w = np.exp(tau * (vals - vals.max()))
    w = w / w.sum()
    grad = np.zeros(n)
    for idx, wt in zip(indices, w):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += wt * 2.0 * a[k - i]
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return a


def _alternating_optimize(a0, deadline, max_iters=5000, verbose=True):
    """Alternating LP optimization with adaptive blending."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)
    best_a = a.copy()
    best_score = _compute_obj(a)
    if verbose:
        print(f"    Start: n={n}, score={best_score:.6f}")

    lp_fail_count = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 15:
            break

        # Get LP solution
        g, M = _lp_step(a, S_target)
        if g is None:
            lp_fail_count += 1
            if lp_fail_count > 20:
                if verbose:
                    print(f"    Too many LP failures, stopping.")
                break
            # Fall back to subgradient step
            a = _subgradient_step(a, 0.05, n_top=3)
            a = _normalize(a, S_target)
            continue
        lp_fail_count = 0

        # Check convergence
        diff = np.max(np.abs(g - a))
        if diff < 1e-7:
            if verbose:
                print(f"    Converged at iter {it}")
            break

        # Adaptive blend parameter
        if it < 20:
            t = 1.0  # Initially take the LP solution fully
        elif it < 100:
            t = max(0.3, 1.0 - 0.7 * (it - 20) / 80.0)
        else:
            t = max(0.05, 0.3 * (1.0 - (it - 100) / 4000.0))

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_target)
        score = _compute_obj(a_new)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and (it < 10 or it % 100 == 0):
                print(f"    iter {it}: score={score:.6f}, M={M:.4f}, t={t:.3f}, diff={diff:.2e}")
        else:
            # Try reduced blends
            improved = False
            for t2 in [t * 0.5, t * 0.1, 0.02, 0.005]:
                a_try = (1.0 - t2) * a + t2 * g
                a_try = np.maximum(a_try, 0.0)
                a_try = _normalize(a_try, S_target)
                s_try = _compute_obj(a_try)
                if s_try < best_score - 1e-10:
                    best_score = s_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    if verbose and (it < 10 or it % 100 == 0):
                        print(f"    iter {it}: score={s_try:.6f} (t={t2:.4f})")
                    break
            if not improved:
                # Try subgradient step to escape
                a_try = _subgradient_step(a, 0.03, n_top=5)
                a_try = _normalize(a_try, S_target)
                s_try = _compute_obj(a_try)
                if s_try < best_score - 1e-10:
                    best_score = s_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    # Use g directly (it minimizes max convolve(a, g))
                    a = _normalize(g, S_target)
                    score = _compute_obj(a)
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()

            stall_count += 1
            if stall_count > 300 and verbose:
                print(f"    Stalled at iter {it}, best={best_score:.6f}")
                stall_count = 0

    # Final check
    final_score = _compute_obj(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    End: best_score={best_score:.6f}")
    return best_a, best_score


def _subgradient_polish(a0, deadline, max_iters=10000, verbose=True):
    """Polish using subgradient method with adaptive learning rate."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    S_target = np.sqrt(2.0 * n)
    a = _normalize(a, S_target)
    best_a = a.copy()
    best_score = _compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        lr = 0.08 * (1.0 + np.cos(np.pi * it / max_iters)) * 0.5
        lr = max(lr, 0.002)
        n_top = min(1 + it // 500, 5)
        a = _subgradient_step(a, lr, n_top=n_top)
        a = _normalize(a, S_target)
        if (it + 1) % 1000 == 0:
            score = _compute_obj(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    print(f"    subgrad iter {it}: score={score:.6f}")

    final_score = _compute_obj(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def _prepare_starts(init_seq, lengths, rng):
    """Prepare starting sequences for different lengths."""
    starts = []
    for L in lengths:
        if len(init_seq) >= L:
            a0 = init_seq[:L].copy()
        else:
            a0 = np.pad(init_seq, (0, L - len(init_seq)), mode='edge')
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-6:
            a0 = rng.random(L) * 0.5 + 0.25
        starts.append(a0)
        # Also a perturbed version
        perturbed = a0 * (1.0 + 0.1 * rng.random(L))
        perturbed = np.maximum(perturbed, 0.0)
        starts.append(perturbed)
    return starts
```
