    # Bounds: b_i in [0, None], t in [0, None]
    lb = [0.0] * n + [0.0]
    ub = [None] * (n + 1)
    bounds = list(zip(lb, ub))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            b = np.maximum(res.x[:n], 0.0)
            bs = b.sum()
            if bs > 1e-15:
                b = b * (S / bs)
            return b
    except Exception:
        pass
    return None


def run_lp_refine(a_init, n, max_time, rng, K_start=8, K_max=40):
    """LP-based active set refinement. Returns (best_a, best_maxconv)."""
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a_init, n)
    best_a = a.copy()
    best_mc = fft_convolve_self(a).max()
    start = time.monotonic()
    K = K_start
    stall = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        mc = c.max()
        if mc < best_mc - 1e-9:
            best_mc = mc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Active set: top K lags
        active = np.argsort(c)[::-1][:K]

        # Solve LP
        b = solve_lp_step(a, active, n, S)
        if b is not None:
            b = np.clip(b, 0, 1000)
            cb = fft_convolve_self(b)
            mb = cb.max()
            if mb < mc:
                # Full or damped move
                for tau in [1.0, 0.7, 0.5, 0.3]:
                    a_try = (1.0 - tau) * a + tau * b
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * (S / s_try)
                    ct = fft_convolve_self(a_try)
                    if ct.max() < mc:
                        a = a_try
                        break
                else:
                    # Try subgradient fallback
                    grad = compute_subgradient(a, c, K)
                    gn = np.linalg.norm(grad)
                    if gn > 1e-14:
                        a = project_simplex(a - 0.05 * S / n * grad / gn, S)
                K = min(K + 1, K_max)
            else:
                # B didn't help; try damped move
                improved = False
                for tau in [0.5, 0.3, 0.1]:
                    a_try = (1.0 - tau) * a + tau * b
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * (S / s_try)
                    ct = fft_convolve_self(a_try)
                    if ct.max() < mc:
                        a = a_try
                        improved = True
                        break
                if not improved:
                    # Subgradient step
                    grad = compute_subgradient(a, c, K)
                    gn = np.linalg.norm(grad)
                    if gn > 1e-14:
                        a = project_simplex(a - 0.05 * S / n * grad / gn, S)

                    if stall > 30:
                        # Perturb and restart
                        a = normalize_seq(a + rng.laplace(0, S / (20.0 * n), n), n)
                        stall = 0
                        K = K_start
        else:
            # LP failed; use subgradient
            grad = compute_subgradient(a, c, K)
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                a = project_simplex(a - 0.05 * S / n * grad / gn, S)
            if stall > 30:
                a = normalize_seq(a + rng.laplace(0, S / (20.0 * n), n), n)
                stall = 0
                K = K_start

        # Periodic print
        if int(time.monotonic() - start) % 60 < 0.1:
            print("  LP iter: mc=%.6f best=%.6f K=%d stall=%d" % (mc, best_mc, K, stall))

    return best_a, best_mc


def run_fine_polish(a_init, n, max_time, rng):
    """Fine polishing with very small Adam steps and top-2 subgradient."""
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a_init, n)
    best_a = a.copy()
    best_mc = fft_convolve_self(a).max()
    start = time.monotonic()
    t = 0
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.99, 1e-8
    base_lr = 0.05 * S / n

    while time.monotonic() - start < max_time:
        # Run batches of 100 steps
        batch_end = min(t + 100, t + 100)
        for _ in range(100):
            t += 1
            c = fft_convolve_self(a)
            mc = c.max()
            if mc < best_mc - 1e-11:
                best_mc = mc
                best_a = a.copy()
            grad = compute_subgradient(a, c, top_k=3)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            grad /= gn
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            lr_t = base_lr / (1.0 + 0.01 * t)
            a = project_simplex(a - lr_t * mh / (np.sqrt(vh) + eps), S)
            if time.monotonic() - start > max_time:
                break
        # Periodic perturbation to escape local minima
        if t % 500 == 0 and t > 0:
            a = project_simplex(a + rng.normal(0, 0.005 * S / n, n), S)
    return best_a, best_mc


def gs_initialization(n, n_iters=200, rng=None):
    """Gerchberg-Saxton style initialization for flat autocorrelation."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = rng.random(n) + 0.5
    for i in range(n_iters):
        A = fft(a)
        mag = np.abs(A)
        ang = np.angle(A)
        # Target: flatten magnitude (keep DC)
        target_mag = np.full(n, mag[1:].mean())
        target_mag[0] = mag[0]
        new_A = target_mag * np.exp(1j * ang)
        a = np.real(ifft(new_A))
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-15:
            a = rng.random(n) + 0.5
    return a


def get_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if given.size >= 64 and np.sum(given) > 0:
            inits.append(("given", normalize_seq(given, given.size)))
    except Exception:
        pass
    # GS
    inits.append(("gs_%d" % n, gs_initialization(n, 300, rng)))
    # Random
    inits.append(("rand_%d" % n, rng.random(n) + 0.1))
    # Smooth (raised cosine)
    tt = np.linspace(0, 1, n)
    inits.append(("smooth_%d" % n, 0.5 + 0.5 * np.cos(2 * np.pi * tt) ** 2))
    # Bimodal
    inits.append(("bimodal_%d" % n,
                  np.exp(-4.0 * ((tt - 0.3) / 0.2) ** 2) + np.exp(-4.0 * ((tt - 0.7) / 0.2) ** 2)))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Returns best sequence as a Python list."""
    global _global_best_score, _global_best_seq
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0  # 15s buffer for finalization

    global_best_score = np.inf
    global_best_a = None

    def update_best(a, score):
        nonlocal global_best_score, global_best_a
        if np.isfinite(score) and score < global_best_score:
            global_best_score = score
            global_best_a = a.copy()
            print("  NEW BEST: %.6f (len=%d) at t=%.0fs" % (score, len(a), time.monotonic() - start_time))

    # === Phase 1: Exploration (40% of time) ===
    phase1_budget = 0.4 * (budget_s - 15.0)
    print("Phase 1: Exploration (budget=%.0fs)" % phase1_budget)

    # Primary length
    n_primary = 8192
    inits = get_initializations(n_primary, rng)

    # Also try a smaller length
    n_small = 4096
    inits.append(("rand_small", rng.random(n_small) + 0.1))

    n_inits = len(inits)
    per_init_time = max(5.0, phase1_budget / n_inits)

    for name, arr in inits:
        if time.monotonic() > deadline - 0.4 * (budget_s - 15.0) - 5.0:
            break
        n = len(arr)
        if n < 64:
            continue
        t0 = time.monotonic()
        remaining = per_init_time - (t0 - t0)
        # Quick Adam
        a_best, mc_best = run_adam(arr, n, remaining, rng, top_k=8)
        S = np.sqrt(2.0 * n)
        score = 2.0 * n * mc_best / S ** 2
        update_best(a_best, score)
        print("  %s: mc=%.4f score=%.6f time=%.1fs" % (name, mc_best, score, time.monotonic() - t0))

    # === Phase 2: LP Refinement (45% of time) ===
    phase2_start = time.monotonic()
    phase2_budget = 0.45 * (budget_s - 15.0) - (phase2_start - start_time)
    phase2_budget = max(60.0, 0.45 * (budget_s - 15.0))
    remaining_for_p2 = min(phase2_budget, deadline - time.monotonic() - 20.0)
    print("\nPhase 2: LP Refinement (budget=%.0fs)" % remaining_for_p2)

    if global_best_a is not None and remaining_for_p2 > 30:
        n = len(global_best_a)
        a_lp, mc_lp = run_lp_refine(global_best_a, n, remaining_for_p2, rng,
                                    K_start=8, K_max=50)
        S = np.sqrt(2.0 * n)
        score = 2.0 * n * mc_lp / S ** 2
        update_best(a_lp, score)
        print("  LP done: mc=%.6f score=%.6f" % (mc_lp, score))

    # === Phase 3: Fine Polishing (remaining time) ===
    remaining_for_p3 = deadline - time.monotonic() - 10.0
    print("\nPhase 3: Fine Polishing (budget=%.0fs)" % remaining_for_p3)

    if global_best_a is not None and remaining_for_p3 > 20:
        n = len(global_best_a)
        # Multiple polish attempts with different perturbations
        n_attempts = max(1, int(remaining_for_p3 / 60))
        for attempt in range(n_attempts):
            if time.monotonic() > deadline - 10:
                break
            a_polish = global_best_a.copy()
            if attempt > 0:
                # Small perturbation for diversity
                a_polish = normalize_seq(a_polish + rng.laplace(0, 0.01 * np.sqrt(2 * n) / n, n), n)
            a_best_p, mc_p = run_fine_polish(a_polish, n, min(remaining_for_p3 / (n_attempts - attempt + 1), deadline - time.monotonic() - 10), rng)
            S = np.sqrt(2.0 * n)
            score_p = 2.0 * n * mc_p / S ** 2
            update_best(a_polish, score_p)  # also check the starting point
            update_best(a_best_p, score_p)
            print("  Polish %d: mc=%.6f score=%.6f" % (attempt, mc_p, score_p))

    # === Phase 4: Final LP pass on the best ===
    remaining = deadline - time.monotonic() - 5.0
    if global_best_a is not None and remaining > 15:
        n = len(global_best_a)
        a_final, mc_final = run_lp_refine(global_best_a, n, remaining, rng,
                                          K_start=10, K_max=40)
        S = np.sqrt(2.0 * n)
        score = 2.0 * n * mc_final / S ** 2
        update_best(a_final, score)
        print("  Final LP: score=%.6f" % score)

    # === Finalization ===
    if global_best_a is None:
        # Fallback
        global_best_a = np.ones(128, dtype=np.float64)

    # Clip and verify
    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(128, dtype=np.float64)

    # Verify with evaluate_sequence
    try:
        final_score = evaluate_sequence([float(x) for x in global_best_a])
        print("\nFinal verified score: %.6f" % final_score)
        print("Sequence length: %d" % len(global_best_a))
    except Exception as e:
        print("Verification error: %s" % str(e))
        # Return a safe fallback
        return [1.0] * 128

    return [float(x) for x in global_best_a]
