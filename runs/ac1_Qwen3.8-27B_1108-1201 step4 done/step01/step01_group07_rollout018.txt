```python
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0"""
    n = len(f)
    # Constraint matrix: A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            if 0 <= k-i < n:
                A[k, i] = f[k-i]
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    return result.x
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
iterations with FFT-accelerated gradient descent as fallback.
The LP finds the best response g maximizing sum(g) subject to (a*g)[k] <= M,
which drives the convolution peak down. Combined with multi-length scanning,
adaptive step sizes, and random restarts.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    k_idx = np.arange(2 * n - 1)[:, None]  # (2n-1, 1)
    i_idx = np.arange(n)[None, :]           # (1, n)
    diff = k_idx - i_idx                    # (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[valid] = f[diff[valid]]
    return A

def lp_response(f, M):
    """Solve LP: max 1^T g s.t. ConvMatrix(f) @ g <= M*1, g >= 0.
    Returns optimal g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None
