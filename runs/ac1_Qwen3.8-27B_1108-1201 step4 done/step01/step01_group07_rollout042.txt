    Adam optimizer on the probability simplex.
    Objective: alpha_l2 * sum(conv^2) + alpha_max * (1/T)*log(sum(exp(T*conv)))
    """
    a = np.maximum(a0, 0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    best_a = a.copy()

    fft_size_grad = 1 << (3 * n - 2 - 1).bit_length()

Wait, I realize there's an issue. The Adam optimizer with the projection (clip + renormalize) is not the same as projected Adam. The renormalization step can undo the optimization progress. Let me think about this more carefully.

The constraint is: a >= 0, sum(a) = 1. The objective is scale-invariant, so we can work with sum(a) = 1 (the simplex). The projection onto the simplex is: clip to 0, then renormalize (divide by sum). This is the Euclidean projection only if we're on the face of the simplex where the non-negativity constraints are active.

Actually, the Euclidean projection onto the simplex {x: x>=0, sum(x)=1} is NOT simply clip+renormalize. The correct projection involves a more complex computation. However, for optimization purposes, clip+renormalize is a common approximation that works well in practice.

Alternatively, I can work in log-space: let a[j] = exp(z[j]) / sum(exp(z[k])). Then I optimize z unconstrained, and a is always on the simplex. This is the "softmax parameterization" and is equivalent to mirror descent with entropy regularization (which is what the previous approach did).

def adam_optimize_log(a0, time_limit, T=50.0, lr=0.05, alpha_l2=1.0, alpha_max=0.1):

    a[j] = exp(z[j]) / sum(exp(z[k]))
    """
    a = np.maximum(a0, 1e-10).astype(np.float64)
    n = len(a)
    a /= a.sum()
    z = np.log(a)  # log-space parameters
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    # Precompute
    size = 2 * n - 1
    fft_size_conv = 1 << (size - 1).bit_length()
    fft_size_grad = 1 << (3 * n - 2 - 1).bit_length()

    best_obj = 2.0 * n * fast_convolve(a).max()

    max_patience = 300
    lr_current = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Get current a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        
        # Compute conv
        A = np.fft.rfft(a, fft_size_conv)
        conv = np.fft.irfft(A * A, fft_size_conv)[:size]
        
        # Compute gradients wrt a
        ar = a[::-1]
        AR = np.fft.rfft(ar, fft_size_grad)
        C = np.fft.rfft(conv, fft_size_grad)
        conv_ac = np.fft.irfft(AR * C, fft_size_grad)
        g_l2 = 4.0 * conv_ac[n-1:2*n-1]
        
        scaled = T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        W = np.fft.rfft(w, fft_size_grad)
        AR2 = np.fft.rfft(ar, fft_size_grad)
        conv_aw = np.fft.irfft(AR2 * W, fft_size_grad)
        g_max = 2.0 * conv_aw[n-1:2*n-1]
        
        grad_a = alpha_l2 * g_l2 + alpha_max * g_max
        
        # Chain rule: da/dz[j] = a[j] * (delta[jk] - a[k])
        # So grad_z[k] = sum_j grad_a[j] * da[j]/dz[k]
        # = sum_j grad_a[j] * a[j]*(delta[jk] - a[k])
        # = a[k]*(grad_a[k] - sum_j grad_a[j]*a[j])
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)
        
        # Adam update on z
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr_current * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        a_eval = np.exp(z - z.max())
        a_eval /= a_eval.sum()
        conv_eval = fast_convolve(a_eval)
        obj = 2.0 * n * conv_eval.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_eval.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            lr_current *= 0.5
            T = min(T + 20, 500)
            if lr_current < 1e-6:
                break
    
    return best_a, best_obj

Explores multiple sequence lengths (128-4096) and diverse initializations

Key mathematical insight: the gradient of sum(conv^2) encourages flat

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

    """
    Compute gradient of alpha_l2*sum(conv^2) + alpha_max*soft_max_T(conv)
    Returns (gradient_vector, convolution).

    A = np.fft.rfft(a, fft_size_conv)

    scaled = T * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws > 1e-30:
        w /= ws
    else:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0

    W = np.fft.rfft(w, fft_size_grad)
    AR2 = np.fft.rfft(ar, fft_size_grad)
    conv_aw = np.fft.irfft(AR2 * W, fft_size_grad)
    g_max = 2.0 * conv_aw[n - 1:2 * n - 1]

    grad = alpha_l2 * g_l2 + alpha_max * g_max
    return grad, conv


def adam_log_optimize(a0, time_limit, T0=30.0, lr0=0.02, alpha_l2=1.0, alpha_max=0.5,
                      beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer in log-space (softmax parameterization).
    a[j] = exp(z[j]) / sum(exp(z[k])). Always stays on the simplex.
    Minimizes alpha_l2*sum(conv^2) + alpha_max*soft_max_T(conv).
    """
    a = np.maximum(a0, 1e-15).astype(np.float64)
    n = len(a)
    a /= a.sum()
    z = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    size = 2 * n - 1
    fft_size_conv = 1 << (size - 1).bit_length()
    fft_size_grad = 1 << (3 * n - 2).bit_length()

    # Initial evaluation
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = obj

    T = T0
    lr = lr0
    t0 = time.monotonic()
    patience = 0
    max_patience = 400
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        n_iter += 1

        # Current a from z (numerically stable)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()

        # Compute gradient
        grad_a, conv = compute_combined_gradient(a, T, alpha_l2, alpha_max,
                                                  fft_size_conv, fft_size_grad, size)

        # Chain rule: grad_z[k] = a[k] * (grad_a[k] - sum_j grad_a[j]*a[j])
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate (every iteration for simplicity)
        z_max = z.max()
        a_eval = np.exp(z - z_max)
        a_eval /= a_eval.sum()
        conv_eval = fast_convolve(a_eval)
        obj = 2.0 * n * conv_eval.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_eval.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive schedule
        if patience >= max_patience:
            patience = 0
            lr *= 0.5
            T = min(T * 1.8, 800.0)
            alpha_max = min(alpha_max * 1.3, 5.0)
            if lr < 1e-7:
                break

        # Safety: check time every 100 iterations
        if n_iter % 100 == 0 and time.monotonic() - t0 >= time_limit:
            break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for a given sequence length."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians with different widths
    for sf in [0.08, 0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Cosine bump
    cos = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos = np.maximum(cos, 1e-10)
    cos /= cos.sum()
    inits['cosine'] = cos

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.2)

    # Multi-Gaussian (random centers)
    n_peaks = rng.integers(2, 5)
    centers = rng.uniform(0.1 * n, 0.9 * n, size=n_peaks)
    widths = rng.uniform(0.03 * n, 0.15 * n, size=n_peaks)
    r = np.zeros(n)
    for c, w in zip(centers, widths):
        r += np.exp(-0.5 * ((x - c) / w) ** 2)
    r = np.maximum(r, 1e-10)
    r /= r.sum()
    inits['multi_gauss'] = r

    # "Checkered" pattern (alternating large/small)
    check = np.ones(n)
    check[n % 2 == 1] = 0.3
    check = np.maximum(check, 1e-10)
    check /= check.sum()
    inits['checkered'] = check

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi = np.maximum(bi, 1e-10)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Parent (if available)
    if parent is not None:
        if parent.size == n:
            pp = np.maximum(parent, 1e-10)
            pp /= pp.sum()
            inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 1e-10)
            pp /= pp.sum()
            inits['parent_interp'] = pp

    return inits


def run_search(seed=42, budget_s=1800):
    """Two-phase search: quick scan then deep optimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30  # Leave 30s safety margin

    best_sequence = None
    best_score = float('inf')

    # Try to access parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    if parent is not None:
        print(f"Parent sequence: length={parent.size}, score={evaluate_sequence(parent.tolist()):.6f}")

    # Define search lengths
    lengths = [128, 256, 512, 1024, 2048]

    # Build config list: (n, init_name, a0, T, lr, alpha_l2, alpha_max)
    configs = []
    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits.items():
            # L2-dominant (encourages flatness)
            configs.append((n, name, a0, 20.0, 0.015, 1.0, 0.3))
            # Balanced
            configs.append((n, name, a0, 50.0, 0.008, 0.7, 0.8))
            # Max-focused
            configs.append((n, name, a0, 100.0, 0.005, 0.3, 2.0))
            # Small learning rate, high T (fine-tuning)
            configs.append((n, name, a0, 80.0, 0.003, 1.0, 1.0))

    print(f"Total configs: {len(configs)}")

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    phase1_time = min(500.0, (deadline - time.monotonic()) * 0.35)
    time_per_config = max(2.0, phase1_time / len(configs))

    scan_results = []
    for i, (n, name, a0, T, lr, a_l2, a_max) in enumerate(configs):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_config, deadline - time.monotonic() - 60)
        if t_alloc < 1.5:
            break

        a_opt, obj_opt = adam_log_optimize(a0, time_limit=t_alloc, T0=T, lr0=lr,
                                            alpha_l2=a_l2, alpha_max=a_max)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt.copy(), T, lr, a_l2, a_max))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                if (i + 1) % 30 == 0 or score < 1.52:
                    print(f"  [{i+1}/{len(configs)}] n={n}, {name}, T={T:.0f}, score={score:.6f}")

    # Phase 1.5: Quick re-scan with top results as new initializations
    print(f"After Phase 1: best={best_score:.6f}")
    scan_results.sort(key=lambda x: x[0])

    # Add cross-length transfer: use best sequences as init for other lengths
    if len(scan_results) > 0:
        top_scores = scan_results[:5]
        for score, n_src, name_src, a_src, T, lr, a_l2, a_max in top_scores:
            for n_tgt in lengths:
                if n_tgt != n_src:
                    # Interpolate to target length
                    a_interp = np.interp(np.linspace(0, 1, n_tgt),
                                         np.linspace(0, 1, n_src.size),
                                         a_src)
                    a_interp = np.maximum(a_interp, 1e-10)
                    a_interp /= a_interp.sum()
                    configs_extra = (n_tgt, f'transfer_{name_src}', a_interp, T, lr * 0.7, a_l2, a_max)
                    configs.append(configs_extra)

    # Phase 2: Deep optimization on best configs
    print(f"Phase 2: Deep optimization. Best so far: {best_score:.6f}")

    # Select top configs with diversity
    top_configs = []
    seen = set()
    for item in scan_results:
        key = (item[1], item[2])  # (n, name)
        if key not in seen:
            top_configs.append(item)
            seen.add(key)
        if len(top_configs) >= 12:
            break

    # Also add the extra transfer configs
    for item in configs[len(lengths) * len(generate_initializations(128, rng, parent)) * 4:]:
        if len(top_configs) < 18:
            top_configs.append(item)

    remaining = deadline - time.monotonic() - 20
    n_deep = len(top_configs)
    time_per_deep = max(10.0, remaining / max(1, n_deep * 3))

    for idx, (score, n, name, a0, T0, lr0, a_l2, a_max) in enumerate(top_configs):
        if time.monotonic() > deadline - 20:
            break
        t_alloc = min(time_per_deep * 3, deadline - time.monotonic() - 20)
        if t_alloc < 5:
            break

        # Try multiple hyperparameter variations
        variations = [
            (T0, lr0, a_l2, a_max),
            (T0 * 1.5, lr0 * 0.7, a_l2, a_max * 1.5),
            (T0 * 0.6, lr0 * 1.5, a_l2 * 1.3, a_max * 0.5),
        ]

        for T_try, lr_try, l2_try, max_try in variations:
            if time.monotonic() > deadline - 20:
                break
            t_single = t_alloc / len(variations)
            a_opt, obj_opt = adam_log_optimize(a0, time_limit=t_single, T0=T_try, lr0=lr_try,
                                                alpha_l2=l2_try, alpha_max=max_try)
            score_try = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score_try) and score_try < best_score:
                best_score = score_try
                best_sequence = a_opt.copy()
                print(f"  Deep: n={n}, {name}, T={T_try:.0f}, score={score_try:.6f}")

    # Phase 3: Final polish on the best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        n = len(best_sequence)
        print(f"Phase 3: Polish. n={n}, best={best_score:.6f}, remaining={remaining:.0f}s")

        polish_configs = [
            (50.0, 0.005, 1.0, 1.0),
            (100.0, 0.003, 0.5, 2.0),
            (30.0, 0.008, 1.5, 0.5),
            (80.0, 0.004, 1.0, 1.5),
            (200.0, 0.002, 0.3, 3.0),
        ]

        for T_try, lr_try, l2_try, max_try in polish_configs:
            if remaining < 3:
                break
            t_single = max(5.0, remaining / len(polish_configs))
            a_opt, obj_opt = adam_log_optimize(best_sequence, time_limit=t_single, T0=T_try,
                                                lr0=lr_try, alpha_l2=l2_try, alpha_max=max_try)
            score_try = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score_try) and score_try < best_score:
                best_score = score_try
                best_sequence = a_opt.copy()
                print(f"  Polish: T={T_try:.0f}, score={score_try:.6f}")
            remaining = deadline - 10 - time.monotonic()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses Adam optimizer in log-space with combined
    L2 flatness and smoothed-max objectives to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    
    Key ideas:
    1. Log-space (softmax) parameterization ensures feasibility.
    2. L2 objective (sum of conv squares) encourages flat autocorrelation.
    3. Smoothed max (log-sum-exp) directly targets the peak.
    4. Multiple sequence lengths and diverse initializations.
    5. Adaptive temperature and learning rate scheduling.
    """
    result = run_search(seed=seed, budget_s=budget_s)
    return [float(max(0.0, x)) for x in result]
