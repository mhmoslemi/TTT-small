except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:

def fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def gradient_surrogate(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 1.0, 0.0
    L = fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64), c.astype(np.float64, copy=False), 1.0, 0.0
    T = max(float(T_factor) * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c, dtype=np.float64)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    G = np.maximum(G, 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    return g, c.astype(np.float64, copy=False), T, surr


def lp_best_response(p, M_override=None):
    """Solve the LP: max sum(g) s.t. conv(p,g) <= M, g >= 0."""
    if not _HAVE_LP:
        return None
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0:
        return None
    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None
    if M_override is not None:
        M = float(M_override)
    
    f = p.copy()
    fs = float(f.sum())
    if fs <= 0:
        return None
    f = f / fs
    
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64, copy=False)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method="highs", options={"disp": False})
    except Exception:
        return None
    
    if res is None:
        return None
    g = getattr(res, "x", None)
    if g is None:
        return None
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return g / s


def make_linear_asym(n, steepness):
    x = np.linspace(0.0, 1.0, n)
    p = 0.1 + steepness * x
    return safe_normalize(p)


def make_power_asym(n, alpha):
    x = np.linspace(1e-4, 1.0, n)

    return safe_normalize(p)


def make_exp_asym(n, rate):
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(rate * x) + 0.01
    return safe_normalize(p)


def make_skew_shape(n, a, b):
    x = np.linspace(0.0, 1.0, n)
    p = np.power(x, a) * np.power(1.0 - x, b) + 1e-4
    return safe_normalize(p)


def make_piecewise_linear(n, kinks):
    x = np.linspace(0.0, 1.0, n)
    p = np.zeros(n)
    for i in range(n):
        x_i = x[i]
        p[i] = 0.1
        for kx, kv in kinks:
            if x_i <= kx:
                p[i] = kv
            else:
                p[i] = kv - 0.01 * (x_i - kx)
    p = np.maximum(p, 0.001)
    return safe_normalize(p)


def make_chirp_amplitude(n, bw):
    """Amplitude of a linear chirp - has good autocorrelation properties."""
    x = np.linspace(0.0, 1.0, n)
    freq = 1.0 + bw * x
    # Use |sin| of cumulative frequency as a non-negative envelope
    phase = 2.0 * np.pi * np.cumsum(freq) / n
    p = np.abs(np.sin(phase)) + 0.1
    return safe_normalize(p)

                prev_p = p
                p = pt
                accepted = True
                break
        
        if not accepted:
            since_improve += 1
            if since_improve > 50:
                sigma = 0.1 / n
                for _ in range(10):
                    pt = project_simplex(p + rng.normal(0, sigma, n))
                    sc = internal_score(pt)
                    if np.isfinite(sc) and sc < current_score:
                        if sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_improve = 0
                        else:
                            since_improve += 1
                        current_score = sc
                        prev_p = p
                        p = pt
                        break
            if since_improve > 200:
                p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0, 0.05/n, n))
                current_score = internal_score(p)
                prev_p = p
                since_improve = 100
        
        if iter_no % 500 == 0:
            print(f"{label} grad iter={iter_no} best={best_score:.10f} T={T_factor:.6f}", flush=True)
    
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def shape_parameterize(theta, n):
    """Chebyshev + log-taper parameterization."""
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    D = min(12, max(1, int(theta.size) - 2))
    
    logp = np.zeros(n, dtype=np.float64)
    if D >= 1:
        T_nm = np.ones(n)
        T_n = z.copy()
        for d in range(D):
            logp += theta[d] * T_n
            T_np = 2.0 * z * T_n - T_nm
            T_nm = T_n
            T_n = T_np
    
    if theta.size > D:
        logp += theta[D] * np.log(x + 1e-3)
    if theta.size > D + 1:
        logp += theta[D + 1] * np.log(1.0 - x + 1e-3)
    
    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n)
    m = float(np.max(logp))
    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)


def shape_objective(theta, n):
    try:
        p = shape_parameterize(theta, n)
        sc = internal_score(p)
        return sc if np.isfinite(sc) else 1e100
    except Exception:
        return 1e100


def global_shape_search(n, rng, budget_s):
    """Differential evolution on shape parameters."""
    dim = 14
    best_p = None
    best_score = np.inf
    
    end = time.monotonic() + max(0.0, float(budget_s))
    
    # Quick evaluations of hand-crafted shapes
    candidates = []
    candidates.append(np.zeros(dim))
    t = np.zeros(dim)
    t[0] = 1.5
    candidates.append(t)
    t = np.zeros(dim)
    t[1] = -1.0
    t[0] = 0.5
    candidates.append(t)
    t = np.zeros(dim)
    t[12] = 3.0
    t[13] = -1.0
    candidates.append(t)
    t = np.zeros(dim)
    t[13] = 3.0
    t[12] = -1.0
    candidates.append(t)
    for _ in range(5):
        candidates.append(rng.uniform(-2, 2, dim))
    
    for x0 in candidates:
        sc = shape_objective(x0, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = shape_parameterize(x0, n)
    
    if _HAVE_MINIMIZE and time.monotonic() < end - 30:
        bounds = [(-3.0, 3.0)] * 12 + [(-4.0, 6.0), (-4.0, 6.0)]
        try:
            res = differential_evolution(
                lambda th: shape_objective(th, n),
                bounds,
                maxiter=30,
                popsize=15,
                tol=0.3,
                seed=int(rng.integers(0, 2**31 - 1)),
                polish=False,
                init="latinhypercube",
            )
            if res is not None and np.isfinite(res.fun):
                sc = res.fun
                if sc < best_score:
                    best_score = sc
                    best_p = shape_parameterize(res.x, n)
                    candidates.append(res.x)
        except Exception:
            pass
        
        for x0 in candidates:
            if time.monotonic() > end - 5:
                break
            try:
                res = minimize(
                    lambda th: shape_objective(th, n),
                    np.asarray(x0, dtype=np.float64),
                    method="Nelder-Mead",
                    options={"maxiter": 200, "xatol": 1e-5, "fatol": 1e-7},
                )
                if res is not None and np.isfinite(res.fun):
                    sc = res.fun
                    if sc < best_score:
                        best_score = sc
                        best_p = shape_parameterize(res.x, n)
            except Exception:
                pass
    
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a nonnegative step function with low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)
    
    # Get parent if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.0:
            parent = parent_arr
    except Exception:
        parent = None
    
    best_p = None
    best_score = np.inf
    
    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = safe_normalize(np.asarray(p, dtype=np.float64))
        if p.size == 0:
            return
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"NEW BEST: {best_score:.10f} (n={p.size}) at t={time.monotonic()-start_time:.1f}s", flush=True)
    
    # Lengths to try
    lengths = []
    if parent is not None and 50 <= parent.size <= 3000:
        lengths.append(int(parent.size))
    for L in [700, 1000, 1200, 1500, 2000]:
        if L not in lengths:
            lengths.append(L)
    # Sort by preference: parent first, then ascending
    lengths = sorted(set(lengths))
    
    print(f"Trying lengths: {lengths}", flush=True)
    
    # Phase 1: Global shape search on a medium length
    if deadline - time.monotonic() > 200:
        shape_n = min(1000, max(500, lengths[0] if lengths else 1000))
        shape_budget = min(150.0, (deadline - time.monotonic()) * 0.12)
        print(f"Phase 1: Shape search n={shape_n} budget={shape_budget:.0f}s", flush=True)
        shape_p, shape_score = global_shape_search(shape_n, rng, shape_budget)
        if shape_p is not None:
            print(f"Shape search result: {shape_score:.10f}", flush=True)
        # Try to record at various lengths
        for L in lengths:
            if shape_p is not None:
                # Resample shape to length L
                x_shape = np.linspace(0, 1, shape_n)
                x_L = np.linspace(0, 1, L)
                p_L = np.interp(x_L, x_shape, shape_p)
                p_L = safe_normalize(p_L)
                sc = internal_score(p_L)
                try_record(p_L, sc)
    else:
        shape_p = None
    
    # Phase 2: Generate and optimize candidates at target lengths
    # Allocate time across lengths
    n_lengths = len(lengths)
    time_per_length = max(60.0, (deadline - time.monotonic() - 120.0) / n_lengths)
    
    for L in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        L_budget = min(time_per_length, deadline - time.monotonic() - 60.0)
        if L_budget < 30:
            continue
        
        print(f"\n=== Length {L}, budget {L_budget:.0f}s ===", flush=True)
        
        # Generate initializations
        inits = generate_initials(L, rng, parent)
        
        # Add shape-based initialization
        if shape_p is not None:
            x_shape = np.linspace(0, 1, shape_p.size)
            x_L = np.linspace(0, 1, L)
            p_shape = np.interp(x_L, x_shape, shape_p)
            inits.append(safe_normalize(p_shape))
        
        # Score and sort
        scored = []
        for p in inits:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
                try_record(p, sc)
        scored.sort(key=lambda x: x[0])
        
        print(f"Top initializations: {[f'{s:.6f}' for s, _ in scored[:5]]}", flush=True)
        
        # Run LP optimization on top candidates
        n_top = min(4, len(scored))
        if n_top == 0:
            continue
        
        lp_budget_each = max(20.0, (L_budget - 30.0) / max(1, n_top))
        
        for i in range(n_top):
            if time.monotonic() > deadline - 60:
                break
            sc, p = scored[i]
            remaining = deadline - time.monotonic() - 60.0
            if remaining < 15:
                break
            limit = min(lp_budget_each, remaining)
            
            p_best, sc_best = lp_optimize(p, rng, limit, label=f"L{L}-{i}")
            try_record(p_best, sc_best)
            
            # Gradient polish
            if time.monotonic() < deadline - 30 and sc_best < best_score * 0.999:
                polish_limit = min(30.0, deadline - time.monotonic() - 30.0)
                if polish_limit > 5:
                    p_pol, sc_pol = gradient_polish(p_best, rng, polish_limit, label=f"L{L}-{i}-pol")
                    try_record(p_pol, sc_pol)
    
    # Phase 3: Final intensive LP on the best found
    if best_p is not None and time.monotonic() < deadline - 20:
        final_budget = max(10.0, deadline - time.monotonic() - 10.0)
        print(f"\n=== Final LP polish, budget {final_budget:.0f}s ===", flush=True)
        n_best = int(best_p.size)
        if n_best <= 1500:
            p_final, sc_final = lp_optimize(best_p, rng, min(final_budget * 0.7, 600.0), label="FINAL-LP")
            try_record(p_final, sc_final)
        
        if time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic() - 10.0
            if rem > 5:
                p_pol, sc_pol = gradient_polish(best_p, rng, rem, label="FINAL-POL")
                try_record(p_pol, sc_pol)
    
    # Verify with evaluate_sequence if available
    if best_p is not None:
        try:
            test_seq = [float(x) for x in best_p]
            if len(test_seq) > 0:
                test_score = evaluate_sequence(test_seq)
                if np.isfinite(test_score):
                    if abs(test_score - best_score) > 0.01:
                        print(f"WARNING: internal={best_score:.10f} vs external={test_score:.10f}", flush=True)
                    best_score = min(best_score, test_score)
        except Exception:
            pass
    
    if best_p is None:
        best_p = np.full(700, 1.0 / 700, dtype=np.float64)
    
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(max(1, best_p.size), 1.0 / max(1, best_p.size), dtype=np.float64)
        s = 1.0
    best_p = best_p / s * s  # keep sum
    best_p = np.minimum(best_p, 1000.0)
    
    print(f"\nFinal answer: n={len(best_p)}, internal_score={best_score:.10f}", flush=True)
    return [float(x) for x in best_p]
