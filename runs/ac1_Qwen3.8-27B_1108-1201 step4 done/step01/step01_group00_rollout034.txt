import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):

    a = np.clip(a, 0.0, 1000.0)

    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)

def build_lp_matrix_fast(f, n):

    A = np.zeros((2 * n - 1, n))
    k_vals = np.arange(2 * n - 1)[:, None]
    j_vals = np.arange(n)[None, :]
    idx = k_vals - j_vals
    mask = (idx >= 0) & (idx < n)
    # f[idx] where mask is true
    safe_idx = np.where(mask, idx, 0)
    A = np.where(mask, f[safe_idx], 0.0)
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: maximize sum(b) s.t. (f*b)_k <= M for all k, b >= 0."""
    A = build_lp_matrix_fast(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def optimize_gradient_lse(a_init, time_limit, rng, verbose=True):
    """

    Parameterization: a = softmax(theta) * sqrt(2n)

    S = np.sqrt(2.0 * n)
    # Initialize theta from a_init
    a_start = a_init / np.sum(a_init)  # normalize to sum 1
    theta = np.log(a_start + 1e-10) * S  # scale
    # Actually, let's work on the simplex directly
    a = a_start.copy()

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_a = a.copy()
    best_score = local_score(a)

    tau_start = 10.0 / n
    tau_end = 0.1 / n
    lr_base = 0.5

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        mean_g = np.mean(g_a)
        g_a = g_a - mean_g

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_a
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_a * g_a)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = lr_base / (1.0 + 0.01 * step)

def polish_local_search(a_init, time_limit, rng, verbose=True):
    """Simulated annealing local search."""
    a = a_init.copy()
    n = len(a)
    current_score = local_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    iterations = 0

    if verbose:
        print(f"    SA start: score={current_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.3:
            # Targeted: reduce peak contributors
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            # Find top contributors to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(2, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(5, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.7:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.85:
            # Partial symmetrization
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
                if verbose and iterations % 200 == 0:
                    print(f"    SA iter={iterations}, new best={best_score:.6f}")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling
        iterations += 1

        if iterations % 500 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    SA iter={iterations}, current={current_score:.6f}, best={best_score:.6f}, temp={temperature:.4f}, time={elapsed:.0f}s")

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 1.1))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Random initial sequence."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k, amp in [(1, 0.2), (2, 0.15), (3, 0.1), (5, 0.08), (7, 0.05), (11, 0.03)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_ramp_init(n, rng, direction=1):
    """Ramp (asymmetric) initial sequence."""
    t = np.linspace(0, 1, n)
    if direction == 1:
        a = 0.3 + 0.7 * t
    else:
        a = 1.0 - 0.7 * t
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_init(n, rng, center_frac=0.3):
    """Gaussian-like initial sequence (off-center)."""
    t = np.arange(n, dtype=np.float64)
    center = center_frac * n
    width = n * 0.2
    a = np.exp(-0.5 * ((t - center) / width) ** 2) + 0.1
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    Multi-phase search:
    Phase 1: LP-based iterative optimization from multiple starting points.
    Phase 2: Gradient-based polishing with LSE smoothing.
    Phase 3: Simulated annealing local search.
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: LP-based optimization
    print("\n=== Phase 1: LP-based optimization ===")

    # Define starting configurations
    n_lp = 256
    if start_seq is not None and len(start_seq) == n_lp:
        lp_starts = [start_seq.copy()]
    else:
        lp_starts = []

    # Add diverse initializations
    lp_starts.append(make_chirp_init(n_lp, 0.5, rng))
    lp_starts.append(make_random_init(n_lp, rng, spread=0.4))
    lp_starts.append(make_multi_tone_init(n_lp, rng))
    lp_starts.append(make_ramp_init(n_lp, rng, direction=1))
    lp_starts.append(make_gaussian_init(n_lp, rng, center_frac=0.25))

    # Allocate time for LP phase
    lp_phase_time = time_budget * 0.45
    time_per_lp = lp_phase_time / min(len(lp_starts), 4)  # limit to 4 best starts

    # Sort initializations by initial score (best first)
    scored_starts = []
    for i, s in enumerate(lp_starts):
        sc = local_score(s)
        scored_starts.append((sc, i, s))
        print(f"  Init {i}: score={sc:.6f}")
    scored_starts.sort()

    # Try top few initializations
    for rank, (init_score, idx, a0) in enumerate(scored_starts[:4]):
        if time.monotonic() > deadline - time_budget * 0.4:
            break
        a0_norm = a0 / np.sum(a0)
        remaining_lp = deadline - time_budget * 0.4 - time.monotonic()
        t_alloc = min(time_per_lp, max(10.0, remaining_lp))

        print(f"\n  LP run {rank+1} (init score={init_score:.6f}), time={t_alloc:.0f}s")
        lp_result, lp_score = lp_optimize(a0_norm, t_alloc, rng, verbose=True)

        # Scale back (score is scale-invariant so normalization doesn't matter)
        actual_score = local_score(lp_result)
        print(f"  LP result: score={actual_score:.6f}")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = lp_result.copy()
            print(f"  *** New best after LP: {best_score:.6f} ***")

    # Phase 2: Gradient-based polish
    print(f"\n=== Phase 2: Gradient polish (best score so far: {best_score:.6f}) ===")
    if best_seq is not None:
        grad_phase_time = time_budget * 0.3
        remaining = deadline - time.monotonic()
        t_grad = min(grad_phase_time, max(30.0, remaining * 0.6))
        print(f"  Gradient polish time: {t_grad:.0f}s")

        # Normalize for gradient method
        a_grad = best_seq / np.sum(best_seq)
        grad_result, grad_score = optimize_gradient_lse(a_grad, t_grad, rng, verbose=True)
        actual_grad_score = local_score(grad_result)
        print(f"  Gradient result: score={actual_grad_score:.6f}")

        if actual_grad_score < best_score:
            best_score = actual_grad_score
            best_seq = grad_result.copy()
            print(f"  *** New best after gradient: {best_score:.6f} ***")
        else:
            # Keep the LP result if gradient didn't help
            print(f"  Keeping LP result (best={best_score:.6f})")

    # Phase 3: Also try gradient from a fresh start with a different length
    if time.monotonic() < deadline - time_budget * 0.2:
        print(f"\n=== Phase 2b: Additional gradient search ===")
        n2 = 384
        a0_2 = make_chirp_init(n2, 0.4, rng)
        a0_2_norm = a0_2 / np.sum(a0_2)
        remaining = deadline - time_budget * 0.2 - time.monotonic()
        t_grad2 = min(120.0, max(20.0, remaining * 0.5))
        print(f"  n={n2}, time={t_grad2:.0f}s")
        grad2_result, grad2_score = optimize_gradient_lse(a0_2_norm, t_grad2, rng, verbose=False)
        actual2 = local_score(grad2_result)
        print(f"  n={n2} gradient result: score={actual2:.6f}")
        if actual2 < best_score:
            best_score = actual2
            best_seq = grad2_result.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Local search polish
    print(f"\n=== Phase 4: Local search polish (best score: {best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(time_budget * 0.2, max(20.0, remaining))
        print(f"  SA time: {sa_time:.0f}s, n={len(best_seq)}")
        sa_result, sa_score = polish_local_search(best_seq, sa_time, rng, verbose=True)
        print(f"  SA result: score={sa_score:.6f}")
        if sa_score < best_score:
            best_score = sa_score
            best_seq = sa_result.copy()
            print(f"  *** New best after SA: {best_score:.6f} ***")

    # Final: one more LP polish if time remains
    if time.monotonic() < deadline - 15 and best_seq is not None:
        print(f"\n=== Phase 5: Final LP polish ===")
        remaining = deadline - time.monotonic() - 5
        a_final = best_seq / np.sum(best_seq)
        lp_final, lp_final_score = lp_optimize(a_final, min(60.0, remaining), rng, verbose=True)
        actual_final = local_score(lp_final)
        if actual_final < best_score:
            best_score = actual_final
            best_seq = lp_final.copy()
            print(f"  *** Final best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        elapsed = time.monotonic() - t_global
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, time={elapsed:.0f}s ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
