import numpy as np
import time

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * np.max(b) / (sum_a ** 2)

def compute_b(a):
    """Compute autocorrelation"""
    return np.convolve(a, a)

def softmax_weights(b, tau):
    """Compute softmax weights for peak selection"""
    b_shifted = b - np.max(b)
    w = np.exp(tau * b_shifted)
    w /= np.sum(w)
    return w

def gradient_at_bk(a, k):
    """Gradient of b[k] w.r.t. a"""
    n = len(a)
    grad = np.zeros(n)
    # b[k] = sum_i a[i]*a[k-i]
    # db[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
    # Equivalent to: grad[j] = a[k-j] + a[j-k] (with zero padding)
    for j in range(n):
        gj = 0.0
        if 0 <= k - j < n:
            gj += a[k - j]
        if 0 <= j - k < n:
            gj += a[j - k]
        grad[j] = gj
    return grad

    """Project onto {x >= 0, sum(x) = target}"""
    v = np.asarray(v, dtype=np.float64)

        return v

    cssv = np.cumsum(u)

        return np.full(n, target / n)
    rho = rho_candidates[-1]

    return np.maximum(v - theta, 0.0)

    n = len(v)

    cssv = np.cumsum(u)
    rho_cand = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]

    rho = rho_cand[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient of smoothed max(b) using FFT"""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)

    else:
        w = np.exp(tau * (b - b_max))
        ws = np.sum(w)
        if ws < 1e-300:

    W = np.fft.rfft(w, N)

    T2 = np.fft.irfft(W * A, N)[:n]

            no_improve += 1

"""
Refined subgradient/proximal gradient optimization for minimizing normalized
autocorrelation peak of non-negative sequences. Uses FFT-based gradient computation,
Nesterov momentum, adaptive smoothing (log-sum-exp annealing), and multi-phase
strategy with diversification to escape local minima.
"""

import numpy as np
import time


def _compute_b(a):
    """Compute autocorrelation b = conv(a, a)."""
    return np.convolve(a, a)


def _score(a):
    """Compute 2*n*max(b)/(sum(a))^2."""
    n = len(a)
    b = _compute_b(a)
    sa = np.sum(a)
    if sa < 1e-10:
        return float('inf')
    return 2.0 * n * np.max(b) / (sa * sa)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_weights(a, w, N):
    """
    Compute gradient of sum_k w[k]*b[k] w.r.t. a using FFT.
    w has length 2n-1, a has length n, N is FFT size >= 3n.
    """
    n = len(a)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    T1 = np.fft.irfft(W * np.conj(A), N)[:n]
    T2 = np.fft.irfft(W * A, N)[:n]
    return T1 + T2


def _optimize_sequence(a_init, deadline, tau, eta_init, momentum, topM, rng):
    """
    Subgradient/gradient descent with momentum on the simplest constraint (sum=1, a>=0).
    
    Parameters:
    - a_init: initial sequence (will be normalized to sum=1)
    - deadline: time.monotonic() deadline
    - tau: smoothing parameter for log-sum-exp (use 0 for hard top-M)
    - eta_init: initial step size
    - momentum: heavy-ball momentum coefficient (0 to 1)
    - topM: number of top peaks to use when tau=0
    - rng: numpy random generator
    
    Returns: (best_sequence, best_score)
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N *= 2

    best = a.copy()
    best_b = _compute_b(best)
    best_score = 2.0 * n * np.max(best_b)

    v = np.zeros(n)
    eta = eta_init
    stalled = 0
    max_stall = 500
    eta_min = 1e-14

    while time.monotonic() < deadline:
        b = _compute_b(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score * (1.0 - 1e-10):
            best_score = cur_score
            best = a.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled >= max_stall:
            eta *= 0.6
            stalled = 0
            if eta < eta_min:
                break

        # Compute weights on autocorrelation lags
        if tau > 1e-6:
            b_shifted = b - b_max
            # Clip for numerical stability
            b_shifted = np.maximum(b_shifted, -50.0 / tau)
            w = np.exp(tau * b_shifted)
            ws = np.sum(w)
            if ws > 1e-300:
                w /= ws
            else:
                w = np.ones(len(b)) / len(b)
        else:
            if topM >= len(b):
                idx = np.arange(len(b))
            else:
                idx = np.argpartition(b, -topM)[-topM:]
            w = np.zeros(len(b))
            w[idx] = 1.0 / topM

        # Compute gradient via FFT
        grad = _grad_from_weights(a, w, N)

        # Center gradient for sum constraint
        grad -= np.mean(grad)

        # Momentum update (heavy ball)
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Take step
        a_new = a - eta * v / v_norm
        a_new = _project_simplex(a_new, 1.0)

        # Check if step is acceptable
        b_new = _compute_b(a_new)
        f_new = 2.0 * n * np.max(b_new)

        if f_new > cur_score * 1.05 and eta > 1e-10:
            # Step too large, reduce and retry
            eta *= 0.4
            a_new = a
            v = momentum * v
        else:
            a = a_new

    return best, best_score


def _make_two_level(n, eps):
    """Two-level anti-symmetric: first half (1+eps), second half (1-eps)."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_smooth_anti(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_four_level(n, eps):
    """Four-level: edges high, center low."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_ramp(n, alpha):
    """Ramp-like: a[i] = 1 + alpha*(i/n - 0.5), normalized."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * (i / n - 0.5)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation and re-project."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 8), size=min(7, n // 4))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        t = np.arange(n)
        noise += amp * np.cos(2 * np.pi * f * t / n + phase)
    noise_max = np.max(np.abs(noise))
    if noise_max > 1e-15:
        noise /= noise_max
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
