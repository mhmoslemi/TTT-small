import numpy as np
import time

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _exact_score(a):
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _make_anti_symmetric(n, profile='cos', eps=0.5):
    i = np.arange(n, dtype=np.float64)
    if profile == 'cos':
        a = 1.0 + eps * np.cos(2 * np.pi * i / n)
    elif profile == 'step':
        a = np.ones(n)
        a[:n//2] = 1.0 + eps
        a[n//2:] = 1.0 - eps
    elif profile == 'ramp':
        a = 1.0 + eps * (1.0 - 2.0 * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _adam_lse_optimize(a, deadline, rng, topK=20, t_init=50.0, t_min=5.0, 
                        lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a)
    a = _project_simplex(a.copy(), 1.0)
    best = a.copy()
    best_score = _exact_score(best)
    
    # ADAM state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    temperature = t_init
    
    while time.monotonic() < deadline:
        # Compute convolution
        b = np.convolve(a, a)
        
        # Get top K peaks
        if topK >= len(b):
            idx = np.arange(len(b))
        else:
            idx = np.argpartition(b, -topK)[-topK:]
        top = b[idx]
        
        # Log-sum-exp weights
        m_val = np.max(top)
        logits = temperature * (top - m_val)
        exp_logits = np.exp(logits)
        sum_exp = np.sum(exp_logits)
        weights = exp_logits / sum_exp
        
        # Compute gradient
        grad = np.zeros(n)
        for ki in range(len(idx)):
            w = weights[ki]
            if w < 1e-10:

            k = int(idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * w * a[k - np.arange(lo, hi+1)]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        a = _project_simplex(a_new, 1.0)

                best = a.copy()

    b = fftconvolve(a, a)
    return 2.0 * n * np.max(b)

def _softmax(x):
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    return e / s

def _logit(a, epsilon=1e-18):
    return np.log(np.maximum(a, epsilon))

def _make_starts(n, rng, parent=None):
    starts = []
    if parent is not None and np.sum(parent) > 1e-10:
        p = parent.copy()
        p = np.clip(p, 0, None)
        p /= np.sum(p)
        starts.append(p)
    
    i = np.arange(n, dtype=np.float64)
    
    for eps in [0.3, 0.5, 0.7, 0.9, 1.0]:
        a = 1.0 + eps * np.cos(2 * np.pi * i / n)
        a = np.maximum(a, 1e-6)
        a /= np.sum(a)
        starts.append(a)
    
    for eps in [0.4, 0.6, 0.8]:
        a = np.ones(n)
        half = n // 2
        a[:half] = 1.0 + eps
        a[half:] = 1.0 - eps
        a = np.maximum(a, 1e-6)
        a /= np.sum(a)
        starts.append(a)
    
    for eps in [0.5, 0.8]:
        a = 1.0 + eps * (1.0 - 2.0 * i / max(1, n - 1))
        a = np.maximum(a, 1e-6)
        a /= np.sum(a)
        starts.append(a)
    
    for _ in range(3):
        a = rng.random(n)
        a = np.maximum(a, 1e-6)
        a /= np.sum(a)
        starts.append(a)
    
    return starts

def _lse_obj_and_grad(x, topK, temperature, n):
    a = _softmax(x)
    b = fftconvolve(a, a)
    
    if topK >= len(b):
        idx = np.arange(len(b))
    else:
        idx = np.argpartition(b, -topK)[-topK:]
    top = b[idx]
    
    m_val = np.max(top)
    t = temperature
    logits = t * (top - m_val)
    exp_logits = np.exp(logits)
    sum_exp = np.sum(exp_logits)
    p = exp_logits / sum_exp
    
    lse = m_val + np.log(sum_exp) / t
    score = 2.0 * n * lse
    
    B = np.dot(p, top)
    
    S = np.zeros(n)
    for ki in range(len(idx)):
        pk = p[ki]
        if pk < 1e-12:
            continue
        k = int(idx[ki])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            S[lo:hi+1] += pk * a[k - np.arange(lo, hi+1)]
    
    grad_x = 4.0 * n * a * (S - B)
    
    return score, grad_x

def _optimize_lbfgs(a_init, deadline, topK=30, temperature=20.0, maxiter=500):
    x0 = _logit(a_init)
    
    eval_count = [0]
    
    def obj(x):
        eval_count[0] += 1
        s, g = _lse_obj_and_grad(x, topK, temperature, len(a_init))
        return s, g
    
    result = minimize(obj, x0, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxfun': 2000,
                               'ftol': 1e-12, 'gtol': 1e-8})
    
    a_opt = _softmax(result.x)
    a_opt = _project_simplex(a_opt, 1.0)
    score = _exact_score(a_opt)
    
    return a_opt, score, result.success, eval_count[0]

def _perturb(a, rng, scale=0.05):
    n = len(a)
    # Low-frequency perturbation
    noise = np.zeros(n)
    n_freqs = min(5, n // 4)
    for _ in range(n_freqs):
        f = rng.integers(1, max(2, n // 5))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
