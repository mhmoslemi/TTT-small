def _compute_obj(a):
    n = len(a)

def _normalize(a, target=None):
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return np.full(n, target / n)
    return a * (target / S)

def _soft_gradient(a, tau, n_peaks=1):
    """Compute gradient of log-sum-exp(tau * conv(a,a)) / tau."""
    n = len(a)
    c = np.convolve(a, a)
    
    if n_peaks >= len(c):
        # Uniform weights
        w = np.ones(len(c)) / len(c)
    elif n_peaks == 1:
        # Softmax
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w /= w.sum()
    else:

        k = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(len(c))
        c_top = c[top_idx]

    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, m, v

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses log-sum-exp surrogate with Adam optimizer, adaptive temperature scheduling,
peak-specific gradient focusing, restart strategies, and LP-based refinement.
Starts from the previous best sequence and explores nearby and different-length sequences.
"""

import numpy as np
import time


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return np.full(n, target / n)
    return a * (target / S)


def _soft_max_weights(c, tau):
    """Compute softmax weights over convolution values."""
    c_shift = c - c.max()
    w = np.exp(tau * c_shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w /= s
    return w


def _top_k_weights(c, tau, k):
    """Compute weights on top-k peaks with softmax."""
    if k >= len(c):
        return _soft_max_weights(c, tau)
    top_idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(len(c))
    c_top = c[top_idx]
    c_top_shift = c_top - c_top.max()
    w_top = np.exp(tau * c_top_shift)
    s = w_top.sum()
    if s < 1e-300:
        w_top = np.zeros(k)
        w_top[c_top.argmax()] = 1.0
    else:
        w_top /= s
    w[top_idx] = w_top
    return w


def _gradient_from_weights(a, w):
    """Compute gradient given weights on convolution output."""
    n = len(a)
    a_rev = a[::-1]
    conv_aw = np.convolve(a_rev, w)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]
    return grad


def _compute_lse_gradient(a, tau, top_k=1):
    """Compute gradient of log-sum-exp(tau * conv(a, a)) / tau."""
    n = len(a)
    c = np.convolve(a, a)
    if top_k > 1:
        w = _top_k_weights(c, tau, top_k)
    else:
        w = _soft_max_weights(c, tau)
    grad = _gradient_from_weights(a, w)
    return grad


def _adam_update(params, grad, m, v, t, lr, b1, b2, eps):
    """Perform one Adam update step."""
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    params[:] = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v


def _optimize_lse_adam(a0, deadline, max_iters=5000, tau_start=3.0, tau_end=200.0,
                       lr_start=0.08, lr_end=0.002, top_k_schedule=None,
                       restart_every=500, restart_scale=0.05, seed=0):
    """Optimize using LSE surrogate with Adam."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a0.astype(np.float64).copy(), S_t)
    
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stagnant = 0
    last_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        progress = it / max_iters
        
        # Adaptive temperature
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5
        lr = lr_start + (lr_end - lr_start) * progress
        
        # Top-k schedule
        if top_k_schedule is not None:
            top_k = top_k_schedule(progress)
        else:
            top_k = 1
        
        # Compute gradient
        grad = _compute_lse_gradient(a, tau, top_k)
        grad = grad - grad.mean()
        
        # Adam step
        _adam_update(a, grad, m, v, t, lr, 0.9, 0.999, 1e-8)
        
        # Project
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)
        
        # Check progress
        if (it + 1) % 30 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                stagnant = 0
            else:

        # Restart if stagnant

            a = _normalize(best_a + rng.random(n) * restart_scale, S_t)
            m[:] = 0
            v[:] = 0
            t = 0
            stagnant = 0
    
    # Final check
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    
    return best_a, best_score


def _optimize_peak_hopping(a0, deadline, max_iters=5000, seed=0):
    """Optimize by directly targeting the current peak with momentum."""

    beta = 0.85
    stagnant = 0

        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        k_star = int(c.argmax())
        
        # Gradient of c[k_star] w.r.t. a
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * a[k_star - indices]

        lr = 0.03 / (1.0 + 0.1 * it)

def _lp_refine(a, deadline, max_iters=80):
    """LP-based refinement: find g maximizing sum(g) s.t. conv(a,g) <= M, g>=0."""
    from scipy.optimize import linprog
    
    n = len(a)
    if n > 2048:
        return a, _compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    c = np.convolve(a, a)
    M = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for m_idx in range(2 * n - 1):
            lo = max(0, m_idx - n + 1)
            hi = min(n - 1, m_idx)
            for j in range(lo, hi + 1):
                A[m_idx, j] = a[m_idx - j]
        
        b = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
            g_sum = float(g.sum())
            if g_sum < S_t * 0.3:
                break
            
            # Try blending
            improved = False
            for t_blend in [0.4, 0.25, 0.15, 0.1]:
                a_new = (1.0 - t_blend) * best_a + t_blend * g
                a_new = np.maximum(a_new, 0.0)
                a_new = _normalize(a_new, S_t)
                s_new = _compute_obj(a_new)
                if s_new < best_score:
                    best_a = a_new.copy()
                    best_score = s_new
                    improved = True
                    break
            
            if not improved:
                # Small perturbation to escape
                break
            
            # Update M
            c = np.convolve(best_a, best_a)
            M = float(c.max())
        
        except Exception:
            break
    
    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=200):
    """Simple coordinate descent: optimize pairs of coordinates."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Optimize a few random pairs
        indices = np.random.default_rng(r).choice(n, size=min(20, n), replace=False)
        for i in indices:
            if i + 1 < n:
                j = i + 1
            else:
                j = i - 1
            # Try shifting mass from i to j and vice versa
            for frac in [0.1, 0.3, 0.5, 0.7]:
                for direction in [1, -1]:
                    a_try = best_a.copy()
                    shift = direction * frac * a_try[i]
                    a_try[i] -= shift
                    a_try[j] += shift
                    if a_try[i] >= 0 and a_try[j] >= 0:
                        a_try = _normalize(a_try, S_t)
                        s = _compute_obj(a_try)
                        if s < best_score:
                            best_a = a_try.copy()
                            best_score = s
                            improved = True
        if not improved:
            break
    
    return best_a, best_score

                seq, score = _optimize_lse_adam(current, sub_dl, max_iters=3000,
                                                tau_start=10.0, tau_end=300.0,
                                                lr_start=0.03, lr_end=0.001,
                                                seed=seed + 100 + pass_i)
            elif pass_i % 3 == 1:
                seq, score = _optimize_multi_peak(current, sub_dl, max_iters=3000,
                                                  seed=seed + 200 + pass_i)
            else:
                seq, score = _optimize_peak_hopping(current, sub_dl, max_iters=3000,
                                                    seed=seed + 300 + pass_i)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                current = seq.copy()
                print(f"    Pass {pass_i+1}: score={score:.6f}")
            else:
                # Try perturbation for next pass
                current = _normalize(current + rng.random(len(current)) * 0.02,
                                     np.sqrt(2.0 * len(current)))
    
    # ===================== PHASE 4: LP refinement and coordinate descent =====================
    print(f"\n=== Phase 4: Refinement (best={best_score:.6f}) ===")
    final_deadline = deadline
    
    if best_seq is not None and len(best_seq) <= 1024:
        if time.monotonic() < final_deadline - 20:
            print("  LP refinement...")
            seq, score = _lp_refine(best_seq, final_deadline - 20, max_iters=60)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    After LP: {score:.6f}")
    
    if best_seq is not None and time.monotonic() < final_deadline - 10:
        print("  Coordinate descent...")
        seq, score = _coordinate_descent(best_seq, final_deadline - 10, max_rounds=100)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    After CD: {score:.6f}")
    
    # Final LSE-Adam polish
    if best_seq is not None and time.monotonic() < final_deadline - 5:
        seq, score = _optimize_lse_adam(best_seq, final_deadline - 3, max_iters=2000,
                                        tau_start=50.0, tau_end=500.0,
                                        lr_start=0.01, lr_end=0.0005,
                                        seed=seed + 999)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    After final polish: {score:.6f}")
    
    # ===================== Final validation =====================
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
        return [float(x) for x in best_seq]
    
    print("Falling back to uniform sequence.")
    return [1.0] * 256
