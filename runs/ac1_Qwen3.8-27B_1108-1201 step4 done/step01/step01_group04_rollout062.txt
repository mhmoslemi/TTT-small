def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based alternating optimization with gradient refinement.
Starts from known good sequences and tries multiple lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def softmax_grad_step(a, lr, tau):
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        if end > 0:
            grad[j] = 2.0 * np.dot(w[j:j + end], a[:end])
    
    grad -= grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            A[m, lo:hi + 1] = f[m - hi:m - lo + 1][::-1]
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, max_iters=2000):

    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = a.copy()
    a = normalize(a, S_t)
    
    c = np.convolve(a, a)
    M = float(c.max())
    best_obj = compute_obj(a)
    best = a.copy()
    
    f = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp(f, M)
        if g is None:
            break
        
        # Try blending
        improved = False
        for t in [0.5, 1.0, 0.8, 0.3]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                a = a_new
                best = a.copy()
                best_obj = obj_new
                c = np.convolve(a, a)
                M = float(c.max())
                f = a.copy()
                improved = True
                break
        
        if not improved:
            # Escape: gradient step
            a = softmax_grad_step(a, 0.03, 20.0)
            c = np.convolve(a, a)
            M = float(c.max())
            f = a.copy()
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best = a.copy()
                best_obj = obj_new
    
    return best, best_obj


def grad_optimize(a, deadline, max_iters=3000):
    """Gradient descent with softmax smoothing."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        tau = 10.0 + 40.0 * (it / max_iters)
        lr = 0.1 * (1.0 + np.cos(np.pi * (it + 1) / max_iters)) * 0.5
        lr = max(lr, 0.005)
        
        a = softmax_grad_step(a, lr, tau)

            if obj < best_obj:

                best = a.copy()

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def softmax_grad_step(a, lr, tau):
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # Avoid overflow
    w = np.exp(np.clip(tau * (c - c_max), -50, 0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        if end > 0:
            grad[j] = 2.0 * np.dot(w[j:j + end], a[:end])
    
    grad -= grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def solve_lp_min_peak(f, S_target):
    """Solve: min M s.t. conv(f,g) <= M, sum(g) = S_target, g >= 0.
    Returns (g, M) or (None, None)."""
    n = len(f)
    # Variables: g[0..n-1], M -> n+1 variables
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality: sum_j f[m-j]*g[j] - M <= 0 for m=0..2n-2
    A_ub = np.zeros((2 * n - 1, n + 1))

        hi = min(n - 1, m)

        A_ub[m, n] = -1.0  # -M
    
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            M = max(0.0, res.x[n])
            return g, M
    except Exception:
        pass
    return None, None

                    best_obj = obj_try

                f = a.copy()
                obj_new = compute_obj(a)
                if obj_new < best_obj:

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')

def normalize(a, target=None):
    """Normalize sequence to have target sum."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def softmax_grad_step(a, lr, tau):
    """One gradient step using softmax-smoothed peak."""
    n = len(a)

    c_max = c.max()
    exponents = np.clip(tau * (c - c_max), -60.0, 0.0)
    w = np.exp(exponents)

    if w_sum < 1e-30:
        w = np.zeros_like(c)

                improved = True
                stagnation = 0
                break

        if not improved:
            stagnation += 1
            if stagnation <= 5:
                # Small gradient escape
                a = softmax_grad_step(a, 0.02 + 0.01 * stagnation, 15.0 + 5.0 * stagnation)
                f = a.copy()
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best = a.copy()
                    best_obj = obj_new
                    stagnation = 0
            elif stagnation <= 15:
                # Larger perturbation
                rng_local = np.random.default_rng(it)
                a_pert = a + 0.05 * rng_local.standard_normal(n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert, S_t)
                if compute_obj(a_pert) < best_obj:
                    a = a_pert
                    best = a.copy()
                    best_obj = compute_obj(a)
                    f = a.copy()
                    stagnation = 0
                else:
                    a = softmax_grad_step(a, 0.1, 10.0)
                    f = a.copy()
            else:
                break

    return best, best_obj


def grad_optimize(a, deadline, max_iters=5000):
    """Gradient descent with adaptive softmax temperature and LR."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(np.maximum(a.copy(), 0.0), S_t)
    best = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        progress = (it + 1) / max_iters
        tau = 5.0 + 60.0 * progress
        lr = 0.12 * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.003)
        a = softmax_grad_step(a, lr, tau)

        if (it + 1) % 300 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best = a.copy()

    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best = a.copy()

    return best, best_obj


def build_lp_matrix(f):
    """Pre-build the LP constraint matrix for f."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A_ub[m, idx] = f[m - idx]
        A_ub[m, n] = -1.0
    return A_ub


def alternating_optimize_v2(a, deadline, max_iters=5000):
    """LP-based alternating with pre-built matrix and adaptive strategy."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(np.maximum(a.copy(), 0.0), S_t)

    best = a.copy()
    best_obj = compute_obj(a)
    f = a.copy()
    A_ub = build_lp_matrix(f)
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_t])
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    stagnation = 0
    rng_local = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = np.maximum(res.x[:n], 0.0)
            g = normalize(g, S_t)
        except Exception:
            break

        improved = False
        for t in [1.0, 0.6, 0.4, 0.8, 0.3, 0.5, 0.7, 0.2, 0.15]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            S_try = a_try.sum()
            if S_try < 1e-12:
                continue
            a_try = a_try * (S_t / S_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-10:
                a = a_try
                best = a.copy()
                best_obj = obj_try
                f = a.copy()
                A_ub = build_lp_matrix(f)
                improved = True
                stagnation = 0
                break

        if not improved:
            stagnation += 1
            if stagnation <= 3:
                a = softmax_grad_step(a, 0.02, 20.0)
                f = a.copy()
                A_ub = build_lp_matrix(f)
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best = a.copy()
                    best_obj = obj_new
                    stagnation = 0
            elif stagnation <= 10:
                noise_scale = 0.03 + 0.01 * (stagnation - 3)
                a_pert = a + noise_scale * rng_local.standard_normal(n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert, S_t)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    a = a_pert
                    best = a.copy()
                    best_obj = obj_pert
                    f = a.copy()
                    A_ub = build_lp_matrix(f)
                    stagnation = 0
                else:
                    a = softmax_grad_step(a, 0.05, 15.0)
                    f = a.copy()
                    A_ub = build_lp_matrix(f)
            else:
                break

    return best, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Collect starting sequences
    starts = []

    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            gs = np.maximum(gs, 0.0)
            for L in [128, 256, 384, 512, 768, 1024, 1536]:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
            # Reversed versions
            for L in [256, 512, 1024]:
                if L <= len(gs):
                    starts.append(gs[:L][::-1].copy())
            # Centered
            for L in [512, 1024]:
                if 2 * L <= len(gs):
                    mid = len(gs) // 2
                    starts.append(gs[mid - L // 2:mid + L // 2].copy())
    except NameError:
        pass

    # Structured starts
    for n in [256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        starts.append(0.1 + 0.9 * i / (n - 1))
        starts.append(np.exp(np.linspace(0, 0.7, n)) + 0.05)
        starts.append(0.3 + 0.4 * i / n + 0.2 * np.sin(np.pi * i / n))
        starts.append(rng.random(n) * 0.5 + 0.2)

    starts.sort(key=len)

    print(f"Starting: {len(starts)} candidates, budget={budget_s}s")

    # Phase 1: Quick gradient scan
    t_phase1 = min(t_start + 90.0, deadline - 120.0)
    for idx, a0 in enumerate(starts):
        if time.monotonic() > t_phase1:
            break
        n = len(a0)
        if n > 2048:
            continue
        t_sub = min(time.monotonic() + 18.0, t_phase1 - 3.0, deadline - 60.0)
        if t_sub < time.monotonic() + 3:
            break
        a0 = np.maximum(a0, 0.0)
        seq, score = grad_optimize(a0, t_sub, max_iters=3000)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  P1[{idx}] n={n}: {score:.6f}")

    # Phase 2: Deep LP optimization
    print(f"\nPhase 2: LP optimization (best={best_score:.6f})")
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq.copy())
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 512:
            gs = np.maximum(gs, 0.0)
            candidates.append(gs[:512].copy())
            if gs.size >= 1024:
                candidates.append(gs[:1024].copy())
    except NameError:
        pass

    # Deduplicate by first few values
    seen = set()
    unique_cands = []
    for c in candidates:
        key = tuple(np.round(c[:5], 1))
        if key not in seen:
            seen.add(key)
            unique_cands.append(c)
    candidates = unique_cands

    t_phase2 = deadline - 45.0
    for ci, c in enumerate(candidates):
        if time.monotonic() > t_phase2:
            break
        n = len(c)
        if n > 2048:
            continue
        print(f"  LP[{ci}] n={n}...", end=' ', flush=True)
        t0 = time.monotonic()
        seq, score = alternating_optimize_v2(c, t_phase2, max_iters=5000)
        el = time.monotonic() - t0
        tag = " *" if score < best_score else ""
        print(f"score={score:.6f} ({el:.1f}s){tag}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Phase 3: Final refinement
    if best_seq is not None:
        print(f"\nPhase 3: Final refinement (score={best_score:.6f})")
        rem = deadline - time.monotonic() - 8.0
        if rem > 15:
            seq, score = alternating_optimize_v2(best_seq, deadline - 8.0, max_iters=10000)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Deep LP: {score:.6f}")
            if time.monotonic() < deadline - 10:
                seq, score = grad_optimize(best_seq, deadline - 8.0, max_iters=3000)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"  Grad polish: {score:.6f}")

    if best_seq is not None:
        final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        fs = compute_obj(np.array(final))
        print(f"\nFinal: n={len(final)}, obj={fs:.6f}")
        return final

    print("Fallback")
    return [1.0] * 256
