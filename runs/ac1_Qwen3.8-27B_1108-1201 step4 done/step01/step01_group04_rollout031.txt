Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based fixed-point iteration (min cross-conv peak), FFT convolution,
gradient refinement, and multi-length exploration within time budget.

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a, b):
    """Compute linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _auto_convolve(a):
    """Compute auto-convolution conv(a,a) via FFT."""
    n = len(a) + 1  # result length is 2*len(a)-1
    size = 1
    base = 2 * len(a) - 1
    while size < base:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:base]


def _compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return float('inf')
    c = _auto_convolve(a)
    return 2.0 * n * c.max() / (S * S)


def _build_toeplitz(a):
    """Build Toeplitz matrix T where (T@x)[k] = (a*x)[k] for cross-conv."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            T[k, j] = a[k - j]
    return T


def _build_toeplitz_fast(a):
    """Build Toeplitz matrix using signal.toeplitz-like approach."""
    n = len(a)
    # Column j of T has a[k-j] at row k (where valid)
    # Column 0: [a[0], a[1], ..., a[n-1], 0, ..., 0]
    # Column j: [0, ..., 0, a[0], a[1], ..., a[n-1-j], 0, ..., 0]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n - j, j] = a[:n - j]
    return T


def _solve_lp_cross(a, M):
    """
    Solve LP: max sum(x) s.t. (a*x)[k] <= M for all k, x >= 0.
    Returns x* or None.
    """
    n = len(a)
    T = _build_toeplitz_fast(a)
    # max sum(x) => min -sum(x)
    c_obj = -np.ones(n)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _solve_lp_minpeak(a):
    """
    Solve LP: min t s.t. (a*x)[k] <= t for all k, sum(x) = 1, x >= 0.
    Returns (x*, t*) or (None, None).
    """
    n = len(a)
    T = _build_toeplitz_fast(a)
    # Variables: x_0,...,x_{n-1}, t  (n+1 total)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    # Inequality: T@x - t <= 0  =>  [T | -1] @ [x; t] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(x) = 1  =>  [1...1 | 0] @ [x; t] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def _gradient_step(a, c, lr=0.05):
    """
    Subgradient step on max of convolution with sum constraint.
    Uses soft-max weighting over peaks.
    """
    n = len(a)
    M = c.max()
    # Soft-max weights on convolution values
    tau = 50.0
    w = np.exp(tau * (c - M))
    w /= w.sum()

    # Gradient: for each j, grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    grad = np.zeros(n)
    for j in range(n):
        # c[k] depends on a[j] through a[k-j] where 0 <= k-j < n
        # i.e., k in [j, j+n-1], intersect [0, 2n-2]
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            # a[k-j] for k in [k_lo, k_hi] => a[0] to a[k_hi - j]
            idx = np.arange(k_hi - k_lo + 1)
            grad[j] = 2.0 * np.dot(a[idx], w[k_lo:k_hi + 1])

    # Project: maintain sum(a) = 1 by subtracting mean
    grad -= grad.mean()

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 0:
        a_new = a_new / S
    else:
        a_new = np.ones(n) / n
    return a_new


def _gradient_step_hard(a, c, lr=0.03):
    """Hard gradient step at the single maximum position."""
    n = len(a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    grad -= grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 0:
        a_new = a_new / S
    else:
        a_new = np.ones(n) / n
    return a_new

                best_a = a.copy()
                best_score = 2.0 * n * M
                stagnant = 0

        # Periodic: try hard gradient to escape
        if stagnant % 50 == 49 and stagnant > 0:
            a_hard = _gradient_step_hard(a, c, lr * 2)
            c_hard = _auto_convolve(a_hard)
            if c_hard.max() < M:
                a = a_hard
                c = c_hard
                M = c.max()
                stagnant = 0

        # Decay learning rate slowly
        if it % 1000 == 999:
            lr *= 0.95

    return best_a, best_M, best_score


def _refine_gradient(a0, deadline, max_iters=5000):
    """Refine using gradient steps with FFT."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    c = _auto_convolve(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M

    lr = 0.02
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        a = _gradient_step(a, c, lr)
        c = _auto_convolve(a)
        M = c.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            best_score = 2.0 * n * M
        if it % 500 == 499:
            lr *= 0.9

    return best_a, best_M, best_score


def _make_structured_start(n, rng, style=0):
    """Create a structured starting sequence of length n, normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    if style == 0:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * i / n)
    elif style == 1:
        # Chirp
        a = 0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n)
    elif style == 2:
        # Gaussian center
        center = n / 2.0
        a = np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2)
    elif style == 3:
        # Modulated
        a = np.abs(np.sin(np.pi * i / n)) + 0.1
    elif style == 4:
        # Linear ramp (reversed)
        a = np.linspace(n, 1, n)
    elif style == 5:
        # Two bumps
        a = (np.exp(-0.5 * ((i - n * 0.25) / (n * 0.15)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.75) / (n * 0.15)) ** 2))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 0:
        a = a / S
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: LP-based fixed-point optimization with gradient refinement.
    Explores multiple sequence lengths and starting points.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)

    best_seq = None
    best_score = float('inf')

    def _try_and_update(a_optimized, n):
        nonlocal best_seq, best_score
        S = a_optimized.sum()
        if S < 1e-12:
            return
        score = 2.0 * n * _auto_convolve(a_optimized).max() / (S * S)
        if score < best_score:
            best_score = score
            best_seq = a_optimized.copy()
            print(f"  [n={n}] New best: score={score:.8f}")

    # Get height_sequence_1 if available
    hs = None
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size < 8 or not np.all(np.isfinite(hs)):
            hs = None
    except Exception:
        hs = None

    # Define search plans: (length, time_alloc, description)
    # Allocate more time to the primary length (512 from previous best)
    if hs is not None and len(hs) >= 512:
        plans = [
            (512, 500, "HS-512-LP"),
            (384, 200, "HS-384-LP"),
            (640, 250, "HS-640-LP"),
            (256, 150, "HS-256-LP"),
            (768, 200, "HS-768-LP"),
        ]
    else:
        plans = [
            (512, 400, "Struct-512"),
            (256, 250, "Struct-256"),
            (384, 250, "Struct-384"),
            (640, 300, "Struct-640"),
        ]

    # Scale time allocations to fit budget
    total_alloc = sum(p[1] for p in plans)
    available = deadline - t_start - 100  # reserve 100s for refinement
    if total_alloc > available:
        scale = available / total_alloc
        plans = [(p[0], int(p[1] * scale), p[2]) for p in plans]

    print(f"Search budget: {budget_s}s, plans: {[(p[0], p[1]) for p in plans]}")

    for n, time_alloc, desc in plans:
        t_plan_start = time.monotonic()
        t_plan_end = t_plan_start + time_alloc

        print(f"\n--- {desc} (n={n}, {time_alloc}s) ---")

        # Create starting sequence
        if hs is not None and len(hs) >= n:
            a0 = hs[:n].copy()
        else:
            a0 = _make_structured_start(n, rng, style=n % 6)

        S0 = a0.sum()
        if S0 > 0:
            a0 = a0 / S0

        # Run LP fixed-point optimization
        result_a, result_M, result_score = _optimize_lp_fixedpoint(
            a0, t_plan_end)
        _try_and_update(result_a, n)

        elapsed = time.monotonic() - t_plan_start
        print(f"  LP done: {elapsed:.1f}s, best_M={_auto_convolve(result_a).max():.6f}")

        # If time remains for this plan, do gradient refinement
        t_now = time.monotonic()
        if t_now < t_plan_end - 5:
            refine_deadline = t_plan_end
            ref_a, ref_M, ref_score = _refine_gradient(
                result_a, refine_deadline, max_iters=3000)
            _try_and_update(ref_a, n)
            print(f"  After refine: score={2.0*n*ref_M:.8f}")

    # --- Final refinement of global best ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n--- Final refinement (current best={best_score:.8f}) ---")
        n = len(best_seq)
        a = best_seq.copy()
        S = a.sum()
        if S > 0:
            a = a / S

        # Multiple LP rounds on the best
        final_deadline = deadline
        result_a, result_M, result_score = _optimize_lp_fixedpoint(
            a, final_deadline, max_iters=50000)
        _try_and_update(result_a, n)

        # Gradient polish
        if time.monotonic() < deadline - 3:
            ref_a, ref_M, ref_score = _refine_gradient(
                result_a, deadline, max_iters=2000)
            _try_and_update(ref_a, n)

    # --- Also try a few random restarts if time permits ---
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n--- Random restarts ---")
        for n_try in [256, 384, 512]:
            t_now = time.monotonic()
            if t_now > deadline - 30:
                break
            t_alloc = min(25, deadline - t_now - 5)
            a0 = _make_structured_start(n_try, rng, style=rng.integers(0, 6))
            S0 = a0.sum()
            if S0 > 0:
                a0 = a0 / S0
            result_a, result_M, result_score = _optimize_lp_fixedpoint(
                a0, t_now + t_alloc, max_iters=50000)
            _try_and_update(result_a, n_try)

    # Final output
    if best_seq is not None:
        final_score = _compute_score(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback to uniform sequence.")
    return [1.0] * 256
