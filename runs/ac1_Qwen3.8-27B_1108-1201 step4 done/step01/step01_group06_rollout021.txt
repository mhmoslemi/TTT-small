import numpy as np

from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

            # Diagonal correction
            if k % 2 == 0:
                i_diag = k // 2
                if lo <= i_diag <= hi:
                    grad[i_diag] -= wk * a[i_diag]
    return grad

def compute_single_grad(a, k):
    """Compute subgradient at a single lag k."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        grad[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        if k % 2 == 0:
            i_diag = k // 2
            if lo <= i_diag <= hi:
                grad[i_diag] = a[i_diag]
    return grad

def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * float(n))
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S/n)
    return a * (S / s)

def run_optimization(a_init, n, max_time, rng, T_init=1.0, T_final=0.01, 
                     lr_init=0.5, top_k=30, momentum_schedule='cosine'):
    """Adam optimizer with log-sum-exp smoothed subgradient."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Anneal temperature
        progress = min(1.0, t / max(1000, max_time * 2))
        T = T_init * (T_final / T_init) ** progress
        
        # Compute objective
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Compute gradient
        grad = compute_smoothed_grad(a, c, T, k_max=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm  # normalize for stable Adam
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Learning rate schedule
        lr = lr_init * np.sqrt(1 - beta2**t) / (1 + 0.001 * t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Periodic perturbation for escaping local minima
        if t % 2000 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.05 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_a, best_score
