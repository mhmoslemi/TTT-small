```python
def solve_lp_peak_reduction(f, time_limit=5.0):
    """Solve: min M' s.t. (f*g)[k] <= M' for all k, sum(g)=1, g>=0."""
    n = len(f)
    # A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # Variables: [g_0, ..., g_{n-1}, M']
    # Objective: minimize M' (coefficient 1 for last variable, 0 for others)
    # Constraints:
    #   (f*g)[k] - M' <= 0 for k = 0,...,2n-2
    #   sum(g) = 1
    #   g_i >= 0, M' >= 0 (or free)
    
    from scipy.optimize import linprog
    import scipy.sparse as sp
    
    # Build A (2n-1 x n)
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A_conv = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
    
    # Full constraint matrix: [A_conv, -I] (2n-1 x (n+1))
    # Each row: (f*g)[k] - M' <= 0
    neg_I = sp.diags([-1.0], 0, shape=(2*n-1, 2*n-1)).tocsr()
    b = np.zeros(2*n-1)
    A_ub = sp.hstack([A_conv, neg_I])
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M'
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Bounds: g_i >= 0, M' >= 0 (or free)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit})
    if res.status == 0:
        g = res.x[:n]
        return np.maximum(g, 0)
    return None
```

def solve_lp_min_peak(f, time_limit=5.0):
    """Solve: min M' s.t. (f*g)[k] <= M' for all k, sum(g)=1, g>=0.
    Returns g (normalized, sum=1) or None."""
    n = len(f)
    # Build sparse matrix A (2n-1 x n): A[k,j] = f[k-j] if valid
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(j, j+n):
            rows.append(k)
            cols.append(j)
            data.append(f[k-j])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    
    # Variables: [g_0, ..., g_{n-1}, M']  (n+1 variables)
    # Minimize M': c = [0,...,0, 1]
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Inequality: A*g - M' <= 0, i.e., [A, -I] * [g, M'] <= 0
    # But -I is (2n-1) x (2n-1), and we only need the diagonal
    # A_ub is (2n-1) x (n+1)
    A_ub = sp.hstack([A, -sp.eye(2*n-1, format='csr')]).tocsr()
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = res.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                g /= s
            return g
    except Exception:
        pass
    return None

    # The last term is a constant: C_max = max(base_conv[k] for k not in [k_lo, k_hi])
    
    k_not_in_range = np.concatenate([np.arange(0, k_lo), np.arange(k_hi+1, 2*n-1)]) if k_lo > 0 or k_hi < 2*n-2 else np.array([])
    C_max = np.max(base_conv[k_not_in_range]) if len(k_not_in_range) > 0 else -np.inf
    
    best_x = x_lo
    best_val = np.inf
    
    # Bisection on the derivative (subgradient)
    # The subgradient of obj(x) w.r.t. x is:
    # - From the quadratic: 2x (if 2i is in range)
    # - From each linear: 2*f[k-i] (if that term is active)
    # At the optimum, 0 is in the subdifferential.
    
    # Simpler: just do a fine grid search + refinement
    # For n up to 4096, this should be fast enough
    
    num_evals = 200
    x_candidates = np.linspace(x_lo, x_hi, num_evals)
    
    if 0 <= 2*i < 2*n-1:
        quad = x_candidates**2 + base_conv[2*i]
    else:
        quad = np.zeros(num_evals) - np.inf
    
    # Linear terms
    k_range = np.arange(k_lo, k_hi+1)
    if len(k_range) > 0:
        coeffs = 2.0 * f[k_range - i]  # f[k-i] for k in range
        # But exclude k = 2i if it's in range (already in quad)
        mask = k_range != 2*i
        if np.any(mask):
            lin = coeffs[mask][:, None] * x_candidates[None, :] + base_conv[k_range[mask]][:, None]
            lin_max = lin.max(axis=0)
        else:
            lin_max = np.zeros(num_evals) - np.inf
    else:
        lin_max = np.zeros(num_evals) - np.inf
    
    obj_vals = np.maximum(np.maximum(quad, lin_max), C_max)
    
    idx = np.argmin(obj_vals)
    best_x = x_candidates[idx]
    
    # Refine with finer search around best_x
    margin = max(x_hi - x_lo, 1e-6) / num_evals
    x_fine = np.linspace(max(0, best_x - 5*margin), best_x + 5*margin, 200)
    x_fine = x_fine[x_fine >= 0]
    
    if 0 <= 2*i < 2*n-1:
        quad_f = x_fine**2 + base_conv[2*i]
    else:
        quad_f = np.zeros(len(x_fine)) - np.inf
    
    if len(k_range) > 0:
        mask = k_range != 2*i
        if np.any(mask):
            lin_f = coeffs[mask][:, None] * x_fine[None, :] + base_conv[k_range[mask]][:, None]
            lin_max_f = lin_f.max(axis=0)
        else:
            lin_max_f = np.zeros(len(x_fine)) - np.inf
    else:
        lin_max_f = np.zeros(len(x_fine)) - np.inf
    
    obj_f = np.maximum(np.maximum(quad_f, lin_max_f), C_max)
    idx_f = np.argmin(obj_f)
    best_x = x_fine[idx_f]
    
    if obj_f[idx_f] >= obj_vals[idx] - 1e-12:
        return None  # No improvement
    
    f_new = f.copy()
    f_new[i] = best_x
    return f_new

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative sequences.

Combines: (1) projected subgradient descent on the peak autocorrelation,
(2) LP-based direction finding (minimize max(f*g)), and (3) randomized
coordinate/local search for final polishing.

Score = 2*n*max(f*f) / (sum(f))^2. Scale-invariant; work with sum(f)=1.
"""
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp
import time


def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def normalize_one(f):
    """Return non-negative copy of f normalized to sum 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def get_peak_indices(conv, tol=1e-12):
    """Return indices where conv achieves its max (within tolerance)."""
    M = conv.max()
    return np.where(conv >= M - tol * max(1.0, M))[0]


def compute_subgradient(f, peak_idx):
    """
    Subgradient of max_k (f*f)[k] w.r.t. f.
    d/d f[i] of (f*f)[k] = 2*f[k-i] if 0 <= k-i < n, else 0.
    Subgradient = sum over peak k of that derivative.
    """
    n = len(f)
    g = np.zeros(n)
    for k in peak_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            g[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1]
    return g


def project_nonneg_sum1(v):
    """Project v onto {x >= 0, sum(x) = 1} via bisection on threshold."""
    n = len(v)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    result = np.maximum(v - tau, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def subgradient_step(f, eta):
    """One projected subgradient step to reduce max(f*f)."""
    conv = np.convolve(f, f)
    peak_idx = get_peak_indices(conv)
    g = compute_subgradient(f, peak_idx)
    # Center the subgradient (Lagrange multiplier for sum constraint)
    g = g - np.mean(g)
    f_new = f - eta * g
    return project_nonneg_sum1(f_new)


def solve_lp_min_peak(f, time_limit=5.0):
    """
    Solve: min M' s.t. (f*g)[k] <= M' for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(f)
    # Build A (2n-1 x n): A[k,j] = f[k-j] if 0 <= k-j < n
    j_range = np.arange(n)
    rows = []
    cols = []
    data = []
    for j in range(n):
        ks = np.arange(j, j + n)
        rows.extend(ks.tolist())
        cols.extend([j] * n)
        data.extend(f.tolist())
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

    # Variables: [g_0,...,g_{n-1}, M']
    c = np.zeros(n + 1)
    c[n] = 1.0

    # [A, -I] * [g, M'] <= 0
    A_ub = sp.hstack([A, -sp.eye(2 * n - 1, format='csr')]).tocsr()
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_max_sum(f, M, time_limit=5.0):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None. (Original LP from reference.)
    """
    n = len(f)
    rows = []
    cols = []
    data = []
    for j in range(n):
        ks = np.arange(j, j + n)
        rows.extend(ks.tolist())
        cols.extend([j] * n)
        data.extend(f.tolist())
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def block_perturb(f, rng, max_width=None, sigma=0.3):
    """Multiplicative random block perturbation, renormalized to sum 1."""
    n = len(f)
    if max_width is None:
        max_width = max(2, n // 10)
    w = int(rng.integers(1, max_width + 1))
    w = min(w, n)
    start = int(rng.integers(0, n - w + 1))
    cand = f.copy()
    cand[start:start + w] *= float(np.exp(rng.normal(0.0, sigma)))
    return normalize_one(cand)


def single_coord_perturb(f, rng, scale=0.3):
    """Perturb a single coordinate multiplicatively."""
    n = len(f)
    cand = f.copy()
    i = int(rng.integers(0, n))
    factor = float(np.exp(rng.normal(0.0, scale)))
    cand[i] *= factor
    return normalize_one(cand)


def make_initial(n, rng, method='random'):
    """Generate initial non-negative sequence of length n, sum=1."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sin_sq':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'two_peaks':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 16)
        a = (np.exp(-0.5 * ((x - n * 0.25) / s) ** 2) +
             np.exp(-0.5 * ((x - n * 0.75) / s) ** 2)) + 0.01
    elif method == 'random_smooth':
        a = rng.random(n) + 0.01
        # Light smoothing
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    return normalize_one(a)


def run_subgradient_phase(f, deadline, eta0=0.05, decay=0.9995, eta_min=0.001):
    """Run subgradient iterations until deadline. Returns best found."""
    best = f.copy()
    best_sc = fast_score(f)
    eta = eta0
    stall = 0
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        f = subgradient_step(f, eta)
        eta = max(eta_min, eta * decay)
        if it % 20 == 0:
            sc = fast_score(f)
            if sc < best_sc:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall > 30:
                # Perturb to escape
                f = block_perturb(f, np.random.default_rng(it * 7919), sigma=0.2)
                stall = 0
    return best, best_sc


def run_lp_phase(f, deadline, use_min_peak=True, t_init=0.3, t_decay=0.99,
                 lp_time=5.0):
    """Run LP-based iterations until deadline. Returns best found."""
    best = f.copy()
    best_sc = fast_score(f)
    t = t_init
    stall = 0
    for it in range(5000):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(f, f)
        M = float(conv.max())
        if use_min_peak:
            g = solve_lp_min_peak(f, time_limit=lp_time)
        else:
            g = solve_lp_max_sum(f, M, time_limit=lp_time)
        if g is None:
            g = block_perturb(f, np.random.default_rng(it * 31 + 7), sigma=0.15)
        f_new = (1.0 - t) * f + t * g
        f_new = normalize_one(f_new)
        if np.linalg.norm(f_new - f) < 1e-10:
            f_new = block_perturb(f_new, np.random.default_rng(it * 13 + 3), sigma=0.1)
        f = f_new
        t = max(0.02, t * t_decay)
        sc = fast_score(f)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f.copy()
            stall = 0
        else:
            stall += 1
        if stall > 40:
            f = block_perturb(f, np.random.default_rng(9999 + it), sigma=0.25,
                              max_width=max(5, len(f) // 20))
            stall = 0
    return best, best_sc


def local_search_polish(f, deadline, max_iters=500000):
    """Randomized local search: try single-coord and small-block perturbations."""
    best = f.copy()
    best_sc = fast_score(f)
    rng = np.random.default_rng(12345)
    n = len(f)
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        r = rng.random()
        if r < 0.5:
            cand = single_coord_perturb(best, rng, scale=0.2)
        elif r < 0.8:
            cand = block_perturb(best, rng, max_width=max(2, n // 50), sigma=0.15)
        else:
            # Two-coordinate perturbation
            cand = best.copy()
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            cand[i1] *= float(np.exp(rng.normal(0.0, 0.2)))
            cand[i2] *= float(np.exp(rng.normal(0.0, 0.2)))
            cand = normalize_one(cand)

        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = cand
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Larger perturbation to escape
                best = block_perturb(best, rng, max_width=max(5, n // 10), sigma=0.3)
                best_sc = fast_score(best)
                stall = 0
    return best, best_sc


def optimize_sequence(f_init, deadline, n_subgrad=200, n_lp=10,
                      use_min_peak=True, seed=42):

    """
    f = normalize_one(f_init)
    best = f.copy()
    best_sc = fast_score(f)
    rng = np.random.default_rng(seed)

    # Phase 1: Subgradient (fast, many steps)
    phase1_deadline = min(deadline, time.monotonic() + 0.3 * (deadline - time.monotonic()))

    if sc1 < best_sc:
        best_sc = sc1
        best = f.copy()

    # Phase 2: LP-based (medium speed)
    phase2_deadline = min(deadline, time.monotonic() + 0.35 * (deadline - time.monotonic()))
    f2, sc2 = run_lp_phase(f, phase2_deadline, use_min_peak=use_min_peak,
                           t_init=0.25, t_decay=0.985, lp_time=5.0)
    if sc2 < best_sc:
        best_sc = sc2
        best = f2.copy()

    # Phase 3: More subgradient from LP result
    phase3_deadline = min(deadline, time.monotonic() + 0.15 * (deadline - time.monotonic()))
    f3, sc3 = run_subgradient_phase(f2, phase3_deadline, eta0=0.03, decay=0.999)
    if sc3 < best_sc:
        best_sc = sc3
        best = f3.copy()

    # Phase 4: LP with different parameters
    phase4_deadline = min(deadline, time.monotonic() + 0.1 * (deadline - time.monotonic()))
    f4, sc4 = run_lp_phase(best, phase4_deadline, use_min_peak=False,

    if remaining > 10.0:
        print("[Final polish] remaining=%.0fs" % remaining, flush=True)
        p_best, p_sc = local_search_polish(best, deadline)
        try_update(p_best, p_sc)

    # Additional LP refinement on best
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        print("[Final LP] remaining=%.0fs" % remaining, flush=True)
        lp_best, lp_sc = run_lp_phase(best, deadline, use_min_peak=True,
                                      t_init=0.1, t_decay=0.995, lp_time=5.0)
        try_update(lp_best, lp_sc)

    # Final subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        sg_best, sg_sc = run_subgradient_phase(best, deadline, eta0=0.01,
                                               decay=0.9995, eta_min=0.0005)
        try_update(sg_best, sg_sc)

    # Final local search
    p2_best, p2_sc = local_search_polish(best, deadline)
    try_update(p2_best, p2_sc)

    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512, dtype=np.float64)
    final_sc = fast_score(best)
    print("FINAL score = %.6f (n=%d)" % (final_sc, best.size), flush=True)
    try:
        auth_sc = float(evaluate_sequence(best.tolist()))
        print("Authoritative score = %.6f" % auth_sc, flush=True)
    except Exception as e:
        print("Warning: evaluator error: %s" % str(e), flush=True)

    return [float(x) for x in best]
