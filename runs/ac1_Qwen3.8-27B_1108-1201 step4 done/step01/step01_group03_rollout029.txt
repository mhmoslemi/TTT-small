This is equivalent to minimizing max_k (f*f)[k], which is the same as minimizing the peak of the inverse FFT of |F|^2.

Since the objective max_k (f*f)[k] is convex but not differentiable (only subdifferentiable at the max), we can use:

- Smoothing: replace max with a log-sum-exp approximation, then use gradient descent

**Log-sum-exp smoothing**:
max_k (f*f)[k] ≈ (1/β) * log(sum_k exp(β * (f*f)[k]))

For large β, this approximates the max. We can then use gradient descent with a projected gradient step (projecting onto the simplex).

The gradient of the smoothed objective:
∇_f L = (1/β) * sum_k (β * exp(β*(f*f)[k])) / (sum_k exp(β*(f*f)[k])) * ∇_f (f*f)[k]

= sum_k w_k * ∇_f (f*f)[k]

where w_k = exp(β*(f*f)[k]) / sum_j exp(β*(f*f)[j]) (softmax weights)

And ∇_f (f*f)[k] = 2 * flip(f, shift by k) (as derived earlier)

So the gradient is: ∇_f L = 2 * sum_k w_k * flip(f, k)

This is a weighted sum of shifted/reversed versions of f.

The projected gradient step on the simplex {f >= 0, sum(f) = 1}:
f_new = proj_simplex(f - eta * ∇_f L)

where proj_simplex is the Euclidean projection onto the simplex.

This is a well-known algorithm (Hurter-Schmeiser or Duchi et al. for simplex projection).

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def greedy_sidon_set(n, target_size=None):
    """
    Greedily construct a Sidon (B_2) set in {0, ..., n-1}.
    Returns the set as a boolean array.
    """
    if target_size is None:
        target_size = int(np.sqrt(2 * n))
    
    S = []
    sums = set()
    
    for j in range(n):
        candidate = True
        new_sums = []
        for existing in S:
            s = j + existing
            if s in sums and s != 2*j:  # Allow j+j only once
                # Check if this sum already exists from a different pair
                if (j, existing) not in sums and (existing, j) not in sums:
                    candidate = False
                    break
            new_sums.append(j + existing)
        if j * 2 not in sums or len(S) == 0:
            new_sums.append(j * 2)
        
        # Proper Sidon check: all pairwise sums must be distinct
        candidate = True
        all_new_sums = [j + s for s in S] + [2*j]
        for s_val in all_new_sums:
            if s_val in sums:
                # Check if it's already from pair (j, s) where s is in S
                # Actually for a Sidon set, we need ALL sums j1+j2 (j1<=j2) to be distinct
                # So if s_val is already in sums (from a pair not involving j), it's a conflict
                candidate = False
                break
        
        if candidate:
            S.append(j)
            sums.update(all_new_sums)
            if len(S) >= target_size:
                break
    
    indicator = np.zeros(n)
    for j in S:
        indicator[j] = 1.0
    return indicator

"""
Multi-strategy convex optimization to minimize score = 2*n*max(conv(a,a))/sum(a)^2
for non-negative sequences a.

Uses Nesterov-accelerated gradient descent with log-sum-exp smoothing of the
max function, multiple diverse initializations (uniform, cosine, Sidon-set-based,
sparse random), and block-perturbation polishing. FFT-based convolution for speed.
"""
import numpy as np
import scipy.fft as spfft
import time


def fft_convolve_full(a):
    """Full linear convolution of a with itself (length 2n-1) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = spfft.rfft(a, size)
    return spfft.irfft(A * A, size)[:2 * n - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    conv = fft_convolve_full(a)
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.ones(n) if n == 1 else v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)
