    c = -np.ones(n)

        s = g.sum()
        if s > 1e-12:

        return g

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based mirror descent.
At each step, solves an LP to find the best direction g that keeps the
convolution peak bounded, then interpolates. Multiple sequence lengths
and initializations are tried.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_solve(f, M):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k in [0, 2n-2], g >= 0.
    Returns normalized g (sum = 1) or None.
    """
    n = len(f)
    c = -np.ones(n)
    
    # A_ub[k, m] = f[k-m] if 0 <= k-m < n
    A_ub = np.zeros((2 * n - 1, n))
    for m in range(n):
        A_ub[m:m + n, m] = f
    
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def optimize_lp(a0, time_limit, verbose=False):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based mirror descent.
At each step, solves a linear program to find the optimal direction g that
minimizes the cross-correlation peak with the current iterate, then interpolates.
Tries multiple sequence lengths and diverse initializations.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def lp_min_cross_peak(f):
    """
    Solve LP: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(f)
    # Variables: g[0..n-1], t  (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    # A_ub: (2n-1) x (n+1)
    # A_ub[k, m] = f[k-m] if 0<=k-m<n for m in [0,n)
    # A_ub[k, -1] = -1 (coefficient of t)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for m in range(n):
        A_ub[m:m + n, m] = f
    A_ub[:, -1] = -1.0

    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g >= 0, t free
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success and res.x is not None:
        g = np.maximum(res.x[:n], 0.0)
        t_opt = res.x[-1]
        s = g.sum()
        if s > 1e-15:
            g = g / s
        return g, t_opt
    return None, None

                j_end = min(n - 1, k)
                if j_start <= j_end:
                    # sg[j] += w[k] * (a[k-j] + a[j-k] if j!=k-j)
                    # Simplified: a[k-j] for j in [j_start, j_end]
                    rev_slice = a[k - j_end:k - j_start + 1][::-1]
                    sg[j_start:j_end + 1] += w[k] * rev_slice

            # Mirror descent update
            sg_centered = sg - sg.mean()
            log_factor = -eta * sg_centered
            log_factor -= log_factor.max()
            a_new = a * np.exp(log_factor)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
                    eta *= 0.8
                    if eta < 0.02:
                        eta = 0.3
            else:
                no_improve += 10

        # Switch phase periodically or on stall
        if no_improve >= max_no_improve:
            # Big restart
            eps = 0.2
            a = (1.0 - eps) * a + eps * rng.dirichlet(np.ones(n))
            a = np.maximum(a, 0)
            a /= a.sum()
            no_improve = 0
            max_no_improve = max(60, max_no_improve // 2)
        elif no_improve >= 60:
            phase = 1 - phase
            if phase == 1:
                grad_power *= 1.5
                eta = 0.3
            no_improve = 0

    return best_a, best_obj


def run_optimization(seed, budget_s):
    """Main optimization routine with multiple phases."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')

    # Access parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent accessible, length={p.size}, "
                  f"score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception as e:
        print(f"No parent: {e}")

    # Build list of (name, initial_array) to try
    candidates = []

    if parent is not None:
        pn = len(parent)
        candidates.append(('parent', parent.copy()))
        # Extended versions
        for mult in [2, 4]:
            tn = pn * mult
            if tn <= 2048:
                pp = np.interp(np.linspace(0, 1, tn),
                               np.linspace(0, 1, pn), parent)
                pp = np.maximum(pp, 0)
                candidates.append((f'parent_x{mult}', pp))

    # Random seeds
    for n in [128, 192, 256, 384, 512]:
        r = rng.dirichlet(np.ones(n))
        candidates.append((f'rand_{n}', r))

    # Structured: alternating strong/weak
    for n in [128, 256]:
        alt = np.ones(n)
        alt[1::2] = 0.2
        candidates.append((f'alt_{n}', alt))

    # Structured: cosine window (non-symmetric)
    for n in [128, 256]:
        x = np.linspace(0, 1.3, n)  # asymmetric
        cw = np.sin(np.pi * x) ** 2
        cw = np.maximum(cw, 0)
        candidates.append((f'coswin_{n}', cw))

    # Phase 1: Quick scan (short time per candidate)
    print("Phase 1: Quick scan...")
    n_cand = len(candidates)
    phase1_budget = min(250.0, (deadline - time.monotonic()) * 0.25)
    time_per = max(3.0, phase1_budget / n_cand)

    results = []
    for i, (name, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 60)
        if t_alloc < 2:
            break

        a_opt, obj_opt = optimize_lp_iter(a0, time_limit=t_alloc, rng=rng)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            results.append((score, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  [{name}] n={len(a_opt)}, score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization on top results using hybrid method
    results.sort(key=lambda x: x[0])
    top_k = min(6, len(results))
    remaining = deadline - time.monotonic() - 30
    time_per_deep = max(10.0, remaining / max(1, top_k))

    print("Phase 2: Deep optimization (hybrid)...")
    for i in range(top_k):
        if time.monotonic() > deadline - 20:
            break
        score, name, a_opt = results[i]
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 20)
        if t_alloc < 5:
            break

        a_new, obj_new = optimize_hybrid(a_opt, time_limit=t_alloc, rng=rng)
        new_score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best_score = new_score
            best_sequence = a_new.copy()
            print(f"  Deep [{name}] n={len(a_new)}, score={new_score:.6f}")

    # Phase 3: More LP iterations on best
    if best_sequence is not None and time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        if remaining > 10:
            print("Phase 3: Final LP polish...")
            a_new, obj_new = optimize_lp_iter(best_sequence,
                                             time_limit=remaining,
                                             verbose=True, rng=rng)
            new_score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(new_score) and new_score < best_score:
                best_score = new_score
                best_sequence = a_new.copy()
                print(f"  Polish: score={new_score:.6f}")

    # Phase 4: Try a few more random restarts if time permits
    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        n_tries = min(3, int(remaining / 30))
        print(f"Phase 4: {n_tries} random restarts...")
        for trial in range(n_tries):
            if time.monotonic() > deadline - 15:
                break
            n = len(best_sequence) if best_sequence is not None else 128
            # Perturb best
            pert = best_sequence.copy()
            eps = 0.3
            pert = (1 - eps) * pert + eps * rng.dirichlet(np.ones(n))
            pert = np.maximum(pert, 0)
            pert /= pert.sum()

            t_alloc = min(40.0, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break
            a_new, obj_new = optimize_lp_iter(pert, time_limit=t_alloc, rng=rng)
            new_score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(new_score) and new_score < best_score:
                best_score = new_score
                best_sequence = a_new.copy()
                print(f"  Restart {trial}: score={new_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128, dtype=np.float64) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses LP-based iterative optimization to minimize
    2*n*max(conv(a,a))/(sum(a))^2. At each step, solves a linear program
    to find the optimal direction minimizing cross-correlation peak.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
