    Given current sequence a (sum=1) and target peak M,
    find g >= 0 that maximizes sum(g) subject to conv(a,g)_k <= M for all k.
    """

    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

        A[j:kend + 1, j] = a[:kend - j + 1]

Actually, I want to think more carefully about the LP direction. The LP I'm solving is:

max sum(g) s.t. A @ g <= M, g >= 0

where A[k,j] = a[k-j] (the cross-correlation matrix).

The solution g* is the "largest" (in L1 norm) non-negative vector whose cross-correlation with a stays below M. After normalizing to sum=1, this gives a direction that "spreads out" as much as possible while being compatible with the current peak.

But there's a subtlety: the LP constraint uses `a` (the current iterate), not `g`. So the constraint is on `conv(a, g)`, not `conv(g, g)`. This means the LP direction is "greedy" - it only looks at the cross-correlation, not the auto-correlation of the new point.

A potentially better approach: solve the LP with the constraint `conv(g, g)_k <= M` for all k. But this is quadratic in `g`, so it's not an LP.

Alternatively, I could use a "predictive" approach: at each step, predict what `conv(g, g)` would be and adjust M accordingly. But this is complex.

    def obj_at(t):
        x = (1 - t) * a + t * g
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-12:
            return float('inf')
        x /= s

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

            hi = d

            fd = fc

            fc = fd
            d = lo + gr * (hi - lo)

"""
LP-based Frank-Wolfe with golden-section line search, combined with
L-BFGS-B smooth gradient polishing, to minimize 2*n*max(conv(a,a))/(sum(a))^2
over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

# Globals for L-BFGS objective/gradient (avoid closures)
_grp_n = [0]
_grp_t = [50.0]

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_conv_mat(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        kend = min(2 * n - 2, j + n - 1)
        A[j:kend + 1, j] = a[:kend - j + 1]
    return A

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_conv_mat(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-12:
            return g / gs
    return None

def ls_obj(x, a, g, n):
    """Objective at point (1-t)*a + t*g, where t=x[0]."""
    t = max(0.0, min(1.0, x[0]))
    v = (1.0 - t) * a + t * g
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:
        return 1e10
    v /= s
    return 2.0 * n * fast_convolve(v).max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = ls_obj(np.array([c]), a, g, n)
    fd = ls_obj(np.array([d]), a, g, n)

            fd = fc
            c = hi - gr * (hi - lo)
            fc = ls_obj(np.array([c]), a, g, n)

            lo = c
            c = d
            fc = fd
            d = lo
