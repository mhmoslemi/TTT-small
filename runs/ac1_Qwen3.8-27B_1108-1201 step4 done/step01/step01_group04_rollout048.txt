If the peak is at position m*, then:
d/d a[j] conv(a,a)[m*] = a[m*-j] (if 0 <= m*-j < n)

So the gradient is just a shifted version of a. This is O(n) per step (after finding the peak location).

For the log-sum-exp version:
d/d a[j] log(sum_m exp(tau * conv(a,a)[m])) = (1/tau) * sum_m exp(tau*conv[m])*conv[m]/(sum exp(tau*conv[m])) * a[m-j]

This requires computing the weighted sum, which is O(n^2) naively but can be done in O(n log n) using FFT.

Let me think about whether FFT actually helps here. For n=512:
- Direct convolution: 512^2 = 262k operations
- FFT convolution: ~512 * log2(512) * 2 = ~18k operations (with padding to 1024)

So FFT is about 10x faster. For n=4096:
- Direct: 4096^2 = 16.7M
- FFT: ~4096 * log2(4096) * 2 = ~1M

So FFT is much more beneficial for larger n.

But the gradient computation is the bottleneck, not the convolution itself. The gradient requires:
- For hard max: O(n) (just extract the relevant part of a)
- For soft max: O(n^2) naively, or O(n log n) with FFT

Wait, I need to double-check the cross-correlation formula.

The circular cross-correlation of x and y is:
corr[j] = sum_{k=0}^{N-1} x[k] * y[(k+j) mod N]

And corr = ifft(conj(FFT(x)) * FFT(y))

So corr[j] = sum_k a[k] * W_full[(k+j) mod N]

For our case, we want: grad[j]/2 = sum_{k=0}^{n-1} W_full[j+k] * a[k]

Since W_full[k] = 0 for k >= 2n-1 and a[k] = 0 for k >= n:
sum_{k=0}^{n-1} W_full[j+k] * a[k] = sum_{k=0}^{N-1} W_full[(j+k) mod N] * a[k] (since the extra terms are 0)

But wait, (j+k) mod N: for j < n and k < n, j+k < 2n <= N+1. If N >= 2n, then j+k < N and no wraparound. If N = 2n-1, then j+k can be up to 2n-2 = N-1, still no wraparound.

But the correlation formula gives corr[j] = sum_k a[k] * W_full[(k+j) mod N], which is sum_k a[k] * W_full[k+j] (no wraparound for our case).

And we want grad[j]/2 = sum_k W_full[j+k] * a[k] = sum_k a[k] * W_full[j+k] = corr[j].

So grad = 2 * corr[:n]. 

Now let me also think about the hard max gradient, which is simpler:

If the peak is at position m*, then:
grad[j] = 2 * a[m*-j] if 0 <= m*-j < n, else 0

This is O(n) and doesn't need FFT.

1. Hard max gradient for the first phase (fast, O(n) per step)

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient descent (hard max + soft max),
LP-based refinement, and diverse multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len


def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def _grad_hardmax(a):
    """Gradient of max(conv(a,a)) w.r.t. a (for the peak position)."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    return grad


def _grad_softmax_fft(a, tau):
    """Gradient of smooth approximation of max(conv(a,a)) using FFT."""
    n = len(a)
    N = next_fast_len(2 * n - 1)
    A = fft(a, N)
    C = np.real(ifft(A * A))
    # C should have length N, first 2n-1 entries are conv(a,a)
    C = C[:2*n-1]
    
    C_shifted = C - C.max()
    W = np.exp(tau * C_shifted)
    W_sum = W.sum()
    if W_sum < 1e-30:
        W = np.zeros(2*n-1)
        W[C.argmax()] = 1.0
    else:
        W /= W_sum
    
    # Build W_full (padded to N)
    W_full = np.zeros(N)
    W_full[:2*n-1] = W
    
    # Cross-correlation: corr[j] = sum_k a[k] * W_full[k+j]
    A_fft = fft(a, N)
    W_fft = fft(W_full, N)
    corr = np.real(ifft(np.conj(A_fft) * W_fft))
    
    grad = 2.0 * corr[:n]
    return grad, C


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 1500:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix using sparse-like approach
    # A[m,j] = f[m-j] if 0<=m-j<n, else 0
    # This is a banded matrix (bandwidth n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _optimize_hardmax(a0, deadline, max_iters=5000):
    """Gradient descent with hard max, cosine annealing."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    
    lr_init = 0.1
    lr_min = 0.001
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Cosine annealing
        progress = it / max_iters
        lr = lr_min + 0.5 * (lr_init - lr_min) * (1.0 + np.cos(np.pi * progress))
        
        grad = _grad_hardmax(a)
        grad -= grad.mean()  # Project out mean (fixed sum)
        
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)
        
        if (it + 1) % 50 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    return best, best_score


def _optimize_softmax(a0, deadline, max_iters=3000, tau_start=10.0, tau_end=50.0):
    """Gradient descent with soft max (FFT-accelerated)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    
    lr = 0.05
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        tau = tau_start + (tau_end - tau_start) * (it / max_iters)
        progress = it / max_iters
        lr = 0.002 + 0.5 * (0.05 - 0.002) * (1.0 + np.cos(np.pi * progress))
        
        grad, C = _grad_softmax_fft(a, tau)
        grad -= grad.mean()
        
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)
        
        if (it + 1) % 100 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    return best, best_score


def _optimize_with_lp(a0, deadline, max_lp_iters=100):
    """LP-based refinement (state of the art approach)."""
    n = len(a0)
    if n > 1200:
        return a0, _compute_obj(a0)
    
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    
    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        g = _lp_solve(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g = _normalize(g, S_t)
        
        # Try different blend factors
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            # Try a small hard-max gradient step
            grad = _grad_hardmax(a)
            grad -= grad.mean()
            a = a - 0.01 * grad
            a = np.maximum(a, 0.0)
            a = _normalize(a, S_t)
            s = _compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
    
    return best, best_score


def _make_leapfrog(a0, n_target, deadline):
    """Try to extend or modify sequence to different length."""
    n = len(a0)
    if n == n_target:
        return a0
    # Truncate or pad
    if n > n_target:
        return a0[:n_target]
    else:
        # Pad with small values
        a = np.zeros(n_target)
        a[:n] = a0
        a[n:] = a0[-1] * 0.1
        return a
```

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Multi-phase: gradient descent -> LP refinement with binary search on M -> polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len


def _compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def _grad_hardmax(a):
    """Gradient of peak autocorrelation w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    return grad


def _grad_softmax_fft(a, tau):
    """FFT-based gradient of smooth peak approximation."""
    n = len(a)
    N = next_fast_len(2 * n - 1)
    A = fft(a, N)
    C = np.real(ifft(A * A))[:2 * n - 1]
    
    C_shifted = C - C.max()
    W = np.exp(tau * C_shifted)
    W_sum = W.sum()
    if W_sum < 1e-30:
        W = np.zeros(2 * n - 1)
        W[int(C.argmax())] = 1.0
    else:
        W /= W_sum
    
    W_full = np.zeros(N)
    W_full[:2 * n - 1] = W
    
    A_fft = fft(a, N)
    W_fft = fft(W_full, N)
    corr = np.real(ifft(np.conj(A_fft) * W_fft))
    
    grad = 2.0 * corr[:n]
    return grad


def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    if n > 1500:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _grad_descent(a0, deadline, max_iters=3000, lr_init=0.1, use_soft=False, tau=20.0):
    """Run gradient descent (hard or soft max) with cosine annealing."""

    if a.sum() < 1e-6:
        a = np.ones(n)

    a = _normalize(a, S_t)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
