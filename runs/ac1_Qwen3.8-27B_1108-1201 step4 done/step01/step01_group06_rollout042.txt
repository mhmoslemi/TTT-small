def make_weights(d, std_d, alpha_var, beta):
    L = d.size
    if L == 0: return np.array([], dtype=float)
    uni = np.full(L, 1.0/L, dtype=np.float64)
    pv = None
    if alpha_var > 0.0:
        s = float(d.sum())
        if s > 1e-300:
            pv = d / s
        else:
            pv = uni
    if alpha_var < 1.0:
        if beta <= 0.0 or std_d <= 1e-10:
            pm = uni
        else:
            dmax = float(d.max())
            if dmax <= 1e-300:
                pm = uni
            else:
                beta_eff = beta / max(0.4, float(std_d))
                if beta_eff < 1.0: beta_eff=1.0
                if beta_eff > 300.0: beta_eff=300.0
                z = beta_eff * (d - dmax)
                zmax = float(z.max())
                if not np.isfinite(zmax):
                    pm = uni
                else:
                    z = z - zmax
                    e = np.exp(z)
                    s = float(e.sum())

                        pm = uni

        p=pm

        p=pv
    else:
        p=alpha_var*pv + (1.0-alpha_var)*pm
        s=float(p.sum())
        if s <=0: p=uni
        else: p/=s
    p = p + 1e-9
    s=p.sum(); p/=s

"""
Search for a non-negative sequence minimizing
    F(a) = 2*n*max_k conv(a,a)_k / (sum(a))^2.

The core optimizer works with sum(a)=1 and uses full dense Fourier weights
for the smooth maximum / variance objectives. It alternates between:

* variance flattening: weights proportional to current convolution values,
* smooth peak reduction: softmax weights over convolution values,
* final peak polishing.

Updates are multiplicative (mirror descent) in the log domain, which keeps
non-negativity and avoids simplex projections.  Multiple starts are used:
the provided best sequence, smooth/Gaussian-family templates, random binary
densities, and a coarse-to-fine chain from uniform.  Candidates are
post-processed by cheap variants (reversal, symmetrization, mixing with
uniform, exponentiation) and finally verified with evaluate_sequence().
"""

import time
import numpy as np
from scipy.fft import fft, ifft

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(8, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        a = np.ones_like(a, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a / s


def upsample_seq(a, n):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(max(1, n))
    if a.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    m = a.size
    if m == n:
        return normalize_sum1(a)
    x_old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-15:
        b = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        b = b / s
    return b


def estimate_score(a):
    """FFT estimate of 2*n*max conv / sum^2, assuming a is normalized to sum 1."""
    try:
        a = normalize_sum1(a)
        n = a.size
        if n == 0 or n > 131072:
            return np.inf
        L = 2 * n - 1
        N = next_power_of_two(L)
        A = fft(a, N)
        C = ifft(A * A).real[:L]
        C = np.maximum(C, 0.0)
        s = float(C.sum())
        if s <= 1e-300:
            return np.inf
        C = C / s
        return float(2.0 * n * max(0.0, float(C.max())))
    except Exception:
        return np.inf


def make_weights(d, std_d, alpha_var, beta):
    """
    Build nonnegative weights over convolution lags, summing to one.

    alpha_var = 1: variance-like weights proportional to d.
    beta = 0: smooth-max weights reduce to uniform.
    0 < alpha_var < 1: convex mix of variance and softmax weights.
    """
    L = d.size
    if L == 0:
        return np.zeros(0, dtype=np.float64)

    uni = np.full(L, 1.0 / L, dtype=np.float64)
    pv = None

    if alpha_var > 0.0:
        s = float(np.sum(d))
        if s > 1e-300:
            pv = d / s
        else:
            pv = uni

    if alpha_var < 1.0:
        if beta <= 0.0 or std_d <= 1e-12:
            pm = uni
        else:
            dmax = float(np.max(d))
            if dmax <= 1e-300:
                pm = uni
            else:
                beta_eff = beta / max(0.4, float(std_d))
                if beta_eff < 1.0:
                    beta_eff = 1.0
                if beta_eff > 300.0:
                    beta_eff = 300.0

                z = beta_eff * (d - dmax)
                zmax = float(np.max(z))
                if not np.isfinite(zmax):
                    pm = uni
                else:
                    z = z - zmax
                    e = np.exp(z)
                    se = float(np.sum(e))
                    if se > 1e-300 and np.isfinite(se):
                        pm = e / se
                    else:
                        pm = uni

    if alpha_var <= 0.0:
        p = pm
    elif alpha_var >= 1.0:
        p = pv
    else:
        p = alpha_var * pv + (1.0 - alpha_var) * pm
        sp = float(np.sum(p))
        if sp > 1e-300 and np.isfinite(sp):
            p = p / sp
        else:
            p = uni

    # Tiny floor so no lag is completely ignored.
    p = p + 1e-9
    p = p / float(np.sum(p))
    return p


def get_stage_params(plan, fr):
    if len(plan) == 0:
        return 0.5, 20.0, 0.30, 0.50
    for end, alpha_var, beta, eta, mu in plan:
        if fr <= end:
            return alpha_var, beta, eta, mu
    last = plan[-1]
    return last[1], last[2], last[3], last[4]


def add_noise_normalize(a, rng, sigma=0.05, floor=-50.0):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.ones(8, dtype=np.float64)
    loga = np.log(np.maximum(a, 1e-300))
    k = min(n, max(8, n // 30))
    if k > 0:
        idx = rng.choice(n, size=k, replace=False)
        loga[idx] += rng.normal(0.0, sigma, size=k)
    maxlog = float(np.max(loga))
    if not np.isfinite(maxlog):
        return np.full(n, 1.0 / n, dtype=np.float64)
    shift = loga - maxlog
    shift = np.maximum(shift, floor)
    b = np.exp(shift)
    s = float(b.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return b / s


def optimize_sequence(a0, n, seconds, plan, seed=0, floor=-50.0):
    """
    Mirror-log optimizer.

    Returns:
        best_est_a, best_est, best_score_a, best_score_est
    where best_est_* minimizes the estimated peak objective and
    best_score_* minimizes a small peak+std score used by the optimizer.
    """
    seconds = float(max(0.0, seconds))
    if n is None or n <= 0:
        n = int(max(1, len(a0)))

    a = np.asarray(a0, dtype=np.float64).reshape(-1)
    if a.size != n:
        if a.size > n:
            a = a[:n]
        else:
            a = upsample_seq(a, n)
    a = normalize_sum1(a)

    N = next_power_of_two(2 * n - 1)
    L = 2 * n - 1

    loga = np.log(np.maximum(a, 1e-300))
    mom = np.zeros(n, dtype=np.float64)

    best_est = np.inf
    best_est_a = a.copy()
    best_score = np.inf
    best_score_a = a.copy()
    best_score_est = np.inf

    start = time.monotonic()
    deadline = start + seconds
    last_improve_t = start
    reset_count = 0

    if seconds <= 1e-6:
        est0 = estimate_score(a)
        std0 = 0.0
        score0 = est0
        return (
            a.copy(),
            est0,
            a.copy(),
            score0,
        )

    while time.monotonic() < deadline:
        fr = (time.monotonic() - start) / max(1e-12, seconds)
        if fr < 0.0:
            fr = 0.0
        if fr > 1.0:
            fr = 1.0

        alpha_var, beta, eta_param, mu = get_stage_params(plan, fr)

        A = fft(a, N)
        C = ifft(A * A).real[:L]
        C = np.maximum(C, 0.0)
        cs = float(np.sum(C))
        if cs <= 1e-300 or not np.isfinite(cs):
            # Numerical failure: restart uniform.
            a = np.full(n, 1.0 / n, dtype=np.float64)
            loga = np.full(n, -np.log(float(n)), dtype=np.float64)
            mom.fill(0.0)
            last_improve_t = time.monotonic()
            continue

        C = C / cs
        d = n * C
        dmax = float(np.max(d))
        est = float(2.0 * dmax)

        mean_d = float(np.mean(d))
        var_d = float(np.dot(d, d) / L - mean_d * mean_d)
        std_d = float(np.sqrt(max(0.0, var_d)))
        score = est + 0.25 * std_d

        improved = False
        if est < best_est - 1e-12:
            best_est = est
            best_est_a = a.copy()
            improved = True
        if score < best_score - 1e-12:
            best_score = score
            best_score_a = a.copy()
            best_score_est = est
            improved = True
        if improved:
            last_improve_t = time.monotonic()
            reset_count = 0

        p = make_weights(d, std_d, alpha_var, beta)
        P = fft(p, N)
        G = ifft(P * np.conjugate(A)).real[:n]

        Gc = G - float(np.mean(G))
        rms = float(np.sqrt(np.mean(Gc * Gc)))

        if rms > 1e-14 and np.isfinite(rms):
            Gc = Gc / rms
            mu = max(0.0, min(0.9, float(mu)))
            mom = mu * mom + (1.0 - mu) * Gc
            step = mom - float(np.mean(mom))

            eta = eta_param * (1.0 + 0.5 * (1.0 - fr))
            if n < 1024:
                eta *= 1.25
            elif n > 16384:
                eta *= 0.75

            loga = loga - eta * step
            maxlog = float(np.max(loga))
            if not np.isfinite(maxlog):
                a = np.full(n, 1.0 / n, dtype=np.float64)
                loga = np.full(n, -np.log(float(n)), dtype=np.float64)
                mom.fill(0.0)
                last_improve_t = time.monotonic()
                continue

            shift = loga - maxlog
            shift = np.maximum(shift, float(floor))
            anew = np.exp(shift)
            s = float(anew.sum())
            if s <= 1e-300 or not np.isfinite(s):
                a = np.full(n, 1.0 / n, dtype=np.float64)
                loga = np.full(n, -np.log(float(n)), dtype=np.float64)
                mom.fill(0.0)
                last_improve_t = time.monotonic()
                continue

            a = anew / s
            s2 = float(a.sum())
            if s2 > 1e-300 and np.isfinite(s2):
                a = a / s2
                loga = shift - np.log(s2)

        stall_threshold = max(3.0, minutes / 25.0 if False else max(3.0, seconds / 25.0))
        if time.monotonic() - last_improve_t > stall_threshold:
            if np.isfinite(best_score) and score > best_score * 1.03 + 1e-8:
                source = best_score_a
                sig = 0.03
            else:
                source = a
                sig = 0.05
            a = add_noise_normalize(source, rng, sigma=sig, floor=floor)
            loga = np.log(np.maximum(a, 1e-300))
            mom.fill(0.0)
            reset_count += 1
            last_improve_t = time.monotonic()
            if reset_count > 80:
                break

    if best_est_a is None:
        best_est_a = a
    if best_score_a is None:
        best_score_a = a
    return best_est_a, best_est, best_score_a, best_score_est


def add_candidate(cands, a, est):
    if a is None or est is None:
        return
    try:
        est = float(est)
    except Exception:
        return
    if not np.isfinite(est):
        return
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return
    s = float(a.sum())
    if s <= 1e-15:
        return
    a = a / s
    if len(cands) < 12:
        cands.append([est, a.copy()])
    else:
        worst_idx = 0
        worst_val = cands[0][0]
        for i in range(1, len(cands)):
            if cands[i][0] > worst_val:
                worst_val = cands[i][0]
                worst_idx = i
        if est < worst_val:
            cands[worst_idx] = [est, a.copy()]


def best_candidate(cands):
    best_val = np.inf
    best_a = None
    for est, a in cands:
        if np.isfinite(est) and est < best_val:
            best_val = est
            best_a = a
    return best_a, best_val


def make_seed_inits(n, rng):
    out = []
    n = int(n)
    if n <= 0:
        return out

    out.append(np.full(n, 1.0 / n, dtype=np.float64))

    x = np.arange(n, dtype=np.float64)
    for c in (0.25, 0.50, 0.75):
        v = 1.0 + c * np.cos(2.0 * np.pi * x / n)
        v = np.maximum(v, 1e-8)
        s = float(v.sum())
        if s > 0.0:
            out.append(v / s)

    for p in (0.60, 0.75, 0.85, 0.92, 0.97):
        v = (rng.random(n) < p).astype(np.float64)
        s = float(v.sum())
        if s > 10.0:
            out.append(v / s)

    tt = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for e in (1.0, 2.0, 3.0):
        v = 1.0 + 0.5 * (tt ** e + (1.0 - tt) ** e)
        v = np.maximum(v, 1e-8)
        s = float(v.sum())
        if s > 0.0:
            out.append(v / s)

    v = 0.5 + rng.random(n)
    v = v / float(v.sum())
    out.append(v)

    return out


def run_refinement_chain(cands, target_n, seconds, seed, start_a=None):
    seconds = float(max(0.0, seconds))
    if seconds <= 10.0 or target_n <= 0:
        return

    if start_a is not None:
        a = normalize_sum1(start_a)
        if a.size != target_n:
            a = upsample_seq(a, target_n)
        steps = [(target_n, None)]
        kind = "start"
    else:
        steps = []
        if target_n > 512:
            steps.append((256, 0.06))
        else:
            steps.append((target_n, None))

        if target_n > 1024:
            if len(steps) == 0 or steps[-1][0] != 1024:
                steps.append((1024, 0.16))
        if target_n > 4096:
            if len(steps) == 0 or steps[-1][0] != 4096:
                steps.append((4096, 0.26))
        if len(steps) == 0 or steps[-1][0] != target_n:
            steps.append((target_n, None))

        a = np.ones(steps[0][0], dtype=np.float64)
        a /= float(a.size)
        kind = "uniform"

    elapsed = 0.0
    m = len(steps)
    for i in range(m):
        n_i, frac = steps[i]
        if i == m - 1:
            sec = max(5.0, seconds - elapsed)
        else:
            sec = frac * seconds
        if sec < 5.0:
            continue

        if i == 0:
            plan = PLAN_FLAT
        elif i == m - 1:
            if kind == "start":
                plan = PLAN_POLISH
            else:
                plan = PLAN_EXPL
        else:
            plan = PLAN_UP

        res = optimize_sequence(a, n_i, sec, plan, seed=seed + 17 * i + 3, floor=-50.0)
        if res[0] is not None:
            add_candidate(cands, res[0], res[1])
            add_candidate(cands, res[2], res[3])

        a_next = res[0]
        if res[1] > res[3] and res[2] is not None:
            a_next = res[2]

        if i < m - 1:
            a = upsample_seq(a_next, steps[i + 1][0])
        else:
            a = a_next
        elapsed += sec


def postprocess_candidate(a):
    a = normalize_sum1(a)
    n = a.size
    best = a.copy()
    best_val = estimate_score(a)

    rev = a[::-1]
    sym = 0.5 * (a + rev)
    uni = np.full(n, 1.0 / n, dtype=np.float64)

    for lam in (0.05, 0.10, 0.20):
        v = (1.0 - lam) * a + lam * uni
        v = normalize_sum1(v)
        val = estimate_score(v)
        if val < best_val:
            best_val = val
            best = v

    for v0 in (rev, sym):
        v = normalize_sum1(v0)
        val = estimate_score(v)
        if val < best_val:
            best_val = val
            best = v

    for ex in (0.80, 0.70):
        v = np.power(np.maximum(a, 1e-300), ex)
        s = float(v.sum())
        if s > 1e-15:
            v = v / s
            val = estimate_score(v)
            if val < best_val:
                best_val = val
                best = v

    return best, best_val


def verify_and_get(a):
    try:
        a = np.asarray(a, dtype=np.float64).reshape(-1)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return np.inf, None
        s = float(a.sum())
        if s < 0.01:
            return np.inf, None
        # Scale to a comfortable machine range. The score is scale invariant.
        scaled = a * (10.0 / s)
        scaled = np.clip(scaled, 0.0, 1000.0)
        val = evaluate_sequence(scaled.tolist())
        val = float(val)
        if np.isfinite(val):
            return val, scaled.tolist()
        return np.inf, None
    except Exception:
        return np.inf, None


def propose_candidate(seed=1234, budget_s=1800, **kwargs):
    t0 = time.monotonic()
    total = float(max(60.0, budget_s))
    search_deadline = t0 + max(60.0, total - 160.0)
    rng = np.random.default_rng(seed)
    cands = []

    # Use the supplied state-of-the-art sequence if present.
    given = globals().get("height_sequence_1")
    if given is not None:
        try:
            g = np.asarray(given, dtype=np.float64).reshape(-1)
            if g.size >= 256:
                g = normalize_sum1(g)
                est = estimate_score(g)
                add_candidate(cands, g, est)
        except Exception:
            pass

    # Candidate-template estimates.
    if cands and time.monotonic() < search_deadline - 180.0:
        a0, _ = best_candidate(cands)
        if a0 is not None:
            t_pol = min(300.0, 0.30 * (search_deadline - time.monotonic()))
            if t_pol > 30.0:
                res = optimize_sequence(
                    a0,
                    a0.size,
                    t_pol,
                    PLAN_POLISH,
                    seed=seed + 1,
                    floor=-50.0,
                )
                add_candidate(cands, res[0], res[1])
                add_candidate(cands, res[2], res[3])

    # Diversify with template inits at 8192, but do not time them out.
    if time.monotonic() < search_deadline - 120.0:
        n_init = 8192
        inits = make_seed_inits(n_init, rng)
        for arr in inits:
            est = estimate_score(arr)
            add_candidate(cands, arr, est)
        # Re-polish the best template if it is much better than the given one.
        a_best, _ = best_candidate(cands)
        if a_best is not None:
            t_mod = min(180.0, 0.18 * (search_deadline - time.monotonic()))
            if t_mod > 30.0:
                res = optimize_sequence(
                    a_best,
                    a_best.size,
                    t_mod,
                    PLAN_POLISH,
                    seed=seed + 11,
                    floor=-50.0,
                )
                add_candidate(cands, res[0], res[1])
                add_candidate(cands, res[2], res[3])

    # Coarse-to-fine uniform search at 8192.
    if time.monotonic() < search_deadline - 200.0:
        t_mg = min(420.0, 0.32 * (search_deadline - time.monotonic()))
        if t_mg > 80.0:
            run_refinement_chain(cands, 8192, t_mg, seed + 100, start_a=None)

    # Direct random-binary/smooth exploration at 8192, if time remains.
    if time.monotonic() < search_deadline - 160.0:
        n_rand = 8192
        rb = (rng.random(n_rand) < 0.82).astype(np.float64)
        s = float(rb.sum())
        if s > 10.0:
            rb = rb / s
        t_rand = min(240.0, 0.20 * (search_deadline - time.monotonic()))
        if t_rand > 60.0:
            res = optimize_sequence(
                rb,
                n_rand,
                t_rand,
                PLAN_EXPL,
                seed=seed + 200,
                floor=-50.0,
            )
            add_candidate(cands, res[0], res[1])
            add_candidate(cands, res[2], res[3])

    # If 8192 found a strong candidate, extend to 16384 and polish.
    if time.monotonic() < search_deadline - 180.0:
        a_best, v_best = best_candidate(cands)
        if a_best is not None and np.isfinite(v_best) and a_best.size < 16384:
            t_16 = min(520.0, search_deadline - time.monotonic())
            if t_16 > 120.0:
                a16 = upsample_seq(a_best, 16384)
                res = optimize_sequence(
                    a16,
                    16384,
                    t_16,
                    PLAN_POLISH,
                    seed=seed + 300,
                    floor=-50.0,
                )
                add_candidate(cands, res[0], res[1])
                add_candidate(cands, res[2], res[3])

    # Post-process the kept candidates with cheap deterministic variants.
    kept = []
    for est, a in cands:
        a2, est2 = postprocess_candidate(a)
        kept.append([min(est, est2), a2])

    # Verify a few best candidates with the authoritative evaluator.
    order = []
    remaining = list(range(len(kept)))
    while remaining:
        mi = remaining[0]
        mv = kept[mi][0]
        for i in remaining[1:]:
            if kept[i][0] < mv:
                mv = kept[i][0]
                mi = i
        order.append(mi)
        remaining.remove(mi)

    max_verify = 4
    if kept and max(a.size for _, a in kept) <= 8192:
        max_verify = 8

    best_exact = np.inf
    best_list = None
    for mi in order[:max_verify]:
        _, a = kept[mi]
        val, lst = verify_and_get(a)
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_list = lst
        if time.monotonic() > t0 + total - 20.0:
            break

    if best_list is None:
        fb = np.ones(128, dtype=np.float64)
        best_list = (fb * (10.0 / float(fb.sum()))).tolist()

    return best_list
