        print(f"  Convex n={n}: best_score={best_sc:.8f}, time={elapsed:.1f}s", flush=True)
    return best_f, best_sc, elapsed


def mirror_descent_step(f, k_star, alpha):
    """One mirror descent (multiplicative weights) step.
    Gradient of c_k at f: grad[j] = 2*f[k-j] if 0<=k-j<n, with factor 1 if 2j=k.
    Update: f_new = f * exp(-alpha * grad) / sum(...)
    """
    n = len(f)
    # Compute gradient of c_{k_star}
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            if 2 * j == k_star:
                grad[j] = f[idx]
            else:
                grad[j] = 2.0 * f[idx]

    # Mirror descent update (exponential tilt)
    log_f = np.log(np.maximum(f, 1e-30))
    log_f_new = log_f - alpha * grad
    log_f_new -= np.max(log_f_new)  # numerical stability
    f_new = np.exp(log_f_new)
    s = f_new.sum()
    if s < 1e-30:
        return f
    return f_new / s


def mirror_descent_search(f, budget_s=120.0, verbose=True):
    """Iterative mirror descent / multiplicative weights for the convex problem."""
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)

    deadline = time.monotonic() + budget_s
    alpha = 1.0 / np.sqrt(n)
    no_improve = 0

    while time.monotonic() < deadline:
        max_c, k_star, _ = compute_conv_max_and_index(f)
        f_new = mirror_descent_step(f, k_star, alpha)
        sc = fast_score(f_new)

        if np.isfinite(sc) and sc < best_sc:
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = f_new.copy()
                no_improve = 0
                if verbose and best_sc < 1.55:
                    print(f"    MD score={best_sc:.8f} alpha={alpha:.6f}", flush=True)
            f = f_new
        else:
            no_improve += 1
            f = f_new  # still take the step

        # Adaptive learning rate
        if no_improve > 200:
            alpha *= 0.7
            no_improve = 0
        elif no_improve > 50:
            alpha *= 0.9

    return best, best_sc


def block_perturb(f, rng, scale=0.05):
    """Random block multiplicative perturbation, renormalized."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(1, max(2, n // 6)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    cand[start:start + w] *= np.exp(rng.normal(0, scale, w))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 0.01:
        return f
    return cand / s


def element_perturb(f, rng, scale=0.05):
    """Perturb a few random elements additively."""
    n = len(f)
    cand = f.copy()
    n_pert = int(rng.integers(1, max(2, n // 32) + 1))
    idxs = rng.integers(0, n, n_pert)
    cand[idxs] += rng.normal(0, scale, n_pert)
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 0.01:
        return f
    return cand / s


def local_search(f, budget_s=60.0, verbose=False):
    """Combined local search with block and element perturbations."""
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    rng = np.random.default_rng(777)

    deadline = time.monotonic() + budget_s
    scale = 0.03
    batch_size = 10

    while time.monotonic() < deadline:
        improved = False
        for _ in range(batch_size):
            if rng.random() < 0.6:
                cand = block_perturb(f, rng, scale)
            else:
                cand = element_perturb(f, rng, scale)

            sc = fast_score(cand)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand.copy()
                f = cand.copy()
                improved = True
                if verbose:
                    print(f"      LS: {best_sc:.8f}", flush=True)

        if not improved:
            scale *= 0.9
            if scale < 0.001:
                scale = 0.03  # occasional restart
        else:
            scale *= 0.995

    return best, best_sc


def make_hann_window(n):
    """Create a normalized Hann-window-like starting point."""
    x = np.arange(n)
    f = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def make_gaussian_window(n):
    """Gaussian-shaped window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def propose_candidate(seed=42, budget_s=1800):
    """Main search: convex optimization + mirror descent + local search."""
    total_start = time.monotonic()
    deadline = total_start + max(1.0, budget_s - 10.0)

    # Try to use parent as starting point
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all() and arr.sum() > 0.01:
            parent = arr / arr.sum()
            print(f"Parent: n={arr.size}, score={float(evaluate_sequence([float(x) for x in arr])):.6f}", flush=True)
    except Exception:
        pass

    if parent is not None:
        best = parent.copy()
        best_sc = float(evaluate_sequence([float(x) for x in best]))
    else:
        best = make_hann_window(512)
        best_sc = fast_score(best)
        print(f"Generated Hann window n=512, score={best_sc:.6f}", flush=True)

    def update_best(f, sc):
        nonlocal best, best_sc
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f.copy()
            print(f"  *** NEW BEST: {best_sc:.8f} (n={len(f)}) ***", flush=True)

    # Phase 1: Convex optimization for multiple n values
    print("\n=== Phase 1: Convex Optimization ===", flush=True)
    for n in [256, 512, 1024, 2048]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        tl = min(120.0, remaining * 0.4)
        print(f"\n[Convex] n={n}, time_limit={tl:.0f}s", flush=True)
        f_conv, sc_conv, _ = solve_convex_n(n, time_limit=tl)
        if f_conv is not None:
            # Verify with evaluator
            sc_verify = float(evaluate_sequence([float(x) for x in f_conv]))
            if np.isfinite(sc_verify):
                sc_conv = min(sc_conv, sc_verify)
            update_best(f_conv, sc_conv)

    # Phase 2: Mirror descent from best convex solution and other starts
    print("\n=== Phase 2: Mirror Descent ===", flush=True)
    md_budgets = []
    remaining = deadline - time.monotonic()
    if remaining > 100:
        # Mirror descent from current best
        md_budgets.append((best.copy(), min(60.0, remaining * 0.25), "best"))
        # From Hann window of best length
        md_budgets.append((make_hann_window(len(best)), min(40.0, remaining * 0.15), "hann"))
        # From Gaussian
        md_budgets.append((make_gaussian_window(len(best)), min(30.0, remaining * 0.1), "gauss"))

        # Try different lengths
        for n_try in [384, 768]:
            if n_try != len(best):
                md_budgets.append((make_hann_window(n_try), min(20.0, remaining * 0.08), f"hann_{n_try}"))

    for f_start, bud, label in md_budgets:
        if time.monotonic() > deadline:
            break
        print(f"[MD] start={label}, budget={bud:.0f}s", flush=True)
        f_md, sc_md = mirror_descent_search(f_start, bud)
        if np.isfinite(sc_md):
            sc_verify = float(evaluate_sequence([float(x) for x in f_md]))
            if np.isfinite(sc_verify):
                sc_md = min(sc_md, sc_verify)
            update_best(f_md, sc_md)

    # Phase 3: Local search polishing
    print("\n=== Phase 3: Local Search Polish ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        polish_bud = min(remaining * 0.4, 300.0)
        print(f"Polishing with budget={polish_bud:.0f}s", flush=True)
        f_pol, sc_pol = local_search(best, polish_bud)
        if np.isfinite(sc_pol):
            sc_verify = float(evaluate_sequence([float(x) for x in f_pol]))
            if np.isfinite(sc_verify):
                sc_pol = min(sc_pol, sc_verify)
            update_best(f_pol, sc_pol)

    # Phase 4: Additional exploration with different strategies
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Additional Exploration ({remaining:.0f}s left) ===", flush=True)
        rng = np.random.default_rng(seed + 999)

        # Try mirror descent with different alpha schedules
        for alpha_init in [2.0 / np.sqrt(len(best)), 5.0 / np.sqrt(len(best))]:
            if time.monotonic() > deadline:
                break
            f_cur = best.copy()
            f_cur = np.maximum(f_cur, 1e-10)
            f_cur = f_cur / f_cur.sum()
            best_md = f_cur.copy()
            best_md_sc = fast_score(f_cur)
            alpha = alpha_init
            md_deadline = min(deadline, time.monotonic() + min(60.0, remaining * 0.15))

            while time.monotonic() < md_deadline:
                max_c, k_star, _ = compute_conv_max_and_index(f_cur)
                f_cur = mirror_descent_step(f_cur, k_star, alpha)
                sc = fast_score(f_cur)
                if np.isfinite(sc) and sc < best_md_sc:
                    best_md_sc = sc
                    best_md = f_cur.copy()
                if (int(time.monotonic()) % 3) == 0:
                    alpha *= 0.98

            if best_md_sc < best_sc:
                sc_verify = float(evaluate_sequence([float(x) for x in best_md]))
                if np.isfinite(sc_verify):
                    best_md_sc = min(best_md_sc, sc_verify)
                update_best(best_md, best_md_sc)

        # Try random restarts with local search
        for _ in range(3):
            if time.monotonic() > deadline:
                break
            n_try = len(best)
            f_rand = np.ones(n_try) + rng.normal(0, 0.3, n_try)
            f_rand = np.maximum(f_rand, 0.01)
            f_rand = f_rand / f_rand.sum()
            sc_rand = fast_score(f_rand)
            if sc_rand < 1.6:
                f_ls, sc_ls = local_search(f_rand, min(30.0, (deadline - time.monotonic()) * 0.1))
                if np.isfinite(sc_ls):
                    update_best(f_ls, sc_ls)

    # Final verification
    final_sc = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_sc) and final_sc < best_sc:
        best_sc = final_sc

    print(f"\nFINAL: n={best.size}, score={best_sc:.8f}", flush=True)
    print(f"Total time: {time.monotonic() - total_start:.1f}s", flush=True)

    # Ensure validity
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or best.sum() < 0.01 or not np.isfinite(best).all():
        best = make_hann_window(256)

    return [float(x) for x in best]
