"""
Minimize 2*n*max(self_conv)/sum^2 using log-sum-exp smoothed continuation + L-BFGS-B.

Key improvements over previous approach:
  - True gradient (via log-sum-exp smoothing) instead of subgradient -> much faster L-BFGS convergence
  - Continuation method: anneal temperature tau from small to large for smooth landscape navigation
  - Scale-invariant objective avoids equality constraint, works natively with L-BFGS-B bound constraints
  - Multi-start with the provided sequence and structured initializations
  - Final subgradient polish for last-mile improvement
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_obj_and_grad(a, tau):
    """Compute smooth objective and its gradient using log-sum-exp."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = np.real(ifft(A * A))[:size]

    # Stabilized log-sum-exp: g = (1/tau)*log(sum exp(tau*B))
    B_max = B.max()
    shifted = tau * (B - B_max)
    # Clip to avoid overflow (shouldn't happen after shift, but safe)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum = B_max * tau + np.log(exp_shifted.sum())
    g = log_sum / tau

    # Softmax weights
    p = exp_shifted / exp_shifted.sum()

    # Scale-invariant objective
    f = 2.0 * n * g / (S * S)

    # Compute h[j] = sum_k p[k] * a[k-j] via FFT cross-correlation
    # h[j] = (p * a_reversed)[n-1 + j]
    fsize2 = 1 << (3 * n - 1).bit_length()
    P = fft(p, fsize2)
    Ar = fft(a[::-1].copy(), fsize2)
    conv_pa = np.real(ifft(P * Ar))
    h = conv_pa[n - 1:n - 1 + n]

    # Gradient: df/da[j] = (4n/S^2)*h[j] - 4n*g/S^3
    grad = (4.0 * n / (S * S)) * h - (4.0 * n * g) / (S ** 3)

    return f, grad


def run_lbfgs_continuation(a_start, n, tau_schedule, time_budget):
    """Run continuation method with L-BFGS-B across increasing tau values."""
    a = np.clip(a_start.copy(), 0.0, 1000.0)
    S = a.sum()
    S_target = np.sqrt(2.0 * n)
    if S > 1e-10:
        a = a * (S_target / S)

    best_a = a.copy()
    best_obj = evaluate_sequence(a.tolist())
    start_time = time.monotonic()

    print("  Continuation start: initial obj = %.6f" % best_obj)

    for i, tau in enumerate(tau_schedule):
        t_stage_start = time.monotonic()
        stage_time = (time_budget / len(tau_schedule)) * (1 + 0.3 * i)
        # Cap remaining time
        remaining = time_budget - (time.monotonic() - start_time)
        stage_time = min(stage_time, remaining - 10)
        if stage_time < 5:
            break

        print("  tau=%6.1f, time budget=%.0fs, current best=%.6f" % (tau, stage_time, best_obj))

        def obj_func(x):
            val, _ = smooth_obj_and_grad(x, tau)
            return val

        def grad_func(x):
            _, g = smooth_obj_and_grad(x, tau)
            return g

        def obj_grad_func(x):
            val, g = smooth_obj_and_grad(x, tau)
            return val, g

        # Track convergence
        n_iter = 0

        def callback(xk):
            nonlocal n_iter
            n_iter += 1
            if n_iter % 50 == 0:
                cur = evaluate_sequence(np.clip(xk, 0, 1000).tolist())
                if cur < best_obj:
                    nonlocal best_a
                    best_a = np.clip(xk, 0, 1000).copy()
                    best_obj = cur
                    print("    iter %d: improved to %.6f" % (n_iter, best_obj))
            return False  # Continue

        # Set a timeout wrapper
        timeout_hit = [False]
        t_start = time.monotonic()

        class TimeoutCallback:
            def __call__(self, grad):
                if time.monotonic() - t_start > stage_time:
                    timeout_hit[0] = True
                    return True
                return None

        try:
            result = minimize(
                obj_grad_func,
                a,

            )

                a = np.clip(result.x, 0.0, 1000.0)

        print("  tau=%6.1f done in %.0fs, best so far=%.6f" % (tau, elapsed, best_obj))

    return best_a, best_obj


def subgradient_polish(a, time_budget, top_k=20):
    """Polyak-step-size subgradient method for final polish."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = np.clip(a copy=0: a if False else a.copy(), 0.0, 1000.0)
    S = a.sum()
    if S > 1e-10:
        a = a * (S_target / S)

    best_a = a.copy()
    best_maxc = fft_convolve_self(a).max()
    best_obj = evaluate_sequence(best_a.tolist())
    start = time.monotonic()
    t = 0
    lr_init = 0.5 * S_target / n

    while time.monotonic() - start < time_budget:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_maxc:
            best_maxc = cur_max
            best_a = a.copy()
            best_obj = evaluate_sequence(best_a.tolist())

        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_indices]
        total_w = top_vals.sum()
        if total_w <= 1e-15:
            break

        grad = np.zeros(n, dtype=np.float64)
        weights = top_vals / total_w
        for w_val, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Polyak step size: lr = best_gap / ||grad||^2 (approximate)
        gap = cur_max - best_maxc
        lr = max(1e-8, min(0.1 * S_target / n, gap / (gnorm ** 2 + 1e-15)))
        lr = lr * (1.0 + 0.5 * np.sin(0.01 * t))  # slight oscillation to escape

        a = a - lr * grad
        # Project to simplex {x >= 0, sum(x) = S_target}
        a = _project_simplex(a, S_target)

    return best_a, best_obj


def _project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
