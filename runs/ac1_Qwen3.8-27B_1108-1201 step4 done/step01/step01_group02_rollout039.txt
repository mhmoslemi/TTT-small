"""
Multi-stage optimization for minimizing peak auto-convolution of non-negative sequences.
Uses LP-based refinement with proper line search, multi-peak subgradient methods,
simulated annealing, and multi-length exploration.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    """Build sparse matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    # For row k, the contribution to conv(a,g)[k] is sum_i a[i]*g[k-i]
    # So C[k, j] = a[j] if 0 <= k-j < n, else 0... 
    # Actually: conv(a,g)[k] = sum_i a[i] * g[k-i] where g is indexed 0..n-1
    # = sum_j g[j] * a[k-j] (reindexing with j = k-i)
    # So C[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row_idx.append(k)
                col_idx.append(j)
                data.append(a[idx])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def build_conv_matrix_fast(a):
    """Build sparse matrix using vectorized approach."""
    n = len(a)
    n_rows = 2 * n - 1
    # C[k, j] = a[k-j] if 0 <= k-j < n
    # For each j, the non-zero entries are at rows j, j+1, ..., j+n-1
    col_starts = np.arange(n)  # first row for each column j
    # Each column j has n non-zero entries (rows j to j+n-1)
    row_idx = np.concatenate([np.arange(j, j+n) for j in range(n)])
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, M, timeout=30.0):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    C = build_conv_matrix_fast(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def line_search_blend(a, g, t_max=1.0, n_trials=10):
    """Find best blend parameter t in [0, t_max] for a_new = (1-t)*a + t*g."""
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    best_t = 0.0
    best_score = score_sequence(a_norm)
    for i in range(1, n_trials+1):
        t = t_max * i / n_trials
        a_new = (1-t) * a_norm + t * g_norm
        a_new = normalize_nonneg(a_new)
        s = score_sequence(a_new)
        if s < best_score:
            best_score = s
            best_t = t
    # Refine around best_t
    if best_t > 0 and best_t < t_max:
        t_center = best_t
        t_width = t_max / n_trials
        for i in range(-5, 6):
            t = t_center + i * t_width / 5
            if 0 < t < t_max:
                a_new = (1-t) * a_norm + t * g_norm
                a_new = normalize_nonneg(a_new)
                s = score_sequence(a_new)
                if s < best_score:
                    best_score = s
                    best_t = t
    return best_t, best_score


def lp_refine(a, time_limit=120.0, verbose=True):

    start = time.monotonic()

        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        
        # Line search for best blend
        best_t, new_score = line_search_blend(a, g, t_max=0.8, n_trials=8)

            a_new = (1-best_t) * a + best_t * (g / sum_g)

                best_score = new_score

                a = a_new
                best_score = s

            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def multi_peak_subgradient(a, top_k=5):
    """Compute a smoothed subgradient using top-k peaks of the auto-convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    total_len = 2 * n - 1
    
    # Get top-k peak positions
    top_positions = np.argsort(conv)[-top_k:]
    weights = np.zeros(total_len)
    for p in top_positions:
        weights[p] = 1.0
    weights /= weights.sum()
    
    # Subgradient: g[j] = sum_k weights[k] * 2*a[k-j] (zero-padded)
    g = np.zeros(n)
    for k in top_positions:
        w = weights[k]
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                g[j] += w * 2.0 * a[idx]
    
    return g


def gradient_refine(a, time_limit=60.0, verbose=True):
    """Gradient-based refinement using multi-peak subgradient."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    lr = 0.1  # Learning rate
    decay = 0.995
    
    if verbose:
        print(f"    Grad start: score={best_score:.6f}")
    
    it = 0
    stagnant = 0
    while time.monotonic() - start < time_limit:
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        
        g = multi_peak_subgradient(a, top_k=5)
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Multiplicative update to maintain non-negativity
        a_new = a * np.exp(-lr * g)
        a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            if verbose and (it < 5 or it % 100 == 0):
                print(f"    Grad iter {it:4d}: score={best_score:.6f}, lr={lr:.5f}")
        else:
            # Try smaller step
            a_new = a * np.exp(-lr * 0.1 * g)
            a_new = normalize_nonneg(a_new)
            s = score_sequence(a_new)
            if s < best_score:
                a = a_new
                best_score = s
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        
        lr *= decay
        it += 1
        
        if stagnant > 100:
            # Reset learning rate
            lr = 0.05
            stagnant = 0
            if verbose:
                print(f"    Grad: restarting lr at iter {it}")
        
        if lr < 1e-6:
            break
    
    if verbose:
        print(f"    Grad done: best={best_score:.6f}, {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def simulated_annealing(a, time_limit=120.0, rng=None, verbose=True):
    """Simulated annealing for local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    T = 0.01  # Initial temperature
    T_min = 1e-6
    cool_rate = 0.998
    
    if verbose:
        print(f"    SA start: score={best_score:.6f}, T={T}")
    
    accept_count = 0
    it = 0
    while time.monotonic() - start < time_limit and T > T_min:
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        
        # Generate perturbation
        width = max(2, int(rng.integers(n // 300, n // 10)))
        start_idx = int(rng.integers(0, max(1, n - width)))
        stop_idx = min(n, start_idx + width)
        
        candidate = a.copy()
        # Multiplicative perturbation
        perturbation = np.exp(rng.normal(0.0, 0.15, size=stop_idx - start_idx))
        candidate[start_idx:stop_idx] *= perturbation
        
        # Occasionally try a larger structural change
        if rng.random() < 0.1:
            # Swap or shift a block
            shift = int(rng.integers(-n//20, n//20 + 1))
            if shift != 0:
                candidate = np.roll(candidate, shift)
                # Re-center
                center = n // 2
                if shift > 0:
                    candidate[:shift] = 0
                else:
                    candidate[shift:] = 0
        
        if rng.random() < 0.2:
            # Symmetrize
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        new_score = score_sequence(candidate)
        
        # Metropolis acceptance
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = candidate
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = candidate.copy()
                accept_count += 1
                if verbose and accept_count % 100 == 0:
                    print(f"    SA accept {accept_count}: score={best_score:.6f}, T={T:.6f}")
        
        T *= cool_rate
        it += 1
    
    if verbose:
        print(f"    SA done: best={best_score:.6f}, {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

"""
Multi-stage optimization for minimizing peak auto-convolution of non-negative sequences.
Uses reformed LP (minimize cross-peak), proper line search, multi-peak subgradient,
simulated annealing, and multi-length exploration. Targets score < 1.5027.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_cross_conv_matrix(a):
    """Build sparse matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_min_peak(a, timeout=20.0):
    """Solve: minimize t s.t. conv(a,g)[k] <= t, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    C = build_cross_conv_matrix(a)
    # Variables: g[0..n-1], t (index n)
    # Minimize: t -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    # Constraints: conv(a,g)[k] - t <= 0 for all k
    # C @ g - t <= 0 => [C, -1] @ [g; t] <= 0
    A_ub = np.hstack((C.toarray(), -np.ones((2 * n - 1, 1))))
    b_ub = np.zeros(2 * n - 1)
    # Sum constraint: sum(g) = 1 => sum(g) <= 1 and -sum(g) <= -1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x[:n])
            t_opt = res.x[n]
            return g, t_opt
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M, timeout=20.0):
    """Solve: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g or None."""
    n = len(a)
    C = build_cross_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def line_search(a, g, n_trials=15):
    """Find best blend t for a_new = (1-t)*a + t*g_normalized."""
    a_n = a / a.sum()
    g_n = g / g.sum()
    best_t = 0.0
    best_s = score_sequence(a_n)
    # Coarse search
    for i in range(1, n_trials + 1):
        t = 0.9 * i / n_trials
        a_new = normalize_nonneg((1 - t) * a_n + t * g_n)
        s = score_sequence(a_new)
        if s < best_s:
            best_s = s
            best_t = t
    # Fine search around best
    if 0 < best_t < 0.9:
        step = 0.9 / n_trials
        for i in range(-8, 9):
            t = best_t + i * step * 0.15
            if 0.001 < t < 0.95:
                a_new = normalize_nonneg((1 - t) * a_n + t * g_n)
                s = score_sequence(a_new)
                if s < best_s:
                    best_s = s
                    best_t = t
    return best_t, best_s


def lp_refine(a, time_limit=120.0, verbose=False):
    """LP-based iterative refinement using min-peak formulation."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0

    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    it = 0
    for it in range(1000):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 8.0:
            break

        conv = np.convolve(a, a)
        M = conv.max()

        # Use min-peak LP
        timeout = max(1.5, min(20.0, remaining * 0.15))
        g, t_opt = solve_lp_min_peak(a, timeout=timeout)

        if g is None:
            # Fallback to max-sum LP
            g = solve_lp_max_sum(a, M, timeout=timeout)
            if g is None:
                break
            if g.sum() <= 1.0 + 1e-10:
                break
        else:
            if t_opt >= M * (1 - 1e-10):
                break

        # Line search
        best_t, new_score = line_search(a, g, n_trials=12)
        if best_t > 0.01 and new_score < best_score - 1e-13:
            a_new = normalize_nonneg((1 - best_t) * a + best_t * (g / g.sum()))
            a = a_new
            best_score = new_score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it < 5 or it % 30 == 0):
                print(f"    LP iter {it:4d}: score={best_score:.6f}, t={best_t:.3f}")
        else:
            stagnant += 1
            # Try smaller step with max-sum LP direction
            g2 = solve_lp_max_sum(a, M, timeout=max(1.0, min(10.0, remaining * 0.1)))
            if g2 is not None and g2.sum() > 1.0 + 1e-10:
                for t_try in [0.05, 0.15, 0.3]:
                    a_try = normalize_nonneg((1 - t_try) * a + t_try * (g2 / g2.sum()))
                    s_try = score_sequence(a_try)
                    if s_try < best_score - 1e-13:
                        a = a_try
                        best_score = s_try
                        best_a = a.copy()
                        stagnant = 0
                        break

        if stagnant > 50:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break

    final_s = score_sequence(a)
    if final_s < best_score:
        best_score = final_s
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def multi_peak_subgradient(a, top_k=5):
    """Compute smoothed subgradient using top-k peaks."""
    n = len(a)
    conv = np.convolve(a, a)
    total_len = 2 * n - 1
    top_positions = np.argsort(conv)[-top_k:]
    # Weight by how close to max
    vals = conv[top_positions]
    max_val = vals.max()
    weights = (vals / max_val) ** 4  # Emphasize the top peaks
    weights /= weights.sum()

    g = np.zeros(n)
    for k, w in zip(top_positions, weights):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                g[j] += w * 2.0 * a[idx]
    return g


def gradient_refine(a, time_limit=60.0, verbose=False):
    """Gradient-based refinement with multiplicative updates."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    lr = 0.08
    decay = 0.997
    stagnant = 0

    if verbose:
        print(f"    Grad start: score={best_score:.6f}")

    it = 0
    while time.monotonic() - start < time_limit:
        if time.monotonic() - start > time_limit:
            break
        g = multi_peak_subgradient(a, top_k=5)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        for lr_try in [lr, lr * 0.3, lr * 0.1]:
            a_new = a * np.exp(-lr_try * g)
            a_new = normalize_nonneg(a_new)
            s = score_sequence(a_new)
            if s < best_score - 1e-13:
                a = a_new
                best_score = s
                best_a = a.copy()
                stagnant = 0
                break
        else:
            stagnant += 1

        lr *= decay
        it += 1
        if stagnant > 200:
            lr = 0.05
            stagnant = 0
        if lr < 1e-7:
            break

    if verbose:
        print(f"    Grad done: best={best_score:.6f}, {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def simulated_annealing(a, time_limit=120.0, rng=None, verbose=False):
    """Simulated annealing with diverse moves."""
    if rng is None:
        rng = np.random.default_rng(0)
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    T = 0.008
    T_min = 1e-7
    cool = 0.9985
    accept_count = 0

    if verbose:
        print(f"    SA start: score={best_score:.6f}")

    it = 0
    while time.monotonic() - start < time_limit and T > T_min:
        if time.monotonic() - start > time_limit:
            break

        candidate = a.copy()
        move_type = rng.random()

        if move_type < 0.5:
            # Local multiplicative perturbation
            width = max(2, int(rng.integers(n // 400, n // 8)))
            s = int(rng.integers(0, max(1, n - width)))
            e = min(n, s + width)
            candidate[s:e] *= np.exp(rng.normal(0.0, 0.12, size=e - s))
        elif move_type < 0.7:
            # Block shift
            width = max(3, int(rng.integers(n // 50, n // 8)))
            s = int(rng.integers(max(0, n // 4), max(1, n - width - n // 4)))
            e = min(n, s + width)
            shift = int(rng.integers(-max(1, n // 30), max(2, n // 30)))
            block = candidate[s:e].copy()
            new_s = max(0, s + shift)
            new_e = min(n, e + shift)
            candidate[s:e] = 0
            w = new_e - new_s
            if w > 0:
                candidate[new_s:new_e] += block[:w]
        elif move_type < 0.85:
            # Symmetric perturbation
            half = n // 2
            width = max(2, int(rng.integers(1, half)))
            s = int(rng.integers(0, max(1, half - width)))
            e = min(half, s + width)
            pert = np.exp(rng.normal(0.0, 0.1, size=e - s))
            candidate[s:e] *= pert
            candidate[n-e:n-s] *= pert  # Mirror
        else:
            # Gaussian bump add/subtract
            center = float(rng.integers(0, n))
            sigma = float(rng.integers(max(2, n // 50), max(3, n // 6)))
            t = np.arange(n, dtype=np.float64)
            bump = np.exp(-0.5 * ((t - center) / sigma) ** 2)
            amplitude = rng.normal(0.0, 0.05)
            candidate += amplitude * bump

        candidate = np.maximum(0, candidate)
        if rng.random() < 0.15:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        new_score = score_sequence(candidate)

        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-max(0, delta) / max(T, 1e-10)):
            a = candidate
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_a = candidate.copy()
                accept_count += 1
                if verbose and accept_count % 200 == 0:
                    print(f"    SA accept {accept_count}: score={best_score:.6f}, T={T:.6f}")

        T *= cool
        it += 1

    if verbose:
        print(f"    SA done: best={best_score:.6f}, {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit=60.0, rng=None, verbose=False):
    """Gentle local search with small perturbations."""
    if rng is None:
        rng = np.random.default_rng(123)
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    while time.monotonic() - start < time_limit:
        width = max(2, int(rng.integers(n // 300, n // 10)))
        s = int(rng.integers(0, max(1, n - width)))
        e = min(n, s + width)
        candidate = a.copy()
        candidate[s:e] *= np.exp(rng.normal(0.0, 0.04, size=e - s))
        candidate = normalize_nonneg(candidate)
        sc = score_sequence(candidate)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = candidate.copy()
            a = candidate
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_flat_topped(n, flat_frac=0.5):
    """Raised cosine with flat top."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half_flat = flat_frac * n / 4
    result = np.zeros(n)
    for i in range(n):
        d = abs(i - center)
        if d <= half_flat:
            result[i] = 1.0
        elif d <= n / 2:
            result[i] = 0.5 * (1 + np.cos(np.pi * (d - half_flat) / (n/2 - half_flat)))
    return result


def initial_rand_smooth(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(3, n // 15), max(4, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return np.maximum(0, a)
