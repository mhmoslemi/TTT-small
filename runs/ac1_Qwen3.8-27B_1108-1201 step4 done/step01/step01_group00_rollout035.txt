    v_hat = v / (1 - beta2**step)
    theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    return theta, m, v

                a[i] = orig * f

                if s < current_score:

                    improved = True

            a[i] = orig  # restore if no improvement found in this round

"""
Optimization of non-negative step functions to minimize the normalized peak of
linear autocorrelation. Uses log-space parameterization with Adam optimizer,
annealed soft-max for the max function, FFT-based gradient computation,
multi-length multi-start search, and coordinate-wise polishing.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score efficiently."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def grad_score_logits(theta, tau):
    """
    Compute score and gradient in log-space (a = exp(theta - max)).
    Returns (score, grad_theta, c, a).
    """
    a = np.exp(theta - np.max(theta))
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    N = 2 * n - 1

    # Softmax approximation of max
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / tau)
    sum_exp = np.sum(exp_c)
    soft_max = c_max + tau * np.log(sum_exp)
    w = exp_c / sum_exp

    # dM/da[i] = 2 * sum_j a[j] * w[i+j] = 2 * cross_correlation(a, w)[i]
    # Using FFT: cross_corr[i] = conv(a, w_rev)[N-1-i]
    w_rev = w[::-1]
    size = 1 << (N + n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fw_rev = np.fft.rfft(w_rev, size)
    conv_aw = np.fft.irfft(fa * fw_rev, size)
    # x[i] = conv_aw[N-1-i] for i=0..n-1
    # = conv_aw[n-1 : 2n-1] reversed
    cross_corr = conv_aw[n - 1:n + n - 1 - 1 + 1][::-1]
    # Let me verify: we need indices conv_aw[N-1], conv_aw[N-2], ..., conv_aw[N-n]
    # N-1 = 2n-2, N-n = n-1
    # So indices are 2n-2, 2n-3, ..., n-1 -> that's conv_aw[n-1:2n-1][::-1]
    cross_corr = conv_aw[n - 1:2 * n - 1][::-1]

    dM_da = 2.0 * cross_corr[:n]

    # df/da[i] = 2n * (dM_da[i] - 2*soft_max/S) / S^2
    df_da = (2.0 * n / (S * S)) * (dM_da - 2.0 * soft_max / S)

    grad_theta = df_da * a

    score = 2.0 * n * soft_max / (S * S)
    return score, grad_theta, c, a

        improved = False
        factors = coarse_factors if pass_num < 3 else fine_factors

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            orig = a[i]
            best_f = 1.0
            best_s = current_score
            for f in factors:
                a[i] = orig * f
                s = compute_score(a)
                if s < best_s:
                    best_s = s
                    best_f = f
            a[i] = orig * best_f
            if best_f != 1.0:
                current_score = best_s
                improved = True

        if not improved:
            break

    return a, current_score


def make_init(n, init_type, rng):
    """Create various initial non-negative sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        phase = rng.uniform(0, np.pi)
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1) + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        n_tones = rng.integers(3, 7)
        for _ in range(n_tones):
            freq = rng.integers(1, n // 3 + 1)
            amp = rng.uniform(0.05, 0.35)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif init_type == 'asymmetric':
        a = 1.0 + 0.4 * np.sin(2 * np.pi * t / n + rng.uniform(0, np.pi))
        a += 0.15 * np.cos(4 * np.pi * t / (n + 1))
    elif init_type == 'exp_decay':
        rate = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.3 * np.exp(-rate * t / n) + 0.2 * np.cos(2 * np.pi * t / n)
    elif init_type == 'random':
        a = 1.0 + 0.25 * rng.standard_normal(n)
    elif init_type == 'rect':
        # Block structure
        a = np.ones(n)
        block = n // rng.integers(3, 7)
        for _ in range(rng.integers(2, 5)):
            start = rng.integers(0, n - block)
            a[start:start + block] *= rng.uniform(0.5, 1.5)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, 10.0)


def try_from_provided(provided, n, rng):
    """Resample provided sequence to target length with noise."""
    x_old = np.linspace(0, 1, len(provided))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function.
    Phase 1: Gradient optimization across multiple lengths and initializations.
    Phase 2: Coordinate-wise polishing.
    Phase 3: Final refinement.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    # Phase 1: Multi-start gradient optimization
    print("\n=== Phase 1: Gradient optimization ===")

    # (n, inits) where inits is list of (type, use_provided)
    search_plan = [
        (256, [('chirp', False), ('multi_tone', False), ('asymmetric', False), ('provided', True)]),
        (384, [('chirp', False), ('multi_tone', False), ('provided', True)]),
        (512, [('chirp', False), ('asymmetric', False), ('provided', True)]),
        (768, [('chirp', False), ('exp_decay', False)]),
        (1024, [('multi_tone', False), ('random', False)]),
        (640, [('chirp', False), ('multi_tone', False)]),
    ]

    total_runs = sum(len(inits) for _, inits in search_plan)
    print(f"  Total runs planned: {total_runs}")

    run_idx = 0
    for n, inits in search_plan:
        if time.monotonic() > deadline - 120:
            break
        for init_type, use_prov in inits:
            if time.monotonic() > deadline - 90:
                break
            run_idx += 1
            remaining = total_runs - run_idx + 1
            time_left = deadline - time.monotonic() - 90
            t_alloc = max(8.0, min(120.0, time_left / max(1, remaining) * 1.3))

            # Generate initialization
            if use_prov and provided is not None:
                a0 = try_from_provided(provided, n, rng)
            elif init_type == 'provided' and best_seq is not None:
                a0 = try_from_provided(best_seq, n, rng)
            else:
                a0 = make_init(n, init_type, rng)

            # Tune hyperparameters based on n
            tau_start = max(0.5, 5.0 / n)
            tau_end = max(0.02, 0.3 / n)
            lr_base = 3.0 + 2.0 * (n / 256)

            print(f"  [{run_idx}/{total_runs}] n={n}, init='{init_type}'{' (+prov)' if use_prov else ''}, t={t_alloc:.0f}s")
            a_best, score = optimize_adam(a0, t_alloc, tau_start, tau_end, lr_base, seed=seed + run_idx * 7)

            actual_score = compute_score(a_best)
            print(f"          score={actual_score:.6f}")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"          *** NEW BEST: {best_score:.6f} (n={n}) ***")

    # Phase 2: Coordinate polish on best
    print(f"\n=== Phase 2: Coordinate polish (current best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        polish_time = min(180.0, deadline - time.monotonic() - 30)
        if polish_time > 15:
            print(f"  Polishing n={len(best_seq)} for {polish_time:.0f}s")
            polished, polish_score = coordinate_polish(best_seq, time_limit=polish_time)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                print(f"  After polish: {best_score:.6f}")
            else:
                print(f"  No improvement from polish.")

    # Phase 3: Final gradient refinement on best sequence
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Phase 3: Final refinement ===")
        remaining = deadline - time.monotonic() - 20
        ref_time = min(remaining * 0.5, 120.0)
        if ref_time > 10:
            n_best = len(best_seq)
            a_ref, ref_score = optimize_adam(
                best_seq, ref_time,
                tau_start=0.3 / n_best,
                tau_end=0.05 / n_best,
                lr_base=1.5,
                seed=seed + 999
            )
            actual = compute_score(a_ref)
            if actual < best_score:
                best_score = actual
                best_seq = a_ref.copy()
                print(f"  After refinement: {best_score:.6f}")

            # Quick coordinate polish
            if time.monotonic() < deadline - 20:
                quick_time = min(30.0, deadline - time.monotonic() - 15)
                if quick_time > 5:
                    polished2, ps2 = coordinate_polish(best_seq, time_limit=quick_time)
                    if ps2 < best_score:
                        best_score = ps2
                        best_seq = polished2.copy()
                        print(f"  After final polish: {best_score:.6f}")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_score(np.array(final_list))
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, time={time.monotonic()-t_global:.1f}s ===")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
