import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def conv_full(p):
    """Full self-convolution of p."""
    n = len(p)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def score(p):
    """Compute the evaluation score."""
    n = len(p)
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = conv_full(p)
    return 2.0 * n * np.max(c) / (s * s)

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssum - 1.0)
    k = np.count_nonzero(cond) - 1
    if k < 0:
        return np.full(n, 1.0/n)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    return out / out.sum()

    data = []

        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            data.append(p[k-j])

    A = np.array(data, dtype=np.float64)

    # where x = [g_0, ..., g_{n-1}, t]
    # Constraint: (p*g)[k] - t <= 0 for all k
    # So A_ub[k, j] = p[k-j] for valid j, A_ub[k, n] = -1
    
    # Build the full A_ub matrix (dense for simplicity, n up to ~2000)

    A_ub = np.zeros((2*n-1, n+1), dtype=np.float64)

        j_end = min(k, n-1)

        A_ub[k, n] = -1.0
    
    # Objective: minimize t, so c = [0, ..., 0, 1]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    # Equality: sum g = 1
    A_eq = np.zeros((1, n+1))

    s = g.sum()
    if s < 1e-12:
        return None, None
    g = g / s
    return g, t

"""
Step-function search for minimum max-autocorrelation score.
Uses smoothed-max gradient descent with periodic LP best-response,
multi-length exploration, and adaptive temperature annealing.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog

# ===== Core utility functions =====

def self_convolve(p):
    """Full self-convolution of p using FFT."""
    n = len(p)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]

def compute_score(p):
    """Compute the evaluation score 2*n*max_c/sum^2."""
    n = len(p)
    s = float(np.sum(p))
    if s < 0.01:
        return float('inf')
    c = self_convolve(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (css - 1.0)
    k = int(np.count_nonzero(cond) - 1)
    if k < 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    theta = (css[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

def gradient_smoothed(p, T_factor):
    """
    Compute gradient of smoothed max surrogate.
    Returns (gradient, weights, convolution, cmax).
    """
    n = len(p)
    c = self_convolve(p)
    cmax = float(np.max(c))
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    w = np.exp(diff)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        wsum = 1.0
    w /= wsum

    # Gradient: grad[j] = 2 * sum_m p[m] * w[j+m]
    # = 2 * ifft(conj(FFT(p)) * FFT(w))[j] for j = 0..n-1
    L = next_fast_len(3 * n - 2)
    P = fft(p, L)
    W = fft(w, L)
    h = np.real(ifft(np.conj(P) * W))[:n]
    grad = 2.0 * h
    return grad, w, c, cmax

def lp_best_response(p):
    """
    Solve LP: maximize sum(g) subject to conv(p,g) <= 1, g >= 0.
    Returns (g_normalized, optimal_t) or (None, None).
    """
    n = len(p)
    if n > 2500:
        return None, None

    # Build A matrix: A[k,j] = p[k-j] if 0 <= k-j <= n-1
    k_idx = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j_idx = np.arange(n)[None, :].astype(np.int64)
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, p[np.clip(diff, 0, n - 1)], 0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'disp': False, 'time_limit': 30})
    except Exception:
        return None, None

    if res is None or not res.success:
        return None, None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None, None
    g = g / s
    t = 1.0 / s
    return g, t

def make_shape_linear(n, a, reverse=True):
    """Linear shape: p[i] = 1 + a*(i/(n-1)) or reversed."""
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + a * x
    if reverse:
        p = p[::-1]
    p = np.maximum(p, 0.01)
    return p / p.sum()

def make_shape_power(n, a, reverse=True):
    """Power shape: p[i] = (i/(n-1))^a or reversed."""
    x = np.linspace(1e-6, 1.0, n)
    p = np.power(x, a)
    if reverse:
        p = p[::-1]
    return p / p.sum()

def make_shape_cap(n, a):
    """Parabolic cap: p[i] = 1 - a*(2x-1)^2."""
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 - a * (2.0 * x - 1.0) ** 2
    p = np.maximum(p, 0.01)
    return p / p.sum()

def make_shape_cup(n, a):
    """Parabolic cup: p[i] = 1 + a*(2x-1)^2."""
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + a * (2.0 * x - 1.0) ** 2
    return p / p.sum()

def make_shape_geometric(n, rate):
    """Geometric decay."""
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-rate * x)
    return p / p.sum()

def make_shape_chebyshev(n, coeffs):
    """Chebyshev polynomial shape with endpoint taper."""
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    logp = np.zeros(n, dtype=np.float64)
    T0 = np.ones(n)
    T1 = z.copy()
    if len(coeffs) > 0:
        logp += coeffs[0] * T1
    for d in range(2, min(len(coeffs), 12)):
        T2 = 2.0 * z * T1 - T0
        logp += coeffs[d - 1] * T2
        T0 = T1
        T1 = T2
    # Endpoint taper
    if len(coeffs) >= 13:
        logp += coeffs[12] * np.log(x + 1e-6)
    if len(coeffs) >= 14:
        logp += coeffs[13] * np.log(1.0 - x + 1e-6)
    logp -= logp.max()
    p = np.exp(logp)
    return p / p.sum()

    # Random Chebyshev shapes
    for _ in range(8):

    return inits

def upsample_sequence(p_coarse, n_target, rng):
    """Upsample a coarse sequence to target length via interpolation + noise."""
    n_coarse = len(p_coarse)
    x_old = np.linspace(0.0, 1.0, n_coarse)
    x_new = np.linspace(0.0, 1.0, n_target)
    p_interp = np.interp(x_new, x_old, p_coarse)
    p_interp = p_interp + rng.normal(0, 0.02 / n_target, n_target)
    p_interp = np.maximum(p_interp, 0.0)
    s = p_interp.sum()
    if s < 1e-12:
        return np.full(n_target, 1.0 / n_target)
    return p_interp / s

            if g is not None:
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
                    p_new = project_simplex((1 - alpha) * p + alpha * g)
                    s_new = compute_score(p_new)
                    if s_new < best_score:
                        best_score = s_new
                        best_p = p_new.copy()
                        p = p_new.copy()
                        current_score = s_new
                        since_improve = 0
                        prev_p = p.copy()
                        break

        # Peak-focused refinement every 500 iterations
        if since_improve > 500 and iter_no % 500 == 0:
            c = self_convolve(p)
            # Find top peaks
            top_k = min(5, len(c))
            peak_indices = np.argsort(c)[::-1][:top_k]
            # Blend weights favoring peaks
            w_peak = np.zeros_like(c)
            for pk in peak_indices:
                w_peak[pk] = 1.0
            w_peak /= w_peak.sum()
            L = next_fast_len(3 * n - 2)
            P = fft(p, L)
            W = fft(w_peak, L)
            h = np.real(ifft(np.conj(P) * W))[:n]
            grad_peak = 2.0 * h
            # Try step in peak gradient direction
            gn = np.max(np.abs(grad_peak))
            if gn > 1e-300:
                lr_peak = (p_scale / gn) * 0.3
                p_new = project_simplex(p - lr_peak * grad_peak)
                s_new = compute_score(p_new)
                if s_new < best_score:
                    best_score = s_new
                    best_p = p_new.copy()
                    p = p_new.copy()
                    current_score = s_new
                    since_improve = 0
                    prev_p = p.copy()

        if iter_no % 2000 == 0:
            print(f"  iter={iter_no} best={best_score:.10f} cur={current_score:.10f} T={T_factor:.5f}", flush=True)

        if best_score < 1.4990 and since_improve > 5000:
            break

    return best_p, best_score

def search_shapes(n, rng, budget_s):
    """
    Low-dimensional shape search using Nelder-Mead on Chebyshev coefficients.
    """
    if budget_s < 10:
        return None, float('inf')

    from scipy.optimize import minimize

    best_theta = None
    best_p = None
    best_score = float('inf')
    end = time.monotonic() + budget_s

    def shape_obj(theta):
        p = make_shape_chebyshev(n, theta)
        s = compute_score(p)
        return s if np.isfinite(s) else 1e6

    # Multiple starts
    starts = [np.zeros(14)]
    for _ in range(5):
        starts.append(rng.uniform(-1.5, 1.5, 14))
    # Tapered starts
    t = np.zeros(14)
    t[12] = 2.0
    t[13] = 2.0
    starts.append(t)
    t = np.zeros(14)
    t[12] = 3.0
    t[13] = 3.0
    t[0] = 1.0
    starts.append(t)
    t = np.zeros(14)
    t[1] = 1.5
    t[3] = -0.8
    t[12] = 1.0
    t[13] = 1.0
    starts.append(t)

    for x0 in starts:
        if time.monotonic() > end - 5:
            break
        sc0 = shape_obj(x0)
        if sc0 < best_score:
            best_score = sc0
            best_theta = x0.copy()
            best_p = make_shape_chebyshev(n, x0)
        try:
            res = minimize(shape_obj, x0, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-7})
            if res.fun < best_score:
                best_score = float(res.fun)
                best_theta = res.x.copy()
                best_p = make_shape_chebyshev(n, res.x)
        except Exception:
            pass

    return best_p, best_score
