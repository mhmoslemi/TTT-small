def compute_score(a):
    """Compute the objective score."""
    n = len(a)
    # Use FFT for convolution
    size = 2 * n
    fa = np.fft.rfft(a, size=size)
    c = np.fft.irfft(fa * fa, size=size)[:2*n-1]
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def fft_convolve(a, a2):
    """Compute full convolution using FFT."""
    n = len(a)
    n2 = len(a2)
    size = 1
    while size < n + n2 - 1:
        size *= 2
    fa = np.fft.rfft(a, size=size)
    fb = np.fft.rfft(a2, size=size)
    return np.fft.irfft(fa * fb, size=size)[:n + n2 - 1]

    """Compute gradient of the score function."""
    n = len(a)
    S = float(a.sum())
    c = fft_convolve(a, a)
    max_c = float(c.max())
    m_star = int(c.argmax())

    grad_c = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)

- a >= 0
- convolve(a, a)[k] <= max_val for all k, where max_val = T / (2n)

This is an LP feasibility problem! We can bisect on max_val to find the minimum achievable.

The LP:
- Variables: a[0], ..., a[n-1]
- Constraints:
  - sum(a) = 1
  - a >= 0
  - For each k in 0..2n-2: sum_{j valid} a[j] * a[k-j] <= max_val

Wait, the convolution constraints are QUADRATIC in a, not linear! So this isn't an LP.

The LP from the previous code works because it's solving for g given a fixed a:
- convolve(a, g) is LINEAR in g (since a is fixed)
- So the constraints are linear in g

That's the key insight. The LP works as a "response" to the current a, finding a g that responds well to a.

    n, m = len(a), len(b)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-accelerated Adam optimization with hard/soft peak targeting,
LP-based refinement, simulated annealing restarts, and diverse initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute objective: 2*n*max(convolve(a,a))/(sum(a))^2 using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size=size)
    c = np.fft.irfft(fa * fa, size=size)[:2 * n - 1]
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    return np.fft.irfft(np.fft.rfft(a, size=size) * np.fft.rfft(b, size=size),
                        size=size)[:n + m - 1]


def hard_grad(a):
    """Gradient of score w.r.t. a at the global peak (projected to zero-mean for fixed-sum)."""
    n = len(a)
    c = fft_convolve(a, a)
    idx = int(c.argmax())
    g = np.zeros(n)
    lo = max(0, idx - n + 1)
    hi = min(n - 1, idx)
    for j in range(lo, hi + 1):
        g[j] = a[idx - j]
    g = g - g.sum() / n
    return g


def soft_grad(a, tau):
    """Gradient using softmax-weighted peaks (smoother)."""
    n = len(a)
    c = fft_convolve(a, a)
    w = np.exp(tau * (c - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for j in range(n):
        kmax = min(n, 2 * n - 1 - j)
        g[j] = np.dot(a[:kmax], w[j:j + kmax])
    g = g - g.sum() / n
    return g


def project_simplex(a, St):
    """Project onto {a >= 0, sum(a) = St}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), St / len(a))
    return a * (St / s)


def adam_optimize(a0, deadline, use_soft=False, max_iters=10000,
                  tau_start=3.0, tau_end=40.0, base_lr=0.08):
    """Adam optimizer with decaying LR and simplex projection."""
    n = len(a0)
    St = np.sqrt(2.0 * n)
    a = project_simplex(a0.copy(), St)
    best = a.copy()
    best_s = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        if use_soft:
            tau = tau_start + (tau_end - tau_start) * (it / max(it, max_iters))
            g = soft_grad(a, tau)
        else:
            g = hard_grad(a)

        t = it + 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        lr = base_lr / ((it + 10) ** 0.35)
        step = lr * m / (np.sqrt(v) + eps)
        a = a - step
        a = project_simplex(a, St)

        if (it + 1) % 200 == 0:
            s = compute_score(a)
            if s < best_s:
                best_s = s
                best = a.copy()

    s = compute_score(a)
    if s < best_s:
        best_s = s
        best = a.copy()
    return best, best_s


def build_lp_matrix(a, n):
    """Build the (2n-1) x n constraint matrix for convolve(a, g) <= M."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_step(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M elementwise, g >= 0."""
    c_obj = -np.ones(n)
    A = build_lp_matrix(a, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, deadline, max_iters=50):
    """Iterative LP refinement: find g, blend, repeat."""
    n = len(a0)
    if n > 2048:
        return a0, compute_score(a0)
    St = np.sqrt(2.0 * n)
    a = project_simplex(a0.copy(), St)
    best = a.copy()
    best_s = compute_score(a)
    M = float(fft_convolve(a, a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        g = lp_step(a, M, n)
        if g is None:
            break
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_new = project_simplex((1.0 - t) * a + t * g, St)
            s = compute_score(a_new)
            if s < best_s - 1e-10:
                a = a_new
                best = a_new.copy()
                best_s = s
                improved = True
                break
        if not improved:
            break
        M = float(fft_convolve(a, a).max())

    return best, best_s
