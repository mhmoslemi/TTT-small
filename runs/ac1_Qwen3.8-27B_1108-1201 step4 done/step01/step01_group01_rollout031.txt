def _fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Smoothed gradient descent with log-sum-exp approximation, FFT-based convolution,
momentum, and multi-start strategy to minimize normalized autocorrelation peak.
"""

import numpy as np
import time


def _fft_convolve(a, b=None):
    """Fast linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = _fft_convolve(a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_smoothed_grad(a, weights):
    """
    Compute gradient of sum_k weights[k] * convolve(a,a)[k] w.r.t. a.
    Uses FFT. weights has length 2n-1.
    """
    n = len(a)
    # grad[j] = 2 * sum_i a[i] * weights[i+j]
    # = 2 * (a * w_rev)[2n-2-j] where w_rev[i] = weights[2n-2-i]
    w_rev = weights[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    C = np.fft.irfft(A * W, size)
    corr = C[n - 1:2 * n - 1][::-1]
    return 2.0 * corr[:n]


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_anti(n, eps, freq=1):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * freq * i / (n - 1))
    a = np.maximum(a, 0.01)
    S = np.sum(a)
    return a / S


def _make_four_level(n, eps):
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_triangular(n, peak=0.5):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + peak * (1.0 - 2.0 * np.abs(i / (n - 1) - 0.5))
    a = np.maximum(a, 0.01)
    S = np.sum(a)
    return a / S


def _make_sine(n, freq=3, amp=0.3):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.01)
    S = np.sum(a)
    return a / S

                           eta_init=None, momentum=0.9):
    """
    Smoothed gradient descent with momentum and log-sum-exp annealing.
    """

    if eta_init is None:
        eta_init = 0.1 / np.sqrt(n)

    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    vel = np.zeros(n)
    best = a.copy()
    best_score = _compute_score(a)
    eta = eta_init
    no_improve = 0
    iter_count = 0
    min_eta = 1e-12

    while time.monotonic() < deadline:
        b = _fft_convolve(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            eta *= 0.7
            if eta < min_eta:
                break
            no_improve = 0

        # Anneal temperature

        t = t_init * (t_max / t_init) ** progress

        # Softmax weights
        exp_b = np.exp(t * (b - b_max))
        weights = exp_b / np.sum(exp_b)

        # Gradient
        grad = _compute_smoothed_grad(a, weights)
        grad -= np.mean(grad)  # center for sum constraint

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        vel = momentum * vel - eta * grad / np.sqrt(n)
        a = a + vel

        # Project onto simplex
        a = _project_simplex(a, 1.0)
        iter_count += 1

    return best, best_score

"""
Multi-start smoothed gradient descent with log-sum-exp annealing, FFT-based
convolution, momentum, and adaptive temperature to minimize the normalized
autocorrelation peak 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Includes diverse initializations, periodic reheating, and final coordinate search.

import numpy as np
import time


def _fft_convolve(a, b=None):
    """Fast linear convolution using FFT (rfft for real inputs)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)≈1."""
    n = len(a)
    b = _fft_convolve(a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target} via sorting (O(n log n))."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_smoothed_grad(a, p):
    """
    Compute grad[j] = 2 * sum_i a[i] * p[i+j]  using FFT cross-correlation.
    p has length 2n-1 (softmax weights over autocorrelation lags).
    """
    n = len(a)
    w_rev = p[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    C = np.fft.irfft(A * W, size)
    corr = C[n - 1:2 * n - 1][::-1]
    return 2.0 * corr[:n]


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_anti(n, eps, freq=1):
    i = np.arange(n, dtype=np.float64)

def _make_four_level(n, eps):

            best = a.copy()
            no_improve = 0

            eta *= 0.6
            if eta < min_eta:
                break

        a = a + vel

    start_time = time.monotonic()
    deadline = start_time + total_budget

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    print(f"n={n}, budget={budget_s}s")

    global_best = None
    global_best_score = np.inf

    # Build diverse starting points
    starts = []

    # 0: Parent (weighted most heavily)
    a0 = parent.copy()
    S0 = np.sum(a0)
    if S0 > 1e-10:
        a0 /= S0
    starts.append((a0, "parent"))

    # Two-level anti-symmetric
    for eps in [0.1, 0.3, 0.5]:
        starts.append((_make_two_level(n, eps), f"2lev_{eps}"))

    # Smooth anti-symmetric
    starts.append((_make_smooth_anti(n, 0.3), "smooth_0.3"))
    starts.append((_make_smooth_anti(n, 0.5), "smooth_0.5"))

    # Four-level
    starts.append((_make_four_level(n, 0.3), "4lev"))

    # Triangular
    starts.append((_make_triangular(n, 0.3), "tri"))

    # Sine waves
    starts.append((_make_sine(n, 3, 0.2), "sine3"))
    starts.append((_make_sine(n, 9, 0.15), "sine9"))

    # Multi-sine random
    starts.append((_make_multi_sine(n, rng), "msine"))

    # Pure random
    a_r = rng.random(n)
    a_r /= np.sum(a_r)
    starts.append((a_r, "rand"))

    n_starts = len(starts)
    print(f"{n_starts} starting points")

    # Phase 1: Multi-start optimization (60% of time)
    phase1_frac = 0.55
    phase1_deadline = start_time + phase1_frac * total_budget

    # Give parent 3x time, rest equal
    parent_time = 0.3 * (phase1_deadline - start_time)
    rest_time = 0.7 * (phase1_deadline - start_time)
    time_per_other = rest_time / (n_starts - 1)

    for i, (a_start, name) in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        if i == 0:
            t_alloc = parent_time
        else:
            t_alloc = time_per_other

        sub_deadline = min(time.monotonic() + t_alloc, phase1_deadline)

        # Varied hyperparameters
        t_init = 3.0 + 1.0 * i
        t_max = 80.0 + 30.0 * i
        eta = 0.08 / np.sqrt(n)
        mom = 0.9

        best_a, best_s = _smoothed_grad_descend(
            a_start, sub_deadline, rng,
            t_init=t_init, t_max=t_max,
            eta_init=eta, momentum=mom
        )

        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            elapsed = time.monotonic() - start_time
            print(f"  [{elapsed:5.1f}s] {name}: BEST {best_s:.8f}")
        else:
            print(f"  [{time.monotonic()-start_time:5.1f}s] {name}: {best_s:.8f}")

    # Phase 2: Polish best with high-temperature (near-exact) descent
    remaining = deadline - time.monotonic()
    print(f"\nPhase 2: polish, {remaining:.1f}s left, score={global_best_score:.8f}")

    if remaining > 20 and global_best is not None:
        polish_deadline = min(time.monotonic() + 0.4 * remaining, deadline)
        best_a, best_s = _smoothed_grad_descend(
            global_best, polish_deadline, rng,
            t_init=200.0, t_max=2000.0,
            eta_init=0.015 / np.sqrt(n), momentum=0.85
        )
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  Polish improved: {global_best_score:.8f}")

    # Phase 3: Diversification - perturb and re-optimize
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_best is not None:
        print(f"\nPhase 3: diversify, {remaining:.1f}s left")
        n_div = max(1, int(remaining / 100))
        per_div = remaining / n_div

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break
            a_pert = _random_perturb(global_best, rng, scale=0.06)
            div_dl = min(time.monotonic() + per_div, deadline)
            best_a_d, best_s_d = _smoothed_grad_descend(
                a_pert, div_dl, rng,
                t_init=50.0, t_max=500.0,
                eta_init=0.04 / np.sqrt(n), momentum=0.9
            )
            if best_s_d < global_best_score:
                global_best = best_a_d.copy()
                global_best_score = best_s_d
                print(f"  Div {d}: improved to {global_best_score:.8f}")

    # Phase 4: Coordinate search
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_best is not None:
        print(f"\nPhase 4: coord search, {remaining:.1f}s left")
        best_a_c, best_s_c = _coordinate_search(global_best, deadline, rng)
        if best_s_c < global_best_score:
            global_best = best_a_c.copy()
            global_best_score = best_s_c
            print(f"  Coord improved: {global_best_score:.8f}")

    # Final
    if global_best is None:
        global_best = a0

    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.8f}")
    print(f"Length: {len(result)}, Sum: {sum(result):.6f}")
    print(f"{'='*50}")
    return result
