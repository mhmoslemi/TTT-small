def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 using FFT for speed."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    # Use FFT for convolution
    N = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    F = np.fft.rfft(f, N_fft)
    conv = np.fft.irfft(F * F, N_fft)[:N]
    return float(2.0 * n * conv.max() / (s * s))

def block_perturb(f, rng, width_range=(2, 50)):
    """Random block multiplicative perturbation."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(width_range[0], min(width_range[1], n)+1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.3)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

def element_perturb(f, rng, num_elements=5):
    """Perturb a few individual elements."""
    n = len(f)
    cand = f.copy()
    indices = rng.integers(0, n, size=min(num_elements, n))
    for i in indices:
        cand[i] *= float(np.exp(rng.normal(0.0, 0.5)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

def make_initial(n, rng, method='uniform'):
    """Create initial sequence."""
    if method == 'uniform':
        a = np.ones(n) + rng.normal(0, 0.3, n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x) + rng.normal(0, 0.1, n)
    elif method == 'sin_sq':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)**2 + rng.normal(0, 0.05, n)
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'peak':
        x = np.arange(n, dtype=float)
        c = n // 2
        s = max(1, n // 8)
        a = np.exp(-0.5 * ((x - c) / s)**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def anneal_search(f, budget_s, rng, method_mix='hybrid'):
    """Simulated annealing with mixed move operators."""
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best = f.copy()
    best_sc = fast_score(f)
    T = 0.5  # initial temperature
    T_min = 0.001
    decay = 0.999
    
    start = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - start < budget_s:
        iter_count += 1
        T = max(T_min, T * decay)
        
        # Choose move operator
        r = rng.random()
        if r < 0.3:
            # Spectral step
            f_new = spectral_step(f, num_iters=3)
        elif r < 0.5:
            # Subgradient step
            f_new = subgradient_step(f, num_top=5, step_size=T)
        elif r < 0.7:
            # Block perturbation
            f_new = block_perturb(f, rng, width_range=(2, max(5, n//10)))
        elif r < 0.85:
            # Element perturbation
            f_new = element_perturb(f, rng, num_elements=max(2, n//50))
        else:
            # Larger block perturbation
            f_new = block_perturb(f, rng, width_range=(n//5, n//2))
        
        sc_new = fast_score(f_new)
        
        # Accept/reject
        if sc_new <= best_sc or rng.random() < np.exp(-(sc_new - best_sc) / max(T, 1e-10)):
            f = f_new
            if sc_new < best_sc - 1e-10:
                best_sc = sc_new
                best = f.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 200:
            # Restart with perturbation
            f = block_perturb(best, rng, width_range=(n//10, n//3))
            stall = 0
            T = min(0.3, T * 5)
    
    return best, best_sc
