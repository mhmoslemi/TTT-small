For a sequence $a$ of length $n$ with $\sum a = 1$, the convolution $c = \text{conv}(a,a)$ has:
- $c[0] = a[0]^2$
- $c[n-1] = \sum_i a[i] a[n-1-i]$ (autocorrelation at lag 0, i.e., inner product with reversed)
- $c[2n-2] = a[n-1]^2$
- $\sum_k c[k] = 1$ (since $\sum a = 1$)

The peak is at least $1/(2n-1)$ (average value). To minimize the peak, we want all $c[k]$ to be as close to $1/(2n-1)$ as possible.

For the constant sequence $a_i = 1/n$:
$c[k] = (n-|k-(n-1)|+1)/n^2$ for $|k-(n-1)| \le n-1$, i.e., a triangle.
$c[n-1] = n/n^2 = 1/n$
Peak = $1/n$
Score = $2n \cdot (1/n) / 1 = 2$

The average is $1/(2n-1) \approx 1/(2n)$. The peak is about $2/(2n) = 1/n$, which is twice the average. So the "peak-to-average ratio" is about 2 for the constant sequence.

        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        value, g_theta, a, c = grad_wrt_theta(theta, S, tau)
        actual_score = 2.0 * n * np.max(c) / (S * S)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_theta = theta.copy()
            best_a = a.copy()

        if verbose and step % 5000 == 0:
            print(f"    step={step}, score={actual_score:.6f}, best={best_score:.6f}, tau={tau:.2e}")

    return best_theta, best_score, best_a


def optimize_lbfgs(a0, time_limit=30):
    """L-BFGS-B refinement using soft-max objective with log parameterization."""
    n = len(a0)
    S = np.sum(a0)

    def obj_and_grad(x):
        a = np.exp(x)
        a = a / np.sum(a) * S
        tau = 0.05 * S / n
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_c = np.exp(shifted)
        sum_exp = np.sum(exp_c)
        w = exp_c / sum_exp
        val = 2.0 * n * (c_max + tau * np.log(sum_exp)) / (S * S)
        # Gradient w.r.t. a
        size = 1
        while size < 4 * n:
            size <<= 1
        W = np.fft.rfft(w, size)
        A = np.fft.rfft(a, size)
        cross = np.fft.irfft(np.conj(W) * A, size)
        g_a = 2.0 * cross[:n]
        # Chain rule through softmax
        inner = np.dot(g_a, a) / S
        g_x = a * (g_a - inner)
        return val, g_x

    x0 = np.log(np.clip(a0, 1e-10, None))
    t0 = time.monotonic()
    try:
        result = _minimize_lbfgs(obj_and_grad, x0, max_iter=200)
        a_best = np.exp(result['x'])
        a_best = a_best / np.sum(a_best) * S
    except Exception:
        a_best = a0.copy()

    return a_best


def _minimize_lbfgs(fun, x0, max_iter=200):
    """Simple L-BFGS implementation (2-point BFGS) to avoid scipy dependency issues."""
    from scipy.optimize import minimize
    return minimize(fun, x0, method='L-BFGS-B', options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})


def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate descent: for each coordinate i, find optimal a[i]
    given all others fixed, by 1D minimization.
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    S = np.sum(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    # Precompute index mapping for efficiency
    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            # Contribution of a[i] to convolution:
            # c[k] = c_rest[k] + 2*a[i]*a[k-i] (k != 2i) or a[i]^2 (k=2i)
            # We'll just try a few values of a[i] and pick the best
            x_old = current[i]
            if x_old < 1e-12:
                candidates = [0.0, S / n * 0.1, S / n * 0.5, S / n, S / n * 2.0, S / n * 4.0]
            else:
                candidates = [0.0, x_old * 0.25, x_old * 0.5, x_old * 0.75,
                              x_old, x_old * 1.25, x_old * 1.5, x_old * 2.0]

            best_x = x_old
            best_sc = current_score

            for xt in candidates:
                current[i] = xt
                s = np.sum(current)
                if s < 0.01:
                    continue
                c = fft_convolve(current, current)
                sc = 2.0 * n * np.max(c) / (s * s)
                if sc < best_sc:
                    best_sc = sc
                    best_x = xt

            current[i] = best_x
            if best_x != x_old:
                improved = True
                s = np.sum(current)
                current_score = 2.0 * n * np.max(fft_convolve(current, current)) / (s * s)
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
        if not improved:
            break

    return best, best_score

        candidate = current.copy()

            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Contributors to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=top_k))
            other = np.setdiff1d(np.arange(n), top_idx)
