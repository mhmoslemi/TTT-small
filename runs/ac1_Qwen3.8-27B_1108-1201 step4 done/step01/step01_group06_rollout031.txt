```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_smooth_gradient(a, c, T):
    """Compute gradient of T*log(sum exp(c/T)) w.r.t. a via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    
    # Softmax of c/T
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    g = exp_c / exp_c.sum()
    
    # Gradient: grad[m] = 2 * sum_j a[j] * g[m+j]
    # = 2 * real(IFFT(conj(FFT(a_padded)) * FFT(g_padded)))[:n]
    a_padded = np.zeros(L, dtype=np.float64)
    a_padded[:n] = a
    g_padded = np.zeros(L, dtype=np.float64)
    g_padded[:size] = g
    
    A = fft(a_padded)
    G = fft(g_padded)
    corr = np.real(ifft(np.conj(A) * G))
    
    return 2.0 * corr[:n], g

def make_freq_init(n, rng):
    """Initialize with flat power spectrum (random phases)."""
    L = 1 << (2*n - 2).bit_length()
    phases = rng.uniform(0, 2*np.pi, L)
    # For real sequence, need conjugate symmetry
    A = np.zeros(L, dtype=np.complex128)
    for k in range(L):
        A[k] = np.exp(1j * phases[k])
    # Enforce conjugate symmetry for real output
    A = (A + np.conj(A[::-1])) / 2  # This isn't quite right for DFT symmetry
    
    # Better: generate random real sequence with flat spectrum
    # Use the fact that if we take a random complex sequence with |A[k]| = 1
    # and enforce A[k] = conj(A[L-k]), the result is real
    A = np.ones(L, dtype=np.complex128)
    for k in range(1, L // 2 + 1):
        A[k] = np.exp(1j * rng.uniform(0, 2*np.pi))
        A[L-k] = np.conj(A[k])
    A[0] = 1.0
    if L % 2 == 0:
        A[L//2] = np.exp(1j * rng.uniform(0, 2*np.pi))  # This will make it non-real
        # Actually for real sequence, A[L//2] must be real
        A[L//2] = rng.choice([-1.0, 1.0])
    
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
```

```python
Uses smooth max (log-sum-exp) approximation of the peak of the self-convolution,

and a final polishing phase maximize the chance of finding a near-optimal solution.

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_gradient(a, c, T):
    """
    Compute the gradient of f_T(a) = T*log(sum_k exp(c[k]/T)) w.r.t. a,
    where c = conv(a,a).
    
    Returns (gradient, softmax_weights).
    """
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    
    # Softmax
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    g = exp_c / exp_c.sum()
    
    # Gradient via FFT cross-correlation
    # grad[m] = 2 * sum_j a[j] * g[m+j] for m = 0..n-1
    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a
    g_pad = np.zeros(L, dtype=np.float64)
    g_pad[:size] = g
    
    A_f = fft(a_pad)
    G_f = fft(g_pad)
    corr = np.real(ifft(np.conj(A_f) * G_f))
    
    return 2.0 * corr[:n], g
```

```python
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)

    Gradient of f_T(a) = T*log(sum exp(c/T)) w.r.t. a, where c = conv(a,a).
    Uses FFT for O(n log n) computation.
    Returns (gradient, softmax_weights).
    """
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    if total < 1e-300:

        g[np.argmax(c)] = 1.0
```
