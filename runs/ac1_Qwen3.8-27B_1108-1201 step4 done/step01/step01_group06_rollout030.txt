```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient(c, a, beta):
    """Compute gradient of LSE(beta*c)/beta w.r.t. a using FFT."""
    n = len(a)
    # Softmax weights
    bc = beta * c
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    
    # Gradient via cross-correlation
    # grad[m] = (2/beta) * sum_{j=0}^{n-1} a[j] * w[m+j]
    N = 1 << (3 * n - 1).bit_length()
    a_rev = a[::-1]
    conv = np.real(ifft(fft(a_rev, N) * fft(w, N)))
    grad = (2.0 / beta) * conv[n-1 : 2*n-1]
    return grad, w

def run_lse_adam(a_init, n, S, max_time, rng, beta_start, beta_end, lr_init):
    """Adam with LSE-smoothed objective and beta annealing."""
    a = np.clip(a_init.astype(np.float64), 0, 1000)
    a = a * (S / a.sum()) if a.sum() > 0 else np.full(n, S/n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    
    total_steps = 10000  # estimate for annealing schedule
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Anneal beta
        frac = min(1.0, t / total_steps)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, w = lse_gradient(c, a, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr = lr_init / (1 + 0.01 * t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Occasional random perturbation
        if t % 500 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.05 * S / n, n), S)
    
    return best_a, best_score
```

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses LSE-smoothed Adam with beta annealing, frequency-domain jolts,
and multiple diverse initializations to escape local minima.
Key improvement over previous: exact LSE gradient (all lags) via FFT cross-correlation,
proper annealing schedule, and periodic spectral peak-compression restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """
    Compute gradient of LSE(beta*c)/beta w.r.t. a using FFT cross-correlation.
    Returns (grad, softmax_weights).
    """
    n = len(a)
    bc = beta * c
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()

    # grad[m] = (2/beta) * sum_{j=0}^{n-1} a[j] * w[m+j]
    # Use FFT: convolve a_rev with w, take appropriate slice
    N = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    conv = np.real(ifft(fft(a_rev, N) * fft(w, N)))
    grad = (2.0 / beta) * conv[n - 1:2 * n - 1]
    return grad, w


def lse_value(c, beta):
    """Compute LSE(beta*c)/beta."""
    bc = beta * c
    return bc.max() + np.log(np.sum(np.exp(bc - bc.max()))) / beta


def freq_jolt(a, rng, strength=0.1):
    """
    Frequency-domain peak compression: reduce large magnitude components
    and project back to non-negative simplex. Inspired by PAPR reduction.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    # Compress large magnitudes
    threshold = np.percentile(mag, 90)
    mag_new = np.where(mag > threshold, threshold + (mag - threshold) * (1 - strength), mag)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    else:
        a_new = project_simplex(a, S)
    return a_new


def run_lse_adam(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=50.0,
                 lr_init=None, jolt_interval=2000, jolt_prob=0.5):
    """
    Adam with LSE-smoothed objective and exponential beta annealing.
    Periodic frequency-domain jolts to escape local minima.
    """
    if lr_init is None:
        lr_init = 0.3 * S / n

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    log_interval = 5000

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if t % log_interval < 100:
                elapsed = time.monotonic() - start
                print("  [adam] t=%d max=%.6f beta=%.1f elapsed=%.0fs" % (t, best_score,
                      beta_start * (beta_end / beta_start) ** min(1.0, t / 20000.0), elapsed))

        # Anneal beta exponentially
        frac = min(1.0, t / 30000.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        grad, w = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad ** 2)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)

        # Learning rate with slow decay
        lr = lr_init / (1.0 + 0.005 * t) ** 0.5
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        # Periodic jolt
        if jolt_interval > 0 and t % jolt_interval == 0 and rng.random() < jolt_prob:
            a_jolted = freq_jolt(a, rng, strength=0.15)
            # Only accept if it doesn't worsen too much
            c_j = fft_convolve_self(a_jolted)
            if c_j.max() < best_score * 1.05:
                a = a_jolted
                m_buf[:] = 0
                v_buf[:] = 0

    return best_a, best_score


def run_mirror_lse(a_init, n, S, max_time, rng, eta_init=0.5):
    """Mirror descent (multiplicative weights) with LSE objective."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    beta = 5.0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        beta_cur = min(20.0, beta * (1.0 + 0.001 * t))
        grad, w = lse_gradient(a, c, beta_cur)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad = grad / gmax

        # Multiplicative update in log-space
        log_a = np.log(a + 1e-30)
        log_a = log_a - eta_init / (1.0 + 0.01 * t) * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)
        else:
            a = np.full(n, S / n)

    return best_a, best_score


def get_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []

    # Try the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)

            idx = np.clip(idx, 0, given.size - 1)

    except Exception:
        pass

    # Random
    inits.append(("rand1", rng.random(n)))
    inits.append(("rand2", rng.random(n)))

    # Structured: raised cosine taper on random
    tt = np.linspace(0, 1, n)
    taper = 0.5 + 0.5 * np.cos(np.pi * (2 * tt - 1))
    taper = taper ** 0.5
    inits.append(("tapered_rand", rng.random(n) * taper + 0.1))

    # Structured: smooth periodic
    inits.append(("periodic", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                  0.2 * np.cos(2 * np.pi * 7 * tt) + 0.1 * np.sin(2 * np.pi * 13 * tt)))

    # Block-structured
    block_len = 128
    n_blocks = n // block_len
    block_vals = 0.5 + 0.5 * rng.random(n_blocks)
    block_seq = np.tile(block_vals, block_len)[:n]
    block_seq = block_seq * taper + 0.1
    inits.append(("block", block_seq))

    # Near-uniform with small perturbation
    inits.append(("near_uniform", np.ones(n) + 0.1 * rng.random(n)))

    return inits


def polish_sequence(a, budget, rng):
    """Final polishing with high beta and small steps."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, 1000)
    a = a * (S / a.sum()) if a.sum() > 0 else np.full(n, S / n)

    # Use very high beta (essentially hard max) with small Adam steps
    best_a, best_score = run_lse_adam(
        a, n, S, budget, rng,
        beta_start=20.0, beta_end=100.0,
        lr_init=0.05 * S / n,
        jolt_interval=3000, jolt_prob=0.3
    )

    # Mirror descent polish
    remaining = budget - (time.monotonic() - (time.monotonic() - budget))
    # Just use half the budget for each
    half = budget / 2.0
    a2, s2 = run_mirror_lse(a, n, S, max(0.1, half - 1.0), rng, eta_init=0.2)
    if s2 < best_score:
        best_a, best_score = a2, s2

    return best_a, best_score
