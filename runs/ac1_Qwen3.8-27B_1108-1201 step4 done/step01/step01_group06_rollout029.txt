```python
"""
Minimize 2*n*max(self-convolution) / (sum)^2 using:
1. L-BFGS with annealed log-sum-exp smoothing of the max-convolution
2. LP-based Frank-Wolfe direction finding for polishing
3. Multiple diverse initializations and lengths

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.signal import fftconvolve

def self_convolve(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def true_objective(a):
    """Compute 2*n*max_b / sum_a^2."""
    n = len(a)
    b = self_convolve(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def smoothed_obj_and_grad(z, alpha):
    """Compute smoothed objective and gradient w.r.t. z (log-softmax parameterization)."""
    # Compute a = softmax(z)
    z_shifted = z - z.max()  # numerical stability
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute b = a * a
    b = self_convolve(a)
    
    # Softmax of b with temperature alpha
    max_b = b.max()
    log_sum_exp = max_b + np.log(np.sum(np.exp(alpha * (b - max_b))))
    obj = log_sum_exp / alpha  # This is the smoothed max
    
    # Compute gradient w.r.t. a: grad_a[m] = 2 * sum_l a[l] * p[m+l]
    p = np.exp(alpha * (b - max_b)) / np.sum(np.exp(alpha * (b - max_b)))
    grad_a = 2.0 * fftconvolve(a[::-1], p, mode='valid')
    
    # Convert to gradient w.r.t. z via softmax Jacobian
    # d(a_i)/d(z_j) = a_i * (delta_ij - a_j)
    # d(obj)/d(z_j) = sum_i grad_a[i] * d(a_i)/d(z_j)
    #                = a_j * (grad_a[j] - sum_i grad_a[i]*a_i)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

def run_lbfgs(z_init, alpha_schedule, max_time, verbose=True):
    """Run L-BFGS-B with annealed smoothing."""
    n = len(z_init)
    best_a = None
    best_true_obj = np.inf
    
    for alpha in alpha_schedule:
        start_t = time.monotonic()
        
        def func(z):
            obj, grad = smoothed_obj_and_grad(z, alpha)
            return obj, grad
        
        options = {
            'maxiter': 200,
            'maxfun': 1000,
            'ftol': 1e-12,
            'gtol': 1e-8,
            'max_ls': 50,
        }
        
        result = minimize(func, z_init, method='L-BFGS-B', jac=True, options=options)
        
        # Extract best a from this run
        a = np.exp(result.x - result.x.max())
        a = a / a.sum()
        obj = true_objective(a)
        
        if obj < best_true_obj:
            best_true_obj = obj
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - start_t
            print(f"  alpha={alpha:.1f}: smoothed_obj={result.fun:.6f}, true_obj={obj:.6f}, "
                  f"iters={result.nit}, time={elapsed:.1f}s")
        
        z_init = result.x - result.x.mean()  # re-center
        if time.monotonic() + 5 > max_time:
            break
    
    return best_a, best_true_obj
```

```python
"""
Minimize 2*n*max(self-convolution)/sum^2 using:
1. L-BFGS with annealed log-sum-exp smoothing (multiple initializations, multiple lengths)
2. Coordinate ascent polishing (exploiting the structure of max-of-linear-functions)
3. Final refinement with high-smoothing L-BFGS
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.signal import fftconvolve


def self_convolve(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def true_objective_normalized(a):
    """Compute max_b when a is normalized to sum=1. (Objective = 2*n*max_b.)"""
    b = self_convolve(a)
    return b.max()


def true_objective(a):
    """Compute 2*n*max_b / sum_a^2 for arbitrary a."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = self_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    exp_z = np.exp(z)
    return exp_z / exp_z.sum()


def smoothed_obj_grad(z, alpha):
    """
    Smoothed objective (LSE of self-convolution) and gradient w.r.t. z.
    a = softmax(z), obj = (1/alpha)*log(sum(exp(alpha*b))), b = a*a
    """
    a = softmax(z)
    b = self_convolve(a)
    
    # Log-sum-exp
    max_b = b.max()
    log_sum = max_b + np.log(np.sum(np.exp(alpha * (b - max_b))))
    obj = log_sum / alpha
    
    # Softmax weights
    p = np.exp(alpha * (b - max_b))
    p_sum = p.sum()
    p = p / p_sum
    
    # grad_a[m] = 2 * sum_l a[l] * p[m+l] = 2 * cross_correlate(p, a)[m]
    # = 2 * fftconvolve(a[::-1], p, mode='valid')
    grad_a = 2.0 * fftconvolve(a[::-1], p, mode='valid')
    if len(grad_a) != len(a):
        grad_a = grad_a[:len(a)]
    
    # grad_z = a * (grad_a - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z


def run_lbfgs_anneal(z_init, alphas, maxiter_per_alpha=300, time_budget=None):
    """Run L-BFGS-B with annealed smoothing parameter."""
    n = len(z_init)
    best_a = softmax(z_init).copy()
    best_obj = true_objective(best_a)
    start = time.monotonic()
    
    for alpha in alphas:
        if time_budget is not None and time.monotonic() - start > time_budget:
            break
        
        alpha_val = float(alpha)
        
        def fun(z):
            return smoothed_obj_grad(z, alpha_val)
        
        try:
            res = minimize(fun, z_init, method='L-BFGS-B', jac=True,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxfun': maxiter_per_alpha * 5})
            z_init = res.x - res.x.mean()
        except Exception as e:
            print(f"  L-BFGS failed at alpha={alpha}: {e}")
            break
        
        a = softmax(z_init)
        obj = true_objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        elapsed = time.monotonic() - start
        if time_budget is not None:
            remaining = time_budget - elapsed
        else:
            remaining = float('inf')
        print(f"  alpha={alpha_val:>6.1f}  obj={best_obj:.8f}  "
              f"last={obj:.8f}  iters={res.nit:>4d}  time={elapsed:6.1f}s  "
              f"remain={min(remaining, 9999):.1f}s")
    
    return best_a, best_obj


def coordinate_ascent(a, n_cycles=10, time_budget=None):
    """
    Coordinate ascent: cycle through coordinates, optimizing each one.
    For coordinate m, max_b is a convex piecewise-linear function of a[m].
    We find the minimizer by checking intersection points of top lines.
    """
    n = len(a)
    b = self_convolve(a)
    best_obj = b.max()
    start = time.monotonic()
    
    for cycle in range(n_cycles):
        if time_budget is not None and time.monotonic() - start > time_budget:
            break
        improved = False
        
        for m in range(n):
            if time_budget is not None and (m % 512 == 0) and (time.monotonic() - start > time_budget):
                break
            
            old_val = a[m]
            if old_val < 1e-15:
                continue
            
            # Affected lags: k in [max(0,m), min(2n-2, m+n-1)]
            k_lo = m  # since m >= 0
            k_hi = min(2 * n - 2, m + n - 1)
            
            # b[k](v) = b[k] + 2*(v - old_val) * a[k-m] for k in [k_lo, k_hi]
            # = (b[k] - 2*old_val*a[k-m]) + 2*v*a[k-m]
            intercepts = b[k_lo:k_hi+1] - 2.0 * old_val * a[0:k_hi - k_lo + 1]
            slopes = 2.0 * a[0:k_hi - k_lo + 1]
            
            # Unaffected max
            # b has length 2n-1. We remove indices k_lo..k_hi.
            if k_lo == 0 and k_hi == 2 * n - 2:
                B_unaff = 0.0
            elif k_lo == 0:
                B_unaff = b[:k_hi+1].max()  # wait, this includes the affected range
                # Need to exclude k_lo..k_hi
                if k_hi + 1 < 2 * n - 1:
                    B_unaff = b[k_hi+1:].max()
                else:
                    B_unaff = 0.0
            elif k_hi == 2 * n - 2:
                B_unaff = b[:k_lo].max() if k_lo > 0 else 0.0
            else:
                B_unaff = max(b[:k_lo].max(), b[k_hi+1:].max())
            
            # Find top-3 lags among affected at current value
            current_vals = b[k_lo:k_hi+1]
            n_affected = len(current_vals)
            top_k = min(3, n_affected)
            top_idx = np.argsort(current_vals)[::-1][:top_k]
            
            # Candidate v values
            candidates = [0.0, old_val]
            for i in range(top_k):
                for j in range(i + 1, top_k):
                    si, sj = top_idx[i], top_idx[j]
                    m1, m2 = slopes[si], slopes[sj]
                    c1, c2 = intercepts[si], intercepts[sj]
                    if abs(m1 - m2) > 1e-20:
                        v_c = (c2 - c1) / (m1 - m2)
                        if 0 <= v_c <= 1000:
                            candidates.append(v_c)
                # Intersection with B_unaff
                m1 = slopes[top_idx[i]]
                c1 = intercepts[top_idx[i]]
                if abs(m1) > 1e-20 and B_unaff > 0:
                    v_c = (B_unaff - c1) / m1
                    if 0 <= v_c <= 1000:
                        candidates.append(v_c)
            
            # Evaluate
            best_v = old_val
            best_max = best_obj  # current max (before change)
            for v in candidates:
                if v < -1e-12 or v > 1000:
                    continue
                v = max(0.0, v)
                new_vals = intercepts + slopes * v
                new_max = max(new_vals.max(), B_unaff)
                if new_max < best_max:
                    best_max = new_max
                    best_v = v
            
            if best_v != old_val and best_max < best_obj - 1e-12:
                a[m] = best_v
                # Update b (expensive, but we need it for next iterations)
                # Actually, let's update b incrementally
                delta = 2.0 * (best_v - old_val) * a[0:k_hi - k_lo + 1]
                b[k_lo:k_hi+1] += delta
                best_obj = max(b.max(), B_unaff)  # b already updated
                improved = True
        
        # Renormalize
        s = a.sum()
        if s > 0:
            a = a / s
        
        if not improved:
            break
    
    return a, true_objective(a)
```
