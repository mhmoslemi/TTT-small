"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses: (1) top-K subgradient descent with line search and momentum,
(2) LP-based peak reduction, (3) multiple starting points and lengths.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def project_fixed_sum_nonneg(b, S):
    """Project b onto {x >= 0, sum(x) = S}."""
    # Use bisection on the threshold
    # We want sum(max(b_i - theta, 0)) = S
    # max(theta) = max(b) - S/(number of positive entries at most)
    # min(theta) = min(b) (or lower)
    
    b = np.asarray(b, dtype=np.float64)
    n = len(b)
    
    # If sum(b) is already close to S and b >= 0, we're done
    if b.min() >= 0 and abs(b.sum() - S) < 1e-10:
        return b
    
    # Bisection
    lo = b.min() - 1.0
    hi = b.max() + 1.0
    
    def f(theta):
        return np.maximum(b - theta, 0.0).sum() - S
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    theta = (lo + hi) / 2.0
    return np.maximum(b - theta, 0.0)

def lp_refine(a0, deadline, max_iters=50):
    """
    LP-based refinement: minimize max(conv(g, a)) over g >= 0, sum(g) = sum(a).
    Then blend.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S = float(a.sum())
    if S < 1e-10:
        return a0, compute_obj(a0)
    
    a = a / S  # normalize
    S_work = 1.0
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(compute_autocorr(a).max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Build LP: minimize t s.t. conv(g, a) <= t, sum(g) = 1, g >= 0
        # Variables: [g_0, ..., g_{n-1}, t]
        # Objective: minimize t -> cost = [0,...,0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Constraints: for each m in 0..2n-2, sum_j g[j] * a[m-j] - t <= 0
        # A_ub has (2n-1) rows, (n+1) columns
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, n] = -1.0  # -t column
        b_ub = np.zeros(2 * n - 1)
        
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A_ub[m, j] = a[m - j]
        
        # Equality constraint: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0.0, None)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x[:n]
        except Exception:
            break
        
        # Blend
        improved = False
        for ta in [0.5, 0.3, 0.2, 0.1, 0.7]:
            a_new = (1.0 - ta) * a + ta * g
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(compute_autocorr(a_new).max())
            if obj_new < best_obj:
                a = a_new
                best_a = a.copy()
                best_obj = obj_new
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_obj

        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def compute_autocorr(a):
    return np.convolve(a, a)

def project_fixed_sum_nonneg(b, S_target):
    b = np.asarray(b, dtype=np.float64)
    n = len(b)
    lo = float(b.min()) - 1.0
    hi = float(b.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.maximum(b - mid, 0.0).sum())
        if val > S_target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    return np.maximum(b - theta, 0.0)

def topk_grad(a, c, K, tau):
    n = len(a)
    # Get top K positions in c
    # c has length 2n-1
    # Use partial argmax for efficiency
    if K >= len(c):
        peaks = np.arange(len(c))
        peak_vals = c
    else:
        # Get top K
        part = np.argpartition(c, -K)[-K:]
        peaks = part
        peak_vals = c[part]
    
    # Softmax weights
    w = np.exp(tau * (peak_vals - peak_vals.max()))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        m = int(peaks[idx])
        wt = 2.0 * w[idx]
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        # grad[j] += wt * a[m-j] for j in [lo, hi]
        # a[m-j] for j in [lo, hi] is a[m-hi : m-lo+1] reversed
        # m-hi to m-lo (inclusive)
        seg = a[m - hi : m - lo + 1]
        grad[lo:hi+1] += wt * seg[::-1]
    
    return grad

def gd_optimize(a0, deadline, K_list, max_iters):
    # a0 should have sum = 1
    n = len(a0)
    a = a0.copy()
    S = float(a.sum())
    a = a / S
    
    c = compute_autocorr(a)
    obj = 2.0 * n * float(c.max())
    best_a = a.copy()
    best_obj = obj
    
    v = np.zeros(n)
    beta = 0.85
    lr = 0.1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max_iters
        
        # Adaptive K
        K_idx = min(int(frac * len(K_list)), len(K_list) - 1)
        K = K_list[K_idx]
        
        # Adaptive tau
        tau = 3.0 + 20.0 * frac
        
        c = compute_autocorr(a)
        grad = topk_grad(a, c, K, tau)
        
        # Momentum
        v = beta * v + (1.0 - beta) * grad
        v = v - v.mean()  # fixed sum
        
        # Try multiple step sizes
        best_candidate = a
        best_candidate_obj = obj
        for factor in [1.0, 0.5, 0.25, 0.1, 2.0]:
            step = lr * factor
            candidate = a - step * v
            candidate = project_fixed_sum_nonneg(candidate, 1.0)
            c_new = compute_autocorr(candidate)
            obj_new = 2.0 * n * float(c_new.max())
            if obj_new < best_candidate_obj:
                best_candidate_obj = obj_new
                best_candidate = candidate
        
        a = best_candidate
        obj = best_candidate_obj
        
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            lr = min(lr * 1.02, 1.0)
        else:
            lr = max(lr * 0.97, 0.001)
    
    return best_a, best_obj

def lp_refine(a0, deadline, max_iters):
    n = len(a0)
    a = a0.copy()
    S = float(a.sum())
    a = a / S
    
    c = compute_autocorr(a)
    best_obj = 2.0 * n * float(c.max())
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # LP: minimize t s.t. conv(g, a) <= t, sum(g) = 1, g >= 0
        # c_obj = [0, ..., 0, 1] (n+1 entries, last is t)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        # Fill in the a values
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            # A_ub[m, j] = a[m-j] for j in [lo, hi]
            # a[m-hi : m-lo+1] reversed
            seg = a[m - hi : m - lo + 1]
            A_ub[m, lo:hi+1] = seg[::-1]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0.0, None)] * (n + 1)
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            break
        
        g = np.maximum(res.x[:n], 0.0)
        
        # Try blending
        improved = False
        for ta in [0.3, 0.5, 0.2, 0.7, 0.1]:
            a_new = (1.0 - ta) * a + ta * g
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = compute_autocorr(a_new)
            obj_new = 2.0 * n * float(c_new.max())
            if obj_new < best_obj:
                a = a_new
                best_a = a.copy()
                best_obj = obj_new
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_obj
