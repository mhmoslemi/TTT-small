"""
Hybrid LP + subgradient optimization for minimizing autocorrelation peak
of non-negative step functions.

Objective: minimize 2*n*max(conv(a,a))/(sum(a))^2 over a >= 0.

Method:
- Min-peak LP: At each step, solve min M' s.t. conv(f,g) <= M', sum(g)=1, g>=0
  to find the best direction g, then blend f with g.
- Subgradient descent with momentum for large n where LP is too expensive.
- Coordinate descent for final polish.
- Multiple diverse starting points and sequence lengths.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    """Compute score = 2*n*max(conv)/(sum^2) without calling evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_conv_matrix_n(n, f):
    """Build sparse (2n-1 x n) matrix A where A@g = conv(f, g, 'full')."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        count = j_hi - j_lo + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(j_lo, j_hi + 1)
        vals[idx:idx+count] = f[k - np.arange(j_lo, j_hi + 1)]
        idx += count
    return sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def solve_min_peak_lp(f, time_limit=5.0):
    """
    Solve: min M'
           s.t. (A@g)[k] <= M' for all k
                sum(g) = 1
                g >= 0
    Returns (g, M_opt) or (None, None).
    """
    n = len(f)
    A = build_conv_matrix_n(n, f)  # (2n-1) x n
    
    # Variables: [g_0, ..., g_{n-1}, M']
    # Minimize: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - M' <= 0 for each row
    # [A, -1] * [g; M'] <= 0
    A_eq_rows = 2 * n - 1
    A_ub = sp.hstack([A, sp.csc_matrix(-np.ones((A_eq_rows, 1)))]).tocsr()
    b_ub = np.zeros(A_eq_rows)
    
    # Equality: sum(g) = 1, M' is free
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(g) = 1, M' coefficient is 0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, M' >= 0 (peak is non-negative for non-negative inputs)
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            M_opt = res.x[n]
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs  # re-normalize for numerical safety
            return g, float(M_opt)
    except Exception as e:
        print(f"  LP error: {e}")
    return None, None

def subgradient_step(f, lr, momentum=0.9):
    """One subgradient step with momentum. f should be normalized to sum 1."""
    conv = np.convolve(f, f)
    M = conv.max()
    # Find peaking indices
    peaks = np.where(conv >= M * 0.999)[0]
    
    # Compute subgradient
    n = len(f)
    s = np.zeros(n)
    for k in peaks:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            s[j] += f[k - j] + f[j]
    
    return s, M

def subgradient_optimize(f_init, budget_s, lr_init=0.01, momentum=0.9):
    """Run subgradient descent with momentum until budget exhausted."""
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        return f_init, fast_score(f_init)
    f = f / s  # normalize
    
    best = f.copy()
    best_sc = fast_score(f)
    lr = lr_init
    v = np.zeros_like(f)
    stall = 0
    
    while time.monotonic() < budget_s:
        sg, M = subgradient_step(f, lr)
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        v = momentum * v - lr * sg
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-10:
            f_new = f
            fs = 1.0
        f_new = f_new / fs
        
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f_new.copy()
            stall = 0
        else:
            stall += 1
            lr *= 0.95
        
        f = f_new
    
    return best, best_sc

def build_conv_matrix_n(n, f):
    """Efficient construction using vectorized operations."""
    # For each k in [0, 2n-2], the valid j range is [max(0,k-n+1), min(n-1,k)]
    # A[k,j] = f[k-j]
    k_vals = np.arange(2*n-1)
    j_lo = np.maximum(0, k_vals - n + 1)
    j_hi = np.minimum(n-1, k_vals)
    counts = j_hi - j_lo + 1
    total = int(counts.sum())
    
    rows = np.repeat(k_vals, counts)
    # For each k, j ranges from j_lo[k] to j_hi[k]
    # We need to compute this efficiently
    cols = np.concatenate([np.arange(j_lo[k], j_hi[k]+1) for k in range(2*n-1)]) if n < 500 else \
         np.empty(total, dtype=np.int32)
    
    if n >= 500:
        # Vectorized approach
        idx = 0
        for k in range(2*n-1):
            lo = int(j_lo[k])
            hi = int(j_hi[k])
            cnt = hi - lo + 1
            cols[idx:idx+cnt] = np.arange(lo, hi+1)
            idx += cnt
    
    vals = f[rows - cols]  # f[k-j]
    return sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

The min-peak LP:
min M'
s.t. (A@g)[k] <= M' for k = 0, ..., 2n-2
     sum(g) = 1
     g >= 0
     M' >= 0

Here A is built from f: A[k,j] = f[k-j] (if valid).

The constraint (A@g)[k] <= M' means conv(f, g)[k] <= M' for all k.

If we set g = f (normalized to sum 1), then conv(f, f) = conv, and M' >= max(conv). So the optimal M' is at most max(conv(f,f)).

If the optimal M' < max(conv(f,f)), then g is a "better" sequence than f in the sense that conv(f,g) has a lower peak than conv(f,f).

After finding g, we update f_new = (1-t)*f + t*g. The new peak of conv(f_new, f_new) should be lower if M' < max(conv).

This is the CCP (convex-concave procedure) idea.

Now, one subtlety: the LP finds g that minimizes max conv(f, g). But we want to minimize max conv(f, f). The CCP guarantees that if f_new is between f and g, and M' < max(conv(f,f)), then max(conv(f_new, f_new)) < max(conv(f,f)) for small enough t. This is because conv(f_new, f_new) is a "quadratic interpolation" between conv(f,f) and conv(g,g), and the peak of conv(f,g) is bounded by M'.

More precisely:
conv(f_new, f_new) = (1-t)² conv(f,f) + 2t(1-t) conv(f,g) + t² conv(g,g)

max(conv(f_new, f_new)) <= (1-t)² * max(conv(f,f)) + 2t(1-t) * M' + t² * max(conv(g,g))

If M' < max(conv(f,f)) and max(conv(g,g)) is not too large, then for small t:
max(conv(f_new, f_new)) < max(conv(f,f))

The issue is max(conv(g,g)). But since g is found by the LP, it's "aligned" with f, so conv(g,g) shouldn't be much larger than conv(f,f).

"""
Hybrid LP + subgradient + coordinate descent optimization for minimizing
autocorrelation peak of non-negative step functions.

Objective: minimize 2*n*max(conv(a,a,'full')) / (sum(a))^2 for a >= 0.

Strategy:
- Multi-start with diverse initializations (uniform, cosine, bimodal, random)
- LP-based direction finding (max-sum and min-peak formulations)
- Relaxed peak constraint to escape local minima
- Subgradient descent with momentum for large dimensions
- Coordinate descent polish for fine-tuning
- Adaptive time allocation across phases

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score = 2*n*max(conv)/(sum^2) faithfully."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * float(conv.max()) / (s * s))


def make_conv_matrix(n, f):
    """Build sparse (2n-1 x n) matrix A where (A@x)[k] = conv(f, x, 'full')[k]."""
    A_k = []
    A_j = []
    A_v = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        A_k.extend([k] * len(js))
        A_j.extend(js.tolist())
        A_v.extend(np.asarray(f[k - js], dtype=np.float64).tolist())
    return sp.coo_matrix((A_v, (A_k, A_j)), shape=(2 * n - 1, n)).tocsr()


def lp_maxsum(f, M, tl=8.0):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    if n > 3000:
        return None

    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0.0, None)] * n, method='highs',
                    options={'time_limit': tl, 'presolve': True})
        if r.status == 0:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def lp_minpeak(f, tl=8.0):
    """Solve min M' s.t. conv(f,g)[k] <= M', sum(g)=1, g>=0. Returns (g, M_opt) or (None, None)."""
    n = len(f)
    if n > 3000:
        return None, None
    A = make_conv_matrix(n, f)
    c = np.zeros(n + 1)
    c[n] = 1.0
    # A_ub: [A, -I] * [g; M'] <= 0
    neg_ones = -np.ones((2 * n - 1, 1))
    A_ub_s = sp.hstack([A, sp.csc_matrix(neg_ones)]).tocsr()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bnd = [(0.0, None)] * n + [(0.0, None)]
    try:
        r = linprog(c, A_ub=A_ub_s, b_ub=np.zeros(2 * n - 1),
                    A_eq=A_eq, b_eq=b_eq, bounds=bnd, method='highs',
                    options={'time_limit': tl, 'presolve': True})
        if r.status == 0:
            g = np.maximum(r.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-12:
                g /= gs
            return g, float(r.x[n])
    except Exception:
        pass
    return None, None


def backtrack_step(f, g, nr_scales=12):
    """Find best t in a set of scales to blend f and g."""
    best_f = f.copy()
    best_sc = fast_score(f)
    scales = np.geomspace(0.03, 1.0, nr_scales)
    for t in scales:
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
    return best_f, best_sc


def block_perturb(f, rng, w_lo=1, w_hi=50):
    """Multiplicative random block perturbation, renormalised to sum 1."""

def subgradient_step(f):
    """Compute subgradient of max(conv(f,f)) w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    peaks = np.where(conv >= M * 0.9999)[0]
    s = np.zeros(n)
    for k in peaks:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            s[j] += f[k - j] + f[j]
    return s, M


def subgradient_optimize(f_init, budget_s, lr_init=0.005, momentum=0.85):
    """Subgradient descent with momentum for large n."""
    n = len(f_init)
    if n > 3000:
        return f_init, fast_score(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        return f_init, fast_score(f_init)
    f = f / s
    best = f.copy()
    best_sc = fast_score(f)
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < budget_s:
        sg, M = subgradient_step(f)
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        v = momentum * v - lr * sg
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-10:
            f_new = f
            fs = 1.0
        f_new = f_new / fs
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f_new.copy()
            stall = 0
        else:
            stall += 1
            lr *= 0.97
        f = f_new
    return best, best_sc

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    scales = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]

    for rnd in range(max_rounds):
        if time.monotonic() - start_time > time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start_time > time_limit:
                break
            f_old = best[j]
            if f_old < 1e-12:
                continue
            for sc in scales:
                trial = best.copy()
                trial[j] = f_old * sc
                s = trial.sum()
                if s < 1e-10:
                    continue
                trial = trial / s
                sc_val = fast_score(trial)
                if sc_val < best_sc:
                    best = trial
                    best_sc = sc_val
    return best, best_sc


def make_initial(n, rng, method):
    """Generate a non-negative sum-1 starting vector of length n."""
    if method == 'uniform':
        a = np.ones(n) * (1.0 + rng.normal(0, 0.3, n))
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x) + rng.normal(0, 0.05, n)
    elif method == 'sine_sq':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.02 + rng.normal(0, 0.02, n)
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 10)
        a = np.exp(-0.5 * ((x - n // 4) / s) ** 2) + np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2)
        a += 0.01 + rng.normal(0, 0.01, n)
    elif method == 'exp_decay':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-x / (n / 3.0)) + 0.01
    elif method == 'triangle':
        x = np.arange(n, dtype=np.float64)
        a = 1.0 - 2.0 * np.abs(x - n / 2.0) / n
        a += 0.01 + rng.normal(0, 0.01, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.005)
    return a / a.sum()


def run_lp_optimize(f_init, budget_s, relaxed=0.0, tl=8.0, use_minpeak_every=5):
    """Main LP-based optimization loop with relaxed peak and backtracking."""
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        return f_init, fast_score(f_init)
    f = f / s
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    stall = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < budget_s:
        it += 1
        if n > 2500:
            # Use subgradient for very large n
            c, sc = subgradient_optimize(f, min(5.0, budget_s - (time.monotonic() - t0)))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = c.copy()
                f = c.copy()
                stall = 0
            continue

        conv = np.convolve(f, f)
        M = float(conv.max())
        M_relaxed = M * (1.0 + relaxed)

        # Choose LP formulation
        g = None
        if use_minpeak_every > 0 and it % use_minpeak_every == 0:
            g, M_opt = lp_minpeak(f, tl=tl)
            if g is not None:
                # Use min-peak result
                pass
            else:
                g = lp_maxsum(f, M_relaxed, tl=tl)
        else:
            g = lp_maxsum(f, M_relaxed, tl=tl)

        if g is None or g.sum() < 1e-12:
            # Fallback: perturb
            rng = np.random.default_rng(it * 7919 + 3)
            f = block_perturb(f, rng, 1, max(2, n // 10))
            continue

        g = g / g.sum()

        # Backtrack to find best blend
        f_new, sc_new = backtrack_step(f, g, nr_scales=10)

        if sc_new < best_sc - 1e-13:
            best = f_new.copy()
            best_sc = sc_new
            f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Escalate exploration
                rng = np.random.default_rng(it * 104729 + 11)
                f = block_perturb(f, rng, 1, max(3, n // 8))
                # Increase relaxation
                relaxed = min(0.3, relaxed + 0.02)
                stall = 0
            elif stall > 10:
                # Mild perturbation
                rng = np.random.default_rng(it * 15485863 + 17)
                f = block_perturb(f, rng, 1, max(2, n // 20))
                stall = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the autocorrelation score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Get baseline from global variable if available
    best = None
    best_sc = np.inf
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0.01:
            best = parent.copy()
            best_sc = float(evaluate_sequence(best.tolist()))
            if not np.isfinite(best_sc):
                best_sc = fast_score(best)
            print(f"Baseline: score={best_sc:.6f}, n={best.size}", flush=True)
    except Exception:
        pass

    if best is None:
        best = np.ones(512, dtype=np.float64)
        best_sc = fast_score(best)
        print(f"No baseline found, using uniform: score={best_sc:.6f}", flush=True)

    methods = ['uniform', 'cosine', 'sine_sq', 'random', 'bimodal', 'exp_decay', 'triangle']
    ns = [256, 384, 512, 768, 1024]

    # If baseline has a different length, add it
    if best.size not in ns and best.size <= 1500:
        ns.append(int(best.size))
    ns = sorted(set(ns))

    # Phase 1: Quick multi-start scan
    total_runs = len(ns) * len(methods)
    time_per = min(25.0, max(4.0, (float(budget_s) * 0.25) / total_runs))
    print(f"\nPhase 1: Quick scan ({len(ns)} lengths x {len(methods)} methods, ~{time_per:.1f}s each)", flush=True)

    for n in ns:
        if time.monotonic() > deadline:
            break
        for m in methods:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 3.0:
                break
            start = make_initial(n, rng, m)
            local_budget = min(time_per, remaining)
            c, sc = run_lp_optimize(start, local_budget, relaxed=0.0, tl=4.0)
            if np.isfinite(sc) and sc < best_sc:
                best = c.copy()
                best_sc = sc
                print(f"  n={n:4d} {m:12s}: score={sc:.6f}", flush=True)

    print(f"\nPhase 1 done. Best: {best_sc:.6f} (n={best.size})", flush=True)

    # Phase 2: Deep optimization starting from best
    print("\nPhase 2: Deep LP optimization", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        deep_budget = min(remaining * 0.35, 500.0)
        c, sc = run_lp_optimize(best.copy(), deep_budget, relaxed=0.0, tl=10.0)
        if np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = sc
            print(f"  Deep: score={sc:.6f} (n={best.size})", flush=True)

    # Phase 3: Relaxed LP for escape
    print("\nPhase 3: Relaxed peak LP", flush=True)
    remaining = deadline - time.monotonic()
    for rel in [0.02, 0.05, 0.1, 0.15]:
        if time.monotonic() > deadline:
            break
        local_budget = min((deadline - time.monotonic()) * 0.15, 150.0)
        if local_budget < 10:
            break
        c, sc = run_lp_optimize(best.copy(), local_budget, relaxed=rel, tl=8.0)
        if np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = sc
            print(f"  relaxed={rel:.2f}: score={sc:.6f}", flush=True)

    # Phase 4: Coordinate descent polish
    print("\nPhase 4: Coordinate descent polish", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        cd_budget = min(remaining * 0.3, 120.0)
        c, sc = coord_descent(best.copy(), max_rounds=4, time_limit=cd_budget, seed=seed + 1)
        if np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = sc
            print(f"  CoordDesc: score={sc:.6f}", flush=True)

    # Phase 5: Final LP + coord alternation
    print("\nPhase 5: Final refinement (alternating LP + coord)", flush=True)
    for it in range(8):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        # LP step
        lp_budget = min(remaining * 0.4, 80.0)
        if lp_budget > 5:
            c, sc = run_lp_optimize(best.copy(), lp_budget, relaxed=0.01, tl=6.0)
            if np.isfinite(sc) and sc < best_sc:
                best = c.copy()
                best_sc = sc
        # Coord descent step
        cd_budget = min(remaining * 0.3, 40.0)
        if cd_budget > 5:
            c2, sc2 = coord_descent(best.copy(), max_rounds=2, time_limit=cd_budget, seed=seed + 100 + it)
            if np.isfinite(sc2) and sc2 < best_sc:
                best = c2.copy()
                best_sc = sc2
        if np.isfinite(best_sc):
            print(f"  iter {it}: score={best_sc:.6f}", flush=True)

    # Phase 6: Final heavy polish
    print("\nPhase 6: Final heavy polish", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 15:
        c, sc = coord_descent(best.copy(), max_rounds=5, time_limit=min(remaining * 0.5, 90.0),
                              seed=seed + 999)
        if np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = sc
            print(f"  FinalCD: score={sc:.6f}", flush=True)

    # Final LP attempt
    remaining = deadline - time.monotonic()
    if remaining > 10:
        c, sc = run_lp_optimize(best.copy(), min(remaining * 0.8, 120.0), relaxed=0.0, tl=8.0)
        if np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = sc
            print(f"  FinalLP: score={sc:.6f}", flush=True)

    # Verify with authoritative evaluator
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256, dtype=np.float64)

    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(256, dtype=np.float64)
        final_sc = float(evaluate_sequence(best.tolist()))

    print(f"\nFINAL: score={final_sc:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
