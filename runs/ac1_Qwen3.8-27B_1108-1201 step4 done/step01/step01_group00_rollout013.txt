from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0
    """
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ g <= b_ub
    # For each k = 0, ..., 2n-2:
    #   sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} a[i] * g[k-i] <= M
    # This is a banded matrix
    
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    b_ub = np.full(n_cons, M)
    
    for k in range(n_cons):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i  # g index
            A_ub[k, j] = a[i]
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(n) / n  # fallback

    """Compute the score: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))

def solve_lp(a, M, n, g_min=0.0):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= g_min
    Returns the optimal g vector.
    """
    # Variables: g_0, ..., g_{n-1}
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix A_ub such that A_ub @ g <= b_ub
    # For each k = 0, ..., 2n-2:
    #   sum_{i: valid} a[i] * g[k-i] <= M
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    b_ub = np.full(n_cons, M)
    
    for k in range(n_cons):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[k, j] = a[i]
    
    bounds = [(g_min, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n * M / np.mean(a)

def alternating_lp_optimize(a_init, time_limit, rng, verbose=True):
    """
    Alternating LP optimization.
    Minimizes max(conv(a,a)) subject to sum(a) = sqrt(2n), a >= 0.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a / np.sum(a) * S
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    t_blend = 0.3
    step = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(a, a))
        
        # Solve LP
        g = solve_lp(a, M, n)
        g = np.clip(g, 0, None)
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g / sum_g * S
        
        # Blend
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.clip(a_new, 0, None)
        
        score_new = local_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stuck_count = 0
            # Slightly increase step
            t_blend = min(0.6, t_blend * 1.02)
        else:
            stuck_count += 1
            if score_new < best_score * 1.05:
                a = a_new
                t_blend = max(0.05, t_blend * 0.95)
            else:
                # Revert
                a = best_a.copy()
                t_blend = max(0.05, t_blend * 0.8)
        
        step += 1
        if verbose and step % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      step={step}, M={M:.4f}, score={score_new:.6f}, best={best_score:.6f}, t={t_blend:.3f}, time={elapsed:.1f}s")
        
        # If stuck for too long, try a perturbation
        if stuck_count > 100:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.clip(a, 0, None)
            t_blend = 0.3
            stuck_count = 0
    
    return best_a, best_score

def polish_local_search(a_init, time_limit, rng):
    """Local search with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce the peak
            conv = fft_convolve(current, current)
            # Find top peaks
            top_indices = np.argsort(conv)[-3:]
            for peak_idx in top_indices:
                # Find which pairs contribute to this peak
                contribs = np.zeros(n)
                for i in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                # Reduce top contributors
                top_k = max(2, n // 16)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.8:
            # Smoothing
            kernel_size = max(3, n // 32)
            kernel = np.ones(kernel_size) / kernel_size
            candidate = np.convolve(candidate, kernel, mode='same')
            candidate *= np.exp(rng.normal(0, 0.005, n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0, None)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score

def make_chirp_init(n, rng):

def make_window_init(n, rng):
    """Initialize with various window functions."""
    t = np.arange(n, dtype=np.float64)
    x = 2 * t / (n - 1) - 1  # [-1, 1]
    window_type = rng.integers(0, 4)
    if window_type == 0:  # Hann
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif window_type == 1:  # Blackman
        a = 0.42 + 0.5 * np.cos(np.pi * x) + 0.08 * np.cos(2 * np.pi * x)
    elif window_type == 2:  # Tukey

        a = np.ones(n)

            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
            a[n - 1 - i] = a[i]

        sigma = 0.2 + 0.3 * rng.random()
        a = np.exp(-x**2 / (2 * sigma**2))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
Alternating LP optimization for minimizing the normalized peak autocorrelation
of non-negative step functions. Combines LP-based dual updates with targeted
local search polishing and multiple restarts at different sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)_k = conv(a,g)_k."""
    n = len(a)
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    for k in range(n_cons):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = a[i]
    return A


def solve_lp(a, M, n):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k=0..2n-2, g >= 0.
    Returns optimal g vector.
    """
    c_obj = -np.ones(n)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return np.maximum(result.x, 0.0)
    else:
        return np.ones(n) / n


def solve_lp_relaxed(a, M, n, relax=1.001):
    """Solve LP with slightly relaxed M to get interior solution."""
    return solve_lp(a, M * relax, n)


def alternating_lp_optimize(a_init, time_limit, rng, verbose=True, relax=1.0):
    """
    Alternating LP optimization minimizing max(conv(a,a)) with sum(a)=sqrt(2n).
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a / np.sum(a) * S

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    t_blend = 0.35
    step = 0
    stuck_count = 0
    best_streak = 0

    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(a, a))

        if relax > 1.0:
            g = solve_lp_relaxed(a, M, n, relax)
        else:
            g = solve_lp(a, M, n)

        g = np.clip(g, 0, None)
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g / sum_g * S

        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) < 1e-10:
            break

        score_new = local_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stuck_count = 0
            best_streak += 1
            t_blend = min(0.55, t_blend * 1.015)
        else:
            stuck_count += 1
            best_streak = 0
            if score_new < best_score * 1.03:
                a = a_new
            else:
                a = best_a.copy()
            t_blend = max(0.08, t_blend * 0.94)

        # Perturbation when stuck
        if stuck_count > 80:
            a = best_a * np.exp(rng.normal(0, 0.08, n))
            a = np.clip(a, 0, None)
            t_blend = 0.3
            stuck_count = 0

        step += 1
        if verbose and step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      step={step}, M={M:.4f}, score={score_new:.6f}, best={best_score:.6f}, t={t_blend:.3f}, time={elapsed:.1f}s")

    return best_a, best_score


def polish_local_search(a_init, time_limit, rng):
    """
    Local search with targeted peak reduction, smoothing, and block perturbations.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99985
    step = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce top peaks
            conv = fft_convolve(current, current)
            top_indices = np.argsort(conv)[-2:]
            for peak_idx in top_indices:
                contribs = np.zeros(n)
                for i in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(3, n // 12)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.012, size=len(top_idx)))
        elif r < 0.55:
            # Anti-peak: boost low-contribution regions
            conv = fft_convolve(current, current)
            peak_val = conv.max()
            # Find elements that contribute least to peak
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            low_idx = np.argsort(contribs)[:max(3, n // 10)]
            candidate[low_idx] *= np.exp(rng.normal(0.025, 0.012, size=len(low_idx)))
        elif r < 0.70:
            # Block perturbation
            n_blocks = rng.integers(2, 6)
            for _ in range(n_blocks):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.82:
            # Smoothing (blend with neighbor average)
            alpha = 0.1 + 0.3 * rng.random()
            smoothed = np.zeros_like(current)
            smoothed[0] = (current[0] + current[1]) / 2
            smoothed[-1] = (current[-1] + current[-2]) / 2
            smoothed[1:-1] = (current[:-2] + 2 * current[1:-1] + current[2:]) / 4
            candidate = (1 - alpha) * current + alpha * smoothed
        elif r < 0.92:
            # Symmetry exploration
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.01, n))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0, 0.008, n))

        candidate = np.clip(candidate, 0, None)
        if np.sum(candidate) < 0.01:
            continue

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling
        step += 1
        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      polish step={step}, score={current_score:.6f}, best={best_score:.6f}, T={temperature:.6f}, time={elapsed:.1f}s")

    return best, best_score


def make_chirp_init(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.2 + 0.5 * rng.random()
    freq = 0.5 + 1.5 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_window_init(n, rng):
    """Window function initialization."""
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    wtype = int(rng.integers(0, 5))
    if wtype == 0:
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif wtype == 1:
        a = 0.42 + 0.5 * np.cos(np.pi * x) + 0.08 * np.cos(2.0 * np.pi * x)
    elif wtype == 2:
        sigma = 0.2 + 0.4 * rng.random()
        a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    elif wtype == 3:
        a = np.abs(np.sinc(x * (1.0 + 2.0 * rng.random()))) + 0.01
    else:
        # Cosine raised to power
        p = 1.0 + 3.0 * rng.random()
        a = np.abs(np.cos(np.pi * x / 2.0)) ** p
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = int(rng.integers(3, 8))
    for _ in range(n_tones):
        k = int(rng.integers(1, n // 4 + 1))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.4):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: (1) alternating LP at multiple lengths/inits,
    (2) local search polish, (3) final LP refinement.
    Returns the best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Define search tasks: (n, init_type, init_i)
    tasks = []

    if start_seq is not None:
        tasks.append((len(start_seq), 'provided', 0))

    # Try various lengths with different initializations
    lengths_to_try = [256, 320, 384, 512, 640, 768]
    init_types = ['chirp', 'window', 'multitone', 'random']

    for n in lengths_to_try:
        if start_seq is not None and n == len(start_seq):
            continue
        for init_t in init_types[:3]:  # top 3 init types per length
            tasks.append((n, init_t, 0))

    print(f"\nTotal tasks: {len(tasks)}")
    print(f"Time budget: {time_budget:.0f}s")

    # Phase 1: Alternating LP
    print("\n=== Phase 1: Alternating LP Optimization ===")
    phase1_deadline = deadline - 180.0  # Reserve 180s for polish + final

    for task_idx, (n, init_type, init_i) in enumerate(tasks):
        if time.monotonic() > phase1_deadline:
            break

        remaining = phase1_deadline - time.monotonic()
        tasks_remaining = len(tasks) - task_idx
        # Allocate time: more time for smaller n (faster LP), less for larger
        base_time = 40.0 if n <= 384 else 25.0
        time_per = min(base_time, remaining / max(1, tasks_remaining))
        time_per = max(15.0, time_per)

        # Generate initial sequence
        if init_type == 'provided' and start_seq is not None:
            if len(start_seq) == n:
                a0 = start_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, start_seq)
        elif init_type == 'chirp':
            a0 = make_chirp_init(n, rng)
        elif init_type == 'window':
            a0 = make_window_init(n, rng)
        elif init_type == 'multitone':
            a0 = make_multitone_init(n, rng)
        else:
            a0 = make_random_init(n, rng)

        # Run alternating LP
        a_best, score = alternating_lp_optimize(a0, time_per, rng, verbose=False)

        # Also try with relaxation
        a_best2, score2 = alternating_lp_optimize(a0, time_per * 0.5, rng, verbose=False, relax=1.002)
        if score2 < score:
            a_best, score = a_best2, score2

        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"  [{task_idx+1}/{len(tasks)}] n={n}, init='{init_type}', score={score:.6f} *** NEW BEST ***")
        else:
            print(f"  [{task_idx+1}/{len(tasks)}] n={n}, init='{init_type}', score={score:.6f}")

    print(f"\n  Best after Phase 1: {best_score:.6f}")

    # Phase 2: Local search polish
    print("\n=== Phase 2: Local Search Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        polish_time = min(200.0, deadline - time.monotonic() - 60)
        print(f"  Polish time: {polish_time:.0f}s, starting score={best_score:.6f}")
        polished, polish_score = polish_local_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  After polish: no improvement (best={best_score:.6f})")

    # Phase 3: Final LP refinement on best
    print("\n=== Phase 3: Final LP Refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 20
        final_time = min(90.0, remaining)
        print(f"  Final LP time: {final_time:.0f}s")
        a_final, score_final = alternating_lp_optimize(best_seq, final_time, rng, verbose=False)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"  After final LP: {best_score:.6f}")

        # Quick polish
        if time.monotonic() < deadline - 20:
            quick_time = min(30.0, deadline - time.monotonic() - 20)
            if quick_time > 5:
                a_polished, score_polished = polish_local_search(best_seq, quick_time, rng)
                if score_polished < best_score:
                    best_score = score_polished
                    best_seq = a_polished.copy()
                    print(f"  After quick polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
