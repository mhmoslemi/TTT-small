For the spectral flattening, here's the algorithm:
```
def spectral_flatten(p, n_iters, alpha):
    L = next_fast_len(2*n)
    for _ in range(n_iters):
        P = fft(p, L)
        power = abs(P)**2
        # Compute current autocorrelation
        c = real(ifft(power))[:2*n-1]
        # Target: make power more uniform (except DC)
        target = mean(power[1:])  # exclude DC
        # Blend
        new_power = power.copy()
        new_power[1:] = alpha * power[1:] + (1-alpha) * target
        # Also handle negative power (shouldn't happen but just in case)
        new_power = max(new_power, 0)
        # Reconstruct
        P_new = sqrt(new_power) * exp(1j * angle(P))
        p_new = real(ifft(P_new))
        # Project to simplex
        p = project_simplex(p_new)
    return p
```

"""
Step-function search using softmax-parameterized Adam optimization with
spectral pre-conditioning and temperature annealing.

Algorithm:
1. Generate diverse asymmetric initial shapes at multiple lengths.
2. Pre-condition via iterative spectral flattening (GS-style).
3. Optimize using Adam on log-parameters with temperature-annealed
   smoothed-max surrogate.
4. Final refinement via coordinate descent on the true max.
"""

import time
import numpy as np

try:
    from scipy.fft import fft, ifft, next_fast_len
except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

def _conv_full(p):
    n = p.size
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    return np.real(ifft(F * F))[:2 * n - 1]

def _score(p):
    n = p.size
    s = p.sum()
    if s < 1e-12:
        return np.inf
    c = _conv_full(p)
    return 2.0 * n * c.max() / (s * s)

def _project_simplex(v):
    n = v.size
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.sum(cond) - 1)
    theta = (css[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    return out / s if s > 1e-300 else np.full(n, 1.0 / n)

def _safe_normalize(p):
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 1e-12:
        return np.full(p.size, 1.0 / p.size)
    return p / s

def _smoothed_max_and_grad(p, T_factor):
    """Compute smoothed max of conv(p,p) and its gradient wrt p.
    
    Returns (surrogate_value, gradient_vector, c, T).
    """
    n = p.size
    c = _conv_full(p)
    cmax = c.max()
    if cmax <= 0:
        return 0.0, np.zeros(n), c, 1.0
    
    T = max(T_factor * cmax, 1e-300)
    # Stable logsumexp
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    surr = cmax + T * np.log(s) if s > 0 else cmax
    
    # Gradient: dS/dp[j] = 2 * sum_k w[k] * p[k-j] for valid k-j
    # Computed via FFT cross-correlation
    L_grad = next_fast_len(3 * n - 2)
    Fp = fft(p, L_grad)
    Fw = fft(w[::-1], L_grad)
    h = np.real(ifft(Fp * Fw))
    # Extract: G[j] = h[2n-2-j]
    G = h[2 * n - 2 : n - 2 : -1]
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    grad = 2.0 * G
    return surr, grad, c, T

def _adam_optimize(p0, time_limit, initial_T_factor=0.1, min_T_factor=1e-5,
                   lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                   decay_rate=0.9995, seed=0):
    """Adam optimizer on softmax(log-parameters) with temperature annealing."""
    rng = np.random.default_rng(seed)
    p = _safe_normalize(p0).copy()
    n = p.size
    
    # Initialize log-parameters (avoid log(0))
    p_init = np.maximum(p, 1e-20)
    q = np.log(p_init)  # log-parameters
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_p = p.copy()
    best_score = _score(p)
    current_score = best_score
    
    T_factor = initial_T_factor
    end_time = time.monotonic() + time_limit
    no_improve = 0
    
    while time.monotonic() < end_time:
        t += 1
        # Anneal temperature
        T_factor = max(min_T_factor, T_factor * decay_rate)
        
        # Forward: p = softmax(q)
        q_max = q.max()
        exp_q = np.exp(q - q_max)
        S = exp_q.sum()
        p = exp_q / S
        
        # Compute smoothed max and gradient wrt p
        surr, grad_p, c, T = _smoothed_max_and_grad(p, T_factor)
        
        # Convert gradient to q-space: dS/dq[j] = p[j] * (grad_p[j] - sum_i p[i]*grad_p[i])
        mean_grad = (p * grad_p).sum()
        grad_q = p * (grad_p - mean_grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_q
        v = beta2 * v + (1 - beta2) * grad_q ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        q = q - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Compute actual score periodically
        if t % 100 == 0:
            current_score = _score(p)
            if current_score < best_score:
                best_score = current_score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Restart if stuck
            if no_improve > 50:
                q = np.log(np.maximum(best_p, 1e-20)) + rng.normal(0, 0.1, n)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                T_factor = initial_T_factor * 0.5
                no_improve = 0
        
        if t % 1000 == 0:
            print(f"  adam: t={t} T_factor={T_factor:.2e} best={best_score:.10f} cur={_score(p):.10f}")
    
    return best_p, best_score

def _spectral_flatten(p0, n_iters=200, alpha=0.7, seed=0):
    """GS-style spectral flattening."""
    p = _safe_normalize(p0).copy()
    n = p.size
    L = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    
    for it in range(n_iters):
        P = fft(p, L)
        power = np.abs(P) ** 2
        # Target: uniform power (excluding DC component)
        target = power[1:].mean() if n > 1 else power[0]
        new_power = power.copy()
        new_power[1:] = alpha * power[1:] + (1 - alpha) * target
        new_power = np.maximum(new_power, 0)
        # Reconstruct with same phase
        P_new = np.sqrt(new_power) * np.exp(1j * np.angle(P))
        p_new = np.real(ifft(P_new))
        p = _project_simplex(p_new)
        
        if it % 50 == 0:
            sc = _score(p)
            if it % 100 == 0:
                print(f"  spectral: it={it} score={sc:.10f}")
    
    return p

def _coord_descent(p0, time_limit, seed=0):
    """Coordinate descent on the true (non-smooth) max."""
    rng = np.random.default_rng(seed)
    p = _safe_normalize(p0).copy()
    n = p.size
    best_score = _score(p)
    best_p = p.copy()
    
    end_time = time.monotonic() + time_limit
    no_improve = 0
    
    while time.monotonic() < end_time:
        # Try moving mass from position i to position j
        # Choose i as argmax of p (most concentrated), j as random
        i = int(np.argmax(p))
        j = int(rng.integers(0, n))
        if i == j:
            j = (i + 1) % n
        
        for frac in (0.1, 0.05, 0.02, 0.01, 0.005, 0.001):
            delta = frac * p[i]
            if delta <= 0:
                continue
            p_new = p.copy()
            p_new[i] -= delta
            p_new[j] += delta
            p_new = _project_simplex(p_new)
            sc = _score(p_new)
            if sc < best_score:
                best_score = sc
                best_p = p_new.copy()
                p = p_new
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 200:
                # Try random pair
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    for frac in (0.05, 0.02, 0.01):
                        delta = frac * p[i]
                        p_new = p.copy()
                        p_new[i] -= delta
                        p_new[j] += delta
                        p_new = _project_simplex(p_new)
                        sc = _score(p_new)
                        if sc < best_score:
                            best_score = sc
                            best_p = p_new.copy()
                            p = p_new
                            no_improve = 0
                            break
    
    return best_p, best_score
