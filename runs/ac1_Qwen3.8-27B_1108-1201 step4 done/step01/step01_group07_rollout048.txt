    sg = np.zeros(n)

    k = min(top_k, len(w))

        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mirror_descent_step(a, conv, t, eta, top_k_frac=0.05):
    """One step of mirror descent with top-k peak focusing."""
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a, conv
    
    # Softmax weights
    scaled = t * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        return a, conv
    w /= w_sum
    
    # Top-k selection
    k = max(1, int(n * top_k_frac))
    top_idx = np.argsort(w)[-k:][::-1]
    
    # Compute subgradient
    sg = np.zeros(n)
    for ki in top_idx:
        k_idx = int(ki)
        wk = w[k_idx]
        j_start = max(0, k_idx - n + 1)
        j_end = min(n - 1, k_idx)
        if j_start <= j_end:
            sg[j_start:j_end+1] += wk * a[k_idx - j_end:k_idx - j_start + 1][::-1]
    
    # Center the subgradient (to preserve sum)
    sg_centered = sg - sg.mean()
    
    # Multiplicative update
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new)
    
    conv_new = fast_convolve(a_new)
    return a_new, conv_new

def run_mirror_descent(a0, time_limit, t_start=30.0, t_end=300.0, eta_start=0.8, 
                       eta_decay=0.95, patience=50, top_k_frac=0.05):
    """
    Mirror descent with adaptive temperature and step size.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    
    eta = eta_start
    t = t_start
    patience_count = 0
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        a, conv = mirror_descent_step(a, conv, t, eta, top_k_frac)
        current_obj = 2.0 * n * conv.max()
        iteration += 1
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience_count = 0
        else:
            patience_count += 1
        
        if patience_count >= patience:
            # Reduce step size and increase temperature
            eta *= eta_decay
            t = min(t * 1.3, t_end)
            patience_count = 0
            
            if eta < 0.01:
                # Reset with new parameters
                eta = 0.3
                t = t_start * np.random.uniform(0.5, 2.0)
                patience = max(20, patience // 2)
        
        # Gradually increase temperature
        progress = (time.monotonic() - t0) / time_limit
        t = t_start + (t_end - t_start) * progress
    
    return best_a, best_obj

def run_pgdescent(a0, time_limit, t=100.0):
    """
    Projected gradient descent with backtracking line search.
    Optimizes the log-sum-exp approximation.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    
    def lse_objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        c = fast_convolve(x)
        scaled = t * c
        scaled -= scaled.max()
        return np.log(np.sum(np.exp(scaled)))
    
    def lse_gradient(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(len(x))
        x = x / s
        c = fast_convolve(x)
        scaled = t * c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        # Subgradient
        sg = np.zeros(len(x))
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-10:
                continue
            j_start = max(0, k_idx - n + 1)
            j_end = min(n - 1, k_idx)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k_idx] * x[k_idx - j_end:k_idx - j_start + 1][::-1]
        return sg
    
    eta = 1.0 / n
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        sg = lse_gradient(a)
        sg_centered = sg - sg.mean()
        
        # Backtracking line search
        alpha = eta
        f_curr = 2.0 * n * conv.max()
        improved = False
        for _ in range(10):
            a_trial = project_simplex(a - alpha * sg_centered)
            conv_trial = fast_convolve(a_trial)
            f_trial = 2.0 * n * conv_trial.max()
            if f_trial < f_curr - 1e-12:
                a = a_trial
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            eta *= 0.5
            no_improve += 1
        else:
            eta = min(eta * 1.1, 1.0)
            no_improve = 0
        
        current_obj = 2.0 * n * fast_convolve(a).max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
        
        if no_improve > 30:
            break
    
    return best_a, best_obj

def run_scipy_polish(a0, time_limit, t=200.0):
    """
    Use scipy.optimize.minimize with L-BFGS-B to polish the solution.
    Optimizes the log-sum-exp approximation.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    
    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        c = fast_convolve(x)
        scaled = t * c
        scaled -= scaled.max()
        return float(np.log(np.sum(np.exp(scaled))))
    
    def gradient(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(len(x))
        x = x / s
        c = fast_convolve(x)
        scaled = t * c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        sg = np.zeros(len(x))
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-10:
                continue
            j_start = max(0, k_idx - n + 1)
            j_end = min(n - 1, k_idx)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k_idx] * x[k_idx - j_end:k_idx - j_start + 1][::-1]
        return sg
    
    t0 = time.monotonic()
    x0 = a.copy()
    
    # Use L-BFGS-B with bounds [0, 1]
    bounds = [(0, 1)] * n
    result = minimize(objective, x0, method='L-BFGS-B', jac=gradient,
                      bounds=bounds, options={'maxiter': 100, 'maxfun': 200})
    
    if result.success or result.fun < objective(x0):
        a_new = normalize(result.x)
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
    
    return best_a, best_obj

def run_scan_and_optimize(seed, budget_s):

            parent = p
    except Exception:
        pass
    
    print(f"Parent available: {parent is not None}, length: {parent.size if parent is not None else 0}")
    
    # Define initializations for a given length n
    def get_initializations(n, rng):
        inits = {}
        x = np.arange(n, dtype=np.float64)
        
        # Uniform
        inits['uniform'] = np.ones(n) / n
        
        # Gaussian (narrow)
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.15 * n)) ** 2)
        inits['gauss_narrow'] = g / g.sum()
        
        # Gaussian (wide)
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.4 * n)) ** 2)
        inits['gauss_wide'] = g / g.sum()
        
        # Random Dirichlet
        inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
        
        # Anti-peak: try to have values that reduce the center peak
        # Start uniform, reduce center, increase edges
        anti = np.ones(n)
        center = n // 2
        for i in range(n):
            anti[i] = 1.0 + 0.5 * abs(i - center) / center
        inits['anti_peak'] = anti / anti.sum()
        
        # Bimodal (edges)
        bi = np.zeros(n)
        bi[:max(1, n // 4)] = 1.0
        bi[-max(1, n // 4):] = 1.0
        inits['bimodal'] = bi / bi.sum()
        
        # From parent
        if parent is not None:
            if parent.size >= n:
                pp = parent[:n].copy()
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_interp'] = pp / pp.sum()

        if parent is not None and parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                noise = rng.normal(0, 0.1, n)
                pp_noisy = np.maximum(pp + noise, 0)

                    inits['parent_noisy'] = pp_noisy / pp_noisy.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with adaptive temperature for main optimization
2. Efficient FFT-based subgradient computation
3. scipy L-BFGS-B polishing with log-sum-exp smoothing
4. Multiple sequence lengths and diverse initializations
5. Parallel-free sequential search with time budgeting
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def efficient_subgradient(w, a):
    """
    Compute sg[i] = sum_j a[j] * w[i+j] using FFT (cross-correlation).
    w has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    return R[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_step(a, conv, t, eta, top_k_frac=0.05):
    """One step of mirror descent with top-k peak focusing."""
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a, conv

    # Softmax weights
    scaled = t * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        return a, conv
    w /= w_sum

    # Top-k selection for efficiency
    k = max(3, int(len(w) * top_k_frac))
    top_idx = np.argsort(w)[-k:][::-1]

    # Compute subgradient from top-k peaks
    sg = np.zeros(n)
    for ki in top_idx:
        k_idx = int(ki)
        wk = w[k_idx]
        j_start = max(0, k_idx - n + 1)
        j_end = min(n - 1, k_idx)
        if j_start <= j_end:
            sg[j_start:j_end + 1] += wk * a[k_idx - j_end:k_idx - j_start + 1][::-1]

    # Center to preserve sum
    sg_centered = sg - sg.mean()

    # Multiplicative update
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new)

    conv_new = fast_convolve(a_new)
    return a_new, conv_new


def run_mirror_descent(a0, time_limit, t_start=30.0, t_end=300.0,
                       eta_start=0.6, patience=50, top_k_frac=0.05,
                       diversify_every=200, diversity_scale=0.01):
    """
    Mirror descent with adaptive temperature, step size, and diversification.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    eta = eta_start
    t = t_start
    patience_count = 0
    t0 = time.monotonic()
    iteration = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        a, conv = mirror_descent_step(a, conv, t, eta, top_k_frac)
        current_obj = 2.0 * n * conv.max()
        iteration += 1

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience_count = 0
        else:
            patience_count += 1

        if patience_count >= patience:
            eta *= 0.7
            t = min(t * 1.5, t_end * 2)
            patience_count = 0

            if eta < 0.02:
                eta = max(0.1, eta_start * 0.5)
                t = t_start * rng.uniform(0.5, 2.0)
                patience = max(20, patience // 2)

        # Gradually increase temperature over time
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        t_adj = t_start + (t_end - t_start) * progress
        t = max(t, t_adj)

        # Periodic diversification to escape local minima
        if iteration % diversify_every == 0 and eta > 0.05:
            noise = rng.normal(0, diversity_scale, n)
            a_noisy = a + noise
            a_noisy = normalize(a_noisy)
            conv_noisy = fast_convolve(a_noisy)
            obj_noisy = 2.0 * n * conv_noisy.max()
            if obj_noisy < best_obj:
                a = a_noisy
                conv = conv_noisy
            else:
                # Accept noisy version temporarily to explore
                a = 0.9 * a + 0.1 * a_noisy
                conv = fast_convolve(a)

    return best_a, best_obj


def run_pgdescent(a0, time_limit, t=100.0):
    """
    Projected gradient descent with backtracking line search on log-sum-exp.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    eta = 2.0 / n
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Compute softmax weights
        scaled = t * conv / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Efficient subgradient
        sg = efficient_subgradient(w, a)
        sg_centered = sg - sg.mean()

        # Backtracking line search
        alpha = eta
        f_curr = 2.0 * n * max_c
        improved = False
        for _ in range(8):
            a_trial = project_simplex(a - alpha * sg_centered)
            conv_trial = fast_convolve(a_trial)
            f_trial = 2.0 * n * conv_trial.max()
            if f_trial < f_curr - 1e-14:
                a = a_trial
                improved = True
                break
            alpha *= 0.5

        if improved:
            eta = min(eta * 1.05, 2.0 / n * 10)
            no_improve = 0
        else:
            eta *= 0.5
            no_improve += 1

        current_obj = 2.0 * n * fast_convolve(a).max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()

        if no_improve > 50:
            # Reset with perturbation
            noise = np.random.default_rng(int(time.monotonic() * 1000)).normal(0, 0.02, n)
            a = normalize(a + noise)
            no_improve = 0
            eta = 2.0 / n

    return best_a, best_obj


def run_scipy_polish(a0, time_limit, t=200.0):
    """
    Use scipy L-BFGS-B to polish with log-sum-exp objective.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        c = fast_convolve(x)
        scaled = t * c
        scaled -= scaled.max()
        return float(np.log(np.sum(np.exp(scaled))))

    def gradient(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(len(x))
        x = x / s
        c = fast_convolve(x)
        max_c = c.max()
        if max_c < 1e-15:
            return np.zeros(len(x))
        scaled = t * c / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            return np.zeros(len(x))
        w /= w_sum
        sg = efficient_subgradient(w, x)
        return sg

    t0 = time.monotonic()
    bounds = [(0, 1)] * n
    options = {'maxiter': 200, 'maxfun': 500, 'ftol': 1e-12, 'gtol': 1e-8}

    try:
        result = minimize(objective, a.copy(), method='L-BFGS-B', jac=gradient,
                          bounds=bounds, options=options)
        if result.fun < objective(a):
            a_new = normalize(result.x)
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_obj


def get_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()

    # Random Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)

    # Anti-peak (higher at edges)
    anti = 1.0 + 0.8 * np.abs(x - n * 0.5) / (n * 0.5)
    inits['anti_peak'] = anti / anti.sum()

    # Bimodal
    bi = np.zeros(n)
 bi[:max(1, n // 4)] = 1.0
    bi[-max(1, n // 4):] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cos_w / cos_w.sum()

    # Parent-based
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

        # Noisy parent
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0)
            if pp.sum() > 1e-10:
                pp = pp / pp.sum()
                for scale in [0.05, 0.1]:
                    noise = rng.normal(0, scale, n)
                    pp_n = np.maximum(pp + noise, 0)
                    if pp_n.sum() > 1e-10:
                        inits[f'parent_noise_{scale}'] = pp_n / pp_n.sum()

    return inits
