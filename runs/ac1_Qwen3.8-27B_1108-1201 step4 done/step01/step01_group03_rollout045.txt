import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time

def fast_score(f):
    """Compute the score without calling evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_conv_matrix(n, f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(f[idx])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def solve_minimax_lp(a, time_limit=5.0):
    """
    Solve: min M s.t. (a*b)[k] <= M for all k, sum(b) = 1, b >= 0.
    Returns (b, M_opt) or (None, None).
    """
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, M -> total n+1 variables
    # Minimize: M -> c = [0,...,0, 1]
    # Constraints:
    #   sum(b) = 1 (equality)
    #   (a*b)[k] - M <= 0 for k = 0, ..., 2n-2 (inequality)
    #   b >= 0
    
    A = build_conv_matrix(n, a)  # shape (2n-1, n)
    
    # Full constraint matrix for (b, M):
    # Inequality: A @ b - M <= 0  ->  [A, -1] @ [b; M] <= 0
    # Equality: sum(b) = 1  ->  [1...1, 0] @ [b; M] = 1
    
    m = 2 * n - 1
    # Inequality constraints: [A, -ones(m,1)]
    A_ineq = sp.hstack([A, sp.csc_matrix(-np.ones((m, 1)))], format='csr')
    b_ineq = np.zeros(m)
    
    # Equality constraints
    A_eq = sp.csc_matrix(np.hstack([np.ones(n), [0.0]]))
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Bounds: b_i >= 0, M is free (but we can bound it)
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            b_opt = np.maximum(res.x[:n], 0.0)
            M_opt = res.x[n]
            # Ensure sum(b_opt) is close to 1
            s = b_opt.sum()
            if s < 1e-10:
                return None, None
            b_opt = b_opt / s
            return b_opt, M_opt
    except Exception:
        pass
    return None, None

import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time

def fast_score(f):

    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def convolve_peak_info(f):
    """Return (conv, max_val, argmax) for f*f."""
    conv = np.convolve(f, f)
    idx = int(np.argmax(conv))
    return conv, float(conv[idx]), idx

def gradient_at_active(f, active_k):
    """
    Gradient of 2*n*c_k/(sum f)^2 w.r.t. f[j], where c_k = (f*f)[k].
    We only need the direction (sign) for subgradient methods.
    Actually let's compute the full gradient for L-BFGS-B.
    """
    n = len(f)
    s = f.sum()
    # c_k = sum_i f[i]*f[i+k] for valid i
    # d(c_k)/df[j] = f[j+k] (if j+k < n) + f[j-k] (if j-k >= 0)
    dc = np.zeros(n)
    # From the "first" term: sum_i f[i]*f[i+k], derivative w.r.t. f[j]:
    # contributes f[j+k] if j+k < n
    for j in range(n):
        if j + active_k < n:
            dc[j] += f[j + active_k]
        if j - active_k >= 0:
            dc[j] += f[j - active_k]
    
    # score_k = 2*n*c_k / s^2
    # d(score_k)/df[j] = 2*n * (dc[j]*s^2 - c_k*2*s) / s^4
    #                   = 2*n * (dc[j]*s - 2*c_k) / s^3
    c_k = np.sum(f[:n-active_k] * f[active_k:]) if active_k < n else 0.0
    
    grad = 2.0 * n * (dc * s - 2.0 * c_k) / (s ** 3)
    return grad

"""
Hybrid optimization for minimizing autocorrelation peak of non-negative step functions.
Combines LP-driven minimax updates, projected gradient descent, diverse random search,
and multi-length exploration to find sequences with low 2n*max(conv)/sum^2 score.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2 quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(n, f):
    """Sparse (2n-1 x n) matrix A with A[k,j] = f[k-j] if 0<=k-j<n."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A

def solve_minimax_lp(a, time_limit=5.0):
    """
    min M s.t. (a*b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(n, a)
    m = 2 * n - 1
    # Variables: b_0..b_{n-1}, M
    # Ineq: A @ b - M <= 0  =>  [A | -1] @ [b; M] <= 0
    # Eq: sum(b) = 1  =>  [1..1 | 0] @ [b; M] = 1
    A_ineq = sp.hstack([A, sp.csc_matrix(-np.ones((m, 1)))], format='csr')
    b_ineq = np.zeros(m)
    A_eq = sp.csc_matrix(np.hstack([np.ones(n), [0.0]]).reshape(1, -1))
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            b_opt = np.maximum(res.x[:n], 0.0)
            M_opt = float(res.x[n])
            s = b_opt.sum()
            if s < 1e-15:
                return None, None
            b_opt = b_opt / s
            return b_opt, M_opt
    except Exception:
        pass
    return None, None

def solve_maxsum_lp(a, M, time_limit=5.0):
    """
    max sum(g) s.t. (a*g)[k] <= M for all k, g>=0.
    Returns g_normalized or None.
    """
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s < 1e-15:
                return None
            return g / s
    except Exception:
        pass
    return None

    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    step = step_init
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        conv = np.convolve(f, f)
        k = int(np.argmax(conv))
        n = len(f)
        # Subgradient of (f*f)[k]
        grad = np.zeros(n)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)

def random_perturb(f, rng, strength=0.1):
    """Random multiplicative perturbation, renormalized."""
    n = len(f)
    cand = f * np.exp(rng.normal(0, strength, n))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-15:
        return f.copy()
    return cand / s

def block_perturb(f, rng, max_block=None):
    """Perturb a random block multiplicatively."""
    n = len(f)
    cand = f.copy()
    width = int(rng.integers(1, max(2, (max_block or n // 10))))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0, 0.3)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-15:
        return f.copy()
    return cand / s

def make_initial(n, rng, method='uniform'):
    """Create initial sequence of length n with given structure."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'bump':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n / 2) / (n / 6)) ** 2)
    elif method == 'two_peak':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 16)
        a = (np.exp(-0.5 * ((x - n / 4) / s) ** 2) +
             np.exp(-0.5 * ((x - 3 * n / 4) / s) ** 2))
    elif method == 'alternating':
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def lp_optimize_loop(f, deadline, rng, lp_time=4.0, max_iters=2000,
                     use_minimax=True):
    """Iterative LP-based optimization loop."""
    n = len(f)
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    t = 0.3
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if use_minimax:
            g, M_opt = solve_minimax_lp(f, time_limit=lp_time)
        else:
            conv = np.convolve(f, f)
            M = float(conv.max())
            g = solve_maxsum_lp(f, M, time_limit=lp_time)
        
        if g is not None:
            f_new = (1.0 - t) * f + t * g
            f_new = project_simplex(f_new)
            sc = fast_score(f_new)
            if np.isfinite(sc) and sc < best_sc - 1e-10:
                best_sc = sc
                best = f_new.copy()
                f = f_new
                stall = 0
            elif np.isfinite(sc) and sc < fast_score(f) + 0.01:
                f = f_new
                stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        t = max(0.05, t * 0.998)
        
        # Periodic random perturbation to escape local minima
        if stall > 15:
            f = random_perturb(f, rng, strength=0.15)
            sc = fast_score(f)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = f.copy()
            stall = 0
    return best, best_sc

def polish_local(f, deadline, rng, n_trials=5000):
    """Fine-grained local search with small perturbations."""
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    for it in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Try different perturbation strategies
        strat = rng.integers(0, 4)
        if strat == 0:
            cand = random_perturb(f, rng, strength=0.02)
        elif strat == 1:
            cand = block_perturb(f, rng, max_block=max(2, n // 50))
        elif strat == 2:
            # Element-wise small change
            idx = int(rng.integers(0, n))
            cand = f.copy()
            cand[idx] = max(0, cand[idx] * float(np.exp(rng.normal(0, 0.1))))
            cand = project_simplex(cand)
        else:
            # Swap two blocks
            w = int(rng.integers(2, max(3, n // 20)))
            i = int(rng.integers(0, max(1, n - 2 * w)))
            j = int(rng.integers(i + w, max(i + w + 1, n - w)))
            cand = f.copy()
            cand[i:i+w], cand[j:j+w] = cand[j:j+w].copy(), cand[i:i+w].copy()
            cand = project_simplex(cand)
        
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-9:
            best_sc = sc
            best = cand
            f = cand
    return best, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    
    best = np.maximum(parent, 0.0).copy()
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        bs = fast_score(best)
    print("Baseline: score=%.6f, n=%d" % (bs, best.size), flush=True)
    
    def try_update(cand, sc):
        nonlocal best, bs
        if np.isfinite(sc) and sc < bs:
            bs = sc
            best = np.maximum(cand, 0.0).copy()
            print("  NEW BEST: score=%.6f, n=%d" % (bs, best.size), flush=True)
    
    # Phase 1: Quick gradient descent from diverse starts
    print("\n[Phase 1] Gradient descent from diverse initializations", flush=True)
    n_starts = deadline - time.monotonic()
    lengths = [256, 384, 512, 640, 768, 1024]
    methods = ['uniform', 'cosine', 'sin2', 'random', 'bump', 'two_peak', 'alternating']
    
    # Include parent as a starting point
    starts = [(best.copy(), 'parent')]
    gp_budget = max(30.0, min(300.0, float(budget_s) * 0.12))
    per_start = gp_budget / (len(lengths) * len(methods) + 1)
    
    for n in lengths:
        for m in methods:
            if time.monotonic() > deadline:
                break
            a0 = make_initial(n, rng, m)
            s_end = min(deadline, time.monotonic() + per_start)
            c, cs = gradient_descent_phase(a0, n_iter=300, step_init=0.008,
                                           step_decay=0.999, deadline=s_end, rng=rng)
            if np.isfinite(cs):
                try_update(c, cs)
    
    # Phase 2: LP-based optimization from best starts
    print("\n[Phase 2] LP-based optimization", flush=True)
    lp_budget = max(60.0, min(900.0, float(budget_s) * 0.45))
    
    # Create diverse LP starting points
    lp_starts = []
    # Best found so far
    lp_starts.append((best.copy(), 'best'))
    # Parent
    lp_starts.append((parent.copy(), 'parent'))
    # A few random starts at promising lengths
    for n in [best.size, 512, 1024]:
        if n not in [s.size for s, _ in lp_starts]:
            lp_starts.append((make_initial(n, rng, 'cosine'), 'cos_%d' % n))
        if n not in [s.size for s, _ in lp_starts]:
            lp_starts.append((make_initial(n, rng, 'sin2'), 'sin2_%d' % n))
    
    per_lp = lp_budget / max(1, len(lp_starts))
    per_lp = max(30.0, min(180.0, per_lp))
    
    for start_f, label in lp_starts:
        if time.monotonic() > deadline:
            break
        n = start_f.size
        if n > 2048 or n < 64:
            continue
        print("  LP: label=%s, n=%d, budget=%.0fs" % (label, n, per_lp), flush=True)
        s_end = min(deadline, time.monotonic() + per_lp)
        
        # Try both LP variants
        for use_mm in [True, False]:
            if time.monotonic() > deadline:
                break
            c, cs = lp_optimize_loop(start_f, s_end, rng, lp_time=3.0,
                                     max_iters=500, use_minimax=use_mm)
            if np.isfinite(cs):
                try_update(c, cs)
                start_f = c  # chain: use result as next start
    
    # Phase 3: More aggressive LP from current best with smaller n
    print("\n[Phase 3] Focused LP optimization on best", flush=True)
    focused_budget = max(60.0, min(400.0, float(budget_s) * 0.3))
    s_end = min(deadline, time.monotonic() + focused_budget)
    
    # Try the best with both LP types, more iterations
    for use_mm in [True, False]:
        if time.monotonic() > deadline:
            break
        c, cs = lp_optimize_loop(best.copy(), s_end, rng, lp_time=5.0,
                                 max_iters=1000, use_minimax=use_mm)
        if np.isfinite(cs):
            try_update(c, cs)
    
    # Phase 4: Polish with local search
    print("\n[Phase 4] Local polishing", flush=True)
    polish_deadline = min(deadline, time.monotonic() + max(60.0, float(budget_s) * 0.15))
    c, cs = polish_local(best.copy(), polish_deadline, rng, n_trials=20000)
    if np.isfinite(cs):
        try_update(c, cs)
    
    # Phase 5: Final gradient polish
    if time.monotonic() < deadline - 10:
        print("\n[Phase 5] Final gradient polish", flush=True)
        c, cs = gradient_descent_phase(best.copy(), n_iter=2000, step_init=0.003,
                                       step_decay=0.9999, deadline=deadline - 10,
                                       rng=rng)
        if np.isfinite(cs):
            try_update(c, cs)
    
    # Final verification
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256, dtype=np.float64)
    
    final_score = float(evaluate_sequence([float(x) for x in best]))
    print("\nFINAL: score=%.6f, n=%d" % (final_score, best.size), flush=True)
    return [float(x) for x in best]
