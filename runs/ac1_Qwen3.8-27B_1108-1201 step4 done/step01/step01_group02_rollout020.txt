def score_sequence(a):

    conv = np.convolve(a, a)
    max_b = conv.max()

    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def compute_gradient_smoothed(a, tau):
    """Compute smoothed gradient of max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    # Softmax weights
    b_max = b.max()
    p = np.exp((b - b_max) / tau)
    p /= p.sum()
    # Cross-correlation: g[i] = 2 * sum_j p[i+j] * a[j]
    N = 1 << (len(b) + n - 1).bit_length()

    aN[:n] = a
    cross = np.real(np.fft.ifft(np.conj(np.fft.fft(pN)) * np.fft.fft(aN)))
    g = 2.0 * cross[:n]
    return np.maximum(0.0, g), b

    if tau_init is None:
        tau_init = spread
    if tau_final is None:
        tau_final = spread * 0.01

    best_a = a.copy()

At each iteration, given the current `a`, the constraint `sum_j a[j] a[k-j] <= t` can be linearized as:
`sum_j (a[j]_current * (a[k-j]_new - a[k-j]_current) + a[k-j]_current * (a[j]_new - a[j]_current) + a[j]_current * a[k-j]_current) <= t`

This simplifies to:
`sum_j (a[j]_current * a[k-j]_new + a[k-j]_current * a[j]_new - a[j]_current * a[k-j]_current) <= t`

Which is linear in the new variables! This is the standard "successive linearization" approach for QPs.

But this is essentially what the LP approach in the previous program does. Let me think of something else.

**Alternative: Use the fact that the problem is "almost" convex**

The function `f(a) = max_k (a*a)[k]` is "nearly convex" in the sense that it's a maximum of quadratic forms. Each quadratic form `a^T Q_k a` is convex if `Q_k` is PSD. Let me check which `Q_k` are PSD.

`(a*a)[k] = sum_j a[j]a[k-j]`. The Hessian of this is `H_k` where:
`(H_k)_{ij} = ∂²(a*a)[k]/∂a[i]∂a[j]`

From earlier: `∂(a*a)[k]/∂a[i] = 2*a[k-i] * 1[0<=k-i<=n-1]` (for `i ≠ k-i`) or `2*a[i]` (for `2i=k`).

So `H_k[i,j] = 2 * δ(j, k-i) * 1[0<=k-i<=n-1]` (for `i ≠ j` case) ... this is getting messy.

`(a*a)[k] = sum_{j: 0<=j<=n-1, 0<=k-j<=n-1} a[j] a[k-j]`

`(M_k)_{j, k-j} = 1/2` for all valid `j` (giving `a[j]*a[k-j]` when we sum over all `j, k-j` pairs).

Actually, `(a*a)[k] = sum_j a[j] a[k-j]`. If I write this as `a^T M_k a`:
`a^T M_k a = sum_{i,j} a[i] (M_k)_{ij} a[j]`

We need `(M_k)_{ij} = 1/2` when `i + j = k` (to get `a[i]a[j] + a[j]a[i] = 2*(1/2)*a[i]*a[j] = a[i]*a[j]`)... 

Wait: `a^T M_k a = sum_i sum_j a[i] M_k[i,j] a[j]`. If `M_k[i,j] = 1/2` when `i+j=k` and 0 otherwise, then:
`a^T M_k a = sum_{i: i+k-i valid} (1/2) a[i] a[k-i] + sum_{j: j+k-j valid} (1/2) a[k-j] a[j]`

Hmm, this gives `sum_i (1/2) a[i] a[k-i]` (since the two sums are the same by substitution). That's half of what I want.

Let me try `M_k[i,j] = 1` when `i+j=k`:
`a^T M_k a = sum_{i,j: i+j=k} a[i] a[j] = sum_i a[i] a[k-i]` ✓

So `M_k` has 1s at positions `(i, k-i)` for valid `i`. This is a permutation-like matrix (a "reversal and shift"). It's symmetric: `M_k[i,j] = M_k[j,i]` since `i+j=k` iff `j+i=k`. ✓

Is `M_k` PSD? `M_k` is a permutation matrix (maps `e_i` to `e_{k-i}`). The eigenvalues of a permutation matrix are roots of unity, so they have magnitude 1. Thus `M_k` has both positive and negative eigenvalues (unless it's the identity or has special structure). So `(a*a)[k]` is NOT a convex function of `a` in general.

For `k = 0`: `M_0` maps `e_i` to `e_{-i}`, which is only valid for `i=0`. So `M_0 = e_0 e_0^T` (just the (0,0) entry is 1). This is PSD. ✓
For `k = n-1`: `M_{n-1}[i,j] = 1` iff `i+j=n-1`. This is the "anti-identity" (reversal) matrix. Its eigenvalues are all ±1. So it's NOT PSD (it has negative eigenvalues). ✗

So the objective is genuinely non-convex. The LP approach and mirror descent are both reasonable for handling this.

"""
Smoothed Mirror Descent + Local Search + LP Polish for minimizing peak auto-convolution.
Key insight: symmetric sequences are local minima (uniform is optimal among symmetric),
so we use asymmetric initializations and fast FFT-based gradient computation.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = b.max()
    p = np.exp((b - b_max) / tau)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[b.argmax()] = 1.0
    else:
        p /= p_sum
    # Cross-correlation: g[i] = 2 * sum_j p[i+j] * a[j]
    N = 1 << (len(b) + n - 1).bit_length()
    pN = np.zeros(N)
    pN[:len(b)] = p
    aN = np.zeros(N)
    aN[:n] = a
    cross = np.real(np.fft.ifft(np.conj(np.fft.fft(pN)) * np.fft.fft(aN)))
    g = 2.0 * cross[:n]
    return np.maximum(0.0, g), b


def mirror_descent_refine(a, time_limit_s=300.0, verbose=True):
    """Smoothed mirror descent on the probability simplex."""
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    b0 = np.convolve(a, a)
    spread = max(1e-6, b0.max() - b0.min())
    
    tau_init = spread
    tau_final = spread * 0.05
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = score_sequence(a)
    
    step_scale = 1.5  # Multiplier for step size
    stagnant = 0
    
    it = 0
    while time.monotonic() - start < time_limit_s:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_limit_s)
        tau = tau_init * (tau_final / tau_init) ** frac
        
        g, b = compute_smoothed_gradient(a, tau)
        max_g = g.max()
        if max_g < 1e-15:
            break
        
        # Mirror descent step
        step = step_scale / max_g
        a_new = a * np.exp(-step * g / max_g)
        a_new = normalize_nonneg(a_new)
        
        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            a = a_new
            if verbose and (it % 100 == 0 or it < 10):
                print(f"      MD iter {it:5d}: score={score:.6f}, tau={tau:.6f}, elapsed={elapsed:.1f}s")
        else:
            stagnant += 1
            if stagnant > 20:
                step_scale *= 0.7
                stagnant = 0
                if step_scale < 0.01:
                    break
        
        it += 1
    
    # Try a few more steps with the best found
    if it > 0:
        final_score = score_sequence(a)
        if final_score < best_score:
            best_score = final_score
            best_a = a.copy()
    
    if verbose:
        print(f"      MD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start:.1f}s")
    
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    accept_count = 0
    
    while time.monotonic() - start < time_limit_s:

        pos = int(rng.integers(0, max(1, n - width)))

        candidate = normalize_nonneg(candidate)

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_refine(a, time_limit_s=120.0, verbose=True):
    """LP-based refinement (from previous approach, used for final polish)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 5.0:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            if verbose and (it % 10 == 0 or it < 3):
                print(f"      LP iter {it:3d}: score={score:.6f}")
        
        M = conv.max()
        timeout = max(2.0, min(30.0, remaining * 0.3))
        
        # Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    alpha = min(0.5, 1.0 / (1.0 + 0.1 * it))
                    a_new = (1.0 - alpha) * a + alpha * g
                    a_new = normalize_nonneg(a_new)
                    new_score = score_sequence(a_new)
                    if new_score <= score + 1e-14:
                        a = a_new
                    else:
                        alpha2 = alpha * 0.3
                        a_new = (1.0 - alpha2) * a + alpha2 * g
                        a_new = normalize_nonneg(a_new)
                        a = a_new
                else:
                    if verbose:
                        print(f"      LP converged at iter {it}")
                    break
            else:
                break
        except Exception:
            break
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"      LP done: best={best_score:.6f} after {it} iters")
    return best_a, best_score
