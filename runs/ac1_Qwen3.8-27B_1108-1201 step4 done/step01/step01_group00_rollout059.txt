Optimization of non-negative step functions to minimize normalized autocorrelation peak.
Uses L-BFGS gradient optimization with annealed soft-max, LP-based direction finding,
and multi-length multi-init search strategy.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (same formula as evaluate_sequence)."""

    """Stable softmax."""
    x_shift = x - x.max()
    e = np.exp(x_shift)
    total = e.sum()
    if total == 0:
        return np.ones_like(x) / len(x)
    return e / total

"""
Optimization of non-negative step functions to minimize normalized autocorrelation peak.
Uses L-BFGS gradient optimization with annealed soft-max, LP-based direction finding
(state-of-the-art approach), and multi-length multi-init search with iterative refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def softmax_normalize(x):
    """Stable softmax returning probabilities summing to 1."""
    x_shift = x - x.max()
    e = np.exp(x_shift)
    total = e.sum()
    if total < 1e-300:
        return np.ones_like(x) / len(x)
    return e / total


def a_from_softmax(x, n):
    """Convert softmax parameters to sequence with sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    return softmax_normalize(x) * S


def objective_and_grad(x, n, tau):
    """
    Compute smooth objective and gradient for softmax parameterization.
    Returns (smooth_max_value, gradient_wrt_x).
    """
    S = np.sqrt(2.0 * n)
    p = softmax_normalize(x)
    a = p * S

    c = fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    smooth_val = c_max + tau * np.log(w_sum)

    # Gradient of smooth_max(conv(a,a)) w.r.t. a:
    # grad_a[j] = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
    grad_a = np.zeros(n)
    for j in range(n):
        grad_a[j] = 2.0 * np.dot(a, w[j:j + n])

    # Chain rule through softmax: x -> p -> a
    inner = np.dot(a, grad_a)
    grad_x = a * grad_a - p * inner

    return smooth_val, grad_x


def run_lbfgs(n, x_init, time_limit, n_phases=25):
    """Run L-BFGS with annealed temperature schedule."""
    t0 = time.monotonic()
    best_x = x_init.copy()
    best_score = compute_score(a_from_softmax(x_init, n))

    tau_start = 50.0 / n
    tau_end = 0.3 / n

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        frac = phase / max(1, n_phases - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        def obj_grad(x):
            val, g = objective_and_grad(x, n, tau)
            return val, g

        try:
            result = minimize(obj_grad, best_x, method='L-BFGS-B',
                              jac=True, options={'maxiter': 300, 'ftol': 1e-14})
            if result.fun < best_score:
                a_check = a_from_softmax(result.x, n)
                actual_score = compute_score(a_check)
                if actual_score < best_score:
                    best_score = actual_score
                    best_x = result.x.copy()
        except Exception:
            pass

    return best_x, best_score

            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k, amp in [(1, 0.25), (2, 0.15), (4, 0.1), (7, 0.07), (11, 0.04)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining gradient optimization, LP-based polishing,
    and local search to minimize the normalized autocorrelation peak.
    """
    t_global = time.monotonic()
    time_budget = max(60.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Gradient optimization at multiple lengths
    print("\n=== Phase 1: L-BFGS Gradient Optimization ===")
    lengths_to_try = [128, 256, 384, 512]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 180:
            break
        time_left = deadline - time.monotonic()
        n_inits = 2 if n > 256 else 3
        time_per_init = min(25.0, time_left / (n_inits + 1.0))

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 90:
                break

            if init_i == 0 and start_seq is not None:
                a0 = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(start_seq)), start_seq)
            elif init_i == 1:
                a0 = make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng)
            else:
                a0 = make_multi_tone_init(n, rng)

            a0 = np.clip(a0, 0.01, 1000.0)
            x_init = np.log(a0 / a0.sum() + 1e-10)

            t0 = time.monotonic()
            x_best, grad_score = run_lbfgs(n, x_init, time_per_init, n_phases=20)
            elapsed = time.monotonic() - t0
            a_best = a_from_softmax(x_best, n)
            actual_score = compute_score(a_best)
            print(f"  n={n}, init={init_i}, score={actual_score:.6f} ({elapsed:.1f}s)")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: LP polishing (state-of-the-art approach)
    print(f"\n=== Phase 2: LP Polishing (SoTA) ===")
    if best_seq is not None:
        polish_time = min(400.0, max(60.0, deadline - time.monotonic() - 90))
        print(f"  Starting SoTA LP polish, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, polish_score = lp_polish_sota(best_seq, polish_time, max_iter=1000)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After SoTA LP polish: {best_score:.6f}")
        else:
            print(f"  SoTA LP polish: {polish_score:.6f} (no improvement)")

    # Phase 3: Alternative LP (minimize peak of cross-correlation)
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: Min-Peak LP Polishing ===")
        alt_time = min(200.0, max(30.0, deadline - time.monotonic() - 60))
        print(f"  Starting min-peak LP, score={best_score:.6f}, time={alt_time:.1f}s")
        polished2, polish2_score = lp_polish_minpeak(best_seq, alt_time, max_iter=300)
        if polish2_score < best_score:
            best_score = polish2_score
            best_seq = polished2.copy()
            print(f"  After min-peak LP: {best_score:.6f}")
        else:
            print(f"  Min-peak LP: {polish2_score:.6f} (no improvement)")

    # Phase 4: Second round of gradient optimization on polished result
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Refinement Gradient ===")
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a_norm = best_seq * (S / best_seq.sum())
        x_init = np.log(a_norm / S + 1e-10)
        refine_time = min(120.0, max(20.0, deadline - time.monotonic() - 40))
        x_best, grad_score = run_lbfgs(n, x_init, refine_time, n_phases=15)
        a_best = a_from_softmax(x_best, n)
        # Rescale to match original sum
        a_best = a_best * (best_seq.sum() / a_best.sum())
        actual_score = compute_score(a_best)
        print(f"  Refinement: score={actual_score:.6f}")
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Local search fine-tuning
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Local Search ===")
        ls_time = min(90.0, max(10.0, deadline - time.monotonic() - 15))
        result, ls_score = local_search(best_seq, ls_time, rng)
        if ls_score < best_score:
            best_score = ls_score
            best_seq = result.copy()
            print(f"  After local search: {best_score:.6f}")
        else:
            print(f"  Local search: {ls_score:.6f} (no improvement)")

    # Phase 6: Final LP polish if time remains
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            print(f"\n=== Phase 6: Final LP Polish ===")
            final_polished, final_score = lp_polish_sota(best_seq, remaining, max_iter=200)
            if final_score < best_score:
                best_score = final_score
                best_seq = final_polished.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")
        return final_list

    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
