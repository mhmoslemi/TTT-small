"""
Optimize non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative weights (mirror descent) with soft subgradients computed via FFT,
LP-based direction finding for periodic corrections, and coordinate descent polishing.
Explores multiple sequence lengths and initializations within a 1800s budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_of(a):
    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def score_normalized(a):
    """Score assuming sum(a)=1 (faster, avoids sum computation)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def soft_subgradient(c, a, tau):
    """
    Compute soft subgradient of max(conv(a,a)) using softmax weights.
    g[j] = 2 * sum_k w[k] * a[k-j] where w is softmax(c/tau).
    Computed as 2 * conv(w, a_rev)[n-1 : 2n-1].
    """
    n = len(a)
    c_shifted = c / tau
    c_shifted -= np.max(c_shifted)
    w = np.exp(c_shifted)
    w /= np.sum(w)
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return g


def hard_subgradient(c, a):
    """Compute hard subgradient at the argmax of c."""
    n = len(a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(k_star, n - 1)
    for j in range(j_lo, j_hi + 1):
        g[j] = 2.0 * a[k_star - j]
    return g


def multiplicative_weights_step(a, g, lr, floor=1e-12):
    """Apply multiplicative weights update and renormalize."""
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, floor)
    a_new = a_new / np.sum(a_new)
    return a_new


def lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum=1) or None on failure.
    """
    n = len(f)
    c_obj = -np.ones(n)
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.full(n_constraints, M)

    for k in range(n_constraints):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A_ub[k, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]

    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            g = result.x
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def coordinate_polish(a, time_limit, rng, n_coords_per_iter=10):
    """
    Coordinate descent polish: for each coordinate, try scaling up/down
    and renormalizing, accepting if score improves.
    """
    n = len(a)
    a = a / np.sum(a)
    best = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    npot = n_coords_per_iter
    npot = min(npot, max(1, n // 4))

    while time.monotonic() - t0 < time_limit:
        idxs = rng.choice(n, size=min(npot, n), replace=False)
        for j in idxs:
            if time.monotonic() - t0 > time_limit:
                break
            base_score = best_score
            # Try several multiplicative factors
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                candidate = best.copy()
                old_val = candidate[j]
                candidate[j] = old_val * factor
                if candidate[j] < 1e-12:
                    candidate[j] = 1e-12
                candidate = candidate / np.sum(candidate)
                s = score_normalized(candidate)
                if s < base_score:
                    best = candidate
                    best_score = s
                    base_score = s
    return best, best_score


def optimize_single(n, a_init, time_limit, rng, use_lp=True):
    """
    Optimize a single sequence using multiplicative weights with soft subgradients.
    Periodically uses LP direction for correction.
    """
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    best = a.copy()
    best_score = score_normalized(a)

    t0 = time.monotonic()
    lr = 2.0
    lr_decay = 0.9995
    tau = 0.1 / n  # Initial temperature for soft subgradient
    tau_decay = 0.999
    step = 0
    last_lp_step = 0
    lp_interval = 200
    plateau = 0
    plateau_best = best_score

    n_evals = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        max_c = np.max(c)

        # Alternate between soft and hard subgradients
        if step % 3 == 0:
            g = hard_subgradient(c, a)
        else:
            g = soft_subgradient(c, a, tau)

        # Multiplicative weights step
        a_new = multiplicative_weights_step(a, g, lr)

        # Periodic LP correction
        if use_lp and step - last_lp_step >= lp_interval:
            M = max_c + 1e-6
            g_lp = lp_direction(a, M)
            if g_lp is not None:
                # Blend with LP direction
                a_blended = 0.7 * a_new + 0.3 * g_lp
                a_blended = np.maximum(a_blended, 1e-10)
                a_blended = a_blended / np.sum(a_blended)
                s_blend = score_normalized(a_blended)
                if s_blend < score_normalized(a_new):
                    a_new = a_blended
            last_lp_step = step

        new_score = score_normalized(a_new)
        a = a_new
        step += 1
        n_evals += 1

        if new_score < best_score:
            best_score = new_score
            best = a.copy()
            plateau = 0
        else:
            plateau += 1
            if plateau % 500 == 0 and plateau > 2000:
                # Perturb to escape local minimum
                a = a * np.exp(0.02 * rng.standard_normal(n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                plateau = 0

        # Anneal
        lr *= lr_decay
        tau *= tau_decay

        # Adaptive: if stuck, increase temperature temporarily
        if plateau > 1000:
            tau = max(tau, 0.5 / n)
            lr = max(lr, 1.0)

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    n={n}, step={step}, score={new_score:.6f}, best={best_score:.6f}, lr={lr:.3f}, tau={tau:.2e}, t={elapsed:.0f}s")

    return best, best_score


def make_init(n, init_type, rng, provided_seq=None):
    """Generate an initial sequence of length n."""
    t = np.arange(n, dtype=np.float64)
    x = np.linspace(0, 1, n)

    if init_type == 'provided' and provided_seq is not None:
        if len(provided_seq) == n:
            a = provided_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(provided_seq))
            a = np.interp(x, x_old, provided_seq)
    elif init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'tapered':
        a = 0.3 + 0.7 * np.sin(np.pi * x)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for k, amp in [(1, 0.2), (3, 0.12), (5, 0.08), (7, 0.05), (11, 0.03)]:
            a += amp * np.cos(2.0 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'gaussian_taper':
        center = n / 2.0
        sig = n / 4.0
        a = np.exp(-0.5 * ((t - center) / sig) ** 2)
        a += 0.05
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)
