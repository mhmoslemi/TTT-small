def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(a, size)
    return np.real(ifft(A * A))[:2 * n - 1]


def logsumexp(b):
    """Numerically stable log-sum-exp."""
    m = np.max(b)
    if m > 100:
        return m + np.log(np.sum(np.exp(b - m)))
    elif m < -100:
        return m + np.log(np.sum(np.exp(b - m)))
    else:
        return np.log(np.sum(np.exp(b)))


def normalize_log(b, S):
    """Normalize log-space parameter so that sum(exp(b)) = S."""
    m = np.max(b)
    log_sum = m + np.log(np.sum(np.exp(b - m)))
    return b + (np.log(S) - log_sum)


def compute_smooth_gradient(a, c, temperature):
    """Compute gradient of smooth max t*log(sum(exp(c/t))) w.r.t. log(a) in log-space.

    W = fft(w, size)

    h = np.real(ifft(W * np.conj(A)))[:n]

        c = fft_convolve_self(a)
        cur = c.max()

        if cur < best_val - 1e-10:
            best_val = cur
            best_b = b.copy()
            last_improve = t

        # Adaptive learning rate: reduce if no improvement for a while
        if t - last_improve > patience:
            lr *= lr_decay_factor
            last_improve = t

        # Temperature schedule
        temp = max(t_final, t_init * np.exp(-t_decay * t))

        # Compute gradient
        if use_smooth:
            grad, _ = compute_smooth_gradient(a, c, temp)
        else:
            grad = compute_topk_subgradient(a, c, top_k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-25:
            break
        grad = grad / gnorm

        # Adam update
        m_v = beta1 * m_v + (1.0 - beta1) * grad
        v_v = beta2 * v_v + (1.0 - beta2) * (grad * grad)
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)

        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        b = normalize_log(b, S)

        # Stochastic exploration
        if t % noise_interval == 0 and rng.random() < noise_prob:
            b = b + rng.normal(0.0, noise_scale, n)
            b = normalize_log(b, S)

    return best_b, best_val


def init_uniform(n, rng, perturb=0.0):
    """Uniform initialization with optional random perturbation."""
    b = np.zeros(n, dtype=np.float64)
    if perturb > 0:
        b += rng.normal(0, perturb, n)
    return b


def init_sinusoidal(n, freq1, amp1, freq2, amp2):
    """Sinusoidal modulation initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + amp1 * np.sin(2 * np.pi * freq1 * t) + amp2 * np.cos(2 * np.pi * freq2 * t)
    a = np.maximum(a, 0.01)
    return np.log(a)


def init_two_scale(n, rng, base_freq, noise_level):
    """Two-scale structure: low-frequency modulation + high-frequency noise."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * base_freq * t)
    a += noise_level * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return np.log(a)


def init_alternating(n, ratio):
    """Alternating high/low pattern."""
    a = np.ones(n, dtype=np.float64)
    a[::2] = ratio
    a = np.maximum(a, 0.01)
    return np.log(a)


def init_random(n, rng, alpha=1.0):
    """Random Dirichlet-like initialization."""
    a = rng.exponential(1.0, n)
    a = np.maximum(a, 0.01)
    return np.log(a)


def init_from_given(given_seq, n_target, rng):
    """Resample or use a given sequence as initialization."""
    a = np.asarray(given_seq, dtype=np.float64).flatten()
    a = np.clip(a, 0.01, 1000.0)
    if len(a) == n_target:
        return np.log(a)
    elif len(a) > n_target:
        # Downsample
        indices = np.linspace(0, len(a) - 1, n_target).astype(int)
        a = a[indices]
        return np.log(np.maximum(a, 0.01))
    else:
        # Upsample via interpolation
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, n_target)
        a_new = np.interp(x_new, x_old, a)
        return np.log(np.maximum(a_new, 0.01))


def evaluate_normalized(a):
    """Evaluate the normalized objective (sum(a) = sqrt(2n) assumed)."""
    n = len(a)
    c = fft_convolve_self(a)
    return c.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(1.0, float(budget_s) - 10.0)

    print("Starting optimized search...")

    # Try multiple sequence lengths
    lengths = [8192, 4096, 2048]

    # Try to use given sequence if available
    given_seq = None
    try:
        given_seq = height_sequence_1
        print(f"Given sequence length: {len(given_seq)}")
    except (NameError, Exception):
        pass

    global_best = np.inf
    global_a = None
    global_n = 0

    # Phase 1: Exploration
    explore_budget = 0.65 * (deadline - time.monotonic())
    print(f"Exploration budget: {explore_budget:.1f}s")

    # Define initializations to try
    inits = []

    # From given sequence
    if given_seq is not None:
        for n_try in lengths:
            inits.append(("given_%d" % n_try, init_from_given(given_seq, n_try, rng)))

    # Uniform with perturbation
    for n_try in lengths:
        inits.append(("uniform_%d" % n_try, init_uniform(n_try, rng, perturb=0.3)))

    # Sinusoidal
    for n_try in (8192, 4096):
        inits.append(("sin1_%d" % n_try, init_sinusoidal(n_try, 3, 0.4, 7, 0.2)))
        inits.append(("sin2_%d" % n_try, init_sinusoidal(n_try, 1, 0.6, 3, 0.3)))
        inits.append(("sin3_%d" % n_try, init_sinusoidal(n_try, 5, 0.3, 11, 0.3)))

    # Two-scale
    for n_try in (8192, 4096):
        inits.append(("two_%d" % n_try, init_two_scale(n_try, rng, 2, 0.1)))

    # Alternating
    for n_try in (8192, 4096):
        inits.append(("alt_%d" % n_try, init_alternating(n_try, 0.3)))

    # Random
    for n_try in (8192, 4096):
        inits.append(("rand_%d" % n_try, init_random(n_try, rng)))

    n_inits = len(inits)
    per_init_budget = explore_budget / n_inits
    print(f"Number of initializations: {n_inits}, per-init budget: {per_init_budget:.1f}s")

    results = []

    for idx, (name, b_init) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - start_total - explore_budget) - 5.0:
            break

        n = len(b_init)
        S = np.sqrt(2.0 * float(n))
        b_norm = normalize_log(b_init, S)
        a_check = np.exp(b_norm)
        c_check = fft_convolve_self(a_check)
        init_val = c_check.max()

        # Run optimizer with smooth max
        t0 = time.monotonic()
        # Temperature: start hot, cool to near-deterministic
        t_init_val = max(0.3, 0.5 * init_val)
        t_final_val = 0.01
        t_decay_val = 5.0 / max(1.0, (time.monotonic() - start_total) / 100.0 + 1.0)
        # Use a fixed decay rate
        t_decay_val = 0.008

        b_best, val_best = run_optimize_log(
            b_init, n, S,
            max_time=max(1.0, per_init_budget * 0.6),
            rng=rng,
            lr_init=0.3 * S / n,
            t_init=t_init_val,
            t_final=t_final_val,
            t_decay=t_decay_val,
            use_smooth=True,
            noise_interval=300,
            noise_prob=0.15,
            noise_scale=0.05
        )
        elapsed = time.monotonic() - t0

        # Also try subgradient method on the result
        remaining = max(0.0, per_init_budget - elapsed - 1.0)
        if remaining > 2.0:
            b_best2, val_best2 = run_optimize_log(
                b_best, n, S,
                max_time=remaining,
                rng=rng,
                lr_init=0.1 * S / n,
                t_init=0.1,
                t_final=0.01,
                t_decay=0.01,
                use_smooth=False,
                top_k=16,
                noise_interval=500,
                noise_prob=0.05,
                noise_scale=0.03
            )
            if val_best2 < val_best:
                b_best, val_best = b_best2, val_best2

        a_best = np.exp(b_best)
        # Verify
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            global_n = n

        results.append((name, n, val_best, ver))
        print(f"  [{idx+1}/{n_inits}] {name} n={n} opt_val={val_best:.6f} verified={ver:.6f} "
              f"best={global_best:.6f} time={time.monotonic()-start_total:.0f}s")

    # Phase 2: Polish the best
    polish_budget = deadline - time.monotonic()
    print(f"\nPolish budget: {polish_budget:.1f}s. Best so far: {global_best:.6f}")

    if global_a is not None and polish_budget > 5.0:
        n = global_n
        S = np.sqrt(2.0 * float(n))
        b_current = np.log(np.maximum(global_a, 1e-15))
        b_current = normalize_log(b_current, S)

        # Multiple polishing rounds with decreasing LR
        rounds = 4
        round_budget = polish_budget / rounds

        for r in range(rounds):
            lr = 0.1 * S / n / (r + 1)
            t_i = max(0.05, 0.5 * global_best / (r + 2))
            b_current, val = run_optimize_log(
                b_current, n, S,
                max_time=round_budget,
                rng=rng,
                lr_init=lr,
                t_init=t_i,
                t_final=0.005,
                t_decay=0.005,
                use_smooth=True,
                noise_interval=200,
                noise_prob=0.03 if r > 0 else 0.08,
                noise_scale=0.02
            )
            a_current = np.exp(b_current)
            ver = evaluate_sequence(a_current.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_current.copy()
            print(f"  Polish round {r+1}: val={val:.6f} verified={ver:.6f} best={global_best:.6f}")

        # Final ultra-fine polish with top-k subgradient
        if time.monotonic() < deadline - 3.0:
            remaining = deadline - time.monotonic() - 3.0
            if remaining > 2.0:
                b_current, val = run_optimize_log(
                    b_current, n, S,
                    max_time=remaining,
                    rng=rng,
                    lr_init=0.01 * S / n,
                    t_init=0.01,
                    t_final=0.005,
                    t_decay=0.002,
                    use_smooth=False,
                    top_k=32,
                    noise_interval=1000,
                    noise_prob=0.01,
                    noise_scale=0.01
                )
                a_final = np.exp(b_current)
                ver = evaluate_sequence(a_final.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_final.copy()
                print(f"  Final subgradient polish: val={val:.6f} verified={ver:.6f} best={global_best:.6f}")

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_ver = evaluate_sequence(global_a.tolist())
        print(f"\nFinal result: {final_ver:.6f}")
    else:
        global_a = np.ones(256, dtype=np.float64)
        print("Fallback to ones. Final:", evaluate_sequence(global_a.tolist()))

    print(f"Total time: {time.monotonic() - start_total:.1f}s")
    return [float(x) for x in global_a]
