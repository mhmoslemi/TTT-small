"""
Convex optimization approach to minimize the autocorrelation peak of a
non-negative step function. Uses FISTA (accelerated projected gradient) on
the convex min-max problem, combined with local search polishing and
multi-start exploration across different sequence lengths.

The key insight: minimizing max_k (f*f)[k] subject to sum(f)=1, f>=0 is a
convex problem (max of quadratic forms). We use accelerated projected
gradient with subgradients from the top-k active constraints.

import numpy as np
from scipy.fft import fft, ifft
import time


def linear_conv(f):
    """Compute linear convolution f*f efficiently."""
    n = len(f)
    if n <= 768:
        return np.convolve(f, f)
    size = 2 * n - 1
    pad = 1
    while pad < size:
        pad *= 2
    F = fft(f, pad)
    return np.real(ifft(F * F))[:size]


def compute_score(f):
    """Compute the evaluation score (scale-invariant)."""
    f = np.maximum(np.asarray(f, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = linear_conv(f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def project_simplex(v, target):
    """Project v onto {x >= 0, sum(x) = target} using water-filling."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1].copy()
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (target - cv) / rho) > 0
    if not np.any(cond):
        return np.full(n, target / n)
    idx = np.where(cond)[0]
    rho_max = idx[-1] + 1
    theta = (target - cv[rho_max - 1]) / rho_max
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    s = result.sum()
    if s > 0:
        result *= target / s
    return result


def subgradient_topk(conv, f, top_k=5):
    """Compute subgradient of max_k conv[k] using top-k active constraints."""
    n = len(f)
    num_constraints = len(conv)
    top_k = min(top_k, num_constraints)
    idx = np.argsort(conv)[::-1][:top_k]
    grad = np.zeros(n)
    for k in idx:
        # d conv[k] / df[m] = 2*f[k-m] if 0 <= k-m < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            m_range = np.arange(lo, hi + 1)
            kj_range = k - m_range
            grad[m_range] += 2.0 * f[kj_range]
    grad /= top_k
    return grad


def fista_step(f, f_old, t, alpha, conv, top_k):
    """One FISTA iteration step."""
    y = f + ((t - 1) / (t + 2)) * (f - f_old)
    grad = subgradient_topk(conv, y, top_k)
    f_new = project_simplex(y - alpha * grad, 1.0)
    return f_new


def optimize_fista(f0, budget_s, top_k=5, alpha_init=None, restart_interval=500):
    """Run FISTA to minimize max(conv) with sum(f)=1, f>=0."""
    n = len(f0)
    f = f0.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    f = project_simplex(f, 1.0)

    f_old = f.copy()
    t = 1.0
    best_f = f.copy()
    best_score = compute_score(f)

    if alpha_init is None:
        alpha_init = 0.5 / max(f.max(), 1e-6)

    alpha = alpha_init
    deadline = time.monotonic() + budget_s
    it = 0
    stall_count = 0
    patience = 200

    while time.monotonic() < deadline:
        conv = linear_conv(f)
        f_new = fista_step(f, f_old, t, alpha, conv, top_k)
        f_old = f
        f = f_new
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        it += 1

        if it % 10 == 0 or it < 10:
            sc = compute_score(f)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > patience:
            # Restart with perturbation
            f = best_f + np.random.normal(0, 0.005, n)
            f = project_simplex(np.maximum(f, 0), 1.0)
            f_old = f.copy()
            t = 1.0
            alpha = alpha_init * 0.8
            stall_count = 0
            patience = max(50, patience // 2)

        if it % restart_interval == 0:
            # Gradual alpha decay
            alpha *= 0.95
            if alpha < 1e-6:
                alpha = alpha_init * 0.5

    return best_f, best_score


def coordinate_descent(f, max_iters=2000, tol=1e-10):
    """Simple coordinate descent to polish the solution."""
    n = len(f)
    f = f.copy().astype(np.float64)
    best = f.copy()
    best_sc = compute_score(f)
    rng = np.random.default_rng(12345)
    order = rng.permutation(n)
    it = 0
    for _iter in range(max_iters):
        if time.monotonic() > deadline_global:
            break
        for i in order:
            orig = f[i]
            # Try a few perturbations
            for delta in [0.5 * orig, -0.5 * orig, 0.1 * orig, -0.1 * orig,
                         2.0 * orig, 0.0, 0.01, -0.01]:
                f[i] = max(0.0, orig + delta)
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                sc = 2.0 * n * linear_conv(f).max() / (s * s)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best = f.copy()
                else:
                    f[i] = orig
                    break
            it += 1
            if it > 50000:
                return best, best_sc
    return best, best_sc


def random_polish(f, budget_s, rng):
    """Random block perturbation polish."""
    n = len(f)
    best = f.copy()
    best_sc = compute_score(f)
    deadline = time.monotonic() + budget_s
    it = 0
    while time.monotonic() < deadline:
        it += 1
        cand = best.copy()
        w = rng.integers(1, max(2, n // 10))
        start = rng.integers(0, n - w + 1)
        scale = np.exp(rng.normal(0, 0.1))
        cand[start:start + w] *= scale
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 0.01:
            continue
        sc = 2.0 * n * linear_conv(cand).max() / (s * s)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand
            print("   polish: %.6f" % best_sc, flush=True)
        if it % 500 == 0:
            print("   polish it=%d best=%.6f" % (it, best_sc), flush=True)
    return best, best_sc


def make_init_cosine(n, rng, n_harmonics=3):
    """Create a cosine-series initialization."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1) * 2 * np.pi
    f = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(-0.3, 0.3) / k
        f += amp * np.cos(k * x)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f


def make_init_peak(n, rng, center_weight=0.3):
    """Create a peaked initialization (high in center, low at edges)."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1)
    f = 1.0 + center_weight * np.cos(np.pi * (2 * x - 1))
    f = np.maximum(f, 0.01)
    f += rng.normal(0, 0.05, n)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f


def make_init_random(n, rng):
    """Random positive initialization."""
    f = rng.random(n) + 0.1
    f = f / f.sum()
    return f


deadline_global = 0.0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the best non-negative sequence minimizing the autocorrelation peak."""
    global deadline_global
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline_global = start_time + max(1.0, float(budget_s) - 10.0)

    # Load parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(512)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(512)

    parent = parent / parent.sum()
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print("Parent score = %.6f (n=%d)" % (best_score, best.size), flush=True)

    def try_update(f, sc):
        nonlocal best, best_score
        if np.isfinite(sc) and sc < best_score:
            # Verify with actual evaluator
            sc_check = float(evaluate_sequence(np.maximum(f, 0).tolist()))
            if np.isfinite(sc_check) and sc_check < best_score:
                best = np.maximum(f, 0).copy()
                best_score = sc_check
                print("  NEW BEST: %.6f (n=%d) at t=%.0fs" % (best_score, best.size,
                      time.monotonic() - start_time), flush=True)

    # Phase 1: FISTA on parent sequence (50% of budget)
    phase1_budget = 0.45 * float(budget_s)
    print("\n=== Phase 1: FISTA on parent (budget=%.0fs) ===" % phase1_budget, flush=True)

    for top_k in [3, 5, 8, 10]:
        if time.monotonic() > deadline_global - 60:
            break
        sub_budget = min(phase1_budget / 4.0, max(10.0, (deadline_global - time.monotonic())))
        f_opt, sc_opt = optimize_fista(parent, sub_budget, top_k=top_k,
                                       alpha_init=0.3 / max(parent.max(), 1e-6))
        if np.isfinite(sc_opt):
            try_update(f_opt, sc_opt)
        parent = best.copy()  # Use best found as new starting point

    # Phase 2: Try different initializations and lengths (30% of budget)
    phase2_budget = 0.30 * float(budget_s)
    print("\n=== Phase 2: Multi-start (budget=%.0fs) ===" % phase2_budget, flush=True)

    lengths_to_try = [best.size]
    if best.size != 512:
        lengths_to_try.append(512)
    for extra in [768, 1024]:
        if extra not in lengths_to_try:
            lengths_to_try.append(extra)

    inits = [
        ('cosine3', lambda n: make_init_cosine(n, rng, 3)),
        ('cosine5', lambda n: make_init_cosine(n, rng, 5)),
        ('peak', lambda n: make_init_peak(n, rng)),
        ('random', lambda n: make_init_random(n, rng)),
    ]

    n_combos = len(lengths_to_try) * len(inits)
    per_combo_budget = max(5.0, phase2_budget / n_combos)

    for n in lengths_to_try:
        for init_name, init_fn in inits:
            if time.monotonic() > deadline_global - 30:
                break
            if time.monotonic() + per_combo_budget > deadline_global:
                per_combo_budget = max(3.0, deadline_global - time.monotonic() - 30)
            f_init = init_fn(n)
            print("  Trying n=%d, init=%s, budget=%.0fs" % (n, init_name, per_combo_budget),
                  flush=True)
            f_opt, sc_opt = optimize_fista(f_init, per_combo_budget, top_k=5,
                                           alpha_init=0.3 / max(f_init.max(), 1e-6))
            if np.isfinite(sc_opt):
                try_update(f_opt, sc_opt)

    # Phase 3: Fine polishing (20% of budget)
    phase3_budget = 0.20 * float(budget_s)
    print("\n=== Phase 3: Polishing (budget=%.0fs) ===" % phase3_budget, flush=True)

    # Coordinate descent polish
    if time.monotonic() < deadline_global - 30:
        cd_budget = min(300.0, phase3_budget * 0.5)
        print("  Coordinate descent polishing...")
        f_cd, sc_cd = coordinate_descent(best, max_iters=100000)
        if np.isfinite(sc_cd):
            try_update(f_cd, sc_cd)

    # Random polish
    if time.monotonic() < deadline_global - 10:
        rp_budget = max(10.0, deadline_global - time.monotonic() - 10)
        rng2 = np.random.default_rng(seed + 999)
        f_rp, sc_rp = random_polish(best, rp_budget, rng2)
        if np.isfinite(sc_rp):
            try_update(f_rp, sc_rp)

    # Final verification and return
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512)
    final_score = float(evaluate_sequence(best.tolist()))
    print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (final_score, best.size,
          time.monotonic() - start_time), flush=True)
    return [float(x) for x in best]
