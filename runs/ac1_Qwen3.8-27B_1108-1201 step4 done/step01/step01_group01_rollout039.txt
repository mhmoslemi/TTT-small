import numpy as np
import time

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    return 2.0 * n * np.max(b) / (sum_a * sum_a)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _make_ramp(n):
    """Linear ramp: a[i] = 2*(n-1-i)/(n*(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = (n - 1 - i)  # n-1, n-2, ..., 0
    S = np.sum(a)
    return a / S

def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n//2] = 1.0 + eps
    a[n//2:] = 1.0 - eps
    a = np.maximum(a, 0)
    S = np.sum(a)
    return a / S

def _make_smooth_anti(n, eps, freq=1):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2 * np.pi * freq * i / n + np.pi)  # anti-symmetric
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _make_sigmoid(n, sharpness):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(sharpness * (i - n/2) / n))

    return a / S

    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)

        a /= S
    else:
        a = np.ones(n) / n
    
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    
    best = a.copy()
    best_score = _compute_score(best)
    eta = eta_init
    no_improve = 0
    t = t_smooth
    
    while time.monotonic() < deadline:

        b_max = np.max(b)
        cur_score = 2.0 * n * b_max / (np.sum(a)**2)

            best = a.copy()
            no_improve = 0
        else:

            no_improve = 0
            if eta < 1e-12:

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses linear ramp (theoretical score ~4/3) as primary start, with subgradient
descent using smooth max approximation, multiple diverse starts, and
adaptive step-size with periodic reheating.
"""

import numpy as np
import time
from scipy.signal import fftconvolve


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    b = fftconvolve(a, a)
    sum_a = np.sum(a)
    return float(2.0 * n * np.max(b) / (sum_a * sum_a))


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_ramp(n):
    """Linear ramp: a[i] = (n-1-i), normalized."""
    a = (n - 1.0 - np.arange(n, dtype=np.float64))
    S = np.sum(a)
    return a / S


def _make_power_law(n, alpha):
    """Power-law decay: a[i] = (n-1-i+1)^alpha, normalized."""
    a = (n - np.arange(n, dtype=np.float64)) ** alpha
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_two_level(n, eps):
    """Two-level anti-symmetric."""
    a = np.ones(n, dtype=np.float64)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_anti(n, eps, freq=1):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_sigmoid(n, sharpness):
    """Sigmoid-like decreasing profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(sharpness * (2.0 * i / n - 1.0)))
    S = np.sum(a)
    return a / S


def _compute_gradient(a, b, top_idx, wts):
    """Compute weighted subgradient of max(c) w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for ki in range(len(top_idx)):
        w = wts[ki]
        if w < 1e-12:
            continue
        k = int(top_idx[ki])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # grad[j] += 2*w*a[k-j] for j in [lo, hi]
            # a[k-j] for j=lo..hi => a[k-lo] down to a[k-hi]
            grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]
    return grad


def _compute_gradient_fft(a, b, t_smooth):
    """Compute gradient using softmax weights over ALL peaks via FFT."""
    n = len(a)
    # Softmax weights
    b_shifted = b - np.max(b)
    wts = np.exp(b_shifted / t_smooth)
    wts_sum = np.sum(wts)
    if wts_sum < 1e-300:
        # Fallback: uniform over top peaks
        top = np.argsort(b)[-10:]
        wts = np.zeros_like(b)
        wts[top] = 1.0 / 10.0
    else:
        wts /= wts_sum

    # grad[j] = 2 * sum_k wts[k] * a[k-j] for valid k-j
    # = 2 * (wts cross-correlated with a)[j]
    # cross[j] = sum_k wts[k] * a[k-j] where 0<=k-j<n
    # Let's compute: cross[j] = sum_{l} wts[j+l] * a[l] where l=k-j, so k=j+l
    # valid: 0<=j+l<len(wts) and 0<=l<n
    # This is a convolution-like operation

    # Use FFT: cross[j] = sum_l a[l] * wts[j+l]
    # Define wts_ext of length 2n-1 (it already has that length)
    # cross[j] = sum_{l=0}^{n-1} a[l] * wts[j+l] (with wts out of bounds = 0)
    # This is the cross-correlation of a and wts at lag j
    # In FFT terms: ifft(fft(a, 2n) * conj(fft(wts, 2n))) gives correlation
    # But we need a specific convention...

    # Simpler: just do it directly for the top peaks
    # For efficiency, use top M peaks
    n_b = len(b)
    M = min(200, n_b)
    top_idx = np.argpartition(b, -M)[-M:]
    b_top = b[top_idx]
    wts_top = np.exp((b_top - np.max(b)) / t_smooth)
    wts_top_sum = np.sum(wts_top)
    if wts_top_sum < 1e-300:
        wts_top = np.ones(M) / M
    else:
        wts_top /= wts_top_sum

    grad = np.zeros(n)
    for ki in range(M):
        w = wts_top[ki]
        if w < 1e-10:
            continue
        k = int(top_idx[ki])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]
    return grad


def _optimize(a, deadline, rng, t_init=0.05, eta_init=None, M_peaks=100,
              cool_patience=300, restart_patience=800):
    """
    Subgradient optimization with smooth max approximation.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 0.03 / np.sqrt(n)

    best = a.copy()
    best_score = _compute_score(best)
    eta = eta_init
    t = t_init
    no_improve = 0
    total_iters = 0
    last_improve_time = time.monotonic()

    while time.monotonic() < deadline:
        b = fftconvolve(a, a)
        b_max = np.max(b)
        sum_a = np.sum(a)
        cur_score = 2.0 * n * b_max / (sum_a * sum_a)

        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            last_improve_time = time.monotonic()
        else:
            no_improve += 1

        total_iters += 1

        # Adaptive: cool down if stuck
        if no_improve >= cool_patience:
            eta *= 0.7
            t = max(t * 0.6, 0.001)
            no_improve = 0
            if eta < 1e-14:
                break

        # If stuck for very long, restart with perturbation
        if no_improve >= restart_patience:
            a = _project_simplex(best + 0.05 * rng.standard_normal(n), 1.0)
            eta = eta_init * 0.5
            t = t_init
            no_improve = 0

        # Compute gradient using top-M peaks with softmax weights
        n_b = len(b)
        M = min(M_peaks, n_b)
        top_idx = np.argpartition(b, -M)[-M:]
        b_top = b[top_idx]
        b_shifted = b_top - np.max(b_top)
        wts = np.exp(b_shifted / t)
        wts_sum = np.sum(wts)
        if wts_sum < 1e-300:
            wts = np.ones(M) / M
        else:
            wts /= wts_sum

        # Compute gradient
        grad = np.zeros(n)
        for ki in range(M):
            w = wts[ki]
            if w < 1e-10:
                continue
            k = int(top_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

        # Center gradient for sum constraint
        grad -= np.mean(grad)

        # Normalized update
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a = a - eta * grad / np.sqrt(n)
        a = _project_simplex(a, 1.0)

        # Safety check
        if not np.all(np.isfinite(a)):
            a = best.copy()
            break

    return best, best_score


def _optimize_mirror(a, deadline, rng, t_init=0.05, M_peaks=100, cool_patience=200):
    """
    Mirror descent with entropy (multiplicative weight update).
    a_new ∝ a * exp(-eta * grad)
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    eta = 0.02 / np.sqrt(n)
    best = a.copy()
    best_score = _compute_score(best)
    t = t_init
    no_improve = 0

    while time.monotonic() < deadline:
        b = fftconvolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max / (np.sum(a) ** 2)

        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= cool_patience:
            eta *= 0.7
            t = max(t * 0.6, 0.001)
            no_improve = 0
            if eta < 1e-14:
                break

        # Compute gradient
        n_b = len(b)
        M = min(M_peaks, n_b)
        top_idx = np.argpartition(b, -M)[-M:]
        b_top = b[top_idx]
        b_shifted = b_top - np.max(b_top)
        wts = np.exp(b_shifted / t)
        wts_sum = np.sum(wts)
        if wts_sum < 1e-300:
            wts = np.ones(M) / M
        else:
            wts /= wts_sum

        grad = np.zeros(n)
        for ki in range(M):
            w = wts[ki]
            if w < 1e-10:
                continue
            k = int(top_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

        # Center gradient
        grad -= np.mean(grad)

        # Mirror descent update (multiplicative)
        # a_new ∝ a * exp(-eta * grad)
        log_a = np.log(a + 1e-30)
        log_a -= eta * grad
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new /= np.sum(a_new)

        if not np.all(np.isfinite(a_new)):
            break
        a = a_new

    return best, best_score


def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    # Low-frequency smooth noise
    n_freqs = min(5, n // 4)
    noise = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    return _project_simplex(a + scale * noise, 1.0)
