import time

def _next_pow2(n):
    m = 1
    while m < n:
        m *= 2
    return m

def _full_conv(a):
    n = len(a)
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2*n-1]

def _conv_grad(a, w):
    n = len(a)
    L = 2 * n - 1
    m = _next_pow2(n + L - 1)
    w_flip = w[::-1]
    fa = np.fft.rfft(a, m)
    fw = np.fft.rfft(w_flip, m)
    conv = np.fft.irfft(fa * fw, m)
    return conv[n-1:2*n-1][::-1].copy()

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _optimize(n, a_init, deadline, tau_init=50.0, eta_init=None, verbose=False):
    """
    Projected gradient descent with momentum and log-sum-exp smoothing.
    Minimizes 2n * max(r) where r = conv(a,a), subject to a >= 0, sum(a) = 1.
    """

        eta_init = 0.1 / np.sqrt(n)
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 1e-15:
        a = a / S
    else:
        a = np.ones(n) / n
    
    # Add small positive floor to avoid log issues
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    
    tau = tau_init
    eta = eta_init
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_full_conv(a))
    
    no_improve = 0
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    
    it = 0
    last_time = time.monotonic()
    
    while time.monotonic() < deadline:
        it += 1
        
        # Check time
        if it % 100 == 0:
            elapsed = time.monotonic() - last_time
            last_time = time.monotonic()
            if verbose and it % 1000 == 0:
                print(f"    it={it}, score={2*n*np.max(_full_conv(a)):.6f}, best={best_score:.6f}, tau={tau:.1f}, eta={eta:.2e}, iter_time={elapsed/100*1000:.1f}ms")
        
        # Forward
        r = _full_conv(a)
        r_max = np.max(r)
        cur_score = 2.0 * n * r_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive tau: increase to sharpen the max approximation
        if no_improve > 200:
            tau = min(tau * 1.5, 50000.0)
            no_improve = 0
            # Also reset momentum
            v = np.zeros(n)

        tau_r = tau * (r - r_max)
        tau_r = np.clip(tau_r, -100, 0)
        weights = np.exp(tau_r)
        w_sum = np.sum(weights)
        if w_sum < 1e-30:

            weights = np.zeros_like(r)
            weights[peak_idx] = 1.0

def _conv_grad(a, w):
    """
    Compute grad[j] = sum_{j'=0}^{n-1} a[j'] * w[j+j'] for j=0,...,n-1.
    w has length 2n-1 (the smoothing weights for each convolution coefficient).
    """
    n = len(a)
    L = 2 * n - 1
    m = _next_pow2(n + L - 1)
    w_flip = w[::-1]
    fa = np.fft.rfft(a, m)
    fw = np.fft.rfft(w_flip, m)
    conv = np.fft.irfft(fa * fw, m)
    return conv[n - 1:2 * n - 1][::-1].copy()


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_smooth(n, eps, power=1.0):
    """Smooth anti-symmetric: 1 + eps * ((2i/n) - 1)^power"""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0  # from -1 to 1
    a = 1.0 + eps * np.sign(x) * np.abs(x) ** power
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_cosine(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

            grad_la = _conv_grad(a_lookahead, w_la)
            grad_la -= np.mean(grad_la)
            v = beta * v_prev + grad_la
            a = a - (eta / (np.linalg.norm(v) + 1e-30)) * v
        else:
            a = a_lookahead

        # Project onto simplex
        a = _project_simplex(a, 1.0)

        if np.sum(a) < 1e-10:
            a = best_a.copy()
            break

        # Adaptive step size
        if no_improve > 300:
            eta *= 0.7
            v = np.zeros(n)
            v_prev = np.zeros(n)
            no_improve = 0
        elif no_improve == 0 and it > 200:
            eta = min(eta * 1.02, eta_start * 10)

        # Increase momentum as we converge
        if no_improve > 100:
            beta = min(beta + 0.001, 0.995)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    it={it}, score={cur_score:.7f}, best={best_score:.7f}, "
                  f"tau={tau:.0f}, eta={eta:.3e}, time={elapsed:.1f}s")

    return best_a, best_score


def _make_two_level(n, eps):
    """Anti-symmetric two-level: first half high, second half low."""
    a = np.ones(n)
    half = n // 2
    a[:half] = 1.0 + eps
    a[half:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_smooth_ramp(n, eps, power=1.0):
    """Smooth anti-symmetric ramp: 1 + eps * sign(x) * |x|^power where x = 2i/n - 1."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / max(1, n - 1) - 1.0
    a = 1.0 + eps * np.sign(x) * np.abs(x) ** power
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_cosine(n, eps):
    """Cosine anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_quarter(n, eps):
    """Four-segment profile: high, mid-high, mid-low, low."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps * 0.3
    a[2 * q:3 * q] = 1.0 - eps * 0.3
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _smooth_noise(n, rng, n_modes=8):
    """Generate smooth random noise using low-frequency Fourier components."""
    i = np.arange(n, dtype=np.float64)
    noise = np.zeros(n)
    max_mode = max(2, n // 10)
    modes = rng.choice(range(1, max_mode), size=min(n_modes, max_mode - 1), replace=False)
    for m in modes:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.sin(2 * np.pi * m * i / n + phase)
    max_abs = np.max(np.abs(noise))
    if max_abs > 1e-15:
        noise /= max_abs
    return noise


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase FFT-based gradient descent to minimize the normalized
    autocorrelation peak of a non-negative sequence of fixed length.
    
    Phase 1: Explore with diverse structural initializations.
    Phase 2: Exploit the best with increasing tau (sharpening max approximation).
    Phase 3: Polish with very fine steps and diversification via perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    global_best = parent.copy()
    S = np.sum(global_best)
    if S > 1e-15:
        global_best = global_best / S
    else:
        global_best = np.ones(n) / n
    global_best_score = 2.0 * n * np.max(_full_conv(global_best))
    print(f"n={n}, initial score={global_best_score:.6f}, budget={budget_s-15:.0f}s")

    # ===== Phase 1: Explore (40% of time) =====
    print("\n=== Phase 1: Explore ===")
    phase1_time = 0.40 * (deadline - time.monotonic())
    phase1_deadline = time.monotonic() + phase1_time

    initializations = [
        ("parent", global_best.copy(), 80.0, 800.0, 0.04 / np.sqrt(n)),
        ("parent_pert", _project_simplex(global_best + 0.05 * (rng.random(n) - 0.5), 1.0), 50.0, 500.0, 0.06 / np.sqrt(n)),
        ("two_level_0.3", _make_two_level(n, 0.3), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("two_level_0.5", _make_two_level(n, 0.5), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("ramp_lin_0.5", _make_smooth_ramp(n, 0.5, power=1.0), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("ramp_quad_0.5", _make_smooth_ramp(n, 0.5, power=2.0), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("cosine_0.5", _make_cosine(n, 0.5), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("cosine_0.8", _make_cosine(n, 0.8), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("quarter_0.5", _make_quarter(n, 0.5), 30.0, 300.0, 0.1 / np.sqrt(n)),
        ("smooth_noise", _project_simplex(global_best + 0.1 * _smooth_noise(n, rng), 1.0), 50.0, 500.0, 0.08 / np.sqrt(n)),
    ]

    n_init = len(initializations)
    per_init = phase1_time / n_init

    for idx, (name, a_init, ts, te, es) in enumerate(initializations):
        if time.monotonic() >= phase1_deadline:
            break
        sub_dl = min(time.monotonic() + per_init, phase1_deadline)
        print(f"  [{idx+1}/{n_init}] {name} (tau {ts}->{te})")
        best_a, best_s = _run_optimizer(n, a_init, sub_dl, ts, te, es, beta=0.9)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"    *** NEW BEST: {global_best_score:.6f} ***")
        else:
            print(f"    score={best_s:.6f} (best={global_best_score:.6f})")

    # ===== Phase 2: Exploit (35% of time) =====
    print(f"\n=== Phase 2: Exploit (best={global_best_score:.6f}) ===")
    phase2_time = 0.35 * (deadline - time.monotonic())
    phase2_deadline = time.monotonic() + phase2_time

    # Multiple optimization runs with different tau schedules and momentum
    schedules = [
        (200.0, 5000.0, 0.015 / np.sqrt(n), 0.93),
        (500.0, 20000.0, 0.008 / np.sqrt(n), 0.95),
        (100.0, 2000.0, 0.02 / np.sqrt(n), 0.97),
        (1000.0, 50000.0, 0.004 / np.sqrt(n), 0.97),
        (50.0, 1000.0, 0.025 / np.sqrt(n), 0.90),
    ]

    per_sched = phase2_time / len(schedules)

    for idx, (ts, te, es, bm) in enumerate(schedules):
        if time.monotonic() >= phase2_deadline:
            break
        sub_dl = min(time.monotonic() + per_sched, phase2_deadline)
        print(f"  [sched {idx+1}/{len(schedules)}] tau {ts:.0f}->{te:.0f}, eta={es:.2e}, beta={bm}")
        best_a, best_s = _run_optimizer(n, global_best.copy(), sub_dl, ts, te, es, beta=bm)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"    *** NEW BEST: {global_best_score:.6f} ***")
        else:
            print(f"    score={best_s:.6f}")

    # ===== Phase 3: Polish and Diversify (remaining time) =====
    print(f"\n=== Phase 3: Polish (best={global_best_score:.6f}) ===")

    # First: very fine polishing of the best
    polish_time = min(60.0, 0.3 * (deadline - time.monotonic()))
    if polish_time > 5:
        polish_dl = time.monotonic() + polish_time
        print(f"  Fine polish ({polish_time:.0f}s):")
        best_a, best_s = _run_optimizer(
            n, global_best.copy(), polish_dl,
            tau_start=5000.0, tau_end=200000.0,
            eta_start=0.002 / np.sqrt(n), beta=0.99
        )
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"    *** NEW BEST: {global_best_score:.6f} ***")

    # Diversification: perturb and re-optimize
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"  Diversification ({remaining:.0f}s remaining):")
        n_pert = max(1, int(remaining / 40))
        per_pert = remaining / n_pert

        for p in range(n_pert):
            if time.monotonic() >= deadline:
                break
            # Decreasing perturbation scale
            scale = 0.03 * (1.0 - 0.5 * p / max(1, n_pert))
            noise = _smooth_noise(n, rng, n_modes=5)
            a_pert = _project_simplex(global_best + scale * noise, 1.0)
            pert_dl = min(time.monotonic() + per_pert, deadline)

            # Vary tau schedule for diversity
            ts_p = 100.0 * (1 + 0.5 * (p % 3))
            te_p = 2000.0 * (1 + 0.5 * (p % 3))
            eta_p = 0.01 / np.sqrt(n)

            best_a_p, best_s_p = _run_optimizer(n, a_pert, pert_dl, ts_p, te_p, eta_p, beta=0.93)
            if best_s_p < global_best_score:
                global_best = best_a_p.copy()
                global_best_score = best_s_p
                print(f"    Perturb {p+1}: *** NEW BEST: {global_best_score:.6f} ***")
            else:
                print(f"    Perturb {p+1}: score={best_s_p:.6f}")

    # Final high-precision polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"  Final polish ({remaining:.0f}s):")
        final_dl = deadline
        best_a, best_s = _run_optimizer(
            n, global_best.copy(), final_dl,
            tau_start=10000.0, tau_end=500000.0,
            eta_start=0.001 / np.sqrt(n), beta=0.995
        )
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"    *** NEW BEST: {global_best_score:.6f} ***")

    # Result
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    return result
