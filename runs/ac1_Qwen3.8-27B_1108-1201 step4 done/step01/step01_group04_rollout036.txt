"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multiplicative weight updates (mirror descent) with near-peak tracking,
LP-based refinement, spectral initialization, and random perturbation restarts.
Explores multiple sequence lengths within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum (default sqrt(2n))."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def compute_pressure(a, active_mask):
    """Compute pressure at each position from active peaks."""
    n = len(a)
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        return np.ones(n)
    pressure = np.zeros(n)
    for k in active_indices:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            pressure[i_lo:i_hi + 1] += a[k - i_lo:k - i_hi - 1: -1]
    pressure /= float(len(active_indices))
    return pressure


def mirror_descent_step(a, lr, K_frac=0.15, K_min=3):
    """One multiplicative weight update step using near-peak pressure."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    min_c = c.min()
    range_c = max_c - min_c
    if range_c < 1e-15:
        return a

    # Select active peaks: those within K_frac of the max
    thresh = max_c - K_frac * range_c
    active_mask = c >= thresh
    # Ensure minimum number of active peaks
    if active_mask.sum() < K_min:
        top_k = np.argsort(c)[-K_min:]
        active_mask = np.zeros(len(c), dtype=bool)
        active_mask[top_k] = True

    pressure = compute_pressure(a, active_mask)

    # Multiplicative update with centering
    p_mean = pressure.mean()
    pos_mask = a > 1e-15
    log_a = np.full(n, np.log(1e-12))
    log_a[pos_mask] = np.log(a[pos_mask])

    log_a -= lr * (pressure - p_mean)

    # Clip for numerical stability
    log_a = np.clip(log_a, -30, 30)
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new)
    return a_new


def grad_step(a, lr):
    """One projected gradient step on the peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize(a)
    return a


def lp_refine(f, M, S_target):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0. Rescale to sum=S_target."""
    n = len(f)
    if n > 1024:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x
            S_g = g.sum()
            if S_g > 1e-10:
                g = g * (S_target / S_g)
                return g
    except Exception:
        pass
    return None


def optimize_sequence(a0, deadline, use_mirror=True, use_grad=False, use_lp=True):
    """
    Optimize a single starting sequence using a combination of methods.
    Returns (best_seq, best_score).
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:

    a = normalize(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    # Phase 1: Primary optimization (mirror descent or gradient)

        lr = 0.4
        K_frac = 0.1
        for it in range(8000):
            if time.monotonic() > deadline - 40.0:

            a = mirror_descent_step(a, lr, K_frac=K_frac, K_min=3)

                if s < best_score:
                    best_score = s
                    best = a.copy()
                # Slowly decay learning rate

                # Periodically increase K_frac to smooth out

    elif use_grad:
        for it in range(8000):
            if time.monotonic() > deadline - 40.0:
                break
            lr = 0.1 * 0.5 * (1.0 + np.cos(np.pi * it / 8000))

                    best = a.copy()

        c = np.convolve(a, a)

                    best = a.copy()
                    best_score = s_new

                    break

                    best_score = s

        lr = 0.15

            a = mirror_descent_step(a, lr, K_frac=0.05, K_min=2)

                s = compute_obj(a)
                if s < best_score:
                    best_score = s
                    best = a.copy()
            lr = max(0.01, lr * 0.999)

    return best, best_score


def spectral_initialize(n, rng):
    """Create a sequence with roughly flat power spectrum using random phases."""
    # Start with random non-negative sequence
    a = rng.random(n) * 0.5 + 0.25
    S_target = np.sqrt(2.0 * n)
    a = normalize(a, S_target)

    # Iterate: modify phases in frequency domain to flatten |A|^2
    N = 2 * n - 1
    for it in range(20):
        A = np.fft.rfft(a, N)
        # Compute |A|^2 and try to flatten it
        mag = np.abs(A)
        # Target: flat magnitude (except DC)
        mag_target = np.ones(N // 2 + 1)
        mag_target[0] = mag[0]  # Keep DC
        # Blend current and target
        mag_new = 0.5 * mag + 0.5 * mag_target * (A[0] / (A[0] + 1e-10))
        # Reconstruct with original phases
        phases = np.angle(A)
        A_new = mag_new * np.exp(1j * phases)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_target)
        a = a_new
    return a


def generate_starts(n, rng):
    """Generate diverse starting sequences of length n."""
    starts = []

    # Uniform
    starts.append(np.ones(n) * np.sqrt(2.0 / n))

    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)

    # Spectral
    starts.append(spectral_initialize(n, rng))

    # Chirp (real part of complex chirp, made non-negative)
    i = np.arange(n, dtype=np.float64)
    chirp = np.cos(2 * np.pi * i * (i + 1) / (2.0 * n))
    starts.append(np.abs(chirp) + 0.3)

    # Raised cosine window
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))

    # Two-frequency
    starts.append(np.abs(np.cos(2 * np.pi * i * 3.0 / n) + 0.5 * np.cos(2 * np.pi * i * 7.0 / n)) + 0.2)

    # Exponential decay from center
    center = n / 2.0
    starts.append(np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2) + 0.1)

    # Bimodal
    c1 = n * 0.25
    c2 = n * 0.75
    starts.append(np.exp(-((i - c1) / (n * 0.15)) ** 2) + np.exp(-((i - c2) / (n * 0.15)) ** 2) + 0.1)

    # Alternating (high-low pattern with noise)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.4)
    starts.append(alt * (0.8 + 0.4 * rng.random(n)))

    return starts
