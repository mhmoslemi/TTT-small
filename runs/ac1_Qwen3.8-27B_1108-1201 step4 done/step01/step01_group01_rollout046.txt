        obj = b_max
    else:

            grad[idx] = 2.0 * a[k_max - idx]
    else:
        sum_bp = np.sum(b**p)
        if sum_bp > 0:
            scale = (sum_bp) ** (1.0/p - 1.0)
        else:
            scale = 1.0
        grad = 2.0 * scale * g
    
    return obj, grad

"""
Multi-stage L_p-norm optimization with FFT-accelerated gradients, Adam optimizer,
and top-k subgradient polishing. Exploits the structure of the non-negative
autocorrelation peak minimization problem using continuation (smooth→sharp),
diverse initializations, and adaptive time allocation.

import numpy as np
from scipy.fft import fft, ifft
import time


def _fft_convolve_aa(a):
    """Compute a*a (full linear convolution) using FFT. Returns real array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = fft(a, fsize)
    b = np.real(ifft(fa * fa, fsize))[:size]
    np.maximum(b, 0, out=b)
    return b


def _gradient_lp(a, b, p):
    """
    Compute gradient of (sum b_k^p)^(1/p) w.r.t. a, where b = conv(a,a).
    Uses FFT to compute the cross-correlation efficiently.
    """
    n = len(a)
    if p == np.inf:
        k_max = int(np.argmax(b))
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * a[k_max - idx]
        b_max = b[k_max]
        obj = b_max
        return obj, grad

    c = b ** (p - 1)
    # g[i] = sum_j a[j] * c[i+j]  (cross-correlation)
    # = conv(a[::-1], c)[n-1 : 2n-1]
    a_rev = a[::-1]
    total_len = n + len(c) - 1
    fsize = 1 << (total_len - 1).bit_length()
    fa_rev = fft(a_rev, fsize)
    fc = fft(c, fsize)
    g_full = np.real(ifft(fa_rev * fc, fsize))
    g = g_full[n - 1:2 * n - 1]

    sum_bp = np.sum(b ** p)
    if sum_bp > 1e-30:
        obj = sum_bp ** (1.0 / p)
        scale = sum_bp ** (1.0 / p - 1.0)
    else:
        obj = 0.0
        scale = 0.0
    grad = 2.0 * scale * g
    return obj, grad


def _topk_gradient(a, b, k):
    """
    Compute gradient of sum of top-k b values w.r.t. a.
    Returns (topk_sum, gradient).
    """
    n = len(a)
    if k >= len(b):
        # Sum of all b = (sum a)^2, gradient is 2*sum(a)*ones... but we normalize so just return
        k = len(b)
    k_idx = np.argpartition(b, -k)[-k:]
    k_sorted = k_idx[np.argsort(b[k_idx])[::-1]]

    # Weight by exponential for smoothness
    b_vals = b[k_sorted]
    spread = b_vals[0] - b_vals[-1]
    if spread > 1e-15:
        wts = np.exp(-3.0 * (b_vals[0] - b_vals) / spread)
    else:
        wts = np.ones(k)
    wts /= np.sum(wts)

    grad = np.zeros(n)
    for ki in range(k):
        w = wts[ki]
        if w < 1e-10:
            continue
        kk = int(k_sorted[ki])
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[kk - idx]

    topk_sum = np.sum(b[k_sorted])
    return topk_sum, grad


def _normalize(a):
    """Normalize a to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _adam_step(params, grad, lr, beta1, beta2, eps, m, v, t):
    """Adam optimizer step. Modifies m, v in place."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v


def _make_ramp(n, eps):
    """Linear ramp profile: (1-eps) to (1+eps)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + eps * t
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_quadratic(n, eps):
    """Quadratic profile: 1 + eps*(2t-1)^2 - eps/3 (centered)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + eps * (t ** 2 - 1.0 / 3.0)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_two_level(n, eps):
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_smooth_cos(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_exponential_taper(n, alpha):
    """Exponential taper: exp(alpha * (2i/(n-1) - 1))."""
    t = np.linspace(-1, 1, n)
    a = np.exp(alpha * t)
    return _normalize(a)


def _make_gaussian_offset(n, center_frac, width):
    """Gaussian bump offset from center."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    a = np.exp(-0.5 * ((i - center) / width) ** 2) + 0.01
    return _normalize(a)


def _optimize_adam(a_init, deadline, p_sequence, lr_init, k_top, rng):
    """
    Multi-stage Adam optimization:
    1. L_p stages with increasing p
    2. Top-k subgradient stage
    Returns best a and score.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve_aa(a))

    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init

    # Stage allocation: give more time to later (sharper) stages
    n_stages = len(p_sequence)
    # Time weights: earlier stages get less time
    weights = np.arange(1, n_stages + 1) ** 1.5
    weights /= np.sum(weights)

    elapsed_start = time.monotonic()
    total_budget = max(1.0, deadline - elapsed_start)

    for stage, p in enumerate(p_sequence):
        if time.monotonic() >= deadline:
            break

        stage_budget = total_budget * weights[stage]
        stage_deadline = min(time.monotonic() + stage_budget, deadline)

        # Reset Adam state at each stage boundary
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = lr_init

        iterations = 0
        patience = 2000
        no_improve = 0
        best_stage_score = best_score
        best_stage_a = best_a.copy()

        while time.monotonic() < stage_deadline:
            b = _fft_convolve_aa(a)
            cur_score = 2.0 * n * np.max(b)

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if cur_score < best_stage_score - 1e-12:
                best_stage_score = cur_score
                best_stage_a = a.copy()

            if no_improve >= patience:
                # Reheat
                lr *= 0.5
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                if lr < 1e-10:
                    break

            # Compute gradient
            if p == np.inf:
                obj, grad = _topk_gradient(a, b, k_top)
            else:
                obj, grad = _gradient_lp(a, b, p)

            # Center gradient (for sum constraint approximation)
            grad -= np.mean(grad)

            # Adam step
            t += 1
            m, v = _adam_step(a, grad, lr, beta1, beta2, eps, m, v, t)

            # Project to non-negative and normalize
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s < 1e-30:
                a = best_a.copy()
                break
            a /= s

            iterations += 1

            if iterations % 5000 == 0:
                print(f"  Stage p={p}, iter={iterations}, score={cur_score:.8f}, lr={lr:.2e}")

    return best_a, best_score


def _subgradient_polish(a_init, deadline, rng, k_top=10):
    """
    Polished subgradient descent on the true max objective.
    Uses adaptive step size and exponential weighting over top-k peaks.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve_aa(a))

    eta = 0.05 / np.sqrt(n)
    min_eta = 1e-10
    no_improve = 0
    reheat_thresh = 500

    while time.monotonic() < deadline:
        b = _fft_convolve_aa(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= reheat_thresh:
            eta *= 0.85
            no_improve = 0
            if eta < min_eta:
                break

        # Get top k peaks with exponential weighting
        if k_top >= len(b):
            kk = len(b)
        else:
            kk = k_top

        k_idx = np.argpartition(b, -kk)[-kk:]
        k_sorted = k_idx[np.argsort(b[k_idx])[::-1]]
        b_vals = b[k_sorted]
        spread = b_vals[0] - b_vals[-1]
        if spread > 1e-15:
            wts = np.exp(-4.0 * (b_vals[0] - b_vals) / spread)
        else:
            wts = np.ones(kk)
        wts /= np.sum(wts)

        grad = np.zeros(n)
        for ki in range(kk):
            w = wts[ki]
            if w < 1e-10:
                continue
            k = int(k_sorted[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += w * 2.0 * a[k - idx]

        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        a = a - eta * grad / gn
        a = _project_simplex(a, 1.0)
        s = np.sum(a)
        if s < 1e-10:
            a = best_a.copy()
            break

    return best_a, best_score


def _perturb(a, rng, scale=0.05):
    """Add smooth low-frequency perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 3), size=min(8, n // 2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    return _normalize(a_new)

    # Process starts
    for i, (name, a_start) in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        if i == 0:
            t_alloc = time_parent
        else:
            t_alloc = time_each_rest

        start_deadline = min(time.monotonic() + t_alloc, deadline)

        if i > 0 and t_alloc < 5:
            continue

        print(f"\nStart '{name}' (t_alloc={t_alloc:.1f}s):")
        best_a, best_s = _optimize_adam(a_start, start_deadline, p_sequence, lr_init, k_top, rng)
        print(f"  Adam stage done: score={best_s:.8f}")

        # Subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 10 and best_s < global_best_score:
            polish_deadline = min(time.monotonic() + min(remaining * 0.3, 120), deadline)
            best_a2, best_s2 = _subgradient_polish(best_a, polish_deadline, rng, k_top=12)
            if best_s2 < best_s:
                best_a, best_s = best_a2, best_s2
                print(f"  Polish done: score={best_s:.8f}")

        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  ★ NEW GLOBAL BEST: {global_best_score:.8f}")
        else:
            print(f"  No improvement (best={global_best_score:.8f})")

    # Final extended polishing on the global best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nExtended polishing with {remaining:.1f}s remaining:")
        # Multiple rounds of perturbation + re-optimization
        n_rounds = max(1, int(remaining / 60))
        per_round = remaining / n_rounds

        for r in range(n_rounds):
            if time.monotonic() >= deadline:
                break

            # Try: direct polish
            round_deadline = min(time.monotonic() + per_round * 0.6, deadline)
            a_pol, s_pol = _subgradient_polish(global_best, round_deadline, rng, k_top=15)
            if s_pol < global_best_score:
                global_best = a_pol.copy()
                global_best_score = s_pol
                print(f"  Round {r+1} polish: {global_best_score:.8f}")

            # Try: perturb + Adam re-optimize
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_pert = _perturb(global_best, rng, scale=0.03)
                pert_deadline = min(time.monotonic() + per_round * 0.4, deadline)
                a_re, s_re = _optimize_adam(a_pert, pert_deadline, [16, 32, np.inf], lr_init * 0.5, k_top, rng)
                if s_re < global_best_score:
                    global_best = a_re.copy()
                    global_best_score = s_re
                    print(f"  Round {r+1} perturb: {global_best_score:.8f}")

    # Try different lengths if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nTrying different lengths with {remaining:.1f}s remaining:")
        for trial_n in [n + 100, n - 100, n + 500, 8000, 4000]:
            if time.monotonic() >= deadline:
                break
            if trial_n <= 0:
                continue
            trial_deadline = min(time.monotonic() + min(remaining / 5, 120), deadline)

            # Use ramp as a diverse start for new lengths
            a_trial = _make_ramp(trial_n, 0.6)
            best_a_t, best_s_t = _optimize_adam(a_trial, trial_deadline, [4, 8, 16, 32, np.inf],
                                                 0.02 / np.sqrt(trial_n), k_top, rng)
            # Note: score for different n is directly comparable
            if best_s_t < global_best_score:
                global_best = best_a_t.copy()
                global_best_score = best_s_t
                n = trial_n
                print(f"  n={trial_n}: ★ NEW BEST {global_best_score:.8f}")
            else:
                print(f"  n={trial_n}: score={best_s_t:.8f}")

    # Final verification
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.8f} (n={len(result)})")
    return result
