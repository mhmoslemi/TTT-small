```python
import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def convolve_full(p):
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def score(p):
    n = len(p)
    s = np.sum(p)
    if s < 0.01:
        return np.inf
    c = convolve_full(p)
    return 2.0 * n * np.max(c) / (s*s)

def lp_best_response(p, M):
    n = len(p)
    # Build constraint matrix: (p*g)[k] <= M for k = 0, ..., 2n-2
    # (p*g)[k] = sum_i p[i] * g[k-i] where 0<=i<n and 0<=k-i<n
    # So for each k, the valid i range is max(0, k-n+1) to min(k, n-1)
    
    rows = []
    cols = []
    data = []
    
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i  # this is the index of g
            rows.append(k)
            cols.append(j)
            data.append(p[i])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x, 0)
        return g
    return None
```

- min max_k (p*g)[k] s.t. sum(g) = 1, g >= 0

This is a linear program too (with an epigraph variable). It directly minimizes the peak of the cross-convolution. This might be a better formulation.

Actually, the first formulation (max sum(g) s.t. (p*g) <= M) is what's used in the state-of-the-art description. Let me stick with that.

Let me also think about convergence. Each LP step gives a "best response" g to the current p. If we set p = g (full update), we're doing best-response dynamics, which might oscillate. The blend (p_new = (1-t)*p + t*g) stabilizes this.

Alternatively, I could use a "mirror descent" type update, but the LP + blend is simpler and proven.

```python
"""
Iterated LP best-response search for low-peak autocorrelation step functions.
Uses Linear Programming to find the best-response direction at each step,
combined with line search and gradient-based polishing.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def full_convolve(p):
    """Full self-convolution using FFT."""
    n = len(p)
    if n == 0:
        return np.array([])
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def compute_score(p):
    """Compute the objective score."""
    n = len(p)
    s = np.sum(p)
    if s < 0.01:
        return np.inf
    c = full_convolve(p)
    cmax = np.max(c)
    return float(2.0 * n * cmax / (s * s))


def build_conv_matrix(p):
    """Build the sparse convolution constraint matrix."""
    n = len(p)
    if n == 0:
        return csr_matrix((0, 0))
    
    # For each output index k (0 to 2n-2), the valid input indices i
    # satisfy: 0 <= i <= n-1 and 0 <= k-i <= n-1
    # So i ranges from max(0, k-n+1) to min(k, n-1)
    # The column index is j = k - i, entry is p[i]
    
    all_rows = []
    all_cols = []
    all_data = []
    
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        num = i_max - i_min + 1
        if num <= 0:
            continue
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        all_rows.append(np.full(num, k, dtype=np.int32))
        all_cols.append(j_vals.astype(np.int32))
        all_data.append(p[i_vals])
    
    rows = np.concatenate(all_rows) if all_rows else np.array([], dtype=np.int32)
    cols = np.concatenate(all_cols) if all_cols else np.array([], dtype=np.int32)
    data = np.concatenate(all_data) if all_data else np.array([], dtype=np.float64)
    
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(p, M):
    """Solve the LP: max sum(g) s.t. (p*g)[k] <= M, g >= 0."""
    n = len(p)
    A = build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.status == 0 and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def normalize(p):
    """Normalize to sum=1, nonnegative."""
    p = np.maximum(p, 0.0)
    s = np.sum(p)
    if s < 1e-12:
        return np.ones(len(p)) / len(p)
    return p / s


def lp_iteration(p, rng, max_blends=None):
    """One LP best-response iteration with line search."""
    n = len(p)
    c = full_convolve(p)
    M = np.max(c)
    
    if M <= 0:
        return p, 0.0
    
    g = solve_lp(p, M)
    if g is None:
        return p, np.max(c)
    
    gs = np.sum(g)
    if gs < 1e-12:
        return p, np.max(c)
    g = g / gs  # normalize g to sum=1
    
    current_score = compute_score(p)
    best_p = p.copy()
    best_score = current_score
    
    blends = [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]
    
    for t in blends:
        p_new = normalize((1 - t) * p + t * g)
        sc = compute_score(p_new)
        if sc < best_score:
            best_score = sc
            best_p = p_new
    
    return best_p, best_score


def gradient_step(p, lr, alpha=0.1):
    """Gradient step using smoothed max surrogate."""
    n = len(p)
    c = full_convolve(p)
    cmax = np.max(c)
    T = alpha * cmax + 1e-300
    
    # softmax weights
    diff = (c - cmax) / T
    e = np.exp(diff)
    w = e / e.sum()
    
    # Gradient of sum_k w_k * c_k where c_k = (p*p)[k]
    # d c_k / d p_j = p[k-j] + p[j-k] (for valid indices)
    # Actually: c_k = sum_i p[i]*p[k-i], so d c_k / d p_j = p[k-j] + p[j-k]
    # (the two terms are the same when we consider both i=j and k-i=j)
    
    # Wait, c_k = sum_i p[i]*p[k-i]. d/dp_j:
    # Term i=j: p[k-j] (if 0<=k-j<=n-1)
    # Term k-i=j, i=k-j: p[k-j] (if 0<=k-j<=n-1)
    # If j = k-j (i.e., k=2j), we get 2*p[j]
    # In general: d c_k / d p_j = p[k-j] * (1 if 0<=k-j<=n-1 else 0) + p[k-j] * (1 if 0<=k-j<=n-1 else 0)
    # Wait, that's not right either.
    
    # c_k = sum_i p[i] * p[k-i]
    # d c_k / d p_j = sum_i [delta(i,j)*p[k-i] + p[i]*delta(k-i,j)]
    # = p[k-j] (if 0<=k-j<=n-1) + p[k-j] (if 0<=k-j<=n-1)
    # = 2*p[k-j] (if 0<=k-j<=n-1)
    # (when j = k-j, i.e., 2j=k, the two terms are the same, giving 2*p[j] which is correct)
    
    # So grad_j = 2 * sum_k w_k * p[k-j] (where 0<=k-j<=n-1, i.e., j <= k <= j+n-1)
    
    # This is 2 * (w * p_reversed) appropriately sliced
    # Let's compute via FFT for efficiency
    
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)  # w reversed for correlation
    h = np.real(ifft(Fp * Fw))
    
    # grad_j = 2 * h[j + n - 1] for j = 0, ..., n-1
    # Wait, let me verify. h[m] = sum_k w[k] * p_rev[k]... 
    
    # Actually, let me just do it directly for correctness with small n,
    # and use FFT for large n.
    
    # h = ifft(fft(p, L) * fft(w_rev, L))
    # h[m] = sum_k p[k] * w_rev[(m-k) mod L]
    # w_rev[i] = w[L-1-i]... no, w_rev = w[::-1] which has length 2n-1 padded to L
    # Hmm, this is getting complicated. Let me just use the direct formula.
    
    # grad_j = 2 * sum_{k: 0<=k-j<=n-1} w[k] * p[k-j]
    # = 2 * sum_{k=j}^{j+n-1} w[k] * p[k-j]  (clipped to valid range)
    
    # This is a filtered version: grad = 2 * correlate(w, p) evaluated at specific points
    
    # Let me compute via convolution:
    # (w * p_full)[k] = sum_i w[i]*p[k-i]
    # We want sum_k w[k]*p[k-j] = (w_reversed * p)[n-1-j]... 
    
    # OK let me just do it the straightforward way with a few FFT tricks.
    # Define: for j = 0,...,n-1, grad[j] = 2 * sum_{i=0}^{n-1} w[j+i] * p[i] (where w[j+i] is 0 if j+i >= 2n-1)
    
    # This is the cross-correlation of w and p, evaluated at specific lags.
    # cross_corr[j] = sum_i w[j+i] * p[i]
    
    # Using FFT: ifft(fft(w, L) * conj(fft(p, L))) gives the cross-correlation
    # But we need to be careful about indexing.
    
    # Actually, the simplest correct approach:
    # grad[j] = 2 * sum_{i=0}^{min(n-1, 2n-2-j)} w[j+i] * p[i]
    
    # Let's compute this via a single FFT-based convolution.
    # Let q[i] = p[i] for i=0,...,n-1 (zero elsewhere)
    # Let r[k] = w[k] for k=0,...,2n-2 (zero elsewhere)
    # Then grad[j] = 2 * sum_i r[j+i] * q[i] = 2 * (r_rev * q)[(2n-2) - j + (n-1)]
    # Hmm, this is getting messy. Let me just use a direct approach for the gradient.
    
    # For n up to ~2000, a direct O(n^2) computation per gradient step is fine
    # if we're doing it infrequently.
    
    # Actually, let me use the relation:
    # grad[j] = 2 * sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} w[k] * p[k-j]
    # Let i = k-j: grad[j] = 2 * sum_{i=max(0,-j)}^{min(n-1, 2n-2-j)} w[j+i] * p[i]
    # Since j >= 0: grad[j] = 2 * sum_{i=0}^{min(n-1, 2n-2-j)} w[j+i] * p[i]
    
    # This is (w shifted left by j) dot p, which is a "sliding dot product".
    # We can compute all of them at once using FFT:
    # Define W = w (length 2n-1), P = p (length n)
    # grad[j]/2 = sum_{i=0}^{min(n-1, 2n-2-j)} W[j+i] * P[i]
    # = <W[j:j+n'], P[:n']> where n' = min(n, 2n-1-j)
    
    # This is the cross-correlation of W and P at lag -j (or P and W at lag j).
    # corr[j] = sum_i W[i+j] * P[i]
    # = sum_i W[i+j] * P[i]
    
    # Using FFT: ifft(conj(fft(P, L)) * fft(W, L))[j] gives sum_i P[i] * W[i+j]? 
    # Let me verify: ifft(conj(F_P) * F_W)[m] = sum_n P[n] * W[(m+n) mod L]
    # For m=j: sum_n P[n] * W[(j+n) mod L]
    # If L is large enough and we're in the valid range, W[(j+n) mod L] = W[j+n]
    # So yes, corr[j] = ifft(conj(fft(P, L)) * fft(W, L))[j]
    
    L = next_fast_len(3 * n)
    FP = fft(p, L)
    FW = fft(w, L)  # w has length 2n-1, zero-padded to L
    corr = np.real(ifft(FP.conjugate() * FW))
    
    grad = 2.0 * corr[:n]
    
    # Projected gradient step on simplex
    p_new = p - lr * grad
    # Project onto simplex {x >= 0, sum(x) = 1}
    p_new = project_simplex(p_new)
    return p_new


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    v = np.array(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = np.sum(out)
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def iterated_lp(p, time_limit, rng, label=""):
    """Run iterated LP best-response for the given time."""
    p = normalize(p.copy())
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    
    end = time.monotonic() + time_limit
    iter_count = 0
    stall_count = 0
    
    while time.monotonic() < end - 5.0:
        c = full_convolve(p)
        M = np.max(c)
        
        if time.monotonic() > end - 10.0:
            break
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp(p, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            stall_count += 1
            if stall_count > 10:
                break
            # Try with slightly relaxed M
            g = solve_lp(p, M * 1.001)
            if g is None:
                break
        
        gs = np.sum(g)
        if gs < 1e-12:
            break
        g = g / gs
        
        # Line search over blend factors
        current_score = best_score
        improved = False
        
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
            p_candidate = normalize((1 - t) * p + t * g)
            sc = compute_score(p_candidate)
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = p_candidate.copy()
                p = p_candidate.copy()
                improved = True
                break
            elif sc < current_score + 1e-6:
                # Accept even without improvement to best (for exploration)
                if t < 0.3:
                    p = p_candidate.copy()
        
        if improved:
            stall_count = 0
        else:
            stall_count += 1
            # Try a gradient step to escape
            if stall_count % 5 == 0:
                lr = 0.01 / n
                p_grad = gradient_step(p, lr)
                sc_grad = compute_score(p_grad)
                if sc_grad < best_score:
                    best_score = sc_grad
                    best_p = p_grad.copy()
                    p = p_grad.copy()

                elif sc_grad < compute_score(p) + 1e-6:
                    p = p_grad.copy()

                p = normalize(best_p + rng.normal(0, 0.01/n, n))

        iter_count += 1
        if iter_count % 50 == 0:
            print(f"  {label} iter={iter_count} score={best_score:.8f} M={M:.6e} lp_time={lp_time*1000:.1f}ms", flush=True)

            p += a * np.sin(2*np.pi*f*x + ph)
```

Let me also think about whether the LP formulation is optimal. The standard form is:
max sum(g) s.t. (p*g)[k] <= M for all k, g >= 0

This gives the "flattest extension" in some sense. An alternative:
min t s.t. (p*g)[k] <= t for all k, sum(g) = 1, g >= 0

This directly minimizes the peak of the cross-convolution. The optimal t tells us the minimum possible peak given the current p. If t < M (the current peak), then there exists a g with lower peak, and we can move towards it.

Let me use this formulation instead:
- Variables: g[0], ..., g[n-1], t
- Constraints: (p*g)[k] <= t for k=0,...,2n-2; sum(g) = 1; g >= 0; t >= 0
- Objective: minimize t

This is a cleaner formulation. The LP has n+1 variables and 2n constraints (+ 1 equality).

```python
"""
Iterated LP best-response search for low-peak self-convolution step functions.
At each step, solves a linear program to find the nonnegative unit-sum vector g
that minimizes the peak of the cross-convolution (p*g), then blends towards g.
Uses multiple sequence lengths and line search for robust convergence.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def full_convolve(p):
    """Full self-convolution using FFT."""
    n = len(p)
    if n == 0:
        return np.array([])
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def compute_score(p):
    """Compute the objective: 2*n*max(p*p)/sum(p)^2."""
    n = len(p)
    s = np.sum(p)
    if s < 0.01:
        return np.inf
    c = full_convolve(p)
    cmax = np.max(c)
    return float(2.0 * n * cmax / (s * s))


def normalize(p):
    """Normalize to sum=1, nonnegative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    p = np.where(np.isfinite(p), p, 0.0)
    s = np.sum(p)
    if s < 1e-12:
        return np.ones(len(p)) / len(p)
    return p / s


def build_A_ub(n):
    """Precompute the row/col indices for the LP constraint matrix (structural part)."""
    all_rows = []
    all_cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            all_rows.append(k)
            all_cols.append(j)
        all_rows.append(k)
        all_cols.append(n)  # t column
    return np.array(all_rows, dtype=np.int32), np.array(all_cols, dtype=np.int32)


def solve_lp_minpeak(p, row_idx, col_idx):
    """Solve: min t s.t. (p*g)[k] <= t, sum(g)=1, g>=0, t>=0."""
    n = len(p)
    # Data for the constraint matrix
    data = np.zeros(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        if j == n:
            data[idx] = -1.0
        else:
            # coefficient of g[j] in (p*g)[k] is p[k-j]
            data[idx] = p[k - j]
    
    A_ub = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            t_opt = float(res.x[n])
            return g, t_opt
    except Exception:
        pass
    return None, np.inf


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    v = np.array(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = np.sum(out)
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def gradient_step(p, lr, alpha=0.05):
    """One step of projected subgradient descent on the smoothed max surrogate."""
    n = len(p)
    c = full_convolve(p)
    cmax = np.max(c)
    T = alpha * cmax + 1e-300
    
    diff = (c - cmax) / T
    e = np.exp(diff)
    w = e / e.sum()
    
    # grad[j] = 2 * sum_{i=0}^{n-1} w[j+i] * p[i] (where w[j+i]=0 if j+i >= 2n-1)
    # This is cross-correlation of w and p evaluated at lag j
    L = next_fast_len(3 * n)
    FP = fft(p, L)
    FW = fft(w, L)
    corr = np.real(ifft(FP.conjugate() * FW))
    grad = 2.0 * corr[:n]
    
    p_new = p - lr * grad
    return project_simplex(p_new)


def iterated_lp(p_init, time_limit, rng, label=""):
    """Run iterated LP best-response optimization."""
    p = normalize(p_init.copy())
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    
    # Precompute structural part of LP matrix
    row_idx, col_idx = build_A_ub(n)
    
    end = time.monotonic() + time_limit
    iter_count = 0
    stall_count = 0
    last_print = time.monotonic()
    
    while time.monotonic() < end - 3.0:
        c = full_convolve(p)
        M = float(np.max(c))
        
        # Solve LP
        t0 = time.monotonic()
        g, t_opt = solve_lp_minpeak(p, row_idx, col_idx)
        lp_time = time.monotonic() - t0
        
        if g is None:
            stall_count += 1
            if stall_count > 20:
                break
            if stall_count % 10 == 0:
                # Perturb and retry
                p = normalize(p + rng.normal(0, 0.005 / n, n))
            continue
        
        g = normalize(g)
        
        # Check if we can improve
        current_score = compute_score(p)
        improved = False
        
        # Try various blend factors
        blends = [1.0, 0.9, 0.75, 0.6, 0.45, 0.3, 0.2, 0.1, 0.05]
        accepted_blend = None
        
        for t in blends:
            if t == 1.0:
                p_candidate = g
            else:
                p_candidate = normalize((1.0 - t) * p + t * g)
            
            sc = compute_score(p_candidate)
            if sc < best_score - 1e-10:
                best_score = sc
                best_p = p_candidate.copy()
                p = p_candidate.copy()
                improved = True
                accepted_blend = t
                break
            elif sc < current_score + 1e-8 and t <= 0.3:
                # Accept neutral/slightly-worse for exploration if blend is small
                p = p_candidate.copy()
                accepted_blend = t
        
        if improved:
            stall_count = 0
        else:
            stall_count += 1
            
            # Try gradient step to escape
            if stall_count >= 3 and stall_count % 4 == 0:
                for lr in [0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n]:
                    p_grad = gradient_step(p, lr)
                    sc_grad = compute_score(p_grad)
                    if sc_grad < best_score - 1e-10:
                        best_score = sc_grad
                        best_p = p_grad.copy()
                        p = p_grad.copy()
                        stall_count = 0
                        improved = True
                        break
                    elif sc_grad < compute_score(p) + 1e-6:
                        p = p_grad.copy()
            
            # Restart from best with perturbation if stuck long
            if stall_count > 40:
                p = normalize(best_p + rng.normal(0, 0.02 / n, n))
                stall_count = 0
            elif stall_count > 20:
                p = normalize(0.8 * best_p + 0.2 * p + rng.normal(0, 0.005 / n, n))
        
        iter_count += 1
        now = time.monotonic()
        if now - last_print > 30.0 or iter_count % 100 == 0:
            print(f"  {label} iter={iter_count} score={best_score:.8f} M={M:.4e} "
                  f"t_opt={t_opt:.4e} blend={accepted_blend} lp_t={lp_time*1000:.0f}ms", flush=True)
            last_print = now
    
    return best_p, best_score


def make_initials(n, rng, parent_seq=None):
    """Generate diverse initial candidates of length n."""
    candidates = []
    x = np.linspace(0.0, 1.0, n)
    
    if parent_seq is not None:
        ps = np.asarray(parent_seq, dtype=np.float64)
        ps = np.maximum(ps, 0.0)
        if ps.size == n and np.sum(ps) > 0.01:
            candidates.append(normalize(ps))
    
    candidates.append(np.ones(n) / n)
    
    # Cosine tapers (various orders)
    for power in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        p = np.cos(np.pi * (x - 0.5)) ** (2 * power)
        p = np.maximum(p, 0)
        if np.sum(p) > 0:
            candidates.append(normalize(p))
    
    # Power laws
    for a in [0.1, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0]:
        p = np.power(x + 1e-10, a)
        candidates.append(normalize(p))
        candidates.append(normalize(p[::-1]))
    
    # Beta-like
    for alpha, beta in [(0.5, 0.5), (1, 1), (2, 1), (1, 2), (3, 1), (1, 3), (4, 1), (1, 4)]:
        p = np.power(x + 1e-10, alpha) * np.power(1 - x + 1e-10, beta)
        candidates.append(normalize(p))
    
    # Gaussians
    for mu in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sig in [0.08, 0.15, 0.25, 0.35]:
            p = np.exp(-0.5 * ((x - mu) / sig) ** 2)
            candidates.append(normalize(p))
    
    # Two-lobe
    candidates.append(normalize(np.exp(-10*(x-0.2)**2) + np.exp(-10*(x-0.8)**2)))
    candidates.append(normalize(np.exp(-20*(x-0.25)**2) + np.exp(-20*(x-0.75)**2)))
    candidates.append(normalize(np.exp(-15*(x-0.2)**2) + np.exp(-15*(x-0.8)**2) + 0.3*np.exp(-5*(x-0.5)**2)))
    
    # Random smooth
    for _ in range(8):
        p = np.ones(n)
        for _ in range(3):
            f = rng.integers(1, max(2, n // 30))
            ph = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.1, 0.6)
            p += a * np.sin(2 * np.pi * f * x + ph)
        p = np.maximum(p, 0.01)
        candidates.append(normalize(p))
    
    return candidates
```
