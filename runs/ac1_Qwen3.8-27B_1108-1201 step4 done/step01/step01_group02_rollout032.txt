        return np.ones(len(a)) / len(a)
    return a / s

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_asymmetric(n, shift=0.3):
    """Asymmetric initialization - shifts the mass to one side."""
    t = np.arange(n, dtype=np.float64)
    center = shift * n
    sigma = 0.2 * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_two_bumps(n, frac=0.4):
    """Two-bump asymmetric initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - 0.3*n) / (0.15*n)) ** 2)
    a += np.exp(-0.5 * ((t - 0.7*n) / (0.15*n)) ** 2)
    return a

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

def build_conv_matrix(a):
    """Build sparse matrix such that M @ g = conv(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    data_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
        data_parts.append(np.full(n, a[j], dtype=np.float64))
    row_idx = np.concatenate(row_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def subgradient_step(a, b, p_power=2.0):
    """
    Compute one subgradient step.
    Returns the new (unnormalized, non-negative) sequence.
    """
    n = len(a)
    # Compute weights
    if p_power > 0:
        w = b ** p_power
        wsum = w.sum()
        if wsum < 1e-30:
            return a.copy()
        w = w / wsum
    else:
        w = np.ones(len(b)) / len(b)
    
    # Compute sg[i] = sum_j w[i+j] * a[j]
    # Use np.correlate: correlate(w, a, 'valid') gives exactly this
    # w has length 2n-1, a has length n, result has length n
    sg = np.correlate(w, a, mode='valid')
    
    # Ensure non-negative
    sg = np.maximum(0.0, sg)
    max_sg = sg.max()
    if max_sg < 1e-30:
        return a.copy()
    
    return sg / max_sg

def mirror_descent_refine(a, time_limit_s=60.0, p_power=2.0, verbose=False):
    """
    Mirror descent with multiplicative updates to minimize max(conv(a,a)).
    """
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    eta = 0.5
    best_eta = eta
    stagnant = 0
    
    for it in range(500000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        b = np.convolve(a, a)
        max_b = b.max()
        current_score = 2.0 * n * max_b / (a.sum() ** 2)
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 200:
            # Try with larger eta
            eta = min(2.0, eta * 2.0)
            stagnant = 0
            if eta > 5.0:
                break
        
        sg = subgradient_step(a, b, p_power)
        
        # Multiplicative update
        a_new = a * (1.0 - eta * sg)
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            eta *= 0.1
            continue
        a_new = a_new / s
        
        new_score = 2.0 * n * np.max(np.convolve(a_new, a_new)) / (a_new.sum() ** 2)
        
        if new_score <= current_score + 1e-14:
            a = a_new
            if new_score < current_score - 1e-10:
                best_eta = eta
                eta = min(2.0, eta * 1.01)
        else:
            eta = max(0.01, eta * 0.5)
    
    return best_a, best_score

def lp_refine(a, time_limit_s=60.0, verbose=False):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    stagnant = 0
    it = 0
    
    # Pre-build the conv matrix structure (it changes with a, but we rebuild each time)
    for it in range(200):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        b = np.convolve(a, a)
        max_b = b.max()
        current_score = 2.0 * n * max_b / (a.sum() ** 2)
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 3):
                print(f"    LP iter {it:3d}: score={current_score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        
        timeout = max(2.0, min(30.0, remaining * 0.3))
        
        # Solve LP: maximize sum(g) s.t. conv(a,g) <= max_b, g >= 0
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, max_b)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    # Mix with current
                    alpha = min(0.5, max(0.05, 2.0 / (1.0 + 0.5 * it)))
                    a_new = (1.0 - alpha) * a + alpha * g
                    a_new = normalize_nonneg(a_new)
                    new_score = compute_score(a_new)
                    if new_score <= current_score + 1e-14:
                        a = a_new
                    else:
                        alpha2 = alpha * 0.1
                        a_new = (1.0 - alpha2) * a + alpha2 * g
                        a = normalize_nonneg(a_new)
                else:
                    if verbose:
                        print(f"    LP: no improvement at iter {it}")
                    break
            else:
                break
        except Exception:
            break
    
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score

def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=False):
    """Random local search with multiplicative mutations."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    accept_count = 0
    
    while time.monotonic() - start_time < time_limit_s:
        # Random window mutation
        width = max(2, int(rng.integers(max(2, n // 100), max(3, n // 4))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        # Multiplicative perturbation
        noise = np.exp(rng.normal(0.0, rng.uniform(0.05, 0.3), size=stop - start))
        candidate[start:stop] *= noise
        
        # Occasionally try global scaling of a random subset
        if rng.random() < 0.2:
            mask = rng.random(n) < 0.5
            candidate[mask] *= np.exp(rng.normal(0.0, 0.2))
        
        # Occasionally try to flip symmetry
        if rng.random() < 0.1:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = compute_score(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    return best_a, best_score

def spectral_refine(a, time_limit_s=30.0, verbose=False):
    """
    Spectral polishing: identify frequency components contributing to
    convolution peaks and dampen them.
    """
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    N = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    N_fft = 1
    while N_fft < 2 * N:
        N_fft *= 2
    
    for it in range(1000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        A = np.fft.rfft(a, N_fft)
        B = A ** 2  # Spectrum of conv(a,a)
        b = np.fft.irfft(B, N_fft)[:N]  # Convolution (zero-padded)
        
        max_b = b.max()
        current_score = 2.0 * n * max_b / (a.sum() ** 2)
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
        
        # Find peak position
        k_star = int(np.argmax(b))
        
        # Compute which frequencies contribute most to the peak
        # b[k_star] = (1/N_fft) * sum_f B[f] * exp(2πi f k_star / N_fft)
        # The contribution of frequency f to b[k_star] is B[f] * exp(2πi f k_star / N_fft)
        freqs = np.fft.rfftfreq(N_fft) * N_fft  # 0, 1, 2, ...
        phases = np.exp(2j * np.pi * freqs * k_star / N_fft)
        contributions = B * phases
        
        # Identify frequencies where contribution is positively real (adds to peak)
        pos_mask = np.real(contributions) > 0
        pos_contrib = np.real(contributions) * pos_mask
        
        if pos_contrib.max() < 1e-15:
            break
        
        # Dampen the most contributing frequencies
        # Scale factor: reduce |A[f]| for high-contribution frequencies
        damping = 1.0
        for freq_idx in np.argsort(pos_contrib)[-10:]:  # Top 10 frequencies
            if pos_contrib[freq_idx] > 0:
                # Reduce A[f] to reduce B[f] = A[f]^2
                # New B[f] = B[f] * (1 - damping * pos_contrib[f] / pos_contrib.max())
                # Since B = A^2, new A = A * sqrt(1 - ...)
                reduce = damping * pos_contrib[freq_idx] / pos_contrib.max()
                reduce = min(0.5, reduce)
                A[freq_idx] *= np.sqrt(1.0 - reduce)
        
        # Reconstruct a
        a_new = np.fft.irfft(A, N_fft)[:n]
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            damping *= 0.5
            continue
        a_new = a_new / s
        
        new_score = compute_score(a_new)
        if new_score < current_score:
            a = a_new
            damping = min(0.5, damping * 1.1)
        else:
            damping *= 0.5
            if damping < 0.01:
                break
    
    return best_a, best_score
