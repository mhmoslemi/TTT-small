def _eval_and_grad(h):

    h = np.clip(h, 0, None)

    dx = 1.0 / (M + 1)

    y[1:M+1] = conv
    
    l2_sq = (dx / 3.0) * float(np.sum(y[:-1]**2 + y[:-1]*y[1:] + y[1:]**2))
    norm_1 = float(np.sum(conv)) / (M + 1)
    norm_inf = float(np.max(conv))
    
    if norm_1 < 1e-15 or norm_inf < 1e-15:
        return 0.0, np.zeros(N)
    
    C = l2_sq / (norm_1 * norm_inf)

    g_y = (dx / 3.0) * (4.0 * y + np.concatenate(([0.0], y[:-1])) + np.concatenate((y[1:], [0.0])))

import numpy as np
import time


def _eval_grad(h):
    N = len(h)
    h = np.clip(h, 0.0, None)
    conv = np.convolve(h, h)
    M = len(conv)
    dx = 1.0 / (M + 1)
    y = np.zeros(M + 2)
    y[1:M + 1] = conv
    l2_sq = (dx / 3.0) * float(np.sum(y[:-1] ** 2 + y[:-1] * y[1:] + y[1:] ** 2))
    norm_1 = float(np.sum(conv)) / (M + 1)
    norm_inf = float(np.max(conv))
    if norm_1 < 1e-15 or norm_inf < 1e-15:
        return 0.0, np.zeros(N)
    C = l2_sq / (norm_1 * norm_inf)
    g_y = (dx / 3.0) * (4.0 * y + np.concatenate(([0.0], y[:-1])) + np.concatenate((y[1:], [0.0])))
    g_conv = g_y[1:M + 1]
    g_n1 = 1.0 / (M + 1)
    mask = (conv == norm_inf)
    g_ninf = mask.astype(np.float64) / max(1, mask.sum())
    denom = norm_1 * norm_inf
    g_Cc = (g_conv * denom - l2_sq * (g_n1 * norm_inf + norm_1 * g_ninf)) / (denom * denom)
    g_h = 2.0 * np.convolve(g_Cc, h[::-1], mode="valid")
    return C, g_h


def _adam_step(params, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * (grad * grad)
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    return params + lr * m_hat / (np.sqrt(v_hat) + eps), m, v

def _upsample(h, factor, rng=None):
    N = len(h)
    N_new = N * factor
    x_old = np.linspace(0.0, 1.0, N)
    x_new = np.linspace(0.0, 1.0, N_new)
    h_new = np.interp(x_new, x_old, h)

    h_new = np.clip(h_new, 0.0, None)
    m = np.mean(h_new)
    if m > 0:
        h_new /= m
    return h_new
