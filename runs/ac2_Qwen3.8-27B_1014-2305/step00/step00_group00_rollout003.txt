def _norms_and_grad(conv, N):
    """Compute C and dC/dconv for non-negative conv (= f*f)."""
    M = conv.size  # 2N-1
    h = 1.0 / (2*N)
    y = np.concatenate(([0.0], conv, [0.0]))  # length M+2 = 2N+1
    
    # l2_sq
    l2_sq = (h/3.0) * np.sum(y[:-1]**2 + y[:-1]*y[1:] + y[1:]**2)
    
    # norm1
    norm1 = np.sum(conv) / (2*N)
    
    # norminf (always at center for non-negative f)
    norminf = conv[N-1]
    
    if norm1 <= 0 or norminf <= 0:
        return 0.0, np.zeros_like(conv)
    
    C = l2_sq / (norm1 * norminf)
    
    # d(l2_sq)/d(conv[j])
    # grad_y[j] = (h/3)(y[j-1] + 4*y[j] + y[j+1]) for interior
    # For conv[j] = y[j+1]:

import numpy as np
import time
from scipy.signal import fftconvolve


def _norms_grad(conv, N):
    M = conv.size
    h = 1.0 / (2.0 * N)
    y = np.empty(M + 2, dtype=np.float64)
    y[0] = 0.0
    y[1:-1] = conv
    y[-1] = 0.0
    l2_sq = (h / 3.0) * float(np.sum(y[:-1]**2 + y[:-1]*y[1:] + y[1:]**2))
    norm1 = float(np.sum(conv)) / (2.0 * N)
    norminf = float(conv[N - 1])
    if norm1 <= 0 or norminf <= 0:
        return 0.0, np.zeros_like(conv)
    C = l2_sq / (norm1 * norminf)
    g_l2 = np.empty_like(conv)
    g_l2[0] = (h / 3.0) * (4.0 * conv[0] + conv[1])
    g_l2[-1] = (h / 3.0) * (conv[-2] + 4.0 * conv[-1])
    if M > 2:
        g_l2[1:-1] = (h / 3.0) * (conv[:-2] + 4.0 * conv[1:-1] + conv[2:])
    g_n1 = np.full_like(conv, 1.0 / (2.0 * N))
    g_ninf = np.zeros_like(conv)
    g_ninf[N - 1] = 1.0
    denom = norm1 * norminf
    g_conv = (g_l2 * denom - l2_sq * (g_n1 * norminf + norm1 * g_ninf)) / (denom * denom)
    return C, g_conv


def _grad_h(h, g_conv):
    h_rev = h[::-1]
    return 2.0 * np.convolve(g_conv, h_rev, mode='valid')


def _eval_fast(h):

    conv = fftconvolve(h, h, mode='full')
    C, _ = _norms_grad(conv, h.size)
    return C

        v = b2 * v + (1 - b2) * g_h ** 2
