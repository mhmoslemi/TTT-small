def _compute_C_and_grad(f):
    """Compute C and dC/df for nonnegative f."""
    f = np.maximum(f, 0)
    N = len(f)
    g = np.convolve(f, f)  # length 2N-1
    M = len(g)
    dx = 1.0 / (M + 1)  # = 1/(2N)
    
    # L2 squared (piecewise linear)
    g_pad = np.concatenate(([0.0], g, [0.0]))  # length M+2
    l2_sq = (dx / 3.0) * np.sum(g_pad[:-1]**2 + g_pad[:-1]*g_pad[1:] + g_pad[1:]**2)
    
    # L1
    n1 = np.sum(g) / (M + 1)
    
    # Linf
    ninf = np.max(g)
    
    if n1 <= 0 or ninf <= 0:
        return 0.0, np.zeros_like(f)
    
    C = l2_sq / (n1 * ninf)
    
    # Gradient of L2
    g_left = np.concatenate(([0.0], g[:-1]))
    g_right = np.concatenate((g[1:], [0.0]))
    d_l2 = (dx / 3.0) * (2.0 * g + g_left + g_right)
    
    # Gradient of n1
    d_n1 = np.full_like(g, 1.0 / (M + 1))
    
    # Gradient of ninf (subgradient)
    max_mask = (g == ninf)
    d_ninf = max_mask.astype(np.float64) / max_mask.sum()
    
    # Chain rule: C = l2_sq / (n1 * ninf)
    denom = n1 * ninf
    denom_sq = denom * denom

    dC_df = 2.0 * np.correlate(dC_dg, f, mode='valid')
    
    return C, dC_df

import time

def _eval_and_grad(f):
    f = np.maximum(f, 0.0)
    N = f.size
    g = np.convolve(f, f)
    M = g.size  # 2N-1
    dx = 1.0 / (M + 1)
    
    gp = np.concatenate(([0.0], g, [0.0]))
    l2_sq = (dx / 3.0) * np.sum(gp[:-1]**2 + gp[:-1]*gp[1:] + gp[1:]**2)
    n1 = np.sum(g) * dx
    ninf = np.max(g)
    
    if n1 < 1e-12 or ninf < 1e-12:
        return 0.0, np.zeros_like(f)
    
    C = l2_sq / (n1 * ninf)
    
    gl = np.concatenate(([0.0], g[:-1]))
    gr = np.concatenate((g[1:], [0.0]))
    d_l2 = (dx / 3.0) * (2.0*g + gl + gr)
    d_n1 = np.full_like(g, dx)
    mask = (g == ninf)
    d_ninf = mask.astype(float) / mask.sum()

    return C, dC_df

def _adam_step(params, grads, state, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1-b1) * grads
    state['v'] = b2 * state['v'] + (1-b2) * grads**2
    m_hat = state['m'] / (1 - b1**t)
    v_hat = state['v'] / (1 - b2**t)
    return np.maximum(params + lr * m_hat / (np.sqrt(v_hat) + eps), 0.0)

def _optimize(f0, max_steps=20000, lr=0.05, deadline=None):

        if C > best_C:
            best_C = C
            best_f = f.copy()
        f = _adam_step(f, grad, state, lr)
    
    return best_f, best_C

def _eval_grad(f):
    f = np.maximum(f, 0.0)
    N = f.size
    g = np.convolve(f, f)
    M = g.size
    dx = 1.0 / (M + 1)
    gp = np.concatenate(([0.0], g, [0.0]))
    l2 = (dx / 3.0) * np.sum(gp[:-1]**2 + gp[:-1]*gp[1:] + gp[1:]**2)
    n1 = np.sum(g) * dx
    ninf = np.max(g)
    if n1 < 1e-15 or ninf < 1e-15:
        return 0.0, np.zeros_like(f)
    C = l2 / (n1 * ninf)
    gl = np.concatenate(([0.0], g[:-1]))
    gr = np.concatenate((g[1:], [0.0]))
    dl2 = (dx / 3.0) * (2*g + gl + gr)
    dn1 = np.full(M, dx)
    mk = (g == ninf)
    dinf = mk.astype(float) / mk.sum()
    dn = n1 * ninf
    dCd = (dl2 * dn - l2 * (dn1 * ninf + n1 * dinf)) / (dn * dn)
    dCdf = 2.0 * np.correlate(dCd, f, mode='valid')
    return C, dCdf


def _adam(params, grad, st, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    st[0] += 1
    t = st[0]
    st[1] = b1*st[1] + (1-b1)*grad
    st[2] = b2*st[2] + (1-b2)*grad*grad
    mhat = st[1] / (1 - b1**t)
    vhat = st[2] / (1 - b2**t)
    return np.maximum(params + lr*mhat/(np.sqrt(vhat)+eps), 0.0)
