"""

Key improvements: adaptive temperature schedule, backtracking line search in LP,

import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=True, max_iter=5000):
    """LP-based optimization with adaptive step size and line search."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()

    n = len(f)

    best_f = f.copy()
    t = 0.25

    iteration = 0

    while time.monotonic() < deadline and iteration < max_iter:
        M = float(full_convolve_fft(f).max())
        if stagnant > 0:
            M = best_M

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Line search: try multiple step sizes
        best_t = 0.0
        best_M_new = M
        for candidate_t in [t, t * 0.5, t * 0.75, t * 1.25]:
            candidate_t = min(max(candidate_t, 0.002), 0.6)
            f_new = (1 - candidate_t) * f + candidate_t * g
            f_new = np.maximum(f_new, 0.0)

            if s_new < 1e-10:

                t = 0.002
            stagnant += 1

            f = (1 - best_t) * f + best_t * g

            stagnant = 0

def soft_mx(c, beta):
    """Smooth max with log-sum-exp. Returns (value, weights)."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-13
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 (gradient of conv)."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def logit_anneal(p0, deadline, rng, lr=0.05, reg=1e-8, n_max_iter=500000):
    """Improved soft-max Adam polish with adaptive temperature and annealing restarts."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    iter_count = 0

    while time.monotonic() < deadline and iter_count < n_max_iter:
        iter_count += 1
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        step = lr * mh / (np.sqrt(vh) + eps)
        # Adaptive step: scale by gradient magnitude
        gnorm = float(np.linalg.norm(step))
        if gnorm > 0:
            step = step * min(1.0, 0.5 / gnorm) * 2.0
        z = z - step

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        # Temperature cooling
        if iter_count % 50 == 0:
            beta = max(1e-14, beta * 0.85)

        # Learning rate decay
        if no_imp % 500 == 0 and no_imp > 0:
            lr = max(0.005, lr * 0.97)

        # Restart logic
        restart_threshold = max(500, 3 * n)
        if no_imp > restart_threshold:
            # Simulated annealing style restart
            temp = 0.15 + 0.1 * (no_imp / (2 * restart_threshold))
            z = best_z + temp * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def local_perturb_search(p0, deadline, rng, n_perturbs=200):
    """Try random local perturbations and keep the best."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    s = p.sum()
    if s < 1e-10:
        return p0, np.inf
    p = p / s

    best_p = p.copy()
    best_score = compute_score(p)

    scale = 0.05 * (1.0 / np.sqrt(n))
    count = 0
    while time.monotonic() < deadline and count < n_perturbs:
        count += 1
        # Try different perturbation strategies
        strategy = count % 3
        if strategy == 0:
            # Gaussian perturbation
            p_new = p + scale * rng.normal(size=n)
        elif strategy == 1:
            # Sparse perturbation (change a few elements)
            p_new = p.copy()
            k = max(1, n // 10)
            idx = rng.choice(n, k, replace=False)
            p_new[idx] *= (1.0 + scale * 3 * rng.uniform(-1, 1, k))
        else:
            # Block perturbation
            p_new = p.copy()
            block_size = max(1, n // 20)
            start = rng.integers(0, n - block_size)
            p_new[start:start + block_size] += scale * 2 * rng.normal(size=block_size)

        p_new = np.maximum(p_new, 0.0)
        s_new = p_new.sum()
        if s_new < 1e-10:
            continue
        p_new = p_new / s_new
        sc = compute_score(p_new)
        if sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            # Decay scale as we find improvements
            scale *= 0.99

    return best_p, best_score


def make_linear_profile(n, eta=1.5):
    """Monotone linear profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_profile(n, alpha=2.0):
    """Power-law profile: (t+eps)^alpha."""
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.01) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_exp_profile(n, rate=2.0):
    """Exponential profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a_coeff=0.0, b_coeff=1.0):
    """Quadratic profile 1 + a*t + b*t^2."""
    t = np.linspace(0.0, 1.0, n)
    a_arr = 1.0 + a_coeff * t + b_coeff * t ** 2
    a_arr = np.maximum(a_arr, 1e-12)
    return a_arr / a_arr.sum()


def make_mixed_profile(n, k1=2, k2=3, amp=0.3):
    """Mixed oscillatory profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2 * np.pi * k1 * t) + 0.5 * amp * np.cos(2 * np.pi * k2 * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def get_starting_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    for eta in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        profiles.append(("lin_{}".format(eta), make_linear_profile(n, eta)))
    for alpha in [1.5, 2.0, 3.0, 4.0]:
        profiles.append(("pow_{}".format(alpha), make_power_profile(n, alpha)))
    for rate in [1.5, 2.5, 4.0]:
        profiles.append(("exp_{}".format(rate), make_exp_profile(n, rate)))
    profiles.append(("quad1", make_quadratic_profile(n, 1.0, -0.5)))
    profiles.append(("quad2", make_quadratic_profile(n, -0.5, 1.0)))
    profiles.append(("mix1", make_mixed_profile(n, 2, 3, 0.3)))
    profiles.append(("mix2", make_mixed_profile(n, 1, 4, 0.2)))
    # Random smooth
    for i in range(4):
        coeffs = rng.uniform(-0.4, 0.4, 5)
        t = np.linspace(0.0, 1.0, n)
        a = 1.0
        for j, cval in enumerate(coeffs[1:], 1):
            a = a + cval * np.cos(2 * np.pi * j * t)
        a = np.maximum(a, 1e-12)
        profiles.append(("rand_{}".format(i), a / a.sum()))
    return profiles


def interpolate_profile(p0, n_new):
    """Cubic interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 1:
        return np.full(n_new, p0[0] / n_new)
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    from scipy.interpolate import PchipInterpolator
    # Clamp to avoid negative values in interpolation
    p0_clamped = np.maximum(p0, 0.0)
    s = p0_clamped.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    p0_clamped = p0_clamped / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    try:
        interp = PchipInterpolator(x0, p0_clamped)
        p_new = interp(x1)
    except Exception:
        p_new = np.interp(x1, x0, p0_clamped)
    p_new = np.maximum(p_new, 0.0)
    s_new = p_new.sum()
    if s_new < 1e-10:
        return np.ones(n_new) / n_new
    return p_new / s_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP descent, soft-max Adam polish with annealing,
    local perturbation search, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ============ Phase 1: LP optimization at n=512 ============
    n_main = 512
    phase1_end = min(deadline, t_start + 350.0)

    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_linear_profile(n_main, 2.0)

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1 start: n={n_main}, score={sc:.10f}")

    # Main LP optimization
    lp_deadline = min(phase1_end, time.monotonic() + 250.0)
    result = lp_optimize(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP: score={sc:.10f}")

    # Try diverse starting profiles
    profiles = get_starting_profiles(n_main, rng)
    for name, p in profiles:
        if time.monotonic() > phase1_end - 3:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.02:
            continue
        per_deadline = min(phase1_end, time.monotonic() + 25.0)
        result = lp_optimize(p, per_deadline, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        if sc < best_score + 0.005:
            print(f"  Profile {name}: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # ============ Phase 2: Soft-max polish at n=512 ============
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 2: polishing at n={n_polish}, current best={best_score:.10f}")

        # Multiple rounds with different hyperparameters
        for round_i, (lr_val, beta_scale) in enumerate([(0.05, 0.15), (0.03, 0.1), (0.02, 0.08)]):
            if time.monotonic() > phase2_end - 5:
                break
            per_deadline = min(phase2_end, time.monotonic() + 120.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.002 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=lr_val)
            update(result, sc)
            print(f"  Adam round {round_i} (lr={lr_val}): score={sc:.10f}")

        # Local perturbation search
        if time.monotonic() < phase2_end - 5:
            per_deadline = min(phase2_end, time.monotonic() + 60.0)
            result, sc = local_perturb_search(best_seq, per_deadline, rng)
            update(result, sc)
            print(f"  Local perturb: score={sc:.10f}")

        # Final short Adam polish
        if time.monotonic() < phase2_end - 5:
            per_deadline = min(phase2_end, time.monotonic() + 30.0)
            p0 = best_seq + 0.001 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=0.02)
            update(result, sc)
            print(f"  Final short Adam: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # ============ Phase 3: LP at n=1024 ============
    n_large = 1024
    phase3_end = min(deadline, t_start + 1100.0)

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 3 start: n={n_large}, score={sc:.10f}")

        lp_deadline2 = min(phase3_end, time.monotonic() + 250.0)
        result = lp_optimize(p_large, lp_deadline2, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After large LP: score={sc:.10f}")

        # Try a few different profiles at large n
        if time.monotonic() < phase3_end - 10:
            for name, p_small in profiles[:5]:
                if time.monotonic() > phase3_end - 5:
                    break
                p_lp = interpolate_profile(p_small, n_large)
                sc_p = compute_score(p_lp)
                if sc_p >= best_score + 0.01:
                    continue
                per_deadline = min(phase3_end, time.monotonic() + 30.0)
                result = lp_optimize(p_lp, per_deadline, rng, verbose=False)
                sc = compute_score(result)
                update(result, sc)

    print(f"After Phase 3: best={best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # ============ Phase 4: Soft-max polish at n=1024 ============
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_polish2 = best_seq.size
        print(f"Phase 4: polishing at n={n_polish2}, current best={best_score:.10f}")

        for round_i, (lr_val, beta_scale) in enumerate([(0.04, 0.12), (0.025, 0.08)]):
            if time.monotonic() > phase4_end - 5:
                break
            per_deadline = min(phase4_end, time.monotonic() + 150.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.001 * rng.normal(size=n_polish2)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=lr_val)
            update(result, sc)
            print(f"  Adam round {round_i} (lr={lr_val}): score={sc:.10f}")

        if time.monotonic() < phase4_end - 5:
            per_deadline = min(phase4_end, time.monotonic() + 40.0)
            result, sc = local_perturb_search(best_seq, per_deadline, rng)
            update(result, sc)
            print(f"  Local perturb: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # ============ Phase 5: Try n=2048 if time remains and score is still high ============
    n_xlarge = 2048
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None and best_score > 1.504:
        p_xl = interpolate_profile(best_seq, n_xlarge)
        sc = compute_score(p_xl)
        update(p_xl, sc)
        print(f"Phase 5: n={n_xlarge}, score={sc:.10f}")

        lp_deadline3 = min(phase5_end, time.monotonic() + 80.0)
        result = lp_optimize(p_xl, lp_deadline3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After xlarge LP: score={sc:.10f}")

        if time.monotonic() < phase5_end - 10:
            per_deadline = min(phase5_end, time.monotonic() + 60.0)
            p0 = best_seq.copy()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=0.03)
            update(result, sc)
            print(f"  Xlarge Adam: score={sc:.10f}")

    print(f"After Phase 5: best={best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # ============ Phase 6: Final intensive polish ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        remaining = deadline - time.monotonic() - 10
        print(f"Phase 6: final polish at n={n_final}, {remaining:.0f}s remaining, score={best_score:.10f}")

        # Split remaining time into chunks
        n_chunks = max(1, int(remaining / 25))
        chunk_time = remaining / n_chunks
        for chunk in range(n_chunks):
            if time.monotonic() > deadline - 10:
                break
            per_deadline = min(deadline - 10, time.monotonic() + chunk_time)
            # Alternate between Adam and perturbation
            if chunk % 2 == 0:
                p0 = best_seq + 0.0005 * rng.normal(size=n_final)
                p0 = np.maximum(p0, 1e-12)
                p0 = p0 / p0.sum()
                result, sc = logit_anneal(p0, per_deadline, rng, lr=0.015)
            else:
                result, sc = local_perturb_search(best_seq, per_deadline, rng, n_perturbs=100)
            update(result, sc)

    print(f"After Phase 6: best={best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 2.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
