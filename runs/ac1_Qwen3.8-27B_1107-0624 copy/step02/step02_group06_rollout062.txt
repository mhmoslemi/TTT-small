def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_value(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    mval = float(c.max())
    if beta <= 1e-14:
        return mval
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        return mval
    return mval + beta * float(np.log(s))


def soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    n = c.size
    if n == 0:
        return np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros(n)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(n)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def corr_weights(w, p):

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

             haiv = float(np.dot(hiv, aiv))
            giv = 2.0 * niv * aiv * (hiv - haiv)
            # Add small regularization to prevent collapse
            giv += 1e-8 * aiv
            return 2.0 * niv * Miv, giv

        res = scipy_minimize(obj_and_grad, z, method='L-BFGS-B',
                             options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8,
                                      'maxls': 50})
        if res.success or res.fun < best_score:
            z = res.x.copy()
        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        sc = float(2.0 * n * c.max())
        if sc < best_score:
            best_score = sc
            best_z = z.copy()

        # Cool down beta
        beta = max(1e-10, beta * 0.5)

        # Check time
        if time.monotonic() > deadline:
            break

    return best_z, best_score


def nesterov_subgradient(a0, deadline, rng, T=None):
    """Nesterov accelerated subgradient method on the simplex."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    if T is None:
        T = max(200, int((deadline - time.monotonic()) * 100))

    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())
    alpha = 1.0

    for t in range(T):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at y
        c = full_convolve_fft(y)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        S = float(y.sum())

        # Gradient: (2*n/S^2) * (2*a_{k*-shifted} - 2*M*a/S)
        # For normalized a (S=1): g_i = 2*n * (2*y[k*-i] * 1_{valid} - 2*M*y[i])
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]
            grad[i] -= 2.0 * M * y[i]
        grad *= 2.0 * n

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        v = (t / (t + 2.0)) * v
        y = y - alpha_t * grad / (float(np.linalg.norm(grad)) + 1e-10)
        # Project onto simplex
        y = project_simplex(y)
        # Momentum step
        a = (t / (t + 2.0)) * a + (1.0 - t / (t + 2.0)) * y

        # Track best
        c_a = full_convolve_fft(a)
        sc = float(2.0 * n * c_a.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.where(sv * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_weights_dual(a0, deadline, rng, n_outer=None):
    """Dual multiplicative weights method."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    # Initialize lambda uniformly
    lam = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())

    if n_outer is None:
        n_outer = int((deadline - time.monotonic()) * 20)
    n_outer = max(100, n_outer)

    eta = 1.0
    for outer in range(n_outer):
        if time.monotonic() > deadline:
            break

        # Inner: minimize a^T Q(lam) a over simplex
        # Use projected gradient with a few iterations
        a_inner = a.copy()
        for inner in range(50):
            # Compute gradient: 2 * H @ a where H_{ij} = lam_{i+j}
            # H @ a = hankel_matvec(lam, a)
            grad = 2.0 * hankel_matvec_fft(lam, a_inner)
            # Projected gradient step with adaptive step size
            lr = 1.0 / (float(np.max(np.abs(grad))) + 1e-10)
            a_new = a_inner - lr * grad
            a_new = project_simplex(a_new)
            # Check improvement
            obj_old = float(a_inner @ hankel_matvec_fft(lam, a_inner))
            obj_new = float(a_new @ hankel_matvec_fft(lam, a_new))
            if obj_new < obj_old:
                a_inner = a_new
            else:
                lr *= 0.5
                a_new = project_simplex(a_inner - lr * grad)
                a_inner = a_new

        a = a_inner

        # Compute autocorrelation
        c = full_convolve_fft(a)

        # Dual update: multiplicative weights
        c_norm = c / (float(c.max()) + 1e-10)
        lam = lam * np.exp(0.5 * c_norm)
        lam = lam / lam.sum()

        # Track best
        sc = float(2.0 * n * c.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adapt eta
        if outer % 50 == 0 and outer > 0:
            eta *= 0.95

    return best_a, best_score


def hankel_matvec_fft(lam, a):
    """Compute (H a)_i = sum_j lam_{i+j} * a_j using FFT."""
    n = len(a)
    # d_i = sum_j a[j] * lam[i+j] = convolve(a[::-1], lam)[n-1 : 2n-1]
    a_rev = a[::-1]
    N = next_pow2(n + len(lam) - 1)
    A = np.fft.rfft(a_rev, N)
    L = np.fft.rfft(lam, N)
    conv = np.fft.irfft(A * L, N)
    return conv[n - 1:2 * n - 1]


def coordinate_descent_exact(a0, deadline, rng):
    """Coordinate descent with exact 1D minimization per coordinate."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())
    no_imp = 0

    n_sweeps = max(2, int((deadline - time.monotonic()) * 1.5))

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]

            def obj_1d(x):
                if x < 0:
                    x = 0.0
                a[i] = x
                s = a.sum()
                if s < 0.01:
                    return 1e10
                a_norm = a / s
                c = full_convolve_fft(a_norm)
                return 2.0 * n * float(c.max())

            # Use golden section search in a neighborhood
            lo = max(0.0, orig * 0.1)
            hi = orig * 5.0
            if hi < 1e-12:
                hi = 1.0
            res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            if res.fun < best_score - 1e-12:
                a[i] = res.x
                a = a / a.sum()
                best_score = res.fun
                best_a = a.copy()
                improved = True
                no_imp = 0
            else:
                a[i] = orig
                no_imp += 1

        if not improved:
            break
        if no_imp > n:
            break

    return best_a, best_score


def make_deterministic_profiles(n):
    profs = []
    t = np.linspace(0.0, 1.0, n)
    # Linear profiles
    for eta in (1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        a = np.maximum(1.0 + eta * (t - 0.5), 1e-8)
        profs.append(a / a.sum())
    # Exponential
    for eta in (1.0, 2.0, 3.0, 5.0, 8.0):
        a = np.exp(eta * t)
        profs.append(a / a.sum())
    # Power
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = t ** p + 1e-8
        profs.append(a / a.sum())
    # Sigmoid
    for eta in (3.0, 6.0, 10.0, 15.0):
        a = 1.0 / (1.0 + np.exp(-eta * (t - 0.5)))
        profs.append(a / a.sum())
    # Cubic
    u = 2.0 * t - 1.0
    for a1, a3 in ((1.5, -1.0), (2.0, -2.0), (2.5, -3.0), (1.0, -0.5)):
        a = np.maximum(1.0 + a1 * u + a3 * u ** 3, 1e-8)
        profs.append(a / a.sum())
    # Gaussian centered
    for sigma in (0.2, 0.3, 0.4, 0.5):
        a = np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2))
        profs.append(a / a.sum())
    # Asymmetric: high on one side
    for alpha in (2.0, 3.0, 5.0):
        a = np.exp(alpha * (t - 1.0))
        profs.append(a / a.sum())
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def soft_max_adam_v2(p0, deadline, rng, lr=0.05, beta_init_frac=0.15):
    """Improved soft-max Adam with Nesterov-like momentum."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        w = soft_max_weights(c, beta)
        L = float(np.dot(w, c))
        h = 2.0 * corr_weights(w, p)
        vp = h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Adaptive beta cooling
        if adam_t % 50 == 0:
            beta = max(1e-14, beta * 0.6)

        # Restart
        if no_imp > max(400, 5 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            no_imp = 0

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining L-BFGS-B, Nesterov subgradient,
    multiplicative weights dual, and exact coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        p = np.asarray(p, dtype=np.float64)
        if sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}]")
        if len(candidates) < 30:
            candidates.append((float(sc), p.copy()))
            candidates.sort(key=lambda x: x[0])
        elif float(sc) < candidates[-1][0]:
            candidates[-1] = (float(sc), p.copy())
            candidates.sort(key=lambda x: x[0])

    # Try to load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        parent = None

    # Phase 0: Initialization (50s)
    print("=== Phase 0: Initialization ===")
    phase0_end = min(deadline, t_start + 50.0)

    if parent is not None:
        sc = compute_score_normalized(parent)
        local_update(parent, sc, "parent")
        for tn in (128, 256, 512):
            if parent.size != tn and time.monotonic() < phase0_end:
                p_i = interpolate_profile(parent, tn)
                if p_i is not None:
                    local_update(p_i, compute_score_normalized(p_i), f"parent_n{tn}")

        for p0 in make_deterministic_profiles(n_init)[:12]:

                break
            local_update(p0, compute_score_normalized(p0), f"init_n{n_init}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: L-BFGS-B at n=128 (300s)
    print("=== Phase 1: L-BFGS-B at n=128 ===")
    n1 = 128

    # Random smooth starts
    for _ in range(10):

        a = 1.0 + rng.uniform(0.5, 3.0) * t + rng.uniform(-1, 1) * t**2
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
        starts_128.append(np.log(a))

    for si, z0 in enumerate(starts_128[:12]):
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + 20.0)
        try:

    # Phase 2: Soft-max Adam at n=128 (300s)
    print("=== Phase 2: Adam at n=128 ===")
