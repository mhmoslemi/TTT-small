    while k < n:

    return max(2, k)

    n = a.size
    if n <= 64:

    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def soft_max_weights(c, beta):

    mval = c.max()
    x = (c - mval) / beta

        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def compute_score(a):

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:

    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution via FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_weights(c, beta):
    """Compute smooth max value and softmax weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def cross_corr(w, a):
    """Compute h[i] = sum_j w[i+j] * a[j] for i=0..n-1 via FFT."""
    n = a.size
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]


def compute_grad_log(a, beta, lambda_l2=0.0):
    """Compute log-space gradient of smooth_max(conv) + lambda*||conv||^2."""
    n = a.size
    c = full_convolve_fft(a)
    _, w = soft_max_weights(c, beta)
    da_grad = 2.0 * cross_corr(w, a)

        da_grad = da_grad + lambda_l2 * da_grad_l2
    dot = float(np.dot(a, da_grad))
    dz = a * (da_grad - dot)
    dz -= float(dz.mean())

    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else np.ones(n_new) / n_new


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    # Cosine profiles
    for k in [1, 2, 3, 5]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    # U-shape (edge-weighted)
    a = 0.3 + 0.7 * (2 * t - 1) ** 2
    profiles.append(a / a.sum())
    # Inverted U (center-weighted)
    a = 1.0 - 0.5 * (2 * t - 1) ** 2
    a = np.maximum(a, 0.01)
    profiles.append(a / a.sum())
    # Random smooth
    for _ in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 6)
        a = 1.0 + coeffs[0]
        for i in range(1, 6):
            a = a + coeffs[i] * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    # Triangular (worst case baseline)
    a = 1.0 - np.abs(2 * t - 1)
    a = np.maximum(a, 0.01)
    profiles.append(a / a.sum())
    return profiles


def softmax_adam_optimize(a_init, deadline, rng, lr=0.06, beta_ratio=0.25,
                          lambda_l2=0.1, restart_stale=150, verbose=False):
    """
    Core soft-max Adam optimizer in log-space with adaptive temperature.
    Returns (best_a, best_score).
    """
    a = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 1e-15)
    s = a.sum()
    if s < 1e-10:
        return a_init.copy(), np.inf
    a = a / s
    n = a.size

    z = np.log(a)
    z -= z.mean()

    c = full_convolve_fft(a)
    beta = beta_ratio * max(c.max(), 1e-10)
    base_beta = beta

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_score = 2.0 * n * float(c.max())
    stale = 0
    iteration = 0

    while time.monotonic() < deadline:
        dz, c = compute_grad_log(a, beta, lambda_l2)

        if not np.all(np.isfinite(dz)):
            z = np.log(best_a) - float(np.log(best_a).mean())
            a = best_a.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * dz
        v = b2 * v + (1 - b2) * dz * dz
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= float(z.mean())

        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        score = 2.0 * n * float(c.max())

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        iteration += 1

        # Temperature decay
        if iteration % 40 == 0:
            beta = max(1e-12, beta * 0.88)

        # Restart if stale
        if stale > restart_stale:
            strategy = rng.integers(0, 6)
            if strategy == 0:
                z = best_a_log = np.log(best_a)
                z = z - float(z.mean())
                z = z + 0.3 * rng.normal(size=n)
            elif strategy == 1:
                z += 0.8 * rng.normal(size=n)
            elif strategy == 2:
                # Morph toward random cosine
                t_arr = np.linspace(0, 1, n)
                k = rng.integers(1, 7)
                target = 1.0 + 0.3 * np.cos(2 * np.pi * k * t_arr + rng.uniform(0, 2 * np.pi))
                target = np.maximum(target, 0.01)
                z_target = np.log(target) - float(np.log(target).mean())
                alpha = float(rng.uniform(0.4, 0.8))
                z = (1 - alpha) * z + alpha * z_target
            elif strategy == 3:
                # Peak shaving
                peak_idx = int(c.argmax())
                z_best = np.log(best_a) - float(np.log(best_a).mean())
                z = z_best.copy()
                for j in range(n):
                    k2 = peak_idx - j
                    if 0 <= k2 < n:
                        z[j] -= 0.2
                        z[k2] -= 0.2
                z -= float(z.mean())
            elif strategy == 4:
                # Anti-peak: boost elements that DON'T contribute to peak
                peak_idx = int(c.argmax())
                contributing = set()
                for j in range(n):
                    k2 = peak_idx - j
                    if 0 <= k2 < n:
                        contributing.add(j)
                        contributing.add(k2)
                non_contrib = [j for j in range(n) if j not in contributing]
                if non_contrib:
                    for j in non_contrib:
                        z[j] += 0.15
                z -= float(z.mean())
            else:
                # Full restart with noise
                a_new = best_a.copy()
                a_new += 0.1 * rng.normal(size=n)
                a_new = np.maximum(a_new, 1e-6)
                a_new = a_new / a_new.sum()
                z = np.log(a_new) - float(np.log(a_new).mean())

            z = np.clip(z, -40.0, 40.0)
            z -= float(z.mean())
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            beta = beta_ratio * max(float(c.max()), 1e-10)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = min(0.12, lr * 1.3)
            stale = 0

        # Adaptive lambda_l2: reduce it as beta decreases
        if iteration % 200 == 0:
            lambda_l2 = max(0.0, lambda_l2 * 0.7)

        if verbose and iteration % 500 == 0:
            print(f"    iter={iteration}, score={best_score:.10f}, beta={beta:.2e}, "
                  f"stale={stale}, lr={lr:.4f}")

    return best_a, best_score


def lp_refine(f_init, deadline, rng, verbose=False):
    """LP-based refinement: find g to minimize max(conv(f,g)) subject to constraints."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0)
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()

    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    t = 0.2
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        g = None
        try:
            c_obj = -np.ones(n)
            b_ub = np.full(2 * n - 1, best_M)
            bounds = [(0.0, None)] * n
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t = max(0.005, t * 0.5)
            continue
        f_new = f_new / s_new

        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-10:
            f = f_new
            if M_new < best_M - 1e-10:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
                # Update A matrix for new f
                for j in range(n):
                    A[j:j + n, j] = f
            else:
                stagnant += 1
        else:
            t = max(0.005, t * 0.5)
            stagnant += 1

        iteration += 1
        t = max(0.003, t * 0.997)

        if verbose and iteration % 20 == 0:
            print(f"    LP iter={iteration}, M={best_M:.10f}, t={t:.4f}")

        if stagnant > 50:
            f = best_f + 0.05 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            for j in range(n):
                A[j:j + n, j] = f
            t = 0.2
            stagnant = 0

    return best_f


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: diverse initialization, soft-max Adam optimization,
    LP refinement, multi-scale interpolation, and final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    print(f"Starting search. budget={budget_s}s, seed={seed}")

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc = compute_score(init_seq)
                update(init_seq, sc)
                print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc:.10f}")
    except Exception:
        pass

    # ============ Phase 1: Optimize at n=512 from multiple starts ============
    n_main = 512
    phase1_end = min(deadline, t_start + 500.0)

    if best_seq is not None and best_seq.size != n_main:
        p_main = interpolate_profile(best_seq, n_main)
    elif best_seq is not None:
        p_main = best_seq.copy()
    else:
        p_main = make_profiles(n_main, rng)[4]  # U-shape as default

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1: n={n_main}, initial score={sc:.10f}")

    # Main optimization run from best initial
    per_deadline = min(phase1_end, time.monotonic() + 250.0)
    result, sc = softmax_adam_optimize(p_main, per_deadline, rng,
                                        lr=0.06, beta_ratio=0.25,
                                        lambda_l2=0.1, restart_stale=120)
    update(result, sc)
    print(f"  After main Adam: score={sc:.10f}")

    # Try other starting profiles
    profiles = make_profiles(n_main, rng)
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end - 10:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.1:
            continue
        per_dl = min(phase1_end, time.monotonic() + 30.0)
        result, sc = softmax_adam_optimize(p, per_dl, rng,
                                            lr=0.05, beta_ratio=0.3,
                                            lambda_l2=0.1, restart_stale=100)
        update(result, sc)
        if sc < best_score + 0.01:
            print(f"  Profile {idx}: score={sc:.10f}")

    # LP refinement on best
    if time.monotonic() < phase1_end - 5:
        per_dl = min(phase1_end, time.monotonic() + 80.0)
        result = lp_refine(best_seq, per_dl, rng)
        sc = compute_score(result)
        update(result, sc)
        print(f"  After LP: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: Optimize at n=1024 ============
    n_large = 1024
    phase2_end = min(deadline, t_start + 1050.0)

    if time.monotonic() < phase2_end - 20 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 2: n={n_large}, initial score={sc:.10f}")

        per_deadline = min(phase2_end, time.monotonic() + 400.0)
        result, sc = softmax_adam_optimize(p_large, per_deadline, rng,
                                            lr=0.05, beta_ratio=0.25,
                                            lambda_l2=0.1, restart_stale=150)
        update(result, sc)
        print(f"  After large Adam: score={sc:.10f}")

        # Quick LP on large
        if time.monotonic() < phase2_end - 10:
            per_dl = min(phase2_end, time.monotonic() + 80.0)
            result = lp_refine(best_seq, per_dl, rng)
            sc = compute_score(result)
            update(result, sc)
            print(f"  After large LP: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Return to best size and diverse optimization ============
    phase3_end = min(deadline, t_start + 1550.0)
    n_best = best_seq.size if best_seq is not None else n_main

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        # Multiple diverse runs at the best size
        for run_id in range(4):
            if time.monotonic() > phase3_end - 10:
                break
            # Different perturbation for each run
            a_pert = best_seq.copy()
            noise_scale = 0.02 * (run_id + 1)
            a_pert = a_pert + noise_scale * rng.normal(size=n_best)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()

            # Different hyperparameters per run
            lr_val = 0.04 + 0.01 * run_id
            beta_r = 0.2 + 0.05 * run_id
            lam = 0.05 * (run_id + 1)

            per_dl = min(phase3_end, time.monotonic() + 100.0)
            result, sc = softmax_adam_optimize(a_pert, per_dl, rng,
                                                lr=lr_val, beta_ratio=beta_r,
                                                lambda_l2=lam, restart_stale=100)
            update(result, sc)
            print(f"  Phase3 run {run_id}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Final polish with very small learning rate ============
    phase4_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        # Very fine polish
        for polish_round in range(3):
            if time.monotonic() > phase4_end - 5:
                break
            a_polish = best_seq + 0.001 * rng.normal(size=best_seq.size)
            a_polish = np.maximum(a_polish, 1e-8)
            a_polish = a_polish / a_polish.sum()

            per_dl = min(phase4_end, time.monotonic() + 60.0)
            result, sc = softmax_adam_optimize(a_polish, per_dl, rng,
                                                lr=0.015, beta_ratio=0.15,
                                                lambda_l2=0.02, restart_stale=300)
            update(result, sc)
            print(f"  Polish round {polish_round}: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Final validation ============
    if best_seq is None:
        best_seq = make_profiles(512, rng)[0]

    final_seq = np.maximum(best_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
