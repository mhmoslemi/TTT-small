    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_M(f):
    """Compute max of linear convolution."""
    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def interp_profile(f, n_new):
    """Interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def make_cosine_profile(n, gamma=1.0):
    """Raised cosine profile."""
    t = np.linspace(0, 1, n)
    f = 0.5 * (1 + np.cos(np.pi * (2 * t - 1) * gamma))
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_linear_profile(n, slope=2.0):
    """Linear ramp."""
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    """Cubic profile."""
    t = np.linspace(0, 1, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_profile(n, mu=0.5, sigma=0.3):
    t = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n):
    t = np.linspace(0, 1, n)
    f = np.exp(-0.5 * (t / 0.12) ** 2) + np.exp(-0.5 * ((t - 1) / 0.12) ** 2) + 0.05
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_fourier_random(n, rng, n_harm=6, amp=0.4):
    """Random smooth Fourier profile."""
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, n_harm + 1):
        a = rng.uniform(-amp, amp)
        b = rng.uniform(-amp, amp)
        f = f + a * np.cos(2 * np.pi * k * t) + b * np.sin(2 * np.pi * k * t)
    f = np.maximum(f, 1e-8)
    return f / f.sum()


def get_starts(n, rng, parent=None, count=12):
    """Generate diverse starting points."""
    starts = []
    starts.append(make_cosine_profile(n, 0.8))
    starts.append(make_cosine_profile(n, 1.0))
    starts.append(make_cosine_profile(n, 1.2))
    starts.append(make_linear_profile(n, 1.5))
    starts.append(make_linear_profile(n, 2.5))
    starts.append(make_cubic_profile(n, 1.5, -1.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    starts.append(make_gaussian_profile(n, 0.5, 0.35))
    starts.append(make_dual_bump(n))
    for _ in range(count):
        starts.append(make_fourier_random(n, rng, n_harm=5, amp=0.35))
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(interp_profile(parent, n))
    return starts


def compute_gradient_fft(f, w, N=None):
    """
    Compute grad[i] = sum_j f(j) * w(i+j) for j=0..n-1, i=0..n-1.
    Uses FFT. w has length 2n-1 (the convolution weights).
    """
    n = len(f)
    if N is None:
        N = 1
        while N < 3 * n - 1:
            N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(f[::-1], N)
    full = np.fft.irfft(fw * fp, N)
    # grad[i] = full[n-1+i]
    return full[n - 1: n - 1 + n]


def adam_optimize(f_init, deadline, rng, base_lr=0.06, verbose=False):
    """
    Adam optimization in log-space with multi-temperature log-sum-exp.
    Returns (best_f, best_M).
    """
    n = len(f_init)
    f = np.maximum(f_init, 1e-15)
    f = f / f.sum()
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.75, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    beta_base = 0.08 * M

    best_f = f.copy()
    best_M = M
    no_imp = 0
    lr = base_lr

    # Precompute FFT size
    N = 1
    while N < 3 * n - 1:
        N <<= 1

    iter_count = 0
    while time.monotonic() < deadline:
        t += 1
        iter_count += 1

        c = np.convolve(f, f)
        m_val = float(c.max())
        M = m_val

        # Multi-temperature gradient
        beta = beta_base * max(0.5, 0.98 ** (t // 200))
        # Primary gradient at current beta
        x = (c - m_val) / beta
        x_max = x.max()
        exp_x = np.exp(x - x_max)
        w1 = exp_x / exp_x.sum()

        # Secondary gradient at 3x beta (flatter signal)
        beta2 = beta * 3.0
        x2 = (c - m_val) / beta2
        x2_max = x2.max()
        exp_x2 = np.exp(x2 - x2_max)
        w2 = exp_x2 / exp_x2.sum()

        # Combine weights
        w_comb = 0.6 * w1 + 0.4 * w2

        # Compute gradient in f-space
        grad = compute_gradient_fft(f, w_comb, N)

        # Project to zero-sum in log-space
        dot = float(np.dot(f, grad))
        gz = f * (grad - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        z = np.clip(z, -80, 80)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()

        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 5e-12:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)

        # Adaptive beta decrease
        if t % 150 == 0:
            beta_base *= 0.85
            beta_base = max(beta_base, 1e-12 * M)

        # Restart on stall
        if no_imp > 200:
            z = np.log(best_f) + 0.25 * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta_base = max(1e-12, 0.1 * M)
            m.fill(0)
            v.fill(0)
            t = 0
            lr = base_lr
            no_imp = 0

        if verbose and iter_count % 1000 == 0:
            score_cur = 2.0 * n * M
            print(f"    adam iter {iter_count}: score={score_cur:.10f}, best={2*n*best_M:.10f}")

    return best_f, best_M


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def lp_step(f, M, deadline, rng):
    """
    Single LP-based step: solve for g maximizing sum(g) s.t. f*g <= M, g >= 0.
    Blend f with normalized g using line search.
    Returns (new_f, new_M, improved).
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
        if not res.success or res.x is None:
            return f, M, False
        g = np.maximum(res.x, 0.0)
    except Exception:
        return f, M, False

    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return f, M, False
    g = g / g_sum

    # Line search
    best_f_try = f
    best_M_try = M
    for t_val in [0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.3, 0.2]:
        f_try = (1.0 - t_val) * f + t_val * g
        f_try = normalize_nonneg(f_try)
        if f_try is None:
            continue
        M_try = compute_M(f_try)
        if M_try < best_M_try - 1e-13:
            best_M_try = M_try
            best_f_try = f_try

    if best_M_try < M - 1e-13:
        return best_f_try, best_M_try, True
    return f, M, False


def lp_optimize(f_init, deadline, rng, max_stall=50):
    """LP-based iterative optimization."""
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    while time.monotonic() < deadline:
        f_new, M_new, improved = lp_step(f, M, deadline, rng)
        if improved:
            f = f_new
            M = M_new
            if M < best_M - 1e-13:
                best_M = M
                best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max_stall:
                # Perturb
                f = 0.8 * best_f + 0.2 * normalize_nonneg(best_f + 0.2 * rng.normal(size=n))
                if f is None:
                    f = best_f.copy()
                M = compute_M(f)
                no_imp = 0

    return best_f, best_M


def coordinate_polish(f, deadline, rng, n_trials=50000):
    """
    Coordinate-wise greedy polish: try small changes to individual coordinates.
    """
    n = len(f)
    f = f.copy()
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    trial = 0

    # Precompute for fast evaluation
    while time.monotonic() < deadline and trial < n_trials:
        # Pick a random coordinate
        i = rng.integers(0, n)
        # Try a few perturbation sizes
        for delta_frac in [0.01, -0.01, 0.05, -0.05, 0.005, -0.005]:
            f_try = f.copy()
            f_try[i] = max(0.0, f_try[i] * (1.0 + delta_frac))
            s = f_try.sum()
            if s < 1e-30:
                continue
            # No need to normalize since score is scale-invariant
            c = np.convolve(f_try, f_try)
            M_try = float(c.max()) / (s * s)
            if M_try < M - 1e-14:
                f = f_try
                M = M_try
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
                break
        else:
            no_imp += 1
        trial += 1

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Adam + LP + coordinate polish.
    Tries multiple lengths and starting points.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.array(f, dtype=np.float64)
            best_score = float(sc)

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass

    if parent is not None:
        print(f"Parent sequence: n={len(parent)}, score={compute_score(parent):.10f}")
        try_update(parent, compute_score(parent))

    # Define lengths to try (prioritize 256 since parent is 256)
    lengths = [256, 384, 128, 512]

    # Phase 1: Adam optimization from multiple starts at n=256
    n1 = 256
    phase1_end = min(deadline, t_start + 600.0)
    print(f"Phase 1: Adam at n={n1}, budget={phase1_end - t_start:.0f}s")

    starts = get_starts(n1, rng, parent, count=10)
    results_phase1 = []

    for i, s in enumerate(starts):
        remaining = phase1_end - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining / max(1, len(starts) - i), 30.0)
        if alloc < 3:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = adam_optimize(s, run_dl, rng, base_lr=0.05)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_phase1.append((sc, f_best))
        if (i + 1) % 3 == 0:
            print(f"  Phase 1 {i+1}/{len(starts)}: best={best_score:.10f}")

    results_phase1.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")

    # Phase 2: LP refinement on top Adam results
    phase2_end = min(deadline, t_start + 1000.0)
    print(f"Phase 2: LP refinement, budget={phase2_end - t_start:.0f}s")

    top_k = min(5, len(results_phase1))
    results_phase2 = []
    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining / max(1, top_k - i), 60.0)
        if alloc < 5:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = lp_optimize(results_phase1[i][1], run_dl, rng)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_phase2.append((sc, f_best))
        print(f"  Phase 2 LP {i+1}/{top_k}: score={sc:.10f}, best={best_score:.10f}")

    results_phase2.sort(key=lambda x: x[0])

    # Phase 3: Adam polish on LP results
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"Phase 3: Adam polish on LP results, budget={phase3_end - t_start:.0f}s")

    top_k3 = min(4, len(results_phase2))
    results_phase3 = []
    for i in range(top_k3):
        remaining = phase3_end - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining / max(1, top_k3 - i), 40.0)
        if alloc < 5:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = adam_optimize(results_phase2[i][1], run_dl, rng, base_lr=0.03)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results_phase3.append((sc, f_best))
        print(f"  Phase 3 Adam {i+1}/{top_k3}: score={sc:.10f}, best={best_score:.10f}")

    results_phase3.sort(key=lambda x: x[0])

    # Phase 4: Try n=384
    n2 = 384
    phase4_end = min(deadline, t_start + 1550.0)
    remaining_for_384 = phase4_end - time.monotonic()
    if remaining_for_384 > 60:
        print(f"Phase 4: n={n2}, budget={remaining_for_384:.0f}s")
        # Interpolate best solution to n=384
        best_src = results_phase3[0][1] if results_phase3 else (
            results_phase2[0][1] if results_phase2 else (
                results_phase1[0][1] if results_phase1 else make_cosine_profile(n1)))
        f_ext = interp_profile(best_src, n2)
        # Adam at n=384
        alloc = min(remaining_for_384 * 0.6, 120.0)
        run_dl = time.monotonic() + alloc
        f_best, M_best = adam_optimize(f_ext, run_dl, rng, base_lr=0.04)
        sc = 2.0 * n2 * M_best
        try_update(f_best, sc)
        print(f"  Phase 4 Adam n={n2}: score={sc:.10f}, best={best_score:.10f}")
        # LP at n=384
        remaining = phase4_end - time.monotonic()
        if remaining > 20:
            f_best2, M_best2 = lp_optimize(f_best, time.monotonic() + min(remaining - 5, 40.0), rng)
            sc2 = 2.0 * n2 * M_best2
            try_update(f_best2, sc2)
            print(f"  Phase 4 LP n={n2}: score={sc2:.10f}, best={best_score:.10f}")
    else:
        print(f"Phase 4: skipped (only {remaining_for_384:.0f}s left)")

    # Phase 5: Try n=128 (sometimes shorter sequences have better ratio)
    n3 = 128
    phase5_end = min(deadline, t_start + 1650.0)
    remaining_for_128 = phase5_end - time.monotonic()
    if remaining_for_128 > 30:
        print(f"Phase 5: n={n3}, budget={remaining_for_128:.0f}s")
        best_src5 = results_phase3[0][1] if results_phase3 else make_cosine_profile(n1)
        f_ext5 = interp_profile(best_src5, n3)
        alloc5 = min(remaining_for_128 * 0.7, 40.0)
        run_dl5 = time.monotonic() + alloc5
        f_best5, M_best5 = adam_optimize(f_ext5, run_dl5, rng, base_lr=0.05)
        sc5 = 2.0 * n3 * M_best5
        try_update(f_best5, sc5)
        print(f"  Phase 5 n={n3}: score={sc5:.10f}, best={best_score:.10f}")
    else:
        print(f"Phase 5: skipped")

    # Phase 6: Coordinate-wise polish on the best solution
    phase6_end = min(deadline, t_start + budget_s - 60.0)
    if best_seq is not None:
        remaining6 = phase6_end - time.monotonic()
        if remaining6 > 10:
            print(f"Phase 6: Coordinate polish, budget={remaining6:.0f}s")
            f_polish, M_polish = coordinate_polish(best_seq, time.monotonic() + remaining6 - 5, rng)
            sc_polish = compute_score(f_polish)
            try_update(f_polish, sc_polish)
            print(f"  Phase 6 result: score={sc_polish:.10f}, best={best_score:.10f}")

    # Phase 7: Final LP on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        remaining7 = deadline - 20 - time.monotonic()
        if remaining7 > 15:
            print(f"Phase 7: Final LP at n={n_best}, budget={remaining7:.0f}s")
            f_final, M_final = lp_optimize(best_seq, time.monotonic() + remaining7 - 5, rng)
            sc_final = 2.0 * n_best * M_final
            try_update(f_final, sc_final)
            print(f"  Phase 7 result: score={sc_final:.10f}, best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = make_cosine_profile(256, 1.0)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_cosine_profile(256, 1.0)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}, best_found={best_score:.10f}")

    # Check with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
