def next_pow2(n):

def full_convolve_fft(a):

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines improved LP-based descent with multi-seed Adam polishing,
diverse initializations, and multi-resolution search to beat 1.5027.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, step0=0.25, verbose=False):
    """LP-based descent optimization."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    best_M = float(full_convolve_fft(f).max())
    best_f = f.copy()
    t = step0
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        current_M = float(full_convolve_fft(f).max())
        # Use tighter M when stagnant to force exploration
        if stagnant > 20:
            M = current_M
        else:
            M = best_M

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            # Try slightly relaxed M
            g = lp_step(f, M * 1.0005)
            if g is None or g.sum() < 1e-10:
                g = lp_step(f, M * 1.005)
                if g is None or g.sum() < 1e-10:
                    break

        g = g / g.sum()

        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new

        M_new = float(full_convolve_fft(f_new).max())

        if M_new <= best_M + 1e-10:
            f = f_new
            if M_new < best_M - 1e-10:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
                t = min(0.4, t * 1.08)
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 50 == 0:
                print(f"    LP iter {iteration}: M={best_M:.12f}, t={t:.4f}, stag={stagnant}")
        else:
            t = max(0.005, t * 0.6)
            stagnant += 1

        t = max(0.005, t * 0.9985)

        if stagnant > 80:
            noise_scale = 0.03 / np.sqrt(n)
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = step0
            stagnant = 0
            if verbose:
                print(f"    LP restart at iter {iteration}, best_M={best_M:.12f}")

    return best_f


def soft_mx(c, beta):
    """Smooth max via log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -80.0, 80.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def adam_optimize(p0, deadline, rng, lr=0.05, reg=1e-8,
                  beta_init_frac=0.15, beta_decay=0.88,
                  restart_every=400, noise_std=0.15):
    """Adam optimizer in log-space with smooth max objective."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 40 == 0:
            beta = max(1e-14, beta * beta_decay)

        if no_imp > restart_every:
            z = 0.75 * best_z + noise_std * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_exponential_profile(n, rate=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    return a / a.sum()


def make_gaussian_profile(n, sigma=0.3, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def make_power_profile(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.01) ** p
    return a / a.sum()


def make_bell_profile(n):
    """Raised cosine bell."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_mixed_osc_profile(n, k1=2, k2=3, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2 * np.pi * k1 * t) + 0.5 * amp * np.cos(2 * np.pi * k2 * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Cubic-like interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def get_diverse_profiles(n, rng):
    """Generate a diverse set of starting profiles."""
    profiles = []
    # Linear family
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(make_linear_profile(n, eta))
    # Cosine family
    for k in [1, 2, 3, 4]:
        profiles.append(make_cosine_profile(n, k))
    # Exponential
    for rate in [1.0, 2.0, 3.0, 4.0]:
        profiles.append(make_exponential_profile(n, rate))
    # Gaussian
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        profiles.append(make_gaussian_profile(n, sigma))
    # Power
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(make_power_profile(n, p))
    # Bell
    profiles.append(make_bell_profile(n))
    # Mixed oscillatory
    profiles.append(make_mixed_osc_profile(n, 2, 3, 0.3))
    profiles.append(make_mixed_osc_profile(n, 1, 4, 0.2))
    profiles.append(make_mixed_osc_profile(n, 3, 5, 0.25))
    # Random smooth (Fourier)
    for _ in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 6)
        t = np.linspace(0.0, 1.0, n)
        a = 1.0
        for i, c_val in enumerate(coeffs[1:], 1):
            a = a + c_val * np.cos(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Random positive (smoothed)
    for _ in range(4):
        raw = rng.uniform(0.3, 1.0, n)
        # Smooth with a small kernel
        kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
        smoothed = np.convolve(raw, kernel, mode='same')
        smoothed = np.maximum(smoothed, 1e-12)
        profiles.append(smoothed / smoothed.sum())
    return profiles


def coordinate_descent(p0, deadline, rng, max_rounds=5000):
    """Simple coordinate descent: try small adjustments to individual elements."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_score = float(2.0 * n * full_convolve_fft(p).max())
    best_p = p.copy()

    step_sizes = [0.001, 0.005, 0.01, 0.02, 0.05]

    for _ in range(max_rounds):
        if time.monotonic() > deadline:
            break
        i = rng.integers(n)
        old_val = p[i]
        improved = False
        for ss in step_sizes:
            for direction in [1.0, -1.0]:
                p_new = p.copy()
                p_new[i] = old_val + direction * ss * old_val
                if p_new[i] < 0:
                    p_new[i] = 1e-12
                p_new = p_new / p_new.sum()
                sc = float(2.0 * n * full_convolve_fft(p_new).max())
                if sc < best_score - 1e-11:
                    p = p_new
                    best_score = sc
                    best_p = p.copy()
                    improved = True
                    break
            if improved:
                break
    return best_p, best_score

            {'lr': 0.05, 'beta_decay': 0.88, 'restart_every': 300, 'noise_std': 0.15},
            {'lr': 0.03, 'beta_decay': 0.90, 'restart_every': 400, 'noise_std': 0.10},
            {'lr': 0.08, 'beta_decay': 0.85, 'restart_every': 250, 'noise_std': 0.20},
            {'lr': 0.02, 'beta_decay': 0.92, 'restart_every': 500, 'noise_std': 0.08},
        ]

        for cfg in adam_configs:
            if time.monotonic() > phase3_end - 10:
                break
            per_dead = min(phase3_end, time.monotonic() + 100.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.0005 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = adam_optimize(p0, per_dead, rng, **cfg)
            update(result, sc)
            print(f"  Adam lr={cfg['lr']}: {sc / (2.0*n_polish) * 2.0 * n_polish / compute_score(result) * compute_score(result):.10f} -> {compute_score(result):.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ===== Phase 4: Coordinate descent + fine Adam =====
    phase4_end = t_start + 1700.0

    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_cd = best_seq.size
        print(f"Phase 4: Coordinate descent at n={n_cd}")

        per_dead = min(phase4_end, time.monotonic() + 150.0)
        result, sc = coordinate_descent(best_seq, per_dead, rng, max_rounds=3000)
        update(result, sc)
        print(f"  After CD: {sc:.10f}")

        # Final fine Adam
        if time.monotonic() < phase4_end - 20:
            per_dead2 = min(phase4_end, time.monotonic() + 80.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.0002 * rng.normal(size=n_cd)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = adam_optimize(p0, per_dead2, rng, lr=0.01,
                                       beta_decay=0.93, restart_every=600, noise_std=0.05)
            update(result, sc)
            print(f"  Fine Adam: {sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ===== Phase 5: Try n=2048 if time and score warrant =====
    phase5_end = min(deadline - 5.0, t_start + 1770.0)

    if (time.monotonic() < phase5_end - 40 and best_seq is not None
            and best_score > 1.5027):
        n5 = 2048
        p_xl = interpolate_profile(best_seq, n5)
        sc = compute_score(p_xl)
        update(p_xl, sc)
        print(f"Phase 5: n={n5}, score={sc:.10f}")

        lp_dead5 = min(phase5_end, time.monotonic() + 80.0)
        result = lp_optimize(p_xl, lp_dead5, rng, step0=0.15, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        print(f"  After xlarge LP: {sc:.10f}")

    print(f"After Phase 5: best={best_score:.10f}")

    # ===== Final validation and cleanup =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        precip = final_seq / s
    else:
        precip = make_linear_profile(256, 1.5)

    final_score = compute_score(precip)
    print(f"FINAL: n={precip.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in precip]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in precip]
