def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def gradient_smooth_max(a, beta):
    """Compute smooth max of conv(a,a) and its gradient wrt a."""
    n = a.size
    c = fft_convolve(a)
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    exp_c = np.exp(np.clip(c_shifted, -50.0, 5.0))
    s_exp = float(exp_c.sum())
    if s_exp < 1e-30:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_c / s_exp
    L = c_max + beta * float(np.log(s_exp))

    # Gradient: grad_a[j] = 2 * sum_m p[j+m] * a[m]
    # Computed via FFT cross-correlation
    N = next_pow2(3 * n)
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fp * np.conj(fa), N)[:n]
    grad_a = 2.0 * h
    return L, grad_a, c


def gradient_descent_logspace(a0, deadline, rng, beta_init=None, lr=0.05,
                              max_iter=200000, anneal_rate=0.9995,
                              restart_threshold=500):
    """
    Gradient descent in log-softmax space using Adam optimizer.
    a0 should be nonnegative with positive sum.
    Returns best_a (normalized to sum=1) and best_M (max of conv).
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-30:
        return a0, np.inf
    a0 = a0 / s0
    n = a0.size

    if beta_init is None:
        M0 = compute_max_conv(a0)
        beta_init = 0.05 * M0 if M0 > 1e-30 else 0.01

    z = np.log(np.maximum(a0, 1e-30))
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0

    beta = float(beta_init)
    best_a = a0.copy()
    best_M = compute_max_conv(a0)
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute a from z (softmax with centering for stability)
        z_centered = z - z.max()
        a = np.exp(z_centered)
        s = a.sum()
        if s < 1e-30:
            z = 0.8 * np.log(np.maximum(best_a, 1e-15)) + 0.5 * rng.normal(size=n)
            z = z - z.mean()
            continue
        a = a / s

        # Compute smooth max and gradient
        c = fft_convolve(a)
        c_max = float(c.max())

        if c_max < best_M - 1e-14:
            best_M = c_max
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Smooth max weights
        c_shifted = (c - c_max) / beta
        exp_c = np.exp(np.clip(c_shifted, -50.0, 5.0))
        s_exp = float(exp_c.sum())
        if s_exp < 1e-30:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = exp_c / s_exp

        # Cross-correlation for gradient
        fp = np.fft.rfft(p, N_corr)
        fa = np.fft.rfft(a, N_corr)
        h = np.fft.irfft(fp * np.conj(fa), N_corr)[:n]
        grad_a = 2.0 * h

        # Convert to z-space gradient (softmax Jacobian)
        # dL/dz_j = a_j * (grad_a_j - sum_l grad_a_l * a_l)
        weighted_mean = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - weighted_mean)

        if not np.all(np.isfinite(grad_z)):
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            m_buf.fill(0)
            v_buf.fill(0)
            t_adam = 0
            no_imp = min(no_imp, restart_threshold // 2)
            continue

        # Adam update
        t_adam += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1**t_adam)
        vh = v_buf / (1.0 - b2**t_adam)

        # Learning rate schedule: warmup then decay
        if t_adam < 100:
            lr_eff = lr * (t_adam / 100.0)
        else:
            lr_eff = lr

        z = z - lr_eff * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()

        # Anneal beta
        if it % 200 == 0:
            beta = max(1e-10, beta * anneal_rate)

        # Restart if stuck
        if no_imp > restart_threshold:
            # Smart restart: perturb toward reducing peak contributions
            peak_idx = int(np.argmax(c))
            # Find which a[j] contribute most to the peak
            contrib = np.abs(a[peak_idx:n] if peak_idx < n else a[:0])
            # Perturb z: push down high contributors, push up low ones
            perturbation = rng.normal(size=n) * 0.3

            z = 0.7 * np.log(np.maximum(best_a, 1e-15)) + 0.3 * z + perturbation

            m_buf.fill(0)
            v_buf.fill(0)
            t_adam = 0
            beta = max(1e-10, 0.05 * compute_max_conv(best_a))
            no_imp = 0

        if no_imp > restart_threshold * 2:

            z = z - z.mean()
            m_buf.fill(0)
            v_buf.fill(0)
            t_adam = 0
            beta = max(1e-10, 0.08 * compute_max_conv(best_a))

def coordinate_descent_polish(a0, deadline, rng, max_sweeps=50):
    """
    Coordinate descent: for each coordinate j, compute the optimal value
    that minimizes the max convolution, given all other coordinates fixed.
    Uses the fact that with sum(a)=1 fixed, changing a_j requires adjusting
    another coordinate. Instead, we use a simpler approach:
    Try small perturbations of individual coordinates and keep improvements.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return a0
    a = a / s

    best_M = compute_max_conv(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try modifying a[j] while keeping sum=1
            # Adjust a[j+1] (mod n) to maintain sum
            for delta_frac in (0.1, -0.1, 0.05, -0.05, 0.2, -0.2):
                a_try = best_a.copy()
                k = (j + 1) % n
                if j == k:
                    k = (j + 2) % n
                a_try[j] += delta_frac * best_a[j]
                a_try[k] -= delta_frac * best_a[j]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-30:
                    continue
                M_try = compute_max_conv(a_try)
                if M_try < best_M - 1e-15:
                    best_M = M_try
                    best_a = a_try
                    improved = True
                    break

        if not improved:
            break

    return best_a


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def multi_start_gradient_search(starts, deadline, rng, n, phase_budget=None):
    """Run gradient descent from multiple starts, return best."""
    if phase_budget is None:
        phase_budget = (deadline - time.monotonic()) / max(1, len(starts))
    
    best_a = None
    best_M = float('inf')

            best_a = p_best
            if i % 5 == 0:
                print(f"    Multi-start {i}: M={M_best:.8f}")

    return best_a, best_M


def peak_targeted_restart(a_best, rng, n, num_peaks=5):
    """Generate perturbations that specifically target reducing peak contributions."""
    c = fft_convolve(a_best)
    peaks = np.argsort(c)[-num_peaks:]
    
    starts = []
    for pk in peaks:
        # Create a perturbation that reduces contribution at peak pk
        a_pert = a_best.copy()
        # Reduce elements that contribute to this peak
        valid_j = np.arange(max(0, pk - n + 1), min(n, pk + 1))
        for j in valid_j:
            a_pert[j] *= (1.0 - rng.uniform(0.05, 0.2))
        # Redistribute
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        starts.append(a_pert)

    # Also try smooth random perturbations
    for _ in range(3):
        a_pert = a_best + rng.uniform(-0.05, 0.05, size=n)
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        starts.append(a_pert)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    1. Gradient descent from parent and diverse starts at n=512
    2. Extended optimization with peak-targeted restarts
    3. Coordinate descent polish
    4. Try n=1024 with interpolated best
    5. Final refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: Gradient descent at n=512 from multiple starts
    n1 = 512
    phase1_end = min(deadline, t_start + 500.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
            # Also try smoothed version
            from scipy.ndimage import uniform_filter1d
            p_smooth = uniform_filter1d(p_int, size=max(3, n1 // 32))
            p_smooth = np.maximum(p_smooth, 1e-10)
            starts.append(p_smooth / p_smooth.sum())

    # Add deterministic profiles
    det_profs = make_deterministic_profiles(n1)
    starts.extend(det_profs[:15])

    # Random smooth profiles
    for _ in range(10):
        starts.append(make_smooth_random(n1, rng, n_harm=12))

    # More aggressive profiles
    for eta in (2.0, 2.5, 3.0, 3.5, 4.0):
        starts.append(make_linear_profile(n1, eta))
    for a1, a3 in ((2.5, -3.5), (3.0, -4.5), (3.5, -5.5), (4.0, -6.0)):
        starts.append(make_cubic_profile(n1, a1, a3))

    print(f"Phase 1: {len(starts)} starts, time budget {phase1_end - t_start:.0f}s")
    best_512, M_512 = multi_start_gradient_search(starts, phase1_end, rng, n1)
    if best_512 is not None:
        sc = 2.0 * n1 * M_512
        try_update(best_512, sc)
        print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Extended optimization on best candidates with different hyperparams
    phase2_end = min(deadline, t_start + 900.0)
    if best_512 is not None:
        # Try different learning rates and beta values
        for lr_val in (0.03, 0.08, 0.12):
            if time.monotonic() > phase2_end:
                break
            per = min(phase2_end, time.monotonic() + 80.0)
            if per - time.monotonic() < 5.0:
                break
            p_best, M_best = gradient_descent_logspace(
                best_512, per, rng, lr=lr_val,
                beta_init=0.03 * compute_max_conv(best_512),
                restart_threshold=300)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"  Phase 2 lr={lr_val}: score={sc:.10f}")

        # Peak-targeted restarts
        restarts = peak_targeted_restart(best_seq, rng, n1, num_peaks=8)
        for rs in restarts[:6]:
            if time.monotonic() > phase2_end:
                break
            per = min(phase2_end, time.monotonic() + 30.0)
            if per - time.monotonic() < 5.0:
                break
            p_best, M_best = gradient_descent_logspace(rs, per, rng, lr=0.04)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)

        print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Coordinate descent polish
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 120.0)
        p_cd = coordinate_descent_polish(best_seq, per, rng, max_sweeps=100)
        sc = compute_score(p_cd)
        try_update(p_cd, sc)
        print(f"After Phase 3 (CD): best={best_score:.10f}")

        # Gradient descent again after CD
        if time.monotonic() < phase3_end - 30:
            per2 = min(phase3_end, time.monotonic() + 60.0)
            p_best, M_best = gradient_descent_logspace(
                p_cd, per2, rng, lr=0.02,
                beta_init=0.02 * compute_max_conv(p_cd))
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"After Phase 3 (GD after CD): best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase4_end - 120:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            sc_ext = compute_score(p_ext)
            try_update(p_ext, sc_ext)
            print(f"Interpolated to n={n2}: score={sc_ext:.10f}")

            per = min(phase4_end, time.monotonic() + 100.0)
            p_best, M_best = gradient_descent_logspace(
                p_ext, per, rng, lr=0.04,
                beta_init=0.04 * compute_max_conv(p_ext))
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 GD (n={n2}): score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                per_cd = min(phase4_end, time.monotonic() + 40.0)
                p_cd2 = coordinate_descent_polish(best_seq, per_cd, rng, max_sweeps=50)
                sc2 = compute_score(p_cd2)
                try_update(p_cd2, sc2)
                print(f"Phase 4 CD (n={n2}): score={sc2:.10f}")

    # Also try direct profiles at n=1024
    if time.monotonic() < deadline - 80:
        for p0 in [make_linear_profile(n2, 2.0), make_cubic_profile(n2, 2.0, -2.5)]:
            if time.monotonic() > deadline - 80:
                break
            per = min(deadline - 80, time.monotonic() + 25.0)
            if per - time.monotonic() < 5.0:
                break
            p_best, M_best = gradient_descent_logspace(p0, per, rng, lr=0.05)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)

    # Phase 5: Final refinement on absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        if per - time.monotonic() > 10:
            # Very fine gradient descent
            p_best, M_best = gradient_descent_logspace(
                best_seq, per, rng, lr=0.01,
                beta_init=0.01 * compute_max_conv(best_seq),
                restart_threshold=2000)
            sc = compute_score(p_best)
            try_update(p_best, sc)
            print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
