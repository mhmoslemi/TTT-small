"""
FFT-accelerated Adam optimization in logit space for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses multiple (n, initialization) configurations
with adaptive softmax temperature and multi-stage time allocation.
"""

import numpy as np
import time


def fft_convolve(a):
    """Compute full linear convolution using FFT for O(n log n) speed."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    c = np.fft.irfft(fa * fa, fsize)
    return c[:size]


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def make_profile(n, etype, param):
    """Generate various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    if etype == 'linear':
        a = 1.0 + param * (2.0 * t - 1.0)
    elif etype == 'power':
        a = t ** param + 1e-10
    elif etype == 'cosine':
        a = np.cos(np.pi * (t - 0.5)) ** 2 + 1e-10
    elif etype == 'sine':
        a = np.sin(np.pi * t ** param) ** 2 + 1e-10
    elif etype == 'exp':
        a = np.exp(param * (t - 0.5))
    elif etype == 'bicubic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u + 0.5 * param * u ** 2 - 0.3 * param * u ** 3
    elif etype == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_seq(a, n_new):
    """Interpolate to new length preserving non-negativity."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize_nonneg(a_new)


def adam_polish(p0, deadline, lr=0.08, reg=1e-10, verbose=True, tag=""):
    """
    Adam gradient descent in logit space with FFT-based convolution.
    Minimizes 2*n*max(conv(p,p)) subject to sum(p)=1, p>=0.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12).astype(np.float64)
    p = p / p.sum()

    z = np.log(p)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = fft_convolve(p)
    max_c = float(c.max())
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-14, 0.15 * max_c)

    while time.monotonic() < deadline:
        it += 1

        c = fft_convolve(p)
        max_c = float(c.max())

        # Softmax weights for differentiating max
        x = (c - max_c) / beta
        x_shifted = x - x.max()
        ex = np.exp(x_shifted)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient: h[i] = sum_j p[j] * w[i+j]
        h = np.empty(n)
        for i in range(n):
            h[i] = float(np.dot(p, w[i:i + n]))

        grad_p = 2.0 * h + reg * p

        # Logit-space gradient (centered for sum constraint)
        dot_val = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot_val)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z -= z.mean()

        p_new = np.exp(z - z.max())
        p_sum = p_new.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p = best_p.copy()
            continue
        p = p_new / p_sum

        c_new = fft_convolve(p)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.001, lr * 0.92)

        # Decay beta to sharpen the softmax
        if it % 300 == 0:
            beta = max(1e-16, beta * 0.7)

        if no_imp > 8000:
            # Restart with perturbation
            rng = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            z = 0.6 * best_z + 0.8 * rng.normal(size=n) * 0.1
            z = np.clip(z, -40, 40)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            p_tmp = np.exp(z - z.max())
            p_tmp = p_tmp / p_tmp.sum()
            beta = max(1e-14, 0.15 * float(fft_convolve(p_tmp).max()))

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"    {tag} iter {it}: score={best_score:.10f}, beta={beta:.2e}, lr={lr:.5f}")

    return best_p, best_score


def lp_direction_fast(f, M, n_constraints_sample=200):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for sampled constraints.
    Uses only a subset of constraints for speed.
    """
    from scipy.optimize import linprog
    n = len(f)
    # Sample constraints: include center and a few others
    all_k = list(range(2 * n - 1))
    if len(all_k) <= n_constraints_sample:
        k_sel = all_k
    else:
        # Include center, edges, and random
        center = n - 1
        k_sel = set([0, 1, 2, n - 2, n - 1, n, 2 * n - 3, 2 * n - 2])
        rng = np.random.default_rng(42)
        extra = rng.choice(2 * n - 1, size=min(200, 2 * n - 1), replace=False)
        k_sel = set(k_sel) | set(extra.tolist())
        k_sel = sorted(k_sel)

    A = np.zeros((len(k_sel), n))
    for idx, k in enumerate(k_sel):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(len(k_sel), M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5.0})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def propose_candidate():
    """
    Multi-stage FFT-accelerated optimization:
    Phase 1: Explore multiple (n, initialization) configurations.
    Phase 2: Polish the best with extended Adam.
    Phase 3: Final fine-tuning with smaller learning rate.
    """
    t_start = time.monotonic()
    total_budget = 1785.0
    deadline_final = t_start + total_budget

    # Load parent sequence if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
    except Exception:
        pass

    best_seq = None
    best_score = np.inf
    best_n = 0

    # Phase 1: Try multiple configurations
    print("Phase 1: Exploring configurations...")
    phase1_deadline = t_start + 600.0
    configs = []

    n_values = [512, 768, 1024]
    if parent is not None:
        # Interpolate parent to various lengths
        for n in n_values:
            p = interpolate_seq(parent, n)
            if p is not None:
                configs.append((n, p, f"parent_n{n}"))

    # Add diverse initializations
    inits = [
        ('linear', 1.5), ('linear', 2.0), ('linear', 1.0),
        ('power', 2.0), ('power', 3.0),
        ('sine', 1.0), ('sine', 1.5),
        ('exp', 1.5), ('exp', 2.5),
        ('bicubic', 1.5), ('bicubic', 2.0),
    ]

    for n in n_values:
        for etype, param in inits:
            p = make_profile(n, etype, param)
            configs.append((n, p, f"{etype}_{param}_n{n}"))

    # Also try parent at different scales of perturbation
    if parent is not None:
        rng0 = np.random.default_rng(123)
        for n in [512, 768]:
            p_base = interpolate_seq(parent, n)
            if p_base is not None:
                for scale in [0.1, 0.3, 0.5]:
                    p_pert = p_base + scale * rng0.normal(size=n) * 0.05
                    p_norm = normalize_nonneg(p_pert)
                    if p_norm is not None:
                        configs.append((n, p_norm, f"parent_pert{n}_s{scale}"))

    # Evaluate all configs quickly and rank
    print(f"  Total configs: {len(configs)}")
    scored_configs = []
    for n, p, tag in configs:
        sc = internal_score(p)
        scored_configs.append((sc, n, p, tag))

    scored_configs.sort(key=lambda x: x[0])

    # Show top 10
    for i, (sc, n, _, tag) in enumerate(scored_configs[:10]):
        print(f"  Top {i+1}: {tag}, n={n}, score={sc:.8f}")

    # Run Adam on top configurations
    top_k = min(6, len(scored_configs))
    per_config_time = (phase1_deadline - time.monotonic()) / top_k - 5.0
    per_config_time = max(per_config_time, 30.0)

    for i in range(top_k):
        if time.monotonic() > phase1_deadline - 10:
            break
        sc0, n, p0, tag = scored_configs[i]
        print(f"\n  Running Adam on {tag} (n={n}, init_score={sc0:.8f}), time_budget={per_config_time:.0f}s")
        t_dead = min(phase1_deadline - 10, time.monotonic() + per_config_time)
        p_best, sc_best = adam_polish(p0, t_dead, lr=0.08, tag=tag)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            best_n = n
            print(f"  New best: {tag}, n={n}, score={best_score:.10f}")

    print(f"\nPhase 1 done. Best: n={best_n}, score={best_score:.10f}")

    # Phase 2: Extended polish on the best
    if best_seq is None:
        best_seq = make_profile(512, 'linear', 1.5)
        best_score = internal_score(best_seq)
        best_n = 512

    print(f"\nPhase 2: Extended polish (n={best_n})...")
    phase2_deadline = min(deadline_final - 300.0, t_start + 1400.0)
    if time.monotonic() < phase2_deadline:
        p_best, sc_best = adam_polish(best_seq, phase2_deadline, lr=0.05, tag="phase2")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 2 best: score={best_score:.10f}")

    # Phase 3: Fine-tuning with smaller learning rate
    print(f"\nPhase 3: Fine-tuning...")
    if time.monotonic() < deadline_final - 30:
        p_best, sc_best = adam_polish(best_seq, deadline_final - 30, lr=0.01, tag="phase3")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 3 best: score={best_score:.10f}")

    # Phase 4: Try a different n if time remains
    remaining = deadline_final - time.monotonic()
    if remaining > 120:
        # Try interpolating best to a different length
        alt_n = best_n + 256 if best_n <= 768 else best_n - 256
        alt_n = max(256, alt_n)
        if alt_n != best_n:
            p_alt = interpolate_seq(best_seq, alt_n)
            if p_alt is not None:
                print(f"\nExtra: Trying n={alt_n}...")
                p_best, sc_best = adam_polish(p_alt, deadline_final - 30, lr=0.03, tag=f"alt_n{alt_n}")
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Alt n={alt_n} best: score={best_score:.10f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_profile(256, 'linear', 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s * 1.0  # Keep normalized (sum=1) so score is 2n*max(c)

    return [float(x) for x in best_seq]
