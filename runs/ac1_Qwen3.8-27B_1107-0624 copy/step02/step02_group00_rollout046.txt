    # This is a 1D convex minimization
    lo = -a[i]
    hi = a[i] * 2  # reasonable upper bound

        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    delta = (lo + hi) / 2
    a[i] += delta
    a[i] = max(a[i], 0.0)

"""
Hybrid LP + Exact Coordinate Descent + Smooth-Max Gradient optimization.
Key innovation: Exact 1D coordinate descent using the analytic structure of the
autocorrelation, combined with LP spreading for coarse optimization.
The coordinate descent minimizes max(conv(a,a))/sum(a)^2 via 1D scalar optimization
per coordinate, exploiting the piecewise-quadratic structure.

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _coord_objective(delta, c, d, e, S):
    """Objective for 1D coordinate optimization: max_k (c[k]+d[k]*delta+e[k]*delta^2) / (S+delta)^2"""
    numer = c + d * delta + e * delta * delta
    denom = (S + delta) ** 2
    if denom < 1e-30:
        return np.inf
    return float(np.max(numer) / denom)


def _coord_descent(a, deadline, max_sweeps=500, random_order=True, rng=None):
    """Exact coordinate descent using 1D scalar optimization per coordinate."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        S = float(a.sum())

        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            # Compute d[k] = a[k-i] + a[i-k] for k in [0, 2n-2]
            d = np.zeros(2 * n - 1)
            # Term 1: a[k-i] nonzero for k in [i, i+n-1]
            d[i:i + n] += a
            # Term 2: a[i-k] nonzero for k in [max(0,i-n+1), i]
            jmax = min(i, n - 1)
            d[i:i - jmax - 1:-1] += a[:jmax + 1]

            # e[k] = 1 if k == 2i, else 0
            e = np.zeros(2 * n - 1)
            if 2 * i <= 2 * n - 2:
                e[2 * i] = 1.0

            # Optimize delta in [-a[i], upper_bound]
            lo = -a[i] + 1e-15
            hi = max(a[i] * 2.0, 0.1)

            # Quick check: try a few points first to narrow the range
            test_deltas = [lo + (hi - lo) * t for t in [0.1, 0.3, 0.5, 0.7, 0.9]]
            test_vals = [_coord_objective(td, c, d, e, S) for td in test_deltas]
            min_test = min(test_vals)
            if min_test >= _coord_objective(0.0, c, d, e, S) - 1e-15:
                # No improvement likely; skip expensive optimization
                continue

            # Use minimize_scalar for the 1D problem
            res = minimize_scalar(
                lambda x: _coord_objective(x, c, d, e, S),
                bounds=(lo, hi),
                method='bounded',
                options={'xatol': 1e-12}

            if res.success:

                a[i] = max(a[i] + delta, 0.0)
                c = _fft_convolve(a)

                    improved = True

                S = float(a.sum())

        else:
            no_imp += 1

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

    base_lr = lr

        if time.monotonic() > deadline:
            break

        a_nest = np.maximum(a + momentum * v, 0)
        a_nest = _normalize(a_nest)
        c = _fft_convolve(a_nest)
        _, w = _smooth_max_weight(c, beta)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h + 1e-8 * a_nest
        grad -= np.mean(grad)

        v = momentum * v - lr * grad
        a_new = _normalize(np.maximum(a + v, 0))
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0

                v = np.zeros(n)

                lr = base_lr
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a

def _interpolate_profile(p0, n_new):

    x0 = np.linspace(0.0, 1.0, n0)

        cd_deadline = min(t_start + 650.0, deadline - 300.0)
        n_cd = best_seq.size
        print(f"Phase 2: Coord descent n={n_cd}, budget~{cd_deadline - time.monotonic():.0f}s")
        a_cd = _coord_descent(best_seq, cd_deadline, max_sweeps=2000, rng=rng)
        sc = _score(a_cd)
        print(f"  CD result: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()

    # Phase 3: More LP + CD alternating
    if best_seq is not None and time.monotonic() < deadline - 200:
        for round_idx in range(3):
            if time.monotonic() > deadline - 200:
                break
            # LP pass
            lp_dead = min(time.monotonic() + 80, deadline - 200)
            a_lp = _lp_refine(best_seq, lp_dead, max_iter=3000, inflate=1.0003, blend=0.2)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Round {round_idx} LP: {sc:.10f}")

            # CD pass
            cd_dead = min(time.monotonic() + 80, deadline - 200)
            if time.monotonic() > deadline - 200:
                break
            a_cd = _coord_descent(best_seq, cd_dead, max_sweeps=500, rng=rng)
            sc = _score(a_cd)
            print(f"  Round {round_idx} CD: {sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_cd.copy()

    # Phase 4: Try different lengths
    if time.monotonic() < deadline - 150:
        for n_try in [256, 768, 1024]:
            if time.monotonic() > deadline - 150:
                break
            if best_seq is not None:
                a0 = _interpolate_profile(best_seq, n_try)
            else:
                a0 = _make_linear_profile(n_try, 1.5)
            sc0 = _score(a0)
            # Quick CD
            cd_dead = min(time.monotonic() + 60, deadline - 150)
            if cd_dead - time.monotonic() < 10:
                break
            a_cd = _coord_descent(a0, cd_dead, max_sweeps=500, rng=rng)
            sc = _score(a_cd)
            print(f"  n={n_try}: init={sc0:.8f} -> CD={sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_cd.copy()

    # Phase 5: Gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_dead = min(time.monotonic() + 80, deadline - 60)
        print(f"Phase 5: Gradient polish")
        a_gp = _gradient_polish(best_seq, gp_dead, max_iter=15000, lr=0.008)
        sc = _score(a_gp)
        print(f"  Gradient result: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_gp.copy()

    # Phase 6: Final coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 20:
        cd_dead = deadline - 20
        print(f"Phase 6: Final CD, budget={cd_dead - time.monotonic():.0f}s")
        a_cd = _coord_descent(best_seq, cd_dead, max_sweeps=3000, rng=rng)
        sc = _score(a_cd)
        print(f"  Final CD: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
