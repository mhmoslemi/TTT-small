def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

"""
Optimized step function coefficient search using smooth gradient descent
with Adam optimizer, random restarts, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_and_weights(c, beta):
    """Compute smooth max and its weights (softmax)."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -500, 500)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    val = mval + beta * np.log(s)
    return val, w


def _gradient_smooth_max(a, beta):
    """Gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_and_weights(c, beta)

    # = 2 * conv(w, a[::-1])[n-1 : 2*n-1]

    conv_result = np.fft.irfft(W * Af, N)
    h = conv_result[n-1 : 2*n-1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        c = _fft_convolve(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0

def _pair_swap_optimize(a, deadline, max_iter=100000):
    """Optimize by random pair swaps (transfer mass between two positions)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Try small transfer from max(i,j) to min(i,j) or vice versa
        # Transfer delta from source to target
        for direction in [0, 1]:
            if direction == 0:
                src, dst = i, j
            else:
                src, dst = j, i
            
            if a[src] <= 1e-15:
                continue
            
            # Try different transfer amounts
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                delta = frac * a[src]
                if delta >= a[src]:
                    continue
                
                a[src] -= delta
                a[dst] += delta
                
                c = _fft_convolve(a)
                sc = float(2.0 * n * c.max())
                
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                else:
                    # Revert
                    a[dst] -= delta
                    a[src] += delta
                    break  # Try next direction
        
        # Periodically reset
        if it % 10000 == 0 and it > 0:
            c = _fft_convolve(a)
            sc = float(2.0 * n * c.max())
            if sc > best_score * 1.01:
                a = best_a.copy()
    
    return best_a


def _make_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Reverse linear
    for eta in [1.0, 2.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Quadratic
    for a2 in [-0.5, -1.0, 0.5, 1.0]:
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + a2 * u * u
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Random smooth
    for _ in range(3):
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.1, 0.4, size=4)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for k in range(4):
            a += amps[k] * np.sin(2 * np.pi * freqs[k] * t + phases[k])
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Interpolated parent
    if parent is not None:
        p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p = np.maximum(p, 1e-10)
        inits.append(p / p.sum())
        
        # Perturbed parent
        for _ in range(3):
            p_pert = p * (1.0 + 0.15 * rng.standard_normal(n))
            p_pert = np.maximum(p_pert, 1e-10)
            inits.append(p_pert / p_pert.sum())
    
    return inits

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -500.0, 500.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weights(c, beta)
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    Af = np.fft.rfft(a[::-1], N)
    conv_result = np.fft.irfft(W * Af, N)
    h = conv_result[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _adam_descent(a, deadline, max_iter=50000, lr=0.01, beta_init=None,
                  beta_decay=0.85, beta_min=1e-12, reset_patience=2000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    c = _fft_convolve(a)
    max_c = float(c.max())
    if beta_init is None:
        beta = max(1e-10, 0.05 * max_c)

    best_a = a.copy()
    best_score = float(2.0 * n * max_c)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, cur_max = _gradient_smooth_max(a, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(beta_min, beta * 2.0)
            continue

        grad_c = grad - np.mean(grad)
        t += 1
        m = b1 * m + (1.0 - b1) * grad_c
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_c * grad_c)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v_arr / (1.0 - b2 ** t)

        a = _project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))

        c = _fft_convolve(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

            beta = max(beta_min, beta * beta_decay)

        if no_imp > reset_patience:
            a = best_a.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            beta = max(beta_init if beta_init else 0.05 * float(_fft_convolve(a).max()), beta_min)
            lr = base_lr
            no_imp = 0

        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = _project_simplex(a)

    return best_a


def _block_descent(a, deadline, block_size=8, max_iter=20000):
    """Optimize blocks of consecutive elements simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(_score(a))
    rng = np.random.default_rng(777)
    n_blocks = n // block_size

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if n_blocks == 0:
            break

        b = rng.integers(0, n_blocks)
        start = b * block_size
        end = start + block_size
        block = a[start:end].copy()
        block_sum = block.sum()
        if block_sum < 1e-15:
            continue

        best_block = block.copy()
        for trial in range(20):
            noise = rng.standard_normal(block_size) * 0.1 * block_sum
            trial_block = block + noise
            trial_block = np.maximum(trial_block, 0.0)
            ts = trial_block.sum()
            if ts < 1e-15:
                continue
            trial_block = trial_block / ts * block_sum

            a[start:end] = trial_block
            sc = _score(a)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                best_block = trial_block.copy()

        a[start:end] = best_block

        if it % 500 == 0 and it > 0:
            sc_cur = _score(a)
            if sc_cur > best_score * 1.005:
                a = best_a.copy()

    return best_a


def _lp_spread(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j:j+n - j] if False else None
    # More efficient construction
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(n):
            A[j + m, m] += a[j]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(_fft_convolve(a).max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate
        g = _lp_spread(a, M, n)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 300:
                break

        a = a_new

    return best_a


def _interpolate_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    s = p_new.sum()
    return p_new / s if s > 1e-30 else np.ones(n_new) / n_new


def _make_inits(n, parent, rng):
    inits = []
    inits.append(np.ones(n) / n)

    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
        a2 = 1.0 + eta * (0.5 - t)
        a2 = np.maximum(a2, 1e-12)
        inits.append(a2 / a2.sum())

    for a2 in [-1.0, -0.5, 0.5, 1.0, 1.5]:
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + a2 * u * u
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    for _ in range(4):
        n_harmonics = rng.integers(2, 6)
        freqs = rng.integers(1, 4, size=n_harmonics)
        phases = rng.uniform(0, 2 * np.pi, size=n_harmonics)
        amps = rng.uniform(0.05, 0.3, size=n_harmonics)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for k in range(n_harmonics):
            a += amps[k] * np.cos(2 * np.pi * freqs[k] * t + phases[k])
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    # Concentrated middle
    t = np.linspace(0.0, 1.0, n)
    for width in [0.2, 0.3, 0.5]:
        a = np.exp(-((t - 0.5) / width) ** 2)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    if parent is not None:
        p = _interpolate_to_length(parent, n)
        inits.append(p)
        for _ in range(4):
            p_pert = p * (1.0 + 0.2 * rng.standard_normal(n))
            p_pert = np.maximum(p_pert, 1e-12)
            inits.append(p_pert / p_pert.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Combines smooth gradient descent, spectral shaping, block descent,
    and LP-based spreading across multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception as e:
        print(f"No parent available: {e}")

    # Lengths to explore
    lengths = [256, 384, 512, 640, 768]
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths.insert(0, pn)

    # Phase 1: Multi-initialization gradient descent (coarse)
    print(f"Phase 1: Multi-init GD, deadline in {deadline - time.monotonic():.0f}s")
    for n in lengths:
        if time.monotonic() > deadline - 80:
            break

        inits = _make_inits(n, parent, rng)

        # Sort by initial score, try best ones first
        scored_inits = []
        for a0 in inits:
            sc = _score(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
            scored_inits.append((sc, a0))

        scored_inits.sort(key=lambda x: x[0])

        # Top initializations get more time
        for rank, (sc0, a0) in enumerate(scored_inits[:12]):
            if time.monotonic() > deadline - 60:
                break

            if rank < 3:
                gd_time = min(50.0, deadline - time.monotonic() - 50.0)
            elif rank < 6:
                gd_time = min(25.0, deadline - time.monotonic() - 50.0)
            else:
                gd_time = min(12.0, deadline - time.monotonic() - 50.0)

            if gd_time < 3.0:
                continue

            gd_end = time.monotonic() + gd_time
            lr = 0.03 if rank < 3 else 0.02
            a_gd, sc_gd = _adam_descent(a0, gd_end, max_iter=50000, lr=lr, beta_decay=0.8)

            if sc_gd < best_score:
                best_score = sc_gd
                best_seq = a_gd.copy()

            if rank < 4:
                print(f"  n={n}, rank={rank}: {sc0:.8f} -> {sc_gd:.10f}")

    # Phase 2: Spectral shaping diversification on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"Phase 2: Spectral shaping, current best={best_score:.10f}")
        spec_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 40.0)
        a_spec = _spectral_shaping(best_seq, n_iter=300, deadline=spec_end)
        sc_spec = _score(a_spec)
        if sc_spec < best_score:
            best_score = sc_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: {sc_spec:.10f}")

        # Gradient polish on spectral result
        if time.monotonic() < deadline - 40:
            gd_end = time.monotonic() + min(20.0, deadline - time.monotonic() - 30.0)
            a_gd2, sc_gd2 = _adam_descent(a_spec, gd_end, max_iter=30000, lr=0.01, beta_decay=0.7)
            if sc_gd2 < best_score:
                best_score = sc_gd2
                best_seq = a_gd2.copy()
                print(f"  Spectral+GD: {sc_gd2:.10f}")

    # Phase 3: Fine gradient descent on best
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"Phase 3: Fine GD, current best={best_score:.10f}")
        gd_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 30.0)
        a_fine, sc_fine = _adam_descent(best_seq, gd_end, max_iter=80000, lr=0.005,
                                        beta_init=0.01 * float(_fft_convolve(best_seq).max()),
                                        beta_decay=0.7, reset_patience=4000)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
            print(f"  Fine GD: {sc_fine:.10f}")

    # Phase 4: Block descent
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"Phase 4: Block descent, current best={best_score:.10f}")
        block_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 20.0)
        a_block = _block_descent(best_seq, block_end, block_size=16, max_iter=30000)
        sc_block = _score(a_block)
        if sc_block < best_score:
            best_score = sc_block
            best_seq = a_block.copy()
            print(f"  Block: {sc_block:.10f}")

    # Phase 5: LP refinement
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_lp = best_seq.size
        if n_lp <= 512:
            print(f"Phase 5: LP refinement (n={n_lp}), current best={best_score:.10f}")
            lp_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 10.0)
            a_lp = _lp_refine(best_seq, lp_end, max_iter=500, inflate=1.001, blend=0.25)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  LP: {sc_lp:.10f}")

            # GD after LP
            if time.monotonic() < deadline - 10:
                gd_end = time.monotonic() + (deadline - time.monotonic() - 10)
                a_post, sc_post = _adam_descent(a_lp, gd_end, max_iter=30000, lr=0.003,
                                                beta_decay=0.75)
                if sc_post < best_score:
                    best_score = sc_post
                    best_seq = a_post.copy()
                    print(f"  LP+GD: {sc_post:.10f}")

    # Phase 6: Random restart and re-optimize
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"Phase 6: Random restart, current best={best_score:.10f}")
        n = best_seq.size
        for attempt in range(3):
            if time.monotonic() > deadline - 15:
                break
            noise_scale = 0.1 + 0.1 * attempt
            a_pert = best_seq * (1.0 + noise_scale * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            gd_end = time.monotonic() + min(20.0, deadline - time.monotonic() - 10.0)
            if gd_end - time.monotonic() < 3.0:
                break
            a_r, sc_r = _adam_descent(a_pert, gd_end, max_iter=30000, lr=0.01, beta_decay=0.8)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {attempt}: {sc_r:.10f}")

    # Phase 7: Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        gd_end = time.monotonic() + (deadline - time.monotonic() - 5)
        if gd_end - time.monotonic() > 3.0:
            a_final, sc_final = _adam_descent(best_seq, gd_end, max_iter=100000, lr=0.002,
                                              beta_init=0.005 * float(_fft_convolve(best_seq).max()),
                                              beta_decay=0.7, reset_patience=5000)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"Phase 7 (final): {sc_final:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
