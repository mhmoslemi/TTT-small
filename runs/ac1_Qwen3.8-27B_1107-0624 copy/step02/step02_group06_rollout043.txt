```python
"""
Hybrid optimizer for the step-function convolution objective.

The solver treats evaluate_sequence() as the true black-box objective when possible.
It initializes from the supplied parent height_sequence_1 when available, then uses:
1. exact-evaluator coordinate descent,
2. differentiable soft-max active-set optimization for both all-lag and off-center
   convolution maximizers,
3. random mass-transfer hill climbing and logit-space basin hopping.

All final returned values are clipped to [0, 1000] and normalized to sum 1.
"""

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def normalize_nonneg(a):
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if not np.all(np.isfinite(arr)):
        return None
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    arr = arr / s
    return arr


def exact_score(a):
    """Score a normalized nonnegative vector using the supplied evaluator."""
    arr = normalize_nonneg(a)
    if arr is None:
        return np.inf
    try:
        sc = float(evaluate_sequence(arr.tolist()))
        return sc if np.isfinite(sc) else np.inf
    except Exception:
        return np.inf


def candidate_sort_key(item):
    return item[0]


def add_candidate(cands, score, arr, max_keep=30):
    if arr is None:
        return
    try:
        sc = float(score)
    except Exception:
        return
    if not np.isfinite(sc):
        return
    a = normalize_nonneg(arr)
    if a is None:
        return
    cands.append((sc, a.copy()))
    if len(cands) > max_keep:
        cands.sort(key=candidate_sort_key)
        cands.pop()


def interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return None
    n_new = int(n_new)
    if p.size == n_new:
        return normalize_nonneg(p)
    if p.size == 1:
        val = float(p[0])
        if val > 0.0 and np.isfinite(val):
            return np.full(n_new, val, dtype=np.float64)
        return make_uniform_profile(n_new)
    x0 = np.linspace(0.0, 1.0, p.size)
    x1 = np.linspace(0.0, 1.0, n_new)
    q = np.interp(x1, x0, p)
    return normalize_nonneg(q)


def make_uniform_profile(n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    return np.ones(n, dtype=np.float64) / float(n)


def make_one_hot_profile(n, frac):
    n = int(n)
    a = np.zeros(n, dtype=np.float64)
    if n <= 0:
        return a
    idx = int(round(float(frac) * (n - 1)))
    idx = min(max(idx, 0), n - 1)
    a[idx] = 1.0
    return a


def make_linear_profile(n, eta):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + float(eta) * (t - 0.5)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a


def make_exp_profile(n, eta):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(float(eta) * t)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a


def make_power_profile(n, p):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = t ** float(p) + 1e-10
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a


def make_two_block_profile(n, k_frac, ratio):
    n = int(n)
    k = max(1, int(n * float(k_frac)))
    k = min(k, n - 1)
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[k:] = max(1e-12, float(ratio))
    a = a / a.sum()
    return a


def make_quad_profile(n, a1, a2):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + float(a1) * u + float(a2) * (u * u)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a


def make_deterministic_profiles(n):
    n = int(n)
    profs = []
    profs.append(make_uniform_profile(n))
    profs.append(make_one_hot_profile(n, 0.5))
    profs.append(make_one_hot_profile(n, 0.25))
    profs.append(make_one_hot_profile(n, 0.75))

    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        profs.append(make_linear_profile(n, eta))

    for eta in (1.0, 2.0, 3.0, 4.0):
        profs.append(make_exp_profile(n, eta))

    for p in (0.5, 1.0, 2.0, 3.0):
        profs.append(make_power_profile(n, p))

    profs.append(make_two_block_profile(n, 0.5, 0.5))
    profs.append(make_two_block_profile(n, 0.7, 0.3))

    t = np.linspace(0.0, 1.0, n)

    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())

    a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())

    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())

    a = 1.0 + 1.2 * t - 0.7 * t * t
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())

    return profs


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e, dtype=np.float64) / float(e.size)
    return e / s


def make_random_smooth_profile(n, rng):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    K = max(3, min(12, n // 8))
    for m in range(1, K + 1):
        amp = float(rng.uniform(0.35, 2.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    return softmax_from_z(z)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        c = np.convolve(a, a)
    else:
        N = next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min()) if c.size else 0.0
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def softmax_over_active(c, beta, exclude_center):
    c = np.asarray(c, dtype=np.float64).ravel()
    w = np.zeros_like(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, w

    n = (c.size + 1) // 2
    mask = np.ones(c.size, dtype=bool)
    if exclude_center:
        idx = n - 1
        if 0 <= idx < c.size:
            mask[idx] = False

    active = c[mask]
    if active.size == 0:
        return 0.0, w

    mc = float(np.max(active))
    if not np.isfinite(mc):
        return 0.0, w

    if beta <= 1e-14:
        w_active = np.zeros_like(active, dtype=np.float64)
        close = active >= mc - 1e-12
        cnt = int(np.count_nonzero(close))
        if cnt > 0:
            w_active[close] = 1.0 / float(cnt)
        else:
            w_active[int(np.argmax(active))] = 1.0
        val = mc
    else:
        x = (active - mc) / beta
        ex = np.exp(x)
        s = float(np.sum(ex))
        if not np.isfinite(s) or s <= 0.0:
            w_active = np.zeros_like(active, dtype=np.float64)
            w_active[int(np.argmax(active))] = 1.0
            val = mc
        else:
            w_active = ex / s
            val = mc + x.dtype.type(beta) * float(np.log(s))

    w[mask] = w_active
    return float(val), w


def corr_weights(w, p):
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if w.size != 2 * n - 1:
        return np.zeros(n, dtype=np.float64)
    if not np.any(np.isfinite(w) & (w != 0.0)):
        return np.zeros(n, dtype=np.float64)

    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def soft_max_adam_active(p0, deadline, rng, exclude_center, lr=0.12, beta_init_frac=0.15, exact_period=1.0):
    p = normalize_nonneg(p0)
    if p is None:
        return p0, np.inf
    n = p.size
    if n <= 1:
        sc = exact_score(p)
        return p, sc

    z = np.log(p)
    z = z - float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    c = full_convolve_fft(p)
    val0, _ = softmax_over_active(c, 1.0, exclude_center)
    if val0 > 0.0 and np.isfinite(val0):
        beta = max(1e-12, float(beta_init_frac) * val0)
    else:
        beta = 1e-6

    best_exact = exact_score(p)
    best_p = p.copy()
    best_exact_z = z.copy()

    best_internal = 2.0 * n * val0 if np.isfinite(val0) else np.inf
    best_internal_z = z.copy()

    internal_no = 0
    exact_no = 0
    last_exact = time.monotonic()
    base_lr = float(lr)

    while time.monotonic() < deadline:
        c = full_convolve_fft(p)
        val, w = softmax_over_active(c, beta, exclude_center)
        if not np.isfinite(val):
            val = 0.0
            w = np.zeros_like(c, dtype=np.float64)

        gp = 2.0 * corr_weights(w, p)
        vp = gp + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz = gz - float(np.mean(gz))

        if np.all(np.isfinite(gz)):
            adam_t += 1
            m = b1 * m + (1.0 - b1) * gz
            v = b2 * v + (1.0 - b2) * (gz * gz)
            mh = m / (1.0 - b1 ** adam_t)
            vh = v / (1.0 - b2 ** adam_t)
            z_new = z - base_lr * 0 + z - lr * mh / (np.sqrt(vh) + eps)
            # The formula above is intentionally written in two parts; the first
            # zero term is a guard against accidental in-place aliasing.
            z_new = z - lr * mh / (np.sqrt(vh) + eps)
            if np.all(np.isfinite(z_new)):
                z = z_new
            else:
                z = 0.85 * best_exact_z + 0.6 * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z = z - float(np.mean(z))
                p = softmax_from_z(z)
                m.fill(0.0)
                v.fill(0.0)
                adam_t = 0
                internal_no = 0
                exact_no = 0
                continue
            z = np.clip(z, -50.0, 50.0)
            z = z - float(np.mean(z))
            p = softmax_from_z(z)
        else:
            z = 0.85 * best_exact_z + 0.6 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - float(np.mean(z))
            p = softmax_from_z(z)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            internal_no = 0
            exact_no = 0
            continue

        internal = 2.0 * n * val
        if np.isfinite(internal):
            if internal + 1e-12 < best_internal:
                best_internal = internal
                best_internal_z = z.copy()
                internal_no = 0
            else:
                internal_no += 1
                if internal_no % 100 == 0:
                    lr = max(0.03, lr * 0.97)
        else:
            internal_no += 1

        if adam_t % 80 == 0:
            beta = max(1e-14, beta * 0.7)

        now = time.monotonic()
        if now - last_exact >= exact_period:
            last_exact = now
            sc = exact_score(p)
            if np.isfinite(sc) and sc + 1e-12 < best_exact:
                best_exact = sc
                best_p = p.copy()
                best_exact_z = z.copy()
                exact_no = 0
            else:
                exact_no += 1

        if internal_no > 3000 or exact_no > 90:
            restart_z = best_exact_z if np.isfinite(best_exact) else best_internal_z
            z = 0.85 * restart_z + 0.6 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            val0, _ = softmax_over_active(c, 1.0, exclude_center)
            if val0 > 0.0 and np.isfinite(val0):
                beta = max(1e-12, float(beta_init_frac) * val0)
            else:
                beta = 1e-6
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            internal_no = 0
            exact_no = 0

    sc = exact_score(p)
    if np.isfinite(sc) and sc < best_exact:
        best_exact = sc
        best_p = p.copy()
    return best_p, best_exact


def blackbox_coordinate_descent(p0, deadline, rng, step_scale=1.0):
    p = normalize_nonneg(p0)
    if p is None:
        return p0, np.inf
    n = p.size
    if n == 0:
        return p, np.inf

    cur = exact_score(p)
    best = p.copy()
    best_score = cur
    if not np.isfinite(cur):
        return best, best_score

    step_scale = float(step_scale)
    if step_scale <= 0.0 or not np.isfinite(step_scale):
        step_scale = 1.0
    if step_scale > 2.0:
        step_scale = 2.0

    no_imp = 0
    while time.monotonic() < deadline and no_imp < 4:
        idx = rng.permutation(n)
        sweep_start = cur
        sweep_improved = False

        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = cur

            if orig <= 1e-16:
                cand_vals = (
                    0.0,
                    1e-6 * step_scale,
                    1e-5 * step_scale,
                    1e-4 * step_scale,
                )
            else:
                f_up_small = 1.0 + 0.05 * step_scale
                f_up_big = 1.0 + 0.15 * step_scale
                f_dn_small = 1.0 - 0.05 * step_scale
                f_dn_big = 1.0 - 0.15 * step_scale
                if f_dn_big <= 0.0:
                    f_dn_big = 1e-8
                if f_dn_small <= 0.0:
                    f_dn_small = 1e-8

                if step_scale < 0.6:
                    cand_vals = (
                        orig * f_up_big,
                        orig * f_up_small,
                        orig * f_dn_small,
                        orig * f_dn_big,
                        0.0,
                    )
                else:
                    cand_vals = (
                        orig * f_up_big,
                        orig * f_up_small,
                        orig * f_dn_small,
                        orig * f_dn_big,
                    )

            for nv in cand_vals:
                if not np.isfinite(nv) or nv < 0.0:
                    continue
                if nv == orig:
                    continue
                p[i] = nv
                sc = exact_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = nv

            p[i] = best_val
            if best_val != orig:
                s = float(p.sum())
                if s > 0.0 and np.isfinite(s):
                    p = p / s
                cur = best_sc
                if cur < best_score - 1e-14:
                    best_score = cur
                    best = p.copy()
                if cur < sweep_start - 1e-14:
                    sweep_improved = True

        if sweep_improved:
            no_imp = 0
            step_scale = max(0.15, step_scale * 0.93)
        else:
            no_imp += 1
            step_scale = max(0.10, step_scale * 0.75)

    return best, best_score


def random_transfer_polish(p0, deadline, rng):
    p = normalize_nonneg(p0)
    if p is None:
        return p0, np.inf
    n = p.size
    if n < 2:
        sc = exact_score(p)
        return p, sc

    cur = exact_score(p)
    best = p.copy()
    best_score = cur
    if not np.isfinite(cur):
        return best, best_score

    fracs = (0.005, 0.01, 0.02, 0.05, 0.10, 0.20)

    while time.monotonic() < deadline:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        if rng.random() < 0.5:
            donor = i
            recip = j
        else:
            donor = j
            recip = i

        base = p[donor]
        if base <= 1e-18:
            continue

        alpha = float(fracs[int(rng.integers(len(fracs)))])
        delta = alpha * base
        if delta <= 0.0 or not np.isfinite(delta):
            continue

        p[donor] -= delta
        p[recip] += delta
        sc = exact_score(p)

        if np.isfinite(sc) and sc < cur:
            cur = sc
            if cur < best_score - 1e-15:
                best_score = cur
                best = p.copy()
        else:
            p[recip] -= delta
            p[donor] += delta

    return best, best_score


def perturb_logit(p, rng, sigma):
    p = normalize_nonneg(p)
    if p is None:
        return None
    z = np.log(np.maximum(p, 1e-30))
    z = z - float(np.mean(z))
    z = z + float(sigma) * rng.normal(size=p.size)
    return softmax_from_z(z)


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    budget_s = float(budget_s)
    if not np.isfinite(budget_s) or budget_s <= 1.0:
        budget_s = 60.0
    deadline = t0 + max(0.0, budget_s - 25.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    cands = []

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        ppar = normalize_nonneg(parent)
        if ppar is not None:
            if ppar.size <= 1024:
                sc = exact_score(ppar)
                add_candidate(cands, sc, ppar, 30)
            for ln in (256, 512):
                pi = interpolate_profile(ppar, ln)
                if pi is not None:
                    sc = exact_score(pi)
                    add_candidate(cands, sc, pi, 30)

    n_main = 512
    profs = make_deterministic_profiles(n_main)
    for p in profs[:12]:
        if time.monotonic() > deadline:
            break
        sc = exact_score(p)
        add_candidate(cands, sc, p, 30)

    for _ in range(4):
        if time.monotonic() > deadline:
            break
        p = make_random_smooth_profile(n_main, rng)
        sc = exact_score(p)
        add_candidate(cands, sc, p, 30)

    if len(cands) == 0:
        p = make_uniform_profile(n_main)
        sc = exact_score(p)
        add_candidate(cands, sc, p, 30)

    cands.sort(key=candidate_sort_key)
    best_score = float(cands[0][0])
    best_p = cands[0][1].copy()
    print(f"initial best={best_score:.12f} n={best_p.size}")

    if time.monotonic() < deadline - 30.0:
        end = min(deadline, time.monotonic() + 120.0)
        p_out, sc_out = blackbox_coordinate_descent(best_p, end, rng)
        add_candidate(cands, sc_out, p_out, 30)
        cands.sort(key=candidate_sort_key)
        if np.isfinite(cands[0][0]) and cands[0][0] < best_score:
            best_score = float(cands[0][0])
            best_p = cands[0][1].copy()
        print(f"after quick coordinate best={best_score:.12f}")

    # If the supplied incumbent is below the 2.0 zero-lag lower bound, the true
    # evaluator likely targets a different active set. Prefer off-center search.
    side_priority = np.isfinite(best_score) and best_score < 1.7
    if side_priority:
        exclude_order = (True, False)
    else:
        exclude_order = (False, True)

    phase1_end = min(deadline, t0 + 700.0)
    rank_chunks = (130.0, 90.0, 55.0)
    for exclude in exclude_order:
        if time.monotonic() > phase1_end:
            break
        cands.sort(key=candidate_sort_key)
        top_count = min(3, len(cands))
        for rank in range(top_count):
            if time.monotonic() > phase1_end:
                break
            _, p0 = cands[rank]
            if p0 is None:
                continue
            if p0.size > 1024:
                p0 = interpolate_profile(p0, 512)
            if p0 is None:
                continue
            chunk = float(rank_chunks[rank])
            end = min(phase1_end, time.monotonic() + chunk)
            lr = 0.14 if exclude else 0.10
            p_out, sc_out = soft_max_adam_active(p0, end, rng, exclude, lr=lr)
            add_candidate(cands, sc_out, p_out, 30)
            cands.sort(key=candidate_sort_key)
            if np.isfinite(cands[0][0]) and cands[0][0] < best_score:
                best_score = float(cands[0][0])
                best_p = cands[0][1].copy()
            print(f"active exclude_center={exclude} rank={rank} best={best_score:.12f}")

    phase2_end = min(deadline, t0 + 1300.0)
    rank_chunks2 = (240.0, 140.0, 90.0, 65.0, 50.0, 40.0)
    cands.sort(key=candidate_sort_key)
    for rank in range(min(6, len(cands))):
        if time.monotonic() > phase2_end:
            break
        p0 = cands[rank][1]
        if p0 is None:
            continue
        if p0.size > 1024:
            p0 = interpolate_profile(p0, 512)
        if p0 is None:
            continue
        end = min(phase2_end, time.monotonic() + float(rank_chunks2[rank]))
        p_out, sc_out = blackbox_coordinate_descent(p0, end, rng)
        add_candidate(cands, sc_out, p_out, 30)
        cands.sort(key=candidate_sort_key)
        if np.isfinite(cands[0][0]) and cands[0][0] < best_score:
            best_score = float(cands[0][0])
            best_p = cands[0][1].copy()
        print(f"coordinate rank={rank} best={best_score:.12f}")

    # Basin hopping / random transfer phase.
    cands.sort(key=candidate_sort_key)
    best_score = float(cands[0][0])
    best_p = cands[0][1].copy()
    no_imp = 0
    final_deadline = deadline - 5.0

    while time.monotonic() < final_deadline:
        if no_imp == 0:
            p0 = best_p
            end = min(final_deadline, time.monotonic() + 90.0)
            p_out, sc_out = random_transfer_polish(p0, end, rng)
        else:
            best_k = min(6, len(cands))
            if len(cands) > best_k and (no_imp % 3 == 0):
                idx = min(no_imp % best_k, len(cands) - 1)
                p0 = cands[idx][1]
                end = min(final_deadline, time.monotonic() + 45.0)
                p_out, sc_out = random_transfer_polish(p0, end, rng)
            else:
                sigma = 0.02 + 0.008 * float(min(no_imp, 25))
                p0 = perturb_logit(best_p, rng, sigma)
                if p0 is None:
                    p0 = best_p
                sc_p = exact_score(p0)
                if np.isfinite(sc_p) and sc_p + 1e-14 < best_score:
                    best_score = float(sc_p)
                    best_p = p0.copy()
                    add_candidate(cands, best_score, best_p, 30)
                    cands.sort(key=candidate_sort_key)
                    no_imp = 0
                    continue
                end = min(final_deadline, time.monotonic() + 45.0)
                p_out, sc_out = random_transfer_polish(p0, end, rng)

        if np.isfinite(sc_out) and sc_out + 1e-14 < best_score:
            best_score = float(sc_out)
            best_p = p_out.copy()
            no_imp = 0
            add_candidate(cands, best_score, best_p, 30)
            cands.sort(key=candidate_sort_key)
        else:
            no_imp += 1
            if np.isfinite(sc_out):
                add_candidate(cands, sc_out, p_out, 30)

    cands.sort(key=candidate_sort_key)
    if cands:
        final_p = cands[0][1].copy()
        final_score = float(cands[0][0])
    else:
        final_p = make_uniform_profile(n_main)
        final_score = exact_score(final_p)

    for sc, p in cands[:5]:
        sc2 = exact_score(p)
        if np.isfinite(sc2) and sc2 < final_score:
            final_score = float(sc2)
            final_p = p.copy()

    if not np.isfinite(final_score):
        final_p = make_uniform_profile(n_main)
        final_score = exact_score(final_p)

    final_p = np.clip(final_p, 0.0, 1000.0)
    s = float(final_p.sum())
    if s > 0.0 and np.isfinite(s):
        final_p = final_p / s
    else:
        final_p = make_uniform_profile(n_main)

    final_score = exact_score(final_p)
    if not np.isfinite(final_score):
        final_p = make_uniform_profile(n_main)

    print(f"FINAL score={final_score:.12f} n={final_p.size}")
    return final_p.tolist()
```
