So for palindromic a, the max is at least 1/n, and score is at least 2.

For non-palindromic a, conv(a,a)[n-1] can be less than sum a[j]^2.

The key question: how much less?

Let's define the "asymmetry" of a as the difference between a and its reverse. If a = r + s where r is the palindromic part and s is the anti-palindromic part (s[i] = -s[n-1-i]), then:
conv(a,a)[n-1] = conv(r+r, r+r)[n-1] = conv(r,r)[n-1] + 2*conv(r,s)[n-1] + conv(s,s)[n-1]

conv(r,r)[n-1] = sum r[j]^2 (since r is palindromic)
conv(s,s)[n-1] = sum s[j]*s[n-1-j] = -sum s[j]^2 (since s is anti-palindromic)
conv(r,s)[n-1] = sum r[j]*s[n-1-j]

So conv(a,a)[n-1] = sum r[j]^2 - sum s[j]^2 + 2*sum r[j]*s[n-1-j]

And sum a[j]^2 = sum(r[j]+s[j])^2 = sum r[j]^2 + 2*sum r[j]*s[j] + sum s[j]^2

The difference: sum a[j]^2 - conv(a,a)[n-1] = 2*sum r[j]*s[j] + 2*sum s[j]^2 - 2*sum r[j]*s[n-1-j]

This is getting complicated. Let me just focus on the practical optimization.

Let me think about what makes a good sequence:
1. It should NOT be palindromic (to reduce the center value).
2. It should have small values at the ends (to reduce conv[0] and conv[2n-2]).
3. It should be "smooth" (to avoid sharp peaks in the autocorrelation).

Combining these: a sequence that rises from 0 on the left, has a broad peak, and falls to 0 on the right, but is ASYMMETRIC (peak is off-center).

For example: a[i] ∝ (i+1)^p * (n-i)^q for appropriate p, q.
If p ≠ q, it's asymmetric.

This rises linearly from the left and falls quadratically on the right. The peak is shifted to the left.

Or p=2, q=1: a[i] ∝ (i+1)^2*(n-i). Peak shifted to the right.

"""
Optimized search for non-negative step functions minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated Adam in softmax/logit space with adaptive softness,
LP-based direction finding, and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=8.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M.
    Returns optimal g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def compute_gradient_fast(p, w, N_fft):
    """
    Compute grad[k] = 2 * sum_j p[j] * w[j+k] for k=0..n-1 using FFT.
    p has length n, w has length 2n-1, N_fft >= 3n-2.
    """
    A = np.fft.rfft(p, N_fft)
    W = np.fft.rfft(w, N_fft)
    R = np.fft.irfft(np.conj(A) * W, N_fft)
    n = len(p)
    return 2.0 * R[:n]


def adam_optimize(p0, deadline, lr=0.05, beta_factor=0.15, verbose=True, label=""):
    """
    Adam gradient descent in logit space to minimize 2*n*max(conv(p,p)) with sum(p)=1.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12).astype(np.float64)
    p = p / p.sum()

    z = np.log(p)
    z -= z.max()

    # Precompute FFT size
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p, p)
    max_c = float(c.max())
    beta = beta_factor * max_c

    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p, p)
        max_c = float(c.max())

        # Soft argmax weights
        x = (c - max_c) / beta
        x -= x.max()
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Compute gradient using FFT
        grad_p = compute_gradient_fast(p, w, N_fft)

        # Project onto tangent space of simplex
        grad_proj = grad_p - np.mean(grad_p)

        # Transform to logit space
        dot_val = float(np.dot(p, grad_proj))
        grad_z = p * (grad_proj - dot_val)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z * grad_z
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= z.mean()

        p_new = np.exp(z - z.max())
        p_sum = p_new.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p = np.exp(z - z.max())
            p = p / p.sum()
            continue
        p_new = p_new / p_sum

        score = 2.0 * n * float(np.convolve(p_new, p_new).max())

        if score < best_score - 1e-14:
            best_score = score
            best_p = p_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adapt beta (sharpen the max approximation)
        if it % 50 == 0:
            beta = max(1e-12, beta * 0.85)

        # Adapt learning rate
        if no_imp > 200:
            lr = max(0.005, lr * 0.95)
        if no_imp > 800:
            lr = max(0.003, lr * 0.9)

        # Restart if stuck
        if no_imp > 2000:
            rng = np.random.default_rng(it * 7 + 13)
            z = best_z + 0.15 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = 0.15 * float(np.convolve(best_p, best_p).max())

        p = p_new

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    {label} Adam iter {it}: score={best_score:.8f}, beta={beta:.2e}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_p, best_score


def lp_refine(f, deadline, max_iters=30, t_init=0.35, verbose=True, label=""):
    """
    LP-based iterative refinement. f must be normalized to sum 1.
    Returns (best_f, best_score).
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=15)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Evaluate full step
        g_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_score < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            # Blended step
            for t in [t_step, 0.5 * t_step, 0.25 * t_step]:
                f_try = (1.0 - t) * f + t * g
                f_try = normalize(f_try)
                if f_try is None:
                    continue
                c_try = np.convolve(f_try, f_try)
                M_try = float(c_try.max())
                score_try = 2.0 * n * M_try
                if score_try < best_score - 1e-12:
                    f = f_try.copy()
                    M = M_try
                    best_f = f.copy()
                    best_score = score_try
                    no_improve = 0
                    break
            else:
                no_improve += 1
                t_step = max(0.02, t_step * 0.7)

        if no_improve > 15:
            t_step = max(0.02, t_step * 0.8)
        if no_improve > 40:
            break

        if verbose and (it % 5 == 0 or it == max_iters - 1):
            print(f"    {label} LP iter {it}: score={best_score:.8f}, M={M:.6f}, t={t_step:.4f}")

    return best_f, best_score


def make_profile(n, func):
    """Create a normalized non-negative profile from a function of t in [0,1]."""
    if n <= 1:
        return np.array([1.0])
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = func(t)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def interp_sequence(a, n_new):
    """Interpolate sequence to new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

    def prof_gauss_skew(t, center=0.4, width=0.28):
        return np.exp(-((t - center) ** 2) / (2.0 * width ** 2))

    def prof_gauss_skew2(t, center=0.6, width=0.3):
        return np.exp(-((t - center) ** 2) / (2.0 * width ** 2))

    def prof_sine_skew(t, p=2.0):
        return np.sin(np.pi * np.maximum(t, 1e-8))**p

    def prof_linear(t, eta=1.5):
        return np.maximum(1.0 + eta * (2.0 * t - 1.0), 1e-8)

    def prof_quadratic(t, a=1.0, b=1.5, c=0.3):
        return np.maximum(a + b * t + c * t * t, 1e-8)

    profiles = [
        prof_exp_ramp,
        prof_exp_ramp2,
        prof_power1,
        prof_power2,
        prof_gauss_skew,
        prof_gauss_skew2,
        prof_sine_skew,
        prof_linear,
        prof_quadratic,
    ]

    candidates = []
    if best_seq is not None:
        a1 = interp_sequence(best_seq, n_target)
        if a1 is not None:
            candidates.append(("parent", a1, internal_score(a1)))

    for i, prof_fn in enumerate(profiles):
        a = make_profile(n_target, prof_fn)
        sc = internal_score(a)
        candidates.append((f"prof_{i}", a, sc))
        # Also try the reversed version
        a_rev = a[::-1].copy()
        sc_rev = internal_score(a_rev)
        if sc_rev < sc:
            candidates.append((f"prof_{i}_rev", a_rev, sc_rev))
        else:
            candidates.append((f"prof_{i}_rev", a, sc))

    # Sort by score and keep top candidates
    candidates.sort(key=lambda x: x[2])
    print(f"\nTop initializations:")
    for name, a, sc in candidates[:8]:
        print(f"  {name}: score={sc:.8f}")

    # Select top 4-5 distinct candidates for optimization
    max_candidates = 5
    selected = []
    for name, a, sc in candidates:
        if len(selected) >= max_candidates:
            break
        # Check for diversity (not too similar to already selected)
        is_diverse = True
        for _, a_sel, _ in selected:
            if float(np.dot(a, a_sel)) > 0.99:
                is_diverse = False
                break
        if is_diverse:
            selected.append((name, a, sc))

    print(f"\nSelected {len(selected)} candidates for optimization")

    # Phase 1: Adam on each candidate with time budget
    phase1_deadline = t_start + 800.0
    time_per_candidate = (phase1_deadline - t_start) / max(1, len(selected))

    all_results = []
    for idx, (name, a_init, init_sc) in enumerate(selected):
        cand_deadline = min(phase1_deadline, t_start + (idx + 1) * time_per_candidate)
        if time.monotonic() > cand_deadline - 5:
            continue

        print(f"\n{'='*50}")
        print(f"Optimizing candidate '{name}' (init score={init_sc:.8f}), budget={cand_deadline - time.monotonic():.0f}s")
        print(f"{'='*50}")

        p_best, sc_best = adam_optimize(a_init, cand_deadline, lr=0.04, beta_factor=0.12,
                                         verbose=True, label=name)
        all_results.append((name, p_best, sc_best))
        print(f"  Result: score={sc_best:.8f}")

        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best

    # Phase 2: LP refinement on the best
    if time.monotonic() < total_deadline - 200:
        print(f"\n{'='*50}")
        print(f"Phase 2: LP refinement, score={best_score:.8f}")
        print(f"{'='*50}")
        lp_deadline = min(total_deadline - 200, time.monotonic() + 300)
        f_best, sc_best = lp_refine(best_seq, lp_deadline, max_iters=20, t_init=0.3,
                                     verbose=True, label="LP")
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  LP improved to: {best_score:.8f}")

    # Phase 3: Final Adam polish
    if time.monotonic() < total_deadline - 10:
        print(f"\n{'='*50}")
        print(f"Phase 3: Final Adam polish, score={best_score:.8f}")
        print(f"{'='*50}")
        polish_deadline = total_deadline - 10
        # Use a fresh Adam run with different hyperparameters for exploration
        p_best, sc_best = adam_optimize(best_seq, polish_deadline, lr=0.02, beta_factor=0.05,
                                         verbose=True, label="polish")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best

    # Phase 4: Try one more LP step if time remains
    if time.monotonic() < total_deadline - 5:
        lp_deadline2 = total_deadline - 5
        f_best, sc_best = lp_refine(best_seq, lp_deadline2, max_iters=5, t_init=0.2,
                                     verbose=True, label="LP_final")
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={final_score:.8f}")
    print(f"Target: 1.5027")
    print(f"{'='*50}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        # Fallback
        t = np.arange(n_target, dtype=np.float64) / (n_target - 1)
        best_seq = np.exp(2.5 * (2.0 * t - 1.0))
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
