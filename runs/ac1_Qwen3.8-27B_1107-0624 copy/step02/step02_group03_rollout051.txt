def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""

    return normalize(a_new)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, entropy-regularized gradient descent with
adaptive temperature, multi-resolution search, and random restarts.
Uses FFT-based convolution for efficiency at larger n.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()

    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def compute_grad_convolution(p, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. p, where c = convolve(p,p).
    Uses the formula: grad_p[i] = 2 * sum_j w[i+j] * p[j]
    """
    n = len(p)
    # Efficient computation using the fact that h[i] = sum_j w[i+j] * p[j]
    # This is a cross-correlation that can be done with FFT
    # w has length 2n-1, p has length n
    # h[i] = sum_{j=0}^{n-1} w[i+j] * p[j] for i=0..n-1
    
    # Use direct method for small n, FFT for large n
    if n <= 512:
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.dot(w[i:i + n], p)
    else:
        # FFT-based: h[i] = sum_j w[i+j]*p[j]
        # This is the cross-correlation of w and p, reversed
        # h = ifft(fft(w) * conj(fft(p))) -- need to be careful with lengths
        # Simpler: just use the direct method but vectorized
        # Actually for n=1024, direct is O(n^2) = 1M, which is fine
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.dot(w[i:i + n], p)
    return 2.0 * h

    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    alpha = alpha_init

    # Track improvements for adaptive scheduling
    recent_improvements = []

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights for differentiating the max
        x = alpha * (c - max_c)
        # Clip for numerical stability
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient computation
        grad_p = compute_grad_convolution(p0, w)

        # Logit space gradient with variance reduction (centering)
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new -= np.mean(z_new)

        p = np.exp(z_new - z_new.max())
        p_sum = p.sum()
        if p_sum < 1e-30 or not np.isfinite(p_sum):
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z_new.copy()
            no_imp = 0
            recent_improvements.append(score)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.003, lr * 0.93)

        # Adaptive temperature: increase as we converge (sharper max approximation)
        if it % 50 == 0:
            if no_imp < 50:
                alpha = min(300.0, alpha * 1.1)
            else:
                alpha = max(10.0, alpha * 0.95)

        if verbose and it % 200 == 0:
            print(f"    {label} iter {it}: score={best_score:.8f}, alpha={alpha:.1f}, lr={lr:.5f}, no_imp={no_imp}")

        if no_imp > 1200:
            # Restart with perturbation - try different magnitudes
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            pert_mag = rng_r.choice([0.3, 0.5, 0.8, 1.0])
            z = best_z + pert_mag * rng_r.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            alpha = alpha_init
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()

        z = z_new
        p0 = p

    return best_p, best_score


def mirror_descent(p0, deadline, lr=0.1, verbose=False, label=""):
    """Mirror descent on the simplex with soft-max loss."""
    n = len(p0)
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()

    best_p = p.copy()
    best_score = internal_score(p)
    it = 0
    alpha = 20.0
    base_lr = lr

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p, p)
        max_c = float(c.max())

        x = alpha * (c - max_c)
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        w = ex / s if s > 1e-30 else np.eye(2*n-1)[np.argmax(c)]

        grad_p = compute_grad_convolution(p, w)
        
        # Mirror descent: p_new[i] proportional to p[i] * exp(-lr * grad_p[i])
        p_new = p * np.exp(-lr * grad_p / max_c)
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()

        score = 2.0 * n * float(np.convolve(p_new, p_new).max())
        if score < best_score:
            best_score = score
            best_p = p_new.copy()

        if it % 200 == 0:
            alpha = min(200.0, alpha * 1.05)
            lr = max(0.01, lr * 0.98)

        p = p_new

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining LP and gradient descent.
    Strategy:
      1. Generate diverse initial profiles
      2. LP refinement at n=512 (fast, many iterations)
      3. Multi-resolution gradient descent (n=512, 768, 1024)
      4. Final polish with advanced gradient method
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent sequence: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    # Generate initial profiles and pick the best
    n_init = 512
    profiles = make_profiles(n_init, rng)
    for p in profiles:
        sc = internal_score(p)
        if sc < best_score:
            best_score = sc
            best_seq = p
    print(f"After profile search: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: LP refinement at n=512
    deadline_p1 = min(deadline, t_start + 350.0)
    if time.monotonic() < deadline_p1 - 30:
        n1 = len(best_seq)
        if n1 != 512:
            best_seq = interpolate_seq(best_seq, 512)
            n1 = 512
            best_score = internal_score(best_seq)
        print(f"\nPhase 1: LP at n={n1}, initial score={best_score:.8f}")
        f_best, sc_best = lp_iterate(best_seq, max_iters=300, t_init=0.35, deadline=deadline_p1)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  Phase 1 result: score={best_score:.8f}")

    # Phase 2: Multi-resolution gradient descent
    # Try n=512 first (refine what LP gave us)
    deadline_p2 = min(deadline, t_start + 700.0)
    if time.monotonic() < deadline_p2 - 100:
        n2 = len(best_seq)
        phase_time = min(200, deadline_p2 - time.monotonic() - 30)
        if phase_time > 30:
            print(f"\nPhase 2a: Gradient at n={n2}, initial score={internal_score(best_seq):.8f}")
            p_best, sc_best = grad_descent_advanced(best_seq, time.monotonic() + phase_time,
                                                     lr=0.025, alpha_init=25.0, label="p2a")
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 2a result: score={best_score:.8f}")

    # Try n=768
    deadline_p3 = min(deadline, t_start + 1100.0)
    if time.monotonic() < deadline_p3 - 100:
        n3 = 768
        phase_time = min(250, deadline_p3 - time.monotonic() - 30)
        if phase_time > 30:
            a3 = interpolate_seq(best_seq, n3)
            if a3 is not None:
                sc3 = internal_score(a3)
                print(f"\nPhase 2b: Gradient at n={n3}, initial score={sc3:.8f}")
                p_best, sc_best = grad_descent_advanced(a3, time.monotonic() + phase_time,
                                                         lr=0.02, alpha_init=30.0, label="p2b")
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Phase 2b result: score={best_score:.8f}")

    # Try n=1024
    if time.monotonic() < deadline - 200:
        n4 = 1024
        phase_time = min(300, deadline - time.monotonic() - 100)
        if phase_time > 30:
            a4 = interpolate_seq(best_seq, n4)
            if a4 is not None:
                sc4 = internal_score(a4)
                print(f"\nPhase 2c: Gradient at n={n4}, initial score={sc4:.8f}")
                p_best, sc_best = grad_descent_advanced(a4, time.monotonic() + phase_time,
                                                         lr=0.015, alpha_init=40.0, label="p2c")
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Phase 2c result: score={best_score:.8f}")

    # Phase 3: Final polish on the best found
    remaining = deadline - time.monotonic() - 10
    if remaining > 30:
        n_final = len(best_seq)
        print(f"\nPhase 3: Final polish at n={n_final}, score={internal_score(best_seq):.8f}")
        p_best, sc_best = grad_descent_advanced(best_seq, time.monotonic() + remaining,
                                                 lr=0.008, alpha_init=50.0, label="final")
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Final result: score={best_score:.8f}")

    # Phase 4: Try a few random restarts from best
    remaining = deadline - time.monotonic() - 10
    if remaining > 60:
        n_r = len(best_seq)
        for restart in range(3):
            if time.monotonic() > deadline - 15:
                break
            rng_r = np.random.default_rng(seed + 1000 * (restart + 1))
            z_r = np.log(np.maximum(best_seq, 1e-12))
            z_r += 0.5 * rng_r.normal(size=n_r)
            z_r = np.clip(z_r, -40, 40)
            z_r -= np.mean(z_r)
            p_r = np.exp(z_r - z_r.max())
            p_r = p_r / p_r.sum()
            
            phase_time = min(150, (deadline - time.monotonic() - 15) / (3 - restart))
            if phase_time < 15:
                break
            
            print(f"\nPhase 4 (restart {restart+1}): initial score={internal_score(p_r):.8f}")
            p_best, sc_best = grad_descent_advanced(p_r, time.monotonic() + phase_time,
                                                     lr=0.02, alpha_init=25.0, label=f"rust{restart}")
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Restart {restart+1} result: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        t = np.linspace(0, 1, 256)
        best_seq = 1.0 + 1.5 * (2 * t - 1)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
