def mirror_descent_step(s, grad, eta):
    """Mirror descent (exponential) step on the simplex."""
    s_new = s * np.exp(-eta * grad / (grad.max() + 1e-15))
    s_new = np.maximum(s_new, 1e-30)
    s_new = s_new / s_new.sum()
    return s_new

def lp_direction(s, M, time_limit=3.0):
    n = len(s)
    # Build constraint matrix: conv(s, b)[k] <= M for all k
    # conv(s, b)[k] = sum_j s[j] * b[k-j] (valid j)
    # This is linear in b.
    # A[k, j] = s[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = s[j] if k-n+1 <= j <= k and 0 <= j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j)  # k-j >= 0 => k >= j
        hi = min(2 * n - 2, j + n - 1)  # k-j < n => k < j+n
        A[lo:hi+1, j] = s[j]
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

One more idea: **iterative reweighting**. Instead of using a fixed top-k, we can iteratively reweight the peaks based on their current values. This is similar to the "交替最小化" (alternating minimization) approach:

1. Start with weights w[k] = 1 for all k.
2. Compute the "weighted" conv: `conv_w[k] = w[k] * conv[k]`.
3. Find the peak of `conv_w`.
4. Update weights: `w[k] *= exp(eta * conv_w[k] / max_conv_w)` (increase weight on peaks).

import numpy as np
import time
from scipy.optimize import linprog

"""
Optimized step function search using:
- Top-k subgradient mirror descent with adaptive step sizes
- LP-based direction finding with tightened peak constraints
- Coordinate descent refinement
- Multiple sequence lengths and random restarts
- Starting from known best (height_sequence_1)
"""


def fast_score(s):
    """Compute 2*n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def compute_subgradient(s, top_k=1):
    """
    Compute the average subgradient of the top-k peaks of conv(s,s).
    Returns (grad, conv, peak_val).
    """
    n = len(s)
    conv = np.convolve(s, s)
    # Get indices of top-k peaks
    if top_k >= len(conv):
        top_indices = np.arange(len(conv))
    else:
        top_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for l in top_indices:
        lo = max(0, l - n + 1)
        hi = min(l, n - 1)
        idx = l - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * s[idx]
    
    peak_val = conv.max()
    return grad, conv, peak_val


def mirror_descent_step(s, grad, eta):
    """Mirror descent (exponential family) step on the simplex."""
    grad_max = grad.max()
    if grad_max < 1e-30:
        return s.copy()
    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-30)
    s_new = s_new / s_new.sum()
    return s_new


def lp_direction_tight(s, M, tight_fraction=0.99, time_limit=3.0):
    """
    Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M*tight_fraction, b >= 0.
    The tighter the constraint, the more aggressive the peak reduction.
    """
    n = len(s)
    M_tight = M * tight_fraction
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        A[lo:hi + 1, j] = s[j]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_tight)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            x = np.maximum(res.x, 0.0)
            if x.sum() > 1e-12:
                return x
    except Exception:
        pass
    return None


def coordinate_descent_step(s, max_iters=100):
    """
    One pass of coordinate descent: for each coordinate, find the best value
    that minimizes the peak autocorrelation.
    """
    n = len(s)
    conv = np.convolve(s, s)
    peak = conv.max()
    
    for i in range(n):
        old_val = s[i]
        # The contribution of s[i] to conv[k] is:
        # s[i] * s[k-i] for each valid k-i
        # So conv_new[k] = conv[k] - s[i]*s[k-i] + new_val*s[k-i]
        # = conv[k] + (new_val - s[i]) * s[k-i]
        # We want to minimize max_k conv_new[k]
        
        # Find the lag where the peak currently is
        peak_lag = np.argmax(conv)
        
        # Try reducing s[i] to reduce the peak
        # If peak is at lag l, and s[i] contributes to it, reducing s[i] helps
        contrib = 0.0
        # s[i] contributes to conv[l] as s[i]*s[l-i] (if valid) and s[l-i]*s[i]
        # Actually conv[l] = sum_j s[j]*s[l-j], so s[i] appears as:
        #   s[i]*s[l-i] (j=i) + s[l-i]*s[i] (j=l-i) if l-i != i
        #   2*s[i]*s[l-i] if l-i == i (i.e., 2i == l), but really it's s[i]^2
        # Let me just compute the derivative: d(conv[l])/ds[i] = 2*s[l-i] (as derived earlier)
        
        # For coordinate descent, try a few candidate values
        best_val = old_val
        best_peak = peak
        
        if old_val < 1e-15:
            continue
        
        # Try reducing
        for frac in [0.0, 0.3, 0.5, 0.7, 0.9]:
            trial_val = old_val * (1.0 - frac)
            s[i] = trial_val
            # Update conv efficiently: conv_new = conv + (trial_val - old_val) * (contribution vector)
            # contribution vector for s[i]: c[k] = 2*s[k-i] (if valid) ... 
            # Actually: d(conv[k])/ds[i] = 2*s[k-i] if 0<=k-i<n, else 0
            # So conv_new[k] ≈ conv[k] + (trial_val - old_val) * 2 * s[k-i] (for small changes)
            # For exact: we need to recompute, but let's use the linear approximation for screening
            # and only recompute for the best few
            
            # Quick estimate using the peak lag
            if 0 <= peak_lag - i < n:
                est_peak = conv[peak_lag] + (trial_val - old_val) * 2.0 * s[peak_lag - i]
            else:
                est_peak = conv[peak_lag]
            
            if est_peak < best_peak - 1e-12:
                # Verify with full computation (expensive but accurate)
                conv_check = np.convolve(s, s)
                actual_peak = conv_check.max()
                if actual_peak < best_peak:
                    best_peak = actual_peak
                    best_val = trial_val
        
        # Also try increasing slightly
        for frac in [0.5, 1.0, 2.0]:
            trial_val = old_val * (1.0 + frac)
            s[i] = trial_val
            if 0 <= peak_lag - i < n:
                est_peak = conv[peak_lag] + (trial_val - old_val) * 2.0 * s[peak_lag - i]
            else:
                est_peak = conv[peak_lag]
            if est_peak < best_peak - 1e-12:
                conv_check = np.convolve(s, s)
                actual_peak = conv_check.max()
                if actual_peak < best_peak:
                    best_peak = actual_peak
                    best_val = trial_val
        
        s[i] = best_val
    
    # Renormalize
    s = np.maximum(s, 0.0)
    total = s.sum()
    if total > 1e-15:
        s = s / total
    return s


def local_optimize(s, deadline, rng, use_lp=False, n_lp_max=512, lp_interval=20):
    """
    Main local optimization loop combining mirror descent, LP, and coordinate descent.
    """
    n = len(s)
    s = s.copy()
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    initial_score = best_score
    
    eta = 1.0  # Step size for mirror descent
    no_improve = 0
    iter_count = 0
    patience = 80
    
    while time.monotonic() < deadline:
        iter_count += 1
        cur_score = fast_score(s)
        
        # --- Phase 1: Mirror descent with top-1 subgradient ---
        if no_improve < patience:
            top_k = 1 if no_improve < patience // 2 else 3
            grad, conv, peak = compute_subgradient(s, top_k)
            
            # Backtracking line search
            improved = False
            for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.125, eta * 0.0625]:
                s_new = mirror_descent_step(s, grad, trial_eta)
                score_new = fast_score(s_new)
                if score_new < cur_score - 1e-12:
                    s = s_new
                    eta = trial_eta
                    improved = True
                    break
                elif score_new < best_score - 1e-12:
                    s = s_new
                    eta = trial_eta
                    improved = True
                    break
            if not improved:
                no_improve += 1
            else:
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_s = s.copy()
                no_improve = 0
        else:
            # --- Phase 2: LP-based direction ---
            if use_lp and n <= n_lp_max and iter_count % lp_interval == 0:
                conv = np.convolve(s, s)
                M = conv.max()
                
                # Try different tightness levels
                for tight in [0.98, 0.99, 0.995, 1.0]:
                    b = lp_direction_tight(s, M, tight, time_limit=2.0)
                    if b is not None:
                        b = b / b.sum()
                        # Blend
                        for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                            cand = (1 - frac) * s + frac * b
                            cand = np.maximum(cand, 0.0)
                            cand_sum = cand.sum()
                            if cand_sum < 1e-12:
                                continue
                            cand = cand / cand_sum
                            sc = fast_score(cand)
                            if sc < best_score - 1e-12:
                                best_score = sc
                                best_s = cand.copy()
                                s = cand
                                no_improve = 0
                                eta = 1.0
                                break
                        if no_improve == 0:
                            break
                
                if no_improve > patience:
                    no_improve = 0
            else:
                # Fallback: perturbation and restart
                perturb_scale = 0.02 * (1 + no_improve / patience)
                s = s + perturb_scale * rng.standard_normal(n) * s
                s = np.maximum(s, 1e-8)
                s = s / s.sum()
                eta = 1.0
                no_improve = 0
        
        # --- Phase 3: Coordinate descent (every 20 iterations) ---
        if iter_count % 20 == 0 and time.monotonic() < deadline - 5:
            s_cd = coordinate_descent_step(s)
            score_cd = fast_score(s_cd)
            if score_cd < best_score - 1e-12:
                best_score = score_cd
                best_s = s_cd.copy()
                s = s_cd
                no_improve = 0
    
    # Return best found
    return best_s, best_score


def generate_initializations(n, rng, hseq):
    """Generate diverse initializations for a given length n."""
    inits = []
    
    # From height_sequence_1 (resized)
    if hseq is not None:
        x_old = np.linspace(0, 1, hseq.size)
        x_new = np.linspace(0, 1, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 0.0)
        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            inits.append((fast_score(h_r), h_r, "hseq_resized"))
    
    # Exact length match
    if hseq is not None and hseq.size == n:
        s = np.maximum(hseq, 0.0)
        if s.sum() > 0.01:
            s = s / s.sum()
            inits.append((fast_score(s), s, "hseq_exact"))
    
    # Uniform
    s = np.ones(n) / n
    inits.append((fast_score(s), s, "uniform"))
    
    # Cosine raised to different powers
    x = np.arange(n) / max(n - 1, 1)
    for p in [2, 4, 6, 8]:
        s = np.cos(np.pi * x / 2) ** p
        s = np.maximum(s, 1e-6)
        s = s / s.sum()
        inits.append((fast_score(s), s, f"cosine_p{p}"))
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            s = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            s = s / s.sum()
            inits.append((fast_score(s), s, f"gauss_{center}_{width}"))
    
    # Two-level patterns
    for frac in [0.3, 0.4, 0.5, 0.6]:
        for gap in [2, 3, 4, 5]:
            s = np.zeros(n)
            for i in range(0, n, gap):
                s[i] = 1.0 / frac
            for i in range(gap, n, gap):
                if i < n:
                    s[i] = 1.0
            s = np.maximum(s, 0.0)
            if s.sum() > 0.01:
                s = s / s.sum()
                inits.append((fast_score(s), s, f"twolvl_{frac}_{gap}"))
    
    # Random
    for _ in range(4):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        inits.append((fast_score(s), s, "random"))
    
    # Smooth random (low-pass filtered)
    for _ in range(3):
        s = rng.random(n) + 0.1
        # Simple moving average smoothing
        kernel = np.ones(min(5, n)) / min(5, n)
        s = np.convolve(s, kernel, mode='same')
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        inits.append((fast_score(s), s, "smooth_rand"))
    
    inits.sort(key=lambda x: x[0])
    return inits
