        M_val = float(c.max())

        # Boltzmann weights: p_k proportional to exp(-(M_val - c_k) / width)
        xc = np.exp(np.clip(-(M_val - c) / width, -60.0, 0.0))
        s_xc = float(xc.sum())
        if s_xc < 1e-300:
            xc = np.zeros_like(c)
            xc[int(np.argmax(c))] = 1.0
            s_xc = 1.0
        p = xc / s_xc

        # h[j] = sum_k p[k] * f[k-j] (cross-correlation of p and f)
        fp = np.fft.rfft(p, N_fft)
        ff = np.fft.rfft(f, N_fft)
        h = np.fft.irfft(np.conj(fp) * ff, N_fft)[:n]

        # E_p[c] = sum_k p_k * c_k
        Ep_c = float(np.dot(p, c))

        # Gradient in logit space: grad_z[i] = 2 * f[i] * (h[i] - Ep_c)
        grad_z = 2.0 * f * (h - Ep_c)

        if not np.all(np.isfinite(grad_z)):
            break

        # Momentum update
        vel = momentum * vel + (1.0 - momentum) * grad_z

        # Try the step
        z_old = z.copy()
        z_new = z - lr * vel
        z_new = np.clip(z_new, -50.0, 50.0)

        f_new = np.exp(z_new - z_new.max())
        f_new = f_new / f_new.sum()
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        c_new = fft_convolve(f_new)
        M_new = float(c_new.max())

        if M_new < M_val - 1e-14:
            f = f_new
            z = z_new
            if M_new < best_M:
                best_M = M_new
                best_f = f_new.copy()
            no_imp = 0
        else:
            # Revert
            z = z_old
            f = np.exp(z - z.max())
            f = f / f.sum()
            no_imp += 1

            if no_imp % 40 == 0:
                lr = max(0.01, lr * 0.65)
            if no_imp > 200:
                # Perturb to escape
                rng_local = np.random.default_rng(seed_base + it * 31)
                z = z_old + 0.08 * rng_local.normal(size=n)
                z -= z.mean()
                vel = np.zeros(n)
                lr = min(2.0, lr * 2.5)
                width = max(min_width, width * 0.6)
                no_imp = 0
            if no_imp > 1500:
                # Hard reset from best
                z = np.log(best_f + 1e-30)
                z -= z.mean()
                f = best_f.copy()
                c = fft_convolve(f)
                M_val = float(c.max())
                vel = np.zeros(n)
                lr = 0.5
                width = c_range * 2.0
                no_imp = 0

        # Anneal width: decrease (sharper focus) over time
        if it % 300 == 0 and it > 0:
            width = max(min_width, width * 0.82)

        # Occasionally widen to escape
        if it % 2000 == 0 and it > 0:
            width = min(max_width, width * 1.5)

        if verbose and it % 2000 == 0:
            print(f"      iter={it}, M={M_val:.12f}, best={best_M:.12f}, "
                  f"width={width:.6f}, lr={lr:.4f}, no_imp={no_imp}")

    return best_f, best_M


def lp_direction_step(f, deadline, max_time_per_lp=5.0):
    """
    One LP-based direction step: find g maximizing sum(g) s.t. (f*g)_k <= M, g>=0.
    Returns the best (f_new, M_new) or None if LP fails.
    """
    from scipy.optimize import linprog

    n = f.size
    M = float(fft_convolve(f).max())

    # Build constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n, else 0
    # (f*g)_k = sum_j f[k-j] * g_j for valid k-j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    t_lp_start = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max_time_per_lp})
    lp_time = time.monotonic() - t_lp_start

    if not res.success or time.monotonic() > deadline:
        return None

    g = res.x
    g_sum = float(g.sum())
    if g_sum <= 1.0 + 1e-7:
        return None

    g = g / g_sum

    # Try blending
    best_blend = None
    best_M_blend = M
    for t in (0.3, 0.1, 0.5, 0.05, 0.2, 0.4, 0.08):
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-30:
            continue
        f_new = f_new / s
        M_new = float(fft_convolve(f_new).max())
        if M_new < best_M_blend:
            best_M_blend = M_new
            best_blend = f_new

    if best_blend is not None:
        return best_blend, best_M_blend
    return None


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search:
    1. Start from parent sequence, run Boltzmann gradient optimization.
    2. Periodically apply LP direction steps for better direction finding.
    3. Try interpolation to larger sequence lengths.
    4. Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent: n={parent.size}, score={sc:.10f}")
    else:
        parent = make_linear_profile(512, 1.5)
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Default start: n={parent.size}, score={sc:.10f}")

    # Phase 1: Main gradient optimization at parent length with periodic LP
    n1 = parent.size
    phase1_end = min(deadline, t_start + 900.0)

    # Alternate between gradient and LP
    f_current = parent.copy()
    grad_deadline = min(phase1_end, t_start + 120.0)

    while time.monotonic() < phase1_end:
        # Gradient optimization burst
        grad_end = min(phase1_end, time.monotonic() + 120.0)
        f_best, M_best = boltzmann_gradient_optimize(f_current, grad_end)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        f_current = f_best
        print(f"  After gradient burst: score={sc:.10f}, time_left={deadline - time.monotonic():.0f}s")

        if time.monotonic() >= phase1_end:
            break

        # LP direction step (a few iterations)
        lp_end = min(phase1_end, time.monotonic() + 60.0)
        f_lp = f_current.copy()
        for _ in range(5):
            if time.monotonic() > lp_end:
                break
            result = lp_direction_step(f_lp, lp_end, max_time_per_lp=8.0)
            if result is not None:
                f_lp, M_lp = result
                sc_lp = 2.0 * n1 * M_lp
                try_update(f_lp, sc_lp)
            else:
                break
        print(f"  After LP steps: score={best_score:.10f}")

        # Set current to best found
        if best_seq is not None and best_seq.size == n1:
            f_current = best_seq.copy()

    print(f"Phase 1 done: best={best_score:.10f} at n={n1}")

    # Phase 2: Try larger sequence length
    if best_seq is not None and time.monotonic() < deadline - 120:
        n2 = n1 * 2
        p2 = interpolate_to(best_seq, n2)
        if p2 is not None:
            grad_end = min(deadline, time.monotonic() + 200.0)
            f2, M2 = boltzmann_gradient_optimize(p2, grad_end)
            sc2 = 2.0 * n2 * M2
            try_update(f2, sc2)
            print(f"Phase 2 (n={n2}): score={sc2:.10f}")

            # A few LP steps at the larger length
            if time.monotonic() < deadline - 60:
                lp_end = min(deadline, time.monotonic() + 40.0)
                f_lp2 = f2.copy()
                for _ in range(3):
                    if time.monotonic() > lp_end:
                        break
                    result = lp_direction_step(f_lp2, lp_end, max_time_per_lp=10.0)
                    if result is not None:
                        f_lp2, M_lp2 = result
                        sc_lp2 = 2.0 * n2 * M_lp2
                        try_update(f_lp2, sc_lp2)
                    else:
                        break
                print(f"Phase 2 after LP: best={best_score:.10f}")

    # Phase 3: Also try n1*1.5 (non-power-of-2)
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_alt = int(n1 * 1.5)
        p_alt = interpolate_to(best_seq, n_alt)
        if p_alt is not None:
            grad_end = min(deadline, time.monotonic() + 80.0)
            f_alt, M_alt = boltzmann_gradient_optimize(p_alt, grad_end)
            sc_alt = 2.0 * n_alt * M_alt
            try_update(f_alt, sc_alt)
            print(f"Phase 3 (n={n_alt}): score={sc_alt:.10f}")

    # Phase 4: Final gradient polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        grad_end = min(deadline, time.monotonic() + 60.0)
        f_final, M_final = boltzmann_gradient_optimize(best_seq, grad_end)
        sc_final = 2.0 * n_best * M_final
        try_update(f_final, sc_final)
        print(f"Phase 4 final polish: score={sc_final:.10f}")

    print(f"\nFinal best: score={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Verify with evaluator on small sequences
    if best_seq is not None and best_seq.size <= 2048:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(sc_eval):
                print(f"Verified with evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
