    # The FFT gives us the "linear" cross-correlation properly if N is large enough.
    return cc[n - 1: n - 1 + n]


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.8) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):
    profs = []
    for eta in (0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5, 3.0):
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in ((1.2, -0.8), (1.5, -1.2), (1.8, -1.5), (2.0, -2.0),
                   (1.0, -0.5), (1.5, -2.5), (2.0, -3.0), (0.8, 0.5),
                   (2.5, -2.0), (3.0, -3.5)):
        profs.append(make_cubic_profile(n, a1, a3))
    t = np.linspace(0, 1, n)
    profs.append((1.0 + 1.3 * t - 0.5 * t * t + 1e-12) / (1.0 + 1.3 * t - 0.5 * t * t + 1e-12).sum())
    profs.append((1.0 + 0.8 * t + 1.2 * t ** 3 + 1e-12) / (1.0 + 0.8 * t + 1.2 * t ** 3 + 1e-12).sum())
    # Add a "bump" profile
    a = 1.0 + 0.5 * np.exp(-5.0 * (t - 0.3) ** 2) - 0.3 * np.exp(-5.0 * (t - 0.7) ** 2)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def annealed_subgradient(f0, deadline, rng, initial_beta=None, max_iter=200000,
                         lr=0.05, verbose=False):
    """Annealed subgradient with Adam for minimizing max(a*a) subject to sum(a)=1, a>=0."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    c = fft_convolve(f)
    M_val = float(c.max())
    if initial_beta is None:
        beta = max(1e-10, 0.05 * M_val)

    # Adam state
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    best_f = f.copy()
    best_M = M_val
    no_imp = 0

    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Recompute convolution periodically or every step if n is small
        if n <= 1024 or it % 1 == 0:
            c = fft_convolve(f)
            M_val = float(c.max())

        # Log-sum-exp weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        # Gradient via cross-correlation
        fa = np.fft.rfft(f, N_corr)
        fw = np.fft.rfft(w, N_corr)
        grad = np.fft.irfft(fw * np.conj(fa), N_corr)[n - 1: n - 1 + n]

        if not np.all(np.isfinite(grad)):
            f = best_f.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        # Backtracking line search
        improved = False
        for t_mult in (1.0, 0.5, 0.25, 0.1, 0.05):
            eta_t = lr * t_mult
            t_adam += 1
            m_adam = b1 * m_adam + (1.0 - b1) * grad
            v_adam = b2 * v_adam + (1.0 - b2) * (grad * grad)
            mh = m_adam / (1.0 - b1 ** t_adam)
            vh = v_adam / (1.0 - b2 ** t_adam)

            f_new = f - eta_t * mh / (np.sqrt(vh) + eps)
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-30:
                continue
            f_new = f_new / s_new

            M_new = compute_max_conv(f_new)
            if M_new < M_val - 1e-15:
                f = f_new
                M_val = M_new
                improved = True
                break
            # Reset adam if backtracking
            m_adam = b1 * m_adam  # partial reset

        if not improved:
            # Small step even if no improvement (to allow exploration)
            t_adam += 1
            m_adam = b1 * m_adam + (1.0 - b1) * grad
            v_adam = b2 * v_adam + (1.0 - b2) * (grad * grad)
            mh = m_adam / (1.0 - b1 ** t_adam)
            vh = v_adam / (1.0 - b2 ** t_adam)
            f_new = f - 0.01 * lr * mh / (np.sqrt(vh) + eps)
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-30:
                f = f_new / s_new
                M_val = compute_max_conv(f)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0

        if M_val < best_M - 1e-14:
            best_M = M_val
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        if it % 200 == 0 and it > 0:
            beta = max(1e-12, beta * 0.85)

        # Structured restart
        if no_imp > max(500, 5 * n):
            # Keep most of the best, add structured perturbation
            noise_scale = 0.1 + 0.1 * (no_imp / (10 * n + 1))
            f = best_f + noise_scale * rng.normal(size=n)
            f = np.maximum(f, 1e-12)
            f = f / f.sum()
            M_val = compute_max_conv(f)
            beta = max(1e-12, 0.05 * M_val)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 5000 == 0:
            sc = 2.0 * n * M_val
            print(f"    AnnealSG iter {it}: M={M_val:.10f}, score={sc:.8f}, beta={beta:.2e}")

    return best_f, best_M


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f0, deadline, rng, max_iter=10000, verbose=False):
    """LP-based iterative optimization. f0 normalized to sum=1."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = compute_max_conv(f)
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                f = best_f + 0.02 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                    M = best_M
                    continue
                f = f / fs
                M = compute_max_conv(f)
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue

        g = g / g_sum
        improved = False

        # Backtracking line search
        for t in (0.5, 0.3, 0.7, 0.2, 0.8, 0.4, 0.1, 0.6, 0.15, 0.25):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = compute_max_conv(f_new)
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = compute_max_conv(f)
                no_imp = 0

        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}")

    return best_f, best_M


def hybrid_optimize(f0, deadline, rng, n_phase_lp=200, n_phase_sg=50000, verbose=False):
    """
    Hybrid: alternate between LP steps and subgradient steps.
    LP gives good directions but is expensive; SG is fast but noisy.
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    beta = max(1e-10, 0.05 * M)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    N_corr = next_pow2(3 * n)

    it_total = 0
    lp_count = 0
    sg_count = 0

    while time.monotonic() < deadline:
        # Decide: LP or SG?
        use_lp = (lp_count < n_phase_lp) and (it_total % max(1, n_phase_sg // 10) == 0 or no_imp > 200)

        if use_lp and time.monotonic() < deadline - 0.5:
            g = solve_lp_direction(f, M, n)
            if g is not None:
                g_sum = float(g.sum())
                if g_sum > 1.0 + 1e-7:
                    g = g / g_sum
                    for t in (0.4, 0.2, 0.6, 0.1, 0.8, 0.3):
                        f_new = (1.0 - t) * f + t * g
                        f_new = np.maximum(f_new, 0.0)
                        fs = f_new.sum()
                        if fs < 1e-30:
                            continue
                        f_new = f_new / fs
                        M_new = compute_max_conv(f_new)
                        if M_new < M - 1e-14:
                            f = f_new
                            M = M_new
                            break

            lp_count += 1
        else:
            # SG step
            c = fft_convolve(f)
            M = float(c.max())

            x = (c - M) / beta
            ex = np.exp(np.clip(x, -100.0, 5.0))
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                ex = np.zeros_like(c)
                ex[int(np.argmax(c))] = 1.0
                s_ex = 1.0
            w = ex / s_ex

            fa = np.fft.rfft(f, N_corr)
            fw = np.fft.rfft(w, N_corr)
            grad = np.fft.irfft(fw * np.conj(fa), N_corr)[n - 1: n - 1 + n]

            if np.all(np.isfinite(grad)):
                improved = False
                for mult in (1.0, 0.5, 0.25, 0.1):
                    lr_t = 0.04 * mult
                    t_adam += 1
                    m_adam = b1 * m_adam + (1.0 - b1) * grad
                    v_adam = b2 * v_adam + (1.0 - b2) * (grad * grad)
                    mh = m_adam / (1.0 - b1 ** t_adam)
                    vh = v_adam / (1.0 - b2 ** t_adam)

                    f_new = f - lr_t * mh / (np.sqrt(vh) + eps)
                    f_new = np.maximum(f_new, 0.0)
                    s_new = f_new.sum()
                    if s_new < 1e-30:
                        continue
                    f_new = f_new / s_new
                    M_new = compute_max_conv(f_new)
                    if M_new < M - 1e-15:
                        f = f_new
                        M = M_new
                        improved = True
                        break
                    m_adam *= b1
                    v_adam *= b2
                    t_adam = max(0, t_adam - 1)

                if not improved:
                    m_adam.fill(0)
                    v_adam.fill(0)
                    t_adam = 0

            sg_count += 1

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it_total % 500 == 0 and it_total > 0:
            beta = max(1e-12, beta * 0.9)

        if no_imp > max(1000, 10 * n):
            f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-8
            f = f / f.sum()
            M = compute_max_conv(f)
            beta = max(1e-12, 0.05 * M)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0

        it_total += 1
        if verbose and it_total % 5000 == 0:
            print(f"    Hybrid iter {it_total}: M={M:.10f}, best={best_M:.10f}, "
                  f"lp={lp_count}, sg={sg_count}, no_imp={no_imp}")

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search combining annealed subgradient, LP refinement,
    and multi-n exploration with adaptive time allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # === Phase 1: Quick exploration at n=256 to find good structure ===
    n1 = 256
    phase1_end = min(deadline, t_start + 120.0)

    if time.monotonic() < phase1_end:
        # Try several starts
        starts_256 = []
        if parent is not None:
            p_int = interpolate_profile(parent, n1)
            if p_int is not None:
                starts_256.append(p_int)
        starts_256.extend(make_deterministic_profiles(n1)[:8])
        for _ in range(6):
            starts_256.append(make_smooth_random(n1, rng))

        per_start = max(3.0, (phase1_end - t_start) / len(starts_256))
        for i, p0 in enumerate(starts_256):
            if time.monotonic() > phase1_end:
                break
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per = min(phase1_end, time.monotonic() + per_start)
            p_best, M_best = annealed_subgradient(p0, per, rng, max_iter=50000, lr=0.04)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)

        print(f"After Phase 1 (n=256): best={best_score:.10f}")

    # === Phase 2: Main optimization at n=512 ===
    n2 = 512
    phase2_end = min(deadline, t_start + 700.0)

    if time.monotonic() < phase2_end:
        starts_512 = []
        # Interpolate from parent
        if parent is not None:
            p_int = interpolate_profile(parent, n2)
            if p_int is not None:
                starts_512.append(p_int)
        # Interpolate best from phase 1
        if best_seq is not None and best_seq.size != n2:
            p_int2 = interpolate_profile(best_seq, n2)
            if p_int2 is not None:
                starts_512.append(p_int2)
        # Deterministic profiles
        starts_512.extend(make_deterministic_profiles(n2)[:6])
        # Random
        for _ in range(4):
            starts_512.append(make_smooth_random(n2, rng))

        candidates_512 = []
        per_start = max(10.0, (phase2_end - t_start) / len(starts_512))

        for i, p0 in enumerate(starts_512):
            if time.monotonic() > phase2_end:
                break
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            per = min(phase2_end, time.monotonic() + per_start)
            # Use annealed subgradient for speed
            p_best, M_best = annealed_subgradient(p0, per, rng, max_iter=100000, lr=0.05)
            sc = 2.0 * n2 * M_best
            candidates_512.append((sc, p_best))
            if i % 3 == 0:
                print(f"  Phase 2 start {i}/{len(starts_512)}: best={best_score:.10f}")

        candidates_512.sort(key=lambda x: x[0])

        # Polish top candidates with hybrid (LP + SG)
        top_k = min(3, len(candidates_512))
        for i in range(top_k):
            if time.monotonic() > phase2_end - 30:
                break
            _, p0 = candidates_512[i]
            per = min(phase2_end - 5, time.monotonic() + 60.0)
            p_best, M_best = hybrid_optimize(p0, per, rng, n_phase_lp=100, n_phase_sg=20000)
            sc = 2.0 * n2 * M_best
            candidates_512[i] = (sc, p_best)

        candidates_512.sort(key=lambda x: x[0])
        if candidates_512:
            try_update(candidates_512[0][1], candidates_512[0][0])

        print(f"After Phase 2 (n=512): best={best_score:.10f}")

    # === Phase 3: Extended LP polish on best n=512 candidate ===
    phase3_end = min(deadline, t_start + 1100.0)

    if best_seq is not None and best_seq.size == n2 and time.monotonic() < phase3_end:
        per = min(phase3_end, time.monotonic() + 200.0)
        p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=50000, verbose=True)
        sc = 2.0 * n2 * M_best
        try_update(p_best, sc)
        print(f"After Phase 3 (LP n=512): best={best_score:.10f}")

        # Follow up with SG polish
        if time.monotonic() < phase3_end - 20:
            per = min(phase3_end - 5, time.monotonic() + 80.0)
            p_best2, M_best2 = annealed_subgradient(p_best, per, rng, max_iter=100000, lr=0.03)
            sc2 = 2.0 * n2 * M_best2
            try_update(p_best2, sc2)
            print(f"After Phase 3b (SG n=512): best={best_score:.10f}")

    # === Phase 4: Try n=1024 ===
    n3 = 1024
    phase4_end = min(deadline, t_start + 1500.0)

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n3)
        if p_ext is not None:
            # Quick SG exploration at n=1024
            per = min(phase4_end - 30, time.monotonic() + 120.0)
            p_best, M_best = annealed_subgradient(p_ext, per, rng, max_iter=50000, lr=0.03)
            sc = 2.0 * n3 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 SG (n=1024): score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                # Hybrid refinement
                per2 = min(phase4_end - 5, time.monotonic() + 120.0)
                p_best2, M_best2 = hybrid_optimize(p_best, per2, rng, n_phase_lp=50, n_phase_sg=10000)
                sc2 = 2.0 * n3 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 4 Hybrid (n=1024): score={sc2:.10f}")

    # Also try different n=1024 starts if time allows
    if time.monotonic() < phase4_end - 30:
        for p0 in make_deterministic_profiles(n3)[:3]:
            if time.monotonic() > phase4_end - 30:
                break
            per = min(phase4_end - 10, time.monotonic() + 40.0)
            p_best, M_best = annealed_subgradient(p0, per, rng, max_iter=20000, lr=0.03)
            sc = 2.0 * n3 * M_best
            try_update(p_best, sc)

    print(f"After Phase 4: best={best_score:.10f}")

    # === Phase 5: Final refinement on the absolute best ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 80.0)

        # Try hybrid on the best
        p_best, M_best = hybrid_optimize(best_seq, per, rng, n_phase_lp=50, n_phase_sg=10000)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"After Phase 5 hybrid: best={best_score:.10f}, n={best_seq.size}")

        # Final SG polish
        if time.monotonic() < deadline - 10:
            per2 = min(deadline - 5, time.monotonic() + 40.0)
            p_best2, M_best2 = annealed_subgradient(p_best, per2, rng, max_iter=50000, lr=0.02)
            sc2 = compute_score(p_best2)
            try_update(p_best2, sc2)
            print(f"After Phase 5b SG: best={best_score:.10f}, n={best_seq.size}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
