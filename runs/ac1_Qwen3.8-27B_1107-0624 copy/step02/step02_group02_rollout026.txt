"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-based smoothed-max gradient descent (Adam in logit space), coordinate descent,
and multi-scale refinement. No LP dependency - pure gradient-based approach with restarts.
"""
import numpy as np
import time
import math


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2 efficiently."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    conv = full_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a):
    """Full linear convolution a * a using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(e) / e.size
    return e / s


def smooth_max(c, beta):
    """Differentiable approximation of max(c) using log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    mval = c.max()
    if beta < 1e-14:
        mask = c >= mval - 1e-12
        cnt = mask.sum()
        w = np.zeros_like(c)
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return mval, w
    val = mval + beta * np.log(s)
    w = ex / s
    return val, w


def corr_grad(w, a):
    """Gradient of conv(a,a) w.r.t. a, weighted by w.
    grad[i] = 2 * sum_k w[k] * a[k-i] (with appropriate indices)
    Using correlation: h[i] = sum_k w[k] * a[k-i] for i=0..n-1
    This equals the cross-correlation, computed via FFT.
    """
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # corr[i] = sum_k w[k+i] * a[i] for the relevant range
    # Actually: d(conv[k])/da[i] = a[k-i] + a[k-i] (symmetric) = 2*a[k-i] for valid k-i
    # So grad[i] = 2 * sum_k w[k] * a[k-i]
    # This is 2 * (w * a_reversed) cross-correlation
    # cross-corr: h[i] = sum_k w[k] * a[k-i]
    # = (w corr a)[i] where corr means sum_k w[k]*a[k-i]
    # In FFT: corr = ifft(conj(fft(w)) * fft(a)) ... no
    # Actually h[i] = sum_k w[k]*a[k-i]. Let's use:
    # h = ifft(fft(w) * conj(fft(a))) gives w * a (convolution)
    # We want cross-correlation: h[i] = sum_k w[k+i]*a[i] = (w_rev corr a)
    # Simpler: h[i] = sum_j w[j] * a[j-i] ... 
    # Let me use: full correlation of w and a
    # corr_full[i] = sum_j w[j] * a[j - i + n - 1] (shifted)
    # Easiest: convolve w with reversed a
    a_rev = a[::-1]
    fw_full = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a_rev, N)
    h = np.fft.irfft(fw_full * fa_rev, N)
    # h has length 2n-1, we want the central part
    # h[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-k+j]
    # We want grad[i] = 2 * sum_k w[k] * a[k-i] for i=0..n-1
    # = 2 * (w * a)[i] in the correlation sense
    # Let me just use direct correlation via FFT:
    # grad[i] = 2 * sum_{k} w[k] * a[k-i] where 0<=k-i<n
    # This is the cross-correlation of w and a at lag i
    # xcorr[i] = sum_j w[j+i] * a[j] = sum_j w[j] * a[j-i]
    # FFT: xcorr = ifft(fft(w) * conj(fft(a)))... 
    # No: xcorr[i] = sum_j w[j] * a[j-i]
    # In circular corr: xcorr = ifft(conj(fft(w)) * fft(a)) gives (a corr w)
    # Let me just do it the simple way:
    # grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * (full cross-correlation of w and a) [i]
    # cross-corr_c[i] = sum_j w[j+i] * a[j]
    # This equals ifft(fft(w) * conjugate(fft(a))) shifted...
    
    # Simplest correct approach:
    # The gradient of (w @ conv) w.r.t. a[i] = 2 * sum_k w[k] * (d conv[k]/da[i])
    # d conv[k]/da[i] = a[k-i] + a[(k-i) mod (2n-1)]... no, for linear conv:
    # conv[k] = sum_j a[j]*a[k-j], so d conv[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-(k-i)] wait
    # d conv[k]/da[i] = a[k-i] (when 0<=k-i<n) + a[i] * d(a[k-j])/da[i]... 
    # Actually conv[k] = sum_j a[j]*a[k-j]. d/d a[i]: 
    #   term j=i gives a[k-i] (if k-i valid)
    #   term k-j=i gives j=k-i, so a[k-i] * 1 (if k-i valid)
    #   So d conv[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] when 0<=k-i<n, else a[k-i] (one term)
    #   Wait no: if j=i then a[j]*a[k-j] = a[i]*a[k-i], derivative wrt a[i] is a[k-i]
    #   If k-j=i (j=k-i) then a[j]*a[k-j] = a[k-i]*a[i], derivative wrt a[i] is a[k-i]
    #   These are the same term when j=i and k-j=i, i.e., k=2i. In that case it's one term with derivative 2*a[i].
    #   Otherwise they're two different terms each contributing a[k-i].
    # So d conv[k]/da[i] = 2*a[k-i] if k != 2i, and 2*a[i] if k = 2i. Both cases: 2*a[k-i] (since k-i=i when k=2i).
    # So: d conv[k]/da[i] = 2*a[k-i] for 0<=k-i<n, plus we need to handle the case where k-j=i and j!=i
    # Actually I think it's just: d conv[k]/da[i] = a[k-i] + a[i+(k-i)]... I'm overcomplicating.
    
    # Let me use the standard result: for C = A*A (convolution), d(A*A)[k] = A[k-i] + A[(k-i)] 
    # In the full convolution: d C[k]/dA[i] = A[k-i] (from the first A at position i) + A[k-i] (from the second A at position k-i)
    # So d C[k]/dA[i] = A[k-i] + A[i-(k-i)]... no.
    
    # C[k] = sum_j A[j] * A[k-j]
    # dC[k]/dA[i] = (d A[j]/dA[i]) * A[k-j] summed over j + A[j] * (d A[k-j]/dA[i]) summed over j
    # = A[k-i] (when j=i) + A[k-i] (when k-j=i, i.e., j=k-i)
    # = 2 * A[k-i] if 0 <= k-i < n and j=i is valid and j=k-i is valid
    # But if i = k-i (i.e., k=2i), it's the same term, so it's 2*A[i].
    # In all cases: dC[k]/dA[i] = 2*A[k-i] for 0 <= k-i < n.
    # (The factor of 2 accounts for A[i] appearing in both the "left" and "right" roles.)
    
    # So: grad[i] = 2 * sum_k w[k] * A[k-i] for 0 <= k-i < n
    # = 2 * (cross_correlation of w and A at lag i)
    # where cross_correlation[i] = sum_k w[k] * A[k-i]
    
    # Compute via FFT: let's compute the full cross-correlation
    N2 = next_pow2(2 * n - 1)
    fw2 = np.fft.rfft(w, N2)
    fa2 = np.fft.rfft(a, N2)
    # ifft(conj(fw) * fa) gives circular cross-correlation: sum_j w[j] * a[(j+i) mod N]
    # We want: xcorr[i] = sum_j w[j] * a[j - i + n - 1] (linear, shifted)
    # 
    # Standard result: ifft(conj(FFT(w)) * FFT(a))[i] = sum_j conj(w[j]) * a[(j+i) mod N]
    # For real: = sum_j w[j] * a[(j+i) mod N]
    # This is the cross-correlation at lag i (with a shifted right by i)
    # We want sum_j w[j] * a[j-i] which is cross-correlation at lag -i = corr at N-i
    # 
    # So: result = ifft(conj(fw2) * fa2)
    # result[i] = sum_j w[j] * a[(j+i) mod N2]
    # For linear (no wraparound), result[i] for i=0..n-1 gives:
    # result[0] = sum_j w[j]*a[j] 
    # result[1] = sum_j w[j]*a[j+1]
    # ...
    # result[i] = sum_j w[j]*a[j+i]
    # 
    # But we want: sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] (substituting k=j+i)
    # = result at lag -i = result at N2-i (circular)
    # Or equivalently: sum_k w[k]*a[k-i] = cross_corr_{-i} 
    # = ifft(fw * conj(fa))[i] (the other conjugation)
    
    # Let me just use: grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * (w cross-corr a)[i] where cross-corr at lag i means sum_k w[k] * a[k-i]
    # = 2 * ifft(FFT(w) * conj(FFT(a)))[i + n - 1] (with proper indexing)
    # 
    # Actually, the cleanest way:
    # Let h = ifft(FFT(w, N2) * conj(FFT(a, N2)), N2)
    # h[i] = sum_j w[j] * a[(j-i) mod N2]  (circular cross-correlation)
    # For linear: h[n-1+i] = sum_j w[j] * a[j-i-n+1]... this is getting confusing.
    
    # Let me just use the direct O(n^2) approach for small n, or the following:
    # grad = 2 * (ifft(fft(w, N) * conj(fft(a, N))) shifted)
    
    r = np.fft.irfft(fw2 * np.conj(fa2), N2)
    # r[i] = sum_j w[j] * a[(j - i + N2) mod N2]
    # For the linear part (no wraparound), when i is in [0, N2-1]:
    # r[i] includes terms where j-i >= 0 (i.e., a[j-i]) and terms where j-i < 0 (wraparound)
    # The linear cross-correlation xcorr[i] = sum_{j=i}^{n-1} w[j] * a[j-i] (for i >= 0)
    # In the circular version: r[i] = sum_{j=i}^{n-1} w[j]*a[j-i] + sum_{j=0}^{i-1} w[j]*a[j-i+N2]
    # The second sum is wraparound garbage (since a is zero-padded, a[j-i+N2] = 0 for j-i+N2 >= n)
    # Wait, a has length n, zero-padded to N2. So a[k] = 0 for k >= n.
    # j-i+N2 >= n when j-i+N2 >= n, i.e., j >= i-n+N2 = i-n+n... 
    # N2 >= 2n-1, so n-1 < N2/2. j-i+N2 >= n means j >= n - N2 + i. Since N2 >= 2n-1 > n, 
    # n-N2+i < 0 for i < N2-n+1 <= 1. So for i >= 1, the wraparound terms have j-i+N2 >= n, 
    # meaning a[j-i+N2] = 0 (since a is zero-padded to N2 and the actual a values are only at indices 0..n-1).
    # Wait no: a is zero-padded to length N2, so a[k] = original_a[k] for k < n, and 0 for k >= n.
    # j - i + N2: if j < i, then j-i+N2 > N2-i >= N2-(n-1) >= 2n-1-n+1 = n. So a[j-i+N2] = 0.
    # Great! So for i in [0, n-1], r[i] = sum_{j=i}^{n-1} w[j] * a[j-i] (no wraparound contamination).
    # This is exactly what we want: r[i] = sum_j w[j] * a[j-i] for j >= i.
    # But we also need terms where j < i: sum_{j=0}^{i-1} w[j] * a[j-i]... but j-i < 0, so a[j-i] doesn't exist.
    # In the linear convolution, a[k-i] is only defined for 0 <= k-i < n.
    # So the correct formula is: grad[i] = 2 * sum_{k=i}^{n-1} w[k] * a[k-i]
    # which is exactly r[i] as computed above! 
    # Wait, but we also need: for k < i, a[k-i] is not defined (negative index), so those terms don't contribute.
    # Hmm, but actually in the gradient formula:
    # grad[i] = 2 * sum_k w[k] * a[k-i] where the sum is over k such that 0 <= k-i < n
    # = 2 * sum_k w[k] * a[k-i] for k >= i (since k-i >= 0) and k-i < n (i.e., k < i+n, always true since k < n <= i+n)
    # = 2 * sum_{k=i}^{n-1} w[k] * a[k-i]
    # = 2 * r[i]  ✓
    
    return 2.0 * r[:n]


def make_uniform(n):
    return np.ones(n) / n


def make_linear(n, slope=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine(n, k=1, amp=0.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic(n):
    t = np.linspace(0, 1, n)
    a = 1.0 + t * t
    return a / a.sum()


def make_random_smooth(n, rng, n_harm=5):
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def interpolate_to_length(a, new_len):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == new_len:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_len)
    return np.interp(x1, x0, a)


def adam_optimize(a0, deadline, rng, lr=0.05, beta_init=0.1, verbose=False):
    """
    Adam optimization in logit space to minimize 2n*max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-12)
    s = a0.sum()
    if s < 1e-10:
        return a0, np.inf
    a0 = a0 / s
    n = a0.size

    # Initialize in logit space
    z = np.log(a0)
    z_mean = z.mean()
    z = z - z_mean

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0

    a = a0.copy()
    conv = full_convolve(a)
    max_c = conv.max()
    beta = max(1e-10, beta_init * max_c)

    best_a = a0.copy()
    best_score = 2.0 * n * max_c  # sum=1 so score = 2n*max_c
    no_improve = 0
    patience = max(500, 3 * n)

    while time.monotonic() < deadline:
        # Compute smooth max and its weights
        L, w = smooth_max(conv, beta)

        # Gradient: dL/da[i] = sum_k w[k] * d(conv[k])/da[i] = corr_grad(w, a)
        grad_a = corr_grad(w, a)

        # Project to simplex tangent space (gradient of sum(a)=1 constraint)
        mean_grad = grad_a.mean()
        grad_tan = grad_a - mean_grad

        # Transform to logit space: dz = a * grad_tan (chain rule)
        grad_z = a * grad_tan
        # Center to avoid drift
        grad_z = grad_z - grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a) - np.log(best_a).mean()
            m[:] = 0
            v[:] = 0
            t_step = 0
            continue

        # Adam update
        t_step += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z ** 2
        m_hat = m / (1 - b1 ** t_step)
        v_hat = v / (1 - b2 ** t_step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip and center
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        # Convert back to probability space
        a = softmax_stable(z)
        conv = full_convolve(a)
        max_c = conv.max()
        score = 2.0 * n * max_c  # sum(a)=1

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Decay temperature
        if t_step % 50 == 0:
            beta = max(1e-14, beta * 0.7)

        # Adaptive learning rate
        if no_improve > patience:
            # Restart with perturbation
            scale = 0.3 + 0.5 * (no_improve / (2 * patience))
            z = scale * np.log(best_a) + 0.5 * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a = softmax_stable(z)
            conv = full_convolve(a)
            beta = max(1e-10, 0.15 * conv.max())
            m[:] = 0
            v[:] = 0
            t_step = 0
            lr = max(0.01, min(0.1, lr * 1.2))
            no_improve = 0

        if verbose and t_step % 200 == 0:
            print(f"    Adam step {t_step}: score={best_score:.8f}, beta={beta:.2e}, lr={lr:.4f}")

    return best_a, best_score


def coordinate_descent(a, deadline, rng, verbose=False):
    """
    Coordinate descent: sweep through coordinates, adjusting each to minimize the objective.
    For coordinate i, the objective as a function of a[i] (with others fixed, normalized) 
    is a rational function that can be minimized analytically.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    s = a.sum()
    a = a / s

    def score_of(a_local):
        sl = a_local.sum()
        if sl < 1e-10:
            return np.inf
        conv = full_convolve(a_local)
        return 2.0 * n * conv.max() / (sl * sl)

    best_score = score_of(a)
    best_a = a.copy()
    epoch = 0

    while time.monotonic() < deadline:
        epoch += 1
        improved = False
        rng_order = rng.permutation(n)

        for i in rng_order:
            if time.monotonic() > deadline:
                break

            # Try different values for a[i]
            # Keep sum constant by not renormalizing (score is scale invariant)
            # Actually, since score is scale invariant, we can just try values
            # and the score handles the normalization implicitly.
            # But for efficiency, let's work with sum=1 constraint.
            # Fix all other elements, vary a[i].
            # a_new = a with a[i] changed to x, then normalize.
            # score(x) = 2n * max(conv(x * a_rest + e_i)) / (x + sum_rest)^2
            # This is complex due to the max. Let's just try a few values.

            old_val = a[i]
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 2.0,
                         old_val + 0.01 / n, max(old_val - 0.01 / n, 1e-10), 1e-8]

            if old_val > 0.1 / n:
                candidates.append(old_val * 0.1)
            candidates.append(1.0 / n)  # uniform value

            best_x = old_val
            best_sc = score_of(a)

            for x in candidates:
                x = max(x, 1e-12)
                a[i] = x
                sc = score_of(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_x = x
            a[i] = best_x

            if best_x != old_val:
                improved = True

        sc_now = score_of(a)
        if sc_now < best_score - 1e-14:
            best_score = sc_now
            best_a = a.copy()

        if not improved:
            # Try random perturbations
            for _ in range(3):
                idx = rng.integers(0, n)
                a[idx] = max(a[idx] * rng.uniform(0.3, 3.0), 1e-10)
                sc = score_of(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                else:
                    a[idx] = best_a[idx]

        if verbose and epoch % 5 == 0:
            print(f"    CD epoch {epoch}: score={best_score:.8f}")

        # Stop if converged
        if sc_now >= best_score - 1e-15 and not improved:
            # Try one more round with different perturbation scale
            perturb = True
            break

    return best_a, best_score


def block_optimization(a, deadline, rng, block_size=8):
    """Optimize blocks of coordinates jointly using Newton's method on the log-sum-exp objective."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    s = a.sum()
    a = a / s

    best_score = 2.0 * n * full_convolve(a).max()
    best_a = a.copy()

    n_blocks = n // block_size

    while time.monotonic() < deadline:
        improved = False
        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size
            end = start + block_size
            block = a[start:end].copy()
            others = np.delete(a, range(start, end))
            sum_others = others.sum()

            # For the block, we want to find values that minimize the score
            # This is hard in general due to the max, so use a few iterations of gradient descent
            for _ in range(10):
                a_block = np.concatenate([others[:n-start] if start > 0 else others, 
                                          block,
                                          others[n-start:] if start + block_size < n else np.array([])])
                # This is getting complicated; skip for now
                pass
        break  # Skip this for simplicity

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: gradient descent in logit space + coordinate descent + multi-scale.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc = compute_score(init_seq)
                update(init_seq, sc)
                print(f"Initial: n={init_seq.size}, score={sc:.10f}")
    except Exception:
        pass

    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.1f}s")

    # Phase 1: Optimize at multiple scales starting from various profiles
    # Allocate time: 60% to main optimization, 25% to coordination, 15% to polish
    t_phase1_end = min(deadline, t_start + 0.55 * (deadline - t_start))
    t_phase2_end = min(deadline, t_start + 0.82 * (deadline - t_start))
    t_phase3_end = deadline

    # Try multiple starting points and lengths
    lengths_to_try = [256, 512, 1024]
    
    if init_seq is not None:
        # Also try at the same length
        n_init = init_seq.size
        if n_init not in lengths_to_try:
            lengths_to_try.insert(0, n_init)

    # Phase 1: Gradient descent from various starts
    for n_try in lengths_to_try:
        if time.monotonic() > t_phase1_end:
            break
        print(f"\n=== Phase 1: n={n_try} ===")
        
        # Generate several starting profiles

        if init_seq is not None:

        starts.append(make_cosine(n_try, 2, 0.3))

        starts.append(make_uniform(n_try))
        starts.append(make_random_smooth(n_try, rng, 4))
        starts.append(make_random_smooth(n_try, rng, 6))
        
        # Time per start
        time_for_starts = max(10.0, (t_phase1_end - time.monotonic()) / len(starts))

        for si, start in enumerate(starts):
            if time.monotonic() > t_phase1_end:
                break
            per_deadline = min(t_phase1_end, time.monotonic() + time_for_starts)
            
            sc_start = compute_score(start)
            update(start, sc_start)
            
            if sc_start < best_score + 0.5:
                # Worth optimizing
                result, sc_res = adam_optimize(
                    start, per_deadline, rng,
                    lr=0.04, beta_init=0.15, verbose=False
                )
                update(result, sc_res)
                if sc_res < best_score + 0.1:
                    print(f"  Start {si} (n={n_try}): {sc_res:.10f}")

    print(f"\nAfter Phase 1: best={best_score:.10f}, n={best_seq.size if best_seq is not None else '?'}")

    # Phase 1.5: Try to improve with different lr and beta
    if best_seq is not None and time.monotonic() < t_phase1_end:
        n_best = best_seq.size
        for lr_try in [0.02, 0.06, 0.01]:
            if time.monotonic() > t_phase1_end:
                break
            per_dl = min(t_phase1_end, time.monotonic() + 60.0)
            a0 = best_seq + 0.002 * rng.standard_normal(n_best)
            a0 = np.maximum(a0, 1e-12)
            a0 = a0 / a0.sum()
            result, sc = adam_optimize(a0, per_dl, rng, lr=lr_try, beta_init=0.1)
            update(result, sc)
            if sc < best_score:
                print(f"  lr={lr_try}: {sc:.10f}")

    # Phase 2: Coordinate descent refinement
    if best_seq is not None and time.monotonic() < t_phase2_end:
        print(f"\n=== Phase 2: Coordinate descent ===")
        cd_deadline = min(t_phase2_end, time.monotonic() + 0.25 * (deadline - t_start))
        result, sc = coordinate_descent(best_seq, cd_deadline, rng, verbose=True)
        update(result, sc)
        print(f"After CD: {sc:.10f}")

        # Try CD from a slightly perturbed version
        if time.monotonic() < t_phase2_end:
            a_pert = best_seq + 0.001 * rng.standard_normal(best_seq.size)
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / a_pert.sum()
            cd_dl2 = min(t_phase2_end, time.monotonic() + 60.0)
            result2, sc2 = coordinate_descent(a_pert, cd_dl2, rng, verbose=False)
            update(result2, sc2)
            print(f"After CD (perturbed): {sc2:.10f}")

    # Phase 3: Final Adam polish with very low temperature
    if best_seq is not None and time.monotonic() < t_phase3_end:
        print(f"\n=== Phase 3: Final polish ===")
        n_polish = best_seq.size
        polish_deadline = min(t_phase3_end, time.monotonic() + 120.0)
        
        for attempt in range(3):
            if time.monotonic() > t_phase3_end:
                break
            a0 = best_seq + 0.0005 * rng.standard_normal(n_polish)
            a0 = np.maximum(a0, 1e-12)
            a0 = a0 / a0.sum()
            per_dl = min(t_phase3_end, time.monotonic() + 40.0)
            result, sc = adam_optimize(a0, per_dl, rng, lr=0.01, beta_init=0.05)
            update(result, sc)
            print(f"  Polish attempt {attempt}: {sc:.10f}")

    # Final: Try a few more coordinate descent passes
    if best_seq is not None and time.monotonic() < deadline - 5:
        final_dl = min(deadline - 5, time.monotonic() + 30.0)
        result, sc = coordinate_descent(best_seq, final_dl, rng, verbose=False)
        update(result, sc)
        print(f"Final CD: {sc:.10f}")

    # Prepare output
    if best_seq is None:
        best_seq = make_uniform(256)

    final_seq = np.maximum(best_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    fs = final_seq.sum()
    if fs < 0.01:
        final_seq = make_uniform(256)
    else:
        # The score is scale-invariant, so we can output at any scale
        # Keep sum = 1 for cleanliness
        final_seq = final_seq / fs

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    # Validate
    try:
        eval_score = evaluate_sequence([float(x) for x in final_seq])
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score < best_score:
                best_score = eval_score
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
