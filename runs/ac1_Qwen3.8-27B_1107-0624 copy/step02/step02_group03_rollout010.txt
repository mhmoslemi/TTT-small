    """Objective in logit space. z is the log of the sequence (centered)."""
    a = np.exp(z)
    a = a / a.sum()
    c = np.convolve(a, a)

    x = x - x.max()  # numerical stability
    return 2.0 * n * beta * np.log(np.exp(x).sum())

def soft_max_grad(z, beta, n):

    a = a / a.sum()
    c = np.convolve(a, a)
    x = c / beta
    x = x - x.max()

    grad_z = a * (grad_a - dot)
    return 2.0 * n * grad_z

import time
from scipy.optimize import minimize, linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def compute_soft_max_grad(a, beta):
    """
    Compute gradient of 2n * soft_max_beta(conv(a,a)) w.r.t. a.
    soft_max_beta(c) = beta * log(sum(exp(c_j / beta)))
    """
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    x = (c - M) / beta
    exp_x = np.exp(x)
    w = exp_x / exp_x.sum()
    # grad_a[i] = 2 * n * sum_{j=0}^{n-1} w[i+j] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(w[i:i + n], a)
    return 2.0 * n * grad_a, M, w


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=30.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M.
    Returns the optimal g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def logit_objective_and_grad(z, beta, n):
    """
    Objective and gradient in logit space.
    z is centered (sum exp(z) after normalization gives the sequence).
    Minimizes 2n * soft_max_beta(conv(a,a)) where a = exp(z - logsumexp(z)).
    """
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum

    c = np.convolve(a, a)
    M = c.max()
    x = (c - M) / beta
    exp_x = np.exp(x)
    w = exp_x / exp_x.sum()

    # Objective value: 2n * (M + beta * log(sum(exp((c-M)/beta))))
    obj = 2.0 * n * (M + beta * np.log(exp_x.sum()))

    # Gradient w.r.t. a: 2n * sum_k w[k] * 2*a[k-i] -> 2n * (correlation)
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(w[i:i + n], a)
    grad_a = 2.0 * n * grad_a

    # Chain rule: z -> a = exp(z)/sum(exp(z))
    # da_i/dz_j = a_i * (delta_{ij} - a_j)
    # grad_z_j = sum_i grad_a_i * da_i/dz_j = sum_i grad_a_i * a_i * (delta_{ij} - a_j)
    # = a_j * grad_a_j - a_j * sum_i (grad_a_i * a_i)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    # Center z to maintain the gauge freedom
    grad_z = grad_z - np.mean(grad_z)

    return obj, grad_z


def lbfgs_optimize(a0, deadline, beta_start=0.1, beta_end=0.001, max_iter=200, verbose=True):
    """
    Run L-BFGS on the soft-max objective in logit space.
    a0 should be normalized to sum 1.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    best_a = a0.copy()
    best_score = internal_score(a0)

    current_beta = beta_start
    lbfgs_iter = 0

    while time.monotonic() < deadline and lbfgs_iter < max_iter:
        # Run L-BFGS for a batch of iterations
        result = minimize(
            fun=lambda z: logit_objective_and_grad(z, current_beta, n)[0],
            x0=z,
            jac=lambda z: logit_objective_and_grad(z, current_beta, n)[1],
            method='L-BFGS-B',
            options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 20}
        )

        z = result.x
        z = z - z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        score = internal_score(a)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose:
            print(f"    L-BFGS iter {lbfgs_iter}: beta={current_beta:.5f}, score={best_score:.8f}")

        # Decrease beta
        if lbfgs_iter % 5 == 0:
            current_beta = max(beta_end, current_beta * 0.5)

        lbfgs_iter += 1

        if not result.success and ' g* > 1.000000* g|' in result.message:
            # Gradient is fine, just hit max iterations - continue with smaller beta
            pass

    return best_a, best_score


def gradient_polish(a0, deadline, beta_start=0.05, lr_start=0.5, verbose=True):
    """
    Gradient descent in logit space with Adam, decreasing beta.
    a0 should be normalized to sum 1.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    best_a = a0.copy()
    best_score = internal_score(a0)

    beta = beta_start
    lr = lr_start
    base_lr = lr

    # Adam state
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        t_adam += 1

        c = np.convolve(a0, a0)
        M = c.max()

        # Soft max weights
        x = (c - M) / beta
        exp_x = np.exp(x)
        w = exp_x / exp_x.sum()

        # Gradient
        grad_a = np.zeros(n)
        for i in range(n):
            grad_a[i] = 2.0 * np.dot(w[i:i + n], a0)
        grad_a = 2.0 * n * grad_a

        # Logit space gradient
        dot = np.dot(a0, grad_a)
        gz = a0 * (grad_a - dot)
        gz = gz - np.mean(gz)

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        score = internal_score(a)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Decay beta periodically
        if it % 100 == 0:
            beta = max(0.0005, beta * 0.5)

        # Adjust LR based on progress
        if no_imp % 500 == 0:
            lr = max(0.01, lr * 0.8)

        # Restart mechanism
        if no_imp > 3000:
            rng = np.random.default_rng(it)
            z = 0.85 * (np.log(np.maximum(best_a, 1e-15)) - np.log(np.maximum(best_a, 1e-15)).mean())
            z = z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            beta = max(0.01, beta * 2)

        a0 = a

        if verbose and it % 500 == 0:
            print(f"    Grad iter {it}: beta={beta:.6f}, lr={lr:.4f}, score={best_score:.8f}")

    return best_a, best_score


def lp_refine(f, deadline, max_iters=100, verbose=True):
    """
    LP-based refinement. f should be normalized to sum 1.
    Returns (best_f, best_score).
    """
    n = len(f)
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M

    t_step = 0.3
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=60.0)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step
        g_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_score < best_score * 0.999:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            # Blended step
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            score_new = 2.0 * n * float(np.convolve(f_new, f_new).max())

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                M = float(np.convolve(f_new, f_new).max())
                f = f_new.copy()
                no_improve = 0
                t_step = max(0.02, t_step * 0.99)
            else:
                no_improve += 1
                f = best_f.copy()
                M = float(np.convolve(f, f).max())
                if no_improve % 10 == 0:
                    t_step = max(0.02, t_step * 0.7)
                if no_improve > 50:
                    break

        if verbose and it % 5 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={M:.8f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_profile(n, eta, skew=0.0):
    """Create a profile: a(t) = (1 + eta*(2t-1) + skew*(2t-1)^2) clipped positive."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u + skew * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()
