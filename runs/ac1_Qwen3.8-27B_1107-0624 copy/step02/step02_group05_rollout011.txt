If we want conv(f,f) to be as flat as possible, we want f to be "anti-concentrated" in the sense that large values of f are paired with small values in the convolution sum.

Actually, let me think about this problem in terms of "balancing". We have a 2n-1 dimensional vector conv(f,f) that we want to make as flat as possible. Each entry conv[k] = sum_j f[j]*f[k-j].

For non-negative f, the best we can do is approximate a flat convolution. The "Golay-like" structure suggests that the optimal f might have a specific pattern of high and low values.

"""
Multi-phase optimization for minimizing 2*n*max(convolve(f,f))/sum(f)^2.
Uses FFT-based natural gradient descent, LP direction injection,
multi-scale exploration, and symmetric/asymmetric variants.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-10:
        return np.inf
    c = fft_convolve(f, f)
    return 2.0 * n * float(c.max()) / (s * s)


def normalize(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f / s


def make_profile(n, ptype, rng):
    """Generate a profile of type ptype with random parameters."""
    t = np.linspace(0, 1, n)
    u = 2.0 * t - 1.0
    if ptype == 0:  # linear
        slope = rng.uniform(0.5, 4.0)
        f = 1.0 + slope * t
    elif ptype == 1:  # quadratic
        a = rng.uniform(-2, 2)
        b = rng.uniform(-1, 1)
        f = 1.0 + a * u + b * u * u
    elif ptype == 2:  # cubic
        a = rng.uniform(-3, 3)
        b = rng.uniform(-1, 2)
        f = 1.0 + a * u + b * u ** 3
    elif ptype == 3:  # gaussian
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    elif ptype == 4:  # u-shape
        c = rng.uniform(0.5, 3.0)
        f = 1.0 + c * u * u
    elif ptype == 5:  # sinusoidal
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.8)
        f = 1.0 + amp * np.sin(2 * np.pi * freq * t + phase)
    elif ptype == 6:  # power law
        p = rng.uniform(0.3, 3.0)
        f = np.maximum(t, 1e-10) ** p + np.maximum(1 - t, 1e-10) ** p + 0.05
    elif ptype == 7:  # asymmetric power
        p1 = rng.uniform(0.2, 2.0)
        p2 = rng.uniform(0.2, 2.0)
        f = np.maximum(t, 1e-10) ** p1 + np.maximum(1 - t, 1e-10) ** p2 + 0.05
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c_neg = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_neg, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def soft_max_weights(c, beta):
    """Compute soft-max weights and value."""
    m = float(c.max())
    x = (c - m) / beta
    x = x - x.max()  # additional stability
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    return m + beta * np.log(s), w


def natural_gradient(f, w, n):
    """Compute natural gradient in logit space for soft-max objective."""
    N = 1
    while N < 2 * n:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(f[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    grad = 2.0 * h[n - 1:2 * n - 1]
    # Natural gradient: gz[j] = f[j] * (grad[j] - dot(f, grad))
    dot = np.dot(f, grad)
    gz = f * (grad - dot)
    return gz

        if symmetric:
            g = 0.5 * (g + g[::-1])
            g = normalize(g)
            if g is None:
                continue

        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.7, 0.5, 0.3, 0.15, 0.08, 0.04]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize(f_try)
            if f_try is None:
                continue
            M_try = float(fft_convolve(f_try, f_try).max())
            if M_try < best_M_try - 1e-14:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 40:
            f = 0.7 * best_f + 0.3 * normalize(best_f + 0.2 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            if symmetric:
                f = 0.5 * (f + f[::-1])
                f = normalize(f)
            M = float(fft_convolve(f, f).max())
            no_imp = 0

    score = 2.0 * n * best_M
    return best_f, score


def combined_optimize(f_init, deadline, rng, symmetric=False):
    """Alternate between gradient and LP steps."""
    f = normalize(f_init.copy())
    if f is None:
        return f_init, compute_score(f_init)
    n = len(f)

    if symmetric:
        f = 0.5 * (f + f[::-1])
        f = normalize(f)

    c = fft_convolve(f, f)
    M = float(c.max())
    beta = max(1e-15, 0.15 * M)

    z = np.log(np.maximum(f, 1e-30))
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = 0.03

    best_f = f.copy()
    best_M = M
    no_imp = 0
    lp_interval = 20  # do LP every 20 gradient steps
    count = 0

    while time.monotonic() < deadline:
        if count % lp_interval == 0 and count > 0:
            # LP step
            g = solve_lp(f, M)
            if g is not None:
                g_sum = float(g.sum())
                if g_sum > 1e-15:
                    g = g / g_sum
                    if symmetric:
                        g = 0.5 * (g + g[::-1])
                        g = normalize(g)
                    if g is not None:
                        for t in [0.5, 0.3, 0.15]:
                            f_try = (1.0 - t) * f + t * g
                            f_try = normalize(f_try)
                            if f_try is None:
                                continue
                            M_try = float(fft_convolve(f_try, f_try).max())
                            if M_try < M:
                                f = f_try
                                M = M_try
                                break
        else:
            # Gradient step
            c = fft_convolve(f, f)
            m, w = soft_max_weights(c, beta)
            gz = natural_gradient(f, w, n)

            if np.all(np.isfinite(gz)):
                t_adam += 1
                m_adam = b1 * m_adam + (1 - b1) * gz
                v_adam = b2 * v_adam + (1 - b2) * gz * gz
                mh = m_adam / (1 - b1 ** t_adam)
                vh = v_adam / (1 - b2 ** t_adam)
                z = z - lr * mh / (np.sqrt(vh) + eps)
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                if symmetric:
                    z = 0.5 * (z + z[::-1])
                e = np.exp(z)
                f = e / e.sum()

        c = fft_convolve(f, f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam > 0 and t_adam % 80 == 0:
            beta = max(1e-18, beta * 0.85)

        if no_imp > 400:
            z_pert = np.log(np.maximum(best_f, 1e-30)) + 0.25 * rng.normal(size=n)
            z_pert = np.clip(z_pert, -60, 60)
            z_pert = z_pert - z_pert.mean()
            if symmetric:
                z_pert = 0.5 * (z_pert + z_pert[::-1])
            e = np.exp(z_pert)
            f = e / e.sum()
            c = fft_convolve(f, f)
            beta = max(1e-15, 0.15 * float(c.max()))
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            lr = 0.03
            no_imp = 0

        count += 1

    score = 2.0 * n * best_M
    return best_f, score


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return result


def extend_symmetric(f, n_new):
    """Extend a sequence to n_new by symmetric addition of smoothed edges."""
    n0 = len(f)
    if n0 >= n_new:
        return interpolate_profile(f, n_new)
    # Simple interpolation
    return interpolate_profile(f, n_new)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: gradient + LP + multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            s = float(np.sum(f))
            if s > 0.01:
                best_seq = f.copy()
                best_score = float(sc)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                print(f"Loaded parent: n={len(parent)}, score={compute_score(parent):.10f}")
    except Exception:
        pass

    # Phase 1: Quick gradient optimization from diverse starts at n=256
    n1 = 256
    phase1_end = t_start + 500.0
    print(f"Phase 1: Gradient optimization at n={n1}")

    starts = []
    if parent is not None:
        if len(parent) == n1:
            starts.append(parent.copy())
            starts.append(interpolate_profile(parent, n1))
        else:
            starts.append(interpolate_profile(parent, n1))
        # Symmetric version of parent
        p_sym = 0.5 * (starts[0] + starts[0][::-1])
        starts.append(normalize(p_sym))

    ptypes = [0, 1, 2, 3, 4, 5, 6, 7]
    for pt in ptypes:
        for _ in range(3):
            starts.append(make_profile(n1, pt, rng))

    # Evaluate all starts
    for s in starts:
        sc = compute_score(s)
        try_update(s, sc)

    print(f"  Initial best: {best_score:.10f}")

    # Run gradient optimization from top starts
    scored_starts = []
    for s in starts:
        sc = compute_score(s)
        scored_starts.append((sc, s))
    scored_starts.sort(key=lambda x: x[0])

    top_k = min(8, len(scored_starts))
    results = []
    for i in range(top_k):
        remaining = phase1_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, top_k - i), 60.0)
        if alloc < 10:
            break
        run_deadline = time.monotonic() + alloc
        sym = (i % 2 == 0)
        f_best, sc = gradient_optimize(scored_starts[i][1], run_deadline, rng,
                                       lr=0.03, symmetric=sym)
        try_update(f_best, sc)
        results.append((sc, f_best))

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")

    # Phase 2: LP refinement on best gradient results
    phase2_end = t_start + 900.0
    print(f"Phase 2: LP refinement")

    top_lp = min(4, len(results))
    for i in range(top_lp):
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining / max(1, top_lp - i), 100.0)
        if alloc < 15:
            break
        run_deadline = time.monotonic() + alloc
        f_best, sc = lp_optimize(results[i][1], run_deadline, rng, max_iters=200)
        try_update(f_best, sc)
        print(f"  LP {i+1}/{top_lp}: score={sc:.10f}, best={best_score:.10f}")

    # Phase 3: Combined gradient + LP polish
    phase3_end = t_start + 1200.0
    print(f"Phase 3: Combined optimization")

    if best_seq is not None:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            f_best, sc = combined_optimize(best_seq, time.monotonic() + remaining, rng,
                                           symmetric=False)
            try_update(f_best, sc)
            print(f"  Combined 1: score={sc:.10f}, best={best_score:.10f}")

        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            f_best, sc = combined_optimize(best_seq, time.monotonic() + remaining, rng,
                                           symmetric=True)
            try_update(f_best, sc)
            print(f"  Combined 2 (sym): score={sc:.10f}, best={best_score:.10f}")

    # Phase 4: Multi-scale exploration
    phase4_end = t_start + 1550.0
    print(f"Phase 4: Multi-scale")

    for n_try in [128, 384, 512]:
        remaining = phase4_end - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining / 3.0, 100.0)
        if alloc < 20:
            break

        f_ext = interpolate_profile(best_seq, n_try)
        f_ext = normalize(f_ext)

        # Quick gradient at new scale
        run_dl = time.monotonic() + alloc * 0.6
        f_best, sc = gradient_optimize(f_ext, run_dl, rng, lr=0.02, symmetric=False)
        try_update(f_best, sc)
        print(f"  n={n_try} gradient: score={sc:.10f}, best={best_score:.10f}")

        # LP at new scale
        remaining = phase4_end - time.monotonic()
        if remaining > 30 and len(f_best) == n_try:
            run_dl = time.monotonic() + min(remaining * 0.7, 60.0)
            f_best2, sc2 = lp_optimize(f_best, run_dl, rng, max_iters=100)
            try_update(f_best2, sc2)
            print(f"  n={n_try} LP: score={sc2:.10f}, best={best_score:.10f}")

    # Phase 5: Final intensive polish at best sequence length
    phase5_end = deadline
    print(f"Phase 5: Final polish")

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = phase5_end - time.monotonic()
        if remaining > 30:
            # Intensive gradient with small learning rate
            run_dl = time.monotonic() + remaining * 0.5
            f_best, sc = gradient_optimize(best_seq, run_dl, rng, lr=0.01,
                                           max_no_imp=300, symmetric=False)
            try_update(f_best, sc)
            print(f"  Final gradient: score={sc:.10f}, best={best_score:.10f}")

        remaining = phase5_end - time.monotonic()
        if remaining > 20:
            run_dl = time.monotonic() + remaining * 0.4
            f_best, sc = combined_optimize(best_seq, run_dl, rng, symmetric=False)
            try_update(f_best, sc)
            print(f"  Final combined: score={sc:.10f}, best={best_score:.10f}")

    # Phase 6: Try random perturbations of best and re-optimize quickly
    if time.monotonic() < deadline - 30:
        print(f"Phase 6: Perturbation search")
        for attempt in range(5):
            if time.monotonic() > deadline - 10:
                break
            pert = best_seq + 0.05 * rng.normal(size=len(best_seq))
            pert = normalize(pert)
            if pert is None:
                continue
            run_dl = min(deadline - 10, time.monotonic() + 25.0)
            f_best, sc = gradient_optimize(pert, run_dl, rng, lr=0.02, max_no_imp=100)
            try_update(f_best, sc)

    # Final validation
    if best_seq is None:
        best_seq = make_profile(256, 0, rng)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_profile(256, 0, rng)
        s = float(best_seq.sum())
    best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score):
            final_score = ev_score
    except Exception as e:
        print(f"evaluate error: {e}")

    print(f"FINAL: n={len(best_seq)}, score={final_score:.10f}")
    print(f"Time used: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_seq]
