"""
Optimized step function search using ramp-initialized Adam optimization.
Key insight: non-symmetric (ramp-like) sequences have their convolution peak 
off-center, allowing much lower peak-to-mean ratios than symmetric ones.
A linear ramp alone gives score ~4/3 vs theoretical bound ~1.0.
We polish with Adam in log-space using sharp soft-max gradients.

def compute_score(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_conv_max(f):
    """Compute max of convolution."""
    return float(np.convolve(f, f).max())


def make_ramp(n, alpha=1.0, offset=0.0):
    """Make a power ramp: f_i = (i+1+offset)^alpha, normalized to sum 1."""
    t = np.arange(n, dtype=np.float64) + 1.0 + offset
    f = np.power(np.maximum(t, 1e-10), alpha)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return f / s


def make_asymmetric_ramp(n, alpha=1.0, gamma=0.5):
    """Asymmetric ramp: rises with power alpha, falls with power gamma."""
    t = np.linspace(0.0, 1.0, n)
    peak_pos = 0.7
    if t <= peak_pos:
        f = np.power(t / peak_pos + 0.1, alpha)
    else:
        f = np.power((1 - t) / (1 - peak_pos) + 0.1, gamma)
    f = np.maximum(f, 1e-15)
    return f / f.sum()


def make_cosine_ramp(n, exponent=1.0):
    """Cosine-weighted ramp."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    f = t + 0.1
    w = np.cos(np.pi * t / 2) ** exponent + 0.1
    f = f * w
    f = np.maximum(f, 1e-15)
    return f / f.sum()


def fft_convolve(f, g):
    """Linear convolution via FFT."""
    n = len(f)
    m = len(g)
    N = 1
    while N < n + m - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    G = np.fft.rfft(g, N)
    return np.fft.irfft(F * G, N)[:n + m - 1]


def soft_topk_weights(c, k, beta):
    """Compute weights for top-k soft-max."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        idx = np.argsort(c)[-k:]
        w[idx] = 1.0 / k
        return w
    w = ex / s
    return w

def propose_candidate(seed=42, budget_s=1800, **kwargs):

    Phase 1: Evaluate ramp profiles at various n and alpha values
    Phase 2: Adam polish on best profiles

    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)
            print(f"  NEW BEST: score={sc:.10f}, n={len(f)}")

    print("Phase 1: Evaluating ramp profiles...")
    phase1_end = min(deadline, t_start + 60.0)

    alpha_values = [0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]
    
    candidates = []
    for n in n_values:
        for alpha in alpha_values:
            f = make_ramp(n, alpha)
            sc = compute_score(f)

            try_update(f, sc)
    
    # Also try asymmetric ramps

        for alpha in [1.0, 1.5, 2.0]:

            f = make_cosine_ramp(n, exp)
            sc = compute_score(f)
            candidates.append((sc, n, exp, f))
            try_update(f, sc)

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best initial: {best_score:.10f}")

    phase2_end = min(deadline, t_start + 600.0)

                                             beta_scale=0.05, restart_patience=150)
        sc = 2.0 * n * M_best  # since sum is normalized to 1
        try_update(f_best, sc)
        results.append((sc, f_best))
        print(f"  Adam {i+1}/{top_k} (n={n}, a={alpha}): score={sc:.10f}, best={best_score:.10f}")

    results.sort(key=lambda x: x[0])

    # Phase 3: Second round of Adam with different hyperparameters on best results
    print("Phase 3: Adam round 2 with different params...")
    phase3_end = min(deadline, t_start + 1200.0)
    
    top_k2 = min(6, len(results))
    for i in range(top_k2):
        if time.monotonic() > phase3_end:
            break
        sc_prev, f_prev = results[i]
        n = len(f_prev)
        remaining = phase3_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, top_k2 - i), 90.0)

            if time.monotonic() > phase3_end:
                break
            run_deadline = min(phase3_end, time.monotonic() + alloc / 2)
            f_best, M_best = adam_ramp_optimize(f_prev, run_deadline, rng, lr=lr_try,
                                                 beta_scale=0.03, restart_patience=300)
            sc = 2.0 * n * M_best
            try_update(f_best, sc)
            results.append((sc, f_best))
            print(f"  Adam2 {i+1}/{top_k2} lr={lr_try}: score={sc:.10f}, best={best_score:.10f}")

    results.sort(key=lambda x: x[0])

    # Phase 4: Try extending best solution to larger n, or reducing to smaller n
    print("Phase 4: Size adjustment...")
    phase4_end = min(deadline, t_start + 1500.0)
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Try interpolating to nearby sizes
        for n_try in [n_best * 2, n_best // 2, n_best + 64, n_best - 64]:
            if n_try < 32 or n_try > 1024 or n_try % 32 != 0:
                continue
            if time.monotonic() > phase4_end - 30:
                break
            t_old = np.linspace(0, 1, n_best)
            t_new = np.linspace(0, 1, n_try)
            f_interp = np.interp(t_new, t_old, best_seq)
            f_interp = np.maximum(f_interp, 0)
            s = f_interp.sum()
            if s > 1e-10:
                f_interp = f_interp / s
            sc = compute_score(f_interp)
            try_update(f_interp, sc)
            
            if sc < best_score + 0.01:  # Close to best, worth polishing
                remaining = phase4_end - time.monotonic()
                if remaining > 30:
                    run_deadline = time.monotonic() + min(remaining, 60)
                    f_pol, M_pol = adam_ramp_optimize(f_interp, run_deadline, rng,
                                                       lr=0.03, restart_patience=200)
                    sc2 = 2.0 * n_try * M_pol
                    try_update(f_pol, sc2)
                    print(f"  Size {n_try}: score={sc2:.10f}, best={best_score:.10f}")

    # Phase 5: Final polish on the overall best
    print("Phase 5: Final polish...")
    phase5_end = deadline
    
    if best_seq is not None:
        remaining = phase5_end - time.monotonic()
        if remaining > 60:
            n_best = len(best_seq)
            # Run multiple restarts from the best point
            for restart in range(3):
                if time.monotonic() > phase5_end - 10:
                    break
                alloc = (phase5_end - 10 - time.monotonic()) / (3 - restart)
                if alloc < 20:
                    break
                # Perturb best
                f_pert = best_seq + 0.05 * rng.normal(size=n_best)
                f_pert = np.maximum(f_pert, 1e-10)
                f_pert = f_pert / f_pert.sum()
                run_deadline = time.monotonic() + alloc
                f_fin, M_fin = adam_ramp_optimize(f_pert, run_deadline, rng,
                                                   lr=0.02, restart_patience=400)
                sc_fin = 2.0 * n_best * M_fin
                try_update(f_fin, sc_fin)
                print(f"  Final restart {restart+1}: score={sc_fin:.10f}, best={best_score:.10f}")
        
        # Also try from the unperturbed best
        if time.monotonic() < phase5_end - 30:
            alloc = phase5_end - 30 - time.monotonic()
            if alloc > 20:
                f_fin, M_fin = adam_ramp_optimize(best_seq, time.monotonic() + alloc,
                                                  rng, lr=0.01, restart_patience=500)
                sc_fin = 2.0 * len(best_seq) * M_fin
                try_update(f_fin, sc_fin)
                print(f"  Final clean: score={sc_fin:.10f}, best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = make_ramp(256, 1.0)
    
    best_seq = np.maximum(best_seq, 0.0).astype(np.float64)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_ramp(256, 1.0)
        s = float(best_seq.sum())
    best_seq = best_seq / s
    
    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")
    
    # Verify with evaluate_sequence
    try:
        ev = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in best_seq]
