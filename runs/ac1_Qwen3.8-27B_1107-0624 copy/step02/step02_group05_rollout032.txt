    """
    Compute approximate subgradient of max(c) where c = conv(f,f).
    Uses top-K weighted average.
    """
    n = len(f)
    M = c.max()

    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]

    vals = c[top_idx]

    grad = np.zeros(n)

For the problem: minimize `max_k (f conv f)[k]` s.t. `f >= 0`, `sum(f) = 1`.

By symmetry, if the solution is unique, it should be symmetric: `f[j] = f[n-1-j]`.

The first-order conditions (KKT) at the optimum:
- For the active set `S = {k: (f conv f)[k] = M}`:
  - `grad_M(f) = lambda * 1 + mu` where `mu_i >= 0` for `f_i = 0` and `mu_i = 0` for `f_i > 0`.
  - Here `grad_M(f) = sum_{k in S} lambda_k * 2*f[k-·]` (weighted average of gradients at active indices).
- Complementarity: `mu_i * f_i = 0`.

This is a system of equations that's hard to solve directly, but it gives insights into the structure of the solution.

If all `f_i > 0` (interior solution), then:
`sum_{k in S} lambda_k * 2*f[k-i] = lambda` for all `i`.

This means the "weighted correlation" of `f` with the active set indicators is constant. In other words, the active set indicators, when correlated with `f`, give a flat response.

This is a strong condition. It suggests that the optimal `f` is such that its correlation with each active index pattern is uniform.

For the symmetric case, if the active set is `{0, n-1, 2n-2}` (the edges of the convolution) or `{n-1}` (the center), the conditions simplify.

Actually, for a "U-shaped" f (high at edges, low in middle), the convolution peak is likely at the center (k=n-1). The edges of the convolution (k=0 and k=2n-2) have small values (they're just `f[0]^2` and `f[n-1]^2`). So the active set is likely just `{n-1}` or a small set around the center.

If the active set is just `{n-1}`, then the KKT condition is:
`2*f[n-1-i] = lambda` for all `i` (assuming interior solution).

This means `f[n-1-i] = lambda/2` for all `i`, i.e., `f` is constant! But a constant `f` gives a triangular convolution with peak at the center, and the score is 2. So this can't be optimal (we can do better).

The issue is that the KKT conditions are necessary but not sufficient for non-convex problems. The "constant f" solution is a stationary point but not a minimum (it's more like a saddle point or maximum in certain directions).

So the actual minimum has some `f_i = 0` at the boundary, or the active set includes multiple indices.

import numpy as np
from scipy.optimize import linprog
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast subgradient method with adaptive smoothing (FFT-based),
LP polishing, and multi-length exploration.
"""


def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def score_of(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def subgradient_of_max(f, c, top_k=20, beta_frac=0.05):
    """Approximate subgradient of max(c) w.r.t. f, where c = conv(f,f)."""
    n = len(f)
    M = float(c.max())
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    vals = c[top_idx]
    beta = beta_frac * M + 1e-30
    weights = np.exp((vals - M) / beta)
    wsum = float(weights.sum())
    if wsum < 1e-30:
        weights = np.zeros(len(top_idx))
        weights[0] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k_val = int(top_idx[i])
        lo = max(0, k_val - n + 1)
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * weights[i] * f[k_val - np.arange(lo, hi + 1)]
    return grad


def adam_subgradient_optimize(f_init, deadline, rng, top_k=20, beta_frac=0.05,
                              init_lr=0.05, max_iters=200000, verbose=False):
    """Adam optimization in log-space using subgradient method with adaptive smoothing."""
    n = len(f_init)
    f = np.maximum(f_init, 1e-30)
    f = f / f.sum()
    z = np.log(f)
    z -= z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = init_lr
    best_f = f.copy()
    best_score = score_of(f)
    no_imp = 0
    cur_beta = beta_frac
    cur_topk = top_k

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = fft_convolve(f, f)
        grad_f = subgradient_of_max(f, c, cur_topk, cur_beta)
        fg = float(f @ grad_f)
        grad_z = f * (grad_f - fg)
        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -100, 100)
        f_new = np.exp(z_new)
        f_new = f_new / f_new.sum()
        sc = score_of(f_new)
        if sc > best_score * 1.02:
            lr = max(0.001, lr * 0.85)
            no_imp += 1
        else:
            z = z_new
            f = f_new
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        # Anneal: sharpen gradient over time
        if it % 500 == 0 and it > 0:
            cur_beta = max(0.005, cur_beta * 0.85)
            cur_topk = max(5, int(cur_topk * 0.9))
        # Restart
        if no_imp > 1500:
            z = 0.8 * np.log(np.maximum(best_f, 1e-30)) + 0.2 * rng.normal(0, 0.3, n)
            z -= z.mean()
            f = np.exp(z)
            f = f / f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr = init_lr
            no_imp = 0
            cur_beta = beta_frac
            cur_topk = top_k
        if verbose and it % 5000 == 0:
            print(f"      iter {it}: sc={sc:.10f}, best={best_score:.10f}, lr={lr:.5f}, beta={cur_beta:.4f}")
    return best_f, best_score


def lp_polish(f_init, deadline, max_iters=500, lp_timeout=3.0):
    """LP-based polishing: find g maximizing sum(g) s.t. conv(f,g)<=M, g>=0."""
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return f, score_of(f)
    f = f / s
    best_f = f.copy()
    best_score = score_of(f)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        cff = fft_convolve(f, f)
        M = float(cff.max())
        # Build LP matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                          method='highs', options={'time_limit': lp_timeout})
            if not res.success or res.x is None:
                no_imp += 1
                if no_imp > 20:
                    break
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            no_imp += 1
            if no_imp > 20:
                break
            continue
        gsum = float(g.sum())
        if gsum < 1e-10:
            break
        g = g / gsum
        # Line search
        best_t = 0.0
        best_sc = score_of(f)
        for t in [0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            st = f_try.sum()
            if st < 1e-10:
                continue
            f_try = f_try / st
            sc_try = score_of(f_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t
        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = f / f.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > 80:
            break
    return best_f, best_score
