"""
Improved multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive top-K soft-max gradient descent in logit space with peak shaving,
multi-length escalation, and diverse restarts.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-10:
        c = c - mn
    return c

def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _score_normalized(p):
    """Score assuming sum(p) = 1, so score = 2*n*max(conv)."""
    n = p.size
    c = _full_convolve_fft(p)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max())

def _corr_weights(w, p):
    """Cross-correlation of w with p (gives gradient of conv weight)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _make_uniform(n):
    return np.ones(n, dtype=np.float64) / n

def _make_zernike_profile(n, order, coeff):
    """Create a profile using shifted Legendre polynomial."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if order == 1:
        poly = u
    elif order == 2:
        poly = 0.5 * (3 * u * u - 1)
    elif order == 3:
        poly = 0.5 * (5 * u ** 3 - 3 * u)
    elif order == 4:
        poly = 0.375 * (35 * u ** 4 - 30 * u ** 2 + 3)
    else:
        poly = u
    a = 1.0 + coeff * poly
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_bimodal_profile(n, width, separation):
    """Create a bimodal profile."""
    t = np.linspace(0.0, 1.0, n)
    center1 = 0.5 - separation
    center2 = 0.5 + separation
    sigma = width
    a = np.exp(-(t - center1) ** 2 / (2 * sigma * sigma)) + np.exp(-(t - center2) ** 2 / (2 * sigma * sigma))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _optimize_logit_topk(p0, deadline, rng, lr=0.05, top_k=5,
                         beta_init=0.05, beta_decay=0.95,

    Logit-space Adam with top-K soft-max awareness.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = beta_init * maxc

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    step_count = 0
    while time.monotonic() < deadline:
        step_count += 1
        # Top-K soft-max
        # Use a two-level approach: soft-max with beta for the "effective max"
        # plus a variance penalty on the top values
        
        # Compute soft-max weights
        c_shifted = c - maxc
        x = c_shifted / beta
        ex = np.exp(x)
        s = float(np.sum(ex))
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # dc_k/dp_i = p[k-i] + p[k-i] (symmetric) ... actually for self-conv:
        # c_k = sum_i p_i * p_{k-i}
        # dc_k/dp_j = p_{k-j} + p_{k-j} if j != k-j, else p_{k-j}
        # Wait, for self-convolution: c_k = sum_i p_i * p_{k-i}
        # dc_k/dp_j = p_{k-j} (from the first factor) + p_{k-j} (from the second factor)
        # But only if both indices are valid
        # Actually: dc_k/dp_j = sum over valid pairs where j appears
        # = p_{k-j} if 0 <= k-j < n (first factor gives p_j * p_{k-j}, deriv wrt p_j gives p_{k-j})
        #   + p_{k-j} if 0 <= k-j < n (second factor: p_{k-j} * p_j, deriv wrt p_j gives p_{k-j})
        # So dc_k/dp_j = 2 * p_{k-j} if 0 <= k-j < n and j != k-j
        #              = p_{k-j} if j == k-j (i.e., k = 2j)
        
        # This is the cross-correlation of w with p (with appropriate handling)
        gp = 2.0 * _corr_weights(w, p)
        
        # Add variance penalty on the central part of the convolution
        if var_weight > 0.0:
            # Penalize high values in the convolution
            c_center = c  # full conv
            c_mean = float(c_center.mean())
            # Gradient of var(c_center) with respect to p
            # This is complex, so use a simpler penalty
            gp += var_weight * (p - p.mean())
        
        gp += reg * p

        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        
        # Cosine annealing of learning rate
        progress = min(1.0, step_count / 5000.0)
        lr_eff = base_lr * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr_eff = max(lr_eff, base_lr * 0.05)
        
        z = z - lr_eff * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)

def _peak_shave(p, n_shaves=10, rng=None):
    """
    Directly identify the argmax of the convolution and reduce contributing elements.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    p = p / p.sum()
    
    for _ in range(n_shaves):
        c = _full_convolve_fft(p)
        max_idx = int(np.argmax(c))
        max_val = float(c[max_idx])
        
        # Find which elements contribute most to c[max_idx]
        # c[max_idx] = sum_i p[i] * p[max_idx - i] for valid i
        lo = max(0, max_idx - n + 1)
        hi = min(n - 1, max_idx)
        contribs = np.zeros(n)
        for i in range(lo, hi + 1):
            j = max_idx - i
            if 0 <= j < n:
                contribs[i] += p[i] * p[j]
        
        # Reduce the elements with highest contribution
        top_contrib_idx = np.argsort(contribs)[::-1][:n // 10 + 1]
        
        # Reduce those elements slightly
        reduction = 0.01
        p[top_contrib_idx] -= reduction * p[top_contrib_idx]
        p = np.maximum(p, 1e-15)
        p = p / p.sum()
    
    return p

"""
Multi-scale optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses corrected FFT-based gradient with top-K soft-max, cosine annealing,
peak shaving heuristics, and diverse multi-scale escalation.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, n)  # Ensure at least n

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < -1e-10:
        c = np.maximum(c, 0)
    return c

def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _score_norm(p):
    n = p.size
    c = _full_convolve_fft(p)
    return float(2.0 * n * c.max())

def _gradient_softmax(p, c, beta, top_k=1):
    """
    Compute gradient of soft-max of convolution with respect to p.
    Uses top-k soft-max for better conditioning.
    """
    n = p.size
    maxc = float(c.max())
    
    # Top-k based weighting
    if top_k <= 1:
        # Standard soft-max
        x = (c - maxc) / beta
        ex = np.exp(x)
        s = float(np.sum(ex))
        if s <= 0 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
    else:
        # Top-k soft-max: only consider top k values
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        max_top = float(c_top.max())
        x_top = (c_top - max_top) / beta
        ex_top = np.exp(x_top)
        s_top = float(np.sum(ex_top))
        w = np.zeros_like(c)
        if s_top > 0 and np.isfinite(s_top):
            w[idx] = ex_top / s_top
        else:
            w[int(np.argmax(c))] = 1.0
    
    # Cross-correlation: grad[j] = 2 * sum_k w[k] * p[k-j]
    # = 2 * (w ⊛ p)[j] = 2 * ifft(FFT(w) * conj(FFT(p)))[:n]
    N = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    C = np.fft.irfft(fw * np.conj(fp), N)
    return 2.0 * C[:n]

def _make_uniform(n):
    return np.ones(n, dtype=np.float64) / n

def _make_poly_profile(n, coeffs):
    """coeffs: list of (order, coeff) for Legendre polynomials."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for order, coeff_val in coeffs:
        if order == 0:
            a += coeff_val
        elif order == 1:
            a += coeff_val * u
        elif order == 2:
            a += coeff_val * (0.5 * (3 * u * u - 1))
        elif order == 3:
            a += coeff_val * (0.5 * (5 * u ** 3 - 3 * u))
        elif order == 4:
            a += coeff_val * (0.375 * (35 * u ** 4 - 30 * u ** 2 + 3))
        elif order == 5:
            a += coeff_val * (0.625 * (63 * u ** 5 - 70 * u ** 3 + 15 * u))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _optimize_logit(p0, deadline, rng, lr=0.05, top_k=3,
                    beta_init_frac=0.08, beta_decay=0.94,
                    var_weight=0.0, reg=1e-7, max_steps=100000):

    Logit-space Adam with corrected gradient and adaptive top-k.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = beta_init_frac * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    step = 0

        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

            z = best_z.copy()
            m[:], v[:] = 0, 0
            adam_t = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
