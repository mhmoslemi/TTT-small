def convolve_fft(a):
    """Full convolution of a with itself using FFT."""
    n = len(a)

        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)

    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15 or not np.isfinite(s):
        return np.inf
    c = convolve_fft(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

    """Euclidean projection onto {x >= 0, sum(x) = 1} using sorting."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return result / s
