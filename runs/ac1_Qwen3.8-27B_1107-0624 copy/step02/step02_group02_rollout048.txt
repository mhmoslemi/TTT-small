"""
Hybrid LP + peak-chasing optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses line search along LP directions, direct peak-chasing gradients, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search_along_direction(f, g, max_t=1.0, n_evals=8):
    """Find best t in [0, max_t] minimizing max(conv((1-t)f+tg, (1-t)f+tg))."""
    n = len(f)
    best_t = 0.0
    best_val = float(np.convolve(f, f).max())
    
    if n <= 512:
        # Direct evaluation
        for t in np.linspace(0.05, max_t, n_evals):
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            val = float(np.convolve(f_new, f_new).max())
            if val < best_val:
                best_val = val
                best_t = t
        # Refine around best
        if best_t > 0:
            lo = max(0.01, best_t * 0.5)
            hi = min(1.0, best_t * 1.5)
            for t in np.linspace(lo, hi, 6):
                f_new = (1 - t) * f + t * g
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s < 1e-10:
                    continue
                f_new = f_new / s
                val = float(np.convolve(f_new, f_new).max())
                if val < best_val:
                    best_val = val
                    best_t = t
    else:
        # Use FFT for larger sequences
        for t in np.linspace(0.05, max_t, n_evals):
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            c = full_convolve_fft(f_new)
            val = float(c.max())
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val


def peak_chasing_step(f, alpha=0.01, top_k=3):
    """Direct gradient step targeting the top autocorrelation peaks."""
    n = len(f)
    c = np.convolve(f, f)
    
    # Find top-k peak positions
    idx = np.argsort(c)[::-1][:top_k]
    weights = c[idx]
    weights = weights / weights.sum()
    
    # Compute gradient: for peak at position k, grad_i = 2*f[k-i] (clipped to valid range)
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, idx):
        # grad_i = 2 * f[k-i] for valid i
        # i ranges from max(0, k-(n-1)) to min(n-1, k)
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            j_lo = 0
            j_hi = k - i_lo
            grad[i_lo:i_hi+1] += w * 2.0 * f[j_lo:j_hi+1]
    
    # Step in negative gradient direction
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    f_new = f_new / s
    return f_new


def lp_optimize_improved(f_init, deadline, rng, verbose=True):
    """LP-based optimization with line search and peak-chasing complement."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)
    use_fft = n > 512
    
    if use_fft:
        conv_f = full_convolve_fft(f)
    else:
        conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t_base = 0.2
    stagnant = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        M = best_M if stagnant > 5 else float(full_convolve_fft(f).max() if use_fft else np.convolve(f, f).max())
        
        # Try LP step
        g = lp_step(f, M)
        if g is not None and g.sum() > 1e-10:
            g = g / g.sum()
            
            # Line search
            best_t, best_val = line_search_along_direction(f, g, max_t=0.5, n_evals=6)
            
            if best_t > 0.01 and best_val < best_M - 1e-10:
                f_new = (1 - best_t) * f + best_t * g
                f_new = np.maximum(f_new, 0.0)
                s_new = f_new.sum()
                if s_new > 1e-10:
                    f_new = f_new / s_new
                    f = f_new
                    if use_fft:
                        M_new = float(full_convolve_fft(f).max())
                    else:
                        M_new = float(np.convolve(f, f).max())
                    if M_new < best_M - 1e-12:
                        best_M = M_new
                        best_f = f.copy()
                        stagnant = 0
                    else:
                        stagnant += 1
                    iteration += 1
                    if verbose and iteration % 50 == 0:
                        print(f"    LP iter {iteration}: M={best_M:.10f}, t={best_t:.4f}")
                    continue
        
        # Peak-chasing fallback
        alpha = 0.02 / (1.0 + stagnant * 0.1)
        f_new = peak_chasing_step(f, alpha=alpha, top_k=3)
        if use_fft:
            M_new = float(full_convolve_fft(f_new).max())
        else:
            M_new = float(np.convolve(f_new, f_new).max())
        
        if M_new < best_M - 1e-10:
            f = f_new
            if M_new < best_M:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            iteration += 1
            if verbose and iteration % 100 == 0:
                print(f"    PC iter {iteration}: M={best_M:.10f}")
        else:
            # Try smaller alpha
            f_new2 = peak_chasing_step(f, alpha=alpha * 0.1, top_k=1)
            if use_fft:
                M_new2 = float(full_convolve_fft(f_new2).max())
            else:
                M_new2 = float(np.convolve(f_new2, f_new2).max())
            if M_new2 < best_M:
                f = f_new2
                best_M = M_new2
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        
        if verbose and iteration % 200 == 0:
            print(f"    iter {iteration}: M={best_M:.10f}, stag={stagnant}")
        
        # Restart when stuck
        if stagnant > 200:
            noise_scale = 0.05 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            stagnant = 0
            if verbose:
                print(f"    Restart at iter {iteration}, M={best_M:.10f}")
                
        # Also try structured restarts
        if stagnant > 400:
            # Random binary-like restart
            f = rng.integers(0, 3, n).astype(np.float64)
            f = f / f.sum()
            stagnant = 0
            
    return best_f


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def logit_anneal_improved(p0, deadline, rng, lr=0.06, reg=1e-8):
    """Logit-space Adam with better scheduling and peak-chasing hybrid."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0
    
    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        # Adaptive beta
        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.75)
        
        # Periodic peak-chasing injection
        if no_imp > 200 and no_imp % 200 == 0:
            f = best_p.copy()
            f_pc = peak_chasing_step(f, alpha=0.005, top_k=2)
            sc_pc = float(2.0 * n * full_convolve_fft(f_pc).max())
            if sc_pc < best_score:
                best_score = sc_pc
                best_p = f_pc.copy()
                z = np.log(np.maximum(best_p, 1e-12))
                z -= float(np.mean(z))
                m.fill(0.0)
                v.fill(0.0)
                adam_t = 0
                no_imp = 0
        
        # Restart
        if no_imp > max(1000, 4 * n):
            strategy = rng.integers(0, 3)
            if strategy == 0:
                z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            elif strategy == 1:
                # Binary-like restart
                p_rand = rng.integers(0, 3, n).astype(np.float64)
                p_rand = p_rand / p_rand.sum()
                z = np.log(np.maximum(p_rand, 1e-12))
                z -= float(np.mean(z))
            else:
                z = best_z + 0.15 * rng.normal(size=n)
            
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
        
        # Decay learning rate slowly
        if no_imp > 500:
            lr = max(0.01, lr * 0.995)
    
    return best_p, best_score


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Cosine profiles
    for k in [1, 2, 3, 4]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Sine squared
    for k in [1, 2, 3]:
        a = 0.5 + 0.5 * np.sin(np.pi * k * t) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential profiles
    for rate in [1, 2, 3]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(6):
        ndf = rng.integers(2, 6)
        coeffs = rng.uniform(-0.4, 0.4, ndf)
        a = 1.0
        for i, c in enumerate(coeffs[1:], 1):
            a = a + c * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Binary-like
    for _ in range(3):
        a = rng.integers(0, 3, n).astype(np.float64)
        s = a.sum()
        if s > 0:
            profiles.append(a / s)
    
    # Peak at edges
    a = 1.0 + 2.0 * (t ** 4)
    profiles.append(a / a.sum())
    
    return profiles


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)
