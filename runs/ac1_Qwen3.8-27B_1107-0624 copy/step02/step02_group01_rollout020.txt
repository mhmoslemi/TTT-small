        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        M_new = float(_fft_convolve(f).max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % beta_decay_every == 0:
            beta = max(1e-14, beta * beta_decay)

        if no_imp > restart_patience:
            z = (1.0 - noise_scale) * np.log(best_f) + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            M_val = float(_fft_convolve(f).max())
            beta = beta_init_frac * M_val
            m_acc.fill(0)
            v_acc.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 10000 == 0:
            sc = 2.0 * n * M_val
            print(f"    SG iter {it}: score={sc:.10f}, best={2*n*best_M:.10f}, "
                  f"beta={beta:.8f}, lr={effective_lr:.6f}")

    return best_f, best_M


def _subgradient_polish(f0, deadline, rng, max_iter=500000, verbose=False):
    """Subgradient method on the exact (non-smooth) max-conv objective."""
    f = np.maximum(np.asarray(f0, dtype=np.float64), 1e-15)
    f = f / f.sum()
    n = f.size
    best_f = f.copy()
    best_M = float(_fft_convolve(f).max())
    alpha = 0.05 / max(1.0, np.sqrt(float(best_M)))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(f)
        M_val = float(c.max())
        if M_val < best_M - 1e-15:
            best_M = M_val
            best_f = f.copy()

        max_val = c.max()
        active = np.where(c >= max_val - 1e-12)[0]
        k_star = active[0]

        if 0 <= k_star < n:
            idx = k_star
        else:
            idx = k_star - n
        if not (0 <= idx < n):
            break

        subgrad = np.zeros(n)
        for k in active:
            j = k
            if 0 <= j < n:
                subgrad[j] += 1.0
            j2 = k - n
            if 0 <= j2 < n:
                subgrad[j2] += 1.0

        subgrad = 2.0 * np.abs(subgrad / len(active))
        subgrad_norm = np.linalg.norm(subgrad)
        if subgrad_norm < 1e-30:
            break

        subgrad = subgrad / subgrad_norm

        f_new = f - alpha * subgrad
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-30:
            break
        f = f_new / fs

        alpha *= 0.9999

        if verbose and it % 50000 == 0:
            sc = 2.0 * n * M_val
            print(f"    SGD iter {it}: score={sc:.10f}, best={2*n*best_M:.10f}, "
                  f"alpha={alpha:.8f}")

    return best_f, best_M


def _lp_optimize(f0, deadline, rng, max_iter=5000, verbose=False):
    """LP-based direction finding to reduce max-conv."""
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    f = f / f.sum()
    n = f.size
    M = float(_fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        for j in range(n):
            A[j:j + n, j] = f

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception:
            no_imp += 1
            if no_imp > 30:
                break
            continue

        if not res.success:
            no_imp += 1
            if no_imp > 50:
                break
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = float(_fft_convolve(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue

        g = res.x
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 200:
                break
            continue

        g = g / g_sum

        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.1, 0.4, 0.6, 0.8, 0.15):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(_fft_convolve(f_new).max())
            if M_new < M - 1e-15:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 100 == 0:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(_fft_convolve(f).max())
                no_imp = 0

        if verbose and it % 20 == 0:
            sc = 2.0 * n * M
            print(f"    LP iter {it}: score={sc:.10f}, best={2*n*best_M:.10f}")

    return best_f, best_M


def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * np.cos(np.pi * t) + a2 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Load parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = _compute_score(parent)
        try_update(parent, sc)
        print(f"Parent: n={parent.size}, score={sc:.10f}")
        n_parent = parent.size
    else:
        print("No parent available, using random start")
        n_parent = 512
        parent = _make_linear_profile(512, 1.5)
        try_update(parent, _compute_score(parent))

    # === Phase 1: LP optimization on parent (n=512) ===
    n1 = 512
    p1 = _interpolate_profile(parent, n1)
    if p1 is None:
        p1 = _make_linear_profile(n1, 1.5)

    phase1_end = min(deadline, t0 + 350.0)
    print(f"\nPhase 1: LP on n={n1}, time budget {phase1_end - t0:.0f}s")
    p1, M1 = _lp_optimize(p1, phase1_end, rng, max_iter=3000, verbose=True)
    sc1 = 2.0 * n1 * M1
    try_update(p1, sc1)
    print(f"Phase 1 done: score={sc1:.10f}, best={best_score:.10f}")

    # === Phase 2: Adam gradient on n=512 ===
    phase2_end = min(deadline, t0 + 900.0)
    print(f"\nPhase 2: Adam gradient on n={n1}, time budget {phase2_end - t0:.0f}s")
    p2, M2 = _adam_softgrad(p1, phase2_end, rng, lr=0.06,
                            beta_init_frac=0.12, beta_decay=0.87,
                            beta_decay_every=120, restart_patience=300,
                            verbose=True)
    sc2 = 2.0 * n1 * M2
    try_update(p2, sc2)
    print(f"Phase 2 done: score={sc2:.10f}, best={best_score:.10f}")

    # === Phase 3: Subgradient polish on n=512 ===
    phase3_end = min(deadline, t0 + 1150.0)
    if best_seq is not None and best_seq.size == n1:
        print(f"\nPhase 3: Subgradient polish on n={n1}, time budget {phase3_end - t0:.0f}s")
        p3, M3 = _subgradient_polish(best_seq, phase3_end, rng, max_iter=1000000,
                                     verbose=True)
        sc3 = 2.0 * n1 * M3
        try_update(p3, sc3)
        print(f"Phase 3 done: score={sc3:.10f}, best={best_score:.10f}")

    # === Phase 4: Try n=1024 ===
    n2 = 1024
    phase4_end = min(deadline, t0 + 1550.0)
    if time.monotonic() < phase4_end - 60:
        p4 = _interpolate_profile(best_seq, n2)
        if p4 is not None:
            print(f"\nPhase 4: Adam gradient on n={n2}, time budget {phase4_end - t0:.0f}s")
            p4, M4 = _adam_softgrad(p4, phase4_end, rng, lr=0.05,
                                    beta_init_frac=0.12, beta_decay=0.88,
                                    beta_decay_every=100, restart_patience=250,
                                    verbose=True)
            sc4 = 2.0 * n2 * M4
            try_update(p4, sc4)
            print(f"Phase 4 done: score={sc4:.10f}, best={best_score:.10f}")

    # === Phase 5: Try n=768 ===
    n3 = 768
    phase5_end = min(deadline, t0 + 1700.0)
    if time.monotonic() < phase5_end - 30:
        p5 = _interpolate_profile(best_seq, n3)
        if p5 is not None:
            print(f"\nPhase 5: Adam gradient on n={n3}, time budget {phase5_end - t0:.0f}s")
            p5, M5 = _adam_softgrad(p5, phase5_end, rng, lr=0.05,
                                    beta_init_frac=0.12, beta_decay=0.88,
                                    beta_decay_every=100, restart_patience=250)
            sc5 = 2.0 * n3 * M5
            try_update(p5, sc5)
            print(f"Phase 5 done: score={sc5:.10f}, best={best_score:.10f}")

    # === Phase 6: Final polish on the absolute best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        print(f"\nPhase 6: Final polish on n={n_best}")
        p6, M6 = _adam_softgrad(best_seq, deadline - 5, rng, lr=0.02,
                                beta_init_frac=0.05, beta_decay=0.92,
                                beta_decay_every=50, restart_patience=200)
        sc6 = 2.0 * n_best * M6
        try_update(p6, sc6)

        if time.monotonic() < deadline - 5:
            p6b, M6b = _subgradient_polish(p6, deadline - 2, rng, max_iter=2000000)
            sc6b = 2.0 * n_best * M6b
            try_update(p6b, sc6b)
            print(f"Phase 6 done: score={sc6b:.10f}, best={best_score:.10f}")

    # Final
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = _compute_score(final)
        print(f"\nFinal: n={final.size}, internal score={sc_final:.10f}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Evaluated score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Eval error: {e}")

        return [float(x) for x in final]

    fb = _make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
