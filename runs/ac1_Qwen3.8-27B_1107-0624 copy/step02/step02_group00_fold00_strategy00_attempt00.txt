**Approach and Rationale**

The previous program used an iterative approach with LP-based spreading and gradient polishing. However, the LP step was expensive for longer sequences. To address this, we use a **Gradient Descent with Simplicial Projection** approach. 

This approach ensures that:
1. The sequence remains in the simplex (positive values sum to 1).
2. We optimize the sequence by iteratively decreasing the maximum convolution term.
3. The optimization leverages accurate gradients to avoid wasting iterations.

**Mathematical Specification**

1. **Representation**: The sequence is represented as a vector in ℝ²ⁿ with each coefficient positive/zero and summing to 1. The simplex projection handles normalization.
2. **Objective**: Minimize the maximum value of the convolution (full mode).
3. **Constraint**: Non-negative coefficients and sum of 1.
4. **Gradient Calculation**: Uses the frequency domain (FFT) to compute accurate gradients.

import time
from scipy.optimize import minimize

def _next_pow2(n):
    """Get the next power of two for FFT convolution."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _project_simplex(v):
    """Project a vector to the simplex (all non-negative, sum of 1)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho_idx = np.nonzero(u * np.arange(1, n + 1) > (cs - 1.0))[0]
    if rho_idx.size == 0:
        x0 = np.ones(n)
        x0 /= x0.sum()
        return x0
    rho_idx = rho_idx[-1]  # -1 means the last index for which condition holds
    theta = (cs[rho_idx] - 1.0) / (rho_idx + 1)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum < 1e-30:
        x0 = np.ones(n)
        x0 /= x0.sum()
        return x0
    return w / w_sum

def _smooth_max_weight(c, beta):
    """Enhanced smooth max bound detection function."""
    c_val = np.asarray(c, dtype=np.float64)
    m = np.max(c_val)
    if np.min(c_val) < 0.0:
        # Found a negative value
        m_val = np.min(c_val)
        if m_val < -1e-10:
            # Coulomb a correction
            return -np.log(1.0 + np.exp(beta * (m - m_val)))
    # Smooth max form: (c_i - max(c_i))/beta, then exponential then normalize
    c_scaled = (c_val - m) / beta
    ex = np.exp(c_scaled)
    s = ex.sum()
    if not np.isfinite(s) or s < 1e-15:
        c_scaled = np.zeros_like(c_scaled)
        c_scaled[np.argmax(c_val)] = 1.0
        s = 1.0
    ex_norm = ex / s
    return beta * np.log(s) + m

    c_val = c.copy()
    m = np.max(c_val)
    if m == 0:
        return np.ones_like(c_val) * 2, m
    c_scaled = (c_val - m) / beta
    ex = np.exp(c_scaled)
    s_val = ex.sum()
    if s_val == 0:
        ex = np.zeros_like(c_scaled)
        ex[np.argmax(c_scaled)] = 1.0
        de = (c_scaled / beta * ex) / s_val  # Avoid division by zero
    else:
        ex_norm = ex / s_val
        de = (c_scaled / beta * ex) / s_val
        de[c_val != m] -= 1.0
        # Weighted combination
        grad = 2.0 * np.sum(ex, axis=1) - np.sum(c_scaled)
        return grad, m

def _score(a):
    """Compute the lower-bound version score."""
    a = np.asarray(a, dtype=np.float64)
    if not np.size(a):
        return np.inf
    s_sum = a.sum()
    if s_sum < 1e-10:  # NaN protection and zero-sum
        return np.inf
    conv = _project_simplex(a)  # Ensure positive and sum to 1
    if s_sum > 0:
        n = len(a)
        conv = np.convolve(a, a, mode="full")
        M_val = np.max(conv)
        if np.min(conv) <= -1e-10:  # If any negative
            return np.inf
        if s_sum < 1e-10:  # After clipping negatives to zero
            return np.inf
        return 2.0 * n * M_val / (s_sum ** 2)
    return np.inf

def _gradient_descent_step(x, gamma=0.1, max_move=1.0, min_move=0.01):
    """One step in gradient descent direction."""
    # Clip the sequence to non-negative
    x_clip = np.clip(x, min_move, gamma * max_move)
    t = gamma * 0.6  # Typical learning rate
    x_new = x + t * (x_clip - gamma * x)  # Adjust with simplex principle
    eps = 1e-6
    x_new = x_new / (np.sum(x_new) + eps)
    return x_new, gamma * (1 + t)

        """Compute gradient for given simplex vector."""

def evaluate_sequence(provided_sequence: list[float]) -> float:
    """Authoritative evaluation of the sequence with strict defense."""
    sequence = [float(x) for x in provided_sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    
    # Fastpath if too small
    s_sum = sum(sequence)
    if s_sum < 0.01:
        return np.inf
    
    # Compute convolution via FFT
    conv = np.convolve(sequence, sequence, 'full')
    M_val = max(conv)
    product = float(2.0 * n * M_val / (s_sum * s_sum))
    
    # If exploding, give feedback (but score calculation is plug-and-play)
    if np.isinf(product):
        return np.inf
        
    return product  # Value as float
