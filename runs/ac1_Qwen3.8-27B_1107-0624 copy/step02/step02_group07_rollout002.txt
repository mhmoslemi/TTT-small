    max_c = conv.max()
    if max_c < 1e-30:

    p = p / p.sum()

def lp_direction(s, M, time_limit=2.0):
    """Solve LP to find direction that maximizes sum while keeping conv peak <= M."""
    n = len(s)
    # conv(s, b)[k] <= M for all k, b >= 0, maximize sum(b)
    # conv(s,b)[k] = sum_{i: 0<=i<n, 0<=k-i<n} s[i]*b[k-i]
    # For k = 0..2n-2
    
    # Build constraint matrix A_ub (2n-1 constraints, n variables)
    # A_ub[k, j] = coefficient of b[j] in conv(s,b)[k]
    # conv(s,b)[k] = sum_i s[i]*b[k-i] = sum_j b[j] * s[k-j] (substituting j = k-i)
    # So A_ub[k, j] = s[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # b[j] appears in conv[s,b][k] for k = j to j+n-1
        # with coefficient s[k-j]
        A[j:j+n, j] = s
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            s_new = s.copy()
        else:

            best_score = score_new
            best_s = s_new.copy()
            no_improve = 0
            s = s_new
            # Maybe increase eta slightly
            eta = min(eta * 1.01, 0.5)
        elif score_new < fast_score(s):
            s = s_new
            no_improve += 1
        else:
            no_improve += 1
            # Decrease eta
            eta *= 0.8

            M = conv.max()
            b = lp_direction(s, M, time_limit=1.5)
            if b is not None and b.sum() > 1e-10:
                b = b / b.sum()
                # Try blending

                for t in [0.1, 0.2, 0.3, 0.5]:
                    cand = (1-t)*s + t*b
                    cand = np.maximum(cand, 0.0)
