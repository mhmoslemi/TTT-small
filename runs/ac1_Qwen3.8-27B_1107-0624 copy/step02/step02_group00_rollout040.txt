import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so just 2n*max_conv."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=20000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _peak_peel(a, deadline, max_iter=100000):
    """
    Greedy peak-peeling: directly reduce the max convolution value
    by adjusting the elements that contribute most to the peak.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())

    step = 0.01

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])

        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Find the pair (i, k_star - i) with the largest product
        # that contributes to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i]
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        if hi < lo:
            break

        best_i = -1
        best_prod = -1
        for i in range(lo, hi + 1):
            j = k_star - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i = i

        if best_i < 0:
            break

        i = best_i
        j = k_star - i

        # Try reducing a[i] and a[j]
        # Reduce the one with larger value (more effective)
        if a[i] >= a[j]:
            a[i] *= (1.0 - step)
        else:
            a[j] *= (1.0 - step)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        # Adaptively adjust step size
        if it % 500 == 0:
            c_check = _fft_convolve(a)
            if float(c_check.max()) > best_M * 1.05:
                step = max(0.001, step * 0.5)
            elif float(c_check.max()) < best_M * 0.95:
                step = min(0.05, step * 1.2)

    return best_a


def _coordinate_descent(a, deadline, max_passes=1000):
    """
    Coordinate descent: for each coordinate, try to find the best value.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())

    rng = np.random.default_rng(12345)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Shuffle coordinate order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break

            c = _fft_convolve(a)
            M = float(c.max())

            # Try several scaled values of a[j]
            factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]
            best_x = a[j]
            best_M_local = M

            for fac in factors:
                x = a[j] * fac
                if x < 0:
                    x = 0.0
                # Compute new conv efficiently
                delta = x - a[j]
                if abs(delta) < 1e-15:
                    continue
                c_new = c + 2.0 * delta * np.concatenate([
                    np.zeros(j), a, np.zeros(j)
                ][:2 * n - 1])
                if 2 * j < 2 * n - 1:
                    c_new[2 * j] += delta * delta
                c_new = np.maximum(c_new, 0)
                M_new = float(c_new.max())
                if M_new < best_M_local:
                    best_M_local = M_new
                    best_x = x

            if best_x != a[j]:
                a[j] = best_x
                s = a.sum()
                if s > 1e-10:
                    a = a / s

        # Check improvement
        c = _fft_convolve(a)
        M = float(c.max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _adam_logit_polish(a, deadline, max_iter=30000, lr=0.05):
    """Adam optimization in logit space with smooth-max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.6)

        if no_imp > max(3000, 5 * n):
            z = 0.7 * best_z + 0.3 * np.random.default_rng(it * 31 + 7).normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _nesterov_polish(a, deadline, max_iter=20000):
    """Nesterov-accelerated gradient polishing with smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-10, 0.05 * float(c.max()))
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = _project_simplex(a_nest)
        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.97)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.7)

    return best_a


def _make_two_hump_profile(n, gap=0.3):
    """Create a two-hump profile: high at both ends, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - gap) / 0.15) ** 2) + np.exp(-((t + 1 - gap) / 0.15) ** 2)
    a = a + 0.3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, curvature=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + curvature * (t - 0.5) ** 2 * 4
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _combined_optimize(a, deadline, phases=None):
    """
    Run a combined optimization: LP -> peak peeling -> coordinate descent -> Adam.
    Each phase uses a fraction of the remaining time.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    elapsed = time.monotonic()
    total_budget = deadline - elapsed

    # Phase A: LP (40% of budget)
    lp_end = elapsed + 0.4 * total_budget
    if lp_end > deadline:
        lp_end = deadline
    a_lp = _lp_refine(a, lp_end, max_iter=10000, inflate=1.001, blend=0.3)
    sc_lp = float(2.0 * n * _fft_convolve(a_lp).max())
    if sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
    a = a_lp
    print(f"    LP: score={sc_lp:.10f}")

    # Phase B: Peak peeling (20%)
    elapsed = time.monotonic()
    peel_end = elapsed + 0.2 * (deadline - elapsed)
    a_peel = _peak_peel(a, peel_end, max_iter=50000)
    sc_peel = float(2.0 * n * _fft_convolve(a_peel).max())
    if sc_peel < best_score:
        best_score = sc_peel
        best_a = a_peel.copy()
    a = a_peel
    print(f"    PeakPeel: score={sc_peel:.10f}")

    # Phase C: Coordinate descent (20%)
    elapsed = time.monotonic()
    cd_end = elapsed + 0.2 * (deadline - elapsed)
    a_cd = _coordinate_descent(a, cd_end, max_passes=500)
    sc_cd = float(2.0 * n * _fft_convolve(a_cd).max())
    if sc_cd < best_score:
        best_score = sc_cd
        best_a = a_cd.copy()
    a = a_cd
    print(f"    CoordDescent: score={sc_cd:.10f}")

    # Phase D: Adam polish (20%)
    elapsed = time.monotonic()
    ad_end = deadline
    a_ad = _adam_logit_polish(a, ad_end, max_iter=20000, lr=0.04)
    sc_ad = float(2.0 * n * _fft_convolve(a_ad).max())
    if sc_ad < best_score:
        best_score = sc_ad
        best_a = a_ad.copy()
    print(f"    Adam: score={sc_ad:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP + peak peeling + coordinate descent + Adam.
    Tries multiple sequence lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Target lengths to try
    lengths = [512]
    if parent is not None and parent.size != 512:
        lengths.insert(0, parent.size)
    lengths.extend([256, 1024])
    # Remove duplicates while preserving order
    seen = set()
    lengths_unique = []
    for ln in lengths:
        if ln not in seen:
            seen.add(ln)
            lengths_unique.append(ln)
    lengths = lengths_unique

    print(f"Trying lengths: {lengths}")

    # Generate initializations for each length
    all_candidates = []
    for n_t in lengths:
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
            # Perturbed parent
            for _ in range(2):
                p_pert = _interpolate_profile(parent, n_t)
                p_pert = p_pert * (1.0 + 0.05 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                p_pert = p_pert / p_pert.sum()
                inits.append(p_pert)
        inits.append(_make_linear_profile(n_t, 1.5))
        inits.append(_make_linear_profile(n_t, 2.0))
        inits.append(_make_two_hump_profile(n_t, 0.3))
        inits.append(_make_two_hump_profile(n_t, 0.2))
        inits.append(_make_cubic_profile(n_t, 1.5, -1.0))
        inits.append(_make_quadratic_profile(n_t, 1.0))

        for a0 in inits:
            sc = _score(a0)
            if np.isfinite(sc):
                all_candidates.append((sc, n_t, a0))

    # Sort by initial score
    all_candidates.sort(key=lambda x: x[0])
    print(f"Generated {len(all_candidates)} candidates, best initial score: {all_candidates[0][0]:.10f}")

    # Allocate time among candidates (more time to better initial scores)
    # Top 3 candidates get full treatment, rest get quick LP only
    num_full = min(3, len(all_candidates))
    time_per_full = (budget_s - 60) / max(num_full, 1)

    for idx in range(min(num_full, len(all_candidates))):
        if time.monotonic() > deadline - 30:
            break

        sc0, n_t, a0 = all_candidates[idx]
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Run combined optimization
        phase_end = min(t_start + (idx + 1) * time_per_full, deadline - 20)
        print(f"\nCandidate {idx+1}/{num_full}: n={n_t}, initial_score={sc0:.10f}, budget={phase_end - time.monotonic():.0f}s")

        a_opt = _combined_optimize(a0, phase_end)
        sc_opt = _score(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
        print(f"  Final: score={sc_opt:.10f}")

    # If we have time, try more candidates with quick LP
    if time.monotonic() < deadline - 30:
        for idx in range(num_full, min(num_full + 5, len(all_candidates))):
            if time.monotonic() > deadline - 30:
                break
            sc0, n_t, a0 = all_candidates[idx]
            if sc0 >= best_score:
                continue
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick LP only
            quick_end = min(time.monotonic() + 60, deadline - 30)
            a_quick = _lp_refine(a0, quick_end, max_iter=3000, inflate=1.001, blend=0.3)
            sc_quick = _score(a_quick)
            if np.isfinite(sc_quick) and sc_quick < best_score:
                best_score = sc_quick
                best_seq = a_quick.copy()
            print(f"  Quick LP candidate {idx}: score={sc_quick:.10f}")

    # Final polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_budget = deadline - time.monotonic() - 5
        if final_budget > 20:
            print(f"\nFinal polish: budget={final_budget:.0f}s")

            # Try Nesterov
            a_ne = _nesterov_polish(best_seq, t_start + time.monotonic() - t_start + final_budget * 0.5, max_iter=10000)
            sc_ne = _score(a_ne)
            if np.isfinite(sc_ne) and sc_ne < best_score:
                best_score = sc_ne
                best_seq = a_ne.copy()
            print(f"  Nesterov: {sc_ne:.10f}")

            # Try one more LP
            remaining = deadline - time.monotonic() - 5
            if remaining > 10:
                a_lpf = _lp_refine(best_seq, t_start + time.monotonic() - t_start + remaining * 0.6,
                                   max_iter=2000, inflate=1.0005, blend=0.2)
                sc_lpf = _score(a_lpf)
                if np.isfinite(sc_lpf) and sc_lpf < best_score:
                    best_score = sc_lpf
                    best_seq = a_lpf.copy()
                print(f"  Final LP: {sc_lpf:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with the authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
