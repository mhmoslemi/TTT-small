def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):

def convolve_full(a, b):
    """Full linear convolution."""
    n_a = len(a)
    n_b = len(b)
    if n_a + n_b <= 512:
        return np.convolve(a, b)
    N = 1
    while N < n_a + n_b - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:n_a + n_b - 1]
    return np.maximum(result, 0)  # Remove tiny negatives

def build_conv_matrix(f):

    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

import numpy as np

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Normalize to sum=1."""
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def convolve_fft(a, b):
    """Full linear convolution using FFT for efficiency."""
    n_a = len(a)
    n_b = len(b)
    if n_a + n_b - 1 <= 256:
        return np.convolve(a, b)
    N = 1
    while N < n_a + n_b - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:n_a + n_b - 1]
    mn = result.min()
    if mn < 0:
        result = np.maximum(result, 0.0)
    return result


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:

        pass

    return float(np.max(val))

def project_simplex(v):

    rho = np.nonzero(cond)[0][-1]

                grad, _ = gradient_at_peak(f)
                gn = float(np.linalg.norm(grad))
                if gn > 1e-15:
                    f = project_simplex(f - 0.02 * grad / gn)
                stagnant += 1
                if stagnant > 200:
                    noise = 0.05 * rng.standard_normal(n)
                    f = project_simplex(best_f + noise)
                    stagnant = 0
                continue

        g = normalize(g)

        # Check if g is essentially the same as f
        diff = float(np.linalg.norm(g - f))
        if diff < 1e-8:
            stagnant += 1
            if stagnant > 50:
                noise = 0.03 * rng.standard_normal(n)
                f = project_simplex(best_f + noise)
                stagnant = 0
            continue

        # Line search
        t_opt = line_search(f, g)
        if t_opt < 1e-6:
            stagnant += 1
            if stagnant > 50:
                noise = 0.03 * rng.standard_normal(n)
                f = project_simplex(best_f + noise)
                stagnant = 0
            continue

        f_new = normalize((1.0 - t_opt) * f + t_opt * g)
        M_new = float(convolve_fft(f_new, f_new).max())

        if M_new <= best_M + 1e-11:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 100 == 0:
                score = 2.0 * n * best_M
                print(f"  LP iter {iteration}: M={best_M:.12f}, score={score:.10f}")
        else:
            stagnant += 1

        if stagnant > 150:
            noise = 0.05 * rng.standard_normal(n)
            f = project_simplex(best_f + noise)
            stagnant = 0
            if verbose:
                print(f"  LP restart at iter {iteration}")

    return best_f


def random_perturb_polish(f_init, deadline, rng, n_trials=20000):
    """Fast random coordinate perturbation polish."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    f = normalize(f)
    n = f.size
    conv_f = convolve_fft(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()

    # Precompute for efficiency
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Pick random coordinate
        i = rng.integers(n)
        # Try a few multiplicative factors
        best_local_M = best_M
        best_local_val = f[i]

        for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            new_val = f[i] * factor
            if new_val < -1e-15:
                continue
            f[i] = new_val
            f = normalize(f)
            conv_tmp = convolve_fft(f, f)
            m = float(conv_tmp.max())
            if m < best_local_M:
                best_local_M = m
                best_local_val = new_val
            f[i] = f[i] * (f_init[i] / max(f[i], 1e-15)) if f[i] > 1e-15 else f[i]
            f = normalize(f)

        # Restore and apply best
        f[i] = best_local_val
        f = normalize(f)
        conv_f = convolve_fft(f, f)
        M = float(conv_f.max())
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()

    return best_f


def make_profile(n, profile_type, rng=None):
    """Create various starting profiles."""
    t = np.linspace(0, 1, n)
    if profile_type == 'linear':
        a = 1.0 + 2.0 * (t - 0.5)
    elif profile_type == 'linear2':
        a = 1.0 + 3.0 * (t - 0.5)
    elif profile_type == 'cosine':
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
    elif profile_type == 'cosine2':
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
    elif profile_type == 'sine_sq':
        a = 0.5 + 0.5 * np.sin(np.pi * t) ** 2
    elif profile_type == 'quadratic':
        a = 1.0 + (2.0 * t - 1.0) ** 2
    elif profile_type == 'bimodal':
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2)
    elif profile_type == 'trimodal':
        a = np.exp(-15.0 * (t - 0.2) ** 2) + np.exp(-15.0 * (t - 0.5) ** 2) + np.exp(-15.0 * (t - 0.8) ** 2)
    elif profile_type == 'plateau':
        a = np.where((t > 0.2) & (t < 0.8), 1.5, 0.5)
    elif profile_type == 'random':
        if rng is None:
            rng = np.random.default_rng()
        k1 = rng.uniform(1, 4)
        k2 = rng.uniform(3, 8)
        a1 = rng.uniform(0.1, 0.5)
        a2 = rng.uniform(0.05, 0.3)
        a = 1.0 + a1 * np.cos(2 * np.pi * k1 * t) + a2 * np.cos(2 * np.pi * k2 * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p, n_new):
    """Linearly interpolate a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return normalize(p)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing 2n*max(conv(a,a))/sum(a)^2.

    Phases:
    1. LP + line search at initial length (main optimizer).
    2. Nesterov projected gradient (alternative local optimizer).
    3. Scale up to larger length and re-optimize.
    4. Multi-start exploration.
    5. Final polish with random perturbations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        raw = np.maximum(raw, 0.0)
        if raw.size > 0 and raw.sum() > 0.01:
            init_seq = normalize(raw)
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s, seed={seed}")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: LP + Line Search =====
    n1 = init_seq.size if init_seq is not None else 512
    if init_seq is not None:
        p1 = init_seq.copy()
    else:
        p1 = make_profile(n1, 'linear', rng)

    # Quick initial LP run to get a good baseline
    phase1a_deadline = min(deadline, t_start + 400.0)
    print(f"\nPhase 1a: LP+LS at n={n1} (main)")
    result = lp_optimize(p1, phase1a_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"Phase 1a done: score={sc:.10f}")

    # ===== Phase 2: Nesterov PG =====
    phase2_deadline = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_deadline - 30 and best_seq is not None:
        print(f"\nPhase 2: Nesterov PG at n={best_seq.size}")
        result = nesterov_pg(best_seq, phase2_deadline, rng, lr_init=0.03)
        sc = compute_score(result)
        update(result, sc)
        print(f"Phase 2 done: score={sc:.10f}")

    # ===== Phase 3: More LP on best found =====
    phase3_deadline = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_deadline - 30 and best_seq is not None:
        print(f"\nPhase 3: LP polish at n={best_seq.size}")
        # Add tiny noise to potentially escape
        p3 = best_seq + 0.001 * rng.standard_normal(best_seq.size)
        p3 = project_simplex(p3)
        result = lp_optimize(p3, phase3_deadline, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"Phase 3 done: score={sc:.10f}")

    # ===== Phase 4: Scale up to 2x length =====
    n2 = n1 * 2
    if n2 <= 2048 and time.monotonic() < deadline - 250 and best_seq is not None:
        print(f"\nPhase 4: Scale up to n={n2}")
        p2 = interpolate_profile(best_seq, n2)
        sc2 = compute_score(p2)
        update(p2, sc2)
        print(f"  Interpolated score: {sc2:.10f}")

        phase4_deadline = min(deadline, time.monotonic() + 350.0)
        result = lp_optimize(p2, phase4_deadline, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"Phase 4 done: score={sc:.10f}")

        if best_seq.size == n1:
            # The original length was still better
            print(f"  Keeping n={n1} (score={best_score:.10f})")

    # ===== Phase 5: Multi-start at best length =====
    if time.monotonic() < deadline - 150 and best_seq is not None:
        n_ms = best_seq.size
        print(f"\nPhase 5: Multi-start at n={n_ms}")

        profiles = [
            ('bimodal', make_profile(n_ms, 'bimodal', rng)),
            ('trimodal', make_profile(n_ms, 'trimodal', rng)),
            ('plateau', make_profile(n_ms, 'plateau', rng)),
            ('rand1', make_profile(n_ms, 'random', rng)),
            ('rand2', make_profile(n_ms, 'random', rng)),
        ]

        for name, p in profiles:
            if time.monotonic() > deadline - 80:
                break
            per_deadline = min(deadline - 80, time.monotonic() + 50.0)
            result = lp_optimize(p, per_deadline, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            print(f"  {name}: score={sc:.10f}")

    # ===== Phase 6: Random perturbation polish =====
    if time.monotonic() < deadline - 20 and best_seq is not None:
        print(f"\nPhase 6: Random perturb polish")
        per_deadline = min(deadline - 10, time.monotonic() + 80.0)
        result = random_perturb_polish(best_seq, per_deadline, rng, n_trials=50000)
        sc = compute_score(result)
        update(result, sc)
        print(f"Phase 6 done: score={sc:.10f}")

    # ===== Phase 7: Final LP refinement =====
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = min(deadline - 3, time.monotonic() + 25.0)
        result = lp_optimize(best_seq, per, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        print(f"Phase 7 (final LP): score={sc:.10f}")

    # ===== Final output =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profile(512, 'linear')

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profile(256, 'linear')
        final_seq = final_seq / final_seq.sum()

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
