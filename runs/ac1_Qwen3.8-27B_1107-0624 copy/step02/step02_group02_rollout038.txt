"""
LP + Soft-max Adam optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based descent for escaping local minima with fast gradient-based
polishing for fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except:
        pass
    return None

def soft_max(c, beta):
    m = float(c.max())
    x = (c - m) / max(beta, 1e-14)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    return m + beta * float(np.log(s)), w

def corr_weights_ff(fw, p, n):
    """Compute h[i] = sum_j w[i+j] * p[j] using precomputed FFT of w."""
    N = next_pow2(2 * n - 1)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    return p / s if s > 0 else np.ones_like(e) / e.size

def profile_linear(n, eta=1.5):

    a = np.maximum(1.0 + eta * (t - 0.5), 1e-12)
    return a / a.sum()

    a = np.maximum(1.0 + amp * np.cos(2 * np.pi * k * t), 1e-12)

    a = np.maximum(1.0 + amp * np.cos(2 * np.pi * k1 * t) + 0.5*amp * np.cos(2 * np.pi * k2 * t), 1e-12)
    return a / a.sum()

def profile_flat_top(n, frac=0.6):
    """Flat top with cosine taper at edges."""
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = frac / 2
    for i in range(n):
        if t[i] < taper:

        elif t[i] > 1 - taper:
            a[i] = 0.5 * (1 + np.cos(np.pi * (1 - t[i]) / taper))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 0 else p
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, p)

def softmax_adam(p0, deadline, rng, lr=0.06, beta_init=0.15, beta_decay=0.95,

    p = np.maximum(p0, 1e-12)
    p = p / p.sum()
    n = p.size
    z = np.log(p)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = fft_convolve(p)
    maxc = float(c.max())
    beta = beta_init * maxc
    
    best_p = p.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    restarts = 0
    
    base_lr = lr
    
    while time.monotonic() < deadline:
        L, w = soft_max(c, beta)
        fw = np.fft.rfft(w, next_pow2(2*n-1))
        gp = 2.0 * corr_weights_ff(fw, p, n)
        vp = gp  # No regularization needed in logit space
        dot = float(p @ vp)
        gz = p * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_p  # restart from best
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz**2
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        
        effective_lr = base_lr * (0.5 ** (restarts * 0.1))
        z = z - effective_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        p = softmax_from_z(z)
        
        c = fft_convolve(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        if t_adam % 100 == 0:
            beta = max(1e-14, beta * beta_decay)

        if no_imp > restart_freq:
            restarts += 1
            if restarts > max_restarts:
                break
            # Perturb from best
            z = np.log(best_p + 1e-12)
            z -= z.mean()
            z += 0.3 * rng.standard_normal(n)

            p = softmax_from_z(z)

            beta = 0.2 * maxc
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            no_imp = 0
            base_lr = max(0.01, base_lr * 0.9)

"""
LP + Soft-max Adam hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based descent for escaping local minima, fast FFT-based soft-max Adam for
fine-tuning, adaptive restarts, and multi-scale (n=512, n=1024) search.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Full linear convolution using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, top_k=None):
    """Solve LP: max sum(b) s.t. conv(f,b)[k] <= M, b >= 0.
    If top_k given, only use top-k constraints for speed."""
    n = len(f)
    A_full = build_conv_matrix(f)
    if top_k is not None and top_k < 2 * n - 1:
        c_ff = np.convolve(f, f)
        idx = np.argsort(c_ff)[::-1][:top_k]
        A = A_full[idx, :]
        b_ub = np.full(len(idx), M)
    else:
        A = A_full
        b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            # Verify feasibility against all constraints
            if top_k is not None:
                c_fg = np.convolve(f, g)
                if c_fg.max() > M * 1.005:
                    return None
            return g
    except Exception:
        pass
    return None


def soft_max_vals(c, beta):
    """Compute soft max value and weights."""
    m = float(c.max())
    x = (c - m) / max(beta, 1e-15)
    ex = np.exp(np.clip(x, -50, 50))
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    return m + beta * float(np.log(s)), w


def corr_weights_fft(w, p, n):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1 using FFT."""
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_stable(z):
    """Numerically stable softmax."""
    z = np.clip(z, -60.0, 60.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + 1e-14
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(e) / e.size
    return p / s


def profile_linear(n, eta=1.5):

def interpolate_profile(p, n_new):
    """Linear interpolation to new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:

        return p / s if s > 1e-10 else p
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, p)


def softmax_adam_optimize(p0, deadline, rng, lr=0.06, beta_scale=0.15,
                          beta_decay=0.97, restart_gap=1500, max_restarts=100):
    """Soft-max Adam in logit space. Returns (best_p, best_score)."""
    p = np.maximum(p0, 1e-12).astype(np.float64)

    v = np.zeros(n)

            z -= float(z.mean())
            m[:] = 0.0

    phase1_end = t_start + 400.0

        p0 = init_seq.copy()
    elif init_seq is not None:
        p0 = interpolate_profile(init_seq, n_main)
    else:
        p0 = profile_linear(n_main, 1.5)

    sc = compute_score(p0)
    if np.isfinite(sc) and sc < best_score:
        best_seq = p0.copy()
        best_score = sc
    print(f"Phase 1 start: n={n_main}, score={sc:.10f}")

    # Main LP optimization
    lp_dl = min(phase1_end, time.monotonic() + 300.0)
    if time.monotonic() < lp_dl:
        res = lp_optimize(p0, lp_dl, rng)
        sc = compute_score(res)
        if np.isfinite(sc) and sc < best_score:
            best_seq = res.copy()
            best_score = sc
        print(f"After main LP: score={sc:.10f}, best={best_score:.10f}")

    # Quick SM polish
    if time.monotonic() < phase1_end - 10 and best_seq is not None:
        sm_dl = min(phase1_end, time.monotonic() + 80.0)
        res, sc_sm = softmax_adam_optimize(best_seq, sm_dl, rng, lr=0.06)
        sc = compute_score(res)
        if np.isfinite(sc) and sc < best_score:
            best_seq = res.copy()
            best_score = sc
        print(f"After SM polish: score={sc:.10f}, best={best_score:.10f}")

    # Try diverse profiles
    profiles = get_profiles(n_main, rng)
    for name, p in profiles:
        if time.monotonic() > phase1_end - 5:
            break
        sc_p = compute_score(p)
        if sc_p > best_score * 1.08:
            continue
        dl = min(phase1_end, time.monotonic() + 25.0)
        if time.monotonic() >= dl:
            continue
        res = lp_optimize(p, dl, rng)
        sc = compute_score(res)
        if np.isfinite(sc) and sc < best_score:
            best_seq = res.copy()
            best_score = sc
            print(f"  Profile {name}: {sc:.10f} (NEW BEST)")

    print(f"Phase 1 complete: best={best_score:.10f}")

    # ============ Phase 2: Intensive optimization at n=512 (500s) ============
    phase2_end = t_start + 950.0

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_cur = best_seq.size
        for rnd in range(8):
            if time.monotonic() > phase2_end - 5:
                break
            # LP round with perturbation
            dl_lp = min(phase2_end, time.monotonic() + 60.0)
            if time.monotonic() < dl_lp:
                noise_scale = 0.015 + 0.005 * rnd
                p_pert = best_seq + noise_scale * rng.standard_normal(n_cur)
                p_pert = np.maximum(p_pert, 1e-6)
                p_pert = p_pert / p_pert.sum()
                res = lp_optimize(p_pert, dl_lp, rng)
                sc = compute_score(res)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = res.copy()
                    best_score = sc
                    print(f"  P2 LP round {rnd}: {sc:.10f} (NEW BEST)")

            # SM polish round
            dl_sm = min(phase2_end, time.monotonic() + 40.0)
            if time.monotonic() < dl_sm and best_seq is not None:
                p_sm = best_seq + 0.005 * rng.standard_normal(n_cur)
                p_sm = np.maximum(p_sm, 1e-6)
                p_sm = p_sm / p_sm.sum()
                res2, sc2 = softmax_adam_optimize(p_sm, dl_sm, rng, lr=0.05, beta_scale=0.12)
                sc = compute_score(res2)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = res2.copy()
                    best_score = sc
                    print(f"  P2 SM round {rnd}: {sc:.10f} (NEW BEST)")

    print(f"Phase 2 complete: best={best_score:.10f}")

    # ============ Phase 3: Try n=1024 (450s) ============
    n_large = 1024
    phase3_end = t_start + 1500.0

    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        if np.isfinite(sc):
            print(f"Phase 3: n={n_large}, interpolated score={sc:.10f}")
            if sc < best_score * 1.03:  # Only proceed if promising
                # LP at larger n
                dl = min(phase3_end, time.monotonic() + 250.0)
                if time.monotonic() < dl:
                    res = lp_optimize(p_large, dl, rng)
                    sc = compute_score(res)
                    if np.isfinite(sc) and sc < best_score:
                        best_seq = res.copy()
                        best_score = sc
                        print(f"  n=1024 LP: {sc:.10f} (NEW BEST)")
                # SM at larger n
                if time.monotonic() < phase3_end - 20 and best_seq is not None:
                    sm_dl = min(phase3_end - 5, time.monotonic() + 150.0)
                    res2, sc2 = softmax_adam_optimize(best_seq, sm_dl, rng, lr=0.04, beta_scale=0.12)
                    sc = compute_score(res2)
                    if np.isfinite(sc) and sc < best_score:
                        best_seq = res2.copy()
                        best_score = sc
                        print(f"  n=1024 SM: {sc:.10f} (NEW BEST)")
            else:
                print(f"  n=1024 interpolated score too high ({sc:.6f}), skipping.")
    else:
        print("Phase 3: skipped (insufficient time or no best_seq)")

    print(f"Phase 3 complete: best={best_score:.10f}")

    # ============ Phase 4: Final intensive polish (remaining time) ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        print(f"Phase 4: final polish at n={n_final}")

        # Multiple short SM rounds with decreasing lr
        for lr in [0.04, 0.025, 0.015]:
            if time.monotonic() > deadline - 5:
                break
            sm_dl = min(deadline - 3, time.monotonic() + 50.0)
            if time.monotonic() >= sm_dl:
                break
            p_sm = best_seq + 0.003 * rng.standard_normal(n_final)
            p_sm = np.maximum(p_sm, 1e-6)
            p_sm = p_sm / p_sm.sum()
            res, sc_r = softmax_adam_optimize(p_sm, sm_dl, rng, lr=lr, beta_scale=0.08,
                                              beta_decay=0.98, restart_gap=800, max_restarts=30)
            sc = compute_score(res)
            if np.isfinite(sc) and sc < best_score:
                best_seq = res.copy()
                best_score = sc
                print(f"  Final SM lr={lr}: {sc:.10f} (NEW BEST)")

        # One more LP round if time permits
        if time.monotonic() < deadline - 15 and best_seq is not None:
            dl = min(deadline - 3, time.monotonic() + 40.0)
            if time.monotonic() < dl:
                p_lp = best_seq + 0.01 * rng.standard_normal(n_final)
                p_lp = np.maximum(p_lp, 1e-6)
                p_lp = p_lp / p_lp.sum()
                res = lp_optimize(p_lp, dl, rng)
                sc = compute_score(res)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = res.copy()
                    best_score = sc
                    print(f"  Final LP: {sc:.10f} (NEW BEST)")

    # ============ Final validation and return ============
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = profile_linear(512, 1.5)
        final_score = compute_score(final_seq)
        print(f"=== FINAL: n={final_seq.size}, score={final_score:.10f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if np.isfinite(eval_score) and eval_score < final_score:
                final_score = eval_score
        except Exception as e:
            print(f"Could not call evaluate_sequence: {e}")
        return [float(x) for x in final_seq]
    else:
        fallback = profile_linear(512, 1.5)
        print(f"FALLBACK: using linear profile, n=512")
        return [float(x) for x in fallback]
