def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""

def lp_direction(f, M, timeout=10.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    Returns the optimal g (or None if infeasible).

    A = build_conv_matrix(f)

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)

    a = 1.0 + eta * u

def make_multi_profile(n, coeffs):
    """Create profile from polynomial coefficients: a(u) = sum coeffs[k] * u^k, u=2t-1."""
    t = np.linspace(0.0, 1.0, n)

    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)

    return a / a.sum()


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1."""
    n = len(p)
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.dot(w[i:i + n], p)
    return h

    v = np.zeros(n)
    it = 0

    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.2 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights for differentiating the max
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient of max(c) w.r.t. p: grad_p[i] = 2 * sum_x w[x] * p[x-i]
        h = corr_weights(w, p0)
        grad_p = 2.0 * h + reg * p0

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

            continue

        score = 2.0 * n * float(c.max())

            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
