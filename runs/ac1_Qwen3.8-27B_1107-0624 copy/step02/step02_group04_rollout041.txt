"""
FISTA-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth soft-max approximation with accelerated gradient (FISTA)
in logit space, adaptive step size, and multi-phase strategy.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = a.size + b.size - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:n]


def _fft_correlate(w, a):
    """Compute h_l = sum_k w_k * a_{k-l} for l=0..n-1."""
    n = a.size
    # h = convolve(w, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    c = _fft_convolve(w, a_rev)
    return c[n-1:2*n-1]


def _softmax_z(z, floor=1e-300):
    """Compute softmax of z (in logit space)."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < floor:
        return np.ones_like(z) / z.size
    return e / s


def _score_normalized(a):
    """Compute score assuming sum(a) = 1."""
    n = a.size
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max()


def _score_from_logits(z):
    a = _softmax_z(z)
    return _score_normalized(a)


def _gradient_logits(z, beta):
    """
    Compute gradient of 2n * softmax(c, beta) w.r.t. logits z.
    
    Returns (gradient, score_approx, c, w, a, h, c_bar)
    """
    n = z.size
    a = _softmax_z(z)
    c = _fft_convolve(a, a)
    
    # Softmax weights
    c_max = c.max()
    x = (c - c_max) / beta
    ex = np.exp(x)
    w = ex / ex.sum()
    
    score_approx = 2.0 * n * (c_max + beta * np.log(ex.sum()))
    
    # h_l = sum_k w_k * a_{k-l}
    h = _fft_correlate(w, a)
    
    # c_bar = sum_k w_k * c_k
    c_bar = float(np.dot(w, c))
    
    # grad_l = 4n * a_l * (h_l - c_bar)
    grad = 4.0 * n * a * (h - c_bar)
    # Center the gradient (sum to zero in logit space)
    grad -= grad.mean()
    
    return grad, score_approx, c, w, a, h, c_bar

    - beta_init: initial soft-max temperature
    - beta_min: minimum beta
    - beta_decay: multiplicative decay factor for beta
    - max_iter_per_beta: max iterations per beta value
    - lr_init: initial learning rate (if None, use backtracking)
    """
    n = z0.size
    z = z0.copy().astype(np.float64)
    
    # Initialize a and compute initial c for beta
    a = _softmax_z(z)
    c = _fft_convolve(a, a)
    max_c = c.max()
    
    beta = min(beta_init * max_c, 0.5)
    if beta < beta_min:
        beta = beta_min
    
    best_score = 2.0 * n * max_c
    best_z = z.copy()
    
    # FISTA variables
    z_prev = z.copy()
    t_fista = 1.0
    lr = lr_init if lr_init is not None else 0.1
    lr_min = 1e-10
    lr_max = 1.0
    
    if verbose:
        print(f"  FISTA: n={n}, beta_init={beta:.6f}, max_c={max_c:.8f}")
    
    while time.monotonic() < deadline and beta >= beta_min:
        z_cur = z.copy()
        t_fista = 1.0
        z_prev = z.copy()
        
        iter_count = 0
        no_improve = 0

            iter_count += 1
            
            # FISTA extrapolation
            z_y = z + ((t_fista - 1) / (t_fista + 2)) * (z - z_prev)
            
            # Compute gradient at z_y
            grad, score_approx, c, w, a, h, c_bar = _gradient_logits(z_y, beta)
            
            if not np.all(np.isfinite(grad)):
                z = z_cur.copy()
                break
            
            # Backtracking line search
            f_y = score_approx
            accepted = False
            for ls_iter in range(20):

                a_new = _softmax_z(z_new)
                c_new = _fft_convolve(a_new, a_new)
                f_new = 2.0 * n * (c_new.max() + beta * np.log(np.exp((c_new - c_new.max()) / beta).sum()))

                z = z_cur.copy()
                t_fista = 1.0

                no_improve += 1

                best_z = z.copy()
                no_improve = 0
            else:
                no_improve += 1

                break
        
        if verbose and iter_count > 0:
            print(f"  FISTA beta={beta:.6e}: {iter_count} iters, score={best_score:.8f}")
        
        # Decrease beta
        beta = max(beta_min, beta * beta_decay)
        # Reset FISTA momentum
        z = best_z.copy()
        z_prev = z.copy()
        t_fista = 1.0
    
    return best_z, best_score

def _lp_correction(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    Returns optimal b (normalized to sum=1).
    """
    n = a.size
    # (a*b)_k = sum_i a_i * b_{k-i}
    # This is a linear constraint in b for each k.
    
    # Use scipy or cvxopt
    from scipy.optimize import linprog
    
    # Maximize sum(b) = -minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # (a*b)_k = sum_i a_i * b_{k-i} (valid i)
    # For fixed k: sum_i a_i * b_{k-i} <= M
    # This is a linear constraint: sum_i A[k,i] * b_i <= M
    # where A[k,i] = a_i if 0 <= k-i <= n-1, else 0
    # Equivalently: A[k,i] = a_{k-i} if 0 <= k-i <= n-1, else 0
    # Wait, let me re-derive.
    # (a*b)_k = sum_{i+j=k} a_i b_j = sum_j a_{k-j} b_j
    # So for each k: sum_j a_{k-j} b_j <= M
    # The coefficient of b_j in constraint k is a_{k-j} (if 0 <= k-j <= n-1)
    
    # Build A matrix (sparse)
    # A[k, j] = a_{k-j} if 0 <= k-j <= n-1, else 0
    # For k = 0, ..., 2n-2 and j = 0, ..., n-1
    
    # Use sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Also b >= 0 (handled by bounds)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        b_opt = res.x
        s = b_opt.sum()
        if s > 0:
            return b_opt / s
    
    return None

We have $a = \text{softmax}(z)$, i.e., $a_i = e^{z_i} / Z$ where $Z = \sum_j e^{z_j}$.

The objective (with $\sum a = 1$) is $f(a) = 2n \cdot \max_k c_k$ where $c = a * a$.

With soft-max approximation: $f_\beta(z) = 2n \cdot \beta \log \sum_k e^{c_k / \beta}$ where $c_k = \sum_i a_i a_{k-i}$.

$\frac{\partial f_\beta}{\partial z_l} = 2n \sum_k \frac{\partial}{\partial z_l} \left[\beta \log \sum_m e^{c_m/\beta}\right]$
$= 2n \sum_k w_k \frac{\partial c_k}{\partial z_l}$

where $w_k = e^{c_k/\beta} / \sum_m e^{c_m/\beta}$.

Now, $\frac{\partial c_k}{\partial z_l} = \sum_m \frac{\partial c_k}{\partial a_m} \frac{\partial a_m}{\partial z_l}$

$\frac{\partial a_m}{\partial z_l} = a_m(\delta_{ml} - a_l)$

$\frac{\partial c_k}{\partial a_m} = 2 a_{k-m}$ (as derived earlier, with $a_j = 0$ for $j \notin [0,n-1]$)

So $\frac{\partial c_k}{\partial z_l} = \sum_m 2a_{k-m} \cdot a_m(\delta_{ml} - a_l)$
$= 2a_{k-l} \cdot a_l - 2a_l \sum_m a_{k-m} a_m$
$= 2a_{k-l} a_l - 2a_l c_k$

Therefore:
$\frac{\partial f_\beta}{\partial z_l} = 2n \sum_k w_k (2a_{k-l} a_l - 2a_l c_k)$
$= 4n a_l \left(\sum_k w_k a_{k-l} - \sum_k w_k c_k\right)$
$= 4n a_l (h_l - \bar{c})$

where $h_l = \sum_k w_k a_{k-l}$ and $\bar{c} = \sum_k w_k c_k$.

This confirms my earlier derivation. Good.

Now, the computation of $h_l = \sum_k w_k a_{k-l}$:
- $w$ has length $2n-1$ (indices 0 to 2n-2)
- $a$ has length $n$ (indices 0 to n-1)
- $h_l = \sum_{k=0}^{2n-2} w_k \cdot a_{k-l}$ where $a_j = 0$ for $j \notin [0, n-1]$

This is the cross-correlation of $w$ and $a$:
$h_l = \sum_k w_k \cdot a(k-l)$

In terms of convolution: if I define $a^R_j = a_{n-1-j}$, then:
$(w * a^R)[m] = \sum_k w_k a^R_{m-k} = \sum_k w_k a_{n-1-(m-k)} = \sum_k w_k a_{k-(m-n+1)}$

Set $m = l + n - 1$: $(w * a^R)[l+n-1] = \sum_k w_k a_{k-l} = h_l$ ✓

So $h = \text{convolve}(w, a^R)[n-1 : 2n-1]$.

"""
FISTA-based convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth soft-max approximation of the max, FISTA acceleration in logit space,
adaptive beta schedule, and LP-based corrections for final refinement.
"""
import numpy as np
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n_out = a.size + b.size - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _fft_cross_corr(w, a):
    """Compute h_l = sum_k w_k * a[k-l] for l=0..n-1.
    This is convolve(w, a_reversed)[n-1 : 2n-1]."""
    n = a.size
    a_rev = a[::-1]
    c = _fft_convolve(w, a_rev)
    return c[n-1:2*n-1]


def _softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / z.size
    return e / s


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-30:
        return np.inf
    a_norm = a / s
    c = _fft_convolve(a_norm, a_norm)
    return 2.0 * n * float(c.max())


def _eval_true(seq_list):
    """Call evaluate_sequence for authoritative score."""
    try:
        return float(evaluate_sequence(seq_list))
    except Exception:
        return np.inf


def _gradient_logits(z, beta):
    """Compute gradient of smooth objective w.r.t. logits z.
    Returns (grad, c, a)."""
    n = z.size
    a = _softmax_stable(z)
    c = _fft_convolve(a, a)
    c_max = c.max()
    x = (c - c_max) / beta
    ex = np.exp(x)
    w = ex / ex.sum()
    h = _fft_cross_corr(w, a)
    c_bar = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - c_bar)
    grad -= grad.mean()
    return grad, c, a


def _fista_run(z0, deadline, beta_start=None, beta_min=1e-9,
               beta_decay=0.65, max_iters_per_beta=20000,
               lr=0.05, verbose=True, label=""):
    """Run FISTA with soft-max approximation.
    Returns (best_z, best_score)."""
    n = z0.size
    z = np.array(z0, dtype=np.float64).copy()
    z = z - z.mean()
    
    a = _softmax_stable(z)
    c = _fft_convolve(a, a)
    max_c = float(c.max())
    
    if beta_start is None:
        beta_start = 0.1 * max_c
    
    beta = max(beta_min, beta_start)
    best_score = 2.0 * n * max_c
    best_z = z.copy()
    
    if verbose:
        print(f"  FISTA [{label}] n={n}, beta0={beta:.6e}, init_score={best_score:.8f}")
    
    total_iters = 0
    
    while time.monotonic() < deadline and beta >= beta_min:
        # FISTA state
        z_curr = z.copy()
        z_prev = z.copy()
        t = 1.0
        no_improve = 0
        beta_iters = 0
        
        while time.monotonic() < deadline and beta_iters < max_iters_per_beta:
            beta_iters += 1
            total_iters += 1
            
            # Extrapolation
            z_y = z_curr + ((t - 1.0) / (t + 2.0)) * (z_curr - z_prev)
            
            # Gradient at z_y
            grad, c_y, a_y = _gradient_logits(z_y, beta)
            
            if not np.all(np.isfinite(grad)):
                z_curr = best_z.copy()
                break
            
            # Take step
            z_new = z_y - lr * grad
            z_new = np.clip(z_new, -50.0, 50.0)
            
            # Evaluate
            a_new = _softmax_stable(z_new)
            c_new = _fft_convolve(a_new, a_new)
            score_new = 2.0 * n * float(c_new.max())
            
            if not np.isfinite(score_new):
                z_curr = best_z.copy()
                t = 1.0
                z_prev = z_curr.copy()
                continue
            
            if score_new < best_score - 1e-15:
                best_score = score_new
                best_z = z_new.copy()
                no_improve = 0
            else:
                # Check if smooth objective decreased (for FISTA validity)
                # If not, reset
                c_smooth = float(c_new.max() + beta * np.log(np.exp((c_new - c_new.max()) / beta).sum()))
                c_y_smooth = float(c_y.max() + beta * np.log(np.exp((c_y - c_y.max()) / beta).sum()))
                if 2.0 * n * c_smooth > 2.0 * n * c_y_smooth + 1e-6:
                    # Function increased too much, reset
                    z_new = z_curr.copy()
                    score_new = 2.0 * n * float(_fft_convolve(_softmax_stable(z_new), _softmax_stable(z_new)).max())
                    no_improve += 1
                
                if no_improve > 500:
                    break
            
            z_prev = z_curr.copy()
            z_curr = z_new.copy()
            t = t + 1.0
        
        z = best_z.copy()
        
        if verbose and beta_iters > 0:
            print(f"    beta={beta:.3e}: {beta_iters} iters, best={best_score:.8f}")
        
        beta = max(beta_min, beta * beta_decay)
        # Adapt lr as beta decreases
        lr = max(0.001, lr * 0.85)
    
    return best_z, best_score


def _lp_correction(a, M, max_time=30.0):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)_k <= M for all k.
    Returns normalized b or None."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, coo_matrix
    
    n = a.size
    t_lp_start = time.monotonic()
    
    # Build constraint matrix A where (a*b)_k = sum_j A[k,j] * b_j <= M
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1 else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For each j, find all k such that 0 <= k-j <= n-1
        # k ranges from j to j+n-1
        k_start = j
        k_end = min(j + n, 2*n-1)
        # a[k-j] for k in [k_start, k_end)
        indices = np.arange(k_start, k_end)
        a_idx = indices - j  # = 0, 1, ..., n-1 (clipped)
        a_idx = a_idx[a_idx < n]
        indices = indices[a_idx < n]
        rows.extend(indices.tolist())
        cols.extend(j * len(indices))
        vals.extend(a[a_idx].tolist())
    
    A = coo_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if time.monotonic() - t_lp_start > max_time:
        return None
    
    if res.success:
        b = res.x
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None


def _make_symmetric(a):
    """Make sequence symmetric while preserving sum."""
    n = a.size
    a_s = 0.5 * (a + a[::-1])
    return _normalize(a_s)


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _make_profile(n, kind='linear', params=None):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        eta = params.get('eta', 1.0) if params else 1.0
        a = 1.0 + eta * (t - 0.5)
    elif kind == 'quadratic':
        a1 = params.get('a1', 0.0) if params else 0.0
        a2 = params.get('a2', 0.5) if params else 0.5
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a2 * u**2
    elif kind == 'cubic':
        a1 = params.get('a1', 1.0) if params else 1.0
        a3 = params.get('a3', -0.5) if params else -0.5
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a3 * u**3
    elif kind == 'cosine':
        a = 1.0 + np.cos(np.pi * t)
    elif kind == 'power':
        p = params.get('p', 0.5) if params else 0.5
        a = t**p + (1-t)**p + 0.1
    else:
        a = np.ones(n)
    return _normalize(np.maximum(a, 1e-12))
