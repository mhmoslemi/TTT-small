Optimization of non-negative sequence to minimize 2*n*max(conv(s,s))/(sum(s))^2.
Combines softmax gradient descent, LP-based direction finding, spectral flattening,
and subgradient methods with momentum. Uses multiple sequence lengths and starting points.

import numpy as np
import time
from scipy.optimize import linprog


def calc_score(s):
    """Compute the objective function value."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def convolve_ff(s):
    """Fast convolution using FFT."""
    n = len(s)
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    conv = np.fft.irfft(S * S, N_fft)[:N]
    return conv


def softmax_gradient(s, conv, beta):
    """Compute softmax-weighted gradient of max(conv)."""
    n = len(s)
    max_conv = np.max(conv)
    if max_conv <= 0:
        return np.zeros(n)
    # Softmax weights
    logits = beta * (conv - max_conv) / (max_conv + 1e-30)
    logits -= np.max(logits)
    weights = np.exp(logits)
    weights /= np.sum(weights)

import numpy as np

from scipy.optimize import linprog

def calc_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def conv_ff(s):
    n = len(s)
    N = 2 * n - 1
    Nf = 1
    while Nf < N:
        Nf *= 2
    S = np.fft.rfft(s, Nf)
    return np.fft.irfft(S * S, Nf)[:N]

def gradient_s(s, w):
    """Compute the 'gradient' for multiplicative update.
    Returns g such that s_new = s * exp(-eta * g / max(g)) is a descent direction.
    """
    n = len(s)
    Nf = 1
    while Nf < 4 * n:
        Nf *= 2
    S = np.fft.rfft(s, Nf)
    W = np.fft.rfft(w, Nf)
    term1 = np.fft.irfft(np.conj(S) * W, Nf)[:n]
    term2 = np.fft.irfft(S * W, Nf)[:n]
    g = term1 + term2
    return np.maximum(g, 0.0)

def lp_solve(s, M, timeout=5.0):
    """Solve LP: max sum(b) s.t. conv(s,b) <= M, b >= 0."""
    n = len(s)
    # conv(s,b)[k] = sum_i s[i]*b[k-i] <= M for all k
    # Number of constraints: 2n-1
    # Build constraint matrix A (size (2n-1) x n)
    # A[k, j] = s[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = s[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = s
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None

def lp_solve_tight(s, M, timeout=5.0):
    """Solve LP with slightly tighter constraint to force improvement."""
    return lp_solve(s, M, timeout)

def multiplicative_update(s, g, eta):
    """Apply multiplicative update."""
    g_max = g.max()
    if g_max < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * g / g_max)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / s_new.sum()
    return s_new

def spectral_flatten(s, n_iter=1000, deadline=None):
    """Spectral flattening: iterate to make autocorrelation flat."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = calc_score(s)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # S^2 in frequency domain corresponds to s*s in time domain
        # We want IFFT(S^2) to be flat
        # Target: constant = mean(s)^2 * n (approximately)
        # Actually, sum(b) = (sum(s))^2, so mean(b) = (sum(s))^2 / (2n-1)
        target = (s.sum() ** 2) / (2 * n - 1)

        # |S(f)|^2 = |S(f)|^2 (real and non-negative since S is from real s)
        # Actually S^2 (not |S|^2) is what appears in the convolution
        # For real s, S(-f) = conj(S(f)), so S^2(-f) = conj(S(f))^2 = conj(S^2(f))
        # The IFFT of S^2 gives the convolution
        
        # Strategy: modify S to make S^2 more uniform
        # S^2 should be approximately target (constant)
        # So S should be approximately sqrt(target) (in magnitude)
        # But S can be complex, so we adjust the magnitude
        
        S_mag = np.abs(S)
        S_phase = np.angle(S)
        
        # Target magnitude: sqrt(target) for all frequencies
        # But we need to be careful: the DC component should be sum(s)
        S_target_mag = np.full_like(S_mag, np.sqrt(target))
        S_target_mag[0] = s.sum()  # DC component
        
        # Interpolate between current and target
        alpha = 0.1  # relaxation
        S_new_mag = S_mag + alpha * (S_target_mag - S_mag)
        S_new = S_new_mag * np.exp(1j * S_phase)
        
        s_new = np.fft.irfft(S_new, 2 * n - 1)[:n]
        s_new = np.maximum(s_new, 0.0)
        if s_new.sum() < 1e-12:
            break
        s_new = s_new / s_new.sum()
        
        sc = calc_score(s_new)
        if sc < best_score:
            best_score = sc
            best_s = s_new.copy()
        
        s = s_new
    
    return best_s, best_score

"""
Optimization of non-negative sequence to minimize 2*n*max(conv(s,s))/(sum(s))^2.
Combines softmax-weighted multiplicative gradient descent, LP-based direction finding,
spectral flattening, and coordinate descent polishing. Uses multiple sequence lengths
and diverse starting points including the provided height_sequence_1.
"""

import numpy as np
import time
from scipy.optimize import linprog


def calc_score(s):
    """Compute the objective: 2*n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def conv_ff(s):
    """Fast full convolution via FFT."""
    n = len(s)
    N = 2 * n - 1
    Nf = 1
    while Nf < N:
        Nf *= 2
    S = np.fft.rfft(s, Nf)
    return np.fft.irfft(S * S, Nf)[:N]


def gradient_s(s, w):
    """Compute combined gradient for multiplicative update using FFT."""
    n = len(s)
    Nf = 1
    while Nf < 4 * n:
        Nf *= 2
    S = np.fft.rfft(s, Nf)
    W = np.fft.rfft(w, Nf)
    term1 = np.fft.irfft(np.conj(S) * W, Nf)[:n]
    term2 = np.fft.irfft(S * W, Nf)[:n]
    g = term1 + term2
    return np.maximum(g, 0.0)


def lp_solve(s, M, timeout=5.0):
    """Solve LP: max sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None


def multiplicative_update(s, g, eta):
    """Apply multiplicative (exponential) update maintaining non-negativity."""
    g_max = g.max()
    if g_max < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * g / g_max)
    s_new = np.maximum(s_new, 1e-15)
    total = s_new.sum()
    if total < 1e-15:
        return s.copy()
    return s_new / total


def coordinate_polish(s, n_trials=200, rng=None):
    """Coordinate descent polishing: try small multiplicative changes to individual elements."""
    n = len(s)
    s = s.copy()
    best_score = calc_score(s)
    if rng is None:
        rng = np.random.default_rng(0)
    for _ in range(n_trials):
        i = rng.integers(0, n)
        for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
            s_try = s.copy()
            s_try[i] *= factor
            if s_try[i] < 0:
                continue
            sc = calc_score(s_try)
            if sc < best_score - 1e-12:
                best_score = sc
                s = s_try
                break
    return s, best_score


def local_optimize(s, deadline, rng, use_lp=True, lp_freq=25):

def spectral_flatten(s, n_iter=800, deadline=None):
    """Spectral domain flattening of autocorrelation."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = calc_score(s)
    alpha = 0.15

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        Nf = 1
        while Nf < 2 * n - 1:
            Nf *= 2
        S = np.fft.rfft(s, Nf)

        # Target: make S^2 have flat real part
        # S^2 magnitude should be ~ (sum(s))^2/(2n-1) for flat autocorr
        target_mag = (s.sum() ** 2) / (2 * n - 1)

        S_mag = np.abs(S)
        S_phase = np.angle(S)

        # Adjust magnitude toward target
        S_target_mag = np.full_like(S_mag, np.sqrt(target_mag))
        S_target_mag[0] = s.sum()  # preserve DC

        S_new_mag = S_mag + alpha * (S_target_mag - S_mag)
        S_new = S_new_mag * np.exp(1j * S_phase)

        s_new = np.fft.irfft(S_new, Nf)[:n]
        s_new = np.maximum(s_new, 0.0)
        tot = s_new.sum()
        if tot < 1e-12:
            break
        s_new = s_new / tot

        sc = calc_score(s_new)
        if sc < best_score:
            best_score = sc
            best_s = s_new.copy()
        s = s_new

    return best_s, best_score


def param_starts(n, rng):
    """Generate diverse parameterized starting points."""
    results = []

    # Uniform
    results.append(np.ones(n))

    # Power laws
    for p in [0.3, 0.7, 1.0, 1.5, 2.0, 3.0, -0.3, -0.7, -1.5]:
        s = np.power(np.arange(1, n + 1, dtype=float), p)
        s = np.maximum(s, 0.05)
        results.append(s.copy())

    # Gaussian
    for center in [0.2, 0.35, 0.5, 0.65, 0.8]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            x = np.arange(n) / max(n - 1, 1)
            s = np.exp(-(x - center) ** 2 / (2.0 * width ** 2))
            s = np.maximum(s, 0.01)
            results.append(s.copy())

    # Cosine
    for a in [0.5, 0.8, 1.0, 1.3, 1.7]:
        x = np.arange(n) / n
        s = np.cos(a * np.pi * (x - 0.5)) ** 2
        s = np.maximum(s, 0.01)
        results.append(s.copy())

    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        k = max(1, int(frac * n))
        for ratio in [0.3, 0.5, 2.0, 3.0]:
            s = np.ones(n)
            s[:k] = ratio
            results.append(s.copy())

    # Random
    for _ in range(8):
        s = rng.random(n) + 0.1
        results.append(s.copy())

    # Smooth random (low-pass)
    for _ in range(4):
        base = rng.random(max(n // 8, 2))
        s = np.interp(np.linspace(0, len(base) - 1, n), np.arange(len(base)), base)
        s = np.maximum(s, 0.1)
        results.append(s.copy())

    return results
