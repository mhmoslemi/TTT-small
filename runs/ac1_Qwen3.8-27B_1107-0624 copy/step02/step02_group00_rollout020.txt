def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())

        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    n = len(a)

    if use_symmetry:
        a = _symmetrize(a)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if use_symmetry:
            gz = _symmetrize(gz)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        if use_symmetry:
            a = _symmetrize(a)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.93)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(2000, 2 * n):
            rng_r = np.random.default_rng(it * 104729)
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            if use_symmetry:
                z = _symmetrize(z)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

        if it % 500 == 0:
            print(f"      Adam iter {it}: score={best_score:.10f}, beta={beta:.2e}, lr={lr:.5f}")

    return best_a


def _make_cosine_profile(n, coeffs=None):
    """Create a symmetric profile using cosine basis."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones(n)
    if coeffs is None:
        coeffs = [0.0, 0.3, -0.1, 0.15, -0.05]
    for k, c in enumerate(coeffs):
        a = a + c * np.cos(2.0 * np.pi * (k + 1) * t)
    a = np.maximum(a, 1e-6)
    a = _symmetrize(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_parabolic_profile(n, curvature=1.0):
    """Parabolic (edge-concentrated) profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + curvature * u * u
    a = np.maximum(a, 1e-6)
    a = _symmetrize(a)
    a = np.maximum(a, 0)
    return a / a.sum()


def _make_trapezoid_profile(n, taper=0.3):
    """Trapezoid profile: flat middle, tapered edges."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    mask = t < taper
    a[mask] = t[mask] / taper
    mask2 = t > (1.0 - taper)
    a[mask2] = (1.0 - t[mask2]) / taper
    a = np.maximum(a, 1e-6)
    a = _symmetrize(a)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _coarse_to_fine(n_target, deadline, use_symmetry=True):
    """Optimize at progressively larger sizes, using the result as initialization."""
    sizes = []
    s = 32
    while s < n_target:
        sizes.append(s)
        s *= 2
    sizes.append(n_target)

    a = None
    for n_cur in sizes:
        if time.monotonic() > deadline:
            break
        if a is None:
            a = _make_parabolic_profile(n_cur, 0.8)
        else:
            a = _interpolate_profile(a, n_cur)

        a = _symmetrize(a)
        a = np.maximum(a, 0)
        a = a / a.sum()

        sc = _score(a)
        print(f"    C2F n={n_cur}: score={sc:.10f}")

        if n_cur >= 64:
            sub_deadline = min(deadline, time.monotonic() + 60.0)
            a = _lp_refine(a, sub_deadline, max_iter=50000, inflate=1.001, blend=0.2, use_symmetry=use_symmetry)
            sc = _score(a)
            print(f"    C2F n={n_cur} after LP: score={sc:.10f}")

    return a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define candidate lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [256, 512, 1024]:
        if ln not in lengths:
            lengths.append(ln)

    print(f"Target lengths: {lengths}")
    print(f"Total time budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")

    # Phase 1: Coarse-to-fine for each target length
    for n_t in lengths:
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic()
        print(f"\n=== Length n={n_t}, remaining={remaining:.0f}s ===")

        # Generate multiple initializations
        inits = []

        # From parent
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))

        # Cosine profiles
        for coeffs in [
            [0.3, -0.1, 0.15, -0.05, 0.03],
            [0.5, -0.15, 0.1, -0.08, 0.05],
            [0.2, 0.2, -0.2, 0.1, -0.05],
            [0.4, -0.2, 0.3, -0.1, 0.1],
        ]:
            inits.append(_make_cosine_profile(n_t, coeffs))

        # Parabolic profiles
        for curv in [0.5, 0.8, 1.2, 1.5]:
            inits.append(_make_parabolic_profile(n_t, curv))

        # Trapezoid profiles
        for taper in [0.2, 0.3, 0.4]:
            inits.append(_make_trapezoid_profile(n_t, taper))

        # Random perturbation of parent
        if parent is not None:
            p_pert = _interpolate_profile(parent, n_t)
            noise = rng.normal(size=n_t) * 0.15
            p_pert = np.maximum(p_pert + noise, 0)
            p_pert = p_pert / p_pert.sum()
            inits.append(p_pert)

        # Score all initializations
        best_init = None
        best_init_score = np.inf
        for a0 in inits:
            a0 = _symmetrize(a0)
            a0 = np.maximum(a0, 0)
            a0 = a0 / a0.sum()
            sc = _score(a0)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a0

        print(f"  Best init: score={best_init_score:.10f}")
        if np.isfinite(best_init_score) and best_init_score < best_score:
            best_score = best_init_score
            best_seq = best_init.copy()

        # LP refinement on best init
        lp_remaining = deadline - time.monotonic()
        lp_budget = min(lp_remaining * 0.5, 500.0)
        if lp_budget > 30:
            lp_end = time.monotonic() + lp_budget
            print(f"  LP refine (budget={lp_budget:.0f}s):")
            a_lp = _lp_refine(best_init, lp_end, max_iter=200000,
                              inflate=1.0002, blend=0.12, use_symmetry=True)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  LP result: score={sc_lp:.10f}")

            # Second LP pass with tighter params
            lp_remaining = deadline - time.monotonic()
            if lp_remaining > 60:
                lp_budget2 = min(lp_remaining * 0.3, 200.0)
                lp_end2 = time.monotonic() + lp_budget2
                print(f"  LP refine 2 (budget={lp_budget2:.0f}s):")
                a_lp2 = _lp_refine(a_lp, lp_end2, max_iter=100000,
                                   inflate=1.0001, blend=0.08, use_symmetry=True)
                sc_lp2 = _score(a_lp2)
                if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                    best_score = sc_lp2
                    best_seq = a_lp2.copy()
                print(f"  LP2 result: score={sc_lp2:.10f}")
                a_lp = a_lp2

            # Adam polish
            ad_remaining = deadline - time.monotonic()
            if ad_remaining > 30:
                ad_budget = min(ad_remaining * 0.3, 150.0)
                ad_end = time.monotonic() + ad_budget
                print(f"  Adam polish (budget={ad_budget:.0f}s):")
                a_ad = _adam_logit_polish(a_lp, ad_end, max_iter=50000,
                                          lr=0.03, use_symmetry=True)
                sc_ad = _score(a_ad)
                if np.isfinite(sc_ad) and sc_ad < best_score:
                    best_score = sc_ad
                    best_seq = a_ad.copy()
                print(f"  Adam result: score={sc_ad:.10f}")

                # Final LP after Adam
                final_remaining = deadline - time.monotonic()
                if final_remaining > 30:
                    final_budget = min(final_remaining * 0.5, 100.0)
                    final_end = time.monotonic() + final_budget
                    print(f"  Final LP (budget={final_budget:.0f}s):")
                    a_fin = _lp_refine(a_ad, final_end, max_iter=50000,
                                       inflate=1.0001, blend=0.05, use_symmetry=True)
                    sc_fin = _score(a_fin)
                    if np.isfinite(sc_fin) and sc_fin < best_score:
                        best_score = sc_fin
                        best_seq = a_fin.copy()
                    print(f"  Final score: {sc_fin:.10f}")

    # Phase 2: Try a few more diverse inits on the best length
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = best_seq.size
        print(f"\n=== Extra exploration at n={n_best} ===")

        extra_inits = []
        # Perturbed versions of best
        for scale in [0.05, 0.1, 0.2]:
            pert = best_seq * (1.0 + scale * rng.normal(size=n_best))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            extra_inits.append(pert)

        # Different profile shapes
        for curv in [0.3, 2.0, 2.5]:
            extra_inits.append(_make_parabolic_profile(n_best, curv))

        for a0 in extra_inits:
            if time.monotonic() > deadline - 30:
                break
            a0 = _symmetrize(a0)
            a0 = np.maximum(a0, 0)
            a0 = a0 / a0.sum()
            sc = _score(a0)

            if sc < best_score:
                sub_deadline = min(deadline, time.monotonic() + 60.0)
                a_ref = _lp_refine(a0, sub_deadline, max_iter=30000,
                                   inflate=1.0005, blend=0.15, use_symmetry=True)
                sc = _score(a_ref)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_ref.copy()
                    print(f"  Extra init improved: score={sc:.10f}")

    # Phase 3: Final Adam polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        ad_budget = min(deadline - time.monotonic() - 10, 120.0)
        if ad_budget > 10:
            ad_end = time.monotonic() + ad_budget
            print(f"\nFinal Adam polish (budget={ad_budget:.0f}s):")
            a_final = _adam_logit_polish(best_seq, ad_end, max_iter=30000,
                                         lr=0.02, use_symmetry=True)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final Adam: score={sc_final:.10f}")

    # Phase 4: One last LP with very tight parameters
    if best_seq is not None and time.monotonic() < deadline - 10:
        lp_budget = min(deadline - time.monotonic() - 5, 60.0)
        if lp_budget > 5:
            lp_end = time.monotonic() + lp_budget
            a_last = _lp_refine(best_seq, lp_end, max_iter=20000,
                                inflate=1.00005, blend=0.05, use_symmetry=True)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Last LP: score={sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_parabolic_profile(256, 1.0)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_parabolic_profile(256, 1.0)
        s = float(final_seq.sum())
    final_seq = final_seq / s

    # Evaluate with the authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence score: {sc_eval:.10f}")
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
