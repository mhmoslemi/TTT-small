"""
Optimized step function search using exact coordinate descent with Nesterov
momentum, multiple n values, structured restarts, and gradient polishing.

import numpy as np
import time
from scipy.optimize import linprog


def compute_conv(f):
    """Compute full linear convolution of f with itself."""
    return np.convolve(f, f)


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def normalize_to_sum1(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def coord_optimize_single(c, a, old_val, n_steps=60):
    """
    Find optimal t >= 0 minimizing max_k (a[k]*t + C[k])
    where C[k] = c[k] - old_val * a[k].
    Returns the optimal t.
    """
    C = c - old_val * a
    g0 = C.max()

    # If all a <= 0, minimum is at t=0
    if a.max() <= 1e-15:
        return 0.0

    # Find upper bound
    t_hi = max(2.0 * old_val, 0.5)
    for _ in range(20):
        g_hi = (a * t_hi + C).max()
        if g_hi > g0:
            break
        t_hi *= 2

    # Bisection on subgradient sign
    t_lo = 0.0
    for _ in range(n_steps):
        t_mid = 0.5 * (t_lo + t_hi)
        vals = a * t_mid + C
        max_val = vals.max()
        # Check subgradient: find all near-max positions
        idx_max = np.argmax(vals)
        a_max = a[idx_max]
        # Also check if there's a negative a at a near-max position
        near_max = vals >= max_val - 1e-12 * max(1.0, abs(max_val))
        a_near = a[near_max]
        if a_near.min() > 1e-12:
            t_hi = t_mid
        elif a_near.max() < -1e-12:
            t_lo = t_mid
        else:
            break

    return max(0.0, 0.5 * (t_lo + t_hi))


def coordinate_descent(f_init, max_sweeps=50000, deadline=None, recompute_every=50):
    """
    Exact coordinate descent minimizing max(conv(f,f)) with sum(f)=1 constraint.
    Uses cyclic coordinate updates with exact 1D optimization.
    """
    n = len(f_init)
    f = f_init.copy()
    c = compute_conv(f)
    M = float(c.max())
    best_M = M
    best_f = f.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Periodically recompute full convolution to avoid drift
        if sweep % recompute_every == 0 and sweep > 0:
            c = compute_conv(f)
            M = float(c.max())

        sweep_improved = False
        for i in range(n):
            old_val = f[i]

            # Compute a[k] = f[k-i] + f[i-k] (with bounds)
            a = np.zeros(2 * n - 1)
            # f[k-i] valid for k in [i, i+n-1]
            a[i:i + n] += f
            # f[i-k] valid for k in [max(0, i-n+1), i]
            lo = max(0, i - n + 1)
            hi = i + 1
            a[lo:hi] += f[i - lo::-1]

            # Find optimal t
            new_val = coord_optimize_single(c, a, old_val, n_steps=50)

            if abs(new_val - old_val) > 1e-14:
                delta = new_val - old_val
                c += delta * a
                f[i] = new_val
                new_M = float(c.max())
                if new_M < M - 1e-14:
                    M = new_M
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
        else:
            no_improve += 1
            if no_improve > 5:
                break

    return best_f, best_M


def nesterov_gradient(f_init, deadline, lr=0.01, momentum=0.9, max_iters=200000):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Uses exact subgradient of max(conv(f,f)).
    """
    n = len(f_init)
    f = f_init.copy()
    f = normalize_to_sum1(f)
    if f is None:
        return f_init, compute_conv(f_init).max()

    v = np.zeros(n)  # momentum
    best_M = float(np.convolve(f, f).max())
    best_f = f.copy()
    z = np.log(f + 1e-30)  # logit representation

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(f, f)
        M = float(c.max())

        # Find all near-max positions
        max_val = M
        idx_max = np.where(c >= max_val - 1e-10 * max(1.0, abs(max_val)))[0]

        # Subgradient of max_c with respect to f[i]
        grad = np.zeros(n)
        for k in idx_max:
            # d(max_c)/df[i] involves f[k-i] and f[i-k]
            # For position k in convolution, d(conv[k])/df[i] = f[k-i] + f[i-k]
            for i in range(n):
                contribution = 0.0
                if 0 <= k - i < n:
                    contribution += f[k - i]
                if 0 <= i - k < n:
                    contribution += f[i - k]
                grad[i] += contribution / len(idx_max)

        # Nesterov lookahead
        f_lookahead = f + momentum * v
        f_lookahead = normalize_to_sum1(f_lookahead)
        if f_lookahead is None:
            f_lookahead = f.copy()

        # Recompute gradient at lookahead
        c_la = np.convolve(f_lookahead, f_lookahead)
        M_la = float(c_la.max())
        idx_la = np.where(c_la >= M_la - 1e-10 * max(1.0, abs(M_la)))[0]

        grad_la = np.zeros(n)
        for k in idx_la:
            for i in range(n):
                contribution = 0.0
                if 0 <= k - i < n:
                    contribution += f_lookahead[k - i]
                if 0 <= i - k < n:
                    contribution += f_lookahead[i - k]
                grad_la[i] += contribution / len(idx_la)

        # Project gradient to tangent space of simplex
        grad_la = grad_la - np.mean(grad_la)

        v = momentum * v - lr * grad_la
        f = f + v
        f = normalize_to_sum1(f)
        if f is None:
            f = best_f.copy()
            v = np.zeros(n)
            continue

        M = float(np.convolve(f, f).max())
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()

    return best_f, best_M

def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * (t / max(n - 1, 1))
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cosine_profile(n, a1=1.0, phase=0.0):
    t = np.linspace(0, 1, n)
    f = 1.0 + a1 * np.cos(2 * np.pi * t + phase)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_quadratic_profile(n, a=1.0):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    f = 1.0 + a * u * u
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_fourier_profile(n, coeffs):
    t = np.linspace(0, 1, n)
    f = 1.0
    for k, (a, b) in enumerate(coeffs):
        f += a * np.cos(2 * np.pi * (k + 1) * t) + b * np.sin(2 * np.pi * (k + 1) * t)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def add_bump(f, center, width, height):

    return normalize_to_sum1(f_new)


def mirror_flip(f):

def lp_optimize_step(f, M, deadline):
    """Single LP-based optimization step."""
    n = len(f)
    # Build constraint matrix: A[i,j] such that sum_j A[i,j]*g[j] = conv(f,g)[i]
    # conv(f,g)[k] = sum_j f[j]*g[k-j]
    # For fixed k: sum_j f[j]*g[k-j] = sum_j f[k-j]*g[j] (reindexing)
    # So A[k,j] = f[k-j] if k-j in [0,n-1], else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def fast_grad_step(f, lr=0.01):

    c = np.convolve(f, f)

    return normalize_to_sum1(f_new)

    for n_val in n_values:
        time_alloc = time_weights.get(n_val, 0.1) * (total_time - 60)

    print(f"Total budget: {budget_s}s, effective: {total_time:.0f}s")

    # Process each n value
    for n_val in n_values:
        if time.monotonic() > deadline - 60:
            break

        n_time = time_weights.get(n_val, 0.1) * (total_time - 60)
        n_deadline = min(deadline, time.monotonic() + n_time)
        print(f"\n=== n={n_val}, time={n_time:.0f}s ===")

        # Generate starting points
        starts = get_starting_points(n_val, rng, count=12)

        # Add parent if available
        if parent is not None:
            starts.append(interpolate_profile(parent, n_val))

        # Add some structured perturbations of the best found so far
        if best_f is not None:
            starts.append(interpolate_profile(best_f, n_val))
            # Add bump perturbations
            for _ in range(3):
                center = rng.uniform(0.2, 0.8)
                width = rng.uniform(0.05, 0.2)
                height = rng.uniform(0.5, 2.0)
                perturbed = add_bump(interpolate_profile(best_f, n_val), center, width, height)
                if perturbed is not None:
                    starts.append(perturbed)
            # Mirror
            if best_f.size == n_val:
                starts.append(mirror_flip(best_f))

        # Evaluate all starts
        start_scores = []
        for s in starts:
            sc = compute_score(s)
            start_scores.append((sc, s))
        start_scores.sort(key=lambda x: x[0])

        n_starts = len(start_scores)
        print(f"  {n_starts} starting points, best initial: {start_scores[0][0]:.10f}")

        # Run optimization from top starts
        top_k = min(8, n_starts)
        results = []
        for i in range(top_k):
            remaining = n_deadline - time.monotonic()
            if remaining < 3:
                break
            alloc = min(remaining / max(1, top_k - i), n_time / max(1, top_k))
            if alloc < 2:
                break
            run_deadline = time.monotonic() + alloc

            sc_init, f_init = start_scores[i]
            f_opt, M_opt = combined_optimize(f_init, run_deadline, rng,
                                             cd_sweeps=30000, grad_iters=20000)
            sc = 2.0 * n_val * M_opt
            try_update(f_opt, sc)
            results.append((sc, f_opt))
            if (i + 1) % 4 == 0 or i == top_k - 1:
                print(f"  After {i+1}/{top_k}: best_n{n_val}={min(r[0] for r in results):.10f}, "
                      f"global_best={best_score:.10f}")

        # LP-based refinement on best result for this n
        if results and time.monotonic() < n_deadline - 10:
            results.sort(key=lambda x: x[0])
            sc_best, f_best_local = results[0]
            remaining = n_deadline - time.monotonic()
            if remaining > 10:
                f_lp, M_lp = lp_iterate(f_best_local, time.monotonic() + remaining * 0.7, max_iters=100)
                sc_lp = 2.0 * n_val * M_lp
                try_update(f_lp, sc_lp)
                print(f"  LP refinement: {sc_lp:.10f}, global_best={best_score:.10f}")

                # Final CD after LP
                if time.monotonic() < n_deadline - 3:
                    f_cd, M_cd = coordinate_descent(f_lp, max_sweeps=50000,
                                                    deadline=time.monotonic() + (n_deadline - time.monotonic()))
                    sc_cd = 2.0 * n_val * M_cd
                    try_update(f_cd, sc_cd)
                    print(f"  Post-LP CD: {sc_cd:.10f}, global_best={best_score:.10f}")

    # Final global polish: take the best and do extended CD + gradient
    if best_f is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final polish ===")
        n_best = len(best_f)
        M_best = float(np.convolve(best_f, best_f).max())
        print(f"  Before polish: n={n_best}, score={2.0*n_best*M_best:.10f}")

        # Extended coordinate descent
        remaining = deadline - time.monotonic() - 20
        f_opt, M_opt = coordinate_descent(best_f, max_sweeps=200000,
                                          deadline=time.monotonic() + remaining * 0.6)
        sc = 2.0 * n_best * M_opt
        try_update(f_opt, sc)

        # Adam gradient
        if time.monotonic() < deadline - 15:
            f_adam, M_adam = adam_gradient(best_f, deadline=time.monotonic() + (deadline - time.monotonic() - 10) * 0.5,
                                           lr=0.01, max_iters=50000, beta_param=0.03)
            sc_adam = 2.0 * n_best * M_adam
            try_update(f_adam, sc_adam)

        # One more CD
        if time.monotonic() < deadline - 5:
            f_final, M_final = coordinate_descent(best_f, max_sweeps=100000,
                                                  deadline=deadline - 3)
            sc_final = 2.0 * n_best * M_final
            try_update(f_final, sc_final)

        print(f"  After polish: global_best={best_score:.10f}")

    # Validate and return
    if best_f is None:
        best_f = make_linear_profile(256, 1.5)
        best_score = compute_score(best_f)

    best_f = np.maximum(best_f, 0.0)
    s = float(best_f.sum())
    if s < 0.01:
        best_f = make_linear_profile(256, 1.5)
    else:
        best_f = best_f / s

    best_f = np.clip(best_f, 0.0, 1000.0)

    final_score = compute_score(best_f)
    print(f"\nFinal: n={len(best_f)}, score={final_score:.10f}")

    # Verify with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_f.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_f]
