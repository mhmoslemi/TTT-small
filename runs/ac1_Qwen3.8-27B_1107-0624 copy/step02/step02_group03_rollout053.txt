def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def softmax_weights(c, temperature):
    """Compute softmax weights over the convolution values."""
    x = (c - c.max()) / temperature
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return ex / s

The key optimization is:
- Minimize `max(conv(a,a))` subject to `a >= 0` and `sum(a) = 1`.

The subgradient at `a` (when the max is achieved at a unique point k*) is:
`g[i] = a[i] + a[k*-i]` (for valid k*-i, i.e., 0 <= k*-i < n)

So `g[i] = a[i] + a[k*-i] * [0 <= k*-i < n]`

This is a very simple formula! The subgradient at index `i` is `a[i]` plus `a[k*-i]` if `k*-i` is a valid index.

For a "flat" convolution, we'd want all `conv[k]` to be equal. The subgradient tells us where to push mass: if `g[i]` is large, we should decrease `a[i]`.

**Optimization algorithm:**

1. Initialize `a` (normalized, non-negative)
2. Repeat:
   a. Compute `c = conv(a,a)`
   b. Find the peak(s) of `c`
   c. Compute subgradient `g`
   d. Take a step: `a_new = a - lr * g` (projected: clip to non-negative, normalize)
   e. Check if improved; if not, reduce `lr`

This is essentially subgradient descent. To make it more effective:
- Use momentum (Polyak averaging or Nesterov)
- Use the "soft" subgradient (softmax over top peaks) for a smoother gradient
- Use adaptive step sizes (Barzilai-Borwein)

Let me also think about the LP approach more carefully:

The LP finds a direction `g` that maximizes `sum(g)` subject to `conv(f, g) <= M` (where `f` is the current solution and `M = max(conv(f,f))`).

The idea is: we want to move in a direction that doesn't increase the max convolution (beyond M), while increasing the sum (which allows us to potentially reduce the max relative to the sum).

But actually, since we're normalizing to sum=1, the LP approach is finding a direction that keeps `conv(f, g) <= M` while maximizing the "spread" of `g`.

Minimize `max_k (conv(f+f*d, f+f*d)[k])` subject to `f+f*d >= 0` and `sum(f+f*d) = 1`.

"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive subgradient/Adam descent with softmax smoothing, LP-based direction finding,
multi-start strategy, and Barzilai-Borwein step sizes. Focuses on n=512 with extended
optimization time and aggressive temperature annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def softmax_weights(c, temperature):
    """Compute softmax weights over the convolution values."""
    x = (c - c.max()) / temperature
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return ex / s


def compute_subgrad(a, w):
    """
    Compute the subgradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    subgrad[i] = a[i] * sum_{k=i}^{i+n-1} w[k] + sum_{j=0}^{n-1} w[i+j]*a[j]
    """
    n = len(a)
    # First part: a[i] * sliding sum of w over [i, i+n-1]
    w_pad = np.concatenate([w, np.zeros(n - 1)])
    cs = np.concatenate([[0.0], np.cumsum(w_pad)])
    sliding = cs[n:n + n] - cs[0:n]
    part1 = a * sliding

    # Second part: corr[i] = sum_{j=0}^{n-1} w[i+j]*a[j]
    # Use direct computation (fast enough for n=512)
    part2 = np.zeros(n)
    for i in range(n):
        part2[i] = np.dot(w[i:i + n], a)

    return part1 + part2


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_profile(n, alpha):
    """Create a cosine-bump profile: 1 + alpha*cos(pi*u), u in [-1,1]."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * np.cos(np.pi * u)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_gaussian_profile(n, sigma):
    """Create a Gaussian-bump profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.exp(-u**2 / (2 * sigma**2))
    a = a + 0.1  # floor to avoid near-zeros
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_profile(n, rng, power=1.0):
    """Create a random non-negative profile with power-law weighting."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = rng.random(n) ** (1.0 / power)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def adam_subgrad(a0, deadline, lr_init=0.05, temp_init=None, temp_min=1e-4,
                 verbose=True, label=""):
    """
    Adam-based subgradient descent with softmax smoothing of the max.
    Optimizes in logit space for non-negativity.
    Returns (best_a, best_score, best_M).
    """
    n = len(a0)
    a = np.maximum(a0, 1e-12).astype(np.float64)
    a = a / a.sum()

    c = np.convolve(a, a)
    max_c = float(c.max())
    if temp_init is None:
        temp_init = 0.1 * max_c

    z = np.log(a)
    z -= np.mean(z)

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    best_M = max_c
    no_imp = 0
    temp = temp_init
    base_lr = lr_init

    # Track for BB step size
    prev_a = a.copy()
    prev_grad = None

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        max_c = float(c.max())

        # Softmax weights
        w = softmax_weights(c, temp)

        # Subgradient
        grad_p = compute_subgrad(a, w)

        # Logit space gradient
        dot = float(np.dot(a, grad_p))
        gz = a * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            temp = 0.1 * best_M
            continue

        # BB step size adaptation (every 10 iterations)
        if prev_grad is not None and it % 10 == 0:
            try:
                s = a - prev_a
                y = gz - prev_grad
                sy = float(np.dot(s, y))
                yy = float(np.dot(y, y))
                if yy > 1e-20 and sy > 1e-15:
                    bb_lr = sy / yy
                    bb_lr = np.clip(bb_lr, 0.001, 0.5)
                    lr = 0.5 * base_lr + 0.5 * bb_lr
                else:
                    lr = base_lr
            except Exception:
                lr = base_lr
        else:
            lr = base_lr

        prev_a = a.copy()
        prev_grad = gz.copy()

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        a_new = np.exp(z - z.max())
        a_sum = a_new.sum()
        if a_sum < 1e-30:
            z = best_z.copy()
            a = np.exp(z - z.max())
            a = a / a.sum()
            continue
        a_new = a_new / a_sum

        c_new = np.convolve(a_new, a_new)
        max_c_new = float(c_new.max())
        score_new = 2.0 * n * max_c_new

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z.copy()
            best_M = max_c_new
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        # Temperature annealing
        temp = max(temp_min, temp * 0.995)

        # Learning rate decay on stagnation
        if no_imp > 500:
            base_lr = max(0.005, base_lr * 0.9)

        if no_imp > 3000:
            # Restart with perturbation
            rng_r = np.random.default_rng(it)
            z = 0.7 * best_z + 0.4 * rng_r.normal(0, 0.3, size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            base_lr = lr_init
            no_imp = 0
            temp = 0.1 * best_M
            a = np.exp(z - z.max())
            a = a / a.sum()
            best_score = 2.0 * n * float(np.convolve(a, a).max())

        if verbose and it % 500 == 1:
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"      {label} Adam iter {it}: score={best_score:.8f}, M={best_M:.6f}, "
                  f"temp={temp:.6e}, lr={base_lr:.4f}, no_imp={no_imp}")

    return best_a, best_score, best_M


def lp_iterate(f, max_iters=200, t_init=0.3, t_min=0.02, deadline=None, verbose=True, label=""):
    """
    LP-based iterations to reduce max(conv(f,f)).
    f must be normalized to sum 1.
    Returns (best_f, best_score, best_M).
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=20)
        if g_raw is None:
            if verbose:
                print(f"    {label} LP failed at iter {it}")
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step first
        g_self_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_self_score < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
            t_step = t_init
        else:
            # Blended step with line search
            best_blend_score = best_score
            best_blend_f = best_f
            for tt in [t_step, t_step * 0.5, t_step * 0.25, t_step * 0.1]:
                f_new = (1.0 - tt) * f + tt * g
                f_new = normalize(f_new)
                if f_new is None:
                    continue
                c_new = np.convolve(f_new, f_new)
                M_new = float(c_new.max())
                score_new = 2.0 * n * M_new
                if score_new < best_blend_score:
                    best_blend_score = score_new
                    best_blend_f = f_new.copy()
                    best_blend_M = M_new

            if best_blend_score < best_score - 1e-12:
                f = best_blend_f.copy()
                M = best_blend_M
                best_f = f.copy()
                best_score = best_blend_score
                best_M = best_blend_M
                no_improve = 0
                t_step = max(t_min, t_step * 0.98)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 10 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 100:
                    break

        if verbose and it % 50 == 0:
            print(f"    {label} LP iter {it}: score={best_score:.8f}, M={best_M:.6f}, t={t_step:.4f}")

    return best_f, best_score, best_M


def top_k_subgrad(a, w, k_top=5):
    """Compute subgradient using only top-k peaks for a sharper direction."""
    n = len(a)
    c = np.convolve(a, a)
    # Get top-k indices
    top_idx = np.argsort(c)[::-1][:k_top]
    # Create sparse weight
    w_sparse = np.zeros_like(c)
    top_vals = c[top_idx]
    # Normalize weights among top-k
    w_sparse[top_idx] = 1.0 / k_top
    return compute_subgrad(a, w_sparse)

        if a_big is not None:
            sc_big_init = internal_score(a_big)
            print(f"  n=1024 initial score: {sc_big_init:.8f}")
            t_big_dl = min(t_phase4_end, time.monotonic() + 200)
            M_init = float(np.convolve(a_big, a_big).max())
            a_opt, sc_opt, _ = adam_subgrad(a_big, t_big_dl, lr_init=0.04,
                                             temp_init=0.08 * M_init, temp_min=1e-5,
                                             verbose=True, label="P4-1024")
            if sc_opt < best_score_val:
                best_score_val = sc_opt
                best_seq = a_opt
                print(f"  Phase 4 (n=1024) result: score={best_score_val:.8f}")
            else:
                print(f"  Phase 4 (n=1024) not better: {sc_opt:.8f} vs {best_score_val:.8f}")

    # --- Phase 5: Final polish on best ---
    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 5:
        M_init = float(np.convolve(best_seq, best_seq).max())
        a_opt, sc_opt, _ = adam_subgrad(best_seq, deadline - 5, lr_init=0.02,
                                         temp_init=0.03 * M_init, temp_min=1e-7,
                                         verbose=True, label="Final")
        if sc_opt < best_score_val:
            best_score_val = sc_opt
            best_seq = a_opt
            print(f"  Final polish result: score={best_score_val:.8f}")

    # --- Final validation ---
    final_score = internal_score(best_seq)
    print(f"\n=== Final: n={len(best_seq)}, score={final_score:.8f} ===")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    best_seq = best_seq / best_seq.sum()

    # Validate with evaluate_sequence
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"  evaluate_sequence score: {eval_score:.8f}")
    except Exception as e:
        print(f"  Evaluation error: {e}")

    return [float(x) for x in best_seq]
