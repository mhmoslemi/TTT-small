    N = next_pow2(2 * n - 1)

"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return c
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def make_structured_profile(n, type_id, param, rng=None):
    t = np.linspace(0.0, 1.0, n)
    if type_id == 0:  # linear
        a = 1.0 + param * (t - 0.5)
    elif type_id == 1:  # quadratic U-shape
        u = 2.0 * t - 1.0
        a = 1.0 + param * u * u - 0.3 * param * u
    elif type_id == 2:  # exponential
        a = np.exp(param * (t - 0.5))
    elif type_id == 3:  # cubic
        u = 2.0 * t - 1.0
        a = 1.0 + param * u + param * 0.6 * u * u * u
    elif type_id == 4:  # quartic asymmetric
        u = 2.0 * t - 1.0
        a = 1.0 + param * u + 0.4 * param * u * u + 0.8 * param * u * u * u * u - 0.5 * param * u * u * u
    elif type_id == 5:  # cosine modulated
        a = 1.0 + param * np.cos(2.0 * np.pi * t) + 0.5 * param * (t - 0.5)
    elif type_id == 6:  # smooth random
        a = np.ones(n)
        for k in range(1, 8):
            amp = rng.uniform(0.05, 0.7) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t + phase)
    elif type_id == 7:  # two-hump asymmetric
        center = 0.3 + 0.4 * param / 3.0
        width = 0.2 + 0.1 * param
        a = np.exp(-((t - center) ** 2) / (2 * width * width)) + 0.5 * np.exp(-((t - (1.0 - center * 0.7)) ** 2) / (2 * (width * 1.3) ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def generate_diverse_starts(n, rng, parent=None):
    starts = []
    if parent is not None:
        if parent.size != n:
            x0 = np.linspace(0.0, 1.0, parent.size)
            x1 = np.linspace(0.0, 1.0, n)
            p_int = np.interp(x1, x0, parent)
            p_int = np.maximum(p_int, 1e-12)
            starts.append(p_int / p_int.sum())
        else:
            starts.append(parent.copy())
        # Mirror and blend
        p_mirror = parent[::-1].copy()
        starts.append(p_mirror / p_mirror.sum())
        for mix in (0.3, 0.7):
            blended = mix * parent + (1.0 - mix) * p_mirror
            blended = np.maximum(blended, 1e-12)
            starts.append(blended / blended.sum())

    # Structured profiles
    for tid in range(6):
        for param in (1.0, 1.5, 2.0, 2.5, 3.0):
            starts.append(make_structured_profile(n, tid, param, rng))
    # Two-hump
    for param in (1.0, 1.5, 2.0, 2.5):
        starts.append(make_structured_profile(n, 7, param, rng))
    # Random smooth
    for _ in range(10):
        starts.append(make_structured_profile(n, 6, 1.0, rng))
    return starts


def gradient_optimize(f0, deadline, rng, beta_init=0.12, beta_decay=0.97,
                      lr=0.08, min_beta_frac=0.003, restart_threshold=400):
    """Fast gradient optimization in log-space with correct gradient and adaptive beta."""
    f = np.asarray(f0, dtype=np.float64).copy()
    n = f.size
    f = np.maximum(f, 1e-15)
    f = f / f.sum()

    z = np.log(f)
    z -= z.mean()
    best_z = z.copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0

    best_M = float(compute_max_conv(f))
    best_f = f.copy()
    no_imp = 0
    it = 0

    N_corr = next_pow2(3 * n + 4)

    while time.monotonic() < deadline:
        c = fft_convolve(f)
        M = float(c.max())

        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Softmax weights
        x = (c - M) / beta_init if False else (c - M) / max(beta_init * (best_M / max(M, 1e-10)), 1e-10)
        # Actually use a running beta
        beta = max(min_beta_frac * best_M, beta_init * best_M * (beta_decay ** (it // 100)))
        x = (c - M) / max(beta, 1e-10)
        ex = np.exp(np.clip(x, -50.0, 4.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # h_i = sum_k w_k * a_{k-i} (cross-correlation)
        Fw = np.fft.rfft(w, N_corr)
        Fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(np.conj(Fw) * Fa, N_corr)[:n]

        # mu = E_w[c] = dot(w, c)
        mu = float(np.dot(w, c))

        # grad_z_i = 2 * a_i * (h_i - mu)
        grad_z = 2.0 * f * (h - mu)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_buf.fill(0)
            v_buf.fill(0)
            t_adam = 0
            it += 1
            continue

        t_adam += 1
        m_buf = 0.9 * m_buf + 0.1 * grad_z
        v_buf = 0.999 * v_buf + 0.001 * grad_z * grad_z
        mh = m_buf / (1.0 - 0.9 ** t_adam)
        vh = v_buf / (1.0 - 0.999 ** t_adam)

        step_size = lr
        z_new = z - step_size * mh / (np.sqrt(vh) + 1e-8)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= z_new.mean()

        f_new = np.exp(z_new - z_new.max())
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        c_new = fft_convolve(f_new)
        M_new = float(c_new.max())

        accepted = False
        if M_new <= M + 1e-12:
            z = z_new
            f = f_new
            accepted = True
        else:
            # Backtrack
            for shrink in (0.5, 0.25, 0.1):
                z_trial = z - step_size * shrink * mh / (np.sqrt(vh) + 1e-8)
                z_trial = np.clip(z_trial, -50.0, 50.0)
                z_trial -= z_trial.mean()
                f_trial = np.exp(z_trial - z_trial.max())
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                c_trial = fft_convolve(f_trial)
                M_trial = float(c_trial.max())
                if M_trial <= M + 1e-12:
                    z = z_trial
                    f = f_trial
                    step_size *= shrink
                    accepted = True
                    break

        if not accepted:
            pass  # no_imp already incremented

        # Restart logic
        if no_imp > restart_threshold:
            strength = min(1.0, 0.2 + 0.1 * (no_imp // restart_threshold))
            z = (1.0 - strength) * best_z + strength * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            m_buf.fill(0)
            v_buf.fill(0)
            t_adam = 0
            no_imp = 0

        it += 1

    return best_f, best_M


def lp_optimize(f0, deadline, rng, max_iter=50000, rebuild_interval=50):
    """LP-based iterative optimization with constraint matrix rebuild."""
    f = np.asarray(f0, dtype=np.float64).copy()
    n = f.size
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s

    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    M = float(compute_max_conv(f))
    best_f = f.copy()
    best_M = M
    no_imp = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        b = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8})

        if not res.success:
            no_imp += 1
            if no_imp > 40:
                break
            f = best_f + 0.02 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = float(compute_max_conv(f))
            if it % rebuild_interval == 0:
                for j in range(n):
                    A[j:j + n, j] = f
            it += 1
            continue

        g = res.x
        g_sum = float(g.sum())

        if g_sum <= 1.0 + 1e-5:
            no_imp += 1
            if no_imp > 200:
                break
            it += 1
            continue

        g = g / g_sum

        # Line search with more t values
        improved = False
        for t in (0.5, 0.3, 0.7, 0.2, 0.4, 0.8, 0.15, 0.6, 0.1, 0.9):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(compute_max_conv(f_new))
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if improved:
            if it % rebuild_interval == 0:
                for j in range(n):
                    A[j:j + n, j] = f
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                f = 0.8 * best_f + 0.2 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(compute_max_conv(f))
                for j in range(n):
                    A[j:j + n, j] = f
                no_imp = 0

        it += 1

    return best_f, best_M

    return p1 / s if s > 1e-30 else None

            best_seq = p_best.copy()
            best_score = sc
        print(f"After Phase 3: best={best_score:.10f}")

    # Also polish second-best
    if len(candidates) > 1 and time.monotonic() < phase3_end - 30:
        _, p_second = candidates[1]
        per = min(phase3_end, time.monotonic() + 60.0)
        p_best, M_best = gradient_optimize(p_second, per, rng,
                                           beta_init=0.04, beta_decay=0.985,
                                           lr=0.03, restart_threshold=400)
        sc = 2.0 * n1 * M_best
        if sc < best_score:
            best_seq = p_best.copy()
            best_score = sc

    print(f"After Phase 3 done: best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1720.0)
    if best_seq is not None and time.monotonic() < phase4_end - 80:
        p_ext = interpolate_to_length(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 70.0)
            p_best, M_best = gradient_optimize(p_ext, per, rng,
                                               beta_init=0.1, beta_decay=0.97,
                                               lr=0.06, restart_threshold=300)
            sc = 2.0 * n2 * M_best
            if sc < best_score:
                best_seq = p_best.copy()
                best_score = sc
            print(f"Phase 4 (n={n2}): score={sc:.10f}, best={best_score:.10f}")

    # Phase 5: Final short LP on absolute best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_b = best_seq.size
        per = min(deadline - 10, time.monotonic() + 50.0)
        p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=5000)
        sc = compute_score(p_best)
        if sc < best_score:
            best_seq = p_best.copy()
            best_score = sc
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # Final
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal: score={sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated: {sc_eval:.10f}")
            except Exception as e:
                print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    t = np.linspace(0, 1, 512)
    fb = np.maximum(1.0 + 1.5 * (t - 0.5), 1e-12)
    return [float(x) for x in fb / fb.sum()]
