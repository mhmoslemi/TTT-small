LP-based and coordinate descent optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Linear Programming to find improvement directions, combined with multi-start
initialization and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Remove tiny negative values from FFT roundoff
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # A[k, j] = f[k-j] for k = j, j+1, ..., min(j+n-1, 2n-2)
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = f[0:k_end - j]
    return A


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M*1."""
    A = build_lp_matrix(f, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def greedy_peak_reduction(f0, deadline, max_rounds=500, verbose=True):
    """Greedy peak reduction: reduce the elements that contribute most to the peak."""
    f = normalize(f0)
    if f is None:
        return f0, internal_score(f0)
    n = f.size
    best_f = f.copy()
    best_score = internal_score(f)
    rng = np.random.default_rng(999)
    
    if verbose:
        print(f"  GR start: score={best_score:.10f}")
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve_fft(f)
        k_star = int(np.argmax(c))
        
        # Find the pair contributing most to position k_star
        # (f*f)(k_star) = sum_i f[i] * f[k_star - i]
        valid_i = range(max(0, k_star - n + 1), min(n, k_star + 1))
        contributions = np.array([f[i] * f[k_star - i] for i in valid_i])
        i_best = valid_i[np.argmax(contributions)]
        j_best = k_star - i_best
        
        # Reduce f[i_best] and f[j_best]
        if i_best == j_best:
            # Only one element to reduce
            delta = 0.1 * f[i_best]
            f[i_best] -= delta
            # Redistribute to a random position with low contribution
            # Find position with minimum f
            redist_idx = int(np.argmin(f))
            f[redist_idx] += delta
            f = np.maximum(f, 0)
            f = f / f.sum()
        else:
            delta = 0.05 * min(f[i_best], f[j_best])
            f[i_best] -= delta
            f[j_best] -= delta
            # Redistribute
            # Find two positions with low local convolution
            redist_idx1 = int(np.argmin(f))
            redist_idx2 = int(np.argmin(f))
            while redist_idx2 == redist_idx1:
                redist_idx2 = int(np.argmin(f))
            f[redist_idx1] += delta
            f[redist_idx2] += delta
            f = np.maximum(f, 0)
            f = f / f.sum()
        
        sc = internal_score(f)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f.copy()
        else:
            # Revert
            f = best_f.copy()
            # Try a smaller reduction
            break
    
    if verbose:
        print(f"  GR end: score={best_score:.10f}")
    
    return best_f, best_score

"""
LP-based and coordinate descent optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Linear Programming to find improvement directions (maximizing sum of g subject to
cross-convolution constraint), combined with multi-start initialization, greedy peak
reduction, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0.0:
        c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s


def build_lp_matrix(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = f[0:k_end - j]
    return A


def solve_lp_for_g(f, M, n):
    A = build_lp_matrix(f, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-15:
            return g
    return None


def lp_optimize(f0, deadline, max_iters=200, t_init=0.15, verbose=True):
    f = normalize_seq(f0)
    if f is None:
        return f0, internal_score(f0)
    n = f.size
    best_f = f.copy()
    best_score = internal_score(f)
    current_t = t_init
    no_imp = 0

    if verbose:
        print(f"    LP start: score={best_score:.10f}, n={n}")

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = full_convolve_fft(f)
        M = float(c.max())
        g = solve_lp_for_g(f, M, n)
        if g is None:
            break
        g = g / g.sum()
        f_new = (1.0 - current_t) * f + current_t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            current_t *= 0.5
            continue
        f_new = f_new / s_new
        sc = internal_score(f_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            no_imp = 0
            current_t = min(0.35, current_t * 1.03)
        else:
            no_imp += 1
            current_t = max(0.03, current_t * 0.85)
            if no_imp > 30:
                break
        if verbose and it % 30 == 0:
            print(f"      iter {it}: score={best_score:.10f}, t={current_t:.4f}")

    if verbose:
        print(f"    LP end: score={best_score:.10f}, iters={it+1}")
    return best_f, best_score


def coordinate_descent(f0, deadline, max_sweeps=100, verbose=True):
    f = normalize_seq(f0)
    if f is None:
        return f0, internal_score(f0)
    n = f.size
    best_f = f.copy()
    best_score = internal_score(f)
    rng = np.random.default_rng(777)
    no_imp_sweeps = 0

    if verbose:
        print(f"    CD start: score={best_score:.10f}")

    factors = [1.4, 1.25, 1.1, 1.05, 0.95, 0.9, 0.8, 0.6]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_val = orig
            best_sc = best_score
            for factor in factors:
                new_val = orig * factor
                if new_val < 0:
                    continue
                f[i] = new_val
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                f = f / s
                sc = internal_score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_val = new_val
                    improved = True
                f[i] = orig
                s = f.sum()
                if s > 0:
                    f = f / s
            if best_val != orig:
                f[i] = best_val
                f = f / f.sum()
                if best_sc < best_score:
                    best_score = best_sc
                    best_f = f.copy()
        if not improved:
            no_imp_sweeps += 1
            if no_imp_sweeps >= 3:
                break
        else:
            no_imp_sweeps = 0

    if verbose:
        print(f"    CD end: score={best_score:.10f}, sweeps={sweep+1}")
    return best_f, best_score
