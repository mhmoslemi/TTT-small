def next_pow2(n):

def fft_convolve(a):

    n = a.size
    if n == 0:

    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    """Compute the evaluation score for a sequence."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def compute_gradient_direct(a, beta):
    """
    Compute gradient of log-sum-exp smoothing of max convolution w.r.t. a.
    L = beta * log(sum_k exp((conv[k] - M)/beta)) + M
    dL/da[j] = 2 * sum_m w[m+j] * a[m]
    """
    n = a.size
    c = fft_convolve(a)
    M = float(c.max())

    shifted = (c - M) / beta
    exp_c = np.exp(np.clip(shifted, -100, 50))
    w = exp_c / exp_c.sum()

    N = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    result = np.fft.irfft(A * np.conj(W), N)

    h = np.zeros(n)
    h[0] = result[0]
    if n > 1:
        h[1:] = result[N - np.arange(1, n)]

    grad = 2.0 * h
    return grad, c, M


def compute_gradient_logspace(a, beta):
    """
    Compute gradient in log-space (z where a = softmax(z)).
    dz_k = a_k * (grad_k - sum_j a_j * grad_j)
    """
    grad, c, M = compute_gradient_direct(a, beta)
    a_grad_mean = float(np.dot(a, grad))
    grad_z = a * (grad - a_grad_mean)
    return grad_z, c, M


def softmax_optimize_direct(a0, deadline, rng, lr=0.1, beta_init=None,
                            max_iter=100000, verbose=False):
    """
    Optimize using softmax-smoothed gradient descent (direct space) with Adam.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a = a / s
    n = a.size

    c = fft_convolve(a)
    M = float(c.max())
    if beta_init is None:
        beta = 0.08 * M

    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    no_imp = 0
    restart_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, c, M_cur = compute_gradient_direct(a, beta)

        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad * grad
        mh = m_adam / (1 - b1 ** t_adam)
        vh = v_adam / (1 - b2 ** t_adam)

        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new / s_new

        c_new = fft_convolve(a_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            best_score = 2.0 * n * best_M
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        # Adaptive beta annealing
        if it % 400 == 0 and it > 0:
            beta = max(1e-7, beta * 0.8)

        # Learning rate decay
        if it % 2000 == 0 and it > 0:
            lr = max(0.005, lr * 0.85)

        # Restart if stuck
        if no_imp > 300:
            restart_count += 1
            no_imp = 300
            # Various restart strategies
            strategy = restart_count % 4
            if strategy == 0:
                pert = np.abs(rng.normal(size=n)) * 0.15
                a = 0.6 * best_a + 0.4 * pert
            elif strategy == 1:
                # Random phase perturbation
                t = np.linspace(0, 1, n)
                phase = rng.uniform(0, 2 * np.pi)
                pert = np.cos(2 * np.pi * rng.integers(1, 8) * t + phase)
                a = best_a + 0.1 * pert
            elif strategy == 2:
                # Swap-based perturbation
                a = best_a.copy()
                for _ in range(n // 10):
                    i, j = rng.integers(0, n, 2)
                    a[i], a[j] = a[j], a[i]
            else:
                # Flatten and re-peak
                a = 0.8 * best_a + 0.2 * np.ones(n) / n

            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = fft_convolve(a)
            M = float(c.max())
            beta = max(1e-7, 0.08 * M)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            lr = 0.1
            no_imp = 0

            if verbose:
                print(f"      restart #{restart_count}, M={M:.10f}, best={best_M:.10f}")

        if verbose and it % 2000 == 0:
            print(f"    iter {it}: M={M_new:.12f}, best={best_M:.12f}, beta={beta:.2e}")

    return best_a, best_score


def softmax_optimize_logspace(a0, deadline, rng, lr=0.5, beta_init=None,
                              max_iter=100000, verbose=False):
    """
    Optimize using softmax-smoothed gradient descent in log-space with Adam.
    a = softmax(z), optimize over z.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size

    z = np.log(a)
    z -= z.max()

    c = fft_convolve(a)
    M = float(c.max())
    if beta_init is None:
        beta = 0.08 * M

    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M
    best_z = z.copy()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    no_imp = 0
    restart_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad_z, c, M_cur = compute_gradient_logspace(a, beta)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            a = best_a.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1 - b1) * grad_z
        v_adam = b2 * v_adam + (1 - b2) * grad_z * grad_z
        mh = m_adam / (1 - b1 ** t_adam)
        vh = v_adam / (1 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z -= z.max()

        a = np.exp(z)
        a = a / a.sum()

        c_new = fft_convolve(a)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            best_z = z.copy()
            best_score = 2.0 * n * best_M
            no_imp = 0
        else:
            no_imp += 1

        if it % 400 == 0 and it > 0:
            beta = max(1e-7, beta * 0.8)

        if it % 2000 == 0 and it > 0:
            lr = max(0.02, lr * 0.85)

        if no_imp > 400:
            restart_count += 1
            no_imp = 400
            strategy = restart_count % 3
            if strategy == 0:
                z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            elif strategy == 1:
                z = best_z + 0.2 * rng.normal(size=n)
            else:
                # Inject a sinusoidal perturbation
                t = np.linspace(0, 1, n)
                freq = rng.integers(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                z = best_z + 0.3 * np.cos(2 * np.pi * freq * t + phase)

            z = np.clip(z, -20, 20)
            z -= z.max()
            a = np.exp(z)
            a = a / a.sum()
            c = fft_convolve(a)
            M = float(c.max())
            beta = max(1e-7, 0.08 * M)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            lr = 0.5
            no_imp = 0

            if verbose:
                print(f"      restart #{restart_count}, M={M:.10f}, best={best_M:.10f}")

        if verbose and it % 2000 == 0:
            print(f"    logspace iter {it}: M={M_new:.12f}, best={best_M:.12f}, beta={beta:.2e}")

    return best_a, best_score


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, c):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_oscillating_profile(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bimodal_profile(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 3.0 * ((t - 0.5) / max(width, 0.05)) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=None):
    if n_harm is None:
        n_harm = max(3, n // 64)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.1, 0.8) / (k ** 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def local_search(a0, deadline, rng, n_trials=10000):
    """
    Simple local search: randomly perturb small groups of elements.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_M = float(fft_convolve(a).max())
    best_score = 2.0 * n * best_M

    c = fft_convolve(a)
    M = float(c.max())

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Pick a random subset of elements to perturb
        k = rng.integers(1, max(2, n // 50) + 1)
        idx = rng.integers(0, n, k)
        deltas = rng.normal(0, 0.05, k)

        a_new = a.copy()
        a_new[idx] = np.maximum(a_new[idx] + deltas, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = float(fft_convolve(a_new).max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            best_score = 2.0 * n * best_M
            a = a_new
            M = M_new
        elif M_new < M:
            # Accept with some probability (simulated annealing style)
            a = a_new
            M = M_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search:
    1. Quick optimization of diverse starts at n=512.
    2. Extended optimization (both direct and log-space) on best candidates.
    3. Interpolation to n=1024 with further refinement.
    4. Local search polishing.
    5. Final evaluation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent sequence
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
                sc = compute_score(parent)
                try_update(parent, sc)
                print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception:
        pass

    # Phase 1: Quick optimization at n=512 from diverse starts
    n1 = 512
    phase1_end = t_start + 0.35 * (deadline - t_start)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)

    # Deterministic profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        starts.append(make_linear_profile(n1, eta))
    for c in (1.0, 2.0, 3.0, 5.0, 8.0):
        starts.append(make_quadratic_profile(n1, c))
    for freq, amp in ((1, 0.3), (2, 0.4), (3, 0.5), (4, 0.3), (5, 0.6), (6, 0.4)):
        starts.append(make_oscillating_profile(n1, freq, amp))
    for w in (0.15, 0.25, 0.35, 0.5):
        starts.append(make_bimodal_profile(n1, w))

    # Random starts
    for _ in range(10):
        starts.append(make_smooth_random(n1, rng, n_harm=6))

    # Extra: "plateau" profiles
    t = np.linspace(0, 1, n1)
    for frac in (0.3, 0.4, 0.5):
        a = np.where(np.abs(t - 0.5) < frac, 2.0, 1.0)
        a = a / a.sum()
        starts.append(a)

    candidates = []
    per_start_time = max(1.5, (phase1_end - t_start) / len(starts))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 1e-15)
        p0 = p0 / p0.sum()

        per = min(phase1_end, time.monotonic() + per_start_time)
        p_best, sc = softmax_optimize_direct(p0, per, rng, lr=0.1, max_iter=3000)
        try_update(p_best, sc)
        candidates.append((sc, p_best.copy()))

        if i % 10 == 0:
            print(f"  Phase 1 start {i}/{len(starts)}: best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Phase 2: Extended optimization on top candidates
    phase2_end = t_start + 0.70 * (deadline - t_start)
    top_k = min(4, len(candidates))
    per_cand = max(5.0, (phase2_end - t_start) / top_k)

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc_i, p_i = candidates[i]
        remaining = phase2_end - time.monotonic()
        if remaining < 5:
            break

        # Direct space optimization
        per1 = min(phase2_end, time.monotonic() + remaining * 0.4)
        p_best, sc = softmax_optimize_direct(p_i, per1, rng, lr=0.08, max_iter=100000)
        try_update(p_best, sc)
        print(f"  Phase 2 cand {i} (direct): score={sc:.10f}")

        # Log-space optimization
        if time.monotonic() < phase2_end - 10:
            per2 = min(phase2_end, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
            p_best2, sc2 = softmax_optimize_logspace(p_best, per2, rng, lr=0.4, max_iter=50000)
            try_update(p_best2, sc2)
            print(f"  Phase 2 cand {i} (logspace): score={sc2:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Try n=1024
    phase3_end = t_start + 0.88 * (deadline - t_start)
    n2 = 1024

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase3_end, time.monotonic() + (phase3_end - time.monotonic()) * 0.6)
            p_best, sc = softmax_optimize_direct(p_ext, per, rng, lr=0.06, max_iter=50000)
            try_update(p_best, sc)
            print(f"Phase 3 (n={n2}, direct): score={sc:.10f}")

            if time.monotonic() < phase3_end - 15:
                per2 = min(phase3_end, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
                p_best2, sc2 = softmax_optimize_logspace(p_best, per2, rng, lr=0.3, max_iter=30000)
                try_update(p_best2, sc2)
                print(f"Phase 3 (n={n2}, logspace): score={sc2:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Local search polishing
    phase4_end = deadline - 10
    if best_seq is not None and time.monotonic() < phase4_end - 20:
        per = min(phase4_end - 5, time.monotonic() + 60.0)
        p_best, sc = local_search(best_seq, per, rng, n_trials=50000)
        try_update(p_best, sc)
        print(f"After Phase 4 local search: best={best_score:.10f}")

        # One more gradient polish after local search
        if time.monotonic() < deadline - 10:
            per2 = min(deadline - 5, time.monotonic() + 30.0)
            p_best2, sc2 = softmax_optimize_direct(best_seq, per2, rng, lr=0.02, max_iter=20000)
            try_update(p_best2, sc2)
            print(f"After Phase 4 final polish: best={best_score:.10f}")

    # Final verification
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
