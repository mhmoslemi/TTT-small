def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _lp_refine_min_t(a, deadline, max_iter=5000, blend=0.5):
    """LP refinement using min-t formulation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        b, t_val = _lp_min_t(a, time_limit=10.0)
        if b is None:
            break

        # Check if using b directly improves
        c_b = _fft_convolve(b)
        score_b = float(2.0 * n * c_b.max())
        
        # Blend
        a_new = (1.0 - blend_cur) * a + blend_cur * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        # Also try using b directly
        if score_b < score_new:
            a_new = b.copy()
            score_new = score_b

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.1, blend_cur * 0.85)
            if no_imp > 200:
                a = best_a.copy()
                blend_cur = blend
                no_imp = 0
                # Try a different blend
                blend = min(0.9, blend * 1.2) if blend < 0.8 else 0.3

        a = a_new

        if it % 50 == 0:
            print(f"    LP-min-t iter {it}: score={best_score:.10f}, t={t_val:.8f}")

    return best_a


def _spectral_shape(a, deadline, max_iter=500):
    """Spectral-domain shaping: try to flatten conv(a,a) via frequency domain operations."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    
    # We want to modify a in the frequency domain to flatten conv(a,a)
    # conv(a,a) in freq domain is A(f)^2
    # Strategy: iteratively adjust A(f) to make A(f)^2 flatter in time domain
    
    alpha = 0.5  # blending parameter
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = c.max()
        c_mean = c.mean()
        
        # Compute "error" - how far from flat
        error = c - c_mean
        
        # In frequency domain, the "gradient" of max(conv) w.r.t. the spectrum
        # is related to the phase of the spectrum at the peak frequency
        # Simple approach: reduce the spectrum at frequencies that contribute to peaks
        
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        
        # The contribution of frequency f to the peak at position p is proportional to
        # |A(f)|^2 * exp(-2j*pi*f*p/N) (roughly). We want to reduce |A(f)| where it
        # contributes to peaks.
        
        # Simpler approach: compute the "peak location" and use phase to identify
        # which frequencies contribute
        peak_pos = int(np.argmax(c))
        
        # Modulate the error signal to get frequency-domain corrections
        error_padded = np.zeros(N)
        error_padded[:2*n-1] = error
        E = np.fft.rfft(error_padded)
        
        # Correction: reduce A where it aligns with the error
        # Using a "matched filter" approach in frequency domain
        phase_align = np.exp(-2j * np.pi * np.arange(len(A)) * peak_pos / N)
        correction = E * phase_align
        correction_val = np.fft.irfft(correction, N)[:n]
        
        # Apply correction
        a_new = a - 0.01 * correction_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            score_new = float(2.0 * n * _fft_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = alpha * a + (1 - alpha) * a_new
        else:
            break
        
        if it % 100 == 0:
            print(f"    Spectral iter {it}: score={best_score:.10f}")

    return best_a


def _gradient_polish(a, deadline, max_iter=20000):
    """Gradient descent with Nesterov momentum and adaptive step size."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    lr = 0.05
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov look-ahead
        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest > 1e-10:
            a_nest = a_nest / s_nest

        # Compute gradient using the correct formula: grad_j = 2 * sum_i w_i * a[i-j]
        n_val = a_nest.size
        c_vals = _fft_convolve(a_nest)
        w = _smooth_max_weight(c_vals, beta)
        
        # h[j] = sum_i w[i] * a[i-j] for j=0,...,n-1
        h = np.zeros(n_val)
        for j in range(n_val):
            # h[j] = sum_{m=0}^{n_val-1-j} a_nest[m] * w[m+j]
            limit = min(n_val - j, len(w))
            if limit > 0:
                h[j] = np.dot(a_nest[:limit], w[:limit])
        
        grad = 2.0 * h
        # Add small regularization to keep interior
        grad += 1e-8 * a_nest

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a
