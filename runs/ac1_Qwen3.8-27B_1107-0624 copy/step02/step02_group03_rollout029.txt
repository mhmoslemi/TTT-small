    """
    n = len(f)
    if n > 1500:
        return None
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_minimize_peak(f, timeout=10.0):
    """
    Alternative LP: minimize t subject to convolve(f,g)[x] <= t,
    sum(g)=1, g>=0. Returns (g, t) or (None, inf).
    """
    n = len(f)
    if n > 1500:
        return None, np.inf
    A = build_conv_matrix(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    # Constraints: A @ g <= t  =>  A @ g - t <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # sum(g) = 1  =>  sum(g) <= 1 and sum(g) >= 1
    b_ub_ext = np.zeros((2 * n - 1, 1))
    # We'll use bounds for g >= 0 and add equality constraint via two inequalities
    bounds = [(0.0, None)] * n + [(None, None)]
    # Add sum(g) = 1 as: sum(g) <= 1 and -sum(g) <= -1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x[:n], 0.0)
        t = res.x[-1]
        if g.sum() > 1e-10:
            g = g / g.sum()
            return g, float(t)
    return None, np.inf


def lp_iterate(f, max_iters=500, t_init=0.4, t_min=0.02, deadline=None,
               verbose=True, lp_timeout=10, use_min_peak=False):
    """LP-based iterations to reduce max(conv(f,f))."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init
    it = 0

    while it < max_iters:
        if deadline is not None and time.monotonic() > deadline:
            break
        it += 1

        if use_min_peak:
            g_raw, M_lp = lp_minimize_peak(f, timeout=lp_timeout)
        else:
            g_raw = lp_direction(f, M, timeout=lp_timeout)
            M_lp = M

        if g_raw is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step
        score_g = 2.0 * n * float(np.convolve(g, g).max())

        if score_g < best_score * 0.998:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            # Blended step with line search
            best_blend_score = best_score
            best_blend_f = None
            for tt in [t_step, 0.5 * t_step, 0.25 * t_step, 0.1 * t_step]:
                f_new = (1.0 - tt) * f + tt * g
                f_new = normalize(f_new)
                if f_new is None:
                    continue
                sc = 2.0 * n * float(np.convolve(f_new, f_new).max())
                if sc < best_blend_score - 1e-14:
                    best_blend_score = sc
                    best_blend_f = f_new

            if best_blend_f is not None:
                f = best_blend_f.copy()
                M = float(np.convolve(f, f).max())
                best_f = f.copy()
                best_score = best_blend_score
                best_M = M
                no_improve = 0
                t_step = max(t_min, t_step * 0.99)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 10 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 80:
                    if verbose:
                        print(f"    LP stalled at iter {it}, score={best_score:.8f}")
                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={best_M:.6f}, "
                  f"t={t_step:.4f}, no_imp={no_improve}")

    return best_f, best_score


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1."""
    n = len(p)
    if n <= 600:
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.dot(w[i:i + n], p)
        return h
    else:
        # FFT-based for larger n
        size = 4 * n
        W = np.fft.rfft(w, size)
        P = np.fft.rfft(p, size)
        # Cross-correlation: h[i] = sum_j w[i+j]*p[j]
        # = (w * flip(p))[i + n - 1] in convolution terms
        # Let's use: ifft(conj(fft(w)) * fft(p))[::-1] or similar
        # Actually: sum_j w[i+j]*p[j] = cross_corr(w, p) at lag i
        # cross_corr(a, b)[k] = sum_j a[j+k] * conj(b[j])
        # Using FFT: cross_corr = ifft(fft(a, N) * conj(fft(b, N)))
        N = size
        X = np.fft.irfft(np.fft.rfft(w, N) * np.conj(np.fft.rfft(p, N)), N)
        # X[k] = sum_j w[j] * p[j-k] (circular)
        # We want h[i] = sum_j w[i+j] * p[j]
        # Let's just use direct for correctness
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.dot(w[i:i + n], p)
        return h


def adam_polish(p0, deadline, lr=0.04, reg=1e-10, verbose=True,
                lp_every=5000, lp_timeout=5, beta_init_frac=0.1):
    """
    Adam gradient descent in logit space to minimize 2*n*max(conv(p,p)).
    Periodically uses LP to find new directions.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    step_count = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-8, beta_init_frac * max_c)

    while time.monotonic() < deadline:
        it += 1
        step_count += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft-max weights with adaptive temperature
        x = (c - max_c) / beta
        x = np.clip(x, -80.0, 0.0)
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient computation
        h = corr_weights(w, p0)
        grad_p = 2.0 * h + reg * p0

        # Logit-space gradient (projection to tangent space of simplex)
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            p0 = best_p.copy()
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        # Adaptive learning rate
        grad_norm = float(np.sqrt(np.sum(mh * mh)))
        effective_lr = lr / (1.0 + 2.0 * grad_norm)

        z = z - effective_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30 or not np.all(np.isfinite(p)):
            z = best_z.copy()
            p0 = best_p.copy()
            continue
        p = p / p_sum

        score = 2.0 * n * float(np.convolve(p, p).max())

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.003, lr * 0.95)

        # Periodic beta annealing
        if step_count % 5000 == 0:
            beta = max(1e-10, beta * 0.5)

        # Periodic LP direction finding
        if lp_every > 0 and step_count % lp_every == 0 and n <= 600:
            g_raw = lp_direction(best_p, best_score / (2.0 * n), timeout=lp_timeout)
            if g_raw is not None:
                g = normalize(g_raw)
                if g is not None:
                    # Try blending
                    for blend in [0.5, 0.3, 0.7]:
                        p_blend = (1.0 - blend) * best_p + blend * g
                        p_blend = normalize(p_blend)
                        if p_blend is not None:
                            sc_blend = 2.0 * n * float(np.convolve(p_blend, p_blend).max())
                            if sc_blend < best_score:
                                best_score = sc_blend
                                best_p = p_blend.copy()
                                z = np.log(np.maximum(best_p, 1e-15))
                                z -= np.mean(z)
                                best_z = z.copy()
                                p0 = best_p.copy()
                                m.fill(0)
                                v.fill(0)
                                it = 0
                                no_imp = 0
                                beta = max(1e-8, 0.1 * float(np.convolve(best_p, best_p).max()))
                                break

        # Restart with perturbation if stuck
        if no_imp > 8000:
            rng = np.random.default_rng(step_count)
            scale = 0.3 + 0.3 * np.sqrt(no_imp / 8000.0)
            perturbation = scale * rng.normal(size=n)
            z = best_z + perturbation
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-8, 0.1 * best_score / (2.0 * n))

        p0 = p

    if verbose:
        print(f"    Adam finished: steps={step_count}, score={best_score:.8f}")

    return best_p, best_score


def coordinate_refine(p0, max_passes=10, deadline=None, verbose=False):
    """
    Coordinate descent refinement: for each pair of coordinates, try
    transferring mass and check if score improves.
    """
    n = len(p0)
    p = p0.astype(np.float64).copy()
    best_score = internal_score(p)
    best_p = p.copy()

    for path in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            if p[i] < 1e-10:
                continue
            # Try transferring some mass from i to j
            for j in range(n):
                if j == i:
                    continue
                for frac in [0.01, 0.005, 0.02]:
                    delta = frac * p[i]
                    if delta > p[i]:
                        continue
                    p_new = p.copy()
                    p_new[i] -= delta
                    p_new[j] += delta
                    sc = internal_score(p_new)
                    if sc < best_score - 1e-14:
                        p = p_new
                        best_score = sc
                        best_p = p.copy()
                        improved = True
                        break
                if improved:
                    break
        if verbose:
            print(f"    Coord pass {path}: score={best_score:.8f}, improved={improved}")
        if not improved:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution LP + Adam search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
      1. Generate diverse initial profiles, select best.
      2. LP refinement at n=128 (fast, many iterations).
      3. LP refinement at n=256.
      4. LP refinement at n=512.
      5. Adam polish at n=512 with periodic LP direction finding.
      6. Coordinate descent refinement.
      7. Optional: upsample to n=1024 and Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # --- Get parent sequence ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent.copy()
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    # --- Generate diverse initial profiles ---
    n_init = 128
    candidates = []

    if best_seq is not None:
        p_par = interpolate_seq(best_seq, n_init)
        if p_par is not None:
            candidates.append(p_par)

    profile_types = ['linear', 'quadratic', 'sinusoidal', 'cosine', 'poly2', 'poly3']
    params = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 2.5, 3.0]
    for pt in profile_types:
        for pm in params:
            candidates.append(make_profile(n_init, pt, pm))

    # Random smooth profiles
    for _ in range(15):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.2, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.linspace(0, 1, n_init)
        a = 1.0 + amp * np.sin(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())

    # Evaluate all candidates
    for c in candidates:
        sc = internal_score(c)
        if sc < best_score:
            best_score = sc
            best_seq = c.copy()

    print(f"After init: n={len(best_seq)}, score={best_score:.8f}")

    # --- Phase 1: LP at n=128 ---
    deadline_p1 = min(deadline, t_start + 250.0)
    a1 = interpolate_seq(best_seq, 128)
    if a1 is not None:
        sc1 = internal_score(a1)
        print(f"\nPhase 1: LP at n=128, initial score={sc1:.8f}")
        f_best, sc_best = lp_iterate(a1, max_iters=3000, t_init=0.4, t_min=0.01,
                                      deadline=deadline_p1, verbose=True, lp_timeout=5)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  Phase 1 result: score={best_score:.8f}")

    # --- Phase 2: LP at n=256 ---
    deadline_p2 = min(deadline, t_start + 650.0)
    if time.monotonic() < deadline_p2 - 30:
        a2 = interpolate_seq(best_seq, 256)
        if a2 is not None:
            sc2 = internal_score(a2)
            print(f"\nPhase 2: LP at n=256, initial score={sc2:.8f}")
            f_best, sc_best = lp_iterate(a2, max_iters=1000, t_init=0.35, t_min=0.01,
                                          deadline=deadline_p2, verbose=True, lp_timeout=10)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 2 result: score={best_score:.8f}")

    # --- Phase 3: LP at n=512 ---
    deadline_p3 = min(deadline, t_start + 1100.0)
    if time.monotonic() < deadline_p3 - 30:
        a3 = interpolate_seq(best_seq, 512)
        if a3 is not None:
            sc3 = internal_score(a3)
            print(f"\nPhase 3: LP at n=512, initial score={sc3:.8f}")
            f_best, sc_best = lp_iterate(a3, max_iters=200, t_init=0.3, t_min=0.01,
                                          deadline=deadline_p3, verbose=True, lp_timeout=15)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 3 result: score={best_score:.8f}")

    # --- Phase 4: Adam polish at n=512 with LP ---
    deadline_p4 = min(deadline, t_start + 1500.0)
    if time.monotonic() < deadline_p4 - 10:
        n4 = len(best_seq)
        if n4 != 512:
            best_seq = interpolate_seq(best_seq, 512)
        print(f"\nPhase 4: Adam at n=512, initial score={internal_score(best_seq):.8f}")
        p_best, sc_best = adam_polish(best_seq, deadline_p4, lr=0.05, reg=1e-10,
                                       verbose=True, lp_every=4000, lp_timeout=5)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 4 result: score={best_score:.8f}")

    # --- Phase 5: Coordinate refinement ---
    deadline_p5 = min(deadline, t_start + 1650.0)
    if time.monotonic() < deadline_p5 - 5 and len(best_seq) <= 512:
        print(f"\nPhase 5: Coordinate refinement")
        p_ref, sc_ref = coordinate_refine(best_seq, max_passes=5, deadline=deadline_p5,
                                           verbose=True)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = p_ref
            print(f"  Phase 5 result: score={best_score:.8f}")

    # --- Phase 6: Try n=1024 ---
    deadline_p6 = min(deadline, t_start + budget_s - 30.0)
    if time.monotonic() < deadline_p6 - 10 and best_score > 1.503:
        a6 = interpolate_seq(best_seq, 1024)
        if a6 is not None:
            sc6 = internal_score(a6)
            print(f"\nPhase 6: Try n=1024, score after interp={sc6:.8f}")
            if sc6 < best_score + 0.005:
                p_best, sc_best = adam_polish(a6, deadline_p6, lr=0.03, reg=1e-10,
                                               verbose=True, lp_every=0, lp_timeout=0)
                if sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best
                    print(f"  Phase 6 result: score={best_score:.8f}")

    # --- Final: one more Adam pass at best resolution ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_final = len(best_seq)
        print(f"\nFinal polish: n={n_final}, score={internal_score(best_seq):.8f}, "
              f"remaining={remaining:.0f}s")
        p_best, sc_best = adam_polish(best_seq, deadline, lr=0.03, reg=1e-10,
                                       verbose=True, lp_every=0, lp_timeout=0)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Final result: score={best_score:.8f}")

    # --- Final validation ---
    final_score = internal_score(best_seq)
    print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.8f} ===")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_profile(256, 'linear', 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
