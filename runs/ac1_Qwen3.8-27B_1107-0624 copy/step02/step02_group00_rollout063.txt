```python
def _lp_min_peak(fixed, n):
    """
    Find g >= 0, sum(g) = 1, minimizing max_k conv(fixed, g)[k].
    Returns (g, t_opt).
    """
    # Variables: g[0], ..., g[n-1], t
    # Total: n+1 variables
    # 
    # Objective: minimize t -> c = [0, ..., 0, 1]
    # 
    # Inequality constraints: conv(fixed, g)[k] - t <= 0 for k = 0, ..., 2n-2
    #   Row k of A_ub: [fixed conv row k, -1]
    #
    # Equality: sum(g) = 1 -> A_eq: [1, ..., 1, 0], b_eq: [1]
    #
    # Bounds: g[i] >= 0, t >= 0 (or t free? t should be >= 0 since conv >= 0)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality: A_ub @ x <= b_ub
    # For each k: sum_j A[k,j] * g[j] - t <= 0
    A_conv = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # with coefficient fixed[k-j]
        A_conv[j:j+n, j] = fixed
    
    # A_ub has shape (2n-1, n+1): [A_conv, -1]
    A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g[i] >= 0, t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        t = res.x[-1]
        return g, t
```

```python
"""
Two-sided LP + Entropic Gradient for minimizing 2n*max(conv(a,a))/sum(a)^2.

Algorithm:
1. Two-sided alternating LP: alternate fixing one side and optimizing the other
   to minimize the peak of their cross-convolution.
2. Entropic (geometric) blending for stable updates.
3. Smooth-max mirror descent for final polish.
4. Multiple sequence lengths explored.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _make_conv_matrix(fixed, n):
    """Build matrix A such that A @ g = conv(fixed, g)."""
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = fixed
    return A

def _lp_min_peak(fixed, n):
    """
    Find g >= 0, sum(g) = 1, minimizing max_k conv(fixed, g)[k].
    Returns (g, t_opt) or (None, None) if infeasible.
    """
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_conv = _make_conv_matrix(fixed, n)
    
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        g = g / g.sum()
        t = float(res.x[-1])
        return g, t
    return None, None

def _entropic_blend(a, b, t, eps=1e-15):
    """Geometric/entropic blend: exp((1-t)*log(a) + t*log(b)), normalized."""
    a = np.maximum(a, eps)
    b = np.maximum(b, eps)
    mixed = np.exp((1.0 - t) * np.log(a) + t * np.log(b))
    return mixed / mixed.sum()

def _linear_blend(a, b, t):
    mixed = (1.0 - t) * a + t * b
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s < 1e-30:
        return a
    return mixed / s

def _two_sided_lp(a_init, deadline, max_iters=5000, blend_rate=0.5):
    """
    Two-sided alternating LP optimization.
    Alternates: fix b, optimize a; fix a, optimize b.
    """
    a = a_init.copy()
    b = a_init.copy()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    no_imp = 0
    t_blend = blend_rate
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Fix b, optimize a
        a_new, t1 = _lp_min_peak(b, n)
        if a_new is None:
            break
        
        # Fix a_new, optimize b
        b_new, t2 = _lp_min_peak(a_new, n)
        if b_new is None:
            b_new = b.copy()
            t2 = t1
        
        # Blend
        a = _entropic_blend(a, a_new, t_blend)
        b = _entropic_blend(b, b_new, t_blend)
        
        # Evaluate
        sc_a = _score(a)
        sc_b = _score(b)
        
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            no_imp = 0
        elif sc_b < best_score - 1e-12:
            best_score = sc_b
            best_a = b.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                t_blend = max(0.05, t_blend * 0.95)
            if no_imp > 500:
                break
        
        # Shrink blend rate gradually
        if it % 100 == 0 and it > 0:
            t_blend = max(0.05, t_blend * 0.8)
    
    return best_a, best_score
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_conv_matrix(fixed, n):
    """Build matrix A (shape 2n-1 x n) such that A @ g = conv(fixed, g)."""
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = fixed
    return A


def _lp_min_peak(fixed, n):
    """
    Find g >= 0, sum(g) = 1, minimizing max_k conv(fixed, g)[k].
    Returns (g, t_opt) or (None, None).
    """
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_conv = _make_conv_matrix(fixed, n)
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        gs = g.sum()
        if gs < 1e-30:
            return None, None
        g = g / gs
        t = float(res.x[-1])
        return g, t
    return None, None


def _lp_spread(fixed, M, n):
    """
    Original LP: maximize sum(g) subject to conv(fixed, g) <= M, g >= 0.
    Returns g or None.
    """
    c_obj = -np.ones(n)
    A_conv = _make_conv_matrix(fixed, n)
    res = linprog(c_obj, A_ub=A_conv, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-30:
            return None
        return g
    return None


def _entropic_blend(a, b, t, eps=1e-15):
    """Geometric/entropic blend preserving non-negativity."""
    a = np.maximum(a, eps)
    b = np.maximum(b, eps)
    log_mixed = (1.0 - t) * np.log(a) + t * np.log(b)
    log_mixed -= float(log_mixed.max())
    mixed = np.exp(log_mixed)
    s = mixed.sum()
    if s < 1e-30:
        return _normalize(a)
    return mixed / s


def _linear_blend(a, b, t):
    mixed = (1.0 - t) * a + t * b
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s < 1e-30:
        return a.copy()
    return mixed / s


def _two_sided_lp(a_init, deadline, max_iters=20000):

    a = a_init.copy()
    b = a_init.copy()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    t_blend = 0.4

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Fix b, optimize a
        a_new, t1 = _lp_min_peak(b, n)
        if a_new is None:
            break

        # Fix a_new, optimize b
        b_new, t2 = _lp_min_peak(a_new, n)
        if b_new is None:
            b_new = b.copy()

        # Entropic blend
        a = _entropic_blend(a, a_new, t_blend)
        b = _entropic_blend(b, b_new, t_blend)

        # Evaluate both
        sc_a = _score(a)
        sc_b = _score(b)

        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            no_imp = 0
        elif sc_b < best_score - 1e-12:

            best_a = b.copy()
            no_imp = 0
        else:

            if no_imp % 30 == 0:

            if no_imp > 300:

def _smooth_max_and_grad(a, beta):
    """Compute smooth max value and gradient of max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    val = mval + beta * float(np.log(s)) if s > 0 else mval
    return val, w, grad, mval


def _mirror_adam_polish(a_init, deadline, max_iter=50000, lr=0.05):
    """Adam optimization in log-space with smooth-max objective."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = _normalize(a)
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        val, w, grad, mval = _smooth_max_and_grad(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        # Convert to log-space gradient
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.93)

        if it % 300 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(3000, 5 * n):
            rng_r = np.random.default_rng(it * 7 + 3)
            z = 0.7 * best_z + 0.3 * rng_r.normal(size=n) * 0.3
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = 0.05
            no_imp = 0

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian_profile(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return _normalize(a)


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_cubic_profile(n, a1=1.5, a3=-1.0):

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return _normalize(a)

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = _normalize(p)
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define target lengths
    lengths = [512, 384, 640]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)

    # Phase 1: Two-sided LP at each length
    for n_t in lengths:
        if time.monotonic() > deadline - 300:
            break

        # Generate initial candidates for this length
        if parent is not None:
            a0 = _interpolate_profile(parent, n_t)
        else:
            a0 = _make_linear_profile(n_t, 1.5)

        sc0 = _score(a0)
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Two-sided LP
        lp_budget = min(400.0, deadline - time.monotonic() - 300.0)
        if lp_budget < 20:
            continue
        lp_end = time.monotonic() + lp_budget
        print(f"Phase 1 Two-sided LP: n={n_t}, start={sc0:.10f}, budget={lp_budget:.0f}s")

        a_lp, sc_lp = _two_sided_lp(a0, lp_end, max_iters=20000)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  Two-sided LP: score={sc_lp:.10f}")

        # Single-sided LP refinement on the result
        ss_budget = min(100.0, deadline - time.monotonic() - 200.0)
        if ss_budget > 10:
            ss_end = time.monotonic() + ss_budget
            a_ss, sc_ss = _lp_refine_single(a_lp, ss_end, max_iters=10000)
            if np.isfinite(sc_ss) and sc_ss < best_score:
                best_score = sc_ss
                best_seq = a_ss.copy()
            print(f"  Single-sided LP: score={sc_ss:.10f}")

    # Phase 2: Try diverse initializations at n=512
    if time.monotonic() < deadline - 200:
        n_try = 512
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_try))
            for _ in range(2):
                p_pert = _interpolate_profile(parent, n_try)
                p_pert = p_pert * (1.0 + 0.15 * rng.normal(size=n_try))
                p_pert = np.maximum(p_pert, 0)
                inits.append(_normalize(p_pert))

        inits.append(_make_gaussian_profile(n_try, 0.25))
        inits.append(_make_gaussian_profile(n_try, 0.35))
        inits.append(_make_linear_profile(n_try, 2.0))
        inits.append(_make_cubic_profile(n_try, 2.0, -2.0))

        for a0 in inits:
            if time.monotonic() > deadline - 200:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            lp_b = min(80.0, deadline - time.monotonic() - 200.0)
            if lp_b < 10:
                break
            a_lp, sc_lp = _two_sided_lp(a0, time.monotonic() + lp_b, max_iters=5000)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  Diverse init: start={sc0:.8f} -> LP={sc_lp:.10f}")

    # Phase 3: Mirror Adam polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        ad_budget = min(300.0, deadline - time.monotonic() - 60.0)
        if ad_budget > 20:
            ad_end = time.monotonic() + ad_budget
            print(f"Phase 3 Mirror Adam: budget={ad_budget:.0f}s")
            a_ad, sc_ad = _mirror_adam_polish(best_seq, ad_end, max_iter=50000, lr=0.04)
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Adam result: score={sc_ad:.10f}")

    # Phase 4: Final LP + gradient cycle
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Quick LP
        lp_b = min(60.0, deadline - time.monotonic() - 30.0)
        if lp_b > 5:
            a_lp, sc_lp = _lp_refine_single(best_seq, time.monotonic() + lp_b, max_iter=5000)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"Phase 4 LP: score={sc_lp:.10f}")

        # Quick Adam
        if time.monotonic() < deadline - 15:
            ad_b = min(40.0, deadline - time.monotonic() - 15.0)
            if ad_b > 5:
                a_ad, sc_ad = _mirror_adam_polish(best_seq, time.monotonic() + ad_b,
                                                  max_iter=10000, lr=0.03)
                if np.isfinite(sc_ad) and sc_ad < best_score:
                    best_score = sc_ad
                    best_seq = a_ad.copy()
                print(f"Phase 4 Adam: score={sc_ad:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
```
