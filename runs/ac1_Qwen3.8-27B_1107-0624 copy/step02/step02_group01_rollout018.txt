"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient descent in log-space (softmax parameterization)
with FFT-based subgradient computation, combined with LP-based polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

def compute_subgradient(a, top_k=10, beta=None):
    """Compute subgradient of max(conv(a,a))[k] w.r.t. a.
    Uses top-k or softmax weighting."""
    n = a.size
    c = fft_convolve(a)
    
    if top_k is not None and top_k < len(c):
        # Top-k subgradient
        idx = np.argpartition(c, -top_k)[-top_k:]
        S = np.zeros(2 * n - 1)
        S[idx] = 1.0 / top_k
    else:
        # Softmax weighting
        if beta is None:
            beta = 0.05 * c.max()
        c_norm = c / max(beta, 1e-30)
        c_shifted = c_norm - c_norm.max()
        w = np.exp(c_shifted)
        S = w / w.sum()
    
    # Compute g[i] = sum_k S[k] * a[k-i] via FFT correlation
    N = next_pow2(3 * n)
    fs = np.fft.rfft(S, N)
    fa = np.fft.rfft(a, N)
    g = np.fft.irfft(fs * np.conj(fa), N)[:n]
    
    return 2.0 * g  # The factor of 2 from the derivative

def nesterov_subgradient(a0, deadline, rng, top_k=20, lr=None, max_iter=100000):
    """Nesterov-accelerated subgradient descent in log-space."""
    n = a0.size
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    
    if lr is None:
        lr = 0.5 / np.sqrt(n)
    
    # Nesterov state
    a_prev = a.copy()
    momentum = 0.9
    
    best_a = a.copy()
    c = fft_convolve(a)
    best_M = c.max()
    
    no_imp = 0
    t_adam = 0
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    z = np.log(a)
    z -= z.mean()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step: evaluate at y = a + (1 - momentum) * (a - a_prev) / momentum
        # Actually, standard Nesterov: y = a + beta_t * (a - a_prev)
        # where beta_t = (1 - sqrt(1 + 4*beta^2)) / 2 for constant momentum beta
        
        beta_t = momentum
        y = a + beta_t * (a - a_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()
        
        a_prev = a.copy()
        
        # Compute subgradient at y
        grad_a = compute_subgradient(y, top_k=top_k)
        
        # Convert to log-space gradient
        mean_grad = np.dot(y, grad_a)
        grad_z = y * (grad_a - mean_grad)
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        
        # Update
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Evaluate
        c = fft_convolve(a)
        M = c.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 500:
            # Random restart from best
            z = np.log(best_a + 1e-15)
            z += 0.1 * rng.normal(size=n)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_a, best_M

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines Nesterov-accelerated subgradient descent (log-space/softmax parameterization,
FFT-based subgradients) with LP-based direction finding for final polishing.
Tries multiple sequence lengths (512, 1024) and diverse starting points.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _fft_convolve(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def _compute_subgrad(a, top_k=30):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a using top-k lags."""
    n = a.size
    c = _fft_convolve(a)
    k = min(top_k, 2 * n - 1)
    idx = np.argpartition(c, -k)[-k:]
    S = np.zeros(2 * n - 1)
    S[idx] = 1.0 / k
    N = _next_pow2(3 * n)
    fs = np.fft.rfft(S, N)
    fa = np.fft.rfft(a, N)
    g = np.fft.irfft(fs * np.conj(fa), N)[:n]
    return 2.0 * g


def _compute_subgrad_smooth(a, beta):
    """Compute smoothed subgradient using softmax weighting."""
    n = a.size
    c = _fft_convolve(a)
    c_shifted = c / max(beta, 1e-30) - (c / max(beta, 1e-30)).max()
    w = np.exp(c_shifted)
    S = w / w.sum()
    N = _next_pow2(3 * n)
    fs = np.fft.rfft(S, N)
    fa = np.fft.rfft(a, N)
    g = np.fft.irfft(fs * np.conj(fa), N)[:n]
    return 2.0 * g, S


def _nesterov_subtype(a0, deadline, rng, top_k=30, lr=None, max_iter=200000):
    """Nesterov-accelerated subgradient in log-space with Adam."""
    n = a0.size
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    if lr is None:
        lr = 1.0 / np.sqrt(n)

    z = np.log(a)
    z -= z.mean()

    c = _fft_convolve(a)
    best_M = float(c.max())
    best_a = a.copy()

    momentum = 0.85
    a_prev = a.copy()
    no_imp = 0
    t_adam = 0
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        a_prev = a.copy()

        grad_a = _compute_subgrad(y, top_k=top_k)
        mean_g = float(np.dot(y, grad_a))
        grad_z = y * (grad_a - mean_g)

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a + 1e-15)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _fft_convolve(a)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 800:
            z = np.log(best_a + 1e-15)
            z += 0.05 * rng.normal(size=n)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            a_prev = a.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            momentum = 0.85

    return best_a, best_M


def _smooth_gradient_polish(a0, deadline, rng, lr=0.08, max_iter=100000):
    """Polish using smoothed gradient with adaptive beta in log-space."""
    n = a0.size
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    c = _fft_convolve(a)
    M_val = float(c.max())
    beta = max(1e-10, 0.03 * M_val)

    best_M = M_val
    best_a = a.copy()
    best_z = z.copy()
    no_imp = 0
    t_adam = 0
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad_a, S = _compute_subgrad_smooth(a, beta)
        mean_g = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - mean_g)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _fft_convolve(a)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 500 == 0:
            beta = max(1e-12, beta * 0.6)

        if no_imp > 600:
            z = 0.9 * best_z + 0.15 * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _fft_convolve(a)
            M_val = float(c.max())
            beta = max(1e-10, 0.05 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_a, best_M


def _solve_lp_direction(f, M, n):
    """LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_optimize(f0, deadline, rng, max_iter=5000):
    """LP-based iterative optimization."""
    n = f0.size
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s

    c = _fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = _solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                continue
            f = f / fs
            M = float(_fft_convolve(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 200:
                break
            continue

        g = g / g_sum

        improved = False
        for t in (0.3, 0.15, 0.5, 0.1, 0.7, 0.05, 0.4, 0.2):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(_fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp > 300:
                break
            if no_imp % 50 == 0:
                f = 0.8 * best_f + 0.2 * np.abs(rng.normal(size=n)) + 1e-10
                f = f / f.sum()
                M = float(_fft_convolve(f).max())
                no_imp = 0

    return best_f, best_M


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n, seed=0):
    rng = np.random.default_rng(seed)
    profs = []
    for eta in (0.6, 0.9, 1.2, 1.5, 1.8, 2.2, 2.8, 3.5):
        profs.append(_make_linear(n, eta))
    for a1, a3 in ((1.0, -0.6), (1.3, -0.9), (1.6, -1.3), (2.0, -1.8),
                    (2.5, -2.2), (1.0, -1.5), (0.7, 0.4), (1.5, -2.5)):
        profs.append(_make_cubic(n, a1, a3))
    for _ in range(6):
        profs.append(_make_smooth_random(n, rng))
    return profs
