"""
LP-based alternating minimization with gradient polish for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses scipy's HiGHS LP solver for the inner LP, combined with 
soft-max gradient descent for fast local optimization.
"""
import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time


def _full_convolve(a):
    """Full convolution using FFT for speed."""
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]


def _score(a):
    """Compute 2*n*max(a*a)/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s*s))


def _build_conv_matrix(f, active_idx=None):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid.
    Only includes rows for active_idx (or all if None)."""
    n = f.size
    m = 2*n - 1
    if active_idx is None:
        active_idx = np.arange(m)
    
    rows = []
    cols = []
    vals = []
    for k in active_idx:
        # A[k, j] = f[k-j] for 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
    # Keep only rows in active_idx
    if len(active_idx) < m:
        A = A[active_idx, :]
    return A


def _solve_lp(f, M, active_idx=None, initial_ub=None):
    """Solve: max sum(b) s.t. f*b <= M, b >= 0."""
    n = f.size
    m = 2*n - 1
    
    A = _build_conv_matrix(f, active_idx)
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(A.shape[0], M),
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - z.max()
    e = np.exp(z)
    p = e + floor
    p = p / p.sum()
    return p


def _logit_gradient_step(f, deadline, lr=0.05, beta_frac=0.1, beta_decay=0.75, 
                          max_iters=200):
    """Quick gradient descent in logit space to polish f."""
    n = f.size
    p = f / f.sum()
    z = np.log(np.maximum(p, 1e-12)) - np.log(np.maximum(p, 1e-12)).mean()
    
    c = _full_convolve(p)
    maxc = c.max()
    beta = beta_frac * maxc
    best_p = p.copy()
    best_score = 2.0 * n * maxc
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Soft-max weights
        x = (c - maxc) / beta
        ex = np.exp(x)
        w = ex / ex.sum()
        
        # Gradient of max(c) w.r.t. p: 2 * (w * p reversed correlation)
        # d/dp_i [sum_k w_k * c_k] = 2 * sum_k w_k * (p * p_reversed)[i+k]
        # = 2 * corr(w, p)[i] where corr is cross-correlation
        p_rev = p[::-1]
        N = 1
        while N < 2*n - 1:
            N <<= 1
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p_rev, N)
        gp = 2.0 * np.fft.irfft(fw * fp, N)[n-1:2*n-1]
        
        # Projected gradient in logit space
        dot = np.dot(p, gp)
        gz = p * (gp - dot)
        gz -= gz.mean()
        
        t_adam += 1
        m_adam = b1 * m_adam + (1-b1) * gz
        v_adam = b2 * v_adam + (1-b2) * gz**2
        mh = m_adam / (1 - b1**t_adam)
        vh = v_adam / (1 - b2**t_adam)
        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        p = _softmax_from_z(z)
        
        c = _full_convolve(p)
        maxc = c.max()
        score = 2.0 * n * maxc
        
        if score < best_score:
            best_score = score
            best_p = p.copy()
        
        if t_adam % 30 == 0:
            beta *= beta_decay
    
    return best_p, best_score

"""
LP-based alternating minimization combined with soft-max gradient descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses scipy HiGHS LP solver for the inner optimization, with active-set
reduction and multi-length search (up to n=4096).
"""
import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = c - mn
    return c


def _score_internal(a):
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_exact(a):
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64)
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _build_conv_matrix(f, active_idx):
    n = f.size
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in active_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    if not rows:
        return sparse.csr_matrix((0, n))
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
    if len(active_idx) < m:
        A = A[active_idx, :]
        A = A.tocsr()
    return A


def _solve_lp(f, M, active_idx=None):
    n = f.size
    m = 2 * n - 1
    if active_idx is None:
        active_idx = np.arange(m)
    active_idx = np.asarray(active_idx, dtype=np.int64)
    A = _build_conv_matrix(f, active_idx)
    if A.nnz == 0:
        return None
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_alternate(f0, deadline, rng, t_step=0.5, active_frac=0.3,
                  min_active=50, max_active=5000, polish_every=20):
    n = f0.size
    f = f0.copy()
    f = _normalize(f)
    if f is None:
        return f0, _score_internal(f0)
    
    c = _full_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * M)
    
    active_idx = np.arange(2 * n - 1)
    iters = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(f)
        M = float(c.max())
        score = float(2.0 * n * M)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        # Determine active set: top constraints + random
        n_active = max(min_active, min(max_active, int(active_frac * (2 * n - 1))))
        top_idx = np.argsort(c)[::-1][:n_active]
        # Also include some random ones
        n_rand = max(5, n_active // 5)
        rand_idx = rng.integers(0, 2 * n - 1, size=n_rand)
        active_idx = np.unique(np.concatenate([top_idx, rand_idx]))
        
        b = _solve_lp(f, M, active_idx)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        f_new = (1.0 - t_step) * f + t_step * g
        f_new = _normalize(f_new)
        if f_new is None:
            break
        f = f_new
        iters += 1
        
        if iters % polish_every == 0:
            # Quick gradient polish
            f, sc = _quick_gradient_polish(f, 30, rng)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
    
    return best_f, best_score


def _quick_gradient_polish(f, max_steps, rng, lr=0.05):
    n = f.size
    p = f / f.sum()
    z = np.log(np.maximum(p, 1e-12))
    z -= z.mean()
    
    c = _full_convolve(p)
    maxc = c.max()
    beta = 0.1 * maxc
    best_p = p.copy()
    best_sc = 2.0 * n * maxc
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    
    for _ in range(max_steps):
        x = (c - maxc) / beta
        ex = np.exp(x - x.max())
        w = ex / ex.sum()
        
        p_rev = p[::-1]
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p_rev, N)
        gp = 2.0 * np.fft.irfft(fw * fp, N)[n - 1:2 * n - 1]
        
        dot = np.dot(p, gp)
        gz = p * (gp - dot)
        gz -= gz.mean()
        
        t_adam += 1
        m_adam = b1 * m_adam + (1 - b1) * gz
        v_adam = b2 * v_adam + (1 - b2) * gz ** 2
        mh = m_adam / (1 - b1 ** t_adam)
        vh = v_adam / (1 - b2 ** t_adam)
        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        p = _softmax_from_z(z)
        
        c = _full_convolve(p)
        maxc = c.max()
        sc = 2.0 * n * maxc
        if sc < best_sc:
            best_sc = sc
            best_p = p.copy()
        
        if t_adam % 10 == 0:
            beta *= 0.8
    
    return best_p, best_sc


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64)
    z = np.clip(z, -50.0, 50.0)
    z = z - z.max()
    e = np.exp(z)
    p = e + floor
    p = p / p.sum()
    return p


def _logit_anneal(p0, deadline, rng, lr=0.06, beta_init_frac=0.12,
                  beta_decay=0.72, no_imp_mult=4):
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    
    z = np.log(p0)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = c.max()
    beta = max(1e-12, beta_init_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    
    while time.monotonic() < deadline:
        x = (c - maxc) / beta
        ex = np.exp(x - x.max())
        w = ex / ex.sum()
        
        p_rev = p[::-1]
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p_rev, N)
        gp = 2.0 * np.fft.irfft(fw * fp, N)[n - 1:2 * n - 1]
        
        dot = np.dot(p, gp)
        gz = p * (gp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        z = np.clip(z, -50, 50)
        z -= z.mean()
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = c.max()
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        if adam_t % 30 == 0:
            beta = max(1e-12, beta * beta_decay)
        
        no_imp_limit = max(100, no_imp_mult * n)
        if no_imp > no_imp_limit:
            noise_scale = 0.15 if best_score < 1.55 else 0.3
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = c.max()
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            no_imp = 0
    
    return best_p, best_score


def _make_profiles(n):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    # Linear
    a = 1.0 + 1.5 * (t - 0.5)
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    a = 1.0 + 2.0 * (t - 0.5)
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    # Quadratic
    a = 1.0 + 1.0 * u + 0.5 * u ** 2
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    a = 1.0 + 1.5 * u - 0.3 * u ** 2
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    # Cubic
    a = 1.0 + 1.5 * u - 1.0 * u ** 3
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    # Cosine bump
    a = 1.0 + 0.5 * np.cos(np.pi * u)
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    # Raised cosine squared
    a = (0.5 * (1 + np.cos(np.pi * u))) ** 2
    profiles.append(np.maximum(a, 1e-12) / (np.maximum(a, 1e-12).sum()))
    # Smooth bump (super-Gaussian)
    a = np.exp(-4.0 * u ** 4)
    profiles.append(a / a.sum())
    return profiles
