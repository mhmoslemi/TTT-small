```python
def solve_lp_step(f, M):
    """
    Solve: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k
    where f is the current sequence and M is the max convolution value.
    """
    n = len(f)
    # (f*b)_k = sum_i f_i * b_{k-i} for valid indices
    # Variable: b (length n)
    # Constraints: for each k in 0..2n-2, sum_{i: 0<=k-i<=n-1} f_i * b_{k-i} <= M
    # Plus: b_j >= 0 for all j
    
    # Build constraint matrix A (shape (2n-1, n))
    # A[k, j] = f_{k-j} if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    # Solve LP
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if result.success:
        b = result.x
        return b
    else:
        return None
```

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if s <= 1e-30:
        return a * target_sum / max(a.size, 1)
    return a * (target_sum / s)

        for k in range(ks, ke + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_direction(f, M):
    """
    Solve: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns the optimal b vector, or None if infeasible.
    """
    n = len(f)
    A = _build_conv_matrix_sparse(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def _lp_optimize(p0, deadline, rng, max_iters=50000, t_init=0.5):
    """
    LP-based iterative optimization.
    p0: initial sequence (will be normalized to sum=sqrt(2n))
    Returns (best_p, best_score_normalized) where score = max(f*f) with sum(f)=sqrt(2n).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p0.size
    target_sum = np.sqrt(2.0 * n)
    f = _normalize(p0, target_sum)

    c = _full_convolve_fft(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * M / (target_sum * target_sum)  # = 2M/(2n) * 2n = ... 
    # Actually score = 2n * M / (sum^2) = 2n * M / (2n) = M when sum=sqrt(2n)
    # Wait: 2n * M / (sqrt(2n))^2 = 2n * M / (2n) = M
    # So the score equals M when normalized to sum=sqrt(2n)!
    best_score = M

    t = t_init
    no_imp = 0
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1

        # Solve LP
        b_star = _solve_lp_direction(f, M)
        if b_star is None:
            # LP failed, fall back to gradient step
            b_star = f.copy()
        else:
            b_star = np.clip(b_star, 0.0, None)
            s_b = float(b_star.sum())
            if s_b < 1e-30:
                b_star = f.copy()
                s_b = target_sum

        # Rescale to match target sum
        g = b_star * (target_sum / s_b)

        # Convex combination update
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        s_new = float(f_new.sum())
        if s_new < 1e-30:
            break
        f_new = f_new * (target_sum / s_new)

        # Evaluate
        c_new = _full_convolve_fft(f_new)
        M_new = float(c_new.max())

        if M_new <= M + 1e-10:
            f = f_new
            M = M_new
            t = min(1.0, t * 1.05)  # slightly increase step
        else:
            # Backtrack: reduce step size
            t = t * 0.5
            if t < 0.01:
                t = 0.01
            # Try with smaller step
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            s_new = float(f_new.sum())
            if s_new < 1e-30:
                break
            f_new = f_new * (target_sum / s_new)
            c_new = _full_convolve_fft(f_new)
            M_new = float(c_new.max())
            if M_new <= M + 1e-10:
                f = f_new
                M = M_new
            else:
                f = f_new  # accept anyway for exploration
                M = M_new

        # Track best
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            best_score = M
            no_imp = 0
        else:
            no_imp += 1

        # Check convergence: if b_star is close to f, we're at a fixed point
        if iter_count % 50 == 0:
            diff = np.max(np.abs(g - f)) / target_sum
            if diff < 1e-8:
                # At fixed point, try a restart
                noise_scale = 0.05 + 0.001 * no_imp
                z = np.log(np.maximum(f, 1e-15))
                z = z + noise_scale * rng.normal(size=n)
                f_new = np.exp(z - z.max())
                f_new = _normalize(f_new, target_sum)
                c_new = _full_convolve_fft(f_new)
                M_new = float(c_new.max())
                if M_new < best_M:
                    f = f_new
                    M = M_new
                else:
                    f = best_f.copy()
                    M = best_M
                t = t_init
                no_imp = 0

        if no_imp > 5000:
            # Aggressive restart
            noise_scale = 0.1
            z = np.log(np.maximum(best_f, 1e-15))
            z = z + noise_scale * rng.normal(size=n)
            f_new = np.exp(z - z.max())
            f_new = _normalize(f_new, target_sum)
            f = f_new
            c = _full_convolve_fft(f)
            M = float(c.max())
            t = t_init
            no_imp = 0

        if iter_count % 200 == 0:
            print(f"    LP iter={iter_count}, M={M:.10f}, t={t:.4f}, no_imp={no_imp}")

    return best_f, best_score


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    """Compute correlation: result[i] = sum_j w[j] * p[j-i+n-1] (circular cross-corr)"""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _gradient_polish(p0, deadline, rng, lr=0.05, beta_init_frac=0.1, beta_decay=0.7):
    """
    Soft-max gradient descent in logit space for fine polishing.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)
    best_p = p0.copy()
    best_score = maxc
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)

        # Add small L2 regularization
        gp += 1e-7 * p

        # Project gradient to simplex (remove mean component)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_p, 1e-15))
            z -= z.mean()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_p, 1e-15))
            z -= z.mean()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())

        if maxc < best_score - 1e-14:
            best_score = maxc
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.92)

        # Adaptive beta
        if adam_t % 50 == 0:
            beta = max(1e-14, beta * beta_decay)

        # Restart
        if no_imp > max(200, 2 * n):
            noise = 0.08
            z = np.log(np.maximum(best_p, 1e-15))
            z = z + noise * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = 0.05
            no_imp = 0

    return best_p, best_score


def _try_lengths_and_optimize(budget_s, seed, parent_seq=None):
    """
    Main optimization loop with multi-length escalation.
    Returns (best_sequence_normalized_to_sum_1, best_score).
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: score={best_score:.10f}, n={p.size}")

    # Generate initial candidates
    initial_candidates = []
    for n in [128, 256, 512]:
        initial_candidates.append((n, _make_flat_profile(n), 'flat'))
        initial_candidates.append((n, _make_cosine_profile(n), 'cosine'))
        initial_candidates.append((n, _make_quadratic_profile(n, 0.5, 0.3), 'quad1'))
        initial_candidates.append((n, _make_quadratic_profile(n, 1.0, -0.2), 'quad2'))
        initial_candidates.append((n, _make_cubic_profile(n, 1.0, -0.5), 'cubic1'))
        initial_candidates.append((n, _make_cubic_profile(n, 1.5, -1.0), 'cubic2'))
        # Random
        for _ in range(3):
            r = np.abs(rng.normal(1.0, 0.5, n))
            r = r / r.sum()
            initial_candidates.append((n, r, 'rand'))

    # Add parent if available
    if parent_seq is not None:
        parent_seq = np.asarray(parent_seq, dtype=np.float64).ravel()
        parent_seq = np.clip(parent_seq, 0.0, None)
        s = parent_seq.sum()
        if s > 0.01:
            parent_norm = parent_seq / s
            pn = parent_norm.size
            initial_candidates.append((pn, parent_norm, 'parent'))
            for n in [256, 512, 1024]:
                if n != pn:
                    p_interp = _interp_profile(parent_norm, n)
                    initial_candidates.append((n, p_interp, f'parent_interp_{n}'))

    # Score and sort initial candidates
    scored = []
    for n, p, label in initial_candidates:
        sc = _internal_score(p)
        if np.isfinite(sc):
            scored.append((sc, n, p, label))
    scored.sort(key=lambda x: x[0])
    print(f"Initial: best score = {scored[0][0]:.6f} ({scored[0][3]}, n={scored[0][1]})")
    _update_best(scored[0][2], scored[0][0])

    # Phase 1: LP at n=512 (fast, good starting point)
    # Use time budget: ~25% for each of 3 length phases, rest for polishing
    phase1_time = 0.20 * budget_s
    phase2_time = 0.35 * budget_s
    phase3_time = 0.30 * budget_s
    phase4_time = 0.10 * budget_s

    phase1_end = t_start + phase1_time
    phase2_end = t_start + phase1_time + phase2_time
    phase3_end = t_start + phase1_time + phase2_time + phase3_time
    phase4_end = deadline

    print(f"\n--- Phase 1: LP at n=512 (budget={phase1_time:.0f}s) ---")
    p_512 = None
    sc_512 = np.inf

    # Get best candidate at or near n=512
    best_at_512 = None
    for sc, n, p, label in scored:
        if n == 512:
            if best_at_512 is None or sc < _internal_score(best_at_512[2]):
                best_at_512 = (sc, n, p, label)
    if best_at_512 is None:
        best_at_512 = (np.inf, 512, _make_cosine_profile(512), 'default')

    # Also try LP from a few different starts at n=512
    starts_512 = []
    for sc, n, p, label in scored:
        if n <= 512:
            if n < 512:
                p = _interp_profile(p, 512)
            starts_512.append((sc, p, label))
            if len(starts_512) >= 4:
                break

    if not starts_512:
        starts_512 = [(0, _make_cosine_profile(512), 'cosine')]

    per_start_time = phase1_time / len(starts_512)
    for i, (sc_init, p_start, label) in enumerate(starts_512):
        if time.monotonic() >= phase1_end:
            break
        per_deadline = min(phase1_end, time.monotonic() + per_start_time)
        print(f"  LP start {i} ({label}):")
        p_result, sc_result = _lp_optimize(p_start, per_deadline, rng, max_iters=10000)
        # Convert: sc_result is M with sum=sqrt(2n), so score = M (since 2n*M/(2n)=M)
        actual_score = sc_result  # = 2n*M/(sum^2) = 2n*M/(2n) = M
        if actual_score < sc_512:
            sc_512 = actual_score
            p_512 = p_result
        _update_best(_normalize(p_result), _internal_score(_normalize(p_result)))

    if p_512 is not None:
        print(f"  Phase 1 best: score={sc_512:.10f}")
    else:
        p_512 = _normalize(best_at_512[2])
        sc_512 = _internal_score(p_512)

    # Phase 2: LP at n=1024
    print(f"\n--- Phase 2: LP at n=1024 (budget={phase2_time:.0f}s) ---")
    p_1024 = None
    sc_1024 = np.inf

    # Build starts for n=1024
    starts_1024 = []
    if p_512 is not None:
        p_interp = _interp_profile(p_512, 1024)
        starts_1024.append((sc_512, p_interp, 'from_512'))
    for sc, n, p, label in scored:
        if n == 1024:
            starts_1024.append((sc, p, label))
            if len(starts_1024) >= 5:
                break
    # Add parent if n=1024
    if parent_seq is not None and np.asarray(parent_seq).size == 1024:
        pn = np.asarray(parent_seq).ravel()
        pn = np.clip(pn, 0, None)
        if pn.sum() > 0.01:
            starts_1024.append((_internal_score(pn / pn.sum()), pn / pn.sum(), 'parent_direct'))

    if not starts_1024:
        starts_1024 = [(0, _make_cosine_profile(1024), 'cosine')]

    # Limit to top 3 starts
    starts_1024.sort(key=lambda x: x[0])
    starts_1024 = starts_1024[:3]

    per_start_time = phase2_time / len(starts_1024)
    for i, (sc_init, p_start, label) in enumerate(starts_1024):
        if time.monotonic() >= phase2_end:
            break
        per_deadline = min(phase2_end, time.monotonic() + per_start_time)
        print(f"  LP start {i} ({label}):")
        p_result, sc_result = _lp_optimize(p_start, per_deadline, rng, max_iters=20000)
        actual_score = sc_result
        if actual_score < sc_1024:
            sc_1024 = actual_score
            p_1024 = p_result
        _update_best(_normalize(p_result), _internal_score(_normalize(p_result)))

    if p_1024 is not None:
        print(f"  Phase 2 best: score={sc_1024:.10f}")

    # Phase 3: LP at n=2048
    print(f"\n--- Phase 3: LP at n=2048 (budget={phase3_time:.0f}s) ---")
    p_2048 = None
    sc_2048 = np.inf

    if p_1024 is not None and time.monotonic() < phase3_end:
        starts_2048 = []
        p_interp = _interp_profile(p_1024, 2048)
        starts_2048.append((sc_1024, p_interp, 'from_1024'))
        # Also try from best overall
        if best_seq is not None and best_seq.size != 2048:
            p_interp2 = _interp_profile(best_seq, 2048)
            sc2 = _internal_score(p_interp2)
            starts_2048.append((sc2, p_interp2, 'from_best'))

        per_start_time = phase3_time / len(starts_2048)
        for i, (sc_init, p_start, label) in enumerate(starts_2048[:2]):
            if time.monotonic() >= phase3_end:
                break
            per_deadline = min(phase3_end, time.monotonic() + per_start_time)
            print(f"  LP start {i} ({label}):")
            p_result, sc_result = _lp_optimize(p_start, per_deadline, rng, max_iters=15000)
            actual_score = sc_result
            if actual_score < sc_2048:
                sc_2048 = actual_score
                p_2048 = p_result
            _update_best(_normalize(p_result), _internal_score(_normalize(p_result)))

    if p_2048 is not None:
        print(f"  Phase 3 best: score={sc_2048:.10f}")

    # Phase 4: Gradient polishing at best length
    print(f"\n--- Phase 4: Gradient polishing (budget={phase4_time:.0f}s) ---")
    if best_seq is not None and time.monotonic() < phase4_end:
        # Polish at the best sequence's length
        p_to_polish = best_seq.copy()
        sc_to_polish = _internal_score(p_to_polish)
        print(f"  Polishing n={p_to_polish.size}, score={sc_to_polish:.10f}")

        # Run gradient polish for most of the time
        polish_deadline = phase4_end - 5.0
        p_result, sc_result_M = _gradient_polish(p_to_polish, polish_deadline, rng,
                                                  lr=0.04, beta_init_frac=0.08, beta_decay=0.75)
        # Convert: sc_result_M is max(c) with sum=1, so score = 2n*sc_result_M
        actual_score = 2.0 * p_result.size * sc_result_M
        _update_best(p_result, actual_score)

        # Also try polishing at a different length if time permits
        if time.monotonic() < phase4_end - 30 and p_2048 is not None:
            p_trial = _interp_profile(p_2048, 2048)
            trial_deadline = phase4_end - 5.0
            p_result2, sc_result2_M = _gradient_polish(p_trial, trial_deadline, rng,
                                                        lr=0.03, beta_init_frac=0.05, beta_decay=0.8)
            actual_score2 = 2.0 * p_result2.size * sc_result2_M
            _update_best(p_result2, actual_score2)

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Phase 1: LP-based optimization at n=512
    Phase 2: Interpolate + LP at n=1024
    Phase 3: Interpolate + LP at n=2048
    Phase 4: Soft-max gradient polishing
    
    The LP direction (max sum(b) s.t. (f*b)_k <= M, b >= 0) provides a
    globally optimal linearization direction, ensuring monotone improvement.
    """
    t_start = time.monotonic()
    total_budget = min(float(budget_s), 1800.0)
    safety_margin = 15.0
    effective_budget = total_budget - safety_margin

    print(f"=== Starting optimization (budget={effective_budget:.0f}s) ===")

    # Get parent sequence if available
    parent_seq = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent_seq = parent_raw
                print(f"Parent sequence found: n={parent_seq.size}, score={_internal_score(parent_seq/parent_seq.sum()):.6f}")
    except Exception:
        print("No parent sequence available.")

    # Run main optimization
    best_seq, best_score = _try_lengths_and_optimize(effective_budget, int(seed) % 1000000, parent_seq)

    # Final validation
    if best_seq is None:
        # Fallback: cosine profile
        best_seq = _make_cosine_profile(512)
        best_score = _internal_score(best_seq)
        print(f"FALLBACK: n=512, score={best_score:.6f}")
    else:
        # Verify with exact convolution for smaller sequences
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s > 0.01:
            best_seq = best_seq / s
        else:
            best_seq = _make_cosine_profile(512)

        n_final = best_seq.size
        if n_final <= 4096:
            c_exact = np.convolve(best_seq, best_seq)
            sc_exact = 2.0 * n_final * float(c_exact.max()) / (float(best_seq.sum()) ** 2)
            print(f"Exact score: {sc_exact:.10f} (n={n_final})")
            if sc_exact < best_score:
                best_score = sc_exact

        # Try to use evaluate_sequence for authoritative check
        if n_final <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in best_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception:
                pass

    elapsed = time.monotonic() - t_start
    print(f"\n=== DONE: score={best_score:.10f}, n={best_seq.size}, time={elapsed:.1f}s ===")
    return [float(x) for x in best_seq]
