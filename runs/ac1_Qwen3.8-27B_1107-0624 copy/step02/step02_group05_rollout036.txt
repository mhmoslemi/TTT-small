The gradient of the smoothed objective (power mean) with respect to f is:
grad = 2 * cross_correlation(w, f)

where w = (f*f)^(p-1) / sum((f*f)^(p-1)) (the weights from the power mean).

And the cross-correlation is computed via FFT.

Now, for the projected gradient descent on the simplex:
1. Compute gradient
2. Center it: grad_centered = grad - mean(grad) (project onto tangent space of simplex)
3. Take step: f_new = f - alpha * grad_centered
4. Project onto simplex: f = max(0, f_new), then renormalize

def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def compute_conv(f):
    """Compute full linear convolution using FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = rfft(f, N)
    conv = irfft(F * F, N)
    return conv[:2 * n - 1]


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = compute_conv(f)
    return float(2.0 * n * c.max() / (s * s))


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def get_grad_weights(conv_vals, p):
    """Get gradient weights from power mean of convolution values."""
    cp = np.maximum(conv_vals, 1e-300)
    cp1 = cp ** (p - 1.0)
    s = float(cp1.sum())
    if s < 1e-300:
        w = np.zeros_like(cp)
        w[np.argmax(cp)] = 1.0
        return w
    return cp1 / s


def compute_gradient(weights, f):
    """
    Compute gradient of sum_k weights[k]*(f*f)[k] w.r.t. f.
    Uses FFT-based cross-correlation.
    """
    n = len(f)
    N = 1
    while N < 3 * n:
        N <<= 1
    fw_rev = rfft(f[::-1], N)
    ww = rfft(weights, N)
    result = irfft(ww * fw_rev, N)
    corr = result[n - 1:2 * n - 1]
    grad = 2.0 * corr
    return grad


def solve_lp_for_g(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def make_linear_profile(n, slope=2.0):
    """Linear ramp profile."""
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    """Anti-symmetric cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3

def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    """Gaussian bump profile."""
    t = np.linspace(0.0, 1.0, n)

def make_dual_bump(n, edge_sigma=0.15):
    """Two bumps at edges."""
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / edge_sigma) ** 2) + np.exp(-0.5 * ((t - 1) / edge_sigma) ** 2) + 0.1

    return f / f.sum()

    x0 = np.linspace(0, 1, n0)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_optimize(f_init, deadline, rng, max_iters=200, time_limit=3.0, verbose=False):

    if f is None:
        return f_init, compute_score(f_init)

    n = len(f)
    M = float(compute_conv(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = solve_lp_for_g(f, M, time_limit=time_limit)
        if g is None:
            f = f + 0.1 * rng.standard_normal(n)
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            M = float(compute_conv(f).max())
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:

            M_try = float(compute_conv(f_try).max())
            if M_try < best_M_try - 1e-13:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-13:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

            f = 0.8 * best_f + 0.2 * normalize_nonneg(best_f + 0.3 * rng.standard_normal(n))
            if f is None:
