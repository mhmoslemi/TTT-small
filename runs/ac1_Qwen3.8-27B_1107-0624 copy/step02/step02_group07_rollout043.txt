def coord_optimize(s, j, S, conv, num_grid=100):

def gradient_step(s, beta=10.0, eta=0.5):
    """One smoothed gradient step (multiplicative update)."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = np.max(conv)
    if max_conv <= 0:
        return s.copy()

    # Softmax weights on conv
    beta_conv = beta * (conv - max_conv) / max_conv
    weights = np.exp(beta_conv)
    weights = weights / weights.sum()

    # Gradient of conv w.r.t. s: grad[j] = 2 * sum_k weights[k] * s[k-j]
    s_rev = s[::-1]
    psr = np.convolve(weights, s_rev, mode='full')
    grad = 2.0 * psr[n - 1:n - 1 + n]
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()

    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-10)
    s_new = s_new / s_new.sum()
    return s_new
