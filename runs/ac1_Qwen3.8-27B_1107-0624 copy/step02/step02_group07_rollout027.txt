Let me also think about the **beta annealing schedule**. Starting with small beta (smooth, easy to optimize) and gradually increasing (sharper, closer to true max). The optimal schedule depends on the problem, but a geometric progression is a good default.

**One concern**: The softmax parameterization means that all entries of s are strictly positive. The true optimum might have some zero entries. However, as beta increases, the L-BFGS should push some entries towards zero (in the x-space, they go to -infinity). In practice, we'll get very small positive values, which is fine for the evaluator.

**Another concern**: L-BFGS-B is a first-order method and might get stuck in local minima. The beta annealing helps with this (starting smooth and getting sharp), but we might need multiple restarts.

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def smooth_max_and_prob(conv, beta):
    m = np.max(conv)
    shifted = beta * (conv - m)
    shifted = shifted - np.max(shifted)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    smax = m + np.log(sum_exp) / beta
    return smax, p

def objective_and_grad(x, beta):
    n = len(x)
    xc = x - np.max(x)
    s = np.exp(xc)
    s = s / s.sum()
    
    conv = np.convolve(s, s)
    smax, p = smooth_max_and_prob(conv, beta)
    val = 2.0 * n * smax
    
    q = s[::-1]
    pq = np.convolve(p, q)
    g = 2.0 * pq[n-1:2*n-1]
    gs = np.dot(g, s)
    grad = 2.0 * n * s * (g - gs)
    
    return val, grad

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def optimize_lbfgs(s_init, deadline, betas, max_iter_per_beta=200):
    n = len(s_init)
    s_init = np.maximum(s_init, 1e-15)
    s_init = s_init / s_init.sum()
    x0 = np.log(s_init)
    
    best_score = fast_score(s_init)
    best_x = x0.copy()
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            return objective_and_grad(x, beta)[0]
        
        def obj_grad(x):
            return objective_and_grad(x, beta)
        
        res = minimize(obj_grad, x0, method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter_per_beta,
                                'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': max_iter_per_beta * 5})
        
        x_new = res.x
        s_new = np.exp(x_new - np.max(x_new))
        s_new = s_new / s_new.sum()
        sc_new = fast_score(s_new)
        
        print(f"    beta={beta}: L-BFGS converged={res.success}, score={sc_new:.6f}, iters={res.nit}")
        
        if sc_new < best_score:
            best_score = sc_new
            best_x = x_new.copy()
        x0 = x_new
    
    s = np.exp(best_x - np.max(best_x))
    s = s / s.sum()
    return s, best_score

def lp_direction(s, M, time_limit=2.0):
    n = len(s)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_min = j
        k_max = min(j+n-1, 2*n-2)
        A[k_min:k_max+1, j] = s[0:k_max-k_min+1]
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None

def lp_polish(s, deadline, max_rounds=20):
    """Polish using LP directions."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = conv.max()
        b = lp_direction(s, M, time_limit=2.0)
        if b is None:
            break
        b = b / b.sum()
        
        best_t = 0.0
        best_sc = fast_score(s)
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            blend = (1-t)*s + t*b
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()
            sc = fast_score(blend)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0:
            break
        
        s = (1-best_t)*s + best_t*b
        s = s / s.sum()
        
        if best_sc < best_score:
            best_score = best_sc
            best_s = s.copy()
    
    return best_s, best_score

def subgradient_polish(s, deadline, max_iter=5000):
    """Polish using exact subgradient with momentum."""
    n = len(s)
    s = s.copy() / s.sum()
    v = np.zeros(n)
    best_s = s.copy()
    best_score = fast_score(s)
    eta = 0.05
    momentum = 0.9
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        k_star = np.argmax(conv)
        
        # Exact subgradient at k_star
        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n-1, k_star)
        idx = np.arange(j_min, j_max + 1)
        g[idx] = 2.0 * s[k_star - idx]
        
        v = momentum * v + (1 - momentum) * g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        s_new = s - eta * v / v_norm
        s_new = project_simplex(s_new)
        
        sc = fast_score(s_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_s = s_new.copy()
        
        s = s_new
        
        # Adaptive eta
        if it % 100 == 99:
            eta *= 0.95
    
    return best_s, best_score

def diverse_init(n, rng, hseq=None):
    inits = []
    
    if hseq is not None:
        h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, hseq.size), hseq)
        h = np.maximum(h, 1e-10)
        inits.append(h /
