"""
FFT-accelerated Nesterov momentum optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses progressive dimension (512 -> 1024), adaptive temperature annealing, and multi-start.
Key improvements over previous approach:
- FFT-based convolution for speed
- Nesterov momentum instead of Adam (better for this smooth objective)
- Aggressive temperature annealing schedule
- Progressive dimension increase (512 -> 1024 -> 2048 if time allows)
- Multi-start with diverse initializations
- Final "sharp" polish phase with near-exact max

import numpy as np
import time


def fft_convolve_same(a):
    """Compute np.convolve(a, a) using FFT for speed."""
    n = len(a)
    n_out = 2 * n - 1
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:n_out]


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve_same(a)
    return float(2.0 * n * c.max() / (s * s))


def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_exp_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def make_power_profile(n, p):
    t = np.linspace(1e-6, 1.0, n)
    a = t ** p
    return a / a.sum()


def nesterov_optimize(p0, deadline, beta_init=0.08, beta_min=5e-5,
                      lr_init=0.012, momentum=0.92, seed=42, verbose=True):
    """
    Nesterov momentum optimization in logit space with smooth max approximation.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1, p>=0.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()

    z = np.log(p)
    z = np.clip(z, -50.0, 50.0)
    z -= z.mean()
    v = np.zeros(n)

    it = 0
    beta = beta_init
    lr_cur = lr_init

    c = fft_convolve_same(p)
    max_c = float(c.max())
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0

    t_start = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve_same(p)
        max_c = float(c.max())

        # Soft max weights for differentiating the max
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_exp = ex.sum()
        if s_exp < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_exp

        # Gradient: grad_p[i] = 2 * sum_k w[i+k] * p[k]
        h = np.correlate(w, p, mode='valid')
        grad_p = 2.0 * h

        # Project to logit space (preserves sum constraint)
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            v = np.zeros(n)
            it = 0
            lr_cur = lr_init
            beta = beta_init
            continue

        # Nesterov step
        v = momentum * v - lr_cur * gz
        z = z + v
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            v = np.zeros(n)
            continue
        p = p / p_sum

        c = fft_convolve_same(p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive learning rate
        if no_imp > 0 and no_imp % 1000 == 0:
            lr_cur = max(0.002, lr_cur * 0.92)

        # Anneal temperature (aggressive schedule)
        if it % 300 == 0:
            beta = max(beta_min, beta * 0.75)

        # Periodic perturbation to escape local minima
        if no_imp > 3000:
            rng = np.random.default_rng(it * 7 + seed)
            z = best_z + 0.08 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            v = np.zeros(n)
            it = 0
            lr_cur = lr_init * 0.8
            no_imp = 0
            beta = max(1e-3, 0.05 * float(fft_convolve_same(best_p).max()))

        # Stronger restart if very stuck
        if no_imp > 8000:
            rng = np.random.default_rng(it * 13 + seed)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            v = np.zeros(n)
            it = 0
            lr_cur = lr_init
            no_imp = 0
            beta = beta_init * 0.5

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      iter {it}: score={best_score:.8f}, beta={beta:.6f}, "
                  f"lr={lr_cur:.4f}, elapsed={elapsed:.0f}s")

    return best_p, best_score


def sharp_polish(p0, deadline, verbose=True):
    """
    Final polish with very low temperature (near-exact max) and small steps.
    Uses a cosine-annealed learning rate schedule.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()

    z = np.log(p)
    z = np.clip(z, -60.0, 60.0)
    z -= z.mean()
    v = np.zeros(n)

    it = 0
    beta = 5e-4
    lr_init = 0.005
    lr_final = 0.0001
    momentum = 0.95

    c = fft_convolve_same(p)
    max_c = float(c.max())
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0

    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)

    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, (time.monotonic() - t_start) / total_time)

        # Cosine annealing for learning rate
        lr_cur = lr_final + 0.5 * (lr_init - lr_final) * (1.0 + np.cos(np.pi * progress))

        c = fft_convolve_same(p)
        max_c = float(c.max())

        # Near-exact max: use very small beta
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_exp = ex.sum()
        if s_exp < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_exp

        h = np.correlate(w, p, mode='valid')
        grad_p = 2.0 * h

        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            v = np.zeros(n)
            continue

        v = momentum * v - lr_cur * gz
        z = z + v
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = fft_convolve_same(p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      polish iter {it}: score={best_score:.8f}, "
                  f"lr={lr_cur:.6f}, elapsed={elapsed:.0f}s")

    return best_p, best_score


def coordinate_refine(p0, deadline, n_trials=20000, seed=123, verbose=True):
    """
    Try random coordinate perturbations to find improvements.
    """
    n = len(p0)
    p = p0.copy()
    best_score = internal_score(p)
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() < deadline and it < n_trials:
        it += 1
        # Perturb 1-3 random coordinates multiplicatively
        n_pert = rng.integers(1, 4)
        idx = rng.integers(0, n, size=n_pert)
        p_trial = p.copy()

        for i in idx:
            factor = 1.0 + 0.15 * rng.normal()
            p_trial[i] = max(0.0, p_trial[i] * factor)

        p_trial = normalize(p_trial)
        if p_trial is None:
            continue

        score = internal_score(p_trial)
        if score < best_score:
            p = p_trial
            best_score = score

        if verbose and it % 5000 == 0:
            print(f"      coord iter {it}: best={best_score:.8f}")

    return p, best_score

            print(f"\nPhase 2: Nesterov at n={n2}, initial score={internal_score(a2):.8f}")
            p2, sc2 = nesterov_optimize(a2, deadline_p2,
                                        beta_init=0.06, beta_min=3e-5,
                                        lr_init=0.01, momentum=0.93,
                                        seed=200, verbose=True)
            if sc2 < best_score:
                best_score = sc2
                best_seq = p2
            print(f"Phase 2 result: score={best_score:.8f}")

    # Phase 3: Try n=2048 if time allows
    n3 = 2048
    deadline_p3 = min(deadline, t_start + 1650.0)
    if time.monotonic() < deadline_p3 - 120:
        a3 = interpolate_seq(best_seq, n3)
        if a3 is not None:
            print(f"\nPhase 3: Nesterov at n={n3}, initial score={internal_score(a3):.8f}")
            p3, sc3 = nesterov_optimize(a3, deadline_p3,
                                        beta_init=0.05, beta_min=2e-5,
                                        lr_init=0.008, momentum=0.94,
                                        seed=300, verbose=True)
            if sc3 < best_score:
                best_score = sc3
                best_seq = p3
            print(f"Phase 3 result: score={best_score:.8f}")

    # Phase 4: Sharp polish on the best sequence found
    if time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        print(f"\nPhase 4: Sharp polish at n={n_final}, score={internal_score(best_seq):.8f}")
        p_final, sc_final = sharp_polish(best_seq, deadline - 5, verbose=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = p_final
        print(f"Phase 4 result: score={best_score:.8f}")

    # Phase 5: Quick coordinate refinement if time remains
    if time.monotonic() < deadline - 5:
        p_coord, sc_coord = coordinate_refine(best_seq, deadline - 2,
                                              n_trials=5000, seed=999, verbose=False)
        if sc_coord < best_score:
            best_score = sc_coord
            best_seq = p_coord
        print(f"Phase 5 result: score={best_score:.8f}")

    # Final validation and cleanup
    final_score = internal_score(best_seq)
    print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.8f} ===")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
