from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternating minimization with subgradient polishing.
Exploits the insight that non-symmetric (monotone) sequences can have
convolution peaks below the uniform baseline of score=2.

def compute_score(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))

def compute_conv_max(f):
    return float(np.convolve(f, f).max())

def normalize(f):
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

def make_power_profile(n, p):
    """f[i] = ((i+1)/n)^p, normalized."""
    t = np.arange(1, n+1, dtype=np.float64) / n
    f = t ** p
    f = np.maximum(f, 1e-15)
    return f / f.sum()

def make_exp_profile(n, alpha):
    """f[i] = exp(alpha * i/(n-1)), normalized."""
    t = np.arange(n, dtype=np.float64) / max(n-1, 1)
    f = np.exp(alpha * t)
    return f / f.sum()

def make_sine_profile(n, p):
    """f[i] = sin(pi*(i+1)/(2n+2))^p, normalized."""
    t = np.arange(1, n+1, dtype=np.float64)
    f = np.sin(np.pi * t / (2*n + 2)) ** p
    f = np.maximum(f, 1e-15)
    return f / f.sum()

def make_log_profile(n, alpha):
    """f[i] = log(1 + alpha*i), normalized."""
    t = np.arange(n, dtype=np.float64)
    f = np.log1p(alpha * t)
    f = np.maximum(f, 1e-15)
    return f / f.sum()

def get_starting_points(n, rng, count=20):
    starts = []
    # Power profiles
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0]:

def build_conv_matrix_sparse(f):
    """Build sparse matrix for linear convolution with f.
    Returns sparse matrix A of shape (2n-1, n) such that A @ g = conv(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi+1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_minimize_peak(f, current_M=None):
    """Solve LP: minimize t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt)."""
    n = len(f)
    A_conv = build_conv_matrix_sparse(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize: t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: (f*g)[k] - t <= 0 for each k
    # A_ub has shape (2n-1, n+1)
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        A_ub[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2*n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x[:n], 0.0)
        t_opt = res.x[n]
        return g, float(t_opt)
    return None, None

    f = normalize(f_init.copy())
    if f is None:

    n = len(f)

                break
            continue
        
        g = normalize(g)

            no_imp += 1
            continue
        
        # Blend f towards g

        best_M_try = M
        best_f_try = f
        for step in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            f_try = (1.0 - step) * f + step * g
            f_try = normalize(f_try)
            if f_try is None:
                continue
            M_try = compute_conv_max(f_try)
            if M_try < best_M_try:
                best_M_try = M_try
                best_f_try = f_try

                best_f = f.copy()

import numpy as np
from scipy.optimize import linprog
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternating minimization (minimize peak formulation) combined
with subgradient Adam polishing in log domain. Exploits the insight that
non-symmetric monotone sequences can achieve convolution peaks well below
the uniform baseline (score=2). Tries multiple sequence lengths and diverse
starting profiles.

def compute_score(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_conv_max(f):
    return float(np.convolve(f, f).max())


def normalize_nonneg(f):
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_power_profile(n, p):
    t = np.arange(1, n + 1, dtype=np.float64) / n
    f = t ** p
    f = np.maximum(f, 1e-15)
    return f / f.sum()


def make_exp_profile(n, alpha):
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    f = np.exp(alpha * t)
    return f / f.sum()


def make_sine_profile(n, p):
    t = np.arange(1, n + 1, dtype=np.float64)
    f = np.sin(np.pi * t / (2 * n + 2)) ** p
    f = np.maximum(f, 1e-15)
    return f / f.sum()


def make_log_profile(n, alpha):
    t = np.arange(n, dtype=np.float64)
    f = np.log1p(alpha * t)
    f = np.maximum(f, 1e-15)
    return f / f.sum()


def get_starting_points(n, rng, count=15):
    starts = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0]:
        starts.append(make_power_profile(n, p))
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 16.0]:
        starts.append(make_exp_profile(n, alpha))
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        starts.append(make_sine_profile(n, p))
    for alpha in [1.0, 5.0, 10.0, 50.0, 200.0]:
        starts.append(make_log_profile(n, alpha))
    # Reversed versions of some
    for idx in [0, 3, 6, 10, 15]:
        if idx < len(starts):
            starts.append(starts[idx][::-1].copy())
    # Random smooth monotone (positive polynomial)
    for _ in range(count):
        deg = int(rng.integers(2, 8))
        coeffs = np.abs(rng.normal(size=deg + 1))
        coeffs = coeffs / coeffs.sum()
        t = np.linspace(0, 1, n)
        f = np.zeros(n)
        for d in range(deg + 1):
            f += coeffs[d] * t ** d
        f = np.maximum(f, 1e-10)
        starts.append(f / f.sum())
    return starts


def lp_minimize_peak(f, time_limit=5.0):
    n = len(f)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            t_opt = float(res.x[n])
            if g.sum() > 1e-15:
                return g / g.sum(), t_opt
    except Exception:
        pass
    return None, None


def lp_maximize_sum(f, M, time_limit=5.0):
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def lp_alternate(f_init, deadline, rng, max_iters=500, use_minpeak=True):

    if f is None:

    n = len(f)

    best_f = f.copy()
    best_M = M
    no_imp = 0
    lp_time = max(1.0, min(5.0, 300.0 / max(n, 1)))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = None
        if use_minpeak:

        if g is None:

            no_imp += 1
            if no_imp > 30:

            no_imp += 1
            continue

def subgradient_corr(w, f):
    f_rev = f[::-1]
    conv_wf = np.convolve(w, f_rev)
    n = len(f)
    return conv_wf[n - 1:2 * n - 1]


def subgradient_polish(f_init, deadline, rng, lr=0.02, max_iters=10000):
    f = np.maximum(f_init, 1e-15)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-12, 0.05 * M)
    best_f = f.copy()
    best_M = M
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    no_imp = 0
    base_lr = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c_max = float(c.max())
        x = np.clip((c - c_max) / beta, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        corr = subgradient_corr(w, f)
        grad_f = 2.0 * corr + 1e-10 * f
        dot = float(np.dot(f, grad_f))
        grad_z = f * (grad_f - dot)
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp += 100
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(best_f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp += 100
            continue

        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-13:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 100 == 0:
            beta = max(1e-12, beta * 0.85)
        if no_imp > 300:
            lr = max(0.002, lr * 0.9)
        if no_imp > 800:
            z = np.log(best_f)
            z = 0.8 * z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.05 * M)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_f, best_M


def interpolate_profile(f, n_new):

        return f.copy()
    x0 = np.linspace(0, 1, n0)

    return np.interp(x1, x0, f)

            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n1))

    for s in starts:
        sc = compute_score(s)
        try_update(s, sc)
    print(f"  Initial best: {best_score:.10f}", flush=True)

    results = []
    n_starts = len(starts)
    for i, s in enumerate(starts):
        remaining = phase1_end - time.monotonic()
        if remaining < 8:
            break
        alloc = min(remaining / max(1, n_starts - i), 25.0)
        if alloc < 4:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = lp_alternate(s, run_dl, rng, max_iters=200)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))
        if (i + 1) % 5 == 0:
            print(f"  {i+1}/{n_starts}: best={best_score:.10f}", flush=True)

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}", flush=True)

    # Phase 2: Extended LP from top results at n=256
    phase2_end = min(deadline, t_start + 800.0)
    print(f"Phase 2: Extended LP at n={n1}", flush=True)

    top_k = min(5, len(results))
    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, top_k - i), 60.0)
        if alloc < 10:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = lp_alternate(results[i][1], run_dl, rng, max_iters=500)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))
        print(f"  LP ext {i+1}/{top_k}: score={sc:.10f}", flush=True)

    results.sort(key=lambda x: x[0])

    # Phase 3: Subgradient polish at n=256
    phase3_end = min(deadline, t_start + 1150.0)
    print(f"Phase 3: Subgradient polish at n={n1}", flush=True)

    top_p = min(8, len(results))
    results2 = []
    for i in range(top_p):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_p - i), 30.0)
        if alloc < 5:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = subgradient_polish(results[i][1], run_dl, rng, lr=0.02)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results2.append((sc, f_best))
        print(f"  Polish {i+1}/{top_p}: score={sc:.10f}, best={best_score:.10f}", flush=True)

    results2.sort(key=lambda x: x[0])
    all_results = sorted(results + results2, key=lambda x: x[0])
    print(f"Phase 3 done. Best: {best_score:.10f}", flush=True)

    # Phase 4: Extend to n=512
    n4 = 512
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"Phase 4: Extension to n={n4}", flush=True)

    results4 = []
    top_e = min(4, len(all_results))
    for i in range(top_e):
        remaining = phase4_end - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining / max(1, top_e - i), 50.0)
        if alloc < 10:
            break
        f_ext = interpolate_profile(all_results[i][1], n4)
        run_dl = time.monotonic() + alloc
        f_best, M_best = lp_alternate(f_ext, run_dl, rng, max_iters=300)
        sc = 2.0 * n4 * M_best
        try_update(f_best, sc)
        results4.append((sc, f_best))
        print(f"  Ext LP {i+1}/{top_e}: score={sc:.10f}, best={best_score:.10f}", flush=True)

    # Polish extended results
    if len(results4) > 0:
        results4.sort(key=lambda x: x[0])
        for i in range(min(3, len(results4))):
            remaining = phase4_end - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining / max(1, min(3, len(results4)) - i), 25.0)
            if alloc < 5:
                break
            run_dl = time.monotonic() + alloc
            f_best, M_best = subgradient_polish(results4[i][1], run_dl, rng, lr=0.015)
            sc = 2.0 * n4 * M_best
            try_update(f_best, sc)
            print(f"  Ext Polish {i+1}: score={sc:.10f}, best={best_score:.10f}", flush=True)

    # Phase 5: Try n=128
    n5 = 128
    if time.monotonic() < deadline - 100:
        phase5_end = min(deadline - 30, time.monotonic() + 80)
        print(f"Phase 5: Try n={n5}", flush=True)
        for i in range(min(3, len(all_results))):
            remaining = phase5_end - time.monotonic()
            if remaining < 8:
                break
            f_ext = interpolate_profile(all_results[i][1], n5)
            run_dl = time.monotonic() + min(remaining / 3, 20.0)
            f_best, M_best = lp_alternate(f_ext, run_dl, rng, max_iters=300)
            sc = 2.0 * n5 * M_best
            try_update(f_best, sc)
            print(f"  n={n5}: score={sc:.10f}, best={best_score:.10f}", flush=True)

    # Phase 6: Final refinement on best solution
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"Phase 6: Final refinement at n={n_best}, time={remaining:.0f}s", flush=True)
        run_dl = time.monotonic() + max(10, remaining * 0.4)
        f_best, M_best = subgradient_polish(best_seq, run_dl, rng, lr=0.01)
        sc = 2.0 * n_best * M_best
        try_update(f_best, sc)
        print(f"  Final polish: {sc:.10f}, best={best_score:.10f}", flush=True)

        if time.monotonic() < deadline - 15:
            run_dl2 = time.monotonic() + max(10, (deadline - time.monotonic()) * 0.5)
            f_best, M_best = lp_alternate(best_seq, run_dl2, rng, max_iters=500)
            sc = 2.0 * n_best * M_best
            try_update(f_best, sc)
            print(f"  Final LP: {sc:.10f}, best={best_score:.10f}", flush=True)

    # Ensure validity
    if best_seq is None:
        best_seq = make_power_profile(256, 1.0)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_power_profile(256, 1.0)
        s = float(best_seq.sum())
    best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}", flush=True)

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}", flush=True)
    except Exception as e:
        print(f"evaluate_sequence error: {e}", flush=True)

    return [float(x) for x in best_seq]
