"""
Improved minimization of 2n*max(conv(a,a))/sum(a)^2 using Frank-Wolfe LP,
subgradient escape, and adaptive gradient polish. Key innovations:
- Minimax LP subproblem (only active constraints) for faster, more focused optimization
- Adaptive active set sizing for stability
- Subgradient-based escape from LP local minima
- Power-based smooth max for better gradient approximation
- Multi-length and multi-init strategy

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _conv_row(a, k):
    """Return the row of the convolution matrix: (a*g)[k] = row @ g."""
    n = len(a)
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    for j in range(lo, hi + 1):
        row[j] = a[k - j]
    return row


def _fw_lp_subproblem(a, active_set):
    """
    Solve the Frank-Wolfe minimax subproblem:
    min t s.t. (a*g)[k] <= t for k in active_set, sum(g)=1, g>=0
    Returns (g, t) or (None, inf) on failure.
    """
    n = len(a)
    m = len(active_set)
    if m == 0:
        return np.ones(n) / n, 0.0

    # Variables: x = [t, g_0, ..., g_{n-1}]
    # Minimize: [1, 0, ..., 0] @ x
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

    # Inequality: (a*g)[k] - t <= 0 for each active k
    # => -t + sum_j A[k,j]*g_j <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    for i, k in enumerate(active_set):
        A_ub[i, 1:] = _conv_row(a, k)
    b_ub = np.zeros(m)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    # Bounds: t in [0, inf], g_j in [0, inf]
    bounds = [(0.0, None)] + [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success and res.x is not None:
        t_val = float(res.x[0])
        g = np.maximum(res.x[1:], 0.0)
        gs = float(g.sum())
        if gs < 1e-15:
            return None, np.inf
        g = g / gs
        return g, t_val
    return None, np.inf


def _full_lp_spread(a, M):
    """Standard LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(m, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = np.maximum(res.x, 0.0)
        return g
    return None


def _fw_optimize(a, deadline, max_iter=50000, active_frac=0.15,
                 blend_init=0.25, min_blend=0.02):
    """
    Frank-Wolfe optimization with adaptive active set.
    """
    a = _normalize(a.copy())
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    blend = blend_init
    no_imp = 0
    stuck_count = 0
    tighten_counter = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Adaptive active set: fraction of top elements
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(int(active_frac * len(c)), len(c) - 1)]
        active = np.where(c >= threshold)[0]
        # Ensure at least 2 active
        if len(active) < 2:
            top2 = np.argsort(c)[::-1][:2]
            active = top2

        # Solve FW subproblem
        g, t_star = _fw_lp_subproblem(a, active)

        if g is None:
            stuck_count += 1
            if stuck_count > 100:
                # Perturb and restart
                a = _normalize(a * (1.0 + 0.05 * np.random.default_rng(it).normal(size=n)))
                stuck_count = 0
            continue

        # Check improvement: t_star < M means the LP found a better direction
        if t_star >= M * (1.0 - 1e-10):
            no_imp += 1
            stuck_count += 1

            if no_imp > 50 and stuck_count % 20 == 0:
                # Subgradient escape
                c = _fft_convolve(a)
                M_cur = float(c.max())
                # Subgradient: average gradient over active peaks
                n_active = int(max(2, 0.05 * len(c)))
                top_k = np.argsort(c)[::-1][:n_active]
                grad = np.zeros(n)
                for k in top_k:
                    grad += _conv_row(a, k)
                grad /= len(top_k)
                # Step in negative gradient direction
                lr = 0.01 * M_cur / (np.linalg.norm(grad) + 1e-15)
                a_new = a - lr * grad
                a_new = np.maximum(a_new, 0.0)
                a = _normalize(a_new)
                no_imp = 0
                stuck_count = 0
            elif no_imp > 300:
                # Bigger perturbation
                rng = np.random.default_rng(hash(it) % (2**31))
                a = _normalize(a * (1.0 + 0.1 * rng.normal(size=n)))
                no_imp = 0
                stuck_count = 0
            continue

        # Good direction found - blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            a = a.copy()
            no_imp += 1
            continue
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            best_score = float(2.0 * n * best_M)
            no_imp = 0
            stuck_count = 0
            # Gradually increase blend as we make progress
            blend = min(0.5, blend * 1.02)
        else:
            no_imp += 1
            blend = max(min_blend, blend * 0.95)

        a = a_new

    return best_a


def _smooth_max_power(c, p):
    """Power-based smooth max: (sum c^p)^(1/p)."""
    c = np.asarray(c, dtype=np.float64)
    c_pos = np.maximum(c, 1e-30)
    cp = c_pos ** p
    s = float(cp.sum())
    if s < 1e-30:
        return 0.0, np.zeros_like(c)
    val = s ** (1.0 / p)
    w = cp / s
    return val, w


def _gradient_polish(a, deadline, max_iter=30000, p_init=20.0):
    """
    Gradient descent on the power-smooth max of conv(a,a),
    projected to simplex {a>=0, sum(a)=1}.
    """
    a = _normalize(a.copy())
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    p = p_init
    lr = 0.015
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        _, w = _smooth_max_power(c, p)

        # Gradient of (sum w_k * c_k) w.r.t. a
        # c_k = sum_i a[i]*a[k-i], grad_c_k = 2*a_rev shifted
        # Use FFT: grad_j = 2 * sum_k w_k * a[k-j] (with bounds)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Small regularization to prevent trivial solutions
        grad += 1e-8 * a

        # Momentum step
        v = momentum * v - lr * grad
        a_new = a + v

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s < 1e-15:
            a = best_a.copy()
            v = np.zeros(n)
            no_imp = 0
            continue
        a_new = a_new / s
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(2000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        # Decay p to make smooth max sharper
        if it % 100 == 0:
            p = max(5.0, p * 0.9)

    return best_a


def _adam_logit_polish(a, deadline, max_iter=20000, lr=0.03):
    """Adam optimization in logit space with power-smooth max."""
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    p = 15.0
    no_imp = 0
    base_lr = lr

    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        _, w = _smooth_max_power(c, p)

        # Gradient in a-space
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        ga = 2.0 * h + 1e-7 * a

        # Transform to z-space: dz = a * (ga - mean(a*ga))
        dot = float(np.dot(a, ga))
        gz = a * (ga - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m, v_arr, t_adam = np.zeros(n), np.zeros(n), 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v_arr / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m, v_arr, t_adam = np.zeros(n), np.zeros(n), 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 150 == 0:
            p = max(5.0, p * 0.85)

        if no_imp > max(3000, 5 * n):
            rng = np.random.default_rng(it * 7 + 3)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            p = 15.0
            m, v_arr, t_adam = np.zeros(n), np.zeros(n), 0
            lr = base_lr
            no_imp = 0

    return best_a


def _make_ramp(n, exponent=1.0):
    """Create a ramp-like profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.5 * t ** exponent
    return a / a.sum()


def _make_arch(n, depth=1.0):
    """Create an arch profile (high at edges, low in middle)."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - depth * (1.0 - 4.0 * (t - 0.5) ** 2)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_cosine(n, freq=1.0, phase=0.0):
    """Cosine-modulated profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _subgradient_round(a, n_rounds=50, lr_scale=0.005):
    """Pure subgradient method for escape."""
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_M = float(_fft_convolve(a).max())

    for r in range(n_rounds):
        c = _fft_convolve(a)
        M = float(c.max())
        # Use top-fraction as active set
        n_act = max(3, int(0.03 * len(c)))
        top_k = np.argsort(c)[::-1][:n_act]
        grad = np.zeros(n)
        for k in top_k:
            grad += _conv_row(a, k)
        grad /= len(top_k)

        lr = lr_scale * M / (np.linalg.norm(grad) + 1e-15)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _normalize(a)

        c = _fft_convolve(a)
        M_new = float(c.max())
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: FW-LP -> Subgradient Escape -> Gradient Polish -> Adam.
    Tries multiple initializations and sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"[Parent] score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Candidate lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [256, 512]:
        if ln not in lengths:
            lengths.append(ln)
    lengths.sort()

    print(f"[Config] Trying lengths: {lengths}, budget={budget_s-25:.0f}s")

    for n_t in lengths:
        if time.monotonic() > deadline - 60:
            break

        # Multiple initializations for this length
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_ramp(n_t, 1.0))
        inits.append(_make_ramp(n_t, 1.5))
        inits.append(_make_arch(n_t, 0.8))
        inits.append(_make_cosine(n_t, 1.0, 0.5))
        # Random init
        inits.append(_normalize(rng.random(size=n_t)))

        best_for_len = None
        best_for_len_score = np.inf

        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 40:
                break

            sc0 = _score(a0)
            print(f"  [n={n_t}] Init {init_idx}: score={sc0:.8f}")

            if not np.isfinite(sc0):
                continue

            # Phase A: FW-LP optimization
            fw_budget = min(180.0, deadline - time.monotonic() - 60.0)
            if fw_budget > 10:
                fw_end = time.monotonic() + fw_budget
                a_fw = _fw_optimize(a0, fw_end, max_iter=30000,
                                    active_frac=0.12, blend_init=0.3)
                sc_fw = _score(a_fw)
                print(f"    FW-LP: score={sc_fw:.10f}")
            else:
                a_fw = a0
                sc_fw = sc0

            # Phase B: Subgradient escape + second FW pass
            if time.monotonic() < deadline - 30 and sc_fw < best_for_len_score:
                a_sg = _subgradient_round(a_fw, n_rounds=30, lr_scale=0.008)
                sc_sg = _score(a_sg)
                if sc_sg < sc_fw:
                    a_fw = a_sg
                    sc_fw = sc_sg
                    print(f"    SG escape: score={sc_fw:.10f}")

                # Second FW pass on improved sequence
                fw2_budget = min(100.0, deadline - time.monotonic() - 30.0)
                if fw2_budget > 10:
                    fw2_end = time.monotonic() + fw2_budget
                    a_fw2 = _fw_optimize(a_fw, fw2_end, max_iter=20000,
                                         active_frac=0.10, blend_init=0.2)
                    sc_fw2 = _score(a_fw2)
                    if sc_fw2 < sc_fw:
                        a_fw = a_fw2
                        sc_fw = sc_fw2
                        print(f"    FW pass 2: score={sc_fw:.10f}")

            # Phase C: Gradient polish
            if time.monotonic() < deadline - 20 and sc_fw < best_for_len_score:
                gp_budget = min(60.0, deadline - time.monotonic() - 20.0)
                if gp_budget > 5:
                    gp_end = time.monotonic() + gp_budget
                    a_gp = _gradient_polish(a_fw, gp_end, max_iter=15000, p_init=25.0)
                    sc_gp = _score(a_gp)
                    if sc_gp < sc_fw:
                        a_fw = a_gp
                        sc_fw = sc_gp
                        print(f"    Grad polish: score={sc_fw:.10f}")

            # Phase D: Adam logit polish
            if time.monotonic() < deadline - 15 and sc_fw < best_for_len_score:
                ad_budget = min(40.0, deadline - time.monotonic() - 15.0)
                if ad_budget > 5:
                    ad_end = time.monotonic() + ad_budget
                    a_ad = _adam_logit_polish(a_fw, ad_end, max_iter=10000, lr=0.02)
                    sc_ad = _score(a_ad)
                    if sc_ad < sc_fw:
                        a_fw = a_ad
                        sc_fw = sc_ad
                        print(f"    Adam: score={sc_fw:.10f}")

            # Track best for this length
            if sc_fw < best_for_len_score:
                best_for_len_score = sc_fw
                best_for_len = a_fw.copy()

            # Update global best
            if sc_fw < best_score:
                best_score = sc_fw
                best_seq = a_fw.copy()

        # If we have a good result for this length, do one more refinement cycle
        if best_for_len is not None and time.monotonic() < deadline - 20:
            print(f"  [n={n_t}] Best: {best_for_len_score:.10f}, refining...")
            ref_budget = min(80.0, deadline - time.monotonic() - 20.0)
            if ref_budget > 10:
                ref_end = time.monotonic() + ref_budget
                a_ref = _fw_optimize(best_for_len, ref_end, max_iter=20000,
                                     active_frac=0.08, blend_init=0.15)
                sc_ref = _score(a_ref)
                if sc_ref < best_for_len_score:
                    best_for_len_score = sc_ref
                    best_for_len = a_ref
                    if sc_ref < best_score:
                        best_score = sc_ref
                        best_seq = a_ref.copy()
                # Quick gradient polish
                if time.monotonic() < deadline - 10:
                    a_ref2 = _gradient_polish(best_for_len, time.monotonic() + 30.0,
                                             max_iter=10000, p_init=30.0)
                    sc_ref2 = _score(a_ref2)
                    if sc_ref2 < best_score:
                        best_score = sc_ref2
                        best_seq = a_ref2.copy()
                        print(f"    Final polish: score={best_score:.10f}")

    # Final global refinement on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_budget = min(60.0, deadline - time.monotonic() - 10.0)
        if final_budget > 10:
            a_final = _fw_optimize(best_seq, time.monotonic() + final_budget,
                                   max_iter=15000, active_frac=0.05, blend_init=0.1)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final
            # One more gradient pass
            if time.monotonic() < deadline - 3:
                a_fin2 = _gradient_polish(best_seq, time.monotonic() + 15.0,
                                         max_iter=5000, p_init=40.0)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_score = sc_fin2
                    best_seq = a_fin2

    # Final cleanup
    if best_seq is None:
        best_seq = _make_ramp(256, 1.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"[Evaluation] score={sc_eval:.10f}, n={final_seq.size}")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
    except Exception as e:
        print(f"[Eval error] {e}")

    print(f"[FINAL] score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
