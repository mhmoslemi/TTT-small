    return a / a.sum()


def _make_bimodal_profile(n, center=0.5, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    a += 0.1 * np.exp(-((t - (1-center)) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal_profile(n, freq=2.0, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _perturb(a, rng, scale=0.05):
    a_pert = a * (1.0 + scale * rng.normal(size=len(a)))
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s
