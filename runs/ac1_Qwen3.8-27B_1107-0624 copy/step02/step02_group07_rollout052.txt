def power_sum_grad(s, p=4):
    """Compute gradient of (sum_k conv[k]^p)^(1/p) w.r.t. s."""
    n = len(s)
    conv = np.convolve(s, s)  # length 2n-1
    w = np.power(conv, p-1)  # weights
    F_p = np.power(np.sum(np.power(conv, p)), 1.0/p)
    if F_p < 1e-15:
        return np.zeros(n)

    W = np.fft.rfft(w, L)
