**Alternative approach: Direct subgradient method**

The function `max_k conv(a,a)(k)` is the max of linear functions (each conv(a,a)(k) is a quadratic form, but if we think of it as a function of a, it's quadratic). Actually each conv(a,a)(k) = a^T C_k a where C_k is a symmetric matrix. So we're maximizing a set of quadratic forms.

The subgradient of max_k q_k(a) at a point a where q_i(a) = max is the subdifferential of q_i, which is {2 C_i a} (for a single active constraint) or the convex hull of {2 C_i a : i in active set}.

So the subgradient of the objective (with sum(a)=1 constraint) is:
∇(max_k conv(a,a)(k)) = 2 * sum_{i in S} w_i * (C_i a) where S is the active set and w_i are weights summing to 1.

The softmax approximation uses w_i ∝ exp((q_i - max)/beta).

    n = len(a)

Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses top-k softmax gradient with L-BFGS continuation and multi-restart.

import numpy as np
import time
from scipy.optimize import minimize

# === Top-level helper functions ===

def _npx2(m):
    k = 1
    while k < m:
        k <<= 1
    return max(2, k)

def _fconv(a):
    """Full linear convolution via FFT."""
    n = a.size
    N = _npx2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c

def _xcorr(w, p):
    """Cross-correlation: result[i] = sum_k w[k]*p[k-i] for i=0..n-1.
    w has length 2n-1, p has length n."""
    n = p.size
    N = _npx2(3*n - 1)
    fw = np.fft.rfft(w, N)
    fr = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fr, N)
    return h[n-1:2*n-1]

def _sfzx(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(e) / e.size
    return e / s

def _score(p):
    """Score of a normalized sequence (sum=1)."""
    n = p.size
    c = _fconv(p)
    return 2.0 * n * float(c.max())

def _ramp(n, alpha):
    """Power-law ramp normalized to sum=1."""
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** alpha
    return a / a.sum()

def _tri(n):
    """Triangle profile."""
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    a = np.maximum(1.0 - np.abs(i - m) / m, 0.01)
    return a / a.sum()

def _interp(p, n_new):
    """Interpolate profile to new length."""
    n0 = p.size
    if n0 == n_new:
        return p.copy() / p.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, p) / np.interp(x1, x0, p).sum()

def _obj_grad(z, n, beta, topk):
    """Objective and gradient for L-BFGS on logit parameters."""
    p = _sfzx(z)
    c = _fconv(p)
    
    if topk < c.size:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(c.size)
        ct = c
    
    m = ct.max()
    x = (ct - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    w = ex / s
    val = m + beta * np.log(s)
    
    fval = 2.0 * n * val
    
    wf = np.zeros(c.size)
    wf[idx] = w
    cross = _xcorr(wf, p)
    dp = np.dot(p, cross)
    g = 4.0 * n * p * (cross - dp)
    
    return fval, g

def _lbfgs(z0, n, beta, topk, maxiter=3000, gtol=1e-10):
    """Run L-BFGS-B on the smoothed objective."""
    def fg(z):
        return _obj_grad(z, n, beta, topk)
    
    res = minimize(fg, z0, method='L-BFGS-B', jac=True,
                   options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-18})
    return res.x, res.fun

def _subgrad_step(p, n, eta, topk=5, rng=None):
    """Projected subgradient step on the simplex."""
    c = _fconv(p)
    if topk < c.size:
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(c.size)
    
    # Subgradient: 4n * sum_{k in idx} (1/topk) * d(c_k)/d(p)
    # d(c_k)/d(p_i) = 2*p[k-i] (if valid)
    # So subgrad = (4n/topk) * sum_{k in idx} 2*p[k-i] for each i
    # = (8n/topk) * sum_{k in idx} p[k-i]
    
    # More efficiently: weight each idx uniformly, compute weighted cross-correlation
    w = np.zeros(c.size)
    w[idx] = 1.0 / len(idx)
    cross = _xcorr(w, p)
    
    grad = 2.0 * cross  # subgradient of max(c) w.r.t. p (up to constant)
    
    # Projected gradient step on simplex: p_new = proj_simplex(p - eta*grad)
    p_new = p - eta * grad
    # Project onto simplex {x >= 0, sum(x) = 1}
    p_new = _proj_simplex(p_new)
    return p_new

def _proj_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u * np.arange(1, n+1) > cssv)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
