Let me think about a completely different approach: **direct optimization of the max using a bisection + LP feasibility check**.

The problem is: minimize `max_k conv(a,a)_k` subject to `sum(a) = 1`, `a >= 0`.

We can binary search on `t` (the max value). For a given `t`, we check if there exists `a >= 0`, `sum(a) = 1`, `conv(a,a)_k <= t` for all k.

The feasibility check is: does there exist `a >= 0`, `sum(a) = 1`, `a^T T_k a <= t` for all k?

This is a set of quadratic constraints, which is non-convex. So we can't just use a standard LP or SDP.

However, we can use the fact that for non-negative `a`, the quadratic constraints `a^T T_k a <= t` are "almost" linear (since `a^T T_k a = sum_j a_j * a_{k-j}` is multi-linear).

Actually, I realize that `conv(a,a)_k = sum_j a_j * a_{k-j}` is NOT a quadratic form in the standard sense. It's a sum of products of pairs of variables. Each term `a_j * a_{k-j}` is a product of two variables. This is a quadratic form: `a^T Q_k a` where `Q_k` has `Q_k[j, k-j] = 1/2` and `Q_k[k-j, j] = 1/2` (for j != k-j) or `Q_k[j,j] = 1` (for j = k-j).

For non-negative `a`, the function `a^T Q_k a` is not necessarily convex or concave. So the feasibility check is NP-hard in general.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w, mval


def _grad_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w, mval = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, mval


def _lp_refine(a, deadline, max_iter=30000, inflate_init=1.001, min_inflate=1.0002):

    a = a / s

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0

                blend = 0.35
                inflate = inflate_init
                no_imp = 0
            continue

        if s_new < 1e-10:

            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

    return best_a


def _adam_logit_polish(a, deadline, max_iter=40000, lr_init=0.1, lr_end=0.002):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.05 * float(c.max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        progress = it / max(1, max_iter - 1)
        lr = lr_end + 0.5 * (lr_init - lr_end) * (1.0 + np.cos(np.pi * progress))

        p = a
        c = _fft_convolve(p)
        w, mval = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-9 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -35.0, 35.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 80 == 0:
            beta = max(1e-9, beta * 0.75)

        if no_imp > max(3000, 4 * n):
            rng_r = np.random.default_rng(it + 12345)
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n) * 0.08
            z = np.clip(z, -35.0, 35.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-6, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp = 0

    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 1e-6
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_random_profile(n, rng):
    a = np.abs(rng.normal(size=n)) + 0.05
    return a / a.sum()

                if ad_budget > 5:
                    a_ad = _adam_logit_polish(a_lp, time.monotonic() + ad_budget,
                                              max_iter=25000, lr_init=0.08, lr_end=0.002)
                    sc_ad = _score(a_ad)
                    if np.isfinite(sc_ad) and sc_ad < best_len_score:
                        best_len_score = sc_ad
                        best_len = a_ad.copy()
                    print(f"          Adam={sc_ad:.10f}")
            elif lp_budget <= 8 and sc0 < best_len_score:
                ad_budget = min(35.0, t_len_end - time.monotonic() - 10.0)
                if ad_budget > 5:
                    a_ad = _adam_logit_polish(a0, time.monotonic() + ad_budget,
                                              max_iter=20000, lr_init=0.06, lr_end=0.003)
                    sc_ad = _score(a_ad)
                    if np.isfinite(sc_ad) and sc_ad < best_len_score:
                        best_len_score = sc_ad
                        best_len = a_ad.copy()
                    print(f"  [{idx}] init={sc0:.8f} Adam={sc_ad:.10f}")

        if best_len is not None:
            if best_len_score < best_score:
                best_score = best_len_score
                best_seq = best_len.copy()
            print(f"  >> Best n={n_t}: {best_len_score:.10f}")

    # Final polishing phase
    if best_seq is not None:
        print(f"\n=== Final polish (score={best_score:.10f}) ===")

        # Coordinate polish
        if time.monotonic() < deadline - 25:
            cd_budget = min(40.0, deadline - time.monotonic() - 20.0)
            if cd_budget > 5:
                a_cd = _coordinate_polish(best_seq, time.monotonic() + cd_budget, max_iter=8000)
                sc_cd = _score(a_cd)
                if np.isfinite(sc_cd) and sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()
                print(f"  Coord: {sc_cd:.10f}")

        # LP on best
        if time.monotonic() < deadline - 20:
            lp_budget = min(30.0, deadline - time.monotonic() - 15.0)
            if lp_budget > 5:
                a_lp = _lp_refine(best_seq, time.monotonic() + lp_budget,
                                  max_iter=8000, inflate_init=1.0005, min_inflate=1.0001)
                sc_lp = _score(a_lp)
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                print(f"  LP: {sc_lp:.10f}")

        # Adam on best
        if time.monotonic() < deadline - 15:
            ad_budget = min(30.0, deadline - time.monotonic() - 10.0)
            if ad_budget > 5:
                a_ad = _adam_logit_polish(best_seq, time.monotonic() + ad_budget,
                                          max_iter=15000, lr_init=0.04, lr_end=0.001)
                sc_ad = _score(a_ad)
                if np.isfinite(sc_ad) and sc_ad < best_score:
                    best_score = sc_ad
                    best_seq = a_ad.copy()
                print(f"  Adam: {sc_ad:.10f}")

        # One more coordinate pass
        if time.monotonic() < deadline - 10:
            cd_budget = min(20.0, deadline - time.monotonic() - 5.0)
            if cd_budget > 3:
                a_cd = _coordinate_polish(best_seq, time.monotonic() + cd_budget, max_iter=5000)
                sc_cd = _score(a_cd)
                if np.isfinite(sc_cd) and sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()
                print(f"  Coord2: {sc_cd:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
    except Exception:
        pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
