"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses polynomial parameterization for fast shape exploration,
LP-based descent for local optimization, and L-BFGS-B polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_conv(c, beta):
    """Compute soft-max and weights of convolution array."""
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

    n = len(p)

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=False):
    """LP-based optimization loop."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = full_convolve_fft(f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = best_M if stagnant > 0 else float(full_convolve_fft(f).max())

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Line search for best t
        best_t = t
        best_M_new = float(full_convolve_fft((1-t)*f + t*g).max())
        
        # Try a few values of t
        for t_try in [t*0.5, t*0.75, t*1.25, t*1.5]:
            t_try = min(1.0, max(0.01, t_try))
            f_try = (1-t_try)*f + t_try*g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                continue
            f_try = f_try / s_try
            M_try = float(full_convolve_fft(f_try).max())
            if M_try < best_M_new:
                best_M_new = M_try
                best_t = t_try

        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new

        M_new = float(full_convolve_fft(f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 50 == 0:
                print(f"  LP iter {iteration}: M={best_M:.10f}, t={best_t:.4f}")
        else:
            t *= 0.5
            if t < 0.002:
                t = 0.002
            stagnant += 1

        t = max(0.005, t * 0.999)

        if stagnant > 100:
            noise_scale = 0.05 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            stagnant = 0
            if verbose:
                print(f"  LP restart at iter {iteration}, M={best_M:.10f}")

    return best_f


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def adam_polish(p0, deadline, rng, lr=0.08, reg=1e-7, beta_init=None):
    """Adam polish in logit space with soft-max objective."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-12, 0.1 * maxc)

    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = soft_max_conv(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_p.copy()
            z = np.log(np.maximum(z, 1e-12))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_p, 1e-12))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.005, lr * 0.95)
        else:
            z = np.log(np.maximum(best_p, 1e-12))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 40 == 0:
            beta = max(1e-14, beta * 0.75)

        if no_imp > max(500, 3 * n):
            z = 0.7 * np.log(np.maximum(best_p, 1e-12)) + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.15 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = lr
            no_imp = 0

    return best_p, best_score
