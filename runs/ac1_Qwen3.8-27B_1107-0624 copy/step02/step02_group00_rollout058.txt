"""
Multi-length LP + Gradient + Peak-shaving optimization for minimizing
2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous:
1. Try multiple sequence lengths (512, 768, 1024) - longer sequences have
   lower theoretical bounds
2. More aggressive LP with adaptive inflation
3. Peak-targeted sub-optimization
4. Better gradient scheduling

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a, n):
    """Build the matrix A such that A @ g = conv(a, g) for g of length n."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a, n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=50000, inflate=1.0005, blend=0.3,
               reduce_blend_every=200):
    """LP-based refinement: iteratively solve LP to find spreading direction."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        
        # Periodically tighten the bound
        if no_imp > 0 and no_imp % reduce_blend_every == 0:
            inflate_cur = max(1.0001, inflate_cur * 0.9)
            blend_cur = max(0.1, blend_cur * 0.85)

        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 1000:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(500, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a, deadline, max_iter=50000, lr=0.03, momentum=0.8):
    """Gradient descent on smooth max of convolution, projected to simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.05 * float(c.max()))
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.93)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 300 == 0:
            beta = max(1e-8, beta * 0.7)

    return best_a


def _peak_shave(a, deadline, n_peaks=5, max_iter=5000):

    a = np.asarray(a, dtype=np.float64).copy()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_smooth_bell(n, width=0.5):
    """Create a bell-shaped profile."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    return a / a.sum()


def _make_sinusoidal(n, freq=1.0, asym=0.3):
    """Create a sinusoidal-modulated profile."""
    t = np.linspace(0, 1, n)
    a = 1.0 + asym * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _adam_polish(a, deadline, max_iter=50000, lr=0.03):
    """Adam optimization in logit space for better convergence."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 800 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 400 == 0:
            beta = max(1e-8, beta * 0.7)

        if no_imp > max(5000, 5 * n):
            rng = np.random.default_rng(it * 7919)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n) * 0.5
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-6, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a, n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=50000, inflate=1.0005, blend=0.3):

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a, deadline, max_iter=50000, lr=0.03, momentum=0.8):

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.05 * float(c.max()))
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()

                no_imp = 0

    return best_a


def _peak_shave(a, deadline, n_peaks=5):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a

    no_imp = 0

        M = float(c.max())

        peak_indices = np.argsort(c)[::-1][:n_peaks]
        w = np.zeros(2 * n - 1)
        for idx in peak_indices:
            w[idx] = 1.0 / n_peaks

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_new = _project_simplex(a - lr * grad)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-10:
            best_score = score_new

            a = a_new

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_smooth_bell(n, width=0.5):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    return a / a.sum()


def _make_sinusoidal(n, freq=1.0, asym=0.3):
    t = np.linspace(0, 1, n)
    a = 1.0 + asym * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _adam_polish(a, deadline, max_iter=50000, lr=0.03):

    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 800 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 400 == 0:

            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-6, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0

def _langevin_explore(a, deadline, max_iter=20000, lr=0.01, sigma=0.05):

    best_a = a.copy()
    c = _fft_convolve(a)

    beta = max(1e-6, 0.1 * float(c.max()))
    no_imp = 0

    base_sigma = sigma

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_new = _project_simplex(a - lr * grad + noise)

            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

            a = best_a.copy()
            sigma = base_sigma
            beta = max(1e-6, 0.1 * float(_fft_convolve(a).max()))
            no_imp = 0

    return best_a

        if time.monotonic() < gp_deadline:
            print(f"  Gradient polish: budget={gp_deadline - time.monotonic():.0f}s")
            a_gp = _gradient_polish(a_lp, gp_deadline, max_iter=30000, lr=0.025)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_init_score:
                best_init_score = sc_gp
                best_init = a_gp.copy()
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  After gradient: {sc_gp:.10f}")
        else:
            a_gp = a_lp
            sc_gp = sc_lp

        # Phase 3: Peak shaving
        ps_deadline = min(phase_deadline - 30, t_start + (time.monotonic() - t_start) + 0.08 * alloc)
        if time.monotonic() < ps_deadline:
            print(f"  Peak shave: budget={ps_deadline - time.monotonic():.0f}s")
            a_ps = _peak_shave(a_gp, ps_deadline, n_peaks=5)
            sc_ps = _score(a_ps)
            if np.isfinite(sc_ps) and sc_ps < best_init_score:
                best_init_score = sc_ps
                best_init = a_ps.copy()
            if np.isfinite(sc_ps) and sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            print(f"  After peak shave: {sc_ps:.10f}")

        # Phase 4: Langevin exploration (to escape local minima)
        if time.monotonic() < phase_deadline - 20:
            lev_deadline = phase_deadline - 20
            lev_budget = lev_deadline - time.monotonic()
            if lev_budget > 30:
                print(f"  Langevin explore: budget={lev_budget:.0f}s")
                a_lev = _langevin_explore(best_init, lev_deadline, max_iter=10000,
                                          lr=0.008, sigma=0.03)
                sc_lev = _score(a_lev)
                if np.isfinite(sc_lev) and sc_lev < best_init_score:
                    best_init_score = sc_lev
                    best_init = a_lev.copy()
                if np.isfinite(sc_lev) and sc_lev < best_score:
                    best_score = sc_lev
                    best_seq = a_lev.copy()
                print(f"  After Langevin: {sc_lev:.10f}")

        print(f"  Best for n={n_t}: score={best_init_score:.10f}")

    # Phase 5: Final Adam polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        ad_deadline = deadline - 5
        ad_budget = ad_deadline - time.monotonic()
        if ad_budget > 20:
            print(f"\n{'='*60}")
            print(f"Final Adam polish: budget={ad_budget:.0f}s, current best={best_score:.10f}")
            print(f"{'='*60}")
            a_ad = _adam_polish(best_seq, ad_deadline, max_iter=50000, lr=0.025)
            sc_ad = _score(a_ad)
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Adam result: {sc_ad:.10f}")

    # Phase 6: One more short LP pass on the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_dead2 = deadline - 2
        if time.monotonic() < lp_dead2:
            a_final_lp = _lp_refine(best_seq, lp_dead2, max_iter=5000,
                                    inflate=1.0002, blend=0.15)
            sc_final = _score(a_final_lp)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final_lp.copy()
            print(f"  Final LP: {sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_smooth_bell(512, 0.5)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_smooth_bell(512, 0.5)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            print(f"Evaluate_sequence returned non-finite: {sc_eval}")
    except Exception as e:
        print(f"Evaluate_sequence error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
