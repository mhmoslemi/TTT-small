```python
"""
Multi-phase optimization: LP direction + L-BFGS-B p-norm + Adam polish.
Improves on previous LP+Adam by adding second-order method and better restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def make_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (2.0*t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def lp_direction(f, M, timeout=10.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_iterate(f, deadline, t_init=0.4, t_min=0.02, verbose=True):
    n = len(f)
    f = normalize(f.astype(np.float64).copy())
    if f is None:
        return None, np.inf
    M = float(np.convolve(f, f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init
    it = 0
    while time.monotonic() < deadline and it < 2000:
        it += 1
        g_raw = lp_direction(f, M)
        if g_raw is None:

        g = normalize(g_raw)
        if g is None:

        best_new_score = best_score
        for t in [t_step, t_step*0.5, t_step*1.5, 0.1, 0.3, 0.7]:

def pnrm_value_grad(a, p):
    """
    Compute 2*n*(sum c^p)^(1/p) and its gradient, where c = conv(a,a).
    a is assumed to be normalized (sum=1).
    """
    n = len(a)
    c = np.convolve(a, a)
    c = np.maximum(c, 1e-30)
    cp = c ** p
    sum_cp = cp.sum()
    pnrm = sum_cp ** (1.0/p)
    # d pnrm / d c_i = c_i^(p-1) / sum_cp^((p-1)/p)
    grad_c = c ** (p-1) / (sum_cp ** ((p-1)/p))
    # d conv[k] / d a[i] = 2*a[k-i] (for valid k-i)
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(grad_c[i:i+n], a)
    value = 2.0 * n * pnrm
    return value, grad_a

def lbfgsb_polish(a, deadline, p_start=4, p_end=64, verbose=True):
    """
    L-BFGS-B optimization in log-space with p-norm smoothing.
    a is normalized (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    best_a = a.copy()
    best_score = internal_score(a)
    
    p_vals = [p_start]
    while p_vals[-1] < p_end:
        p_vals.append(min(p_vals[-1] * 2, p_end))
    
    for p in p_vals:
        if time.monotonic() > deadline - 30:
            break
        
        def obj(z):
            a_curr = np.exp(z - z.max())
            a_curr = a_curr / a_curr.sum()
            c = np.convolve(a_curr, a_curr)
            c = np.maximum(c, 1e-30)
            return 2.0 * n * (c**p).sum() ** (1.0/p)
        
        def grad(z):
            a_curr = np.exp(z - z.max())
            a_curr = a_curr / a_curr.sum()
            c = np.convolve(a_curr, a_curr)
            c = np.maximum(c, 1e-30)
            cp = c ** p
            sum_cp = cp.sum()
            grad_c = c ** (p-1) / (sum_cp ** ((p-1)/p))
            grad_a = np.zeros(n)
            for i in range(n):
                grad_a[i] = 2.0 * np.dot(grad_c[i:i+n], a_curr)
            dot = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - dot)
            return 2.0 * n * (sum_cp ** (1.0/p - 1.0)) * grad_z
        
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8, 'maxfun': 1000})
        
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        sc = internal_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new
            z = result.x - result.x.mean()
            if verbose:
                print(f"    L-BFGS-B p={p}: score={best_score:.8f}")
        else:
            if verbose:
                print(f"    L-BFGS-B p={p}: no improvement (score={sc:.8f} vs {best_score:.8f})")
    
    return best_a, best_score

def adam_polish(a, deadline, lr=0.05, verbose=True):
    """Adam in logit space with adaptive softmax temperature."""
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    best_a = a.copy()
    best_z = z.copy()
    best_score = internal_score(a)
    no_imp = 0
    base_lr = lr
    beta = 0.1 * float(np.convolve(a, a).max())
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        c_max = float(c.max())
        x = (c - c_max) / beta
        ex = np.exp(x)
        w = ex / ex.sum()
        
        # gradient of smooth max
        grad_a = np.zeros(n)
        for i in range(n):
            grad_a[i] = 2.0 * np.dot(w[i:i+n], a)
        
        dot = np.dot(grad_a, a)
        gz = a * (grad_a - dot)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); it = 0
            continue
        
        m = b1 * m + (1-b1) * gz
        v = b2 * v + (1-b2) * gz * gz
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        sc = 2.0 * n * float(np.convolve(a_new, a_new).max())
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.01, lr * 0.9)
```

```python
"""
Multi-phase optimization combining LP direction-finding, L-BFGS-B with p-norm smoothing,
and Adam polish. Improves on previous LP+Adam by adding a second-order phase, adaptive
step sizes in LP, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def make_profile(n, eta=1.5):
    """Linear profile 1+eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_quad_profile(n, c1, c2):
    """Quadratic profile 1+c1*(2t-1)+c2*(2t-1)^2, normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c2 * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length, preserving non-negativity."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix for convolution with f."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, deadline, t_init=0.5, t_min=0.02, verbose=True):
    """LP-based iterations to reduce max(conv(f,f)) with adaptive step sizes."""
    n = len(f)
    f = normalize(f.astype(np.float64).copy())
    if f is None:
        return None, np.inf
    M = float(np.convolve(f, f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init
    it = 0
    while time.monotonic() < deadline and it < 3000:
        it += 1
        g_raw = lp_direction(f, M)
        if g_raw is None:
            if no_improve > 50:
                break
            f = best_f.copy()
            M = best_M
            continue
        g = normalize(g_raw)
        if g is None:
            break
        # Try multiple step sizes for better convergence
        best_new = None
        best_new_score = best_score
        candidates_t = [t_step, t_step * 0.5, t_step * 1.5, 0.1, 0.2, 0.4, 0.6, 0.8]
        for t in candidates_t:
            if t <= 0 or t > 1:
                continue
            f_new = normalize((1.0 - t) * f + t * g)
            if f_new is None:
                continue
            sc = 2.0 * n * float(np.convolve(f_new, f_new).max())
            if sc < best_new_score:
                best_new_score = sc
                best_new = f_new
                best_t = t
        if best_new is not None and best_new_score < best_score - 1e-13:
            best_f = best_new
            best_score = best_new_score
            M = float(np.convolve(best_f, best_f).max())
            f = best_new
            no_improve = 0
            t_step = max(t_min, best_t * 1.05)
        else:
            no_improve += 1
            f = best_f.copy()
            M = best_M
            if no_improve % 20 == 0:
                t_step = max(t_min, t_step * 0.7)
            if no_improve > 200:
                break
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={t_step:.4f}")
    return best_f, best_score


def pnrm_grad_in_logspace(z, p, n):
    """
    Compute value and gradient of 2*n*(sum c^p)^(1/p) in log-space z.
    a = softmax(z), c = conv(a,a).
    Returns (value, grad_z).
    """
    a = np.exp(z - z.max())
    s_a = a.sum()
    a = a / s_a
    c = np.convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros(n)
    u = c / c_max
    up = u ** p
    S = up.sum()
    if S < 1e-30:
        pnrm = c_max
        grad_u = np.zeros_like(u)
    else:
        pnrm = c_max * S ** (1.0 / p)
        # d pnrm / d u_k = S^(1/p - 1) * u_k^(p-1)
        grad_u = S ** (1.0 / p - 1.0) * (u ** (p - 1))
    value = 2.0 * n * pnrm
    # d pnrm / d c_k = grad_u[k] / c_max
    grad_c = grad_u / c_max
    # d F / d a[i] = 2 * sum_k grad_c[k] * a[k-i]
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = 2.0 * np.dot(grad_c[i:i + n], a)
    # Chain through softmax: dF/dz[j] = a[j]*(grad_a[j] - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return value, grad_z


def lbfgsb_polish(a, deadline, p_list=None, verbose=True):
    """L-BFGS-B in log-space with increasing p for sharper max approximation."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    best_a = a.copy()
    best_score = internal_score(a)

    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128]

    for p in p_list:
        if time.monotonic() > deadline - 30:
            break

        def obj(z):
            return pnrm_grad_in_logspace(z, p, n)[0]

        def grad(z):
            return pnrm_grad_in_logspace(z, p, n)[1]

        t0 = time.monotonic()
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10,
                                   'maxfun': 2000, 'maxls': 50})
        elapsed = time.monotonic() - t0
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        sc = internal_score(a_new)
        if verbose:
            print(f"    L-BFGS-B p={p}: score={sc:.8f} (best={best_score:.8f}), "
                  f"iter={result.nit}, time={elapsed:.1f}s")
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = result.x - result.x.mean()

    return best_a, best_score


def adam_polish(a, deadline, lr=0.05, verbose=True):
    """Adam gradient descent in logit space with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    best_a = a.copy()
    best_z = z.copy()
    best_score = internal_score(a)
    no_imp = 0
    base_lr = lr
    c_init = np.convolve(a, a)
    beta = 0.15 * float(c_init.max())

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        c_max = float(c.max())
        x = (c - c_max) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        grad_a = np.zeros(n)
        for i in range(n):
            grad_a[i] = 2.0 * np.dot(w[i:i + n], a)

        dot = np.dot(grad_a, a)
        gz = a * (grad_a - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - np.mean(z)

        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        sc = 2.0 * n * float(np.convolve(a_new, a_new).max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.01, lr * 0.92)

        if it % 1000 == 0:
            beta = max(1e-12, beta * 0.6)

        if no_imp > 8000:
            rng = np.random.default_rng(it * 7 + 3)
            z = 0.6 * best_z + 0.8 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            a_tmp = np.exp(z - z.max())
            a_tmp = a_tmp / a_tmp.sum()
            beta = 0.15 * float(np.convolve(a_tmp, a_tmp).max())

        a = a_new

    return best_a, best_score


def perturb_sequence(a, amount, rng):
    """Randomly perturb a sequence while maintaining non-negativity."""
    a_new = a * (1.0 + amount * rng.normal(size=len(a)))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def coordinate_descent_polish(a, deadline, verbose=True):
    """
    Cyclic coordinate descent with golden-section search in log-space.
    For each coordinate, find best value via 1D search.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = internal_score(a)
    rng = np.random.default_rng(123)

    # Precompute: for coordinate i, the "base" convolution (with a[i]=0)
    # and the "contribution" vector
    # conv[k] = base_conv[k] + 2*a[i]*a[k-i] (for k != 2i) + a[i]^2 (at k=2i)
    # After changing a[i] to x and normalizing:
    # a_new = a, a_new[i] = x, then normalize
    # This is complex, so use a simpler approach:
    # Propose new a[i] = a[i] * exp(delta) for various delta values

    it = 0
    no_improve = 0
    while time.monotonic() < deadline and it < 50000:
        it += 1
        i = it % n
        # Try a few multiplicative perturbations of a[i]
        improvements = False
        for delta in [0.3, -0.3, 0.5, -0.5, 0.1, -0.1, 0.8, -0.8]:
            a_try = a.copy()
            a_try[i] *= np.exp(delta)
            if a_try[i] < 0:
                continue
            a_try = normalize(a_try)
            if a_try is None:
                continue
            sc = internal_score(a_try)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_try.copy()
                a = a_try
                improvements = True
                no_improve = 0
                break
        if not improvements:
            no_improve += 1
            if no_improve > n * 3:
                break

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase search: LP + L-BFGS-B + Adam + coordinate descent.
    Explores multiple sequence lengths and uses diverse initialization.
    """
    t_start = time.monotonic()
    deadline = t_start + 1785
    rng = np.random.default_rng(42)

    # Get initial sequence
    best_seq = None
    best_score = np.inf
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0 and parent.sum() > 1e-10:
            parent = parent / parent.sum()
            best_seq = parent
            best_score = internal_score(parent)
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception as e:
        print(f"No parent sequence available: {e}")

    if best_seq is None:
        # Generate diverse initial profiles
        candidates = [
            make_profile(512, 1.0),
            make_profile(512, 1.5),
            make_profile(512, 2.0),
            make_quad_profile(512, 1.0, 0.3),
            make_quad_profile(512, 1.5, -0.5),
            make_quad_profile(512, 0.5, 0.8),
        ]
        for p in candidates:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.8f}")

    n = len(best_seq)
    print(f"Starting optimization: n={n}, score={best_score:.8f}")

    # Phase 1: LP refinement at current length (budget ~400s)
    dl1 = min(deadline, t_start + 400)
    if time.monotonic() < dl1 - 30:
        print(f"\n=== Phase 1: LP refinement (n={n}, budget {dl1 - time.monotonic():.0f}s) ===")
        f_best, sc_best = lp_iterate(best_seq, dl1, t_init=0.5, t_min=0.02, verbose=True)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  After LP: score={best_score:.8f}")

    # Phase 2: L-BFGS-B polish with p-norm smoothing (budget ~400s)
    dl2 = min(deadline, t_start + 900)
    if time.monotonic() < dl2 - 60:
        print(f"\n=== Phase 2: L-BFGS-B polish (n={n}, budget {dl2 - time.monotonic():.0f}s) ===")
        a_best, sc_best = lbfgsb_polish(best_seq, dl2, p_list=[4, 8, 16, 32, 64], verbose=True)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best
            n = len(a_best)
            print(f"  After L-BFGS-B: score={best_score:.8f}")

    # Phase 3: Adam polish (budget ~400s)
    dl3 = min(deadline, t_start + 1400)
    if time.monotonic() < dl3 - 60:
        print(f"\n=== Phase 3: Adam polish (n={n}, budget {dl3 - time.monotonic():.0f}s) ===")
        a_best, sc_best = adam_polish(best_seq, dl3, lr=0.04, verbose=False)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best
            n = len(a_best)
            print(f"  After Adam: score={best_score:.8f}")
        else:
            print(f"  Adam no improvement (sc={sc_best:.8f} vs best={best_score:.8f})")

    # Phase 4: Try different lengths (budget ~200s)
    dl4 = min(deadline, t_start + 1650)
    if time.monotonic() < dl4 - 60:
        print(f"\n=== Phase 4: Multi-length exploration ===")
        lengths_to_try = [384, 448, 640, 768]
        for n_try in lengths_to_try:
            if time.monotonic() > dl4 - 60:
                break
            if n_try == n:
                continue
            a_try = interpolate_seq(best_seq, n_try)
            if a_try is None:
                continue
            sc_try = internal_score(a_try)
            if sc_try < best_score * 1.001:  # only pursue if close
                print(f"  n={n_try}: initial score={sc_try:.8f} (best={best_score:.8f})")
                # Quick optimization at this length
                dl_sub = min(dl4, time.monotonic() + 100)
                f_b, sc_b = lp_iterate(a_try, min(dl_sub, time.monotonic() + 40), verbose=False)
                if f_b is not None:
                    a_try = f_b
                    sc_try = internal_score(a_try)
                if sc_try < best_score:
                    a_try2, sc_try2 = lbfgsb_polish(a_try, dl_sub, p_list=[4, 16, 64], verbose=False)
                    if sc_try2 < best_score:
                        best_score = sc_try2
                        best_seq = a_try2
                        n = n_try
                        print(f"  Improved at n={n_try}: score={best_score:.8f}")

    # Phase 5: Final Adam polish with low learning rate (remaining time)
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 5: Final Adam polish ===")
        a_best, sc_best = adam_polish(best_seq, deadline - 5, lr=0.02, verbose=False)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best
            print(f"  Final Adam: score={best_score:.8f}")

    # Phase 6: One more L-BFGS-B round if time remains
    if time.monotonic() < deadline - 20:
        a_best, sc_best = lbfgsb_polish(best_seq, deadline - 5, p_list=[32, 64, 128], verbose=False)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best
            print(f"  Final L-BFGS-B: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={final_score:.8f}")
    print(f"{'='*50}")

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01 or not np.isfinite(s):
        best_seq = make_profile(512, 1.5)
    return [float(x) for x in best_seq]
```
