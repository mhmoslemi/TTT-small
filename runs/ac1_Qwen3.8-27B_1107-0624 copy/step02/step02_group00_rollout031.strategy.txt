    t = np.linspace(0, 1, n) - 0.5  # Centrar en tiempo=-1
    t = t * 1.5 + np.random.randn(n) * 0.1  # Añadir ruido aleatorio
    t = t / np.sum(t)  # Normalizar
    return t

def propose_candidate(seed=42, budget_s=1800):
    try:
        parent = np.asarray(height_sequence_1)
        if parent.size <= 4096 and np.all(np.isfinite(parent)) and np.sum(parent) >= 0.01:
            parent = build_sequence_from_parent(parent)
    except:
        pass

    seq_length = 4096
    initial_sequence = np.ones(seq_length) / seq_length
    initial_score = evaluate_sequence(list(initial_sequence))

    def objective_function(x):
        try:
            s_val = np.sum(x)
            if s_val < 0.01:
                return np.inf
            x = np.maximum(x, 0.0)
            x = np.minimum(x, 1000.0)
            convolution_result = np.convolve(x, x, mode='full')
            max_val = np.max(convolution_result)
            score = float(2 * seq_length * max_val / (s_val * s_val))
            return score
        except:
            return np.inf

    try:
        optimization_result = scipy.optimize.minimize(objective_function, initial_sequence, method='CG', options={'maxiter': 1000})
        optimized_seq = optimization_result.x
        optimized_score = evaluate_sequence(list(optimized_seq))
    except:
        optimized_seq = np.ones(seq_length) / seq_length
        optimized_score = np.inf

    return list(optimized_seq)
