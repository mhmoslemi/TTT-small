def make_conv_matrix(f):

    ex = np.exp(x)
    s = float(ex.sum())

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # More precisely: d/d(f_j) [sum_k w_k * (f*f)_k] = 2 * (w * f)_j (cross-correlation)
    # (w*f)_j = sum_k w_k * f_{k-j}
    n = len(f)
    # Compute cross-correlation of w (length 2n-1) with f (length n)

    h = np.fft.irfft(fw * fp, N)
