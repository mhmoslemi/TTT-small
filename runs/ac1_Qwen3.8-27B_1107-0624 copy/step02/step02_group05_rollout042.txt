def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_score_normalized(f):
    """Compute 2*n*max(conv) assuming sum(f)=1."""
    n = len(f)
    c = np.convolve(f, f)
    return float(2.0 * n * c.max())


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ g = np.convolve(f, g)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            A[k, indices] = f[k - indices]
    return A


def solve_lp_min_t(f):
    """
    Solve LP: minimize t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0, t>=0.
    Returns g (normalized) or None on failure.
    """
    n = len(f)
    m = 2 * n - 1
    A = build_conv_matrix(f)

    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    # A_ub @ x <= b_ub: A @ g - t <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    # A_eq @ x = b_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def solve_lp_max_sum(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    m = 2 * n - 1
    A = build_conv_matrix(f)

    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def lp_optimize_v2(f_init, deadline, rng, max_iters=500, use_min_t=True):
    """
    LP-based optimization with new formulation.
    Uses minimize-t LP and line search.
    """
    f = normalize_nonneg(f_init.copy())

    n = len(f)

            continue

        # Line search
        best_step = 0.0
        best_f_try = f
        best_score_try = score
        for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
            f_try = (1.0 - t) * f + t * g

            sc = compute_score_normalized(f_try)
            if sc < best_score_try - 1e-10:

                best_step = t

        if best_step > 0:
            f = best_f_try
            score = best_score_try
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # Stall detection
        if no_imp > 20:
            # Structured perturbation
            if no_imp % 10 == 0:
                # Random direction
                f = 0.8 * best_f + 0.2 * normalize_nonneg(
                    np.abs(rng.normal(size=n)) + 0.1)

            no_imp = 0

            grad_f = (2.0 / beta) * (A.T @ w)
            # Add second term approximately
            A_flip = build_conv_matrix(f[::-1])
            # A_flip[k, i] = f[n-1-(k-i)]
            # sum_k w[k] * f[n-1-(k-i)] ... this gives sum_k w[k]*f[i-k+n-1]
            # which is shifted version. Let me just do it properly:
            # Second term = sum_k w[k] * f[i-k] for 0<=i-k<n
            # = sum_{j=0}^{n-1} w[i-j] * f[j] (letting j=i-k, k=i-j)
            # = (w shifted by i) dot f
            # This is the i-th element of the correlation of w and f
            # corr[i] = sum_j w[i+j] * f[j] ... hmm
            # Let's just use: second_term[i] = sum_{k: 0<=i-k<n} w[k]*f[i-k]
            # = sum_{j: 0<=j<n} w[i-j]*f[j] where i-j in [0, 2n-2]
            # This is (f * flip(w))[i] (convolution of f and flip(w))
            w_flip = w[::-1]
            N3 = 1
            while N3 < 2 * n - 1:
                N3 <<= 1
            ff = np.fft.rfft(f, N3)
            wf_flip = np.fft.rfft(w_flip, N3)
            conv_fw = np.fft.irfft(ff * wf_flip, N3)
            # (f * w_flip)[i] = sum_j f[j] * w_flip[i-j] = sum_j f[j] * w[2n-2-(i-j)] = sum_j f[j] * w[2n-2-i+j]
            # Hmm, not quite what I want. Let me abandon the FFT for the second term
            # and just note that by symmetry of the problem, the gradient should be:
            # grad_f[i] = (2/beta) * sum_k w[k] * (f[k-i] + f[i-k])
            # The two terms are: (A^T w)[i] and (A^T w_rev)[i] where A is built from f
            # and w_rev is... I'm overcomplicating this.
            # For n > 512, just do it in a loop (slow but correct)
            grad_f = np.zeros(n)
            for i in range(n):
                s1 = 0.0
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                if k_lo <= k_hi:
                    s1 = float(np.dot(w[k_lo:k_hi+1], f[k_lo-i:k_hi-i+1]))
                s2 = 0.0
                k_lo2 = max(0, i-n+1)
                k_hi2 = min(2*n-2, i)
                if k_lo2 <= k_hi2:
                    s2 = float(np.dot(w[k_lo2:k_hi2+1], f[i-k_lo2:i-k_hi2+1][::-1]))
                grad_f[i] = (2.0 / beta) * (s1 + s2)

        # Project to simplex in log-space
        dot = float(np.dot(f, grad_f))
        gz = f * (grad_f - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = np.log(best_f + 1e-30)
            z -= z.mean()
            m_buf.fill(0)
            v_buf.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m_buf = b1 * m_buf + (1 - b1) * gz
        v_buf = b2 * v_buf + (1 - b2) * (gz * gz)
        mh = m_buf / (1 - b1 ** adam_t)
        vh = v_buf / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(best_f + 1e-30)
            z -= z.mean()
            m_buf.fill(0)
            v_buf.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        sc = compute_score_normalized(f)

        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)

        if adam_t % 200 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(500, 5 * n):
            z = 0.7 * np.log(best_f + 1e-30) + 0.3 * rng.normal(size=n) * 0.5
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            beta = max(1e-12, 0.5 * float(np.convolve(f, f).max()))
            m_buf.fill(0)
            v_buf.fill(0)
            adam_t = 0
            lr = 0.05
            no_imp = 0

    return best_f, best_score


def get_starting_points(n, rng, count=20):
    """Generate diverse starting points."""
    starts = []

    # Linear profiles
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        t = np.arange(n, dtype=np.float64)
        f = 1.0 + slope * t / max(n - 1, 1)
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())

    # Exponential decay
    for lam in [0.5, 1.0, 2.0, 3.0, 5.0]:
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        f = np.exp(-lam * t)
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1) + 1e-6
        f = t ** p
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())

    # Quadratic
    for a, b in [(-1, 1), (1, -1), (-2, 3), (2, -3)]:
        t = np.linspace(0, 1, n)
        f = 1.0 + a * t + b * t * t
        f = np.maximum(f, 1e-12)
        starts.append(f / f.sum())

    # Random smooth
    for _ in range(count):
        n_harm = rng.integers(2, 7)
        coeffs = rng.normal(-1, 1, 2 * n_harm)
        t = np.linspace(0, 1, n)
        f = np.ones(n)
        for k in range(n_harm):
            f += coeffs[2*k] * np.cos(2*np.pi*(k+1)*t)
            f += coeffs[2*k+1] * np.sin(2*np.pi*(k+1)*t)
        f = np.maximum(f, 1e-6)
        starts.append(f / f.sum())

    return starts


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal step function sequence.
    Uses improved LP formulation, gradient polish, and multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass

    if parent is not None:
        print(f"Parent sequence: n={len(parent)}, score={compute_score(parent):.10f}")

    # ============================================================
    # Phase 1: LP optimization from parent at n=256
    # ============================================================
    n1 = 256
    phase1_end = min(deadline, t_start + 600.0)
    print(f"Phase 1: LP from parent at n={n1}")

    if parent is not None:
        f_start = interpolate_profile(parent, n1) if len(parent) != n1 else parent.copy()
        f_start = normalize_nonneg(f_start)
    else:
        f_start = get_starting_points(n1, rng, 1)[0]

    # Run LP optimization
    results_p1 = []
    alloc = phase1_end - time.monotonic()
    if alloc > 30:
        f_best, sc = lp_optimize_v2(f_start, time.monotonic() + alloc * 0.8, rng, max_iters=1000, use_min_t=True)
        sc_full = compute_score(f_best)
        try_update(f_best, sc_full)
        results_p1.append((sc_full, f_best))
        print(f"  LP min_t from parent: score={sc_full:.10f}")

        # Also try old formulation
        if time.monotonic() < phase1_end:
            f_best2, sc2 = lp_optimize_v2(f_start, phase1_end, rng, max_iters=500, use_min_t=False)
            sc2_full = compute_score(f_best2)
            try_update(f_best2, sc2_full)
            results_p1.append((sc2_full, f_best2))
            print(f"  LP max_sum from parent: score={sc2_full:.10f}")

    # ============================================================
    # Phase 2: LP from diverse starts at n=256
    # ============================================================
    phase2_end = min(deadline, t_start + 1000.0)
    print(f"Phase 2: LP from diverse starts at n={n1}")

    starts = get_starting_points(n1, rng, count=10)
    if parent is not None:
        starts.append(interpolate_profile(parent, n1))

    results_p2 = []
    for i, s in enumerate(starts):
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining / max(1, len(starts) - i), 80.0)
        if alloc < 10:
            break
        f_best, sc = lp_optimize_v2(s, time.monotonic() + alloc, rng, max_iters=300, use_min_t=True)
        sc_full = compute_score(f_best)
        try_update(f_best, sc_full)
        results_p2.append((sc_full, f_best))

    results_p2.sort(key=lambda x: x[0])
    print(f"  Phase 2 best: {results_p2[0][0]:.10f} (overall: {best_score:.10f})")

    # ============================================================
    # Phase 3: Adam polish on best results
    # ============================================================
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"Phase 3: Adam polish")

    all_results = results_p1 + results_p2
    all_results.sort(key=lambda x: x[0])
    top_k = min(5, len(all_results))

    for i in range(top_k):
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining / max(1, top_k - i), 120.0)
        if alloc < 15:
            break
        f_best, sc = adam_polish_logit(all_results[i][1], time.monotonic() + alloc, rng, lr=0.04)
        sc_full = compute_score(f_best)
        try_update(f_best, sc_full)
        print(f"  Adam {i+1}/{top_k}: score={sc_full:.10f}")

    # ============================================================
    # Phase 4: Try n=512
    # ============================================================
    n4 = 512
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"Phase 4: Extension to n={n4}")

    results_p4 = []
    top_for_ext = min(3, len(all_results))
    for i in range(top_for_ext):
        remaining = phase4_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining / max(1, top_for_ext - i), 100.0)
        if alloc < 15:
            break
        f_ext = interpolate_profile(all_results[i][1], n4)
        f_best, sc = lp_optimize_v2(f_ext, time.monotonic() + alloc, rng, max_iters=500, use_min_t=True)
        sc_full = compute_score(f_best)
        try_update(f_best, sc_full)
        results_p4.append((sc_full, f_best))
        print(f"  n={n4} LP {i+1}: score={sc_full:.10f}")

    # Adam polish on best n=512 result
    if results_p4:
        results_p4.sort(key=lambda x: x[0])
        remaining = phase4_end - time.monotonic()
        if remaining > 30:
            f_best, sc = adam_polish_logit(results_p4[0][1], time.monotonic() + min(remaining - 10, 80), rng, lr=0.03)
            sc_full = compute_score(f_best)
            try_update(f_best, sc_full)
            print(f"  n={n4} Adam polish: score={sc_full:.10f}")

    # ============================================================
    # Phase 5: Try n=384 and n=128 (alternative lengths)
    # ============================================================
    phase5_end = min(deadline - 30, time.monotonic() + 120)
    if phase5_end > time.monotonic() + 20 and best_seq is not None:
        for n_alt in [384, 128]:
            remaining = phase5_end - time.monotonic()
            if remaining < 20:
                break
            f_alt = interpolate_profile(best_seq, n_alt)
            f_best, sc = lp_optimize_v2(f_alt, time.monotonic() + min(remaining - 5, 50), rng, max_iters=300, use_min_t=True)
            sc_full = compute_score(f_best)
            try_update(f_best, sc_full)
            print(f"  n={n_alt}: score={sc_full:.10f}")

    # ============================================================
    # Phase 6: Final refinement on best solution
    # ============================================================
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            nbest = len(best_seq)
            print(f"Phase 6: Final refinement at n={nbest}, time={remaining:.0f}s")
            # LP refinement
            alloc_lp = min(remaining * 0.5, 60.0)
            if alloc_lp > 10:
                f_best, sc = lp_optimize_v2(best_seq, time.monotonic() + alloc_lp, rng, max_iters=1000, use_min_t=True)
                sc_full = compute_score(f_best)
                try_update(f_best, sc_full)
                print(f"  Final LP: score={sc_full:.10f}")

            # Adam polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20 and best_seq is not None:
                alloc_adam = min(remaining2 - 10, 80.0)
                if alloc_adam > 10:
                    f_best, sc = adam_polish_logit(best_seq, time.monotonic() + alloc_adam, rng, lr=0.03)
                    sc_full = compute_score(f_best)
                    try_update(f_best, sc_full)
                    print(f"  Final Adam: score={sc_full:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = get_starting_points(256, rng, 1)[0]

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = get_starting_points(256, rng, 1)[0]
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")
    print(f"Best score found: {best_score:.10f}")

    return [float(x) for x in best_seq]
