def objective_and_grad(x, n, beta):
    """Compute objective and gradient for softmax parameterization."""
    # Softmax with numerical stability
    x_shifted = x - x.max()
    s = np.exp(x_shifted)
    s = s / s.sum()
    
    # Compute convolution
    c = np.convolve(s, s)
    
    # Log-sum-exp
    max_c = c.max()
    log_sum_exp = max_c + np.log(np.sum(np.exp(beta * (c - max_c))))
    obj = (2.0 * n / beta) * log_sum_exp
    
    # Gradient w.r.t. s
    p = np.exp(beta * (c - max_c)) / np.sum(np.exp(beta * (c - max_c)))
    # grad_s[j] = (4n/beta) * sum_m p[m+j] * s[m]
    # This is cross-correlation of s and p
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    P = np.fft.rfft(p, N_fft)
    grad_s = np.fft.irfft(np.conj(S) * P, N_fft)[0:n] * (4.0 * n / beta)
    
    # Gradient w.r.t. x: grad_x[j] = s[j] * (grad_s[j] - sum_i grad_s[i] * s[i])
    dot = np.sum(grad_s * s)
    grad_x = s * (grad_s - dot)
    
    return obj, grad_x

Actually, I realize I should also try a completely different approach: **direct minimization of the max using a "peak-fighting" heuristic**.

The idea: identify the lag $k^*$ that achieves the maximum of the convolution, and then try to reduce $c_{k^*}$ specifically. This can be done by:
1. Computing the gradient of $c_{k^*}$ w.r.t. $s$.
2. Taking a step in the direction that decreases $c_{k^*}$ (while keeping $s \geq 0$ and $\sum s_i = 1$).

The gradient of $c_k$ w.r.t. $s_j$ is $2s_{k-j}$ (as derived earlier). So to decrease $c_k$, we want to decrease $s_j$ for those $j$ where $s_{k-j}$ is large.

But we also need to maintain $\sum s_i = 1$ and $s_i \geq 0$. A simple approach:
- Compute $g_j = 2 s_{k^*-j}$ (for valid $j$).
- Update: $s_j \leftarrow s_j - \eta g_j$ (for all $j$).
- Project onto $\{s \geq 0, \sum s_i = 1\}$.

This is essentially a projected gradient step targeting a specific peak.

I could alternate between:
- Soft log-sum-exp steps (to reduce all peaks simultaneously)
- Peak-targeting steps (to specifically reduce the worst peak)

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    c = np.convolve(s, s)
    return 2.0 * n * np.max(c) / (total * total)

def softmax_obj_grad(x, n, beta):
    x = x - x.max()  # stability
    s = np.exp(x)
    s = s / s.sum()
    
    c = np.convolve(s, s)
    
    max_c = c.max()
    e = np.exp(beta * (c - max_c))
    e_sum = e.sum()
    log_se = max_c + np.log(e_sum)
    obj = (2.0 * n / beta) * log_se
    
    p = e / e_sum
    
    # Gradient via FFT
    N = 1
    while N < 3 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    P = np.fft.rfft(p, N)
    grad_s = np.fft.irfft(np.conj(S) * P, N)[:n]
    grad_s *= (4.0 * n / beta)
    
    dot = np.dot(grad_s, s)
    grad_x = s * (grad_s - dot)
    
    return obj, grad_x

def peak_targeting_step(s, step_size=0.1):
    """Take a step to reduce the maximum of the convolution."""
    n = len(s)
    c = np.convolve(s, s)
    k_star = np.argmax(c)
    
    # Gradient of c[k_star] w.r.t. s[j] = 2 * s[k_star - j] (if valid)
    # k_star is in [0, 2n-2], j is in [0, n-1]
    # k_star - j is valid if 0 <= k_star - j < n
    grad = np.zeros(n)
    for j in range(n):
        k_j = k_star - j
        if 0 <= k_j < n:
            grad[j] = 2.0 * s[k_j]
    
    # Update: s_new = s - eta * grad, then project onto {s >= 0, sum = 1}
    s_new = s - step_size * grad
    s_new = np.maximum(s_new, 0)
    s_new = s_new / s_new.sum()
    return s_new

def lp_correct(s, max_time=5.0):
    """LP-based correction: find direction b that maximizes sum(b) s.t. conv(s,b) <= max(conv(s,s))."""
    n = len(s)
    c = np.convolve(s, s)
    M = c.max()
    
    # LP: maximize sum(b) s.t. (s*b)_k <= M for all k, b >= 0
    # (s*b)_k = sum_i s_i * b_{k-i} (for valid indices)
    # This is a linear constraint in b.
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = s[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(max(0, j), min(2*n-1, j+n)):
            A[k, j] = s[k-j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': max_time})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-12:
                b = b / b.sum()
                return b
    except:
        pass
    return None
