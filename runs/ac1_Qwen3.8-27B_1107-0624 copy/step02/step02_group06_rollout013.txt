def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-14
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def make_basis_mixed(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u, u**2, u**3]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
    return np.column_stack(cols[:K])

def make_varied_profiles(n, rng):
    profs = []

    # Uniform
    profs.append(np.ones(n) / n)

    # Linear ramps
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 8.0):
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * t
        a = np.maximum(a, 1e-12)

        profs.append(a / a.sum())

    # Cubic profiles

def gradient_soft_max(p, beta):
    """Compute gradient of soft-max(conv(p,p)) w.r.t. logit space."""
    c = full_convolve_fft(p)
    L, w = soft_mx(c, beta)
    gp = 2.0 * corr_weights(w, p)
    return L, gp, c


def optimize_nesterov(p0, deadline, rng, lr=0.12, beta_init_frac=0.15,
                      momentum=0.9, restart_thresh=400):
    """Nesterov-accelerated gradient descent in logit space with soft-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    v_nest = np.zeros(n, dtype=np.float64)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    it = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, gp, c = gradient_soft_max(p, beta)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            v_nest.fill(0.0)
            continue

        # Nesterov: look-ahead step
        z_look = z + momentum * v_nest
        p_look = softmax_from_z(z_look)
        L, gp_look, _ = gradient_soft_max(p_look, beta)

        # Convert to gradient in z-space
        vp = gp_look
        dot = float(np.dot(p_look, vp))
        gz = p_look * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            v_nest.fill(0.0)
            continue

        v_nest = momentum * v_nest - lr * gz
        z = z + v_nest
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        it += 1

        # Adaptive beta cooling
        if it % 50 == 0:
            beta = max(1e-14, beta * 0.75)

        # Learning rate adaptation
        if no_imp > 100 and no_imp % 100 == 0:
            lr = max(0.02, lr * 0.9)
        elif no_imp == 0 and lr < base_lr:
            lr = min(base_lr, lr * 1.05)

        # Restart
        if no_imp > restart_thresh:
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            v_nest.fill(0.0)
            lr = base_lr
            no_imp = 0
            restart_thresh = max(200, restart_thresh * 0.8)

    return best_p, best_score


def optimize_adam(p0, deadline, rng, lr=0.10, beta_init_frac=0.2,

    c = full_convolve_fft(p)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        vp = gp
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1**adam_t)
        vh = v / (1.0 - b2**adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.7)

        if no_imp % 200 == 0 and no_imp > 0:

            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def active_set_equalize(p, deadline, rng, n_active_frac=0.05, n_iters=500):
    """
    Try to equalize the top convolution values by direct adjustment.
    Uses the KKT insight: at optimum, gradient should be constant on support.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = float(2.0 * n * float(full_convolve_fft(p).max()))
    lr = 0.05

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = full_convolve_fft(p)
        mval = float(c.max())
        score = float(2.0 * n * mval)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()

        # Find active set: positions within fraction of max
        threshold = mval * (1.0 - n_active_frac)
        active_mask = c >= threshold
        n_active = max(1, int(np.sum(active_mask)))

        # Compute average gradient over active set
        w = np.zeros_like(c)
        w[active_mask] = 1.0 / n_active
        gp = 2.0 * corr_weights(w, p)

        # Project gradient to logit space
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        # Take step
        p_new = p * (1.0 - lr * gz / (np.abs(gz).max() + 1e-12))
        p_new = np.maximum(p_new, 1e-12)
        s = p_new.sum()
        if s < 1e-10:
            break
        p_new = p_new / s

        c_new = full_convolve_fft(p_new)
        score_new = float(2.0 * n * float(c_new.max()))

        if np.isfinite(score_new) and score_new < score:
            p = p_new
            lr = min(0.2, lr * 1.05)
        else:
            lr = max(0.01, lr * 0.7)
            if lr < 0.01:
                break

    return best_p, best_score
