```python
from scipy.optimize import linprog

def lp_refine(a, M):
    """Find g that maximizes sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    # conv(a, g)_k = sum_i a_{i} * g_{i-k} (need to be careful about indexing)
    # Actually, np.convolve(a, g)[k] = sum_i a[i] * g[k-i]
    # We need to express this as a linear function of g.
    # np.convolve(a, g)[k] = sum_{i} a[i] * g[k-i]
    # For k from 0 to 2n-2:
    #   np.convolve(a,g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i]
    # Let j = k-i, so g[j] where j = k-i
    #   = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * g[j]
    
    # So the matrix A has A[k,j] = a[k-j] if k-j is in [0,n-1] and j is in [0,n-1]
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    # Constraint: A @ g <= M
    # g >= 0 handled by bounds
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()

    z = z - float(np.max(z))
    e = np.exp(z)

def build_conv_matrix(a):
    """Build the convolution matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def lp_refine_step(a, M, t):
    """Single LP refinement step. Returns new a or None if LP fails."""
    n = len(a)
    A = build_conv_matrix(a)
    # Maximize sum(g) subject to A@g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    g = res.x
    s_g = float(g.sum())
    if s_g < 1e-10:
        return None
    # Rescale g to have same sum as a
    s_a = float(a.sum())
    g = g / s_g * s_a
    # Step
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    return a_new
```
