import numpy as np
from scipy.optimize import linprog
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(convolve(f,f))/sum(f)^2.
Uses diverse asymmetric starting points, LP-based iterative refinement,
spectral initialization, and Adam polishing across multiple sequence lengths.
"""


def compute_score_normalized(f):
    """Compute 2*n*max(conv) for f normalized to sum=1."""
    n = len(f)
    c = np.convolve(f, f)
    return float(2.0 * n * c.max())


def compute_M(f):
    """Compute max of linear convolution of f with itself."""
    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 4.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def make_linear_profile(n, alpha=1.0, beta=0.0, gamma=0.0):
    """Polynomial profile: alpha*t^2 + beta*t + 1, clamped to positive."""
    t = np.linspace(0, 1, n)
    f = 1.0 + beta * t + alpha * t * t
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_exp_profile(n, rate=2.0, shift=0.0):
    """Exponential profile: exp(rate * t + shift)."""
    t = np.linspace(0, 1, n)
    f = np.exp(rate * t + shift)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_power_profile(n, p=1.0):
    """Power law: (t + eps)^p."""
    t = np.linspace(0, 1, n)
    eps = 1.0 / n
    f = (t + eps) ** p
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_skew_gaussian(n, mu=0.3, sigma=0.3, skew=1.0):
    """Skewed Gaussian-like profile."""
    t = np.linspace(0, 1, n)
    z = (t - mu) / sigma
    f = np.exp(-0.5 * z * z + skew * z)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

    N = 2 * n - 1

    best_score = 2.0 * n * M
    no_imp = 0

        if g is None:
            f = 0.8 * f + 0.2 * normalize_nonneg(f + 0.1 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = compute_M(f_try)
            if M_try < best_M_try - 1e-14:

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f.copy()
                best_M = M
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # Stall: try different strategies
        if no_imp > 40:
            strategy = rng.integers(3)
            if strategy == 0:
                f = normalize_nonneg(best_f + 0.2 * rng.normal(size=n))

def soft_max_grad(c, beta):
    """Soft-max approximation and its gradient weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x - x.max())
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j]*p[j] for i=0..len(p)-1 via FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1 + len(w) - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1: 2 * n - 1]


def adam_polish(f_init, deadline, rng, lr=0.05, max_iters=8000):
    """Adam optimization in logit space to minimize max convolution."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-10, 0.1 * M)

    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    best_score = 2.0 * n * M
    no_imp = 0
    base_lr = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        _, w = soft_max_grad(c, beta)
        gp = 2.0 * corr_weights_fft(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())
        sc = 2.0 * n * M

        if sc < best_score - 1e-12:
            best_score = sc
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)

        # Adapt beta (temperature for soft-max)
        if adam_t % 200 == 0:
            beta = max(1e-12, beta * 0.7)

        # Restart
        if no_imp > max(500, 2 * n):
            z = 0.7 * best_z + 0.3 * rng.normal(size=n) * 0.5
            z = np.clip(z, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.1 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse exploration at small n,
    focused LP optimization at medium n, Adam polish, and extension.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # === Phase 1: Rapid exploration at n=128 ===
    n1 = 128
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: Explore at n={n1}, budget={phase1_end - t_start:.0f}s")

    starts1 = generate_diverse_starts(n1, rng, count=35)
    if parent is not None:
        starts1.append(interpolate_profile(parent, n1))

    # Quick eval all starts
    scored_starts = []
    for s in starts1:
        sc = compute_score_normalized(s)
        scored_starts.append((sc, s))
    scored_starts.sort(key=lambda x: x[0])
    for sc, s in scored_starts:
        try_update(s, sc)
    print(f"  Best init: {best_score:.8f}")

    # Run LP on top starts
    top_k1 = min(12, len(scored_starts))
    results1 = []
    for i in range(top_k1):
        remaining = phase1_end - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining / max(1, top_k1 - i), 40.0)
        run_dl = time.monotonic() + alloc
        f_best, sc = lp_optimize(scored_starts[i][1], run_dl, rng, max_iters=500)
        try_update(f_best, sc)
        results1.append((sc, f_best))
        if (i + 1) % 4 == 0:
            print(f"  After {i+1}/{top_k1}: best={best_score:.8f}")

    results1.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Focused optimization at n=256 ===
    n2 = 256
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"Phase 2: Optimize at n={n2}, budget={phase2_end - t_start:.0f}s")

    results2 = []
    top_k2 = min(8, len(results1))
    for i in range(top_k2):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k2 - i), 100.0)
        run_dl = time.monotonic() + alloc
        f_ext = interpolate_profile(results1[i][1], n2)
        f_best, sc = lp_optimize(f_ext, run_dl, rng, max_iters=1000)
        try_update(f_best, sc)
        results2.append((sc, f_best))
        print(f"  n={n2} run {i+1}/{top_k2}: score={sc:.8f}, best={best_score:.8f}")

    # Also try direct LP from parent at n=256
    if parent is not None:
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            run_dl = time.monotonic() + min(remaining, 80.0)
            f_best, sc = lp_optimize(parent, run_dl, rng, max_iters=1000)
            try_update(f_best, sc)
            results2.append((sc, f_best))
            print(f"  n={n2} parent: score={sc:.8f}, best={best_score:.8f}")

    # Also try diverse starts at n=256
    for i in range(min(5, len(scored_starts))):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, 5 - i), 40.0)
        run_dl = time.monotonic() + alloc
        f_ext = interpolate_profile(scored_starts[i][1], n2)
        f_best, sc = lp_optimize(f_ext, run_dl, rng, max_iters=500)
        try_update(f_best, sc)
        results2.append((sc, f_best))

    results2.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: Adam polish ===
    phase3_end = min(deadline, t_start + 1500.0)
    print(f"Phase 3: Adam polish, budget={phase3_end - t_start:.0f}s")

    results3 = list(results2)
    top_k3 = min(5, len(results2))
    for i in range(top_k3):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k3 - i), 80.0)
        run_dl = time.monotonic() + alloc
        f_best, sc = adam_polish(results2[i][1], run_dl, rng, lr=0.04)
        try_update(f_best, sc)
        results3.append((sc, f_best))
        print(f"  Adam {i+1}/{top_k3}: score={sc:.8f}, best={best_score:.8f}")

    results3.sort(key=lambda x: x[0])

    # === Phase 4: Extend to n=512 ===
    n4 = 512
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"Phase 4: Extend to n={n4}, budget={phase4_end - t_start:.0f}s")

    results4 = list(results3)
    top_k4 = min(4, len(results3))
    for i in range(top_k4):
        remaining = phase4_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, top_k4 - i), 60.0)
        run_dl = time.monotonic() + alloc
        f_ext = interpolate_profile(results3[i][1], n4)
        f_best, sc = lp_optimize(f_ext, run_dl, rng, max_iters=500)
        try_update(f_best, sc)
        results4.append((sc, f_best))
        print(f"  n={n4} LP {i+1}/{top_k4}: score={sc:.8f}, best={best_score:.8f}")

    # Adam on best n=512
    for i in range(min(2, len(results4))):
        remaining = phase4_end - time.monotonic()
        if remaining < 10:
            break
        run_dl = time.monotonic() + min(remaining, 40.0)
        f_best, sc = adam_polish(results4[i][1], run_dl, rng, lr=0.03)
        try_update(f_best, sc)
        results4.append((sc, f_best))
        print(f"  n={n4} Adam {i+1}: score={sc:.8f}, best={best_score:.8f}")

    # === Phase 5: Final refinement ===
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 30:
        n_best = len(best_seq)
        print(f"Phase 5: Final refinement at n={n_best}, budget={remaining:.0f}s")
        run_dl = time.monotonic() + min(remaining - 5, 120.0)
        f_best, sc = lp_optimize(best_seq, run_dl, rng, max_iters=2000)
        try_update(f_best, sc)
        print(f"  Final LP: score={sc:.8f}, best={best_score:.8f}")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            f_best, sc = adam_polish(best_seq, time.monotonic() + min(remaining2 - 5, 60.0),
                                     rng, lr=0.02)
            try_update(f_best, sc)
            print(f"  Final Adam: score={sc:.8f}, best={best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = make_linear_profile(256, alpha=2.0)
        best_score = compute_score_normalized(best_seq)

    # Final validation
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, alpha=2.0)
    best_seq = best_seq / best_seq.sum()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score_normalized(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
