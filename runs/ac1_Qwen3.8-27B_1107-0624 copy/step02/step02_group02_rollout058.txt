    # Actually conv(p,p)[k] = sum_{i+j=k} p[i]p[j], and this is NOT symmetric in general.
    # Wait, conv(p,p)[k] = sum_i p[i]*p[k-i]. And conv(p,p)[2n-2-k] = sum_i p[i]*p[2n-2-k-i].
    # Let j = 2n-2-k-i: = sum_j p[2n-2-k-j]*p[j] = sum_j p[j]*p[2n-2-k-j]
    # This equals conv(p,p)[k] by symmetry of the sum (just relabel). Yes!
    # So conv(p,p) IS symmetric: conv(p,p)[k] = conv(p,p)[2n-2-k].
    
    # OK so let me just compute the gradient directly:
    # g[i] = sum_k w[k] * d_k_i where d_k_i = d/dp[i] conv(p,p)[k]
    # conv(p,p)[k] = sum_m p[m]*p[k-m]
    # d/dp[i]: contributes when m=i (gives p[k-i]) or k-m=i i.e. m=k-i (gives p[i])
    # So d_k_i = p[k-i] + p[k-i] = 2*p[k-i] if k-i != i, else p[k-i]
    # (the factor of 2 disappears when k-i = i because the same term is counted once)
    
    # For simplicity (and since the exact factor barely matters for optimization), let's use:
    # g[i] ≈ 2 * sum_k w[k] * p[k-i]  (with p[j]=0 for j outside [0,n-1])
    # sum_k w[k] * p[k-i] = sum_{k-i=0}^{n-1} w[k] * p[k-i] = sum_{j=0}^{n-1} w[i+j] * p[j]
    # This is the cross-correlation of w (padded) and p, at offset i.
    
    # Using FFT: 
    # Let W be w padded to length L >= 2n-1+n-1 = 3n-2
    # Let P be p padded to length L
    # cross_corr[i] = sum_j w[i+j]*p[j] = sum_k w[k]*p[k-i] where k=i+j
    # This is (w reversed) convolved with p, at a specific index.
    # Actually: sum_j w[i+j]*p[j] = (w * p_reversed)[i + n - 1]... 
    # Let me just use: it's the correlation, which can be computed as conv(w, p[::-1])
    # conv(w, p[::-1])[i + n - 1] = sum_k w[k]*p[::-1][i+n-1-k] = sum_k w[k]*p[n-1-i-... ]

    # = sum_j p[j] * w[i+j]
    # If I define W_ext[m] = w[m] for m in [0, 2n-2], 0 otherwise

    # Let j = n-1-k: = sum_j p[j] * W_ext[i+j]

def lp_flatten(f, M, max_iter=1):
    """LP: find g that maximizes sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    # Build the constraint matrix implicitly - for large n, this is too big
    # Instead, use a simplified approach: just find the direction that most reduces the peak
    # For now, skip if n is too large
    if n > 2048:
        return None
    
    # Build A where A[k, j] = contribution of g[j] to conv(f, g)[k]
    # conv(f, g)[k] = sum_j f[j] * g[k-j] (with bounds)
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(2*n-1, j+n)
        A[start:end, j] = f[:end-start]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None

            m.fill(0); v.fill(0); t = 0

        t += 1
        m = b1 * m + (1-b1) * gz

            beta = max(beta_min, 0.3 * float(full_convolve_fft(p).max()))

            no_imp = 0

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logit-space Adam with Nesterov momentum, adaptive smooth-max temperature,
peak-focused mass transfer, and periodic LP refinement. Multi-resolution: 512 -> 1024.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-9:
        c = c - mn
    return c


def soft_max_weights(c, beta):
    """Compute smooth max value and softmax weights over convolution entries."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_grad_p(w, p):
    """Compute gradient of smooth_max(conv(p,p)) wrt p.
    Returns approximately 2 * sum_k w[k] * p[k-i] for each i.
    """
    n = len(p)
    L = 2 * n - 1
    N = next_pow2(3 * n + 4)
    W = np.zeros(N)
    W[:L] = w
    pr = p[::-1]
    P = np.zeros(N)
    P[:n] = pr
    C = np.fft.irfft(np.fft.rfft(P, N) * np.fft.rfft(W, N), N)
    h = C[n - 1:2 * n - 1]
    return 2.0 * h


def softmax_stable(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / float(e.size)
    return e / s


def interp_profile(p, n_new):
    """Linearly interpolate a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p)


def make_linear_profile(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_profile(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_hamming_profile(n):
    n_f = max(n, 3)
    a = 0.54 + 0.46 * np.cos(2.0 * np.pi * np.arange(n_f) / (n_f - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_rand_smooth(n, rng, norder=5):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i in range(1, norder + 1):
        coeff = rng.uniform(-0.4, 0.4)
        a = a + coeff * np.cos(2 * np.pi * i * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def peak_mass_transfer(p, rng, n_iter=5, frac=0.02):
    """Transfer mass from peak-contributing positions to reduce the max."""
    n = p.size
    p = p.copy()
    for _ in range(n_iter):
        c = full_convolve_fft(p)
        peak_k = int(np.argmax(c))
        # Positions that contribute to the peak lag
        contrib = []
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contrib.append((i, p[i] * p[j]))
        if len(contrib) < 2:
            break
        contrib.sort(key=lambda x: -x[1])
        # Pick a top contributor and transfer to a non-contributor
        top_i = contrib[0][0]
        contrib_set = set(x[0] for x in contrib)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        if not non_contrib:
            non_contrib = list(range(n))
        dst = int(rng.integers(0, len(non_contrib)))
        dst_idx = non_contrib[dst]
        transfer = frac * p[top_i]
        p[top_i] -= transfer
        p[dst_idx] += transfer
        p = np.maximum(p, 1e-12)
        s = p.sum()
        if s > 1e-10:
            p = p / s
    return p


def lp_descent(f, M, timeout_s=2.0):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(f)
    if n > 4096:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout_s})
        if time.monotonic() - t0 > timeout_s:
            return None
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def adam_optimize(p0, deadline, rng, lr=0.12, beta_start_frac=0.35,
                  beta_min=1e-9, cd_every=80, cd_iters=4,
                  lp_every=400, restart_after=600):
    """Logit-space Adam with adaptive smooth max, mass transfer, and LP refinement."""
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-12)
    p = p / p.sum()
    n = p.size

    z = np.log(p)
    z -= float(np.mean(z))

    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c0 = full_convolve_fft(p)
    beta = beta_start_frac * float(c0.max())

    best_score = float(2 * n * c0.max())
    best_p = p.copy()
    best_z = z.copy()
    no_imp = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(p)
        L_val, w = soft_max_weights(c, beta)

        gp = compute_grad_p(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100.0, 100.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)

        # Periodic mass transfer
        if t % cd_every == 0:
            p = peak_mass_transfer(p, rng, n_iter=cd_iters, frac=0.015)
            z = np.log(np.maximum(p, 1e-30))
            z -= float(np.mean(z))

        # Periodic LP refinement
        if t % lp_every == 0:
            c_check = full_convolve_fft(p)
            M_cur = float(c_check.max())
            g = lp_descent(p, M_cur, timeout_s=1.5)
            if g is not None:
                blend = 0.3
                p_new = (1.0 - blend) * p + blend * g
                p_new = np.maximum(p_new, 1e-12)
                p_new = p_new / p_new.sum()
                sc_new = float(2 * n * full_convolve_fft(p_new).max())
                if sc_new < best_score:
                    p = p_new
                    z = np.log(np.maximum(p, 1e-30))
                    z -= float(np.mean(z))

        # Evaluate
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        if t % 25 == 0:
            beta = max(beta_min, beta * 0.82)

        # Decay learning rate slowly
        lr = max(0.01, lr * 0.9995)

        # Restart if stuck
        if no_imp > restart_after:
            mix = 0.6 + 0.4 * rng.random()
            z = mix * best_z + (1.0 - mix) * rng.normal(0, 0.5, n)
            z = np.clip(z, -100.0, 100.0)
            z -= float(np.mean(z))
            p = softmax_stable(z)
            beta = max(beta_min, 0.3 * float(full_convolve_fft(p).max()))
            m.fill(0)
            v.fill(0)
            t = 0
            no_imp = 0
            lr = 0.12

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Optimize at n=512 with diverse profiles.
    Phase 2: Interpolate to n=1024 and optimize.
    Phase 3: Final polish at best resolution.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Search start. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        try_update(init_seq, sc_init)
    else:
        print("No initial sequence available.")

    # ============ Phase 1: n=512 ============
    n1 = 512
    phase1_end = min(deadline, t_start + 550.0)
    print(f"Phase 1: n={n1}, budget to {phase1_end - t_start:.0f}s")

    # Generate starting profiles
    profiles = []
    if init_seq is not None:
        p_init = interp_profile(init_seq, n1) if init_seq.size != n1 else init_seq.copy()
        profiles.append(("init", p_init))
    profiles.append(("lin_2", make_linear_profile(n1, 2.0)))
    profiles.append(("lin_3", make_linear_profile(n1, 3.0)))
    profiles.append(("cos_1", make_cosine_profile(n1, 1, 0.5)))
    profiles.append(("cos_2", make_cosine_profile(n1, 2, 0.3)))
    profiles.append(("gauss", make_gaussian_profile(n1, 0.35)))
    profiles.append(("hamm", make_hamming_profile(n1)))
    for i in range(3):
        profiles.append((f"rand_{i}", make_rand_smooth(n1, rng, 5)))

    # Try each profile with a time-limited optimization
    per_time = 40.0
    profile_results = []
    for name, p0 in profiles:
        if time.monotonic() > phase1_end - 10:
            break
        sc0 = compute_score(p0)
        try_update(p0, sc0)
        dl = min(phase1_end, time.monotonic() + per_time)
        res, sc = adam_optimize(p0, dl, rng, lr=0.12, beta_start_frac=0.3,
                                beta_min=1e-10, cd_every=60, cd_iters=3,
                                lp_every=0, restart_after=300)
        try_update(res, sc)
        profile_results.append((name, sc, res))
        print(f"  P1 {name}: score={sc:.10f}")

    # Take the best profile result and polish further
    if profile_results:
        profile_results.sort(key=lambda x: x[1])
        best_name, best_sc, best_p1 = profile_results[0]
        print(f"  Phase 1 best: {best_name} = {best_sc:.10f}")
        # Extra polish on the best
        if time.monotonic() < phase1_end - 5:
            dl2 = phase1_end
            res2, sc2 = adam_optimize(best_p1, dl2, rng, lr=0.08, beta_start_frac=0.2,
                                      beta_min=1e-10, cd_every=50, cd_iters=4,
                                      lp_every=200, restart_after=400)
            try_update(res2, sc2)
            print(f"  P1 polish: score={sc2:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: n=1024 ============
    n2 = 1024
    phase2_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase2_end - 20 and best_seq is not None:
        print(f"Phase 2: n={n2}, budget to {phase2_end - (time.monotonic()):.0f}s")
        p2 = interp_profile(best_seq, n2)
        sc2 = compute_score(p2)
        try_update(p2, sc2)
        print(f"  Interpolated: score={sc2:.10f}")

        # Main optimization at n=1024
        dl3 = min(phase2_end - 5, time.monotonic() + 700.0)
        res3, sc3 = adam_optimize(p2, dl3, rng, lr=0.1, beta_start_frac=0.3,
                                  beta_min=1e-10, cd_every=80, cd_iters=4,
                                  lp_every=300, restart_after=500)
        try_update(res3, sc3)
        print(f"  P2 main: score={sc3:.10f}")

        # Try a second starting point at n=1024
        if time.monotonic() < phase2_end - 30:
            p2b = interp_profile(make_linear_profile(n1, 2.5), n2)
            sc2b = compute_score(p2b)
            if sc2b < best_score + 0.1:
                dl4 = min(phase2_end - 5, time.monotonic() + 100.0)
                res4, sc4 = adam_optimize(p2b, dl4, rng, lr=0.1,
                                          beta_start_frac=0.3, beta_min=1e-10,
                                          cd_every=80, cd_iters=4,
                                          lp_every=300, restart_after=400)
                try_update(res4, sc4)
                print(f"  P2 alt: score={sc4:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Final polish ============
    phase3_end = deadline
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        n_best = best_seq.size
        print(f"Phase 3: polish at n={n_best}, budget={phase3_end - time.monotonic():.0f}s")

        # Adam polish with very small beta
        p3 = best_seq.copy()
        p3 = p3 + 0.001 * rng.normal(size=n_best)
        p3 = np.maximum(p3, 1e-12)
        p3 = p3 / p3.sum()

        dl5 = min(phase3_end - 10, time.monotonic() + 350.0)
        res5, sc5 = adam_optimize(p3, dl5, rng, lr=0.06, beta_start_frac=0.15,
                                  beta_min=1e-11, cd_every=40, cd_iters=5,
                                  lp_every=200, restart_after=300)
        try_update(res5, sc5)
        print(f"  P3 polish: score={sc5:.10f}")

        # Another restart with different perturbation
        if time.monotonic() < phase3_end - 10:
            p3b = best_seq.copy()
            p3b = p3b + 0.005 * rng.normal(size=n_best)
            p3b = np.maximum(p3b, 1e-12)
            p3b = p3b / p3b.sum()
            dl6 = min(phase3_end - 5, time.monotonic() + 150.0)
            res6, sc6 = adam_optimize(p3b, dl6, rng, lr=0.08, beta_start_frac=0.2,
                                      beta_min=1e-11, cd_every=40, cd_iters=5,
                                      lp_every=250, restart_after=250)
            try_update(res6, sc6)
            print(f"  P3 restart: score={sc6:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
