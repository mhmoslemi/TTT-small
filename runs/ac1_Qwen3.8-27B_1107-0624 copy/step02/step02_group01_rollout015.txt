Combines L-BFGS on smoothed objective, coordinate descent, and LP-based methods.

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def make_smoothed_objective(a, beta):
    """Return (value, gradient) of the smoothed max-conv objective."""
    n = a.size
    c = fft_convolve(a)
    # Soft-max smoothing
    c_shift = c / beta
    c_shift -= c_shift.max()
    exp_c = np.exp(c_shift)
    log_sum = np.log(exp_c.sum())
    value = beta * (log_sum + c.max() / beta)  # = beta * log_sumexp(c/beta)
    
    # Weights
    w = exp_c / exp_c.sum()
    
    # Gradient: grad_m = 2 * sum_k w[k] * a[k-m] (with bounds)
    # This is 2 * cross_correlation(w, a)[m]
    N_corr = next_pow2(3 * n)
    fw = rfft(w, N_corr)
    fa = rfft(a, N_corr)
    h = irfft(fw * np.conj(fa), N_corr)[:n]
    
    grad = 2.0 * h
    return value, grad
