def compute_score(f):

    s = float(f.sum())
    if s < 1e-30:

    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))

def soft_max_and_weights(c, beta):
    """Compute soft-max value and softmax weights of convolution."""
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    w = ex / s
    val = m + beta * np.log(s)
    return val, w

"""
Soft-max gradient Adam optimization for minimizing 2*n*max(conv(f,f))/sum(f)^2.

Key innovations over previous approach:
1. Efficient O(n log n) gradient via FFT-based cross-correlation
2. Multi-scale search (n=128 to 2048) with progressive refinement
3. Adaptive beta (temperature) cooling schedule
4. Diversity-preserving restarts
5. Better starting points based on theoretical analysis
"""

import numpy as np
import time

def compute_score(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))

def compute_conv(f):
    return np.convolve(f, f)

def soft_max_val(c, beta):
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    return m + beta * np.log(ex.sum())

def soft_max_weights(c, beta):
    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    return ex / ex.sum()

def gradient_f(w, f):
    """Compute dL/df[i] = 2 * sum_j w[i+j] * f[j]"""
    n = len(f)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(f[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return 2.0 * h[n-1:2*n-1]

def normalize(f):
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-30:
        return None
    return f / s

def make_uniform(n):
    return np.ones(n) / n

def make_linear(n, slope=1.0, offset=1.0):
    t = np.linspace(0, 1, n)
    f = offset + slope * t
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_exp(n, rate=1.0):
    t = np.linspace(0, 1, n)
    f = np.exp(-rate * t)
    return f / f.sum()

def make_sine(n, freq=1.0, phase=0.0, floor=0.3):
    t = np.linspace(0, 1, n)
    f = 1.0 + (1.0-floor) * np.sin(2*np.pi*freq*t + phase)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_quadratic(n, a=1.0):
    t = np.linspace(0, 1, n)
    f = 1.0 + a * (t - 0.5)**2
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_random_smooth(n, rng, n_harmonics=5, amp=0.3):
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, n_harmonics + 1):
        a = rng.normal(0, amp)
        b = rng.normal(0, amp)
        f = f + a * np.cos(2*np.pi*k*t) + b * np.sin(2*np.pi*k*t)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)

def adam_optimize(f_init, deadline, rng, lr=0.05, beta_init=None, max_iters=10000):
    """Adam optimization with soft-max gradient."""
    f = normalize(f_init.copy())
    if f is None:
        return f_init, compute_score(f_init)
    
    n = len(f)
    c = compute_conv(f)
    M = float(c.max())
    if beta_init is None:
        beta = 0.1 * M
    else:
        beta = beta_init
    
    # Work in log-space for non-negativity
    z = np.log(f + 1e-30)
    z0 = z.copy()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_f = f.copy()
    best_score = compute_score(f)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr
    
    # Track beta history
    beta_decay = 0.97
    min_beta = max(1e-8, M * 1e-6)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Recompute convolution and soft-max
        c = compute_conv(f)
        w = soft_max_weights(c, beta)
        M = float(c.max())
        
        # Compute gradient
        grad = gradient_f(w, f)
        
        # Project gradient to simplex tangent (remove mean component)
        # Since score is scale-invariant, gradient along (1,1,...,1) doesn't matter
        grad -= grad.mean()
        
        # Clip gradient for stability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 100:
            grad = grad / grad_norm * 100
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)
        
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back
        z_new = np.clip(z_new, -30, 30)
        e = np.exp(z_new)
        f_new = e / e.sum()
        
        # Check validity
        if not np.all(np.isfinite(f_new)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        # Evaluate
        c_new = compute_conv(f_new)
        score_new = 2.0 * n * c_new.max() / f_new.sum()**2
        
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_f = f_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        f = f_new
        z = z_new
        
        # Decay beta (anneal)
        if it % 50 == 0:
            beta = max(min_beta, beta * beta_decay)
        
        # Adaptive learning rate
        if no_imp > 200:
            lr = max(lr * 0.8, 0.001)
        
        # Restart if stuck
        if no_imp > 500:
            # Perturb and restart
            z = best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -30, 30)
            e = np.exp(z)
            f = e / e.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            beta = max(0.05 * compute_conv(f).max(), min_beta)
            lr = base_lr
            no_imp = 0
    
    return best_f, best_score

def adam_optimize_v2(f_init, deadline, rng, lr=0.05, beta_init=None, max_iters=10000):
    """Improved Adam with cycle restarts and multiple beta scales."""
    f = normalize(f_init.copy())
    if f is None:
        return f_init, compute_score(f_init)
    
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    # Multiple passes with different beta scales
    c = compute_conv(f)
    M0 = float(c.max())
    
    beta_list = [M0 * s for s in [0.3, 0.1, 0.05, 0.02, 0.01, 0.005]]

    elapsed = time.monotonic()

    time_per_pass = total_time / (len(beta_list) + 2)
    
    for beta_scale in beta_list:
        if time.monotonic() > deadline:
            break
        
        # Reset to best
        f = best_f.copy()
        z = np.log(f + 1e-30)
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        beta = beta_scale
        min_beta = max(1e-10, M0 * 1e-7)
        no_imp = 0
        
        pass_deadline = min(deadline, time.monotonic() + time_per_pass)
        
        while time.monotonic() < pass_deadline:
            c = compute_conv(f)
            w = soft_max_weights(c, beta)
            M = float(c.max())
            
            grad = gradient_f(w, f)
            grad -= grad.mean()
            
            gn = np.linalg.norm(grad)
            if gn > 1000:
                grad = grad / gn * 1000
            
            t_adam += 1
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = np.clip(z, -40, 40)
            e = np.exp(z)
            f = e / e.sum()
            
            if not np.all(np.isfinite(f)):

                z = np.log(f + 1e-30)

            score = 2.0 * n * compute_conv(f).max()

                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
            
            if no_imp > 300:
                z = np.log(best_f + 1e-30) + 0.05 * rng.normal(size=n)

                e = np.exp(z)
                f = e / e.sum()
                m.fill(0); v.fill(0); t_adam = 0
                no_imp = 0

    return best_f, best_score
