```python
def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
```

```python
def gradient_step_simple(p, eta=0.1, M_top=5):
    """Simple gradient step focusing on top M lags."""
    n = len(p)
    c = np.convolve(p, p)
    max_c = c.max()
    
    # Find top M lags
    top_m = np.argsort(c)[-M_top:]
    
    # Subgradient: for each active lag k, the gradient w.r.t. p is G[k,:]
    G = compute_G(p)
    grad = np.sum(G[top_m, :], axis=0)
    grad = np.maximum(grad, 0)  # Only decrease p where gradient is positive
    
    # Normalized multiplicative update
    grad_max = grad.max()
    if grad_max < 1e-15:
        return p.copy()
    
    p_new = p * np.exp(-eta * grad / grad_max)
    p_new = np.maximum(p_new, 1e-10)
    p_new = p_new / p_new.sum()
    return p_new
```

```python
def compute_G(p):
    """Compute the matrix G[k,i] = d c[k] / d p[i]."""
    n = len(p)
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    i_idx = np.arange(n)[None, :]  # (1, n)
    J = k_idx - i_idx  # (2n-1, n), J[k,i] = k - i
    
    valid = (J >= 0) & (J <= n-1)  # (2n-1, n)
    
    # Clip J to valid range for indexing
    J_clipped = np.clip(J, 0, n-1)
    pJ = np.where(valid, p[J_clipped], 0.0)  # (2n-1, n)
    
    G = 2.0 * pJ
    
    # Fix diagonal: where J[k,i] == i (i.e., k = 2*i)
    # J[k,i] = k-i = i means k = 2*i
    diag_mask = (J == i_idx) & valid
    G[diag_mask] = pJ[diag_mask]
    
    return G
```

Now let me think about the polishing step. After the LP iterations converge (or slow down), we can do a fine-grained polish:

1. **Coordinate descent**: For each index i, optimize p[i] (with sum constraint) to minimize max c[k]. This is a 1D problem.

Actually, for coordinate descent with the sum constraint, if I change p[i] to p[i] + δ, I need to distribute -δ to the other elements. The simplest is to distribute uniformly: p[j] += -δ/(n-1) for j ≠ i. But this might not be optimal.

Alternatively, I can parameterize as: p[i] = x, and p[j] = (1-x) * p[j] / (1-p[i]) for j ≠ i. This maintains the sum constraint and is a 1D optimization in x.

For a given x, c[k] = sum_j p_new[j]*p_new[k-j] is a quadratic function of x (since p_new[i] = x and p_new[j] is linear in x for j ≠ i). So max_k c[k] is the max of quadratic functions, which is a convex function of x. We can minimize it with ternary search or by finding the "intersection points."

This is doable but might be slow to implement correctly. Let me skip it for now and use the gradient method for polishing.

```python
def gradient_polish(p0, deadline, rng):
    """Fine-grained gradient polishing."""
    n = len(p0)
    p = p0.copy() / p0.sum()
    best_p = p.copy()
    best_score = fast_score(p)
    
    eta = 0.05
    beta = 20.0  # Sharp focus on peaks
    
    no_improve = 0
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(p, p)
        max_c = c.max()
        
        # Soft weighting
        w = np.exp(beta * (c - max_c) / max_c)
        
        # Gradient via FFT
        N_fft = 1
        while N_fft < 4*n:
            N_fft *= 2
        P = np.fft.rfft(p, N_fft)
        W = np.fft.rfft(w, N_fft)
        term1 = np.fft.irfft(np.conj(P) * W, N_fft)[:n]
        term2 = np.fft.irfft(P * W, N_fft)[:n]
        grad = term1 + term2
        grad = np.maximum(grad, 0)
        
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        
        p_new = p * np.exp(-eta * grad / grad_max)
        p_new = np.maximum(p_new, 1e-10)
        p_new = p_new / p_new.sum()
        
        sc = fast_score(p_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
            no_improve = 0
        elif sc < fast_score(p):
            p = p_new
            no_improve += 1
        else:
            no_improve += 1
            eta *= 0.95
        
        if no_improve >= 200:
            # Perturb
            p = p + 0.005 * rng.standard_normal(n)
            p = np.maximum(p, 0.001)
            p = p / p.sum()
            eta = 0.05
            no_improve = 0
    
    return best_p
```
