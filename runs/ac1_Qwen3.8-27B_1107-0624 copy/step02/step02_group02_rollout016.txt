def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / e.size
    else:
        p = p / s
    return p


def logit_adam_optimize(p0, deadline, rng, lr=0.05, reg=1e-8, verbose=False):
    """Optimize in log-space using Adam with soft-max surrogate for max."""
    p0 = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-6, 0.1 * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr
    iteration = 0
    
    while time.monotonic() < deadline:
        L_val, w = soft_max_and_weights(c, beta)
        
        # Gradient: dL/da[i] = (2/beta) * sum_j w[i+j] * a[j] / (1) [normalized]
        # Actually: dL/da[i] = (1/beta) * sum_k (c_k/S) * d(c_k)/da[i]
        #           = (1/beta) * sum_k w_k * 2*a[k-i]  (since w_k = exp(c_k/beta)/S)
        #           = (2/beta) * sum_k w_k * a[k-i]
        
        # Using the cross-correlation computed above
        grad_a = (2.0 / beta) * corr(a=p, w=w)
        
        # Project to tangent space of simplex
        grad_a = grad_a - float(np.dot(p, grad_a))
        
        # Convert to log-space gradient
        grad_z = p * grad_a
        grad_z = grad_z - float(np.mean(grad_z))
        grad_z = grad_z + reg * p  # small regularization
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        
        iteration += 1
        if verbose and iteration % 200 == 0:
            print(f"  Adam iter {iteration}: score={best_score:.10f}, beta={beta:.6f}")
        
        # Anneal beta
        if iteration % 100 == 0:
            beta = max(1e-8, beta * 0.85)
        
        # Restart if stuck
        if no_imp > max(500, 3 * n):
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-6, 0.2 * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_anti_sym_profile(n, alpha=1.0):
    """Strongly anti-symmetric: large at one end, small at other."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * (0.5 - t))  # large at t=0, small at t=1
    return a / a.sum()


def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_profile(n, p=2.0):
    """a[j] = (j+1)^p / sum."""
    t = np.arange(n, dtype=np.float64) + 1
    a = t ** p
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)
