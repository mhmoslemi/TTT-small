import numpy as np
from scipy.optimize import linprog
import time


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_M(f):
    """Compute max of linear convolution of f with itself (f normalized to sum=1)."""
    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def solve_lp(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    A = make_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase LP-based optimization to minimize 2*n*max(convolve(a,a))/sum(a)^2.
    Uses diverse starting points, LP iteration with adaptive steps, smoothing,
    greedy local search, and multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass

    # ============================================
    # Phase A: LP optimization at n=256 (main)
    # ============================================
    n_a = 256
    phase_a_end = min(deadline, t_start + 800.0)
    print(f"Phase A: LP optimization at n={n_a}, time budget={phase_a_end - t_start:.0f}s")

    starts_a = get_starting_points(n_a, rng, count=12)
    if parent is not None:
        if parent.size == n_a:
            starts_a.insert(0, parent.copy())
        else:
            starts_a.insert(0, interpolate_profile(parent, n_a))

    # Evaluate all starts
    scored_starts = []
    for i, s in enumerate(starts_a):
        sc = compute_score(s)
        scored_starts.append((sc, s))
        try_update(s, sc)
    scored_starts.sort(key=lambda x: x[0])
    print(f"  Initial best: {best_score:.10f}")

    # Run LP optimization from top starts
    results_a = []
    n_starts_to_optimize = min(12, len(started_starts)) if False else min(12, len(scored_starts))
    for i in range(n_starts_to_optimize):
        remaining = phase_a_end - time.monotonic()
        if remaining < 10:
            break
        # Allocate time: earlier starts get more time
        n_remaining_starts = n_starts_to_optimize - i
        alloc = min(remaining / max(1, n_remaining_starts), 90.0)
        if alloc < 8:
            alloc = 8
        run_deadline = time.monotonic() + alloc
        f_best, M_best = lp_optimize(scored_starts[i][1], run_deadline, rng, max_iters=2000)
        sc = 2.0 * n_a * M_best
        try_update(f_best, sc)
        results_a.append((sc, f_best.copy()))
        if (i + 1) % 3 == 0:
            print(f"  After {i+1}/{n_starts_to_optimize} LP runs: best={best_score:.10f}")

    results_a.sort(key=lambda x: x[0])
    print(f"Phase A done. Best: {best_score:.10f}")

    # ============================================
    # Phase B: Greedy local search on top results
    # ============================================
    phase_b_end = min(deadline, t_start + 1000.0)
    print(f"Phase B: Greedy local search, time budget={phase_b_end - time.monotonic():.0f}s")

    top_b = min(5, len(results_a))
    for i in range(top_b):
        remaining = phase_b_end - time.monotonic()
        if remaining < 10:
            break
        run_deadline = time.monotonic() + min(remaining - 5, 40.0)
        f_greedy = greedy_local_search(results_a[i][1].copy(), run_deadline, max_passes=30)
        sc = compute_score(f_greedy)
        try_update(f_greedy, sc)
        if sc < results_a[i][0]:
            results_a[i] = (sc, f_greedy.copy())
            print(f"  Greedy improved run {i}: {results_a[i][0]:.10f}")

    results_a.sort(key=lambda x: x[0])
    print(f"Phase B done. Best: {best_score:.10f}")

    # ============================================
    # Phase C: More LP iterations on best solutions
    # ============================================
    phase_c_end = min(deadline, t_start + 1300.0)
    print(f"Phase C: Refined LP on top solutions")

    top_c = min(4, len(results_a))
    for i in range(top_c):
        remaining = phase_c_end - time.monotonic()
        if remaining < 15:
            break
        run_deadline = time.monotonic() + min(remaining - 5, 70.0)
        f_best, M_best = lp_optimize(results_a[i][1].copy(), run_deadline, rng, max_iters=2000)
        sc = 2.0 * n_a * M_best
        try_update(f_best, sc)
        if sc < results_a[i][0]:
            results_a[i] = (sc, f_best.copy())
        print(f"  Phase C run {i}: {sc:.10f}, best={best_score:.10f}")

    results_a.sort(key=lambda x: x[0])
    print(f"Phase C done. Best: {best_score:.10f}")

    # ============================================
    # Phase D: Try different n values
    # ============================================
    phase_d_end = min(deadline, t_start + 1600.0)
    print(f"Phase D: Different sequence lengths, time budget={phase_d_end - time.monotonic():.0f}s")

    # Try n=512
    if time.monotonic() < phase_d_end - 120:
        n_d = 512
        print(f"  Trying n={n_d}")
        f_ext = interpolate_profile(results_a[0][1], n_d)
        run_deadline = min(phase_d_end - 60, time.monotonic() + 100.0)
        f_best, M_best = lp_optimize(f_ext, run_deadline, rng, max_iters=1500)
        sc = 2.0 * n_d * M_best
        try_update(f_best, sc)
        print(f"  n={n_d} LP result: {sc:.10f}, best={best_score:.10f}")

        # Greedy on n=512
        if time.monotonic() < phase_d_end - 20:
            run_deadline = min(phase_d_end - 5, time.monotonic() + 40.0)
            f_greedy = greedy_local_search(f_best.copy(), run_deadline, max_passes=20)
            sc = compute_score(f_greedy)
            try_update(f_greedy, sc)
            print(f"  n={n_d} greedy result: {sc:.10f}, best={best_score:.10f}")

    # Try n=128
    if time.monotonic() < phase_d_end - 40:
        n_e = 128
        print(f"  Trying n={n_e}")
        f_ext = interpolate_profile(results_a[0][1], n_e)
        run_deadline = min(phase_d_end - 10, time.monotonic() + 50.0)
        f_best, M_best = lp_optimize(f_ext, run_deadline, rng, max_iters=1500)
        sc = 2.0 * n_e * M_best
        try_update(f_best, sc)
        print(f"  n={n_e} LP result: {sc:.10f}, best={best_score:.10f}")

    # Try n=384
    if time.monotonic() < phase_d_end - 60:
        n_f = 384
        print(f"  Trying n={n_f}")
        f_ext = interpolate_profile(results_a[0][1], n_f)
        run_deadline = min(deadline - 30, time.monotonic() + 60.0)
        f_best, M_best = lp_optimize(f_ext, run_deadline, rng, max_iters=1000)
        sc = 2.0 * n_f * M_best
        try_update(f_best, sc)
        print(f"  n={n_f} LP result: {sc:.10f}, best={best_score:.10f}")

    print(f"Phase D done. Best: {best_score:.10f}")

    # ============================================
    # Phase E: Final polish on the best solution
    # ============================================
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"Phase E: Final polish")
        n_final = len(best_seq)
        # One more round of greedy
        run_deadline = min(deadline - 5, time.monotonic() + 30.0)
        f_greedy = greedy_local_search(best_seq.copy(), run_deadline, max_passes=30)
        sc = compute_score(f_greedy)
        try_update(f_greedy, sc)

        # One more short LP round
        if time.monotonic() < deadline - 10:
            f_norm = normalize_nonneg(best_seq)
            run_deadline2 = min(deadline - 2, time.monotonic() + 20.0)
            f_best, M_best = lp_optimize(f_norm, run_deadline2, rng, max_iters=500)
            sc = 2.0 * len(f_best) * M_best
            try_update(f_best, sc)

    # ============================================
    # Final validation and return
    # ============================================
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_score(best_seq)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = float(best_seq.sum())
    best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")

    # Validate with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
        if np.isfinite(ev_score):
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
