def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def fft_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def soft_max(c, T):
    """Smooth max: T * log(sum(exp(c/T)))."""
    c_shifted = (c - c.max()) / T
    return T * (c.max() + np.log(np.sum(np.exp(c_shifted))))

def soft_max_grad(c, T):
    """Gradient of soft max w.r.t. c."""
    c_shifted = (c - c.max()) / T
    w = np.exp(c_shifted)
    w /= w.sum()
    return w
