def next_pow2(n):
    k = 1
    while k < n:

def full_convolve(a):

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:

    return c

"""
Multi-phase optimization combining power iteration, soft-max gradient Adam,
and coordinate descent to minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses parent solution for initialization, explores multiple lengths, and applies
adaptive cooling schedules with momentum-based restarts.

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-10:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def score_unit(a):
    """Score assuming sum(a)=1, so score = 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    c = full_convolve(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def power_iteration(a, max_iters, T_init, T_min, T_decay, momentum=0.3):
    """
    Power iteration for the optimality condition:
    a is a fixed point of normalize(max(correlate(softmax(conv(a,a)/T), a), 0)).
    With momentum for stability.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = a.size
    T = T_init
    best_a = a.copy()
    best_score = score_unit(a)
    
    for it in range(max_iters):
        b = full_convolve(a)
        maxb = float(b.max())
        if maxb <= 0:
            break
        sc = score_unit(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Soft-max weights over convolution
        if T < 1e-12:
            threshold = maxb * (1.0 - 1e-6)
            mask = b >= threshold
            cnt = mask.sum()
            if cnt == 0:
                mask[int(np.argmax(b))] = True
                cnt = 1
            w = mask.astype(np.float64) / cnt
        else:
            x = b / T
            x = x - x.max()
            e = np.exp(x)
            se = e.sum()
            if se < 1e-30 or not np.isfinite(se):
                w = np.zeros_like(b)
                w[int(np.argmax(b))] = 1.0
            else:
                w = e / se
        
        # Correlation: g_j = sum_i a_i * w_{i+j}
        g = np.correlate(w, a, mode='valid')[:n]
        
        # Update with momentum and thresholding
        a_new = np.maximum(g, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Momentum
        a_mom = (1.0 - momentum) * a_new + momentum * a
        a_mom = np.maximum(a_mom, 0)
        s_mom = a_mom.sum()
        if s_mom < 1e-15:
            a = a_new
        else:
            a = a_mom / s_mom
        
        # Convergence check
        if it > 10 and np.max(np.abs(a - a_new)) < 1e-10:
            sc_final = score_unit(a)
            if sc_final < best_score:
                best_score = sc_final
                best_a = a.copy()
            break
        
        T = max(T_min, T * T_decay)
    
    return best_a


def soft_max_adam(p0, deadline, rng, lr=0.08, beta_frac=0.12, patience=400):
    """Soft-max gradient Adam in logit space with adaptive beta."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.90, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    
    p = p0.copy()
    c = full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Soft-max weights
        x = (c - maxc) / beta
        ex = np.exp(x)
        se = float(ex.sum())
        if not np.isfinite(se) or se <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            se = 1.0
        w = ex / se
        
        # Gradient: g_j = 2 * sum_i p_i * w_{i+j}
        g = 2.0 * np.correlate(w, p, mode='valid')[:n]
        vp = g + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_buf.fill(0)
            v_buf.fill(0)
            t = 0
            continue
        
        t += 1
        m_buf = b1 * m_buf + (1 - b1) * gz
        v_buf = b2 * v_buf + (1 - b2) * (gz * gz)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m_buf.fill(0)
            v_buf.fill(0)
            t = 0
            continue
        
        z_new = np.clip(z_new, -50, 50)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        p = e / e.sum()
        
        c = full_convolve(p)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)
        
        if np.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = p.copy()
                best_z = z_new.copy()
                no_imp = 0
                z = z_new.copy()
            else:
                no_imp += 1
                if no_imp > 150:
                    lr = max(0.02, lr * 0.94)
                # Revert z if stuck
                if no_imp > patience // 2:
                    z = best_z.copy()
                    z = z_new
        else:
            z = best_z.copy()
            m_buf.fill(0)
            v_buf.fill(0)
            t = 0
            continue
        
        # Cooling schedule
        if t % 80 == 0:
            beta = max(1e-14, beta * 0.65)
        
        # Restart
        if no_imp > patience:
            z = best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            e = np.exp(z)
            p = e / e.sum()
            c = full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m_buf.fill(0)
            v_buf.fill(0)
            t = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score


def coordinate_descent(p, deadline, rng):
    """Coordinate descent with multiple trial factors per coordinate."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    
    best_p = p.copy()
    best_score = score_unit(p)
    
    factors = np.array([1.15, 1.08, 1.04, 0.96, 0.92, 0.85])
    no_imp = 0
    
    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        sweep_improved = False
        
        for i in idx:
            if time.monotonic() > deadline:
                return best_p, best_score
            
            orig = p[i]
            
            if orig < 1e-8:
                # Try activating this coordinate
                trial_val = float(np.median(p[np.arange(n) != i])) * 0.1
                if trial_val < 1e-10:
                    trial_val = 1e-4
                p[i] = trial_val
                s = p.sum()
                p = p / s
                sc = score_unit(p)
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    sweep_improved = True
                    no_imp = 0
                else:
                    p[i] = 0.0
                    p = p / p.sum()
                    no_imp += 1
                continue
            
            for f in factors:
                new_val = orig * f
                if new_val < 1e-15:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                p = p / s
                sc = score_unit(p)
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    sweep_improved = True
                    no_imp = 0
                    orig = p[i]
                else:
                    p[i] = orig
                    p = p / p.sum()
                    no_imp += 1
                    break
        
        if no_imp > 3 * n:
            break
    
    return best_p, best_score


def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0, 1, n)
    
    # Linear
    for eta in [1.0, 2.0, 3.0, 5.0, 8.0]:
        a = 1.0 + eta * t
        profiles.append(a / a.sum())
    
    # Exponential
    for eta in [1.0, 2.0, 4.0, 8.0, 12.0]:
        a = np.exp(eta * t)
        profiles.append(a / a.sum())
    
    # Power
    for p in [0.3, 0.5, 1.0, 2.0, 4.0]:
        a = t**p + 1e-8
        profiles.append(a / a.sum())
    
    # Inverted parabola
    for c in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 - c * (t - 1.0)**2
        a = np.maximum(a, 1e-8)
        profiles.append(a / a.sum())
    
    # Cubic
    for a1, a3 in [(1.5, -1.0), (2.0, -1.5), (2.5, -2.0), (3.0, -2.5)]:
        u = 2*t - 1
        a = 1.0 + a1*u + a3*u**3
        a = np.maximum(a, 1e-8)
        profiles.append(a / a.sum())
    
    # Gaussian
    for center in [0.2, 0.4, 0.6, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - center)**2) / (2*width**2))
            profiles.append(a / a.sum())
    
    # Sigmoid
    for eta in [5, 10, 20]:
        a = 1.0 / (1.0 + np.exp(-eta*(t - 0.5)))
        profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(8):
        z = rng.normal(0, 1, n)
        k = max(1, n // 6)
        kernel = np.ones(k) / k
        z = np.convolve(z, kernel, mode='same')
        a = np.exp(z * rng.uniform(0.5, 2.0))

            k = int(n * frac)
            a[k:] = ratio
            a = np.maximum(a, 1e-8)
            profiles.append(a / a.sum())
    
    return profiles


def interpolate_to(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-15:
            return np.ones(n_new) / n_new
        return p / s
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()

        per = min(phase3_end, time.monotonic() + max(20, rem * 0.7))
        p_best, sc = soft_max_adam(p_c, per, rng, lr=0.04, beta_frac=0.06, patience=500)
        sc_full = score_unit(p_best)
        update_best(p_best, sc_full, f"ext_adam_{i}_n{p_best.size}")
    
    print(f"  After phase 3: best={best_score:.10f}")
    
    # Phase 4: Coordinate descent polish
    print("Phase 4: Coordinate descent polish")
    phase4_end = min(deadline, t_start + 1720)
    
    for i in range(min(8, len(candidates))):
        if time.monotonic() > phase4_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 768:
            continue
        per = min(phase4_end, time.monotonic() + 25)
        p_best, sc = coordinate_descent(p_c, per, rng)
        sc_full = score_unit(p_best)
        update_best(p_best, sc_full, f"cd_{i}_n{p_best.size}")
    
    print(f"  After phase 4: best={best_score:.10f}")
    
    # Phase 5: Final combined polish - power iter + CD on very best
    print("Phase 5: Final polish")
    phase5_end = min(deadline, t_start + 1770)
    
    if best_seq is not None:
        # One more round of power iteration on the best
        a_final = power_iteration(best_seq, 300, T_init=0.02, T_min=0.0003, T_decay=0.98, momentum=0.5)
        sc_final_pi = score_unit(a_final)
        if sc_final_pi < best_score:
            update_best(a_final, sc_final_pi, "final_pi")
        
        # Coordinate descent on the best
        if best_seq.size <= 768:
            per = min(phase5_end, time.monotonic() + 20)
            p_cd, sc_cd = coordinate_descent(best_seq, per, rng)
            sc_cd_full = score_unit(p_cd)
            update_best(p_cd, sc_cd_full, "final_cd")
    
    # Final selection and verification
    print("Final selection")
    if best_seq is not None:
        final_seq = np.asarray(best_seq, dtype=np.float64)
        final_seq = np.clip(final_seq, 0, 1000)
        s = float(final_seq.sum())
        if s < 1e-10:
            final_seq = np.ones(256)
            s = 256.0
        final_seq = final_seq / s
        
        # Verify with exact evaluation
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  Evaluated score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print(f"  Warning: evaluate returned non-finite: {sc_eval}")
        except Exception as e:
            print(f"  Warning: evaluation failed: {e}")
        
        print(f"  FINAL: score={best_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]
    
    # Fallback
    print("  FALLBACK: using linear profile")
    fallback = np.arange(1, 257, dtype=np.float64)
    fallback = fallback / fallback.sum()
    print(f"  Fallback score: {score_unit(fallback):.10f}")
    return [float(x) for x in fallback]
