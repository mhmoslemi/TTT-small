def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _smooth_max_gradient(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max_and_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, val

def _make_conv_matrix_sparse_rows(a, n_constraints=None):
    """Build constraint matrix for cross-convolution (a * g)."""
    n = len(a)

    rows = []
    cols = []
    vals = []

            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return rows, cols, vals


def _lp_spread_improved(a, b_current, deadline, inflate=1.002):
    """Improved LP: maximize sum(g) s.t. conv(a,g)_k <= b_k*inflate + slack."""
    n = len(a)
    # Use per-constraint bounds: tighter near peak, looser elsewhere
    max_b = float(b_current.max())
    # M_k = max_b * (1 + inflate) for peak, max_b * (1 + inflate*3) for others
    # Use a smooth interpolation
    ratio = b_current / max_b
    M = max_b * (1.0 + inflate * (1.0 + 2.0 * (1.0 - ratio)))
    M = np.maximum(M, max_b * (1.0 + inflate))

    # Build the constraint matrix: (a * g)_k <= M_k
    # (a * g)_k = sum_j a_j * g_{k-j} for valid j, k-j
    # So row k of the matrix has: entry (k, j) = a_j if 0 <= k-j < n
    n_rows = 2 * n - 1

        end = min(j + n, n_rows)
        A_list[start:end, j] = a[j]

                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None

def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, np.pi, n)

def _make_tukey_profile(n, r=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    half = r / 2.0
    for i in range(n):
        if t[i] < half:
            a[i] = 0.5 * (1 + np.cos(np.pi * (t[i] / half - 1)))
        elif t[i] > 1.0 - half:
            a[i] = 0.5 * (1 + np.cos(np.pi * ((t[i] - (1.0 - half)) / half)))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    if n0 == n_new:

        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _peak_shave(a, deadline, max_iters=10000):

        print(f"  LP result: score={sc_lp:.10f}")

    # Phase 5: Peak shaving
    if best_seq is not None and time.monotonic() < deadline - 100:
        print("Phase 5: Peak shaving")
        ps_deadline = min(time.monotonic() + 100, deadline - 60)
        a_ps = _peak_shave(best_seq, ps_deadline, max_iters=20000)
        sc_ps = _score(a_ps)
        if np.isfinite(sc_ps) and sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
        print(f"  Peak shave: score={sc_ps:.10f}")

    # Phase 6: Final FISTA polish with very small beta
    if best_seq is not None and time.monotonic() < deadline - 40:
        print("Phase 6: Final FISTA polish")
        c_final = _fft_convolve(best_seq)
        max_c = float(c_final.max())
        polish_deadline = min(time.monotonic() + 60, deadline - 10)
        a_pol, sc_pol = _fista_optimize(best_seq, polish_deadline,
                                         max_iter=30000,
                                         beta_init=0.01 * max_c,
                                         beta_min=1e-12)
        if np.isfinite(sc_pol) and sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
        print(f"  Final polish: score={sc_pol:.10f}")

    # Phase 7: One more LP pass
    if best_seq is not None and time.monotonic() < deadline - 10:
        print("Phase 7: Final LP")
        a_last = _lp_refine_loop(best_seq, deadline - 5, max_rounds=15)
        sc_last = _score(a_last)
        if np.isfinite(sc_last) and sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"  Final LP: score={sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_cosine_profile(512, 1.0)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_cosine_profile(512, 1.0)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
