    import numpy as np
    import time

    def fast_conv(s):
        """Compute full convolution of s with itself using FFT."""
        n = len(s)
        size = 1
        while size < 2 * n:
            size *= 2
        S = np.fft.rfft(s, size)
        conv = np.fft.irfft(S * S, size)[:2*n-1]
        return conv

    def compute_subgrad(s, active):
        """Compute subgradient of max(c) with respect to s, averaged over active lags."""
        n = len(s)
        g = np.zeros(n)
        s_rev = s[::-1]  # s_rev[j] = s[n-1-j]
        for k in active:
            # grad_i = 2 * s[k-i] for 0 <= k-i < n
            # s[k-i] = s_rev[n-1-(k-i)] = s_rev[n-1-k+i]
            lo = max(0, k - (n-1))  # i such that k-i <= n-1, i >= k-(n-1)
            hi = min(n-1, k)         # i such that k-i >= 0, i <= k
            if lo <= hi:
                g[lo:hi+1] += 2.0 * s_rev[n-1-k+lo : n-1-k+hi+1]
        g /= len(active)
        return g

        """Multiplicative weight update."""
        s_new = s * np.exp(-alpha * grad)
        s_new = np.maximum(s_new, 1e-30)
        s_new /= s_new.sum()
        return s_new

        total = s.sum()
        if total < 1e-15:
            return float('inf')
        c = fast_conv(s)
        return 2.0 * n * c.max() / (total * total)
