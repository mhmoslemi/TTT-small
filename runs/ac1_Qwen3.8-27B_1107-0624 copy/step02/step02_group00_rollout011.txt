def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _reweight_step(a, deadline, max_iter=1000):
    """
    Reweighting step: identify peak positions in convolution,
    then reduce mass at positions contributing to those peaks.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        
        # Find positions where conv is near max
        threshold = M * 0.95
        peak_positions = np.where(c >= threshold)[0]
        
        # For each peak position k, the contributing positions are j and k-j
        # We want to reduce a at positions that appear in many peak contributions
        contrib = np.zeros(n)
        for k in peak_positions:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    contrib[j] += a[j] * a[kj]
        
        # Reduce a at high-contribution positions
        if contrib.max() > 0:
            # Scale down high-contribution positions
            scale = 1.0 - 0.1 * contrib / contrib.max()
            a_new = a * scale
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                sc = _score(a_new)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
    
    return best_a

        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_maxsum(a, M):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_minpeak(a):
    """LP: min t s.t. conv(a,b) <= t, sum(b) = 1, b >= 0.
    Returns (b_opt, t_opt) or (None, inf)."""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    # Constraints: conv(a,b) <= t -> A*b - t <= 0 for each row
    #              sum(b) = 1 (equality)
    #              b >= 0, t free
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A @ b - t <= 0 for all 2n-1 rows
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = _make_conv_matrix(a)
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, np.inf


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_grad(a, beta):
    """Compute gradient of smooth_max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _nesterov_gd(a, deadline, max_iter=20000):
    """Nesterov accelerated projected gradient on simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-6, 0.08 * float(c.max()))
    lr = 0.015
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        y = a + momentum * v
        y = _project_simplex(y)
        
        grad = _smooth_grad(y, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            v = np.zeros(n)
            continue
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        grad = grad / gnorm
        
        v = momentum * v - lr * grad
        a = _project_simplex(a + v)
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.93)
            if no_imp > max(3000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.08 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
        
        if it % 300 == 0:
            beta = max(1e-8, beta * 0.7)
    
    return best_a


def _lp_refine(a, deadline, max_iter=20000, inflate=1.001, blend=0.25, use_minpeak=False):
    """LP-based refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    base_blend = blend
    base_inflate = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        if use_minpeak:
            g, t_opt = _lp_minpeak(a)
            if g is None:
                break
            sum_g = float(g.sum())
            if sum_g < 1.0 - 1e-8:
                g = g / max(sum_g, 1e-10)
            else:
                break
        else:
            c = _fft_convolve(a)
            M = float(c.max())
            M_lp = M * inflate_cur
            g = _lp_maxsum(a, M_lp)
            if g is None:
                break
            sum_g = float(g.sum())
            if sum_g <= 1.0 + 1e-8:
                no_imp += 1
                if no_imp % 10 == 0:
                    inflate_cur = min(1.01, inflate_cur * 1.08)
                if no_imp > 300:
                    break
                continue
            g = g / sum_g
        
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Gradually tighten
            if it > 200:
                inflate_cur = max(1.0003, inflate_cur * 0.995)
                blend_cur = max(0.05, blend_cur * 0.998)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = base_blend
                inflate_cur = base_inflate
                no_imp = 0
        
        a = a_new
    
    return best_a


def _logit_adam(a, deadline, max_iter=25000, lr=0.025):
    """Adam in logit space with adaptive smoothness."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.04 * float(c.max()))
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()
        z = z_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a
