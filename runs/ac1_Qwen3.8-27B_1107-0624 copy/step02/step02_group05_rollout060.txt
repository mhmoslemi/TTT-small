    Ff = np.fft.rfft(f_flipped, N)
    conv = np.fft.irfft(Fw * Ff, N)
    # conv[k] = (w * f_flipped)[k] for k = 0..len(w)+n-2
    # h[i] = conv[n-1-i] for i = 0..n-1
    h = conv[n-1::-1]  # h[i] = conv[n-1-i]
    return h

def gradient_step(c, beta, f, n):
    """
    Compute gradient of soft_max(conv(f,f), beta) w.r.t. f (unconstrained).
    Returns (soft_max_value, grad) where grad has length n.
    """
    mval, w = soft_max_value_and_weights(c, beta)
    h = compute_gradient_fft(w, f)
    grad = 2.0 * h
    return mval, grad

def adam_optimize(f_init, deadline, rng, beta_init_scale=0.1, max_iters=50000, 
                  lr=0.1, beta_decay=0.95, beta_min_scale=0.01):
    """
    Adam optimization in log-space to minimize soft-max of auto-convolution.
    f_init should be a probability vector (sum=1, non-negative).
    Returns best_f, best_M.
    """
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).copy()
    f = f / f.sum()
    
    # Initialize log params
    z = np.log(f)
    z -= z.mean()  # Centering for numerical stability
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Compute initial convolution
    c = np.convolve(f, f)
    M = float(c.max())
    beta = beta_init_scale * M
    
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        # Compute gradient
        mval, grad = gradient_step(c, beta, f, n)
        
        # Project gradient to tangent space of simplex (in log-param space)
        # dL/dz[i] = f[i] * (dL/df[i] - sum_j f[j] dL/df[j])
        dot = float(np.dot(f, grad))
        gz = f * (grad - dot)
        gz -= gz.mean()  # Extra centering for stability
        
        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_f, 1e-15))
            z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            continue
        
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        # Convert back to probability
        z = np.clip(z, -50, 50)
        z -= z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-30 or not np.isfinite(s):
            z = np.log(np.maximum(best_f, 1e-15))
            z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            continue
        f = e / s
        
        # Compute new M
        c = np.convolve(f, f)
        M = float(c.max())
        
        if M < best_M - 1e-10:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta periodically
        if it % 100 == 0:
            beta = max(beta_min_scale * best_M, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > 200:
            # Perturb in log space
            z = np.log(np.maximum(best_f, 1e-15))
            z += 0.3 * rng.normal(size=n)
            z -= z.mean()
            z = np.clip(z, -50, 50)
            e = np.exp(z - z.max())
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = beta_init_scale * best_M
            m.fill(0); v.fill(0); t = 0
            lr = 0.1
            no_imp = 0
        
        # Adaptive learning rate
        if no_imp > 100:
            lr = max(0.01, lr * 0.9)
    
    return best_f, best_M

def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u**3
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_dual_bump(n):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15)**2) + np.exp(-0.5 * ((t - 1) / 0.15)**2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_cosine_profile(n, freq=1.0, depth=0.5):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 - depth * np.cos(2 * np.pi * freq * t / n)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def get_starting_points(n, rng, count=20):
    starts = []
    starts.append(make_linear_profile(n, 1.0))
    starts.append(make_linear_profile(n, 2.0))
    starts.append(make_linear_profile(n, 3.0))
    starts.append(make_cubic_profile(n, 1.5, -1.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    starts.append(make_dual_bump(n))
    starts.append(make_cosine_profile(n, 1.0, 0.5))
    starts.append(make_cosine_profile(n, 2.0, 0.5))
    starts.append(make_cosine_profile(n, 3.0, 0.3))
    # Random smooth profiles
    for _ in range(count):
        theta = rng.uniform(-2, 2, 12)
        t = np.linspace(0, 1, n)
        f = 1.0
        for k in range(6):
            f = f + 0.2 * theta[2*k] * np.cos(2 * np.pi * (k+1) * t)
            f = f + 0.2 * theta[2*k+1] * np.sin(2 * np.pi * (k+1) * t)
        f = np.maximum(f, 1e-6)
        f = f / f.sum()
        starts.append(f)
    return starts

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)

def lp_polish(f_init, deadline, max_iters=200):
    """LP-based polishing step."""
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    M = compute_conv_max(f)
    best_f = f.copy()
    best_M = M
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        # Build constraint matrix: A @ g = conv(f, g), A is (2n-1) x n
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        # (f*g)[k] = sum_j A[k,j] * g[j]
        
        # For efficiency, only use "active" constraints (where conv is close to M)
        c = np.convolve(f, f)
        # Use all constraints for correctness
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k - n + 1)
            hi = min(n-1, k)
            A[k, lo:hi+1] = f[lo:hi+1][::-1]  # f[k-j] for j=lo..hi
        
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 3.0})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Line search
                    best_t = 0
                    best_M_try = M
                    for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                        f_try = (1-t)*f + t*g
                        f_try = np.maximum(f_try, 0)
                        s = f_try.sum()
                        if s < 1e-15:
                            continue
                        f_try = f_try / s
                        M_try = compute_conv_max(f_try)
                        if M_try < best_M_try - 1e-12:
                            best_M_try = M_try
                            best_t = t
                    if best_t > 0:
                        f = (1-best_t)*f + best_t*g
                        f = np.maximum(f, 0)
                        f = f / f.sum()
                        M = compute_conv_max(f)
                        if M < best_M - 1e-12:
                            best_M = M
                            best_f = f.copy()
                    else:
                        break
        except Exception:
            break
    
    return best_f, best_M

import numpy as np
import time
from scipy.optimize import linprog

"""
Multi-phase optimization to minimize 2*n*max(conv(f,f))/sum(f)^2.
Uses FFT-based gradient descent with Adam in log-space (fast iterations),
combined with LP polishing for local refinement.
Key improvements over previous approach:
- O(n log n) gradient computation via FFT instead of O(n^2) LP solves
- Thousands of gradient iterations vs hundreds of LP iterations
- Smooth max approximation with adaptive temperature scheduling
- Multiple sequence lengths and diverse starting points
"""


def compute_conv_max(f):
    """Compute max of linear auto-convolution of f."""
    c = np.convolve(f, f)
    return float(c.max())


def compute_score_normalized(f):
    """Compute 2n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def soft_max_weights(c, beta):
    """Compute soft-max weights for vector c (returns weights summing to 1)."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def compute_grad_fft(w, f):
    """
    Compute h[i] = sum_m w[i+m] * f[m] for i=0..n-1 using FFT.
    w has length 2n-1, f has length n.
    Uses: h[i] = (f_rev * w)[i + n - 1] where f_rev = f[::-1].
    """
    n = len(f)
    f_rev = f[::-1]
    N = 1
    while N < n + (2 * n - 1) - 1:
        N <<= 1
    F = np.fft.rfft(f_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F * W, N)
    return conv[n - 1:2 * n - 1]


def adam_optimize(f_init, deadline, rng, beta_scale=0.1, lr=0.1,
                  max_iters=100000, beta_decay=0.97, beta_min_scale=0.005,
                  restart_patience=150, var_reg=0.0):
    """
    Adam optimization in log-space to minimize soft-max of auto-convolution.
    f_init: probability vector (sum=1, non-negative).
    Returns (best_f, best_M).
    """
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).copy()
    f = f / f.sum()

    z = np.log(f)
    z -= z.mean()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = np.convolve(f, f)
    M = float(c.max())
    beta = beta_scale * M

    best_f = f.copy()
    best_M = M
    no_imp = 0
    it = 0

    while it < max_iters:
        if time.monotonic() >= deadline:
            break

        # Compute gradient of soft-max(c, beta) w.r.t. f
        w = soft_max_weights(c, beta)
        h = compute_grad_fft(w, f)
        grad = 2.0 * h

        # Add variance regularization if requested
        if var_reg > 0:
            c_mean = float(c.mean())
            c_var_grad = 2.0 * (c - c_mean)  # d/d f of var(c) through c
            # Actually var = mean(c^2) - mean(c)^2
            # d var/df[i] = 2*(mean(c) - c_mean)*... this is complex.
            # Skip for simplicity; soft-max already handles it.
            pass

        # Project to tangent space (log-param gradient)
        dot = float(np.dot(f, grad))
        gz = f * (grad - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_f, 1e-15))
            z -= z.mean()
            m_adam.fill(0)
            v_adam.fill(0)
            t = 0
            it += 1
            continue

        # Adam update
        t += 1
        m_adam = b1 * m_adam + (1 - b1) * gz
        v_adam = b2 * v_adam + (1 - b2) * (gz * gz)
        mh = m_adam / (1 - b1 ** t)
        vh = v_adam / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        # Convert to probability
        z = np.clip(z, -50, 50)
        z -= z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-30 or not np.isfinite(s):
            z = np.log(np.maximum(best_f, 1e-15))
            z -= z.mean()
            m_adam.fill(0)
            v_adam.fill(0)
            t = 0
            it += 1
            continue
        f = e / s

        # Evaluate
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-11:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Decay beta
        if it % 50 == 0:
            beta = max(beta_min_scale * best_M, beta * beta_decay)

        # Adaptive learning rate
        if no_imp > 80:
            lr = max(0.01, lr * 0.95)

        # Restart
        if no_imp > restart_patience:
            noise_scale = 0.5 + 0.3 * (no_imp - restart_patience) / restart_patience
            z = np.log(np.maximum(best_f, 1e-15))
            z += noise_scale * rng.normal(size=n)
            z -= z.mean()
            z = np.clip(z, -50, 50)
            e = np.exp(z - z.max())
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = beta_scale * best_M
            m_adam.fill(0)
            v_adam.fill(0)
            t = 0
            lr = 0.1
            no_imp = 0

        it += 1

    return best_f, best_M


def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    f = (np.exp(-0.5 * (t / width) ** 2) +
         np.exp(-0.5 * ((t - 1) / width) ** 2) + 0.05)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cosine_profile(n, freq=1.0, depth=0.5):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 - depth * np.cos(2 * np.pi * freq * t / n)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_trapezoid_profile(n, frac=0.3):
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    idx_rise = int(frac * n)
    idx_fall = n - idx_rise
    f[:idx_rise] = np.linspace(0.3, 1.0, idx_rise)
    f[idx_fall:] = np.linspace(1.0, 0.3, n - idx_fall)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def get_starting_points(n, rng, count=10):

    starts.append(make_linear_profile(n, 1.0))

def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def lp_polish(f_init, deadline, max_iters=300):
    """LP-based polishing: alternate between f and g to reduce M."""
    n = len(f_init)
    f = np.maximum(f_init, 0).copy()
    s = f.sum()
    if s < 1e-15:
        return f_init, compute_conv_max(f_init)
    f = f / s
    M = compute_conv_max(f)
    best_f = f.copy()
    best_M = M

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # Build convolution matrix A: (2n-1) x n, A[k,j] = f[k-j] if valid
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 4.0})
            if not res.success or res.x is None:
                break
            g = np.maximum(res.x, 0.0)
            g_sum = float(g.sum())
            if g_sum < 1e-15:
                break
            g = g / g_sum

            # Line search
            best_t = 0.0
            best_M_try = M
            best_f_try = f
            for t in [0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
                f_try = (1.0 - t) * f + t * g
                f_try = np.maximum(f_try, 0)
                st = f_try.sum()
                if st < 1e-15:
                    continue
                f_try = f_try / st
                M_try = compute_conv_max(f_try)
                if M_try < best_M_try - 1e-13:
                    best_M_try = M_try
                    best_t = t
                    best_f_try = f_try

            if best_t > 0:
                f = best_f_try
                M = best_M_try
                if M < best_M - 1e-12:
                    best_M = M
                    best_f = f.copy()
            else:
                break
        except Exception:
            break

    return best_f, best_M


def heuristic_push_away(f_init, deadline, rng, max_iters=5000):
    """
    Heuristic: identify peak locations in convolution, reduce f at contributing
    positions, increase at non-contributing positions. Faster than LP.
    """
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).copy()
    f = f / f.sum()
    M = compute_conv_max(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        c = np.convolve(f, f)
        peak_thresh = M - 0.05 * M  # locations close to peak
        peak_locs = np.where(c >= peak_thresh)[0]

        if len(peak_locs) == 0:
            break

        # For each peak location k, the contributing pairs (j, k-j) should be reduced
        # Compute "pressure" on each position
        pressure = np.zeros(n)
        for k in peak_locs:
            weight = c[k] - (1.0 / (2 * n - 1))  # excess over uniform
            if weight <= 0:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                # f[j] * f[k-j] contributes to c[k]
                # Reduce both f[j] and f[k-j]
                contrib = f[j] * f[k - j]
                # Distribute reduction proportional to contribution
                if contrib > 1e-15:
                    frac = contrib / max(weight, 1e-15)
                    frac = min(frac, 0.5)
                    pressure[j] += frac
                    if k - j != j:
                        pressure[k - j] += frac

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Move mass: reduce where pressure is high, increase where low
        step = 0.05
        f_new = f - step * pressure + step * pressure.mean()
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        M_new = compute_conv_max(f_new)
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                # Try with random perturbation
                f = f + 0.1 * rng.normal(size=n)
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                no_imp = 0

    return best_f, best_M

        all_r.sort(key=lambda x: x[0])
        for i in range(min(3, len(all_r))):
            remaining = phase5_end - time.monotonic()
            if remaining < 30:
                break
            f_ext = interpolate_profile(all_r[i][1], n2)
            alloc = min(remaining / max(1, 3 - i), 60.0)
            run_deadline = time.monotonic() + alloc
            f_best, M_best = adam_optimize(f_ext, run_deadline, rng,
                                            beta_scale=0.1, lr=0.06)
            sc = 2.0 * n2 * M_best
            try_update(f_best, sc)
            results5.append((sc, f_best))
            print(f"  n=512 Adam {i+1}: score={sc:.10f} | best={best_score:.10f}")

        # Quick LP on n=512
        for i in range(min(2, len(results5))):
            remaining = phase5_end - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining / max(1, 2 - i), 30.0)
            run_deadline = time.monotonic() + alloc
            f_best, M_best = lp_polish(results5[i][1], run_deadline, max_iters=40)
            sc = 2.0 * n2 * M_best
            try_update(f_best, sc)
            print(f"  n=512 LP {i+1}: score={sc:.10f} | best={best_score:.10f}")

    # === Phase 6: Final LP polish ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            print(f"\nPhase 6: Final LP polish (n={len(best_seq)}, budget {remaining:.0f}s)")
            f_best, M_best = lp_polish(best_seq, time.monotonic() + remaining, max_iters=200)
            sc = compute_score_normalized(f_best)
            try_update(f_best, sc)
            print(f"  Final: {sc:.10f}")

    # Ensure valid output
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_score_normalized(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = best_seq * (1.0 / max(s, 1e-15))
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score_normalized(best_seq)
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={final_score:.10f}")
    print(f"{'='*60}")

    # Validate with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence confirms: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score != final_score:
            print(f"  (note: discrepancy between local and evaluate)
            "
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
