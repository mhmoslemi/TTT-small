"""
Continuation method with L-BFGS-B for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing with increasing sharpness, scipy L-BFGS-B at each level,
explicit subgradient polish, and multi-restart with diverse perturbations.

import numpy as np
import time
from scipy.optimize import minimize


def _fft_convolve(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    need = 2 * n - 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:need]
    mn = c.min()
    if mn < -1e-10:
        c -= mn
    return c


def _score_normalized(a):
    """Score = 2*n*max(c)/sum(a)^2 for normalized a (sum=1)."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _true_score(a):
    """Exact score using np.convolve."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    mval = float(np.max(c))
    return float(2.0 * n * mval / (s * s))


def _softmax(x, floor=1e-300):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < floor:
        return np.ones_like(x) / x.size
    return e / s

def _continuation_lbfgs(a0, n, max_time, rng, verbose=True):

    Continuation method: minimize log-sum-exp of convolution with increasing beta.
    a0: initial point (sum=1, non-negative)
    Returns best a and score.
    """
    t_start = time.monotonic()
    
    # Initialize in logit space
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    x = np.log(a)
    x -= x.mean()  # center for numerical stability
    
    c = _fft_convolve(a)
    max_c = c.max()
    
    best_x = x.copy()
    best_a = a.copy()
    best_score = float(2.0 * n * max_c)

    # beta * max_c ranges from ~3 to ~200

    current_scale = 3.0

        current_scale *= 1.8

    if verbose:

    for level, beta in enumerate(beta_levels):
        if time.monotonic() - t_start > max_time:
            break
        
        level_time = (max_time - (time.monotonic() - t_start)) / (len(beta_levels) - level)
        level_timeout = min(level_time, 30.0)  # cap per-level time
        
        def obj_and_grad(x_val):
            a_val = _softmax(x_val)
            c_val = _fft_convolve(a_val)
            
            # Log-sum-exp with beta
            bc = beta * c_val
            bc_max = bc.max()
            shifted = bc - bc_max
            exp_shifted = np.exp(shifted)
            log_sum = bc_max + np.log(exp_shifted.sum())
            val = log_sum / beta
            
            # Weights
            w = exp_shifted / exp_shifted.sum()
            
            # Gradient wrt a
            grad_a = 2.0 * np.convolve(w, a_val[::-1])[n - 1: 2 * n - 1]
            
            # Gradient wrt x
            dot = np.dot(a_val, grad_a)
            grad_x = a_val * (grad_a - dot)
            
            return val, grad_x
        
        # Run L-BFGS-B
        t_level_start = time.monotonic()
        result = minimize(
            obj_and_grad,
            x,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': 500,
                'maxfun': 500,
                'gtol': 1e-10,
                'ftol': 1e-14,
                'maxls': 50,
            }
        )
        
        x = np.clip(result.x, -50, 50)
        x -= x.mean()
        a = _softmax(x)
        c = _fft_convolve(a)
        max_c = c.max()
        score = float(2.0 * n * max_c)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_x = x.copy()
        
        if verbose and level % 3 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Level {level}: beta*max_c={beta*max_c:.1f}, score={score:.10f}, "
                  f"time={elapsed:.1f}s, success={result.success}")
    
    return best_a, best_score


def _subgradient_polish(a0, n, max_time, rng, num_steps=5000, step_scale=1e-4):
    """
    Explicit subgradient method with momentum for final polish.
    Uses exact max (not soft-max) for the subgradient.
    """
    t_start = time.monotonic()
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    # Momentum
    v = np.zeros(n)
    momentum = 0.9
    
    # Adaptive step size
    step = step_scale
    no_improve = 0
    
    for step_i in range(num_steps):
        if time.monotonic() - t_start > max_time:
            break
        
        c = _fft_convolve(a)
        max_idx = np.argmax(c)
        
        # Subgradient: direction that increases c[max_idx]
        # To DECREASE c[max_idx], we go in the negative subgradient direction
        # c[max_idx] = sum_j a[j] * a[max_idx - j]
        # dc/dx = 2 * a_correlated_with_delta
        # Subgradient direction (to increase max): 2 * a[max_idx - i] (for valid i)
        
        # The subgradient of max(c) at a is: 2 * v where v[i] = a[max_idx - i] (if valid)
        sg = np.zeros(n)
        valid_start = max(0, max_idx - (n - 1))
        valid_end = min(n - 1, max_idx)
        for i in range(valid_start, valid_end + 1):
            j = max_idx - i
            if 0 <= j < n:
                sg[i] = a[j]
        
        sg = 2.0 * sg
        
        # Project gradient to tangent space of simplex
        dot_a = np.dot(a, sg)
        sg_tangent = sg - dot_a
        
        # Update with momentum
        v = momentum * v - step * sg_tangent
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Maybe increase step slightly
            if step < 1e-3:
                step = min(step * 1.05, 1e-3)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                step *= 0.8
                v *= 0.5
                # Occasionally add noise
                if step < 1e-5:
                    a = best_a.copy()
                    noise = rng.normal(0, 0.01, n)
                    a = a + noise
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    step = 1e-4
                    no_improve = 0
    
    return best_a, best_score


def _multi_beta_lbfgs(a0, n, max_time, rng, verbose=True):
    """
    Multi-start continuation L-BFGS. Try several perturbations and betas.
    """
    best_a = a0.copy()
    best_score = _score_normalized(a0)
    
    t_start = time.monotonic()
    n_tries = 3
    
    for try_i in range(n_tries):
        if time.monotonic() - t_start > max_time:
            break
        
        per_time = (max_time - (time.monotonic() - t_start)) / (n_tries - try_i)
        
        if try_i == 0:
            a_try = a0.copy()
        else:
            # Perturb in logit space
            x = np.log(np.maximum(a0, 1e-15))
            x -= x.mean()
            noise_scale = 0.1 * (try_i + 1)
            x += noise_scale * rng.normal(size=n)
            x = np.clip(x, -50, 50)
            x -= x.mean()
            a_try = _softmax(x)
        
        a_best, sc = _continuation_lbfgs(a_try, n, per_time, rng, verbose=(try_i == 0))
        if sc < best_score:
            best_score = sc
            best_a = a_best.copy()
    
    return best_a, best_score


def _interp_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()

"""
Continuation L-BFGS-B + subgradient polish for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key ideas: (1) log-sum-exp smoothed objective with systematic beta schedule,
(2) scipy L-BFGS-B for efficient convex-like optimization at each smoothing level,

import numpy as np
import time
from scipy.optimize import minimize


def _fft_convolve(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:need]
    mn = c.min()
    if mn < -1e-10:
        c = c - mn
    return c


def _score_norm(a):
    """Score for normalized (sum=1) non-negative array."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _true_score(a):
    """Exact score using np.convolve."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / x.size
    return e / s


def _grad_lse(x, beta, n):
    """Compute log-sum-exp smoothed objective and gradient in logit space.
    x: logit vector (length n)
    beta: smoothing parameter
    Returns (value, gradient).
    """
    a = _softmax(x)
    c = _fft_convolve(a)
    
    bc = beta * c
    bc_max = bc.max()
    shifted = bc - bc_max
    exp_shifted = np.exp(shifted)
    log_s = exp_shifted.sum()
    
    if log_s < 1e-300:
        val = c.max()
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        val = (bc_max + np.log(log_s)) / beta
        w = exp_shifted / log_s
    
    # Gradient wrt a: grad_a[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * convolve(w, a_reversed)[n-1 : 2n-1]
    conv_wa = np.convolve(w, a[::-1])
    grad_a = 2.0 * conv_wa[n - 1: 2 * n - 1]
    
    # Gradient wrt x (softmax Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return val, grad_x

def _subgrad_polish(a0, n, max_time, rng, step_scale=1e-4):
    """
    Subgradient method with momentum for polishing near the optimum.
    Uses exact argmax of convolution for the subgradient.
    Returns (best_a, best_score).
    """
    t_start = time.monotonic()
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    
    v = np.zeros(n)
    mu = 0.85  # momentum
    step = step_scale
    no_imp = 0
    patience = 200
    
    while time.monotonic() - t_start < max_time:
        c = _fft_convolve(a)
        max_idx = int(np.argmax(c))
        
        # Subgradient: sg[i] = 2 * a[max_idx - i] for valid i
        sg = np.zeros(n)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        for i in range(lo, hi + 1):
            j = max_idx - i
            if 0 <= j < n:
                sg[i] = a[j]
        sg *= 2.0
        
        # Project to tangent of simplex
        dot_a = np.dot(a, sg)
        sg_t = sg - dot_a
        
        # Momentum update
        v = mu * v - step * sg_t
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())
        
        if sc_new < best_score - 1e-15:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            step = min(step * 1.02, 5e-3)
        else:
            no_imp += 1
            step *= 0.97
            if no_imp > patience:
                # Restart with noise
                a = best_a + 0.005 * rng.normal(size=n)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                step = step_scale
                no_imp = 0
    
    return best_a, best_score


def _fast_subgrad(a0, n, max_time, rng, step_scale=5e-5):
    """
    Vectorized subgradient polish (faster than loop version).
    Batches the subgradient computation.
    """
    t_start = time.monotonic()

        c = _fft_convolve(a)
        max_idx = int(np.argmax(c))
        
        # Build subgradient efficiently
        # sg[i] = 2*a[max_idx - i] for i in [max(0,max_idx-n+1), min(n-1,max_idx)]
        lo = max(0, max_idx - n + 1)

        sg = np.zeros(n)
        sg[lo:hi+1] = a[max_idx - (hi):max_idx - lo - 1:-1] * 2.0
        
        dot_a = np.dot(a, sg)
        sg_t = sg - dot_a
        
        v = mu * v - step * sg_t
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)

            no_imp = 0
            if step < 1e-3:

            step *= 0.98
            if no_imp > 300:
                a = best_a + 0.003 * rng.normal(size=n)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                step = step_scale
                no_imp = 0
    
    return best_a, best_score


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-10)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s

            x += noise * rng.normal(size=n_b)
            x = np.clip(x, -50, 50)
            x -= x.mean()
            a_start = _softmax(x)
            
            a_try, sc_try = _continuation_opt(a_start, n_b, per_t, rng, verbose=False)
            _upd(a_try, sc_try, f"restart-{i}")
    
    # === Phase 4: Try different lengths ===
    if time.monotonic() < deadline:
        p4_time = (deadline - time.monotonic()) * 0.3
        print(f"\nPhase 4: Multi-length exploration, time={p4_time:.0f}s")
        
        # Try n=512
        if time.monotonic() < deadline - 30:
            n_try = 512
            if best_seq is not None:
                a_512 = _interp_profile(best_seq, n_try)
            else:
                a_512 = np.ones(512) / 512
            t_alloc = min(60.0, p4_time * 0.3)
            a_try, sc_try = _continuation_opt(a_512, n_try, t_alloc, rng, verbose=False)
            _upd(a_try, sc_try, "n=512")
            if time.monotonic() < deadline - 10:
                a_pol, sc_pol = _fast_subgrad(a_try, n_try, min(20.0, (deadline - time.monotonic()) * 0.1), rng)
                _upd(a_pol, sc_pol, "n=512-polish")
        
        # Try n=2048
        if time.monotonic() < deadline - 60:
            n_try = 2048
            if best_seq is not None:
                a_2048 = _interp_profile(best_seq, n_try)
            else:
                a_2048 = np.ones(2048) / 2048
            sc_interp = _score_norm(a_2048)
            print(f"  n=2048 interp: {sc_interp:.10f}")
            _upd(a_2048, sc_interp, "n=2048-interp")
            
            t_alloc = min(90.0, (deadline - time.monotonic()) * 0.25)
            a_try, sc_try = _continuation_opt(a_2048, n_try, t_alloc, rng, verbose=False)
            _upd(a_try, sc_try, "n=2048")
            
            if time.monotonic() < deadline - 10:
                a_pol, sc_pol = _fast_subgrad(a_try, n_try, min(30.0, (deadline - time.monotonic()) * 0.1), rng)
                _upd(a_pol, sc_pol, "n=2048-polish")
    
    # === Phase 5: Final intensive optimization on best ===
    if best_seq is not None and time.monotonic() < deadline:
        n_b = best_seq.size
        p5_time = deadline - time.monotonic()
        print(f"\nPhase 5: Final optimization, n={n_b}, time={p5_time:.0f}s")
        
        # L-BFGS continuation
        t_alloc = p5_time * 0.4
        a_fin, sc_fin = _continuation_opt(best_seq, n_b, t_alloc, rng, verbose=False)
        _upd(a_fin, sc_fin, "final-cont")
        
        # Subgradient polish
        if time.monotonic() < deadline - 5:
            t_alloc2 = min(p5_time * 0.5, deadline - time.monotonic() - 5)
            a_pol, sc_pol = _fast_subgrad(best_seq, n_b, t_alloc2, rng, step_scale=1e-5)
            _upd(a_pol, sc_pol, "final-subgrad")
        
        # One more L-BFGS
        if time.monotonic() < deadline - 5:
            t_alloc3 = min(30.0, deadline - time.monotonic() - 5)
            a_fin2, sc_fin2 = _continuation_opt(best_seq, n_b, t_alloc3, rng, verbose=False)
            _upd(a_fin2, sc_fin2, "final-cont2")
    
    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = np.ones(256) / 256
        else:
            final_seq = final_seq / s
        
        sc_exact = _true_score(final_seq)
        print(f"\nFinal exact score (np.convolve): {sc_exact:.10f}")
        print(f"Final best_score: {best_score:.10f}")
        print(f"Length: {final_seq.size}")
        
        # Check with evaluator if possible
        if final_seq.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirms: {sc_eval:.10f}")
            except Exception:
                pass
        
        return [float(x) for x in final_seq]
    
    # Fallback
    print("FALLBACK: uniform n=256")
    fb = np.ones(256) / 256
    return [float(x) for x in fb]
