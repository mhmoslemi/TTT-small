- `L(t) = max_{k ∈ S} (c[k] - 2*a[i]*a[k-i] + 2*a[k-i]*t)` (max of linear functions)
- `P(t) = c[2i] - a[i]^2 + t^2` (parabola, if `2i` is in range)
- `C = max_{k ∉ [i,i+n-1]} c[k]` (constant)

The minimum of `f(t)` for `t >= 0` can be found efficiently:
1. `L(t)` is convex piecewise linear, minimum at a "kink" or boundary
2. `P(t)` has minimum at `t = 0` (since coefficient of `t^2` is positive)
3. `C` is constant

So the minimum of `f` is at the minimum of `max(L(t), P(t), C)` for `t >= 0`.

Since `P(t)` = `t^2 + const` is increasing for `t > 0`, and `L(t)` is the max of lines (some increasing, some decreasing in `t`), the interesting region is where `L(t)` and `P(t)` cross, or where `L(t) = C`, or at `t = 0`.

For a practical implementation, I'll use binary search on the value level:
- Binary search on `M`: is there a `t >= 0` such that `f(t) <= M`?
  - This is equivalent to: is there a `t >= 0` such that all constraints are satisfied?
  - For each `k ∈ S`: `c[k] + 2*a[k-i]*(t-a[i]) <= M` → `t <= (M - c[k]) / (2*a[k-i]) + a[i]` (if `a[k-i] > 0`) or `t >= (M - c[k]) / (2*a[k-i]) + a[i]` (if `a[k-i] < 0`, but `a >= 0` so this doesn't happen)
  - For `k = 2i`: `t^2 + c[2i] - a[i]^2 <= M` → `t <= sqrt(M - c[2i] + a[i]^2)` (if the expression under sqrt is non-negative)
  - For `k ∉ [i,i+n-1]`: `c[k] <= M` (must already be satisfied)

So the feasible set for `t` is an interval `[0, T_max(M)]` where `T_max(M) = min over all constraints of the upper bound`. We need `T_max(M) >= 0`.

The minimum `M` is the smallest value such that `T_max(M) >= 0`, i.e., all upper bounds are non-negative.

This is a clean bisection approach! Let me implement it:

    k_lo = max(0, i)
    k_hi = min(2*n-2, i + n - 1)
    
    if k_lo > k_hi:
        return a[i], float(c.max())

    if k_lo == 0 and k_hi == 2*n-2:
        C = 0.0
    else:
        parts = []

        if k_hi < 2*n-2:
            parts.append(c[k_hi+1:].max())

    M_hi = M_lo  # we'll try to reduce M_lo

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based descent, coordinate descent, and soft-max Adam polishing.
Key improvements: coordinate descent with 1D least-max optimization,
better time management, multi-n exploration, and adaptive step sizes.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=False):
    """LP-based optimization with adaptive step size and Nesterov momentum."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    best_M = float(np.convolve(f, f).max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0
    v_f = np.zeros(n)  # Nesterov momentum

    while time.monotonic() < deadline:
        M = float(np.convolve(f, f).max())

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            # Try with slightly relaxed M
            g = lp_step(f, M * 1.01)
            if g is None or g.sum() < 1e-10:
                break
        g = g / g.sum()

        # Nesterov: use momentum in the direction
        direction = (1 - t) * f + t * g
        v_f = 0.85 * v_f + (direction - f)
        f_new = f + v_f
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                t *= 0.5
                continue
        f_new = f_new / s_new
        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 200 == 0:
                print(f"    LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
        else:
            v_f = 0.5 * v_f
            t = max(0.005, t * 0.9)
            stagnant += 1

        t = max(0.005, t * 0.9985)

        if stagnant > 100:
            noise_scale = 0.15 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.25
            stagnant = 0
            v_f = np.zeros(n)
            if verbose:
                print(f"    LP restart at iter {iteration}, M={best_M:.10f}")

    return best_f


def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def logit_anneal(p0, deadline, rng, lr=0.06, reg=1e-8):

    n = p0.size

    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999

    c = full_convolve_fft(p)

    while time.monotonic() < deadline:

        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))

        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)

        adam_t += 1

        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
                lr_patience = 0
            else:
                no_imp += 1
                lr_patience += 1
                if lr_patience % 500 == 0:

        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0

            z = mix * best_z + (1 - mix) * rng.normal(size=n) * 0.5
            z = np.clip(z, -60.0, 60.0)

            p = softmax_from_z(z)
            c = full_convolve_fft(p)

            beta = max(1e-12, 0.2 * maxc)

            v.fill(0.0)
            adam_t = 0

def coord_descent(a_init, deadline, rng, max_sweeps=200):
    """
    Coordinate descent for minimizing max(conv(a,a)) with a >= 0.
    Score is scale-invariant so we work without sum constraint.
    """
    a = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 0.0).copy()
    n = a.size
    if n == 0:
        return a

    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = full_convolve_fft(a)
    best_M = float(c.max())
    best_a = a.copy()
    sweep = 0
    no_imp_sweeps = 0

    while time.monotonic() < deadline and sweep < max_sweeps:
        sweep += 1
        # Random order for robustness
        order = rng.permutation(n)
        imp_this_sweep = 0

        for i in order:
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-14:
                continue

            old_val = a[i]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Compute unaffected max
            if k_lo == 0 and k_hi == 2 * n - 2:
                C = 0.0
            else:
                C = 0.0
                if k_lo > 0:
                    C = max(C, float(c[:k_lo].max()))
                if k_hi < 2 * n - 2:
                    C = max(C, float(c[k_hi + 1:].max()))

            # Precompute linear coefficients for affected range
            # c[k] + 2*(t - old_val)*a[k-i] for k in [k_lo, k_hi], k != 2i
            # c[2i] + t^2 - old_val^2 for k = 2i (if in range)
            offsets = np.arange(k_lo, k_hi + 1)

            c_slice = c[offsets]
            lin_base = c_slice - 2.0 * old_val * a_slice
            lin_slope = 2.0 * a_slice

            if has_quad:

                quad_base = c[2 * i] - old_val * old_val

            # Find optimal t using bisection on M
            # f(t) = max(max(lin_base + lin_slope * t for linear terms), quad_base + t^2, C)
            # We want min over t >= 0

            f_current = float(c.max())

            # Try t = 0
            def eval_f(t):
                vals = lin_base + lin_slope * t
                if has_quad:
                    vals[quad_idx] = quad_base + t * t
                return max(float(vals.max()), C)

            # Initial candidates
            candidates = [0.0, old_val * 0.1, old_val * 0.25, old_val * 0.5,
                          old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5]
            # Add some random candidates
            candidates.append(old_val * rng.uniform(0.0, 2.0))

            best_t = old_val
            best_f_val = f_current

            for t in candidates:
                t = max(0.0, t)
                fv = eval_f(t)
                if fv < best_f_val - 1e-15:
                    best_f_val = fv
                    best_t = t

            # Refine around best_t with finer grid
            if best_t > 0:
                step = max(best_t * 0.05, 1e-10)
            else:
                step = max(old_val * 0.01, 1e-10)

            for frac in np.array([0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]):
                for sign in [1, -1]:
                    t = best_t + sign * frac * step
                    if t < 0:
                        continue
                    fv = eval_f(t)
                    if fv < best_f_val - 1e-15:
                        best_f_val = fv
                        best_t = t

            if best_f_val < f_current - 1e-14:
                # Update a and c
                a[i] = best_t
                c[k_lo:k_hi + 1] = lin_base + lin_slope * best_t
                if has_quad:
                    c[2 * i] = quad_base + best_t * best_t
                imp_this_sweep += 1

        # Periodic recomputation for accuracy
        if sweep % 5 == 0:
            c = full_convolve_fft(a)
            new_M = float(c.max())
            if new_M < best_M - 1e-13:
                best_M = new_M
                best_a = a.copy()
                no_imp_sweeps = 0
            else:
                no_imp_sweeps += 1
        else:
            new_M = float(c.max())
            if new_M < best_M - 1e-13:
                best_M = new_M
                best_a = a.copy()
                no_imp_sweeps = 0

        if no_imp_sweeps > 30:
            # Restart with perturbation
            noise = 0.02 * rng.normal(size=n)
            a = best_a + noise * a
            a = np.maximum(a, 1e-14)
            a = a / a.sum()
            c = full_convolve_fft(a)
            no_imp_sweeps = 0

    # Normalize back
    s = best_a.sum()
    if s > 1e-10:
        best_a = best_a / s
    return best_a


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, phase=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_exponential_profile(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 1.0))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bimodal_profile(n, center1=0.3, center2=0.7, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2) + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def get_starting_profiles(n, rng):
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        profiles.append((f"lin_{eta}", make_linear_profile(n, eta)))
    for k in [1, 2, 3, 4]:
        for phase in [0.0, np.pi / 3, 2 * np.pi / 3]:
            profiles.append((f"cos_{k}_{phase:.2f}", make_cosine_profile(n, k, phase)))
    for rate in [2.0, 3.0, 5.0, 7.0]:
        profiles.append((f"exp_{rate}", make_exponential_profile(n, rate)))
        profiles.append((f"exp_rev_{rate}", make_exponential_profile(n, -rate)))
    profiles.append(("bimodal1", make_bimodal_profile(n, 0.3, 0.7, 0.12)))
    profiles.append(("bimodal2", make_bimodal_profile(n, 0.2, 0.8, 0.18)))
    # Random smooth
    for trial in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 7)
        t = np.linspace(0.0, 1.0, n)
        a = 1.0
        for i in range(1, 7):
            a = a + coeffs[i] * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append((f"rand_{trial}", a / a.sum()))
    return profiles


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 0.0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP descent, coordinate descent, soft-max polish,
    multi-n exploration. Optimized time management for maximum improvement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ============ Phase 1: LP refinement at n=512 ============
    n_main = 512
    phase1_end = min(deadline, t_start + 350.0)

    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_linear_profile(n_main, 2.0)

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1 start: n={n_main}, score={sc:.10f}")

    # LP from main starting point
    lp_deadline = min(phase1_end, time.monotonic() + 250.0)
    result = lp_optimize(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP: score={sc:.10f}")

    # Try a few alternative profiles with short LP
    profiles = get_starting_profiles(n_main, rng)
    for name, p in profiles[:8]:
        if time.monotonic() > phase1_end - 3:
            break
        sc_p = compute_score(p)
        if sc_p < best_score + 0.02:
            per_dl = min(phase1_end, time.monotonic() + 20.0)
            result = lp_optimize(p, per_dl, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            if sc < best_score + 0.01:
                print(f"  Alt profile {name}: {sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f}, n={best_seq is not None and best_seq.size or 'N/A'}")

    # ============ Phase 2: Coordinate descent ============
    phase2_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_cd = best_seq.size
        print(f"Phase 2: coord descent at n={n_cd}, current best={best_score:.10f}")

        # CD round 1
        cd_deadline = min(phase2_end, time.monotonic() + 250.0)
        p_cd = best_seq.copy()
        result = coord_descent(p_cd, cd_deadline, rng, max_sweeps=100)
        sc = compute_score(result)
        update(result, sc)
        print(f"  CD round 1: score={sc:.10f}")

        # CD round 2 with perturbation
        if time.monotonic() < phase2_end - 20:
            p_cd2 = best_seq + 0.005 * rng.normal(size=best_seq.size)
            p_cd2 = np.maximum(p_cd2, 1e-12)
            p_cd2 = p_cd2 / p_cd2.sum()
            cd_deadline2 = min(phase2_end, time.monotonic() + 150.0)
            result = coord_descent(p_cd2, cd_deadline2, rng, max_sweeps=80)
            sc = compute_score(result)
            update(result, sc)
            print(f"  CD round 2: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Soft-max Adam polish ============
    phase3_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 3: soft-max polish at n={n_polish}, current best={best_score:.10f}")

        # Multiple polish rounds
        for round_idx, (lr, duration) in enumerate([(0.06, 150), (0.04, 120), (0.03, 100)]):
            if time.monotonic() > phase3_end - 5:
                break
            per_dl = min(phase3_end, time.monotonic() + duration)
            p0 = best_seq.copy()
            # Small perturbation to avoid exact local minimum
            noise_scale = 0.002 / (1 + round_idx)
            p0 = p0 + noise_scale * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-14)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_dl, rng, lr=lr)
            update(result, sc)
            print(f"  Adam round {round_idx+1} (lr={lr}): score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Multi-n exploration ============
    phase4_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_end - 20 and best_seq is not None:
        print(f"Phase 4: multi-n exploration, best={best_score:.10f}")

        for n_try in [384, 640, 768, 1024, 1280]:
            if time.monotonic() > phase4_end - 15:
                break
            if n_try == best_seq.size:
                continue
            p_interp = interpolate_profile(best_seq, n_try)
            sc_interp = compute_score(p_interp)
            update(p_interp, sc_interp)

            # Quick CD at this n
            if sc_interp < best_score + 0.01:
                per_dl = min(phase4_end, time.monotonic() + 40.0)
                result = coord_descent(p_interp, per_dl, rng, max_sweeps=30)
                sc = compute_score(result)
                update(result, sc)
                print(f"  n={n_try}: CD score={sc:.10f}")

            # Quick Adam at this n if CD helped
            if best_seq.size != n_try and time.monotonic() < phase4_end - 10:
                per_dl2 = min(phase4_end, time.monotonic() + 30.0)
                p0 = best_seq + 0.001 * rng.normal(size=best_seq.size)
                p0 = np.maximum(p0, 1e-14)
                p0 = p0 / p0.sum()
                result, sc = logit_anneal(p0, per_dl2, rng, lr=0.04)
                update(result, sc)
                print(f"  n={best_seq.size}: Adam score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 'N/A'}")

    # ============ Phase 5: Final coordinate descent polish ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = min(deadline - 10, time.monotonic() + 60.0)
        n_final = best_seq.size
        print(f"Phase 5: final CD at n={n_final}, best={best_score:.10f}")
        result = coord_descent(best_seq, per, rng, max_sweeps=50)
        sc = compute_score(result)
        update(result, sc)
        print(f"  Final CD: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 2.0)
        s = final_seq.sum()
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
