from scipy.optimize import linprog
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FISTA subgradient, LP-based direction finding, and Adam polishing.
Multi-scale approach: optimize at n=128, upsample to n=256, then n=512.

def compute_score_normalized(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f (assumes sum=1)."""
    n = len(f)
    c = np.convolve(f, f)
    return float(2.0 * n * c.max())

def compute_M(f):
    """Compute max of linear convolution of f with itself."""
    return float(np.convolve(f, f).max())

def normalize_simplex(f):
    """Project onto simplex: f >= 0, sum(f) = 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the standard projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def subgradient_max_conv(f):
    """Compute subgradient of max(conv(f,f)) with respect to f (flat norm, sum=1).
    Returns the gradient vector g such that g[i] = d(max_k conv_k)/df_i.
    """
    n = len(f)
    c = np.convolve(f, f)
    k_star = int(np.argmax(c))
    # grad_k[i] = f[k-i] + f[k-i] if i != k-i, else f[i]
    # More precisely: d(conv_k)/df_i where conv_k = sum_j f_j * f[k-j]
    # = f[k-i] (from j=i term) + f[k-i] (from k-j=i term)
    # But if i = k-i (i.e., 2i = k), it's just f[i] (single term)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] += f[j]
            else:
                grad[i] += 2.0 * f[j]
    return grad

def subgradient_multi_max_conv(f, k_active=5):
    """Compute averaged subgradient over top-k active convolution peaks."""
    n = len(f)
    c = np.convolve(f, f)
    # Get top k indices
    top_k = np.argsort(c)[::-1][:k_active]
    grad = np.zeros(n)
    for k_star in top_k:
        k_star = int(k_star)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] += f[j]
                else:
                    grad[i] += 2.0 * f[j]
    grad /= k_active
    return grad

def fista_optimize(f_init, deadline, n_iter_max=5000, lr_init=0.01, verbose=False):
    """FISTA (accelerated proximal gradient) for minimizing max(conv(f,f)) on the simplex."""
    f = normalize_simplex(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    
    n = len(f)
    y = f.copy()
    t_k = 1.0
    lr = lr_init
    best_f = f.copy()
    best_M = compute_M(f)
    no_imp = 0
    
    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        g = subgradient_multi_max_conv(y, k_active=3)
        
        # Gradient step
        y_new = y - lr * g
        # Project onto simplex
        x_new = project_simplex(y_new)
        
        # Check improvement
        M_new = compute_M(x_new)
        if M_new < best_M - 1e-10:
            best_M = M_new
            best_f = x_new.copy()
            no_imp = 0
        else:

        elif no_imp > 50:
            lr = lr * 0.95

        y = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
        y = project_simplex(y)  # Keep y on simplex for stability
        t_k = t_new
        
        if verbose and it % 500 == 0:
            print(f"    FISTA iter {it}: M={M_new:.10f}, best={best_M:.10f}, lr={lr:.6f}")

def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A

def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        g = solve_lp_for_g(f, M)

def soft_max_gradient(c, beta):
    """Compute gradient of soft-max of c."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 50))
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def adam_polish(f_init, deadline, rng, lr=0.05, max_iters=5000, verbose=False):
    """Adam optimization in log-space for maximizing flatness."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-12, 0.3 * M)
    
    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        L, w = soft_max_gradient(c, beta)
        gp = 2.0 * corr_weights(w, f)
        # Add small regularization to prevent collapse
        vp = gp + 1e-6 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())
        
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.01, lr * 0.9)
        
        # Anneal beta
        if adam_t % 200 == 0:
            beta = max(1e-12, beta * 0.75)
        
        if no_imp > max(400, 2 * n):
            # Restart with perturbation
            pert_scale = 0.2 + 0.2 * rng.random()
            z = (1.0 - pert_scale) * best_z + pert_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.3 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
        
        if verbose and it % 500 == 0:
            print(f"    Adam iter {it}: M={M:.10f}, best={best_M:.10f}, beta={beta:.6f}")
    
    return best_f, best_M

def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return result

def make_linear_profile(n, slope=2.0):
    """Linear ramp profile."""
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_cubic_profile(n, a1=1.5, a3=-1.0):
    """Anti-symmetric cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    """Gaussian bump profile."""
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_dual_bump(n):
    """Two bumps at edges."""
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_beta_profile(n, a=2.0, b=2.0):
    """Beta distribution shape: t^a * (1-t)^b."""
    t = np.linspace(0.01, 0.99, n)
    f = t ** a * (1.0 - t) ** b
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_raised_cosine(n):
    """Raised cosine: 0.5*(1 - cos(2*pi*t))."""
    t = np.linspace(0.0, 1.0, n)
    f = 0.5 * (1.0 - np.cos(2.0 * np.pi * t))
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_sine_bump(n):
    """Sine bump: sin(pi*t)."""
    t = np.linspace(0.0, 1.0, n)
    f = np.sin(np.pi * t)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_asym_profile(n, alpha=0.3, beta=0.7):
    """Asymmetric profile: peaks at position alpha."""
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - alpha) / 0.2) ** 2) + 0.3 * np.exp(-0.5 * ((t - beta) / 0.15) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def make_random_fourier(n, rng, n_modes=6, amplitude=0.4):
    """Random smooth profile via Fourier series."""
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, n_modes + 1):
        a = rng.uniform(-amplitude, amplitude)
        b = rng.uniform(-amplitude, amplitude)
        f = f + a * np.cos(2 * np.pi * k * t) + b * np.sin(2 * np.pi * k * t)
    f = np.maximum(f, 1e-6)
    return f / f.sum()

import numpy as np
from scipy.optimize import linprog
import time
import math

"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses LP-based direction finding, FISTA subgradient, Adam polishing,
simulated annealing for escape, and multi-scale refinement.
"""


def compute_score_normalized(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f (assumes sum=1)."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_M(f):
    """Compute max of linear convolution of f with itself."""
    return float(np.convolve(f, f).max())


def normalize_simplex(f):
    """Project onto simplex: f >= 0, sum(f) = 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def subgradient_max_conv(f, k_active=3):
    """Compute averaged subgradient over top-k active convolution peaks."""
    n = len(f)
    c = np.convolve(f, f)
    top_k = np.argsort(c)[::-1][:k_active]
    grad = np.zeros(n)
    for idx in top_k:
        k_star = int(idx)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] += f[j]
                else:
                    grad[i] += 2.0 * f[j]
    grad /= k_active
    return grad


def fista_optimize(f_init, deadline, lr_init=0.005, max_iters=5000):
    """FISTA for minimizing max(conv(f,f)) on the simplex."""
    f = normalize_simplex(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    n = len(f)
    y = f.copy()
    t_k = 1.0
    lr = lr_init
    best_f = f.copy()
    best_M = compute_M(f)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        g = subgradient_max_conv(y, k_active=3)
        y_step = y - lr * g
        x_new = project_simplex(y_step)
        M_new = compute_M(x_new)
        if M_new < best_M - 1e-10:
            best_M = M_new
            best_f = x_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        if M_new > best_M * 1.1:
            lr *= 0.7
        elif no_imp > 100:
            lr *= 0.95
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
        y = project_simplex(y)
        t_k = t_new

    return best_f, best_M


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 4.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, max_iters=500):
    """LP-based iterative optimization with line search."""
    f = normalize_simplex(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        g = solve_lp_for_g(f, M)
        if g is None:
            f = normalize_simplex(f + 0.05 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            continue
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum
        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.9, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_simplex(f_try)
            if f_try is None:
                continue
            M_try = compute_M(f_try)
            if M_try < best_M_try - 1e-12:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t
        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > 15:
            pert = 0.15 + 0.15 * rng.random()
            pert_f = normalize_simplex(best_f + 0.3 * rng.normal(size=n))
            if pert_f is None:
                pert_f = best_f
            f = (1.0 - pert) * best_f + pert * pert_f
            f = normalize_simplex(f)
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            no_imp = 0

    return best_f, best_M


def soft_max_gradient(c, beta):
    """Compute gradient of soft-max of c."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 50))
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def adam_polish(f_init, deadline, rng, lr=0.04, max_iters=5000):
    """Adam optimization in log-space with adaptive beta annealing."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-12, 0.3 * M)
    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        L, w = soft_max_gradient(c, beta)
        gp = 2.0 * corr_weights(w, f)
        vp = gp + 1e-6 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.008, lr * 0.9)
        if adam_t % 150 == 0:
            beta = max(1e-12, beta * 0.7)
        if no_imp > max(500, 2 * n):
            pert_scale = 0.15 + 0.25 * rng.random()
            z = (1.0 - pert_scale) * best_z + pert_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.3 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_f, best_M


def simulated_annealing_optimize(f_init, deadline, rng, iterations=5000):
    """Simulated annealing with LP-guided moves and random perturbations."""
    f = normalize_simplex(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    T = 0.05
    T_min = 0.001
    cooling = 0.999

    for it in range(iterations):
        if time.monotonic() > deadline:
            break
        # Choose move type
        r = rng.random()
        if r < 0.5:
            # LP-guided move
            g = solve_lp_for_g(f, M)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                t = rng.uniform(0.1, 0.6)
                f_new = (1.0 - t) * f + t * g
                f_new = normalize_simplex(f_new)
            else:
                f_new = None
        elif r < 0.8:
            # Smooth random perturbation
            delta = rng.normal(size=n)
            # Apply low-pass filter
            delta_k = np.fft.rfft(delta)
            delta_k[4:] *= 0.3
            delta_k[8:] *= 0.1
            delta = np.fft.irfft(delta_k, n)
            scale = rng.uniform(0.02, 0.15)
            f_new = normalize_simplex(f + scale * delta)
        else:
            # Small local perturbation at random positions
            n_pert = max(1, n // 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            f_new = f.copy()
            f_new[idx] *= np.exp(rng.normal(size=n_pert) * 0.3)
            f_new = normalize_simplex(f_new)

        if f_new is None:
            continue
        M_new = compute_M(f_new)
        delta_M = M_new - M
        if delta_M < 0 or rng.random() < math.exp(-delta_M / max(T, 1e-10)):
            f = f_new
            M = M_new
            if M < best_M:
                best_M = M
                best_f = f.copy()
        T = max(T_min, T * cooling)

    return best_f, best_M


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def make_linear_profile(n, slope=2.0):
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n):
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_beta_profile(n, a=2.0, b=2.0):
    t = np.linspace(0.01, 0.99, n)
    f = t ** a * (1.0 - t) ** b
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_sine_bump(n):
    t = np.linspace(0.0, 1.0, n)
    f = np.sin(np.pi * t)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_random_fourier(n, rng, n_modes=6, amplitude=0.4):
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, n_modes + 1):
        a = rng.uniform(-amplitude, amplitude)
        b = rng.uniform(-amplitude, amplitude)
        f = f + a * np.cos(2 * np.pi * k * t) + b * np.sin(2 * np.pi * k * t)
    f = np.maximum(f, 1e-6)
    return f / f.sum()


def get_starting_points(n, rng, count=15):
    """Generate diverse starting points."""
    starts = []
    starts.append(make_linear_profile(n, 0.5))
    starts.append(make_linear_profile(n, 1.0))
    starts.append(make_linear_profile(n, 2.0))
    starts.append(make_linear_profile(n, 3.0))
    starts.append(make_linear_profile(n, 4.0))
    starts.append(make_cubic_profile(n, 1.0, -0.5))
    starts.append(make_cubic_profile(n, 1.5, -1.0))
    starts.append(make_cubic_profile(n, 2.0, -1.5))
    starts.append(make_cubic_profile(n, 2.5, -2.0))
    starts.append(make_dual_bump(n))
    starts.append(make_gaussian_bump(n, 0.3, 0.25))
    starts.append(make_gaussian_bump(n, 0.7, 0.25))
    starts.append(make_beta_profile(n, 2.0, 2.0))
    starts.append(make_beta_profile(n, 3.0, 1.0))
    starts.append(make_beta_profile(n, 1.5, 2.5))
    starts.append(make_sine_bump(n))
    for _ in range(count):
        starts.append(make_random_fourier(n, rng, n_modes=int(rng.integers(3, 10)),
                                          amplitude=rng.uniform(0.2, 0.6)))
    return starts

        if f_norm is not None:
            sc_final = compute_score_normalized(f_norm)
            update_best(f_norm, sc_final)
            print(f"    Final polish: score={sc_final:.10f}, best={best_score:.10f}")

    # Ensure valid output
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score_normalized(best_seq)
    print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.10f} ===")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
