Uses mirror descent with Nesterov momentum in logit space, with aggressive
beta annealing and multiple restarts at different sequence lengths.

import numpy as np

import math


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn if mn > -1e-9 else np.maximum(c, 0.0)
    return c


def soft_max_val(c, beta):
    """Compute log-sum-exp approximation of max(c)."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    return mval + beta * float(np.log(s))


def soft_max_weights(c, beta):
    """Compute soft-max weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    return ex / s


def corr_grad(w, p):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. p, where c = conv(p,p).
    grad[i] = sum_k w[k] * (p[k-i] + p[i-k]) for valid indices

    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    # conv(w, p) gives us the correlation
    # Actually, dc[k]/dp[i] = p[k-i] if 0<=k-i<n
    # So grad[i] = sum_k w[k] * p[k-i] = (w * p)[i] where * is cross-correlation
    # cross-correlation: (w ⊛ p)[i] = sum_k w[k] * p[k-i]
    # In terms of convolution: (w ⊛ p)[i] = conv(w[::-1], p)[i + len(w) - 1]

    fp_rev = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp_rev, N)

    return A


def lp_descent_step(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def logit_adam(p_init, deadline, rng, lr=0.1, beta_start=0.05,
               b1=0.9, b2=0.999, restart_every=3000, beta_decay=0.97,
               beta_period=50, noise_scale=0.6):
    """Adam optimizer in logit space with soft-max objective."""
    n = len(p_init)
    p = np.maximum(np.asarray(p_init, dtype=np.float64), 1e-15).copy()
    p = p / p.sum()

    z = np.log(p)
    z -= float(np.mean(z))

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    eps = 1e-8

    beta = beta_start
    best_p = p.copy()
    c = full_convolve_fft(p)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(p)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1

        _, w = logsumexp_max_weights(c, beta)
        g_p = 2.0 * corr_grad_fft(w, p)
        dot = float(np.dot(p, g_p))
        g_z = p * (g_p - dot)

        if not np.all(np.isfinite(g_z)):
            z = np.log(best_p)
            z -= float(np.mean(z))
            p = best_p.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * g_z
        v = b2 * v + (1.0 - b2) * (g_z * g_z)
        m_h = m / (1.0 - b1 ** t_adam)
        v_h = v / (1.0 - b2 ** t_adam)
        z = z - lr * m_h / (np.sqrt(v_h) + eps)
        z -= float(np.mean(z))
        z = np.clip(z, -45.0, 45.0)
        p = softmax_stable(z)

        it += 1
        if it % beta_period == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp >= restart_every:
            z_base = np.log(np.maximum(best_p, 1e-15))
            noise = rng.normal(0, noise_scale, n)
            z = z_base + noise
            z -= float(np.mean(z))
            z = np.clip(z, -45.0, 45.0)
            p = softmax_stable(z)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            beta = beta_start
            no_imp = 0
            lr = max(0.03, lr * 0.85)

    return best_p, best_score


def lp_polish(p_init, deadline, rng, max_lp_iters=200, t_start=0.25):
    """LP-based descent with adaptive step size."""
    f = np.maximum(np.asarray(p_init, dtype=np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return p_init.copy()
    f = f / s
    n = len(f)

    c = full_convolve_fft(f)
    best_M = float(c.max())
    best_f = f.copy()
    t = t_start
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        M_curr = float(full_convolve_fft(f).max())
        if stagnant > 0:
            M = best_M
        else:
            M = M_curr

        g = lp_descent_step(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new

        M_new = float(full_convolve_fft(f_new).max())

        if M_new <= best_M + 1e-11:
            f = f_new
            if M_new < best_M - 1e-11:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            it += 1
        else:
            t *= 0.5
            t = max(t, 0.003)
            stagnant += 1

        t = max(0.005, t * 0.998)

        if stagnant > 80:
            noise = 0.08 * rng.normal(0, 1, n)
            f = best_f + noise
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = t_start
            stagnant = 0

    return best_f


def make_diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = np.maximum(1.0 + eta * (t - 0.5), 1e-10)
        profiles.append(a / a.sum())

    for k in [1, 2, 3, 4]:
        a = np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * k * t), 1e-10)
        profiles.append(a / a.sum())

    for alpha in [0.1, 0.15, 0.2, 0.3]:
        a = np.maximum(np.exp(-((t - 0.5) / alpha) ** 2), 1e-10)
        profiles.append(a / a.sum())

    for c1 in [0.2, 0.3, 0.35]:
        for sig in [0.05, 0.1, 0.15]:
            a = np.maximum(
                np.exp(-((t - c1) / sig) ** 2) + np.exp(-((t - (1 - c1)) / sig) ** 2),
                1e-10
            )
            profiles.append(a / a.sum())

    for _ in range(6):
        a = np.ones(n) * 1.0
        for i in range(1, 8):
            amp = rng.uniform(-0.4, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            a = a + amp * np.cos(2 * np.pi * i * t + phase)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP warm-start + logit Adam polish at multiple scales.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start_global = time.monotonic()
    deadline = t_start_global + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1) + 1)

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc = compute_score(init_seq)
                update(init_seq, sc)
                print(f"Initial: n={init_seq.size}, score={sc:.10f}")
    except Exception:
        pass

    # ========== Phase 1: LP warm-start at n=512 ==========
    n512 = 512
    phase1_end = min(deadline, t_start_global + 300.0)

    print(f"\n=== Phase 1: LP at n={n512} ===")
    if init_seq is not None:
        p512 = interp_profile(init_seq, n512)
    else:
        p512 = make_diverse_profiles(n512, rng)[2]  # linear eta=1.5

    sc = compute_score(p512)
    update(p512, sc)

    # LP optimization
    lp_dl1 = min(phase1_end, time.monotonic() + 280.0)
    p512_lp = lp_polish(p512, lp_dl1, rng, max_lp_iters=300, t_start=0.25)
    sc = compute_score(p512_lp)
    update(p512_lp, sc)
    print(f"After LP n={n512}: score={sc:.10f}")

    # ========== Phase 2: Logit Adam polish at n=512 ==========
    print(f"\n=== Phase 2: Adam polish at n={n512} ===")
    phase2_end = min(deadline, t_start_global + 600.0)

    if time.monotonic() < phase2_end - 10:
        p0 = p512_lp.copy() + 0.001 * rng.normal(size=n512)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()

        dl2 = min(phase2_end, time.monotonic() + 250.0)
        p_best, sc = logit_adam(p0, dl2, rng, lr=0.1, beta_start=0.05)
        sc_full = compute_score(p_best)
        update(p_best, sc_full)
        print(f"After Adam n={n512}: score={sc_full:.10f}")

        # Second round with different params
        if time.monotonic() < phase2_end - 20:
            p0b = best_seq.copy() + 0.0005 * rng.normal(size=n512)
            p0b = np.maximum(p0b, 1e-12)
            p0b = p0b / p0b.sum()
            dl2b = min(phase2_end, time.monotonic() + 200.0)
            p_best2, sc2 = logit_adam(p0b, dl2b, rng, lr=0.05, beta_start=0.02,
                                      restart_every=5000, beta_decay=0.95)
            sc_full2 = compute_score(p_best2)
            update(p_best2, sc_full2)
            print(f"After Adam2 n={n512}: score={sc_full2:.10f}")

    # ========== Phase 3: Try n=1024 ==========
    print(f"\n=== Phase 3: Multi-scale n=1024 ===")
    n1024 = 1024
    phase3_end = min(deadline, t_start_global + 1000.0)

    if time.monotonic() < phase3_end - 20 and best_seq is not None:
        p1024 = interp_profile(best_seq, n1024)
        sc = compute_score(p1024)
        update(p1024, sc)
        print(f"Interpolated to n={n1024}: score={sc:.10f}")

        # Quick LP at n=1024
        dl3_lp = min(phase3_end, time.monotonic() + 150.0)
        p1024_lp = lp_polish(p1024, dl3_lp, rng, max_lp_iters=100, t_start=0.2)
        sc = compute_score(p1024_lp)
        update(p1024_lp, sc)
        print(f"After LP n={n1024}: score={sc:.10f}")

        # Adam polish at n=1024
        if time.monotonic() < phase3_end - 30:
            p0_1024 = p1024_lp.copy() + 0.001 * rng.normal(size=n1024)
            p0_1024 = np.maximum(p0_1024, 1e-12)
            p0_1024 = p0_1024 / p0_1024.sum()
            dl3_adam = min(phase3_end, time.monotonic() + 150.0)
            p1024_best, sc_adam = logit_adam(p0_1024, dl3_adam, rng, lr=0.08,
                                             beta_start=0.04, restart_every=4000)
            sc_full = compute_score(p1024_best)
            update(p1024_best, sc_full)
            print(f"After Adam n={n1024}: score={sc_full:.10f}")

    # ========== Phase 4: Try different starting points at n=512 ==========
    print(f"\n=== Phase 4: Diverse starts at n={n512} ===")
    phase4_end = min(deadline, t_start_global + 1350.0)

    if time.monotonic() < phase4_end - 20:
        profiles = make_diverse_profiles(n512, rng)
        for i, prof in enumerate(profiles):
            if time.monotonic() > phase4_end - 10:
                break
            sc_p = compute_score(prof)
            if sc_p < best_score + 0.02:
                # Quick Adam from this profile
                dl = min(phase4_end, time.monotonic() + 60.0)
                p_res, sc_res = logit_adam(prof, dl, rng, lr=0.08, beta_start=0.05)
                sc_full = compute_score(p_res)
                update(p_res, sc_full)
                if sc_full < best_score + 0.01:
                    print(f"  Profile {i}: {sc_full:.10f}")
            else:
                continue

    # ========== Phase 5: Try n=2048 briefly ==========
    print(f"\n=== Phase 5: Try n=2048 ===")
    n2048 = 2048
    phase5_end = min(deadline, t_start_global + 1550.0)

    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        p2048 = interp_profile(best_seq, n2048)
        sc = compute_score(p2048)
        update(p2048, sc)
        print(f"Interpolated to n={n2048}: score={sc:.10f}")

        if sc < best_score + 0.01:
            dl5 = min(phase5_end, time.monotonic() + 150.0)
            p0_2048 = p2048.copy() + 0.001 * rng.normal(size=n2048)
            p0_2048 = np.maximum(p0_2048, 1e-12)
            p0_2048 = p0_2048 / p0_2048.sum()
            p2048_best, sc_adam = logit_adam(p0_2048, dl5, rng, lr=0.06,
                                             beta_start=0.03, restart_every=3000)
            sc_full = compute_score(p2048_best)
            update(p2048_best, sc_full)
            print(f"After Adam n={n2048}: score={sc_full:.10f}")

    # ========== Phase 6: Final intensive polish ==========
    print(f"\n=== Phase 6: Final polish ===")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        print(f"Polishing at n={n_final}, current best={best_score:.10f}")

        # Multiple rounds with decreasing lr
        rounds = [
            (0.04, 0.02, 4000),
            (0.02, 0.01, 6000),
            (0.01, 0.005, 8000),
        ]
        for lr_r, beta_r, restart_r in rounds:
            if time.monotonic() > deadline - 5:
                break
            p0 = best_seq.copy() + 0.0003 * rng.normal(size=n_final)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            dl = min(deadline - 5, time.monotonic() + 120.0)
            p_res, sc_res = logit_adam(p0, dl, rng, lr=lr_r, beta_start=beta_r,
                                       restart_every=restart_r, beta_decay=0.96)
            sc_full = compute_score(p_res)
            update(p_res, sc_full)
            print(f"  Polish lr={lr_r}: {sc_full:.10f}")

    # ========== Final validation ==========
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_diverse_profiles(512, rng)[2]

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence: {eval_score:.10f}")
    except Exception:
        pass

    return [float(x) for x in final_seq]
