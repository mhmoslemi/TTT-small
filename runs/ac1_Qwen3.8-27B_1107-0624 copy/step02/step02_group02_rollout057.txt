    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cosine profiles
    for k in [1, 2, 3, 4, 5]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    # Exponential decay
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.exp(-rate * t)
        profiles.append(a / a.sum())
        a = np.exp(-rate * (1.0 - t))
        profiles.append(a / a.sum())
    # Two-sided exponential
    for rate in [1.0, 2.0, 3.0]:
        a = np.exp(-rate * np.abs(t - 0.5))
        profiles.append(a / a.sum())
    # Quadratic
    for a_c, b_c in [(0, 1), (1, -0.5), (-0.5, 1), (1, 0), (0, -0.3), (2, -1), (-1, 2)]:
        arr = 1.0 + a_c * t + b_c * t ** 2
        arr = np.maximum(arr, 1e-12)
        profiles.append(arr / arr.sum())
    # V-shape
    for width in [0.2, 0.3, 0.5, 0.7]:
        a = 1.0 + 3.0 * np.abs(t - 0.5) / width
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # U-shape
    for width in [0.2, 0.3, 0.5]:
        a = 1.0 + 4.0 * ((t - 0.5) / width) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Random smooth
    for _ in range(8):
        coeffs = rng.uniform(-0.6, 0.6, 7)
        a = 1.0
        for i, c in enumerate(coeffs[1:], 1):
            a = a + c * np.cos(2.0 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def lp_optimize(f_init, deadline, rng, verbose=True):
    """LP-based optimization with iterative M tightening, line search, and smart restarts."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline - 1.0:
        current_conv = np.convolve(f, f)
        current_M = float(current_conv.max())

        # Try progressively tighter M values
        g = None
        for factor in [1.0, 0.998, 0.995, 0.99, 0.985]:
            M_try = current_M * factor
            g = lp_step(f, M_try)
            if g is not None and g.sum() > 1e-10:
                break

        if g is None or g.sum() < 1e-10:
            stagnant += 5
            if stagnant > 50:
                f = best_f + rng.normal(0, 0.03 * max(np.std(best_f), 1e-6), n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                stagnant = 0
            continue

        g = g / g.sum()

        # Line search: try several step sizes
        best_t = t
        best_M_new = current_M
        t_candidates = [t, t * 0.5, t * 1.5, 0.1, 0.2, 0.4, 0.5, 0.7, 0.9]
        for t_try in t_candidates:
            t_try = min(1.0, max(0.01, t_try))
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                continue
            f_try = f_try / s_try
            M_try_new = float(np.convolve(f_try, f_try).max())
            if M_try_new < best_M_new - 1e-15:
                best_M_new = M_try_new
                best_t = t_try

        if best_M_new >= current_M - 1e-15:
            # No improvement found, shrink step
            t = max(0.01, t * 0.6)
            stagnant += 1
            continue

        # Accept the step
        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        M_new = float(np.convolve(f, f).max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            stagnant = 0
            t = min(0.5, t * 1.03)
        else:
            stagnant += 1
            t = max(0.01, t * 0.8)

        iteration += 1
        now = time.monotonic()
        if verbose and now - last_print > 5.0:
            print(f"  LP iter {iteration}: M={best_M:.10f}, score={2*n*best_M:.8f}, t={t:.4f}, stag={stagnant}")
            last_print = now

        # Smart restart when stuck
        if stagnant > 60:
            r = rng.random()
            if r < 0.3:
                noise_scale = 0.02 + 0.03 * (stagnant / 60.0)
                f = best_f + rng.normal(0, noise_scale * np.std(best_f), n)
            elif r < 0.5:
                f = best_f.copy()
                n_perturb = max(10, n // 10)
                idx = rng.choice(n, size=n_perturb, replace=False)
                f[idx] *= (1.0 + rng.normal(0, 0.15, n_perturb))
            elif r < 0.7:
                # Targeted: increase elements that contribute to peak
                peak_lag = int(np.argmax(np.convolve(best_f, best_f)))
                f = best_f.copy()
                for j in range(n):
                    other = peak_lag - j
                    if 0 <= other < n:
                        f[j] *= (1.0 - 0.05 * rng.random())
                        f[other] *= (1.0 - 0.05 * rng.random())
            else:
                # Blend with a random profile
                profs = make_profiles(n, rng)
                p_rand = profs[int(rng.integers(0, len(profs)))]
                f = 0.8 * best_f + 0.2 * p_rand
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            stagnant = 0
            if verbose:
                print(f"  LP restart at iter {iteration}, M={best_M:.10f}")

    return best_f


def logit_polish(p0, deadline, rng, lr=0.06, reg=1e-8):
    """Logit-space Adam optimization with annealing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-14)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-14, 0.3 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    last_print = time.monotonic()

    while time.monotonic() < deadline - 1.0:
        L, w = soft_max(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.005, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Anneal beta periodically
        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.85)

        if time.monotonic() - last_print > 10.0:
            print(f"    Polish iter {adam_t}: score={best_score:.10f}, beta={beta:.2e}, lr={lr:.4f}")
            last_print = time.monotonic()

        # Restart when stuck
        if no_imp > max(1000, 5 * n):
            r = rng.random()
            if r < 0.5:
                z = 0.85 * best_z + 0.3 * rng.normal(size=n)
            else:
                z = best_z + rng.normal(0, 0.5, n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-14, 0.3 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_polish(f, deadline, rng, max_passes=50):
    """Coordinate descent: adjust one element at a time to reduce peak."""
    f = np.maximum(np.asarray(f, dtype=np.float64).ravel(), 0.0).copy()
    f = f / f.sum()
    n = f.size

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    passes = 0

    while time.monotonic() < deadline - 1.0 and passes < max_passes:
        improved = False
        conv_f = np.convolve(f, f)
        current_M = float(conv_f.max())

        for j in range(n):
            if time.monotonic() > deadline - 2.0:
                break
            # Try modifying f[j]
            original_val = f[j]
            # Binary search for optimal f[j]
            lo = 0.0
            hi = original_val * 3.0 + 1.0 / n
            best_val = original_val
            best_m = current_M

            for _ in range(8):
                mid = (lo + hi) / 2.0
                f[j] = mid
                s = f.sum()
                if s < 1e-10:
                    f[j] = original_val
                    break
                f_norm = f / s
                m_new = float(np.convolve(f_norm, f_norm).max())
                if m_new < best_m - 1e-15:
                    best_m = m_new
                    best_val = mid
                    lo = mid
                else:
                    hi = mid

            f[j] = best_val
            if best_val != original_val:
                f = f / f.sum()
                improved = True

        passes += 1
        conv_f = np.convolve(f, f)
        new_M = float(conv_f.max())
        if new_M < best_M - 1e-15:
            best_M = new_M
        elif not improved:
            break
        if passes % 10 == 0:
            print(f"    Coord pass {passes}: M={best_M:.10f}")

    return f, 2.0 * n * best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP descent at n=512, LP at n=1024, gradient polish,
    coordinate descent, and final refinement.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ============ Phase 1: LP optimization at n=512 ============
    n_main = 512
    phase1_end = min(deadline, t_start + 380.0)

    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_profiles(n_main, rng)[2]  # a good linear profile

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1 start: n={n_main}, score={sc:.10f}")

    # Main LP optimization
    lp_deadline = min(phase1_end, time.monotonic() + 280.0)
    result = lp_optimize(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP (n=512): score={sc:.10f}")

    # Try a few other starting profiles if time allows
    if time.monotonic() < phase1_end - 20:
        profiles = make_profiles(n_main, rng)
        # Pick the top 5 by initial score
        scored = [(compute_score(p), i) for i, p in enumerate(profiles)]
        scored.sort()
        for sc_p, idx in scored[:5]:
            if time.monotonic() > phase1_end - 5:
                break
            if sc_p >= best_score + 0.1:
                continue
            per_deadline = min(phase1_end, time.monotonic() + 25.0)
            result = lp_optimize(profiles[idx], per_deadline, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            if sc < best_score + 0.05:
                print(f"  Alt profile: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: LP at n=1024 ============
    n_large = 1024
    phase2_end = min(deadline, t_start + 900.0)

    if time.monotonic() < phase2_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 2 start: n={n_large}, score={sc:.10f}")

        lp_deadline2 = min(phase2_end, time.monotonic() + 450.0)
        result = lp_optimize(p_large, lp_deadline2, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After LP (n=1024): score={sc:.10f}")

        # Try a couple more profiles at n=1024
        if time.monotonic() < phase2_end - 30:
            profiles_1024 = make_profiles(n_large, rng)
            scored = [(compute_score(p), i) for i, p in enumerate(profiles_1024[:20])]
            scored.sort()
            for sc_p, idx in scored[:3]:
                if time.monotonic() > phase2_end - 10:
                    break
                if sc_p >= best_score + 0.08:
                    continue
                per_deadline = min(phase2_end, time.monotonic() + 80.0)
                result = lp_optimize(profiles_1024[idx], per_deadline, rng, verbose=False)
                sc = compute_score(result)
                update(result, sc)

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Gradient polishing ============
    phase3_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase3_end - 20 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 3: polishing at n={n_polish}, current best={best_score:.10f}")

        # Multiple polish rounds with different hyperparameters
        lr_list = [0.06, 0.04, 0.08, 0.03]
        for lr in lr_list:
            if time.monotonic() > phase3_end - 10:
                break
            per_deadline = min(phase3_end, time.monotonic() + 100.0)
            p0 = best_seq.copy()
            # Add tiny noise for diversity
            p0 = p0 + 0.002 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-14)
            p0 = p0 / p0.sum()
            result, sc = logit_polish(p0, per_deadline, rng, lr=lr)
            update(result, sc)
            print(f"  Polish lr={lr}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Coordinate descent polish ============
    phase4_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        n_cd = best_seq.size
        print(f"Phase 4: coordinate descent at n={n_cd}, current best={best_score:.10f}")
        per_deadline = min(phase4_end - 5, time.monotonic() + 150.0)
        result, sc = coordinate_polish(best_seq, per_deadline, rng, max_passes=80)
        update(result, sc)
        print(f"  After coord descent: score={sc:.10f}")

        # One more short LP round after coordinate descent
        if time.monotonic() < phase4_end - 30 and best_seq is not None:
            per_deadline2 = min(phase4_end - 5, time.monotonic() + 100.0)
            result = lp_optimize(best_seq, per_deadline2, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            print(f"  After post-coord LP: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Final polish ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        p0 = best_seq + 0.001 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-14)
        p0 = p0 / p0.sum()
        result, sc = logit_polish(p0, per, rng, lr=0.02)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profiles(512, rng)[2]

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profiles(256, rng)[2]

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"WARNING: evaluate_sequence returned {eval_score}, using fallback")
            final_seq = make_profiles(512, rng)[2]
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
