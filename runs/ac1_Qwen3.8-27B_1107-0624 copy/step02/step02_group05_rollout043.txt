        hi = min(n - 1, k)

    return A

    if best_g is None:
        # Last resort: try full LP with shorter time limit
        best_g = solve_lp_full(f, M, time_limit=min(2.0, time_limit))

    return best_g


def lp_optimize(f_init, deadline, rng, max_iters=10000, use_cutting=True):
    """
    LP-based iterative optimization.
    Returns (best_f, best_M).
    """
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)

    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP for direction g
        if use_cutting:
            g = solve_lp_cutting(f, M, time_limit=2.0)
        else:
            g = solve_lp_full(f, M, time_limit=2.0)

        if g is None:
            # Fallback
            f = f + 0.1 * rng.normal(size=n)
            f = normalize_nonneg(f)
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            no_imp += 1
            if no_imp > 50:
                f = best_f.copy()
                M = best_M
                f = f + 0.2 * rng.normal(size=n)
                f = normalize_nonneg(f)
                if f is None:
                    f = best_f.copy()
                M = compute_M(f)
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = compute_M(f_try)
            if M_try < best_M_try - 1e-14:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # Stall detection
        if no_imp > 40:
            f = 0.6 * best_f + 0.4 * normalize_nonneg(best_f + 0.3 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            no_imp = 0

        if it % 500 == 0 and it > 0:
            print(f"    LP iter {it}: M={M:.12f}, best_M={best_M:.12f}")

    return best_f, best_M


def soft_max_weights(c, beta):
    """Compute soft-max weights for the convolution peaks."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 (using FFT)."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def adam_polish(f_init, deadline, rng, lr=0.04, max_iters=20000, beta_init=None):
    """Adam polish in logit space using soft-max gradient."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    if beta_init is None:
        beta = max(1e-6, 0.15 * M)
    else:
        beta = beta_init

    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        w = soft_max_weights(c, beta)
        gp = 2.0 * corr_weights(w, f)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_f.copy()
            z = np.log(np.maximum(z, 1e-12))
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            f2 = best_f.copy()
            z = np.log(np.maximum(f2, 1e-12))
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adapt beta
        if adam_t % 200 == 0:
            beta = max(1e-8, beta * 0.85)

        # Restart on stall
        if no_imp > max(500, 5 * n):
            z_pert = 0.7 * np.log(np.maximum(best_f, 1e-12)) + 0.3 * rng.normal(size=n)
            z = np.clip(z_pert, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-8, 0.15 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            no_imp = 0

    return best_f, best_M


def make_antisymmetric_profile(n, kind='linear', eps=0.8):
    """Create a profile with antisymmetric structure: a[i]+a[n-1-i] = const."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if kind == 'linear':
        s = (t - center) / center if center > 0 else np.zeros(n)
    elif kind == 'tanh':
        s = np.tanh(2.0 * (t - center) / center) if center > 0 else np.zeros(n)
    elif kind == 'steep_tanh':
        s = np.tanh(4.0 * (t - center) / center) if center > 0 else np.zeros(n)
    elif kind == 'cubic':
        u = (t - center) / center if center > 0 else np.zeros(n)
        s = u - 0.3 * u ** 3
    elif kind == 'sinusoidal':
        s = np.sin(np.pi * (t - center) / center) if center > 0 else np.zeros(n)
    elif kind == 'stepped':
        s = np.where(t < center, 1.0, -1.0)
    else:
        s = (t - center) / center if center > 0 else np.zeros(n)

    # Scale s to have max abs = 1
    s_max = np.max(np.abs(s))
    if s_max > 0:
        s = s / s_max

    f = 1.0 + eps * s
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_random_antisymmetric(n, rng, smoothness=3):
    """Random antisymmetric profile with smoothness control."""
    # Generate random antisymmetric: s[i] = -s[n-1-i]
    half = n // 2
    s_half = rng.normal(size=half)
    # Smooth
    for _ in range(smoothness):
        s_half = (s_half + np.roll(s_half, 1)) / 2.0
        s_half = (s_half + np.roll(s_half, -1)) / 2.0
    s = np.zeros(n)
    s[:half] = s_half
    s[half:] = -s_half[::-1]
    if n % 2 == 1:
        s[half] = 0.0

    s_max = np.max(np.abs(s))
    if s_max > 0:
        s = s / s_max

    eps = rng.uniform(0.5, 1.0)
    f = 1.0 + eps * s
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def get_diverse_starts(n, rng, count=20):
    """Generate diverse starting points exploiting antisymmetric structure."""
    starts = []

    # Antisymmetric profiles with different eps values
    for eps in [0.3, 0.5, 0.7, 0.9, 0.95]:
        for kind in ['linear', 'tanh', 'steep_tanh', 'cubic']:
            starts.append(make_antisymmetric_profile(n, kind, eps))

    # Random antisymmetric
    for i in range(min(count, 15)):
        starts.append(make_random_antisymmetric(n, rng, smoothness=rng.integers(1, 6)))

    # Fourier-based smooth profiles
    t = np.linspace(0, 1, n)
    for i in range(min(count, 10)):
        coeffs = rng.uniform(-1, 1, 6)
        f = 1.0
        for k in range(3):
            f = f + 0.4 * coeffs[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
            f = f + 0.4 * coeffs[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
        # Add antisymmetric component
        f = f + 0.5 * rng.uniform(0.5, 1.0) * (t - 0.5) * 2.0
        f = np.maximum(f, 1e-6)
        f = f / f.sum()
        starts.append(f)

    return starts


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP cutting plane, antisymmetric profiles,
    and Adam polish to minimize 2*n*max(convolve(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # Try to use parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                print(f"Parent sequence loaded: n={parent.size}, score={compute_score(parent):.10f}")
                try_update(parent, compute_score(parent))
    except Exception:
        pass

    # Phase 1: Diverse exploration with quick LP (first 400s)
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: Diverse exploration, {phase1_end - t_start:.0f}s budget")

    lengths = [128, 256, 512]
    all_results = []

    for n in lengths:
        remaining = phase1_end - time.monotonic()
        if remaining < 20:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")

        starts = get_diverse_starts(n, rng, count=10)
        if parent is not None:
            starts.append(interpolate_profile(parent, n))

        # Quick LP from each start (limited iterations)
        per_start_time = max(5.0, remaining / (len(starts) + 1))
        for i, s in enumerate(starts):
            remaining = phase1_end - time.monotonic()
            if remaining < 3:
                break
            run_dl = time.monotonic() + min(per_start_time, remaining - 1)
            sc_init = compute_score(s)
            try_update(s, sc_init)

            f_best, M_best = lp_optimize(s, run_dl, rng, max_iters=2000)
            sc = 2.0 * n * M_best
            try_update(f_best, sc)
            all_results.append((sc, f_best, n))
            if (i + 1) % 5 == 0:
                print(f"    start {i+1}/{len(starts)}: best={best_score:.10f}")

    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Top score: {best_score:.10f}")

    # Phase 2: Deep LP optimization on top candidates (next 700s)
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"Phase 2: Deep LP optimization, {phase2_end - t_start:.0f}s budget")

    top_k = min(8, len(all_results))
    phase2_results = []
    for i in range(top_k):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, top_k - i), 120.0)
        if alloc < 5:
            break
        run_dl = time.monotonic() + alloc
        sc, f_init, n = all_results[i]
        print(f"  Deep LP {i+1}/{top_k}: n={n}, init_score={sc:.10f}, alloc={alloc:.0f}s")
        f_best, M_best = lp_optimize(f_init, run_dl, rng, max_iters=20000)
        sc_new = 2.0 * n * M_best
        try_update(f_best, sc_new)
        phase2_results.append((sc_new, f_best, n))
        print(f"    result: {sc_new:.10f}, best={best_score:.10f}")

    phase2_results.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Adam polish on best solutions (next 300s)
    phase3_end = min(deadline, t_start + 1450.0)
    print(f"Phase 3: Adam polish, {phase3_end - t_start:.0f}s budget")

    combined = phase2_results[:5] + all_results[:5]
    # Deduplicate by keeping best per (n, approximate)
    combined.sort(key=lambda x: x[0])
    top_polish = combined[:6]

    for i in range(len(top_polish)):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(remaining / max(1, len(top_polish) - i), 60.0)
        if alloc < 5:
            break
        run_dl = time.monotonic() + alloc
        sc, f_init, n = top_polish[i]
        print(f"  Adam polish {i+1}: n={n}, init={sc:.10f}, alloc={alloc:.0f}s")
        f_best, M_best = adam_polish(f_init, run_dl, rng, lr=0.03)
        sc_new = 2.0 * n * M_best
        try_update(f_best, sc_new)
        print(f"    result: {sc_new:.10f}, best={best_score:.10f}")

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # Phase 4: Final LP refinement on the single best (remaining time)
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(best_seq)
            print(f"Phase 4: Final refinement, n={n_best}, {remaining:.0f}s")
            f_best, M_best = lp_optimize(best_seq, deadline, rng, max_iters=50000)
            sc = 2.0 * n_best * M_best
            try_update(f_best, sc)
            print(f"  Final LP: {sc:.10f}")

            # Quick Adam polish if time remains
            remaining = deadline - time.monotonic()
            if remaining > 20:
                f_best2, M_best2 = adam_polish(best_seq, deadline, rng, lr=0.02)
                sc2 = 2.0 * n_best * M_best2
                try_update(f_best2, sc2)
                print(f"  Final Adam: {sc2:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = make_antisymmetric_profile(256, 'linear', 0.8)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_antisymmetric_profile(256, 'linear', 0.8)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f} (tracked: {best_score:.10f})")

    # Validate with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
