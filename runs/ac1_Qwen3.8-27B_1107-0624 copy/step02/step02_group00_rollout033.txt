Optimized step function search for minimizing 2n*max(conv(a,a))/sum(a)^2.

Uses a combination of:

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

# Parametric initializations
def _gaussian_profile(n, sigma=0.3):
    t = np.linspace(0, 1, n) - 0.5
    a = np.exp(-t**2 / (2 * sigma**2))
    return a / a.sum()

def _cosine_profile(n, power=1.0):
    t = np.linspace(0, 1, n) - 0.5
    a = np.cos(np.pi * t * 0.999) ** power  # avoid zero at endpoints
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _parabolic_profile(n, curvature=1.0):
    t = np.linspace(0, 1, n) - 0.5
    a = 1.0 - curvature * (2*t)**2
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _linear_profile(n, slope=1.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _bimodal_profile(n, center=0.3):
    t = np.linspace(0, 1, n) - 0.5
    a = np.exp(-(t - center)**2 / 0.05) + np.exp(-(t + center)**2 / 0.05)
    return a / a.sum()

def _semicircle_profile(n):
    t = np.linspace(0, 1, n) - 0.5
    a = np.sqrt(np.maximum(1 - (2*t)**2, 1e-10))
    return a / a.sum()

def _power_profile(n, p=2.0):
    t = np.linspace(0, 1, n) - 0.5
    a = (1 - (2*t)**p) ** (1/p) if p > 0 else np.ones(n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

Let me implement a **mirror descent with adaptive step size**:

The problem: minimize f(a) = max_k (a*a)_k subject to a >= 0, sum(a) = 1.

Mirror descent with entropy mirror:
- a^{t+1}_j = a^t_j * exp(-eta * g_j) / Z
where g is the subgradient at a^t.

The subgradient at a^t for f(a) = max_k (a*a)_k:
If k* is the argmax, then g_j = a[k*-j] (for 0 <= k*-j < n), else 0.

But this subgradient is very sparse (only n non-zero entries out of n total, actually it's a shifted copy of a). And the max might be achieved at multiple points, in which case we average.

A better approach: use the **smooth max** with a small temperature:
f_smooth(a) = (1/β) * log(sum_k exp(β * (a*a)_k))

The gradient of this is:
∇_j f_smooth = (1/sum_k exp(β*c_k)) * sum_k exp(β*c_k) * ∂c_k/∂a_j
= sum_k w_k * a[k-j] (for valid k-j)

where w_k = exp(β*c_k) / sum_l exp(β*c_l)

This is a weighted average of shifts of a.

For mirror descent:
a^{t+1} = a^t * exp(-eta * ∇f_smooth(a^t)) / Z

This is equivalent to:
log(a^{t+1}_j) = log(a^t_j) - eta * ∇_j f_smooth - log(Z)

And log(Z) = log(sum_j a^t_j * exp(-eta * ∇_j f_smooth))

This maintains the simplex constraint automatically (after normalization).

The step size eta should be small enough for convergence but large enough for progress. A common choice is eta = O(1/sqrt(T)) for T total iterations, but for practical purposes we can use a fixed small eta.

        # np.correlate(w, a, 'full')[k] = sum_i w[i+k-(n-1)] * a[i] for valid i
        # Hmm this is getting confusing. Let me just verify numerically for small case.
        
        # Actually, let me use the definition directly with a simple loop-free approach:
        # grad_j = 2 * sum_{u=0}^{n-1} a[u] * w[u+j]
        # Build a (n x n) matrix where M[j, u] = w[u+j], then grad = 2 * M @ a
        # But that's O(n^2) memory. For n=512, that's 256K entries, fine.
        # For n=2048, that's 4M entries = 32MB, still OK.
        
        # Better: use the FFT approach correctly.
        # We want h[j] = sum_{u=0}^{n-1} a[u] * w[u+j] for j = 0, ..., n-1
        # This is the cross-correlation of a and w at "negative lag" j
        # = convolution of a_rev and w at specific positions
        
        # a_rev = a[::-1], length n
        # conv(a_rev, w)[k] = sum_i a_rev[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]
        # Let u = n-1-i: = sum_u a[u] * w[k-n+1+u]
        # We want this to equal sum_u a[u] * w[u+j], so k-n+1 = j, k = j+n-1
        # So h[j] = conv(a_rev, w)[j+n-1]
        
        a_rev = a[::-1]
        N2 = _next_pow2(n + 2 * n - 1 - 1)  # = next_pow2(3n-2)
        # Actually we need conv of a_rev (len n) and w (len 2n-1), result has len n+2n-1-1 = 3n-2
        N2 = _next_pow2(3 * n - 2)
        fa_rev = np.fft.rfft(a_rev, N2)
        fw = np.fft.rfft(w, N2)
        conv = np.fft.irfft(fa_rev * fw, N2)[:3 * n - 2]
        h = conv[n - 1:2 * n - 1]  # h[j] = conv[j+n-1] for j = 0, ..., n-1
        return 2.0 * h
    
    # Fallback for very large n (shouldn't happen)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, w[j:j+n])
    return grad


def _mirror_descent(a, deadline, max_iter=50000, eta=1.0, beta_start=None, beta_end=None, sym=False):
    """Mirror descent with entropy regularization on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    if sym:
        # Enforce symmetry
        if n % 2 == 0:
            half = n // 2
            a = np.concatenate([a[:half], a[:half][::-1]])
        else:
            half = n // 2
            a = np.concatenate([a[:half], [a[half]], a[:half][::-1]])
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    
    if beta_start is None:
        beta_start = 0.5 * c_max
    if beta_end is None:
        beta_end = 10.0 * c_max
    
    beta = beta_start
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    
    no_imp = 0
    eta_cur = eta
    best_no_imp_action = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        score = float(2.0 * n * c_max)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                eta_cur = max(0.1, eta_cur * 0.95)
                if no_imp > 5000:
                    a = best_a.copy()
                    best_no_imp_action += 1
                    if best_no_imp_action > 5:
                        break
                    no_imp = 0
                    eta_cur = eta
        
        # Compute smooth max weights
        _, w = _smooth_max_and_weights(c, beta)
        
        # Compute gradient
        grad = _gradient_smooth_max(a, w)
        
        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            continue
        
        # Mirror descent update: a_new_j ∝ a_j * exp(-eta * grad_j)
        log_update = -eta_cur * grad
        log_update -= float(np.max(log_update))  # numerical stability
        exp_update = np.exp(log_update)
        a_new = a * exp_update
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a = best_a.copy()
            continue
        a = a_new / s_new
        
        # Enforce symmetry if needed
        if sym:
            if n % 2 == 0:
                half = n // 2
                a_new_sym = np.concatenate([(a[:half] + a[half:][::-1]) / 2, (a[:half] + a[half:][::-1]) / 2][::-1])
            else:
                half = n // 2
                mid = a[half]
                a_new_sym = np.concatenate([(a[:half] + a[half+1:][::-1]) / 2, [mid], (a[:half] + a[half+1:][::-1]) / 2][::-1])
            a_new_sym = np.maximum(a_new_sym, 1e-30)
            a = a_new_sym / a_new_sym.sum()
        
        # Adapt beta: increase over time to sharpen
        if it % 500 == 0:
            progress = min(1.0, it / max(1, max_iter))
            beta = beta_start * (beta_end / beta_start) ** progress
    
    return best_a


def _lp_sprefine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    """LP-based refinement: find direction to spread mass while respecting conv bound."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    best_a = a.copy()
    
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend
    
    # Precompute constraint matrix structure info
    # A[k, j] = a[j] if j <= k <= j+n-1
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        
        # Build LP: max sum(g) s.t. conv(a, g) <= M_lp, g >= 0
        # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid indices
        # Constraint k: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * g[k-j] <= M_lp
        # Let i = k-j: constraint k is sum_{i=k-(n-1)}^{k} a[i] * g[k-i] <= M_lp
        # Hmm, let me just build the matrix directly.
        
        # A_ub shape: (2n-1, n)
        # A_ub[k, j] = a[j] if 0 <= k-j < n, else 0
        # i.e., A_ub[k, j] = a[j] if j <= k <= j+n-1
        
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a[j]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.1, blend_cur * 0.9)
                inflate_cur = min(1.005, max(1.001, inflate_cur * 0.95))
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
                if no_imp > 500:
                    break
        
        a = a_new
    
    return best_a


def _parametric_search(n, deadline):
    """Search over parametric families to find good initializations."""
    candidates = []
    
    # Gaussian profiles
    for sigma in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        t = np.linspace(0, 1, n) - 0.5
        a = np.exp(-t**2 / (2 * sigma**2))
        a = a / a.sum()
        candidates.append(a)
    
    # Cosine profiles
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        t = np.linspace(0, 1, n) - 0.5
        a = np.cos(np.pi * t * 0.999) ** p
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        candidates.append(a)
    
    # Parabolic profiles
    for curv in [0.3, 0.5, 0.7, 0.9, 1.0, 1.2]:
        t = np.linspace(0, 1, n) - 0.5
        a = 1.0 - curv * (2 * t)**2
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        candidates.append(a)
    
    # Semicircle
    t = np.linspace(0, 1, n) - 0.5
    a = np.sqrt(np.maximum(1 - (2 * t)**2, 1e-10))
    a = a / a.sum()
    candidates.append(a)
    
    # Power profiles
    for p in [1.5, 2, 3, 4]:
        t = np.linspace(0, 1, n) - 0.5
        r = 2 * t
        a = np.power(np.maximum(1 - r**p, 1e-15), 1.0/p)
        a = a / a.sum()
        candidates.append(a)
    
    # Linear profiles
    for slope in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        t = np.linspace(0, 1, n)
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        candidates.append(a)
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Bimodal
    for center in [0.2, 0.25, 0.3]:
        t = np.linspace(0, 1, n) - 0.5
        a = np.exp(-(t - center)**2 / 0.03) + np.exp(-(t + center)**2 / 0.03)
        a = a / a.sum()
        candidates.append(a)
    
    # Score all
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def _local_search(a, deadline, n_trials=2000):
    """Local search: random perturbations and accept if better."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    
    rng = np.random.default_rng(12345)
    temp = 0.01
    no_imp = 0
    
    for it in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        # Random perturbation
        a_trial = a.copy()
        # Add noise to a random subset
        n_pert = max(1, n // 10)
        idx = rng.choice(n, size=n_pert, replace=False)
        noise = rng.normal(0, temp * np.max(a), size=n_pert)
        a_trial[idx] += noise
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
            no_imp = 0
        else:
            no_imp += 1
            # Annealing: reduce temp
            if no_imp % 100 == 0:
                temp *= 0.7
                a = best_a.copy()
            if no_imp > 1000:
                break
    
    return best_a


def _adam_polish(a, deadline, max_iter=30000, lr=0.03):
    """Adam optimization in logit space for final polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    best_score = float(2.0 * n * c_max)
    best_a = a.copy()
    best_z = z.copy()
    
    beta = 0.3 * c_max
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        score = float(2.0 * n * c_max)
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5000:
                a = best_a.copy()
                z = best_z.copy()
                m = np.zeros(n)
                v_arr = np.zeros(n)
                t_adam = 0
                beta = 0.3 * float(_fft_convolve(a).max())
                no_imp = 0
                if no_imp > 10000:
                    break
        
        # Compute gradient via smooth max
        _, w = _smooth_max_and_weights(c, beta)
        grad = _gradient_smooth_max(a, w)
        grad += 1e-6 * a  # regularization to avoid zeroing out
        
        # Project gradient to tangent space of simplex
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * gz**2
        mh = m / (1 - b1**t_adam)
        vh = v_arr / (1 - b2**t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue
        
        z = np.clip(z, -40, 40)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        if it % 1000 == 0:
            beta = max(1e-6, beta * 0.8)
    
    return best_a


def _interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    if n_old == 1:
        return np.ones(n_new) / n_new
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search for optimal step function heights.
    Combines parametric initialization, mirror descent, LP refinement,
    Adam polish, and local search across multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s_p = p.sum()
            if s_p > 0.01:
                parent = p / s_p
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Sequence lengths to try
    n_values = [512, 1024, 2048]
    
    # Phase 1: Parametric initialization search (~150s budget)
    print(f"Phase 1: Parametric search, start={time.monotonic()-t_start:.1f}s")
    phase1_deadline = t_start + 150.0
    
    for n in n_values:
        if time.monotonic() > phase1_deadline:
            break
        
        # Parametric search
        a_param, sc_param = _parametric_search(n, phase1_deadline)
        if np.isfinite(sc_param) and sc_param < best_score:
            best_score = sc_param
            best_seq = a_param.copy()
        print(f"  n={n}: parametric best={sc_param:.10f}")
        
        # Add parent if available
        if parent is not None:
            a_parent = _interpolate_to_length(parent, n)
            sc_parent = _score_norm(a_parent)
            if sc_parent < best_score:
                best_score = sc_parent
                best_seq = a_parent.copy()
                print(f"  n={n}: parent interpolated={sc_parent:.10f}")
    
    # Phase 2: Mirror descent optimization (~600s budget)
    print(f"\nPhase 2: Mirror descent, start={time.monotonic()-t_start:.1f}s")
    phase2_deadline = t_start + 750.0
    
    # For each n value, run mirror descent on the best initialization
    for n in n_values:
        if time.monotonic() > phase2_deadline - 100:
            break
        
        # Get best initialization for this n
        if parent is not None:
            a_init = _interpolate_to_length(parent, n)
        else:
            a_init, _ = _parametric_search(n, phase2_deadline)
        
        sc_init = _score_norm(a_init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()
        
        # Run mirror descent (asymmetric)
        md_deadline = min(phase2_deadline - 50, time.monotonic() + 200)
        a_md = _mirror_descent(a_init, md_deadline, max_iter=100000, eta=1.5,
                               beta_start=0.3 * float(_fft_convolve(a_init).max()))
        sc_md = _score_norm(a_md)
        if sc_md < best_score:
            best_score = sc_md
            best_seq = a_md.copy()
        print(f"  n={n}: mirror descent={sc_md:.10f} (init={sc_init:.10f})")
        
        # Run mirror descent (symmetric) if time allows
        if time.monotonic() < phase2_deadline - 100:
            md_sym_deadline = min(phase2_deadline - 50, time.monotonic() + 150)
            a_md_sym = _mirror_descent(a_init, md_sym_deadline, max_iter=80000, eta=1.5,
                                       sym=True, beta_start=0.3 * float(_fft_convolve(a_init).max()))
            sc_md_sym = _score_norm(a_md_sym)
            if sc_md_sym < best_score:
                best_score = sc_md_sym
                best_seq = a_md_sym.copy()
            print(f"  n={n}: symmetric MD={sc_md_sym:.10f}")
    
    # Phase 3: LP refinement (~400s budget)
    print(f"\nPhase 3: LP refinement, start={time.monotonic()-t_start:.1f}s")
    phase3_deadline = t_start + 1150.0
    
    if best_seq is not None and time.monotonic() < phase3_deadline - 50:
        n_best = len(best_seq)
        # Run LP on the best so far
        lp_deadline = min(phase3_deadline - 30, time.monotonic() + 300)
        a_lp = _lp_sprefine(best_seq, lp_deadline, max_iter=3000, inflate=1.001, blend=0.3)
        sc_lp = _score_norm(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  LP result: {sc_lp:.10f} (was {best_score:.10f} before)")
        
        # Also try LP on parent-interpolated if different
        if parent is not None and n_best != len(parent):
            a_p = _interpolate_to_length(parent, n_best)
            if time.monotonic() < phase3_deadline - 30:
                lp_deadline2 = min(phase3_deadline - 30, time.monotonic() + 150)
                a_lp2 = _lp_sprefine(a_p, lp_deadline2, max_iter=2000, inflate=1.001, blend=0.3)
                sc_lp2 = _score_norm(a_lp2)
                if sc_lp2 < best_score:
                    best_score = sc_lp2
                    best_seq = a_lp2.copy()
                print(f"  LP on parent: {sc_lp2:.10f}")
    
    # Phase 4: Local search (~150s budget)
    print(f"\nPhase 4: Local search, start={time.monotonic()-t_start:.1f}s")
    phase4_deadline = t_start + 1300.0
    
    if best_seq is not None and time.monotonic() < phase4_deadline - 20:
        ls_deadline = min(phase4_deadline - 20, time.monotonic() + 130)
        a_ls = _local_search(best_seq, ls_deadline, n_trials=5000)
        sc_ls = _score_norm(a_ls)
        if sc_ls < best_score:
            best_score = sc_ls
            best_seq = a_ls.copy()
        print(f"  Local search: {sc_ls:.10f}")
        
        # Try local search with different perturbation scales
        if time.monotonic() < phase4_deadline - 10:
            a_pert = best_seq * (1 + 0.05 * rng.normal(size=len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            ls_deadline2 = min(phase4_deadline - 10, time.monotonic() + 60)
            a_ls2 = _local_search(a_pert, ls_deadline2, n_trials=2000)
            sc_ls2 = _score_norm(a_ls2)
            if sc_ls2 < best_score:
                best_score = sc_ls2
                best_seq = a_ls2.copy()
            print(f"  Local search (perturbed): {sc_ls2:.10f}")
    
    # Phase 5: Adam polish (~200s budget)
    print(f"\nPhase 5: Adam polish, start={time.monotonic()-t_start:.1f}s")
    phase5_deadline = t_start + 1500.0
    
    if best_seq is not None and time.monotonic() < phase5_deadline - 30:
        ad_deadline = min(phase5_deadline - 30, time.monotonic() + 180)
        a_ad = _adam_polish(best_seq, ad_deadline, max_iter=50000, lr=0.02)
        sc_ad = _score_norm(a_ad)
        if sc_ad < best_score:
            best_score = sc_ad
            best_seq = a_ad.copy()
        print(f"  Adam result: {sc_ad:.10f}")
    
    # Phase 6: Final mirror descent polish (~150s budget)
    print(f"\nPhase 6: Final mirror descent, start={time.monotonic()-t_start:.1f}s")
    phase6_deadline = t_start + 1650.0
    
    if best_seq is not None and time.monotonic() < phase6_deadline - 20:
        md_deadline = min(phase6_deadline - 20, time.monotonic() + 130)
        c_peak = float(_fft_convolve(best_seq).max())
        a_final = _mirror_descent(best_seq, md_deadline, max_iter=100000, eta=0.5,
                                  beta_start=2.0 * c_peak, beta_end=50.0 * c_peak)
        sc_final = _score_norm(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final MD: {sc_final:.10f}")
    
    # Phase 7: One more LP pass
    print(f"\nPhase 7: Final LP, start={time.monotonic()-t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 20:
        lp_deadline = min(deadline - 20, time.monotonic() + 80)
        a_last = _lp_sprefine(best_seq, lp_deadline, max_iter=2000, inflate=1.0005, blend=0.2)
        sc_last = _score_norm(a_last)
        if sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"  Final LP: {sc_last:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = _score(best_seq)
    
    # Ensure non-negative and valid
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(512) / 512
    else:
        # Keep as is (don't renormalize - the score is scale-invariant)
        pass
    
    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
