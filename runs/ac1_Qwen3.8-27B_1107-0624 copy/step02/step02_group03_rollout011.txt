def internal_score(a):

    s = a.sum()
    if s < 1e-30:

    return float(2.0 * n * c.max() / (s * s))

def smooth_max(c, beta):
    """Smooth approximation of max using log-sum-exp."""
    c_shifted = c - c.max()
    return c.max() + np.log(np.exp(beta * c_shifted).sum()) / beta

def smooth_max_grad(c, beta):
    """Gradient of smooth max w.r.t. c."""
    c_shifted = c - c.max()
    ex = np.exp(beta * c_shifted)
    w = ex / ex.sum()
    return w  # d/dc of smooth_max = w (up to the factor that makes it a probability)

def log_sigmoid(z):
    """Compute log-softmax(z) = z - log(sum(exp(z))) (centered)."""
    z_max = z.max()
    z_centered = z - z_max
    return z_centered - np.log(np.exp(z_centered).sum())

def neg_log_softmax_grad(z):
    """Gradient of -log(sum(exp(z))) w.r.t. z = 1 - softmax(z)."""
    z_max = z.max()
    zc = z - z_max
    p = np.exp(zc)
    p = p / p.sum()
    return 1.0 - p

    # Since x <= 0, exp(x) <= 1, sum <= len(c), so log(sum) <= log(len(c))
    # This is fine numerically.


def smooth_max_and_weights(c, beta):
    """Return (smooth_max_value, softmax_weights)."""
    c_max = c.max()
    x = beta * (c - c_max)
    # x <= 0 always
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return c_max, w
    w = ex / s
    val = c_max + np.log(s) / beta
    return val, w


def obj_grad_logspace(z, beta):
    """
    Compute (f, grad_z) where f = soft_max(conv(softmax(z), softmax(z))) * 2*n
    and grad_z is the gradient w.r.t. z.
    """
    n = len(z)
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()

    c = np.convolve(a, a)
    val, w = smooth_max_and_weights(c, beta)

    # grad_a[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * correlate(w, a, 'full')[:n]
    grad_a_full = np.correlate(w, a, mode='full')
    grad_a = 2.0 * grad_a_full[:n]

    # grad_z = a * (grad_a - dot(a, grad_a))
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)

    # Scale: the actual objective is 2*n*max_b/sum^2, but with sum(a)=1 it's 2*n*max_b
    # So f_actual = 2*n*val
    f_actual = 2.0 * n * val
    grad_z_actual = 2.0 * n * grad_z

    return f_actual, grad_z_actual


def lbfgs_smooth_max(a_init, deadline, beta_schedule=None, max_iter=5000):
    """
    Optimize using L-BFGS on the smooth max in log-space.
    Returns (best_a, best_score).
    """
    from scipy.optimize import minimize

    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    z0 = np.log(a_init)
    z0 -= z0.mean()

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 100, 500, 2000, 10000]

    best_a = a_init.copy()
    best_score = internal_score(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def fun_grad(z):
            f, g = obj_grad_logspace(z, beta)
            return f, g

        res = minimize(fun_grad, z0, jac=True, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': max_iter,
                                'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 50})

        if res.x is not None:
            z_new = res.x
            z_new = np.clip(z_new, -50, 50)
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            sc = internal_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z0 = z_new.copy()
            else:
                # Check with the rescaled version
                pass
        else:
            break

    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    # convolve(f, g)[k] = sum_j f[j] * g[k-j] (valid indices)
    # For fixed f, this is linear in g.
    # (A @ g)[k] = sum_j g[j] * f[k-j] where 0 <= k-j < n
    # = sum_j g[j] * f[k-j] (when 0 <= k-j < n, i.e., j <= k < j+n)
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=30.0):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_refine(f, deadline, max_iters=200, t_init=0.4, verbose=True):
    """LP-based iterative refinement. f normalized to sum=1."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=30)
        if g_raw is None:
            if verbose:
                print(f"  LP failed at iter {it}")
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step
        g_score = 2.0 * n * float(np.convolve(g, g).max())
        if g_score < best_score * 0.999:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            # Blended step
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            score_new = 2.0 * n * float(c_new.max())

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                best_M = float(c_new.max())
                f = f_new.copy()
                M = best_M
                no_improve = 0
                t_step = max(0.02, t_step * 0.998)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 50 == 0:
                    t_step = max(0.01, t_step * 0.7)
                if no_improve > 300:
                    break

        if verbose and it % 50 == 0:
            print(f"  LP iter {it}: score={best_score:.10f}, t={t_step:.4f}")

    return best_f, best_score


def multiplicative_update(a, deadline, eta=0.1, max_iters=10000, beta_init=10.0,
                          beta_end=200.0, verbose=True):
    """
    Mirror descent (multiplicative weights) on the simplex.
    a_new ∝ a * exp(-eta * grad)
    """
    n = len(a)
    a = normalize(a)
    if a is None:
        return None, np.inf
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = internal_score(a)
    beta = beta_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        _, w = smooth_max_and_weights(c, beta)

        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] - 2*mean
        grad_a_full = np.correlate(w, a, mode='full')
        grad_a = 2.0 * grad_a_full[:n]
        grad_centered = grad_a - grad_a.mean()

        # Multiplicative update
        log_update = -eta * grad_centered
        a_new = a * np.exp(log_update)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = internal_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                eta = max(0.01, eta * 0.8)
            if no_improve > 2000:
                break

        if it % 500 == 1:
            beta = min(beta_end, beta * 1.2)

        if verbose and it % 2000 == 0:
            print(f"  MU iter {it}: score={best_score:.10f}, eta={eta:.4f}, beta={beta:.1f}")

    return best_a, best_score


def adam_polish(a, deadline, lr=0.01, max_iters=50000):
    """Adam gradient descent in log-space with adaptive beta."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = internal_score(a)
    no_imp = 0
    beta = 50.0
    base_lr = lr

    while time.monotonic() < deadline:
        it += 1
        a = np.exp(z - z.max())
        a = a / a.sum()

        c = np.convolve(a, a)
        _, w = smooth_max_and_weights(c, beta)

        grad_a_full = np.correlate(w, a, mode='full')
        grad_a = 2.0 * grad_a_full[:n]
        dot_ag = np.dot(a, grad_a)
        gz = a * (grad_a - dot_ag)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()

        sc = internal_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 1000 == 0:
            beta = min(10000.0, beta * 1.3)
            if no_imp > 500:
                lr = max(0.001, lr * 0.85)

        if no_imp > 3000:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            z = best_z + 0.1 * rng.normal(size=n)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = 20.0

    return best_a, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def propose_candidate(seed=42):
    """
    Multi-strategy search: diverse init + L-BFGS + LP + multiplicative update + Adam.
    """
    t_start = time.monotonic()
    total_budget = 1800.0
    deadline_final = t_start + total_budget - 30.0

    # Get parent if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                print(f"Parent: n={len(parent)}, score={internal_score(parent):.10f}")
    except Exception:
        pass

    # Phase 1: Diverse initialization (30s)
    deadline_p1 = t_start + 30.0
    candidates = []
    scores = []

    if parent is not None:
        # Downsample/upsample parent to various sizes
        for n_try in [256, 384, 512, 640, 768, 1024]:
            p = interpolate_seq(parent, n_try)
            if p is not None:
                candidates.append(p)
                scores.append(internal_score(p))

    # Geometric initializations
    for n_try in [256, 512, 768]:
        for r in [1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 8.0]:
            a = make_geometric(n_try, r)
            candidates.append(a)
            scores.append(internal_score(a))

    # Linear profiles
    for n_try in [256, 512]:
        for eta in [1.0, 1.5, 2.0, 3.0]:
            a = make_linear(n_try, eta)
            candidates.append(a)
            scores.append(internal_score(a))

    # Polynomial profiles
    for n_try in [512]:
        for coeffs in [[1.5, -0.5], [2.0, -1.0], [1.5, -1.0, 0.5], [2.0, -1.5, 0.3]]:
            a = make_polynomial(n_try, coeffs)
            candidates.append(a)
            scores.append(internal_score(a))

    # Sort and keep top
    order = np.argsort(scores)
    print(f"\nPhase 1: {len(candidates)} candidates, best score = {scores[order[0]]:.10f}")
    for i in order[:10]:
        print(f"  n={len(candidates[i])}, score={scores[i]:.10f}")

    best_a = candidates[order[0]].copy()
    best_score = scores[order[0]]

    # Phase 2: L-BFGS optimization on top candidates (600s)
    deadline_p2 = t_start + 630.0
    top_n = min(8, len(candidates))
    beta_schedule = [5, 20, 100, 500, 2000]

    for idx in order[:top_n]:
        if time.monotonic() > deadline_p2:
            break
        a_init = candidates[idx].copy()
        n_init = len(a_init)
        if n_init != 512:
            a_init = interpolate_seq(a_init, 512)
            if a_init is None:
                continue
        a_init = np.maximum(a_init, 1e-15)
        a_init = a_init / a_init.sum()

        print(f"\nPhase 2: L-BFGS on candidate {idx} (n={len(a_init)}, init_score={internal_score(a_init):.10f})")
        t_phase = time.monotonic()

        # Run L-BFGS with progressive beta
        z = np.log(a_init)
        z -= z.mean()
        for beta in beta_schedule:
            if time.monotonic() > deadline_p2:
                break

            def fun_grad(z_in, b=beta):
                f, g = obj_grad_logspace(z_in, b)
                return f, g

            try:
                from scipy.optimize import minimize
                res = minimize(fun_grad, z, jac=True, method='L-BFGS-B',
                               options={'maxiter': 3000, 'maxfun': 3000,
                                        'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 30})
                if res.x is not None:
                    z = np.clip(res.x, -50, 50)
                    z -= z.mean()
                    a_new = np.exp(z - z.max())
                    a_new = a_new / a_new.sum()
                    sc = internal_score(a_new)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        print(f"    beta={beta}: score={sc:.10f}")
            except Exception as e:
                print(f"    L-BFGS error at beta={beta}: {e}")
                break

    print(f"\nAfter Phase 2: best_score={best_score:.10f}, n={len(best_a)}")

    # Phase 3: LP refinement (500s)
    deadline_p3 = t_start + 1130.0
    if time.monotonic() < deadline_p3 - 30:

        if a512 is not None:

            a_lp, sc_lp = lp_refine(a512, deadline_p3, max_iters=200, t_init=0.35)
            if a_lp is not None and sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp
                print(f"  LP result: score={best_score:.10f}")

    # Phase 4: Multiplicative update polish (350s)
    deadline_p4 = t_start + 1480.0
    if time.monotonic() < deadline_p4 - 30:
        a_cur = interpolate_seq(best_a, 512)
        if a_cur is not None:

            a_mu, sc_mu = multiplicative_update(a_cur, deadline_p4, eta=0.05,
                                                 beta_init=10.0, beta_end=500.0)
            if a_mu is not None and sc_mu < best_score:
                best_score = sc_mu
                best_a = a_mu

    # Phase 5: Final Adam polish (250s)

    if time.monotonic() < deadline_p5 - 10:

    # Phase 6: Try different lengths for the final sequence
    if time.monotonic() < deadline_final:
        print(f"\nPhase 6: Multi-length exploration")
        for n_try in [384, 448, 512, 576, 640, 768, 896, 1024]:
            if time.monotonic() > deadline_final:
                break
            a_try = interpolate_seq(best_a, n_try)
            if a_try is None:

            sc = internal_score(a_try)
            if sc < best_score:

    return [float(x) for x in best_a]
