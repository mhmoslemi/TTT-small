    conv = np.fft.irfft(faf * fw, N)

    return 2.0 * h

Multi-scale logit-space gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based soft-max gradients with adaptive beta sharpening, multi-resolution

import numpy as np

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_{m=0}^{n-1} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:

        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w

    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))

        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def logit_lambdad_optimize(a0, deadline, rng, lr=0.02, beta_factor=0.05,
                           beta_decay=0.90, beta_min=1e-10, restart_period=1000):
    """
    Logit-space optimization with LAMBDA-style adaptive step (no first moment).
    Simpler and sometimes more stable for this problem.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # RMSprop-style (no momentum)
        rms = np.sqrt(np.mean(dz * dz)) + 1e-10
        step = current_lr * dz / rms
        step = np.clip(step, -0.1, 0.1)
        z = z + step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 60 == 0:
            beta = max(beta_min, beta * beta_decay)
            current_lr = max(0.003, current_lr * 0.99)

        if no_improve > restart_period:
            z = best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.06 * max_c)
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def peak_targeting_refine(a0, deadline, rng, n_peaks=5, lr=0.03):
    """
    Refine by specifically targeting the top peaks in the convolution.
    Creates correction vectors that push down the identified peaks.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()
    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = float(2.0 * n * c.max())

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        # Identify top peaks
        peak_indices = np.argsort(c)[-n_peaks:]

        # Create targeted weights: uniform over top peaks
        w = np.zeros_like(c)
        for idx in peak_indices:
            w[idx] = 1.0 / n_peaks

        # Gradient
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score:
            best_score = score
            best_a = a0.copy()

    return best_a, best_score


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_linear_profile(n, eta))
    # Cosine
    for k in [1, 2, 3]:
        profiles.append(make_cosine_profile(n, k, 0.4))
    # Smooth bumps
    for width in [0.3, 0.5, 0.7]:
        for offset in [0.2, 0.4, 0.6]:
            profiles.append(make_smooth_bump(n, width, offset))
    # Power ramps
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(make_ramp_profile(n, p))
    # Random smooth (low-frequency)
    for _ in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 4)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, c_val in enumerate(coeffs, 1):
            a = a + c_val * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: explore at small n, refine at medium n,
    polish at large n. Uses logit-space Adam with adaptive soft-max beta,
    peak-targeting refinement, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting multi-phase search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick exploration at n=256 =====
    n1 = 256
    phase1_deadline = min(deadline, t_start + 150.0)
    print(f"\n--- Phase 1: n={n1} (exploration) ---")

    profiles_1 = diverse_profiles(n1, rng)
    if init_seq is not None:
        profiles_1.insert(0, interpolate_profile(init_seq, n1))

    best_p1 = None
    best_s1 = np.inf
    for idx, p in enumerate(profiles_1):
        if time.monotonic() > phase1_deadline - 3:
            break
        sc = compute_score(p)
        if sc >= best_s1 + 0.1:
            continue
        dl = min(phase1_deadline, time.monotonic() + 12.0)
        result, sc_r = logit_adam_optimize(p, dl, rng, lr=0.05,
                                            beta_factor=0.1, beta_decay=0.90)
        sc_final = compute_score(result)
        if sc_final < best_s1:
            best_s1 = sc_final
            best_p1 = result.copy()
        update(result, sc_final)
        if idx < 5 or sc_final < best_s1 + 0.01:
            print(f"  Profile {idx}: score={sc_final:.8f}")

    if best_p1 is not None:
        print(f"  Phase 1 best: {best_s1:.10f}")
        update(best_p1, best_s1)

    # ===== Phase 2: Refinement at n=512 =====
    n2 = 512
    phase2_deadline = min(deadline, t_start + 550.0)
    print(f"\n--- Phase 2: n={n2} (refinement) ---")

    if best_p1 is not None:
        p2_base = interpolate_profile(best_p1, n2)
    elif init_seq is not None and init_seq.size == n2:
        p2_base = init_seq.copy()
    else:
        p2_base = make_linear_profile(n2, 1.5)

    # Main optimization at n=512
    dl2a = min(phase2_deadline, time.monotonic() + 200.0)
    result, sc = logit_adam_optimize(p2_base, dl2a, rng, lr=0.03,
                                      beta_factor=0.06, beta_decay=0.93)
    sc_final = compute_score(result)
    update(result, sc_final)
    print(f"  After main Adam: {sc_final:.10f}")

    # Peak targeting
    if time.monotonic() < phase2_deadline - 10:
        dl2b = min(phase2_deadline, time.monotonic() + 120.0)
        result2, sc2 = peak_targeting_refine(result, dl2b, rng, n_peaks=3, lr=0.02)
        sc2_final = compute_score(result2)
        update(result2, sc2_final)
        print(f"  After peak targeting: {sc2_final:.10f}")
        if sc2_final < sc_final:
            result = result2

    # Try alternative profiles
    profiles_2 = diverse_profiles(n2, rng)
    for idx, p in enumerate(profiles_2[:6]):
        if time.monotonic() > phase2_deadline - 5:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.08:
            continue
        dl = min(phase2_deadline, time.monotonic() + 25.0)
        r, sc_r = logit_adam_optimize(p, dl, rng, lr=0.04, beta_factor=0.08)
        sc_f = compute_score(r)
        update(r, sc_f)
        if sc_f < best_score + 0.01:
            print(f"  Alt profile {idx}: {sc_f:.8f}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Large n refinement =====
    n3 = 1024
    phase3_deadline = min(deadline, t_start + 1200.0)
    print(f"\n--- Phase 3: n={n3} (large scale) ---")

    if best_seq is not None:
        p3_base = interpolate_profile(best_seq, n3)
    else:
        p3_base = interpolate_profile(p2_base, n3)

    sc3_init = compute_score(p3_base)
    update(p3_base, sc3_init)
    print(f"  Interpolated to n={n3}: {sc3_init:.10f}")

    # Long optimization at n=1024
    dl3a = min(phase3_deadline, time.monotonic() + 400.0)
    result3, sc3 = logit_adam_optimize(p3_base, dl3a, rng, lr=0.02,
                                        beta_factor=0.04, beta_decay=0.94,
                                        restart_period=2000)
    sc3_final = compute_score(result3)
    update(result3, sc3_final)
    print(f"  After large-scale Adam: {sc3_final:.10f}")

    # Peak targeting at large scale
    if time.monotonic() < phase3_deadline - 20:
        dl3b = min(phase3_deadline, time.monotonic() + 150.0)
        r3p, sc3p = peak_targeting_refine(result3, dl3b, rng, n_peaks=5, lr=0.015)
        sc3p_final = compute_score(r3p)
        update(r3p, sc3p_final)
        print(f"  After large-scale peak targeting: {sc3p_final:.10f}")
        if sc3p_final < sc3_final:
            result3 = r3p

    # Also try n=2048 if time allows and we're not yet below target
    n4 = 2048
    if time.monotonic() < phase3_deadline - 100 and best_score > 1.503:
        print(f"\n  Trying n={n4}...")
        p4 = interpolate_profile(best_seq if best_seq is not None else p3_base, n4)
        sc4_init = compute_score(p4)
        update(p4, sc4_init)
        print(f"  Interpolated to n={n4}: {sc4_init:.10f}")
        dl4 = min(phase3_deadline, time.monotonic() + 100.0)
        result4, sc4 = logit_adam_optimize(p4, dl4, rng, lr=0.015,
                                            beta_factor=0.03, beta_decay=0.95)
        sc4_final = compute_score(result4)
        update(result4, sc4_final)
        print(f"  After n={n4} Adam: {sc4_final:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Final polish at best n =====
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Final polish ---")

    if best_seq is not None:
        n_polish = best_seq.size
        # Multiple polish rounds with different strategies
        for round_idx, (lr_val, beta_f, method) in enumerate([
            (0.015, 0.02, "adam"),
            (0.01, 0.015, "lambd"),
            (0.008, 0.01, "adam"),
        ]):
            if time.monotonic() > phase4_deadline - 5:
                break
            dl = min(phase4_deadline, time.monotonic() + 100.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.0005 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            if method == "adam":
                r, sc_r = logit_adam_optimize(p0, dl, rng, lr=lr_val,
                                               beta_factor=beta_f, beta_decay=0.95)
            else:
                r, sc_r = logit_lambdad_optimize(p0, dl, rng, lr=lr_val,
                                                  beta_factor=beta_f, beta_decay=0.93)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Polish round {round_idx} ({method}, lr={lr_val}): {sc_f:.10f}")

        # Peak targeting final
        if time.monotonic() < phase4_deadline - 10:
            dl_pt = min(phase4_deadline, time.monotonic() + 40.0)
            r_pt, sc_pt = peak_targeting_refine(best_seq, dl_pt, rng,
                                                 n_peaks=3, lr=0.008)
            sc_pt_f = compute_score(r_pt)
            update(r_pt, sc_pt_f)
            print(f"  Final peak targeting: {sc_pt_f:.10f}")

    # ===== Try smaller n in case it's actually better =====
    if best_seq is not None and best_seq.size > 512:
        for n_try in [256, 384, 512]:
            if time.monotonic() > deadline - 10:
                break
            p_small = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_small)
            update(p_small, sc_s)
            if sc_s < best_score + 0.01:
                dl_s = min(deadline, time.monotonic() + 30.0)
                r_s, _ = logit_adam_optimize(p_small, dl_s, rng, lr=0.02,
                                              beta_factor=0.03, beta_decay=0.93)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Smaller n={n_try}: {sc_s2:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    # ===== Final validation and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
