Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

def next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()


def fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3

    A = np.zeros((2 * n - 1, n), dtype=np.float64)

            print(f"      iter={it_count}, M={M:.10f}, best={best_M:.10f}, tau={tau:.1f}")

    return best_a, best_M


def optimize_lp_refine(a0, deadline, rng, max_iter=10000):
    """
    LP-based iterative refinement.
    Alternates between LP direction finding and line search.
    """
    a = np.maximum(a0, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)

    c = fft_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    no_imp = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        g = solve_lp_direction(a, M, n, time_limit=10)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            a = best_a + 0.02 * np.abs(rng.normal(size=n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = fft_convolve(a)
            M = c.max()
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                a = best_a + 0.02 * np.abs(rng.normal(size=n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                c = fft_convolve(a)
                M = c.max()
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    no_imp = 0
                continue

        g = g / g_sum
        improved = False
        for t in (0.5, 0.3, 0.7, 0.2, 0.4, 0.6, 0.1, 0.8, 0.15, 0.25):
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-30:
                continue
            a_new = a_new / s
            c_new = fft_convolve(a_new)
            M_new = float(c_new.max())
            if M_new < M - 1e-14:
                a = a_new
                M = M_new
                c = c_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                a = 0.8 * best_a + 0.2 * (np.abs(rng.normal(size=n)) + 0.1)
                a = a / a.sum()
                c = fft_convolve(a)
                M = c.max()
                no_imp = 0

    return best_a, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search:
    1. Quick exploration at n=512 from diverse starts (log-sum-exp gradient).
    2. Extended LP optimization on best candidates.
    3. Final log-sum-exp polish at low temperature.
    4. Interpolation to n=1024 with further refinement.
    5. Return best sequence found.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: Quick exploration at n=512
    n1 = 512
    phase1_time = 240.0
    phase1_end = min(deadline, t_start + phase1_time)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
            starts.append(p_int + 0.02 * rng.normal(size=n1))
            starts[-1] = np.maximum(starts[-1], 1e-12)
            starts[-1] = starts[-1] / starts[-1].sum()

    for p in make_deterministic_profiles(n1)[:8]:
        starts.append(p)
    for _ in range(6):
        starts.append(make_smooth_random(n1, rng))
    for eta in (1.4, 1.8, 2.2, 2.8):
        starts.append(make_linear_profile(n1, eta))

    per_start = max(5.0, (phase1_end - t_start) / max(1, len(starts)))
    candidates = []

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 1e-12)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = optimize_logsumexp(p0, per, rng)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1 (n={n1}): best={best_score:.10f}, {len(candidates)} candidates")

    # Phase 2: Extended LP + log-sum-exp on top candidates
    phase2_time = 480.0
    phase2_end = min(deadline, t_start + phase1_time + phase2_time)
    top_k = min(4, len(candidates))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        time_per = max(15.0, (phase2_end - time.monotonic()) / max(1, top_k - i))

        # LP refinement
        half_time = time_per * 0.5
        per_lp = min(phase2_end - half_time * 0.2, time.monotonic() + half_time)
        p_best_lp, M_best_lp = optimize_lp_refine(p0, per_lp, rng, max_iter=5000)
        sc_lp = 2.0 * n1 * M_best_lp
        try_update(p_best_lp, sc_lp)

        # Log-sum-exp polish on the LP result
        per_ls = min(phase2_end, time.monotonic() + half_time)
        p_best_ls, M_best_ls = optimize_logsumexp(p_best_lp, per_ls, rng)
        sc_ls = 2.0 * n1 * M_best_ls
        try_update(p_best_ls, sc_ls)
        print(f"  Phase 2 cand {i}: LP={sc_lp:.10f}, LS={sc_ls:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Very low temperature polish on absolute best
    phase3_time = 180.0
    phase3_end = min(deadline, t_start + phase1_time + phase2_time + phase3_time)

    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 60.0)
        p_best, M_best = optimize_logsumexp(best_seq, per, rng)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)

        # Alternate with LP
        if time.monotonic() < phase3_end - 30:
            per2 = min(phase3_end, time.monotonic() + 45.0)
            p_best2, M_best2 = optimize_lp_refine(best_seq, per2, rng, max_iter=8000)
            sc2 = 2.0 * n1 * M_best2
            try_update(p_best2, sc2)

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_time = 240.0
    phase4_end = min(deadline, t_start + phase1_time + phase2_time + phase3_time + phase4_time)

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end - 30, time.monotonic() + 80.0)
            p_best, M_best = optimize_logsumexp(p_ext, per, rng)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"  Phase 4 (n={n2}): score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                per2 = min(phase4_end, time.monotonic() + 50.0)
                p_best2, M_best2 = optimize_lp_refine(p_best, per2, rng, max_iter=3000)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"  Phase 4 LP (n={n2}): score={sc2:.10f}")

    # Also try a fresh start at n=1024 with a good profile
    if time.monotonic() < phase4_end - 30:
        for p0 in [make_linear_profile(n2, 2.0), make_smooth_random(n2, rng)]:
            if time.monotonic() > phase4_end - 30:
                break
            per = min(phase4_end - 30, time.monotonic() + 30.0)
            p_best, M_best = optimize_logsumexp(p0, per, rng)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final refinement if time remains
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 30.0)
        p_best, M_best = optimize_logsumexp(best_seq, per, rng)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.8)
    print(f"Fallback score: {compute_score(fb):.10f}")
    return [float(x) for x in fb]
