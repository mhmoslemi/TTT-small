def compute_score(a):

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def smooth_max_and_weights(c, beta):
    """Compute smooth max and softmax weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))

        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    val = mval + beta * np.log(s)
    return val, w

We know conv(a,a)[k] <= M and conv(a,g)[k] <= M. But we don't control conv(g,g). So the new peak could be (1-t)^2*M + 2t(1-t)*M + t^2*max(conv(g,g)) = M*(1-t)^2 + 2tM(1-t) + t^2*max(conv(g,g)).

If max(conv(g,g)) > M, the new peak could be higher. So we need to be careful.

A safer approach: after finding g from the LP, check the new score directly and only accept if it improved. This is what the previous code does.

"""
Hybrid smooth-gradient + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses soft-max smoothing of peak autocorrelation for efficient FFT-based gradients,
combined with periodic LP-based descent steps. Multiple starting points and
adaptive annealing for robustness.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def smooth_max_and_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def compute_smooth_gradient(a, beta):
    n = a.size
    S = float(a.sum())
    c = full_convolve_fft(a)
    M, w = smooth_max_and_weights(c, beta)
    L = next_pow2(2 * n - 1 + n)
    W = np.fft.rfft(w, L)
    A_rev = np.fft.rfft(a[::-1], L)
    s_full = np.fft.irfft(W * A_rev, L)
    s = s_full[n - 1:n - 1 + n]
    grad = (4.0 * n / (S * S)) * (s - M / S)
    return grad, M, S


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def gradient_optimize(a0, deadline, rng, beta_init=None, lr_init=0.15, verbose=False, lp_every=200):
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 1e-12).copy()
    a = a / a.sum()
    n = a.size

    if beta_init is None:
        c0 = full_convolve_fft(a)
        beta_init = 0.15 * float(c0.max())

    beta = beta_init
    lr = lr_init
    base_lr = lr
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    step_count = 0

    while time.monotonic() < deadline:
        grad, M, S = compute_smooth_gradient(a, beta)

        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-10:
            a = best_a.copy()
            continue
        a = a / S
        step_count += 1

        if step_count % 5 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if step_count % lp_every == 0 and n <= 1024:
            c = full_convolve_fft(a)
            M_cur = float(c.max())
            g = lp_step(a, M_cur)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                for t_blend in [0.4, 0.6, 0.8]:
                    a_try = (1 - t_blend) * a + t_blend * g
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st > 1e-10:
                        a_try = a_try / st
                        sc_try = compute_score(a_try)
                        if sc_try < best_score - 1e-12:
                            best_score = sc_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            no_improve = 0
                            break

        if step_count % 80 == 0:
            beta = max(beta * 0.8, 1e-8 * float(c.max()))

        if no_improve > 300:
            lr = max(0.02, lr * 0.95)
        if no_improve > 800:
            scale = 0.05 + 0.05 * rng.random()
            a = best_a + scale * rng.normal(size=n)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            m.fill(0); v.fill(0); t_adam = 0
            lr = base_lr
            no_improve = 0
            c_tmp = full_convolve_fft(a)
            beta = 0.15 * float(c_tmp.max())
            if verbose:
                print(f"    Restart at step {step_count}, score={best_score:.10f}")

        if verbose and step_count % 1000 == 0:
            print(f"    GradOpt step {step_count}: score={best_score:.10f}, beta={beta:.2e}, lr={lr:.4f}")

    return best_a, best_score


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    return np.maximum(a, 1e-12) / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    return np.maximum(a, 1e-12) / a.sum()


def make_random_smooth(n, rng, num_harmonics=5):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i in range(1, num_harmonics + 1):
        a += rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
    return np.maximum(a, 1e-12) / a.sum()


def make_two_level_profile(n, rng, fill=0.5):
    a = np.zeros(n)
    mask = rng.random(n) < fill
    a[mask] = 1.0
    a[~mask] = rng.uniform(0.1, 0.5, n - int(mask.sum()))
    # Smooth slightly
    kernel = np.array([0.25, 0.5, 0.25])
    a_smooth = np.convolve(a, kernel, mode='same')
    return np.maximum(a_smooth, 1e-12) / a_smooth.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def get_starting_profiles(n, rng):
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_linear_profile(n, eta))
    for k in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            profiles.append(make_cosine_profile(n, k, amp))
    for _ in range(6):
        profiles.append(make_random_smooth(n, rng))
    for _ in range(4):
        profiles.append(make_two_level_profile(n, rng))
    # Symmetric profiles
    half = n // 2
    for _ in range(3):
        h = np.abs(rng.normal(1.0, 0.3, half + 1))
        a = np.concatenate([h, h[-2::-1]])
        if len(a) < n:
            a = np.pad(a, (0, n - len(a)), mode='edge')
        else:
            a = a[:n]
        profiles.append(np.maximum(a, 1e-12) / a.sum())
    return profiles


def lp_deep_optimize(f_init, deadline, rng, verbose=False):
    """Extended LP-based optimization with tighter M tracking."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 1e-12).copy()
    f = f / f.sum()
    n = len(f)

    best_M = float(full_convolve_fft(f).max())
    best_f = f.copy()
    t = 0.4
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = float(full_convolve_fft(f).max())
        if stagnant > 0:
            M = best_M

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new
        M_new = float(full_convolve_fft(f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 50 == 0:
                print(f"    LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
        else:
            t *= 0.5
            if t < 0.003:
                t = 0.003
            stagnant += 1

        t = max(0.01, t * 0.995)

        if stagnant > 100:
            noise = 0.1 * rng.normal(size=n)
            f = best_f + noise
            f = np.maximum(f, 1e-12)
            f = f / f.sum()
            t = 0.4
            stagnant = 0

    return best_f
