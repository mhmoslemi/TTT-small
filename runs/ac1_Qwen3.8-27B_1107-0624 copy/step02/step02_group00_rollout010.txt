    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy().ravel()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _conv_matrix_sparse(a):
    """Build sparse matrix A where (A@b)[k] = sum_j a[j]*b[k-j] (valid indices)."""
    n = len(a)
    # Each column j has nonzeros at rows j, j+1, ..., j+n-1 with value a[j]
    total_nz = 0
    for j in range(n):
        total_nz += n  # each column has exactly n nonzeros
    # Actually row j of column j goes from j to j+n-1, all within [0, 2n-2]
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    vals = np.repeat(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_tight(a, time_limit=3.0):
    """Solve max sum(b) s.t. conv_matrix(a) @ b <= max(conv(a,a)), b >= 0."""
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max())
    if M <= 0:
        return a.copy()
    A = _conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def _smooth_max_info(c, beta):
    """Compute smooth max value and weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def _grad_smooth_max(a, beta):
    """Gradient of smooth max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_info(c, beta)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is 2 * conv(w_reversed, a) truncated to length n
    # Actually: grad[j] = 2 * (w * a_rev)[j] where a_rev = a[::-1]
    # (w * a_rev)[j] = sum_k w[k] * a_rev[k-j] = sum_k w[k] * a[n-1-k+j]
    # Hmm, let me think more carefully.
    # d(conv(a,a)[k])/da[j] = a[k-j] (if 0<=k-j<=n-1)
    # So grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # Let i = k-j, k = i+j: grad[j] = 2 * sum_i w[i+j] * a[i] (for valid i, i+j)
    # This is 2 * (cross_correlation of w and a)[j]
    
    # Using FFT: cross_correlation(w, a)[j] = sum_i w[i+j] * a[i]
    # = IFFT(FFT(w) * conj(FFT(a)))[j]... 
    # Actually, let's use the convolution property:
    # sum_i w[i+j] * a[i] = (w_rev * a)[n-1+j] where w_rev = w[::-1]
    # No wait, let me just compute it directly using FFT.
    
    N = _next_pow2(2 * n - 1)
    # We want h[j] = sum_k w[k] * a[(k-j) mod ...] 
    # Let's use: h = ifft(fft(w, N) * fft(a_reversed, N))[n-1 : 2n-1]
    # where a_reversed = a[::-1]
    # Actually: sum_k w[k] * a[k-j] for 0<=k-j<n
    # = sum_i w[i+j] * a[i] for 0<=i<n and 0<=i+j<2n-1
    # Let's define: for each j, h[j] = sum_{i=0}^{n-1} w[i+j] * a[i]
    # This is the "anti-diagonal" sum of the outer product w ⊗ a.
    # Equivalently: h[j] = (a * w_shifted)[n-1-j]... 
    # Let me just use the convolution directly.
    
    # Method: h[j] = sum_i a[i] * w[i+j]
    # Define v = w (length 2n-1 padded to N)
    # h[j] = sum_i a[i] * w[i+j] = (a reversed) convolved with w, evaluated at specific point
    # Actually: let a_rev = a[::-1], length n
    # (a_rev * w)[k] = sum_i a_rev[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]
    # We want h[j] = sum_i a[i] * w[i+j]
    # Let i' = n-1-i: h[j] = sum_{i'} a[n-1-i'] * w[n-1-i'+j]
    # = (a_rev * w)[n-1+j] where the convolution is defined as (a_rev*w)[k] = sum_{i'} a_rev[i'] * w[k-i']
    # So h[j] = conv(a_rev, w)[n-1+j]
    
    a_rev = a[::-1]
    N2 = _next_pow2(2 * n - 1 + 2 * n - 1)  # enough for full conv of a_rev (len n) and w (len 2n-1)
    fa_rev = np.fft.rfft(a_rev, N2)
    fw = np.fft.rfft(w, N2)
    h_full = np.fft.irfft(fa_rev * fw, N2)
    # h[j] = h_full[n-1+j] for j=0,...,n-1
    h = h_full[n-1 : 2*n-1]
    grad = 2.0 * h
    return grad, float(c.max())

def _gradient_polish(a, deadline, max_iter=50000):
    """Gradient polish using smooth max approximation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-6, 0.05 * float(c.max()))
    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s = a_nest.sum()
        if s < 1e-10:
            a_nest = a.copy()
        else:
            a_nest = a_nest / s
        
        grad, maxc = _grad_smooth_max(a_nest, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.93)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
        
        if it % 100 == 0:
            beta = max(1e-8, beta * 0.8)
        
        if it % 1000 == 0:
            print(f"    Grad iter {it}: score={score:.10f}, beta={beta:.2e}, lr={lr:.4f}")
    
    return best_a
