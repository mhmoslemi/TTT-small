"""
Hybrid LP + Coordinate Descent + Gradient optimization.
Key improvements over previous: LP with line search (no fixed blend),
cyclic coordinate descent for fine-tuning, and refined Adam polish.
"""

import numpy as np
import time
from scipy.optimize import linprog

# === Core utilities ===

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

# === LP machinery ===

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _solve_lp(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

# === LP refinement with line search ===

def _lp_refine(a, deadline, max_iter=50000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    no_imp = 0
    alpha = 0.3  # initial blend
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max()) * 1.0001  # tiny inflation for numerical safety
        
        g = _solve_lp(a, M)
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-10:
            no_imp += 1
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        
        # Line search: try several blend values
        best_t = 0.0
        best_s = best_score
        for t in [alpha * 0.5, alpha, alpha * 1.5, alpha * 2, 0.1, 0.5]:
            t = min(1.0, max(0.01, t))
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            s_new = _score(a_new)
            if s_new < best_s:
                best_s = s_new
                best_t = t
        
        if best_t == 0.0:
            no_imp += 1
            # Try smaller steps
            for t in [0.01, 0.005, 0.001]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                s_new = _score(a_new)
                if s_new < best_s:
                    best_s = s_new
                    best_t = t
        
        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            a = a_new
            if best_s < best_score:
                best_score = best_s
                best_a = a.copy()
                no_imp = 0
            alpha = min(0.5, alpha * 1.1)
        else:
            no_imp += 1
            alpha = max(0.01, alpha * 0.8)
            if no_imp > 500:
                a = best_a.copy()
                alpha = 0.3
                no_imp = 0
                if _score(a) > best_score * 1.01:
                    break
    
    return best_a

def _coord_descent(a, deadline, max_passes=1000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for pass_idx in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try modifying a[i]
            ai = a[i]
            if ai < 1e-15:
                # Try small positive values
                candidates = [1e-4, 1e-3, 0.005, 0.01]
            else:
                candidates = [ai * 0.5, ai * 0.75, ai * 1.25, ai * 1.5, ai * 2.0, 0.0]
            
            best_val = ai
            best_s = _score(a)  # current score (no change)
            
            for x in candidates:
                a[i] = max(0.0, x)
                s = a.sum()
                if s < 1e-10:
                    a[i] = ai
                    continue
                a_norm = a / s
                s_val = _score(a_norm)
                if s_val < best_s:
                    best_s = s_val
                    best_val = x
            
            if best_val != ai:
                a[i] = best_val
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    improved = True
                    if _score(a) < best_score:
                        best_score = _score(a)
                        best_a = a.copy()
            else:
                a[i] = ai
        
        if not improved:
            break
    
    return best_a

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w

    # We want h[i] for i = 0, ..., n-1
    h = h[:n]
