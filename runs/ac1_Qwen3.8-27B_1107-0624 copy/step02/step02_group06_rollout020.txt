```python
from scipy.optimize import linprog

def lp_step(a, M):
    """
    Solve: max sum(g) s.t. a*g <= M, g >= 0
    """
    n = len(a)
    # Build the constraint matrix A such that A @ g <= M
    # (a*g)_k = sum_j a_{k-j} * g_j for valid j
    # For k = 0, ..., 2n-2:
    #   (a*g)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_{k-j} * g_j
    
    # Number of constraints: 2n-1
    # Number of variables: n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return None, None
```

```python
"""
LP-Gradient hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based blending steps, soft-max gradient optimization, and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = c - mn
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def build_convolve_matrix_sparse(a):
    """Build sparse matrix A such that (A @ g)_k = (conv(a, g))_k"""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_blend_step(a, M, method='highs'):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
    Returns the optimal g (unnormalized).
    """
    n = len(a)
    A = build_convolve_matrix_sparse(a)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        return g
    return None

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e / e.sum()
    return p

def soft_max_adam(p0, deadline, rng, lr=0.08, beta_init_frac=0.15):
    """Soft-max gradient descent in logit space with Adam."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    
    z = np.log(p)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.90, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0
    
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)
    
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m_adam = b1 * m_adam + (1 - b1) * gz
        v_adam = b2 * v_adam + (1 - b2) * (gz * gz)
        mh = m_adam / (1 - b1 ** adam_t)
        vh = v_adam / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue
        
        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.6)
        
        if no_imp > max(300, 3 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score

        profiles.append(a / a.sum())
    
    # Two-level
    for k_frac in [0.3, 0.4, 0.5]:
        for ratio in [0.1, 0.3, 0.5]:
            a = np.ones(n)
            a[:int(n*k_frac)] = 1.0
            a[int(n*k_frac):] = ratio
            profiles.append(a / a.sum())

def coordinate_descent_polish(p, deadline, rng, step_frac=0.1):
    """Coordinate descent to polish a sequence."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    
    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    sf = step_frac
    
    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for factor in [1+sf, 1+0.5*sf, 1+2*sf, 1-sf, 1-0.5*sf, 1-0.3*sf]:
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                # Quick score check
                c = full_convolve_fft(p)
                sc = 2.0 * n * c.max() / (s * s)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if improved:
            sf = max(0.02, sf * 0.9)
        if no_imp > n:
            break
    
    return best_p, best_score
```

```python
"""
LP-Gradient hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Primary driver: LP-based blending (find best g s.t. conv(a,g)<=M).
Secondary: soft-max gradient in logit space with Adam.
Polish: coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def build_convolve_matrix_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_blend_step(a, M):
    n = len(a)
    A = build_convolve_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-15:
            return g
    return None


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            mask[int(np.argmax(c))] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(e) / e.size
    return e / s


def soft_max_adam(p0, deadline, rng, lr=0.06, beta_init_frac=0.12):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    z = np.log(p)
    z -= float(np.mean(z))

    b1, b2, eps = 0.90, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0

    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m_adam = b1 * m_adam + (1 - b1) * gz
        v_adam = b2 * v_adam + (1 - b2) * (gz * gz)
        mh = m_adam / (1 - b1 ** adam_t)
        vh = v_adam / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 120 == 0:
            beta = max(1e-14, beta * 0.6)

        if no_imp > max(400, 3 * n):
            z = 0.6 * best_z + 0.8 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def make_profiles(n, rng):
    t = np.linspace(0.0, 1.0, n)
    profiles = []

    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    for eta in [1.0, 2.0, 3.0, 5.0, 7.0]:
        a = np.exp(eta * (t - 0.5))
        profiles.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = t**p + 1e-8
        profiles.append(a / a.sum())

    for k in [1, 2, 3, 5]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    for k in [2, 3, 5, 7, 11]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k * t) + 0.2 * np.sin(2 * np.pi * (k+1) * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    for _ in range(15):
        nfreq = rng.integers(2, 8)
        a = np.ones(n)
        for _ in range(nfreq):
            f = rng.uniform(0.5, 8.0)
            ph = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.8)
            a += amp * np.cos(2 * np.pi * f * t + ph)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    for _ in range(10):
        a = np.maximum(rng.normal(1.0, 0.3, n), 0.01)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    for k_frac in [0.25, 0.35, 0.45, 0.55]:
        for ratio in [0.1, 0.3, 0.5, 0.7]:
            a = np.ones(n)
            k = int(n * k_frac)
            a[:k] = 1.0
            a[k:] = ratio
            profiles.append(a / a.sum())

    # Random binary-like (low autocorrelation inspired)
    for _ in range(5):
        a = (rng.random(n) > 0.5).astype(float)
        # Smooth slightly
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())

    return profiles


def coordinate_descent_polish(p, deadline, rng, step_frac=0.08):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    sf = step_frac

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            c = full_convolve_fft(p)
            base_max = c.max()
            s_base = p.sum()
            factors = [1+sf, 1+0.5*sf, 1+2*sf, 1-sf, 1-0.5*sf, 1-0.3*sf, 1+3*sf, 1-0.2*sf]
            for factor in factors:
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                c2 = full_convolve_fft(p)
                sc = 2.0 * n * c2.max() / (s * s)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if improved:
            sf = max(0.02, sf * 0.85)
        if no_imp > n:
            break

    return best_p, best_score


def lp_optimize(p, deadline, rng, max_iters=500):
    """LP-based optimization: iterate LP blending step."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = normalize_nonneg(p)
    if p is None:
        return None, np.inf
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = full_convolve_fft(p)
        M = float(c.max())
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_p = p.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        g = lp_blend_step(p, M)
        if g is None:
            # Try with slightly relaxed M
            g = lp_blend_step(p, M * 1.001)
            if g is None:
                if stuck_count > 20:
                    # Restart with perturbation
                    p = 0.7 * best_p + 0.3 * np.maximum(rng.normal(1, 0.2, n), 0.01)
                    p = normalize_nonneg(p)
                    if p is None:
                        break
                    stuck_count = 0
                continue
            if g is None:
                break

        s_g = float(g.sum())
        if s_g < 1e-10:
            break
        g = g / s_g

        # Line search over step sizes
        best_t = 0.1
        best_sc_t = score
        for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            p_try = (1 - t_try) * p + t_try * g
            p_try = normalize_nonneg(p_try)
            if p_try is None:
                continue
            sc_try = internal_score(p_try)
            if sc_try < best_sc_t:
                best_sc_t = sc_try
                best_t = t_try

        if best_t < 0.04:
            # Try smaller steps
            for t_try in [0.02, 0.03, 0.04]:
                p_try = (1 - t_try) * p + t_try * g
                p_try = normalize_nonneg(p_try)
                if p_try is None:
                    continue
                sc_try = internal_score(p_try)
                if sc_try < best_sc_t:
                    best_sc_t = sc_try
                    best_t = t_try

        if best_t < 0.01:
            if stuck_count > 10:
                p = 0.8 * best_p + 0.2 * np.maximum(rng.normal(1, 0.3, n), 0.01)
                p = normalize_nonneg(p)
                if p is None:
                    break
                stuck_count = 0
            continue

        p = (1 - best_t) * p + best_t * g
        p = normalize_nonneg(p)
        if p is None:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 25.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} {label}")

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = normalize_nonneg(parent)
            if pn is not None:
                parent = pn
    except Exception:
        parent = None

    N = 512

    # Phase 1: Initialization
    print("Phase 1: Initialization")
    if parent is not None:
        sc = internal_score(parent)
        update(parent, sc, "parent")
        print(f"  Parent score: {sc:.10f}")
        if parent.size != N:
            x0 = np.linspace(0, 1, parent.size)
            x1 = np.linspace(0, 1, N)
            pi = np.interp(x1, x0, parent)
            pi = normalize_nonneg(pi)
            if pi is not None:
                sc = internal_score(pi)
                update(pi, sc, "parent_interp")

    profiles = make_profiles(N, rng)
    scored_profiles = []
    for p in profiles:
        sc = internal_score(p)
        scored_profiles.append((sc, p))
        update(p, sc, "init")
    scored_profiles.sort(key=lambda x: x[0])
    print(f"  Best init: {scored_profiles[0][0]:.10f}")

    # Phase 2: LP optimization on top candidates
    print("Phase 2: LP optimization")
    phase2_end = min(deadline, t_start + 700.0)

    top_k = min(6, len(scored_profiles))
    for ci in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = scored_profiles[ci]
        per = min(phase2_end, time.monotonic() + (120.0 if ci < 2 else 60.0))
        p_best, sc_best = lp_optimize(p_c, per, rng, max_iters=300)
        if p_best is not None:
            sc_final = internal_score(p_best)
            update(p_best, sc_final, f"lp_{ci}")

    print(f"  After LP phase: best={best_score:.10f}")

    # Phase 3: Gradient optimization
    print("Phase 3: Gradient optimization")
    phase3_end = min(deadline, t_start + 1200.0)

    # Gather candidates: best so far + top inits
    grad_candidates = []
    if best_seq is not None:
        grad_candidates.append(best_seq.copy())
    for sc_c, p_c in scored_profiles[:4]:
        if sc_c < best_score * 1.15:
            grad_candidates.append(p_c.copy())

    for ci, p_c in enumerate(grad_candidates[:6]):
        if time.monotonic() > phase3_end:
            break
        per = min(phase3_end, time.monotonic() + 40.0)
        p_best, sc = soft_max_adam(p_c, per, rng, lr=0.05, beta_init_frac=0.1)
        sc = internal_score(p_best)
        update(p_best, sc, f"grad_{ci}")

    print(f"  After gradient: best={best_score:.10f}")

    # Phase 4: Extended LP on best
    print("Phase 4: Extended LP")
    phase4_end = min(deadline, t_start + 1600.0)

    if best_seq is not None:
        p_best, sc_best = lp_optimize(best_seq, phase4_end, rng, max_iters=1000)
        if p_best is not None:
            sc_final = internal_score(p_best)
            update(p_best, sc_final, "lp_ext")

    # Try LP on a few more diverse starts if time remains
    if time.monotonic() < phase4_end - 60:
        for ci in range(4, min(10, len(scored_profiles))):
            if time.monotonic() > phase4_end - 30:
                break
            per = min(phase4_end - 30, time.monotonic() + 30.0)
            p_best, sc_best = lp_optimize(scored_profiles[ci][1], per, rng, max_iters=100)
            if p_best is not None:
                sc_final = internal_score(p_best)
                update(p_best, sc_final, f"lp_late_{ci}")

    print(f"  After extended LP: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1760.0)

    if best_seq is not None:
        p_cd, sc_cd = coordinate_descent_polish(best_seq, phase5_end, rng, step_frac=0.05)
        update(p_cd, sc_cd, "cd")

    # Phase 6: Final short gradient polish
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print("Phase 6: Final gradient polish")
        per = min(deadline - 10, time.monotonic() + 60.0)
        p_best, sc = soft_max_adam(best_seq, per, rng, lr=0.03, beta_init_frac=0.05)
        sc = internal_score(p_best)
        update(p_best, sc, "final_grad")

    # Final verification
    if best_seq is not None:
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s > 0:
            best_seq = best_seq / s
        sc_final = internal_score(best_seq)
        print(f"  FINAL internal: {sc_final:.10f} n={best_seq.size}")
        try:
            sc_eval = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("  Using fallback")
    fallback = make_profiles(512, rng)[0]
    return [float(x) for x in fallback]
```
