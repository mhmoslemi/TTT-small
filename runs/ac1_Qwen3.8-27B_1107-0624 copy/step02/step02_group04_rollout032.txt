```python
import numpy as np
import time

def _fft_convolve(a):
    """Full linear convolution using FFT."""
    n = len(a)
    N = 1
    while N < 2*n-1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0)  # Remove negative artifacts

def _score(a):
    """Compute the score 2n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2*n*c.max()/s**2

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    if target_sum <= 0:
        return np.zeros_like(v)
    n = len(v)
    if np.all(v <= 0):
        return np.full(n, target_sum/n)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgradient_step(a, lr, n_peaks=3):
    """One step of projected subgradient descent."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = c.max()
    # Find top peaks
    threshold = max_c * 0.95
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = [int(np.argmax(c))]
    # Average gradient over active peaks
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += 2.0 * a[ki]
    grad /= len(active)
    # Projected gradient step
    a_new = a - lr * grad
    a_new = _project_simplex(a_new, 1.0)
    return a_new, c, max_c
```

```python
"""
Optimized step function search using projected subgradient descent with exact max.
Key insight: For non-negative sequences, the max of the auto-convolution is NOT
necessarily at the center. A monotone ramp (linear) gives score ~4/3 asymptotically.
We use exact subgradient of the max (not soft-max approximation) with backtracking
line search, multi-peak averaging, and periodic restarts to escape local minima.
"""
import numpy as np
import time


def _fft_convolve(a):
    """Full linear convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_from_conv(a, c):
    """Compute score given precomputed convolution."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _score(a):
    """Compute the score 2n*max(conv(a,a))/sum(a)^2."""
    c = _fft_convolve(a)
    return _score_from_conv(a, c)


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient(a, c, frac=0.90):
    """Compute average subgradient over active (near-max) peaks."""
    n = len(a)
    max_c = float(c.max())
    if max_c <= 0:
        return np.zeros(n)
    threshold = max_c * frac
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            # grad[i_lo..i_hi] += 2 * a[k-i_hi .. k-i_lo] (reversed)
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo - 1:-1]
    grad /= float(len(active))
    return grad


def _projected_subgrad_descent(a0, deadline, rng, max_iters=5000000,
                                init_lr=0.01, frac=0.90,
                                patience=3000, restart_scale=0.3):
    """
    Projected subgradient descent for minimizing max(conv(a,a)) subject to
    a >= 0, sum(a) = 1. Uses backtracking and periodic restarts.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s

    n = a.size
    c = _fft_convolve(a)
    max_c = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * max_c)

    lr = init_lr
    no_improve = 0
    iters = 0

    while time.monotonic() < deadline and iters < max_iters:
        iters += 1
        # Compute subgradient
        grad = _subgradient(a, c, frac=frac)

        # Backtracking line search
        alpha = lr
        found = False
        for _ in range(20):
            a_trial = a - alpha * grad
            a_trial = _project_simplex(a_trial, 1.0)
            c_trial = _fft_convolve(a_trial)
            max_c_trial = float(c_trial.max())
            score_trial = float(2.0 * n * max_c_trial)

            if score_trial < best_score or score_trial < best_score * (1.0 + 1e-10):
                # Accept
                a = a_trial
                c = c_trial
                max_c = max_c_trial
                if score_trial < best_score - 1e-14:
                    best_score = score_trial
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                found = True
                lr = min(lr * 1.05, 0.5)
                break
            else:
                alpha *= 0.5

        if not found:
            # All backtracks failed, take a small random step
            a = a + 0.001 * rng.normal(size=n)
            a = _project_simplex(a, 1.0)
            c = _fft_convolve(a)
            max_c = float(c.max())
            no_improve += 1
            lr = max(lr * 0.95, 0.0001)

        # Periodic restart to escape local minima
        if no_improve > patience:
            # Add noise proportional to current solution scale
            noise = restart_scale * (no_improve / patience) * 0.1
            a = best_a + noise * rng.normal(size=n)
            a = _project_simplex(np.maximum(a, 0.0), 1.0)
            c = _fft_convolve(a)
            max_c = float(c.max())
            no_improve = 0
            lr = init_lr

        # Periodic shrink of frac (sharpen focus on max)
        if iters % 5000 == 0:
            frac = max(0.95, frac + 0.005)

    return best_a, best_score


def _coordinate_descent(a0, deadline, rng, max_rounds=50):
    """
    Coordinate descent: optimize one element at a time.
    For element i, the optimal value given all others is found by
    minimizing the resulting max convolution.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, np.inf
    a = a / s
    n = a.size

    best_a = a.copy()
    best_score = _score(a)
    order = rng.permutation(n)

    rounds = 0
    while time.monotonic() < deadline and rounds < max_rounds:
        rounds += 1
        improved = False
        for idx in order:
            i = int(idx)
            # Try a range of values for a[i]
            # The optimal a[i] given others can be found by bisection
            # since the objective is nearly unimodal in a[i]
            a_old = a[i]
            # Binary search for best a[i]
            lo = 0.0
            hi = 2.0 * best_score / (2.0 * n) * s  # rough upper bound
            # Actually, just try a few values
            candidates = [0.0, a_old * 0.5, a_old * 0.8, a_old, a_old * 1.2, a_old * 1.5]
            # Add the value that would make sum = 1 if all others stay
            if np.sum(a) - a_old > 0:
                candidates.append(max(0, 1.0 - (np.sum(a) - a_old)))
            best_val = a_old
            best_sc = best_score
            for v in candidates:
                v = max(0.0, min(1.0, v))
                a[i] = v
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = a_old
                    continue
                a_norm = a / s_new
                sc = _score(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_val = v
            a[i] = best_val
            s_new = a.sum()
            if s_new > 1e-10:
                a = a / s_new
                sc = _score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Perturb to escape
            i_rand = int(rng.integers(0, n))
            a[i_rand] = max(0.0, a[i_rand] * (1.0 + 0.1 * rng.standard_normal()))
            a = _project_simplex(a, 1.0)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _make_ramp(n, power=1.0):
    """Create a monotone ramp: a[i] = (i+1)^power, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    a = a / a.sum()
    return a


def _make_bimodal(n, center=0.5, width=0.3):
    """Bimodal profile."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.15) / width) ** 2) + np.exp(-((t - 0.85) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_custom(n, coeffs):
    """Polynomial profile: a[i] = sum_k coeffs[k] * (i/(n-1))^k."""
    t = np.linspace(0, 1, n) if n > 1 else np.array([1.0])
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * t ** k
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _parametric_search(n, deadline, rng):
    """
    Search over polynomial parameters to find good shape.
    a[i] = (1 + a1*u + a2*u^2 + a3*u^3 + a4*u^4) where u = 2*i/(n-1) - 1
    """
    best_a = None
    best_score = np.inf

    # Grid search over key parameters
    a1_vals = [0.5, 1.0, 1.5, 2.0, 2.5]
    a2_vals = [-0.5, -0.3, 0.0, 0.3, 0.5]
    a3_vals = [-0.3, 0.0, 0.3]

    t = np.linspace(-1, 1, n)

    for a1 in a1_vals:
        for a2 in a2_vals:
            for a3 in a3_vals:
                if time.monotonic() > deadline:
                    return best_a, best_score
                a = 1.0 + a1 * t + a2 * t ** 2 + a3 * t ** 3
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score
```
