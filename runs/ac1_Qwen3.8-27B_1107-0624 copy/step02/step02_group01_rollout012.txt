"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, adaptive soft-max gradient with

sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):

    for eta in (0.5, 0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5, 3.0):

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    if res.success:
        return res.x
    return None


def lp_optimize(f0, deadline, rng, max_iter=20000):
    """Run LP-based iterative optimization."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue

        g = g / g_sum

        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.4, 0.1, 0.6, 0.8, 0.05):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                no_imp = 0

    return best_f, best_M


def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=50000):
    """Polish using soft-max gradient in logit space with adaptive temperature."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, 0.1 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Soft-max weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        # Gradient via cross-correlation
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
            M_val = M_new
        else:
            no_imp += 1

        if t_adam % 200 == 0:
            beta = max(1e-12, beta * 0.6)

        if no_imp > max(200, 2 * n):
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M

def combined_optimize(f0, deadline, rng):
    """Combine LP and soft-max gradient with adaptive switching."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M

    # Alternate between LP and gradient
    phase = 'lp'
    lp_budget = 0.4
    grad_budget = 0.4
    
    t_start = time.monotonic()
    total_budget = max(1.0, deadline - t_start)
    
    lp_deadline = t_start + lp_budget * total_budget
    grad_deadline = deadline
    
    # Phase 1: LP
    if M > 0:
        f, M = lp_optimize(f, lp_deadline, rng, max_iter=10000)
    
    if M < best_M:
        best_M = M
        best_f = f.copy()

    # Phase 2: Gradient polish
    f, M = softmax_gradient_polish(f, grad_deadline, rng, lr=0.05, max_iter=30000)
    
    if M < best_M:
        best_M = M
        best_f = f.copy()

    return best_f, best_M

"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, adaptive soft-max gradient with
backtracking, subgradient averaging, and multi-start at various sequence lengths.
Key improvements: better time management, more diverse initializations,
adaptive temperature scheduling, and combined LP+gradient phases.

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a2, phase=0.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a2 * (u - phase) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):

        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def solve_lp_direction(f, M, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 15})
    if res.success:
        return res.x
    return None


def lp_optimize(f0, deadline, rng, max_iter=20000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 80:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue

        g = g / g_sum

        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.4, 0.1, 0.6, 0.8, 0.05, 0.15):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 40 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = float(fft_convolve(f).max())
                no_imp = 0

    return best_f, best_M


def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=50000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, 0.1 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
            M_val = M_new
        else:
            no_imp += 1

        if t_adam % 150 == 0:
            beta = max(1e-12, beta * 0.55)

        if no_imp > max(200, 2 * n):
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def subgradient_optimize(f0, deadline, rng, max_iter=20000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    best_f = f.copy()
    best_M = float(fft_convolve(f).max())

    step = 0.03
    no_imp = 0
    f_avg = f.copy()
    avg_count = 0
    restart_count = 0
    max_restarts = 10

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(f)
        M_val = float(c.max())
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        km = k_star - np.arange(n)
        valid = (km >= 0) & (km < n)
        grad[valid] = 2.0 * f[km[valid]]

        decay = 1.0 / (1.0 + 0.002 * it)
        step_cur = step * decay

        f_new = f - step_cur * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-30:
            f_new = f.copy()
        else:
            f_new = f_new / s

        f = f_new
        avg_count += 1
        f_avg = f_avg + (f - f_avg) / avg_count

        if M_val < best_M - 1e-14:
            best_M = M_val
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        M_avg = float(fft_convolve(f_avg).max())
        if M_avg < best_M - 1e-14:
            best_M = M_avg
            best_f = f_avg.copy()
            f = f_avg.copy()
            no_imp = 0

        if no_imp > max(80, n // 4):
            restart_count += 1
            if restart_count > max_restarts:
                break
            step *= 0.6
            f = 0.8 * best_f + 0.2 * np.abs(rng.normal(size=n)) + 1e-8
            f = f / f.sum()
            f_avg = f.copy()
            avg_count = 0
            no_imp = 0

    return best_f, best_M


def combined_lp_gradient(f0, deadline, rng):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M

    t_start = time.monotonic()
    total = max(1.0, deadline - t_start)

    lp_deadline = t_start + 0.5 * total
    f, M = lp_optimize(f, lp_deadline, rng, max_iter=15000)
    if M < best_M:
        best_M = M
        best_f = f.copy()

    if time.monotonic() < deadline:
        f, M = softmax_gradient_polish(f, deadline, rng, lr=0.04, max_iter=20000)
        if M < best_M:
            best_M = M
            best_f = f.copy()

    if time.monotonic() < deadline and (deadline - time.monotonic()) > 20:
        f, M = lp_optimize(f, min(deadline, time.monotonic() + 30), rng, max_iter=5000)
        if M < best_M:
            best_M = M
            best_f = f.copy()

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: Quick exploration at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 350.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)

    det_profs = make_deterministic_profiles(n1)
    starts.extend(det_profs[:10])
    for _ in range(8):
        starts.append(make_smooth_random(n1, rng, n_harm=12))

    per_start = max(4.0, (phase1_end - t_start) / max(1, len(starts)))

    results_512 = []
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = lp_optimize(p0, per, rng, max_iter=2500)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        results_512.append((sc, p_best))
        if i % 4 == 0:
            print(f"  Phase 1 [{i}/{len(starts)}]: score={sc:.10f}, best={best_score:.10f}")

    results_512.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Deep optimization on top candidates
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(4, len(results_512))
    per_cand = max(30.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        _, p0 = results_512[i]
        per = min(phase2_end - 30, time.monotonic() + per_cand)
        p_best, M_best = combined_lp_gradient(p0, per, rng)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 2 cand {i}: score={sc:.10f}, best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Gradient polish + subgradient on the best
    phase3_end = min(deadline, t_start + 1250.0)
    if best_seq is not None and best_seq.size == n1 and time.monotonic() < phase3_end - 20:
        per = min(phase3_end, time.monotonic() + 80.0)
        p_best, M_best = softmax_gradient_polish(best_seq, per, rng, lr=0.03, max_iter=15000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"Phase 3 gradient: score={sc:.10f}")

        if time.monotonic() < phase3_end - 20:
            per = min(phase3_end, time.monotonic() + 50.0)
            p_best, M_best = subgradient_optimize(best_seq, per, rng, max_iter=8000)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"Phase 3 subgradient: score={sc:.10f}")

        if time.monotonic() < phase3_end - 10:
            per = min(phase3_end, time.monotonic() + 40.0)
            p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=8000)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"Phase 3 LP: score={sc:.10f}, best={best_score:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 100.0)
            p_best, M_best = lp_optimize(p_ext, per, rng, max_iter=8000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 LP (n={n2}): score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                per2 = min(phase4_end, time.monotonic() + 70.0)
                p_best2, M_best2 = softmax_gradient_polish(p_best, per2, rng, lr=0.03, max_iter=8000)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 4 polish (n={n2}): score={sc2:.10f}")

    # Phase 5: Also try n=256 (sometimes shorter is better for the ratio)
    n3 = 256
    if time.monotonic() < deadline - 120:
        if parent is not None:
            p_int3 = interpolate_profile(parent, n3)
        else:
            p_int3 = interpolate_profile(best_seq, n3) if best_seq is not None else None
        if p_int3 is not None:
            per = min(deadline - 80, time.monotonic() + 60.0)
            p_best, M_best = combined_lp_gradient(p_int3, per, rng)
            sc = 2.0 * n3 * M_best
            try_update(p_best, sc)
            print(f"Phase 5 (n={n3}): score={sc:.10f}")

    # Phase 6: Final refinement on absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        p_best, M_best = softmax_gradient_polish(best_seq, per, rng, lr=0.015, max_iter=10000)
        sc = compute_score(p_best)
        try_update(p_best, sc)

        if time.monotonic() < deadline - 5:
            per2 = min(deadline - 3, time.monotonic() + 20.0)
            p_best2, M_best2 = lp_optimize(best_seq, per2, rng, max_iter=3000)
            sc2 = compute_score(p_best2)
            try_update(p_best2, sc2)

    print(f"After Phase 6: best={best_score:.10f}")

    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
