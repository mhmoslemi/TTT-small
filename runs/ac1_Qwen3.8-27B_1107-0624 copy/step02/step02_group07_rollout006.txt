import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def circular_conv_score(s):
    """Approximate score using circular convolution (faster)."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    S = np.fft.rfft(s, 2*n)
    circ = np.fft.irfft(S * S, 2*n)
    # For circular conv, the "wraparound" adds some energy
    # But for sequences that taper, it's a good approximation
    return 2.0 * n * np.max(circ[:n]) / (total * total)

def spectral_flatten(s, n_iters=100, n_peaks=10):
    """GS-style spectral flattening for non-negative sequences."""
    n = len(s)
    s = s.copy()
    for i in range(n_iters):
        # Compute DFT
        S = np.fft.rfft(s)
        mag = np.abs(S)
        phase = S / np.maximum(mag, 1e-15)
        
        # Find peaks in the power spectrum
        power = mag ** 2
        # Target: make power spectrum as flat as possible
        # Use top-k peaks for hard selection
        top_k = min(n_peaks, len(power))
        peak_indices = np.argsort(power)[-top_k:]
        
        # Reduce the peak magnitudes towards the mean
        target = np.mean(power)
        for idx in peak_indices:
            if power[idx] > target:
                reduction = 1.0 - 0.1 * (power[idx] - target) / target
                S[idx] *= np.sqrt(max(reduction, 0.5))
        
        s_new = np.fft.irfft(S)
        s_new = np.maximum(s_new, 0.0)
        if s_new.sum() < 1e-15:
            break
        s_new = s_new / s_new.sum()
        s = s_new
    return s

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    theta = (cssv[rho[0]] - target_sum) / (rho[0] + 1)
    result = np.maximum(v - theta, 0)
    return result / result.sum() * target_sum  # Ensure exact sum

def compute_subgradient(s, peak_indices):
    """Compute subgradient of sum_{p in peaks} conv(s,s)[p] w.r.t. s."""
    n = len(s)
    grad = np.zeros(n)
    for p in peak_indices:
        # d(conv[p])/d(s[m]) = 2*s[p-m] for 0 <= p-m < n
        m_min = max(0, p - (n - 1))
        m_max = min(p, n - 1)
        m_range = np.arange(m_min, m_max + 1)
        pm = p - m_range
        grad[m_range] += 2.0 * s[pm]
    return grad

    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    n = len(s)
    # Build constraint matrix: A[k, j] = coefficient of b_j in conv(s,b)[k]
    # conv(s,b)[k] = sum_j s[k-j] * b_j (for valid indices)
    # So A[k, j] = s[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # b_j appears in conv(s,b)[k] for k = j, j+1, ..., j+n-1
        # coeff = s[k-j]
        A[j:j+n, j] = s
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True, 'tol': 1e-8})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        peak_indices = np.argsort(conv)[-k:]

        grad = compute_subgradient(s, peak_indices)

        v_momentum = momentum * v_momentum - eta * grad

            s = s_new
        else:

            v_momentum *= 0.5

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_subgradient(s, peak_indices):
    """Subgradient of sum_{p in peaks} conv(s,s)[p]."""
    n = len(s)
    grad = np.zeros(n)
    for p in peak_indices:
        m_min = max(0, p - (n - 1))
        m_max = min(p, n - 1)
        if m_min > m_max:
            continue
        m_range = np.arange(m_min, m_max + 1)
        pm = p - m_range
        grad[m_range] += 2.0 * s[pm]
    return grad
