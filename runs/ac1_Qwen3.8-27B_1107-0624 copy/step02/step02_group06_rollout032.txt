def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = np.maximum(c, 0.0)
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

    z = np.clip(z, -50.0, 50.0)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def golden_section_minimize(f, a, b, tol=1e-10, max_iter=50):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x_opt = (a + b) / 2.0
    return x_opt, f(x_opt)

            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            
            # Pre-compute the "base" affected autocorrelation (without delta)
            # c_affected[k] = c[k] for k in [k_lo, k_hi]
            c_aff_base = c[k_lo:k_hi + 1].copy()
            # a[k-j] for k in [j, j+n-1] is a[0], a[1], ..., a[n-1]
            a_shifted = a[0:n]  # length n
            # Check if 2j is in range
            has_2j = (2 * j <= 2 * n - 2)
            
            unchanged_max = -np.inf
            if k_lo > 0:
                unchanged_max = max(unchanged_max, float(c[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unchanged_max = max(unchanged_max, float(c[k_hi + 1:].max()))
            
            def obj(val):
                val = max(0.0, val)
                S_new = 1.0 - aj + val
                if S_new < 1e-10:
                    return 1e10
                delta = val - aj
                if abs(delta) < 1e-15:
                    return best_score if best_score < 1e9 else float(2.0 * n * c.max())
                # New affected autocorrelation
                c_aff = c_aff_base + 2.0 * delta * a_shifted
                if has_2j:
                    idx_2j = 2 * j - k_lo  # = j
                    c_aff[idx_2j] = c[2 * j] + delta * (val + aj)
                chg_max = float(c_aff.max())
                new_max = max(unchanged_max, chg_max)
                return float(2.0 * n * new_max / (S_new * S_new))
            
            # Search range: [0, max(2*aj, 5*mean(a), 0.01)]
            upper = max(2.0 * aj, 5.0 * (1.0 / n) * step_widen, 0.02)
            upper = min(upper, 1.0)  # Can't exceed 1 since sum=1
            
            val_opt, score_opt = golden_section_minimize(obj, 0.0, upper, tol=1e-12, max_iter=40)
            
            if score_opt < best_score - 1e-14:
                # Update
                delta = val_opt - aj
                a[j] = val_opt
                # Update c
                c[k_lo:k_hi + 1] += 2.0 * delta * a_shifted
                if has_2j:
                    # Need to subtract the old special term and add new one
                    # Actually we already added 2*delta*a[j] for k=2j, but we need delta*(val+aj)
                    # Correction: subtract 2*delta*a[j], add delta*(val_opt+aj)
                    c[2 * j] -= 2.0 * delta * a[j]
                    c[2 * j] += delta * (val_opt + aj)
                S_new = 1.0 - aj + val_opt
                # Renormalize
                a = a / S_new
                S = 1.0
                c = c / (S_new * S_new)
                best_score = score_opt
                best_a = a.copy()
                best_c = c.copy()
                improved_this_sweep = True
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 3 * n:
            # Try perturbations
            perturbation = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2])
            a_pert = best_a * (1.0 + perturbation * rng.normal(size=n))
            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / a_pert.sum()
            sc = internal_score(a_pert)
            if sc < best_score:
                best_score = sc
                best_a = a_pert.copy()
                best_c = full_convolve_fft(a_pert)
                a = a_pert.copy()
                c = best_c.copy()
            no_improve = 0
            step_widen = min(3.0, step_widen * 1.2)
        elif improved_this_sweep:
            step_widen = max(0.5, step_widen * 0.95)
        
        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: score={best_score:.10f}")
    
    return best_a, best_c, best_score


def soft_max_gradient_step(a, c, deadline, rng, lr=0.05, max_steps=200):
    """A few soft-max gradient steps to escape local minima."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    c = full_convolve_fft(a)
    maxc = float(c.max())
    beta = 0.1 * maxc
    
    z = np.log(np.maximum(a, 1e-15))
    z -= float(np.mean(z))
    
    best_score = float(2.0 * n * maxc)
    best_a = a.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Soft-max weights
        x = (c - maxc) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if s <= 0 or not np.isfinite(s):
            w = np.zeros(n * 2 - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        
        # Gradient of max(c) w.r.t. a: 2 * (w * a)[n-1 : 2n-1]
        # Actually, d/dc_k [max] = w_k, and dc_k/da_i = a[k-i] + a[i-k] (symmetric)
        # So da = 2 * (w convolved with a reversed), taking appropriate slice
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa, N)
        grad = 2.0 * h[n - 1:2 * n - 1]
        
        # Project gradient to simplex tangent
        grad = grad - float(np.mean(a * grad)) * a
        # Actually for softmax parameterization:
        # gz = a * (grad - <a, grad>)
        grad_proj = a * (grad - float(np.dot(a, grad)))
        
        t_adam += 1
        m = b1 * m + (1 - b1) * grad_proj
        v = b2 * v + (1 - b2) * grad_proj ** 2
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        
        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if step % 30 == 0:
            beta = max(1e-10, beta * 0.8)
            maxc = float(c.max())
    
    return best_a, best_score


def block_coord_update(a, c, j1, j2, deadline, max_iters=30):
    """Optimize two coordinates simultaneously."""
    n = len(a)
    aj1 = a[j1]
    aj2 = a[j2]
    
    def obj_2d(vals):
        v1 = max(0.0, vals[0])
        v2 = max(0.0, vals[1])
        S_new = 1.0 - aj1 - aj2 + v1 + v2
        if S_new < 1e-10:
            return 1e10
        # New autocorrelation
        c_new = c.copy()
        d1 = v1 - aj1
        d2 = v2 - aj2
        # Effect of j1
        k_lo1 = j1
        k_hi1 = min(j1 + n - 1, 2 * n - 2)
        c_new[k_lo1:k_hi1 + 1] += 2.0 * d1 * a[0:n]
        if 2 * j1 <= 2 * n - 2:
            c_new[2 * j1] -= 2.0 * d1 * a[j1]
            c_new[2 * j1] += d1 * (v1 + aj1)
        # Effect of j2
        k_lo2 = j2
        k_hi2 = min(j2 + n - 1, 2 * n - 2)
        c_new[k_lo2:k_hi2 + 1] += 2.0 * d2 * a[0:n]
        if 2 * j2 <= 2 * n - 2:
            c_new[2 * j2] -= 2.0 * d2 * a[j2]
            c_new[2 * j2] += d2 * (v2 + aj2)
        
        max_val = float(c_new.max())
        return float(2.0 * n * max_val / (S_new * S_new))
    
    # Try a grid of 2D values
    best_score = internal_score(a)
    best_v1, best_v2 = aj1, aj2
    
    # Coarse grid
    for f1 in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        for f2 in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
            v1 = aj1 * f1
            v2 = aj2 * f2
            sc = obj_2d([v1, v2])
            if sc < best_score:
                best_score = sc
                best_v1, best_v2 = v1, v2
    
    # Fine grid around best
    for f1 in np.linspace(0.5, max(1.5, best_v1 / aj1 * 1.2), 8):
        for f2 in np.linspace(0.5, max(1.5, best_v2 / aj2 * 1.2), 8):
            v1 = aj1 * f1
            v2 = aj2 * f2
            sc = obj_2d([v1, v2])
            if sc < best_score:
                best_score = sc
                best_v1, best_v2 = v1, v2
    
    if best_score < internal_score(a) - 1e-12:
        a[j1] = best_v1
        a[j2] = best_v2
        S_new = a.sum()
        a = a / S_new
        c = full_convolve_fft(a)
    
    return a, c, best_score


def search_single_length(n, deadline, rng, initial_profiles=None):
    """Search at a single length n."""
    print(f"  Searching at n={n}...")
    
    best_a = None
    best_score = np.inf
    
    # Collect initial candidates
    candidates = []
    if initial_profiles:
        for p in initial_profiles:
            p = normalize_nonneg(p)
            if p is not None and len(p) == n:
                sc = internal_score(p)
                candidates.append((sc, p.copy()))
    
    # Add analytic profiles
    for p in make_analytic_profiles(n):
        sc = internal_score(p)
        candidates.append((sc, p.copy()))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    
    # Take top candidates
    top_k = min(10, len(candidates))
    
    for ci in range(top_k):
        if time.monotonic() > deadline:
            break
        sc_init, p_init = candidates[ci]
        print(f"    Candidate {ci}: initial score={sc_init:.10f}")
        
        # Phase 1: Gradient steps to get close
        g_deadline = min(deadline, time.monotonic() + max(2.0, 30.0 / n * 10))
        p_grad, sc_grad = soft_max_gradient_step(p_init, full_convolve_fft(p_init), g_deadline, rng, lr=0.03, max_steps=100)
        
        # Phase 2: Coordinate descent polish
        cd_deadline = min(deadline, time.monotonic() + max(5.0, 60.0 / n * 10))
        p_cd, c_cd, sc_cd = coord_descent_polish(p_grad, full_convolve_fft(p_grad), 1.0, 
                                                   cd_deadline, rng, max_sweeps=200)
        
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = p_cd.copy()
            print(f"    New best: {sc_cd:.10f}")
    
    # Final aggressive polish on the best
    if best_a is not None and time.monotonic() < deadline:
        final_deadline = deadline
        best_a, _, best_score = coord_descent_polish(best_a, full_convolve_fft(best_a), 1.0,
                                                      final_deadline, rng, max_sweeps=1000)
        print(f"    Final polish: {best_score:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining coordinate descent with line search,
    gradient steps, and block updates to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(a, sc, label=""):
        nonlocal best_seq, best_score
        if a is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(a, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(a)}) {label}")
    
    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
            if parent is not None:
                sc = internal_score(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent score: {sc:.10f} (n={parent.size})")
    except Exception:
        pass
    
    # Try parent at different lengths
    if parent is not None:
        for tn in (128, 256, 512, 1024):
            if parent.size != tn:
                p_interp = interpolate_profile(parent, tn)
                if p_interp is not None:
                    sc = internal_score(p_interp)
                    update_best(p_interp, sc, f"parent_n{tn}")
    
    # Phase 1: Quick search at n=256 (fast, good for initialization)
    print("Phase 1: n=256")
    p1_deadline = min(deadline, t_start + 300.0)
    if parent is not None:
        p256 = interpolate_profile(parent, 256)
        inits_256 = [p256] if p256 is not None else []
    else:
        inits_256 = []
    
    p_best_256, sc_best_256 = search_single_length(256, p1_deadline, rng, inits_256)
    if p_best_256 is not None:
        update_best(p_best_256, sc_best_256, "n256_search")
    
    # Phase 2: Search at n=512 (main target)
    print("Phase 2: n=512")
    p2_deadline = min(deadline, t_start + 1200.0)
    inits_512 = []
    if parent is not None:
        p512 = interpolate_profile(parent, 512)
        if p512 is not None:
            inits_512.append(p512)
    if p_best_256 is not None:
        p512_from_256 = interpolate_profile(p_best_256, 512)
        if p512_from_256 is not None:
            inits_512.append(p512_from_256)
    
    p_best_512, sc_best_512 = search_single_length(512, p2_deadline, rng, inits_512)
    if p_best_512 is not None:
        update_best(p_best_512, sc_best_512, "n512_search")
    
    # Phase 3: Try n=1024 if time remains
    if time.monotonic() < t_start + 1500.0 and best_seq is not None:
        print("Phase 3: n=1024")
        p3_deadline = min(deadline, t_start + 1600.0)
        inits_1024 = [interpolate_profile(best_seq, 1024)]
        p_best_1024, sc_best_1024 = search_single_length(1024, p3_deadline, rng, inits_1024)
        if p_best_1024 is not None:
            update_best(p_best_1024, sc_best_1024, "n1024_search")
    
    # Phase 4: Final aggressive polish on the overall best
    if best_seq is not None and time.monotonic() < deadline:
        print("Phase 4: Final polish")
        final_deadline = deadline
        # Get autocorrelation
        a_final = best_seq.copy()
        a_final = normalize_nonneg(a_final)
        c_final = full_convolve_fft(a_final)
        
        # Run coordinate descent with remaining time
        a_pol, c_pol, sc_pol = coord_descent_polish(a_final, c_final, 1.0,
                                                     final_deadline, rng, max_sweeps=5000)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"  Final polish improved to: {best_score:.10f}")
    
    # Final verification
    if best_seq is not None:
        # Clip and normalize
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s > 0.01:
            best_seq = best_seq / s
        
        # Verify with true score
        if best_seq.size <= 2048:
            sc_true = true_score(best_seq)
            if np.isfinite(sc_true):
                best_score = min(best_score, sc_true)
        
        # Try evaluate_sequence
        try:
            sc_eval = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(sc_eval):
                best_score = min(best_score, sc_eval)
        except Exception:
            pass
        
        print(f"  FINAL: score={best_score:.10f}, n={best_seq.size}")
        return [float(x) for x in best_seq]
    
    # Fallback
    print("  Using fallback")
    fb = make_raised_cosine(256)
    print(f"  Fallback score: {internal_score(fb):.10f}")
    return [float(x) for x in fb]
