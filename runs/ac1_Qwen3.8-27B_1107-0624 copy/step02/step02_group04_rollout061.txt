"""
Optimized step function search using top-K FFT gradient with adaptive K,
parametric initialization (tilted Gaussians), variance-aware objective,
and structured restarts. exploits the fact that the gradient of conv(a,a)_k
is 2*reversed-shifted(a), enabling efficient FFT-based gradient computation.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 0.01:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _make_tilted_gaussian(n, center_frac, width_frac, tilt):
    t = np.arange(n, dtype=np.float64)
    c = center_frac * n
    w = width_frac * n
    x = (t - c) / max(w, 1.0)
    a = np.exp(-0.5 * x * x + tilt * x)
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _make_sigmoid_ramp(n, sharpness, base):
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = base + (1.0 - base) / (1.0 + np.exp(-sharpness * (t - 0.5)))
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    return a / s


def _make_combined_profile(n, center_frac, width_frac, tilt, ramp_sharp):
    t = np.arange(n, dtype=np.float64)
    c = center_frac * n
    w = width_frac * n
    x = (t - c) / max(w, 1.0)
    g = np.exp(-0.5 * x * x + tilt * x)
    ramp = 1.0 / (1.0 + np.exp(-ramp_sharp * (t / n - 0.5)))
    a = g * (0.5 + 0.5 * ramp)
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return None
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _topk_fft_gradient(c, p, K, beta):
    """
    Compute soft-max objective estimate and gradient using top-K positions.
    Gradient of c_k w.r.t. p_i = 2*p[k-i] (if valid).
    Uses FFT for efficient computation.
    Returns (L, grad) where L is soft-max estimate and grad is gradient array.
    """
    n = len(p)
    c_len = c.size  # 2n-1
    K = min(K, c_len)

    if K == 1:
        k_star = int(np.argmax(c))
        val = float(c[k_star])
        # Gradient: 2 * p[k_star - i] for valid i
        grad = np.zeros(n)
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        if i_min <= i_max:
            j_min = k_star - i_max
            j_max = k_star - i_min
            grad[i_min:i_max + 1] = 2.0 * p[j_min:j_max + 1][::-1]
        return val, grad

    # Find top K
    if K >= c_len:
        top_idx = np.arange(c_len)
        top_vals = c.copy()
    else:
        part_idx = np.argpartition(c, -K)
        top_idx = part_idx[-K:]
        top_vals = c[top_idx]

    # Soft weights
    if beta <= 1e-14:
        w = np.zeros(K)
        w[int(np.argmax(top_vals))] = 1.0
        L = float(top_vals.max())
    else:
        x = (top_vals - top_vals.max()) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(K)
            w[int(np.argmax(top_vals))] = 1.0
        else:
            w = ex / s
        L = float(top_vals.max()) + beta * float(np.log(s if s > 0 else 1))

    # Build weight vector of length 2n-1
    w_full = np.zeros(c_len)
    w_full[top_idx] = w

    # Gradient via FFT: grad = 2 * convolve(w_full, p_reversed)[n-1 : 2n-1]
    p_rev = p[::-1]
    N = _next_pow2(c_len + n - 1)
    fw = np.fft.rfft(w_full, N)
    fp = np.fft.rfft(p_rev, N)
    conv_res = np.fft.irfft(fw * fp, N)
    grad = 2.0 * conv_res[n - 1:2 * n - 1]

    # Clean up numerical noise
    grad = np.abs(grad) * np.sign(grad)  # ensure clean zeros
    grad[~np.isfinite(grad)] = 0.0

    return L, grad


def _variance_gradient(c, p, n_active):
    """
    Approximate gradient of variance of the top n_active convolution values.
    d(var)/da_i = (2/M) * sum_j (c_j - mean) * dc_j/da_i
    Approximate using top-M positions.
    """
    n = len(p)
    M = min(n_active, c.size)
    if M < 2:
        return np.zeros(n)

    top_idx = np.argpartition(c, -M)[-M:]
    top_vals = c[top_idx]
    mean_c = float(top_vals.mean())

    # Weight each top position by (c_j - mean)
    w_full = np.zeros(c.size)
    w_full[top_idx] = (top_vals - mean_c) / (2.0 * M)

    p_rev = p[::-1]
    N = _next_pow2(c.size + n - 1)
    fw = np.fft.rfft(w_full, N)
    fp = np.fft.rfft(p_rev, N)
    conv_res = np.fft.irfft(fw * fp, N)
    grad = conv_res[n - 1:2 * n - 1]
    grad[~np.isfinite(grad)] = 0.0
    return grad


def _optimize_logit(p0, deadline, rng, lr=0.05, var_weight=0.0,
                    K_init=3, K_max=16, beta_frac=0.1,
                    no_impust_limit=300, name=""):
    """
    Main optimizer: Adam in logit space with top-K FFT gradient.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    score = 2.0 * n * maxc
    beta = max(1e-10, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = score
    no_imp = 0
    K_cur = K_init

    while time.monotonic() < deadline:
        # Compute top-K gradient
        L, grad_c = _topk_fft_gradient(c, p, K_cur, beta)

        # Add variance penalty
        if var_weight > 0.0:
            n_act = min(50, c.size)
            grad_v = _variance_gradient(c, p, n_act)
            grad_c = grad_c + var_weight * grad_v

        # Convert to logit-space gradient (DPP constraint: sum p = 1)
        dot = float(np.dot(p, grad_c))
        gz = p * (grad_c - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        # Adam update
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            # Adaptive K and beta
            if no_imp % 50 == 0:
                # Check top-2 gap
                if c.size >= 2:
                    sorted_c = np.sort(c)[::-1]
                    gap = sorted_c[0] - sorted_c[1]
                    if gap < beta * 0.1:
                        K_cur = min(K_max, K_cur + 2)
                    else:
                        K_cur = max(K_init, K_cur - 1)
                beta = max(1e-12, beta * 0.85)

        # Learning rate decay on stagnation
        if no_imp > 100 and no_imp % 50 == 0:
            lr = max(0.01, lr * 0.92)

        # Restart
        if no_imp > no_impust_limit:
            noise_scale = 0.15 if best_score < 1.52 else 0.25
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            K_cur = K_init
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = 0.05
            no_imp = 0

    return best_p, best_score


def _quick_parametric_search(n, deadline, rng):
    """
    Random search over tilted Gaussian profiles.
    Returns list of (score, profile) sorted by score.
    """
    results = []
    n_iter = 0
    while time.monotonic() < deadline:
        n_iter += 1
        cf = rng.uniform(0.25, 0.75)
        wf = rng.uniform(0.15, 0.45)
        tilt = rng.uniform(-3.0, 3.0)
        p = _make_tilted_gaussian(n, cf, wf, tilt)
        sc = _internal_score(p)
        if np.isfinite(sc):
            results.append((sc, p))
        # Also try combined profiles
        if n_iter % 3 == 0:
            rs = rng.uniform(2.0, 8.0)
            p2 = _make_combined_profile(n, cf, wf, tilt, rs)
            sc2 = _internal_score(p2)
            if np.isfinite(sc2):
                results.append((sc2, p2))
        if n_iter % 5 == 0:
            sh = rng.uniform(2.0, 10.0)
            base = rng.uniform(0.1, 0.5)
            p3 = _make_sigmoid_ramp(n, sh, base)
            sc3 = _internal_score(p3)
            if np.isfinite(sc3):
                results.append((sc3, p3))

    results.sort(key=lambda x: x[0])
    return results

        z2 = np.log(np.maximum(best_seq, 1e-12))
        z2 -= z2.mean()
        z2 += 0.3 * rng.normal(size=n_best)
        z2 = np.clip(z2, -60, 60)
        z2 -= z2.mean()
        restart_profiles.append(("perturb_med", _softmax_from_z(z2)))

        # Restart 3: New profile at best length
        cf = rng.uniform(0.3, 0.7)
        wf = rng.uniform(0.2, 0.4)
        tilt = rng.uniform(-2, 2)
        restart_profiles.append(("new_gauss", _make_tilted_gaussian(n_best, cf, wf, tilt)))

        # Restart 4: Sigmoid ramp
        sh = rng.uniform(3, 8)
        base = rng.uniform(0.1, 0.4)
        restart_profiles.append(("new_ramp", _make_sigmoid_ramp(n_best, sh, base)))

        # Restart 5: Combined
        restart_profiles.append(("new_comb", _make_combined_profile(
            n_best, rng.uniform(0.3, 0.7), rng.uniform(0.2, 0.4),
            rng.uniform(-2, 2), rng.uniform(2, 6))))

        n_restarts = len(restart_profiles)
        per_restart = min(80.0, (p4_end - time.monotonic()) / max(1, n_restarts))

        for label, p_start in restart_profiles:
            if time.monotonic() > p4_end:
                break
            per_end = min(p4_end, time.monotonic() + per_restart)
            p_try, sc_try = _optimize_logit(p_start, per_end, rng, lr=0.05,
                                            var_weight=0.005, K_init=4,
                                            K_max=12, beta_frac=0.08,
                                            no_impust_limit=200)
            _update_best(p_try, sc_try, f"restart {label}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # --- Phase 5: Final polish on the best ---
    print("Phase 5: Final polish")
    if best_seq is not None:
        p5_end = deadline
        remaining = p5_end - time.monotonic()
        per_final = min(remaining, 200.0)
        if per_final > 30:
            p_fin, sc_fin = _optimize_logit(best_seq, time.monotonic() + per_final,
                                            rng, lr=0.02, var_weight=0.0,
                                            K_init=1, K_max=4, beta_frac=0.05,
                                            no_impust_limit=400)
            _update_best(p_fin, sc_fin, "final_polish")

    # --- Validate and return ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s * s  # keep same scale, just clip
            final = np.clip(final, 0.0, 1000.0)
        else:
            final = _make_tilted_gaussian(512, 0.5, 0.3, 1.0)

        # Exact score check
        n_fin = final.size
        if n_fin <= 4096:
            sc_exact = _true_score(final)
            print(f"Exact score: {sc_exact:.10f}")
            if np.isfinite(sc_exact):
                best_score = min(best_score, sc_exact)

        # Try evaluator
        if n_fin <= 4096:
            try:
                sc_ev = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_ev):
                    best_score = min(best_score, sc_ev)
                    print(f"Evaluator score: {sc_ev:.10f}")
            except Exception:
                pass

        print(f"FINAL: best_score={best_score:.10f}, n={n_fin}")
        return [float(x) for x in final]

    # Fallback
    print("FALLBACK: using tilted Gaussian n=512")
    fb = _make_tilted_gaussian(512, 0.5, 0.3, 1.0)
    return [float(x) for x in fb]
