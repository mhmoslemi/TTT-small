"""
Hybrid gradient + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive softmax-gradient descent as the primary optimizer, with LP polishing
for final precision. Explores multiple sequence lengths and diverse initial profiles.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_smooth_random(n, rng, n_harm=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.8) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5, 3.0):
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in ((1.0, -0.5), (1.2, -0.8), (1.5, -1.2), (1.8, -1.5),
                   (2.0, -2.0), (2.0, -3.0), (2.5, -3.5), (1.5, -2.5),
                   (0.8, 0.5), (1.0, 1.0), (0.5, -2.0)):
        profs.append(make_cubic_profile(n, a1, a3))
    t = np.linspace(0, 1, n)
    # Quadratic profiles
    for b in (-0.5, -1.0, -1.5, 0.5, 1.0):
        a = 1.0 + b * (t - 0.5)**2 * 4
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    # Sinusoidal
    for freq, amp in ((1, 0.5), (1, 1.0), (2, 0.5), (2, 1.0), (3, 0.5), (1, 1.5)):
        a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    return profs

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None

def gradient_optimize(f0, deadline, rng, max_iter=200000, top_k=5, 
                      lr_init=0.02, temp_init=0.05, verbose=False):
    """
    Adaptive softmax gradient descent with top-k subgradient.
    f0 must be normalized (sum=1) and positive.
    Returns (best_f, best_M).
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    z = np.log(f)
    z -= z.mean()
    
    c = fft_convolve(f)
    M_val = float(c.max())
    
    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    
    # Adam state
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    # Adaptive temperature for top-k selection
    temp = temp_init
    lr = lr_init
    
    # Track the "excess" for adaptive temperature
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Recompute convolution
        c = fft_convolve(f)
        M_val = float(c.max())
        
        # Select top-k positions for subgradient
        # Use soft weighting: weight proportional to exp((c[k] - M) / temp)
        x = (c - M_val) / temp
        x = np.clip(x, -50.0, 0.0)
        w = np.exp(x)
        s_w = float(w.sum())
        if s_w < 1e-30:
            w = np.zeros_like(c)
            idx = np.argsort(c)[-top_k:]
            w[idx] = 1.0 / top_k
            s_w = 1.0
        w = w / s_w
        
        # Gradient: grad_a[i] = 2 * sum_k w[k] * a[k-i]
        # This is 2 * (w * a)[i] where * is cross-correlation
        # (w * a)[i] = sum_k w[k] * a[k-i]
        # We can compute this via FFT: it's the cross-correlation of w and a
        N_corr = next_pow2(3 * n)
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        # Cross-correlation: ifft(conj(FW) * FA) gives (w * a)
        # Actually, (w * a)[i] = sum_k w[k] * a[k-i]
        # This is the cross-correlation, which is ifft(conj(FW) * FA)
        # Wait, let me be more careful.
        # FFT of w convolved with a: FFT(w) * FFT(a) = FFT(w * a) where * is convolution
        # But we want cross-correlation: (w ⋆ a)[i] = sum_k w[k] * a[k-i]
        # Cross-correlation = convolution with reversed: (w ⋆ a)[i] = (w * flip(a))[i]
        # In FFT: FFT(w ⋆ a) = conj(FFT(w)) * FFT(a)
        # But we need to be careful with indexing.
        
        # Actually, let me use a different approach.
        # grad_a[i] = 2 * sum_k w[k] * a[k-i] (where a[m]=0 for m outside [0,n-1])
        # Let h[i] = sum_k w[k] * a[k-i]
        # This is the cross-correlation of w and a, evaluated at lag i.
        # h = np.correlate(w, a, mode='full')[n-1:n-1+n] ... hmm, this is tricky.
        
        # Let me use FFT:
        # Define W = fft(w, N) and A = fft(a, N) with N >= n + len(w) - 1
        # The cross-correlation h[i] = sum_k w[k] * a[k-i] 
        # = sum_k w[k] * a_flipped[i-k] where a_flipped[m] = a[-m]
        # This is the convolution of w with a_flipped.
        # FFT(h) = FFT(w) * FFT(a_flipped) = FFT(w) * conj(FFT(a)) (for real a)
        
        # Actually, for real signals: conv(w, flip(a)) has FFT = FFT(w) * conj(FFT(a))
        # And conv(w, flip(a))[i] = sum_k w[k] * flip(a)[i-k] = sum_k w[k] * a[k-i]
        # Yes! That's exactly what we want.
        
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]
        grad_a = 2.0 * h
        
        # Convert to z-gradient: grad_z = a * (grad_a - dot(grad_a, a))
        dot_ga = float(np.dot(grad_a, f))
        grad_z = f * (grad_a - dot_ga)
        
        if not np.all(np.isfinite(grad_z)):
            # Reset to best
            f = best_f.copy()
            z = np.log(f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        
        # Adaptive learning rate
        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z -= z.mean()
        
        # Softmax to get new f
        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Check improvement
        c = fft_convolve(f)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive temperature and learning rate
        if t_adam % 500 == 0:
            temp = max(1e-6, temp * 0.95)
            lr = max(0.001, lr * 0.99)
        
        # Restart if stuck
        if no_imp > max(500, 5 * n):
            # Try different restart strategies
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Mix with best
                z = 0.7 * z + 0.3 * (np.log(best_f) - np.log(best_f).mean())
            elif strategy == 1:
                # Add noise
                z = z + 0.5 * rng.normal(size=n)
            elif strategy == 2:
                # Mix with random
                z_rand = rng.normal(size=n)
                z = 0.8 * z + 0.2 * z_rand
            else:
                # Reset to best with small perturbation
                z = np.log(best_f)
                z -= z.mean()
                z = z + 0.1 * rng.normal(size=n)
            
            z = np.clip(z, -80.0, 80.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            temp = max(0.01, temp * 2.0)
            lr = min(0.05, lr * 1.5)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
        
        if verbose and it % 1000 == 0:
            score = 2.0 * n * M_val
            print(f"    Grad iter {it}: M={M_val:.10f}, best={best_M:.10f}, score={score:.10f}, temp={temp:.6f}")
    
    return best_f, best_M


def lp_optimize(f0, deadline, rng, max_iter=5000, verbose=False):
    """LP-based optimization (simplified version for polishing)."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, float('inf')
    f = f / s
    n = f.size
    
    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Build LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        # (f*g)[k] = sum_j f[j] * g[k-j]
        # For each k, the constraint is: sum_j f[j] * g[k-j] <= M
        # where k-j must be in [0, n-1]
        
        # Build constraint matrix A of shape (2n-1, n)
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f
        
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if not res.success:
            no_imp += 1
            if no_imp > 30:
                break
            continue
        
        g = res.x
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        g = g / g_sum
        
        # Line search
        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.4, 0.6, 0.1, 0.8):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break
        
        if not improved:
            no_imp += 1
            if no_imp > 50:
                break
    
    return best_f, best_M

Hybrid gradient + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive softmax-gradient descent with top-k subgradient as primary optimizer,

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5, 3.0, 3.5):
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in ((1.0, -0.5), (1.2, -0.8), (1.5, -1.2), (1.8, -1.5),
                   (2.0, -2.0), (2.0, -3.0), (2.5, -3.5), (1.5, -2.5),
                   (0.8, 0.5), (1.0, 1.0), (0.5, -2.0), (2.2, -2.8)):
        profs.append(make_cubic_profile(n, a1, a3))
    t = np.linspace(0, 1, n)
    for b in (-0.5, -1.0, -1.5, -2.0, 0.5, 1.0, 1.5):
        a = 1.0 + b * (t - 0.5) ** 2 * 4
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    for freq, amp in ((1, 0.5), (1, 1.0), (2, 0.5), (2, 1.0), (3, 0.5),
                      (1, 1.5), (2, 1.5), (1, 2.0), (3, 1.0)):
        for phase in (0.0, 0.5, 1.0):
            a = 1.0 + amp * np.sin(2 * np.pi * freq * t + phase)
            a = np.maximum(a, 1e-12)
            profs.append(a / a.sum())
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def compute_cross_correl(w, a):
    """Compute h[i] = sum_k w[k] * a[k-i] for i=0..n-1 using convolution."""
    n = a.size
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    h = conv_result[n - 1:2 * n - 1]
    return h


def gradient_optimize(f0, deadline, rng, max_iter=200000, top_k=5,
                      lr_init=0.02, temp_init=0.05, sparsify_interval=5000,
                      sparsify_thresh=1e-8, verbose=False):
    """Adaptive softmax gradient descent with top-k soft subgradient and sparsification."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    c = fft_convolve(f)
    M_val = float(c.max())

    best_f = f.copy()
    best_M = M_val
    no_imp = 0

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    temp = temp_init
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(f)
        M_val = float(c.max())

        # Soft weights for subgradient
        x = (c - M_val) / temp
        x = np.clip(x, -50.0, 0.0)
        w = np.exp(x)
        s_w = float(w.sum())
        if s_w < 1e-30:
            w = np.zeros_like(c)
            idx = np.argsort(c)[-top_k:]
            w[idx] = 1.0 / max(top_k, 1)
            s_w = 1.0
        w = w / s_w

        # Compute grad_a[i] = 2 * sum_k w[k] * a[k-i]
        h = compute_cross_correl(w, f)
        grad_a = 2.0 * h

        # Convert to z-gradient: grad_z = a * (grad_a - dot(grad_a, a))
        dot_ga = float(np.dot(grad_a, f))
        grad_z = f * (grad_a - dot_ga)

        if not np.all(np.isfinite(grad_z)):
            f = best_f.copy()
            z = np.log(f)
            z -= z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Sparsification: set very small elements to near-zero
        if sparsify_interval > 0 and it > 0 and it % sparsify_interval == 0:
            thresh = sparsify_thresh * f.max()
            mask = f < thresh
            if mask.any() and not mask.all():
                f_sp = f.copy()
                f_sp[mask] = 0.0
                f_sp = f_sp / f_sp.sum()
                c_sp = fft_convolve(f_sp)
                M_sp = float(c_sp.max())
                if M_sp < M_new + 1e-10:
                    f = f_sp
                    z = np.log(np.maximum(f, 1e-15))
                    z -= z.mean()

        # Adaptive schedule
        if t_adam % 1000 == 0:
            temp = max(1e-6, temp * 0.92)
            lr = max(0.001, lr * 0.995)

        # Restart if stuck
        if no_imp > max(1000, 5 * n):
            strategy = rng.integers(0, 5)
            if strategy == 0:
                z = 0.7 * z + 0.3 * (np.log(best_f) - np.log(best_f).mean())
            elif strategy == 1:
                z = z + 0.5 * rng.normal(size=n)
            elif strategy == 2:
                z_rand = rng.normal(size=n)
                z = 0.8 * z + 0.2 * z_rand
            elif strategy == 3:
                z = np.log(best_f)
                z -= z.mean()
                z = z + 0.1 * rng.normal(size=n)
            else:
                # Try sparsification restart
                f_sp = best_f.copy()
                thresh = 0.01 * f_sp.max()
                f_sp[f_sp < thresh] = 0.0
                f_sp = f_sp / f_sp.sum()
                z = np.log(np.maximum(f_sp, 1e-15))
                z -= z.mean()

            z = np.clip(z, -80.0, 80.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            temp = max(0.01, temp * 2.0)
            lr = min(0.05, lr * 1.5)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 2000 == 0:
            score = 2.0 * n * M_val
            print(f"    Grad iter {it}: M={M_val:.10f}, best={best_M:.10f}, "
                  f"score={score:.10f}, temp={temp:.6f}, no_imp={no_imp}")

    return best_f, best_M


def lp_optimize(f0, deadline, rng, max_iter=5000, verbose=False):
    """LP-based block coordinate descent for polishing."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, float('inf')
    f = f / s
    n = f.size

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if not res.success:
            no_imp += 1
            if no_imp > 40:
                break
            if no_imp % 10 == 0:
                f = best_f + 0.01 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                    M = best_M
                    continue
                f = f / fs
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
            continue

        g = res.x
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 150:
                break
            continue

        g = g / g_sum

        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.4, 0.6, 0.1, 0.8, 0.45, 0.35):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp > 60:
                break

        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}")

    return best_f, best_M


def multi_phase_gradient(f0, deadline, rng, n_phases=3):
    """Run gradient optimization in multiple phases with decreasing parameters."""
    f = f0
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        phase_frac = 1.0 / (n_phases - phase)
        phase_deadline = min(deadline, time.monotonic() + phase_frac * 100)
        lr = 0.02 * (0.5 ** phase)
        temp = 0.05 * (0.5 ** phase)
        f, M = gradient_optimize(f, phase_deadline, rng,
                                 max_iter=100000, top_k=3 + 2 * phase,
                                 lr_init=lr, temp_init=temp)
    return f, M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage hybrid optimization:
    1. Diverse gradient optimization at n=512 (800s)
    2. Extended gradient + multi-phase on top candidates (500s)
    3. Try n=1024 with gradient (300s)
    4. LP polishing (150s)
    5. Final gradient polish (50s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        pass

    if parent is not None:
        sc = compute_score(parent)
        update_best(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # ========== Phase 1: Diverse gradient optimization at n=512 ==========
    n1 = 512
    phase1_end = min(deadline, t_start + 750.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
        for pert_scale in (0.02, 0.05, 0.1):
            p_pert = parent + pert_scale * rng.normal(size=parent.size)
            p_pert = np.maximum(p_pert, 1e-12)
            p_pert = p_pert / p_pert.sum()
            p_int_pert = interpolate_profile(p_pert, n1)
            if p_int_pert is not None:
                starts.append(p_int_pert)

    det_profs = make_deterministic_profiles(n1)
    starts.extend(det_profs[:15])

    for _ in range(12):
        starts.append(make_smooth_random(n1, rng, n_harm=15))

    print(f"Phase 1: {len(starts)} starts, budget={phase1_end - t_start:.0f}s")

    candidates = []
    per_start = max(5.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 1e-15)
        p0 = p0 / p0.sum()

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = gradient_optimize(p0, per, rng,
                                           max_iter=150000, top_k=5,
                                           lr_init=0.02, temp_init=0.05,
                                           sparsify_interval=10000,
                                           sparsify_thresh=1e-6)
        sc = 2.0 * n1 * M_best
        update_best(p_best, sc)
        candidates.append((sc, p_best))

        if (i + 1) % 5 == 0 or i == 0:
            elapsed = time.monotonic() - t_start
            print(f"  P1 [{i+1}/{len(starts)}]: best={best_score:.10f}, t={elapsed:.0f}s")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # ========== Phase 2: Extended multi-phase gradient on top candidates ==========
    phase2_end = min(deadline, t_start + 1200.0)
    top_k = min(6, len(candidates))
    per_cand = max(20.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)

        # Multi-phase: coarse to fine
        p_best, M_best = gradient_optimize(p0, per, rng,
                                           max_iter=200000, top_k=3,
                                           lr_init=0.005, temp_init=0.01,
                                           sparsify_interval=8000,
                                           sparsify_thresh=5e-7,
                                           verbose=(i == 0))
        sc = 2.0 * n1 * M_best
        update_best(p_best, sc)

        if sc < candidates[i][0] - 1e-10:
            print(f"  P2 cand {i}: {candidates[i][0]:.10f} -> {sc:.10f}")
        else:
            print(f"  P2 cand {i}: {sc:.10f} (no improvement)")

    print(f"After Phase 2: best={best_score:.10f}")

    # ========== Phase 3: Try n=1024 ==========
    n2 = 1024
    phase3_end = min(deadline, t_start + 1580.0)

    if best_seq is not None and time.monotonic() < phase3_end - 120:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase3_end, time.monotonic() + 180.0)
            p_best, M_best = gradient_optimize(p_ext, per, rng,
                                               max_iter=150000, top_k=5,
                                               lr_init=0.015, temp_init=0.04,
                                               sparsify_interval=10000,
                                               sparsify_thresh=1e-6,
                                               verbose=True)
            sc = 2.0 * n2 * M_best
            update_best(p_best, sc)
            print(f"Phase 3 (n={n2}): score={sc:.10f}")

    if time.monotonic() < phase3_end - 40:
        det_profs_n2 = make_deterministic_profiles(n2)
        for p0 in det_profs_n2[:5]:
            if time.monotonic() > phase3_end - 20:
                break
            per = min(phase3_end - 20, time.monotonic() + 25.0)
            p_best, M_best = gradient_optimize(p0, per, rng,
                                               max_iter=80000, top_k=5,
                                               lr_init=0.02, temp_init=0.05)
            sc = 2.0 * n2 * M_best
            update_best(p_best, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # ========== Phase 4: LP polishing ==========
    phase4_end = min(deadline, t_start + 1740.0)
    if best_seq is not None and time.monotonic() < phase4_end - 10:
        n_best = best_seq.size
        if n_best <= 1024:
            per = min(phase4_end - 5, time.monotonic() + 120.0)
            p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=3000,
                                         verbose=True)
            sc = compute_score(p_best)
            update_best(p_best, sc)
            print(f"After Phase 4 LP: best={best_score:.10f}, n={n_best}")

    # ========== Phase 5: Final gradient polish ==========
    if best_seq is not None and time.monotonic() < deadline - 10:
        per = min(deadline - 5, time.monotonic() + 50.0)
        p_best, M_best = gradient_optimize(best_seq, per, rng,
                                           max_iter=100000, top_k=10,
                                           lr_init=0.003, temp_init=0.003,
                                           sparsify_interval=5000,
                                           sparsify_thresh=1e-7)
        sc = compute_score(p_best)
        update_best(p_best, sc)
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluator score: {sc_eval:.10f}")
                    if sc_eval < sc_final:
                        pass  # evaluator clips negatives, might differ slightly
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
