Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
1. Optimal step size via golden section search (instead of fixed decaying t)
2. Faster gradient polishing with adaptive parameters

5. Better time management

import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return np.maximum(c, 0.0)  # Remove tiny negative values from FFT

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})

def optimal_step_size(c_ff, c_fg, c_gg):
    """Find optimal t in [0,1] minimizing max_k cv((1-t)f+tg, (1-t)f+tg)."""
    def obj(t):
        vals = (1-t)**2 * c_ff + 2*t*(1-t) * c_fg + t**2 * c_gg
        return vals.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = 0.0, 1.0
    c_ = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c_)

            break

            fc = obj(c_)
        else:

def soft_max_and_weights(c, beta):
    """Compute soft max and weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j]."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements:
1. Optimal step size via golden section search (replaces fixed decaying t)

5. Local search refinement for final polishing

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a, b):
    """Full linear convolution using FFT with clipping of tiny negatives."""
    n = len(a)
    m = len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return np.maximum(c, 0.0)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M, time_limit=20):
    """Solve LP: max sum(b) s.t. conv_matrix(f) @ b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def golden_step(c_ff, c_fg, c_gg, tol=1e-10, max_iter=80):
    """Find optimal t in [0,1] minimizing max_k cv((1-t)f+tg, (1-t)f+tg) via golden section."""
    def obj(t):
        return np.maximum((1 - t) ** 2 * c_ff + 2 * t * (1 - t) * c_fg + t ** 2 * c_gg).max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = obj(c_pt)
    fd = obj(d_pt)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj(d_pt)
    return (a + b) / 2.0

    best_f = f.copy()
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = best_M
        g = lp_solve(f, M, time_limit=15)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        c_fg = fft_convolve(f, g)
        c_gg = fft_convolve(g, g)
        c_ff = np.convolve(f, f)

        t_opt = golden_step(c_ff, c_fg, c_gg)

        if t_opt < 1e-7:
            stagnant += 1
            if stagnant > 80:
                noise = 0.08 * max(1.0 / n, float(np.std(best_f)))
                f = best_f + noise * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                stagnant = 0
            continue

        f_new = (1 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        c_new = np.convolve(f_new, f_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-11:
            f = f_new
            best_M = M_new
            best_f = f.copy()
            stagnant = 0
            iteration += 1
            if verbose and iteration % 15 == 0:
                print(f"    LP iter {iteration}: M={best_M:.10f}, t_opt={t_opt:.4f}")
        else:
            stagnant += 1
            if stagnant > 120:
                noise = 0.1 * max(1.0 / n, float(np.std(best_f)))
                f = best_f + noise * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
            stagnant = 0 if stagnant > 120 else stagnant

    return best_f


def soft_max_weights(c, beta):
    """Compute soft max value and normalized weights."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def logit_adam(p0, deadline, rng, lr=0.05, reg=1e-8, verbose=False):
    """Adam optimization in logit space with soft-max smoothing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    p = p0.copy()
    c = np.convolve(p, p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = soft_max_weights(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Cosine decay learning rate
        frac = 0.0
        if t > 5000:
            frac = min(1.0, (t - 5000) / 20000.0)
        current_lr = lr * (1.0 - 0.7 * frac)

        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        ex = np.exp(z)
        p = ex / ex.sum()

        c = np.convolve(p, p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
            if verbose and t % 5000 == 0:
                print(f"      Adam t={t}: score={score:.10f}")
        else:
            no_imp += 1

        if t % 40 == 0:
            beta = max(1e-14, beta * 0.82)

        if no_imp > 600:
            z = 0.75 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            ex = np.exp(z)
            p = ex / ex.sum()
            c = np.convolve(p, p)
            beta = max(1e-12, 0.2 * float(c.max()))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_imp = 0

    return best_p, best_score


def local_search(p, deadline, rng, max_rounds=20):
    """Local search: perturb elements with high gradient contribution."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return p
    p = p / s
    n = len(p)
    best_score = compute_score(p)
    best_p = p.copy()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = np.convolve(p, p)
        k_star = int(np.argmax(c))
        # Gradient direction: elements contributing to peak
        grad = np.zeros(n)
        for j in range(n):
            val = p[j]
            if 0 <= k_star - j < n:
                val += p[k_star - j]
            grad[j] = val

        # Top-k elements by gradient magnitude
        top_k = min(20, n)
        idx = np.argsort(grad)[-top_k:]

        improved = False
        for j in idx:
            for delta in [0.01, 0.005, -0.01, -0.005, 0.02, -0.02]:
                p_trial = p.copy()
                p_trial[j] = max(0.0, p_trial[j] + delta * p.mean())
                s_t = p_trial.sum()
                if s_t < 1e-10:
                    continue
                p_trial = p_trial / s_t
                sc = compute_score(p_trial)
                if sc < best_score - 1e-12:
                    p = p_trial
                    best_p = p_trial.copy()
                    best_score = sc
                    improved = True
                    break
            if time.monotonic() > deadline:
                break
        if not improved and rnd > 5:
            break

    return best_p


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(1.0 + eta * (t - 0.5), 1e-12)
    return a / a.sum()


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = np.maximum(1.0 + eta * (t - 0.5), 1e-12)
        profiles.append(a / a.sum())

    # Cosine
    for k in [1, 2, 3, 4, 5]:
        for amp in [0.3, 0.5, 0.7]:
            a = np.maximum(1.0 + amp * np.cos(2 * np.pi * k * t), 1e-12)
            profiles.append(a / a.sum())

    # Sin^2
    for k in [1, 2, 3]:
        a = np.maximum(0.3 + 0.7 * np.sin(np.pi * k * t) ** 2, 1e-12)
        profiles.append(a / a.sum())

    # Bimodal
    for c1 in [0.2, 0.3, 0.35]:
        a = np.maximum(0.3 + np.exp(-20 * (t - c1) ** 2) + np.exp(-20 * (t - 1 + c1) ** 2), 1e-12)
        profiles.append(a / a.sum())

    # Random low-frequency
    for _ in range(8):
        deg = rng.integers(3, 8)
        coeffs = rng.uniform(-0.4, 0.4, deg)
        coeffs[0] = 0
        a = 1.0
        for i, c in enumerate(coeffs[1:], 1):
            a = a + c * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.maximum(np.exp(-rate * (t - 0.5) ** 2), 1e-12)
        profiles.append(a / a.sum())

    return profiles


def interpolate_profile(p0, n_new):
    """Linear interpolation to new length."""
    n0 = len(p0)
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phase 1: LP with golden-section step at n=512
    Phase 2: Adam logit polish at n=512
    Phase 3: LP + polish at n=1024
    Phase 4: Local search + final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc:.10f}")
        update(init_seq, sc)

    # ============ Phase 1: LP at n=512 ============
    n1 = 512
    phase1_end = min(deadline, t_start + 550.0)

    if init_seq is not None:
        p_main = interpolate_profile(init_seq, n1)
    else:
        p_main = make_linear_profile(n1, 1.5)

    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"Phase 1 start: n={n1}, score={sc:.10f}")

    # Main LP optimization
    lp_dl = min(phase1_end, time.monotonic() + 380.0)
    result = lp_optimize(p_main, lp_dl, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"  Main LP done: score={sc:.10f}")

    # Try diverse profiles
    profiles = make_profiles(n1, rng)
    profiles.sort(key=lambda p: compute_score(p))
    for i, p in enumerate(profiles[:12]):
        if time.monotonic() > phase1_end - 5:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.08:
            continue
        per_dl = min(phase1_end, time.monotonic() + 25.0)
        res = lp_optimize(p, per_dl, rng, verbose=False)
        sc = compute_score(res)
        update(res, sc)
        if sc < best_score + 0.02:
            print(f"  Profile {i}: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # ============ Phase 2: Adam polish at n=512 ============
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end - 15 and best_seq is not None:
        n_p = best_seq.size
        print(f"Phase 2: Adam polish at n={n_p}, score={best_score:.10f}")

        for lr in [0.05, 0.03, 0.02]:
            if time.monotonic() > phase2_end - 10:
                break
            p0 = best_seq.copy() + 0.001 * rng.normal(size=n_p)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per_dl = min(phase2_end - 5, time.monotonic() + 80.0)
            res, sc = logit_adam(p0, per_dl, rng, lr=lr)
            update(res, sc)
            print(f"  Adam lr={lr}: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # ============ Phase 3: n=1024 ============
    n3 = 1024
    phase3_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase3_end - 50 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n3)
        sc_interp = compute_score(p_large)
        update(p_large, sc_interp)
        print(f"Phase 3: n={n3}, interpolated score={sc_interp:.10f}")

        lp_dl3 = min(phase3_end - 60, time.monotonic() + 250.0)
        result = lp_optimize(p_large, lp_dl3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"  LP at n={n3}: score={sc:.10f}")

        # Quick Adam polish at n=1024
        if time.monotonic() < phase3_end - 20:
            p0 = result + 0.0005 * rng.normal(size=n3)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per_dl = min(phase3_end - 5, time.monotonic() + 50.0)
            res2, sc2 = logit_adam(p0, per_dl, rng, lr=0.03)
            update(res2, sc2)
            print(f"  Adam at n={n3}: score={sc2:.10f}")

    print(f"After Phase 3: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # ============ Phase 4: Local search + final polish ============
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_f = best_seq.size
        # Local search
        ls_dl = min(deadline - 15, time.monotonic() + 30.0)
        if n_f <= 1024:
            res_ls = local_search(best_seq, ls_dl, rng, max_rounds=10)
            sc_ls = compute_score(res_ls)
            update(res_ls, sc_ls)
            print(f"  Local search: score={sc_ls:.10f}")

        # Final Adam polish
        if time.monotonic() < deadline - 5:
            p0 = best_seq + 0.0003 * rng.normal(size=n_f)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            per = min(deadline - 3, time.monotonic() + 15.0)
            res, sc = logit_adam(p0, per, rng, lr=0.015)
            update(res, sc)
            print(f"  Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is None:
        best_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(best_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score < final_score:
                final_score = eval_score
    except Exception as e:
        print(f"eval call error: {e}")

    return [float(x) for x in final_seq]
