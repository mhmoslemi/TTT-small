from scipy.optimize import linprog
import time

def compute_score_normalized(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f, assuming sum(f)=1."""
    n = len(f)
    c = np.convolve(f, f)
    return 2.0 * n * float(c.max())

def softmax_gradient(c, beta):
    """Compute gradient of (1/beta)*log(sum(exp(c/beta))) w.r.t. c."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 50))
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s

def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1 using FFT."""
    n = len(p)
    N = 1
    needed = 2 * n - 1
    while N < needed:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]

    """Adam optimization in logit space with softmax-smoothed max objective."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    
    if beta_init is None:
        beta_init = 0.2 * float(np.convolve(f, f).max())
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = np.convolve(f, f)
    M = float(c.max())
    beta = beta_init
    
    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    beta_decay = 0.95
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute softmax weights
        w = softmax_gradient(c, beta)
        
        # Gradient of max(conv) w.r.t. f
        gp = 2.0 * corr_weights(w, f)
        
        # Add small regularization to prevent f from going to 0
        vp = gp + 1e-8 * f
        
        # Project gradient (remove mean component for logit space)
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz

        mh = m / (1 - b1**t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0

        c = np.convolve(f, f)

            best_z = z.copy()

def peak_shave(f, n_steps=100, rng=None):
    """Directly shave peaks in the convolution by redistributing mass."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    f = np.maximum(f, 1e-12).copy()
    f = f / f.sum()
    c = np.convolve(f, f)
    M = float(c.max())
    
    for step in range(n_steps):
        # Find peak lag
        k_peak = int(np.argmax(c))
        
        # Find which pairs contribute most to c[k_peak]
        # c[k_peak] = sum_j f[j] * f[k_peak - j]
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        contribs = f[lo:hi+1] * f[k_peak - np.arange(lo, hi+1)]
        
        # Find the top contributing pair
        idx_local = int(np.argmax(contribs))
        j1 = lo + idx_local
        j2 = k_peak - j1
        
        # Find minimum of c to identify where to add mass
        k_min = int(np.argmin(c))
        
        # Transfer a small amount from the peak-contributing pair to min-contributing pair
        # We need to find a pair (j3, j4) with j3+j4 = k_min
        lo_m = max(0, k_min - n + 1)
        hi_m = min(n - 1, k_min)
        # Pick the pair with smallest product
        if hi_m >= lo_m:
            products = f[lo_m:hi_m+1] * f[k_min - np.arange(lo_m, hi_m+1)]
            idx_m = int(np.argmin(products))
            j3 = lo_m + idx_m
            j4 = k_min - j3

Multi-phase optimization for minimizing 2*n*max(convolve(f,f))/sum(f)^2.
Combines Adam gradient descent, LP-based best response, and coordinate descent.

    return np.convolve(f, f)

    """Compute softmax weights for smoothing max."""
    m = float(c.max())
    x = np.clip((c - m) / beta, -100, 100)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:

        w[int(np.argmax(c))] = 1.0

    w = softmax_weights(c, beta)
