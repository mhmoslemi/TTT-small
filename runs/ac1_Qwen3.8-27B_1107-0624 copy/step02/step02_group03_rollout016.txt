"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP direction-finding with FFT-accelerated Adam in logit space,
multiple restarts, and multi-resolution strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M, timeout=5.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def fft_gradient(w, p):
    """Compute grad[i] = 2 * sum_j w[i+j] * p[j] using FFT.
    w has length 2n-1, p has length n. Returns gradient of length n.
    """
    n = len(p)
    L = 1
    while L < 3 * n:
        L *= 2
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    # Cross-correlation: ifft(conj(W) * P)
    corr = np.fft.irfft(np.conj(W) * P, L)[:n]
    return 2.0 * corr

def adam_optimize(p0, deadline, beta_init=None, lr=0.05, reg=1e-8, max_restarts=5, verbose=True):
    n = len(p0)
    if beta_init is None:
        beta_init = 0.02 * float(np.convolve(p0, p0).max())
    
    best_p = p0.copy()
    best_score = 2.0 * n * float(np.convolve(p0, p0).max())
    
    for restart in range(max_restarts):
        if time.monotonic() > deadline:
            break
        
        p = best_p.copy()
        z = np.log(np.maximum(p, 1e-12))
        z -= np.mean(z)
        
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = np.zeros(n)
        v = np.zeros(n)
        it = 0
        beta = beta_init
        cur_lr = lr
        no_imp = 0
        
        while time.monotonic() < deadline:
            it += 1
            c = np.convolve(p, p)
            max_c = float(c.max())
            score = 2.0 * n * max_c
            
            # Soft-max weights
            x = (c - max_c) / beta
            ex = np.exp(x)
            s_ex = ex.sum()
            if s_ex < 1e-30:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ex / s_ex
            
            # FFT gradient
            grad = fft_gradient(w, p) + reg * p
            
            # Logit gradient
            dot = float(np.dot(p, grad))
            gz = p * (grad - dot)
            
            if not np.all(np.isfinite(gz)):
                z = np.log(np.maximum(best_p, 1e-12))
                z -= np.mean(z)
                m.fill(0)
                v.fill(0)
                it = 0
                continue
            
            m = b1 * m + (1 - b1) * gz
            v = b2 * v + (1 - b2) * gz * gz
            mh = m / (1.0 - b1 ** it)
            vh = v / (1.0 - b2 ** it)
            z = z - cur_lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50.0, 50.0)
            z -= np.mean(z)
            
            p_new = np.exp(z - z.max())
            s_p = p_new.sum()
            if s_p < 1e-30:
                z = np.log(np.maximum(best_p, 1e-12))
                z -= np.mean(z)
                m.fill(0); v.fill(0); it = 0
                continue
            p_new = p_new / s_p
            
            c_new = np.convolve(p_new, p_new)
            score_new = 2.0 * n * float(c_new.max())
            
            if score_new < best_score - 1e-10:
                best_score = score_new
                best_p = p_new.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 2000 == 0:
                    cur_lr = max(0.01, cur_lr * 0.9)
            
            if it % 300 == 0:
                beta = max(1e-8, beta * 0.5)

                    print(f"        Adam r{restart} it{it}: score={best_score:.8f}, beta={beta:.2e}, lr={cur_lr:.4f}")
            
            if no_imp > 3000:
                break
            
            p = p_new
        
        if time.monotonic() > deadline:
            break
        # Perturb for next restart
        if restart < max_restarts - 1:
            rng = np.random.default_rng(42 + restart)
            z_current = np.log(np.maximum(best_p, 1e-12))
            z_pert = z_current + rng.normal(0, 0.15 + 0.05*restart, size=n)
            z_pert = np.clip(z_pert, -50, 50)
            z_pert -= np.mean(z_pert)
            p_pert = np.exp(z_pert - z_pert.max())
            p_pert = p_pert / p_pert.sum()
            sc_pert = 2.0 * n * float(np.convolve(p_pert, p_pert).max())
            if sc_pert < best_score:
                best_p = p_pert
                best_score = sc_pert
            beta_init = 0.02 * best_score / (2.0 * n)
    
    return best_p, best_score

def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def make_profile(n, func):
    t = np.linspace(0.0, 1.0, n)
    a = func(t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def lp_refine(f, max_iters=100, t_init=0.3, deadline=None, verbose=True):
    """LP-based refinement."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf
    
    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    t_step = t_init
    no_improve = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g_raw = lp_direction(f, M, timeout=5.0)
        if g_raw is None:
            break
        
        g = normalize(g_raw)
        if g is None:
            break
        
        # Try full step
        g_score = 2.0 * n * float(np.convolve(g, g).max())
        
        if g_score < best_score * 0.999:
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            # Blended step with binary search
            best_blended_score = best_score
            best_blended_f = f.copy()
            lo, hi = 0.0, 1.0
            for _ in range(8):
                mid = (lo + hi) / 2
                f_try = (1.0 - mid) * f + mid * g
                f_try = normalize(f_try)
                if f_try is None:
                    hi = mid
                    continue
                sc = 2.0 * n * float(np.convolve(f_try, f_try).max())
                if sc < best_blended_score:
                    best_blended_score = sc
                    best_blended_f = f_try.copy()
                    lo = mid
                else:
                    hi = mid
            
            if best_blended_score < best_score - 1e-12:
                best_f = best_blended_f.copy()
                best_score = best_blended_score
                f = best_blended_f.copy()
                M = float(np.convolve(f, f).max())
                no_improve = 0
            else:
                no_improve += 1
                if no_improve % 50 == 0:
                    t_step = max(0.05, t_step * 0.7)
                if no_improve > 100:
                    break
        
        if verbose and it % 20 == 0:
            print(f"      LP it{it}: score={best_score:.8f}")
    
    return best_f, best_score

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated Adam in logit space with soft-max smoothing,
LP direction-finding with binary line search, multiple diverse
initializations, and adaptive restarts. Targets n=512 primarily.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def fft_conv_gradient(w, p):
    """
    Compute grad[i] = 2 * sum_j w[i+j] * p[j] using FFT.
    w has length 2n-1, p has length n. Returns array of length n.
    This is the gradient of sum_k w[k]*conv(p,p)[k] w.r.t. p.
    """
    n = len(p)
    # Need L >= len(w) + len(p) - 1 = 3n - 2
    L = 1
    while L < 3 * n + 2:
        L <<= 1
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    # Cross-correlation (w ⋆ p)[i] = sum_j w[i+j] * p[j]
    # = ifft(conj(W) * P)
    corr = np.fft.irfft(np.conj(W) * P, L)[:n]
    return 2.0 * corr


def adam_optimize(p0, deadline, beta_init=None, lr=0.04, reg=1e-8,
                  max_restarts=10, verbose=True):
    """
    Adam optimization in logit space with soft-max smoothing.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1, p>=0.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    c0 = np.convolve(p0, p0)
    if beta_init is None:
        beta_init = 0.03 * float(c0.max())

    best_p = p0.copy()
    best_score = 2.0 * n * float(c0.max())

    for restart in range(max_restarts):
        if time.monotonic() > deadline:
            break

        # Initialize from best (with perturbation for restarts > 0)
        if restart == 0:
            p = best_p.copy()
        else:
            rng = np.random.default_rng(1000 + restart * 7)
            z_base = np.log(np.maximum(best_p, 1e-12))
            perturb_scale = 0.08 + 0.03 * restart
            z_pert = z_base + rng.normal(0, perturb_scale, size=n)
            z_pert = np.clip(z_pert, -50, 50)
            z_pert -= np.mean(z_pert)
            p = np.exp(z_pert - z_pert.max())
            p = p / p.sum()
            sc_pert = 2.0 * n * float(np.convolve(p, p).max())
            if sc_pert < best_score:
                best_score = sc_pert
                best_p = p.copy()

        z = np.log(np.maximum(p, 1e-12))
        z -= np.mean(z)

        b1, b2, eps = 0.9, 0.999, 1e-8
        m = np.zeros(n)
        v = np.zeros(n)
        it = 0
        beta = beta_init
        cur_lr = lr
        no_imp = 0
        stall_count = 0

        while time.monotonic() < deadline:
            it += 1
            c = np.convolve(p, p)
            max_c = float(c.max())
            score = 2.0 * n * max_c

            # Soft-max weights (temperature = beta)
            x = (c - max_c) / beta
            ex = np.exp(x)
            s_ex = ex.sum()
            if s_ex < 1e-30:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ex / s_ex

            # FFT-based gradient
            grad = fft_conv_gradient(w, p) + reg * p

            # Logit-space gradient: gz = p * (grad - p.dot(grad))
            dot_val = float(np.dot(p, grad))
            gz = p * (grad - dot_val)

            if not np.all(np.isfinite(gz)):
                z = np.log(np.maximum(best_p, 1e-12))
                z -= np.mean(z)
                m.fill(0)
                v.fill(0)
                it = 0
                continue

            m = b1 * m + (1 - b1) * gz
            v = b2 * v + (1 - b2) * gz * gz
            mh = m / (1.0 - b1 ** it)
            vh = v / (1.0 - b2 ** it)
            z = z - cur_lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50.0, 50.0)
            z -= np.mean(z)

            p_new = np.exp(z - z.max())
            s_p = p_new.sum()
            if s_p < 1e-30:
                z = np.log(np.maximum(best_p, 1e-12))
                z -= np.mean(z)
                m.fill(0)
                v.fill(0)
                it = 0
                continue
            p_new = p_new / s_p

            c_new = np.convolve(p_new, p_new)
            score_new = 2.0 * n * float(c_new.max())

            if score_new < best_score - 1e-11:
                best_score = score_new
                best_p = p_new.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 3000 == 0:
                    cur_lr = max(0.008, cur_lr * 0.92)

            # Anneal beta: decrease slowly to sharpen the max
            if it % 400 == 0:
                beta = max(1e-7, beta * 0.6)

            if verbose and it % 3000 == 0:
                print(f"        Adam r{restart} it{it}: score={best_score:.8f}, "
                      f"beta={beta:.2e}, lr={cur_lr:.5f}")

            # Break on stall
            if no_imp > 5000:
                stall_count += 1
                if stall_count > 3:
                    break
                break

            p = p_new

    return best_p, best_score


def lp_refine(f, max_iters=150, deadline=None, verbose=True):
    """LP-based direction finding with binary line search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=5.0)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Binary line search: find best t in [0, 1]
        best_t = 0.0
        best_trial_score = best_score
        best_trial_f = f.copy()

        lo, hi = 0.0, 1.0
        for _ in range(10):
            mid = (lo + hi) / 2.0
            f_try = (1.0 - mid) * f + mid * g
            f_try = normalize(f_try)
            if f_try is None:
                hi = mid
                continue
            sc = 2.0 * n * float(np.convolve(f_try, f_try).max())
            if sc < best_trial_score - 1e-13:
                best_trial_score = sc
                best_trial_f = f_try.copy()
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_trial_score < best_score - 1e-12:
            best_f = best_trial_f.copy()
            best_score = best_trial_score
            f = best_trial_f.copy()
            M = float(np.convolve(f, f).max())
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

        if verbose and it % 30 == 0:
            print(f"      LP it{it}: score={best_score:.8f}, t={best_t:.3f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def generate_diverse_inits(n, rng):
    """Generate several diverse initial sequences."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    inits = []

    # Linear ramps
    for eta in [1.0, 1.5, 2.0, 2.5, 3.0]:
        a = np.maximum(1.0 + eta * u, 1e-10)
        inits.append(a / a.sum())

    # Gaussian bump
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        a = np.exp(-u**2 / (2 * sigma**2))
        inits.append(a / a.sum())

    # U-shape (high at ends)
    for alpha in [1.0, 2.0, 3.0]:
        a = 1.0 + alpha * u**2
        inits.append(a / a.sum())

    # Cosine window
    a = 0.5 * (1 + np.cos(np.pi * u))
    inits.append(np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum())

    # Random smooth
    for _ in range(5):
        base = rng.normal(0, 1, size=n)
        # Smooth with a Gaussian kernel
        kernel_size = max(3, n // 32 | 1)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * np.linspace(-3, 3, kernel_size)**2)
        kernel /= kernel.sum()
        a = np.convolve(base, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    return inits

            if sc1024 < best_score:
                best_score = sc1024
                best_seq = p1024
                print(f"  n=1024 result: {best_score:.8f}")
            else:
                print(f"  n=1024 not better: {sc1024:.8f}")

    # Phase 4: Final aggressive Adam on the best
    if time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        print(f"\nPhase 4: Final polish at n={n_final}, score={best_score:.8f}")
        # Use very small beta for precise optimization
        beta_final = 0.005 * float(np.convolve(best_seq, best_seq).max())
        p_final, sc_final = adam_optimize(best_seq, deadline,
                                           beta_init=beta_final, lr=0.015,
                                           max_restarts=5, verbose=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = p_final
            print(f"  Final result: {best_score:.8f}")

    # Validate
    final_score = internal_score(best_seq)
    print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.8f} ===")

    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        t = np.linspace(0, 1, 512)
        best_seq = np.maximum(1.0 + 1.5 * (2 * t - 1), 1e-10)
        best_seq = best_seq / best_seq.sum()
    best_seq = best_seq / float(best_seq.sum())

    return [float(x) for x in best_seq]
