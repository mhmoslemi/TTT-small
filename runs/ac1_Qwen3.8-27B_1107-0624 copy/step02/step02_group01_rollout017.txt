def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):

    if n == 0:

    if n <= 64:

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    m = n - 1
    alpha = 2.0 * k / m - 1.0 if m > 0 else np.zeros(n)
    w = iv(0, beta * np.sqrt(1 - alpha**2)) / iv(0, beta)

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def make_gaussian(n, sigma_frac):
    """Create a Gaussian-shaped profile."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac * 0.5
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return normalize(a)

def make_raised_cosine(n, width):
    """Raised cosine window (peaked at center)."""
    x = np.linspace(0, 1, n)
    # width controls how wide the peak is (1 = full width, 0.5 = half width)
    a = np.cos(np.pi * (x - 0.5) / width) ** 2
    a = np.where(np.abs(x - 0.5) < width, a, 0.0)
    return normalize(a)

def make_power_law(n, p):
    """a_j proportional to (j+1)^p * (n-j)^p (bell shape controlled by p)."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** p * (n - j + 1) ** p
    return normalize(a)

def make_parabola(n, eta):
    """a_j proportional to 1 + eta * (1 - 2*(j/(n-1) - 0.5))^2."""
    x = np.linspace(0, 1, n)
    u = 2.0 * x - 1.0
    a = 1.0 + eta * u ** 2
    return normalize(a)

def make_inverted_parabola(n, eta):
    """a_j proportional to 1 + eta * (1 - (2*(j/(n-1)) - 1)^2) = 1 + eta*4*x*(1-x)."""
    x = np.linspace(0, 1, n)
    a = 1.0 + eta * 4.0 * x * (1.0 - x)
    return normalize(a)

def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.1, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    return normalize(a)

def softmax_optimize(a0, deadline, rng, max_iter=50000, lr=0.05):
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    
    z = np.log(a) - np.log(a).mean()
    
    c = fft_convolve(a)
    M_val = float(c.max())
    beta = 0.1 * M_val
    
    best_a = a.copy()
    best_M = M_val
    no_imp = 0
    
    N_corr = next_pow2(3 * n)
    a_rev = a[::-1].copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute soft-max weights
        x = (c - M_val) / beta
        x = np.clip(x, -50.0, 5.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros(2*n-1)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * np.log(s_ex)
        
        # Compute h = cross-correlation of w with a
        a_rev = a[::-1]
        fa = np.fft.rfft(a_rev, N_corr)
        fw = np.fft.rfft(w, N_corr)
        conv = np.fft.irfft(fa * fw, N_corr)
        h = conv[n-1 : 2*n-1]
        
        # Gradient in log-space: grad_z = 2*h - 2*L*a (approximately)
        # For the mirror descent: a <- a * exp(-eta * (h - L) / norm)
        # Actually, the gradient of L w.r.t. a_m is 2*h_m
        # The "constraint" gradient (from sum(a)=1) is L (Lagrange multiplier)
        # So the effective gradient is 2*h - 2*L (constant)
        # For mirror descent: a_m <- a_m * exp(-eta * 2 * h_m)
        # Then renormalize
        
        grad = 2.0 * h
        # Normalize gradient for stable updates
        g_norm = float(np.sqrt(np.mean(grad**2)))
        if g_norm < 1e-15:
            g_norm = 1.0
        
        # Multiplicative update
        a = a * np.exp(-lr * grad / g_norm)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Update convolution
        c = fft_convolve(a)
        M_new = float(c.max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive beta
        if it % 100 == 0:
            M_val = M_new
            beta = max(1e-10, beta * 0.8)
        
        # Restart if stuck
        if no_imp > max(200, 2 * n):
            # Perturb in log-space
            z = np.log(best_a)
            z += 0.3 * rng.normal(size=n)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = fft_convolve(a)
            M_val = float(c.max())
            beta = 0.1 * M_val
            no_imp = 0
    
    return best_a, best_M

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def make_gaussian(n, sigma_frac):
    x = np.linspace(0, 1, n)
    sigma = sigma_frac * 0.5
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return normalize(a)


def make_raised_cosine(n, width):
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.where(d < width, np.cos(np.pi * d / width) ** 2, 0.0)
    return normalize(a)


def make_power_bell(n, p):
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** p * (n - j + 1) ** p
    return normalize(a)


def make_inverted_parabola(n, eta):
    x = np.linspace(0, 1, n)
    a = 1.0 + eta * 4.0 * x * (1.0 - x)
    return normalize(a)


def make_two_peak(n, center_sep, width):
    x = np.linspace(0, 1, n)
    c = center_sep
    a = (np.exp(-((x - 0.5 + c) / width) ** 2) + np.exp(-((x - 0.5 - c) / width) ** 2))
    return normalize(a)


def make_flat_center(n, frac):
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.where(d < frac, 1.5, 0.5)
    return normalize(a)


def make_smooth_random(n, rng, n_harm=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    return normalize(a)


def make_all_profiles(n, rng):
    profs = []
    for sf in (0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5, 0.6):
        profs.append(make_gaussian(n, sf))
    for w in (0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0):
        profs.append(make_raised_cosine(n, w))
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0):
        profs.append(make_power_bell(n, p))
    for eta in (1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        profs.append(make_inverted_parabola(n, eta))
    for cs in (0.1, 0.15, 0.2, 0.25):
        for w2 in (0.08, 0.12, 0.15, 0.2):
            profs.append(make_two_peak(n, cs, w2))
    for fr in (0.2, 0.3, 0.4, 0.5):
        profs.append(make_flat_center(n, fr))
    for _ in range(6):
        profs.append(make_smooth_random(n, rng))
    return profs


def solve_lp_direction(f, M, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5, 'output_flag': 0})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f0, deadline, rng, max_iter=100000, verbose=False):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, compute_score(f0)
    f = f / s
    n = f.size

    M = float(fft_convolve(f).max())
    best_f = f.copy()
    best_M = M
    no_imp = 0
    step_attempts = [0.5, 0.3, 0.7, 0.2, 0.4, 0.6, 0.1, 0.8]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                f = best_f + 0.02 * rng.normal(size=n)
                f = normalize(f)
                if f is None:
                    f = best_f.copy()
                    M = best_M
                    continue
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n))
                f = normalize(f)
                if f is None:
                    f = best_f.copy()
                    M = best_M
                    no_imp = 0
                    continue
                M = float(fft_convolve(f).max())
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                no_imp = 0
            continue

        g = g / g_sum

        improved = False
        for t in step_attempts:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = float(fft_convolve(f_new).max())
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-10
                f = normalize(f)
                if f is None:
                    f = best_f.copy()
                    M = best_M
                else:
                    M = float(fft_convolve(f).max())
                no_imp = 0

        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}")

    return best_f, best_M


def adam_softmax_polish(f0, deadline, rng, lr=0.05, max_iter=100000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, 0.1 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0

    N_corr = next_pow2(3 * n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        x = (c - M_val) / beta
        x = np.clip(x, -50.0, 5.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros(2 * n - 1)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        a_rev = f[::-1]
        fa = np.fft.rfft(a_rev, N_corr)
        fw = np.fft.rfft(w, N_corr)
        conv = np.fft.irfft(fa * fw, N_corr)
        h = conv[n - 1:2 * n - 1]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * grad_z
        v_adam = b2 * v_adam + (1.0 - b2) * (grad_z * grad_z)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 200 == 0:
            M_val = M_new
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(300, 2 * n):
            z = 0.8 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize(p0)
    if n0 < 2:
        return None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    return normalize(p1)


def thin_sequence(a, threshold_frac=0.01):
    """Set values below threshold_frac of max to zero, then renormalize."""
    a = a.copy()
    thresh = threshold_frac * a.max()
    a[a < thresh] = 0.0
    return normalize(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to use parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # === Phase 1: Diverse LP exploration at n=512 (400s) ===
    n1 = 512
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: LP exploration at n={n1}, budget={phase1_end - t_start:.0f}s")

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)

    all_profiles = make_all_profiles(n1, rng)
    # Pick a diverse subset
    import random
    random.seed(int(seed))
    selected = random.sample(range(len(all_profiles)), min(16, len(all_profiles)))
    for idx in selected:
        starts.append(all_profiles[idx])

    per_start = max(3.0, (phase1_end - t_start) / max(1, len(starts)))
    candidates = []

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = normalize(p0)
        if p0 is None:
            continue
        t_end = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = lp_optimize(p0, t_end, rng, max_iter=50000)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        candidates.append((sc, p_best))
        if i % 4 == 0:
            print(f"  P1 [{i+1}/{len(starts)}]: best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f} (n={best_seq.size if best_seq is not None else 0})")

    # === Phase 2: Focused LP on top candidates (500s) ===
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(5, len(candidates))
    per_cand = max(30.0, (phase2_end - t_start) / max(1, top_k))
    print(f"Phase 2: Focused LP, {top_k} candidates, budget={phase2_end - t_start:.0f}s")

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        t_end = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = lp_optimize(p0, t_end, rng, max_iter=100000)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"  P2 [{i+1}/{top_k}]: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # === Phase 3: Adam-softmax polish (300s) ===
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"Phase 3: Adam polish, budget={phase3_end - t_start:.0f}s")

    if best_seq is not None:
        # Polish the best at its native length
        n_best = best_seq.size
        t_end = min(phase3_end - 60, time.monotonic() + 150.0)
        p_pol, M_pol = adam_softmax_polish(best_seq, t_end, rng, lr=0.04)
        sc = compute_score(p_pol)
        try_update(p_pol, sc)
        print(f"  P3 polish n={n_best}: score={sc:.10f}")

    # Polish a couple more candidates
    for i in range(min(3, len(candidates))):
        if time.monotonic() > phase3_end - 30:
            break
        _, p0 = candidates[i]
        t_end = min(phase3_end, time.monotonic() + 80.0)
        p_pol, M_pol = adam_softmax_polish(p0, t_end, rng, lr=0.05)
        sc = compute_score(p_pol)
        try_update(p_pol, sc)

    # Try thinning
    if best_seq is not None:
        for th in (0.005, 0.01, 0.02, 0.03, 0.05):
            thinned = thin_sequence(best_seq, th)
            if thinned is not None:
                sc_th = compute_score(thinned)
                try_update(thinned, sc_th)

    print(f"After Phase 3: best={best_score:.10f}")

    # === Phase 4: Multi-length exploration (300s) ===
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"Phase 4: Multi-length, budget={phase4_end - t_start:.0f}s")

    if best_seq is not None:
        for n_try in (768, 1024, 384):
            if time.monotonic() > phase4_end - 60:
                break
            p_ext = interpolate_profile(best_seq, n_try)
            if p_ext is None:
                continue
            sc_ext = compute_score(p_ext)
            try_update(p_ext, sc_ext)

            t_end = min(phase4_end - 60, time.monotonic() + 100.0)
            p_best, M_best = lp_optimize(p_ext, t_end, rng, max_iter=50000)
            sc = compute_score(p_best)
            try_update(p_best, sc)
            print(f"  P4 n={n_try}: score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                t_end2 = min(phase4_end - 30, time.monotonic() + 50.0)
                p_pol, _ = adam_softmax_polish(p_best, t_end2, rng, lr=0.04)
                sc2 = compute_score(p_pol)
                try_update(p_pol, sc2)
                print(f"  P4 n={n_try} polished: score={sc2:.10f}")

    # Also try direct profiles at other lengths
    if time.monotonic() < phase4_end - 30:
        for n_try in (640, 768):
            if time.monotonic() > phase4_end - 30:
                break
            for prof in [make_gaussian(n_try, 0.3), make_raised_cosine(n_try, 0.6),
                         make_inverted_parabola(n_try, 3.0)]:
                if time.monotonic() > phase4_end - 30:
                    break
                t_end = min(phase4_end - 30, time.monotonic() + 40.0)
                p_best, _ = lp_optimize(prof, t_end, rng, max_iter=20000)
                sc = compute_score(p_best)
                try_update(p_best, sc)

    print(f"After Phase 4: best={best_score:.10f}")

    # === Phase 5: Final LP polish on absolute best (200s) ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        phase5_end = deadline - 10
        t_end = min(phase5_end, time.monotonic() + 150.0)
        n_b = best_seq.size
        print(f"Phase 5: Final LP polish at n={n_b}, budget={t_end - time.monotonic():.0f}s")
        p_final, M_final = lp_optimize(best_seq, t_end, rng, max_iter=100000, verbose=True)
        sc_final = compute_score(p_final)
        try_update(p_final, sc_final)

        # Try thinning the final
        for th in (0.005, 0.01, 0.02):
            thinned = thin_sequence(p_final, th)
            if thinned is not None:
                sc_th = compute_score(thinned)
                try_update(thinned, sc_th)

        # One more Adam polish
        if time.monotonic() < deadline - 20:
            t_end2 = min(deadline - 20, time.monotonic() + 50.0)
            p_pol2, _ = adam_softmax_polish(best_seq, t_end2, rng, lr=0.03)
            sc2 = compute_score(p_pol2)
            try_update(p_pol2, sc2)

    print(f"After Phase 5: best={best_score:.10f}")

    # Final
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_gaussian(512, 0.3)
    if fb is None:
        fb = np.ones(512) / 512.0
    return [float(x) for x in fb]
