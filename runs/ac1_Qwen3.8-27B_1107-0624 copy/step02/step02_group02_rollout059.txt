def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def cross_corr(w, a):
    """Compute h[m] = sum_j w[m+j] * a[j] for m=0..n-1."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def softmax_stable(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

    m = c.max()
    x = (c - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    val = m + beta * np.log(s)
    w = ex / s
    return val, w

def peak_reduction(a, k_star, alpha=0.1):
    """Reduce peak by redistributing mass away from peak-contributing positions."""
    n = len(a)
    c = full_convolve_fft(a)
    peak_val = c[k_star]
    
    # Compute contributions to the peak
    contrib = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]
    
    # Normalize contributions
    total = contrib.sum()
    if total < 1e-12:
        return a
    
    # Positions with high contribution to peak should be reduced
    # Positions with low contribution should be increased
    # Use a "temperature" to control the aggressiveness
    weights = contrib / total  # probability of each position contributing
    
    # Target: make a proportional to (1 - alpha * weights)
    a_target = a * (1 - alpha * weights / weights.max() if weights.max() > 0 else 1)
    a_target = np.maximum(a_target, 0)
    
    # Also boost positions that don't contribute to this peak
    peak_mask = np.zeros(n, dtype=bool)
    for i in range(n):
        if 0 <= k_star - i < n:
            peak_mask[i] = True
    
    # Positions NOT in the peak's support can be boosted
    non_peak = ~peak_mask
    if np.any(non_peak):
        a_target[non_peak] += alpha * a.sum() * non_peak.sum() / n * 0.1
    
    a_new = a_target / a_target.sum()
    return a_new

"""
Multi-resolution gradient-based optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses logit-space Adam with adaptive soft-max annealing, diverse parametric
initialization, direct peak-reduction steps, and multi-resolution interpolation.
"""
import numpy as np
import time


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def cross_corr_w_a(w, a):
    """Compute h[m] = sum_j w[m+j] * a[j] for m=0..n-1 via FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.max(z))
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s


def soft_max_with_weights(c, beta):
    """Compute soft max and weights."""
    m = float(np.max(c))
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s <= 0 or not np.isfinite(s):
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = m + beta * np.log(s)
    w = ex / s
    return val, w


def logit_adam_optimize(a0, deadline, rng, lr=0.06, beta_init=None,
                        b1=0.9, b2=0.999, restart_thresh=None,
                        peak_reduce_every=200, verbose=False):
    """Optimize in logit-space using Adam with soft-max annealing."""
    n = a0.size
    a = np.maximum(a0, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))

    c = full_convolve_fft(a)
    if beta_init is None:
        beta = max(1e-10, 0.15 * float(c.max()))
    else:
        beta = beta_init

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    best_z = z.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    base_lr = lr
    if restart_thresh is None:
        restart_thresh = max(400, 2 * n)

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        L, w = soft_max_with_weights(c, beta)
        grad_a = 2.0 * cross_corr_w_a(w, a)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        a = softmax_stable(z)

        c = full_convolve_fft(a)
        score = float(2.0 * n * c.max())

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        if t % 40 == 0:
            beta = max(1e-10, beta * 0.82)

        # Direct peak reduction every peak_reduce_every steps
        if peak_reduce_every > 0 and t % peak_reduce_every == 0 and no_imp > 50:
            c_full = full_convolve_fft(a)
            k_star = int(np.argmax(c_full))
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            cmax = float(contrib.max())
            if cmax > 0:
                # Reduce high-contribution positions, boost low ones
                alpha = 0.03
                reduction = contrib / float(contrib.sum())
                a_new = a * (1.0 - alpha * reduction / reduction.max())
                # Boost positions not in peak support
                peak_support = np.zeros(n, dtype=bool)
                for i in range(n):
                    if 0 <= k_star - i < n:
                        peak_support[i] = True
                non_peak = ~peak_support
                if np.any(non_peak):
                    boost = alpha * 0.3 * a.sum()
                    a_new[non_peak] += boost * (a[non_peak] / a[non_peak].sum())
                a_new = np.maximum(a_new, 1e-12)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    c_new = full_convolve_fft(a_new)
                    sc_new = float(2.0 * n * c_new.max())
                    if sc_new < best_score:
                        a = a_new
                        z = np.log(np.maximum(a, 1e-12))
                        z -= float(np.mean(z))
                    else:
                        a = best_a.copy()
                        z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            t = 0

        # Restart
        if no_imp > restart_thresh:
            z = 0.8 * best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            a = softmax_stable(z)
            c = full_convolve_fft(a)
            beta = max(1e-10, 0.2 * float(c.max()))
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = base_lr
            no_imp = 0

        # Decay learning rate slowly
        if no_imp > restart_thresh // 2:
            lr = max(0.005, lr * 0.995)

        if verbose and t % 200 == 0:
            print(f"    iter={t}, score={best_score:.10f}, beta={beta:.2e}, lr={lr:.4f}")

    return best_a, best_score


def make_profiles(n, rng):
    """Generate diverse parametric profiles."""
    profiles = []
    t_cont = np.linspace(0.0, 1.0, n)
    idx = np.arange(n, dtype=np.float64)

    # Linear ramps (various slopes)
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
        a = 1.0 + eta * (t_cont - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Power law (i+1)^p
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (idx + 1.0) ** p
        profiles.append(a / a.sum())

    # Reverse power law
    for p in [0.5, 1.0, 2.0]:
        a = (n - idx) ** p
        profiles.append(a / a.sum())

    # Gaussian bumps
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - center) / (sigma * n)) ** 2)
        profiles.append(a / a.sum())

    # Off-center Gaussian
    for center_frac in [0.3, 0.4, 0.6, 0.7]:
        center = center_frac * (n - 1)
        sigma = 0.3 * n
        a = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        profiles.append(a / a.sum())

    # Cosine windows
    for k in [1, 2, 3]:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * k * t_cont))
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Blackman-like
    a = 0.42 + 0.5 * np.cos(2 * np.pi * t_cont) + 0.08 * np.cos(4 * np.pi * t_cont)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())

    # Triangular
    a = np.minimum(idx + 1, n - idx)
    a = np.maximum(a, 1.0)
    profiles.append(a / a.sum())

    # Sqrt ramp
    a = np.sqrt(idx + 1.0)
    profiles.append(a / a.sum())

    # Log ramp
    a = np.log(idx + 2.0)
    profiles.append(a / a.sum())

    # Mixed smooth (low-frequency Fourier)
    for _ in range(8):
        coeffs = rng.normal(size=6) * 0.25
        a = np.ones(n)
        for i in range(1, len(coeffs)):
            a = a + coeffs[i] * np.cos(2.0 * np.pi * i * t_cont)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    return profiles


def interpolate_profile(a, n_new):
    """Linear interpolation of a profile to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == n_new:
        s = a.sum()
        return a / s if s > 1e-10 else a
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, a)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-10:
        result = result / s
    return result
