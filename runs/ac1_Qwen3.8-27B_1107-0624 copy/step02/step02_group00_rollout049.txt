"""
Optimized step function search for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive smooth-max L-BFGS with analytical gradients, multi-start, and LP refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, N=None):
    """FFT-based convolution of array with itself."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if N is None:
        N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Small negative values from FFT roundoff
    if c.min() < 0:
        c = c - c.min()
    return c


def _score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max(c, beta):
    """Compute smooth max and weights."""
    c_max = float(c.max())
    x = (c - c_max) / beta
    # Clip for numerical stability
    x = np.clip(x, -500, 500)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = c_max + beta * np.log(s)
    return val, w


def _grad_smooth_max_conv(a, beta):
    """
    Compute gradient of 2*n*soft_beta(conv(a,a)) w.r.t. a (not z).
    Returns (value, grad_a) where grad_a has same shape as a.
    """
    n = len(a)
    c = _fft_conv(a)
    val_sm, w = _smooth_max(c, beta)
    
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    # = 2 * cross_correlation(w, a)[m]
    # Compute via FFT:
    # sum_t a[t] * w[t+m] = (a * w_flipped)[len(w)-1-m]
    w_flipped = w[::-1]
    conv_len = n + len(w) - 1  # = n + 2n-2 = 3n-2
    N = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w_flipped, N)
    conv = np.fft.irfft(Fa * Fw, N)[:conv_len]
    # We need conv[len(w)-1-m] = conv[2n-2-m] for m=0,...,n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad_a = 2.0 * conv[n-1:2*n-1][::-1]
    
    val = 2.0 * n * val_sm  # since sum(a)=1, score = 2n*max(c)
    return val, grad_a


def _obj_grad_z(z, beta):
    """
    Objective and gradient in log-space z (where a = softmax(z)).
    """
    n = len(z)
    # Compute a = softmax(z)
    z_shifted = z - z.max()
    ez = np.exp(z_shifted)
    a = ez / ez.sum()
    
    # Compute smooth max value and gradient w.r.t. a
    val, grad_a = _grad_smooth_max_conv(a, beta)
    
    # Convert gradient from a-space to z-space
    # da[m]/dz[j] = a[m](delta_{mj} - a[j])
    # dz[j] = sum_m grad_a[m] * a[m] * (delta_{mj} - a[j])
    #        = a[j] * (grad_a[j] - sum_m grad_a[m]*a[m])
    dot = float(np.dot(grad_a, a))
    gz = a * (grad_a - dot)
    
    return val, gz


def _lbfgs_smooth_max(a0, beta_start, beta_end, max_iter, deadline, beta_decay=0.95):
    """
    L-BFGS optimization of smooth max with adaptive beta.
    Starts from a0 (on simplex), optimizes in log-space.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z = z - z.mean()
    
    best_a = a.copy()
    best_val = _score(a)
    
    beta = beta_start
    total_iter = 0
    stuck_count = 0
    last_best_iter = 0
    
    while time.monotonic() < deadline and total_iter < max_iter and beta > beta_end:
        # Define the objective and gradient
        def fun_and_grad(z_loc):
            val, grad = _obj_grad_z(z_loc, beta)
            if not np.all(np.isfinite(grad)):
                return val + 1e10, np.zeros_like(z_loc)
            return val, grad
        
        # L-BFGS optimization
        res = minimize(fun_and_grad, z, method='L-BFGS-B', 
                      options={'maxiter': 100, 'gtol': 1e-8, 'ftol': 1e-10})
        
        total_iter += 1
        z_new = res.x
        if not np.all(np.isfinite(z_new)):
            z_new = z
            stuck_count += 1
        else:
            z_new = z_new - z_new.mean()  # Center for numerical stability
            z_new = np.clip(z_new, -50, 50)
            ez = np.exp(z_new - z_new.max())
            a_new = ez / ez.sum()
            
            val_new = _score(a_new)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_a = a_new.copy()
                last_best_iter = total_iter
                stuck_count = 0
            else:
                stuck_count += 1
            
            z = z_new
        
        # Check for stuck
        if stuck_count > 30:
            # Perturb and restart
            z = z + 0.1 * np.random.default_rng(total_iter).standard_normal(n)
            z = z - z.mean()
            stuck_count = 0
        
        # Decay beta
        beta = beta_decay * beta
        
        if total_iter % 10 == 0:
            print(f"  L-BFGS: iter={total_iter}, beta={beta:.6f}, best_val={best_val:.10f}")
    
    return best_a


def _lp_spread(a, M, tol=1e-6):
    """LP: maximize sum(g) s.t. conv(a*g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: (a * g)[k] <= M for k=0,...,2n-2
    # (a * g)[k] = sum_j A[k,j] * g[j] where A[k,j] = a[j] if valid
    # Use the convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def _lp_refine_fast(a, deadline, max_refines=500):
    """Fast LP refinement with fewer, more targeted LP calls."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_refines):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(a)
        M = c.max()
        M_lp = M * 1.001  # Slight inflation
        
        g = _lp_spread(a, M_lp)
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 50:
                break
            continue
        
        g = g / sum_g
        # Blend
        blend = 0.3
        a_new = (1 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        a = a_new
        
        if it % 50 == 0:
            print(f"  LP: iter={it}, score={sc:.10f}")
    
    return best_a
