            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                blend_cur = max(0.03, blend_cur * 0.85)
                inflate_cur = min(1.003, inflate_cur * 0.92)
            if no_imp > max(500, 4 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
                # Simulated annealing: add small random perturbation
                rng = np.random.default_rng(it + n)
                pert = 0.05 * rng.normal(size=n)
                a = a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()

        a = a_new

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _adam_logit_polish(a, deadline, max_iter=30000, lr=0.08):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        _, w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        # Gradient in a-space
        vp = 2.0 * h + 1e-8 * a
        dot = float(np.dot(a, vp))
        # Gradient in logit space: dz_j = a_j * (vp_j - sum(a*vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, beta * 2.0)
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        # Cosine annealing for learning rate
        progress = min(1.0, it / max(1, max_iter))
        lr_cur = base_lr * (0.5 + 0.5 * np.cos(np.pi * progress))
        lr_cur = max(lr_cur, base_lr * 0.1)

        z = z - lr_cur * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

        if no_imp > max(2000, 4 * n):
            # Restart with perturbation
            rng = np.random.default_rng(10000 + it)
            z = 0.9 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-6, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp = 0

    return best_a


def _gradient_polish(a, deadline, max_iter=20000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    lr = 0.015
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = np.maximum(a + momentum * v, 0)
        a_nest = a_nest / a_nest.sum()
        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.001, lr * 0.93)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.7)

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, alpha=1.0):
    t = np.linspace(0.01, 1.0, n)
    a = t ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Multi-length, multi-profile search
    lengths = [512, 256]
    if parent is not None:
        # Put parent length first
        if parent.size in (256, 512, 1024):
            lengths = [parent.size] + [l for l in lengths if l != parent.size]

    print(f"Attempting lengths: {lengths}")
    print(f"Deadline: {deadline - t_start:.1f}s remaining")

    # Phase 1: Quick screening of profiles at multiple lengths
    for n_t in lengths:
        if time.monotonic() > deadline - 200:
            break

        profiles = []
        # Linear profiles
        for eta in [0.8, 1.0, 1.2, 1.5, 2.0]:
            profiles.append(_make_linear_profile(n_t, eta))
        # Power profiles
        for alpha in [0.3, 0.5, 0.8, 1.2, 1.5]:
            profiles.append(_make_power_profile(n_t, alpha))
        # Cosine profiles
        for eta in [0.8, 1.2, 1.5]:
            profiles.append(_make_cosine_profile(n_t, eta))
        # Quadratic profiles
        for eta in [0.5, 1.0, 1.5]:
            profiles.append(_make_quadratic_profile(n_t, eta))

        # Add parent-interpolated
        if parent is not None:
            profiles.append(_interpolate_profile(parent, n_t))
            # Perturbed parent
            for _ in range(2):
                pp = _interpolate_profile(parent, n_t)
                pp = pp * (1.0 + 0.15 * rng.normal(size=n_t))
                pp = np.maximum(pp, 0)
                pp = pp / pp.sum()
                profiles.append(pp)

        # Quick LP screening: 20s per profile
        per_profile_budget = 20.0
        phase1_deadline = min(deadline - 200, t_start + (time.monotonic() - t_start) + per_profile_budget * len(profiles))

        for pi, a0 in enumerate(profiles):
            if time.monotonic() > phase1_deadline:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick LP pass
            lp_end = min(time.monotonic() + per_profile_budget, phase1_deadline)
            a_lp = _lp_refine(a0, lp_end, max_iter=3000, inflate=1.001, blend=0.3)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()

            if pi % 5 == 0:
                print(f"  Phase1 n={n_t} prof[{pi}]: init={sc0:.8f} -> LP={sc_lp:.8f} (best={best_score:.8f})")

        print(f"Phase 1 done for n={n_t}, best so far: {best_score:.10f}")

    # Phase 2: Extended LP + gradient on best candidate
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = best_seq.size
        a_work = best_seq.copy()

        # Extended LP
        lp_deadline = min(time.monotonic() + 200, deadline - 120)
        print(f"Phase 2: Extended LP, n={n_best}, start={best_score:.10f}, budget={lp_deadline - time.monotonic():.0f}s")
        a_lp2 = _lp_refine(a_work, lp_deadline, max_iter=20000, inflate=1.0005, blend=0.2)
        sc_lp2 = _score(a_lp2)
        if np.isfinite(sc_lp2) and sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()
        print(f"  Extended LP: {sc_lp2:.10f}")

        # Gradient polish
        if time.monotonic() < deadline - 60:
            gp_deadline = min(time.monotonic() + 60, deadline - 60)
            a_gp = _gradient_polish(a_lp2, gp_deadline, max_iter=10000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  Gradient polish: {sc_gp:.10f}")

    # Phase 3: Adam logit polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        ad_deadline = min(time.monotonic() + 250, deadline - 60)
        print(f"Phase 3: Adam logit polish, budget={ad_deadline - time.monotonic():.0f}s")
        a_ad = _adam_logit_polish(best_seq, ad_deadline, max_iter=30000, lr=0.06)
        sc_ad = _score(a_ad)
        if np.isfinite(sc_ad) and sc_ad < best_score:
            best_score = sc_ad
            best_seq = a_ad.copy()
        print(f"  Adam result: {sc_ad:.10f}")

    # Phase 4: Final tight LP + gradient sprint
    if best_seq is not None and time.monotonic() < deadline - 20:
        # Tight LP
        lp3_deadline = min(time.monotonic() + 100, deadline - 20)
        print(f"Phase 4: Final tight LP, budget={lp3_deadline - time.monotonic():.0f}s")
        a_final = _lp_refine(best_seq, lp3_deadline, max_iter=15000, inflate=1.0003, blend=0.15)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final LP: {sc_final:.10f}")

        # Final gradient
        if time.monotonic() < deadline - 5:
            gp3_deadline = min(time.monotonic() + 50, deadline - 5)
            a_fgp = _gradient_polish(best_seq, gp3_deadline, max_iter=15000)
            sc_fgp = _score(a_fgp)
            if np.isfinite(sc_fgp) and sc_fgp < best_score:
                best_score = sc_fgp
                best_seq = a_fgp.copy()
            print(f"  Final gradient: {sc_fgp:.10f}")

    # Phase 5: Another Adam pass if time allows
    if best_seq is not None and time.monotonic() < deadline - 10:
        ad2_deadline = min(time.monotonic() + 80, deadline - 5)
        a_ad2 = _adam_logit_polish(best_seq, ad2_deadline, max_iter=20000, lr=0.03)
        sc_ad2 = _score(a_ad2)
        if np.isfinite(sc_ad2) and sc_ad2 < best_score:
            best_score = sc_ad2
            best_seq = a_ad2.copy()
        print(f"Phase 5: Final Adam: {sc_ad2:.10f}")

    # Final cleanup
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    # Validate with evaluate_sequence
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final_seq]
