   import time

       while k < n:

           return np.array([])

           return c

       if not np.isfinite(mval):

   def simplex_project(v):
       """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
       n = len(v)
       if n == 0:
           return v

       theta = cssv[rho] / (rho + 1.0)
       return np.maximum(v - theta, 0.0)

def next_pow2(n):

def hankel_matvec(p, a):
    """Compute Q @ a where Q(i,j) = p[i+j]. p has length 2n-1, a has length n.
    Uses stride tricks for efficient vectorized computation."""
    n = len(a)

        return np.array([p[0] * a[0]])
    # Sliding window of p: windows[i] = p[i:i+n]
    windows = as_strided(p, shape=(n, n), strides=(p.strides[0], p.strides[0]))
    return windows @ a


def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    idx = np.arange(1, n + 1, dtype=np.float64)
    positive = u - cssv / idx > 0
    if not np.any(positive):
        return np.ones(n) / n
    rho = np.max(np.where(positive)[0])
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    a_new = np.exp(z)

    if s < 1e-30:

    a_new /= s
    return a_new

        if no_imp > 10000:
            a = best_a + 0.02 * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            a_prev = a.copy()
            no_imp = 0

    return best_a, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def make_varied_starts(n, rng):
    """Create diverse starting points."""
    starts = []
    # Uniform
    starts.append(np.ones(n) / n)
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0):
        t = np.linspace(0, 1, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    # Quadratic / bell shapes
    t = np.linspace(0, 1, n)
    for center, width in ((0.5, 0.3), (0.3, 0.4), (0.7, 0.4), (0.5, 0.2), (0.4, 0.25)):
        a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
        starts.append(a / a.sum())
    # Sinusoidal modulation
    for freq, amp in ((1, 0.3), (2, 0.4), (3, 0.5)):
        a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    # Random smooth
    for _ in range(4):
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * t + rng.uniform(0, 2 * np.pi))
        a += 0.3 * np.sin(2 * np.pi * rng.uniform(3.0, 6.0) * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimization:
    1. Saddle-point method at n=512 from diverse starts
    2. Subgradient polishing with top-K averaging
    3. Accelerated mirror descent
    4. Interpolation to n=1024 with refinement
    5. Final exact evaluation
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    # Phase 1: Saddle-point optimization at n=512
    n1 = 512
    phase1_end = min(deadline, t_start + 480.0)

    # Build starts
    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
            sc = compute_score(p_int)
            try_update(p_int, sc)
            print(f"Parent score: {sc:.10f}, n={parent.size}")

    starts.extend(make_varied_starts(n1, rng)[:10])

    candidates = []
    per_start = max(10.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = saddle_point_optimize(p0, n1, per, rng, max_iter=2000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))

        if i % 3 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  Phase 1 start {i}/{len(starts)}: best={best_score:.10f} ({elapsed:.0f}s)")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1 (SP): best={best_score:.10f}")

    # Phase 2: Subgradient polishing on top candidates
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(4, len(candidates))
    per_cand = max(20.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = subgradient_optimize(p0, n1, per, rng, max_iter=200000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        # Add to candidates if improved
        if sc < candidates[0][0]:
            candidates.insert(0, (sc, p_best))
        print(f"  Phase 2 cand {i}: score={sc:.10f}, overall best={best_score:.10f}")

    print(f"After Phase 2 (SG): best={best_score:.10f}")

    # Phase 3: Accelerated mirror descent on the best
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 200.0)
        p_best, M_best = accelerated_mirror_descent(best_seq, n1, per, rng, max_iter=100000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 3 (AMDD): score={sc:.10f}, best={best_score:.10f}")

    # Also run accelerated on second best
    if len(candidates) > 1:
        per = min(phase3_end, time.monotonic() + 100.0)
        p_best, M_best = accelerated_mirror_descent(candidates[1][1], n1, per, rng, max_iter=80000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Saddle-point again on the current best (with different params)
    phase4_end = min(deadline, t_start + 1440.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase4_end, time.monotonic() + 150.0)
        p_best, M_best = saddle_point_optimize(best_seq, n1, per, rng, max_iter=3000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 4 (SP2): score={sc:.10f}, best={best_score:.10f}")

    # Phase 5: Interpolate to n=1024 and refine
    n2 = 1024
    phase5_end = min(deadline, t_start + 1680.0)
    if best_seq is not None and time.monotonic() < phase5_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            sc_ext = compute_score(p_ext)
            try_update(p_ext, sc_ext)
            print(f"  Interpolated to n={n2}: score={sc_ext:.10f}")

            # Saddle point at n=1024
            per = min(phase5_end, time.monotonic() + 100.0)
            p_best, M_best = saddle_point_optimize(p_ext, n2, per, rng, max_iter=1000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)

            # Subgradient at n=1024
            if time.monotonic() < phase5_end - 30:
                per2 = min(phase5_end, time.monotonic() + 80.0)
                p_best2, M_best2 = subgradient_optimize(p_best, n2, per2, rng, max_iter=100000)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"  Phase 5 (n={n2}): score={sc2:.10f}, best={best_score:.10f}")

    # Phase 6: Final subgradient polish on absolute best with remaining time
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 120.0)
        p_best, M_best = subgradient_optimize(best_seq, n_best, per, rng, max_iter=300000)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"  Phase 6 (final SG): score={sc:.10f}, best={best_score:.10f}, n={n_best}")

    # Try also a fresh saddle-point at n=256 and see if it helps
    if time.monotonic() < deadline - 120:
        n3 = 256
        p0 = np.ones(n3) / n3
        per = min(deadline - 60, time.monotonic() + 60.0)
        p_best, M_best = saddle_point_optimize(p0, n3, per, rng, max_iter=3000)
        sc = 2.0 * n3 * M_best
        try_update(p_best, sc)

        if sc < best_score * 0.999:
            # Good enough to interpolate up
            p_ext = interpolate_profile(p_best, n1)
            if p_ext is not None:
                sc_ext = compute_score(p_ext)
                per2 = min(deadline - 30, time.monotonic() + 40.0)
                p_best2, M_best2 = subgradient_optimize(p_ext, n1, per2, rng, max_iter=100000)
                sc2 = 2.0 * n1 * M_best2
                try_update(p_best2, sc2)
        print(f"  n={n3} attempt: best now={best_score:.10f}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        # Evaluate with the actual evaluator
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
                        best_seq = final
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = np.ones(512) / 512
    return [float(x) for x in fb]
