import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

"""
Optimized step function search combining LP spreading, peak-targeted gradient
descent, Fourier-domain diversification, and logit-space Adam optimization.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _conv_gradient(a):
    """Compute gradient of max smooth convolution w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    # Use top-k weighting for sharper focus on peaks
    k = max(1, min(n, int(0.05 * (2*n-1))))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    # Also add smooth-max weight
    _, sw = _smooth_max_weight(c, max(1e-6, 0.02 * mval))
    w = 0.5 * w + 0.5 * sw
    w = w / w.sum()
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, mval

    c = _fft_convolve(a)

        M = float(c.max())

        if sum_g <= 1.0 + 1e-8:

                blend_cur = blend

def _gradient_polish(a, deadline, max_iter=20000, lr=0.015, momentum=0.9):

    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    beta = max(1e-8, 0.05 * float(c.max()))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _conv_gradient(a_nest)

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            beta = max(beta, 0.1 * float(c.max()))
            continue

        v = momentum * v - lr * grad / (np.linalg.norm(grad) + 1e-12)
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_geometric(n, ratio=0.95):
    """Geometric sequence (asymmetric)."""
    a = ratio ** np.arange(n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_sine_modulated(n, freq=1.0, asym=0.3):
    """Sine-modulated profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + asym * np.sin(2 * np.pi * freq * t)
    a = 1.0 + asym * (t - 0.5)  # linear asymmetry
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_bump(n, pos1=0.3, pos2=0.7, width=0.15):
    """Two Gaussian bumps."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - pos1) / width) ** 2) + np.exp(-((t - pos2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _fourier_flatten(a, strength=0.3):
    """Apply Fourier-domain flattening to reduce spectral peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    mag = np.abs(fa)
    # Target: make magnitudes more uniform
    # Use a soft compression: mag_new = mag / (1 + strength * mag^2)^0.5
    phase = np.angle(fa)
    mag_compressed = mag / np.sqrt(1.0 + strength * mag * mag)
    fa_new = mag_compressed * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a / a.sum()
    return a_new / s
