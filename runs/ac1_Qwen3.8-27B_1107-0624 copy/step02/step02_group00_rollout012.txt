"""
Optimized step function search: sparse LP + Polyak subgradient + logit Adam.
Key innovations: sparse LP constraints, tighter inflation, U-shaped inits,
Polyak step size using theoretical lower bound, adaptive parameter scheduling.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _conv_matrix_sparse(a):
    """Build sparse convolution matrix A where (A*b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # Column j: a[j] at rows j, j+1, ..., j+n-1
        r = np.arange(j, j + n)
        rows.extend(r)
        cols.extend([j] * n)
        vals.extend([a[j]] * n)
    A = sparse.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsc()


def _smooth_max_and_grad(a, beta):
    """Returns (smooth_max_value, weights, gradient)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    # = 2 * (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_r = np.fft.rfft(a_rev, N)
    b_full = np.fft.irfft(fw * fa_r, N)
    b = b_full[n - 1:n - 1 + n]
    grad = 2.0 * b
    return mval, w, grad


def _lp_spread(a, M, A_sparse):
    """Solve LP: max sum(b) s.t. (a*b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_sparse, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def _subgradient_optimize(a, deadline, max_iter=100000):
    """Subgradient method with Polyak step size."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    c = _fft_convolve(a)
    current_val = float(c.max())
    best_val = current_val
    best_a = a.copy()

    # Lower bound: sum(c) = (sum(a))^2 = 1, so max(c) >= 1/(2n-1)
    lower_bound = 1.0 / (2.0 * n - 1.0)

    beta = max(1e-10, 0.03 * current_val)
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    step_count = 0
    best_step = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        mval, w, grad = _smooth_max_and_grad(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(beta * 2.0, 1e-6)
            continue

        # Polyak step size
        gap = mval - lower_bound
        gn_sq = float(np.dot(grad, grad))
        if gn_sq > 1e-20:
            lr = gap / gn_sq
            lr = min(lr, 0.3)
        else:
            lr = 0.01

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        val = float(c.max())

        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            no_imp = 0
            best_step = step_count
        else:
            no_imp += 1

        if it % 80 == 0:
            beta = max(1e-12, beta * 0.75)

        if no_imp > max(500, 2 * n):
            a = best_a.copy()
            v = np.zeros(n)
            beta = max(1e-12, 0.03 * best_val)
            momentum = min(0.95, momentum + 0.02)
            no_imp = 0

        step_count += 1

    return best_a


def _adam_logit_polish(a, deadline, max_iter=50000, lr=0.04):
    """Adam in logit space for final polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))
    best_z = z.copy()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    best_val = float(c.max())
    best_a = a.copy()
    beta = max(1e-10, 0.03 * best_val)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        mval, w, grad = _smooth_max_and_grad(a, beta)

        if not np.all(np.isfinite(grad)):
            continue

        # Natural gradient: project grad to tangent space of simplex
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        val = float(c.max())

        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.003, lr * 0.97)

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(2000, 4 * n):
            rng_local = np.random.default_rng((it * 7919 + n) % (2**31))
            z = 0.7 * best_z + 0.5 * rng_local.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-10, 0.03 * best_val)
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabolic_profile(n, curvature=2.0):
    """U-shaped: high at edges, low at center."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + curvature * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_shape_profile(n, depth=1.5):
    """W-shaped profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: sparse LP spreading + Polyak subgradient + logit Adam.
    Tries multiple lengths and diverse initializations (linear, parabolic, U-shaped, W-shaped).
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    try_lengths = [384, 448, 512, 576, 640, 768, 1024]
    if parent is not None:
        pn = parent.size
        try_lengths.append(pn)
        # Sort by distance to parent length
        try_lengths.sort(key=lambda x: abs(x - pn))
    else:
        try_lengths.sort()

    # Remove duplicates, keep top 5
    seen = set()
    unique_lengths = []
    for ln in try_lengths:
        if ln not in seen and 128 <= ln <= 2048:
            seen.add(ln)
            unique_lengths.append(ln)
    try_lengths = unique_lengths[:5]

    print(f"Trying lengths: {try_lengths}")

    # Phase 1: For each length, try diverse inits + LP + subgradient
    for n_t in try_lengths:
        if time.monotonic() > deadline - 150:
            break

        # Generate initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
            # Perturbed parent
            for scale in [0.1, 0.2, 0.3]:
                pert = _interpolate_profile(parent, n_t) * (1.0 + scale * rng.normal(size=n_t))
                pert = np.maximum(pert, 0.0)
                ps = pert.sum()
                if ps > 1e-10:
                    inits.append(pert / ps)
        # Structured profiles
        inits.append(_make_linear_profile(n_t, 1.5))
        inits.append(_make_linear_profile(n_t, 2.5))
        inits.append(_make_parabolic_profile(n_t, 2.0))
        inits.append(_make_parabolic_profile(n_t, 3.0))
        inits.append(_make_cubic_profile(n_t, 2.0, -1.5))
        inits.append(_make_w_shape_profile(n_t, 1.5))

        for a0 in inits:
            if time.monotonic() > deadline - 100:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # LP refinement
            remaining = deadline - time.monotonic() - 80
            if remaining < 30:
                break
            lp_budget = min(180.0, remaining * 0.6)
            lp_end = time.monotonic() + lp_budget
            a_lp = _lp_refine(a0, lp_end, max_iter=30000,
                              inflate_init=1.0005, blend_init=0.25)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()

            # Subgradient polish
            remaining = deadline - time.monotonic() - 60
            if remaining < 15:
                break
            sg_budget = min(80.0, remaining * 0.5)
            sg_end = time.monotonic() + sg_budget
            a_sg = _subgradient_optimize(a_lp, sg_end, max_iter=50000)
            sc_sg = _score(a_sg)
            if np.isfinite(sc_sg) and sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()

            print(f"  n={n_t}: init={sc0:.8f} -> LP={sc_lp:.10f} -> SG={sc_sg:.10f}")

    # Phase 2: Adam polish on the best found
    if best_seq is not None and time.monotonic() < deadline - 30:
        ad_budget = min(150.0, deadline - time.monotonic() - 20)
        if ad_budget > 20:
            ad_end = time.monotonic() + ad_budget
            print(f"Phase 2: Adam polish, budget={ad_budget:.0f}s")
            a_ad = _adam_logit_polish(best_seq, ad_end, max_iter=50000, lr=0.04)
            sc_ad = _score(a_ad)
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Adam: {sc_ad:.10f}")

    # Phase 3: One more LP pass on the very best (with tighter params)
    if best_seq is not None and time.monotonic() < deadline - 15:
        lp_budget = min(80.0, deadline - time.monotonic() - 10)
        if lp_budget > 10:
            lp_end = time.monotonic() + lp_budget
            a_final = _lp_refine(best_seq, lp_end, max_iter=20000,
                                 inflate_init=1.0003, blend_init=0.15)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"Phase 3: Final LP: {sc_final:.10f}")

    # Phase 4: Quick subgradient on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 5:
        sg_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 3)
        a_sgf = _subgradient_optimize(best_seq, sg_end, max_iter=30000)
        sc_sgf = _score(a_sgf)
        if np.isfinite(sc_sgf) and sc_sgf < best_score:
            best_score = sc_sgf
            best_seq = a_sgf.copy()
        print(f"Phase 4: Final SG: {sc_sgf:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(512, 1.5)
        s = float(final_seq.sum())
    # Normalize to get good score (the score is scale-invariant, but ensure valid range)
    final_seq = final_seq / s

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
