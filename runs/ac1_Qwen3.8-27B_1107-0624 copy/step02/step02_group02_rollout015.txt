"""
Optimization of step function heights to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses combination of LP-based descent, coordinate descent, and soft-max Adam polish.
Key improvements: aggressive step-size scheduling, coordinate descent fine-tuning,
and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def fft_convolve(a):
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2*n-1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:2*n-1]
    return np.maximum(c, 0)

def lp_descent(f, M, deadline, rng, verbose=False):
    """LP-based descent: find g maximizing sum(g) s.t. conv(f,g)<=M, g>=0."""
    n = len(f)
    # Build convolution matrix A where A@g = conv(f,g)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    t = 0.1
    best_M = M
    best_f = f.copy()
    stagnant = 0
    
    while time.monotonic() < deadline:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'max_iter': 10000})
        except Exception:
            break
        
        if not res.success or res.x is None or res.x.sum() < 1e-10:
            break
        
        g = np.maximum(res.x, 0)
        g = g / g.sum()
        
        # Line search for step size
        best_t = 0
        best_M_new = best_M
        for trial_t in [t, t*0.5, t*0.25, t*2]:
            if trial_t < 0.001 or trial_t > 1:
                continue
            f_new = (1-trial_t)*best_f + trial_t*g
            f_new = np.maximum(f_new, 0)
            if f_new.sum() < 1e-10:
                continue
            f_new = f_new / f_new.sum()
            c_new = fft_convolve(f_new)
            M_new = float(c_new.max())
            if M_new < best_M_new:
                best_M_new = M_new
                best_t = trial_t
        
        if best_t > 0:
            f_new = (1-best_t)*best_f + best_t*g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            best_M = best_M_new
            best_f = f_new
            stagnant = 0
            # Rebuild A with new f
            A = np.zeros((2*n-1, n))
            for j in range(n):
                A[j:j+n, j] = best_f
        else:
            stagnant += 1
            t *= 0.5
            if t < 0.001:
                # Random restart
                noise = 0.05 * rng.normal(size=n)
                f_try = best_f + noise
                f_try = np.maximum(f_try, 0)
                if f_try.sum() > 1e-10:
                    f_try = f_try / f_try.sum()
                    M_try = float(fft_convolve(f_try).max())
                    if M_try < best_M:
                        best_f = f_try
                        best_M = M_try
                        A = np.zeros((2*n-1, n))
                        for j in range(n):
                            A[j:j+n, j] = best_f
                t = 0.1
                stagnant = 0
        
        if verbose and stagnant % 50 == 0:
            print(f"    LP: M={best_M:.10f}, t={t:.5f}, stagnant={stagnant}")
    
    return best_f, best_M

def coordinate_descent(a, deadline, rng, step_sizes=None):
    """Coordinate descent to reduce peak autocorrelation."""
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    c = fft_convolve(a)
    best_peak = float(c.max())
    best_a = a.copy()
    
    if step_sizes is None:
        step_sizes = [0.1, 0.05, 0.01, 0.005, 0.001]
    
    iteration = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        improved = False
        # Randomize the order for better results
        idx = rng.permutation(n)
        
        for i in idx:
            if time.monotonic() > deadline:
                break
            
            base_val = a[i]
            for step in step_sizes:
                # Try increasing
                a[i] = base_val + step * a.mean()
                a[i] = max(0, a[i])
                c = fft_convolve(a)
                peak = float(c.max())
                if peak < best_peak - 1e-12:
                    best_peak = peak
                    best_a = a.copy()
                    improved = True
                    break
                # Try decreasing
                a[i] = max(0, base_val - step * a.mean())
                c = fft_convolve(a)
                peak = float(c.max())
                if peak < best_peak - 1e-12:
                    best_peak = peak
                    best_a = a.copy()
                    improved = True
                    break
                a[i] = base_val
            
            a = best_a  # Reset to best
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Add noise and try again
                a = best_a + 0.01 * a.std() * rng.normal(size=n)
                a = np.maximum(a, 0)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                else:
                    a = best_a.copy()
                no_improve = 0
        else:
            no_improve = 0
            a = best_a.copy()
        
        iteration += 1
        if verbose and iteration % 10 == 0:
            print(f"    CD iter {iteration}: peak={best_peak:.10f}")
    
    return best_a, best_peak

    block_frac = 0.3  # fraction of positions to perturb
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Perturb a random subset
        n_perturb = max(1, int(n * block_frac))
        idx = rng.choice(n, size=n_perturb, replace=False)
        
        f_new = f.copy()
        f_new[idx] += sigma * f.mean() * rng.standard_normal(n_perturb)
        f_new = normalize(f_new)
        
        c_new = fft_conv(f_new)
        M_new = float(c_new.max())
        
        if M_new < M - 1e-14:
            f = f_new
            M = M_new
            if M < best_M:
                best_M = M
                best_f = f.copy()
            no_improve = 0
            sigma = min(sigma * 1.01, 0.2)  # Slightly increase
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                sigma *= 0.95
            if no_improve > 500:
                # Reset from best
                f = best_f + sigma * 0.5 * rng.standard_normal(n)
                f = normalize(f)
                c = fft_conv(f)
                M = float(c.max())
                no_improve = 0
                sigma = max(sigma * 0.8, 0.001)
        
        if verbose and (no_improve % 1000 == 0 or best_M < float('inf')):
            pass  # too verbose
    
    return best_f, best_M

def coord_refine(a, deadline, rng, verbose=False):
    """Single-coordinate refinement."""
    n = len(a)
    a = normalize(a.copy())
    c = fft_conv(a)
    best_M = float(c.max())
    best_a = a.copy()
    
    scales = [0.5, 0.2, 0.1, 0.05, 0.01, 0.005]
    no_improve = 0
    iters = 0
    
    while time.monotonic() < deadline:
        improved = False
        for i in rng.permutation(n):
            if time.monotonic() > deadline:
                break
            base = a[i]
            for s in scales:
                for sign in [1, -1]:
                    a[i] = max(0, base + sign * s * best_a.mean())
                    c = fft_conv(a)
                    M = float(c.max())
                    if M < best_M - 1e-14:
                        best_M = M
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        a = best_a.copy()
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                a = best_a + 0.005 * best_a.std() * rng.standard_normal(n)
                a = normalize(a)
                c = fft_conv(a)
                if float(c.max()) < best_M:
                    best_M = float(c.max())
                    best_a = a.copy()
                no_improve = 0
        iters += 1
    
    return best_a, best_M

def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0, 1, n)
    
    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for sign in [1, -1]:
            a = 1.0 + sign * eta * (t - 0.5)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    
    # Quadratic
    for a_coef in [-1, -0.5, 0, 0.5, 1]:
        for b_coef in [-1, -0.5, 0, 0.5, 1]:
            a = 1.0 + a_coef * (t - 0.5) + b_coef * (t - 0.5)**2
            a = np.maximum(a, 1e-12)
            if a.sum() > 0.1:
                profiles.append(a / a.sum())
    
    # Cosine
    for k in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2*np.pi*k*t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1, 2, 3, 5]:
        for sign in [1, -1]:
            a = np.exp(sign * rate * (t - 0.5))
            profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(10):
        coeffs = rng.uniform(-1, 1, 6)
        a = np.ones(n)
        for k in range(1, 6):
            a += coeffs[k] * np.cos(2*np.pi*k*t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-12)
        if a.sum() > 0.1:
            profiles.append(a / a.sum())
    
    return profiles

def interpolate_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

Wait, I realize there's an issue with the scaling. The objective is scale-invariant:
score = 2n * max(conv(a,a)) / sum(a)^2

If we scale a by c: conv(ca, ca) = c^2 * conv(a,a), sum(ca) = c * sum(a)
score = 2n * c^2 * max(conv(a,a)) / (c^2 * sum(a)^2) = same!

So the scaling doesn't matter. I should just return the sequence as-is (normalized to sum=1 or any positive value).

Let me also reconsider the GS phase. The issue with my current implementation is that the sigma (noise level) needs to be tuned carefully. If it's too large, we'll rarely find improvements. If it's too small, we'll make very slow progress.

A better approach: use the "1/5 rule" from evolution strategies. If the improvement rate is about 1/5 (20%), the sigma is about right. Otherwise, increase or decrease.

Let me also add a more sophisticated local search: instead of perturbing a random subset, I'll perturb the positions that are most "important" (i.e., positions where the convolution peak is achieved).

Actually, let me think about this more carefully. The convolution peak is at some position k*. The value b[k*] = sum_i a[i]*a[k*-i]. To reduce this, we need to reduce the products a[i]*a[k*-i] for the i that contribute most. This suggests we should perturb a[i] for i near 0 and i near k* (since those are the pairs that contribute to b[k*]).

1. LP for coarse optimization

Let me also consider: what if the optimal solution has a specific structure? For instance, what if it's a "two-bump" structure (large at both ends, small in the middle)? Or "three-bump"?

For a two-bump structure: a is large at positions near 0 and near n-1, small in the middle.
- b[0] = a[0]^2 (large)
- b[n-1] = sum a[i]*a[n-1-i] (the "cross" terms: large*small + small*small + small*large)
- b[2n-2] = a[n-1]^2 (large)

So the peaks would be at the edges, not the center. The center would be relatively small.

For a uniform sequence, the peak is at the center with value 1/n.
For a two-bump sequence, the peak is at the edges.

The question is: can we make all peaks (including edges) less than 1/n?

For the edge peak b[0] = a[0]^2. To have a[0]^2 < 1/n, we need a[0] < 1/sqrt(n). With sum(a) = 1 and n elements, the average is 1/n. So a[0] < 1/sqrt(n) means a[0] < sqrt(n) * (average). For n=512, sqrt(n) ≈ 22.6, so a[0] < 22.6 * (1/512) ≈ 0.044. The average is 1/512 ≈ 0.002, so a[0] < 0.044 is a[0] < 22.6 * average. That's quite a large element relative to the average.

If we set a[0] = a[n-1] = 0.04 and distribute the rest uniformly:
Remaining sum = 1 - 0.08 = 0.92, over n-2 elements, each ≈ 0.92/510 ≈ 0.0018.

b[0] = 0.04^2 = 0.0016
b[n-1] ≈ 0.04 * 0.0018 * (n-2) + ... ≈ 0.04 * 0.92 = 0.0368 (rough estimate)

Hmm, that's much larger than 1/n = 0.002. So the center is still the problem.

Actually, let me compute b[n-1] more carefully. b[n-1] = sum_{i=0}^{n-1} a[i]*a[n-1-i].

If a[0] = a[n-1] = 0.04, and a[i] = c for 1 <= i <= n-2 where c = 0.92/(n-2):

b[n-1] = a[0]*a[n-1] + a[n-1]*a[0] + sum_{i=1}^{n-2} a[i]*a[n-1-i]
= 2*0.04^2 + sum_{i=1}^{n-2} c * c  (since a[n-1-i] = c for i in [1, n-2])
= 0.0032 + (n-2)*c^2
= 0.0032 + 510 * (0.92/510)^2
= 0.0032 + 510 * 0.00180^2
= 0.0032 + 510 * 0.00000325
= 0.0032 + 0.00166
= 0.00486

1/n = 1/512 ≈ 0.00195

So b[n-1] ≈ 0.00486 > 1/n. The center is still the peak!

And b[0] = 0.0016 < 1/n. So the center dominates.
