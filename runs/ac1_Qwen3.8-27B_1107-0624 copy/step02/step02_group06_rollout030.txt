```python
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines projected gradient (FISTA) in the simplex, multi-frequency oscillation
search, and coordinate descent polishing to find near-flat autocorrelation sequences.

import time

# --- Core utility functions ---

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def soft_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        w[mask] = 1.0 / max(1, float(np.count_nonzero(mask)))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def autocorr_gradient(w, a):
    """Compute the gradient of sum_k w_k * c_k w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    # c_k = sum_j a[j] * a[k-j] (with bounds)
    # dc_k/da_i = a[k-i] + a[i-k] (with a[j]=0 for j out of [0,n-1])
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    # corr_weights: h[i] = sum_k w[k] * a[k-i] (correlation, not convolution)
    # This is convolve(w, a_reversed) appropriately indexed
    fa = np.fft.rfft(a, N)
    # We need: grad_i = sum_k w_k * (a[k-i] + a[i-k])
    # = sum_k w_k * a[k-i] + sum_k w_k * a[i-k]
    # First term: conv(w, a)[i] (with appropriate indexing)
    # Actually, let's think in terms of correlation.
    # sum_k w[k] * a[k-i]: this is (w * a_flipped)[i+n-1] where a_flipped = a[::-1]
    # Or equivalently, it's the cross-correlation of w and a.
    
    # Let's use FFT: 
    # term1[i] = sum_k w[k] * a[k-i] for valid k,i
    # This is the "cross-correlation" of w and a.
    # In FFT terms: ifft(fft(w) * conj(fft(a))) gives cross-correlation
    # But we need to be careful with the indexing.
    
    # Alternative: just compute it directly using the convolution structure
    # grad_i = sum_k w[k]*(a[(k-i) mod n] if in bounds + a[(i-k) mod n] if in bounds)
    # But this is circular; we need linear.
    
    # Let's use the direct FFT approach:
    # h1[i] = sum_k w[k] * a[k-i] where both are 0-padded to length 2n-1
    # This is the convolution of w with a reversed: conv(w, a[::-1])
    # Then h1[i] = conv(w, a_rev)[i + n - 1]
    
    a_rev = a[::-1]
    h1 = np.fft.irfft(fw * np.fft.rfft(a_rev, N), N)
    # h1 has length N, we want indices [n-1, 2n-2] for the "valid" part
    h1_valid = h1[n-1:2*n-1]  # length n
    
    # h2[i] = sum_k w[k] * a[i-k] = sum_k w[k-n+1+k] * ... 
    # Actually h2[i] = h1[n-1-i] by symmetry (since a[i-k] = a_rev[n-1-(i-k)] = a_rev[n-1-i+k])
    # Hmm, let me just compute it directly.
    # h2[i] = sum_k w[k] * a[i-k] where a[i-k] is valid when 0 <= i-k < n
    # = sum_{k=max(0,i-n+1)}^{min(i, n-1)} w[k] * a[i-k]
    # This is conv(w, a)[i] (the "valid" part starting from index 0)
    # In full convolution: conv(w, a)[i] for i in [0, n-1]
    
    h2 = np.fft.irfft(fw * fa, N)
    h2_valid = h2[0:n]  # length n
    
    grad = h1_valid + h2_valid
    return grad

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s

def project_simplex(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)

def fista_optimize(a0, deadline, beta0=0.1, beta_min=1e-6, beta_decay=0.95,
                   beta_period=50, L_factor=1.5, use_momentum=True):
    """
    FISTA (accelerated proximal gradient) on the simplex.
    Minimizes f(a) = soft_max(conv(a,a), beta) with projected gradient steps.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size
    
    a = a0.copy()
    y = a0.copy()
    t = 1.0
    beta = beta0
    
    c = full_convolve_fft(y)
    maxc = float(c.max())
    beta = max(beta_min, min(beta0, 0.15 * maxc))
    
    # Estimate Lipschitz constant L
    # The Hessian of the smooth part is bounded by ~2 * max(w) * 2 (from the quadratic structure)
    # A safe upper bound: L = 2 * n (since sum of all a is 1, each a[i] <= 1)
    # More tightly: L = 2 * max(w) * 2 * max(a) ... let's just use 2*n as a safe bound
    L = 2.0 * n
    step = 1.0 / L
    
    best_a = a0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    max_iter = 50000
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        cY = full_convolve_fft(y)
        _, w = soft_max_and_weights(cY, beta)
        gy = autocorr_gradient(w, y)
        
        # Proximal gradient step
        a_new = project_simplex(y - step * gy)
        
        # Update momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = float(y.sum())
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = a_new.copy()
        a = a_new
        t = t_new
        
        # Evaluate
        cA = full_convolve_fft(a)
        maxcA = float(cA.max())
        score = float(2.0 * n * maxcA)
        
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta periodically
        if it > 0 and it % beta_period == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Backtracking if no improvement for too long
        if no_imp > 500:
            L = max(1.0, L * 0.7)
            step = 1.0 / L
            no_imp = 0
        
    return best_a, best_score


def logit_adam_optimize(a0, deadline, lr=0.08, beta_frac=0.15, beta_decay=0.85,
                       beta_period=60, b1=0.9, b2=0.999, eps=1e-8):
    """
    Adam optimization in logit space with soft-max surrogate.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size
    
    z = np.log(a0 + 1e-15)
    z -= float(np.mean(z))
    
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    a = a0.copy()
    c = full_convolve_fft(a)
    maxc = float(c.max())
    beta = max(1e-8, beta_frac * maxc)
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        _, w = soft_max_and_weights(c, beta)
        ga = autocorr_gradient(w, a)
        
        # Gradient in logit space: grad_z = a * (grad_a - sum(a * grad_a))
        dot = float(np.dot(a, ga))
        gz = a * (ga - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        a = softmax_from_z(z)
        
        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        if adam_t % beta_period == 0:
            beta = max(1e-10, beta * beta_decay)
        
        if no_imp > max(400, 3 * n):
            # Restart with perturbation
            z = 0.7 * best_z + 0.5 * np.random.default_rng(adam_t).standard_normal(n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            maxc = float(c.max())
            beta = max(1e-8, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

def coordinate_descent_polish(a0, deadline, n_sweeps=None, step_frac=0.08):

    best_score = internal_score(a0)
    no_imp = 0

               1.0 + 2.0*step_frac, 1.0 - 0.3*step_frac]
    
    for sweep in range(n_sweeps):
```

def next_pow2(n):

            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        a = softmax_from_z(z)

        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        if adam_t % beta_period == 0:
            beta = max(1e-10, beta * beta_decay)

        if no_imp > max(400, 3 * n):
            z = restart_keep * best_z + restart_perturb * rng.standard_normal(n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            maxc = float(c.max())
            beta = max(1e-8, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def fista_optimize(a0, deadline, beta0=0.1, beta_min=1e-8, beta_decay=0.93,
                   beta_period=40, L_init_factor=2.0):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size

    a = a0.copy()
    y = a0.copy()
    t_k = 1.0

    c = full_convolve_fft(a)
    maxc = float(c.max())
    beta = max(beta_min, min(beta0, 0.2 * maxc))

    L = L_init_factor * n
    step = 1.0 / L

    best_a = a0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    max_iter = 100000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        cY = full_convolve_fft(y)
        _, w = soft_max_and_weights(cY, beta)
        gy = autocorr_gradient(w, y)

        a_new = project_simplex(y - step * gy)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = float(y.sum())
        if sy > 1e-30:
            y = y / sy
        else:
            y = a_new.copy()
        a = a_new
        t_k = t_new

        cA = full_convolve_fft(a)
        maxcA = float(cA.max())
        score = float(2.0 * n * maxcA)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it > 0 and it % beta_period == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_imp > 800:
            L = max(1.0, L * 0.7)
            step = 1.0 / L
            no_imp = 0

    return best_a, best_score


def oscillation_refine(a0, deadline, max_freq=25, n_phases=48, amps=None):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    if amps is None:
        amps = (0.005, 0.01, 0.02, 0.04, 0.06, 0.1, 0.15, 0.2, 0.3)

    best_a = a0.copy()
    best_score = internal_score(a0)

    idx = np.arange(n)

    for freq in range(1, max_freq + 1):
        if time.monotonic() > deadline:
            break
        for phase_i in range(n_phases):
            if time.monotonic() > deadline:
                break
            phase = 2.0 * np.pi * phase_i / n_phases
            os = np.sin(2.0 * np.pi * freq * idx / n + phase)
            for eps in amps:
                a_new = a0 + eps * os
                if np.min(a_new) < 1e-10:
                    continue
                a_new = a_new / a_new.sum()
                sc = internal_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()

    return best_a, best_score


def multi_sine_refine(a0, deadline, num_sines=3, max_freq=15, n_phases=12, amps=None):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    if amps is None:
        amps = (0.02, 0.05, 0.1, 0.15)

    best_a = a0.copy()
    best_score = internal_score(a0)
    idx = np.arange(n)
    rng = np.random.default_rng(777)

    max_trials = 2000
    trials = 0
    while time.monotonic() < deadline and trials < max_trials:
        trials += 1
        pert = np.zeros(n)
        for _ in range(num_sines):
            f = rng.integers(1, max_freq + 1)
            ph = rng.uniform(0, 2.0 * np.pi)
            am = rng.uniform(0.02, 0.15)
            pert += am * np.sin(2.0 * np.pi * f * idx / n + ph)

        for eps in amps:
            a_new = a0 + eps * pert
            if np.min(a_new) < 1e-10:
                break
            a_new = a_new / a_new.sum()
            sc = internal_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    return best_a, best_score


def targeted_redistribution(a0, deadline, n_iters=500, step_size=0.1):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size

    best_a = a0.copy()
    best_score = internal_score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = full_convolve_fft(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc / (a.sum() ** 2)

        # Find top few lags
        top_k = np.argsort(c)[::-1][:5]
        k_star = int(top_k[0])

        # Gradient of c[k_star] w.r.t. a[i] = a[k_star - i] + a[i - k_star] (with bounds)
        contrib = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            contrib[i] = val

        # Move mass from high-contrib to low-contrib positions
        mean_c = float(np.mean(contrib))
        high_mask = contrib > 1.5 * mean_c
        low_mask = contrib < 0.5 * mean_c

        if not np.any(high_mask) or not np.any(low_mask):
            continue

        total_high = float(a[high_mask].sum())
        total_low = float(a[low_mask].sum())

        if total_high < 1e-15:
            continue

        # Redistribute: reduce high positions, increase low positions proportionally
        move = step_size * total_high

        a_new = a.copy()
        a_new[high_mask] = a_new[high_mask] * (1.0 - step_size)
        if total_low > 1e-15:
            low_weights = a[low_mask] / total_low
            a_new[low_mask] = a[low_mask] + move * low_weights

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc_new = internal_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            step_size = max(0.01, step_size * 0.95)

    return best_a, best_score


def coordinate_descent_polish(a0, deadline, n_sweeps=None, step_frac=0.06):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size

    if n_sweeps is None:
        n_sweeps = max(2, int((deadline - time.monotonic()) * 2))

    rng = np.random.default_rng(99999)

    best_a = a.copy()
    best_score = internal_score(a)
    no_imp = 0

    factors = [1.0 + step_frac, 1.0 - step_frac,
               1.0 + 0.5 * step_frac, 1.0 - 0.5 * step_frac,
               1.0 + 2.0 * step_frac, 1.0 - 0.3 * step_frac,
               1.0 + 0.2 * step_frac, 1.0 - 0.2 * step_frac]

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        sweep_improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            best_val = orig
            best_sc = best_score
            for f in factors:
                new_val = orig * f
                if new_val < 1e-15:
                    continue
                a[i] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[i] = orig
                    continue
                a = a / s
                sc = internal_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            a[i] = best_val
            a = a / a.sum()
            if best_sc < best_score - 1e-15:
                best_score = best_sc
                best_a = a.copy()
                sweep_improved = True
                no_imp = 0
            else:
                no_imp += 1

        if sweep_improved:
            step_frac = max(0.015, step_frac * 0.93)
        if no_imp > 4 * n:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 31 - 1))

    best_seq = None
    best_score = np.inf
    hall_of_fame = []

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print("  NEW BEST: {:.10f} n={} {}".format(best_score, best_seq.size, label))
        if len(hall_of_fame) < 30:
            hall_of_fame.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            hall_of_fame.sort()
        elif float(sc) < hall_of_fame[-1][0]:
            hall_of_fame[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
            hall_of_fame.sort()

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        parent = None

    print("Phase 0: Initialization")
    if parent is not None:
        sc = internal_score(parent)
        update_best(parent, sc, "parent_n{}".format(parent.size))
        for tn in (256, 384, 512, 768):
            if parent.size != tn:
                p_i = try_interpolate(parent, tn)
                if p_i is not None:
                    sc_i = internal_score(p_i)
                    update_best(p_i, sc_i, "parent_interp_n{}".format(tn))

    n_main = 512
    if parent is not None and parent.size in (256, 384, 768, 1024):
        n_main = parent.size

    # Generate diverse initial profiles at main length
    profiles = make_profiles(n_main, rng)
    for i, p0 in enumerate(profiles):
        sc = internal_score(p0)
        update_best(p0, sc, "init_{}".format(i))

    print("  After init: best={:.10f}".format(best_score))

    # Phase 1: Logit Adam optimization from top candidates at n_main
    print("Phase 1: Logit Adam at n={}".format(n_main))
    p1_end = min(deadline, t_start + 350.0)
    sorted_hof = sorted(hall_of_fame, key=lambda x: x[0])
    n_starts = min(15, len(sorted_hof))
    for i in range(n_starts):
        if time.monotonic() > p1_end:
            break
        sc_c, p_c = sorted_hof[i]
        if p_c.size != n_main:
            p_c = try_interpolate(p_c, n_main)
            if p_c is None:
                continue
        per = min(p1_end, time.monotonic() + 15.0)
        p_best, sc = logit_adam_optimize(p_c, per, lr=0.08, beta_frac=0.15)
        sc_full = internal_score(p_best)
        update_best(p_best, sc_full, "p1_{}".format(i))

    print("  After phase 1: best={:.10f}".format(best_score))

    # Phase 2: FISTA optimization
    print("Phase 2: FISTA at n={}".format(n_main))
    p2_end = min(deadline, t_start + 550.0)
