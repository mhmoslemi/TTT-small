```python
import numpy as np
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_conv_gradient(s, k):
    """Compute gradient of conv[k] with respect to s.
    Result[j] = 2*s[k-j] if 0 <= k-j < n, else 0."""
    n = len(s)
    j = np.arange(n)
    idx = k - j
    grad = np.zeros(n)
    valid = (idx >= 0) & (idx < n)
    grad[valid] = 2.0 * s[idx[valid]]
    return grad

def softmax_weighted_gradient(s, T=0.5):
    """Compute gradient using softmax weighting over lags."""
    n = len(s)
    conv = np.convolve(s, s)
    # Softmax weights
    w = np.exp(conv / (T * conv.max() + 1e-15))
    w /= w.sum()
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] > 1e-10:
            grad += w[k] * compute_conv_gradient(s, k)
    return grad

    """Optimize sequence using projected gradient descent."""
    n = len(s0)
    s = s0.copy()
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    
    eta = 0.1  # step size
    T = 1.0  # softmax temperature
    momentum = 0.5
    v = np.zeros(n)  # momentum buffer
    no_improve = 0
    patience = 200
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        
        if method == 'soft':
            max_conv = conv.max()
            T = max_conv * 0.1  # adaptive temperature
            w = np.exp((conv - max_conv) / (T + 1e-15))
            w /= w.sum()
            # Compute weighted gradient using FFT or direct
            grad = np.zeros(n)
            for k in range(len(conv)):
                if w[k] > 1e-8:
                    j = np.arange(n)
                    idx = k - j
                    valid = (idx >= 0) & (idx < n)
                    grad[valid] += w[k] * 2.0 * s[idx[valid]]
        else:  # hard max

            grad = compute_conv_gradient(s, k_star)
```

```python
def compute_soft_gradient(s, w):
    n = len(s)
    L = len(w)  # = 2n-1
    # For each j, compute sum_k w[k] * s[k-j] where 0 <= k-j < n
    # = sum_i w[i+j] * s[i] where 0 <= i < n and 0 <= i+j < L
    # This is the (n-point) cross-correlation of w and s at lag -j
    # Or equivalently, the convolution of w_flipped and s at a specific index
    
    # Use FFT for efficiency
    N_fft = 1
    while N_fft < n + L:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    # cross_corr[j] = sum_i w[i+j] * s[i]
    # This is the same as: (w reversed) convolved with s, at a specific index
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, N_fft)
    cross = np.fft.irfft(S * W_rev, N_fft)
    # cross[m] = sum_i s[i] * w_rev[m-i] = sum_i s[i] * w[L-1-(m-i)] = sum_i s[i] * w[L-1-m+i]
    # We want: sum_i s[i] * w[i+j]
    # So L-1-m+i = i+j => m = L-1-j
    # Thus cross[L-1-j] = sum_i s[i] * w[i+j]
    grad = 2.0 * cross[L-1-np.arange(n)]
    return grad
```

```python
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_soft_gradient_fft(s, w):
    """Compute gradient of sum_k w[k]*conv[k] with respect to s."""
    n = len(s)
    L = len(w)  # should be 2n-1
    N_fft = 1
    while N_fft < n + L:
        N_fft *= 2
    S = np.fft.rfft(s, N_fft)
    W_rev = np.fft.rfft(w[::-1], N_fft)
    cross = np.fft.irfft(S * W_rev, N_fft)
    grad = 2.0 * cross[L - 1 - np.arange(n)]
    return grad

def geometric_sequence(n, r):
    s = np.power(r, np.arange(n))
    return s / s.sum()

def optimize_soft(s0, deadline, rng):
    """Optimize using soft (softmax-weighted) gradient descent."""
    n = len(s0)
    s = s0.copy()
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_score = fast_score(s)
    
    eta = 0.05
    momentum_coeff = 0.7
    v = np.zeros(n)
    no_improve = 0
    patience = 300
    T_scale = 0.05  # temperature scale
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Softmax weights
        T = T_scale * max_conv
        w = np.exp((conv - max_conv) / T)
        w /= w.sum()
        
        # Compute gradient
        grad = compute_soft_gradient_fft(s, w)
        
        # Momentum
        v = momentum_coeff * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm * np.linalg.norm(grad)
        
        # Step
        s_new = s - eta * grad / (np.max(np.abs(grad)) + 1e-15)
        s_new = np.maximum(s_new, 0)
        s_sum = s_new.sum()
        if s_sum < 1e-15:
            eta *= 0.5
            continue
        s_new = s_new / s_sum
        
        score = fast_score(s_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_s = s_new.copy()
            no_improve = 0
            s = s_new
        else:
            no_improve += 1
            if no_improve > patience:
                # Try smaller step or perturbation
                eta *= 0.7
                no_improve = 0
                if eta < 0.005:
                    # Perturb
                    s = best_s + 0.05 * rng.standard_normal(n)
                    s = np.maximum(s, 0.001)
                    s = s / s.sum()
                    eta = 0.05
                    v = np.zeros(n)
                    no_improve = 0
                    T_scale = rng.uniform(0.02, 0.2)
        
    return best_s, best_score
```
