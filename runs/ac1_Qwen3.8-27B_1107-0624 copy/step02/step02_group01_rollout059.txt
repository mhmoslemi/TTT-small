def _fft_convolve(a):

    if n == 0:
        return np.array([])
    if n <= 32:
        return np.maximum(np.convolve(a, a), 0.0)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

def _cross_corr(w, a):
    """h[j] = sum_k w[k] * a[k-j] for j = 0..n-1"""
    n = a.size
    N = 1
    while N < len(w) + n - 1:
        N <<= 1

def _fft_convolve(a):
