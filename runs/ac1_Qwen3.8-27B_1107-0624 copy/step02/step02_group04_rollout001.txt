"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logit-space Adam with soft-max focusing on the max convolution point,
multi-restart strategy, and adaptive beta annealing.
Starts from uniform sequences and the parent, then refines via gradient descent.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _softmax_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _correlation(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _logit_optimize(p0, deadline, rng, lr=0.06, beta_frac=0.12, beta_decay=0.72,
                    no_imp_limit=400, reg=1e-8):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / float(p0.sum())
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _convolve_fft(p)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _softmax_weights(c, beta)
        gp = 2.0 * _correlation(w, p)
        gp += reg * p

        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            no_imp += 100
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)

        score = float(2.0 * n * max_c)

            best_score = score

            best_z = z.copy()
            no_imp = 0

            no_imp += 1

            c = _convolve_fft(p)
            max_c = float(c.max())
            beta = max(1e-10, beta_frac * max_c)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            no_imp = 0

    return best_p, best_score


def _coordinate_descent(p0, deadline, rng, n_restarts=5):
    """Coordinate descent: randomly perturb a block of coordinates and keep if improves."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / float(p0.sum())
    n = p0.size

    best_p = p0.copy()
    best_score = float(2.0 * n * _convolve_fft(p0).max())
    p = best_p.copy()

    block_sizes = [4, 8, 16, 32, 64]

    while time.monotonic() < deadline:
        bs = block_sizes[int(rng.integers(0, len(block_sizes)))]
        bs = min(bs, n // 2)
        start = int(rng.integers(0, n - bs + 1))
        end = start + bs

        scale = 0.01 + 0.02 * rng.random()
        p_new = p.copy()
        p_new[start:end] += scale * rng.normal(size=bs)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / float(p_new.sum())

        c_new = _convolve_fft(p_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            p = p_new
        elif rng.random() < 0.1:
            p = p_new

    return best_p, best_score


def _make_polynomial_profile(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * t**k
    a = np.maximum(a, 1e-10)
    a = a / float(a.sum())
    return a
