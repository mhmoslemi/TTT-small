```python
def fast_score(s):

    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)
```

```python
def solve_1d_subproblem(si, S_other, si2_conv, M_other, conv_affected, s_all, n, i):
    """
    Solve the 1D subproblem for coordinate i.
    
    Parameters:
    - si: current value of s[i]
    - S_other: sum of all other elements
    - si2_conv: conv[2*i] (the value at the quadratic position)
    - M_other: max of unaffected conv entries
    - conv_affected: conv[i:i+n] (the affected conv values)
    - s_all: the full s array
    - n: sequence length
    - i: coordinate index
    
    Returns: optimal value of a = s[i]
    """
    # Linear terms: for j=0..n-1, j≠i:
    #   c_j = conv[i+j] - 2*si*s[j]
    #   d_j = 2*s[j]
    # Quadratic: q = conv[2*i] - si^2
    
    q = si2_conv - si*si
    
    # Candidate a values
    candidates = [0.0]
    
    # From quadratic: a = sqrt(S_other^2 + q) - S_other
    disc = S_other*S_other + q
    if disc >= 0:
        a_quad = np.sqrt(disc) - S_other
        if a_quad > 0:
            candidates.append(a_quad)
    
    # From each linear constraint j≠i:
    # a_j = S_other - 2*c_j/d_j (if d_j > 0)
    for j in range(n):
        if j == i:
            continue
        d_j = 2.0 * s_all[j]
        if d_j > 1e-15:
            c_j = conv_affected[j] - 2.0 * si * s_all[j]
            a_j = S_other - 2.0 * c_j / d_j
            if a_j > 0:
                candidates.append(a_j)
    
    # Evaluate f at all candidates and pick the best
    best_a = 0.0
    best_val = np.inf
    
    for a in candidates:
        # Compute N(a) = max(M_other, max_j (c_j + d_j*a), q + a^2)
        # Linear terms
        vals = (conv_affected - 2.0*si*s_all) + 2.0*a*s_all
        vals[i] = q + a*a  # quadratic term
        N_a = max(M_other, vals.max())
        denom = (S_other + a)**2
        if denom < 1e-30:
            continue
        val = 2.0 * n * N_a / denom
        if val < best_val:
            best_val = val
            best_a = a
    
    # Also check a slightly above 0 and at a few other points for safety
    # (in case the true minimum is between candidates)
    return best_a
```

```python
Coordinate descent with analytical 1D subproblem solving for minimizing
2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Uses vectorized numpy operations and multiple sequence lengths/initializations.
"""

import numpy as np
from scipy.optimize import minimize_scalar, linprog
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def update_conv_after_change(conv, s, i, old_val, new_val, n):
    """Update conv in-place after changing s[i] from old_val to new_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return
    k_start = i
    k_end = i + n  # exclusive
    # For j = 0, 1, ..., n-1: k = i + j
    # If j != i: conv[k] += 2 * delta * s[j]
    # If j == i: conv[k] += 2 * delta * old_val + delta^2  (i.e., (old+delta)^2 - old^2)
    # Vectorized:
    conv[k_start:k_end] += 2.0 * delta * s[:n]
    # Fix the quadratic term at k = 2*i (which is at index i in [k_start, k_end))
    if k_start <= 2 * i < k_end:
        # We added 2*delta*s[i] = 2*delta*old_val, but need delta^2 more
        conv[2 * i] += delta * delta

def _compute_M_other(conv, i, n):
    """Compute max of conv entries not affected by coordinate i."""
    # Affected: [i, i+n-1]
```

```python
"""
Coordinate descent with analytical 1D subproblem solving for minimizing
2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Uses vectorized numpy operations, multiple sequence lengths, diverse initialization,
and LP-based refinement. Key insight: the 1D subproblem for each coordinate can be
solved by evaluating a small set of critical points (from derivative=0 conditions
and grid points), making each sweep very fast (~8ms for n=384).
"""

import numpy as np
from scipy.optimize import minimize_scalar, linprog
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def _update_conv(conv, s, i, old_val, new_val, n):
    """Update conv in-place after changing s[i] from old_val to new_val. s[i] must still be old_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return
    conv[i:i + n] += 2.0 * delta * s[:n]
    if 0 <= 2 * i <= 2 * n - 2:
        conv[2 * i] += delta * delta


def _compute_m_other(conv, i, n):
    """Max of conv entries unaffected by coordinate i (affected: [i, i+n-1])."""
    m = -np.inf
    if i > 0:
        m = max(m, conv[:i].max())
    end = i + n
    if end <= 2 * n - 1:
        m = max(m, conv[end:].max())
    if m == -np.inf:
        return 0.0
    return m


def coord_descent_optimize(s, deadline, rng, verbose=False):
    """
    Run coordinate descent to minimize the score.
    Returns (best_s, best_score).
    """
    n = len(s)
    s = np.maximum(s.copy(), 0.0)
    S = s.sum()
    if S < 1e-15:
        s = np.ones(n) / n
        S = 1.0

    conv = np.convolve(s, s)
    best_s = s.copy()
    best_score = fast_score(s)

    sweep = 0
    no_improve = 0
    patience = 4
    max_patience = 12

    while time.monotonic() < deadline:
        sweep += 1
        S = s.sum()

        for i in range(n):
            si = s[i]
            S_other = S - si
            M_other = _compute_m_other(conv, i, n)

            # Quadratic coefficient: q = conv[2*i] - si^2
            quad_pos = 2 * i
            if 0 <= quad_pos <= 2 * n - 2:
                q = conv[quad_pos] - si * si
            else:
                q = 0.0

            # Affected conv values
            conv_aff = conv[i:i + n]

            # Linear coefficients: c_j = conv[i+j] - 2*si*s[j], d_j = 2*s[j]
            d = 2.0 * s[:n]
            c = conv_aff - 2.0 * si * s[:n]

            # Build candidate set for optimal a
            candidates_list = [0.0]

            # From quadratic: a = sqrt(S^2 + q) - S
            disc = S_other * S_other + q
            if disc >= 0:
                a_quad = float(np.sqrt(disc)) - S_other
                if a_quad > 1e-10:
                    candidates_list.append(a_quad)

            # From linear constraints: a_j = S_other - 2*c_j/d_j
            mask = d > 1e-12
            if np.any(mask):
                a_lin = S_other - 2.0 * c[mask] / d[mask]
                a_lin = a_lin[(a_lin > 0) & (a_lin < 1e6)]
                if a_lin.size > 0:
                    # Include min, max, and a few quantiles
                    candidates_list.append(float(a_lin.min()))
                    candidates_list.append(float(a_lin.max()))
                    if a_lin.size > 1:
                        q25 = float(np.percentile(a_lin, 25))
                        q50 = float(np.percentile(a_lin, 50))
                        q75 = float(np.percentile(a_lin, 75))
                        candidates_list.extend([q25, q50, q75])
                    # Add a few random ones
                    if a_lin.size > 5:
                        sample_idx = rng.choice(a_lin.size, size=min(5, a_lin.size), replace=False)
                        candidates_list.extend(a_lin[sample_idx].tolist())

            # Grid points
            upper_bound = max(S_other * 1.5, 2.0 * si + 0.1, 0.1)
            grid = np.linspace(0, upper_bound, 15)
            candidates_list.extend(grid.tolist())

            candidates = np.array(candidates_list)
            candidates = np.clip(candidates, 0.0, upper_bound * 2.0)
            # Remove duplicates approximately
            candidates = np.unique(np.round(candidates, 10))
            if candidates.size == 0:
                candidates = np.array([0.0, upper_bound])

            # Limit candidates for performance
            if candidates.size > 600:
                idx = np.linspace(0, candidates.size - 1, 600).astype(int)
                candidates = candidates[idx]

            m_cand = candidates.size

            # Evaluate f(a) = 2n * max(M_other, max_j(c_j + d_j*a), q + a^2) / (S_other + a)^2
            # Vectorized: (m_cand, n) matrix
            lin_vals = c[None, :] + d[None, :] * candidates[:, None]
            lin_max = lin_vals.max(axis=1)
            quad_vals = q + candidates ** 2
            N_vals = np.maximum(M_other, np.maximum(lin_max, quad_vals))
            denom = (S_other + candidates) ** 2
            scores = 2.0 * n * N_vals / (denom + 1e-30)

            best_idx = scores.argmin()
            a_new = float(candidates[best_idx])

            if abs(a_new - si) > 1e-12:
                _update_conv(conv, s, i, si, a_new, n)
                s[i] = a_new
                S = S_other + a_new

        # End of sweep
        if sweep % 50 == 0:
            conv = np.convolve(s, s)  # Recorrect any numerical drift

        new_score = fast_score(s)
        if new_score < best_score - 1e-10:
            best_score = new_score
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                if no_improve >= max_patience:
                    break
                # Perturbation to escape local minimum
                n_pert = max(3, n // 15)
                idx = rng.choice(n, size=n_pert, replace=False)
                s[idx] = s[idx] + rng.uniform(0.02, 0.15, size=n_pert) * (s[idx] + 0.01)
                # Also try zeroing some elements
                if rng.random() < 0.3:
                    z_idx = rng.choice(n, size=max(2, n // 25), replace=False)
                    s[z_idx] *= 0.1
                s = np.maximum(s, 0.0)
                S = s.sum()
                if S < 1e-10:
                    s = best_s.copy()
                    S = s.sum()
                conv = np.convolve(s, s)
                no_improve = 0

        if verbose and sweep % 20 == 0:
            print(f"    Sweep {sweep}: best_score={best_score:.8f}")

    return best_s, best_score


def lp_refine(s, deadline, rng):
    """
    Use LP to find a direction that reduces peak autocorrelation.
    Blend with current sequence.
    """
    n = len(s)
    if n > 600:
        return s, fast_score(s)

    conv = np.convolve(s, s)
    M = conv.max()
    S = s.sum()

    # LP: maximize sum(b) s.t. (s*b)_k <= M, b >= 0
    # (s*b)_k = sum_j s[j] * b[k-j] for valid j
    # A matrix: A[k, j] = s[j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 3.0, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            b_sum = b.sum()
            if b_sum > 1e-10:
                b = b / b_sum
                # Try blending
                best_s = s.copy()
                best_sc = fast_score(s)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    cand = (1 - t) * s + t * b
                    cand = np.maximum(cand, 0.0)
                    cs = fast_score(cand)
                    if cs < best_sc:
                        best_sc = cs
                        best_s = cand.copy()
    except Exception:
        pass

    return best_s, fast_score(best_s)


def generate_initializations(n, rng, hseq=None):
    """Generate diverse initializations of length n. Returns list of (score, sequence)."""
    results = []

    # Uniform
    s = np.ones(n) / n
    results.append(s.copy())

    # Cosine windows
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.linspace(0, 1, n)
        s = np.cos(alpha * np.pi * (x - 0.5)) ** 2
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.4]:
            x = np.linspace(0, 1, n)
            s = np.exp(-(x - center) ** 2 / (2 * width ** 2))
            s = np.maximum(s, 0.01)
            s = s / s.sum()
            results.append(s.copy())

    # Linear ramps
    for beta in [-0.5, -0.2, 0.2, 0.5]:
        x = np.linspace(0, 1, n)
        s = 1.0 + beta * (2.0 * x - 1.0)
        s = np.maximum(s, 0.05)
        s = s / s.sum()
        results.append(s.copy())

    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6]:
        k = max(1, int(frac * n))
        for ratio in [0.5, 1.0, 2.0, 3.0]:
            s = np.ones(n)
            s[:k] = ratio
            s = s / s.sum()
            results.append(s.copy())

    # Random (smoothed)
    for _ in range(4):
        s = np.abs(rng.standard_normal(n)) + 0.1
        # Smooth
        kernel = np.ones(5) / 5
        s = np.convolve(s, kernel, mode='same')
        s = s / s.sum()
        results.append(s.copy())

    # Pure random
    for _ in range(3):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        results.append(s.copy())

    # Resized height_sequence_1
    if hseq is not None and hseq.size > 10:
        x_old = np.linspace(0.0, 1.0, hseq.size)
        x_new = np.linspace(0.0, 1.0, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 0.0)
        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            results.insert(0, h_r)

    # If hseq has exactly length n, use directly
    if hseq is not None and hseq.size == n:
        h_direct = np.maximum(hseq, 0.0)
        if h_direct.sum() > 0.01:
            results.insert(0, h_direct / h_direct.sum())

    # Score and sort
    scored = []
    for s in results:
        sc = fast_score(s)
        if np.isfinite(sc):
            scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    return scored
```
