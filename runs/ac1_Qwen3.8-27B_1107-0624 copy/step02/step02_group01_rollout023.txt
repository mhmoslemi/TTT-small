Smooth gradient descent with FFT-based convolution for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax-smoothed max objective, Adam optimizer in logit space, multi-phase search

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

    Compute R[i] = sum_m w[m+i] * a[m] for i=0..len(a)-1.
    w has length 2n-1, a has length n.
    """
    n = a.size
    ar = a[::-1]
    N = next_pow2(3 * n)
    far = np.fft.rfft(ar, N)
    fw = np.fft.rfft(w, N)
    conv_result = np.fft.irfft(far * fw, N)
    result = conv_result[n - 1:2 * n - 1]
    return result


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def smooth_grad_descent(f0, deadline, rng, beta_frac=0.1, beta_frac_min=0.003,
                        lr=0.008, max_iter=200000, top_k_mode=False, top_k_size=20):
    """
    Softmax-smoothed gradient descent to minimize max(conv(f,f)) subject to f>=0, sum(f)=1.
    Works in logit space with Adam optimizer.
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f + 1e-30)
    z -= z.max()

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = beta_frac * M_val

    best_f = f.copy()
    best_M = M_val
    no_imp = 0

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        if top_k_mode and top_k_size < c.size:
            # Top-k gradient: use only top-k convolution values
            thresh = np.partition(c, c.size - top_k_size)[c.size - top_k_size]
            mask = c >= thresh
            w = mask.astype(np.float64)
            s_w = float(w.sum())
            if s_w > 0:
                w = w / s_w
            else:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
        else:
            # Softmax gradient
            x = (c - M_val) / beta
            ex = np.exp(np.clip(x, -200.0, 50.0))
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                ex = np.zeros_like(c)
                ex[int(np.argmax(c))] = 1.0
                s_ex = 1.0
            w = ex / s_ex

        grad = 2.0 * fft_cross_corr(w, f)

        if not np.all(np.isfinite(grad)):
            no_imp += 1
            if no_imp > 100:
                z = 0.85 * np.log(best_f + 1e-30) + 0.5 * rng.normal(size=n)
                z -= z.max()
                f = np.exp(z - z.max())
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                c = fft_convolve(f)
                M_val = float(c.max())
                beta = beta_frac * M_val
                m_state.fill(0)
                v_state.fill(0)
                t_adam = 0
                no_imp = 0
            continue

        # Adam update
        t_adam += 1
        m_state = b1 * m_state + (1.0 - b1) * grad
        v_state = b2 * v_state + (1.0 - b2) * (grad * grad)
        mh = m_state / (1.0 - b1 ** t_adam)
        vh = v_state / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)

        f = np.exp(z - z.max())
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-13:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Anneal beta
        if it > 0 and it % 150 == 0:
            beta = max(beta_frac_min * M_val, beta * 0.85)

        # Restart if stuck
        if no_imp > max(300, n):
            z = 0.85 * np.log(best_f + 1e-30) + 0.4 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.max()
            f = np.exp(z - z.max())
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(beta_frac_min * M_val * 5, 0.08 * M_val)
            m_state.fill(0)
            v_state.fill(0)
            t_adam = 0
            no_imp = 0

        M_val = M_new

    return best_f, best_M


def peak_shave(f0, deadline, rng, num_tries=5000):
    """
    Try to reduce peaks by moving mass from elements contributing to high peaks
    to elements contributing to low peaks.
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M

    for _ in range(num_tries):
        if time.monotonic() > deadline:
            break

        # Find peak positions
        peak_idx = int(np.argmax(c))
        # Find elements that contribute most to this peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # Contribution of a[i] to c[peak_idx] involves a[peak_idx - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = 2.0 * f[j]
            else:
                contrib[i] = 0.0

        # Find low-convolution region
        min_idx = int(np.argmin(c))

        # Try moving mass from high-contributor to low-peak contributor
        src = int(np.argmax(contrib * f))
        # Find a good destination: element that contributes to low peaks
        dest_contrib = np.zeros(n)
        for i in range(n):
            j = min_idx - i
            if 0 <= j < n:
                dest_contrib[i] = 2.0 * f[j]
            else:
                dest_contrib[i] = 0.0
        dest = int(np.argmin(dest_contrib))

        if src == dest:
            continue

        # Try different transfer amounts
        for frac in [0.01, 0.005, 0.02, 0.002, 0.05]:
            amt = f[src] * frac
            if amt < 1e-15:
                continue
            f_new = f.copy()
            f_new[src] -= amt
            f_new[dest] += amt
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-30:
                continue
            f_new = f_new / s
            c_new = fft_convolve(f_new)
            M_new = float(c_new.max())
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f_new.copy()
                f = f_new.copy()
                c = c_new
                M = M_new
                break

    return best_f, best_M


def make_diverse_starts(n, rng):
    """Generate diverse initial profiles."""
    starts = []
    t = np.linspace(0, 1, n)

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())

    # Sine profiles
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a = 1.0 + amp * np.sin(2 * np.pi * freq * (t + rng.uniform(0, 1)))
            a = np.maximum(a, 1e-10)
            starts.append(a / a.sum())

    # Bi-modal
    for center_spread in [0.15, 0.2, 0.3, 0.4]:
        a = np.exp(-((t - 0.2 * center_spread) ** 2) / (2 * 0.01)) + \
            np.exp(-((t - 1 + 0.2 * center_spread) ** 2) / (2 * 0.01)) + 0.1
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())

    # Random smooth
    for _ in range(10):
        a = np.ones(n)
        for k in range(1, 7):
            a += rng.uniform(0.05, 0.8) / k * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())

    # Power-law-like
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = t ** p + (1 - t) ** p + 0.1
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())

    return starts


def interpolate_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n_old = p.size
    if n_old == n_new:
        p = np.maximum(p, 0.0)
        s = p.sum()
        return p / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p)
    p1 = np.maximum(p1, 1e-10)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search:
    1. Smooth gradient descent from diverse starts at n=512
    2. Extended optimization on best candidates with varied hyperparameters
    3. Peak-shaving polish
    4. Try longer sequences (n=1024, 2048)
    5. Final refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Load parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent: score={sc:.10f}, n={parent.size}")

    n1 = 512

    # Phase 1: Diverse starts with quick gradient descent
    phase1_end = t_start + (deadline - t_start) * 0.35
    starts = []

    if parent is not None:
        p_int = interpolate_to_length(parent, n1)
        if p_int is not None:
            starts.append(p_int)

    starts.extend(make_diverse_starts(n1, rng)[:25])

    n_starts = len(starts)
    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + 8.0)
        p_best, M_best = smooth_grad_descent(p0, per, rng,
                                             beta_frac=0.1, beta_frac_min=0.01,
                                             lr=0.01, max_iter=30000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        if i % 5 == 0:
            print(f"  P1 [{i}/{n_starts}]: best={best_score:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Extended optimization on best with varied hyperparameters
    phase2_end = t_start + (deadline - t_start) * 0.55
    if best_seq is not None:
        hyperparams = [
            (0.05, 0.005, 0.006, False, 0),
            (0.08, 0.004, 0.008, False, 0),
            (0.03, 0.003, 0.004, False, 0),
            (0.06, 0.005, 0.007, True, 15),
            (0.06, 0.005, 0.007, True, 30),
            (0.1, 0.006, 0.01, True, 10),
        ]
        for hp in hyperparams:
            if time.monotonic() > phase2_end:
                break
            bf, bfm, lr, tk_mode, tk_size = hp
            per = min(phase2_end, time.monotonic() + 30.0)
            p_best, M_best = smooth_grad_descent(best_seq, per, rng,
                                                 beta_frac=bf, beta_frac_min=bfm,
                                                 lr=lr, max_iter=100000,
                                                 top_k_mode=tk_mode, top_k_size=tk_size)
            sc = 2.0 * n1 * M_best
            try_update(p_best, sc)
            print(f"  P2 (bf={bf}, lr={lr}, tk={tk_mode}): score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Peak shaving
    phase3_end = t_start + (deadline - t_start) * 0.65
    if best_seq is not None and time.monotonic() < phase3_end:
        per = min(phase3_end, time.monotonic() + 25.0)
        p_best, M_best = peak_shave(best_seq, per, rng, num_tries=10000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"After Phase 3 (peak shave): best={best_score:.10f}")

    # Phase 4: Try longer sequences
    for n_try in [1024, 2048]:
        if time.monotonic() > deadline - 180:
            break
        if best_seq is None:
            break

        p_ext = interpolate_to_length(best_seq, n_try)
        if p_ext is None:
            continue

        per = min(deadline - 120, time.monotonic() + 60.0)
        p_best, M_best = smooth_grad_descent(p_ext, per, rng,
                                             beta_frac=0.08, beta_frac_min=0.005,
                                             lr=0.006, max_iter=50000)
        sc = 2.0 * n_try * M_best
        try_update(p_best, sc)
        print(f"Phase 4 (n={n_try}): score={sc:.10f}, best={best_score:.10f}")

        # Quick polish
        if time.monotonic() < deadline - 90:
            per2 = min(deadline - 90, time.monotonic() + 30.0)
            p_best2, M_best2 = smooth_grad_descent(p_best, per2, rng,
                                                   beta_frac=0.04, beta_frac_min=0.002,
                                                   lr=0.004, max_iter=30000)
            sc2 = 2.0 * n_try * M_best2
            try_update(p_best2, sc2)
            print(f"Phase 4 polish (n={n_try}): score={sc2:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final refinement on absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 45.0)

        # Multiple final attempts with different settings
        final_configs = [
            (0.03, 0.001, 0.003, False, 0),
            (0.05, 0.002, 0.004, True, 10),
            (0.04, 0.001, 0.002, True, 5),
        ]
        for fc in final_configs:
            if time.monotonic() > deadline - 10:
                break
            bf, bfm, lr, tk, tks = fc
            per = min(deadline - 5, time.monotonic() + 15.0)
            p_best, M_best = smooth_grad_descent(best_seq, per, rng,
                                                 beta_frac=bf, beta_frac_min=bfm,
                                                 lr=lr, max_iter=50000,
                                                 top_k_mode=tk, top_k_size=tks)
            sc = compute_score(p_best)
            try_update(p_best, sc)

        # Peak shave one more time
        if time.monotonic() < deadline - 5:
            per = min(deadline - 3, time.monotonic() + 10.0)
            p_best, M_best = peak_shave(best_seq, per, rng, num_tries=5000)
            sc = compute_score(p_best)
            try_update(p_best, sc)

    print(f"After Phase 5: best={best_score:.10f}")

    # Final output
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final: score={sc_final:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fb = 1.0 + 1.5 * (np.linspace(0, 1, 512) - 0.5)
    fb = np.maximum(fb, 1e-10)
    fb = fb / fb.sum()
    return [float(x) for x in fb]
