from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    N = next_pow2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    a_norm = a / s
    c = fft_convolve(a_norm)
    return 2.0 * n * c.max()

    """Compute h[j] = sum_k w[k+j]*a[k] for j=0..n-1 using FFT."""
    n = a.size
    N = next_pow2(3*n)  # enough to avoid wrap-around
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return h

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient descent with softmax smoothing, LP direction finding,
and multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    """Compute the objective value."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    a_norm = a / s
    c = fft_convolve(a_norm)
    return float(2.0 * n * c.max())


def cross_correlation(w, a):
    """Compute h[j] = sum_k w[k+j]*a[k] for j=0..n-1 via FFT."""
    n = a.size
    N = next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return h


def make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0, 1, n)
    
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = 1.0 + eta * t
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    
    # Power profiles
    for p in (0.5, 1.0, 2.0, 3.0, 5.0):
        a = (t + 1e-6) ** p
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    
    # Exponential profiles
    for rate in (1.0, 2.0, 3.0, 5.0):
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    
    # Smooth random
    for _ in range(5):
        a = np.ones(n)
        for k in range(1, 10):
            amp = rng.uniform(0.1, 1.0) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    
    return profiles

    a = np.maximum(a0, 1e-15).astype(np.float64)

    if s < 1e-30:

    a = a / s

    t_adam = 0

    M = float(c.max())
    beta = 0.3 * M
    
    best_a = a.copy()
    best_M = M
