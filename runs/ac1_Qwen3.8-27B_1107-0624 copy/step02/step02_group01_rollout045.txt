"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax gradient descent with Adam optimizer, adaptive temperature,
simulated annealing restarts, and LP-based final polishing across multiple
sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a2):

    u = 2.0 * t - 1.0
    a = 1.0 + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):

    a = np.ones(n)
    for k in range(1, n_harm + 1):

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-15)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def softmax_gradient_optimize(f0, deadline, rng, lr=0.05, max_iter=50000,
                              beta_init=None, beta_min=1e-8, beta_decay=0.85,
                              beta_period=200, anneal_prob=0.0, anneal_scale=0.3,
                              verbose=False):
    """
    Softmax gradient descent in logit space with Adam optimizer.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>0.
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())

    if beta_init is None:
        beta = max(1e-10, 0.12 * M_val)
    else:
        beta = beta_init

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(2 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Softmax weights: w_k ∝ exp(c_k / beta)
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        # Smoothed max approximation
        L = M_val + beta * float(np.log(s_ex))

        # Gradient in z-space: grad_z = 2 * f * (h - L)
        # where h[j] = sum_p w[p] * f[p-j] (cross-correlation of w and f)
        fw = np.fft.rfft(w, N_corr)
        ff = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(ff), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        # Adaptive lr schedule
        effective_lr = lr * (1.0 + 0.5 * np.exp(-it / 1000.0))
        z = z - effective_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Temperature decay
        if t_adam > 0 and t_adam % beta_period == 0:
            beta = max(beta_min, beta * beta_decay)

        # Simulated annealing: occasionally perturb
        if anneal_prob > 0 and rng.random() < anneal_prob:
            z = best_z + anneal_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(beta_min, 0.12 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

        # Hard restart if stuck
        if no_imp > max(500, 4 * n):
            z = 0.8 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(beta_min, 0.12 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"      iter {it}: M={M_new:.12f}, best={best_M:.12f}, beta={beta:.4e}, no_imp={no_imp}")

    return best_f, best_M

    M_val = float(c.max())
    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    N_corr = next_pow2(2 * n)

    # Track which lags are near-max for subgradient
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(f)
        M_val = float(c.max())

        # Find active lags (within 1% of max)
        threshold = M_val * 0.999
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        # Subgradient: grad_j = 2 * sum_{k in active} a_{k-j} * I(0<=k-j<n)
        # Compute via FFT: h[j] = sum_k active[k] * a_{k-j}
        fa = np.fft.rfft(f, N_corr)
        fb = np.fft.rfft(active.astype(np.float64), N_corr)
        h = np.fft.irfft(fb * np.conj(fa), N_corr)[:n]
        grad = 2.0 * h / n_active

        # Projected gradient step on simplex
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-30:
            f_new = best_f.copy()
            continue
        f_new = f_new / fs

        M_new = float(fft_convolve(f_new).max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f_new.copy()
            no_imp = 0
            f = f_new
            # Decrease lr slightly
            lr *= 0.999
        else:
            no_imp += 1
            # Average step (Polyak averaging)
            f = 0.95 * f + 0.05 * f_new
            f = np.maximum(f, 1e-15)
            f = f / f.sum()

        if no_imp > patience:
            f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n))
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            lr = 0.02
            no_imp = 0

        if verbose and it % 500 == 0:
            print(f"      subgrad iter {it}: M={M_val:.12f}, best={best_M:.12f}, n_active={n_active}")

    return best_f, best_M


def lp_direction(f, M, n, time_limit=15):
    """Solve LP for direction: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    if n > 2048:
        # Too large for direct LP, use subsampled approach
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x
    return None


def lp_polish(f0, deadline, rng, max_iter=5000, verbose=False):
    """LP-based iterative polishing."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = lp_direction(f, M, n, time_limit=10)
        if g is None:
            no_imp += 1
            if no_imp > 30:
                break
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 100:
                break
            continue

        g = g / g_sum
        improved = False
        for t in (0.3, 0.15, 0.5, 0.1, 0.7, 0.2, 0.05, 0.4):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = compute_max_conv(f_new)
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                improved = True
                break

        if not improved:
            no_imp += 1
            if no_imp % 50 == 0:
                f = 0.8 * best_f + 0.2 * np.abs(rng.normal(size=n))
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                M = compute_max_conv(f)
                no_imp = 0

        if verbose and it % 200 == 0:
            print(f"      LP iter {it}: M={M:.12f}, best={best_M:.12f}")

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search combining softmax gradient descent, subgradient method,
    and LP polishing across multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # ============ Phase 1: Softmax gradient at n=512 ============
    n1 = 512
    phase1_end = min(deadline, t_start + 500.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)

    # Add diverse deterministic starts
    profs = make_deterministic_profiles(n1)
    starts.extend(profs[:10])

    # Add random smooth starts
    for _ in range(6):
        starts.append(make_smooth_random(n1, rng, n_harm=10))

    per_start = max(3.0, (phase1_end - t_start) / max(1, len(starts)))
    candidates = []

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = softmax_gradient_optimize(
            p0, per, rng, lr=0.06, max_iter=30000,
            beta_decay=0.82, beta_period=150)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))

        if i % 5 == 0:
            print(f"  Phase 1 start {i}/{len(starts)}: best={best_score:.10f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1 (n={n1}): best={best_score:.10f}")

    # ============ Phase 2: Extended softmax on top candidates ============
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(4, len(candidates))
    per_cand = max(10.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = softmax_gradient_optimize(
            p0, per, rng, lr=0.03, max_iter=50000,
            beta_init=None, beta_decay=0.88, beta_period=300,
            anneal_prob=0.02, anneal_scale=0.2)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 2 cand {i}: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Subgradient polishing ============
    phase3_end = min(deadline, t_start + 1100.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 80.0)
        p_best, M_best = subgradient_optimize(
            best_seq, per, rng, lr=0.015, max_iter=20000, patience=300)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"After Phase 3 (subgrad): best={best_score:.10f}")

    # ============ Phase 4: LP polishing ============
    phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and best_seq.size <= 1024:
        per = min(phase4_end, time.monotonic() + 120.0)
        p_best, M_best = lp_polish(
            best_seq, per, rng, max_iter=10000)
        sc = 2.0 * best_seq.size * M_best
        try_update(p_best, sc)
        print(f"After Phase 4 (LP): best={best_score:.10f}")

    # ============ Phase 5: Try larger sequences (n=1024) ============
    n2 = 1024
    phase5_end = min(deadline, t_start + 1650.0)

    if best_seq is not None and time.monotonic() < phase5_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase5_end, time.monotonic() + 80.0)
            p_best, M_best = softmax_gradient_optimize(
                p_ext, per, rng, lr=0.04, max_iter=30000,
                beta_decay=0.85, beta_period=200)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 5 (n={n2} softmax): score={sc:.10f}")

            if time.monotonic() < phase5_end - 30 and sc < best_score + 1e-6:
                per2 = min(phase5_end, time.monotonic() + 50.0)
                p_best2, M_best2 = softmax_gradient_optimize(
                    p_best, per2, rng, lr=0.02, max_iter=20000,
                    beta_decay=0.9, beta_period=400)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 5 polish (n={n2}): score={sc2:.10f}")

    # Also try n=1024 from scratch with a few starts
    if time.monotonic() < phase5_end - 30:
        for p0 in (make_linear_profile(n2, 1.2), make_cubic_profile(n2, 1.5, -1.2)):
            if time.monotonic() > phase5_end - 30:
                break
            per = min(phase5_end - 30, time.monotonic() + 25.0)
            p_best, M_best = softmax_gradient_optimize(
                p0, per, rng, lr=0.05, max_iter=15000,
                beta_decay=0.8, beta_period=100)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)

    print(f"After Phase 5: best={best_score:.10f}")

    # ============ Phase 6: Final refinement on the absolute best ============
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)

        if n_best <= 1024:
            p_best, M_best = softmax_gradient_optimize(
                best_seq, per, rng, lr=0.01, max_iter=30000,
                beta_decay=0.92, beta_period=500)
            sc = 2.0 * n_best * M_best
            try_update(p_best, sc)
            print(f"Phase 6 final (softmax): best={best_score:.10f}")

        # Try LP if still time
        if time.monotonic() < deadline - 10 and best_seq.size <= 1024:
            per2 = min(deadline - 5, time.monotonic() + 30.0)
            p_best, M_best = lp_polish(
                best_seq, per2, rng, max_iter=5000)
            sc = 2.0 * best_seq.size * M_best
            try_update(p_best, sc)
            print(f"Phase 6 final (LP): best={best_score:.10f}")

    # ============ Final evaluation ============
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s

        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        # Verify with evaluate_sequence
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
                    # Use the evaluator's score as ground truth
                    if sc_eval < sc_final:
                        print(f"  (evaluator gives lower: {sc_eval:.10f} vs {sc_final:.10f})")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    print("Using fallback. Score:", compute_score(fb))
    return [float(x) for x in fb]
