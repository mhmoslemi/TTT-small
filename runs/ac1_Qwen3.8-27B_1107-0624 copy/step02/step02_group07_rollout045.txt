import numpy as np
import time
from scipy.optimize import linprog

"""
Improved optimization for minimizing 2*n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Key improvements over previous approach:
1. Power-law peak weighting (sharper than exponential) for gradient computation
2. Spectral equalization steps to reduce power spectrum dynamism
3. Direct subgradient method targeting top-k autocorrelation peaks
4. Adaptive backtracking line search
5. LP-guided direction finding with tighter constraints
6. Multi-scale approach: optimize at coarse resolution, then refine
"""

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2 efficiently."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * conv.max() / (total * total))

def compute_gradient_weighted(s, w):
    """
    Compute gradient of sum_k w[k] * conv(s,s)[k] with respect to s.
    Uses FFT for efficiency.
    w should be of length 2*n-1 (same as conv).
    Returns array of length n.
    """
    n = len(s)
    N_fft = 1
    while N_fft < 2 * (2 * n - 1):
        N_fft *= 2
    
    S = np.fft.rfft(s, N_fft)
    W = np.fft.rfft(w, N_fft)
    
    # conv(s,s)[k] = sum_i s[i]*s[k-i]
    # d/ds[j] conv(s,s)[k] = 2*s[k-j] (if 0<=k-j<n)
    # So grad[j] = 2*sum_k w[k]*s[k-j] = 2 * (w cross-correlated with s)[j]
    # = 2 * ifft(conj(fft(w)) * fft(s))[j] ... 
    
    # Actually: sum_k w[k]*s[k-j] = (w * s)[j] in the cross-correlation sense
    # Let me define: sum_k w[k]*s[k-j] for k where s[k-j] is valid (0<=k-j<n)
    # This is the cross-correlation of w and s at lag j.
    # cross_corr(w, s)[j] = sum_k w[k]*s[k-j]
    
    # Using FFT: cross_corr = ifft(conj(FFT(w)) * FFT(s))
    # But we need to be careful with the length.
    
    # Alternative: use the fact that grad[j] = 2 * sum_{k} w[k] * s[k-j] * [0<=k-j<n]
    # = 2 * sum_{i=0}^{n-1} w[j+i] * s[i]  (substituting k=j+i)
    # = 2 * (w_conv * s)[j] where w_conv is w padded appropriately
    
    # Let me use a direct FFT approach:
    # Define x = w (length 2n-1), y = s (length n)
    # We want z[j] = sum_i x[j+i] * y[i] for i=0..n-1
    # This is a cross-correlation.
    
    # Using FFT: z = ifft(conj(FFT(x, N)) * FFT(y, N))[0:n]
    # Wait, I need to be more careful.
    
    # cross_corr[x,y][j] = sum_i x[i+j]*y[i]
    # This equals ifft(FFT(x) * conj(FFT(y)))[j]... no.
    
    # Let me just use the convolution theorem properly.
    # cross_corr[x,y][j] = sum_i x[i+j]*y[i] = (x flipped, convolved with y)[something]
    
    # Actually the simplest: x[i+j]*y[i] summed over i.
    # Let's define x_flipped[i] = x[-i] (or equivalently, reverse x).
    # Then cross_corr[j] = sum_i x[i+j]*y[i] = sum_i x_flipped[-i-j]*y[i]... this is getting confusing.
    
    # Let me just do it with the standard FFT cross-correlation formula:
    # cross_correlation(a, b)[k] = sum_n a[n+k] * b[n]
    # = ifft(FFT(a, N) * conj(FFT(b, N)))[k]  ... No, that's not right either.
    
    # The correct formula: 
    # If we define C[k] = sum_n a[n] * b[n-k] (cross-correlation of a and b at lag k)
    # Then C = ifft(FFT(a) * conj(FFT(b)))
    
    # In our case: grad[j] = 2 * sum_k w[k] * s[k-j]
    # Let a = w, b = s: C[j] = sum_k w[k] * s[k-j] ... this is cross-correlation of w and s at lag j
    # C = ifft(FFT(w) * conj(FFT(s)))
    
    W_fft = np.fft.rfft(w, N_fft)
    S_fft = np.fft.rfft(s, N_fft)
    cross = np.fft.irfft(W_fft * np.conj(S_fft), N_fft)[0:n]
    
    return 2.0 * cross

def gradient_step_power(s, p=16, eta=0.2):
    """
    Gradient step with power-law peak weighting.
    w[k] = (conv[k] / max_conv)^p
    """
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return s.copy()
    
    w = (conv / max_conv) ** p
    grad = compute_gradient_weighted(s, w)
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()
    
    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-10)
    s_new = s_new / s_new.sum()
    return s_new

def spectral_equalize(s, alpha=0.5):
    """
    Spectral equalization: reduce peaks in the power spectrum.
    S_new(f) = S(f) * (|S(f)| / mean|S(f)|)^(-alpha)
    """
    n = len(s)
    S = np.fft.rfft(s)
    mag = np.abs(S)
    mag_mean = mag.mean()
    if mag_mean < 1e-15:
        return s.copy()
    
    # Reduce peaks, boost valleys
    scale = (mag / mag_mean) ** (-alpha)
    scale[0] = 1.0  # Don't modify DC component (preserves sum)
    S_new = S * scale
    
    # Enforce Hermitian symmetry (should already be there since we used rfft)
    s_new = np.fft.irfft(S_new, n)
    s_new = np.maximum(s_new, 0.0)
    if s_new.sum() < 1e-15:
        return s.copy()
    s_new = s_new / s_new.sum()
    return s_new

def lp_direction(s, M, time_limit=3.0):
    """
    Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0.
    Returns optimal b or None.
    """
    n = len(s)
    if n > 512:
        return None  # Too large for LP
    
    # conv(s,b)[k] = sum_i s[i]*b[k-i] <= M
    # This is a linear constraint in b.
    # A_ub[b] = conv(s,b) <= M
    # A_ub has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    
    c = -np.ones(n)  # Minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None

                # Line search

                best_ts = cur_score

                    cand = (1 - t) * s + t * b

                    if ts < best_ts:

                        best_t = t
                if best_t > 0:
                    s = ((1 - best_t) * s + best_t * b)
                    s = np.maximum(s, 0.0) / s.sum()
                    if best_ts < best_score - 1e-10:
