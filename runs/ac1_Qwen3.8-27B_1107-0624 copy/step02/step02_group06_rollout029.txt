def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 50.0))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def interpolate_profile(p0, n_new):

def build_conv_mat(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi + 1] = f[k - i_hi:k - i_lo + 1][::-1]
    return A


def lp_solve(f, M):
    n = len(f)
    A_ub = build_conv_mat(f)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status != 0:
        return None
    return res.x

def coordinate_descent_polish(p, deadline, rng, n_sweeps=None, step_frac=0.1):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    if n_sweeps is None:
        n_sweeps = max(3, int((deadline - time.monotonic()) * 3))

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    current_step = step_frac

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved_this_sweep = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            factors = [1.0 + current_step, 1.0 + 0.5 * current_step, 1.0 + 2.0 * current_step,
                       1.0 - current_step, 1.0 - 0.5 * current_step, 1.0 - 0.3 * current_step]
            best_val = orig
            best_sc = internal_score(p)
            for factor in factors:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                p[i] = new_val
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved_this_sweep = True
                no_imp = 0
            else:
                no_imp += 1
        if improved_this_sweep:
            current_step = max(0.02, current_step * 0.9)
        if no_imp > 3 * n:
            break

    return best_p, best_score

        for tn in (128, 256, 512):
            if parent.size != tn:
                p_interp = interpolate_profile(parent, tn)
                sc_i = internal_score(p_interp)
                local_update(p_interp, sc_i, "parent_n{}".format(tn))

    for n_init in (128, 256):
        profs = make_deterministic_profiles(n_init)
        for p0 in profs[:10]:
            sc = internal_score(p0)
            local_update(p0, sc, "det_n{}".format(n_init))
        for _ in range(5):
            p0 = make_random_smooth_profile(n_init, rng)
            sc = internal_score(p0)
            local_update(p0, sc, "rand_n{}".format(n_init))

    print("  After init: best={:.10f}".format(best_score))

    # Phase 1: Hybrid optimization at n=256
    print("Phase 1: Hybrid opt at n=256")
    n1 = 256
    phase1_end = min(deadline, t_start + 250.0)
    top_n1 = min(8, len(candidates))
    for i in range(top_n1):
        if time.monotonic() > phase1_end:
            break
        sc_c, p_c = candidates[i]
        p_c256 = interpolate_profile(p_c, n1) if p_c.size != n1 else p_c
        p_c256 = normalize_nonneg(p_c256)
        if p_c256 is None:
            continue
        per = min(phase1_end, time.monotonic() + 20.0)
        p_best, sc = hybrid_optimize(p_c256, per, rng, use_lp=True, lr=0.08, lp_every=30)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p1_{}".format(i))

    # Also try pure LP-FW at n=256
    for i in range(min(4, len(candidates))):
        if time.monotonic() > phase1_end:
            break
        sc_c, p_c = candidates[i]
        p_c256 = interpolate_profile(p_c, n1) if p_c.size != n1 else p_c
        p_c256 = normalize_nonneg(p_c256)
        if p_c256 is None:
            continue
        per = min(phase1_end, time.monotonic() + 10.0)
        p_best, sc = lp_frank_wolfe(p_c256, per, rng, max_iters=5000)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p1lp_{}".format(i))

    print("  After phase 1: best={:.10f}".format(best_score))

    # Phase 2: Hybrid optimization at n=512
    print("Phase 2: Hybrid opt at n=512")
    n2 = 512
    phase2_end = min(deadline, t_start + 750.0)
    for i in range(min(10, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        p_c512 = interpolate_profile(p_c, n2) if p_c.size != n2 else p_c
        p_c512 = normalize_nonneg(p_c512)
        if p_c512 is None:
            continue
        per = min(phase2_end, time.monotonic() + 30.0)
        p_best, sc = hybrid_optimize(p_c512, per, rng, use_lp=True, lr=0.06, lp_every=40)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p2_{}".format(i))

    # LP-FW at n=512 for top candidates
    for i in range(min(5, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        p_c512 = interpolate_profile(p_c, n2) if p_c.size != n2 else p_c
        p_c512 = normalize_nonneg(p_c512)
        if p_c512 is None:
            continue
        per = min(phase2_end, time.monotonic() + 15.0)
        p_best, sc = lp_frank_wolfe(p_c512, per, rng, max_iters=10000)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p2lp_{}".format(i))

    print("  After phase 2: best={:.10f}".format(best_score))

    # Phase 3: Try n=1024
    print("Phase 3: Hybrid opt at n=1024")
    n3 = 1024
    phase3_end = min(deadline, t_start + 1300.0)
    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        p_c1024 = interpolate_profile(p_c, n3) if p_c.size != n3 else p_c
        p_c1024 = normalize_nonneg(p_c1024)
        if p_c1024 is None:
            continue
        per = min(phase3_end, time.monotonic() + 40.0)
        p_best, sc = hybrid_optimize(p_c1024, per, rng, use_lp=True, lr=0.05, lp_every=60)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p3_{}".format(i))

    # LP-FW at n=1024
    for i in range(min(4, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        p_c1024 = interpolate_profile(p_c, n3) if p_c.size != n3 else p_c
        p_c1024 = normalize_nonneg(p_c1024)
        if p_c1024 is None:
            continue
        per = min(phase3_end, time.monotonic() + 20.0)
        p_best, sc = lp_frank_wolfe(p_c1024, per, rng, max_iters=5000)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p3lp_{}".format(i))

    print("  After phase 3: best={:.10f}".format(best_score))

    # Phase 4: Extended polish of top candidates
    print("Phase 4: Extended polish")
    phase4_end = min(deadline, t_start + 1600.0)
    for i in range(min(8, len(candidates))):
        if time.monotonic() > phase4_end:
            break
        sc_c, p_c = candidates[i]
        per = min(phase4_end, time.monotonic() + (25.0 if p_c.size <= 512 else 40.0))
        p_best, sc = hybrid_optimize(p_c, per, rng, use_lp=True, lr=0.04, lp_every=50)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p4_{}".format(i))

    print("  After phase 4: best={:.10f}".format(best_score))

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1750.0)
    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase5_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size <= 1024:
            per = min(phase5_end, time.monotonic() + 20.0)
            p_best, sc = coordinate_descent_polish(p_c, per, rng, step_frac=0.08)
            sc_full = internal_score(p_best)
            local_update(p_best, sc_full, "p5_{}".format(i))

    print("  After phase 5: best={:.10f}".format(best_score))

    # Phase 6: Final exact scoring
    print("Phase 6: Final selection")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        if final_seq.size <= 4096:
            sc = true_score(final_seq)
            if np.isfinite(sc):
                final_score = min(final_score, sc)

        for sc_c, p_c in candidates:
            if p_c.size <= 4096:
                sc_cx = true_score(p_c)
                if np.isfinite(sc_cx) and sc_cx < final_score:
                    final_score = sc_cx
                    final_seq = p_c

        if final_seq is not None and final_seq.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval):
                    final_score = min(final_score, sc_eval)
            except Exception:
                pass

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print("  FINAL: score={:.10f} n={}".format(final_score, final_seq.size))
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_linear_profile(256, 1.5)
    print("  FALLBACK: score={:.10f}".format(internal_score(fallback)))
    return [float(x) for x in fallback]
