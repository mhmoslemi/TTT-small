import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_gradient(c, p, beta):
    """Compute gradient of soft-max smoothed objective using FFT correlation."""
    _, w = soft_mx(c, beta)
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    gp = 2.0 * h[n - 1:2 * n - 1]
    return gp

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:

    return normalize_nonneg(np.interp(x1, x0, p0))


def smooth_max_adam_nesterov(p0, deadline, rng, lr=0.12, reg=1e-7, beta_init_frac=0.15,

def coordinate_descent_polish(p, deadline, rng, step_modes=None):
    """
    Coordinate descent with multi-step try and adaptive step size.
    More aggressive than previous: tries larger steps and uses golden section.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    total_sweeps = 0

    # Step factors to try (more aggressive than before)
    if step_modes is None:
        step_modes = [1.2, 0.8, 1.5, 0.5, 1.8, 0.4, 2.5, 0.3]

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved_this_sweep = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score

            for factor in step_modes:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                p[i] = new_val
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                # Normalize
                p = p / s
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
                    improved_this_sweep = True

            if best_val != orig:
                p[i] = best_val
                p = p / p.sum()
                best_score = best_sc
                best_p = p.copy()
                no_imp = 0
            else:
                p[i] = orig
                p = p / p.sum()
                no_imp += 1

        total_sweeps += 1
        if no_imp > 3 * n:
            # Try increasing step size
            step_modes = [x * 1.3 for x in step_modes]
            no_imp = 0
        if total_sweeps > 2000:
            break

    return best_p, best_score


def multi_start_gradient_search(p0, deadline, rng, n_starts=5, per_start_frac=0.7):
    """
    Run multiple gradient descent starts with perturbations around p0.
    Returns the best result.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    p0 = normalize_nonneg(p0)
    if p0 is None:
        return p0, np.inf

    n = p0.size
    best_p = p0.copy()
    best_score = internal_score(p0)

    per_start = min(deadline - time.monotonic(), (deadline - time.monotonic()) * per_start_frac / max(1, n_starts))

    for s in range(n_starts):
        if time.monotonic() > deadline:
            break
        # Perturbation scale decreases with each start
        scale = 0.3 * (0.7 ** s)
        p_start = p0 + scale * rng.standard_normal(n)
        p_start = np.maximum(p_start, 1e-12)
        p_start = p_start / p_start.sum()

        end_t = min(deadline, time.monotonic() + per_start)
        if end_t - time.monotonic() < 5.0:
            break

        p_res, sc = smooth_max_adam_nesterov(p_start, end_t, rng, lr=0.10, beta_init_frac=0.12)
        sc_check = internal_score(p_res)
        if sc_check < best_score:
            best_score = sc_check
            best_p = p_res.copy()

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Nesterov-accelerated Adam with adaptive soft-max, population-based
    multi-start, multi-length exploration, and coordinate descent polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} [{label}]")
        if len(candidates) < 25:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            if float(sc) < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])

    # Load parent solution
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        parent = None

    # Phase 0: Initialization
    print("Phase 0: Initialization")
    if parent is not None:
        sc = internal_score(parent)
        local_update(parent, sc, "parent")
        print(f"  Parent: score={sc:.10f} n={parent.size}")

    # Try deterministic profiles at n=512
    n_main = 512
    for p0 in make_deterministic_profiles(n_main)[:10]:
        sc = internal_score(p0)
        local_update(p0, sc, f"det_n{n_main}")

    # If parent exists, interpolate to main length
    if parent is not None and parent.size != n_main:
        p_interp = interpolate_profile(parent, n_main)
        if p_interp is not None:
            sc = internal_score(p_interp)
            local_update(p_interp, sc, f"parent_interp_n{n_main}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: Intensive gradient optimization at main length (n=512)
    print("Phase 1: Gradient optimization at n=512")
    phase1_end = min(deadline, t_start + 700.0)

    if parent is not None:
        p_work = interpolate_profile(parent, n_main) if parent.size != n_main else parent.copy()
        p_work = normalize_nonneg(p_work)
        if p_work is not None:
            per = min(phase1_end, time.monotonic() + 200.0)
            p_best, sc = smooth_max_adam_nesterov(p_work, per, rng, lr=0.12, beta_init_frac=0.15)
            sc_check = internal_score(p_best)
            local_update(p_best, sc_check, "p1_parent")

    # Multi-start from top candidates
    top_k = min(8, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase1_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n_main:
            p_c = interpolate_profile(p_c, n_main)
        p_c = normalize_nonneg(p_c)
        if p_c is None:
            continue
        per = min(phase1_end, time.monotonic() + 40.0)
        if per - time.monotonic() < 10.0:
            break
        p_best, sc = smooth_max_adam_nesterov(p_c, per, rng, lr=0.10, beta_init_frac=0.12)
        sc_check = internal_score(p_best)
        local_update(p_best, sc_check, f"p1_cand_{i}")

    # Also try optimizing deterministic profiles directly
    for p0 in make_deterministic_profiles(n_main)[:6]:
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + 35.0)
        if per - time.monotonic() < 10.0:
            break
        p_best, sc = smooth_max_adam_nesterov(p0, per, rng, lr=0.12, beta_init_frac=0.15)
        sc_check = internal_score(p_best)
        local_update(p_best, sc_check, "p1_det")

    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: Multi-length exploration
    print("Phase 2: Multi-length exploration")
    phase2_end = min(deadline, t_start + 1200.0)
    target_lengths = [256, 384, 640]

    for tn in target_lengths:
        if time.monotonic() > phase2_end:
            break

        # Interpolate top candidates to this length
        for i in range(min(5, len(candidates))):
            if time.monotonic() > phase2_end:
                break
            sc_c, p_c = candidates[i]
            p0 = interpolate_profile(p_c, tn)
            p0 = normalize_nonneg(p0)
            if p0 is None:
                continue
            per = min(phase2_end, time.monotonic() + 30.0)
            if per - time.monotonic() < 10.0:
                break
            p_best, sc = smooth_max_adam_nesterov(p0, per, rng, lr=0.10, beta_init_frac=0.12)
            sc_check = internal_score(p_best)
            local_update(p_best, sc_check, f"p2_n{tn}_c{i}")

        # Deterministic at this length
        for p0 in make_deterministic_profiles(tn)[:4]:
            if time.monotonic() > phase2_end:
                break
            per = min(phase2_end, time.monotonic() + 25.0)
            if per - time.monotonic() < 10.0:
                break
            p_best, sc = smooth_max_adam_nesterov(p0, per, rng, lr=0.12, beta_init_frac=0.15)
            sc_check = internal_score(p_best)
            local_update(p_best, sc_check, f"p2_n{tn}_det")

    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: Extended polish of top candidates
    print("Phase 3: Extended polish")
    phase3_end = min(deadline, t_start + 1550.0)

    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        n_c = p_c.size
        per = min(phase3_end, time.monotonic() + (40.0 if n_c <= 384 else 50.0))
        if per - time.monotonic() < 10.0:
            break
        p_best, sc = smooth_max_adam_nesterov(p_c, per, rng, lr=0.06, beta_init_frac=0.08,
                                              nesterov_alpha=0.95)
        sc_check = internal_score(p_best)
        local_update(p_best, sc_check, f"p3_{i}")

    print(f"  After phase 3: best={best_score:.10f}")

    # Phase 4: Coordinate descent polish
    print("Phase 4: Coordinate descent")
    phase4_end = min(deadline, t_start + 1750.0)

    polish_targets = []
    for i in range(min(6, len(candidates))):
        sc_c, p_c = candidates[i]
        if p_c.size <= 512:
            polish_targets.append(p_c)

    for pi, p0 in enumerate(polish_targets):
        if time.monotonic() > phase4_end:
            break
        per = min(phase4_end, time.monotonic() + (30.0 if p0.size <= 256 else 45.0))
        if per - time.monotonic() < 5.0:
            break
        p_best, sc = coordinate_descent_polish(p0, per, rng)
        sc_check = internal_score(p_best)
        local_update(p_best, sc_check, f"p4_cd_{pi}")

    print(f"  After phase 4: best={best_score:.10f}")

    # Phase 5: Final intensive polish on the absolute best
    print("Phase 5: Final polish on best")
    phase5_end = min(deadline, t_start + 1780.0)

    if best_seq is not None and time.monotonic() < phase5_end:
        p_best_now = best_seq.copy()
        per = phase5_end
        p_res, sc = smooth_max_adam_nesterov(p_best_now, per, rng, lr=0.04, beta_init_frac=0.05,
                                             nesterov_alpha=0.97)
        sc_check = internal_score(p_res)
        local_update(p_res, sc_check, "p5_final")

        # Coordinate descent on the result
        if time.monotonic() < phase5_end and p_res.size <= 512:
            per2 = min(phase5_end, time.monotonic() + 30.0)
            p_cd, sc_cd = coordinate_descent_polish(p_res, per2, rng)
            sc_check2 = internal_score(p_cd)
            local_update(p_cd, sc_check2, "p5_cd")

    # Phase 6: Final selection with exact scoring
    print("Phase 6: Final selection")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        # Exact score for final candidate
        if final_seq.size <= 2048:
            sc = true_score(final_seq)
            if np.isfinite(sc):
                final_score = sc

        # Check all candidates exactly
        for sc_c, p_c in candidates:
            if p_c.size <= 2048:
                sc_cx = true_score(p_c)
                if np.isfinite(sc_cx) and sc_cx < final_score:
                    final_score = sc_cx
                    final_seq = p_c.copy()

        # Verify with evaluator
        if final_seq is not None and final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval):
                    final_score = min(final_score, sc_eval)
            except Exception:
                pass

        # Final normalization
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_linear_profile(256, 2.0)
    print(f"  FALLBACK: score={internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
