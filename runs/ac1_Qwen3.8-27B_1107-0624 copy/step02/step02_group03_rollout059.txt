def lp_direction(f, M, timeout=10.0):

    return None

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP direction-finding with backtracking, Adam logit-space polish,
and simplex-projected gradient descent. Explores n=512 and n=1024.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f, g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)<=M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(f, deadline, max_iters=2000, verbose=True):
    """LP iterations with backtracking line search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=10)
        if g_raw is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = normalize(g_raw)
        if g is None:
            break

        # Backtracking line search
        found = False
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = normalize(f_new)
            if f_new is None:
                continue
            c_new = np.convolve(f_new, f_new)
            score_new = 2.0 * n * float(c_new.max())
            if score_new < best_score - 1e-10:
                best_f = f_new.copy()
                best_score = score_new
                M = float(np.convolve(best_f, best_f).max())
                f = f_new.copy()
                no_improve = 0
                found = True
                break

        if not found:
            no_improve += 1
            f = best_f.copy()
            M = float(np.convolve(f, f).max())
            if no_improve > 50:
                break

        if verbose and it % 50 == 0:
            print(f"    LP {it}: score={best_score:.10f}, M={M:.8e}")

    return best_f, best_score


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j]."""
    n = len(p)
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.dot(w[i:i + n], p)
    return h


def softmax_weights(c, beta):
    """Compute softmax weights for the max of c."""
    max_c = float(c.max())
    x = (c - max_c) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    return w, max_c


def adam_polish(p0, deadline, lr=0.03, verbose=True):
    """Adam in logit space to minimize max(conv(p,p)) with sum(p)=1."""
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.05 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        w, _ = softmax_weights(c, beta)
        h = corr_weights(w, p0)
        grad_p = 2.0 * h

        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.93)

        if it % 2000 == 0:
            beta = max(1e-12, beta * 0.4)
            if verbose:
                print(f"    Adam {it}: score={best_score:.10f}, lr={lr:.4f}, beta={beta:.2e}")

        if no_imp > 4000:
            rng = np.random.default_rng(it)
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            beta = max(1e-12, 0.05 * float(np.convolve(p0, p0).max()))

        p0 = p

    return best_p, best_score


def simplex_project(v, target_sum=1.0):
    """Project v onto {x: x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v
    x_min = v.min()
    x_max = v.max()
    lo = x_min
    hi = x_max
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(v - mid, 0.0).sum()
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(v - tau, 0.0)


def grad_polish_simplex(p0, deadline, lr_init=0.005, verbose=True):
    """Projected gradient on simplex to minimize max(conv(p,p))."""
    n = len(p0)
    p = p0.copy()
    c = np.convolve(p, p)
    best_score = 2.0 * n * float(c.max())
    best_p = p.copy()
    beta = max(1e-12, 0.05 * float(c.max()))
    lr = lr_init
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p, p)
        max_c = float(c.max())

        w, _ = softmax_weights(c, beta)
        grad = 2.0 * corr_weights(w, p)

        p_new = p - lr * grad
        p_new = simplex_project(p_new, 1.0)

        score_new = 2.0 * n * float(np.convolve(p_new, p_new).max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_p = p_new.copy()
            p = p_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(1e-7, lr * 0.9)
            p = best_p.copy()

        if it % 3000 == 0:
            beta = max(1e-12, beta * 0.5)
            if verbose:
                print(f"    SimGrad {it}: score={best_score:.10f}, lr={lr:.6f}")

        if no_imp > 3000:
            rng = np.random.default_rng(it)
            perturbation = 0.005 * rng.normal(size=n)
            p = simplex_project(best_p + perturbation, 1.0)
            lr = lr_init
            no_imp = 0
            beta = max(1e-12, 0.05 * float(np.convolve(p, p).max()))

    return best_p, best_score


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_profile(n, type_id, **kw):
    """Generate various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if type_id == 0:
        eta = kw.get('eta', 1.5)
        a = 1.0 + eta * u
    elif type_id == 1:
        eta = kw.get('eta', 2.0)
        a = 1.0 + eta * u
    elif type_id == 2:
        a = 1.0 + 0.8 * (1.0 - u * u)
    elif type_id == 3:
        a = 1.0 + 1.5 * u - 0.5 * u * u
    elif type_id == 4:
        a = 1.0 + 2.0 * u - 1.5 * u * u + 0.5 * u**3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
