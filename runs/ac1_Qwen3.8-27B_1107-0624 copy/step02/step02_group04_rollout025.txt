    n = len(a)

        order = np.arange(n)

        # Constraint: for k in [i, min(i+n-1, 2n-2)]:
        #   conv[k] + a[k-i] * g[i] <= M
        #   => g[i] <= (M - conv[k]) / a[k-i]

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        a_new = a_new * np.sqrt(2.0 * n) / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())

        if M_new < M - 1e-14:
            a = a_new
            M = M_new
            no_imp = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                best_score = 2.0 * n * best_M / (best_a.sum() ** 2)
            if it % 300 == 0:
                t = max(t_min, t * 0.98)
        else:
            t = max(t_min, t * 0.6)
            no_imp += 1
            if no_imp > 300:
                # Perturb and restart
                a = a + 0.05 * np.random.randn(n) * np.sqrt(2.0 * n) / n
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a * np.sqrt(2.0 * n) / s
                t = t_init
                no_imp = 0

        it += 1

    score = 2.0 * n * best_M / (best_a.sum() ** 2)
    if verbose:
        print(f"    {label} done: best_M={best_M:.8f}, score={score:.10f}, iters={it}")
    return best_a, score


def _gradient_polish(a_init, time_limit, rng, lr=0.06, beta_init_frac=0.12,
                     beta_decay=0.72, var_weight=0.0, name=""):
    """Logit-space Adam with adaptive beta, backtracking, and restarts."""
    p0 = np.asarray(a_init, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _fft_convolve(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    start_time = time.monotonic()

    while time.monotonic() - start_time < time_limit:
        L, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)

        if var_weight > 0.0:
            center = c[n - 1:2 * n - 1] if c.size >= 2 * n - 1 else c
            gp += var_weight * (p - p.mean())

        vp = gp + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        p_new = _softmax_from_z(z_new)
        c_new = _fft_convolve(p_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)

        if np.isfinite(score_new) and score_new < best_score:
            z = z_new
            p = p_new
            c = c_new
            maxc = maxc_new
            best_score = score_new
            best_p = p_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.003, lr * 0.9)

        if adam_t % 50 == 0:
            beta = max(1e-14, beta * beta_decay)

        no_imp_limit = max(200, 4 * n)
        if no_imp > no_imp_limit:
            if best_score < 1.52:
                noise_scale = 0.08
            elif best_score < 1.55:
                noise_scale = 0.15
            else:
                noise_scale = 0.3
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _fft_convolve(p)
            maxc = float(c.max())
            beta = max(1e-12, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_profile(n, kind="smooth"):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == "linear":
        a = 1.0 + 1.2 * u
    elif kind == "cubic":
        a = 1.0 + 1.5 * u - 0.8 * u ** 3
    elif kind == "quartic":
        a = 1.0 + 1.0 * u + 0.5 * u ** 2 - 0.3 * u ** 4
    elif kind == "gauss":
        a = 1.0 + 0.8 * np.exp(-4.0 * u ** 2)
    else:
        a = 1.0 + 0.5 * u + 0.3 * u ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0)
    s = p.sum()
    if s > 1e-10:
        p = p / s
    return p


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimizer:
    Phase 1: Coordinate-wise LP iteration (polar method) at multiple lengths
    Phase 2: Gradient-based polishing with adaptive soft-max
    Phase 3: Multi-restart exploration
    Phase 4: Final refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  *** NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    print(f"=== Starting optimization with budget={budget_s}s ===")

    # Get parent sequence
    parent = None
    parent_n = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                parent_n = parent.size
                parent_score = _score(parent)
                update_best(parent, parent_score, "parent")
                print(f"Parent: n={parent_n}, score={parent_score:.10f}")
    except Exception as e:
        print(f"Parent not available: {e}")
        parent = None

    # ============ PHASE 1: LP iteration ============
    print(f"\n--- Phase 1: LP iteration ---")
    phase1_end = min(deadline, t_start + 400.0)

    # Try parent at its native length first, then other lengths
    lp_lengths = set()
    if parent is not None and parent_n <= 2048:
        lp_lengths.add(parent_n)
    lp_lengths.update([512, 1024])
    lp_lengths = sorted(lp_lengths)

    for n in lp_lengths:
        if time.monotonic() > phase1_end - 30:
            break

        # Initial sequence for this length
        if parent is not None:
            if parent_n == n:
                a_init = parent.copy()
            else:
                a_init = _interp_profile(parent, n)
        else:
            a_init = _make_profile(n, "linear")

        a_init = np.maximum(a_init, 0)
        s = a_init.sum()
        if s > 0:
            a_init = a_init / s

        sc_init = _score(a_init)
        update_best(a_init, sc_init, f"lp_init n={n}")

        t_alloc = min(180.0, phase1_end - time.monotonic() - 10.0)
        if t_alloc < 10:
            break

        a_best, sc_best = _lp_iterate(a_init, t_alloc, t_init=0.35,
                                       verbose=True, label=f"LP n={n}")
        update_best(a_best, sc_best, f"LP n={n}")

    print(f"Phase 1 done. Best: {best_score:.10f} (n={best_seq.size if best_seq is not None else 0})")

    # ============ PHASE 2: Gradient polishing ============
    print(f"\n--- Phase 2: Gradient polishing ---")
    phase2_end = min(deadline, t_start + 900.0)

    if best_seq is not None and time.monotonic() < phase2_end:
        n_best = best_seq.size

        # Try gradient polish at current best length
        t_alloc = min(150.0, phase2_end - time.monotonic() - 10.0)
        if t_alloc > 20:
            p_b, sc_b = _gradient_polish(best_seq, t_alloc, rng, lr=0.05,
                                         beta_init_frac=0.10, beta_decay=0.73,
                                         var_weight=0.0, name="polish_main")
            update_best(p_b, sc_b, "gradient_polish")

        # Try at a slightly different length
        for alt_n in [n_best * 2, n_best // 2]:
            if time.monotonic() > phase2_end - 30:
                break
            if alt_n < 128 or alt_n > 4096:
                continue
            a_alt = _interp_profile(best_seq, alt_n)
            if a_alt is None or a_alt.size != alt_n:
                continue
            sc_alt = _score(a_alt)
            update_best(a_alt, sc_alt, f"interp n={alt_n}")

            t_alloc = min(60.0, phase2_end - time.monotonic() - 5.0)
            if t_alloc > 15:
                p_b2, sc_b2 = _gradient_polish(a_alt, t_alloc, rng, lr=0.04,
                                               beta_init_frac=0.08, beta_decay=0.75)
                update_best(p_b2, sc_b2, f"gradient n={alt_n}")

    print(f"Phase 2 done. Best: {best_score:.10f} (n={best_seq.size if best_seq is not None else 0})")

    # ============ PHASE 3: Multi-restart ============
    print(f"\n--- Phase 3: Multi-restart ---")
    phase3_end = min(deadline, t_start + 1400.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        n_best = best_seq.size
        n_starts = 4
        remaining = phase3_end - time.monotonic()
        per_start = min(80.0, remaining / max(1, n_starts))

        for i in range(n_starts):
            if time.monotonic() > phase3_end - 5:
                break

            # Different perturbation scales
            noise_scales = [0.1, 0.2, 0.35, 0.5]
            ns = noise_scales[i % len(noise_scales)]

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_start = _softmax_from_z(z)

            t_alloc = min(per_start, phase3_end - time.monotonic() - 5.0)
            if t_alloc < 10:
                break

            # Alternate between LP and gradient
            if i % 2 == 0:
                p_b, sc_b = _lp_iterate(p_start, t_alloc, t_init=0.3,
                                         verbose=False, label=f"restart{i}")
            else:
                p_b, sc_b = _gradient_polish(p_start, t_alloc, rng, lr=0.05,
                                             beta_init_frac=0.09, beta_decay=0.74,
                                             var_weight=0.01)
            update_best(p_b, sc_b, f"restart{i}")

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # ============ PHASE 4: Final refinement ============
    print(f"\n--- Phase 4: Final refinement ---")
    phase4_end = deadline

    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = best_seq.size
        t_alloc = min(60.0, phase4_end - time.monotonic() - 5.0)

        if t_alloc > 10:
            # Very small learning rate for fine-tuning
            p_fin, sc_fin = _gradient_polish(best_seq, t_alloc, rng, lr=0.015,
                                             beta_init_frac=0.04, beta_decay=0.82,
                                             var_weight=0.0, name="final")
            update_best(p_fin, sc_fin, "final_finetune")

    # ============ FINAL: Validation and return ============
    print(f"\n--- Final ---")
    if best_seq is not None:
        final_seq = np.asarray(best_seq, dtype=np.float64).copy()
        final_n = final_seq.size

        # Validate with exact convolution for small sequences
        if final_n <= 2048:
            c_exact = np.convolve(final_seq, final_seq)
            s_exact = float(final_seq.sum())
            sc_exact = float(2.0 * final_n * c_exact.max() / (s_exact * s_exact))
            print(f"Exact score (n={final_n}): {sc_exact:.10f}")

            # Try evaluate_sequence for authoritative check
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
            except Exception:
                pass

        # Ensure valid output
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            fallback = _make_profile(256, "linear")
            final_seq = fallback

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    print("Using fallback profile")
    fallback = _make_profile(512, "linear")
    return [float(x) for x in fallback]
