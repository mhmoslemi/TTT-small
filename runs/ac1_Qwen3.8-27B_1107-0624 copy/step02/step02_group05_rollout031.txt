    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_direction(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_min_peak(f):
    """LP: find g minimizing max_k (f*g)[k] subject to g>=0, sum(g)=1."""
    n = len(f)
    # Variables: g[0..n-1], t (the peak value)
    # Minimize t
    # Constraints: (f*g)[k] <= t for all k, sum(g) = 1, g >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    # Inequality: (f*g)[k] - t <= 0 for each k
    A_conv = make_conv_matrix(f)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x[:n], 0.0), float(res.x[-1])
    except Exception:
        pass
    return None, None


def interpolate_profile(f, n_new):
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def spline_from_control_points(ctrl, n):
    """Evaluate a cubic B-spline or simple piecewise-linear from control points."""
    m = len(ctrl)
    if m == 1:
        return np.full(n, ctrl[0])
    x_ctrl = np.linspace(0, 1, m)
    x_eval = np.linspace(0, 1, n)
    return np.interp(x_eval, x_ctrl, ctrl)


def make_profile(n, profile_type, param=0.0):
    t = np.linspace(0, 1, n)
    if profile_type == 'uniform':
        f = np.ones(n)
    elif profile_type == 'ramp':
        f = 1.0 + param * t
    elif profile_type == 'ramp2':
        f = 1.0 + param * t * t
    elif profile_type == 'U':
        f = 1.0 + param * (4 * (t - 0.5) ** 2)
    elif profile_type == 'W':
        f = 1.0 + param * np.cos(4 * np.pi * t) ** 2
    elif profile_type == 'gauss':
        f = np.exp(-8 * (t - 0.5) ** 2) + 0.1
    elif profile_type == 'dual':
        f = np.exp(-20 * t ** 2) + np.exp(-20 * (t - 1) ** 2) + 0.05
    elif profile_type == 'tri':
        f = np.minimum(t, 1 - t) + 0.1
    elif profile_type == 'asym_ramp':
        f = t ** param + 0.01
    elif profile_type == 'zigzag':
        f = 1.0 + param * np.abs(np.sin(3 * np.pi * t))
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def lp_optimize(f_init, deadline, rng, max_iters=1000, verbose=False):
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # Alternating LP strategies
        if it % 3 == 0:
            g, peak_t = solve_lp_min_peak(f)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
            else:
                g = solve_lp_direction(f, M)
                if g is None:
                    break
                g_sum = g.sum()
                if g_sum < 1e-15:
                    break
                g = g / g_sum
        else:
            g = solve_lp_direction(f, M)
            if g is None:
                # Fallback
                g = f + 0.1 * rng.normal(size=n)
                g = normalize_nonneg(g)
                if g is None:
                    break
            else:
                g_sum = g.sum()
                if g_sum < 1e-15:
                    break
                g = g / g_sum

        # Line search
        best_t = 0.0
        best_f_try = f
        best_M_try = M
        for t_val in [0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:
            f_try = (1.0 - t_val) * f + t_val * g
            f_try = normalize_nonneg(f_try)
            if f_try is None:
                continue
            M_try = compute_M(f_try)
            if M_try < best_M_try - 1e-14:
                best_M_try = M_try
                best_f_try = f_try
                best_t = t_val

        if best_t > 0:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 25:
            # Perturb and restart
            noise_scale = 0.2 + 0.3 * (no_imp / 100.0)
            f_new = best_f + noise_scale * rng.standard_normal(n)
            f_new = normalize_nonneg(f_new)
            if f_new is not None and compute_M(f_new) < best_M * 1.5:
                f = f_new
                M = compute_M(f)
            else:
                f = best_f + 0.1 * rng.standard_normal(n)
                f = normalize_nonneg(f)
                if f is None:
                    break
                M = compute_M(f)
            no_imp = 0

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best_M={best_M:.10f}")

    return best_f, best_M


def soft_max_objective(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def adam_optimize(f_init, deadline, rng, lr=0.05, max_iters=10000, beta_init=None):
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = np.convolve(f, f)
    M = float(c.max())
    if beta_init is None:
        beta = 0.3 * M
    else:
        beta = beta_init

    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline and adam_t < max_iters:
        L, w = soft_max_objective(c, beta)
        # Gradient: sum_k w_k * C_k * f
        # C_k * f is the "correlation" of w with f, shifted
        # (C_k @ f)[i] = sum_j w[k-j]*f[j] ... let me use the convolution formulation
        # d/d f[i] of (f*f)[k] = f[k-i] + (if k-i != i) f[k-i] ... 
        # Actually (f*f)[k] = sum_i f[i]*f[k-i]
        # d/d f[i] = f[k-i] + f[k-i] (if i != k-i) or 2*f[k-i] (if i = k-i)
        # More carefully: d/d f[i] sum_j f[j]f[k-j] = f[k-i] (from j=i term) + f[i] (from k-j=i, j=k-i term)
        # = f[k-i] + f[k-i] = 2*f[k-i] (when i != k-i)
        # = 2*f[i] (when i = k-i, i.e., k = 2i)
        # Actually for i = k-i: the term f[i]*f[k-i] = f[i]^2, derivative = 2*f[i]
        # For i != k-i: derivative of f[i]*f[k-i] w.r.t. f[i] is f[k-i], and derivative of f[k-i]*f[i] w.r.t. f[i] is f[k-i]
        # But wait, in the sum sum_j f[j]*f[k-j], the term involving f[i] is:
        #   j=i: f[i]*f[k-i]
        #   k-j=i, i.e., j=k-i: f[k-i]*f[i]
        # These are the same term! So d/d f[i] = f[k-i] (from j=i) + f[k-i] (from j=k-i) = 2*f[k-i]
        # Wait no: d/d f[i] of f[j]*f[k-j] is nonzero only when j=i (gives f[k-i]) or k-j=i i.e. j=k-i (gives f[i]... no)
        # Let me be very careful. (f*f)[k] = sum_{j} f[j] * f[k-j] (with valid indices)
        # d/d f[i] (f*f)[k] = sum_j [delta(j,i)*f[k-j] + f[j]*delta(k-j,i)]
        # = f[k-i] + f[k-i] (if k-i is a valid index and k-i != i)
        # = 2*f[k-i] generally
        # But if k-i = i (i.e., k=2i), then we get f[i] + f[i] = 2*f[i], same thing.
        # So d/d f[i] (f*f)[k] = 2*f[k-i] (when 0 <= k-i <= n-1), else 0.
        
        # So gradient of L = sum_k w[k] * 2 * f shifted by k
        # grad[i] = 2 * sum_k w[k] * f[k-i] (valid k-i)
        # This is 2 * (w_reversed * f)[i] in some sense... 
        # Actually let's compute it as: for each k, the contribution to grad[i] is 2*w[k]*f[k-i]
        # = 2 * sum_k w[k] * f[k-i]
        # Let j = k-i: = 2 * sum_j w[i+j] * f[j] (when 0 <= i+j <= 2n-2)
        # This is 2 * (correlation of w and f)[i]
        
        # Compute using FFT
        N_fft = 1
        while N_fft < 2 * n - 1:
            N_fft <<= 1
        fw = np.fft.rfft(w, N_fft)
        fp = np.fft.rfft(f, N_fft)
        # corr[i] = sum_j w[i+j]*f[j] = ifft(w * conj(f)) ... 
        # Actually: sum_j w[i+j]*f[j]. Let's define g[m] = w[m] for m=0..2n-2.
        # corr[i] = sum_j g[i+j]*f[j] where j=0..n-1 and 0<=i+j<=2n-2
        # = sum_j g[i+j]*f[j]
        # This is the cross-correlation of g and f, evaluated at i.
        # Using FFT: corr = ifft(conj(fft(g)) * fft(f))[...] 
        # Or: corr[i] = sum_j g[i+j]*f[j] = (g * f_reversed)[n-1-i] ... 
        # Let me just use direct computation for clarity (n is small enough)
        
        # Actually for n up to 512, direct is 512*512 = 262K per iteration, which is fine.
        # But FFT is faster. Let me use the standard cross-correlation formula.
        # corr[i] = sum_j w[i+j] * f[j] for j=0..n-1, i=0..n-1
        # This equals sum_j w[i+j] * f[j]
        # = cross_correlation(w, f)[i] where cross_correlation(w,f)[i] = sum_j w[i+j]*f[j]
        
        # Using FFT: cross_correlation can be computed as ifft(fft(w) * conj(fft(f)))
        # Wait, that gives sum_j w[j+i]*conj(f[j]) which for real f is sum_j w[j+i]*f[j]
        # Hmm, let me just be careful.
        
        # corr[i] = sum_{j=0}^{n-1} w[i+j] * f[j]
        # Define W = fft(w, N), F = fft(f, N)
        # ifft(W * conj(F))[i] = sum_j w[j] * f[(j-i) mod N]
        # That's not quite what we want.
        
        # Let me use: corr[i] = sum_j w[i+j]*f[j]
        # = sum_j w[i+j]*f[j]
        # Let's define h[k] = w[k] for k=0..2n-2, 0 otherwise
        # corr[i] = sum_j h[i+j]*f[j] = (h * f_rev)[?] 
        # = cross-correlation of h and f
        # In numpy: np.correlate(h, f, mode='valid') gives sum_j h[i+j]*f[j] for i=0..len(h)-len(f)
        # But that's O(n^2). For n=512, that's 262K which is fine for a few thousand iterations.
        
        # Actually let's just use the direct method for n <= 512:
        grad_direct = np.zeros(n)
        # grad[i] = 2 * sum_j w[i+j] * f[j] for j = 0..n-1, 0<=i+j<=2n-2
        # w has length 2n-1, f has length n
        # For each i: sum over j=0..min(n-1, 2n-2-i) of w[i+j]*f[j]
        for k_idx in range(2 * n - 1):
            wk = w[k_idx]
            if wk < 1e-15:
                continue
            # Contribution to grad[i] from this k: 2*wk*f[k-i] for i where 0<=k-i<=n-1
            # i ranges from max(0, k-n+1) to min(n-1, k)
            i_lo = max(0, k_idx - n + 1)
            i_hi = min(n - 1, k_idx)
            if i_lo <= i_hi:
                grad_direct[i_lo:i_hi+1] += 2.0 * wk * f[k_idx - np.arange(i_lo, i_hi + 1)]
        
        # Add regularization to prevent f from going to 0
        vp = grad_direct + 1e-6 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-13:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta
        if adam_t % 200 == 0:
            beta *= 0.85
        if no_imp > 500:
            beta *= 1.5  # increase smoothness to escape
        if no_imp > max(500, 2 * n):
            # Restart
            z = 0.7 * best_z + 0.3 * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-6, 0.3 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_f, best_M


def simulated_annealing(f_init, deadline, rng, max_iters=50000):
    f = normalize_nonneg(f_init.copy())
    if f is None:
        return f_init, compute_M(f_init)
    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    T = 0.5 * M
    T_min = 1e-10
    cool_rate = 0.9995
    no_imp = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # Random perturbation in log-space
        z = np.log(np.maximum(f, 1e-12))
        # Perturb a random subset
        n_perturb = max(1, n // 10)
        idx = rng.choice(n, size=n_perturb, replace=False)
        z[idx] += rng.standard_normal(n_perturb) * 0.3
        z = np.clip(z, -30, 30)
        z -= z.mean()
        e = np.exp(z)
        f_try = e / e.sum()
        M_try = compute_M(f_try)

        delta = M_try - M
        if delta < 0 or rng.random() < math.exp(-max(0, delta) / max(T, T_min)):
            f = f_try
            M = M_try
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        T = max(T_min, T * cool_rate)

        if no_imp > 2000:
            T = max(T, 0.3 * M)
            no_imp = 0

    return best_f, best_M


def de_optimize(n, n_ctrl, deadline, rng, popsize=40, maxiter=50):
    """Differential evolution on control points, interpolating to length n."""
    def objective(ctrl):
        f = spline_from_control_points(np.maximum(ctrl, 0.01), n)
        f = f / f.sum()
        return compute_score(f)

    bounds = [(0.0, 5.0)] * n_ctrl
    try:
        result = differential_evolution(
            objective, bounds, seed=rng.integers(10000),
            maxiter=maxiter, popsize=popsize, tol=1e-8,
            mutation=(0.3, 1.0), recombination=0.9,
            updating='deferred', polish=False,
            init='latinhypercube'
        )
        if result.success or result.fun < np.inf:
            f = spline_from_control_points(np.maximum(result.x, 0.01), n)
            f = f / f.sum()
            return f, compute_score(f)
    except Exception:
        pass
    return None, np.inf


def get_starting_points(n, rng, count=25):
    starts = []
    # Analytical profiles
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        starts.append(make_profile(n, 'ramp', p))
        starts.append(make_profile(n, 'asym_ramp', p))
    for p in [1.0, 2.0, 3.0, 5.0]:
        starts.append(make_profile(n, 'U', p))
        starts.append(make_profile(n, 'W', p))
    starts.append(make_profile(n, 'uniform'))
    starts.append(make_profile(n, 'dual'))
    starts.append(make_profile(n, 'tri'))
    starts.append(make_profile(n, 'gauss'))
    for p in [1.0, 2.0, 3.0]:
        starts.append(make_profile(n, 'zigzag', p))
    for p in [0.5, 1.0, 1.5, 2.0]:
        starts.append(make_profile(n, 'ramp2', p))
    # Random smooth profiles via Fourier
    for _ in range(count):
        n_fourier = 6
        coeffs = rng.uniform(-1, 1, 2 * n_fourier + 1)
        t = np.linspace(0, 1, n)
        f = np.ones(n) * 2.0
        for k in range(n_fourier):
            f += coeffs[2*k+1] * np.cos(2 * np.pi * (k+1) * t)
            f += coeffs[2*k+2] * np.sin(2 * np.pi * (k+1) * t)
        f = np.maximum(f, 0.01)
        f = f / f.sum()
        starts.append(f)
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass

    # ============ PHASE 0: Quick evaluation of starting points ============
    print("Phase 0: Evaluating starting points...")
    n0 = 256
    if parent is not None:
        n0 = len(parent)
        parent_interp = interpolate_profile(parent, 256)
        sc = compute_score(parent_interp)
        try_update(parent_interp, sc)
        print(f"  Parent at n=256: {sc:.10f}")

    starts = get_starting_points(256, rng, count=20)
    for s in starts:
        sc = compute_score(s)
        try_update(s, sc)
    print(f"  Phase 0 best: {best_score:.10f}")

    # ============ PHASE 1: Differential Evolution (global) ============
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\nPhase 1: DE global search, time budget {phase1_end - t_start:.0f}s")

    # DE on 20 control points for n=256
    n_ctrl = 20
    alloc1 = (phase1_end - time.monotonic()) / 2
    f_de, sc_de = de_optimize(256, n_ctrl, time.monotonic() + max(5, alloc1), rng,
                              popsize=30, maxiter=40)
    if f_de is not None:
        try_update(f_de, sc_de)
        print(f"  DE n=256, ctrl={n_ctrl}: {sc_de:.10f}, best={best_score:.10f}")

    # DE with more control points
    n_ctrl2 = 32
    alloc1b = phase1_end - time.monotonic() - 5
    if alloc1b > 30:
        f_de2, sc_de2 = de_optimize(256, n_ctrl2, time.monotonic() + alloc1b * 0.6, rng,
                                    popsize=30, maxiter=30)
        if f_de2 is not None:
            try_update(f_de2, sc_de2)
            print(f"  DE n=256, ctrl={n_ctrl2}: {sc_de2:.10f}, best={best_score:.10f}")

    # DE at n=128
    if time.monotonic() < phase1_end - 20:
        f_de3, sc_de3 = de_optimize(128, 16, phase1_end - 10, rng,
                                    popsize=25, maxiter=30)
        if f_de3 is not None:
            f_de3_256 = interpolate_profile(f_de3, 256)
            try_update(f_de3_256, compute_score(f_de3_256))
            print(f"  DE n=128->256: {sc_de3:.10f}, best={best_score:.10f}")

    print(f"  Phase 1 best: {best_score:.10f}")

    # ============ PHASE 2: LP-based refinement ============
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\nPhase 2: LP refinement, time budget {phase2_end - time.monotonic():.0f}s")

    # Collect top starting points from all sources
    candidates = []
    if parent is not None:
        candidates.append(interpolate_profile(parent, 256))
    if f_de is not None:
        candidates.append(f_de)
    if f_de2 is not None:
        candidates.append(f_de2)
    if best_seq is not None:
        candidates.append(interpolate_profile(best_seq, 256))
    # Add a few random starts
    candidates.append(make_profile(256, 'ramp', 2.0))
    candidates.append(make_profile(256, 'U', 2.0))

    # Deduplicate (keep unique-ish)
    unique_candidates = []
    for c in candidates:
        if c is None:
            continue
        is_unique = True
        for uc in unique_candidates:
            if np.linalg.norm(c - uc) / max(np.linalg.norm(c), 1e-10) < 0.1:
                is_unique = False
                break
        if is_unique:
            unique_candidates.append(c)

    # Run LP on each candidate with time allocation
    n_cands = min(6, len(unique_candidates))
    time_per = (phase2_end - time.monotonic()) / max(1, n_cands) - 2
    time_per = max(5, min(time_per, 180))

    lp_results = []
    for i in range(n_cands):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(time_per, remaining - 5)
        if alloc < 5:
            break
        f_best, M_best = lp_optimize(unique_candidates[i], time.monotonic() + alloc, rng,
                                     max_iters=500)
        sc = compute_score(f_best)
        try_update(f_best, sc)
        lp_results.append((sc, f_best))
        print(f"  LP cand {i+1}/{n_cands}: {sc:.10f}, best={best_score:.10f}")

    lp_results.sort(key=lambda x: x[0])
    print(f"  Phase 2 best: {best_score:.10f}")

    # ============ PHASE 3: Adam polish on best LP results ============
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\nPhase 3: Adam polish, time budget {phase3_end - time.monotonic():.0f}s")

    top_k = min(4, len(lp_results))
    if top_k == 0:
        top_k = 1
        lp_results = [(compute_score(best_seq), best_seq)] if best_seq is not None else [(2.0, make_profile(256, 1.0))]

    time_per_adam = (phase3_end - time.monotonic()) / max(1, top_k) - 2
    time_per_adam = max(10, min(time_per_adam, 200))

    adam_results = []
    for i in range(top_k):
        remaining = phase3_end - time.monotonic()
        if remaining < 10:
            break
        alloc = min(time_per_adam, remaining - 5)
        if alloc < 5:
            break
        f_best, M_best = adam_optimize(lp_results[i][1], time.monotonic() + alloc, rng,
                                       lr=0.04, max_iters=20000)
        sc = compute_score(f_best)
        try_update(f_best, sc)
        adam_results.append((sc, f_best))
        print(f"  Adam {i+1}/{top_k}: {sc:.10f}, best={best_score:.10f}")

    adam_results.sort(key=lambda x: x[0])

    # Also run Adam on the DE results directly
    if f_de is not None and time.monotonic() < phase3_end - 20:
        f_best, M_best = adam_optimize(f_de, phase3_end - 10, rng, lr=0.05, max_iters=10000)
        sc = compute_score(f_best)
        try_update(f_best, sc)
        print(f"  Adam on DE: {sc:.10f}, best={best_score:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # ============ PHASE 4: Simulated annealing / random walk ============
    phase4_end = min(deadline, t_start + 1600.0)
    print(f"\nPhase 4: Simulated annealing, time budget {phase4_end - time.monotonic():.0f}s")

    if best_seq is not None:
        remaining = phase4_end - time.monotonic()
        if remaining > 30:
            f_best, M_best = simulated_annealing(best_seq, time.monotonic() + remaining * 0.7,
                                                 rng, max_iters=100000)
            sc = compute_score(f_best)
            try_update(f_best, sc)
            print(f"  SA round 1: {sc:.10f}, best={best_score:.10f}")

            # Another SA round from the updated best
            if time.monotonic() < phase4_end - 10:
                f_best2, M_best2 = simulated_annealing(best_seq, phase4_end - 5,
                                                       rng, max_iters=50000)
                sc2 = compute_score(f_best2)
                try_update(f_best2, sc2)
                print(f"  SA round 2: {sc2:.10f}, best={best_score:.10f}")

    print(f"  Phase 4 best: {best_score:.10f}")

    # ============ PHASE 5: Multi-n exploration ============
    phase5_end = deadline
    print(f"\nPhase 5: Multi-n exploration, time budget {phase5_end - time.monotonic():.0f}s")

    if best_seq is not None:
        for n_try in [128, 192, 384, 512]:
            remaining = phase5_end - time.monotonic()
            if remaining < 30:
                break
            f_interp = interpolate_profile(best_seq, n_try)
            # Quick Adam on this
            allocate = min(remaining * 0.4, 120)
            if allocate < 10:
                break
            f_best, M_best = adam_optimize(f_interp, time.monotonic() + allocate, rng,
                                           lr=0.04, max_iters=10000)
            sc = compute_score(f_best)
            try_update(f_best, sc)
            print(f"  n={n_try}: {sc:.10f}, best={best_score:.10f}")

            # Quick LP
            remaining = phase5_end - time.monotonic()
            if remaining > 30:
                allocate_lp = min(remaining * 0.5, 100)
                if allocate_lp > 10:
                    f_best_lp, M_best_lp = lp_optimize(f_best, time.monotonic() + allocate_lp,
                                                       rng, max_iters=300)
                    sc_lp = compute_score(f_best_lp)
                    try_update(f_best_lp, sc_lp)
                    print(f"  n={n_try} LP: {sc_lp:.10f}, best={best_score:.10f}")

    # ============ PHASE 6: Final LP refinement on best ============
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\nPhase 6: Final LP, {remaining:.0f}s remaining")
        n_best = len(best_seq)
        f_best, M_best = lp_optimize(best_seq, deadline, rng, max_iters=1000)
        sc = compute_score(f_best)
        try_update(f_best, sc)
        print(f"  Final LP: {sc:.10f}")

    # Validate and return
    if best_seq is None:
        best_seq = make_profile(256, 1.5)
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_profile(256, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
