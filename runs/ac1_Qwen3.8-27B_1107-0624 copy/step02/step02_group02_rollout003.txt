"""

Uses LP-based descent, subgradient steps, and soft-max Adam polishing.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=True):
    """Main LP-based optimization loop with improved step control."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = float(np.convolve(f, f).max())
        if stagnant > 0:
            M = best_M

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Try multiple step sizes (backtracking)
        best_f_new = None
        best_M_new = M
        best_t = None
        for t_try in [t, t * 0.5, t * 0.75, t * 1.25]:
            t_try = max(0.001, min(1.0, t_try))

            s_new = f_new.sum()
            if s_new < 1e-10:
                continue

            M_new = float(np.convolve(f_new, f_new).max())

            if best_M_new < best_M - 1e-13:

                stagnant = 0
            else:

            iteration += 1

                print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
        else:

def subgrad_step(f, lr, deadline_check=None):
    """Take a subgradient step on the max convolution value."""
    conv_f = np.convolve(f, f)
    idx = int(np.argmax(conv_f))
    n = len(f)
    # Subgradient of conv[idx] w.r.t. f[i]
    # conv[idx] = sum_{i} f[i]*f[idx-i] (valid indices)
    sg = np.zeros(n)
    for i in range(n):
        j = idx - i
        if 0 <= j < n:
            sg[i] += f[j]  # derivative of f[i]*f[j] w.r.t. f[i]
        if i != j and 0 <= i < n:
            sg[i] += 0  # already counted above if j==i
    # Actually, conv[idx] = sum_i f[i]*f[idx-i] where both indices valid
    # d(conv[idx])/df[i] = f[idx-i] (if valid) + f[i-idx] (if valid and i-idx != idx-i)
    sg = np.zeros(n)
    for i in range(n):
        j1 = idx - i
        if 0 <= j1 < n:
            sg[i] += f[j1]
        j2 = i - idx
        if 0 <= j2 < n and j2 != j1:
            sg[i] += f[j2]
    # Normalize and step
    sg = sg / (np.max(np.abs(sg)) + 1e-15)
    f_new = f - lr * sg
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    return f_new


def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

                if no_imp > 500:

        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0

def make_linear_profile(n, eta=1.5):
    """Monotone linear profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    """Cosine-based profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_hann_profile(n):
    """Hann window profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_profile(n, sigma=0.3):
    """Gaussian profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_triangular_profile(n, peak=0.5):
    """Triangular profile with adjustable peak."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(t - peak) / max(peak, 1 - peak, 1e-6)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

    # Main LP optimization
    lp_deadline = min(phase1_end, time.monotonic() + 400.0)
    result = lp_optimize(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"After main LP: score={sc:.10f}")

    # Try other starting profiles with shorter time each
    profiles = get_starting_profiles(n_main, rng)
    for name, p in profiles:
        if time.monotonic() > phase1_end - 5:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.1:
            continue
        per_deadline = min(phase1_end, time.monotonic() + 30.0)
        result = lp_optimize(p, per_deadline, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: Soft-max Adam polish (200s) ============
    phase2_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 2: polishing at n={n_polish}, current best={best_score:.10f}")

        for lr, beta_init in [(0.1, None), (0.05, 0.0005), (0.08, 0.001)]:
            if time.monotonic() > phase2_end - 5:
                break
            per_deadline = min(phase2_end, time.monotonic() + 60.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.001 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=lr, beta_init=beta_init)
            update(result, sc)
            print(f"  Adam lr={lr}, beta_init={beta_init}: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: n=1024 optimization (500s) ============
    n_large = 1024
    phase3_end = min(deadline, t_start + 1400.0)

    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 3 start: n={n_large}, score={sc:.10f}")

        # LP at n=1024
        lp_deadline3 = min(phase3_end, time.monotonic() + 350.0)
        result = lp_optimize(p_large, lp_deadline3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After large LP: score={sc:.10f}")

        # Soft-max polish at n=1024
        if time.monotonic() < phase3_end - 10:
            per_deadline = min(phase3_end, time.monotonic() + 100.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.0005 * rng.normal(size=n_large)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=0.06)
            update(result, sc)
            print(f"  Large Adam polish: score={sc:.10f}")

        # Try a few different starts at n=1024
        for name, p_small in profiles[:6]:
            if time.monotonic() > phase3_end - 5:
                break
            p_lp = interpolate_profile(p_small, n_large)
            sc_p = compute_score(p_lp)
            if sc_p >= best_score + 0.1:
                continue
            per_deadline = min(phase3_end, time.monotonic() + 40.0)
            result = lp_optimize(p_lp, per_deadline, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Final polish on best (remaining time) ============
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = best_seq.size
        per = min(deadline - 5, time.monotonic() + 80.0)
        print(f"Phase 4: final polish at n={n_final}, time_left={per:.0f}s")

        p0 = best_seq.copy()
        p0 = p0 + 0.0003 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = logit_anneal(p0, per, rng, lr=0.03)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 2.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
