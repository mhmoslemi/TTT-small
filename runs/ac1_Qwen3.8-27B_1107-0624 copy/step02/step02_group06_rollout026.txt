    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))

        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]

    # In scipy: min -1^T g s.t. A g <= M*1, 0 <= g

    if res.success:
        return res.x
    return None

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    elif mn < 0:
        c = c - mn
    return c


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve(a)
    mv = float(c.max())
    return float(2.0 * n * mv / (s * s))


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[k, j] = a[k - j]
    return A


def mp_lp_step(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)

        return res.x
    return None

def soft_max_adam_polish(a0, deadline, rng, lr=0.08, beta_frac=0.2):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= np.mean(z)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve(a)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)
    best_a = a.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Soft max
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s
        L = mval + beta * np.log(s)

        # Gradient: corr_weights
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        gp = 2.0 * h[n - 1:2 * n - 1]

        vp = gp + 1e-7 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= np.mean(gz)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= np.mean(z)

        e = np.exp(z)
        a = e / e.sum()
        c = full_convolve(a)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.02, lr * 0.95)

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.7)

        if no_imp > max(300, 3 * n):

            z -= np.mean(z)
            a = np.exp(z)
            a = a / a.sum()
            c = full_convolve(a)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, compute_score(best_a)

            orig = a[i]
            best_val = orig
            best_sc = best_score
            for factor in [1.3, 1.15, 0.85, 0.7, 1.5, 0.5]:
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                a[i] = new_val
                a = a / a.sum()
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            a[i] = best_val
            a = a / a.sum()
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_a = a.copy()
                improved = True
                no_imp_total = 0
            else:
                no_imp_total += 1
        if improved:
            step_frac = max(0.03, step_frac * 0.9)
        if no_imp_total > 3 * n:
            break
    return best_a, best_score


def make_profiles(n, rng):
    profs = []
    t = np.linspace(0, 1, n)
    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    # Exponential
    for eta in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(eta * t)
        profs.append(a / a.sum())
    # Power
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = t ** p + 1e-8
        profs.append(a / a.sum())
    # Quadratic
    u = 2.0 * t - 1.0
    for a1, a2 in [(1.5, -1.0), (2.0, 0.5), (1.0, -0.5), (2.5, -2.0)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    # Sigmoid
    for eta in [5.0, 10.0, 20.0]:
        a = 1.0 / (1.0 + np.exp(-eta * (t - 0.5)))
        profs.append(a / a.sum())
    # Random smooth
    for _ in range(10):
        coeffs = rng.normal(0, 1, 8)
        a = np.ones(n)
        for k, ck in enumerate(coeffs, 1):
            a += ck * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-8)
        profs.append(a / a.sum())
    return profs


def interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, a)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(a, sc, label=""):
        nonlocal best_seq, best_score
        if a is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_score = float(sc)
            best_seq = np.asarray(a, dtype=np.float64).copy()
            print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} ({label})")

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_seq(parent)
    except Exception:
        pass

    print("Phase 0: Initialization")
    candidates = []  # list of (score, sequence)

    if parent is not None:
        sc = compute_score(parent)
        update_best(parent, sc, "parent")
        candidates.append((sc, parent.copy()))
        # Interpolate to different lengths
        for tn in [256, 1024]:
            if parent.size != tn:
                pi = interpolate_to_length(parent, tn)
                pi = normalize_seq(pi)
                if pi is not None:
                    sc_i = compute_score(pi)
                    update_best(pi, sc_i, f"parent_n{tn}")
                    candidates.append((sc_i, pi.copy()))

    # Try diverse profiles at n=512
    n_main = 512
    if parent is None or parent.size != n_main:
        for p0 in make_profiles(n_main, rng)[:15]:
            sc = compute_score(p0)
            candidates.append((sc, p0))
            if sc < best_score:
                update_best(p0, sc, "profile")

    candidates.sort(key=lambda x: x[0])
    print(f"  After init: best={best_score:.10f}, {len(candidates)} candidates")

    # Phase 1: MP optimization at n=512 (main phase)
    print("Phase 1: MP optimization at n=512")
    phase1_end = min(deadline, t_start + 800.0)
    top_k = min(5, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > phase1_end:
            break
        sc_c, a_c = candidates[ci]
        if a_c.size != n_main:
            a_c = normalize_seq(interpolate_to_length(a_c, n_main))
            if a_c is None:
                continue
        per_iter = min(phase1_end, time.monotonic() + max(60.0, (phase1_end - time.monotonic()) / max(1, top_k - ci) * 0.8))
        print(f"  MP on candidate {ci} (score={sc_c:.6f}), budget={per_iter - time.monotonic():.0f}s")
        a_best, sc_best = mp_optimize(a_c, per_iter, max_iters=3000, verbose=(ci == 0))
        sc_final = compute_score(a_best)
        update_best(a_best, sc_final, f"mp_c{ci}")
        candidates.append((sc_final, a_best.copy()))
        candidates.sort(key=lambda x: x[0])

    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: Gradient polish on top candidates
    print("Phase 2: Gradient polish")
    phase2_end = min(deadline, t_start + 1200.0)
    top_g = min(4, len(candidates))
    for gi in range(top_g):
        if time.monotonic() > phase2_end:
            break
        sc_c, a_c = candidates[gi]
        if a_c.size != n_main:
            a_c = normalize_seq(interpolate_to_length(a_c, n_main))
            if a_c is None:
                continue
        per_iter = min(phase2_end, time.monotonic() + max(30.0, (phase2_end - time.monotonic()) / max(1, top_g - gi) * 0.7))
        a_g, sc_g = soft_max_adam_polish(a_c, per_iter, rng, lr=0.06, beta_frac=0.15)
        sc_final = compute_score(a_g)
        update_best(a_g, sc_final, f"grad_c{gi}")
        candidates.append((sc_final, a_g.copy()))
        candidates.sort(key=lambda x: x[0])

    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: Try n=1024
    print("Phase 3: Optimization at n=1024")
    phase3_end = min(deadline, t_start + 1500.0)
    n_large = 1024
    for ci in range(min(3, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, a_c = candidates[ci]
        a_big = normalize_seq(interpolate_to_length(a_c, n_large))
        if a_big is None:
            continue
        per_iter = min(phase3_end, time.monotonic() + 60.0)
        a_best, sc_best = mp_optimize(a_big, per_iter, max_iters=1000, verbose=False)
        sc_final = compute_score(a_best)
        update_best(a_best, sc_final, f"n1024_c{ci}")
        # Quick gradient polish
        per_grad = min(phase3_end, time.monotonic() + 30.0)
        a_g, sc_g = soft_max_adam_polish(a_best, per_grad, rng, lr=0.05, beta_frac=0.1)
        sc_final2 = compute_score(a_g)
        update_best(a_g, sc_final2, f"n1024_grad_c{ci}")
        candidates.append((sc_final2, a_g.copy()))
        candidates.sort(key=lambda x: x[0])

    print(f"  After phase 3: best={best_score:.10f}")

    # Phase 4: Coordinate descent polish
    print("Phase 4: Coordinate descent polish")
    phase4_end = min(deadline, t_start + 1720.0)
    for ci in range(min(3, len(candidates))):
        if time.monotonic() > phase4_end:
            break
        sc_c, a_c = candidates[ci]
        if a_c.size > 1024:
            continue
        per_iter = min(phase4_end, time.monotonic() + 40.0)
        a_cd, sc_cd = coordinate_descent_polish(a_c, per_iter, rng, max_sweeps=20)
        sc_final = compute_score(a_cd)
        update_best(a_cd, sc_final, f"cd_c{ci}")
        candidates.append((sc_final, a_cd.copy()))
        candidates.sort(key=lambda x: x[0])

    print(f"  After phase 4: best={best_score:.10f}")

    # Phase 5: Final MP polish on absolute best
    print("Phase 5: Final MP polish")
    phase5_end = min(deadline, t_start + 1770.0)
    if best_seq is not None and best_seq.size <= 512:
        per_iter = phase5_end
        a_final, sc_final = mp_optimize(best_seq, per_iter, max_iters=2000, verbose=True)
        sc_final2 = compute_score(a_final)
        update_best(a_final, sc_final2, "final_mp")

    print(f"  After phase 5: best={best_score:.10f}")

    # Final verification
    print("Phase 6: Final verification")
    if best_seq is not None:
        # Clamp and normalize
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s > 0:
            best_seq = best_seq / s
        # Exact score
        c_exact = np.convolve(best_seq, best_seq)
        sc_exact = float(2.0 * len(best_seq) * float(c_exact.max()) / (s * s))
        print(f"  Exact score: {sc_exact:.10f}")
        # Try evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(sc_eval):
                print(f"  Evaluator score: {sc_eval:.10f}")
                sc_exact = min(sc_exact, sc_eval)
        except Exception as e:
            print(f"  Evaluator error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("  FALLBACK: using linear profile")
    fallback = np.array([1.0 + 1.5 * i / 255.0 for i in range(256)])
    fallback = fallback / fallback.sum()
    return [float(x) for x in fallback]
