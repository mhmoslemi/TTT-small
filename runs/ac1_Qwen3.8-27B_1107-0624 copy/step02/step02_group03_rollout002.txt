```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: LP at small n (128, 256), gradient-based optimization at larger n (512, 1024).
Uses FFT-accelerated convolution and log-sum-exp smoothing for the max.
"""

import numpy as np
import time
from scipy.optimize import linprog

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M, timeout=10.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_iterate(f, max_iters=500, t_init=0.4, t_min=0.02, deadline=None, verbose=True):
    n = len(f)
    f = np.maximum(f, 0.0).astype(np.float64)
    f = f / f.sum()
    
    M = float(np.convolve(f, f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = t_init
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g_raw = lp_direction(f, M)
        if g_raw is None:
            break
        g = normalize(g_raw)
        if g is None:
            break
        
        g_score = 2.0 * n * float(np.convolve(g, g).max())
        if g_score < best_score * 0.999:
            f = g
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            f_new = normalize((1-t_step)*f + t_step*g)
            if f_new is None:
                break
            M_new = float(np.convolve(f_new, f_new).max())
            score_new = 2.0 * n * M_new
            if score_new < best_score - 1e-12:
                best_f = f_new
                best_score = score_new
                best_M = M_new
                f = f_new
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.995)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 20 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 200:
                    break
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")
    
    return best_f, best_score

def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def make_profile(n, shape='linear', param=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0  # in [-1, 1]
    if shape == 'linear':
        a = 1.0 + param * u
    elif shape == 'quadratic':
        a = 1.0 + param * u**2
    elif shape == 'triangle':
        a = 1.0 - param * np.abs(u)
    elif shape == 'bell':
        a = 1.0 + param * np.exp(-3.0 * u**2)
    elif shape == 'bathtub':
        a = 1.0 + param * u**2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def grad_logsumexp_fft(p, beta):
    """
    Compute gradient of logsumexp(convolve(p,p), beta) w.r.t. p.
    Uses FFT for efficiency.
    Returns gradient of length n.
    """
    n = len(p)
    c = np.convolve(p, p)  # length 2n-1
    c_max = c.max()
    x = (c - c_max) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    
    # grad[j] = 2 * sum_{m=0}^{n-1} w[m+j] * p[m]
    # Use FFT: this is (p cross-correlated with w) [first n values]
    # (p ⋆ w)[j] = sum_m p[m] * w[m+j]
    # FFT: ifft(conj(FFT(p)) * FFT(w))
    
    N = 1
    while N < 3 * n:
        N *= 2
    
    P = np.fft.rfft(p, N)
    W = np.fft.rfft(w, N)
    H = np.fft.irfft(np.conj(P) * W, N)
    grad = 2.0 * H[:n]
    
    return grad

def adam_optimize(p0, deadline, lr=0.05, verbose=True, beta_init=0.01, beta_min=0.001):
    """
    Adam gradient descent in logit space to minimize 2*n*max(conv(p,p)) with sum(p)=1.
    Uses log-sum-exp for smooth max approximation with annealed temperature.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-10)
    p = p / p.sum()
    
    z = np.log(p)
    z -= np.mean(z)
    
    b1, b2, eps = 0.9, 0.999, 1e-10
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    
    c = np.convolve(p, p)
    max_c = float(c.max())
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    beta = beta_init
    base_lr = lr
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p, p)
        
        grad = grad_logsumexp_fft(p, beta)
        
        # Add small regularization
        grad += 1e-10 * p
        
        # Logit space gradient
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m[:] = 0
            v[:] = 0
            it = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        
        p_new = np.exp(z - z.max())
        p_sum = p_new.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p = np.exp(z - z.max())
            p = p / p.sum()
            continue
        p_new = p_new / p_sum
        
        c_new = np.convolve(p_new, p_new)
        score = 2.0 * n * float(c_new.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_p = p_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Anneal beta
        if it % 100 == 0:
            beta = max(beta_min, beta * 0.9)
        
        # Periodic restart
        if no_imp > 2000:
            rng = np.random.default_rng(it)
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m[:] = 0
            v[:] = 0
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(0.005, 0.1 * float(np.convolve(
                np.exp(z - z.max()) / np.exp(z - z.max()).sum(),
                np.exp(z - z.max()) / np.exp(z - z.max()).sum()
            ).max()))
        
        p = p_new
        
        if verbose and it % 500 == 0:
            print(f"    Adam iter {it}: score={best_score:.8f}, beta={beta:.6f}, lr={lr:.6f}")
    
    return best_p, best_score
