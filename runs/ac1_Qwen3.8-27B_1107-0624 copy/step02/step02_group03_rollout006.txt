def build_conv_matrix(f):

def lp_direction(f, M, timeout=10.0):

def interpolate_seq(a, n_new):

    x_new = np.linspace(0.0, 1.0, n_new)

    return normalize(a_new)


def cosine_basis_profile(n, coeffs):

"""
Hybrid LP + Nesterov soft-max gradient optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses adaptive temperature annealing,
momentum-accelerated gradient steps, and multi-phase strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """Solve LP: maximize sum(g) s.t. g >= 0, convolve(f,g)[x] <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def soft_max_weights(c, beta):
    """Compute soft-max weights for correlation values c."""
    max_c = c.max()
    x = (c - max_c) / beta
    x = np.clip(x, -200, 200)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    return w


def compute_grad(a, beta):
    """
    Compute gradient of 2n * soft_max(conv(a,a), beta) w.r.t. a.
    Returns (grad, soft_max_value).
    """
    n = len(a)
    c = np.convolve(a, a)
    w = soft_max_weights(c, beta)
    # h[k] = sum_x w[x] * a[x-k] = correlate(w, a, 'valid')
    h = np.correlate(w, a, mode='valid')
    grad = 4.0 * n * h
    max_c = c.max()
    soft_max_val = max_c + beta * np.log(np.exp((c - max_c) / beta).sum() / len(c))
    return grad, soft_max_val


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return None
    return v / s


def nesterov_polish(a0, deadline, beta_init=0.08, lr_init=0.003, momentum=0.9,
                     beta_decay=0.997, restart_every=400, verbose=True):
    """
    Nesterov-accelerated gradient descent on soft-max objective.
    a0 normalized to sum=1.
    Returns (best_a, best_max_c).
    """
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)

    c_init = np.convolve(a, a)
    max_init = float(c_init.max())
    beta = beta_init if beta_init is not None else 0.05 * max_init

    best_a = a.copy()
    best_max_c = max_init
    lr = lr_init
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        # Nesterov lookahead
        x = a + momentum * v

        # Project to simplex
        x = project_simplex(x)
        if x is None:
            x = a.copy()
            v = np.zeros(n)
            continue

        # Gradient at x
        grad, sm_val = compute_grad(x, beta)

        # Update velocity
        v = momentum * v - lr * grad
        a_new = x + v

        # Project
        a_new = project_simplex(a_new)
        if a_new is None:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(1e-7, lr * 0.5)
            continue

        # Evaluate
        c_new = np.convolve(a_new, a_new)
        max_new = float(c_new.max())

        # Adaptive lr: reduce if making progress is negative
        if max_new > best_max_c * 1.005:
            lr = max(1e-7, lr * 0.7)
            v *= 0.5

        if max_new < best_max_c - 1e-13:
            best_max_c = max_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Decay beta (cooling)
        beta = max(1e-6, beta * beta_decay)

        # Periodic restart with perturbation
        if no_improve >= restart_every:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            for attempt in range(5):
                scale = 0.05 * (attempt + 1)
                pert = rng.normal(0, scale, size=n)
                a_try = project_simplex(best_a + pert)
                if a_try is not None:
                    c_try = np.convolve(a_try, a_try)
                    if float(c_try.max()) < best_max_c:
                        best_max_c = float(c_try.max())
                        best_a = a_try.copy()
                        a = best_a.copy()
                        break
            # Reset optimizer state
            a = project_simplex(best_a + 0.02 * rng.normal(0, 1, size=n))
            if a is None:
                a = best_a.copy()
            v = np.zeros(n)
            lr = lr_init
            beta = 0.05 * float(np.convolve(a, a).max())
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov iter {it}: max_c={best_max_c:.8f}, "
                  f"beta={beta:.6f}, lr={lr:.2e}, no_imp={no_improve}, t={elapsed:.0f}s")

        # Termination
        if beta < 1e-5 and no_improve > 200:
            break

    return best_a, best_max_c


def logit_adam_polish(a0, deadline, lr=0.01, beta_init=None, verbose=True):
    """
    Adam gradient descent in logit (softplus) space.
    a0 normalized to sum=1.
    Returns (best_a, best_max_c).
    """
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-15)
    a0_safe = a0_safe / a0_safe.sum()

    # Initialize logit: z = log(a), centered
    z = np.log(a0_safe)
    z -= z.mean()

    c_init = np.convolve(a0_safe, a0_safe)
    max_init = float(c_init.max())
    if beta_init is None:
        beta_init = 0.05 * max_init
    beta = beta_init

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    vv = np.zeros(n)
    it = 0

    best_a = a0_safe.copy()
    best_max_c = max_init
    no_improve = 0
    lr_cur = lr
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        it += 1

        # Compute softmax probabilities from z
        z_shifted = z - z.max()
        p = np.exp(z_shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = np.log(best_a)
            z -= z.mean()
            m = np.zeros(n)
            vv = np.zeros(n)
            it = 0
            continue
        p = p / p_sum

        # Gradient of soft-max objective w.r.t. a
        c = np.convolve(p, p)
        w = soft_max_weights(c, beta)
        h = np.correlate(w, p, mode='valid')
        grad_a = 4.0 * n * h

        # Convert to logit-space gradient: dz = p * (grad_a - sum(p*grad_a))
        dot = float(np.dot(p, grad_a))
        gz = p * (grad_a - dot)

        if not np.all(np.isfinite(gz)):
            z = np.log(best_a)
            z -= z.mean()
            m = np.zeros(n)
            vv = np.zeros(n)
            it = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        vv = b2 * vv + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = vv / (1.0 - b2 ** it)
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= z.mean()

        # Get new p
        p = np.exp(z - z.max())
        ps = p.sum()
        if ps < 1e-30:
            z = np.log(best_a)
            z -= z.mean()
            continue
        p = p / ps

        c = np.convolve(p, p)
        max_c = float(c.max())

        if max_c < best_max_c - 1e-13:
            best_max_c = max_c
            best_a = p.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 300 == 0:
                lr_cur = max(0.002, lr_cur * 0.9)

        # Decay beta
        beta = max(1e-5, beta * 0.998)

        # Restart
        if no_improve > 1500:
            rng = np.random.default_rng(it)
            z = 0.7 * np.log(np.maximum(best_a, 1e-15)) + 0.5 * rng.normal(size=n)
            z -= z.mean()
            z = np.clip(z, -60, 60)
            m = np.zeros(n)
            vv = np.zeros(n)
            it = 0
            lr_cur = lr
            beta = 0.05 * max_c
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Logit-Adam iter {it}: max_c={best_max_c:.8f}, "
                  f"beta={beta:.6f}, lr={lr_cur:.4f}, no_imp={no_improve}, t={elapsed:.0f}s")

        if beta < 1e-4 and no_improve > 500:
            break

    return best_a, best_max_c


def lp_refine(a, deadline, max_iters=500, verbose=True):
    """LP-based refinement with multi-step-size search."""
    n = len(a)
    a = a.copy()
    c = np.convolve(a, a)
    M = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * M

    no_improve = 0
    t0 = time.monotonic()
    it = 0

    # Try to time one LP solve
    t_lp0 = time.monotonic()
    _ = lp_direction(a, M, timeout=5)
    t_lp_est = time.monotonic() - t_lp0
    if verbose:
        print(f"    LP est. time: {t_lp_est*1000:.0f}ms")

    while it < max_iters and time.monotonic() < deadline:
        it += 1
        g_raw = lp_direction(a, M, timeout=30)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Multi-step-size search
        best_local = None
        best_local_score = np.inf
        best_local_M = M
        for t in [0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            if a_try is None:
                continue
            c_try = np.convolve(a_try, a_try)
            M_try = float(c_try.max())
            sc_try = 2.0 * n * M_try
            if sc_try < best_local_score:
                best_local_score = sc_try
                best_local = a_try
                best_local_M = M_try

        if best_local is None:
            break

        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best_a = best_local.copy()
            a = best_local.copy()
            M = best_local_M
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: score={best_score:.8f}, M={M:.6f}, t={elapsed:.0f}s")

    return best_a, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_pwu_profile(n, order=3):
    """
    Power-weighted uniform profile: a_i ∝ (1 + α_i)
    where α_i comes from a Prolate-like shape.
    """
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    # Use a raised-cosine * linear combination
    a = 1.0 + 1.2 * u + 0.3 * u**2 - 0.1 * u**3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimization:
      Phase 1: LP at n=256 (fast convergence)
      Phase 2: Nesterov soft-max gradient at n=512
      Phase 3: LP at n=512 (precise direction)
      Phase 4: Logit-Adam polish at n=512
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 10.0)

    # Get initial sequence
    best_seq = None
    best_score = np.inf

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except (NameError, Exception):
        pass

    if best_seq is None:
        n_init = 512
        t = np.linspace(0, 1, n_init)
        u = 2 * t - 1
        profiles = []
        for eta in [0.8, 1.2, 1.5, 2.0, 2.5, 3.0]:
            a = 1.0 + eta * u
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())
        # Quadratic profiles
        for c1, c2 in [(1.0, 0.3), (1.5, -0.2), (0.8, 0.5)]:
            a = 1.0 + c1 * u + c2 * u**2
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())
        for p in profiles:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.8f}")

    # Ensure we work at n=512
    target_n = 512
    if len(best_seq) != target_n:
        best_seq = interpolate_seq(best_seq, target_n)
        best_score = internal_score(best_seq)

    # ========== Phase 1: LP at n=256 ==========
    deadline_p1 = min(deadline, t_start + 250.0)
    if time.monotonic() < deadline_p1 - 10:
        a256 = interpolate_seq(best_seq, 256)
        if a256 is not None:
            sc256 = internal_score(a256)
            print(f"\n=== Phase 1: LP at n=256, initial score={sc256:.8f} ===")
            a256_best, sc256_best = lp_refine(a256, deadline_p1, max_iters=500)
            if sc256_best < 2.0:
                a512_from_lp = interpolate_seq(a256_best, target_n)
                if a512_from_lp is not None:
                    sc512 = internal_score(a512_from_lp)
                    print(f"  Phase 1 done: n=256 score={sc256_best:.8f}, "
                          f"upsampled to 512: {sc512:.8f}")
                    if sc512 < best_score:
                        best_score = sc512
                        best_seq = a512_from_lp
                    else:
                        print(f"  Upampling did not help, keeping previous best.")

    # ========== Phase 2: Nesterov soft-max at n=512 ==========
    deadline_p2 = min(deadline, t_start + 950.0)
    if time.monotonic() < deadline_p2 - 10:
        print(f"\n=== Phase 2: Nesterov soft-max at n=512, score={internal_score(best_seq):.8f} ===")
        a_best, max_c_best = nesterov_polish(
            best_seq, deadline_p2,
            beta_init=0.10, lr_init=0.004, momentum=0.85,
            beta_decay=0.9965, restart_every=350, verbose=True)
        sc = 2.0 * target_n * max_c_best
        print(f"  Nesterov result: max_c={max_c_best:.8f}, score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_best

    # ========== Phase 3: LP at n=512 ==========
    deadline_p3 = min(deadline, t_start + 1450.0)
    if time.monotonic() < deadline_p3 - 10:
        print(f"\n=== Phase 3: LP at n=512, score={internal_score(best_seq):.8f} ===")
        a_best, sc_best = lp_refine(best_seq, deadline_p3, max_iters=300)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best
            print(f"  Phase 3 result: score={best_score:.8f}")
        else:
            print(f"  Phase 3: no improvement (best={internal_score(best_seq):.8f})")

    # ========== Phase 4: Final Logit-Adam polish ==========
    deadline_p4 = deadline
    if time.monotonic() < deadline_p4 - 5:
        print(f"\n=== Phase 4: Logit-Adam polish, score={internal_score(best_seq):.8f} ===")
        a_best, max_c_best = logit_adam_polish(
            best_seq, deadline_p4,
            lr=0.015, beta_init=0.04, verbose=True)
        sc = 2.0 * target_n * max_c_best
        print(f"  Logit-Adam result: max_c={max_c_best:.8f}, score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_best
            print(f"  New best: {best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        t = np.linspace(0, 1, 512)
        best_seq = np.maximum(1.0 + 1.5 * (2 * t - 1), 0.0)
        s = best_seq.sum()
    best_seq = best_seq / s

    # Verify with internal_score one more time
    final_check = internal_score(best_seq)
    print(f"Output validation: score={final_check:.8f}")

    return [float(x) for x in best_seq]
