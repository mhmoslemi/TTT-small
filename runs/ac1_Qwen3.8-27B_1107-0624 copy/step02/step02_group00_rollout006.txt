def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_scaled(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    """Full score: 2n * max(conv(a,a)) / sum(a)^2"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-20:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def _lp_min_peak(a):
    """
    Solve: minimize t subject to conv(a,b)_i <= t for all i, sum(b)=1, b>=0, t>=0.
    Returns (b, t, dual_info) where b is the optimal direction.
    """
    n = len(a)
    A = _make_conv_matrix(a)
    # Variables: [b_0, ..., b_{n-1}, t]
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[n] = 1.0  # minimize t

    # Inequality: A @ b - t <= 0 for each row
    A_ub = np.zeros((2 * n - 1, nvar))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b = res.x[:n]
        t_val = res.x[n]
        # Extract dual info (which constraints are active)
        duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
        return b, t_val, duals
    return None, None, None

def _lbfgs_polish(a, deadline, max_iter=5000):
    """
    L-BFGS style optimization on smooth max of conv(a,a), with simplex projection.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _score_scaled(a)
    
    beta = 0.05 * float(_fft_convolve(a).max())
    beta_min = 1e-6
    lr = 1.0  # We'll use a fixed-point iteration style
    
    # Simple momentum-based method with smooth max gradient
    v = np.zeros(n)
    momentum = 0.9
    no_imp = 0
    base_momentum = momentum

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        smooth_val, grad, c, w = _smooth_max_and_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            continue

        # Project gradient onto simplex tangent space
        grad_proj = grad - float(np.mean(grad))
        
        v = momentum * v - lr * grad_proj
        a_new = _project_simplex(a + v)
        
        # Also try a few different step sizes
        best_local = _score_scaled(a_new)
        if best_local >= _score_scaled(a):
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = _project_simplex(a + frac * v)
                sc_try = _score_scaled(a_try)
                if sc_try < best_local:
                    best_local = sc_try
                    a_new = a_try
                    break

        if best_local < best_score - 1e-12:
            best_score = best_local
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                momentum = max(0.5, momentum * 0.95)
                lr = max(0.01, lr * 0.9)
            if no_imp > 2000:
                a = best_a.copy()
                v = np.zeros(n)
                momentum = base_momentum
                lr = 1.0
                no_imp = 0

        a = a_new

        # Anneal beta
        if it % 100 == 0:
            beta = max(beta_min, beta * 0.7)

    return best_a


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _lp_refine_improved(a, deadline, max_iter=50000):
    """
    LP-based refinement using direct peak minimization LP + line search.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-20:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_scaled(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Solve LP to find direction
        b, t_val, _ = _lp_min_peak(a)
        if b is None:
            break

        # Check if the LP solution is meaningfully different from a
        diff = float(np.max(np.abs(b - a)))
        if diff < 1e-8:
            no_imp += 1
            if no_imp > 50:
                break
            continue

        # Line search for best blend
        best_alpha = _line_search_blend(a, b)
        
        a_new = (1.0 - best_alpha) * a + best_alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = _score_scaled(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break

        a = a_new

    return best_a


def _hann_window(n):
    t = np.linspace(0.0, np.pi, n)
    a = 0.5 * (1.0 + np.cos(t))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _gaussian_window(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _triangular_window(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - 2.0 * np.abs(t - 0.5)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Optimized search for minimum 2n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Multiple diverse initializations
    2. LP-based refinement (direct peak minimization)
    3. L-BFGS/gradient polish
    4. Best result returned
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define initializations for n=512
    n_target = 512
    inits = []
    inits_names = []

    if parent is not None:
        p512 = _interpolate_to_length(parent, n_target)
        inits.append(p512)
        inits_names.append("parent_512")
        
        # Perturbed parent
        for _ in range(2):
            p_pert = p512 * (1.0 + 0.05 * rng.normal(size=n_target))
            p_pert = np.maximum(p_pert, 0)
            p_pert = p_pert / p_pert.sum()
            inits.append(p_pert)
            inits_names.append("parent_perturb")

    inits.append(_hann_window(n_target))
    inits_names.append("hann")
    
    inits.append(_gaussian_window(n_target, 0.35))
    inits_names.append("gauss_0.35")
    
    inits.append(_gaussian_window(n_target, 0.25))
    inits_names.append("gauss_0.25")
    
    inits.append(_triangular_window(n_target))
    inits_names.append("triangular")

    # Also try n=256 (smaller, faster LP)
    n_small = 256
    if parent is not None:
        p256 = _interpolate_to_length(parent, n_small)
        inits.append(p256)
        inits_names.append("parent_256")
    inits.append(_hann_window(n_small))
    inits_names.append("hann_256")

    # Try n=1024 for one initialization
    n_large = 1024
    if parent is not None:
        p1024 = _interpolate_to_length(parent, n_large)
        inits.append(p1024)
        inits_names.append("parent_1024")

    # Phase 1: LP refinement on each initialization
    # Allocate time based on number of inits
    n_inits = len(inits)
    time_per_lp = min(200.0, (deadline - time.monotonic() - 120.0) / max(1, n_inits))
    
    print(f"Phase 1: LP refinement, {n_inits} inits, ~{time_per_lp:.0f}s each")
    
    for idx, (a0, name) in enumerate(zip(inits, inits_names)):
        if time.monotonic() > deadline - 120:
            break
        
        sc0 = _score_scaled(a0)
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
        
        lp_budget = time_per_lp
        lp_end = time.monotonic() + lp_budget
        print(f"  [{name}] n={a0.size}, init_score={sc0:.8f}, budget={lp_budget:.0f}s")
        
        a_lp = _lp_refine_improved(a0, lp_end, max_iter=20000)
        sc_lp = _score_scaled(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  [{name}] -> LP score={sc_lp:.10f}")

    # Phase 2: Gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_budget = min(180.0, deadline - time.monotonic() - 60.0)
        if gp_budget > 10:
            gp_end = time.monotonic() + gp_budget
            print(f"Phase 2: Gradient polish, budget={gp_budget:.0f}s")
            a_gp = _lbfgs_polish(best_seq, gp_end, max_iter=10000)
            sc_gp = _score_scaled(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  Gradient polish: score={sc_gp:.10f}")

    # Phase 3: Another LP pass on the polished result
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp_budget = min(120.0, deadline - time.monotonic() - 30.0)
        if lp_budget > 10:
            lp_end = time.monotonic() + lp_budget
            print(f"Phase 3: Final LP pass, budget={lp_budget:.0f}s")
            a_final = _lp_refine_improved(best_seq, lp_end, max_iter=10000)
            sc_final = _score_scaled(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP: score={sc_final:.10f}")

    # Phase 4: One more gradient polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        gp_budget = min(60.0, deadline - time.monotonic() - 15.0)
        if gp_budget > 5:
            gp_end = time.monotonic() + gp_budget
            a_p2 = _lbfgs_polish(best_seq, gp_end, max_iter=5000)
            sc_p2 = _score_scaled(a_p2)
            if np.isfinite(sc_p2) and sc_p2 < best_score:
                best_score = sc_p2
                best_seq = a_p2.copy()
            print(f"  Final polish: score={sc_p2:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _hann_window(256)
        best_score = _score(best_seq)

    # Ensure non-negative and valid
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        # Normalize for evaluate_sequence (it handles any sum, but let's keep it reasonable)
        pass
    else:
        final_seq = _hann_window(256)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
