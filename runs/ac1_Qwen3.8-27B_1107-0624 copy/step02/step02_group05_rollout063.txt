def lse_gradient(f, beta):

def make_conv_matrix(f):

def solve_lp(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    n = len(f)

            return np.maximum(res.x, 0.0)
    except:
        pass
    return None

And dc_k/df_i = 2*f_{k-i} if 0 <= k-i < n, else 0.

So dL/df_i = 2 * sum_{k: 0<=k-i<n} w_k * f_{k-i} = 2 * sum_{j=0}^{n-1} f_j * w_{j+i}

To compute all gradients at once:
grad[i] = 2 * sum_j f_j * w_{j+i} for i = 0, ..., n-1

This is 2 * cross_correlation(f, w)[0:n] where cross_correlation(f, w)[i] = sum_j f_j * w_{j+i}.

Using convolution: cross_correlation(f, w)[i] = convolve(f[::-1], w)[i + n - 1]

        if g is None:
            no_imp += 1
            if no_imp > 30:
                # Perturb and retry
                f = simplex_project(f + 0.1 * rng.normal(size=n))
                M = float(np.max(np.convolve(f, f)))
                no_imp = 0
            it += 1
            continue

        g = g / g.sum()

        # Line search on blending parameter
        best_t = 0.0
        best_M_try = M
        for t in [0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
            f_try = simplex_project((1.0 - t) * f + t * g)
            M_try = float(np.max(np.convolve(f_try, f_try)))
            if M_try < best_M_try - 1e-12:
                best_M_try = M_try
                best_t = t

        if best_t > 0:
            f = simplex_project((1.0 - best_t) * f + best_t * g)
            M = best_M_try
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 40:
            f = simplex_project(0.8 * best_f + 0.2 * np.abs(rng.normal(size=n)))
            M = float(np.max(np.convolve(f, f)))
            no_imp = 0

        it += 1
        if verbose and it % 100 == 0:
            print(f"    LP it={it}: M={M:.10f}, best_M={best_M:.10f}")

    best_score = compute_score(best_f)
    return best_f, best_score


def make_profiles(n, rng):
    """Generate diverse starting profiles, each normalized to sum=1, non-negative."""
    profiles = []
    t = np.linspace(0, 1, n)

    # Uniform
    profiles.append(np.ones(n) / n)

    # Linear ramps (various slopes and directions)
    for slope in [0.5, 1.0, 2.0, 4.0, 8.0]:
        f = 1.0 + slope * t
        profiles.append(f / f.sum())
        f = 1.0 + slope * (1.0 - t)
        profiles.append(f / f.sum())

    # Quadratic (U-shaped and inverted-U)
    for a in [1, 2, 4, -1, -2]:
        f = 1.0 + a * (2 * t - 1) ** 2
        f = np.maximum(f, 0.01)
        profiles.append(f / f.sum())

    # Cosine profiles
    for k in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            f = 1.0 + amp * np.cos(2 * np.pi * k * t)
            f = np.maximum(f, 0.01)
            profiles.append(f / f.sum())

    # Alternating (comb-like)
    for period in [2, 3, 4]:
        f = 1.0 + 0.3 * np.cos(2 * np.pi * np.arange(n) / period)
        f = np.maximum(f, 0.05)
        profiles.append(f / f.sum())

    # Gaussian bumps
    for mu in [0.2, 0.5, 0.8]:
        for sig in [0.1, 0.2, 0.35]:
            f = np.exp(-0.5 * ((t - mu) / sig) ** 2) + 0.01
            profiles.append(f / f.sum())

    # Dual bump (edges high)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.05
    profiles.append(f / f.sum())

    # Random smooth profiles (Fourier-based)
    for _ in range(15):
        coeffs = rng.normal(size=10)
        f = 1.0
        for k in range(5):
            f = f + 0.4 * coeffs[2 * k] * np.cos(2 * np.pi * (k + 1) * t)
            f = f + 0.4 * coeffs[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t)
        f = np.maximum(f, 0.01)
        profiles.append(f / f.sum())

    # Random with structure (low-frequency + high-frequency)
    for _ in range(5):
        low = 1.0 + 0.5 * rng.normal() * np.cos(2 * np.pi * t * rng.uniform(1, 4))
        high = 0.2 * rng.normal() * np.cos(2 * np.pi * t * rng.uniform(10, 30))
        f = low + high
        f = np.maximum(f, 0.01)
        profiles.append(f / f.sum())

    return profiles


def interpolate_to_length(f, n_new):
    """Linearly interpolate a profile to a new length, preserving sum=1."""
    n_old = len(f)
    if n_old == n_new:
        return f.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = float(f_new.sum())
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return f_new / s


def adam_polish(f_init, deadline, rng, verbose=False):
    """Adam optimizer in log-space for final polishing."""
    f = np.maximum(f_init, 1e-15).astype(np.float64)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = 0.05
    beta = max(1e-10, 0.5 * float(np.max(np.convolve(f, f))))

    best_f = f.copy()
    best_score = compute_score(f)
    no_imp = 0

    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        c_max = float(np.max(c))
        z_lse = beta * (c - c_max)
        ez = np.exp(z_lse)
        w = ez / float(np.sum(ez))

        # Gradient of LSE w.r.t. f
        conv = np.convolve(f[::-1], w)
        grad_f = 2.0 * conv[n - 1:2 * n - 1]

        # Convert to log-space gradient: dz_i = f_i * (grad_f_i - sum(f*grad_f))
        dot = float(np.dot(f, grad_f))
        grad_z = f * (grad_f - dot)
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(best_f)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        z = np.clip(z, -50, 50)
        z -= z.mean()
        f = np.exp(z)
        f = f / f.sum()

        score = compute_score(f)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(lr * 0.9, 0.005)

        if t_adam % 200 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > 500:
            z = 0.8 * np.log(best_f) + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            f = np.exp(z)
            f = f / f.sum()
            beta = max(1e-10, 0.5 * float(np.max(np.convolve(f, f))))
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr = 0.05
            no_imp = 0

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2*n*max(convolve(f,f))/sum(f)^2.
    
    Phase 1: LSE gradient descent from diverse starts at n=256
    Phase 2: LP-based refinement on top candidates
    Phase 3: Adam polish
    Phase 4: Try other sequence lengths
    Phase 5: Final combined refinement
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.array(f, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: score={sc:.10f}, n={len(f)} ***")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                par_score = compute_score(parent)
                print(f"Parent: n={len(parent)}, score={par_score:.10f}")
                try_update(parent, par_score)
    except Exception:
        pass

    # === Phase 1: LSE-GD at n=256 from diverse starting points ===
    n1 = 256
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: LSE-GD at n={n1} (time budget: {phase1_end - t_start:.0f}s) ===")

    starts1 = make_profiles(n1, rng)
    if parent is not None:
        if len(parent) == n1:
            starts1.append(parent.copy())
        else:
            starts1.append(interpolate_to_length(parent, n1))

    results1 = []
    time_per = (phase1_end - t_start) / len(starts1)
    for i, s in enumerate(starts1):
        if time.monotonic() >= phase1_end:
            break
        alloc = max(time_per, 3.0)
        run_end = min(phase1_end, time.monotonic() + alloc)
        f_best, sc = lse_gd(s, run_end, rng)
        try_update(f_best, sc)
        results1.append((sc, f_best))
        if (i + 1) % 10 == 0:
            print(f"  {i+1}/{len(starts1)} done. Current best: {best_score:.10f}")

    results1.sort(key=lambda x: x[0])
    print(f"Phase 1 complete. Best: {best_score:.10f}")

    # === Phase 2: LP refinement on top LSE-GD results ===
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: LP refinement (time budget: {phase2_end - t_start:.0f}s) ===")

    top_k2 = min(10, len(results1))
    results2 = []
    if top_k2 > 0:
        time_per2 = (phase2_end - t_start) / top_k2
        for i in range(top_k2):
            if time.monotonic() >= phase2_end:
                break
            alloc = max(time_per2, 10.0)
            run_end = min(phase2_end, time.monotonic() + alloc)
            f_best, sc = lp_optimize(results1[i][1], run_end, rng)
            try_update(f_best, sc)
            results2.append((sc, f_best))
            if (i + 1) % 3 == 0:
                print(f"  LP {i+1}/{top_k2}: score={sc:.10f}, best={best_score:.10f}")

    # Merge and sort
    all_results = results1 + results2
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 2 complete. Best: {best_score:.10f}")

    # === Phase 3: Adam polish on best candidates ===
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\n=== Phase 3: Adam polish (time budget: {phase3_end - t_start:.0f}s) ===")

    top_k3 = min(6, len(all_results))
    results3 = []
    if top_k3 > 0:
        time_per3 = (phase3_end - t_start) / top_k3
        for i in range(top_k3):
            if time.monotonic() >= phase3_end:
                break
            alloc = max(time_per3, 10.0)
            run_end = min(phase3_end, time.monotonic() + alloc)
            f_best, sc = adam_polish(all_results[i][1], run_end, rng)
            try_update(f_best, sc)
            results3.append((sc, f_best))
            print(f"  Adam {i+1}/{top_k3}: score={sc:.10f}, best={best_score:.10f}")

    all_results = all_results + results3
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 3 complete. Best: {best_score:.10f}")

    # === Phase 4: Alternate LSE-GD and LP on the best solution ===
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 4: Alternating refinement (time budget: {phase4_end - t_start:.0f}s) ===")

    if best_seq is not None:
        f_cur = best_seq.copy()
        n_cur = len(f_cur)
        round_num = 0
        while time.monotonic() < phase4_end:
            round_num += 1
            # LSE-GD step
            if time.monotonic() < phase4_end - 20:
                run_end = min(phase4_end - 20, time.monotonic() + 60.0)
                f_best, sc = lse_gd(f_cur, run_end, rng)
                try_update(f_best, sc)
                f_cur = f_best
                print(f"  Round {round_num} LSE-GD: {sc:.10f}")

            # LP step
            if time.monotonic() < phase4_end - 10:
                run_end = min(phase4_end - 10, time.monotonic() + 40.0)
                f_best, sc = lp_optimize(f_cur, run_end, rng)
                try_update(f_best, sc)
                f_cur = f_best
                print(f"  Round {round_num} LP: {sc:.10f}")

            # Adam step
            if time.monotonic() < phase4_end - 5:
                run_end = min(phase4_end - 5, time.monotonic() + 30.0)
                f_best, sc = adam_polish(f_cur, run_end, rng)
                try_update(f_best, sc)
                f_cur = f_best
                print(f"  Round {round_num} Adam: {sc:.10f}")

    # === Phase 5: Try different sequence lengths ===
    phase5_end = deadline
    print(f"\n=== Phase 5: Other lengths (time budget: {phase5_end - t_start:.0f}s) ===")

    if best_seq is not None:
        for n_try in [512, 128, 1024]:
            if time.monotonic() >= phase5_end - 30:
                break
            f_ext = interpolate_to_length(best_seq, n_try)
            print(f"  Trying n={n_try}...")

            # Quick LSE-GD
            run_end = min(phase5_end - 20, time.monotonic() + 50.0)
            f_best, sc = lse_gd(f_ext, run_end, rng)
            try_update(f_best, sc)
            print(f"    LSE-GD n={n_try}: {sc:.10f}")

            # Quick LP
            if time.monotonic() < phase5_end - 15:
                run_end = min(phase5_end - 15, time.monotonic() + 30.0)
                f_best2, sc2 = lp_optimize(f_best, run_end, rng)
                try_update(f_best2, sc2)
                print(f"    LP n={n_try}: {sc2:.10f}")

    # === Final: validate and return ===
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = compute_score(best_seq)

    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = np.ones(256)
        s = 256.0
    best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal answer: n={len(best_seq)}, score={final_score:.10f}")

    # Verify with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence confirms: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    # One last attempt: direct LSE-GD from the best solution with remaining time
    if time.monotonic() < deadline and best_seq is not None:
        run_end = deadline
        f_best, sc = lse_gd(best_seq, run_end, rng)
        if np.isfinite(sc) and sc < final_score:
            best_seq = f_best
            final_score = sc
            print(f"Extra LSE-GD improved to: {final_score:.10f}")

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s > 0:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Returning sequence of length {len(best_seq)}, score={compute_score(best_seq):.10f}")
    return [float(x) for x in best_seq]
