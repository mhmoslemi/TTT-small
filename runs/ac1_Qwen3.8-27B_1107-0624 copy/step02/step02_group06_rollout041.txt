import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)  # numerical safety
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def solve_lp(lambda_vec, n):
    """
    Solve: max 1^T a s.t. A a <= 1, a >= 0
    where A[j,k] = lambda[j+k] for valid indices.
    """
    # Build constraint matrix A
    # Row j: A[j,k] = lambda[j+k] for k=0,...,n-1
    # lambda has length 2n-1 (indices 0 to 2n-2)
    A = np.zeros((n, n), dtype=np.float64)
    for j in range(n):
        A[j, :] = lambda_vec[j:j+n]
    
    # linprog: minimize c^T x s.t. A_ub x <= b_ub, x >= 0
    # We want to maximize 1^T a, so minimize -1^T a
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs', 
                  options={'presolve': True})
    
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        return None

        return a, internal_score(a)
    
    a_new = a_new / a_new.sum()
    score = internal_score(a_new)
    
    return a_new, score

    """
    Run the LP-based iterative optimization.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:

    a = a / s
    
    n = a.size
    best_a = a.copy()

    beta = beta_init * best_score  # scale with the objective
    
    no_improve = 0
    
    while time.monotonic() < deadline:

            no_improve = 0
    
    return best_a, best_score

    where A[j,m] = lambda[j+m].

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def gradient_optimize(a0, deadline, rng, lr=0.08):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a = normalize(a0)
    if a is None:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    base_lr = lr

    c = fft_convolve(a)
    c_max = float(c.max())
    beta = 0.15 * c_max

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * c_max)
    no_imp = 0

    while time.monotonic() < deadline:
        c_shifted = (c - c_max) / beta
        exp_c = np.exp(c_shifted)
        w = exp_c / exp_c.sum()

        grad = 2.0 * corr_weights(w, a)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue

        z = np.clip(z, -50, 50)
        z -= z.mean()
        a = softmax_from_z(z)
        c = fft_convolve(a)
        c_max = float(c.max())
        sc = float(2.0 * n * c_max)

        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.02, lr * 0.95)

        if t_adam % 60 == 0:
            beta = max(1e-14, beta * 0.85)

        if no_imp > max(300, 4 * n):
            z = 0.7 * best_z + 0.3 * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = softmax_from_z(z)
            c = fft_convolve(a)
            c_max = float(c.max())
            beta = 0.15 * c_max
            m.fill(0); v.fill(0); t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_a, float(score(best_a))


def coord_descent_polish(p, deadline, rng, step_frac=0.05):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = score(p)
    no_imp = 0

    factors = [1.03, 1.015, 1.06, 0.97, 0.985, 0.94]

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for f in factors:
                nv = orig * f
                if nv < 1e-15 or nv > 1000.0:
                    continue
                p[i] = nv
                s = p.sum()
                if s < 0.01:
                    p[i] = orig
                    continue
                p = p / s
                sc = score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = nv
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if improved:
            step_frac = max(0.01, step_frac * 0.95)
        if no_imp > 3 * n:
            break

    return best_p, best_score


def make_initializations(n, rng):
    profs = []
    t = np.linspace(0, 1, n)
    profs.append(np.ones(n) / n)
    profs.append((1.0 + 1.5 * (t - 0.5)) / (1.0 + 1.5 * (t - 0.5)).clip(0, None).sum())
    profs.append((1.0 + 2.0 * (t - 0.5)) / (1.0 + 2.0 * (t - 0.5)).clip(0, None).sum())
    profs.append((1.0 - 1.5 * (t - 0.5)) / (1.0 - 1.5 * (t - 0.5)).clip(0, None).sum())
    profs.append((t ** 2 + 1e-10) / (t ** 2 + 1e-10).sum())
    profs.append(((1 - t) ** 2 + 1e-10) / ((1 - t) ** 2 + 1e-10).sum())
    a = np.exp(2.0 * t) / np.exp(2.0 * t).sum()
    profs.append(a)
    a = np.exp(-2.0 * t) / np.exp(-2.0 * t).sum()
    profs.append(a)
    a = 1.0 / (1.0 + np.exp(-8.0 * (t - 0.5)))
    profs.append(a / a.sum())
    a = 1.0 / (1.0 + np.exp(8.0 * (t - 0.5)))
    profs.append(a / a.sum())
    for _ in range(6):
        z = rng.standard_normal(n)
        z = np.convolve(z, np.ones(20) / 20, mode='same')
        a = np.exp(z) / np.exp(z).sum()
        profs.append(a)
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize(p0)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} [{label}]")

    # Phase 0: Load parent and quick inits
    print("Phase 0: Initialization")
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize(parent)
    except Exception:
        parent = None

    n_primary = 512

    if parent is not None:
        sc = score(parent)
        update(parent, sc, "parent")
        if parent.size != n_primary:
            p_interp = interpolate_profile(parent, n_primary)
            if p_interp is not None:
                update(p_interp, score(p_interp), "parent_interp")

    # Diverse inits at n_primary
    inits = make_initializations(n_primary, rng)
    for p in inits:
        p = normalize(p)
        if p is not None:
            sc = score(p)
            update(p, sc, "init")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: LP-based optimization at n=512
    print("Phase 1: LP optimization at n=512")
    phase1_end = min(deadline, t_start + 700.0)

    # Use parent (or best init) as starting point for LP
    if parent is not None and parent.size == n_primary:
        lp_start = parent
    elif parent is not None:
        lp_start = interpolate_profile(parent, n_primary)
    else:
        lp_start = inits[0].copy()

    per = min(phase1_end, time.monotonic() + 350.0)
    p_lp, sc_lp = lp_optimize(lp_start, per, beta_init=None, rng=rng, verbose=True)
    update(p_lp, sc_lp, "LP_main")

    # Second LP run from a different init
    if time.monotonic() < phase1_end:
        for p0 in inits[1:5]:
            if time.monotonic() >= phase1_end:
                break
            per = min(phase1_end, time.monotonic() + 60.0)
            p_lp2, sc_lp2 = lp_optimize(p0, per, beta_init=None, rng=rng, verbose=False)
            update(p_lp2, sc_lp2, "LP_alt")

    print(f"  After Phase 1: best={best_score:.10f}")

    # Phase 2: Gradient polish at n=512
    print("Phase 2: Gradient polish at n=512")
    phase2_end = min(deadline, t_start + 1100.0)

    gradient_starts = []
    if best_seq is not None and best_seq.size == n_primary:
        gradient_starts.append(best_seq)
    if p_lp is not None:
        gradient_starts.append(p_lp)
    gradient_starts.extend(inits[:3])

    for gi, gs in enumerate(gradient_starts[:5]):
        if time.monotonic() >= phase2_end:
            break
        gs = normalize(gs)
        if gs is None:
            continue
        per = min(phase2_end, time.monotonic() + 50.0)
        p_g, sc_g = gradient_optimize(gs, per, rng, lr=0.08)
        update(p_g, sc_g, f"grad_{gi}")

    print(f"  After Phase 2: best={best_score:.10f}")

    # Phase 3: Try n=1024
    print("Phase 3: Multi-length exploration (n=1024)")
    phase3_end = min(deadline, t_start + 1450.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        p_1024 = interpolate_profile(best_seq, 1024)
        if p_1024 is not None:
            sc_1024 = score(p_1024)
            update(p_1024, sc_1024, "interp_1024")

            # LP at 1024
            per = min(phase3_end, time.monotonic() + 150.0)
            p_lp1024, sc_lp1024 = lp_optimize(p_1024, per, beta_init=None, rng=rng, verbose=True)
            update(p_lp1024, sc_lp1024, "LP_1024")

            # Gradient at 1024
            if time.monotonic() < phase3_end:
                per = min(phase3_end, time.monotonic() + 100.0)
                p_g1024, sc_g1024 = gradient_optimize(p_lp1024, per, rng, lr=0.05)
                update(p_g1024, sc_g1024, "grad_1024")

    print(f"  After Phase 3: best={best_score:.10f}")

    # Phase 4: Additional LP runs with different betas at n=512
    print("Phase 4: Additional LP refinement")
    phase4_end = min(deadline, t_start + 1600.0)

    if best_seq is not None and best_seq.size == n_primary:
        for beta_mult in [0.5, 1.5, 0.2]:
            if time.monotonic() >= phase4_end:
                break
            c = fft_convolve(best_seq)
            c_range = float(c.max() - c.min())
            b_init = max(1e-10, c_range * 0.25 * beta_mult)
            per = min(phase4_end, time.monotonic() + 40.0)
            p_b, sc_b = lp_optimize(best_seq, per, beta_init=b_init, rng=rng, verbose=False)
            update(p_b, sc_b, f"LP_beta{beta_mult}")

    # Also try gradient from best with different lr
    if best_seq is not None and best_seq.size == n_primary and time.monotonic() < phase4_end:
        per = min(phase4_end, time.monotonic() + 50.0)
        p_g2, sc_g2 = gradient_optimize(best_seq, per, rng, lr=0.03)
        update(p_g2, sc_g2, "grad_fine")

    print(f"  After Phase 4: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1730.0)

    if best_seq is not None and best_seq.size <= 1024:
        per = min(phase5_end, time.monotonic() + 80.0)
        p_cd, sc_cd = coord_descent_polish(best_seq, per, rng, step_frac=0.04)
        update(p_cd, sc_cd, "CD")

        # Another round with smaller steps
        if time.monotonic() < phase5_end:
            per = min(phase5_end, time.monotonic() + 50.0)
            p_cd2, sc_cd2 = coord_descent_polish(p_cd, per, rng, step_frac=0.02)
            update(p_cd2, sc_cd2, "CD_fine")

    print(f"  After Phase 5: best={best_score:.10f}")

    # Phase 6: Final verification
    print("Phase 6: Final selection")
    if best_seq is not None:
        # Use exact convolution for final score
        final_sc = score(best_seq)
        print(f"  Final internal score: {final_sc:.10f}, n={best_seq.size}")

        # Verify with evaluate_sequence
        try:
            final_eval = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(final_eval):
                print(f"  Evaluator score: {final_eval:.10f}")
                if final_eval < final_sc:
                    final_sc = final_eval
        except Exception as e:
            print(f"  Evaluator error: {e}")

        # Ensure valid output
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s < 0.01:
            best_seq = np.ones(n_primary) / n_primary

        result = [float(x) for x in best_seq]
        print(f"  RETURNING: score={final_sc:.10f}, n={len(result)}")
        return result

    # Fallback
    fallback = np.ones(512) / 512
    print(f"  FALLBACK score: {score(fallback):.10f}")
    return [float(x) for x in fallback]
