"""
Multi-phase optimization for minimizing 2*n*max(convolve(f,f))/sum(f)^2.
Uses direct gradient descent in log-space (Adam) with adaptive softmax temperature,
diverse initializations, multi-resolution refinement, and simulated annealing.

def compute_exact_score(f):
    """Compute exact score 2n*max(conv)/sum^2."""
    n = len(f)
    S = float(f.sum())
    if S < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (S * S))


def logspace_grad(z, beta):
    """Compute gradient of 2n*softmax_beta(conv(exp(z),exp(z)))/sum(exp(z))^2 w.r.t. z."""
    n = len(z)
    z_shift = z - z.max()
    f = np.exp(z_shift)
    S = f.sum()

    # Convolution
    g = np.convolve(f, f)

    # Softmax
    g_max = float(g.max())
    g_norm = (g - g_max) / beta
    ex = np.exp(g_norm)
    ex_sum = float(ex.sum())
    w = ex / ex_sum
    L = g_max + beta * np.log(ex_sum)

    # h[i] = sum_j w[i+j] * f[j]  (i=0..n-1)
    # Compute via FFT: h = conv(w, f_reversed)[n-1 : 2n-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    Fw = np.fft.rfft(w, N)
    Ff_rev = np.fft.rfft(f[::-1], N)
    h_conv = np.fft.irfft(Fw * Ff_rev, N)
    h = h_conv[n - 1:2 * n - 1]

    # dL/df_i = 2*h_i / beta
    # dphi/df_i = 2n * (dL/df_i * S - 2*L) / S^3
    # dphi/dz_i = dphi/df_i * f_i
    dL_df = 2.0 * h / beta
    dphi_df = 2.0 * n * (dL_df * S - 2.0 * L) / (S ** 3)
    grad = dphi_df * f

    # Center gradient (objective is shift-invariant in z)
    grad -= grad.mean()

    return grad


def compute_score_from_z(z):
    """Compute exact score from log-space parameter z."""
    z_shift = z - z.max()
    f = np.exp(z_shift)
    return compute_exact_score(f)


def z_to_f(z):
    """Convert log-space to normalized sequence."""
    z_shift = z - z.max()
    f = np.exp(z_shift)

def simulated_annealing(z_init, deadline, rng, n_steps=50000, T_init=0.1, T_final=0.001):
    """Simulated annealing in log-space."""
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()

    f = z_to_f(z)
    current_score = compute_exact_score(f)
    best_z = z.copy()
    best_score = current_score
    best_f = f.copy()

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        alpha = step / max(n_steps - 1, 1)
        T = T_init * (T_final / T_init) ** alpha
        noise_scale = 0.05 + 0.1 * T / T_init

        z_trial = z + noise_scale * rng.normal(size=n)
        z_trial = np.clip(z_trial, -50, 50)
        z_trial -= z_trial.mean()

        f_trial = z_to_f(z_trial)
        trial_score = compute_exact_score(f_trial)

        delta = trial_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            z = z_trial
            f = f_trial
            current_score = trial_score
            if current_score < best_score:
                best_score = current_score
                best_z = z.copy()
                best_f = f.copy()

    return best_z, best_score, best_f

        f = 1.0 + 0.5 * smooth
        f = np.maximum(f, 1e-10)
        starts.append(f / f.sum())

    return starts


def interpolate_to_length(f, n_new):
    """Interpolate sequence to new length."""
    n_old = len(f)
    if n_old == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, f)
    return result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2n*max(conv(f,f))/sum(f)^2.
    Phases:
      A: Adam optimization at n=256 from diverse starts
      B: Continue best solutions at n=512
      C: Simulated annealing on best solutions
      D: Try n=1024 for final improvement
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf
    best_z = None
    best_n = 0

    def try_update(f, score, n):
        nonlocal best_seq, best_score, best_z, best_n
        if f is not None and np.isfinite(score) and score < best_score:
            best_seq = f.copy()
            best_score = float(score)
            best_n = n
            z_shift = np.log(np.maximum(f, 1e-30))
            best_z = z_shift - z_shift.mean()

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except Exception:
        pass

    # Phase A: Optimization at n=256
    n_a = 256
    phase_a_deadline = min(deadline, t_start + 700.0)
    print(f"Phase A: n={n_a}, time_budget={phase_a_deadline - t_start:.1f}s")

    starts_a = get_diverse_starts(n_a, rng, parent)
    print(f"  {len(starts_a)} starting points")

    # Evaluate all starts
    for s in starts_a:
        sc = compute_exact_score(s)
        try_update(s, sc, n_a)

    print(f"  Best initial: {best_score:.10f}")

    # Run Adam from each start (time allocation)
    results_a = []
    n_starts = len(starts_a)

    for i, s in enumerate(starts_a):
        remaining = phase_a_deadline - time.monotonic()
        if remaining < 5:
            break
        alloc = remaining / (n_starts - i)
        run_deadline = time.monotonic() + alloc

        z_init = np.log(np.maximum(s, 1e-30))
        z_init -= z_init.mean()

        z_best, score = adam_logspace(
            z_init, run_deadline, rng,
            lr=0.08, beta_init=1.0 / n_a, beta_min=0.02 / n_a,
            beta_decay_period=300, beta_decay_factor=0.93,
            restart_patience=600, max_iters=500000
        )
        f_best = z_to_f(z_best)
        # Recompute exact score
        score_exact = compute_exact_score(f_best)
        try_update(f_best, score_exact, n_a)
        results_a.append((score_exact, f_best, z_best))

        if (i + 1) % 5 == 0:
            print(f"  After {i+1}/{n_starts}: best={best_score:.10f}")

    results_a.sort(key=lambda x: x[0])
    print(f"  Phase A done. Best: {best_score:.10f}")

    # Phase B: Extend to n=512 and optimize
    n_b = 512
    phase_b_deadline = min(deadline, t_start + 1300.0)
    print(f"Phase B: n={n_b}, time_budget={phase_b_deadline - time.monotonic():.1f}s")

    results_b = []
    top_b = min(8, len(results_a))
    for i in range(top_b):
        remaining = phase_b_deadline - time.monotonic()
        if remaining < 10:
            break
        alloc = remaining / (top_b - i)
        run_deadline = time.monotonic() + alloc

        f_ext = interpolate_to_length(results_a[i][1], n_b)
        z_init = np.log(np.maximum(f_ext, 1e-30))
        z_init -= z_init.mean()

        z_best, score = adam_logspace(
            z_init, run_deadline, rng,
            lr=0.06, beta_init=1.0 / n_b, beta_min=0.02 / n_b,
            beta_decay_period=300, beta_decay_factor=0.93,
            restart_patience=800, max_iters=500000
        )
        f_best = z_to_f(z_best)
        score_exact = compute_exact_score(f_best)
        try_update(f_best, score_exact, n_b)
        results_b.append((score_exact, f_best, z_best))
        print(f"  B[{i}]: score={score_exact:.10f}, best={best_score:.10f}")

    results_b.sort(key=lambda x: x[0])
    print(f"  Phase B done. Best: {best_score:.10f}")

    # Phase C: Simulated annealing on best solutions
    phase_c_deadline = min(deadline, t_start + 1550.0)
    print(f"Phase C: Simulated annealing, time_budget={phase_c_deadline - time.monotonic():.1f}s")

    all_results = results_a + results_b
    all_results.sort(key=lambda x: x[0])
    top_c = min(4, len(all_results))

    for i in range(top_c):
        remaining = phase_c_deadline - time.monotonic()
        if remaining < 5:
            break
        alloc = remaining / (top_c - i)
        run_deadline = time.monotonic() + alloc

        n_c = len(all_results[i][1])
        z_init = all_results[i][2]
        z_best, score, f_best = simulated_annealing(
            z_init, run_deadline, rng,
            n_steps=50000, T_init=0.08, T_final=0.005
        )
        score_exact = compute_exact_score(f_best)
        try_update(f_best, score_exact, n_c)
        print(f"  C[{i}]: score={score_exact:.10f}, best={best_score:.10f}")

    # Phase D: Try n=1024 if time permits
    if time.monotonic() < deadline - 100:
        n_d = 1024
        phase_d_deadline = min(deadline - 30, time.monotonic() + 180.0)
        print(f"Phase D: n={n_d}, time_budget={phase_d_deadline - time.monotonic():.1f}s")

        # Take the overall best and interpolate
        f_best_current = best_seq
        f_ext = interpolate_to_length(f_best_current, n_d)
        z_init = np.log(np.maximum(f_ext, 1e-30))
        z_init -= z_init.mean()

        z_best, score = adam_logspace(
            z_init, phase_d_deadline, rng,
            lr=0.05, beta_init=1.0 / n_d, beta_min=0.02 / n_d,
            beta_decay_period=500, beta_decay_factor=0.92,
            restart_patience=1000, max_iters=300000
        )
        f_best = z_to_f(z_best)
        score_exact = compute_exact_score(f_best)
        try_update(f_best, score_exact, n_d)
        print(f"  Phase D: score={score_exact:.10f}, best={best_score:.10f}")

    # Phase E: Final polish on the absolute best
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_z is not None:
        print(f"Phase E: Final polish, time_budget={remaining:.1f}s")
        n_final = best_n
        z_init = best_z.copy()

        z_best, score = adam_logspace(
            z_init, deadline - 5, rng,
            lr=0.02, beta_init=0.5 / n_final, beta_min=0.01 / n_final,
            beta_decay_period=500, beta_decay_factor=0.95,
            restart_patience=500, max_iters=200000
        )
        f_best = z_to_f(z_best)
        score_exact = compute_exact_score(f_best)
        try_update(f_best, score_exact, n_final)
        print(f"  Phase E: score={score_exact:.10f}, best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = make_linear_profile(256, 1.5)
        best_score = compute_exact_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_exact_score(best_seq)
    elapsed = time.monotonic() - t_start
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}, time={elapsed:.1f}s")

    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate error: {e}")

    return [float(x) for x in best_seq]
