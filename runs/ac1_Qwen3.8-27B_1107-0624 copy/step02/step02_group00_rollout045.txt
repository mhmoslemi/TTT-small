from scipy.signal import correlate


def _next_pow2(n):
    k = 1
    while k < n:

def _fft_convolve(a):

    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)

def _score(a):

    n = a.size
    s = float(a.sum())
    if s < 1e-10:

    a_n = a / s
    c = _fft_convolve(a_n)
    return float(2.0 * n * c.max())


def _softmax(x, tau):
    x = (x - x.max()) / tau
    ex = np.exp(x)
    return ex / ex.sum()

def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build constraint matrix: A[j, i] = a[i] if j <= i < j+n (i.e., conv position j has a[i] at offset i-j)
    # conv(a, b)_k = sum_i a[i] * b[k-i]
    # For fixed a, this is linear in b.
    # (A @ b)_k = sum_i A[k,i] * b[i] = sum_i a[i] * b[k-i]
    # So A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # Equivalently: A[k, i] = a[k-i] if max(0, k-n+1) <= i <= min(k, n-1)
    
    # Build A as a sparse-like structure
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = a
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _make_cosine_profile(n):
    """Cosine-squared profile."""
    j = np.arange(n)
    a = np.cos(np.pi * (j + 0.5) / (2 * n)) ** 2
    a = a / a.sum()

    j = np.arange(n)
    x = 2.0 * (j + 0.5) / n - 1.0
    a = np.sqrt(np.maximum(0.0, 1.0 - x**2))

    return a

    """Discrete arcsine profile (concentrated at edges)."""
    j = np.arange(n)
    x = (j + 0.5) / n
    a = 1.0 / np.sqrt(np.maximum(1e-10, x * (1.0 - x)))
    a = a / a.sum()
    return a


def _make_uniform_profile(n):
    return np.ones(n) / n


def _make_linear_profile(n, eta):

def _interpolate_profile(p0, n_new):

        s = p0.sum()
        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

                pp = _interpolate_profile(parent, n_t)
                pp = pp * (1.0 + 0.15 * rng.normal(size=n_t))
                pp = np.maximum(pp, 0)
                pp = pp / pp.sum()
                inits.append(pp)
                names.append("parent_pert")
        
        # Analytical profiles
        inits.append(_make_cosine_profile(n_t))
        names.append("cosine")
        inits.append(_make_semicircle_profile(n_t))
        names.append("semicircle")
        inits.append(_make_arcsine_profile(n_t))
        names.append("arcsine")
        inits.append(_make_uniform_profile(n_t))
        names.append("uniform")
        inits.append(_make_linear_profile(n_t, 1.5))
        names.append("linear15")
        inits.append(_make_linear_profile(n_t, 2.5))
        names.append("linear25")
        
        best_for_n = None
        best_score_n = np.inf
        
        # Run mirror descent on each initialization
        for a0, name in zip(inits, names):
            if time.monotonic() > deadline - 100:
                break
            
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            # Mirror descent
            md_deadline = min(time.monotonic() + 180.0, deadline - 100.0)
            a_md = _mirror_descent(a0, md_deadline, max_iter=300000,
                                   eta=0.08, beta=0.9, tau0=0.05, tau_decay=0.9998)
            sc_md = _score(a_md)
            
            if np.isfinite(sc_md) and sc_md < best_score:
                best_score = sc_md
                best_seq = a_md.copy()
            if np.isfinite(sc_md) and sc_md < best_score_n:
                best_score_n = sc_md
                best_for_n = a_md.copy()
            
            print(f"  n={n_t}, {name}: init={sc0:.8f} -> MD={sc_md:.10f}")
        
        # LP refinement on best for this length
        if best_for_n is not None and time.monotonic() < deadline - 50:
            lp_deadline = min(time.monotonic() + 200.0, deadline - 50.0)
            a_lp = _lp_refine(best_for_n, lp_deadline, max_iter=5000,
                              inflate=1.001, blend=0.3)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  n={n_t}, LP: {sc_lp:.10f}")
        
        # Final mirror descent polish on LP result
        if best_for_n is not None and time.monotonic() < deadline - 30:
            md2_deadline = min(time.monotonic() + 100.0, deadline - 30.0)
            a_md2 = _mirror_descent(best_for_n if best_for_n is not best_seq else best_seq,
                                    md2_deadline, max_iter=200000,
                                    eta=0.03, beta=0.9, tau0=0.02, tau_decay=0.9999)
            sc_md2 = _score(a_md2)
            if np.isfinite(sc_md2) and sc_md2 < best_score:
                best_score = sc_md2
                best_seq = a_md2.copy()
            print(f"  n={n_t}, MD polish: {sc_md2:.10f}")

        best_score = _score(best_seq)
