```python
    # This is the (j+1)-th element of the "upper" correlation

    # S1[j] = dot(w[1:n-j], a[j+1:n])

    # Or I can use scipy.signal.fftconvolve
    
    # Actually, the simplest efficient approach:
    # S2 = np.convolve(a, w, 'full')[:n] - w[0]*a  (this gives sum_{i=0}^{j} w[i]*a[j-i] - w[0]*a[j] = sum_{i=1}^{j} w[i]*a[j-i])

    # If I reverse both a and w: a_rev[i] = a[n-1-i], w_rev[i] = w[n-1-i]
    # S1[j] = sum_{i=1}^{n-1-j} w[i]*a[j+i]

    # = sum_{p=j}^{n-2} w_rev[n-1-(n-1-p)]*a_rev[n-1-(p-j)]
    # = sum_{p=j}^{n-2} w[n-1-p]*a_rev[n-1-p+j]

    # S1[j] = sum_{i=1}^{n-1-j} w[i] * a[j+i]
    # = sum_{m=j+1}^{n-1} w[m-j] * a[m]

    # Then S1 = T @ a
    
    # But that's O(n^2) memory. For n=512, that's 512^2 * 8 = 2MB, totally fine.

    # Actually, for n=512, let me just use a for loop with numpy dot:
    S1 = np.zeros(n)
    for j in range(n):
        if n-1-j > 0:
            S1[j] = np.dot(w[1:n-j], a[j+1:n])
    # And S2:
    full_conv = np.convolve(w, a, 'full')
    S2 = full_conv[:n] - w[0]*a
    
    grad = 2.0*w[0]*a + S1 + S2
    return grad
```

For n=512, the for loop has 512 iterations, each doing a dot product of average length ~256. Total: ~512*256 = 131K multiply-adds. With numpy overhead, this might take ~1-2ms per call. If I do 10000 iterations, that's 10-20 seconds just for gradients. That's acceptable.

For n=1024, it's ~4x more work, so ~40ms per 10000 iterations. Still fine.

But actually, I can do better. Let me use the FFT approach properly this time.

The key insight: `grad[j] = sum_k V[k] * a[j+k-(n-1)]` is a **convolution** (not cross-correlation) if I flip one of the signals.

Let me define:
- `V` of length 2n-1, where `V[k]` is the weight for lag `k-(n-1)`
- I want: `grad[j] = sum_{k=0}^{2n-2} V[k] * a[j + k - (n-1)]` (out-of-bounds = 0)

Let `t = j - (n-1)`. Then `grad[j] = sum_k V[k] * a[t + k]`.

This is the **cross-correlation** of V and a at lag t, or equivalently, the **convolution** of flipped-V and a at position t + (2n-2):

`conv(flip(V), a)[m] = sum_k flip(V)[k] * a[m-k] = sum_k V[2n-2-k] * a[m-k]`

Let `l = 2n-2-k`: `= sum_l V[l] * a[m-2n+2+l]`

I want this to equal `sum_k V[k] * a[t+k]`. So I need `m-2n+2+l = t+k` when `l=k`, giving `m = t + 2n - 2`.

So: `grad[j] = conv(flip(V), a)[t + 2n - 2]` where `t = j - (n-1)`.
`= conv(flip(V), a)[j - (n-1) + 2n - 2] = conv(flip(V), a)[j + n - 1]`

So: `grad = convolve(V[::-1], a, 'full')[n-1 : 2*n-1]`

Let me verify with a tiny example: n=2, a=[a0, a1], w=[w0, w1].

V (length 3): V[0] = w1 (lag -1), V[1] = 2*w0 (lag 0), V[2] = w1 (lag +1)
V_rev = [w1, 2*w0, w1]

conv(V_rev, a) = [w1*a0, w1*a1 + 2*w0*a0, 2*w0*a1 + w1*a0, w1*a1]

Wait, full conv of [w1, 2w0, w1] and [a0, a1]:
= [w1*a0, w1*a1 + 2w0*a0, 2w0*a1 + w1*a0, w1*a1]

Length = 3+2-1 = 4. Index 0 to 3.

I want grad[0] and grad[1].
grad[j] = conv(V_rev, a)[j + n - 1] = conv(V_rev, a)[j + 1]

grad[0] = conv[1] = w1*a1 + 2*w0*a0
grad[1] = conv[2] = 2*w0*a1 + w1*a0

Let me verify directly:
grad[0] = sum_k V[k] * a[0 + k - 1] = V[0]*a[-1] + V[1]*a[0] + V[2]*a[1] = 0 + 2*w0*a0 + w1*a1 = 2*w0*a0 + w1*a1 ✓

grad[1] = sum_k V[k] * a[1 + k - 1] = V[0]*a[0] + V[1]*a[1] + V[2]*a[2] = w1*a0 + 2*w0*a1 + 0 = w1*a0 + 2*w0*a1 ✓

So the formula is: `grad = convolve(V[::-1], a, 'full')[n-1 : 2*n-1]`

In FFT terms: `grad = ifft(FFT(V[::-1], N) * FFT(a, N))[n-1 : 2*n-1]` where N >= 3n-2.

Great! Now I have an efficient O(n log n) gradient computation.

```python
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= z_new.mean()

        # Update f
        f_new = np.exp(z_new - z_new.max())
        f_new = f_new / f_new.sum()
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Check improvement
        c_new = fft_convolve(f_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        f = f_new
        z = z_new
        total_iter += 1

        # Restart if stuck
        if no_imp > max(200, 2 * n):
            z = 0.7 * np.log(best_f + 1e-15) + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
            beta = 0.3 * float(fft_convolve(f).max())

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - (deadline - 3600)
            print(f"    {label} iter {it}: M={M_new:.12f}, best={best_M:.12f}, beta={beta:.6f}")

    return best_f, best_M


def gradient_optimize_annealed(f0, deadline, rng, verbose=False, label=""):
    """Multi-phase gradient optimization with aggressive beta annealing."""
    n = f0.size
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s

    c = fft_convolve(f)
    M_init = float(c.max())

    best_M = M_init
    best_f = f.copy()

    # Phase A: Coarse (high beta) - find basin
    phase_a_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    f, M_a = gradient_optimize(f, phase_a_end, rng,
                               beta_start=0.4 * M_init, beta_end=0.1 * M_init,
                               lr=0.1, label=label + "_coarse")
    if M_a < best_M:
        best_M = M_a
        best_f = f.copy()

    # Phase B: Medium
    phase_b_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.35)
    f, M_b = gradient_optimize(best_f, phase_b_end, rng,
                               beta_start=0.1 * M_init, beta_end=0.03 * M_init,
                               lr=0.06, label=label + "_med")
    if M_b < best_M:
        best_M = M_b
        best_f = f.copy()

    # Phase C: Fine
    f, M_c = gradient_optimize(best_f, deadline, rng,
                               beta_start=0.03 * M_init, beta_end=0.005 * M_init,
                               lr=0.03, label=label + "_fine")
    if M_c < best_M:
        best_M = M_c
        best_f = f.copy()

    return best_f, best_M


def coordinate_polish(f0, deadline, max_rounds=200):
    """Coordinate-wise optimization: for each j, find optimal f[j] given others fixed."""
    n = f0.size
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0
    f = f / s
    N_fft = next_pow2(2 * n - 1)

    best_M = float(fft_convolve(f).max())
    best_f = f.copy()

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        perm = np.random.permutation(n) if rd > 0 else np.arange(n)
        for j in perm:
            # c[k] = sum_i f[i]*f[k-i]
            # dc[k]/df[j] = 2*f[k-j] (0<=k-j<n)
            # We want to minimize max_k (c[k] + delta * 2*f[k-j]) where delta is change in f[j]
            # For simplicity, use the smoothed max and do a 1D line search
            base_c = fft_convolve(f)
            M_cur = float(base_c.max())

            # Gradient at j: grad_j = 2 * sum_k w[k] * f[k-j] (0<=k-j<n)
            # Approximate with uniform weights on top few
            top_k = np.argsort(base_c)[-max(1, n // 4):]
            w_approx = np.zeros(2 * n - 1)
            w_approx[top_k] = 1.0 / len(top_k)
            g_j = 2.0 * sum(w_approx[k] * f[k - j] for k in range(j, min(j + n, 2 * n - 1)))

            # Try multiplicative update
            for factor in (0.95, 0.85, 1.05, 1.15, 0.7, 1.3):
                f_new = f.copy()
                f_new[j] = f[j] * factor
                f_new = np.maximum(f_new, 0.0)
                s_new = f_new.sum()
                if s_new < 1e-30:
                    continue
                f_new = f_new / s_new
                M_new = float(fft_convolve(f_new).max())
                if M_new < M_cur - 1e-15:
                    f = f_new
                    M_cur = M_new
                    improved = True
                    break

        if float(fft_convolve(f).max()) < best_M:
            best_M = float(fft_convolve(f).max())
            best_f = f.copy()

        if not improved:
            break

    return best_f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage gradient-based search for optimal sequence.
    Phase 1: Diverse quick optimization at n=512
    Phase 2: Extended annealed optimization on best candidates
    Phase 3: Coordinate polish
    Phase 4: Extend to n=1024 and optimize
    Phase 5: Final fine-tuning
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: Quick optimization at n=512 from diverse starts
    n1 = 512
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: n={n1}, budget to {phase1_end - t_start:.0f}s")

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)
            # Also add a perturbed version
            p_pert = p_int + 0.1 * rng.normal(size=n1)
            p_pert = np.maximum(p_pert, 1e-10)
            starts.append(p_pert / p_pert.sum())

    starts.extend(make_deterministic_starts(n1))
    for _ in range(6):
        starts.append(make_smooth_random(n1, rng, n_harm=12))

    # Deduplicate (keep first 16 unique-ish starts)
    starts = starts[:16]

    candidates = []
    per_start = max(10.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 1e-15)
        p0 = p0 / p0.sum()

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = gradient_optimize(p0, per, rng,
                                           beta_start=0.3 * float(fft_convolve(p0).max()),
                                           beta_end=0.02 * float(fft_convolve(p0).max()),
                                           lr=0.08, label=f"P1_{i}")
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))
        print(f"  Phase 1 start {i+1}/{len(starts)}: local={sc:.8f}, global_best={best_score:.8f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Extended annealed optimization on top candidates
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(5, len(candidates))
    per_cand = max(30.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = gradient_optimize_annealed(p0, per, rng,
                                                     label=f"P2_{i}")
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 2 cand {i}: score={sc:.10f}, global_best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Coordinate polish on best
    phase3_end = min(deadline, t_start + 1100.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per = min(phase3_end, time.monotonic() + 80.0)
        p_pol = coordinate_polish(best_seq, per, max_rounds=100)
        sc = compute_score(p_pol)
        try_update(p_pol, sc)
        print(f"After Phase 3 (coord polish): best={best_score:.10f}")

    # Additional gradient fine-tune
    if best_seq is not None and time.monotonic() < phase3_end:
        per = min(phase3_end, time.monotonic() + 60.0)
        c_init = float(fft_convolve(best_seq).max())
        p_best, M_best = gradient_optimize(best_seq, per, rng,
                                           beta_start=0.05 * c_init,
                                           beta_end=0.005 * c_init,
                                           lr=0.02, label="P3_fine")
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"After Phase 3 (fine grad): best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 120.0)
            p_best, M_best = gradient_optimize_annealed(p_ext, per, rng, label="P4_1024")
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 (n=1024): score={sc:.10f}, global_best={best_score:.10f}")

            # Fine-tune in 1024
            if time.monotonic() < phase4_end - 20:
                c_init = float(fft_convolve(p_best).max())
                per2 = min(phase4_end - 20, time.monotonic() + 80.0)
                p_best2, M_best2 = gradient_optimize(p_best, per2, rng,
                                                     beta_start=0.05 * c_init,
                                                     beta_end=0.005 * c_init,
                                                     lr=0.02, label="P4_fine")
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 4 fine (n=1024): score={sc2:.10f}, global_best={best_score:.10f}")

    # Also try n=768 as intermediate
    n3 = 768
    if best_seq is not None and time.monotonic() < deadline - 100:
        p_ext3 = interpolate_profile(best_seq, n3)
        if p_ext3 is not None:
            per = min(deadline - 80, time.monotonic() + 80.0)
            p_best3, M_best3 = gradient_optimize_annealed(p_ext3, per, rng, label="P4_768")
            sc3 = 2.0 * n3 * M_best3
            try_update(p_best3, sc3)
            print(f"Phase 4 (n=768): score={sc3:.10f}, global_best={best_score:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = best_seq.size
        c_init = float(fft_convolve(best_seq).max())
        per = min(deadline - 5, time.monotonic() + 45.0)
        p_best, M_best = gradient_optimize(best_seq, per, rng,
                                           beta_start=0.03 * c_init,
                                           beta_end=0.003 * c_init,
                                           lr=0.01, label="P5")
        sc = 2.0 * n_best * M_best
        try_update(p_best, sc)
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Final: coordinate polish if time remains
    if best_seq is not None and time.monotonic() < deadline - 5:
        per = min(deadline - 3, time.monotonic() + 20.0)
        p_pol = coordinate_polish(best_seq, per, max_rounds=50)
        sc = compute_score(p_pol)
        try_update(p_pol, sc)
        print(f"After final polish: best={best_score:.10f}")

    # Return best
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final score: {sc_final:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 2.0)
    return [float(x) for x in fb]
```
