"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based spreading, mirror descent subgradient, coordinate descent,
and smooth-max gradient polish. Key insight: non-symmetric (ramp) sequences
can achieve score < 2 by reducing the center of the autocorrelation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_ramp_profile(n, power=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.1) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    """LP-based refinement with adaptive line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search: find best blend alpha
        best_alpha = blend_cur
        best_line_score = np.inf
        for alpha in [blend_cur * 0.5, blend_cur * 0.75, blend_cur, blend_cur * 1.25, blend_cur * 1.5]:
            alpha = min(alpha, 1.0)
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = _fft_convolve(a_new)
            sc = float(2.0 * n * c_new.max())
            if sc < best_line_score:
                best_line_score = sc
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        c_new = _fft_convolve(a)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > 200:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

    return best_a


def _mirror_descent(a, deadline, max_iter=2000000):
    """Mirror descent (exponential weights) with subgradient of max autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    eta = 0.5
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_c = float(c.max())
        # Find peaks (within tolerance)
        tol = 1e-8 * max_c
        K = np.where(c >= max_c - tol)[0]
        # Limit to top few peaks for stability
        if len(K) > 20:
            top_idx = np.argsort(c)[::-1][:10]
            K = top_idx

        # Compute subgradient: grad_j = (2/|K|) * sum_{k in K} a_{|j-k|} for |j-k| < n
        grad = np.zeros(n)
        for k in K:
            # For each j in [0, n-1], add a_{|k-j|} if |k-j| < n
            # j ranges over [max(0, k-n+1), min(n-1, k+n-1)]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k + n - 1)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            lags = np.abs(k - js)
            valid = lags < n
            if np.any(valid):
                grad[js[valid]] += a[lags[valid]]

        grad *= 2.0 / len(K)

        # Mirror descent update (exponential weights / multiplicative weights)
        logits = np.log(a) - eta * grad
        logits = np.clip(logits, -500, 500)
        logits -= logits.max()
        a_new = np.exp(logits)
        s = a_new.sum()
        if s < 1e-300 or not np.isfinite(s):
            eta *= 0.5
            continue
        a_new = a_new / s
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            # Slightly increase eta if improving
            eta = min(2.0, eta * 1.01)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                eta *= 0.95
            if no_imp > 5000:
                # Restart from best
                a = best_a.copy()
                eta = min(2.0, eta * 1.5)
                no_imp = 0

    return best_a


def _coordinate_descent(a, deadline, max_sweeps=50000):
    """Coordinate descent: minimize max(c) by optimizing one variable at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Random order for diversity
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break

            a_j = a[j]
            rest_sum = s - a_j

            # c_k as function of x = a_j:
            # c_k = u_k * x^2 + v_k * x + w_k
            # where for k in [0, 2n-2]:
            #   u_k = 1 if k == 2*j else 0
            #   v_k = a[(k-j) % n] if 0 <= k-j < n else 0, plus a[j-k] if 0 <= j-k < n else 0
            #         (but need to be careful about double counting)
            # Actually: c_k = sum_i a_i * a_{k-i}
            # Terms with a_j:
            #   i=j: a_j * a_{k-j} (if 0<=k-j<n)
            #   k-i=j => i=k-j: a_{k-j} * a_j (if 0<=k-j<n and k-j != j)
            # So coefficient of x (linear) in c_k:
            #   a_{k-j} if 0<=k-j<n and k!=2j
            #   a_{k-j} if 0<=j-k<n and j!=k  (same as above since |k-j| is symmetric)
            #   0 if k=2j (that's the quadratic term)
            # Coefficient of x^2: 1 if k=2j else 0

            # Precompute: for each k, the coefficient of x and x^2 in c_k
            # c_k = (x * b_k + c0_k) * x + c0_k * ... 
            # Let me redefine:
            # c_k = x * (a_{k-j} if valid) + x * (a_{k-j} if valid and k!=2j) + (terms without x)
            #     = x^2 (if k=2j) + x * 2*a_{|k-j|} (if |k-j|<n and k!=2j) + w_k

            # More precisely:
            # c_k = x^2 * [k==2j] + x * (a_{k-j} [0<=k-j<n] + a_{j-k} [0<=j-k<n]) [k!=2j] + w_k
            # But a_{k-j} and a_{j-k}: if k>j, then k-j>0 and j-k<0 (invalid). So a_{j-k}=0.
            # If k<j, then k-j<0 (invalid) and j-k>0. So a_{k-j}=0.
            # If k==j, both are a_0.
            # So: coefficient of x (linear) = a_{|k-j|} if |k-j| < n and k != 2j, plus a_{|k-j|} if |k-j| < n and k != 2j... 
            # Wait, I think the total linear coefficient is:
            #   If k > j and k-j < n: a_{k-j} (from i=j term) + a_{k-j} (from i=k-j term, since k-j < n and k-j != j when k > j) = 2*a_{k-j}
            #   If k < j and j-k < n: a_{k-j}=a_{negative}=0 (from i=j) + a_{j-k} (from i=k-j=j-k... wait)
            
            # I'm overcomplicating this. Let me just compute c_k at x=0 and x=1 and x=0.5 to get the quadratic.
            # Actually, simplest: compute the gradient of c_k w.r.t. a_j, which is:
            # d(c_k)/d(a_j) = a_{k-j} + a_{j-k} (with a_x=0 outside [0,n-1])
            # This is the value at the current a_j.
            # c_k is quadratic in a_j: c_k = 0.5*u_k*a_j^2 + v_k*a_j + w_k where:
            #   u_k = 1 if k=2j else 0  (coefficient of a_j^2)
            #   v_k = a_{k-j} if 0<=k-j<n else 0 (this is d(c_k)/d(a_j) when a_j=0, plus the symmetric part)
            
            # OK let me just do it numerically. The cleanest way:
            # Set a_j = 0, compute c_k for all k -> gives w_k
            # Set a_j = 1, compute c_k -> gives u_k + v_k + w_k
            # So u_k + v_k = c_k(a_j=1) - c_k(a_j=0)
            # And u_k = c_k(a_j=1) - 2*c_k(a_j=0.5) + c_k(a_j=0) (second difference)
            
            # But this requires 3 convolutions, which is expensive to do for each j.
            
            # Alternative: compute the gradient once per sweep and use it.
            # d(c_k)/d(a_j) at current a = a_{k-j} + a_{j-k} (with bounds)
            # This gives us the linear coefficient v_k + u_k * a_j.
            # So v_k = d(c_k)/d(a_j) - u_k * a_j.
            
            # Let me compute the "direction" vector d = d(c)/d(a_j) for the current a:
            # d_k = a_{k-j} + a_{j-k} (with a_x = 0 outside [0,n-1])
            
            # d_k = a[k-j] if 0<=k-j<n else 0, plus a[j-k] if 0<=j-k<n else 0
            # Note: when k-j == j-k (i.e., k=2j), both terms are a_0... no, a[k-j]=a[j] and a[j-k]=a[j]... 
            # Wait no. When k=2j: k-j=j and j-k=-j. a[k-j]=a[j]=x (the variable), a[j-k]=a[-j]=0 (invalid).
            # So d_k at k=2j: a[j] + 0 = x. But the actual d(c_{2j})/d(a_j) = 2*a_j (from the a_j^2 term).
            
            # I think the correct formula is:
            # d(c_k)/d(a_j) = 2 * a_{k-j} if 0 <= k-j < n and k != 2j
            #                 2 * a_j if k = 2j  (since c_k includes a_j^2, derivative is 2*a_j)
            #                 2 * a_{j-k} if 0 <= j-k < n and k != 2j  (this is the same as 2*a_{k-j} by symmetry when |k-j|<n)
            #                 0 otherwise
            
            # Actually the simplest correct formula (verified earlier):
            # d(c_k)/d(a_j) = 2 * a_{|k-j|} if |k-j| < n, else 0.
            # (With the understanding that a_{|k-j|} means a[|k-j|] as an array index.)
            # Wait, that's not right either because a is not symmetric.
            
            # Let me just verify with the example:
            # a = [a0, a1, a2, a3], j=1.
            # d(c_0)/d(a1) = 0 (c_0 = a0^2, no a1). Formula: 2*a_{|0-1|} = 2*a_1. WRONG.
            
            # So the formula 2*a_{|k-j|} is INCORRECT. Let me go back to basics.
            
            # c_k = sum_{i=0}^{n-1} a_i * a_{k-i} (a_x = 0 for x outside [0,n-1])
            # d(c_k)/d(a_j) = a_{k-j} (from i=j, if 0<=k-j<n) + a_{k-j} (from k-i=j, i.e., i=k-j, if 0<=k-j<n and i=k-j is in [0,n-1] and k-j != j)
            
            # Case 1: k-j is in [0, n-1] and k-j != j (i.e., k != 2j):
            #   d(c_k)/d(a_j) = a_{k-j} + a_{k-j} = 2 * a_{k-j}
            # Case 2: k-j is in [0, n-1] and k-j == j (i.e., k = 2j):
            #   d(c_k)/d(a_j) = a_j (only one term, since i=j=k-j)
            #   But wait, the term is a_j * a_{k-j} = a_j * a_j = a_j^2, derivative is 2*a_j.
            #   So d(c_k)/d(a_j) = 2 * a_j = 2 * a_{k-j} (since k-j=j).
            #   So actually it's 2*a_{k-j} in all cases where 0<=k-j<n!
            #   But wait, in case 2, there's only ONE term in the sum (i=j), not two.
            #   The derivative of a_j^2 is 2*a_j. And 2*a_{k-j} = 2*a_j. So it's consistent!
            
            # Case 3: k-j is NOT in [0, n-1] (i.e., k-j < 0 or k-j >= n):
            #   From i=j: a_{k-j} = 0, no contribution.
            #   From i=k-j: k-j < 0 or k-j >= n, so i is not a valid index. No contribution.
            #   d(c_k)/d(a_j) = 0.
            
            # Conclusion: d(c_k)/d(a_j) = 2 * a[k-j] if 0 <= k-j < n, else 0.
            
            # Verify: a=[a0,a1,a2,a3], j=1, k=0: 0<=0-1<4? No (-1<0). So d(c_0)/d(a1)=0. ✓
            # j=1, k=1: 0<=1-1<4? Yes (0). d(c_1)/d(a1) = 2*a[0] = 2*a0. 
            #   c_1 = 2*a0*a1. d/d(a1) = 2*a0. ✓
            # j=1, k=2: 0<=2-1<4? Yes (1). d(c_2)/d(a1) = 2*a[1] = 2*a1.
            #   c_2 = 2*a0*a2 + a1^2. d/d(a1) = 2*a1. ✓
            # j=1, k=3: 0<=3-1<4? Yes (2). d(c_3)/d(a1) = 2*a[2] = 2*a2.
            #   c_3 = 2*a0*a3 + 2*a1*a2. d/d(a1) = 2*a2. ✓
            # j=1, k=4: 0<=4-1<4? No (3<4 is true!). Wait, 4-1=3, 0<=3<4 is yes.
            #   d(c_4)/d(a1) = 2*a[3] = 2*a3.
            #   c_4 = 2*a1*a3 + a2^2. d/d(a1) = 2*a3. ✓
            # j=1, k=5: 0<=5-1<4? No (4>=4). d(c_5)/d(a1) = 0.
            #   c_5 = 2*a2*a3. d/d(a1) = 0. ✓
            
            # Great! So the formula is: d(c_k)/d(a_j) = 2*a[k-j] if 0 <= k-j < n, else 0.
            # This means c_k is:
            #   If 0 <= k-j < n: c_k = 2*a[k-j]*a_j + (terms not involving a_j) when k != 2j
            #                     c_k = a_j^2 + ... when k = 2j
            #   Actually, c_k as a function of x=a_j:
            #     If k=2j: c_k = x^2 + w_k (quadratic)
            #     If 0<=k-j<n, k!=2j: c_k = 2*a[k-j]*x + w_k (linear in x)
            #     If not (0<=k-j<n): c_k = w_k (constant in x)
            
            # So to minimize max_k c_k over x in [0, rest_sum]:
            # c_k(x) = u_k * x^2 + v_k * x + w_k
            # where u_k = 1 if k==2j else 0
            #       v_k = 2*a[k-j] if (0<=k-j<n and k!=2j) else 0
            #       w_k = c_k - u_k*x^2 - v_k*x (at current x=a_j)
            
            # The function f(x) = max_k (u_k*x^2 + v_k*x + w_k) is convex in x.
            # We can minimize it using ternary search or by finding where the subgradient is zero.
            
            # Compute v_k for all k:
            # v_k = 2*a[k-j] if 0<=k-j<n else 0
            # This is a shifted version of 2*a
            v = np.zeros(2 * n - 1)
            start = j  # k-j >= 0 => k >= j
            end = j + n  # k-j < n => k < j+n
            v[start:end] = 2.0 * a  # a[0], a[1], ..., a[n-1]
            # Override: if k = 2j is in range, v[2j] should be 0 (it's the quadratic term)
            if 0 <= 2 * j < 2 * n - 1:
                v[2 * j] = 0.0

            # w_k = c_k_current - u_k * a_j^2 - v_k * a_j
            c_current = _fft_convolve(a)
            u = np.zeros(2 * n - 1)
            if 0 <= 2 * j < 2 * n - 1:
                u[2 * j] = 1.0
            w = c_current - u * a_j**2 - v * a_j

            # Now minimize f(x) = max_k (u_k*x^2 + v_k*x + w_k) over x in [0, rest_sum]
            # f is convex. Use ternary search.
            lo = 0.0
            hi = max(rest_sum, 0.0)
            if hi <= 0:
                continue

            # Ternary search
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(u * m1**2 + v * m1 + w)
                f2 = np.max(u * m2**2 + v * m2 + w)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            x_new = (lo + hi) / 2.0
            x_new = max(0.0, min(x_new, hi))
            
            if abs(x_new - a_j) < 1e-12:
                continue
                
            a[j] = x_new
            # Renormalize
            s = a.sum()
            if s < 1e-10:
                a[j] = a_j
                continue
            a = a / s
            # Update rest_sum for next iteration
            s = a.sum()

        # End of sweep
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

    return best_a


def _gradient_polish(a, deadline, max_iter=50000):
    """Smooth-max gradient descent with line search and restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    lr = 0.05
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest < 1e-10:
            a_nest = a
        else:
            a_nest = a_nest / s_nest

        grad, maxc = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            a = best_a.copy()
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))
                lr = 0.05
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _logit_adam_polish(a, deadline, max_iter=50000, lr=0.03):
    """Adam optimization in logit space with smooth-max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        e = np.exp(z - z.max())
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(2000, 4 * n):
            z = 0.7 * best_z + 0.3 * np.random.default_rng(it + n).normal(size=n) * 0.1
            z = np.clip(z, -50.0, 50.0)
            e = np.exp(z - z.max())
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Parametric initialization + LP refinement
    2. Mirror descent subgradient
    3. Coordinate descent (1D minimization per variable)
    4. Logit-space Adam polish
    5. Final gradient polish
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine target lengths to try
    lengths = [512]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)

    # Phase 1: LP refinement with multiple initializations
    print(f"Phase 1: LP refinement, n={lengths[0]}", flush=True)
    for n_t in lengths[:2]:  # Limit to 2 lengths
        if time.monotonic() > deadline - 300:
            break

        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
            p_t = _interpolate_profile(parent, n_t)
            for scale in [0.1, 0.2, 0.3]:
                pert = p_t * (1.0 + scale * rng.normal(size=n_t))
                pert = np.maximum(pert, 0)
                ps = pert.sum()
                if ps > 1e-10:
                    inits.append(pert / ps)
        inits.append(_make_linear_profile(n_t, 1.5))
        inits.append(_make_linear_profile(n_t, 2.0))
        inits.append(_make_ramp_profile(n_t, 1.0))
        inits.append(_make_exp_profile(n_t, 2.0))

        for a0 in inits:
            if time.monotonic() > deadline - 300:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # LP refinement
            lp_budget = min(150.0, deadline - time.monotonic() - 300.0)
            if lp_budget < 10:
                break
            lp_end = time.monotonic() + lp_budget
            a_lp = _lp_refine(a0, lp_end, max_iter=5000, inflate=1.001, blend=0.3)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  Init({sc0:.8f}) -> LP({sc_lp:.10f})", flush=True)

    # Phase 2: Mirror descent subgradient on best
    if best_seq is not None and time.monotonic() < deadline - 200:
        md_budget = min(300.0, deadline - time.monotonic() - 200.0)
        if md_budget > 10:
            md_end = time.monotonic() + md_budget
            print(f"Phase 2: Mirror descent, budget={md_budget:.0f}s", flush=True)
            a_md = _mirror_descent(best_seq, md_end, max_iter=1000000)
            sc_md = _score(a_md)
            if np.isfinite(sc_md) and sc_md < best_score:
                best_score = sc_md
                best_seq = a_md.copy()
            print(f"  Mirror descent: score={sc_md:.10f}", flush=True)

    # Phase 3: Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 150:
        cd_budget = min(400.0, deadline - time.monotonic() - 150.0)
        if cd_budget > 10:
            cd_end = time.monotonic() + cd_budget
            print(f"Phase 3: Coordinate descent, budget={cd_budget:.0f}s", flush=True)
            a_cd = _coordinate_descent(best_seq, cd_end, max_sweeps=50000)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  Coordinate descent: score={sc_cd:.10f}", flush=True)

    # Phase 4: Logit Adam polish
    if best_seq is not None and time.monotonic() < deadline - 100:
        ad_budget = min(200.0, deadline - time.monotonic() - 100.0)
        if ad_budget > 10:
            ad_end = time.monotonic() + ad_budget
            print(f"Phase 4: Logit Adam, budget={ad_budget:.0f}s", flush=True)
            a_ad = _logit_adam_polish(best_seq, ad_end, max_iter=50000, lr=0.03)
            sc_ad = _score(a_ad)
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Adam result: score={sc_ad:.10f}", flush=True)

    # Phase 5: Another round of coordinate descent on the polished result
    if best_seq is not None and time.monotonic() < deadline - 50:
        cd_budget = min(150.0, deadline - time.monotonic() - 50.0)
        if cd_budget > 10:
            cd_end = time.monotonic() + cd_budget
            print(f"Phase 5: Final coordinate descent, budget={cd_budget:.0f}s", flush=True)
            a_cd2 = _coordinate_descent(best_seq, cd_end, max_sweeps=30000)
            sc_cd2 = _score(a_cd2)
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
            print(f"  Final CD: score={sc_cd2:.10f}", flush=True)

    # Phase 6: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        gp_budget = min(80.0, deadline - time.monotonic() - 20.0)
        if gp_budget > 5:
            gp_end = time.monotonic() + gp_budget
            a_final = _gradient_polish(best_seq, gp_end, max_iter=20000)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"Phase 6: Final gradient polish: score={sc_final:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in final_seq]
