def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_matrix_sparse(a):

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)

    x = (c - mval) / beta

        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):

    c = _fft_convolve(a)

    N = _next_pow2(2 * n - 1)

    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _line_search_simplex(a, grad, max_step=1.0, n_trials=20):
    """Simple line search: try several step sizes, pick best."""
    n = len(a)
    best_score = _score(a)
    best_a = a.copy()
    # Direction: project negative gradient onto simplex tangent
    neg_grad = -grad
    # Remove mean to stay on simplex
    neg_grad = neg_grad - np.mean(neg_grad)
    for i in range(n_trials):
        alpha = max_step * (0.5 ** i)
        a_trial = a + alpha * neg_grad
        a_trial = _project_simplex(a_trial)
        sc = _score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial
        else:
            break
    return best_a, best_score

def _subgradient_polyak(a, deadline, max_iter=10000):
    """Subgradient method with Polyak averaging for non-smooth objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    # Polyak step size: alpha_k = (f_k - f*) / ||g_k||^2
    # We don't know f*, so use a running best as estimate
    f_star = best_score
    avg_a = a.copy()
    count = 0
    infeasible = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        # Find active constraints (top few entries of convolution)
        top_k = max(1, min(10, len(c) // 10))
        active_idx = np.argsort(c)[-top_k:]
        active_weights = np.zeros(len(c))
        active_weights[active_idx] = 1.0 / top_k

        # Compute subgradient
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(active_weights, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Polyak step
        f_curr = float(2.0 * n * M)
        gap = f_curr - f_star
        if gap <= 1e-14:
            gap = 1e-10
        gnorm2 = float(np.dot(grad, grad)) + 1e-12
        alpha = gap / gnorm2
        alpha = min(alpha, 0.1)

        a_new = a - alpha * grad
        a_new = _project_simplex(a_new)

        # Update average
        count += 1
        avg_a = ((count - 1) * avg_a + a_new) / count

        c_avg = _fft_convolve(avg_a)
        score_avg = float(2.0 * n * c_avg.max())
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            f_star = best_score

        a = a_new

    return best_a

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine which n values to try
    n_candidates = [256, 512, 1024]
    if parent is not None:
        n_parent = parent.size
        if n_parent not in n_candidates:
            n_candidates.insert(0, n_parent)
        # Sort by preference: parent n first, then others
        n_candidates = [n_parent] + [x for x in n_candidates if x != n_parent]
    else:
        n_candidates = [512, 256, 1024]

    # Phase 1: LP refinement for each n
    for n_t in n_candidates:
        if time.monotonic() > deadline - 100:
            break

        if parent is not None:
            a0 = _interpolate_profile(parent, n_t)
        else:
            a0 = _make_linear_profile(n_t, 1.5)

        sc0 = _score(a0)
        print(f"n={n_t}: initial score={sc0:.10f}")
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # LP refinement
        lp_budget = min(400.0, deadline - time.monotonic() - 150.0)
        if lp_budget < 15:
            continue
        lp_end = time.monotonic() + lp_budget
        print(f"  LP phase for n={n_t}, budget={lp_budget:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=15000)
        sc_lp = _score(a_lp)
        print(f"  LP result: score={sc_lp:.10f}")
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

        # Gradient polish on LP result
        gp_budget = min(80.0, deadline - time.monotonic() - 80.0)
        if gp_budget > 10:
            gp_end = time.monotonic() + gp_budget
            a_gp = _gradient_polish(a_lp, gp_end, max_iter=8000)
            sc_gp = _score(a_gp)
            print(f"  Gradient polish: score={sc_gp:.10f}")
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()

    # Phase 2: Adam polish on best found so far
    if best_seq is not None and time.monotonic() < deadline - 60:
        ad_budget = min(200.0, deadline - time.monotonic() - 60.0)
        if ad_budget > 10:
            ad_end = time.monotonic() + ad_budget
            print(f"Phase 2: Adam polish, budget={ad_budget:.0f}s, n={best_seq.size}")
            a_ad = _adam_polish(best_seq, ad_end, max_iter=20000, lr=0.03)
            sc_ad = _score(a_ad)
            print(f"  Adam result: score={sc_ad:.10f}")
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()

    # Phase 3: Try diverse initializations for the best n
    if time.monotonic() < deadline - 100:
        n_best = best_seq.size if best_seq is not None else 512
        inits = []
        for eta in [1.0, 1.5, 2.0, 3.0]:
            inits.append(_make_linear_profile(n_best, eta))
        for gamma in [1.0, 2.0, 3.0]:
            inits.append(_make_parabolic_profile(n_best, gamma))
        for a1, a3 in [(1.5, -1.0), (2.0, -2.0), (1.0, -0.5)]:
            inits.append(_make_cubic_profile(n_best, a1, a3))

        if parent is not None:
            p_n = _interpolate_profile(parent, n_best)
            inits.append(p_n)
            for _ in range(2):
                p_pert = p_n * (1.0 + 0.15 * rng.normal(size=n_best))
                p_pert = np.maximum(p_pert, 0)
                p_pert = p_pert / p_pert.sum()
                inits.append(p_pert)

        for a0 in inits:
            if time.monotonic() > deadline - 80:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick LP + gradient
            quick_budget = min(40.0, deadline - time.monotonic() - 80.0)
            if quick_budget < 5:
                break
            quick_end = time.monotonic() + quick_budget
            a_q = _lp_refine(a0, quick_end, max_iter=3000)
            sc_q = _score(a_q)
            if np.isfinite(sc_q) and sc_q < best_score:
                best_score = sc_q
                best_seq = a_q.copy()

            # Quick gradient
            if time.monotonic() < deadline - 80:
                g_end = time.monotonic() + min(15.0, deadline - time.monotonic() - 80.0)
                a_g = _gradient_polish(a_q, g_end, max_iter=3000)
                sc_g = _score(a_g)
                if np.isfinite(sc_g) and sc_g < best_score:
                    best_score = sc_g
                    best_seq = a_g.copy()

    # Phase 4: Final fine-tuning with very small gradient steps
    if best_seq is not None and time.monotonic() < deadline - 20:
        fin_budget = min(80.0, deadline - time.monotonic() - 20.0)
        if fin_budget > 5:
            fin_end = time.monotonic() + fin_budget
            print(f"Phase 4: Final fine-tune, budget={fin_budget:.0f}s")
            a_fin = _gradient_polish(best_seq, fin_end, max_iter=15000)
            sc_fin = _score(a_fin)
            print(f"  Final gradient: score={sc_fin:.10f}")
            if np.isfinite(sc_fin) and sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()

            # One more Adam pass
            if time.monotonic() < deadline - 20:
                ad2_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 20.0)
                a_ad2 = _adam_polish(best_seq, ad2_end, max_iter=10000, lr=0.01)
                sc_ad2 = _score(a_ad2)
                print(f"  Final Adam: score={sc_ad2:.10f}")
                if np.isfinite(sc_ad2) and sc_ad2 < best_score:
                    best_score = sc_ad2
                    best_seq = a_ad2.copy()

    # Phase 5: Subgradient with Polyak averaging on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        sub_budget = min(30.0, deadline - time.monotonic() - 10.0)
        if sub_budget > 3:
            sub_end = time.monotonic() + sub_budget
            a_sub = _subgradient_polyak(best_seq, sub_end, max_iter=5000)
            sc_sub = _score(a_sub)
            print(f"Phase 5: Subgradient: score={sc_sub:.10f}")
            if np.isfinite(sc_sub) and sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.copy()

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        pass  # Keep as is, score is scale-invariant
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print("WARNING: evaluate_sequence returned inf, using fallback")
                final_seq = _make_linear_profile(256, 1.5)
                best_score = _score(final_seq)
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
