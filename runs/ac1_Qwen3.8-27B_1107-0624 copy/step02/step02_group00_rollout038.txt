def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):

    n = a.size

    if mn < -1e-12:

    return c

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a, mode='full')
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-10:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            A[k, idx] = a[k - idx]
    return A


def _solve_lp_min_t(a):
    n = len(a)
    A = _make_conv_matrix(a)
    # Variables: [b_0, ..., b_{n-1}, t]
    # Minimize: t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    # Inequality: A @ b - t <= 0 for each row
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.x[-1])
    return None, None


def _solve_lp_weighted(a, weights):
    n = len(a)
    A = _make_conv_matrix(a)
    c_obj = A.T @ weights
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_and_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    val = mval + beta * float(np.log(s))
    # Gradient: 2 * sum_k w[k] * a[k-p] for valid k-p
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a, deadline, max_iter=50000, lr=0.03):
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    c0 = _fft_convolve(a)
    beta = max(1e-10, 0.05 * float(c0.max()))
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        val, grad, maxc = _smooth_max_and_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 300 == 0:
            beta = max(1e-12, beta * 0.7)

    return best_a


def _logit_adam_polish(a, deadline, max_iter=50000, lr=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-10, 0.1 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_a * 1.0  # reset
            z = np.log(np.maximum(best_a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(5000, 5 * n):
            rng = np.random.default_rng(it * 7919)
            z = 0.7 * np.log(np.maximum(best_a, 1e-12)) + 0.3 * rng.normal(size=n)
            z -= float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-10, 0.1 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _lp_block_coordinate(a, deadline, max_iter=500, alpha_start=0.8, alpha_min=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(np.maximum(a, 0))
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    alpha = alpha_start
    no_imp = 0
    stagnant = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        b, t_opt = _solve_lp_min_t(a)
        if b is None:
            break

        b = np.maximum(b, 0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

            no_imp = 0
        else:

        if no_imp > 100:
            alpha = max(alpha_min, alpha * 0.8)
            no_imp = 0

        if stagnant > 300:
            break

        a = a_new

    return best_a

def _make_gaussian(n, sigma=0.25):

    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)

def _make_cosine(n):

    return a / a.sum()

def _make_symmetric_ramp(n):

def _make_fourier_seq(n, coeffs=None, n_modes=6):
    i = np.arange(n)
    a = np.ones(n)
    if coeffs is None:
        rng = np.random.default_rng(12345)
        coeffs = rng.normal(size=n_modes) * 0.3
    for k in range(n_modes):
        freq = (k + 1)
        a += coeffs[k] * np.cos(2 * np.pi * freq * i / n)
        if k + 1 < n_modes:
            a += 0.5 * coeffs[k] * np.sin(2 * np.pi * freq * i / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _anneal_perturb(a, temp, rng):
    n = len(a)
    # Add structured perturbation
    freq = rng.integers(1, n // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi)
    i = np.arange(n)
    perturb = temp * np.cos(2 * np.pi * freq * i / n + phase)
    # Also add some noise
    perturb += temp * 0.5 * rng.normal(size=n)
    a_new = a + perturb
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

        lengths.append(256)

    for n_t in lengths:
        if time.monotonic() > deadline - 60:
            break

        print(f"\n=== Processing length n={n_t} ===")

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(('parent', _interpolate_profile(parent, n_t)))
            # Perturbed parents
            for _ in range(3):
                pp = _interpolate_profile(parent, n_t)
                noise = 0.05 * rng.normal(size=n_t)
                pp = np.maximum(pp + noise, 0)
                pp = pp / pp.sum()
                inits.append(('parent_pert', pp))
        inits.append(('gauss', _make_gaussian(n_t, 0.25)))
        inits.append(('gauss_wide', _make_gaussian(n_t, 0.35)))
        inits.append(('cosine', _make_cosine(n_t)))
        inits.append(('raised_cos', _make_raised_cosine(n_t)))
        inits.append(('linear', _make_linear_ramp(n_t)))
        inits.append(('sym_ramp', _make_symmetric_ramp(n_t)))
        inits.append(('fourier1', _make_fourier_seq(n_t, n_modes=4)))
        inits.append(('fourier2', _make_fourier_seq(n_t, n_modes=8)))

        # Keep only the best few initializations (by initial score)
        scored_inits = []
        for name, a0 in inits:
            sc0 = _score(a0)
            scored_inits.append((sc0, name, a0))
        scored_inits.sort(key=lambda x: x[0])
        # Take top 5
        scored_inits = scored_inits[:5]

        for sc0, name, a0 in scored_inits:
            if time.monotonic() > deadline - 60:
                break

            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            print(f"  [{name}] init score={sc0:.10f}", end="")

            # Phase 1: LP block coordinate
            lp_deadline = min(time.monotonic() + 120, deadline - 60)
            if time.monotonic() < lp_deadline:
                a_lp = _lp_block_coordinate(a0, lp_deadline, max_iter=200, alpha_start=0.7, alpha_min=0.1)
                sc_lp = _score(a_lp)
                print(f" -> LP={sc_lp:.10f}", end="")
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                a0 = a_lp

            # Phase 2: Weighted LP
            wl_deadline = min(time.monotonic() + 60, deadline - 60)
            if time.monotonic() < wl_deadline:
                a_wl = _lp_weighted_iterate(a0, wl_deadline, max_iter=100, gamma=3.0)
                sc_wl = _score(a_wl)
                print(f" -> WL={sc_wl:.10f}", end="")
                if sc_wl < best_score:
                    best_score = sc_wl
                    best_seq = a_wl.copy()
                a0 = a_wl

            # Phase 3: Gradient polish
            gp_deadline = min(time.monotonic() + 60, deadline - 60)
            if time.monotonic() < gp_deadline:
                a_gp = _gradient_polish(a0, gp_deadline, max_iter=20000, lr=0.02)
                sc_gp = _score(a_gp)
                print(f" -> GP={sc_gp:.10f}", end="")
                if sc_gp < best_score:
                    best_score = sc_gp
                    best_seq = a_gp.copy()
                a0 = a_gp

            # Phase 4: Logit Adam
            ad_deadline = min(time.monotonic() + 40, deadline - 60)
            if time.monotonic() < ad_deadline:
                a_ad = _logit_adam_polish(a0, ad_deadline, max_iter=20000, lr=0.03)
                sc_ad = _score(a_ad)
                print(f" -> AD={sc_ad:.10f}", end="")
                if sc_ad < best_score:
                    best_score = sc_ad
                    best_seq = a_ad.copy()
                a0 = a_ad

            print()

    # Final polishing phases on the overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final polish on best (score={best_score:.10f}, n={best_seq.size}) ===")

        # Final LP pass
        lp_deadline = min(time.monotonic() + 80, deadline - 30)
        if time.monotonic() < lp_deadline:
            a_final_lp = _lp_block_coordinate(best_seq, lp_deadline, max_iter=100, alpha_start=0.4, alpha_min=0.05)
            sc_flp = _score(a_final_lp)
            print(f"  Final LP: {sc_flp:.10f}")
            if sc_flp < best_score:
                best_score = sc_flp
                best_seq = a_final_lp.copy()

        # Final gradient
        gp_deadline = min(time.monotonic() + 60, deadline - 20)
        if time.monotonic() < gp_deadline:
            a_final_gp = _gradient_polish(best_seq, gp_deadline, max_iter=30000, lr=0.01)
            sc_fgp = _score(a_final_gp)
            print(f"  Final GP: {sc_fgp:.10f}")
            if sc_fgp < best_score:
                best_score = sc_fgp
                best_seq = a_final_gp.copy()

        # Final Adam
        ad_deadline = min(time.monotonic() + 40, deadline - 15)
        if time.monotonic() < ad_deadline:
            a_final_ad = _logit_adam_polish(best_seq, ad_deadline, max_iter=30000, lr=0.02)
            sc_fad = _score(a_final_ad)
            print(f"  Final AD: {sc_fad:.10f}")
            if sc_fad < best_score:
                best_score = sc_fad
                best_seq = a_final_ad.copy()

    # Simulated annealing restarts on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Annealing restarts ===")
        n_b = best_seq.size
        temps = [0.1, 0.05, 0.02, 0.01, 0.005]
        for temp in temps:
            if time.monotonic() > deadline - 30:
                break
            a_pert = _anneal_perturb(best_seq, temp, rng)
            # Quick optimization
            opt_deadline = min(time.monotonic() + 60, deadline - 30)
            a_opt = _lp_block_coordinate(a_pert, opt_deadline, max_iter=50, alpha_start=0.5, alpha_min=0.1)
            a_opt = _gradient_polish(a_opt, min(time.monotonic() + 30, deadline - 30), max_iter=10000, lr=0.015)
            sc_an = _score(a_opt)
            print(f"  temp={temp}: score={sc_an:.10f}")
            if sc_an < best_score:
                best_score = sc_an
                best_seq = a_opt.copy()

    # Final validation
    if best_seq is None:
        best_seq = _make_gaussian(256, 0.25)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_gaussian(256, 0.25)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print(f"\nEvaluate_sequence returned non-finite: {sc_eval}")
        except Exception as e:
            print(f"Evaluation error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
