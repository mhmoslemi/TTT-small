def compute_score(f):
    n = len(f)
    s = float(f.sum())

def soft_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(-x)  # flip for numerical stability (larger c -> smaller ex)
    # Actually let's be careful: we want weights proportional to exp(c[k]/beta)
    # For numerical stability: w[k] = exp((c[k]-max)/beta) / sum(exp((c[j]-max)/beta))
    x2 = (c - mval) / beta  # all <= 0
    ex = np.exp(x2)
    s = float(ex.sum())
    w = ex / s

Actually, let me think about this differently. The key insight from the LP is: we want to find a direction d (can be positive or negative) such that:
- f + d ≥ 0
- max_k (f + d)*(f + d)[k] ≤ M (or approximately)
- sum(d) ≥ 0 (we want to increase the sum, or at least not decrease it)

Since we're working in the normalized space (sum = 1), the constraint is sum(d) = 0.

The LP replaces this with: find g ≥ 0 such that f*g ≤ M, then blend f with g. The blending (1-t)f + t*g has sum = 1 (since both have sum 1 after normalization), and the convolution is bounded.

Actually, I think the key advantage of the LP is that it finds a completely different point (g) that satisfies the constraint, rather than making small steps. This allows larger moves.

Let me try a different approach: **iterative water-filling**.

Algorithm:
1. Start with f (sum = 1)
2. Compute conv(f,f), find the peak value M
3. For each position k where conv(f,f)[k] is close to M:
   - The peak is sum_i f[i] f[k-i]
   - We want to reduce this by adjusting f
   - The most effective adjustment is to reduce the f[i] that appear as large terms in the sum
4. "Pour" the reduced mass into positions that don't contribute to peaks

This is similar to the LP but done greedily. Let me implement this:

```python
def water_filling_step(f, M, rng):
    """One step of water-filling: reduce contribution to peaks."""
    n = len(f)
    c = np.convolve(f, f)
    
    # Find positions where c is close to M
    threshold = M * 0.95
    peak_positions = np.where(c >= threshold)[0]
    
    if len(peak_positions) == 0:
        return f.copy()
    
    f_new = f.copy()
    
    for k in peak_positions:
        # Contribution of f[i] to c[k] is f[i] * f[k-i] (for valid i)
        # We want to reduce the largest contributions
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        indices = np.arange(lo, hi + 1)
        contributions = f[indices] * f[k - indices]
        
        # Find the index with the largest contribution
        max_idx = indices[np.argmax(contributions)]
        
        # Reduce f[max_idx] and increase some other position
        reduction = 0.01 * f[max_idx]
        f_new[max_idx] -= reduction
        
        # Find positions to increase (those that don't contribute to peaks)
        # Simply increase a random position
        inc_idx = rng.integers(0, n)
        f_new[inc_idx] += reduction
    
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s
```

Actually, here's an idea: instead of the full LP, solve a smaller problem. At each step, we have the current f and the current M = max(conv(f,f)). We want to find a direction d (with ||d||=1, and f+d ≥ 0) that minimizes max_k (f+d)*(f+d)[k].

This is a min-max problem, which is hard. But we can approximate it:
- Linearize: (f+d)*(f+d)[k] ≈ (f*f)[k] + 2*(f*d)[k] (first order in d)
- We want to minimize max_k [(f*f)[k] + 2*(f*d)[k]]
- Subject to d ≥ -f (i.e., f+d ≥ 0) and ||d|| ≤ ε

This is still a min-max, but with a linear objective. We can use the soft-max approximation:
- minimize soft_max_k [(f*f)[k] + 2*(f*d)[k]]
- This is smooth in d, so we can use gradient methods

But this is essentially what Adam is already doing (with the soft-max of the full nonlinear objective).

The problem is: minimize max_k sum_i a[i] a[k-i] subject to sum a = 1, a ≥ 0.

This is equivalent to: find a probability distribution on {0, ..., n-1} that minimizes the maximum probability of the sum of two independent draws.

This is related to the "addition theorem" in combinatorics. For a uniform distribution, the sum distribution is triangular (peaks in the middle). We want to "flatten" this triangular shape.

One way to flatten: use a distribution that's concentrated at the edges (like a U-shape or bimodal). For example, a distribution that puts mass at 0 and n-1:
- a[0] = a[n-1] = 1/2, rest = 0
- conv: peak at 0 (value 1/4), n-1 (value 1/2), 2n-2 (value 1/4)
- max = 1/2, score = 2n * 1/2 = n. Very bad.

import numpy as np
import time

def compute_conv_max(f):
    """Compute max of auto-convolution."""
    return float(np.convolve(f, f).max())

def compute_score(f):
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    return 2.0 * n * compute_conv_max(f) / (s * s)

def soft_max_weights(c, beta):
    """Return softmax weights for the max approximation."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    return ex / ex.sum(), mval

def grad_soft_max_conv(f, w):
    """Gradient of (w @ conv(f,f)) w.r.t. f."""
    return 2.0 * np.correlate(w, f, mode='valid')

def f_to_z(f):
    """Convert probability vector to logits (centered)."""
    f = np.maximum(f, 1e-30)
    z = np.log(f)
    return z - z.mean()

def z_to_f(z):
    """Convert logits to probability vector."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def adam_main(f_init, deadline, rng, lr=0.06, beta_start=None,

              name=""):

    Main Adam optimizer in log-space.
    """

    f = np.maximum(f_init, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-30:
        return f_init, compute_score(f_init)
    f = f / s
    
    z = f_to_z(f)
    c = np.convolve(f, f)
    M0 = float(c.max())

        beta = M0 * 0.3
    else:
        beta = beta_start
    
    # Adam
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_f = f.copy()
    best_M = M0
    no_imp = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = float(c.max())
        w, _ = soft_max_weights(c, beta)
        g_f = grad_soft_max_conv(f, w)
        g_z = f * (g_f - float(np.dot(f, g_f)))
        
        t += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * g_z ** 2
        mhat = m / (1 - b1**t)
        vhat = v / (1 - b2**t)
        z = z - lr * mhat / (np.sqrt(vhat) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()
        f = z_to_f(z)

        M = float(c.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Beta annealing
        if t % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Restart

            z -= z.mean()

            M = float(c.max())
            beta = max(beta_min, M * 0.25)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_imp = 0
    
    score = 2.0 * n * best_M
    return best_f, score
