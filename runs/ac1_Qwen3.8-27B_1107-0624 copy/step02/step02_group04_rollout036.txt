    c = np.maximum(c, 0.0)
    return c

"""
Optimized step-function search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses correct gradient computation via FFT cross-correlation, top-K focused
soft-max, variance regularization, multi-scale escalation, and local search.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _conv_grad(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a. Returns array length n."""
    n = len(a)
    N = _next_pow2(3 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)
    return 2.0 * corr[:n]


def _var_gradient(a):
    """Gradient of var(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _full_conv(a)
    c_centered = c - c.mean()
    return (2.0 / (2 * n - 1)) * _conv_grad(a, c_centered)


def _softmax_vec(z, floor=1e-300):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / float(z.size)
    return p / s


def _soft_max_approx(c, beta):
    """Compute (soft_max_value, softmax_weights) for the approximation of max."""
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    if beta <= 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    x = (c - m) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    val = m + beta * float(np.log(s))
    return val, w


def _score(a):
    """Compute 2*n*max(conv)/sum^2 for non-negative a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    a_norm = a / s
    c = _full_conv(a_norm)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _make_uniform(n):
    return np.ones(n) / n


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.exp(-0.5 * (u / sigma) ** 2)
    return a / a.sum()


def _make_raised_cosine(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * np.cos(np.pi * u)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _optimize_logit(a0, deadline, rng, lr=0.06, beta_frac=0.12, beta_decay=0.75,
                    var_weight=0.0, no_imp_mult=5, topk=0, name=""):
    """
    Logit-space Adam optimization with soft-max gradient.
    a0: initial sequence (will be normalized internally)
    Returns (best_a, best_score).
    """
    a0 = _normalize(a0)
    n = a0.size

    z = np.log(np.maximum(a0, 1e-15))
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    a = a0.copy()
    c = _full_conv(a)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    iter_count = 0

    while time.monotonic() < deadline:
        # Compute soft-max weights
        if topk > 0 and c.size > topk:
            # Top-K focused soft-max
            idx = np.argsort(c)[::-1][:topk]
            c_k = c[idx]
            m_k = float(c_k.max())
            x_k = (c_k - m_k) / beta
            ex_k = np.exp(x_k)
            s_k = float(ex_k.sum())
            if s_k > 0 and np.isfinite(s_k):
                w = np.zeros_like(c)
                w[idx] = ex_k / s_k
            else:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
        else:
            _, w = _soft_max_approx(c, beta)

        # Compute gradient
        g = _conv_grad(a, w)

        # Add variance regularization
        if var_weight > 0.0:
            g = g + var_weight * _var_gradient(a)

        # Add small L2 regularization
        g = g + 1e-7 * a

        # Gradient in logit space: gz = a * (g - dot(g, a))
        dot_ga = float(np.dot(g, a))
        gz = a * (g - dot_ga)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        a = _softmax_vec(z)
        c = _full_conv(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        iter_count += 1

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.92)
        else:
            z = best_z.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue

        # Adaptive beta decay
        if iter_count % 30 == 0:
            beta = max(1e-14, beta * beta_decay)
            if c.size >= 2:
                sorted_c = np.sort(c)[::-1]
                gap = sorted_c[0] - sorted_c[1]
                if gap < beta * 0.02:
                    beta = max(1e-14, beta * 0.3)

        # Restart if stuck
        no_imp_limit = max(100, no_imp_mult * n)
        if no_imp > no_imp_limit:
            noise_scale = 0.08 if best_score < 1.52 else (0.15 if best_score < 1.55 else 0.3)
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            a = _softmax_vec(z)
            c = _full_conv(a)
            maxc = float(c.max())
            beta = max(1e-12, beta_frac * maxc)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            lr = lr if lr > 0.01 else 0.06
            no_imp = 0

    return best_a, best_score


def _local_search(a0, deadline, rng, n_max_iter=200000):
    """
    Patch-based local search for fine-tuning.
    Tries small random perturbations and keeps improvements.
    """
    a = _normalize(a0)
    n = a.size
    c = _full_conv(a)
    maxc = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * maxc)

    patch_min = 3
    patch_max = max(10, n // 32)
    T = 0.02
    T_min = 0.0005
    cool = 0.9997
    iters = 0
    improvements = 0

    while time.monotonic() < deadline and iters < n_max_iter:
        iters += 1
        ps = int(np.clip(rng.integers(patch_min, min(patch_max, n // 2) + 1), patch_min, n // 2))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        scale = T * 0.3
        a_new = a.copy()
        a_new[start:end] += rng.normal(0, scale, ps)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        c_new = _full_conv(a_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)

        delta = score_new - best_score
        if delta < -1e-15:
            a = a_new
            best_a = a_new.copy()
            best_score = score_new
            c = c_new
            maxc = maxc_new
            T = max(T_min, T * 0.995)
            improvements += 1
        else:
            if T > T_min:
                accept = np.exp(-delta / T) if delta < T * 10 else 0.0
                if rng.random() < accept:
                    a = a_new
                    c = c_new
            T = max(T_min, T * cool)

        if iters % 20000 == 0:
            print(f"    LS: iter={iters}, T={T:.6f}, best={best_score:.10f}, imp={improvements}")

    return best_a, best_score


def _multi_start_optimize(a0, deadline, rng, n_starts, lr, beta_frac, var_weight, name):
    """Run optimization from multiple perturbed starts of a0."""
    a0 = _normalize(a0)
    n = a0.size
    best_a = a0.copy()
    best_score = _score(a0)

    per_start = (deadline - time.monotonic()) / max(1, n_starts)
    for i in range(n_starts):
        if time.monotonic() >= deadline:
            break
        per_end = min(deadline, time.monotonic() + per_start)
        if i == 0:
            start_a = a0
        else:
            z = np.log(np.maximum(a0, 1e-15))
            z -= z.mean()
            noise = 0.1 * (1.0 + 0.4 * i)
            z = z + noise * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            start_a = _softmax_vec(z)

        p_best, sc = _optimize_logit(start_a, per_end, rng, lr=lr, beta_frac=beta_frac,
                                     var_weight=var_weight, name=f"{name}_start{i}")
        if sc < best_score:
            best_score = sc
            best_a = p_best.copy()
            print(f"    {name} start{i}: {sc:.10f}")

    return best_a, best_score

    if time.monotonic() < phase4_end - 120 and best_seq is not None:
        p_2048 = _interp_profile(best_seq, 2048)
        sc_2048 = _score(p_2048)
        update_best(p_2048, sc_2048, "interp_2048")
        print(f"  n=2048 interp: {sc_2048:.10f}")

        if sc_2048 < best_score * 1.02:  # Only optimize if interpolation didn't hurt much
            per_end = min(phase4_end, time.monotonic() + 250.0)
            p_2048_opt, sc_2048_opt = _optimize_logit(p_2048, per_end, rng, lr=0.04,
                                                      beta_frac=0.06, beta_decay=0.75,
                                                      var_weight=0.002, name="p4_2048")
            update_best(p_2048_opt, sc_2048_opt, "opt_2048")

    print(f"Phase 4 done: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 5: Local search on best ---
    phase5_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        print(f"  Local search on n={best_seq.size}, score={best_score:.10f}")
        p_ls, sc_ls = _local_search(best_seq, phase5_end, rng, n_max_iter=300000)
        update_best(p_ls, sc_ls, "local_search")

    # --- Phase 6: Final fine-tuning ---
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        remaining = phase6_end - time.monotonic()
        if remaining > 20:
            per_end = time.monotonic() + min(remaining - 5, 80.0)
            p_fin, sc_fin = _optimize_logit(best_seq, per_end, rng, lr=0.02,
                                            beta_frac=0.04, beta_decay=0.82,
                                            var_weight=0.0, topk=8,
                                            no_imp_mult=3, name="final")
            update_best(p_fin, sc_fin, "final_polish")

    # --- Final validation ---
    if best_seq is not None:
        final = best_seq.copy()
        final_n = final.size
        final = np.clip(final, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s

        # Exact score
        if final_n <= 4096:
            sc_exact = _score(final)
            print(f"Exact final score: {sc_exact:.10f}")
            # Try evaluator
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception:
                pass

        print(f"FINAL: score={best_score:.10f}, n={final_n}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_uniform(256)
    print(f"FALLBACK: n=256, score={_score(fallback):.4f}")
    return [float(x) for x in fallback]
