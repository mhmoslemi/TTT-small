def next_pow2(n):

    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):

    n = a.size
    if n == 0:

    if n <= 128:
        return np.convolve(a, a)

    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute the gradient via correlation of weights with p."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -80.0, 80.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def internal_score(a):
    """Compute 2*n*max(conv)/sum^2 using FFT convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def exact_score(a):
    """Compute exact score using np.convolve."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_basis(n, K):
    """Create a mixed trigonometric + polynomial basis."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])


def profile_from_basis(theta, basis):
    z = basis @ np.asarray(theta, dtype=np.float64)
    return softmax_from_z(z)


def make_profiles(n, rng):
    """Generate a diverse set of initial profiles."""
    profs = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    
    # Quadratic profiles
    for a1 in (0.5, 1.0, 1.5, 2.0, 3.0):
        for a2 in (-0.5, 0.0, 0.5, 1.0):
            a = 1.0 + a1 * (t - 0.5) + a2 * (2*t-1)**2
            a = np.maximum(a, 1e-12)
            profs.append(a / a.sum())
    
    # Exponential profiles
    for eta in (1.0, 2.0, 3.0, 5.0):
        a = np.exp(eta * t)
        profs.append(a / a.sum())
    
    # Power profiles
    for p in (0.5, 1.0, 2.0, 3.0):
        a = t**p + 1e-8
        profs.append(a / a.sum())
    
    # Sigmoid profiles
    for eta in (3.0, 6.0, 10.0, 15.0):
        a = 1.0 / (1.0 + np.exp(-eta * (t - 0.5)))
        profs.append(a / a.sum())
    
    # Two-block profiles
    for k_frac in (0.3, 0.4, 0.5, 0.6):
        for ratio in (0.3, 0.5, 0.7, 1.5, 2.0):
            k = max(1, int(n * k_frac))
            a = np.ones(n)
            a[:k] = 1.0
            a[k:] = ratio
            profs.append(a / a.sum())
    
    # Cubic profiles
    for a1 in (1.0, 1.5, 2.0, 2.5):
        for a3 in (-0.5, -1.0, -1.5, -2.0):
            u = 2*t - 1
            a = 1.0 + a1*u + a3*u**3
            a = np.maximum(a, 1e-12)
            profs.append(a / a.sum())
    
    # Random smooth profiles via basis
    basis = make_basis(n, 20)
    for _ in range(20):
        th = rng.uniform(-3.0, 3.0, basis.shape[1])
        profs.append(profile_from_basis(th, basis))
    
    # Random perturbations of flat
    for _ in range(10):
        a = 1.0 + 0.3 * rng.normal(size=n)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def lp_direction(f, M):
    """
    Solve the LP: maximize sum(b) subject to b >= 0 and (f*b)_j <= M for all j.
    Returns the optimal b (normalized to sum 1) or None if infeasible/trivial.
    """
    n = len(f)
    # Build constraint matrix A where A[j,l] = f_{j-l} if valid
    # This is a (2n-1) x n matrix
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(2*n-1):
        for l in range(n):
            k = j - l
            if 0 <= k < n:
                A[j, l] = f[k]
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    b = res.x
    s = float(b.sum())
    if s <= 1e-10:
        return None
    return b / s


def soft_max_adam(p0, deadline, rng, lr=0.10, beta_init_frac=0.25, use_lp=True, lp_interval=200):
    """
    Soft-max Adam optimization with periodic LP restarts.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            p = best_p.copy()
            c = full_convolve_fft(p)
            maxc = float(c.max())
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Decay beta
        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.7)

        # LP-based restart
        if use_lp and adam_t > 0 and adam_t % lp_interval == 0:
            b_dir = lp_direction(best_p, float(best_score / (2.0 * n)))
            if b_dir is not None:
                # Blend
                for t_blend in (0.3, 0.5, 0.7, 1.0):
                    p_try = (1.0 - t_blend) * best_p + t_blend * b_dir
                    p_try = np.maximum(p_try, 1e-15)
                    p_try = p_try / p_try.sum()
                    sc_try = internal_score(p_try)
                    if np.isfinite(sc_try) and sc_try < best_score:
                        best_score = sc_try
                        best_p = p_try.copy()
                        z = np.log(np.maximum(best_p, 1e-15))
                        z -= float(np.mean(z))
                        p = best_p.copy()
                        c = full_convolve_fft(p)
                        maxc = float(c.max())
                        beta = max(1e-12, 0.25 * maxc)
                        m.fill(0.0)
                        v.fill(0.0)
                        adam_t = 0
                        no_imp = 0
                        break

        # Random restart if stuck
        if no_imp > max(200, 3 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.3 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
            # Decrease learning rate slightly
            base_lr = max(0.02, base_lr * 0.95)

    return best_p, best_score


def subgradient_polish(p0, deadline, rng, n_iter_max=5000):
    """
    Pure subgradient (active set) mirror descent polish.
    """
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    z = np.log(p)
    z -= float(np.mean(z))

    best_p = p.copy()
    best_score = internal_score(p)
    eta = 0.01  # Initial step size
    eta_min = 1e-6
    no_imp = 0

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if score < best_score:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 200:
            break
        
        # Find active set
        active = c >= maxc - 1e-12
        n_active = int(np.count_nonzero(active))
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        # Compute subgradient: sum over active j of 2*shift(p, j)
        # shift(p, j)_m = p_{j-m} if valid
        g = np.zeros(n, dtype=np.float64)
        active_indices = np.where(active)[0]
        for j in active_indices:
            # (f*b)_j gradient contribution: 2 * p shifted
            # d/d(p_m) [conv(p,p)_j] = 2 * p_{j-m} if 0 <= j-m < n
            # So g_m += 2 * p_{j-m}
            # This is equivalent to: g += 2 * circulant shift
            lo = max(0, j - (n - 1))
            hi = min(j, n - 1)
            # For m in [lo, hi]: j-m is in [0, n-1]
            # g[m] += 2 * p[j-m]
            # j-m ranges from j-hi to j-lo
            if lo <= hi:
                g[lo:hi+1] += 2.0 * p[j-hi:j-lo+1][::-1]
        
        # Normalize subgradient
        g_norm = float(np.sqrt(np.dot(g, g)))
        if g_norm > 0:
            g = g / g_norm
        
        # Mirror descent update
        z = z - eta * g
        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        
        # Adaptive step size
        c_new = full_convolve_fft(p)
        score_new = float(2.0 * n * c_new.max())
        if score_new > score * 1.01:
            eta = max(eta_min, eta * 0.8)
        elif score_new < score * 0.99:
            eta = min(0.1, eta * 1.05)

    return best_p, best_score


def coordinate_descent_polish(p, deadline, rng, n_sweeps=None):
    """Coordinate descent polish."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    if n_sweeps is None:
        n_sweeps = max(3, int((deadline - time.monotonic()) * 2))

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    step_frac = 0.1

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for factor in (1.0 + step_frac, 1.0 + 0.5*step_frac, 1.0 + 2.0*step_frac,
                          1.0 - step_frac, 1.0 - 0.5*step_frac, 1.0 - 0.3*step_frac):
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000.0:
                    continue
                p[i] = new_val
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if improved:
            step_frac = max(0.02, step_frac * 0.85)
        if no_imp > 2 * n:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining soft-max Adam, LP direction finding,
    subgradient polish, and coordinate descent to minimize
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []  # List of (score, array)

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} [{label}]")
        if len(candidates) < 30:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            if float(sc) < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])

    # Phase 0: Load parent and quick initialization
    print("Phase 0: Initialization")
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
            if parent is not None:
                sc = internal_score(parent)
                local_update(parent, sc, "parent")
                print(f"  Parent score: {sc:.10f} n={parent.size}")
                # Try parent at different lengths
                for tn in (128, 192, 256, 384, 512, 768):
                    if parent.size != tn:
                        p_i = interpolate_profile(parent, tn)
                        if p_i is not None:
                            sc_i = internal_score(p_i)
                            local_update(p_i, sc_i, f"parent_n{tn}")
    except Exception:
        pass

    # Quick profile evaluation
    for n_init in (128, 256):
        for p0 in make_profiles(n_init, rng)[:20]:
            sc = internal_score(p0)
            local_update(p0, sc, f"init_n{n_init}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: Soft-max Adam at small lengths
    print("Phase 1: Soft-max Adam at n=128,192,256")
    phase1_end = min(deadline, t_start + 500.0)
    lengths_p1 = [128, 192, 256]
    
    for n in lengths_p1:
        # Get top candidates at this length
        init_p = []
        for sc_c, p_c in candidates[:10]:
            if p_c.size != n:
                p_i = interpolate_profile(p_c, n)
            else:
                p_i = p_c
            if p_i is not None:
                init_p.append(p_i)
        # Add random profiles
        for p0 in make_profiles(n, rng)[:5]:
            init_p.append(p0)
        
        for pi, p0 in enumerate(init_p[:8]):
            if time.monotonic() > phase1_end:
                break
            per = min(phase1_end, time.monotonic() + 20.0)
            p_best, sc = soft_max_adam(p0, per, rng, lr=0.10, use_lp=True, lp_interval=150)
            sc_full = internal_score(p_best)
            local_update(p_best, sc_full, f"p1_n{n}_{pi}")

    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: Soft-max Adam at medium lengths
    print("Phase 2: Soft-max Adam at n=384,512")
    phase2_end = min(deadline, t_start + 1000.0)
    lengths_p2 = [384, 512]
    
    for n in lengths_p2:
        init_p = []
        for sc_c, p_c in candidates[:8]:
            p_i = interpolate_profile(p_c, n) if p_c.size != n else p_c
            if p_i is not None:
                init_p.append(p_i)
        for p0 in make_profiles(n, rng)[:4]:
            init_p.append(p0)
        
        for pi, p0 in enumerate(init_p[:6]):
            if time.monotonic() > phase2_end:
                break
            per = min(phase2_end, time.monotonic() + 30.0)
            p_best, sc = soft_max_adam(p0, per, rng, lr=0.08, use_lp=True, lp_interval=200)
            sc_full = internal_score(p_best)
            local_update(p_best, sc_full, f"p2_n{n}_{pi}")

    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: LP-based refinement on best candidates
    print("Phase 3: LP refinement")
    phase3_end = min(deadline, t_start + 1300.0)
    
    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        n = p_c.size
        if n > 768:
            continue
        
        # Multiple LP iterations
        p_cur = p_c.copy()
        M_cur = float(best_score * p_cur.size / (2.0 * n)) if n == p_c.size else internal_score(p_c) / (2.0 * n) * p_c.size
        # Actually M = max(conv(p_c, p_c))
        c_c = full_convolve_fft(p_c)
        M_cur = float(c_c.max())
        
        for lp_iter in range(5):
            if time.monotonic() > phase3_end:
                break
            b_dir = lp_direction(p_cur, M_cur)
            if b_dir is None:
                break
            found = False
            for t_blend in (0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
                p_try = (1.0 - t_blend) * p_cur + t_blend * b_dir
                p_try = np.maximum(p_try, 1e-15)
                p_try = p_try / p_try.sum()
                sc_try = internal_score(p_try)
                if np.isfinite(sc_try) and sc_try < internal_score(p_cur):
                    p_cur = p_try
                    M_cur = float(full_convolve_fft(p_cur).max())
                    local_update(p_cur, sc_try, f"lp_{i}_{lp_iter}_t{t_blend}")
                    found = True
                    break
            if not found:
                break

    print(f"  After phase 3: best={best_score:.10f}")

    # Phase 4: Subgradient polish
    print("Phase 4: Subgradient polish")
    phase4_end = min(deadline, t_start + 1600.0)
    
    for i in range(min(5, len(candidates))):
        if time.monotonic() > phase4_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 512:
            continue
        per = min(phase4_end, time.monotonic() + 30.0)
        p_best, sc = subgradient_polish(p_c, per, rng)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"sg_{i}")

    print(f"  After phase 4: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1750.0)
    
    for i in range(min(4, len(candidates))):
        if time.monotonic() > phase5_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 512:
            continue
        per = min(phase5_end, time.monotonic() + 25.0)
        p_best, sc = coordinate_descent_polish(p_c, per, rng)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"cd_{i}")

    print(f"  After phase 5: best={best_score:.10f}")

    # Phase 6: Final evaluation
    print("Phase 6: Final evaluation")
    
    final_seq = best_seq
    final_score = best_score
    
    if final_seq is not None:
        # Check all candidates with exact score
