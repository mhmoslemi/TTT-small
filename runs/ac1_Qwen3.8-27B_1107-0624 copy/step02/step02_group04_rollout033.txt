import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(p):
    """Score for a sequence normalized to sum=1: 2*n*max(conv(p,p))."""
    n = p.size
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s < 1e-12:
        return np.inf
    c = _fft_convolve(p)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def _smooth_max_and_grad(c, beta):
    """Compute log-sum-exp and its gradient weights."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w


def _conv_grad_wrt_a(w, a):
    """Gradient of smooth_max(conv(a,a)) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] (a zero-padded outside [0,n-1])
    """
    n = a.size
    # v[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    # v = conv(a[::-1], w)[n-1:2*n-1]
    b = a[::-1]
    N = _next_pow2(n + (2 * n - 1) - 1)
    fb = np.fft.rfft(b, N)
    fw = np.fft.rfft(w, N)
    conv_bw = np.fft.irfft(fb * fw, N)
    v = conv_bw[n - 1:2 * n - 1]
    return 2.0 * v


def _obj_and_grad(a, beta):
    """Objective and gradient for sum-normalized sequence.
    f(a) = 2*n*smooth_max(conv(a,a)) / sum(a)^2
    """
    n = a.size
    s = float(a.sum())
    if s < 1e-12:
        return 1e10, np.ones_like(a)
    c = _fft_convolve(a)
    smax, w = _smooth_max_and_grad(c, beta)
    grad_smax = _conv_grad_wrt_a(w, a)
    # df/da = 2*n * (grad_smax * s^2 - smax * 2*s) / s^4
    #       = 2*n * (grad_smax * s - 2*smax) / s^3
    obj = 2.0 * n * smax / (s * s)
    grad = 2.0 * n * (grad_smax * s - 2.0 * smax) / (s ** 3)
    return obj, grad


def _make_obj_grad(beta):
    """Return (objective, gradient) pair for scipy minimize."""
    def obj(a):
        val, _ = _obj_and_grad(a, beta)
        return val

    def grad(a):
        _, g = _obj_and_grad(a, beta)
        return g

    return obj, grad

def _adam_optimize(p0, beta_init, deadline, lr=0.05, beta_decay=0.85,

    from scipy.optimize import minimize
    
    n = p0.size
    p = p0 / p0.sum()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    
    beta = beta_init
    best_p = p.copy()
    best_score = _score_normalized(p)
    no_imp = 0
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for iteration in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(p)
        smax, w = _smooth_max_and_grad(c, beta)
        grad_smax = _conv_grad_wrt_a(w, p)
        s = float(p.sum())
        grad = 2.0 * n * (grad_smax * s - 2.0 * smax) / (s ** 3)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 1.5)
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        p_new = p - lr * mh / (np.sqrt(vh) + eps)
        p_new = np.maximum(p_new, 1e-14)
        p_new = p_new / p_new.sum()
        
        score = _score_normalized(p_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_p = p_new.copy()
            no_imp = 0
            p = p_new
        else:
            no_imp += 1
            p = p_new
        
        # Adaptive beta
        if t % 30 == 0:
            beta = max(1e-10, beta * beta_decay)
        
        # Restart
        if no_imp > no_imp_limit:
            p = best_p + 0.1 * np.random.default_rng(t).normal(size=n)
            p = np.maximum(p, 1e-14)
            p = p / p.sum()
            beta = max(1e-8, beta * 3)
            m.fill(0)
            v.fill(0)
            t = 0
            no_imp = 0
    
    return best_p, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_profiles(n):
    """Generate structured initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = {}
    
    # Linear
    for amp in [1.0, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + amp * u
        a = np.maximum(a, 1e-12)
        profiles[f'lin{amp}'] = a / a.sum()
    
    # Quadratic
    for a2 in [-0.5, 0.0, 0.5, 1.0]:
        a = 1.0 + a2 * u * u
        a = np.maximum(a, 1e-12)
        profiles[f'quad{a2}'] = a / a.sum()
    
    # Cubic
    for a1, a3 in [(1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (1.0, -1.0)]:
        a = 1.0 + a1 * u + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profiles[f'cubic{a1},{a3}'] = a / a.sum()
    
    # Cosine
    for freq, amp in [(1, 0.5), (1, 1.0), (1, 1.5), (2, 0.5), (2, 1.0), (3, 0.5)]:
        a = 1.0 + amp * np.cos(freq * np.pi * u)
        a = np.maximum(a, 1e-12)
        profiles[f'cos{freq}_{amp}'] = a / a.sum()
    
    # Lorentzian (U-shaped)
    for gamma in [1.0, 2.0, 3.0, 5.0]:
        a = 1.0 / (1.0 + gamma * u * u)
        profiles[f'lor{gamma}'] = a / a.sum()
    
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    1. L-BFGS with multi-temperature on parent
    2. Adam with adaptive beta
    3. Top-lag equalization
    4. Multi-length exploration
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(int(seed) % (2**31))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    print(f"Parent: n={parent.size if parent is not None else 'N/A'}")
    if parent is not None:
        sc_parent = _score_normalized(parent)
        print(f"Parent score: {sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    
    # Phase 1: L-BFGS on parent at n=1024 with multiple temperatures
    phase1_end = min(deadline, t_start + 400.0)
    n_work = 1024
    
    if parent is not None:
        p1024 = _interp_profile(parent, n_work) if parent.size != n_work else parent.copy()
    else:
        p1024 = np.ones(n_work) / n_work
        profiles = _make_profiles(n_work)
        best_prof = min(profiles.items(), key=lambda x: _score_normalized(x[1]))
        p1024 = best_prof[1]
        update_best(p1024, _score_normalized(p1024), "best_profile_init")
    
    print(f"\nPhase 1: L-BFGS multi-temperature (n={n_work})")
    
    # Try several beta values relative to the current max
    c1024 = _fft_convolve(p1024)
    maxc = float(c1024.max())
    s1024 = float(p1024.sum())
    
    # Normalize to sum=1 for cleaner objective
    p1024_n = p1024 / s1024
    maxc_n = float(_fft_convolve(p1024_n).max())
    
    beta_values = [0.05 * maxc_n, 0.02 * maxc_n, 0.01 * maxc_n, 0.005 * maxc_n, 0.001 * maxc_n]
    
    p_current = p1024_n.copy()
    for i, beta in enumerate(beta_values):
        if time.monotonic() > phase1_end:
            break
        per_deadline = min(phase1_end, time.monotonic() + 70.0)
        print(f"  L-BFGS beta={beta:.6f} (iter {i+1}/{len(beta_values)})")
        p_res, sc_res = _lbfgs_optimize(p_current, beta, per_deadline, maxiter=300)
        sc_full = _score_normalized(p_res)
        update_best(p_res, sc_full, f"lbfgs_beta{i}")
        if sc_full < _score_normalized(p_current):
            p_current = p_res.copy()
    
    # Phase 2: Adam optimization
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\nPhase 2: Adam optimization (deadline={time.monotonic()-t_start:.0f}s)")
    
    if best_seq is not None:
        p_adam = _interp_profile(best_seq, n_work) if best_seq.size != n_work else best_seq.copy()
        p_adam = p_adam / p_adam.sum()
    else:
        p_adam = p1024_n.copy()
    
    c_adam = _fft_convolve(p_adam)
    maxc_adam = float(c_adam.max())
    
    for beta_init in [0.03 * maxc_adam, 0.01 * maxc_adam, 0.003 * maxc_adam]:
        if time.monotonic() > phase2_end:
            break
        per_deadline = min(phase2_end, time.monotonic() + 80.0)
        p_res, sc_res = _adam_optimize(p_adam, beta_init, per_deadline,
                                        lr=0.03, beta_decay=0.88, no_imp_limit=100)
        sc_full = _score_normalized(p_res)
        update_best(p_res, sc_full, f"adam_beta={beta_init:.4f}")
        if sc_full < _score_normalized(p_adam):
            p_adam = p_res.copy()
    
    # Phase 3: Top-lag equalization
    phase3_end = min(deadline, t_start + 900.0)
    print(f"\nPhase 3: Top-lag equalization")
    
    if best_seq is not None:
        p_eq = best_seq.copy() / best_seq.sum()
    else:
        p_eq = p1024_n.copy()
    
    for top_k in [1, 2, 3, 4, 5]:
        if time.monotonic() > phase3_end:
            break
        per_deadline = min(phase3_end, time.monotonic() + 30.0)
        p_res, sc_res = _top_lag_equalize(p_eq, top_k=top_k, n_iter=300, lr=0.005)
        update_best(p_res, sc_res, f"equalize_k{top_k}")
        if sc_res < _score_normalized(p_eq):
            p_eq = p_res.copy()
    
    # Phase 4: Multi-length exploration
    phase4_end = min(deadline, t_start + 1300.0)
    print(f"\nPhase 4: Multi-length exploration")
    
    if best_seq is not None:
        for target_n in [1280, 1536]:
            if time.monotonic() > phase4_end:
                break
            p_long = _interp_profile(best_seq, target_n)
            sc_long = _score_normalized(p_long)
            update_best(p_long, sc_long, f"interp_n{target_n}")
            
            # Quick L-BFGS on the longer sequence
            c_long = _fft_convolve(p_long)
            maxc_long = float(c_long.max())
            beta_long = 0.02 * maxc_long
            per_deadline = min(phase4_end, time.monotonic() + 100.0)
            p_res, sc_res = _lbfgs_optimize(p_long, beta_long, per_deadline, maxiter=200)
            update_best(p_res, sc_res, f"lbfgs_n{target_n}")
    
    # Phase 5: Fine-tuned multi-restart
    phase5_end = min(deadline, t_start + 1650.0)
    print(f"\nPhase 5: Fine-tuned multi-restart (n={n_work})")
    
    if best_seq is not None:
        p_base = best_seq.copy() / best_seq.sum()
    else:
        p_base = p1024_n.copy()
    
    c_base = _fft_convolve(p_base)
    maxc_base = float(c_base.max())
    
    n_restarts = 8
    per_restart = min(35.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
    
    for i in range(n_restarts):
        if time.monotonic() > phase5_end:
            break
        per_deadline = min(phase5_end, time.monotonic() + per_restart)
        
        # Structured perturbation: modify a random contiguous block
        block_size = rng.integers(16, 64)
        start = rng.integers(0, n_work - block_size)
        p_pert = p_base.copy()
        p_pert[start:start + block_size] *= (1.0 + 0.2 * rng.normal(size=block_size))
        p_pert = np.maximum(p_pert, 1e-14)
        p_pert = p_pert / p_pert.sum()
        
        # L-BFGS with very small beta
        c_p = _fft_convolve(p_pert)
        beta_p = 0.002 * float(c_p.max())
        p_res, sc_res = _lbfgs_optimize(p_pert, beta_p, per_deadline, maxiter=200)
        update_best(p_res, sc_res, f"restart{i}")
    
    # Phase 6: Final ultra-fine polish
    phase6_end = deadline
    print(f"\nPhase 6: Final polish")
    
    if best_seq is not None:
        p_final = best_seq.copy() / best_seq.sum()
        # Adam with very small learning rate and beta
        c_f = _fft_convolve(p_final)
        maxc_f = float(c_f.max())
        p_res, sc_res = _adam_optimize(p_final, 0.001 * maxc_f, phase6_end,
                                        lr=0.01, beta_decay=0.95, no_imp_limit=500,
                                        max_iter=50000)
        update_best(p_res, sc_res, "final_polish")
        
        # Also try direct top-1 equalization
        p_res2, sc_res2 = _top_lag_equalize(p_final, top_k=1, n_iter=200, lr=0.001)
        update_best(p_res2, sc_res2, "final_eq1")
    
    # Final validation
    if best_seq is not None:
        final_seq = best_seq.copy()
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        
        # Exact score
        sc_exact = _true_score(final_seq)
        print(f"\nExact score (n={final_seq.size}): {sc_exact:.10f}")
        
        # Try evaluate_sequence
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  eval error: {e}")
        
        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]
    
    # Fallback
    print("FALLBACK: using uniform sequence")
    fallback = np.ones(256) / 256
    return [float(x) for x in fallback]
