                        continue

                bases[k] = base

            # Let me use a smarter approach.
            
            # Actually, conv[k] is already computed. The change when we modify a_i:
            # delta_conv[k] = (new_conv[k] - old_conv[k])
            # For k where a_i appears:
            #   if k = 2i: delta = new_ai^2 - old_ai^2
            #   if k != 2i and 0 <= k-i <= n-1: delta = 2*(new_ai - old_ai)*a_{k-i}

            # We want to find new_ai >= 0 that minimizes max_k new_conv[k]

                    continue

                    # We need base = conv[k] - old_ai^2
                    base = conv[k] - old_ai * old_ai
                    if base + 0 >= 0:  # target = 0
                        pass  # a_i = 0 already in candidates
                    # Try making this equal to the second-highest value
                else:
                    ki = k - i
                    if 0 <= ki <= n - 1 and ki != i:
                        # conv[k] = base + 2*a_i*a_{ki}
                        # base = conv[k] - 2*old_ai*a_{ki}

                        base = conv[k] - 2.0 * old_ai * a[ki]
                        # Try target = 0

                        if a_cand > 0:
                            candidates.append(a_cand)
                        # Try target = current_max * 0.99
                        T = current_max * 0.99
                        a_cand = (T - base) / (2.0 * a[ki]) if a[ki] > 1e-15 else 0.0
                        if a_cand > 0:
                            candidates.append(a_cand)
            
            # Try each candidate
            for new_ai in candidates:
                new_ai = max(0.0, new_ai)
                # Compute new max
                # Only affected lags change
                new_max = 0.0
                # Non-affected lags: max is at most current_max (unchanged)
                new_max = current_max  # upper bound from non-affected
                # Actually, non-affected lags keep their values, so their max is:
                non_affected_max = 0.0
                for k in range(2 * n - 1):
                    if k < k_min or k > k_max:
                        non_affected_max = max(non_affected_max, conv[k])
                # Affected lags:
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        val = conv[k] - old_ai * old_ai + new_ai * new_ai
                    else:
                        ki = k - i
                        if 0 <= ki <= n - 1 and ki != i:
                            val = conv[k] + 2.0 * (new_ai - old_ai) * a[ki]
                        elif ki == i:
                            # This means k = 2i, already handled
                            val = conv[k] - old_ai * old_ai + new_ai * new_ai
                        else:
                            val = conv[k]
                    new_max = max(new_max, val)
                
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_val = new_ai
            
            if best_val != old_ai:
                a[i] = best_val
                # Update convolution
                delta = best_val - old_ai
                for k in range(k_min, k_max + 1):
                    ki = k - i
                    if k == 2 * i:
                        conv[k] += best_val * best_val - old_ai * old_ai
                    elif 0 <= ki <= n - 1 and ki != i:
                        conv[k] += 2.0 * delta * a[ki]
                current_max = conv.max()
                improved = True
        
        if not improved:
            break
    
    # Re-normalize
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a, compute_score(a)

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses improved LP-based descent with adaptive step sizes, coordinate descent
polishing, and logit-space Adam refinement.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_step(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            x = np.maximum(res.x, 0.0)
            if x.sum() > 1e-10:
                return x
    except Exception:
        pass
    return None

def lp_step_tight(f, M, tightness=0.95):
    """LP with tighter constraint to force more improvement."""
    M_tight = M * tightness
    g = lp_step(f, M_tight)
    if g is not None:
        return g
    # Fallback: use original M
    return lp_step(f, M)

def line_search(f, g, M_f, M_fg, M_g, base_t=0.3):
    """Find best step size t for blend h = (1-t)*f + t*g."""
    best_t = base_t
    best_M = np.inf
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
        h = (1 - t) * f + t * g
        h = np.maximum(h, 0.0)
        s = h.sum()
        if s < 1e-10:
            continue
        h = h / s
        M_h = float(np.convolve(h, h).max())
        if M_h < best_M:
            best_M = M_h
            best_t = t
    return best_t, best_M

        return f_init.copy()

def coordinate_descent(a, max_rounds=50, deadline=None):
    """Coordinate descent polishing."""
    n = a.size
    a = np.array(a, dtype=np.float64).copy()
    s = a.sum()
    a = a / s
    
    conv = np.convolve(a, a)
    current_max = float(conv.max())
    
    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        any_improved = False
        for i in range(n):
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            old_ai = a[i]
            
            # Compute the max of non-affected lags
            # Non-affected lags are those k where k < k_min or k > k_max
            if k_min > 0:
                non_aff_max_left = float(conv[:k_min].max())
            else:
                non_aff_max_left = 0.0
            if k_max < 2 * n - 2:
                non_aff_max_right = float(conv[k_max + 1:].max())
            else:
                non_aff_max_right = 0.0
            non_aff_max = max(non_aff_max_left, non_aff_max_right)
            
            # Try candidate values for a_i
            candidates = [0.0, old_ai]
            
            # For peak lags, try values that would reduce them
            peak_threshold = current_max * 0.95
            for k in range(k_min, k_max + 1):
                if conv[k] < peak_threshold:
                    continue
                if k == 2 * i:
                    # conv[k] = base + a_i^2
                    base = conv[k] - old_ai * old_ai
                    # Try a_i that makes conv[k] = non_aff_max (if possible)
                    if base <= non_aff_max:
                        a_cand = np.sqrt(max(0, non_aff_max - base))
                        candidates.append(a_cand)
                    # Try a_i that makes conv[k] = current_max * 0.9
                    target = current_max * 0.9
                    if target >= base:
                        a_cand = np.sqrt(target - base)
                        candidates.append(a_cand)
                else:
                    ki = k - i
                    if 0 <= ki <= n - 1:
                        if ki == i:
                            continue  # Shouldn't happen since k != 2i
                        a_ki = a[ki]
                        if a_ki < 1e-15:
                            continue
                        base = conv[k] - 2.0 * old_ai * a_ki
                        # Try a_i that makes conv[k] = current_max * 0.9
                        target = current_max * 0.9
                        a_cand = (target - base) / (2.0 * a_ki)
                        if a_cand > 0:
                            candidates.append(a_cand)
                        # Try a_i = 0
                        # Already in candidates

            best_val = old_ai
            best_new_max = current_max
            
            for new_ai in candidates:
                new_ai = max(0.0, min(new_ai, 10.0))
                # Compute new max
                new_max = non_aff_max
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        val = conv[k] - old_ai * old_ai + new_ai * new_ai
                    else:
                        ki = k - i
                        if 0 <= ki <= n - 1 and ki != i:

                        else:
                            val = conv[k]
                    if val > new_max:

                    best_val = new_ai

                a[i] = best_val
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:

                    else:

                any_improved = True

    if s > 1e-10:

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def logit_anneal(p0, deadline, rng, lr=0.05, reg=1e-8):
    """Logit-space Adam polishing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0
    
    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.1 * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    
    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        
        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.85)
        
        if no_imp > max(500, 3 * n):
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            no_imp = 0
    
    return best_p, best_score

    return arr / arr.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP optimization -> coordinate descent -> logit Adam.
    Tries multiple n values and starting profiles.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    print(f"Starting search. budget={budget_s}s, deadline at t={deadline - t_start:.1f}s")
    
    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass
    
    if init_seq is not None:
        sc = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc:.10f}")
        update(init_seq, sc)
    
    # Phase 1: LP optimization at n=512
    n_main = 512
    phase1_end = min(deadline, t_start + 400.0)
    print(f"\n=== Phase 1: LP at n={n_main} ===")
    
    if init_seq is not None and init_seq.size != n_main:
        p_main = interpolate_profile(init_seq, n_main)
    elif init_seq is not None:
        p_main = init_seq.copy()
    else:
        p_main = make_linear_profile(n_main, 1.5)
    
    sc = compute_score(p_main)
    update(p_main, sc)
    print(f"  Start: score={sc:.10f}")
    
    lp_deadline = min(phase1_end, time.monotonic() + 250.0)
    result = lp_optimize_v2(p_main, lp_deadline, rng, verbose=True)
    sc = compute_score(result)
    update(result, sc)
    print(f"  After LP: score={sc:.10f}")
    
    # Try other starting profiles
    profiles = [
        ("lin_1.0", make_linear_profile(n_main, 1.0)),
        ("lin_2.0", make_linear_profile(n_main, 2.0)),
        ("lin_3.0", make_linear_profile(n_main, 3.0)),
        ("cos_1", make_cosine_profile(n_main, 1, 0.5)),
        ("cos_2", make_cosine_profile(n_main, 2, 0.4)),
        ("tri", make_triangle_profile(n_main)),
        ("para1", make_parabolic_profile(n_main, 1.0, -0.5)),
        ("para2", make_parabolic_profile(n_main, -0.5, 1.0)),
    ]
    
    for name, p in profiles:
        if time.monotonic() > phase1_end - 5:
            break
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.1:
            continue
        per_dl = min(phase1_end, time.monotonic() + 30.0)
        result = lp_optimize_v2(p, per_dl, rng, verbose=False)
        sc = compute_score(result)
        update(result, sc)
        if sc < best_score + 0.02:
            print(f"  Profile {name}: score={sc:.10f}")
    
    # Coordinate descent on current best
    if best_seq is not None and time.monotonic() < phase1_end - 10:
        print(f"  Coordinate descent on n={best_seq.size}...")
        cd_deadline = min(phase1_end, time.monotonic() + 60.0)
        cd_result, cd_score = coordinate_descent(best_seq, max_rounds=20, deadline=cd_deadline)
        update(cd_result, cd_score)
        print(f"  After CD: score={cd_score:.10f}")
    
    print(f"  Phase 1 best: {best_score:.10f}")
    
    # Phase 2: Try n=1024
    n_large = 1024
    phase2_end = min(deadline, t_start + 800.0)
    print(f"\n=== Phase 2: LP at n={n_large} ===")
    
    if time.monotonic() < phase2_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"  Start: score={sc:.10f}")
        
        lp_dl2 = min(phase2_end, time.monotonic() + 300.0)
        result = lp_optimize_v2(p_large, lp_dl2, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"  After LP: score={sc:.10f}")
        
        # CD on large
        if best_seq is not None and time.monotonic() < phase2_end - 10:
            cd_dl = min(phase2_end, time.monotonic() + 60.0)
            cd_result, cd_score = coordinate_descent(best_seq, max_rounds=10, deadline=cd_dl)
