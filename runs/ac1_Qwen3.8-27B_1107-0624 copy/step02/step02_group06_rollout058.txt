"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences of length n.
Key insight: optimal shape is asymmetric power-law with high-end taper (beta-distribution-like).
Uses diverse initialization, log-space Adam/Nesterov optimization, and coordinate descent polish.
"""
import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def make_beta_profile(n, alpha, beta_p):
    """a[j] proportional to (j+1)^(alpha-1) * (n-j)^(beta_p-1)"""
    j = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    if alpha != 1.0:
        a *= (j + 1.0) ** (alpha - 1.0)
    if beta_p != 1.0:
        a *= (n - j) ** (beta_p - 1.0)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def make_power_profile(n, p):
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0) ** p
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    return a / s


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_trapped_profile(n, p, taper):
    """Power law with exponential taper at the high end."""
    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    a = (j + 1.0) ** p
    if taper > 0.01:
        a *= np.exp(-taper * t ** 3)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    return a / s


def make_diverse_profiles(n):
    """Generate diverse initial profiles."""
    profs = []
    # Beta distribution profiles
    for alpha in [2.0, 2.5, 3.0, 3.5, 4.0, 4.5, 5.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
            profs.append(make_beta_profile(n, alpha, beta_p))
    # Power profiles
    for p in [1.5, 2.0, 2.5, 3.0, 3.5, 4.0]:
        profs.append(make_power_profile(n, p))
    # Trapped profiles
    for p in [2.0, 2.5, 3.0, 3.5, 4.0]:
        for taper in [0.5, 1.0, 2.0, 3.0, 5.0]:
            profs.append(make_trapped_profile(n, p, taper))
    # Linear
    for eta in [1.0, 1.5, 2.0, 2.5, 3.0]:
        profs.append(make_linear_profile(n, eta))
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))

                   b1=0.9, b2=0.999, nesterov=True,
                   restart_every=150, restart_scale=0.5):
    """Optimize in log-space using Adam or Nesterov+Adam hybrid."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size

    log_a = np.log(a0)
    log_a -= float(np.mean(log_a))

    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
