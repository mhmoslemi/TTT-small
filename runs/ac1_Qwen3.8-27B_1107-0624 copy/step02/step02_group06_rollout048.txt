"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses L-BFGS-B in logit space with soft-max objective, adaptive soft-max Adam,
golden-section coordinate descent, and multi-length search starting from known best.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):

def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s


def soft_max_val_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -50, 50))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def conv_gradient_p(p, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. p, where c = conv(p,p)."""
    n = p.size
    g = np.zeros(n, dtype=np.float64)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fw * fp, N)
    return 2.0 * h[:n]


def objective_and_grad_logits(z, beta):
    """Compute soft-max objective and gradient w.r.t. logits z."""
    p = softmax_from_z(z)
    n = p.size
    c = full_convolve_fft(p)
    L, w = soft_max_val_weights(c, beta)
    score = 2.0 * n * L
    if not np.isfinite(score):
        return score, np.zeros_like(z)
    gp = conv_gradient_p(p, w)
    dot = float(np.dot(p, gp))
    gz = p * (gp - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        gz = np.zeros_like(z)
    return score, gz


def lbfgs_optimize(p0, deadline, beta_init=0.15, beta_decay=0.95, max_iter=2000):
    """L-BFGS-B optimization in logit space with decreasing beta."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z0 = np.log(p0)
    z0 -= float(np.mean(z0))

    best_score = float(2.0 * n * float(full_convolve_fft(p0).max()))
    best_p = p0.copy()

    beta = beta_init * best_score
    t_start = time.monotonic()

    for iteration in range(max_iter):
        if time.monotonic() > deadline:
            break

        def obj_fun(z):
            sc, _ = objective_and_grad_logits(z, beta)
            if not np.isfinite(sc):
                return 1e10
            return sc

        def grad_fun(z):
            _, g = objective_and_grad_logits(z, beta)
            return g

        result = scipy_minimize(
            obj_fun, z0, jac=grad_fun,
            method='L-BFGS-B',
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )

        z_opt = result.x
        z_opt = np.clip(z_opt, -100.0, 100.0)
        z_opt -= float(np.mean(z_opt))
        p_opt = softmax_from_z(z_opt)
        sc = internal_score(p_opt)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_opt.copy()
            z0 = z_opt.copy()
        else:
            z0 = z_opt.copy()

        beta *= beta_decay

    return best_p, best_score


def soft_max_adam_v2(p0, deadline, rng, lr=0.05, beta_frac=0.15, beta_decay=0.85):
    """Improved soft-max Adam with adaptive beta and better restarts."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve_fft(p0)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_max_val_weights(c, beta)
        gp = conv_gradient_p(p0, w)
        dot = float(np.dot(p0, gp))
        gz = p0 * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -100.0, 100.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.01, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        p0 = p

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * beta_decay)

        if no_imp > max(200, 3 * n):
            scale = max(0.1, 1.0 / np.sqrt(n))
            z = best_z + scale * rng.standard_normal(n)
            z = np.clip(z, -100.0, 100.0)
            z -= float(np.mean(z))
            p0 = softmax_from_z(z)
            c = full_convolve_fft(p0)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_descent_golden(p0, deadline, rng, max_sweeps=20):
    """Coordinate descent with golden section search along each coordinate."""
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    best_p = p0.copy()
    best_score = internal_score(p0)
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    no_imp_total = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        sweep_improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p0[i]
            # Golden section search for optimal p[i]
            lo = orig * 0.1
            hi = orig * 5.0 + 0.1
            a = hi - GR * (hi - lo)
            b = lo + GR * (hi - lo)

            pa = p0.copy()
            pa[i] = a
            pa = pa / pa.sum()
            fa = internal_score(pa)

            pb = p0.copy()
            pb[i] = b
            pb = pb / pb.sum()
            fb = internal_score(pb)

            for _ in range(12):
                if fa < fb:
                    hi = b
                    b = a
                    fb = fa
                    a = hi - GR * (hi - lo)
                    pa = p0.copy()
                    pa[i] = a
                    pa = pa / pa.sum()
                    fa = internal_score(pa)
                else:
                    lo = a
                    a = b
                    fa = fb
                    b = lo + GR * (hi - lo)
                    pb = p0.copy()
                    pb[i] = b
                    pb = pb / pb.sum()
                    fb = internal_score(pb)

            opt_x = (lo + hi) / 2.0
            p_new = p0.copy()
            p_new[i] = opt_x
            p_new = np.maximum(p_new, 1e-15)
            p_new = p_new / p_new.sum()
            sc = internal_score(p_new)

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p_new.copy()
                p0 = p_new.copy()
                sweep_improved = True

        if not sweep_improved:
            no_imp_total += 1
            if no_imp_total >= 2:
                break
        else:
            no_imp_total = 0

    return best_p, best_score


def perturb_and_optimize(p0, deadline, rng, n_perturb=5, perturb_scale=0.05):
    """Apply random perturbations and re-optimize."""
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    best_p = p0.copy()
    best_score = internal_score(p0)

    per = (deadline - time.monotonic()) / max(1, n_perturb)

    for k in range(n_perturb):
        if time.monotonic() > deadline:
            break
        noise = rng.standard_normal(p0.size) * perturb_scale
        p_pert = p0 + noise
        p_pert = np.maximum(p_pert, 1e-12)
        p_pert = p_pert / p_pert.sum()

        # Quick Adam from perturbed point
        sub_deadline = min(deadline, time.monotonic() + per * 0.7)
        p_opt, sc = soft_max_adam_v2(p_pert, sub_deadline, rng, lr=0.03, beta_frac=0.1)
        sc_full = internal_score(p_opt)
        if np.isfinite(sc_full) and sc_full < best_score:
            best_score = sc_full
            best_p = p_opt.copy()

    return best_p, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_to_sum1(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_to_sum1(np.interp(x1, x0, p0))


def make_analytical_profiles(n, rng):
    """Generate diverse analytical profiles for initialization."""
    profs = []
    t = np.linspace(0.0, 1.0, n)

    for eta in (1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        a = np.exp(eta * t)
        profs.append(a / a.sum())

    for eta in (-1.0, -1.5, -2.0, -2.5, -3.0):
        a = np.exp(eta * t)
        profs.append(a / a.sum())

    for p in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        a = (t + 1e-8) ** p
        profs.append(a / a.sum())

    for p in (0.3, 0.5, 1.0, 1.5, 2.0):
        a = ((1 - t) + 1e-8) ** p
        profs.append(a / a.sum())

    for a1 in (0.5, 1.0, 1.5, 2.0):
        for a2 in (-0.5, 0.0, 0.5):
            u = 2.0 * t - 1.0
            a = 1.0 + a1 * u + a2 * u ** 2
            a = np.maximum(a, 1e-12)
            profs.append(a / a.sum())

    # Bump functions
    for width in (0.2, 0.3, 0.4, 0.5):
        for center in (0.3, 0.5, 0.7):
            a = np.exp(-((t - center) ** 2) / (2.0 * width ** 2))
            profs.append(a / a.sum())

    # Random smooth profiles
    for _ in range(10):
        coeffs = rng.uniform(-2, 2, 8)
        a = sum(c * np.cos(2.0 * np.pi * (k + 1) * t) for k, c in enumerate(coeffs)) + 3.0
        a = np.maximum(a, 1e-8)
        profs.append(a / a.sum())

    return profs

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print("  NEW BEST: {:.10f} n={} {}".format(best_score, best_seq.size, label))
        if len(candidates) < 30:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        elif float(sc) < candidates[-1][0]:
            candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
            candidates.sort(key=lambda x: x[0])

    # Phase 0: Load parent and initialize
    print("Phase 0: Initialization")
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_to_sum1(parent)
    except Exception:
        parent = None

    if parent is not None:
        sc = internal_score(parent)
        local_update(parent, sc, "parent")
        print("  Parent score: {:.10f} n={}".format(sc, parent.size))

    # Try analytical profiles at n=512
    n_main = 512
    init_deadline = min(deadline, t_start + 20.0)
    if parent is not None and parent.size != n_main:
        p_interp = interpolate_profile(parent, n_main)
        if p_interp is not None:
            sc = internal_score(p_interp)
            local_update(p_interp, sc, "parent_interp_512")

    profs_512 = make_analytical_profiles(n_main, rng)
    for p0 in profs_512:
        if time.monotonic() > init_deadline:
            break
        sc = internal_score(p0)
        local_update(p0, sc, "analytical_512")

    print("  After init: best={:.10f}".format(best_score))

    # Phase 1: L-BFGS-B at n=512
    print("Phase 1: L-BFGS-B at n=512")
    phase1_end = min(deadline, t_start + 500.0)
    starts_512 = []
    if best_seq is not None and best_seq.size == n_main:
        starts_512.append(best_seq.copy())
    elif parent is not None:
        p_i = interpolate_profile(parent, n_main)
        if p_i is not None:
            starts_512.append(p_i)
    for sc_c, p_c in candidates[:8]:
        if p_c.size != n_main:
            p_c = interpolate_profile(p_c, n_main)
        if p_c is not None:
            starts_512.append(p_c)

    seen_starts = 0
    for si, p_start in enumerate(starts_512[:6]):
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + 60.0)
        p_opt, sc = lbfgs_optimize(p_start, per, beta_init=0.15, beta_decay=0.93)
        sc_full = internal_score(p_opt)
        local_update(p_opt, sc_full, "lbfgs_512_{}".format(si))
        seen_starts += 1

    print("  After phase 1: best={:.10f}".format(best_score))

    # Phase 2: Soft-max Adam at n=512
    print("Phase 2: Soft-max Adam at n=512")
    phase2_end = min(deadline, t_start + 850.0)
    for i in range(min(10, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n_main:
            p_c = interpolate_profile(p_c, n_main)
        if p_c is None:
            continue
        p_c = normalize_to_sum1(p_c)
        per = min(phase2_end, time.monotonic() + 25.0)
        p_opt, sc = soft_max_adam_v2(p_c, per, rng, lr=0.04, beta_frac=0.12, beta_decay=0.88)
        sc_full = internal_score(p_opt)
        local_update(p_opt, sc_full, "adam_512_{}".format(i))

    print("  After phase 2: best={:.10f}".format(best_score))

    # Phase 3: Coordinate descent with golden section at n=512
    print("Phase 3: Golden-section coordinate descent at n=512")
    phase3_end = min(deadline, t_start + 1200.0)
    for i in range(min(5, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n_main:
            p_c = interpolate_profile(p_c, n_main)
        if p_c is None:
            continue
        p_c = normalize_to_sum1(p_c)
        per = min(phase3_end, time.monotonic() + 40.0)
        p_opt, sc = coordinate_descent_golden(p_c, per, rng, max_sweeps=8)
        sc_full = internal_score(p_opt)
        local_update(p_opt, sc_full, "cd_512_{}".format(i))

    print("  After phase 3: best={:.10f}".format(best_score))

    # Phase 4: Try n=1024
    print("Phase 4: Extended search at n=1024")
    phase4_end = min(deadline, t_start + 1550.0)
    n_ext = 1024
    if best_seq is not None:
        p_ext = interpolate_profile(best_seq, n_ext)
        if p_ext is not None:
            sc = internal_score(p_ext)
            local_update(p_ext, sc, "interpolated_1024")
            # Optimize at n=1024
            per = min(phase4_end, time.monotonic() + 60.0)
            p_opt, sc2 = soft_max_adam_v2(p_ext, per, rng, lr=0.03, beta_frac=0.10, beta_decay=0.90)
            sc_full = internal_score(p_opt)
            local_update(p_opt, sc_full, "adam_1024")

            # L-BFGS at n=1024
            if time.monotonic() < phase4_end:
                per2 = min(phase4_end, time.monotonic() + 60.0)
                p_opt2, sc3 = lbfgs_optimize(p_opt, per2, beta_init=0.10, beta_decay=0.94)
                sc_full2 = internal_score(p_opt2)
                local_update(p_opt2, sc_full2, "lbfgs_1024")

    print("  After phase 4: best={:.10f}".format(best_score))

    # Phase 5: Perturbation and re-optimization
    print("Phase 5: Perturbation exploration")
    phase5_end = min(deadline, t_start + 1720.0)
    if best_seq is not None:
        n_b = best_seq.size
        per = min(phase5_end, time.monotonic() + 100.0)
        p_opt, sc = perturb_and_optimize(best_seq, per, rng, n_perturb=4, perturb_scale=0.03)
        sc_full = internal_score(p_opt)
        local_update(p_opt, sc_full, "perturb")

    print("  After phase 5: best={:.10f}".format(best_score))

    # Phase 6: Final L-BFGS polish on best
    print("Phase 6: Final polish")
    phase6_end = deadline
    if best_seq is not None:
        per = min(phase6_end, time.monotonic() + 50.0)
        p_opt, sc = lbfgs_optimize(best_seq, per, beta_init=0.08, beta_decay=0.92)
        sc_full = internal_score(p_opt)
        local_update(p_opt, sc_full, "final_lbfgs")

        # Quick coordinate descent
        if time.monotonic() < phase6_end:
            per2 = min(phase6_end, time.monotonic() + 30.0)
            p_opt2, sc2 = coordinate_descent_golden(p_opt, per2, rng, max_sweeps=5)
            sc_full2 = internal_score(p_opt2)
            local_update(p_opt2, sc_full2, "final_cd")

    print("  After phase 6: best={:.10f}".format(best_score))

    # Final selection and verification
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        # Exact score using np.convolve
        if final_seq.size <= 2048:
            sc_exact = true_score(final_seq)
            if np.isfinite(sc_exact):
                final_score = sc_exact

        # Check all candidates
        for sc_c, p_c in candidates[:15]:
            if p_c.size <= 2048:
                sc_cx = true_score(p_c)
                if np.isfinite(sc_cx) and sc_cx < final_score:
                    final_score = sc_cx
                    final_seq = p_c

        # Verify with evaluate_sequence
        if final_seq is not None and final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass

        print("  FINAL: score={:.10f} n={}".format(final_score, final_seq.size))
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(512) / 512.0
    print("  FALLBACK: score={:.10f}".format(internal_score(fallback)))
    return [float(x) for x in fallback]
