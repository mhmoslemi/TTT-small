    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    m = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[m - 1] - 1.0) / m
    return np.maximum(v - theta, 0.0)

                noise = 0.01 * rng.normal(size=n)
                a_pert = a + noise
                a = project_simplex(a_pert)
                step = initial_step
                no_imp = 0
                prev_a = None
                prev_grad = None
        
        it += 1
        if verbose and it % 1000 == 0:
            print(f"  SGD iter {it}: max_conv={best_max:.10f}, step={step:.6f}")
        
        if no_imp > 500:
            break
    
    return best_a


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def lp_optimize(f_init, deadline, rng, verbose=False):
    """LP-based optimization."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)
    
    conv_f = full_convolve_fft(f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        M = best_M if stagnant > 0 else float(full_convolve_fft(f).max())
        
        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new
        
        M_new = float(full_convolve_fft(f_new).max())
        
        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
        else:
            t *= 0.5
            if t < 0.002:
                t = 0.002
            stagnant += 1
        
        t = max(0.005, t * 0.998)
        
        if stagnant > 100:
            noise_scale = 0.05 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            stagnant = 0
    
    return best_f
