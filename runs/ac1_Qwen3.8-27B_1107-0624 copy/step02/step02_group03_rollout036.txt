"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses p-power annealing with Adam, soft-argmax polish, and multi-scale search.
"""

import numpy as np
import time

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def softmax(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def p_power_grad(a, p):
    """
    Compute gradient of sum(conv(a,a)[k]^p) w.r.t. a.
    Returns (loss, grad) where grad has same shape as a.
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    loss = np.sum(c**p)
    
    w = p * c**(p-1)  # length 2n-1
    
    # grad[j] = sum_m a[m]*w[m+j] + a[j]*sum_{k=j}^{j+n-1} w[k]
    # First term: cross-correlation
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(a, w[j:j+n])
    
    # Second term: sliding window sum
    cumsum_w = np.concatenate([[0.0], np.cumsum(w)])
    window_sums = cumsum_w[n:] - cumsum_w[:n]  # length n
    grad += a * window_sums
    
    return loss, grad

def adam_p_power(p0, deadline, p_start=4, p_end=512, verbose=True):
    """
    Adam optimization in softmax space, minimizing p-power loss with annealed p.
    p0: initial sequence (will be normalized internally).
    Returns (best_p, best_score, best_z).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    
    z = np.log(p0)
    z -= z.mean()  # This is the initial logit (softmax(z) ≈ p0)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    
    best_p = p0.copy()
    best_score = 2.0 * n * float(np.convolve(p0, p0).max())
    best_z = z.copy()
    
    lr = 0.05
    no_imp = 0
    
    # p annealing schedule: stay at each p for a certain number of iterations
    p_schedule = [4, 8, 16, 32, 64, 128, 256, 512]
    p_dur = 200  # iterations per p level
    
    while time.monotonic() < deadline:
        it += 1
        
        # Determine current p
        p_idx = min((it - 1) // p_dur, len(p_schedule) - 1)
        p = p_schedule[p_idx]
        
        # Compute a from z
        a = softmax(z)
        
        # Compute gradient of p-power loss
        _, grad_a = p_power_grad(a, p)
        
        # Convert to z-space gradient (softmax Jacobian)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); it = 0
            continue
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z * grad_z
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= z_new.mean()
        
        # Check if the new point is valid
        a_new = softmax(z_new)
        if a_new.sum() < 1e-30 or not np.all(np.isfinite(a_new)):
            continue
        
        z = z_new
        
        # Evaluate actual score
        c = np.convolve(a_new, a_new)
        score = 2.0 * n * float(c.max())
        
        if score < best_score - 1e-10:
            best_score = score
            best_p = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Learning rate schedule
        if no_imp > 500:
            lr = max(0.005, lr * 0.95)
            if no_imp > 2000:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                z = best_z + 0.1 * rng.normal(size=n)
                z -= z.mean()
                m.fill(0); v.fill(0); it = 0
                lr = 0.05
                no_imp = 0
        
        if verbose and it % 500 == 0:
            print(f"  p-power it={it}, p={p}, score={best_score:.8f}, lr={lr:.4f}")
    
    return best_p, best_score, best_z

def soft_argmax_polish(p0, deadline, verbose=True):
    """
    Adam optimization in softmax space, minimizing log-sum-exp (soft argmax) of conv(a,a).
    This is a direct approach to minimizing max(conv(a,a)).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    
    z = np.log(p0)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    
    best_p = p0.copy()
    best_score = 2.0 * n * float(np.convolve(p0, p0).max())
    best_z = z.copy()
    
    lr = 0.03
    beta = 0.01  # temperature for soft argmax (start small for sharpness)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        it += 1
        
        a = softmax(z)
        c = np.convolve(a, a)
        max_c = float(c.max())
        
        # Soft argmax weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        
        # Gradient: sum_k w[k] * dc[k]/da[j]
        # dc[k]/da[j] = (a[k-j] + a[j]) * 1[0<=k-j<n]
        grad_a = np.zeros(n)
        for j in range(n):
            # sum_{k: 0<=k-j<n} w[k]*(a[k-j] + a[j])
            indices = np.arange(max(0, j), min(2*n-1, j+n))
            vals = w[indices]
            # a[k-j] for k in indices: a[0], a[1], ..., a[n-1]
            # a[j] for all k in indices: a[j] * len(indices)
            grad_a[j] = np.dot(a, vals) + a[j] * np.sum(vals)
        
        # Convert to z-space
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); it = 0
            continue
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z * grad_z
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= z_new.mean()
        
        a_new = softmax(z_new)
        z = z_new
        
        score = 2.0 * n * float(np.convolve(a_new, a_new).max())
        
        if score < best_score - 1e-10:
            best_score = score
            best_p = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Anneal beta (make soft argmax sharper)
        if it % 200 == 0:
            beta = max(1e-6, beta * 0.7)
        
        # Learning rate schedule
        if no_imp > 300:
            lr = max(0.003, lr * 0.95)
        if no_imp > 1500:
            # Restart
            rng = np.random.default_rng(it)
            z = best_z + 0.05 * rng.normal(size=n)
            z -= z.mean()
            m.fill(0); v.fill(0); it = 0
            lr = base_lr
            no_imp = 0
            beta = 0.01
        
        if verbose and it % 500 == 0:
            print(f"  soft-argmax it={it}, beta={beta:.6f}, score={best_score:.8f}")
    
    return best_p, best_score, best_z

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses p-power annealing with Adam in softmax space, soft-argmax polish,
and multi-scale search (n=512 and n=1024).
"""

import numpy as np
import time


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def softmax_stable(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(z, 1.0 / len(z))
    return e / s


def compute_conv_grad(a, w):
    """
    Compute sum_k w[k] * dc[k]/da[j] for all j.
    where c = conv(a,a), dc[k]/da[j] = (a[k-j]+a[j]) * 1[0<=k-j<n].
    Result: grad[j] = sum_m a[m]*w[m+j] + a[j]*sum(w[j:j+n]).
    Uses matrix form for speed.
    """
    n = len(a)
    # W[j,m] = w[j+m], then grad_first = W @ a
    idx = np.add.outer(np.arange(n), np.arange(n))
    W = w[idx]
    grad_first = W @ a
    # Second term: a[j] * sum(w[j:j+n])
    cumsum_w = np.concatenate([[0.0], np.cumsum(w)])
    window_sums = cumsum_w[n:] - cumsum_w[:n]
    grad_second = a * window_sums
    return grad_first + grad_second


def p_power_loss_grad(a, p):
    """Compute sum(c[k]^p) and gradient w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    # Normalize to avoid overflow
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros(n)
    c_norm = c / c_max
    loss = np.sum(c_norm ** p)
    # Gradient of sum(c_norm[k]^p) w.r.t. a, then scale back
    # d/d(a) sum((c[k]/cmax)^p) = (p/cmax) * c_norm[k]^(p-1) * dc[k]/da
    # The factor (p/cmax) is constant, doesn't affect direction
    w = p * (c_norm ** (p - 1))
    grad = compute_conv_grad(a, w)
    return loss, grad


def adam_p_power_optimize(p0, deadline, p_schedule=None, iters_per_p=300, verbose=True, seed=0):
    """
    Adam optimization in softmax space with p-power annealing.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= z.mean()

    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512]

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    lr = 0.04
    base_lr = lr

    best_p = p0.copy()
    best_score = 2.0 * n * float(np.convolve(p0, p0).max())
    best_z = z.copy()
    no_imp = 0

    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        it += 1
        p_idx = min((it - 1) // iters_per_p, len(p_schedule) - 1)
        p = p_schedule[p_idx]

        a = softmax_stable(z)
        _, grad_a = p_power_loss_grad(a, p)

        # Softmax Jacobian
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            it = 0
            continue

            best_z = z.copy()

    no_imp = 0

def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def coordinate_descent_polish(a, max_sweeps=500, verbose=True):
    """
    Coordinate descent: try decreasing each element and redistributing mass.
    """
    n = len(a)
    a = a.copy()
    best_score = internal_score(a)
    no_imp_sweeps = 0

    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            if a[j] < 1e-6:
                continue
            # Try reducing a[j] by delta and adding to a random other position
            delta = a[j] * 0.01
            if delta < 1e-8:
                continue
            a_new = a.copy()
            a_new[j] -= delta
            # Add delta to position k that minimizes the increase in max conv
            # Simple heuristic: spread delta to a few neighbors
            k = (j + n // 2) % n  # far position
            a_new[k] += delta
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = internal_score(a_new)
            if sc < best_score - 1e-12:
                a = a_new
                best_score = sc
                improved = True

        if improved:
            no_imp_sweeps = 0
        else:
            no_imp_sweeps += 1
            if no_imp_sweeps > 5:
                break

        if verbose and sweep % 100 == 0:
            print(f"    coord-descent sweep {sweep}: score={best_score:.8f}")

    return a, best_score

        print(f"Phase 3: scale to n={n_large}")
        print(f"  Time budget: {remaining:.0f}s")
        print(f"{'='*60}")

        a_large = interpolate_seq(best_seq, n_large)
        if a_large is not None:
            sc_large = internal_score(a_large)
            print(f"  Interpolated score: {sc_large:.8f}")

            # p-power at n=1024
            p3_deadline = min(deadline - 100, time.monotonic() + 0.4 * remaining)
            p_best3, sc_best3, _ = adam_p_power_optimize(
                a_large, p3_deadline,
                p_schedule=[16, 32, 64, 128, 256],
                iters_per_p=300, verbose=True, seed=seed + 2
            )
            if sc_best3 < best_score:
                best_score = sc_best3
                best_seq = p_best3
                print(f"  Phase 3 p-power: score={best_score:.8f}")
            elif sc_large < best_score:
                best_score = sc_large
                best_seq = a_large
                print(f"  Phase 3 interpolated: score={best_score:.8f}")

            # Soft-argmax at n=1024
            if time.monotonic() < deadline - 50:
                p4_deadline = min(deadline - 30, time.monotonic() + 0.3 * remaining)
                p_best4, sc_best4, _ = soft_argmax_optimize(
                    best_seq, p4_deadline, beta_init=0.01, verbose=True, seed=seed + 3
                )
                if sc_best4 < best_score:
                    best_score = sc_best4
                    best_seq = p_best4
                    print(f"  Phase 3 polish: score={best_score:.8f}")

    # === Phase 4: Final polish on best ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*60}")
        print(f"Phase 4: Final polish, n={len(best_seq)}, score={best_score:.8f}")
        print(f"  Time budget: {deadline - time.monotonic():.0f}s")
        print(f"{'='*60}")

        p5_deadline = deadline
        p_best5, sc_best5, _ = soft_argmax_optimize(
            best_seq, p5_deadline, beta_init=0.005, verbose=True, seed=seed + 4
        )
        if sc_best5 < best_score:
            best_score = sc_best5
            best_seq = p_best5
            print(f"  Phase 4: score={best_score:.8f}")

    # Final validation
    final_score = internal_score(best_seq)
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={final_score:.8f}")
    print(f"{'='*60}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        t = np.linspace(0, 1, 512)
        best_seq = np.maximum(1.0 + 1.5 * (2 * t - 1), 1e-10)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
