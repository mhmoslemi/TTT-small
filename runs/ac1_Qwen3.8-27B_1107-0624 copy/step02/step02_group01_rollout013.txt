"""
Multi-stage gradient-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing, softmax parameterization, Adam optimizer,
multi-scale refinement, and adaptive temperature scheduling.
"""

import numpy as np
import time
import math


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    if n <= 32 and m <= 32:
        return np.convolve(a, b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_gradient_smoothed(a, beta):
    """
    Compute the gradient of the log-sum-exp smoothed max of conv(a,a).
    Returns (gradient, L_value, weights).
    """
    n = a.size
    c = fft_convolve(a)  # length 2n-1
    # Log-sum-exp for numerical stability
    c_max = float(c.max())
    x = (c - c_max) / beta
    ex = np.exp(np.clip(x, -200.0, 50.0))
    s_ex = float(ex.sum())
    if s_ex < 1e-300:
        # Fallback: uniform weights
        w = np.ones_like(c) / c.size
    else:
        w = ex / s_ex
    
    L = c_max + beta * math.log(s_ex)
    
    # Compute gradient: grad[i] = sum_m w[m] * (a[m-i] + a[i-m])
    N_corr = next_pow2(3 * n)
    fa = np.fft.rfft(a, N_corr)
    fw = np.fft.rfft(w, N_corr)
    
    # u[i] = sum_j w[i+j] * a[j] = cross-correlation of w and a at lag i
    u = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]
    # v[i] = sum_j w[i-j] * a[j] = convolution of w and a at position i
    v = np.fft.irfft(fw * fa, N_corr)[:n]
    
    grad = u + v
    return grad, L, w


def softmax(z):
    z = z - z.max()
    a = np.exp(z)
    a = a / a.sum()
    return a


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=10):

    a = np.ones(n)

        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def adam_optimize(f0, deadline, rng, max_iter=200000, lr=None, beta0=None,
                  beta_decay=0.9999, restart_after=2000, verbose=False):
    """
    Adam-based optimization in log-space with log-sum-exp smoothing.
    f0: initial feasible point (non-negative, sum=1)
    Returns (best_f, best_M).
    """
    n = f0.size
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    f = f / s
    
    # Initialize logits
    z = np.log(f)
    z -= z.mean()
    
    if lr is None:
        lr = 0.01
    if beta0 is None:
        c = fft_convolve(f)
        beta = float(c.max()) * 0.1
    else:
        beta = beta0
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_f = f.copy()
    best_L = float(fft_convolve(f).max())
    no_imp = 0
    
    N_corr = next_pow2(3 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_adam += 1
        
        # Compute gradient
        grad_a, L_val, w = compute_gradient_smoothed(f, beta)
        
        # Log-space gradient
        S = float(np.dot(grad_a, f))
        grad_z = f * (grad_a - S)
        
        if not np.all(np.isfinite(grad_z)):
            # Reset
            z = best_f.size and np.log(np.maximum(best_f, 1e-15))
            z -= z.mean()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            f = best_f.copy()
            beta = float(fft_convolve(f).max()) * 0.1
            continue
        
        # Adam update
        m_adam = b1 * m_adam + (1.0 - b1) * grad_z
        v_adam = b2 * v_adam + (1.0 - b2) * (grad_z * grad_z)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100.0, 100.0)
        z -= z.mean()
        
        # Convert back to a
        f = softmax(z)
        
        # Evaluate
        c = fft_convolve(f)
        M_new = float(c.max())
        
        if M_new < best_L - 1e-12:
            best_L = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        beta *= beta_decay
        beta = max(beta, best_L * 0.001)
        
        # Restart if stuck
        if no_imp > restart_after:
            noise_scale = max(0.1, 0.5 * no_imp / restart_after)
            z_noise = 0.8 * z + noise_scale * rng.normal(size=n)
            z = np.clip(z_noise, -50.0, 50.0)
            z -= z.mean()
            f = softmax(z)
            c = fft_convolve(f)
            M_cur = float(c.max())
            beta = max(0.05 * M_cur, best_L * 0.05)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            no_imp = 0
            if M_cur < best_L:
                best_L = M_cur
                best_f = f.copy()
        
        if verbose and it % 10000 == 0:
            print(f"      iter {it}: M={M_new:.10f}, best={best_L:.10f}, beta={beta:.6f}")
    
    return best_f, best_L


def coordinate_refine(f0, deadline, max_sweeps=10):
    """
    Coordinate-wise refinement: for each coordinate, find optimal value
    via ternary search on the 1D objective.
    """
    n = f0.size
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0
    f = f / s
    
    best_score = compute_score(f)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Optimize coordinate i
            # (a*a)_k is linear in a_i with coefficient (a_{i+k} + a_{i-k})
            # so max_k (a*a)_k is piecewise-linear convex in a_i
            # Find minimum via ternary search or VP algorithm
            
            a_i_old = f[i]
            
            # Try a few values
            # The optimal a_i is where the "binding" constraints balance
            # Simple approach: ternary search on [0, a_i_old * 3]
            lo = 0.0
            hi = a_i_old * 3.0 + 0.1 / n
            
            # Evaluate at current
            def obj_at(val):
                f[i] = val
                f[i+0] = val  # make sure
                # Recompute max conv
                c = fft_convolve(f)
                return c.max()
            
            # Save
            save_i = f[i]
            
            # Ternary search
            for _ in range(20):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                f[i] = m1
                o1 = fft_convolve(f).max()
                f[i] = m2
                o2 = fft_convolve(f).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            f[i] = (lo + hi) / 2
            new_val = fft_convolve(f).max()
            
            if new_val < obj_at(save_i) - 1e-14:
                improved = True
            else:
                f[i] = save_i
        
        if not improved:
            break
    
    s = f.sum()
    f = f / s
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage gradient-based search:
    1. Generate diverse initial profiles at multiple lengths
    2. Optimize with Adam + log-sum-exp smoothing
    3. Multi-scale refinement
    4. Coordinate refinement
    5. Final evaluation
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")
    else:
        print("No parent available, starting from scratch")
    
    # Define lengths to try
    lengths = [256, 384, 512, 640, 768, 1024]
    
    # Phase 1: Quick optimization at n=512 from parent
    n1 = 512
    if parent is not None and parent.size != n1:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            sc = compute_score(p_int)
            try_update(p_int, sc)
    
    # Generate diverse starts
    starts_512 = []
    if parent is not None:
        p_use = interpolate_profile(parent, n1) if parent.size != n1 else parent.copy()
        if p_use is not None:
            starts_512.append(("parent", p_use))
    
    for idx, p in enumerate(make_deterministic_profiles(n1)[:12]):
        starts_512.append((f"det{idx}", p))
    
    for idx in range(6):
        starts_512.append((f"rand{idx}", make_smooth_random(n1, rng)))
    
    # Additional diverse starts
    t = np.linspace(0, 1, n1)
    for alpha in (1.5, 2.0, 2.5, 3.0):
        a = 1.0 + alpha * np.cos(np.pi * t)  # dips in middle
        a = np.maximum(a, 1e-12)
        starts_512.append((f"cos{alpha}", a / a.sum()))
    
    a = 1.0 + 1.5 * t - 1.0 * t**2
    a = np.maximum(a, 1e-12)
    starts_512.append(("quad1", a / a.sum()))
    
    a = 0.3 + 1.4 * t + 0.3 * t**2
    a = np.maximum(a, 1e-12)
    starts_512.append(("quad2", a / a.sum()))
    
    print(f"Phase 1: {len(starts_512)} starts at n={n1}")
    
    # Phase 1: Optimize each start with Adam
    phase1_budget = min(deadline, t_start + 600.0)
    per_start = max(5.0, (phase1_budget - t_start) / len(starts_512))
    
    candidates = []
    for i, (name, p0) in enumerate(starts_512):
        if time.monotonic() > phase1_budget:
            break
        p0 = np.maximum(p0, 1e-15)
        p0 = p0 / p0.sum()
        
        per = min(phase1_budget, time.monotonic() + per_start)
        p_best, M_best = adam_optimize(p0, per, rng, max_iter=100000, lr=0.02,
                                       beta0=float(fft_convolve(p0).max()) * 0.15)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates.append((sc, p_best))
        
        if i % 3 == 0:
            print(f"  Phase 1 [{i}/{len(starts_512)}] ({name}): score={sc:.10f}, global_best={best_score:.10f}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")
    
    # Phase 2: Extended optimization on top candidates at n=512
    phase2_end = min(deadline, t_start + 1080.0)
    top_k = min(4, len(candidates))
    per_cand = max(20.0, (phase2_end - t_start) / max(1, top_k))
    
    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break

                                       beta0=float(fft_convolve(p0).max()) * 0.05,
                                       beta_decay=0.99995, restart_after=5000)
