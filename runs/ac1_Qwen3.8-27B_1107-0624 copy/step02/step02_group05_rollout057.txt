"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses parametric polynomial initialization + accelerated proximal gradient (FISTA/Adam)
with softmax approximation of the max, FFT-based gradients, and multi-scale search.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * len(a) * np.max(c) / (s * s))

def make_poly_profile(n, coeffs):
    """
    Generate polynomial profile: a[i] = max(0, sum_j coeffs[j] * P_j(2*i/(n-1) - 1))
    where P_j are Legendre polynomials.
    """
    x = 2.0 * np.arange(n, dtype=np.float64) / max(n - 1, 1) - 1.0
    f = np.zeros(n)
    for j, c in enumerate(coeffs):
        f += c * legendre_poly(x, j)
    f = np.maximum(f, 1e-12)
    return f / f.sum()

def legendre_poly(x, n):
    """Compute Legendre polynomial P_n(x) using recurrence."""
    if n == 0:
        return np.ones_like(x)
    if n == 1:
        return x.copy()
    p0 = np.ones_like(x)
    p1 = x.copy()
    for k in range(1, n):
        p2 = ((2 * k + 1) * x * p1 - k * p0) / (k + 1)
        p0, p1 = p1, p2
    return p1

def poly_score(n, coeffs):
    """Score of a polynomial profile."""
    a = make_poly_profile(n, coeffs)
    return compute_score(a)

def optimize_poly(n, degree, rng, max_iter=2000):
    """Optimize polynomial coefficients to minimize score."""
    def obj(params):
        coeffs = [1.0] + list(params[:degree])
        # Add small regularizer to keep coefficients bounded
        score = poly_score(n, coeffs)
        if not np.isfinite(score):
            return 1e10
        return score
    
    x0 = rng.uniform(-2, 2, degree)
    res = scipy_minimize(obj, x0, method='Nelder-Mead', 
                         options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
    best_coeffs = [1.0] + list(res.x[:degree])
    a = make_poly_profile(n, best_coeffs)
    return a, compute_score(a)

def fft_cross_corr(a, w):
    """
    Compute h[i] = sum_k w[k] * a[k-i] for i=0..n-1 (cross-correlation).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    m = len(w)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation: (a ⋆ w)[i] = sum_j a[j] * w[j+i]
    # = (a * w_reversed)[i + m - 1] in convolution terms
    # Using FFT: ifft(conj(F_a) * F_w) gives the cross-correlation
    h = np.fft.irfft(np.conj(fa) * fw, N)
    # h[i] = sum_j a[j] * w[j+i] for valid j
    # We want h[0:n]
    return h[:n]

def soft_max_and_weights(c, beta):
    """Compute soft max value and softmax weights."""
    m = float(np.max(c))
    x = (c - m) / beta
    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    s = float(np.sum(ex))

"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses parametric polynomial initialization + accelerated proximal gradient (FISTA/Adam)
with adaptive softmax approximation of max, FFT-based cross-correlation gradients,
and multi-scale (multi-n) search with diverse restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * len(a) * np.max(c) / (s * s))


def legendre_poly(x, deg):
    """Compute Legendre polynomial P_deg(x) via Clenshaw recurrence."""
    if deg == 0:
        return np.ones_like(x)
    if deg == 1:
        return x.copy()
    p0 = np.ones_like(x)
    p1 = x.copy()
    for k in range(1, deg):
        p2 = ((2 * k + 1) * x * p1 - k * p0) / (k + 1)
        p0, p1 = p1, p2
    return p1


def make_poly_profile(n, coeffs):
    """Generate polynomial profile on [0,1] using Legendre basis."""
    x = 2.0 * np.arange(n, dtype=np.float64) / max(n - 1, 1) - 1.0
    f = np.zeros(n)
    for j, c in enumerate(coeffs):
        f += c * legendre_poly(x, j)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def poly_score(n, coeffs):
    """Score of a polynomial profile."""
    a = make_poly_profile(n, coeffs)
    return compute_score(a)


def optimize_poly(n, degree, rng, max_iter=1000):
    """Optimize polynomial coefficients via Nelder-Mead."""
    def obj(params):
        coeffs = [1.0] + list(params[:degree])
        try:
            sc = poly_score(n, coeffs)
        except Exception:
            return 1e10
        if not np.isfinite(sc):
            return 1e10
        return sc

    best_a = None
    best_sc = np.inf
    for trial in range(3):
        x0 = rng.uniform(-2, 2, degree)
        try:
            res = scipy_minimize(obj, x0, method='Nelder-Mead',
                                 options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-11})
            sc = float(res.fun)
            if sc < best_sc:
                best_sc = sc
                coeffs = [1.0] + list(res.x[:degree])
                best_a = make_poly_profile(n, coeffs)
        except Exception:
            pass
    if best_a is None:
        best_a = np.ones(n) / n
        best_sc = compute_score(best_a)
    return best_a, best_sc


def fft_cross_corr(a, w):
    """
    Compute h[i] = sum_k w[k] * a[k-i] for i=0..n-1 (cross-correlation).
    Uses FFT for O(n log n) efficiency.
    """
    n = len(a)
    m = len(w)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation via FFT: ifft(conj(F_a) * F_w)
    h = np.fft.irfft(np.conj(fa) * fw, N)
    return h[:n]


def soft_max_weights(c, beta):
    """Compute softmax weights over convolution values (for gradient computation)."""
    m = float(np.max(c))
    x = (c - m) / beta
    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s

def interpolate_to_length(a, n_new):
    """Linearly interpolate a sequence to a new length."""
    n0 = len(a)
    if n0 == n_new:

    x0 = np.linspace(0, 1, n0)

        nonlocal best_seq, best_score
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_seq = a.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: score={sc:.10f}, n={len(a)} ***")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                parent_score = compute_score(parent)
                print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
                try_update(parent, parent_score)
    except Exception:
        pass

    # === Phase 1: Parametric polynomial optimization ===
    n1 = 256
    phase1_end = min(deadline, t_start + 90.0)
    print(f"\n=== Phase 1: Parametric poly at n={n1} (budget {phase1_end - t_start:.0f}s) ===")

    poly_results = []
    for degree in [1, 2, 3, 4, 5]:
        if time.monotonic() > phase1_end:
            break
        a, sc = optimize_poly(n1, degree, rng, max_iter=800)
        poly_results.append((sc, a))
        print(f"  degree {degree}: score={sc:.10f}")

    poly_results.sort(key=lambda x: x[0])
    for sc, a in poly_results[:4]:
        try_update(a, sc)
    print(f"Phase 1 best: {best_score:.10f}")

    # === Phase 2: Full optimization at n=256 ===
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 2: Full prox-grad at n={n1} (budget {phase2_end - t_start:.0f}s) ===")

    starts = make_diverse_starts(n1, rng, count=8)
    if parent is not None:
        if len(parent) != n1:
            starts.append(interpolate_to_length(parent, n1))
        else:
            starts.append(parent.copy())
    for sc, a in poly_results[:3]:
        starts.append(a.copy())

    # Evaluate and track starts
    start_scores = []
    for s in starts:
        sc = compute_score(s)
        start_scores.append((sc, s))
        try_update(s, sc)
    start_scores.sort(key=lambda x: x[0])

    results2 = []
    n_starts = len(starts)
    for i, (sc_s, s) in enumerate(start_scores):
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        alloc = remaining / max(1, n_starts - i)
        alloc = min(alloc, 150.0)
        if alloc < 10:
            break
        run_deadline = time.monotonic() + alloc
        f_best, M_best = prox_grad_optimize(s, n1, run_deadline, rng,
                                            initial_lr=0.08, initial_beta=0.04)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results2.append((sc, f_best))
        print(f"  Start {i+1}/{n_starts}: score={sc:.10f} (best={best_score:.10f})")

    results2.sort(key=lambda x: x[0])
    print(f"Phase 2 best: {best_score:.10f}")

    # === Phase 3: Extension to n=512 ===
    n3 = 512
    phase3_end = min(deadline, t_start + 1250.0)
    print(f"\n=== Phase 3: Extension to n={n3} (budget {phase3_end - t_start:.0f}s) ===")

    candidates3 = []
    top_results = results2[:4]
    if best_seq is not None and best_seq.size == n1:
        top_results.append((best_score, best_seq))
    seen = set()
    for sc, a in sorted(top_results, key=lambda x: x[0]):
        if a is None:
            continue
        key = tuple(np.round(a[:5], 4)) if len(a) >= 5 else tuple(a)
        if key in seen:
            continue
        seen.add(key)
        candidates3.append(interpolate_to_length(a, n3))

    results3 = []
    for i, s in enumerate(candidates3):
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        alloc = remaining / max(1, len(candidates3) - i)
        alloc = min(alloc, 180.0)
        if alloc < 15:
            break
        run_deadline = time.monotonic() + alloc
        f_best, M_best = prox_grad_optimize(s, n3, run_deadline, rng,
                                            initial_lr=0.05, initial_beta=0.025)
        sc = 2.0 * n3 * M_best
        try_update(f_best, sc)
        results3.append((sc, f_best))
        print(f"  n={n3} run {i+1}/{len(candidates3)}: score={sc:.10f} (best={best_score:.10f})")

    results3.sort(key=lambda x: x[0])
    print(f"Phase 3 best: {best_score:.10f}")

    # === Phase 4: Try n=128 ===
    n4 = 128
    phase4_end = min(deadline, t_start + 1450.0)
    print(f"\n=== Phase 4: Try n={n4} (budget {phase4_end - t_start:.0f}s) ===")

    candidates4 = []
    all_good = []
    if best_seq is not None:
        all_good.append(best_seq)
    for sc, a in results2[:2] + results3[:2]:
        if a is not None:
            all_good.append(a)

    for a in all_good[:4]:
        candidates4.append(interpolate_to_length(a, n4))

    results4 = []
    for i, s in enumerate(candidates4):
        remaining = phase4_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, len(candidates4) - i), 60.0)
        if alloc < 8:
            break
        run_deadline = time.monotonic() + alloc
        f_best, M_best = prox_grad_optimize(s, n4, run_deadline, rng,
                                            initial_lr=0.1, initial_beta=0.06)
        sc = 2.0 * n4 * M_best
        try_update(f_best, sc)
        results4.append((sc, f_best))
        print(f"  n={n4} run {i+1}: score={sc:.10f} (best={best_score:.10f})")

    print(f"Phase 4 best: {best_score:.10f}")

    # === Phase 5: Final refinement ===
    phase5_end = deadline
    remaining = phase5_end - time.monotonic()
    print(f"\n=== Phase 5: Final refinement (budget {remaining:.0f}s) ===")

    if best_seq is not None and remaining > 60:
        n_best = len(best_seq)
        f_best, M_best = prox_grad_optimize(best_seq, n_best, phase5_end, rng,
                                            initial_lr=0.03, initial_beta=0.01,
                                            beta_decay=0.9999)
        sc = 2.0 * n_best * M_best
        try_update(f_best, sc)
        print(f"  Final refinement: score={sc:.10f}")

    # Also try a fresh optimization from the best at the same n with different params
    if best_seq is not None and (deadline - time.monotonic()) > 40:
        n_best = len(best_seq)
        perturbed = best_seq + 0.02 * rng.normal(size=n_best)
        perturbed = project_simplex(perturbed)
        f_best, M_best = prox_grad_optimize(perturbed, n_best, deadline, rng,
                                            initial_lr=0.05, initial_beta=0.02)
        sc = 2.0 * n_best * M_best
        try_update(f_best, sc)
        print(f"  Perturbed restart: score={sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = compute_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = np.ones(256) / 256
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.10f}")

    # Verify with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
