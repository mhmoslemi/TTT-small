Exact-subgradient simplex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements: exact subgradient with tie averaging, squared-param for zeros,
Nesterov momentum, backtracking line search, larger n exploration, structural restarts.

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    if n <= 32 or m <= 32:
        return np.convolve(a, b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    # Clean up numerical noise
    c[np.abs(c) < 1e-14] = 0.0
    return c

def _convolve_self(a):
    return _fft_convolve(a, a)

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(c)."""
    n = len(a)
    c = _convolve_self(a)
    return 2.0 * n * float(c.max())

def _full_score(a):
    """Full score: 2*n*max(c)/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = _convolve_self(a)
    return 2.0 * n * float(c.max()) / (s * s)

def _subgradient_at_max(a, c, beta=0.0):
    """
    Compute subgradient of max(c) w.r.t. a.
    If beta=0: exact subgradient at argmax positions.
    If beta>0: soft-max weighted average.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    if beta <= 1e-14:
        # Find positions at (or near) the max
        c_max = float(c.max())
        # Find all positions within a tiny tolerance of max
        tol = 1e-12 * c_max
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Average subgradients over active set
        g = np.zeros(n)
        for k in active:
            # g[j] += 2 * a[k-j] if 0 <= k-j < n
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                # a[k - j] for j in [j_min, j_max]
                # This is a[k-j_max] : a[k-j_min] reversed
                idx = np.arange(k - j_max, k - j_min + 1)
                g[j_min:j_max+1] += 2.0 * a[idx]
        g /= len(active)
        return g
    else:
        # Soft-max
        c_max = float(c.max())
        x = (c - c_max) / beta
        ex = np.exp(x)
        w = ex / ex.sum()
        g = np.zeros(n)
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-15:

            j_min = max(0, k_idx - n + 1)
            j_max = min(k_idx, n - 1)
            if j_min <= j_max:
                idx = np.arange(k_idx - j_max, k_idx - j_min + 1)
                g[j_min:j_max+1] += w[k_idx] * 2.0 * a[idx]
        return g

def _subgradient_fast(a, c, top_k=5):
    """
    Fast subgradient: average over top-k peaks of c.
    """
    n = len(a)
    c_max = float(c.max())
    # Get top-k indices
    if len(c) <= top_k:
        active = np.arange(len(c))
    else:
        # Use partial sort for efficiency
        idx = np.argpartition(c, -top_k)[::-1][:top_k]
        # Only include those close to max
        active = idx[c[idx] >= c_max * 0.999]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
    
    g = np.zeros(n)
    for k in active:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            idx = np.arange(k - j_max, k - j_min + 1)
            g[j_min:j_max+1] += 2.0 * a[idx]
    g /= len(active)
    return g

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # We want max(0, v - tau) to sum to target_sum
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64)
    n_old = len(p)
    if n_old == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s > 0:
        p_new /= s
    return p_new

    best_a = a.copy()
    c = _convolve_self(a)
    best_score = 2.0 * n * float(c.max())

    momentum_val = momentum
    
    lr = lr_init
    no_improve = 0
    iter_count = 0
