import numpy as np

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    k = 1
    while k < 2 * n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _fft_score(a):

            t_step = max(0.03, t_step * 0.8)
            if no_imp > 200:
                break

        b = _lp_direction(f, M)
        if b is None:
            break
        lp_count += 1

        g = b / b.sum()
        f_new = (1.0 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        c_new = _full_convolve_fft(f_new)
        maxc_new = float(c_new.max())
        score_new = 2.0 * n * maxc_new

            f = f_new
            if lp_count % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FW: iter={lp_count}, score={best_score:.10f}, "
                      f"cur={score_new:.10f}, t={t_step:.4f}, elapsed={elapsed:.1f}s")
        else:
            f = f_new  # still accept for exploration
            t_step = max(0.03, t_step * 0.95)

    return best_f, best_score

    # We want this for m=0,...,n-1, which is our "second term"

    # For the "first term": sum_k w[k]*f[k-m]
    # = sum_k w[k]*f_pad[k-m]
    # Let's compute conv of w with reversed f_pad:
    # Actually: sum_k w[k]*f[k-m] = sum_j w[j+m]*f[j] = (w reversed convolved with f)[m]
    # Or: = ifft(F * conj(W))[m] ... let me check
    # ifft(F_f * conj(F_w))[m] = sum_k f[k]*conj(w[k-m]) = sum_k f[k]*w[k-m] (real)
    # = sum_k f[k]*w[k-m]: let j=k-m, = sum_j f[j+m]*w[j]
    # That's NOT what we want.

    # Let me try: ifft(conj(F_f) * F_w)[m] = sum_k f[k+m]*w[k] = sum_j f[j]*w[j-m] (let j=k+m)
    # = sum_k f[k]*w[k-m] ... same thing.

    # OK different approach:
    # sum_k w[k]*f[k-m] = sum_k w[k]*f_pad[k-m]
    # Define g[k] = w[N-1-k] (reversed w)
    # Then sum_k w[k]*f[k-m] = sum_k g[N-1-k]*f[k-m]
    # Let j = N-1-k: = sum_j g[j]*f[N-1-j-m]
    # = (g * f)[N-1-m] (circular convolution)
    # So term1[m] = (g * f_pad)[N-1-m]

    w_rev = np.zeros(N)
    w_rev[:2*n-1] = w[::-1]
    wr_hat = np.fft.rfft(w_rev)
    conv_wr_f = np.fft.irfft(wr_hat * f_hat, N)
    # conv_wr_f[k] = sum_j w_rev[j]*f[k-j] = sum_j w[2n-2-j]*f[k-j]
    # Let i = 2n-2-j: = sum_i w[i]*f[k-2n+2+i]
    # conv_wr_f[N-1-m] = sum_i w[i]*f[N-1-m-2n+2+i] = sum_i w[i]*f[i+N-2n+1-m]
    # With N >= 3n-1: N-2n+1 >= n-1, so f[i+N-2n+1-m] is mostly 0
    # This is getting too complicated. Let me just use the O(n^2) direct method.

    # DIRECT METHOD (fine for n <= 4096):
    grad = np.zeros(n)
    for m in range(n):
        # term1: sum_k w[k]*f[k-m], valid k: max(0,m) <= k <= min(2n-2, m+n-1)
        k_lo = max(0, m)
        k_hi = min(2*n-1, m+n)
        if k_lo < k_hi:
            grad[m] += np.dot(w[k_lo:k_hi], f[k_lo-m:k_hi-m])
        # term2: sum_k w[k]*f[m-k], valid k: max(0, m-n+1) <= k <= min(m, 2n-2)
        k_lo = max(0, m-n+1)
        k_hi = min(m+1, 2*n-1)
        if k_lo < k_hi:
            grad[m] += np.dot(w[k_lo:k_hi], f[m-k_lo:m-k_hi+1])
    return grad


def _adam_optimize(f0, deadline, rng, lr=0.03, beta_lse=0.02, beta_decay=0.98,
                   var_weight=0.0, name=""):
    """Adam optimization with LSE-smoothed max of convolution."""
    f = np.asarray(f0, dtype=np.float64).ravel().copy()
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    n = f.size

    b1, b2, eps = 0.85, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_iter = 0

    beta = beta_lse
    c = _full_convolve_fft(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0

    while time.monotonic() < deadline:
        t_iter += 1
        c = _full_convolve_fft(f)
        cm = float(c.max())
        score = 2.0 * n * cm

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 800:
            # Restart from best with perturbation
            f = best_f + 0.05 * rng.normal(size=n)
            f = np.maximum(f, 1e-10)
            f = f / f.sum()
            m_buf.fill(0)
            v_buf.fill(0)
            t_iter = 0
            beta = beta_lse
            no_imp = 0
            continue

        # LSE weights
        x = (c - cm) / beta
        ex = np.exp(np.clip(x, -50, 0))
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient
        grad = _grad_convolve(f, w)

        # Variance penalty (optional): penalize variance of center convolution
        if var_weight > 0:
            center = c[n//4:3*n//4] if c.size >= n//2 else c
            c_mean = center.mean()
            # Approximate: push f toward uniform
            grad += var_weight * (f - 1.0/n)

        # Adam update (in direct space with projection)
        m_buf = b1 * m_buf + (1-b1) * grad
        v_buf = b2 * v_buf + (1-b2) * grad**2
        mh = m_buf / (1 - b1**t_iter)
        vh = v_buf / (1 - b2**t_iter)
        f_new = f - lr * mh / (np.sqrt(vh) + eps)
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            f_new = best_f.copy()
        else:
            f_new = f_new / s_new

        f = f_new

        # Decay beta
        if t_iter % 50 == 0:
            beta = max(0.001, beta * beta_decay)

        if t_iter % 500 == 0:
            elapsed = time.monotonic() - (deadline - (time.monotonic()))
            print(f"    Adam {name}: iter={t_iter}, best={best_score:.10f}, "
                  f"cur={score:.10f}, beta={beta:.6f}")

    return best_f, best_score


def _coord_descent(f0, deadline, rng, n_sweeps=100):
    """Coordinate descent: optimize one element at a time."""
    f = np.asarray(f0, dtype=np.float64).ravel().copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-20:
        return f0, np.inf
    f = f / s
    n = f.size

    c = _full_convolve_fft(f)
    best_score = float(2.0 * n * c.max())
    best_f = f.copy()
    t0 = time.monotonic()

    sweep = 0
    while time.monotonic() < deadline and sweep < n_sweeps:
        sweep += 1
        improved = False
        for m in range(n):
            if time.monotonic() < deadline - 1.0:
                pass
            else:
                return best_f, best_score

            # Try a few values for f[m]
            # Optimal f[m] for minimizing max(c) with other elements fixed:
            # c changes linearly in f[m]: c[k] += f[m] * (f[k-m] + f[m-k]) - (f[m])^2 * something
            # Actually c[k] = sum_i f[i]*f[k-i], so d c[k]/d f[m] = f[k-m] + f[m-k]
            # (both terms, with zero-padding)
            # c_new[k] = c[k] + delta * (f[k-m] + f[m-k]) where delta = f_new[m] - f_old[m]
            # We want to minimize max(c_new)

            old = f[m]
            if old < 1e-15:
                continue

            # Try scaling f[m] by factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                new_val = old * factor
                delta = new_val - old
                # Compute effect on c
                # c[k] += delta * (f[k-m] + f[m-k])
                # Need to compute f[k-m] + f[m-k] for all k
                contrib = np.zeros(2*n-1)
                # f[k-m] is nonzero for k in [m, m+n-1]
                k_lo = m
                k_hi = min(m + n, 2*n-1)
                if k_lo < k_hi:
                    contrib[k_lo:k_hi] += f[:k_hi-m]
                # f[m-k] is nonzero for k in [m-n+1, m]
                k_lo2 = max(0, m-n+1)
                k_hi2 = min(m+1, 2*n-1)
                if k_lo2 < k_hi2:
                    contrib[k_lo2:k_hi2] += f[m-k_hi2+1:m-k_lo2+1][::-1]

                c_new = c + delta * contrib
                # Also need to account for the f[m]^2 term in c[m+m]=c[2m] (if valid)
                # c[2m] includes f[m]*f[m] = f[m]^2, so when we change f[m],
                # c[2m] changes by: new_f[m]^2 - old_f[m]^2 = (new-val - old)^2 + 2*(new-val-oh)*oh...
                # Actually c[2m] = sum_i f[i]*f[2m-i], and the term i=m gives f[m]*f[m]
                # So c[2m] = f[m]^2 + sum_{i≠m} f[i]*f[2m-i]
                # When we change f[m] to f[m]+delta:
                # new c[2m] = (f[m]+delta)^2 + sum_{i≠m} f[i]*f[2m-i]
                # = c[2m] + 2*delta*f[m] + delta^2
                # But our linear approx gives: c[2m] + delta*(f[2m-m]+f[m-2m+m]) = c[2m] + delta*(f[m]+f[m]) = c[2m]+2*delta*f[m]
                # So we're missing the delta^2 term. For small delta it's fine.

                if c_new.max() < c.max() - 1e-15:
                    f[m] = new_val
                    f = f / f.sum()  # renormalize
                    c = _full_convolve_fft(f)
                    sc = 2.0 * n * c.max()
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                    improved = True
                    break

        if not improved and sweep > 2:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid LP (Frank-Wolfe) + Adam gradient + coordinate descent
    for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-scale escalation from n=256 to n=2048.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    print(f"Parent: {'n=' + str(parent.size) if parent is not None else 'not available'}")

    # --- Phase 0: Generate initial candidates ---
    t_phase0_end = t_start + 30.0
    candidates = []  # (score, profile, n, label)

    shapes = {
        128: [_make_hann(128), _make_blackman(128), _make_asymmetric(128, 1.5),
              _make_asymmetric(128, 2.5), _make_bipolar_smooth(128, 0.15)],
        256: [_make_hann(256), _make_blackman(256), _make_asymmetric(256, 1.5),
              _make_asymmetric(256, 2.5), _make_bipolar_smooth(256, 0.15),
              _make_symmetric_ramp(256, 0.5)],
        512: [_make_hann(512), _make_blackman(512), _make_asymmetric(512, 1.5),
              _make_asymmetric(512, 2.5), _make_bipolar_smooth(512, 0.15),
              _make_symmetric_ramp(512, 0.5)],
        1024: [_make_hann(1024), _make_blackman(1024), _make_asymmetric(1024, 1.5),
               _make_asymmetric(1024, 2.5), _make_bipolar_smooth(1024, 0.15)],
    }

    labels = ['hann', 'blackman', 'asym1.5', 'asym2.5', 'bipolar', 'ramp0.5']

    for n, profs in shapes.items():
        for i, p in enumerate(profs):
            if p is not None and p.size == n:
                sc = _fft_score(p)
                if np.isfinite(sc):
                    candidates.append((sc, p, n, f"{labels[i] if i < len(labels) else 'init'}_{n}"))

    # Add parent at its native length and interpolated
    if parent is not None:
        pn = parent.size
        sc = _fft_score(parent)
        candidates.append((sc, parent.copy(), pn, "parent"))
        for n in [256, 512, 1024]:
            if pn != n:
                p_interp = _interp_profile(parent, n)
                if p_interp is not None:
                    sc = _fft_score(p_interp)
                    candidates.append((sc, p_interp, n, f"parent_interp_{n}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    update_best(candidates[0][1], candidates[0][0], candidates[0][3])

    # --- Phase 1: LP-based FW optimization at n=256 ---
    t_fw1_end = min(deadline, t_start + 180.0)
    print(f"\nPhase 1: FW at n=256 (budget {t_fw1_end - time.monotonic():.0f}s)")

    # Pick best n=256 candidate
    fw_start = None
    for sc, p, n, label in candidates:
        if n == 256:
            fw_start = p
            break
    if fw_start is None:
        fw_start = _make_hann(256)

    f_fw, sc_fw = _fw_optimize(fw_start, t_fw1_end, rng, n_max_lp=3000)
    update_best(f_fw, sc_fw, "FW_256")

    # Also try FW from a different start
    if time.monotonic() < t_fw1_end - 30:
        fw_start2 = _make_asymmetric(256, 2.0)
        t_sub = min(t_fw1_end - 5, time.monotonic() + (t_fw1_end - time.monotonic()) * 0.6)
        f_fw2, sc_fw2 = _fw_optimize(fw_start2, t_sub, rng, n_max_lp=1000)
        update_best(f_fw2, sc_fw2, "FW_256b")

    # --- Phase 2: Escalate to n=512 and optimize ---
    t_esc1_end = min(deadline, t_start + 400.0)
    print(f"\nPhase 2: Escalate to n=512")

    if best_seq is not None:
        p_512 = _interp_profile(best_seq, 512)
    else:
        p_512 = _make_hann(512)
    if p_512 is not None:
        sc_512 = _fft_score(p_512)
        update_best(p_512, sc_512, "interp_512")

        # FW at 512
        t_fw2_end = min(t_esc1_end - 60, time.monotonic() + 120.0)
        f_fw512, sc_fw512 = _fw_optimize(p_512, t_fw2_end, rng, n_max_lp=1500)
        update_best(f_fw512, sc_fw512, "FW_512")

        # Adam at 512
        if time.monotonic() < t_esc1_end - 20:
            t_adam512 = min(t_esc1_end - 10, time.monotonic() + 80.0)
            f_ad512, sc_ad512 = _adam_optimize(f_fw512 if f_fw512 is not None else p_512,
                                                t_adam512, rng, lr=0.03,
                                                beta_lse=0.02, name="512")
            update_best(f_ad512, sc_ad512, "Adam_512")

    # --- Phase 3: Escalate to n=1024 and optimize ---
    t_esc2_end = min(deadline, t_start + 1000.0)
    print(f"\nPhase 3: Escalate to n=1024 (budget {t_esc2_end - time.monotonic():.0f}s)")

    if best_seq is not None:
        p_1024 = _interp_profile(best_seq, 1024)
    else:
        p_1024 = _make_hann(1024)
    if p_1024 is not None:
        sc_1024 = _fft_score(p_1024)
        update_best(p_1024, sc_1024, "interp_1024")

        # Also try parent directly if available
        if parent is not None and parent.size == 1024:
            update_best(parent, _fft_score(parent), "parent_1024")

        # Adam optimization at 1024 (multiple rounds)
        current = p_1024.copy()
        for round_i in range(3):
            if time.monotonic() >= t_esc2_end - 30:
                break
            per_round = min(120.0, (t_esc2_end - 30 - time.monotonic()))
            if per_round < 10:
                break
            t_round_end = time.monotonic() + per_round
            lr = 0.03 / (1 + 0.3 * round_i)
            beta_init = 0.015 / (1 + 0.5 * round_i)
            f_new, sc_new = _adam_optimize(current, t_round_end, rng,
                                            lr=lr, beta_lse=beta_init,
                                            var_weight=0.0 if round_i == 0 else 0.005,
                                            name=f"r{round_i}")
            update_best(f_new, sc_new, f"Adam_1024_r{round_i}")
            current = f_new

        # FW at 1024 (a few iterations)
        if time.monotonic() < t_esc2_end - 60:
            t_fw1024 = min(time.monotonic() + 180.0, t_esc2_end - 30)
            f_fw1024, sc_fw1024 = _fw_optimize(current, t_fw1024, rng, n_max_lp=500)
            update_best(f_fw1024, sc_fw1024, "FW_1024")
            current = f_fw1024 if sc_fw1024 < _fft_score(current) else current

            # More Adam after FW
            if time.monotonic() < t_esc2_end - 20:
                t_more = min(t_esc2_end - 10, time.monotonic() + 80.0)
                f_more, sc_more = _adam_optimize(current, t_more, rng,
                                                  lr=0.015, beta_lse=0.008, name="postFW")
                update_best(f_more, sc_more, "Adam_1024_postFW")
                current = f_more

    # --- Phase 4: Try n=2048 ---
    t_esc3_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 4: Try n=2048 (budget {t_esc3_end - time.monotonic():.0f}s)")

    if best_seq is not None and time.monotonic() < t_esc3_end - 60:
        p_2048 = _interp_profile(best_seq, 2048)
        if p_2048 is not None:
            sc_2048 = _fft_score(p_2048)
            update_best(p_2048, sc_2048, "interp_2048")

            # Adam at 2048
            t_adam2048 = min(t_esc3_end - 20, time.monotonic() + 300.0)
            f_2048, sc_2048 = _adam_optimize(p_2048, t_adam2048, rng,
                                              lr=0.02, beta_lse=0.01, name="2048")
            update_best(f_2048, sc_2048, "Adam_2048")

    # --- Phase 5: Coordinate descent polish ---
    t_cd_end = min(deadline, t_start + 1750.0)
    print(f"\nPhase 5: Coordinate descent polish")

    if best_seq is not None and time.monotonic() < t_cd_end:
        # Try CD at the best length, and also try CD on a 1024 version
        n_cd = best_seq.size
        if n_cd > 2048:
            n_cd = 1024
        p_cd_in = _interp_profile(best_seq, n_cd) if best_seq.size != n_cd else best_seq.copy()
        if p_cd_in is not None:
            t_cd = min(t_cd_end - 5, time.monotonic() + 200.0)
            f_cd, sc_cd = _coord_descent(p_cd_in, t_cd, rng, n_sweeps=50)
            update_best(f_cd, sc_cd, f"CD_{n_cd}")

    # --- Phase 6: Final Adam polish ---
    t_final_end = deadline
    print(f"\nPhase 6: Final polish")

    if best_seq is not None and time.monotonic() < t_final_end:
        n_fin = best_seq.size
        t_fin = min(t_final_end - 5, time.monotonic() + 60.0)
        if t_fin - time.monotonic() > 20:
            f_fin, sc_fin = _adam_optimize(best_seq, time.monotonic() + (t_fin - time.monotonic()),
                                            rng, lr=0.005, beta_lse=0.003, name="final")
            update_best(f_fin, sc_fin, "final_adam")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s

        # Exact score
        n_final = final.size
        sc_exact = _exact_score(final)
        print(f"\nExact score (n={n_final}): {sc_exact:.10f}")

        # Try evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
            if np.isfinite(sc_eval) and sc_eval < best_score:
                best_score = sc_eval
        except Exception as e:
            print(f"Evaluator error: {e}")

        print(f"FINAL: score={min(best_score, sc_exact):.10f}, n={n_final}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_hann(256)
    print(f"FALLBACK: n=256, score={_exact_score(fallback):.6f}")
    return [float(x) for x in fallback]
