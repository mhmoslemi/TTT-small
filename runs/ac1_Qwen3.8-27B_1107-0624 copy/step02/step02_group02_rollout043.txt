"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient descent with adaptive smooth-max temperature,
combined with LP-based descent for large jumps, and structured restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog


# ============ Core computation functions ============

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a, b=None):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).ravel()
    m = b.size
    if n <= 64 and m <= 64:
        return np.convolve(a, b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn if mn > -1e-9 else np.maximum(c, 0.0)
    return c


def autocorr_fft(a):
    """Autocorrelation (full) using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn if mn > -1e-9 else np.maximum(c, 0.0)
    return c


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = autocorr_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def smooth_max_and_weights(c, beta):
    """Compute smooth max and softmax weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = np.count_nonzero(mask)
        w[mask] = 1.0 / max(1, cnt)
        return mval, w
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w


def gradient_smooth_max(a, c, w, beta):
    """
    Compute gradient of 2n*smooth_max(conv(a,a)) w.r.t. a.
    ∇[i] = 4n * sum_k w[k] * a[k-i] (for valid k-i in [0,n-1])
    This is a cross-correlation of w and a.
    """
    n = a.size
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: h[i] = sum_k w[k]*a[k-i]
    # In FFT: h = ifft(fft(w) * conj(fft(a)))... 
    # Actually, cross-correlation of w and a:
    # h[i] = sum_k w[k] * a[k-i]
    # This is the same as (w * a_reversed)[n-1+i]... 
    # Let me think differently.
    # h[i] = sum_{k} w[k] * a[k-i] where a[j]=0 for j outside [0,n-1]
    # = sum_{j} w[i+j] * a[j]  (substituting j = k-i)
    # This is a cross-correlation: h = cross_corr(w, a)
    # h[i] = sum_j w[i+j] * a[j]
    # In FFT terms: h = ifft( conj(fft(a)) * fft(w) )... 
    # Actually: cross_corr(w, a)[i] = sum_j w[i+j]*a[j] = (a_reversed * w)[n-1+i]
    # Or equivalently: h = ifft(fft(w) * conj(fft(a_reversed)))... 
    
    # Let me use a simpler approach:
    # h[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1
    # This is the (n-1)-th to (2n-2)-th element of conv(w_padded, a_reversed)
    # Or: h = conv(w, a)[n-1 : 2n-1] reversed... 
    
    # Simplest: use the fact that h[i] = (w * a_cross)[i] where a_cross is the correlation
    # In FFT: h = ifft(fft(w, N) * conj(fft(a, N)))[:n]
    # Let's verify: ifft(fft(w)*conj(fft(a)))[i] = sum_k w[k]*a[i-k]... 
    # Hmm, that's circular correlation, not what we want.
    
    # Let me just use direct computation via FFT:
    # h[i] = sum_j a[j] * w[i+j]
    # Create w_ext of length 2n-1 (w is already length 2n-1 from the convolution)
    # h[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1
    # This is a "sliding dot product" which can be computed as:
    # h = conv(a, w_reversed)[n-1 : 2n-1]... 
    
    # Actually, the cleanest way:
    # h[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = [conv(a, w_flipped)] where w_flipped[j] = w[n-1-j]... 
    
    # Let me just do it with the matrix approach for clarity, then optimize:
    # Actually for n up to 2048, even O(n^2) would be 4M which is fast enough per iteration.
    # But let's use FFT for efficiency.
    
    # h[i] = sum_j a[j]*w[i+j] = cross-correlation of a and w, at lag i
    # = ifft(fft(a) * conj(fft(w))) at position i (circular)
    # But we need linear, not circular. With zero-padding to N >= 2n-1, it works.
    
    # The circular cross-correlation: r[i] = sum_j a[j]*w[(i+j) mod N]
    # With N >= 2n-1 and w zero-padded: r[i] = sum_j a[j]*w[i+j] for i+j < N
    # Since i < n and j < n, i+j < 2n <= N (if N >= 2n). Good.
    
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(fw * np.conj(fa), N)[:n]
    
    return 4.0 * n * h


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return e / s


# ============ LP-based step ============

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, max_time=10.0):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    if n > 2048:
        return None  # Too large for LP
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


# ============ Main optimization: Adam with smooth max ============

def adam_optimize(a_init, deadline, rng, lr=0.05, beta1=0.9, beta2=0.999,
                  eps=1e-8, verbose=False):
    """
    Adam optimization in logit space with smooth max objective.
    Minimizes 2n * max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    Parameterization: a = softmax(z), optimize z.
    """
    n = a_init.size
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Initial beta for smooth max
    c = autocorr_fft(a)
    max_c = float(c.max())
    beta = max(1e-10, 0.1 * max_c)
    
    best_a = a.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        c = autocorr_fft(a)
        L, w = smooth_max_and_weights(c, beta)
        
        # Gradient w.r.t. a: ∇_a smooth_max = 2 * cross_corr(w, a) (factor 2 from chain rule through conv)
        # Actually: ∂(max c)/∂a[i] = sum_k w[k] * ∂c[k]/∂a[i] = sum_k w[k] * 2*a[k-i]
        # So ∇_a (2n * smooth_max) = 2n * 2 * h[i] = 4n * h[i] where h = cross_corr(w, a)
        h = gradient_smooth_max(a, c, w, beta)
        # h[i] = 4n * sum_k w[k]*a[k-i]
        # But we also have the factor from the objective: 2n * L
        # The gradient of 2n*L w.r.t. a is: 2n * sum_k w[k] * ∂c[k]/∂a[i] = 2n * 2 * (cross_corr)(w,a)[i]
        # = 4n * (cross_corr)(w,a)[i]
        # But gradient_smooth_max already includes the 4n factor, so h[i] IS the gradient of 2n*L w.r.t. a[i].
        
        # Map gradient from a-space to z-space
        # a = softmax(z), so da/dz = a * (I - a*a^T) (for each i: da[i]/dz[j] = a[i](δij - a[j]))
        # ∂f/∂z[j] = sum_i (∂f/∂a[i]) * (da[i]/∂z[j])
        # = sum_i grad_a[i] * a[i] * (δij - a[j])
        # = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
        # = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
        
        dot = float(np.dot(a, h))
        grad_z = a * (h - dot)
        
        # Center gradient (invariance to adding constant to z)
        grad_z -= grad_z.mean()
        
        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a)
            z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            continue
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z * grad_z
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if not np.all(np.isfinite(z)):
            z = np.log(best_a)
            z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        a = softmax_stable(z)
        
        c = autocorr_fft(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive beta schedule
        if t % 200 == 0:
            beta = max(1e-12, beta * 0.85)
        
        # Adaptive LR
        if no_improve > 500:
            lr = max(0.01, lr * 0.95)
        
        # Restart
        if no_improve > max(2000, 4 * n):
            # Structured restart: mix best with random perturbation
            noise = rng.normal(0, 0.3, n)
            z_new = 0.7 * np.log(best_a + 1e-15) + noise
            z_new = np.clip(z_new, -50, 50)
            z_new -= z_new.mean()
            z = z_new
            a = softmax_stable(z)
            m.fill(0); v.fill(0); t = 0
            beta = max(1e-10, 0.15 * float(autocorr_fft(a).max()))
            lr = base_lr
            no_improve = 0
            if verbose:
                print(f"  Restart at t={t}, score={best_score:.10f}")
        
        if verbose and t % 5000 == 0:
            print(f"  Adam t={t}, score={best_score:.10f}, beta={beta:.2e}, lr={lr:.4f}")
    
    return best_a, best_score


# ============ FISTA-style optimization ============

def fista_optimize(a_init, deadline, rng, verbose=False):
    """
    FISTA (accelerated proximal gradient) on the simplex.
    Uses smooth max approximation with adaptive temperature.
    """
    n = a_init.size
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    
    y = a.copy()
    t_k = 1.0
    
    c = autocorr_fft(a)
    max_c = float(c.max())
    beta = max(1e-10, 0.05 * max_c)
    
    # Estimate Lipschitz constant
    # ||∇f||_L <= 4n * ||w||_1 * max|a| * ... 
    # For now, use adaptive step size
    step = 1.0 / (4.0 * n * max_c + 1e-6)
    
    best_a = a.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        c = autocorr_fft(y)
        L, w = smooth_max_and_weights(c, beta)
        
        # Gradient at y
        h = gradient_smooth_max(y, c, w, beta)
        
        # Gradient step
        z = y - step * h
        
        # Project onto simplex (a >= 0, sum(a) = 1)
        a = project_simplex(z)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_new * (a - y)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-15)
        t_k = t_new
        it += 1
        
        c = autocorr_fft(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive beta
        if it % 500 == 0:
            beta = max(1e-12, beta * 0.9)
        
        # Adaptive step
        if no_improve > 1000:
            step *= 0.9
            if step < 1e-10:
                break
        
        # Restart
        if no_improve > max(5000, 4 * n):
            noise = 0.5 * rng.normal(size=n)
            a_new = best_a + noise
            a_new = project_simplex(a_new)
            a = a_new
            y = a.copy()
            t_k = 1.0
            step = 1.0 / (4.0 * n * float(autocorr_fft(a).max()) + 1e-6)
            beta = max(1e-10, 0.1 * float(autocorr_fft(a).max()))
            no_improve = 0
            if verbose:

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines FFT-based gradient descent (subgradient of true max + smooth max),
LP-based descent for large jumps, regression-to-the-mean for monotone profiles,
FISTA acceleration, and structured restarts with momentum reset.
"""
import numpy as np
import time
from scipy.optimize import linprog


# ============ Core computation ============

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def autocorr_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn if mn > -1e-9 else np.maximum(c, 0.0)
    return c


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = autocorr_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = max(1, int(np.count_nonzero(mask)))
        w[mask] = 1.0 / cnt
        return mval, w
    x = np.clip((c - mval) / beta, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def grad_cross_corr(w, a):
    h = np.fft.irfft(np.fft.rfft(w, next_pow2(2 * w.size - 1)) *
                     np.conj(np.fft.rfft(a, next_pow2(2 * a.size - 1))),
                     next_pow2(2 * max(w.size, a.size) - 1))
    return h[:a.size] * 2.0


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return e / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = int(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ============ LP step ============

def lp_descend(a, max_iter=80, max_time=120.0, safety=1.05, rng=None):
    n = a.size
    if n > 1200:
        return a.copy()
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s0 = float(a.sum())
    if s0 < 1e-12:
        return a.copy()
    a = a / s0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    best = a.copy()
    best_M = float(autocorr_fft(best).max())
    t0 = time.monotonic()
    t = 0.25
    for it in range(max_iter):
        if time.monotonic() - t0 > max_time:
            break
        M = best_M * safety
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(A.shape[0], M),
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 8.0})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        g = np.maximum(res.x, 0.0)
        sg = float(g.sum())
        if sg < 1e-12:
            break
        g = g / sg
        f_new = (1.0 - t) * best + t * g
        f_new = project_simplex(f_new)
        Mn = float(autocorr_fft(f_new).max())
        if Mn <= best_M + 1e-12:
            best = f_new
            if Mn < best_M - 1e-12:
                best_M = Mn
            else:
                t *= 0.7
        else:
            t *= 0.5
        if t < 0.01:
            break
    return best

def adam_train(a0, deadline, rng, lr=0.08, beta1=0.9, beta2=0.999,
               eps=1e-8, warmup=400, use_smooth=True, verbose=False):
    n = a0.size
    a = np.maximum(a0, 1e-15).astype(np.float64).copy()
    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = max(1e-10, 0.08 * float(autocorr_fft(a).max()))
    best = a.copy()
    best_sc = float(2.0 * n * autocorr_fft(a).max())
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        c = autocorr_fft(a)
        exact_max = float(c.max())
        cur = float(2.0 * n * exact_max)
        if use_smooth:
            _, w = smooth_max_and_weights(c, beta)
            ga = grad_cross_corr(w, a)
        else:
            top = np.argpartition(c, -6)[-6:]
            w = np.zeros(c.size)
            w[top] = 1.0
            ga = grad_cross_corr(w, a)

        dot = float(np.dot(a, ga))
        gz = a * (ga - dot)
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best, 1e-15)); z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            continue
        t += 1
        m = beta1 * m + (1.0 - beta1) * gz
        v = beta2 * v + (1.0 - beta2) * gz * gz
        step = lr * (t / (t + warmup))
        z = z - step * (m / (1.0 - beta1 ** t)) / (np.sqrt(v / (1.0 - beta2 ** t)) + eps)
        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best, 1e-15)); z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            continue
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        a = softmax_stable(z)
        mc = float(autocorr_fft(a).max())
        sc = float(2.0 * n * mc)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        if t % 300 == 0:
            beta = max(1e-12, beta * 0.85)
        if t % 2000 == 0:
            lr = max(0.01, lr * 0.97)
        if no_imp > max(1500, 3 * n):
            sig = 0.4 if rng is not None else 0.0
            z = np.log(np.maximum(best, 1e-15))
            if sig > 0:
                z = z + sig * rng.standard_normal(n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            m.fill(0); v.fill(0); t = 0
            beta = max(1e-10, 0.1 * float(autocorr_fft(softmax_stable(z)).max()))
            lr = base_lr
            no_imp = 0
        if verbose and t % 5000 == 0:
            print(f"    Adam t={t} score={best_sc:.10f} beta={beta:.1e}")
    return best, best_sc


def fista_train(a0, deadline, rng, use_smooth=True, verbose=False):
    n = a0.size
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    y = a.copy()
    tk = 1.0
    beta = max(1e-10, 0.05 * float(autocorr_fft(a).max()))
    step = 0.5 / max(1e-8, float(autocorr_fft(a).max()) + 1.0 / n)
    best = a.copy()
    best_sc = float(2.0 * n * autocorr_fft(a).max())
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        c = autocorr_fft(y)
        if use_smooth:
            _, w = smooth_max_and_weights(c, beta)
            gy = grad_cross_corr(w, y)
        else:
            top = np.argpartition(c, -6)[-6:]
            w = np.zeros(c.size)
            w[top] = 1.0
            gy = grad_cross_corr(w, y)
        z = y - step * gy
        a = project_simplex(z)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) * 0.5
        y = a + (tk - 1.0) / tn * (y - a)
        y = project_simplex(y)
        tk = tn
        it += 1
        mc = float(autocorr_fft(a).max())
        sc = float(2.0 * n * mc)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        if it % 400 == 0:
            beta = max(1e-12, beta * 0.9)
            step *= 0.995
        if no_imp > max(3000, 3 * n):
            a = project_simplex(best + 0.4 * rng.standard_normal(n))
            y = a.copy()
            tk = 1.0
            beta = max(1e-10, 0.1 * float(autocorr_fft(a).max()))
            step = 0.5 / max(1e-8, beta + 1.0 / n)
            no_imp = 0
            if verbose:

    return best, best_sc


def rtm_train(a0, deadline, rng, n_iter=4000, verbose=False):
    """Regression-to-the-mean for monotone-like profiles."""
    n = a0.size
    L = 2 * n - 1
    P = np.log(np.maximum(a0, 1e-12))
    A_reg = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A_reg[j:j + n, j] = 1.0
    A_reg = A_reg / np.sum(a0)
    b_reg = autocorr_fft(np.maximum(a0, 0.0))
    per_iter = max(1, int(n_iter * max(0.0, deadline - time.monotonic()) / 60.0))
    for _ in range(per_iter):
        if time.monotonic() > deadline:
            break
        b = A_reg @ P
        resp = b_reg / np.maximum(b, 1e-8)
        P -= 0.3 * (1.0 - resp)
    return P

        d6 = min(hard_deadline - 5, time.monotonic() + 40.0)
        res, sc = adam_train(p0, d6, rng, lr=0.01, use_smooth=False, verbose=False)
        update(res, sc)
        print(f"[P5] final micro polish score={sc:.10f}")

    # ===== Finalize =====
    if best_seq is None:
        best_seq = make_linear_profile(512, 2.0)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s = float(best_seq.sum())
    if s > 0.01:
        best_seq = best_seq / s
    final_score = compute_score(best_seq)
    print(f"[FINAL] n={best_seq.size}  score={final_score:.10f}")

    try:
        es = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(es):
            print(f"[FINAL] evaluate_sequence confirms: {es:.10f}")
    except Exception as e:
        print(f"[FINAL] could not call evaluate_sequence: {e}")

    return [float(x) for x in best_seq]
