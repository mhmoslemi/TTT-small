def fft_convolve(a):

    n = a.size
    if n <= 64:
        return np.maximum(np.convolve(a, a), 0.0)
    N = 1
    while N < 2*n-1:

    c = np.fft.irfft(fa * fa, N)[:2*n-1]

def compute_score(a):
    """Compute the evaluation score."""
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s*s)

def build_conv_matrix(f):
    """Build the constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = f.size
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp(f, M, n, critical_lags=None, delta=0.01):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M_k, g >= 0."""

    return None

    """Use scipy minimize with the max-conv objective."""
    f = f0.copy().astype(np.float64)

    f = f / s
    n = f.size

        s = a.sum()

        c = fft_convolve(a)
        return float(c.max())

        a = np.maximum(x, 0.0)
        s = a.sum()
        if s < 1e-30:
            return np.zeros_like(x)
        a = a / s
        c = fft_convolve(a)
        k_max = int(np.argmax(c))
        # Gradient of conv(a,a)[k_max] w.r.t. a
        # d/d(a[j]) conv(a,a)[k] = a[k-j] + a[k-j] (if k-j is valid)
        # Actually: conv(a,a)[k] = sum_i a[i]*a[k-i]
        # d/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[k-j] (from the other term)
        # Wait: d/d(a[j]) sum_i a[i]*a[k-i]
        # = a[k-j] (differentiating a[i] at i=j) + a[j] * 0 ... 
        # No: sum_i a[i]*a[k-i]. Derivative w.r.t. a[j]:
        # Term where i=j: a[k-j] (if valid)
        # Term where k-i=j, i.e., i=k-j: a[k-j] (if valid)
        # But these are the same term if j = k-j, i.e., j = k/2
        # More carefully: d/d(a[j]) sum_i a[i]*a[k-i]
        # = sum_i [delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j}]
        # = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n)
        # = 2*a[k-j] if 0<=k-j<n, else a[k-j] if j=k-j (i.e., 2j=k)
        
        grad = np.zeros(n)
        kj = k_max - np.arange(n)
        valid1 = (kj >= 0) & (kj < n)
        grad[valid1] += a[kj[valid1]]
        valid2 = (kj >= 0) & (kj < n)
        grad[valid2] += a[kj[valid2]]
        # Handle the double-counting at j = k/2
        if k_max % 2 == 0:
            j_mid = k_max // 2
            if 0 <= j_mid < n:
                grad[j_mid] -= a[j_mid]  # subtract one copy
        
        # Also need to account for the 1/s normalization and a/s
        # For simplicity, just use the raw gradient
        return grad

    t_start = time.monotonic()
    for restart in range(3):
        if time.monotonic() > deadline:
            break
        x0 = best_f + 0.05 * rng.normal(size=n)
        x0 = np.maximum(x0, 0.0)
        x0 = x0 / x0.sum()
        
        res = minimize(objective, x0, method='CG', jac=jac,
                       options={'maxiter': 200, 'disp': False,
                                'iprint': -1, 'gtol': 1e-10})
        if res.success and res.fun < best_score:
            best_f = np.maximum(res.x, 0.0)
            best_f = best_f / best_f.sum()
            best_score = float(res.fun)
    
    return best_f, best_score


def softmax_gradient_polish(f0, deadline, rng, lr=0.05, max_iter=50000):
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, 0.1 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 500 == 0:
            beta = max(1e-12, beta * 0.6)

        if no_imp > max(400, 4 * n):
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: Exploration across multiple lengths
    phase1_end = min(deadline, t_start + 400.0)
    n_candidates = [256, 384, 512, 640, 768, 1024]
    
    # Filter to reasonable sizes
    n_candidates = [n for n in n_candidates if n >= 64 and n <= 2048]
    
    # If we have a parent, prioritize its length
    if parent is not None:
        parent_n = parent.size
        if parent_n not in n_candidates:
            n_candidates.insert(0, parent_n)
    
    results = []
    per_n = max(30.0, (phase1_end - t_start) / len(n_candidates))
    
    print(f"Phase 1: Exploring n in {n_candidates}, ~{per_n:.0f}s each")
    
    for n in n_candidates:
        if time.monotonic() > phase1_end:
            break
        t_n = time.monotonic()
        t_n_end = min(phase1_end, t_n + per_n * len(n_candidates) // n_candidates.index(n) if n in n_candidates else per_n)
        
        # Generate starts
        if parent is not None:
            p_int = interpolate_profile(parent, n)
        else:
            p_int = make_linear_profile(n, 1.5)
        
        starts = [p_int]
        starts.append(make_linear_profile(n, 1.0))
        starts.append(make_linear_profile(n, 2.0))
        starts.append(make_cubic_profile(n, 1.5, -1.0))
        starts.append(make_smooth_random(n, rng))
        starts.append(make_smooth_random(n, rng))
        
        per_start = max(3.0, (min(phase1_end, t_n + per_n) - time.monotonic()) / max(1, len(starts)))
        
        for p0 in starts:
            if time.monotonic() > phase1_end:
                break
            p0 = np.maximum(p0, 0.0)
            s = p0.sum()
            if s < 1e-30:
                continue
            p0 = p0 / s
            
            t_per = min(phase1_end, time.monotonic() + per_start)
            p_best, M_best = lp_optimize(p0, t_per, rng, max_iter=5000)
            sc = 2.0 * n * M_best
            try_update(p_best, sc)
            results.append((sc, n, p_best.copy()))
        
        print(f"  n={n}: best so far={best_score:.10f}")
    
    results.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Focused LP optimization on best candidates
    phase2_end = min(deadline, t_start + 1100.0)
    top_k = min(3, len(results))
    per_cand = max(30.0, (phase2_end - time.monotonic()) / max(1, top_k))
    
    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc_i, n_i, p_i = results[i]
        print(f"  Phase 2: optimizing candidate {i} (n={n_i}, score={sc_i:.10f})")
        
        t_per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = lp_optimize(p_i, t_per, rng, max_iter=50000, verbose=True)
        sc = 2.0 * n_i * M_best
        try_update(p_best, sc)
        
        # Also try scipy
        if time.monotonic() < phase2_end - 20:
            t_scipy = min(phase2_end, time.monotonic() + 30.0)
            p_sc, sc_sc = scipy_optimize(p_best, t_scipy, rng)
            sc2 = 2.0 * n_i * sc_sc
            try_update(p_sc, sc2)
            print(f"    After scipy: best={best_score:.10f}")
        
        print(f"  Phase 2 done: best={best_score:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Gradient polish
    phase3_end = min(deadline, t_start + 1500.0)
    if best_seq is not None:
        per_polish = min(phase3_end - time.monotonic(), 200.0)
        if per_polish > 30:
            p_best, M_best = softmax_gradient_polish(best_seq, time.monotonic() + per_polish, rng, lr=0.04)
            sc = 2.0 * best_seq.size * M_best
            try_update(p_best, sc)
            print(f"After Phase 3 polish: best={best_score:.10f}")
        
        # Second polish with different lr
        if time.monotonic() < phase3_end - 30:
            per2 = min(phase3_end - time.monotonic(), 100.0)
            p_best2, M_best2 = softmax_gradient_polish(best_seq, time.monotonic() + per2, rng, lr=0.02)
            sc2 = 2.0 * best_seq.size * M_best2
            try_update(p_best2, sc2)
            print(f"After Phase 3 polish 2: best={best_score:.10f}")

    # Phase 4: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        per_final = min(deadline - time.monotonic() - 10, 200.0)
        if per_final > 30:
            print(f"Phase 4: Final LP on n={n_best}")
            p_best, M_best = lp_optimize(best_seq, time.monotonic() + per_final, rng, max_iter=50000)
            sc = 2.0 * n_best * M_best
            try_update(p_best, sc)
            print(f"After Phase 4: best={best_score:.10f}")
        
        # One more gradient polish
        if time.monotonic() < deadline - 20:
            per_g = min(deadline - time.monotonic() - 5, 60.0)
            if per_g > 10:
                p_best_g, M_best_g = softmax_gradient_polish(best_seq, time.monotonic() + per_g, rng, lr=0.02)
                sc_g = 2.0 * best_seq.size * M_best_g
                try_update(p_best_g, sc_g)
                print(f"After final polish: best={best_score:.10f}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")
        
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")
        
        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
