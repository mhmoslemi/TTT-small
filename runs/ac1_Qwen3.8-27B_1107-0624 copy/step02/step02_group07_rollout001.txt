import time
from scipy.optimize import linprog

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def subgradient_step(s, s, eta):
    """
    Compute one multiplicative subgradient step.
    The peak of conv(s,s) is at position k*.
    Gradient g[m] = s[k*-m] + s[m-k*] (with out-of-range = 0).
    Update: s_new[m] = s[m] * exp(-eta * g[m] / max_g)
    """
    n = len(s)
    conv = np.convolve(s, s)
    k_star = np.argmax(conv)
    
    # Compute gradient
    g = np.zeros(n)
    # s[k_star - m] for valid m
    for m in range(n):
        idx1 = k_star - m
        idx2 = m - k_star
        val = 0.0
        if 0 <= idx1 < n:
            val += s[idx1]
        if 0 <= idx2 < n:
            val += s[idx2]
        g[m] = val
    
    max_g = np.max(g)
    if max_g < 1e-15:
        return s.copy()
    
    s_new = s * np.exp(-eta * g / max_g)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / s_new.sum()
    return s_new

The reversed s should be constant, i.e., s should be symmetric (s[j] = s[n-1-j]).

So the KKT analysis tells us that if the only active constraint is c[n-1] <= M, the optimal s is symmetric and uniform, giving score = 2. To do better, we need more constraints to be active (i.e., multiple c[k] values at the maximum).

This means the optimal solution has a "flat" autocorrelation: multiple peaks at the same level. The subgradient method naturally drives toward this because it always attacks the current maximum.

        return float('inf')

    ms = max(0, k - n + 1)
    me = min(n - 1, k)
    if ms <= me:

    me2 = min(n - 1, k + n - 1)
    if ms2 <= me2:
        g[ms2:me2+1] += s[:me2-ms2+1]
    return g

def subgrad_iter(s, eta, K=1, warm_start_conv=None):
    """One subgradient step. Returns (s_new, conv, k_star, score)."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = np.max(conv)
    score = 2.0 * n * max_conv / (np.sum(s)**2)
    
    if K == 1:
        k_star = np.argmax(conv)
        g = compute_peak_grad(s, k_star)

        g = np.zeros(n)
        for k in top_idx:
            g += compute_peak_grad(s, k)
        g /= K
    
    max_g = np.max(g)
    if max_g < 1e-15:
        return s.copy(), conv, score
    s_new = s * np.exp(-eta * g / max_g)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / s_new.sum()
