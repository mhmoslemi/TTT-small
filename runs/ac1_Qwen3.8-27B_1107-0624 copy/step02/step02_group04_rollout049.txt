```python
"""
Optimized search for minimal 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive top-k soft-max gradient in log-space, DFT-based global corrections,
and structured restarts. Builds on multi-scale escalation from n=256 to n=1024.

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = e / s
    return p

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _make_profile(n, profile_type, param1=1.0, param2=0.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if profile_type == 'const':
        a = np.ones(n)
    elif profile_type == 'linear':
        a = 1.0 + param1 * u
    elif profile_type == 'quadratic':
        a = 1.0 + param1 * u + param2 * u * u
    elif profile_type == 'cubic':
        a = 1.0 + param1 * u + param2 * u ** 3
    elif profile_type == 'cosine_bump':
        a = 1.0 + param1 * np.cos(np.pi * u)
    elif profile_type == 'gaussian_bump':
        a = 1.0 + param1 * np.exp(-param2 * u * u)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = float(a.sum())
    return a / s

def _dft_notch_correction(p, top_k=3, eta=0.1):
    """Apply a DFT-based notch filter to reduce spectral peaks."""
    n = p.size
    N = _next_pow2(2 * n - 1)
    F = np.fft.rfft(p, N)
    # Power spectrum
    P = np.abs(F) ** 2
    # Find top-k frequency bins (excluding DC)
    P_dc = P[0]
    P[0] = -1  # exclude DC
    top_idx = np.argsort(P)[::-1][:top_k]
    # Reduce those frequency bins
    for idx in top_idx:
        F[idx] *= (1.0 - eta)
    # Transform back
    p_new = np.fft.irfft(F, N)[:n]
    p_new = np.maximum(p_new, 1e-12)
    p_new = p_new / p_new.sum()
    return p_new

def _logit_optimize(p0, deadline, rng, lr=0.08, top_k_start=1, top_k_max=10,
                    beta_init_frac=0.12, beta_decay=0.72, no_imp_limit=500,
                    use_dft=False, dft_interval=2000, dft_eta=0.05):
    """
    Logit-space Adam optimization with adaptive top-k soft-max.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)
    
    # Top-k scheduling
    top_k = top_k_start
    top_k_schedule = 500  # increase top_k every this many iterations
    iter_count = 0

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Update top-k
        iter_count += 1
        if iter_count % top_k_schedule == 0 and top_k < top_k_max:
            top_k = min(top_k_max, top_k + 1)
            # Reset beta when changing top_k
            beta = max(1e-12, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0

        # Compute soft-max weights (top-k version)
        c_clipped = c.copy()
        # Zero out all but top-k positions
        if top_k < c.size:
            threshold = np.partition(c, c.size - top_k)[c.size - top_k]
            c_clipped[c_clipped < threshold - 1e-15] = -np.inf
        
        mval = float(c.max())
        x = (c_clipped - mval) / beta
        # Handle -inf
        x = np.where(np.isfinite(x), x, -100.0)
        ex = np.exp(x)
        s_w = float(np.sum(ex))
        if not np.isfinite(s_w) or s_w <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_w

        # Gradient: correlation of w with p (reversed)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp_rev, N)
        gp = h[n - 1:2 * n - 1]
        
        # Regularization
        vp = 2.0 * gp + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Adam update
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # DFT correction
        if use_dft and iter_count % dft_interval == 0 and best_score < 1.52:
            p_dft = _dft_notch_correction(p, top_k=2, eta=dft_eta)
            sc_dft = float(2.0 * n * float(_full_convolve_fft(p_dft).max()))
            if sc_dft < best_score:
                p = p_dft
                z = np.log(np.maximum(p, 1e-12))
                z -= z.mean()
                c = _full_convolve_fft(p)
                maxc = float(c.max())
                best_p = p.copy()
                best_z = z.copy()
                best_score = sc_dft
                no_imp = 0
                m.fill(0.0)
                v.fill(0.0)
                adam_t = 0

        # Adaptive beta decay
        if adam_t % 40 == 0:
            beta = max(1e-12, beta * beta_decay)

        # Restart
        if no_imp > no_imp_limit:
            if best_score < 1.51:
                noise_scale = 0.08
            elif best_score < 1.53:
                noise_scale = 0.15
            elif best_score < 1.56:
                noise_scale = 0.25
            else:
                noise_scale = 0.4
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
            top_k = top_k_start

    return best_p, best_score

def _interp_profile(p0, n_new):

    x1 = np.linspace(0.0, 1.0, n_new)
```

```python
def _lp_direction(p, top_k=5):
    """Solve LP to find best direction to reduce max of convolution."""
    from scipy.optimize import linprog
    n = p.size
    c = _full_convolve_fft(p)
    
    # Get top-k positions
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Gradients: dc[k]/da[i] = a[k-i] (for valid k-i)
    # For position k, gradient is [a[k-0], a[k-1], ..., a[k-(n-1)]] (padded)
    # i.e., gradient[i] = a[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((len(top_idx), n))
    b = np.zeros(len(top_idx))
    for j, k in enumerate(top_idx):
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                A[j, i] = p[ki]
        b[j] = c[k]
    
    # Objective: minimize t (but we add t as a variable)
    # Actually, let's minimize sum of (a - p) to get a direction
    # Or: minimize c_new_max = max_j (b_j + A_j @ (a - p))
    # This is: minimize t s.t. A_j @ a - A_j @ p + b_j <= t for all j
    # Variables: a (n vars) + t (1 var) = n+1 vars
    
    # c_lp @ x = t (index n)
    c_lp = np.zeros(n + 1)
    c_lp[n] = 1.0
    
    # Inequality: A_j @ a - t <= A_j @ p - b_j
    # => A_j @ a + (-1) * t <= A_j @ p - b_j
    A_ub = np.hstack([A, -np.ones((len(top_idx), 1))])
    b_ub = A @ p - c[top_idx]
    
    # Equality: sum(a) = 1
    A_eq = np.hstack([np.ones(n), np.zeros(1)])
    b_eq = np.array([1.0])
    
    # Bounds: a[i] >= 0, t unbounded
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            a_new = res.x[:n]
            return a_new
    except Exception:
        pass
    return None
```
