Combines LP-based direction finding, refined coordinate descent,

import numpy as np
from scipy.optimize import linprog
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0.0)
    return c

    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)

    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

    A = np.zeros((2*n-1, n))

        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = f[k - i]
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:

def lp_iterate(f, deadline, max_iter=200, t_init=0.1):
    """Run LP-based iteration."""
    f = normalize(f)
    n = f.size
    M = fft_convolve(f).max()
    t = t_init
    best_f = f.copy()
    best_score = score(f)

        if time.monotonic() > deadline:
            break
        g = lp_direction(f, M, n)
        if g is None or g.sum() < 1e-15:

                best_t = trial_t
        f = (1-best_t)*f + best_t*g
        f = np.maximum(f, 0)
        new_M = fft_convolve(f).max()
        if new_M < M * (1 - 1e-8):

        else:
            t = max(0.01, t * 0.7)
        if best_new_score < best_score:
            best_score = best_new_score
            best_f = f.copy()
    return best_f, best_score

"""
Hybrid LP + Adam + coordinate descent optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key improvements over previous approach:
1. LP-based direction finding (from state of the art) for better descent directions

4. Multi-phase search with diverse initialization and restarts

import numpy as np
from scipy.optimize import linprog
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def score_fn(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def make_asymmetric_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_asymmetric_exp(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    return a / a.sum()


def make_asymmetric_power(n, p, shift=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.abs(t - shift) ** p + 0.01
    return a / a.sum()


def make_two_hump(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a += 0.01 * np.exp(-((t - (1 - center)) ** 2) / (2 * (width * 0.7) ** 2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_custom_asymmetric(n, params):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0
    for i, (amp, freq, phase) in enumerate(params):
        a += amp * np.sin(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_diverse_profiles(n, rng, count=30):
    profs = []
    # Asymmetric linear
    for s in (1.0, 1.5, 2.0, 2.5, 3.0, -1.0, -1.5, -2.0):

    # Asymmetric exponential

    # Two hump

    # Custom asymmetric
    for _ in range(count):
        nparams = rng.integers(1, 5)
        params = []
        for _ in range(nparams):
            params.append((rng.uniform(0.1, 0.8), rng.uniform(1.0, 6.0), rng.uniform(0, 2 * np.pi)))
        profs.append(make_custom_asymmetric(n, params))
    return profs


def build_lp_matrix(f, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = f[k - i]
    return A


def lp_direction(f, M, n):
    """Solve LP: maximize sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    c = -np.ones(n)
    A = build_lp_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            g = res.x
            if g is not None and np.all(np.isfinite(g)) and g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_optimize(f, deadline, max_iters=300, verbose=False):
    """LP-based optimization with line search."""
    f = normalize_nonneg(f)
    if f is None:
        return f, np.inf
    n = f.size
    c = fft_convolve(f)
    M = float(c.max())
    best_f = f.copy()
    best_score = score_fn(f)
    t = 0.15
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        g = lp_direction(f, M, n)
        if g is None:
            break
        g = g / g.sum()  # normalize g to sum=1
        f_sum = f.sum()
        g = g * f_sum  # rescale to match sum of f

        # Line search
        best_t = t
        best_sc = best_score
        for trial_t in (t, t * 0.5, t * 0.25, t * 1.5, t * 2.0):
            if trial_t > 0.9:
                continue
            f_try = (1 - trial_t) * f + trial_t * g
            f_try = np.maximum(f_try, 0.0)
            sc = score_fn(f_try)
            if sc < best_sc:
                best_sc = sc
                best_t = trial_t

        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        ss = f.sum()
        if ss < 1e-30:
            break
        f = f / ss

        c = fft_convolve(f)
        new_M = float(c.max())
        if new_M < M * 0.999:
            M = new_M
            t = min(0.5, t * 1.05)
        else:
            t = max(0.02, t * 0.8)

        sc = best_sc
        if sc < best_score - 1e-10:
            best_score = sc
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart with noise
                f = best_f + 0.05 * np.random.default_rng(it).normal(size=n)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                c = fft_convolve(f)
                M = float(c.max())
                t = 0.15
                no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}, M={M:.6f}, t={t:.4f}")

    return best_f, best_score


def soft_max_adam(p0, deadline, rng, lr=0.08, beta_frac=0.15):
    """Soft-max Adam optimization in logit space."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    p = p0.copy()
    c = fft_convolve(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Soft max and weights
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s = float(ex.sum())
        if s <= 0 or not np.isfinite(s):
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s

        # Gradient: corr_weights(w, p)
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        gp = 2.0 * h[n - 1:2 * n - 1]

        vp = gp + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        p = e / e.sum()
        c = fft_convolve(p)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)

        if np.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 60 == 0:
            beta = max(1e-14, beta * 0.75)

        if no_imp > max(400, 3 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = fft_convolve(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def softmax_from_z(z):
    z = np.clip(z, -40, 40)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def coord_descent(p, deadline, rng, max_sweeps=100):
    """Coordinate descent with multiple candidate values per coordinate."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = score_fn(p)
    no_imp_total = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved_sweep = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            if orig < 1e-10:
                # Try small positive values
                candidates = [1e-6, 1e-5, 1e-4, orig + 1e-5]
            else:
                candidates = [
                    orig * 1.2, orig * 1.1, orig * 1.05,
                    orig * 0.95, orig * 0.9, orig * 0.8,
                    orig * 1.5, orig * 0.5
                ]
            best_val = orig
            best_sc = best_score
            for val in candidates:
                if val < 0 or val > 1000:
                    continue
                p[i] = val
                ss = p.sum()
                if ss < 0.01:
                    p[i] = orig
                    continue
                sc = score_fn(p)
                if sc < best_sc:
                    best_sc = sc
                    best_val = val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_p = p.copy()
                improved_sweep = True
                no_imp_total = 0
            else:
                no_imp_total += 1

        if not improved_sweep and no_imp_total > n:
            break
        if no_imp_total > 5 * n:
            break

    return best_p, best_score


def interpolate_to(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return normalize_nonneg(p)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return normalize_nonneg(np.interp(x1, x0, p))


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} {label}")
        if len(candidates) < 25:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        elif float(sc) < candidates[-1][0]:
            candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
            candidates.sort(key=lambda x: x[0])

    # Phase 0: Initialization
    print("Phase 0: Initialization")
    # Try parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        pass

    if parent is not None:
        sc = score_fn(parent)
        update(parent, sc, "parent")
        print(f"  Parent score: {sc:.10f}, n={parent.size}")

    # Diverse profiles at multiple lengths
    for n_init in (128, 256, 512):
        profs = make_diverse_profiles(n_init, rng, count=20)
        for i, p in enumerate(profs[:15]):
            sc = score_fn(p)
            update(p, sc, f"init_n{n_init}_{i}")
        print(f"  n={n_init}: best init so far = {best_score:.10f}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: LP-based optimization (main improvement)
    print("Phase 1: LP-based optimization")
    phase1_end = min(deadline, t_start + 700.0)
    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    
    n_lp = 0
    for ci in range(min(10, len(candidates))):
        if time.monotonic() > phase1_end:
            break
        sc_c, p_c = candidates[ci]
        if p_c.size > 512:
            p_c = interpolate_to(p_c, 512)
        per = min(phase1_end, time.monotonic() + 60.0)
        p_best, sc = lp_optimize(p_c, per, max_iters=200, verbose=True)
        if p_best is not None and np.isfinite(sc):
            sc_full = score_fn(p_best)
            update(p_best, sc_full, f"lp_{ci}_n{p_best.size}")
            n_lp += 1
        # Also try at different length
        for tn in (256, 512):
            if time.monotonic() > phase1_end:
                break
            p_tn = interpolate_to(p_c, tn)
            if p_tn is None:
                continue
            per2 = min(phase1_end, time.monotonic() + 30.0)
            p_tn_best, sc_tn = lp_optimize(p_tn, per2, max_iters=100)
            if p_tn_best is not None and np.isfinite(sc_tn):
                sc_full2 = score_fn(p_tn_best)
                update(p_tn_best, sc_full2, f"lp_{ci}_n{tn}")
    
    print(f"  After LP: best={best_score:.10f} (ran {n_lp} LP optimizations)")

    # Phase 2: Adam-based refinement
    print("Phase 2: Adam refinement")
    phase2_end = min(deadline, t_start + 1200.0)
    candidates.sort(key=lambda x: x[0])
    for ci in range(min(8, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[ci]
        per = min(phase2_end, time.monotonic() + 30.0)
        p_best, sc = soft_max_adam(p_c, per, rng, lr=0.08)
        if p_best is not None:
            sc_full = score_fn(p_best)
            update(p_best, sc_full, f"adam_{ci}")
        # Try extended version
        if p_c.size <= 256:
            p_ext = interpolate_to(p_c, 512)
            if p_ext is not None:
                per2 = min(phase2_end, time.monotonic() + 25.0)
                p_ext_best, sc_ext = soft_max_adam(p_ext, per2, rng, lr=0.06)
                if p_ext_best is not None:
                    sc_full_ext = score_fn(p_ext_best)
                    update(p_ext_best, sc_full_ext, f"adam_ext_{ci}")

    print(f"  After Adam: best={best_score:.10f}")

    # Phase 3: Coordinate descent polish
    print("Phase 3: Coordinate descent")
    phase3_end = min(deadline, t_start + 1700.0)
    candidates.sort(key=lambda x: x[0])
    for ci in range(min(6, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[ci]
        per = min(phase3_end, time.monotonic() + 40.0)
        p_best, sc = coord_descent(p_c, per, rng, max_sweeps=50)
        if p_best is not None:
            sc_full = score_fn(p_best)
            update(p_best, sc_full, f"cd_{ci}_n{p_best.size}")

    print(f"  After CD: best={best_score:.10f}")

    # Phase 4: Final LP polish on absolute best
    print("Phase 4: Final polish")
    phase4_end = min(deadline, t_start + budget_s - 50.0)
    if best_seq is not None:
        per = phase4_end - time.monotonic()
        if per > 30:
            # LP polish
            p_lp, sc_lp = lp_optimize(best_seq, time.monotonic() + per * 0.5, max_iters=100)
            if p_lp is not None and np.isfinite(sc_lp):
                sc_lp_full = score_fn(p_lp)
                update(p_lp, sc_lp_full, "final_lp")
            # CD polish
            per2 = phase4_end - time.monotonic()
            if per2 > 20:
                p_cd, sc_cd = coord_descent(best_seq, time.monotonic() + per2 * 0.5, rng, max_sweeps=30)
                if p_cd is not None:
                    sc_cd_full = score_fn(p_cd)
                    update(p_cd, sc_cd_full, "final_cd")

    # Final selection
    print("Phase 5: Final selection")
    if best_seq is not None:
        final_seq = best_seq.copy()
        # Exact scoring with numpy convolve
        if final_seq.size <= 2048:
            n = final_seq.size
            s = float(final_seq.sum())
            if s >= 0.01:
                c_exact = np.convolve(final_seq, final_seq)
                final_score = float(2 * n * c_exact.max() / (s * s))
                print(f"  Exact score: {final_score:.10f} n={n}")
                # Verify with evaluator
                try:
                    sc_eval = float(evaluate_sequence(final_seq.tolist()))
                    if np.isfinite(sc_eval) and sc_eval < final_score:
                        final_score = sc_eval
                        print(f"  Evaluator score: {sc_eval:.10f}")
                except Exception as e:
                    print(f"  Evaluator error: {e}")
        
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        ss = float(final_seq.sum())
        if ss > 0:
            final_seq = final_seq / ss
        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    print("  FALLBACK: using best candidate or linear profile")
    if candidates:
        sc_c, p_c = candidates[0]
        p_c = np.clip(p_c, 0.0, 1000.0)
        ss = p_c.sum()
        if ss > 0:
            return [float(x) for x in (p_c / ss)]
    fl = make_asymmetric_linear(256, 2.0)
    return [float(x) for x in fl]
