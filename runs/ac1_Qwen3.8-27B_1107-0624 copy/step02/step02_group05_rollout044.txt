def make_smooth_ramp(n, alpha=1.0):
    """Smooth asymmetric ramp using sigmoid blending."""
    t = np.linspace(0.0, 1.0, n)
    # Asymmetric: high at one end
    a = 0.3 + 0.7 * t ** alpha
    return a / a.sum()


def make_poly_profile(n, coeffs):
    """Polynomial profile: a(t) = sum coeffs[k] * t^k, normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c in coeffs:
        a += c
    # Actually let's do: a(t) = sum_k coeffs[k] * t^k
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * (t ** k)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_dual_asym(n, mu1=0.2, mu2=0.8, s1=0.15, s2=0.1, w1=1.0, w2=0.6):
    """Two asymmetric Gaussian bumps."""
    t = np.linspace(0.0, 1.0, n)
    a = w1 * np.exp(-0.5 * ((t - mu1) / s1) ** 2) + w2 * np.exp(-0.5 * ((t - mu2) / s2) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_fourier_profile(n, amps, phases, n_terms=6):
    """Sum of cosines with given amplitudes and phases."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(min(n_terms, len(amps))):
        a += amps[k] * np.cos(2 * np.pi * (k + 1) * t + phases[k])
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def get_diverse_starts(n, rng, count=30):
    """Generate diverse starting profiles."""
    starts = []
    # Power-law ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        starts.append(make_ramp_profile(n, p=p))
    # Smooth ramps
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        starts.append(make_smooth_ramp(n, alpha))
    # Polynomial profiles
    starts.append(make_poly_profile(n, [1, 2, -1]))
    starts.append(make_poly_profile(n, [1, 3, -2]))
    starts.append(make_poly_profile(n, [1, 1, -0.5, 0.3]))
    starts.append(make_poly_profile(n, [0.5, 2, 1, -1]))
    # Dual asymmetric
    starts.append(make_dual_asym(n, 0.1, 0.7, 0.12, 0.08, 1.0, 0.8))
    starts.append(make_dual_asym(n, 0.2, 0.8, 0.15, 0.1, 1.0, 0.5))
    starts.append(make_dual_asym(n, 0.05, 0.5, 0.1, 0.2, 1.0, 0.7))
    # Random Fourier
    for _ in range(count - len(starts)):
        amps = rng.uniform(-0.5, 0.5, 8)
        phases = rng.uniform(0, 2 * np.pi, 8)
        starts.append(make_fourier_profile(n, amps, phases))
    # Ensure all valid
    valid = []
    for s in starts:
        ns = normalize_to_sum1(s)
        if ns is not None:
            valid.append(ns)
    return valid


def make_conv_matrix_rows(f, active_lags):
    """Build constraint matrix rows for given active lags only."""
    n = len(f)
    rows = []
    for k in active_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            row[i] = f[k - i]
        rows.append(row)
    return np.array(rows) if rows else None


def solve_lp_full(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    # Build full constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_active(f, M, active_lags, time_limit=2.0):
    """Solve LP using only active constraints (faster)."""
    n = len(f)
    c = -np.ones(n)
    A = make_conv_matrix_rows(f, active_lags)
    if A is None or len(A) == 0:
        return None
    b = np.full(len(A), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize_v2(f_init, deadline, rng, max_iters=5000, verbose=False):
    """
    Improved LP-based iterative optimization with active set management.
    Uses full LP initially, then switches to active-set LP for speed.
    """
    f = normalize_to_sum1(f_init.copy())
    if f is None:
        return f_init, compute_score_normalized(f_init)

    n = len(f)
    M = compute_M(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    use_active = n <= 512  # Use active set for reasonable sizes

    # Track active lags
    c = np.convolve(f, f)
    active_lags = list(np.argsort(c)[-min(10, len(c)):])  # Top 10

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP
        if use_active and it > 20:
            g = solve_lp_active(f, M, active_lags)
        else:
            g = solve_lp_full(f, M)

        if g is None:
            # Try full LP as fallback
            g = solve_lp_full(f, M)

        if g is None:
            # Fallback: random perturbation
            f = normalize_to_sum1(f + 0.1 * rng.normal(size=n))
            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            no_imp += 1
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check if g is essentially the same as f (convergence)
        if float(np.linalg.norm(g - f, ord=1)) < 1e-8:
            no_imp += 1
        else:
            # Line search
            best_t = 0.0
            best_f_try = f
            best_M_try = M
            for t in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                f_try = normalize_to_sum1((1.0 - t) * f + t * g)
                if f_try is None:
                    continue
                M_try = compute_M(f_try)
                if M_try < best_M_try - 1e-14:
                    best_M_try = M_try
                    best_f_try = f_try
                    best_t = t

            if best_t > 0:
                f = best_f_try
                M = best_M_try
                if M < best_M - 1e-14:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1

        # Update active set periodically
        if it % 10 == 0:
            c = np.convolve(f, f)
            # Use top K lags where K grows as we iterate
            K = min(10 + it // 50, 2 * n - 1)
            active_lags = list(np.argsort(c)[-K:])

        # Stall detection with smart restart
        if no_imp > 20:
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Mirror
                f = normalize_to_sum1(best_f[::-1].copy())
            elif strategy == 1:
                # Shift
                shift = rng.integers(1, max(2, n // 8))
                f_shifted = np.roll(best_f, shift)
                f = normalize_to_sum1(f_shifted)
            elif strategy == 2:
                #Frequency perturbation
                freq = rng.integers(1, max(2, n // 4))
                phase = rng.uniform(0, 2 * np.pi)
                t = np.linspace(0, 1, n)
                pert = 0.2 * np.cos(2 * np.pi * freq * t + phase)
                f = normalize_to_sum1(best_f + pert)
            else:
                # Gaussian perturbation
                f = normalize_to_sum1(best_f + 0.3 * rng.normal(size=n))

            if f is None:
                f = best_f.copy()
            M = compute_M(f)
            no_imp = 0

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}: M={M:.12f}, best={best_M:.12f}")

    return best_f, best_M


def adam_polish_v2(f_init, deadline, rng, lr=0.05):
    """
    Improved Adam polish with adaptive soft-max temperature and
    frequent restarts.
    """
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f + 1e-30)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = np.convolve(f, f)
    M = float(c.max())
    M_mean = float(c.mean())
    beta = max(1e-10, 0.3 * (M - M_mean))

    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    patience = 0

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        # Compute soft-max weights
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x - x.max())
        s = float(ex.sum())
        if s < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Gradient: d/dz of sum_k w_k * (f*f)[k] (through softmax on f)
        # (f*f)[k] = sum_i f[i]*f[k-i]
        # d(f*f)[k]/d f[m] = f[k-m] + f[m] (with bounds)
        # In log domain: d f/m / d z[m'] = f[m] * (delta_{m,m'} - f[m'])
        # So grad_z[m'] = f[m'] * (sum_k w_k * (d(f*f)[k]/d f[m'])) - f[m'] * sum_z grad_f

        # Compute grad_f first
        grad_f = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad_f[i] += w[k] * (f[k - i] + f[i])

        # Convert to log domain
        grad_z = f * (grad_f - float(np.dot(f, grad_f)))
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        z = np.clip(z, -40, 40)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
            patience = 0
        else:
            no_imp += 1
            patience += 1

        # Anneal beta
        if t_adam % 200 == 0:
            beta *= 0.85
            beta = max(beta, 0.01 * M)

        # Decrease learning rate
        if no_imp > 100:
            lr = max(base_lr * 0.1, lr * 0.95)

        # Restart
        if no_imp > max(500, 2 * n):
            restart_type = patience % 3
            if restart_type == 0:
                z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            elif restart_type == 1:
                # Try mirror
                z = best_z[::-1].copy() + 0.3 * rng.normal(size=n)
            else:
                z = best_z.copy() + rng.normal(size=n) * 0.5
            z = np.clip(z, -40, 40)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.3 * max(M - float(c.mean()), 1e-10))
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            patience = 0

    return best_f, best_M


def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def try_update(best_info, f, score):
    """Try to update the best found sequence."""
    if f is not None and np.isfinite(score) and score < best_info['score']:
        best_info['score'] = float(score)
        best_info['seq'] = f.copy()
        best_info['n'] = len(f)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    A: Diverse LP optimization at n=256
    B: Adam polish on best LP results
    C: Multi-length exploration (128, 512, 1024)
    D: Final intensive refinement
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_info = {'score': np.inf, 'seq': None, 'n': 0}

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
    except NameError:
        pass
    except Exception:
        pass

    # === Phase A: LP optimization at n=256 ===
    n_a = 256
    phase_a_deadline = min(deadline, t_start + 720.0)
    print(f"Phase A: LP opt at n={n_a}, time budget={phase_a_deadline - t_start:.0f}s")

    starts = get_diverse_starts(n_a, rng, count=25)
    if parent is not None:
        if parent.size == n_a:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n_a))

    # Evaluate all starts
    for s in starts:
        sc = compute_score_normalized(s)
        try_update(best_info, s, sc)
    print(f"  Initial best: {best_info['score']:.10f}")

    # Run LP from top starts
    scored_starts = sorted([(compute_score_normalized(s), s) for s in starts], key=lambda x: x[0])
    n_starts = min(10, len(scored_starts))

    results_a = []
    for i in range(n_starts):
        remaining = phase_a_deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, n_starts - i), 90.0)
        run_dl = time.monotonic() + alloc
        f_best, M_best = lp_optimize_v2(scored_starts[i][1], run_dl, rng, max_iters=3000)
        sc = 2.0 * n_a * M_best
        try_update(best_info, f_best, sc)
        results_a.append((sc, f_best))
        if (i + 1) % 3 == 0:
            print(f"  LP {i+1}/{n_starts}: score={sc:.10f}, best={best_info['score']:.10f}")

    results_a.sort(key=lambda x: x[0])
    print(f"Phase A done. Best: {best_info['score']:.10f}")

    # === Phase B: Adam polish ===
    phase_b_deadline = min(deadline, t_start + 1080.0)
    print(f"Phase B: Adam polish, budget={phase_b_deadline - t_start:.0f}s")

    results_b = []
    n_polish = min(5, len(results_a))
    for i in range(n_polish):
        remaining = phase_b_deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining / max(1, n_polish - i), 60.0)
        run_dl = time.monotonic() + alloc
        f_best, M_best = adam_polish_v2(results_a[i][1], run_dl, rng, lr=0.04)
        sc = 2.0 * n_a * M_best
        try_update(best_info, f_best, sc)
        results_b.append((sc, f_best))
        print(f"  Adam {i+1}/{n_polish}: score={sc:.10f}, best={best_info['score']:.10f}")

    # Merge results
    all_results = results_a + results_b
    all_results.sort(key=lambda x: x[0])
    print(f"Phase B done. Best: {best_info['score']:.10f}")

    # === Phase C: Multi-length exploration ===
    phase_c_deadline = min(deadline, t_start + 1500.0)
    print(f"Phase C: Multi-length, budget={phase_c_deadline - t_start:.0f}s")

    for n_try in [512, 128, 1024]:
        remaining = phase_c_deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining, 200.0)
        run_dl = time.monotonic() + alloc

        # Interpolate best profile to new length
        base = all_results[0][1]
        f_ext = interpolate_profile(base, n_try)
        f_ext = normalize_to_sum1(f_ext)

        # LP optimize at new length
        f_best, M_best = lp_optimize_v2(f_ext, run_dl - 30, rng, max_iters=2000)
        sc = 2.0 * n_try * M_best
        try_update(best_info, f_best, sc)
        print(f"  n={n_try} LP: score={sc:.10f}, best={best_info['score']:.10f}")

        # Adam polish
        remaining2 = run_dl - time.monotonic()
        if remaining2 > 15:
            f_best2, M_best2 = adam_polish_v2(f_best, run_dl, rng, lr=0.03)
            sc2 = 2.0 * n_try * M_best2
            try_update(best_info, f_best2, sc2)
            print(f"  n={n_try} Adam: score={sc2:.10f}, best={best_info['score']:.10f}")

    print(f"Phase C done. Best: {best_info['score']:.10f}")

    # === Phase D: Final intensive refinement ===
    phase_d_deadline = deadline
    remaining = phase_d_deadline - time.monotonic()
    if remaining > 30 and best_info['seq'] is not None:
        print(f"Phase D: Final refinement, budget={remaining:.0f}s")
        best_n = best_info['n']
        f_cur = best_info['seq']

        # More LP iterations
        f_best, M_best = lp_optimize_v2(f_cur, time.monotonic() + remaining * 0.5, rng, max_iters=5000)
        sc = 2.0 * best_n * M_best
        try_update(best_info, f_best, sc)
        print(f"  LP refine: score={sc:.10f}")

        # Adam polish
        remaining2 = phase_d_deadline - time.monotonic()
        if remaining2 > 15:
            f_best2, M_best2 = adam_polish_v2(f_cur, phase_d_deadline, rng, lr=0.02)
            sc2 = 2.0 * best_n * M_best2
            try_update(best_info, f_best2, sc2)
            print(f"  Adam refine: score={sc2:.10f}")

    # === Phase E: Try mirror + re-optimize ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_info['seq'] is not None:
        f_mirror = best_info['seq'][::-1].copy()
        f_mirror = normalize_to_sum1(f_mirror)
        run_dl = time.monotonic() + min(remaining * 0.5, 120.0)
        f_best, M_best = lp_optimize_v2(f_mirror, run_dl, rng, max_iters=2000)
        sc = 2.0 * len(f_best) * M_best
        try_update(best_info, f_best, sc)
        if sc < best_info['score']:
            print(f"  Mirror improvement: {sc:.10f}")

    # Final validation
    if best_info['seq'] is None:
        best_info['seq'] = make_smooth_ramp(n_a, 1.5)
        best_info['score'] = compute_score_normalized(best_info['seq'])

    best_seq = np.maximum(best_info['seq'], 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_smooth_ramp(n_a, 1.5)
    else:
        best_seq = best_seq / s
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = compute_score_normalized(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")

    # Validate with evaluate_sequence
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence score: {ev_score:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return [float(x) for x in best_seq]
