    n = p0.size
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    topk = max(2, int(topk_frac * (2 * n - 1)))
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _topk_softmax_weights(c, beta, k=topk)
        gp = 2.0 * _corr_weights(w, p)
        gp = gp + reg * p

        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > no_imp_limit:
            if best_score < 1.52:
                noise_scale = 0.08
            elif best_score < 1.55:
                noise_scale = 0.15
            else:
                noise_scale = 0.3
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _coordinate_descent(p0, deadline, rng, n_sweeps_max=500):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    c = _full_convolve_fft(p)
    maxc = float(c.max())
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    sweep = 0

    while time.monotonic() < deadline and sweep < n_sweeps_max:
        sweep += 1
        improved = False
        nn = n
        for i in range(nn):
            if time.monotonic() > deadline:
                break
            # Try adjusting p[i] slightly
            for delta_frac in [0.01, -0.01, 0.03, -0.03, 0.005, -0.005]:
                p_try = p.copy()
                old_val = p_try[i]
                p_try[i] = max(1e-15, old_val * (1.0 + delta_frac))
                s = p_try.sum()
                p_try = p_try / s
                c_try = _full_convolve_fft(p_try)
                sc_try = float(2.0 * n * c_try.max())
                if sc_try < best_score - 1e-14:
                    p = p_try.copy()
                    c = c_try
                    maxc = float(c_try.max())
                    best_score = sc_try
                    best_p = p.copy()
                    improved = True
            if sweep % 5 == 0 and i % (n // 10) == 0:
                # Periodic random perturbation
                j = int(rng.integers(0, n))
                p_try = p.copy()
                p_try[j] *= (1.0 + 0.02 * rng.standard_normal())
                p_try = np.maximum(p_try, 1e-15)
                p_try = p_try / p_try.sum()
                c_try = _full_convolve_fft(p_try)
                sc_try = float(2.0 * n * c_try.max())
                if sc_try < best_score - 1e-14:
                    p = p_try.copy()
                    best_score = sc_try
                    best_p = p.copy()
                    improved = True
        if not improved and sweep > 20:
            break

    return best_p, best_score


def _simulated_anneal(p0, deadline, rng):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    c = _full_convolve_fft(p)
    maxc = float(c.max())
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)

    T = 0.03
    T_min = 0.001
    cool_rate = 0.9997
    patch_min = 3
    patch_max = min(60, n // 4)

    while time.monotonic() < deadline and T > T_min:
        ps = int(np.clip(rng.integers(patch_min, patch_max + 1), patch_min, n // 2))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        p_new = p.copy()
        p_new[start:end] += T * 0.3 * rng.standard_normal(ps)
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()

        c_new = _full_convolve_fft(p_new)
        score_new = float(2.0 * n * c_new.max())

        delta = score_new - best_score
        if delta < 0:
            p = p_new
            best_p = p_new.copy()
            best_score = score_new
            c = c_new
            T = max(T_min, T * 0.995)
        else:
            if T > T_min:
                accept_prob = np.exp(-delta / T)
                if rng.random() < accept_prob:
                    p = p_new
                    c = c_new
            T = max(T_min, T * cool_rate)

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} (n={p.size}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    print(f"Parent: {'n=' + str(parent.size) if parent is not None else 'not available'}")

    # Phase 0: Initial candidates
    candidates = []
    lengths_to_try = [256, 512, 1024]

    for n in lengths_to_try:
        profiles = [
            ('lin1.5', _make_profile(n, 'linear', {'slope': 1.5})),
            ('lin2.0', _make_profile(n, 'linear', {'slope': 2.0})),
            ('edge2', _make_profile(n, 'edge_high', {'amp': 2.0})),
            ('cos_edge', _make_profile(n, 'cos_edge', {'amp': 0.8})),
            ('quad_mid', _make_profile(n, 'quad_mid', {'a2': 0.3})),
            ('cubic', _make_profile(n, 'cubic', {'a1': 1.5, 'a3': -1.0})),
            ('gumbel', _make_profile(n, 'gumbel', {'scale': 0.8})),
        ]
        if parent is not None:
            p_int = _interp_to_length(parent, n)
            profiles.append(('parent_interp', p_int))

        for label, p in profiles:
            sc = _compute_score(p)
            if np.isfinite(sc):
                candidates.append((sc, p, n, label))
                update_best(p, sc, f"init {label} n={n}")

    # Also try parent at its own length
    if parent is not None and parent.size not in lengths_to_try:
        sc = _compute_score(parent)
        if np.isfinite(sc):
            candidates.append((sc, parent, parent.size, 'parent'))
            update_best(parent, sc, f"parent n={parent.size}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best={best_score:.10f}, {len(candidates)} candidates")

    # Phase 1: Polish top candidates
    phase1_end = min(deadline, t_start + 300.0)
    seen = set()
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase1_end or count >= 10:
            break
        key = (n, label)
        if key in seen:
            continue
        seen.add(key)
        per_end = min(phase1_end, time.monotonic() + 25.0)
        p_best, sc_best = _logit_descent(p, per_end, rng, lr=0.06, beta_frac=0.10,
                                          beta_decay=0.70, topk_frac=0.12,
                                          no_imp_limit=200, name=label)
        update_best(p_best, sc_best, f"polish {label} n={n}")
        candidates.append((sc_best, p_best, n, label + "_v2"))
        count += 1

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: best={best_score:.10f}")

    # Phase 2: Multi-scale escalation
    phase2_end = min(deadline, t_start + 900.0)
    escalation_lengths = [512, 1024, 2048]

    for target_n in escalation_lengths:
        if time.monotonic() > phase2_end - 60.0:
            break
        if best_seq is None:
            continue
        if best_seq.size == target_n:
            current = best_seq.copy()
            sc_current = best_score
        else:
            current = _interp_to_length(best_seq, target_n)
            sc_current = _compute_score(current)
            update_best(current, sc_current, f"interp n={target_n}")
            print(f"  Escalating to n={target_n}: score={sc_current:.10f}")

        # Polish at this length - first pass
        if time.monotonic() < phase2_end - 30.0:
            per_end = min(phase2_end - 30.0, time.monotonic() + 80.0)
            p_b, sc_b = _logit_descent(current, per_end, rng, lr=0.05, beta_frac=0.08,
                                        beta_decay=0.72, topk_frac=0.10,
                                        no_imp_limit=300, name=f"esc{target_n}")
            update_best(p_b, sc_b, f"esc_polish1 n={target_n}")

        # Second pass with finer settings
        if time.monotonic() < phase2_end - 20.0 and best_seq is not None:
            per_end = min(phase2_end - 20.0, time.monotonic() + 60.0)
            p_b2, sc_b2 = _logit_descent(best_seq, per_end, rng, lr=0.03, beta_frac=0.05,
                                          beta_decay=0.78, topk_frac=0.05,
                                          no_imp_limit=250, name=f"fin{target_n}")
            update_best(p_b2, sc_b2, f"esc_polish2 n={target_n}")

    print(f"Phase 2: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Phase 3: Multi-restart at best length
    phase3_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        n_best = best_seq.size
        n_starts = 6
        per_start_budget = (phase3_end - time.monotonic()) / max(1, n_starts)

        for i in range(n_starts):
            if time.monotonic() > phase3_end:
                break
            per_end = min(phase3_end, time.monotonic() + max(20.0, per_start_budget))

            # Different restart strategies
            z = np.log(np.maximum(best_seq, 1e-15))
            if i % 3 == 0:
                # Gaussian noise
                noise = 0.12 * (1.0 + 0.3 * i)
                z = z + noise * rng.standard_normal(n_best)
            elif i % 3 == 1:
                # Structural: shift and blend
                shift = int(rng.integers(1, max(2, n_best // 8)))
                z_shifted = np.roll(z, shift)
                z = 0.7 * z + 0.3 * z_shifted
                z = z + 0.05 * rng.standard_normal(n_best)
            else:
                # Blend with a different profile
                p_alt = _make_profile(n_best, rng.choice(['edge_high', 'cos_edge', 'gumbel']),
                                       {'amp': rng.uniform(0.5, 2.0), 'scale': rng.uniform(0.5, 1.5)})
                z_alt = np.log(np.maximum(p_alt, 1e-15))
                z = (1.0 - 0.3 * (1 + i * 0.1)) * z + 0.3 * (1 + i * 0.1) * z_alt

            z = np.clip(z, -60, 60)
            p_start = _softmax_from_z(z)
            p_try, sc_try = _logit_descent(p_start, per_end, rng, lr=0.04, beta_frac=0.06,
                                            beta_decay=0.74, topk_frac=0.08,
                                            no_imp_limit=200, name=f"restart{i}")
            update_best(p_try, sc_try, f"restart{i} n={n_best}")

    print(f"Phase 3: best={best_score:.10f}")

    # Phase 4: Coordinate descent + SA
    phase4_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        # Coordinate descent (limited time)
        cd_end = min(phase4_end - 60.0, time.monotonic() + 120.0)
        p_cd, sc_cd = _coordinate_descent(best_seq, cd_end, rng, n_sweeps_max=100)
        update_best(p_cd, sc_cd, "coord_descent")

        # Simulated annealing
        if time.monotonic() < phase4_end:
            sa_end = phase4_end
            p_sa, sc_sa = _simulated_anneal(best_seq, sa_end, rng)
            update_best(p_sa, sc_sa, "SA")

    print(f"Phase 4: best={best_score:.10f}")

    # Phase 5: Final polish
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            p_fin, sc_fin = _logit_descent(best_seq, deadline - 2.0, rng, lr=0.01,
                                            beta_frac=0.03, beta_decay=0.82,
                                            topk_frac=0.03, no_imp_limit=150,
                                            name="final")
            update_best(p_fin, sc_fin, "final_polish")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(256, 'linear', {'slope': 1.5})

        # Try evaluate_sequence for authoritative score
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as e:
            print(f"Evaluator error: {e}")

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(256, 'linear', {'slope': 1.5})
    print(f"FALLBACK: n=256, score={_compute_score(fallback):.10f}")
    return [float(x) for x in fallback]
