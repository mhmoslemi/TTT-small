"""
Multi-scale gradient + LP hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses correct FFT-based gradients with Adam optimizer, LP-based direction finding for
local minimum escape, and multi-scale interpolation from n=256 to n=1024.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def compute_conv_and_max(a):
    c = fft_convolve(a)
    return c, float(c.max())

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_quadratic_profile(n, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0):
        profs.append(make_linear_profile(n, eta))
    for a2 in (0.5, 1.0, 1.5, 2.0, 2.5):
        profs.append(make_quadratic_profile(n, a2))
    for a1, a3 in ((1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (2.5, -2.0),
                   (1.0, 0.5), (1.5, 1.0), (2.0, -2.5), (2.5, -3.0)):
        profs.append(make_cubic_profile(n, a1, a3))
    return profs

def gradient_descent(f0, deadline, rng, lr=None, beta_init=None, max_iter=100000):
    """
    Adam-based gradient descent in log-space on the log-sum-exp smooth approximation.
    f0 must be a non-negative array with sum > 0.
    Returns (best_f_normalized, best_M).
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    if lr is None:
        lr = 0.05
    if beta_init is None:
        _, M0 = compute_conv_and_max(f)
        beta_init = 0.2 * M0
    
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t_adam = 0
    
    c, M_val = compute_conv_and_max(f)
    beta = beta_init
    
    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    
    # For FFT cross-correlation and convolution, we need padding
    # c has length 2n-1, a has length n
    # For T1 = cross-corr(w, a): need N >= (2n-1) + n - 1 = 3n-2
    # For T2 = conv(a, w): need N >= n + (2n-1) - 1 = 3n-2
    N_fft = next_pow2(3 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute softmax weights
        x = (c - M_val) / beta
        # For numerical stability
        ex = np.exp(np.clip(x, -50.0, 0.0))
        # Actually, since x <= 0 (c - M <= 0), exp(x) <= 1, so no overflow
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex
        
        # Compute T1 = cross-correlation of w and f: T1[i] = sum_k w[k] * f[k-i]
        fw = np.fft.rfft(w, N_fft)
        ff = np.fft.rfft(f, N_fft)
        T1 = np.fft.irfft(fw * np.conj(ff), N_fft)[:n]
        
        # Compute T2 = convolution of f and w: T2[i] = sum_k f[k] * w[i-k]
        T2 = np.fft.irfft(ff * fw, N_fft)[:n]
        
        # Total gradient of L w.r.t. a: h = T1 + T2
        h = T1 + T2
        
        # Gradient w.r.t. z: grad_z = f * (h - mean_h)
        mean_h = float(np.dot(f, h))
        grad_z = f * (h - mean_h)
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_.fill(0)
            v_.fill(0)
            t_adam = 0
            continue
        
        # Adam update
        t_adam += 1
        m_ = b1 * m_ + (1.0 - b1) * grad_z
        v_ = b2 * v_ + (1.0 - b2) * (grad_z * grad_z)
        mh = m_ / (1.0 - b1 ** t_adam)
        vh = v_ / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        
        # Convert back to f
        f = np.exp(z - z.max())
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Recompute convolution and max
        c, M_new = compute_conv_and_max(f)
        M_val = M_new
        
        # Update best
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Periodically reduce beta (sharpen the approximation)
        if t_adam % 500 == 0:
            beta = max(1e-12, beta * 0.8)
        
        # Restart if stuck
        if no_imp > max(500, 5 * n):
            # Random restart in z-space
            z = 0.9 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            c, M_val = compute_conv_and_max(f)
            beta = max(1e-12, 0.2 * M_val)
            m_.fill(0)
            v_.fill(0)
            t_adam = 0
            no_imp = 0
        
        # If very stuck, try a more aggressive restart
        if no_imp > max(2000, 20 * n):
            z = best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            c, M_val = compute_conv_and_max(f)
            beta = max(1e-12, 0.2 * M_val)
            m_.fill(0)
            v_.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_f, best_M

def solve_lp_direction(f, M, n, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit})
    if res.success and res.x is not None:
        g = res.x
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s > 1e-30:
            g = g / s
        return g
    return None

def lp_step(f, M, n, rng, deadline, max_iter=5000):
    """LP-based direction finding and line search."""
    f = f.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp_direction(f, M, n, time_limit=5)
        if g is None:
            no_imp += 1
            if no_imp > 20:
                # Perturb and retry
                f = best_f + 0.02 * rng.normal(size=n)
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                _, M = compute_conv_and_max(f)
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
            if no_imp > 100:
                break
            continue
        
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 50:
                f = 0.9 * best_f + 0.1 * np.abs(rng.normal(size=n)) + 1e-10
                f = f / f.sum()
                _, M = compute_conv_and_max(f)
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
            if no_imp > 200:
                break
            continue
        
        # Line search
        improved = False
        for t in (0.3, 0.5, 0.2, 0.7, 0.4, 0.1, 0.8, 0.15):
            f_new = (1.0 - t) * f + t * g
