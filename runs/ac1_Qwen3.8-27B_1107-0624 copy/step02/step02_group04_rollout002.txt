def _full_convolve(a):
    """Compute full convolution using FFT for efficiency."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Remove small negative values from numerical error
    if c.min() < 0:
        c = c - c.min()
    return c

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logit-space mirror descent with exact subgradient, Nesterov momentum,
adaptive active-set tracking, and multi-length escalation.
Exploits the convexity of the problem (for fixed sum) to guarantee convergence.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _softmax(z, floor=1e-30):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(z.max())
    z = np.clip(z, -100.0, 100.0)
    e = np.exp(z) + floor
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return e / s


def _score_normalized(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s < 1e-30:
        return np.inf
    pn = p / s
    c = _full_convolve(pn)
    return float(2.0 * n * c.max())


def _exact_subgrad(p, c, eps_frac=0.01):
    """
    Compute exact subgradient of max(conv(p,p)) w.r.t. p.
    Uses near-active set (lags within eps_frac of max) for stability.
    Returns (gradient vector of size n, num_active).
    """
    n = p.size
    max_c = float(c.max())
    threshold = max_c - eps_frac * max_c
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[lo:hi + 1] += p[k - idx]

    grad /= float(len(active))
    return grad, len(active)


def _mirror_descent(p0, deadline, rng, lr_init=0.06, eps_start=0.01,
                    nesterov=True, max_restarts=20, name=""):
    """
    Logit-space mirror descent with exact subgradient and Nesterov momentum.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z = z - float(z.mean())

    best_p = p0.copy()
    best_score = float(2.0 * n * _full_convolve(p0).max())

    momentum = 0.0
    momentum_target = 0.85
    momentum_rate = 0.0005

    z_outer = z.copy()
    t = 0
    no_improve = 0
    patience = max(300, 2 * n)
    current_eps = eps_start
    restarts = 0

    while time.monotonic() < deadline:
        t += 1

        if nesterov:
            momentum = min(momentum_target, momentum + momentum_rate)
            p_outer = _softmax(z_outer)
            c_outer = _full_convolve(p_outer)
            grad, n_act = _exact_subgrad(p_outer, c_outer, eps=current_eps)
        else:
            p = _softmax(z)
            c = _full_convolve(p)
            grad, n_act = _exact_subgrad(p, c, eps=current_eps)

        # Center gradient for simplex constraint
        p_cur = _softmax(z)
        grad_centered = grad - float(np.dot(grad, p_cur))

        # Adaptive learning rate with cosine-like decay
        lr = lr_init / (1.0 + 0.005 * t)

        if nesterov:
            z_outer = z_outer - lr * grad_centered
            z = (1.0 - momentum) * z_outer + momentum * z
        else:
            z = z - lr * grad_centered

        z = np.clip(z, -60.0, 60.0)
        z = z - float(z.mean())

        p = _softmax(z)
        c = _full_convolve(p)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt epsilon
        if t % 30 == 0 and n_act > 0:
            sorted_c = np.sort(c)[::-1]
            if len(sorted_c) >= 2:
                gap = sorted_c[0] - sorted_c[1]
                current_eps = max(0.001, min(eps_start, gap * 0.3))

        # Restart
        if no_improve > patience and restarts < max_restarts:
            restarts += 1
            noise_scale = 0.08 + 0.04 * restarts
            z = np.log(np.maximum(best_p, 1e-15))
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z = z - float(z.mean())
            z_outer = z.copy()
            momentum = 0.0
            no_improve = 0
            current_eps = eps_start
            if restarts % 5 == 0:
                lr_init = max(0.01, lr_init * 0.8)

    return best_p, best_score


def _make_profile(n, shape):
    t = np.linspace(0.0, 1.0, n)
    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'linear':
        a = 1.0 + 1.8 * (t - 0.5)
    elif shape == 'triangular':
        a = 1.0 - 0.7 * np.abs(2 * t - 1)
    elif shape == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2 * np.pi * t))
        a = a + 0.3
    elif shape == 'quadratic':
        u = 2 * t - 1
        a = 1.0 + 0.6 * u - 0.4 * u ** 2
    elif shape == 'cubic':
        u = 2 * t - 1
        a = 1.0 + 1.0 * u - 0.6 * u ** 3
    elif shape == 'edge_boost':
        a = 1.0 + 0.5 * np.cos(np.pi * t)
    elif shape == 'w':
        a = 1.0 + 0.4 * np.cos(2 * np.pi * t) - 0.2 * np.cos(4 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n_old = p.size
    if n_old == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()
