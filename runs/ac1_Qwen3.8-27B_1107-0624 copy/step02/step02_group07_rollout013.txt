    """Build constraint matrix for LP: (s*b)[k] for all k."""

        return s.copy()

import time
from scipy.optimize import linprog

"""
Minimizes 2n*max(conv(s,s))/(sum(s))^2 using:
1. Diverse asymmetric initialization with beta-distribution and geometric shapes
2. FFT-based multiplicative gradient descent with adaptive top-K peak weighting
3. LP-guided direction finding with careful line search
4. Coordinate descent for 1D minimax refinement
5. Multi-phase search with adaptive time allocation
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def fft_grad(s, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. s using FFT."""
    n = len(s)
    N = 1
    while N < 4 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    W = np.fft.rfft(w, N)
    t1 = np.fft.irfft(np.conj(S) * W, N)[:n]
    t2 = np.fft.irfft(S * W, N)[:n]
    return t1 + t2


def build_lp_matrix(s):
    """Build constraint matrix A where A[k,i] gives coefficient of b[i] in (s*b)[k]."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = s
    return A


def lp_direction(s, M, timeout=5.0):
    """Solve LP: max sum(b) s.t. (s*b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_lp_matrix(s)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': timeout})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def grad_step(s, K=5, alpha=1.0, eta=0.3, floor=1e-10):
    """Multiplicative gradient step with top-K peak weighting."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return s.copy()

    # Top-K exponential weighting
    sorted_idx = np.argsort(conv)[::-1]
    w = np.zeros(len(conv))
    k_use = min(K, len(conv))
    for rank in range(k_use):
        w[sorted_idx[rank]] = np.exp(-alpha * rank)
    # Small uniform floor to avoid completely ignoring lower peaks
    w += 0.005

    grad = fft_grad(s, w)
    grad = np.maximum(grad, 0)
    gmax = grad.max()
    if gmax < 1e-15:
        return s.copy()

    s_new = s * np.exp(-eta * grad / gmax)
    s_new = np.maximum(s_new, floor)
    s_new = s_new / s_new.sum()
    return s_new


def coordinate_refine(s, n_passes=2):
    """Coordinate descent: for each position, try to reduce max convolution."""
    n = len(s)
    s = s.copy()
    current_max = np.convolve(s, s).max()

    for _ in range(n_passes):
        for i in range(n):
            old_val = s[i]
            # Try several candidate values
            candidates = [0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75,
                          old_val, old_val * 1.5, old_val * 2.0]
            # Also try value that would make conv[i] = conv[2n-2-i] (symmetry hint)
            best_val = old_val
            best_m = current_max

            for val in candidates:
                s[i] = val
                conv = np.convolve(s, s)
                m = conv.max()
                if m < best_m - 1e-12:
                    best_m = m
                    best_val = val

            s[i] = best_val
            if best_m < current_max:
                current_max = best_m

    return s, current_max

                        s = (1 - best_t) * s + best_t * b
                        s = np.maximum(s, 0) / s.sum()
                        sc = best_sc
                        improved = True

            # Coordinate refinement
            if n <= 450:
                s_coord, _ = coordinate_refine(s, n_passes=1)
                sc_coord = fast_score(s_coord)
                if sc_coord < sc - 1e-12:
                    s = s_coord
                    sc = sc_coord
                    improved = True

            # Small gradient nudge
            s_nudge = grad_step(s, K=2, alpha=0.5, eta=0.05)
            sc_nudge = fast_score(s_nudge)
            if sc_nudge < sc - 1e-12:
                s = s_nudge
                sc = sc_nudge
                improved = True

            if sc < global_best_score - 1e-10:
                global_best_score = sc
                global_best_seq = s.copy()
                if rnd % 5 == 0:
                    print(f"  Round {rnd}: score={sc:.10f} ***NEW BEST***")
            elif rnd % 5 == 0:
                print(f"  Round {rnd}: score={sc:.10f}")

            if not improved and sc == global_best_score:
                # Stuck - try perturbation
                s = s + 0.003 * rng.standard_normal(n)
                s = np.maximum(s, 0.001)
                s = s / s.sum()

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = evaluate_sequence(result)
    print(f"\n{'='*60}")
    print(f"FINAL: score={final_score:.10f}, length={len(result)}")
    print(f"{'='*60}")

    return result
