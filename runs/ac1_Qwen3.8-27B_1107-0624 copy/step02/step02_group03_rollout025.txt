"""
Multi-strategy optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP direction-finding, log-sum-exp gradient polishing with
FFT-accelerated convolutions, and multi-resolution search.
Improvements over previous: better line search, adaptive softmax temperature,
FFT-based gradients, multi-resolution (512->1024), and random restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def fft_convolve(a, b):
    """FFT-based full convolution."""
    n = len(a) + len(b) - 1
    m = 1
    while m < n:
        m *= 2
    fa = fft(a, m)
    fb = fft(b, m)
    c = ifft(fa * fb)
    return c[:n].real


def fft_cross_corr(w, a):
    """
    Compute s[j] = sum_k w[k] * a[k-j] for j=0..len(a)-1.
    Uses (w is length 2n-1, a is length n, both non-negative).
    """
    n = len(a)
    n_c = len(w)
    L = 1
    while L < n_c + n - 1:
        L *= 2
    W = fft(w, L)
    A = fft(a, L)
    # ifft(W * conj(A)) gives r[m] = sum_k w[k] * a[(k-m) mod L]
    # We want s[j] = sum_k w[k] * a[k-j] = r[j] (for j in [0, n-1], no wrap if L large enough)
    r = ifft(W * np.conj(A))
    return r[:n].real


def lse_objective_and_grad(a, beta):
    """
    Compute log-sum-exp of conv(a,a) with temperature beta, and its gradient.
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fft_convolve(a, a)
    x = c / beta
    x_max = x.max()
    x_shift = x - x_max
    exp_x = np.exp(x_shift)
    s_exp = exp_x.sum()
    if s_exp < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(x)] = 1.0
        lse_val = beta * x_max
    else:
        w = exp_x / s_exp
        lse_val = beta * (x_max + np.log(s_exp))
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
    s = fft_cross_corr(w, a)
    grad = 2.0 * s
    return lse_val, grad


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=15.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and conv(f,g)[x] <= M.
    Returns optimal g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_with_linesearch(f, max_iters=500, deadline=None, verbose=True):
    """
    LP-based iterations with backtracking line search.
    f must be normalized to sum 1.
    Returns (best_f, best_score).
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    score = 2.0 * n * M
    best_f = f.copy()
    best_score = score
    no_improve = 0
    t_step = 0.5

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M, timeout=10.0)
        if g_raw is None:
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Backtracking line search
        best_t = 0.0
        best_trial = np.inf
        for t in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = normalize(f_new)
            if f_new is None:
                continue
            sc = 2.0 * n * float(np.convolve(f_new, f_new).max())
            if sc < best_trial:
                best_trial = sc
                best_t = t

        if best_t > 0 and best_trial < best_score - 1e-14:
            f = normalize((1.0 - best_t) * f + best_t * g)
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            no_improve += 1
            f = best_f.copy()
            M = float(np.convolve(f, f).max())
            if no_improve > 100:
                break

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}, M={M:.8f}")

    return best_f, best_score


def adam_polish_lse(p0, deadline, lr=0.05, verbose=True):
    """
    Adam gradient descent in logit space using log-sum-exp objective.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1, using LSE as smooth surrogate.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = 0.1 * max_c  # initial temperature

    while time.monotonic() < deadline:
        it += 1
        lse_val, grad_p = lse_objective_and_grad(p0, beta)

        # Project gradient: remove component in direction of ones (to maintain sum=1 in logit space)
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adapt beta: keep it related to the "width" of the peak
        if it % 100 == 0:
            # Compute effective number of peaks
            x = (c - max_c) / max(beta, 1e-30)
            exp_x = np.exp(x - x.max())
            H = -float(np.sum((exp_x / exp_x.sum()) * np.log(exp_x / exp_x.sum() + 1e-30)))
            eff_n = float(np.exp(H))
            # Adapt beta: smaller beta when peaks are more concentrated
            target_beta = max_c / max(eff_n, 1.0) * 2.0
            beta = 0.7 * beta + 0.3 * target_beta
            beta = max(beta, max_c * 0.01)

        if it % 500 == 0:
            if no_imp > 200:
                lr = max(0.01, lr * 0.9)
            
        if no_imp > 3000:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            beta = 0.1 * float(np.convolve(p0, p0).max())
            continue

        p0 = p
        if verbose and it % 1000 == 0:
            print(f"    LSE-Adam iter {it}: score={best_score:.10f}, beta={beta:.8f}, lr={lr:.6f}")

    return best_p, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    if time.monotonic() < deadline - 10 and best_score > 1.503:

        n_try = len(best_seq)
        for _ in range(3):

            pert = normalize(pert)
            if pert is None:

            dl = min(deadline - 10, time.monotonic() + 30)

            if sc_try < best_score:

    if final_score < best_score:
        best_score = final_score
