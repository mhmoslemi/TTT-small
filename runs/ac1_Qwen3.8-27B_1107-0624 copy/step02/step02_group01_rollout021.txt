def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def solve_lp_direction(f, M, n, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x
    return None

def softmax_polish(f0, deadline, rng, lr=0.05, max_iter=30000):
    """Softmax-based gradient polishing in logit space."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size
    
    z = np.log(f)
    z -= z.mean()
    
    c = fft_convolve(f)
    M_val = float(c.max())
    beta = 0.1 * M_val
    
    best_f = f.copy()
    best_M = M_val
    no_imp = 0
    N_corr = next_pow2(3 * n)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(f)
        M_val = float(c.max())
        
        # Softmax weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))
        
        # Gradient in logit space
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]
        grad_z = 2.0 * f * (h - L)
        
        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f) - np.log(best_f).mean()
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= z.mean()
        
        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        M_new = float(fft_convolve(f).max())
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if t_adam % 200 == 0:
            beta = max(1e-12, beta * 0.8)
        
        if no_imp > max(200, 2 * n):
            z = 0.8 * np.log(best_f) - 0.8 * np.log(best_f).mean() + 0.5 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            M_val = float(fft_convolve(f).max())
            beta = max(1e-12, 0.1 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_f, best_M

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines: cvxpy QP solves at small n for exact shapes, subgradient pre-optimization,
LP-based direction finding for precise polishing, and softmax gradient refinement.
Uses KKT-condition-based verification and active-set LP reduction for speed.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        rho = np.array([0])
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_exponential_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()

def make_smooth_random(n, rng, n_harm=10):

    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.6) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cubic_profile(n, a1, a3):

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def subgradient_direction(f, n):
    """Compute the average subgradient of max(a*a) at point f."""
    c = fft_convolve(f)
    M_val = float(c.max())
    active = np.where(c >= M_val - 1e-12)[0]
    
    # Compute grad[j] = 2/|S| * sum_{k in S} f[k-j]
    # Using FFT cross-correlation
    N_c = next_pow2(3 * n)
    ind = np.zeros(2 * n - 1)
    ind[active] = 1.0
    fi = np.fft.rfft(ind, N_c)
    ff = np.fft.rfft(f, N_c)
    cross = np.fft.irfft(fi * np.conj(ff), N_c)[:n]
    grad = 2.0 * cross / len(active)
    return grad, M_val, active

def subgradient_optimize(a0, deadline, max_iter=200000, verbose=False):
    """Subgradient method with adaptive step size and averaging."""
    f = a0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return a0
    f = f / s
    n = f.size
    
    best_f = f.copy()
    best_M = compute_max_conv(f)
    
    # Initialize step size
    eta = 1.0 / np.sqrt(n)
    avg_f = f.copy()
    total_weight = 1.0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, M_val, active = subgradient_direction(f, n)
        
        # Take step
        f_new = f - eta * grad
        f_new = project_simplex(f_new)
        
        M_new = compute_max_conv(f_new)
        
        # Weighted averaging (Polyak-Ruppert)
        w = it + 1
        avg_f = (avg_f * total_weight + f_new * w) / (total_weight + w)
        total_weight += w
        
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            eta = min(eta * 1.02, 5.0 / n)
        else:
            eta *= 0.98
            f_new = project_simplex(f - 0.5 * eta * grad)
            M_new2 = compute_max_conv(f_new)
            if M_new2 < best_M:
                best_M = M_new2
                best_f = f_new.copy()
        
        f = f_new
        avg_f = project_simplex(avg_f)
        M_avg = compute_max_conv(avg_f)
        if M_avg < best_M:
            best_M = M_avg
            best_f = avg_f.copy()
        
        if verbose and it % 5000 == 0:
            print(f"      Subgrad iter {it}: M={M_new:.10f}, best={best_M:.10f}, eta={eta:.6f}")
    
    return best_f

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_u
