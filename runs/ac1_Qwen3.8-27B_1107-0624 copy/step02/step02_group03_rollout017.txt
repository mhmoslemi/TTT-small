from scipy.optimize import linprog

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_direction(f, M, timeout=10.0):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_iterate(f, max_iters=200, t_init=0.4, t_min=0.02, deadline=None):

    f = f.astype(np.float64).copy()
    f = normalize(f)
    if f is None:
        return None, np.inf
    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M
    no_improve = 0
    t_step = t_init
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g_raw = lp_direction(f, M, timeout=min(10, (deadline - time.monotonic()) if deadline else 10))
        if g_raw is None:
            break
        g = normalize(g_raw)
        if g is None:
            break
        f_new = (1.0 - t_step) * f + t_step * g
        f_new = normalize(f_new)
        if f_new is None:

            best_score = score_new

    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= np.mean(z)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    c = np.convolve(p0, p0)
    max_c = float(c.max())

    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.1 * max_c)  # Start with smaller beta for sharper focus
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())
        x = (c - max_c) / beta
        ex = np.exp(x)

            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        h = corr_weights(w, p0)
        grad_p = 2.0 * h + reg * p0
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum
        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())
        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        # Anneal beta
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.5)
        if no_imp > 1000:
            z = best_z + 0.1 * np.random.default_rng(it).normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            beta = max(1e-10, 0.1 * float(np.convolve(np.exp(z - z.max()), np.exp(z - z.max())).max()))
        p0 = p
    return best_p, best_score

def spssa_optimize(p0, deadline, A=0.1, a=10, gamma=0.6, c=0.01, max_iters=50000):
    n = len(p0)
    p = p0.copy()
    p = p / p.sum()
    best_p = p.copy()
    best_score = internal_score(p)
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # Generate random direction
        delta = np.random.choice([-1, 1], size=n)
        # Compute c_k
        c_k = c / (it ** gamma)
        p_plus = p + c_k * delta
        p_plus = np.maximum(p_plus, 0)
        p_plus = p_plus / p_plus.sum()
        p_minus = p - c_k * delta
        p_minus = np.maximum(p_minus, 0)
        p_minus = p_minus / p_minus.sum()
        f_plus = internal_score(p_plus)
        f_minus = internal_score(p_minus)
        # SPSA update
        a_k = A / (it + a) ** a
        g = (f_plus - f_minus) / (2 * c_k)
        p = p - a_k * g * delta
        p = np.maximum(p, 0)
        p = p / p.sum()
        score = internal_score(p)
        if score < best_score:
            best_score = score
            best_p = p.copy()
    return best_p, best_score

Multi-stage optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding, Adam gradient descent with annealing,
and SPSA for final non-smooth optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog

def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

    A = build_conv_matrix(f)

                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def corr_weights(w, p):
    n = len(p)
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.dot(w[i:i + n], p)
    return h

def adam_polish(p0, deadline, lr=0.05, reg=1e-9):
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= np.mean(z)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.05 * max_c)
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        h = corr_weights(w, p0)
        grad_p = 2.0 * h + reg * p0
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); it = 0
            continue
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum
        score = 2.0 * n * float(np.convolve(p, p).max())
        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        # Aggressive annealing
        if it % 50 == 0:
            beta = max(1e-12, beta * 0.4)
        if no_imp > 500:
            rng = np.random.default_rng(it * 7 + 13)
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0); v.fill(0); it = 0
            no_imp = 0
            lr = base_lr
            beta = max(1e-10, 0.05 * float(np.convolve(np.exp(z-z.max()), np.exp(z-z.max())).max()))
        p0 = p
    return best_p, best_score

def interpolate_seq(a, n_new):

    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def make_profile(n, type='linear'):
    if type == 'linear':
        a = np.arange(1, n + 1, dtype=np.float64)
    elif type == 'quadratic':
        a = np.arange(1, n + 1, dtype=np.float64) ** 1.5
    elif type == 'exp':
        a = np.exp(np.linspace(0, 1, n))
    elif type == 'cosine':
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t)
    elif type == 'sqrt':
        a = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

"""
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines polynomial initialization, multi-scale LP direction finding,
Adam gradient descent with aggressive annealing, and SPSA for non-smooth polish.
Key insight: asymmetric (ramp-like) sequences minimize auto-convolution peak.

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=5.0):
    """Solve LP: maximize sum(g) s.t. g>=0, convolve(f,g)<=M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for soft gradient."""

    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    z = np.log(p0)
    z -= np.mean(z)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, beta_init_factor * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex
        h = corr_weights(w, p0)
        grad_p = 2.0 * h + reg * p0
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            p0 = np.exp(z - z.max())
            p0 = p0 / p0.sum()
            continue
        p = p / p_sum
        score = 2.0 * n * float(np.convolve(p, p).max())
        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        # Aggressive annealing schedule
        if it % 40 == 0:
            beta = max(1e-12, beta * 0.35)
        # Restart on stagnation
        if no_imp > 400:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.02 + 0.03 * (no_imp / 1000.0)
            z = best_z + scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            lr = base_lr
            p_tmp = np.exp(z - z.max())
            p_tmp = p_tmp / p_tmp.sum()
            beta = max(1e-10, beta_init_factor * float(np.convolve(p_tmp, p_tmp).max()))
        p0 = p
    return best_p, best_score


def spssa_optimize(p0, deadline, A=0.3, gamma_a=0.6, c0=0.005, gamma_c=0.5):
    """SPSA for direct non-smooth optimization."""
    n = len(p0)
    p = p0.copy().astype(np.float64)
    p = p / p.sum()
    best_p = p.copy()
    best_score = internal_score(p)
    it = 0
    rng = np.random.default_rng(99999)
    while time.monotonic() < deadline:
        it += 1
        delta = rng.choice([-1.0, 1.0], size=n)
        c_k = c0 / (1 + it * 0.005) ** gamma_c
        p_plus = np.maximum(p + c_k * delta, 1e-15)
        p_plus = p_plus / p_plus.sum()
        p_minus = np.maximum(p - c_k * delta, 1e-15)
        p_minus = p_minus / p_minus.sum()
        f_plus = internal_score(p_plus)
        f_minus = internal_score(p_minus)
        a_k = A / (it + 30) ** gamma_a
        g = (f_plus - f_minus) / (2.0 * c_k)
        p = p - a_k * g * delta
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s < 1e-30:
            p = best_p.copy()
            continue
        p = p / s
        score = internal_score(p)
        if score < best_score:
            best_score = score
            best_p = p.copy()
        if it % 5000 == 0:
            print(f"    SPSA iter {it}: score={best_score:.8f}")
    return best_p, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_poly_profile(n, coeffs):
    """Create polynomial profile: a(t) = sum coeffs[k] * t^k, t in [0,1]."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * (t ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_profile(n, ptype):
    """Create various analytic profiles."""
    if ptype == 'linear':
        a = np.arange(1, n + 1, dtype=np.float64)
    elif ptype == 'quad':
        a = (np.arange(1, n + 1, dtype=np.float64)) ** 1.5
    elif ptype == 'exp':
        a = np.exp(np.linspace(0, 0.8, n))
    elif ptype == 'sqrt':
        a = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    elif ptype == 'cos_taper':
        t = np.linspace(0, 1, n)
        a = t + 0.3 * (1.0 - np.cos(np.pi * t))
    elif ptype == 'log':
        a = np.log(np.arange(2, n + 2, dtype=np.float64))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate():
    """
    Multi-resolution LP + Adam + SPSA search for non-negative sequences
    minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    total_budget = 1800.0
    deadline = t_start + total_budget - 15.0

    # Phase 0: Generate diverse initializations
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-10:
                parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={internal_score(parent):.8f}")

    candidates = []
    if parent is not None:
        candidates.append(parent)
        candidates.append(parent[::-1].copy())
        for seed in range(8):
            rng = np.random.default_rng(seed * 101)
            scale = 0.02 + 0.02 * seed
            perturbed = parent + scale * rng.normal(size=len(parent))
            perturbed = normalize(perturbed)
            if perturbed is not None:
                candidates.append(perturbed)

    # Analytic profiles at n=512
    for ptype in ['linear', 'quad', 'exp', 'sqrt', 'cos_taper', 'log']:
        candidates.append(make_profile(512, ptype))

    # Polynomial profiles (asymmetric)
    poly_set = [
        [1, 1.5, 0.0],
        [1, 2.0, -0.5],
        [1, 1.0, 0.5, -0.2],
        [1, 2.5, -1.0],
        [0.5, 2.0, 0.5],
        [1, 1.5, -0.3, 0.1],
        [0.8, 2.2, -0.8, 0.1],
        [1, 3.0, -2.0, 0.5],
    ]
    for coeffs in poly_set:
        candidates.append(make_poly_profile(512, coeffs))

    # Evaluate all candidates
    best_seq = None
    best_score = np.inf
    for c in candidates:

    print(f"Best initial: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: LP at n=64 (fast, many iterations)

        n1 = 64
        a1 = interpolate_seq(best_seq, n1)
        if a1 is not None:
            print(f"\nPhase 1: LP at n={n1}, score={internal_score(a1):.8f}")
            t0 = time.monotonic()
            M_init = float(np.convolve(a1, a1).max())
            _ = lp_direction(a1, M_init, timeout=2)
            t_lp = time.monotonic() - t0
            print(f"  LP time: {t_lp*1000:.1f}ms")
            max_iters_1 = min(500, max(30, int((deadline_p1 - time.monotonic()) / max(t_lp + 0.005, 0.02))))
            f_best, sc_best = lp_iterate(a1, max_iters=max_iters_1, t_init=0.4, t_min=0.02, deadline=deadline_p1)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 1 result: score={best_score:.8f}")

    # Verify

        print(f"evaluate_sequence confirms score: {eval_score:.8f}")
    except Exception as e:
