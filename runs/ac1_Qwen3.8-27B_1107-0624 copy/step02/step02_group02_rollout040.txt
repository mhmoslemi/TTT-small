The autocorrelation conv(a,a) has specific structure:

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based smooth subgradient optimization with Adam in log-space,
combined with trust-region active-set polishing and multi-n exploration.
"""
import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fft_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def safe_softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -30.0, 30.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s


def cross_corr_wa(w, a):
    """Compute G[i] = sum_k w[k] * a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    G = np.fft.irfft(np.conj(fw) * fa, N)
    return G[:n]


def smooth_max_value_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-14
        cnt = float(np.count_nonzero(mask))
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return float(val), w


def adam_update(state, grad, t, lr):
    """Adam optimizer state update. Returns new z and updated state."""
    z, m, v = state
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * (grad * grad)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    z_new = np.clip(z_new, -30.0, 30.0)
    z_new = z_new - float(np.mean(z_new))
    return z_new, (z_new, m, v)


def optimize_smooth(a0, deadline, rng, verbose=True):
    """
    Main smooth optimization in log-space with Adam.
    Adapts beta (sharpness) over time.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size

    z = np.log(a0)
    z = z - float(np.mean(z))
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0

    beta = 0.3 * float(fft_autocorr(a0).max())
    beta_min = 1e-6
    lr = 0.05

    best_a = a0.copy()
    best_score = float(2.0 * n * fft_autocorr(a0).max())
    no_improve = 0
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() < deadline:
        a = safe_softmax(z)
        c = fft_autocorr(a)
        val, w = smooth_max_value_weights(c, beta)

        # Compute gradient
        g = 2.0 * cross_corr_wa(w, a)
        dot = float(np.dot(a, g))
        grad_z = a * (g - dot)

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a)
            z = z - float(np.mean(z))
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t_adam = 0
            beta = 0.3 * float(fft_autocorr(best_a).max())
            continue

        t_adam += 1
        z, (z, m, v) = adam_update((z, m, v), grad_z, t_adam, lr)
        a = safe_softmax(z)
        c = fft_autocorr(a)
        cur_score = float(2.0 * n * c.max())

        if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1
            stagnation_count += 1

        # Anneal beta
        if t_adam % 50 == 0:
            beta = max(beta_min, beta * 0.92)

        # Adapt learning rate
        if no_improve > 500:
            lr = max(0.005, lr * 0.95)
        if no_improve > 2000:
            # Restart with noise
            z = np.log(best_a) + 0.1 * rng.standard_normal(n)
            z = z - float(np.mean(z))
            z = np.clip(z, -30.0, 30.0)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t_adam = 0
            beta = max(0.1 * float(fft_autocorr(best_a).max()), beta_min)
            lr = 0.05
            no_improve = 0

        iter_count += 1
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic()
            print(f"    smooth opt: iter={iter_count}, score={best_score:.10f}, beta={beta:.6f}, lr={lr:.4f}")

    return best_a, best_score


def active_set_polish(a0, deadline, rng, verbose=False):
    """
    Polish using active-set identification and targeted LP-like steps.
    Identifies lags near the maximum and performs targeted adjustments.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size

    best_a = a0.copy()
    best_M = float(fft_autocorr(a0).max())
    best_score = float(2.0 * n * best_M)

    tol = 0.01
    step_size = 0.05

    while time.monotonic() < deadline:
        a = best_a.copy()
        c = fft_autocorr(a)
        M = float(c.max())

        # Find active lags (within tol*M of maximum)
        active = np.where(c >= M - tol * M)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute combined subgradient from active lags
        # For each active lag k, the subgradient component at position i is 2*a[k-i] (if valid)
        # We want to find a direction d that reduces all active lags simultaneously
        # Simple approach: average the subgradients and take a projected step

        g_sum = np.zeros(n, dtype=np.float64)
        for k in active:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    g_sum[i] += 2.0 * a[j]
        g_sum /= len(active)

        # Projected gradient step on the simplex
        # d[i] = g_sum[i] - mean(g_sum) (to maintain sum constraint)
        d = g_sum - float(np.mean(g_sum))

        # Line search for step size
        best_step = 0.0
        best_new_M = M
        for alpha in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            a_new = a - alpha * d
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            c_new = fft_autocorr(a_new)
            M_new = float(c_new.max())
            if M_new < best_new_M:
                best_new_M = M_new
                best_step = alpha

        if best_step > 0 and best_new_M < best_M - 1e-14:
            a_new = best_a - best_step * d
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_a = a_new
            best_M = best_new_M
            best_score = float(2.0 * n * best_M)
        else:
            # Try random perturbation
            if rng.random() < 0.3:
                a_new = best_a + 0.02 * rng.standard_normal(n)
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                c_new = fft_autocorr(a_new)
                if c_new.max() < best_M:
                    best_a = a_new
                    best_M = float(c_new.max())
                    best_score = float(2.0 * n * best_M)
            # Decrease tolerance
            tol = max(0.001, tol * 0.8)
            step_size = max(0.005, step_size * 0.8)

        if verbose:
            print(f"    active set polish: M={best_M:.10f}, |active|={len(active)}, tol={tol:.4f}")

    return best_a, best_score


def generate_smooth_random(n, rng, n_harmonics=8):
    """Generate a random smooth positive sequence via Fourier series."""
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.4) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def generate_asymmetric_power(n, rng):
    """Generate an asymmetric power-law profile."""
    t = np.linspace(0.0, 1.0, n)
    alpha = rng.uniform(0.3, 2.5)
    center = rng.uniform(0.2, 0.8)
    a = 1.0 + alpha * np.abs(t - center) ** 0.7
    # Add some asymmetry
    a += 0.3 * rng.uniform(-1, 1) * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(a, n_new):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-10 else a
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: smooth subgradient + active-set polish + multi-n.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ============ Phase 1: Smooth optimization at multiple n values ============
    n_values = [512, 384, 640, 768]
    phase1_deadline = min(deadline, t_start + 900.0)

    for n in n_values:
        if time.monotonic() > phase1_deadline - 20:
            break

        # Get starting point for this n
        if init_seq is not None and init_seq.size == n:
            a_start = init_seq.copy()
        elif init_seq is not None:
            a_start = interpolate_sequence(init_seq, n)
        else:
            a_start = generate_asymmetric_power(n, rng)

        sc = compute_score(a_start)
        update(a_start, sc)

        # Smooth optimization
        per_deadline = min(phase1_deadline, time.monotonic() + 200.0)
        result, sc_res = optimize_smooth(a_start, per_deadline, rng, verbose=False)
        update(result, sc_res)
        print(f"  n={n}: after smooth opt, score={sc_res:.10f}")

        # Active set polish
        if time.monotonic() < phase1_deadline - 30:
            per_deadline2 = min(phase1_deadline, time.monotonic() + 60.0)
            result2, sc_res2 = active_set_polish(result, per_deadline2, rng, verbose=False)
            update(result2, sc_res2)
            print(f"  n={n}: after active polish, score={sc_res2:.10f}")

        # Try a few random restarts
        for restart in range(3):
            if time.monotonic() > phase1_deadline - 10:
                break
            a_rand = generate_smooth_random(n, rng)
            a_mix = 0.5 * best_seq[:n] if best_seq is not None and best_seq.size >= n else a_rand
            if best_seq is not None:
                a_mix = 0.5 * interpolate_sequence(best_seq, n) + 0.5 * a_rand
                a_mix = np.maximum(a_mix, 1e-10)
                a_mix = a_mix / a_mix.sum()
            per_deadline3 = min(phase1_deadline, time.monotonic() + 50.0)
            result3, sc_res3 = optimize_smooth(a_mix, per_deadline3, rng, verbose=False)
            update(result3, sc_res3)

    print(f"After Phase 1: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # ============ Phase 2: Extended optimization at best n ============
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = best_seq.size
        phase2_deadline = min(deadline, t_start + 1400.0)

        # Multiple smooth optimization runs with different parameters
        for trial in range(5):
            if time.monotonic() > phase2_deadline - 10:
                break
            a_start = best_seq + 0.005 * rng.standard_normal(n_best)
            a_start = np.maximum(a_start, 1e-10)
            a_start = a_start / a_start.sum()

            per_deadline = min(phase2_deadline, time.monotonic() + 80.0)
            result, sc_res = optimize_smooth(a_start, per_deadline, rng, verbose=False)
            update(result, sc_res)

            if sc_res < best_score:
                # Follow up with active set polish
                if time.monotonic() < phase2_deadline - 20:
                    per2 = min(phase2_deadline, time.monotonic() + 40.0)
                    result2, sc_res2 = active_set_polish(result, per2, rng, verbose=False)
                    update(result2, sc_res2)

        print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Try n=1024 and n=256 ============
    for n_try in [1024, 256]:
        if time.monotonic() > deadline - 40:
            break
        if best_seq is not None:
            a_start = interpolate_sequence(best_seq, n_try)
        else:
            a_start = generate_asymmetric_power(n_try, rng)

        sc = compute_score(a_start)
        update(a_start, sc)

        per_deadline = min(deadline - 20, time.monotonic() + 100.0)
        result, sc_res = optimize_smooth(a_start, per_deadline, rng, verbose=False)
        update(result, sc_res)
        print(f"  n={n_try}: score={sc_res:.10f}")

    # ============ Phase 4: Final intensive polish ============
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_final = best_seq.size
        # Short smooth optimization with very small beta
        a_start = best_seq.copy()
        per = min(deadline - 10, time.monotonic() + 60.0)
        result, sc_res = optimize_smooth(a_start, per, rng, verbose=True)
        update(result, sc_res)

        # Active set polish
        if time.monotonic() < deadline - 10:
            per2 = min(deadline - 10, time.monotonic() + 30.0)
            result2, sc_res2 = active_set_polish(result, per2, rng, verbose=True)
            update(result2, sc_res2)

    print(f"FINAL: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Final validation
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = generate_asymmetric_power(512, rng)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    final_score = compute_score(final_seq)
    print(f"Final computed score: {final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
