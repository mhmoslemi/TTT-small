def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M, time_limit=30.0):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0. Returns (g, status)."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0), 1
    except Exception:
        pass
    return None, 0

def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

        # Brief smooth
        if time.monotonic() < phase1_end - 5:
            per_dead2 = min(phase1_end, time.monotonic() + 20.0)
            r2, sc2 = smooth_optimize(r1, per_dead2, rng, lr=0.06, verbose=False)
            update(r2, sc2)
        if sc < best_score + 0.005:
            print(f"  Profile {name}: best so far={best_score:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: Smooth polish at n=512 (500-900s) ============
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end - 20 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 2: smooth polish at n={n_polish}, best={best_score:.10f}")

        # Multiple rounds with different LR
        for attempt, lr_val in enumerate([0.06, 0.04, 0.08]):
            if time.monotonic() > phase2_end - 10:
                break
            per_dead = min(phase2_end, time.monotonic() + 120.0)
            p0 = best_seq.copy()
            # Small perturbation for diversity
            noise = 0.002 * rng.normal(size=n_polish)
            p0 = p0 + noise
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result_p, result_sc = smooth_optimize(p0, per_dead, rng, lr=lr_val, verbose=True)
            update(result_p, result_sc)
            print(f"  Smooth round {attempt} (lr={lr_val}): score={result_sc:.10f}")

        # Interleave with LP
        if time.monotonic() < phase2_end - 20:
            per_dead = min(phase2_end, time.monotonic() + 60.0)
            result = lp_descent(best_seq, per_dead, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            print(f"  LP round 2: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Try n=1024 (900-1450s) ============
    n2 = 1024
    phase3_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n2)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 3 start: n={n2}, score={sc:.10f}")

        # LP at n=1024
        lp_dead3 = min(phase3_end, time.monotonic() + 350.0)
        result = lp_descent(p_large, lp_dead3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After LP (n={n2}): score={sc:.10f}")

        # Smooth polish at n=1024
        if time.monotonic() < phase3_end - 20:
            per_dead = min(phase3_end, time.monotonic() + 80.0)
            result_p, result_sc = smooth_optimize(result, per_dead, rng, lr=0.04, verbose=False)
            update(result_p, result_sc)
            print(f"After smooth (n={n2}): score={result_sc:.10f}")

        # Try another profile at n=1024
        if time.monotonic() < phase3_end - 30:
            p_alt = make_raised_cosine(n2, 1.0)
            # Scale to match best
            per_dead = min(phase3_end, time.monotonic() + 60.0)
            result = lp_descent(p_alt, per_dead, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)
            if sc < best_score + 0.01:
                print(f"  Alt profile n={n2}: score={sc:.10f}")
                if time.monotonic() < phase3_end - 10:
                    per_dead2 = min(phase3_end, time.monotonic() + 30.0)
                    result_p, result_sc = smooth_optimize(result, per_dead2, rng, lr=0.05)
                    update(result_p, result_sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Final refinement (1450-1780s) ============
    phase4_end = deadline
    if time.monotonic() < phase4_end - 15 and best_seq is not None:
        n_final = best_seq.size
        print(f"Phase 4: final polish at n={n_final}, best={best_score:.10f}")

        # Aggressive final polish with multiple strategies
        strategies = [
            ("smooth_lr03", lambda p0, d: smooth_optimize(p0, d, rng, lr=0.03, verbose=False)),
            ("smooth_lr05", lambda p0, d: smooth_optimize(p0, d, rng, lr=0.05, verbose=False)),
            ("lp_final", lambda p0, d: lp_descent(p0, d, rng, verbose=False)),
        ]

        for strat_name, strat_fn in strategies:
            if time.monotonic() > phase4_end - 10:
                break
            per_dead = min(phase4_end, time.monotonic() + 60.0)
            p0 = best_seq.copy()
            # Tiny noise
            p0 = p0 + 0.0005 * rng.normal(size=n_final)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            if isinstance(strat_fn, type(lambda: None)):
                continue
            result = strat_fn(p0, per_dead)
            if isinstance(result, tuple):
                result_p, result_sc = result
            else:
                result_p = result
                result_sc = compute_score(result_p)
            update(result_p, result_sc)
            print(f"  {strat_name}: score={result_sc:.10f}")

        # One more LP + smooth combo on the overall best
        if time.monotonic() < phase4_end - 20:
            per_dead = min(phase4_end, time.monotonic() + 40.0)
            r_lp = lp_descent(best_seq, per_dead, rng, verbose=False)
            sc_lp = compute_score(r_lp)
            update(r_lp, sc_lp)

            if time.monotonic() < phase4_end - 10:
                per_dead2 = min(phase4_end, time.monotonic() + 20.0)
                r_sm, sc_sm = smooth_optimize(r_lp, per_dead2, rng, lr=0.03, verbose=False)
                update(r_sm, sc_sm)
                print(f"  Final combo: score={sc_sm:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Final validation and return ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_raised_cosine(512, 1.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_uniform(512)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    # Validate with evaluate_sequence
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score < best_score:
                best_score = eval_score
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s")

    return [float(x) for x in final_seq]
