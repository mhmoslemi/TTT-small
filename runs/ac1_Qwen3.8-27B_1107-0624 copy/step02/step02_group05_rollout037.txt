        deltas = [0.0, 0.005, 0.01, 0.02, 0.03]
        if no_imp > 20:
            deltas = [0.01, 0.02, 0.03, 0.05, 0.08]
        
        for delta in deltas:
            if time.monotonic() > deadline:
                break
            g = solve_lp_variant(f, M, delta)
            if g is None:
                continue
            gs = g.sum()
            if gs < 1e-15:
                continue
            g = g / gs
            candidates.append(g)
        
        if not candidates:
            # Fallback: random perturbation
            f = f + 0.1 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            if f.sum() > 1e-30:
                f /= f.sum()
            M = float(np.convolve(f, f).max())
            no_imp += 1
            continue
        
        # Pick best candidate via line search
        best_M_try = M
        best_f_try = f.copy()
        
        for g in candidates:
            # Blend with momentum
            g_mom = 0.7 * g + 0.3 * momentum_dir
            g_mom = np.maximum(g_mom, 0.0)
            if g_mom.sum() < 1e-30:
                g_mom = g
            else:
                g_mom = g_mom / g_mom.sum()
            
            for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03]:
                f_try = (1.0 - t) * f + t * g_mom
                f_try = np.maximum(f_try, 0.0)
                if f_try.sum() < 1e-30:
                    continue
                f_try = f_try / f_try.sum()
                M_try = float(np.convolve(f_try, f_try).max())
                if M_try < best_M_try - 1e-14:
                    best_M_try = M_try
                    best_f_try = f_try.copy()
                    momentum_dir = g_mom.copy()
                    break
        
        if best_M_try < M - 1e-14:
            f = best_f_try
            M = best_M_try
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        
        # Stall detection
        if no_imp > 40:
            # Try larger perturbation
            scale = 0.15 + 0.1 * min(no_imp / 100, 1.0)
            f = best_f + scale * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            if f.sum() > 1e-30:
                f /= f.sum()
                M = float(np.convolve(f, f).max())
            else:
                f = best_f.copy()
                M = best_M
            momentum_dir = np.zeros(n)
            no_imp = 0
        
        # Periodic larger restarts
        if no_imp > 100:
            # Very different restart: keep 50% of best, randomize rest
            mask = rng.random(n) < 0.5
            f_new = best_f.copy()
            f_new[~mask] = np.abs(rng.normal(size=n[~mask].sum() if n[~mask].sum() > 0 else 1))
            f_new = np.maximum(f_new, 0.0)
            if f_new.sum() > 1e-30:
                f = f_new / f_new.sum()
                M = float(np.convolve(f, f).max())
            else:
                f = best_f.copy()
                M = best_M
            momentum_dir = np.zeros(n)
            no_imp = 0
    
    return best_f, best_M


def adam_polish(f_init, deadline, rng, lr=0.03):
    """Adam-based polish using soft-max approximation of max function."""
    f = np.maximum(f_init, 1e-12)
    f = f / f.sum()
    n = len(f)
    z = np.log(f)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    c = np.convolve(f, f)
    M = float(c.max())
    beta = max(1e-12, 0.15 * M)
    
    best_f = f.copy()
    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr
    
    for it in range(10000):
        if time.monotonic() > deadline:
            break
        
        # Soft-max gradient
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        ss = float(ex.sum())
        if ss < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / ss
        
        # Gradient of soft_max wrt f
        # d/d f[i] of c[k] = (f*k)[k] contribution
        np_ = len(f)
        gp = np.zeros(np_)
        for i in range(np_):
            # c[k] = sum_j f[j]*f[k-j], dc[k]/df[i] = f[k-i] + f[k-i] (if k!=2i) or f[i] (if k=2i)
            # Sum over k: gp[i] = sum_k w[k] * dc[k]/df[i]
            # Efficient: gp[i] = sum_k w[k] * (f[k-i] if 0<=k-i<n and k!=2i contributes 2*f[k-i])
            # Simplification using correlation
            pass
        
        # Use FFT-based correlation for efficiency
        N_fft = 1
        while N_fft < 2 * n - 1:
            N_fft <<= 1
        fw = np.fft.rfft(w, N_fft)
        fp = np.fft.rfft(np.concatenate([f[::-1], np.zeros(N_fft - n)]), N_fft)
        h = np.fft.irfft(fw * fp, N_fft)
        gp = h[n - 1:2 * n - 1]
        
        # Gradient projection (maintain sum constraint)
        vp = gp + 1e-8 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())
        
        if M < best_M - 1e-13:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Anneal beta
        if adam_t % 200 == 0:
            beta = max(1e-14, beta * 0.85)
        
        # Learning rate decay
        if no_imp % 500 == 0 and no_imp > 0:
            lr = max(0.005, lr * 0.9)
        
        if no_imp > max(400, 2 * n):
            z = 0.85 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-12, 0.15 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_f, best_M


def make_ramp_profile(n, alpha=1.0):
    """Power-law ramp: f[i] = (i+1)^alpha, normalized."""
    f = (np.arange(1, n + 1, dtype=np.float64)) ** alpha
    return f / f.sum()


def make_exp_ramp(n, rate=2.0):
    """Exponential ramp."""
    f = np.exp(rate * np.arange(n, dtype=np.float64) / max(n - 1, 1))
    return f / f.sum()


def make_poly_profile(n, coeffs):
    """Polynomial profile: f[i] = sum coeffs[k] * (i/(n-1))^k."""
    t = np.linspace(0, 1, n)
    f = np.zeros(n)
    for k, c in enumerate(coeffs):
        f += c * t ** k
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_random_smooth(n, rng, n_freq=12):
    """Random smooth profile via Fourier coefficients."""
    t = np.linspace(0, 1, n)
    f = np.ones(n)
    for k in range(1, n_freq):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        f += amp * np.cos(2 * np.pi * k * t + phase)
    f = np.maximum(f, 1e-8)
    return f / f.sum()


def interpolate_profile(f, n_new):
    """Interpolate profile to new length using cubic-like interpolation."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)


def get_starting_points(n, rng, parent=None, count=20):
    """Generate diverse starting points for optimization."""
    starts = []
    
    # Power-law ramps with various exponents
    for alpha in [0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 2.5]:
        starts.append(make_ramp_profile(n, alpha))
    
    # Exponential ramps
    for rate in [1.0, 2.0, 3.0, 5.0]:
        starts.append(make_exp_ramp(n, rate))
    
    # Polynomial profiles
    starts.append(make_poly_profile(n, [1.0, 2.0, -1.0]))
    starts.append(make_poly_profile(n, [1.0, 3.0, -2.0]))
    starts.append(make_poly_profile(n, [0.5, 1.0, 2.0]))
    starts.append(make_poly_profile(n, [1.0, -1.0, 2.0]))
    
    # Random smooth profiles
    for _ in range(count):
        starts.append(make_random_smooth(n, rng))
    
    # Parent if available
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_profile(parent, n))
    
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase LP and Adam optimization to minimize 2*n*max(convolve(a,a))/sum(a)^2.
    Uses improved peel-LP, multi-n search, and extended refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = float(sc)
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            pn = np.maximum(p, 0)
            if pn.sum() > 1e-10:
                parent = pn / pn.sum()
                sc = compute_score_norm(parent)
                try_update(parent, sc)
                print(f"Parent: n={len(parent)}, score={sc:.10f}")
    except Exception:
        pass
    
    # Phase 1: LP optimization at n=256 (primary target length)
    n1 = 256
    phase1_end = min(deadline, t_start + 500.0)
    print(f"Phase 1: LP at n={n1}, time allocated: {phase1_end - t_start:.0f}s")
    
    starts = get_starting_points(n1, rng, parent, count=10)
    
    # Evaluate starting points
    scored_starts = []
    for s in starts:
        sc = compute_score_norm(s)
        scored_starts.append((sc, s))
    scored_starts.sort(key=lambda x: x[0])
    print(f"  Best start: {scored_starts[0][0]:.10f}")
    
    # LP optimize from top starts
    results = []
    top_k = min(8, len(scored_starts))
    for i in range(top_k):
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining / max(1, top_k - i), 90.0)
        if alloc < 15:
            break
        run_dl = time.monotonic() + alloc
        f_best, M_best = lp_optimize(scored_starts[i][1], run_dl, rng)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        results.append((sc, f_best))
        print(f"  Start {i+1}/{top_k}: score={sc:.10f}, best={best_score:.10f}")
    
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.10f}")
    
    # Phase 2: Extended LP refinement on best at n=256
    phase2_end = min(deadline, t_start + 800.0)
    print(f"Phase 2: Extended LP refinement, time={phase2_end - time.monotonic():.0f}s")
    
    if results:
        f_best, M_best = lp_optimize(results[0][1], phase2_end, rng, max_iters=5000)
        sc = 2.0 * n1 * M_best
        try_update(f_best, sc)
        print(f"  Phase 2 result: {sc:.10f}, best={best_score:.10f}")
    
    # Phase 3: Try n=512
    if time.monotonic() < deadline - 200:
        n3 = 512
        phase3_end = min(deadline, t_start + 1100.0)
        print(f"Phase 3: n={n3}, time={phase3_end - time.monotonic():.0f}s")
        
        best_source = best_seq if best_seq is not None else results[0][1]
        f_ext = interpolate_profile(best_source, n3)
        f_best3, M_best3 = lp_optimize(f_ext, phase3_end, rng, max_iters=3000)
        sc3 = 2.0 * n3 * M_best3
        try_update(f_best3, sc3)
        print(f"  n={n3} result: {sc3:.10f}, best={best_score:.10f}")
        
        # Also try a fresh start at n=512
        if time.monotonic() < phase3_end - 30:
            for alpha in [1.0, 1.5]:
                f_ramp = make_ramp_profile(n3, alpha)
                remaining = phase3_end - time.monotonic()
                if remaining < 30:
                    break
                f_b, M_b = lp_optimize(f_ramp, time.monotonic() + remaining, rng, max_iters=1500)
                sc_r = 2.0 * n3 * M_b
                try_update(f_b, sc_r)
                print(f"  n={n3} ramp(a={alpha}): {sc_r:.10f}, best={best_score:.10f}")
    
    # Phase 4: Try n=128 (sometimes smaller n gives better ratio)
    if time.monotonic() < deadline - 150:
        n4 = 128
        phase4_end = min(deadline, t_start + 1250.0)
        print(f"Phase 4: n={n4}")
        
        best_source = best_seq if best_seq is not None else results[0][1]
        f_ext4 = interpolate_profile(best_source, n4)
        remaining = phase4_end - time.monotonic()
        if remaining > 20:
            f_b, M_b = lp_optimize(f_ext4, time.monotonic() + remaining, rng, max_iters=2000)
            sc4 = 2.0 * n4 * M_b
            try_update(f_b, sc4)
            print(f"  n={n4} result: {sc4:.10f}, best={best_score:.10f}")
    
    # Phase 5: Try n=384
    if time.monotonic() < deadline - 150:
        n5 = 384
        phase5_end = min(deadline, t_start + 1350.0)
        print(f"Phase 5: n={n5}")
        
        best_source = best_seq if best_seq is not None else results[0][1]
        f_ext5 = interpolate_profile(best_source, n5)
        remaining = phase5_end - time.monotonic()
        if remaining > 30:
            f_b, M_b = lp_optimize(f_ext5, time.monotonic() + remaining, rng, max_iters=2000)
            sc5 = 2.0 * n5 * M_b
            try_update(f_b, sc5)
            print(f"  n={n5} result: {sc5:.10f}, best={best_score:.10f}")
    
    # Phase 6: Adam polish on best solution
    if time.monotonic() < deadline - 100:
        phase6_end = min(deadline, t_start + 1500.0)
        print(f"Phase 6: Adam polish on best, time={phase6_end - time.monotonic():.0f}s")
        
        if best_seq is not None:
            n_best = len(best_seq)
            remaining = phase6_end - time.monotonic()
            if remaining > 30:
                f_ap, M_ap = adam_polish(best_seq, time.monotonic() + remaining, rng, lr=0.02)
                sc_ap = 2.0 * n_best * M_ap
                try_update(f_ap, sc_ap)
                print(f"  Adam polish: {sc_ap:.10f}, best={best_score:.10f}")
    
    # Phase 7: Final LP refinement on the overall best
    if time.monotonic() < deadline - 30:
        phase7_end = deadline
        print(f"Phase 7: Final LP refinement, time={phase7_end - time.monotonic():.0f}s")
        
        if best_seq is not None:

                try_update(f_final, sc_final)
                print(f"  Final LP: {sc_final:.10f}, best={best_score:.10f}")

        best_seq = make_ramp_profile(256, 1.0)
        best_score = compute_score_norm(best_seq)
    
    # Ensure validity: non-negative, sum >= 0.01, values <= 1000
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_ramp_profile(256, 1.0)
        s = float(best_seq.sum())
    # Scale to have reasonable magnitude (score is scale-invariant, but ensure sum >= 0.01)
    if s < 1.0:
        best_seq = best_seq * (1.0 / s)
        s = 1.0
    
    # Final score verification
    final_score = compute_score_norm(best_seq)
    print(f"Final: n={len(best_seq)}, score={final_score:.10f}")
    
    # Try evaluate_sequence for double-check
    try:
        ev_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"evaluate_sequence: {ev_score:.10f}")
        if np.isfinite(ev_score) and ev_score < final_score:
            final_score = ev_score
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return [float(x) for x in best_seq]
