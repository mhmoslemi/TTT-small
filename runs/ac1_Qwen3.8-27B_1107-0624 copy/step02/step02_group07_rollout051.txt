```python
    n = len(s)
    grad = np.zeros(n)
    # Term 1: s[k_star - j] for valid j
    # j ranges 0 to n-1, k_star - j ranges k_star to k_star-(n-1)
    # Valid when 0 <= k_star - j < n, i.e., k_star - n + 1 <= j <= k_star
    j_min1 = max(0, k_star - n + 1)
    j_max1 = min(n - 1, k_star)
    if j_min1 <= j_max1:
```

```python
import numpy as np
import time
from scipy.optimize import linprog

"""
Minimize 2n*max(conv(s,s))/(sum(s))^2 using exact subgradient descent with momentum,
adaptive p-norm weighting, LP guidance, and multiple sequence lengths.
Key improvements: exact peak subgradients (not soft approximations), Nesterov momentum,
adaptive step-size with backtracking, periodic L2 smoothing for escaping local minima,
and exploitation of the existing height_sequence_1 starting point.
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    b = np.convolve(s, s)
    return 2.0 * n * b.max() / (total * total)


def compute_b(s):
    """Full autocorrelation via linear convolution."""
    return np.convolve(s, s)


def peak_subgradient(s, k_star):
    """O(n) gradient of b[k_star] = sum_i s[i]*s[k_star-i] w.r.t. s."""
    n = len(s)
    grad = np.zeros(n)
    lo1 = max(0, k_star - n + 1)
    hi1 = min(n - 1, k_star)
    if lo1 <= hi1:
        idx = k_star - np.arange(lo1, hi1 + 1)
        grad[lo1:hi1 + 1] += s[idx]
    lo2 = max(0, k_star)
    hi2 = min(n - 1, k_star + n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1) - k_star
        grad[lo2:hi2 + 1] += s[idx]
    return grad


def weighted_gradient_fft(s, w):
    """Compute sum_k w[k]*(s[k-j] + s[j-k]) for all j via FFT. w has length 2n-1."""
    n = len(s)
    N = 1
    while N < 4 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    W = np.fft.rfft(w, N)
    t1 = np.fft.irfft(np.conj(S) * W, N)[:n]
    t2 = np.fft.irfft(S * W, N)[:n]
    return t1 + t2


def lp_direction(s, M, time_limit=2.0):
    """LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def generate_inits(n, hseq, rng):
    """Generate diverse initializations for sequence of length n."""
    results = []

    # Uniform
    s = np.ones(n) / n
    results.append((fast_score(s), s.copy()))

    # Cosine windows
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = np.cos(alpha * np.pi * x) ** 2
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    # Sine-based
    for freq in [1.0, 2.0, 3.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = 0.5 + 0.5 * np.sin(freq * np.pi * x)
        s = np.maximum(s, 0.05)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6]:
        k = max(1, int(frac * n))
        for ratio in [0.3, 0.7, 2.0, 3.0]:
            s = np.ones(n)
            s[:k] = ratio
            s = s / s.sum()
            results.append((fast_score(s), s.copy()))

    # Random smooth
    for _ in range(4):
        s = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        s = np.convolve(s, kernel, mode='same')
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    # From height_sequence_1
    if hseq is not None:
        x_old = np.linspace(0, 1, hseq.size)
        x_new = np.linspace(0, 1, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 1e-6)
        h_r = h_r / h_r.sum()
        results.append((fast_score(h_r), h_r.copy()))
        if hseq.size == n:
            s = np.maximum(hseq, 1e-6)
            s = s / s.sum()
            results.append((fast_score(s), s.copy()))

    results.sort(key=lambda x: x[0])
    return results


def optimize(s0, deadline, rng, use_lp=True, verbose=False):
    """
    Main optimization: multiplicative subgradient descent with momentum.
    Alternates between exact peak subgradient and L2-smoothed gradient.
    """
    n = len(s0)
    s = s0.copy()
    s = np.maximum(s, 1e-10)
    s = s / s.sum()

    best_s = s.copy()
    best_score = fast_score(s)

    eta = 0.4
    mu = 0.7
    momentum = np.zeros(n)
    no_improve = 0
    patience = 150
    it = 0
    max_iters = 100000

    # Track phase: 0=exact subgrad, 1=L2-smoothed
    phase = 0
    phase_count = 0

    while it < max_iters:
        if time.monotonic() > deadline:
            break
        it += 1

        b = compute_b(s)
        max_b = b.max()
        mean_b = b.sum() / len(b)
        cur_score = 2.0 * n * max_b / (s.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Patience-based adaptation
        if no_improve >= patience:
            eta *= 0.6
            if eta < 0.01:
                # Restart with perturbation
                s = s + 0.005 * rng.standard_normal(n)
                s = np.maximum(s, 1e-8)
                s = s / s.sum()
                momentum = np.zeros(n)
                eta = 0.3
                mu = 0.7
                no_improve = 0
                patience = 80
            continue

        # Choose gradient type based on phase cycling
        phase_count += 1
        if phase_count % 200 < 50:
            phase = 1  # L2-smoothed phase
        else:
            phase = 0  # Exact subgradient phase

        if phase == 0:
            # Exact subgradient: use top peaks
            top_k = min(5, len(b))
            peak_idx = np.argsort(b)[-top_k:]
            grad = np.zeros(n)
            for k in peak_idx:
                grad += peak_subgradient(s, k)
            grad /= top_k
        else:
            # L2-smoothed: weight by (b - mean)^2
            excess = np.maximum(b - mean_b, 0) ** 2
            if excess.sum() < 1e-15:
                break
            grad = weighted_gradient_fft(s, excess)

        # Clip negative gradient components (we don't want to increase s where grad < 0
        # in the context of multiplicative updates... actually we do, it's fine)
        # But ensure gradient is non-negative for stability
        grad = np.maximum(grad, 0.0)
        grad_max = grad.max()
        if grad_max < 1e-15:
            grad_max = 1e-15

        # Momentum update (heavy ball)
        momentum = mu * momentum + (1 - mu) * grad
        mom_max = momentum.max()
        if mom_max < 1e-15:
            break

        # Multiplicative update with backtracking
        improved = False
        for attempt in range(6):
            step = eta / (1 + 0.1 * attempt)
            s_new = s * np.exp(-step * momentum / mom_max)
            s_new = np.maximum(s_new, 1e-15)
            s_new = s_new / s_new.sum()
            new_score = fast_score(s_new)
            if new_score < cur_score:
                s = s_new
                improved = True
                break

        if not improved:
            eta *= 0.8

        # Slowly recover eta
        if it % 200 == 0 and eta < 0.5:
            eta = min(eta * 1.05, 0.5)

        # LP step (for small n, periodically)
        if use_lp and n <= 384 and it % 300 == 150:
            M = max_b * (s.sum() ** 2)  # Scale back to actual b values
            direction = lp_direction(s, M, time_limit=1.5)
            if direction is not None and direction.sum() > 1e-10:
                direction = direction / direction.sum()
                cur_sc = fast_score(s)
                for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    cand = (1 - t) * s + t * direction

                    cand = cand / cand.sum()
                    if fast_score(cand) < cur_sc:
                        s = cand
                        if fast_score(s) < best_score:
                            best_score = fast_score(s)
                            best_s = s.copy()
                            no_improve = 0
                        break

    return best_s, best_score

        if mom_max < 1e-15:
            break

        # Multiplicative update with backtracking
        improved = False
        for attempt in range(5):
            step = eta / (1 + 0.2 * attempt)
            s_new = s * np.exp(-step * momentum / mom_max)
            s_new = np.maximum(s_new, 1e-15)
            s_new = s_new / s_new.sum()
            new_score = fast_score(s_new)
            if new_score < cur_score * (1 - 1e-10):
                s = s_new
                improved = True
                break

        if not improved:
            eta *= 0.75
        elif eta < 0.5:
            eta = min(eta * 1.01, 0.6)

    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Uses exact subgradient descent with Nesterov momentum, adaptive p-norm weighting,
    LP guidance, multiple sequence lengths, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 40.0

    best_score = float('inf')
    best_seq = np.ones(256) / 256.0

    # Load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, score={fast_score(h/h.sum()):.6f}")
    except Exception:
        pass

    # Priority 1: Optimize from height_sequence_1 at its native length
    if hseq is not None:
        n0 = hseq.size
        s0 = np.maximum(hseq, 1e-8)
        s0 = s0 / s0.sum()
        print(f"\n=== Optimizing height_sequence_1 (n={n0}) ===")
        t_start = time.monotonic()
        alloc = min(400.0, (deadline - time.monotonic()) * 0.3)
        s_best, sc_best = optimize_refined(s0, time.monotonic() + alloc, rng,
                                           use_lp=(n0 <= 384))
        if sc_best < best_score:
            best_score = sc_best
            best_seq = s_best.copy()
            print(f"  After refine: {sc_best:.6f} (took {time.monotonic()-t_start:.0f}s)")

        # Also try the basic optimizer from the same start
        t_start = time.monotonic()
        alloc2 = min(200.0, (deadline - time.monotonic()) * 0.2)
        s_best2, sc_best2 = optimize(s0, time.monotonic() + alloc2, rng,
                                     use_lp=(n0 <= 384))
        if sc_best2 < best_score:
            best_score = sc_best2
            best_seq = s_best2.copy()
            print(f"  After basic: {sc_best2:.6f}")

    # Priority 2: Try various lengths with diverse initializations
    lengths = [192, 256, 320, 384, 448, 512, 640]
    if hseq is not None:
        lengths.insert(0, hseq.size)

    for n in lengths:
        if time.monotonic() > deadline - 90:
            break
        if hseq is not None and n == hseq.size:
            continue  # Already done above

        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        inits = generate_inits(n, hseq, rng)

        # Run on top 2-3 candidates
        num_cands = min(3, len(inits))
        for i in range(num_cands):
            if time.monotonic() > deadline - 30:
                break
            sc0, s0 = inits[i]
            if sc0 >= best_score and i > 0:
                continue  # Skip if already worse than best and not the best init

            time_per = max(15.0, (deadline - time.monotonic()) / (num_cands - i + 0.5))
            t_start = time.monotonic()

            # Use refined optimizer
            s_best, sc_best = optimize_refined(s0, time.monotonic() + time_per * 0.6,
                                               rng, use_lp=(n <= 384))
            if sc_best < best_score:
                best_score = sc_best
                best_seq = s_best.copy()
                print(f"  Init {i} (sc0={sc0:.4f}): new best {sc_best:.6f} "
                      f"({time.monotonic()-t_start:.0f}s)")

            # Also try basic optimizer if time remains
            if time.monotonic() < deadline - 30 and sc0 < best_score + 0.1:
                t2 = time.monotonic()
                s_best2, sc_best2 = optimize(s0, time.monotonic() + 30,
                                             rng, use_lp=(n <= 384))
                if sc_best2 < best_score:
                    best_score = sc_best2
                    best_seq = s_best2.copy()
                    print(f"  Init {i} basic: new best {sc_best2:.6f} "
                          f"({time.monotonic()-t2:.0f}s)")

    # Priority 3: Final intensive refinement on the best found
    if time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement (score={best_score:.6f}, n={len(best_seq)}) ===")
        n_best = len(best_seq)
        alloc = deadline - time.monotonic() - 5
        s_best, sc_best = optimize_refined(best_seq, time.monotonic() + alloc * 0.7,
                                           rng, use_lp=(n_best <= 384))
        if sc_best < best_score:
            best_score = sc_best
            best_seq = s_best.copy()
            print(f"  Refined: {sc_best:.6f}")

        # One more pass with basic optimizer
        if time.monotonic() < deadline - 10:
            alloc2 = deadline - time.monotonic() - 5
            s_best2, sc_best2 = optimize(best_seq, time.monotonic() + alloc2,
                                         rng, use_lp=(n_best <= 384))
            if sc_best2 < best_score:
                best_score = sc_best2
                best_seq = s_best2.copy()
                print(f"  Basic refine: {sc_best2:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(result)}")

    return result
```
