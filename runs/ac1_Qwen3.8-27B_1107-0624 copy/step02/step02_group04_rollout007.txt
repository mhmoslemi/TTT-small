"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses improved initialization (Hann/raised-cosine windows), 
adaptive soft-max gradient descent with active-set focusing,
frequency-domain perturbations for escape, and extended final polish.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _internal_score(a):

    n = a.size

    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def _make_hann_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-12)

    a = np.cos(np.pi * beta_r * u / (2.0 * beta_r)) ** 2 if beta_r > 0 else np.ones(n)
    # Actually let's use a different raised cosine

    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 - alpha * u ** 2
    a = np.maximum(a, 1e-12)

    a = np.maximum(a, 1e-12)

    a = np.exp(-t**2 / (2*sigma**2))
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

def _logit_anneal(p0, deadline, rng, lr=0.08, reg=1e-7, var_weight=0.0,
                  beta_init_frac=0.10, beta_decay=0.70,
                  no_imp_mult=4, freq_perturb_interval=500):

    Logit-space Adam with adaptive beta, optional variance penalty,
    and periodic frequency-domain perturbations for better exploration.
    """

    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    perturbation_counter = 0

    while time.monotonic() < deadline:
        L, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)

        if var_weight > 0.0:
            c_center = c[n - 1:2 * n - 1] if c.size >= n else c
            c_mean = float(c_center.mean())
            gp += var_weight * (p - p.mean())

        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)

            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
                perturbation_counter = 0
            else:
                no_imp += 1
                perturbation_counter += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.92)
                    
                    # Periodic frequency-domain perturbation
                    if perturbation_counter >= freq_perturb_interval:
                        perturbation_counter = 0
                        # Perturb in frequency domain
                        fa = np.fft.rfft(p, n)
                        freq_mask = np.ones(n // 2 + 1)
                        # Target mid-frequencies for perturbation
                        mid_start = n // 8
                        mid_end = 3 * n // 8
                        freq_mask[mid_start:mid_end] = 1.0 + 0.3 * rng.normal(size=mid_end - mid_start)
                        fa_perturbed = fa * freq_mask
                        p_perturbed = np.fft.irfft(fa_perturbed, n)
                        p_perturbed = np.maximum(p_perturbed, 1e-12)
                        p_perturbed = p_perturbed / p_perturbed.sum()
                        c_check = _full_convolve_fft(p_perturbed)
                        sc_check = float(2.0 * n * c_check.max())
                        if sc_check < best_score:
                            p = p_perturbed
                            z = np.log(p)
                            z -= z.mean()
                            c = c_check
                            maxc = float(c_check.max())
                        else:
                            # Just add small noise
                            z = best_z + 0.1 * rng.normal(size=n)
                            z = np.clip(z, -50, 50)
                            z -= z.mean()
                            p = _softmax_from_z(z)
                            c = _full_convolve_fft(p)
                            maxc = float(c.max())
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Adaptive beta decay
        if adam_t % 30 == 0:
            beta = max(1e-12, beta * beta_decay)
            if c.size >= 2:
                sorted_idx = np.argsort(c)[::-1]
                if sorted_idx[0] != sorted_idx[1]:
                    gap = c[sorted_idx[0]] - c[sorted_idx[1]]
                    if gap < beta * 0.03:
                        beta = max(1e-12, beta * 0.3)

        # Restart with noise
        no_imp_limit = max(200, no_imp_mult * n)
        if no_imp > no_imp_limit:
            if best_score < 1.51:
                noise_scale = 0.05
            elif best_score < 1.53:
                noise_scale = 0.1
            elif best_score < 1.56:
                noise_scale = 0.2
            else:
                noise_scale = 0.4
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
            perturbation_counter = 0

    return best_p, best_score


def _frequency_anneal(p0, deadline, rng):

    # Work with the amplitude spectrum
    pa = np.fft.rfft(p0, n)
    # Parameterize as log amplitude + phase
    mag = np.abs(pa) + 1e-20
    phase = np.angle(pa)
    log_mag = np.log(mag)

def _local_search(p0, deadline, rng, patch_size_range=(3, 20)):
    """
    Greedy local search: try small patches and keep improvements.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    best_p = p0.copy()
    best_score = float(2.0 * n * _full_convolve_fft(p0).max())
    no_imp = 0

    while time.monotonic() < deadline:
        ps = int(rng.integers(patch_size_range[0], patch_size_range[1] + 1))
        ps = min(ps, n // 2)
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        p_new = best_p.copy()
        p_new[start:end] += rng.normal(0, 0.02, ps)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()

        c_new = _full_convolve_fft(p_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score - 1e-15:
            best_p = p_new
            best_score = sc_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                # Try a different patch size
                no_imp = 0

    return best_p, best_score
