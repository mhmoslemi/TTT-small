```python
def fast_score(s):
    n = len(s)

    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def build_conv_matrix(s):
    """Build the matrix A such that A @ b = convolve(s, b)."""
    n = len(s)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = s
    return A

    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)

    bounds = [(0.0, None)] * n

                  options={'time_limit': 5.0, 'presolve': True})
```

```python
"""
Optimized step function search using LP-guided descent with subgradient refinement.
Key improvements over baseline: LP every iteration, parameterized shape search initialization,
FFT-based subgradient, golden-section line search, and multi-length warm-starting.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_score(s):
    """Compute 2*n*max(conv(s,s))/(sum(s))^2 efficiently."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def build_conv_matrix(s):
    """Build matrix A such that A @ b = full_convolve(s, b). Shape (2n-1, n)."""
    n = len(s)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def lp_direction(s, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(s,b)[k] <= M, b >= 0. Returns b or None."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_subgrad(s, w):
    """Compute subgradient: g[j] = 2 * sum_k w[k] * s[k-j] (s[x]=0 outside [0,n-1])."""
    n = len(s)
    h = np.convolve(w, s[::-1], mode='full')
    g = 2.0 * h[n - 1: n - 1 + n]
    return g


def subgrad_step(s, eta, beta=5.0):
    """Weighted subgradient descent step with peak emphasis."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv <= 0:
        return s.copy()
    # Soft weighting on peaks
    w = np.exp(beta * (conv - max_conv) / max_conv)
    w = w / w.sum()
    grad = compute_subgrad(s, w)
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()
    grad_norm = grad / grad_max
    s_new = s - eta * grad_norm
    s_new = np.maximum(s_new, 1e-12)
    s_new = s_new / s_new.sum()
    return s_new


def golden_line_search(s, direction, best_score, max_iters=30):
    """Golden section line search over t in [0, 1] for s + t*(direction - s)."""
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - inv_phi * (b - a)
    d = a + inv_phi * (b - a)
    s_c = (1 - c) * s + c * direction
    s_c = np.maximum(s_c, 0.0)
    sc_c = fast_score(s_c)
    s_d = (1 - d) * s + d * direction
    s_d = np.maximum(s_d, 0.0)
    sc_d = fast_score(s_d)

    for _ in range(max_iters):
        if sc_c < sc_d:
            b = d
            d = c
            sc_d = sc_c
            c = b - inv_phi * (b - a)
            s_c = (1 - c) * s + c * direction
            s_c = np.maximum(s_c, 0.0)
            sc_c = fast_score(s_c)
        else:
            a = c
            c = d
            sc_c = sc_d
            d = a + inv_phi * (b - a)
            s_d = (1 - d) * s + d * direction
            s_d = np.maximum(s_d, 0.0)
            sc_d = fast_score(s_d)

    t_best = (a + b) / 2.0
    s_best = (1 - t_best) * s + t_best * direction
    s_best = np.maximum(s_best, 0.0) / s_best.sum()
    score_best = fast_score(s_best)
    return s_best, score_best, t_best


def param_initializations(n, rng):
    """Generate diverse parameterized initial sequences. Returns list of (score, s)."""
    candidates = []
    x = np.arange(n) / max(n - 1, 1)

    # Uniform
    s = np.ones(n) / n
    candidates.append((fast_score(s), s.copy()))

    # Power law decay: s[i] = (1 - i/n)^p
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0]:
        s = (1.0 - x) ** p
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Exponential decay
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 8.0, 12.0]:
        s = np.exp(-alpha * x)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Cosine window
    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        s = np.cos(np.pi * x / 2.0) ** p
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Linear tilt
    for tilt in [0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.5, 2.0]:
        s = 1.0 + tilt * (0.5 - x)
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Combined: exp decay with cosine
    for alpha in [1.0, 2.0, 3.0]:
        for cp in [1.0, 2.0]:
            s = np.exp(-alpha * x) * (np.cos(np.pi * x / 2.0)) ** cp
            s = np.maximum(s, 0.01)
            s = s / s.sum()
            candidates.append((fast_score(s), s.copy()))

    # Sine arch
    for p in [0.5, 1.0, 2.0, 3.0]:

        candidates.append((fast_score(s), s.copy()))

        candidates.append((fast_score(s), s.copy()))

def lp_optimize(s, deadline, rng, n_max_lp=512):
    """Main LP-based optimization loop. Returns (best_s, best_score)."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    no_improve = 0
    patience = 200
    eta = 0.1
    beta = 5.0
    use_lp = (n <= n_max_lp)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        cur_score = fast_score(s)

        improved = False

        # LP step
        if use_lp:
            conv = np.convolve(s, s)
            M = conv.max()
            if M > 0:
                b = lp_direction(s, M, time_limit=5.0)
                if b is not None and b.sum() > 1e-12:
                    b = b / b.sum()
                    # Line search
                    s_new, sc_new, t_best = golden_line_search(s, b, best_score, max_iters=20)
                    if sc_new < best_score - 1e-10:
                        best_score = sc_new
                        best_s = s_new.copy()
                        s = s_new
                        no_improve = 0
                        improved = True
                    elif sc_new < cur_score:
                        s = s_new
                        no_improve += 1
                        improved = True

        # Subgradient step (if LP didn't help or as complement)
        if not improved or it % 3 == 0:
            s_sg = subgrad_step(s, eta=eta, beta=beta)
            sc_sg = fast_score(s_sg)
            if sc_sg < best_score - 1e-10:
                best_score = sc_sg
                best_s = s_sg.copy()
                s = s_sg
                no_improve = 0
            elif sc_sg < cur_score:
                s = s_sg
                no_improve += 1
            else:
                no_improve += 1

        # Adaptive parameters
        if no_improve == patience:
            eta *= 0.7
            no_improve = 0
            if eta < 0.01:
                # Perturbation restart
                noise = 0.05 * rng.standard_normal(n)
                s = s + noise
                s = np.maximum(s, 0.01)
                s = s / s.sum()
                eta = 0.1
                beta = rng.choice([3.0, 5.0, 8.0, 12.0, 20.0])
                no_improve = 0
        elif no_improve == patience // 2:
            beta = min(beta * 1.5, 50.0)

        # Gradually increase beta to focus on peaks
        if it % 500 == 0:
            beta = min(beta + 1.0, 40.0)

    return best_s, best_score

        x_new = np.linspace(0.0, 1.0, n)
        s_new = np.interp(x_new, x_old, global_best_seq)
        s_new = np.maximum(s_new, 0.0)
        s_new = s_new / s_new.sum()

        cand_deadline = time.monotonic() + 60.0
        best_s, best_sc = lp_optimize(s_new, cand_deadline, rng)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** n={n}: new best {best_sc:.6f}")
        else:
            print(f"  n={n}: best={best_sc:.6f} (no improvement)")

    # Phase 4: Final polish on the global best
    print(f"\n=== Phase 4: Final Polish, score={global_best_score:.6f} ===")
    if time.monotonic() < deadline - 10:
        s_polish, sc_polish = lp_optimize(global_best_seq, deadline - 5, rng)
        if sc_polish < global_best_score:
            global_best_score = sc_polish
            global_best_seq = s_polish.copy()
            print(f"  ** Polished: {global_best_score:.6f}")

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(final_list)}")

    return final_list
```
