import numpy as np
import time

def fast_score(s):
    """Compute 2*n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_pressure(s, beta):
    """Compute the pressure (gradient-like) vector for multiplicative updates."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv <= 0:
        return np.zeros(n)
    # Soft weighting
    w = np.exp(beta * (conv - max_conv) / max_conv)
    w = w / w.sum()
    
    # Compute gradient of weighted autocorrelation
    # grad[j] = sum_k w[k] * (s[k-j] + s[j-k]) (with valid indices)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for k in range(2*n-1):
            ki = k - j  # index in s for the "forward" term
            if 0 <= ki < n:
                val += w[k] * s[ki]
            kj = j - k  # index in s for the "backward" term
            if 0 <= kj < n:
                val += w[k] * s[kj]
        grad[j] = val
    return grad
