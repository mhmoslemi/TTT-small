def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):

    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):

                  bounds=[(0, None)] * n, method='highs')
    if res.success:

    return None

def _smooth_max_and_grad(a, beta):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    smooth_val = mval + beta * float(np.log(s))
    # Gradient: d(smooth_max)/d(a[i]) = 2 * sum_k w[k] * a[k-i]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return smooth_val, grad, w, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.001, lr * 0.95)
                top_frac = max(0.01, top_frac * 0.9)
            if no_imp > max(1000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                top_frac = 0.1
                no_imp = 0
                # Random perturbation
                if np.random.random() < 0.5:
                    rng_p = np.random.default_rng(it + 1000)
                    pert = 0.02 * rng_p.normal(size=n)
                    a = np.maximum(a + pert, 0)
                    a = _project_simplex(a)

    return best_a


def _smooth_max_polish(a, deadline, max_iter=20000, lr_start=0.03):
    """Smooth-max gradient polish with adaptive beta and momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    lr = lr_start
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, grad, _, _ = _smooth_max_and_grad(a, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        v = momentum * v - lr * grad
        a = _project_simplex(a + v)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _logit_adam_polish(a, deadline, max_iter=20000, lr=0.04):
    """Adam optimization in logit space with smooth-max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_and_grad(p, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(3000, 5 * n):
            z = 0.7 * best_z + 0.5 * np.random.default_rng(it + 999).normal(size=n) * 0.1
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric_profile(n, eta=2.0, gamma=0.5):
    """Asymmetric profile: heavily weighted toward one end."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * ((t - gamma) ** 2) + 0.5 * eta * (t - gamma)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_bump_profile(n, center1=0.25, center2=0.75, width=0.15, amp=2.0):
    """Two-bump profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.exp(-((t - center1) / width) ** 2) + amp * np.exp(-((t - center2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_ramp_profile(n, power=1.0):
    """Monotone ramp with adjustable power."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + power * t ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _refine_single(a, deadline, n_lp, n_sg, n_sm, n_ad):
    """Run a sequence of refinement methods on a starting point."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)

    # LP phase
    if n_lp > 0:
        lp_end = min(deadline, time.monotonic() + n_lp)
        a_lp = _lp_refine_v2(a, lp_end)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
        a = best_a.copy()

    # Subgradient phase
    if n_sg > 0:
        sg_end = min(deadline, time.monotonic() + n_sg)
        a_sg = _subgradient_polish(a, sg_end)
        sc_sg = _score(a_sg)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
        a = best_a.copy()

    # Smooth-max phase
    if n_sm > 0:
        sm_end = min(deadline, time.monotonic() + n_sm)
        a_sm = _smooth_max_polish(a, sm_end)
        sc_sm = _score(a_sm)
        if sc_sm < best_score:
            best_score = sc_sm
            best_a = a_sm.copy()
        a = best_a.copy()

    # Adam phase
    if n_ad > 0:
        ad_end = min(deadline, time.monotonic() + n_ad)
        a_ad = _logit_adam_polish(a, ad_end)
        sc_ad = _score(a_ad)
        if sc_ad < best_score:
            best_score = sc_ad
            best_a = a_ad.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: LP + Subgradient + Smooth-Max + Adam.
    Explores multiple lengths and diverse initializations with stochastic restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024]:
        if ln not in lengths:
            lengths.append(ln)
    lengths.sort()

    print(f"Target lengths: {lengths}")
    print(f"Budget: {budget_s}s, deadline at t={deadline - t_start:.1f}s")

    # For each length, try multiple strategies
    for n_t in lengths:
        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"\n=== Length {n_t} === (remaining: {remaining:.0f}s)")

        # Generate diverse initializations
        inits = []
        init_names = []

        if parent is not None:
            p_interp = _interpolate_profile(parent, n_t)
            inits.append(p_interp)
            init_names.append("parent_interp")
            # Perturbed parent
            for _ in range(2):
                pp = p_interp * (1.0 + 0.05 * rng.normal(size=n_t))
                pp = np.maximum(pp, 0)
                pp = pp / pp.sum()
                inits.append(pp)
                init_names.append("parent_pert")

        # Linear profiles
        for eta in [1.5, 2.0, 2.5, 3.0]:
            inits.append(_make_linear_profile(n_t, eta))
            init_names.append(f"linear_eta{eta}")

        # Asymmetric profiles
        for eta in [1.5, 2.0, 3.0]:
            for gamma in [0.2, 0.5, 0.8]:
                inits.append(_make_asymmetric_profile(n_t, eta, gamma))
                init_names.append(f"asym_eta{eta}_g{gamma}")

        # Ramp profiles
        for pw in [1.0, 2.0, 3.0]:
            inits.append(_make_ramp_profile(n_t, pw))
            init_names.append(f"ramp_pw{pw}")

        # Two-bump
        inits.append(_make_two_bump_profile(n_t, 0.3, 0.7, 0.2, 1.5))
        init_names.append("two_bump")

        # Random smooth
        for _ in range(2):
            r = rng.normal(size=n_t)
            # Smooth it
            from scipy.ndimage import gaussian_filter1d
            r = gaussian_filter1d(r, sigma=n_t // 20)
            r = np.maximum(r + 1.0, 0.1)
            r = r / r.sum()
            inits.append(r)
            init_names.append("random_smooth")

        # Allocate time per initialization
        n_inits = len(inits)
        time_per_init = max(10.0, (deadline - time.monotonic() - 60.0) / (n_inits * 1.5))

        for idx, (a0, name) in enumerate(zip(inits, init_names)):
            if time.monotonic() > deadline - 30:
                break

            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Determine refinement budget based on initial score
            if sc0 < best_score + 0.01:
                # Good start, give more time
                t_lp = min(time_per_init * 0.4, remaining * 0.3)
                t_sg = min(time_per_init * 0.2, remaining * 0.2)
                t_sm = min(time_per_init * 0.2, remaining * 0.2)
                t_ad = min(time_per_init * 0.2, remaining * 0.2)
            else:
                # Average start, give less time
                t_lp = min(time_per_init * 0.3, remaining * 0.25)
                t_sg = min(time_per_init * 0.15, remaining * 0.15)
                t_sm = min(time_per_init * 0.15, remaining * 0.15)
                t_ad = min(time_per_init * 0.15, remaining * 0.1)

            if t_lp + t_sg + t_sm + t_ad < 5:
                continue

            a_ref, sc_ref = _refine_single(
                a0, deadline,
                n_lp=t_lp, n_sg=t_sg, n_sm=t_sm, n_ad=t_ad
            )
            if np.isfinite(sc_ref) and sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"  [{name}] n={n_t}: {sc0:.8f} -> {sc_ref:.10f} (BEST)")
            else:
                print(f"  [{name}] n={n_t}: {sc0:.8f} -> {sc_ref:.8f}")

    # Final polish on the overall best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Final polish (remaining: {remaining:.0f}s) ===")
            n = best_seq.size
            # Interpolate to best length for final refinement
            a_final = best_seq.copy()

            # Try a few different final refinement strategies
            strategies = [
                ("lp", lambda a, d: _lp_refine_v2(a, d, inflate_start=1.0003, blend_start=0.2)),
                ("smooth", lambda a, d: _smooth_max_polish(a, d, lr_start=0.02)),
                ("adam", lambda a, d: _logit_adam_polish(a, d, max_iter=30000, lr=0.03)),
                ("subgrad", lambda a, d: _subgradient_polish(a, d, lr_start=0.005)),
            ]

            for strat_name, strat_fn in strategies:
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                t_strat = min(rem * 0.7, 150.0)
                print(f"  Final {strat_name} (budget {t_strat:.0f}s)...")
                a_res = strat_fn(a_final, t_start + (time.monotonic() - t_start) + t_strat)
                sc_res = _score(a_res)
                if np.isfinite(sc_res) and sc_res < best_score:
                    best_score = sc_res
                    best_seq = a_res.copy()
                    a_final = a_res.copy()
                    print(f"    {strat_name}: {sc_res:.10f} (NEW BEST)")
                else:
                    print(f"    {strat_name}: {sc_res:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
