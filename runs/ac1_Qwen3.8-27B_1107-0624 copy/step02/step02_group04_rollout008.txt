"""
Multi-resolution optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses coarse-to-fine search, adaptive soft-max with subgradient refinement,
structured perturbations, and coordinate descent final polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size + b.size - 1
    if a.size <= 32 and b.size <= 32:
        return np.convolve(a, b)
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _self_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * a.size - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * a.size - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _softmax_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_from_weights(w, p):
    """Compute the cross-correlation of w with p (gives gradient of c[k] w.r.t. p)."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score(a):
    """Compute 2n*max(conv)/sum^2 (scale invariant, so assume sum=1)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _self_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _score_unit(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _self_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval)


def _upsample(p, new_n):
    p = np.asarray(p, dtype=np.float64).ravel()
    old_n = p.size
    if old_n == new_n:
        return _normalize_nonneg(p)
    x0 = np.linspace(0.0, 1.0, old_n)
    x1 = np.linspace(0.0, 1.0, new_n)
    return _normalize_nonneg(np.interp(x1, x0, p))


def _make_profile(n, kind='hann', **kw):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == 'hanning_asym':
        slope = kw.get('slope', 1.0)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
    elif kind == 'cosine_window':
        alpha = kw.get('alpha', 0.5)
        a = np.cos(alpha * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
    elif kind == 'raised_cos':
        beta = kw.get('beta', 1.0)
        a = np.cos(np.pi * beta * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
    elif kind == 'exponential':
        rate = kw.get('rate', 2.0)
        a = np.exp(-rate * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
    elif kind == 'linear':
        slope = kw.get('slope', 1.5)
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
    elif kind == 'quadratic':
        a = 1.0 + kw.get('a2', 0.5) * (2.0 * t - 1.0) ** 2 + kw.get('a1', 0.0) * (2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
    elif kind == 'bimodal':
        center1 = kw.get('c1', 0.3)
        center2 = kw.get('c2', 0.7)
        width = kw.get('width', 0.15)
        a = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2)
        a = np.maximum(a, 1e-10)
    elif kind == 'sawtooth':
        a = (t % 0.25) / 0.25
        a = 1.0 + kw.get('amp', 0.5) * (a - 0.5)
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n)
    return _normalize_nonneg(a)


def _optimize_logit(p0, deadline, rng, lr=0.06, reg=1e-7,

                    no_imp_limit=400, use_subgrad=True, subgrad_mix=0.3):

    Logit-space Adam with adaptive soft-max and optional subgradient.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-14)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _self_convolve(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    iters = 0

    while time.monotonic() < deadline:
        iters += 1

        # Soft-max gradient
        L, w = _softmax_weights(c, beta)
        gp = 2.0 * _corr_from_weights(w, p)

        # Subgradient correction (true max direction)
        if use_subgrad and beta < 0.01 * maxc + 1e-10:
            top_idx = int(np.argmax(c))
            w_sub = np.zeros_like(c)
            w_sub[top_idx] = 1.0
            gp_sub = 2.0 * _corr_from_weights(w_sub, p)
            gp = (1.0 - subgrad_mix) * gp + subgrad_mix * gp_sub

        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _self_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-15:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        # Adaptive beta
        if adam_t % 30 == 0:
            beta = max(beta_floor, beta * beta_decay)
            # Check gap for faster convergence
            if c.size >= 2:
                sorted_idx = np.argsort(c)[::-1]
                gap = c[sorted_idx[0]] - c[sorted_idx[1]]
                if gap < beta * 0.1:
                    beta = max(beta_floor, beta * 0.3)

        # Learning rate decay on stagnation
        if no_imp > 0 and no_imp % 100 == 0:
            lr = max(0.003, lr * 0.95)

        # Restart
        if no_imp > no_imp_limit:
            noise_scale = 0.15 if best_score < 1.52 else 0.3
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _self_convolve(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _coordinate_descent(p0, deadline, rng, max_rounds=200):
    """
    Coordinate descent: for each index, try small perturbations.
    Returns (best_p, best_score).
    """
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    best_p = p.copy()
    best_score = _score_unit(p)

    for rnd in range(max_rounds):
        if time.monotonic() >= deadline:
            break
        improved = False
        indices = rng.permutation(n)
        for i in indices:
            if time.monotonic() >= deadline:
                break
            orig = p[i]
            # Try several multiplicative perturbations
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 2.0]:
                p[i] = orig * mult
                if p[i] < 0:
                    p[i] = 0.0
                s = p.sum()
                if s <= 0:
                    p[i] = orig
                    continue
                p_n = p / s
                sc = _score_unit(p_n)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_p = p_n.copy()
                    p = p_n.copy()
                    improved = True
                    break
            p[i] = orig  # revert if not improved
        if not improved and rnd > 20:
            break

    return best_p, best_score


def _patch_optimize(p0, deadline, rng, patch_size=16, n_trials=500):
    """
    Optimize patches of the sequence using local search.
    Returns (best_p, best_score).
    """
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    best_p = p.copy()
    best_score = _score_unit(p)

    for trial in range(n_trials):
        if time.monotonic() >= deadline:
            break
        ps = min(patch_size, n // 4)
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        orig_patch = p[start:end].copy()
        # Try Gaussian perturbation of the patch
        for scale in [0.1, 0.3, 0.5, 1.0]:

            p_new[start:end] = orig_patch + scale * rng.normal(size=ps)

                p = p_new.copy()
                break

def _structured_perturbations(p0, rng):
    """
    Generate structured perturbations of p0 for multi-start.
    Returns list of perturbed sequences.
    """
    p = np.asarray(p0, dtype=np.float64).ravel()
    n = p.size
    results = []

    # 1. Circular shifts
    for shift in [1, 2, n // 16, n // 8]:
        p_shift = np.roll(p, shift)
        # Re-normalize since shift breaks the structure
        p_shift = np.maximum(p_shift, 1e-14)
        results.append(p_shift / p_shift.sum())

    # 2. Mirror
    p_mirror = p[::-1].copy()
    results.append(p_mirror / p_mirror.sum())

    # 3. Compression/expansion
    for factor in [0.9, 1.1]:
        new_n = int(n * factor)
        new_n = max(64, min(2048, new_n))
        if new_n != n:
            p_comp = _upsample(p, new_n)
            if p_comp is not None and p_comp.size == n:
                results.append(p_comp)

    # 4. Add/remove a bump
    for pos in [0.25, 0.5, 0.75]:
        bump_pos = int(pos * n)
        bump_width = max(5, n // 32)
        bump = np.zeros(n)
        for i in range(n):
            d = i - bump_pos
            bump[i] = np.exp(-d * d / (2.0 * bump_width ** 2))
        for amp in [0.1, 0.3, -0.1, -0.3]:
            p_bump = p + amp * bump
            p_bump = np.maximum(p_bump, 1e-14)
            results.append(p_bump / p_bump.sum())

    # 5. Low-pass filtering
    for alpha in [0.3, 0.5]:
        kernel = np.array([1.0, alpha, 1.0, alpha, 1.0])
        kernel = kernel / kernel.sum()
        p_filt = np.convolve(p, kernel, mode='same')
        p_filt = np.maximum(p_filt, 1e-14)
        results.append(p_filt / p_filt.sum())

    # 6. Random noise (baseline)
    for scale in [0.05, 0.1, 0.2]:
        p_noise = p + scale * rng.normal(size=n)
        p_noise = np.maximum(p_noise, 1e-14)
        results.append(p_noise / p_noise.sum())

    return results

            ('lin2.0', 'linear', {'slope': 2.0}),
            ('lin4.0', 'linear', {'slope': 4.0}),
            ('bimodal', 'bimodal', {}),
        ],
    }

    for n, profs in length_profiles.items():
        for prof in profs:
            label = prof[0]
            kind = prof[1]
            kw = prof[2] if len(prof) > 2 else {}
            p = _make_profile(n, kind=kind, **kw)
            if p is not None:
                sc = _score(p)
                init_candidates.append((sc, p, n, label))

    # Add parent-based candidates
    if parent is not None:
        for n in [128, 256, 512, 1024]:
            if parent_n != n:
                p_interp = _upsample(parent, n)
                if p_interp is not None:
                    sc = _score(p_interp)
                    init_candidates.append((sc, p_interp, n, f'parent_n{n}'))
            else:
                sc = _score(parent)
                init_candidates.append((sc, parent.copy(), n, 'parent'))

    init_candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: {len(init_candidates)} candidates, best={init_candidates[0][0]:.10f}")
    update_best(init_candidates[0][1], init_candidates[0][0], "init")

    # --- Phase 1: Coarse-to-fine optimization ---
    print("Phase 1: Coarse-to-fine optimization...")
    phase1_end = min(deadline, t_start + 600.0)

    # Optimize at 128
    target_n = 128
    # Find best at 128
    best_128_p = None
    best_128_sc = np.inf
    for sc, p, n, label in init_candidates:
        if n == target_n and sc < best_128_sc:
            best_128_p = p
            best_128_sc = sc
    if best_128_p is None:
        best_128_p = _make_profile(128, 'hann')
        best_128_sc = _score(best_128_p)

    per_end = min(phase1_end - 300.0, t_start + 60.0)
    p_128, sc_128 = _optimize_logit(best_128_p, per_end, rng, lr=0.1,
                                     beta_frac=0.15, beta_decay=0.75,
                                     no_imp_limit=200)
    update_best(p_128, sc_128, f"128-opt")

    # Upsample to 256
    p_256 = _upsample(p_128, 256)
    if p_256 is not None:
        sc_256 = _score(p_256)
        # Also consider best direct 256 candidate
        for sc, p, n, label in init_candidates:
            if n == 256 and sc < sc_256:
                p_256 = p
                sc_256 = sc
        per_end = min(phase1_end - 180.0, time.monotonic() + 60.0)
        p_256, sc_256 = _optimize_logit(p_256, per_end, rng, lr=0.08,
                                         beta_frac=0.12, beta_decay=0.73,
                                         no_imp_limit=300)
        update_best(p_256, sc_256, f"256-opt")

        # Upsample to 512
        p_512 = _upsample(p_256, 512)
        if p_512 is not None:
            sc_512 = _score(p_512)
            for sc, p, n, label in init_candidates:
                if n == 512 and sc < sc_512:
                    p_512 = p
                    sc_512 = sc
            per_end = min(phase1_end - 60.0, time.monotonic() + 90.0)
            p_512, sc_512 = _optimize_logit(p_512, per_end, rng, lr=0.06,
                                             beta_frac=0.10, beta_decay=0.72,
                                             no_imp_limit=400)
            update_best(p_512, sc_512, f"512-opt")

            # Upsample to 1024
            p_1024 = _upsample(p_512, 1024)
            if p_1024 is not None:
                sc_1024 = _score(p_1024)
                for sc, p, n, label in init_candidates:
                    if n == 1024 and sc < sc_1024:
                        p_1024 = p
                        sc_1024 = sc
                # Also compare with parent
                if parent is not None and parent_n == 1024:
                    sc_parent = _score(parent)
                    if sc_parent < sc_1024:
                        p_1024 = parent.copy()
                        sc_1024 = sc_parent
                per_end = min(phase1_end, time.monotonic() + 240.0)
                p_1024, sc_1024 = _optimize_logit(p_1024, per_end, rng, lr=0.05,
                                                   beta_frac=0.08, beta_decay=0.72,
                                                   no_imp_limit=500)
                update_best(p_1024, sc_1024, f"1024-opt")

    # If we never got to 1024, go directly
    if best_seq is None or best_seq.size != 1024:
        if parent is not None:
            p_1024 = _upsample(parent, 1024) if parent_n != 1024 else parent.copy()
        else:
            p_1024 = _make_profile(1024, 'hann')
        sc_1024 = _score(p_1024)
        per_end = min(phase1_end, time.monotonic() + 120.0)
        p_1024, sc_1024 = _optimize_logit(p_1024, per_end, rng, lr=0.05,
                                           beta_frac=0.10, beta_decay=0.72,
                                           no_imp_limit=400)
        update_best(p_1024, sc_1024, f"1024-direct")

    print(f"Phase 1 done: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 2: Intensive optimization at best length ---
    print("Phase 2: Intensive optimization...")
    phase2_end = min(deadline, t_start + 1200.0)
    if best_seq is not None:
        n_best = best_seq.size
        # Multiple rounds of optimization with different parameters
        current = best_seq.copy()
        current_sc = best_score

        for round_i in range(4):
            if time.monotonic() >= phase2_end:
                break
            per_end = min(phase2_end, time.monotonic() + 100.0)

            # Vary parameters per round
            lr = 0.04 - 0.005 * round_i
            beta_frac = 0.06 - 0.01 * round_i
            beta_decay = 0.73 + 0.01 * round_i
            sg_mix = 0.2 + 0.1 * round_i

            p_new, sc_new = _optimize_logit(current, per_end, rng,
                                             lr=lr, beta_frac=beta_frac,
                                             beta_decay=beta_decay,
                                             no_imp_limit=600,
                                             use_subgrad=True,
                                             subgrad_mix=sg_mix)
            if sc_new < current_sc:
                current = p_new
                current_sc = sc_new
            update_best(current, sc_new, f"round{round_i} n={n_best}")

    # --- Phase 3: Coordinate descent ---
    print("Phase 3: Coordinate descent...")
    phase3_end = min(deadline, t_start + 1400.0)
    if best_seq is not None:
        p_cd, sc_cd = _coordinate_descent(best_seq, phase3_end, rng, max_rounds=100)
        update_best(p_cd, sc_cd, "coord_descent")

    # --- Phase 4: Patch optimization ---
    print("Phase 4: Patch optimization...")
    phase4_end = min(deadline, t_start + 1550.0)
    if best_seq is not None:
        p_patch, sc_patch = _patch_optimize(best_seq, phase4_end, rng,
                                             patch_size=16, n_trials=300)
        update_best(p_patch, sc_patch, "patch")

    # --- Phase 5: Multi-start with structured perturbations ---
    print("Phase 5: Multi-start structured perturbations...")
    phase5_end = min(deadline, t_start + 1700.0)
    if best_seq is not None:
        perturbations = _structured_perturbations(best_seq, rng)
        # Score all perturbations
        pert_candidates = []
        for i, p_pert in enumerate(perturbations):
            sc_pert = _score(p_pert)
            pert_candidates.append((sc_pert, p_pert, i))
        pert_candidates.sort(key=lambda x: x[0])

        # Optimize top few perturbations
        n_opt = min(4, len(pert_candidates))
        per_pert = max(20.0, (phase5_end - time.monotonic()) / max(1, n_opt))
        for i in range(n_opt):
            if time.monotonic() >= phase5_end:
                break
            sc_pert, p_pert, idx = pert_candidates[i]
            if sc_pert >= best_score:
                continue
            per_end = min(phase5_end, time.monotonic() + per_pert)
            p_new, sc_new = _optimize_logit(p_pert, per_end, rng,
                                             lr=0.04, beta_frac=0.05,
                                             beta_decay=0.78, no_imp_limit=300,
                                             use_subgrad=True, subgrad_mix=0.3)
            update_best(p_new, sc_new, f"pert{i}")

    # --- Phase 6: Final very careful polish ---
    print("Phase 6: Final polish...")
    if best_seq is not None:
        per_final = min(60.0, deadline - time.monotonic())
        if per_final > 5.0:
            p_fin, sc_fin = _optimize_logit(best_seq, time.monotonic() + per_final, rng,
                                             lr=0.02, beta_frac=0.03,
                                             beta_decay=0.85, no_imp_limit=500,
                                             use_subgrad=True, subgrad_mix=0.5)
            update_best(p_fin, sc_fin, "final_polish")

            # Final coordinate descent
            if time.monotonic() < deadline - 5.0:
                p_cd2, sc_cd2 = _coordinate_descent(best_seq, deadline - 2.0, rng, max_rounds=50)
                update_best(p_cd2, sc_cd2, "final_cd")

    # --- Validate and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(256, 'hann')

        # Verify with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print("WARNING: Evaluator returned inf!")
                final_seq = _make_profile(256, 'hann')
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Fallback evaluator score: {sc_eval:.10f}")
        except Exception as e:
            print(f"Evaluator error: {e}")

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(256, 'hann')
    print(f"FALLBACK: n=256")
    return [float(x) for x in fallback]
