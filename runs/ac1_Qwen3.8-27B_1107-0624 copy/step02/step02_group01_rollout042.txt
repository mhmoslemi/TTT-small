    a = softmax(z)
    c = fft_convolve(a)

    xc = c / beta
    xc = xc - xc.max()

def lse_smoothed_gradient(z, beta):
    """Gradient of LSE-smoothed max w.r.t. z."""
    a = softmax(z)
    c = fft_convolve(a)
    n = a.size
    
    # Softmax weights
    xc = c / beta
    xc = xc - xc.max()
    w = np.exp(xc)
    w = w / w.sum()
    
    # E_w[c] = dot(w, c)
    E_wc = float(np.dot(w, c))
    
    # grad_a[j] = conv(a,w)[j] + conv(a,w_rev)[2n-2-j]
    w_rev = w[::-1]
    aw = fft_convolve2(a, w)       # length n + (2n-1) - 1 = 3n-2
    awr = fft_convolve2(a, w_rev)  # length 3n-2

    grad_a = aw[:n] + awr[2*n-2 : 2*n-2 + n : -1]
    
    # grad_z = (a/beta) * (grad_a - 2*E_wc)
    grad_z = (a / beta) * (grad_a - 2.0 * E_wc)
    
    return lse_smoothed_objective(z, beta), grad_z

    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    return None

Hybrid LSE + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Phase 1: LSE-smoothed L-BFGS-B from multiple diverse starts.
Phase 2-5: LP-based direction finding with adaptive step sizes.
Explores n=256, 512, 1024 to find the best sequence.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

# Global state
_G_T_START = 0.0
_G_DEADLINE = 0.0
_G_BEST_SEQ = None
_G_BEST_SCORE = np.inf
_G_LSE_BETA = 0.1


def _setup_time(budget_s):
    global _G_T_START, _G_DEADLINE, _G_BEST_SEQ, _G_BEST_SCORE
    _G_T_START = time.monotonic()
    _G_DEADLINE = _G_T_START + max(1.0, float(budget_s) - 30.0)
    _G_BEST_SEQ = None
    _G_BEST_SCORE = np.inf


def _try_update(p, sc):
    global _G_BEST_SEQ, _G_BEST_SCORE
    if p is not None and np.isfinite(sc) and sc < _G_BEST_SCORE:
        _G_BEST_SEQ = np.asarray(p, dtype=np.float64).copy()
        _G_BEST_SCORE = float(sc)


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv_sq(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _fft_conv(a, b):
    na, nb = a.size, b.size
    if na == 0 or nb == 0:
        return np.array([])
    if na * nb <= 32768:
        return np.convolve(a, b)
    N = _next_pow2(na + nb - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:na + nb - 1]


def _compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = _fft_conv_sq(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _lse_obj_and_grad(z, beta):
    a = _softmax_fn(z)
    n = a.size
    c = _fft_conv_sq(a)
    xc = c / beta
    xc_max = float(xc.max())
    xc_sh = xc - xc_max
    log_sum = float(np.log(np.sum(np.exp(xc_sh))))
    val = beta * (xc_max + log_sum)
    w = np.exp(xc_sh)
    w = w / w.sum()
    E_wc = float(np.dot(w, c))
    w_rev = w[::-1]
    aw = _fft_conv(a, w)
    awr = _fft_conv(a, w_rev)
    grad_a = aw[:n].copy()
    idx = 2 * n - 2 - np.arange(n)
    grad_a += awr[idx]
    grad_z = (a / beta) * (grad_a - 2.0 * E_wc)
    return val, grad_z


def _lse_objective_wrapper(z):
    global _G_LSE_BETA
    z_centered = z - z.mean()
    return _lse_obj_and_grad(z_centered, _G_LSE_BETA)


def _lse_lbfgs_optimize(a0, beta_start, beta_end, beta_factor, deadline, max_iter_per_beta):

    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()
    best_a = a0.copy()
    best_sc = _compute_score(a0)
    beta = beta_start
    while beta >= beta_end and time.monotonic() < deadline:
        _G_LSE_BETA = beta
        res = minimize(_lse_objective_wrapper, z, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter_per_beta, 'gtol': 1e-8,
                                'maxfun': max_iter_per_beta * 3, 'maxls': 40})
        if res.x is not None and np.all(np.isfinite(res.x)):
            z = res.x
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        beta *= beta_factor
    return best_a, best_sc


def _solve_lp_direction(f, M, n):

            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
