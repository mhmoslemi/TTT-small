import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    """Fast score computation."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def load_fft(s, w):
    """Compute d/ds sum_k w[k]*conv[k] using FFT."""
    n = len(s)
    N = 1
    while N < 4 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    W = np.fft.rfft(w, N)
    t1 = np.fft.irfft(np.conj(S) * W, N)[:n]
    t2 = np.fft.irfft(S * W, N)[:n]
    return t1 + t2

def step_softmax(s, beta, eta):
    """Soft-max gradient step."""
    n = len(s)
    conv = np.convolve(s, s)
    mx = conv.max()
    if mx < 1e-15:
        return s.copy()
    w = np.exp(-beta * (mx - conv) / mx)  # Note: inverted from before
    # Actually, we want to emphasize HIGH conv values
    # w should be large where conv is large
    w = np.exp(beta * (conv / mx - 1.0))  # w in [exp(-beta), 1]
    w = w / w.sum()
    load = load_fft(s, w)
    load = np.maximum(load, 0)
    lm = load.max()
    if lm < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * load / lm)
    s_new = np.maximum(s_new, 1e-15)
    return s_new / s_new.sum()

def step_target(s, eta, margin=0.05):
    """Target-based gradient step."""
    n = len(s)
    conv = np.convolve(s, s)
    avg = conv.sum() / len(conv)  # = 1/(2n-1) for sum=1
    target = avg * (1 + margin)
    w = np.maximum(conv - target, 0)
    ws = w.sum()
    if ws < 1e-15:
        # All below target, use soft-max
        return step_softmax(s, 10.0, eta)
    w = w / ws
    load = load_fft(s, w)
    load = np.maximum(load, 0)
    lm = load.max()
    if lm < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * load / lm)
    s_new = np.maximum(s_new, 1e-15)
    return s_new / s_new.sum()

def spectral_step(s, alpha=0.3):
    """Flatten the spectrum."""
    n = len(s)
    N = 1
    while N < 4 * n:
        N *= 2
    S = np.fft.rfft(s, N)
    P = np.abs(S) ** 2
    Pm = P.mean()
    if Pm < 1e-15:
        return s.copy()
    # Reduce where P is high
    scale = (Pm / (P + 1e-15)) ** alpha
    S_new = S * scale
    s_new = np.fft.irfft(S_new, N)[:n]
    s_new = np.maximum(s_new, 0)
    ss = s_new.sum()
    if ss < 1e-15:
        return s.copy()
    return s_new / ss

def optimize(s0, deadline, rng):
    """Main optimization loop."""
    n = len(s0)
    s = s0.copy() / s0.sum()
    best_s = s.copy()
    best_sc = fast_score(s)
    
    eta = 0.5
    beta = 10.0
    use_target = True
    no_imp = 0
    patience = 80
    
    while time.monotonic() < deadline:
        cur_sc = fast_score(s)
        
        # Choose update method
        if use_target:
            s_new = step_target(s, eta)
        else:
            s_new = step_softmax(s, beta, eta)
        
        new_sc = fast_score(s_new)
        
        if new_sc < best_sc - 1e-10:
            best_sc = new_sc
            best_s = s_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Accept if improved over current or with some probability
        if new_sc < cur_sc:
            s = s_new
        elif no_imp > 10:
            # Try smaller step
            s_half = 0.5 * s + 0.5 * s_new
            s_half = np.maximum(s_half, 1e-15)
            s_half = s_half / s_half.sum()
            sc_half = fast_score(s_half)
            if sc_half < cur_sc:
                s = s_half
            else:
                s = s  # stay
        
        # Spectral step every 40 iterations
        if no_imp % 40 == 39:
            s_spec = spectral_step(s, alpha=0.2)
            sc_spec = fast_score(s_spec)
            if sc_spec < fast_score(s):
                s = s_spec
                if sc_spec < best_sc:
                    best_sc = sc_spec
                    best_s = s.copy()
        
        # Adaptive parameters
        if no_imp >= patience:
            if eta > 0.1:
                eta *= 0.7
            else:
                # Restart
                s = s + 0.03 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s = s / s.sum()
                eta = 0.5
                beta = rng.choice([5, 10, 15, 20])
                use_target = rng.random() < 0.5
            no_imp = 0
        elif no_imp >= patience // 2:

            use_target = True  # Switch to target-based
        
        # Occasionally switch method
        if no_imp == 40:
            use_target = not use_target
    
    return best_s, best_sc
