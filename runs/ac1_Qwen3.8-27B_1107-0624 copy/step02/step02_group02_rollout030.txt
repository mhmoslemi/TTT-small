def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def build_G_matrix(w):
    """Build the n×n matrix G where G[j,i] = w[min(i+j, 2n-2)] + w[|j-i|]."""
    n = len(w) // 2  # w has length 2n-1, so n = (len(w)+1)//2
    # Actually, w has length 2n-1
    n = (len(w) + 1) // 2
    G = np.zeros((n, n), dtype=np.float64)
    for i in range(n):
        for j in range(n):

            ij_w = w[ij] if ij <= 2 * n - 2 else 0.0
            ji = abs(i - j)
            ji_w = w[ji] if ji <= 2 * n - 2 else 0.0
            G[i, j] = ij_w + ji_w

def build_G_matrix(w):
    n = (len(w) + 1) // 2
    # Hankel part: H[j,i] = w[i+j]
    # First column of H: w[0], w[1], ..., w[n-1]
    # First row of H: w[0], w[1], ..., w[2n-2]
    h_first_col = w[:n]
    h_first_row = w[:2*n-1]
    H = hankel(h_first_col, h_first_row)
    # Toeplitz part: T[j,i] = w[|j-i|]
    # First column: w[0], w[1], ..., w[n-1]
    # First row: w[0], w[1], ..., w[n-1]
    t_col = w[:n]
    T = toeplitz(t_col, t_col)
    return H + T

G x = 1 (where G is built from the autocorrelation weights), which is faster
than the LP approach and allows more optimization iterations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz, hankel

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def build_G_matrix(w):
    """Build n×n matrix G where G[j,i] = w[min(i+j,2n-2)] + w[abs(j-i)]."""
    n = (len(w) + 1) // 2
    # Hankel part: H[j,i] = w[i+j] for i+j <= 2n-2
    h_first_col = w[:n].copy()
    h_first_row = w[:2*n-1].copy()
    try:
        H = hankel(h_first_col, h_first_row)
    except Exception:
        H = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                idx = i + j
                if idx < len(w):
                    H[i, j] = w[idx]
    # Toeplitz part: T[j,i] = w[abs(j-i)]
    t_vec = w[:n].copy()
    try:
        T = toeplitz(t_vec, t_vec)
    except Exception:
        T = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                T[i, j] = w[abs(i-j)]
    return H + T

def kkt_optimize(a_init, deadline, rng, verbose=True):
    """KKT-based optimization using the linear system G x = 1."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a_init.copy()
    a = a / s
    n = a.size
    
    best_a = a.copy()
    c = full_convolve_fft(a)
    best_M = float(c.max())
    
    beta = 20.0
    t_blend = 0.5
    stagnant = 0
    reg = 1e-6
    iteration = 0
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        M = float(c.max())
        
        # Compute weights (softmax on convolution values)
        shifted = c - M
        # Clip to avoid underflow
        shifted = np.maximum(shifted, -50.0 / beta)
        w = np.exp(beta * shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-10:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Build G and solve
        try:
            G = build_G_matrix(w)
            G_reg = G + reg * np.eye(n)
            x = np.linalg.solve(G_reg, np.ones(n))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(G + reg * np.eye(n), np.ones(n), rcond=None)[0]
            except Exception:
                x = a.copy()
        except Exception:
            x = a.copy()
        
        if not np.all(np.isfinite(x)):
            x = a.copy()
        
        # Project and normalize
        x = np.maximum(x, 0.0)
        s_x = x.sum()
        if s_x < 1e-10:
            t_blend *= 0.5
            if t_blend < 0.005:
                a = best_a + rng.normal(0, 0.02, n)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                beta = 20.0
                t_blend = 0.5
                stagnant = 0
            iteration += 1
            continue
        x = x / s_x
        
        # Blend
        a_new = (1.0 - t_blend) * a + t_blend * x
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            t_blend *= 0.5
            continue
        a_new = a_new / s_new
        
        # Evaluate
        c_new = full_convolve_fft(a_new)
        M_new = float(c_new.max())
        
        if M_new < best_M - 1e-14:
            a = a_new
            best_M = M_new
            best_a = a_new.copy()
            stagnant = 0
            # Gradually increase beta
            beta = min(beta * 1.05, 200.0)
        else:
            stagnant += 1
            t_blend *= 0.5
            if t_blend < 0.005:
                if stagnant % 20 == 0:
                    beta = max(5.0, beta * 0.7)
                t_blend = 0.3
                if stagnant > 80:
                    # Restart
                    scale = 0.01 + 0.03 * (stagnant // 100)
                    a = best_a + rng.normal(0, scale, n)
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-10:
                        a = a / s
                    stagnant = 0
                    beta = 20.0
                    t_blend = 0.5
        
        iteration += 1
        if verbose and iteration % 200 == 0:
            print(f"  KKT iter {iteration}: score={2*n*best_M:.10f}, beta={beta:.1f}, t={t_blend:.3f}")
    
    return best_a

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def logit_anneal(p0, deadline, rng, lr=0.05, reg=1e-7):
    """Soft-max Adam polish in log-space."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.1 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.005, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 100 == 0:
            beta = max(1e-14, beta * 0.85)

        if no_imp > max(500, 3 * n):
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0.copy() if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

The problem is: min M s.t. c[k] <= M for all k, sum(a) = 1, a >= 0.

Lagrangian: L = M + sum_k λ_k (c[k] - M) + ν(sum(a) - 1) - sum_j μ_j a[j]

KKT stationarity w.r.t. a[j]:
∂L/∂a[j] = sum_k λ_k * ∂c[k]/∂a[j] + ν - μ_j = 0

Now, c[k] = sum_i a[i] a[k-i] (linear convolution).
∂c[k]/∂a[j] = a[k-j] + a[j-k] (with a[i] = 0 for i outside [0,n-1])

So: sum_k λ_k (a[k-j] + a[j-k]) + ν - μ_j = 0

If a[j] > 0, then μ_j = 0, so:
sum_k λ_k (a[k-j] + a[j-k]) = -ν

Since the LHS is a linear function of a (for fixed λ), we write:
G_λ * a = -ν * 1

where G_λ[j, i] = λ[i+j] + λ[|j-i|] (as derived before).

Since ν can be any real number (it's the multiplier for an equality constraint), and we can always scale the solution, we can just solve:
G_λ * a = 1 (absorbing -ν into the normalization)

Then normalize: a = a / sum(a).

This confirms my approach. The solution of G_λ * x = 1, projected to non-negativity and normalized, gives a candidate that satisfies the KKT conditions (approximately, due to the projection).

One subtlety: the sign. We have G_λ * a = -ν * 1. If ν < 0 (which is typical for a minimization with equality constraint), then G_λ * a = |ν| * 1 > 0. So solving G_λ * x = 1 (with all positive RHS) should give a solution with positive entries (if G_λ is M-matrix-like, which it should be for non-negative λ).

"""
KKT-based + LP + Adam optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Main innovation: KKT-based solver that directly solves the linear system G x = 1
(where G is built from autocorrelation weights via Hankel + Toeplitz structure),
which is much faster than LP and allows significantly more optimization iterations.
Combined with adaptive softmax weighting, restarts, and final Adam polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz, hankel


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_G_matrix(w):
    """Build n x n matrix G where G[j,i] = w[i+j] + w[|j-i|]."""
    n = (len(w) + 1) // 2
    # Hankel part: H[j,i] = w[i+j]
    h_first_col = w[:n].copy()
    h_first_row = w[:2 * n - 1].copy()
    H = hankel(h_first_col, h_first_row)
    # Toeplitz part: T[j,i] = w[|j-i|]
    t_vec = w[:n].copy()
    T = toeplitz(t_vec, t_vec)
    return H + T


def kkt_optimize(a_init, deadline, rng, verbose=True):
    """KKT-based optimization: solve G x = 1 each iteration."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a_init.copy()
    a = a / s
    n = a.size

    best_a = a.copy()
    c = full_convolve_fft(a)
    best_M = float(c.max())

    beta = 15.0
    t_blend = 0.5
    stagnant = 0
    reg = 1e-6
    iteration = 0
    best_beta = 15.0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        M = float(c.max())

        # Compute softmax weights
        shift = c - M
        shift = np.maximum(shift / beta, -40.0 / beta)
        w = np.exp(shift)
        w_sum = float(w.sum())
        if w_sum < 1e-10:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Build G and solve linear system
        try:
            G = build_G_matrix(w)
            G_reg = G + reg * np.eye(n)
            x = np.linalg.solve(G_reg, np.ones(n))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(G + reg * np.eye(n), np.ones(n), rcond=None)[0]
            except Exception:
                x = a.copy()
        except Exception:
            x = a.copy()

        if not np.all(np.isfinite(x)):
            x = a.copy()

        # Project and normalize
        x = np.maximum(x, 0.0)
        s_x = float(x.sum())
        if s_x < 1e-10:
            t_blend *= 0.5
            if t_blend < 0.005:
                a = best_a + rng.normal(0, 0.02, n)
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a / s2
                beta = 15.0
                t_blend = 0.5
                stagnant = 0
            iteration += 1
            continue
        x = x / s_x

        # Blend with current
        a_new = (1.0 - t_blend) * a + t_blend * x
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            t_blend *= 0.5
            iteration += 1
            continue
        a_new = a_new / s_new

        # Evaluate improvement
        c_new = full_convolve_fft(a_new)
        M_new = float(c_new.max())

        if M_new < best_M - 1e-14:
            a = a_new
            best_M = M_new
            best_a = a_new.copy()
            best_beta = beta
            stagnant = 0
            beta = min(beta * 1.03, 150.0)
        else:
            stagnant += 1
            t_blend *= 0.6
            if t_blend < 0.01:
                t_blend = 0.4
                beta = max(5.0, beta * 0.75)
                if stagnant > 60:
                    scale = 0.01 + 0.02 * min(stagnant // 100, 3.0)
                    a = best_a + rng.normal(0, scale, n)
                    a = np.maximum(a, 0.0)
                    s2 = a.sum()
                    if s2 > 1e-10:
                        a = a / s2
                    stagnant = 0
                    beta = best_beta
                    t_blend = 0.5
                    reg = 1e-6

        iteration += 1
        if verbose and iteration % 200 == 0:
            print(f"  KKT iter {iteration}: score={2.0*n*best_M:.10f}, beta={beta:.1f}, t={t_blend:.3f}")

    return best_a


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=True):
    """LP-based optimization loop."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = float(np.convolve(f, f).max())

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new
        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 50 == 0:
                print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
        else:
            t *= 0.5
            if t < 0.002:
                t = 0.002
            stagnant += 1

        t = max(0.005, t * 0.998)

        if stagnant > 120:
            noise_scale = 0.1 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            stagnant = 0

    return best_f


def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p
