Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based descent, L-BFGS soft-max polishing, and coordinate descent peak shaving.

import numpy as np
import time
from scipy.optimize import linprog, minimize

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k

def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_step(f, M, secondary=False):

    A = build_conv_matrix(f)
    if secondary:

    bounds = [(0.0, None)] * n
    try:

def soft_max_value(c, beta):
    mval = float(c.max())
    if beta <= 1e-14:
        return mval
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        return mval
    return mval + beta * float(np.log(s))

def l_bfgs_polish(p0, deadline, rng, beta_init=None):
    """L-BFGS-B optimization of soft-max of conv(p,p) with p>=0, sum(p)=1."""
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    
    if beta_init is None:
        c0 = full_convolve_fft(p0)
        beta_init = 0.1 * float(c0.max())
    
    beta = beta_init
    best_p = p0.copy()
    best_score = compute_score(p0)
    
    # Work in unconstrained space: p = softmax(z)
    z = np.log(p0) - np.log(p0).mean()
    
    def objective(z):
        p = np.exp(z - z.max())
        p = p / p.sum()
        c = full_convolve_fft(p)
        return soft_max_value(c, beta)
    
    def gradient(z):
        p = np.exp(z - z.max())
        p = p / p.sum()
        c = full_convolve_fft(p)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(np.sum(ex))
        w = ex / s
        
        # d(soft_max)/dz = d(soft_max)/dp * dp/dz
        # dp/dz = p * (I - p*p^T) (Jacobian of softmax)
        # d(soft_max)/dp = 2 * C_w * p (computed above)
        gp = soft_max_grad(p, w)  # This is d(soft_max)/dp... wait, I need to redo this
        
        # Actually, my soft_max_grad function computes d(sum_k w[k]*conv[k])/dp = 2*C_w*p
        # But I need d(soft_max)/dp = sum_k (dw[k]/dp) * conv[k] + sum_k w[k] * d(conv[k])/dp
        # The first term (dw/dp) is small when beta is small, so I'll ignore it.
        # So d(soft_max)/dp ≈ sum_k w[k] * d(conv[k])/dp = 2*C_w*p = gp
        
        # Now convert to z-space: d(soft_max)/dz = (dp/dz)^T * gp = p * (gp - sum(p*gp))
        zg = p * (gp - float(np.dot(p, gp)))
        return zg
    
    # Use L-BFGS-B
    no_imp = 0
    start_beta = beta
    patience = 2000
    
    while time.monotonic() < deadline:
        try:
            res = minimize(objective, z, jac=gradient, method='L-BFGS-B', 
                          options={'maxiter': 500, 'gtol': 1e-8, 'maxls': 50})
            z_new = res.x
        except Exception:
            break
        
        p = np.exp(z_new - z_new.max())
        p = p / p.sum()
        score = compute_score(p)
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        beta = max(1e-10, beta * 0.5)
        
        if no_imp > patience or beta < 1e-12:
            # Restart with noise
            z = np.log(best_p) - np.log(best_p).mean()
            z += 0.1 * rng.normal(size=n)
            beta = start_beta * 0.5
            no_imp = 0
        
        z = z_new
    
    return best_p, best_score

def coordinate_descent_peak_shave(p0, deadline, rng, max_sweeps=100):
    """Coordinate descent: for each j, optimize a[j] to minimize the peak."""
    n = p0.size
    p = np.maximum(p0, 0.0).copy()
    s = p.sum()
    if s < 1e-10:
        return p
    p = p / s
    
    best_p = p.copy()
    best_score = compute_score(p)
    rng_order = rng.permutation(n)
    
    sweep = 0
    while time.monotonic() < deadline and sweep < max_sweeps:
        improved = False
        for idx in rng_order:
            # Try to optimize p[idx]
            save = p[idx]
            # The peak as a function of x = p[idx] is convex
            # Try golden section search
            # x in [0, 2*save+0.1] (or [0, 1] as an upper bound)
            lo = 0.0
            hi = min(1.0, 3.0 * save + 0.05)
            
            def peak_at(x):
                p[idx] = x
                # Rescale? No, keep sum fixed by adjusting... 
                # Actually, for coordinate descent, we should keep sum(p) = 1.
                # If we change p[idx] to x, the sum changes by (x - save).
                # To keep sum = 1, we'd need to adjust another element.
                # Simpler: just work with the unnormalized score.
                c = full_convolve_fft(p)
                return c.max() / (p.sum() ** 2)
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            a_ = lo + (1 - gr) * (hi - lo)
            b_ = lo + gr * (hi - lo)
            fa = peak_at(a_)
            fb = peak_at(b_)
            for _ in range(30):
                if fa < fb:
                    hi = b_
                    b_ = a_
                    fb = fa
                    a_ = lo + (1 - gr) * (hi - lo)
                    fa = peak_at(a_)
                else:
                    lo = a_
                    a_ = b_
                    fa = fb
                    b_ = lo + gr * (hi - lo)
                    fb = peak_at(b_)
            
            opt_x = (lo + hi) / 2
            p[idx] = opt_x
            
            # Check if improved
            sc = compute_score(p)
            if sc < best_score - 1e-14:
                best_score = sc
                best_p = p.copy()
                improved = True
            else:
                p[idx] = save  # revert
        
        sweep += 1
        if not improved and sweep > 5:
            break
    
    return best_p

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)

def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear
    for eta in [1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Cosine
    for k in [1, 2, 3, 5]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Quadratic (bump)
    for width in [0.5, 1.0, 1.5]:
        a = np.exp(-((t - 0.5) * 2) ** 2 / width)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Tukey-like (flat middle, tapered edges)
    for r in [0.3, 0.5, 0.7]:
        a = np.ones(n)
        edge = int(r * n / 2)
        for i in range(edge):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
            a[n-1-i] = a[i]
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(3):
        coeffs = rng.uniform(-0.3, 0.3, 4)
        a = 1.0
        for i, c in enumerate(coeffs, 1):
            a = a + c * np.cos(2 * np.pi * i * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    return profiles

    """
    Multi-phase optimization: LP descent + L-BFGS soft-max polish + coordinate descent.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    print(f"Starting search. budget={budget_s}s")
    
    # Try to get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass
    
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
