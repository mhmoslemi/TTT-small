from scipy.optimize import linprog
import time

def compute_score_from_f(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f normalized to sum 1."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))

def compute_M_from_f(f):
    """Compute max of linear convolution of f with itself."""
    c = np.convolve(f, f)
    return float(c.max())

def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s

def soft_max_and_weights(c, beta):
    """Compute soft-max value and weights."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w

def compute_gradient(f, w, n):
    """
    Compute grad_i = 2 * sum_k w[k] * f[k-i] for valid k-i.
    This is the derivative of soft-max(M) with respect to f_i (times 1, 
    since d(soft_max)/d(f_i) = sum_k w_k * dc_k/d(f_i) = sum_k w_k * 2*f_{k-i})
    """
    # g_i = 2 * sum_{k: 0<=k-i<=n-1} w[k] * f[k-i]
    # Let j = k-i, so k = i+j, j in [0, n-1], i+j in [0, 2n-2]
    # g_i = 2 * sum_{j=0}^{n-1} w[i+j] * f[j]
    # This is the cross-correlation of w and f, computed via FFT
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # We need h_i = sum_{j=0}^{n-1} w[i+j] * f[j]
    # This is (w * f_rev)[n-1 + i] where * is circular convolution
    # Let's use: h = ifft(fft(w, N) * fft(f_rev, N))[n-1 : n-1+n]
    
    f_rev = f[::-1].copy()
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f_rev, N)
    h_full = np.fft.irfft(fw * ff, N)
    h = h_full[n-1 : n-1+n]
    
    return 2.0 * h

def interpolate_profile(f, n_new):
