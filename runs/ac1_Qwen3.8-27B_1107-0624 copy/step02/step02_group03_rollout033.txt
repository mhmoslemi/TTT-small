import numpy as np
import time

def fft_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fft_cross_corr(w, a):
    """Compute sum_k w[k] * a[k-i] for i = 0, ..., len(a)-1 using FFT."""
    na = len(a)
    nw = len(w)
    size = 1 << (na + nw - 2).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(W * np.conj(A), size)[:na]

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def make_profile(n, func):
    """Create a profile from a function on [0,1]."""
    t = np.linspace(0.0, 1.0, n)
    a = np.array([func(x) for x in t])
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def adam_fft_polish(p0, deadline, lr=0.02, beta_init=None, verbose=True, seed=0):
    """
    Adam gradient descent in logit space with FFT-accelerated gradients.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    
    if beta_init is None:
        c_init = fft_autocorr(p0)
        beta_init = 0.1 * c_init.max()
    
    z = np.log(p0)
    z -= np.mean(z)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    
    c = fft_autocorr(p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    beta = beta_init
    rng = np.random.default_rng(seed)
    
    while time.monotonic() < deadline:
        it += 1
        
        c = fft_autocorr(p0)
        max_c = float(c.max())
        
        # Soft weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        
        # Gradient via cross-correlation (FFT)
        h = fft_cross_corr(w, p0)
        grad_p = 2.0 * h
        
        # Logit space gradient (centered)
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue
        
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        
        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30 or not np.isfinite(p_sum):
            z = best_z.copy()
            continue
        p = p / p_sum
        
        c_new = fft_autocorr(p)
        score = 2.0 * n * float(c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp % 200 == 0:
            beta = max(1e-10, beta * 0.7)
        
        if no_imp > 1500:
            # Restart with perturbation
            z = best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= np.mean(z)
            m.fill(0)

            it = 0

            beta = 0.1 * float(fft_autocorr(np.exp(z - z.max())).max())
            lr = max(0.005, lr * 0.8)
        
        if it % 500 == 0:
            lr = max(0.005, lr * 0.97)
        
        p0 = p
        
    return best_p, best_score

FFT-accelerated Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-resolution strategy with adaptive temperature scheduling.

import numpy as np
import time

def fft_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fft_cross_corr(w, a):
    """Compute R[i] = sum_k w[k] * a[k-i] for i=0..len(a)-1 using FFT."""
    na = len(a)
    nw = len(w)
    if na == 0:
        return np.array([])
    size = 1 << (na + nw - 2).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(W * np.conj(A), size)[:na]

def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def interpolate_seq(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    result = normalize(a_new)
    return result

    """Generate several initial profiles at size n."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []

    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:

    # Cosine-like

    # Exponential tilt
    a = np.exp(0.5 * u)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    return profiles

    """

"""
FFT-accelerated Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-resolution strategy (512→1024→2048) with adaptive temperature scheduling,
multiple initializations, and smart restarts to escape local minima.
Key improvement over prior approach: FFT-based convolution enables much larger n values
within the time budget, providing more degrees of freedom to flatten the autocorrelation.
"""

import numpy as np
import time


def fft_autocorr(a, fft_size=None):
    """Fast autocorrelation via FFT."""
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_profiles(n):
    """Generate several initial profiles at size n."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5]:
        a = 1.0 + eta * u
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Quadratic
    a = 1.0 + 1.5 * u - 0.4 * u ** 2
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    # Cosine shape
    a = 1.0 + 0.8 * np.cos(np.pi * t)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    # Exponential tilt
    a = np.exp(0.8 * u)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    # Watershed (high in middle, low at edges)
    a = 1.0 - 0.7 * np.abs(u)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    return profiles


def adam_fft_polish(p0, deadline, lr=0.02, beta_init=None, verbose=True, seed=0,
                    max_no_imp=3000, restart_perturb=0.05):
    """
    Adam gradient descent in logit space with FFT-accelerated gradients.
    Minimizes 2*n*max(conv(p,p)) subject to sum(p)=1, p>=0.
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    fft_size = 1 << (3 * n - 2).bit_length()

    # Initial autocorrelation
    fa = np.fft.rfft(p0, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]
    c = np.maximum(c, 0.0)  # remove tiny negative values from FFT
    max_c0 = float(c.max())

    if beta_init is None:
        beta_init = max(1e-6, 0.08 * max_c0)

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c0
    no_imp = 0
    beta = beta_init
    rng = np.random.default_rng(seed)
    base_lr = lr

    while time.monotonic() < deadline:
        it += 1

        # Autocorrelation (fa was computed at end of previous iteration)
        c = np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        max_c = float(c.max())

        # Soft-argmax weights
        x = (c - max_c) / beta
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30 or not np.isfinite(s_ex):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Cross-correlation gradient: grad[i] = 2 * sum_k w[k] * p[k-i]
        fw = np.fft.rfft(w, fft_size)
        h = np.fft.irfft(fw * np.conj(fa), fft_size)[:n]
        grad_p = 2.0 * h

        # Logit-space gradient (centered)
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            p0 = best_p.copy()
            fa = np.fft.rfft(p0, fft_size)
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        # Adam update
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z -= np.mean(z)

        # Compute new p
        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30 or not np.isfinite(p_sum):
            z = best_z.copy()
            p0 = best_p.copy()
            fa = np.fft.rfft(p0, fft_size)
            continue
        p = p / p_sum

        # Compute score using new FFT
        fa = np.fft.rfft(p, fft_size)
        c_new = np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]
        c_new = np.maximum(c_new, 0.0)
        score = 2.0 * n * float(c_new.max())

        if score < best_score - 1e-13:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adaptive beta: decrease to sharpen max approximation
        if no_imp > 0 and no_imp % 150 == 0:
            beta = max(1e-10, beta * 0.82)

        # Learning rate decay
        if it % 2000 == 0:
            lr = max(0.002, lr * 0.93)

        # Restart if stuck for too long
        if no_imp > max_no_imp:
            perturb_scale = restart_perturb * (1.0 + 0.5 * min(no_imp / max_no_imp, 2.0))
            z = best_z + perturb_scale * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            no_imp = 0
            lr = base_lr
            p0_tmp = np.exp(z - z.max())
            p0_tmp = p0_tmp / p0_tmp.sum()
            fa_tmp = np.fft.rfft(p0_tmp, fft_size)
            c_tmp = np.fft.irfft(fa_tmp * fa_tmp, fft_size)[:2 * n - 1]
            beta = max(1e-8, 0.04 * float(c_tmp.max()))
            p0 = p0_tmp
            fa = fa_tmp
            continue

        p0 = p

        if verbose and it % 10000 == 0:
            print(f"      iter {it}: score={best_score:.8f}, beta={beta:.2e}, lr={lr:.5f}, no_imp={no_imp}")

    return best_p, best_score

            if cand is None:
                continue
            sc = internal_score(cand)
            if sc < a3_sc:
                a3_sc = sc
                a3_best = cand

        if a3_best is not None:
            print(f"  Best init score: {a3_sc:.8f}")
            
            p_best, sc_best = adam_fft_polish(a3_best, deadline_p3, lr=0.015,
                                              seed=301, max_no_imp=4000)
            if sc_best < best_score:
                best_score = sc_best
                best_seq = p_best
                print(f"  Phase 3 result: score={best_score:.8f}")
            else:
                print(f"  Phase 3 no improvement (best={best_score:.8f})")

    # --- Final validation ---
    print(f"\n{'='*50}")
    print(f"Final: n={len(best_seq)}, internal_score={internal_score(best_seq):.8f}")
    
    # Validate with evaluate_sequence if available
    try:
        ev_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"evaluate_sequence score: {ev_score:.8f}")
    except Exception as e:
        print(f"eval check failed: {e}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        n = 256
        t = np.linspace(0, 1, n)
        a = 1 + 1.5 * (2 * t - 1)
        a = np.maximum(a, 1e-10)
        best_seq = a / a.sum()
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
