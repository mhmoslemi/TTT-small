Multi-resolution Adam optimization with LP corrections for minimizing
2*n*max(conv(a,a))/sum(a)^2.

from scipy.optimize import linprog

def internal_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def compute_gradient_fft(p, w):
    """Compute grad[i] = 2 * sum_j p[j] * w[i+j] using FFT."""
    n = len(p)
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p[::-1], N)
    c = np.fft.irfft(W * P, N)
    return 2.0 * c[n-1:2*n-1]

def make_profile(n, kind='linear', **kwargs):
    t = np.linspace(0, 1, n)
    u = 2*t - 1
    if kind == 'linear':
        eta = kwargs.get('eta', 1.5)
        a = 1.0 + eta * u
    elif kind == 'quad':
        a = 1.0 + kwargs.get('c2', 0.3) * u**2 + kwargs.get('c1', 0.5) * u
    elif kind == 'exp':
        rate = kwargs.get('rate', 2.0)
        a = np.exp(rate * t)
    elif kind == 'cos':
        a = 1.0 + kwargs.get('amp', 0.5) * np.cos(np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

def lp_direction(f, M, timeout=30.0):
    n = len(f)
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                  method='highs', options={'time_limit': timeout})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

        z -= z.mean()

                m = np.zeros(n); v = np.zeros(n)

"""
Multi-resolution Adam optimization with FFT-based gradients and LP corrections
for minimizing 2*n*max(conv(a,a))/sum(a)^2. Uses adaptive soft-max smoothing,
logit-space parameterization, and progressive dimension increase (512->1024->2048).
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def compute_gradient_fft(p, w):
    """Compute grad[i] = 2 * sum_j p[j] * w[i+j] using FFT.
    p has length n, w has length 2n-1. Returns array of length n.
    """
    n = len(p)
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p[::-1], N)
    c = np.fft.irfft(W * P, N)
    return 2.0 * c[n - 1:2 * n - 1]


def compute_gradient_direct(p, w):
    """Direct O(n^2) gradient computation for verification/small n."""
    n = len(p)
    grad = np.zeros(n)
    for i in range(n):
        length = min(n, 2 * n - 1 - i)
        if length > 0:
            grad[i] = 2.0 * np.dot(p[:length], w[i:i + length])
    return grad


def make_profile(n, kind='linear', **kwargs):
    """Generate initial profiles of various shapes."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    if kind == 'linear':
        eta = kwargs.get('eta', 1.5)
        a = 1.0 + eta * u
    elif kind == 'quad':
        c1 = kwargs.get('c1', 0.5)
        c2 = kwargs.get('c2', 0.3)
        a = 1.0 + c1 * u + c2 * u ** 2
    elif kind == 'cubic':
        c1 = kwargs.get('c1', 1.0)
        c2 = kwargs.get('c2', 0.5)
        c3 = kwargs.get('c3', 0.2)
        a = 1.0 + c1 * u + c2 * u ** 2 + c3 * u ** 3
    elif kind == 'exp':
        rate = kwargs.get('rate', 2.0)
        a = np.exp(rate * (t - 0.5))
    elif kind == 'exp_neg':
        rate = kwargs.get('rate', 2.0)
        a = np.exp(-rate * (t - 0.5))
    elif kind == 'cos':
        amp = kwargs.get('amp', 0.5)
        a = 1.0 + amp * np.cos(np.pi * t)
    elif kind == 'bump':
        width = kwargs.get('width', 0.4)
        center = kwargs.get('center', 0.3)
        a = np.exp(-((t - center) / width) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_to_n(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def lp_direction(f, M, timeout=30.0):
    """Solve LP: maximize sum(g) subject to conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def adam_phase(p0, deadline, lr=0.04, beta0=0.05, n_restarts=3,
               lp_every=0, verbose=True):
    """
    Adam gradient descent in logit space to minimize 2*n*max(conv(p,p)).
    p0 is normalized (sum=1, non-negative).
    """
    n = len(p0)
    best_p = p0.copy()
    best_score = internal_score(p0)
    rng = np.random.default_rng(777)

    remaining = max(30.0, deadline - time.monotonic())
    time_per_restart = remaining / max(1, n_restarts)

            p_start = p0.copy()
        elif restart == 1:

            p_start = p0 + noise
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:

            p_start = p0 + noise

        z = np.log(p_start)
        z -= z.mean()
        m = np.zeros(n)

            t += 1

            M = float(c.max())

            w = np.exp(x)
            ws = w.sum()
            if ws < 1e-30:

            else:
                w = w / ws

            grad += 1e-8 * p

            # Check for NaN

                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                continue

            # Adam update
            b1, b2, eps = 0.9, 0.999, 1e-8
            m = b1 * m + (1 - b1) * gz
            v = b2 * v + (1 - b2) * gz * gz
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            z = z - cur_lr * mh / (np.sqrt(vh) + eps)
            z -= z.mean()
            z = np.clip(z, -45.0, 45.0)

            # Score check
            score = 2.0 * n * M
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adaptive learning rate

                cur_lr = max(0.005, cur_lr * 0.85)

            # LP correction periodically

                if time.monotonic() < restart_dl - 20:
                    c_best = np.convolve(best_p, best_p)
                    M_best = float(c_best.max())

                                    m = np.zeros(n)
                                    v = np.zeros(n)
                                    t = 0
                                    beta = beta0
                                    no_improve = 0
                                    break

            # Reset if severely stuck
            if no_improve > 5000:
                z = np.log(p_start)
                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                beta = beta0
                cur_lr = lr
                no_improve = 0
                break

        if verbose:
            print(f"    Adam restart {restart}: best={best_score:.8f}")

    return best_p, best_score


def lp_phase(f0, deadline, max_iters=200, t_init=0.3, t_min=0.02, verbose=True):
    """LP-based iterative refinement. f0 must be normalized."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    t_step = t_init

    # Time one LP solve
    t0 = time.monotonic()
    _ = lp_direction(f, M, timeout=10)
    t_lp = time.monotonic() - t0
    if verbose:
        print(f"    LP time estimate: {t_lp*1000:.0f}ms")

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break

        g = lp_direction(f, M, timeout=min(60, (deadline - time.monotonic()) * 0.2))
        if g is None:
            break

        # Try full step
        sc_g = 2.0 * n * float(np.convolve(g, g).max())

        if sc_g < best_score * 0.998:
            f = g.copy()
            c = np.convolve(f, f)
            M = float(c.max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            no_improve = 0
        else:
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            sc_new = 2.0 * n * M_new

            if sc_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = sc_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                t_step = max(t_min, t_step * 0.99)
            else:
                no_improve += 1
                f = best_f.copy()
                M = 2.0 * n * M / (2.0 * n)  # M stays the same
                if no_improve % 30 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 300:
                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")

    return best_f, best_score
