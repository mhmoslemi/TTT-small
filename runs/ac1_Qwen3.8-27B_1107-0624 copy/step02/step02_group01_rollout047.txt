Multi-stage optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines softmax gradient descent, LP direction finding, and local search.
Uses parent sequence as warm start with exploration from alternative profiles.

import numpy as np

from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def softmax_obj_grad(f, beta):
    """Compute softmax-smoothed max convolution value and gradient."""
    n = f.size
    c = fft_convolve(f)
    M = float(c.max())
    x = (c - M) / beta
    ex = np.exp(np.clip(x, -100.0, 5.0))
    s_ex = float(ex.sum())
    if s_ex < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s_ex
    # Gradient: dL/df[i] = 2 * sum_k w[k] * dc[k]/df[i]
    # dc[k]/df[i] gives cross-correlation structure
    N = next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)
    h = np.fft.irfft(fw * np.conj(ff), N)[:n]
    grad = 2.0 * h
    return M, grad, c

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 15})
    if res.success and res.x is not None:
        return res.x
    return None

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def make_smooth_random(n, rng, n_harm=10):
    t = np.linspace(0.0, 1.0, n)

        a += amp * np.cos(2 * np.pi * k * t + phase)

    return a / a.sum()
