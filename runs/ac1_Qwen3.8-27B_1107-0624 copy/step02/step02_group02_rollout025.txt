from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def fast_lp_step(f, f_conv, K=30):
    """
    Fast LP step using only top K constraints.
    f: current sequence (normalized to sum=1)
    f_conv: convolution of f with itself
    K: number of active constraints to use
    Returns: g (direction) or None
    """
    n = len(f)
    M = float(f_conv.max())
    
    # Find top K positions where f_conv is closest to M
    # We want constraints where f_conv[k] is large (close to M)
    # But we also want to include the constraint that's actually binding
    # Use the top K values
    top_k_indices = np.argsort(f_conv)[::-1][:K]
    
    # Build the constraint matrix for these K rows
    # A[k, j] = f[j] for the appropriate shift
    # conv(f, g)[k] = sum_j f[j] * g[k-j] (for valid k-j)
    # So for a fixed k, A[k, j] = f[j] if k-j is in [0, n-1], else 0
    # i.e., A[k, j] = f[j] if 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
    
    A = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(top_k_indices):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[idx, lo:hi+1] = f[lo:hi+1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

import numpy as np

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_sparse_conv_rows(f, rows):
    """Build constraint matrix rows for given indices."""
    n = len(f)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, k in enumerate(rows):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[idx, lo:hi+1] = f[lo:hi+1]
    return A


def fast_lp_step(f, f_conv, K):
    """Fast LP step using only top K constraints."""
    n = len(f)
    M = float(f_conv.max())
    
    # Select top K indices
    if K >= 2 * n - 1:
        rows = np.arange(2 * n - 1)
    else:
        rows = np.argsort(f_conv)[::-1][:K]
    
    A = build_sparse_conv_rows(f, rows)
    c_obj = -np.ones(n)
    b_ub = np.full(len(rows), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def soft_max_and_weights(c, beta):
    """Compute soft max and weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        else:
            w[int(np.argmax(c))] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 50.0))
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def logit_adam_polish(p0, deadline, rng, lr=0.05, reg=1e-8):

    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-10, 0.15 * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        L, w = soft_max_and_weights(c, beta)
        h = cross_corr_weights(w, p)
        vp = 2.0 * h + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        
        step = lr * mh / (np.sqrt(vh) + eps)
        z = z - step
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        
        # Softmax
        z_clip = z - float(np.max(z))
        e = np.exp(z_clip)
        p = e / e.sum()

        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score) and score < best_score - 1e-13:

            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0

            no_imp += 1

            beta = max(1e-12, beta * 0.85)

            p = p / p.sum()
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.2 * maxc)
            m.fill(0.0)

def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-rate * t)
        profiles.append(a / a.sum())
        a = np.exp(rate * (t - 1.0))
        profiles.append(a / a.sum())
    
    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            a = np.exp(-((t - center) / width) ** 2)
            profiles.append(a / a.sum())
    
    # Cosine
    for k in [1, 2, 3]:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    # Triangle
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(3):
        freqs = rng.uniform(0.5, 5.0, 4)
        amps = rng.uniform(-0.3, 0.3, 4)
        a = np.ones(n)
        for fr, am in zip(freqs, amps):
            a = a + am * np.cos(2.0 * np.pi * fr * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    
    return profiles


def interpolate_profile(p, n_new):
    """Linear interpolation of a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-10 else p
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p)
    s = result.sum()
    return result / s if s > 1e-10 else result

def compute_score(a):

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf

    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def cross_corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1 using FFT."""
    n = len(p)
    N = next_pow2(3 * n)
    pf = p[::-1].copy()
    fpf = np.fft.rfft(pf, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fpf * fw, N)
    return conv[n - 1:2 * n - 1]


def build_conv_rows(f, row_indices):
    """Build constraint matrix rows for given convolution indices."""
    n = len(f)
    A = np.zeros((len(row_indices), n), dtype=np.float64)
    for idx, k in enumerate(row_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[idx, lo:hi + 1] = f[lo:hi + 1]
    return A


def fast_lp_step(f, f_conv, K):
    """Fast LP: max sum(g) s.t. conv(f,g)[k] <= M for top-K k, g >= 0."""
    n = len(f)
    M = float(f_conv.max())
    total_len = 2 * n - 1
    if K >= total_len:
        rows = np.arange(total_len)
    else:
        rows = np.argsort(f_conv)[::-1][:K]
    A = build_conv_rows(f, rows)
    c_obj = -np.ones(n)
    b_ub = np.full(len(rows), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

        else:
            w[int(np.argmax(c))] = 1.0
        return mval, w
    x = (c - mval) / beta
    x = np.minimum(x, 50.0)
    ex = np.exp(x)

    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def interpolate_profile(p, n_new):

    result = np.interp(x1, x0, p)

        p1 = np.maximum(1.0 + 1.5 * (t_arr - 0.5), 1e-10)
        p1 = p1 / p1.sum()

    sc1 = compute_score(p1)
    record(p1, sc1)
    print(f"Phase 1: n={n1}, start score={sc1:.10f}")

    lp_dl1 = min(deadline, t_start + 550.0)
    result = lp_optimize(p1, lp_dl1, rng, K_init=40, verbose=True, label="n512")
    sc = compute_score(result)
    record(result, sc)
    print(f"After main LP n=512: score={sc:.10f}")

    # Try diverse starting points at n=512
    profiles_512 = make_profiles(n1, rng)
    for i, p in enumerate(profiles_512):
        if time.monotonic() > deadline - 300:
            break
        sc_p = compute_score(p)
        if sc_p < best_score * 1.02:
            per_dl = min(deadline - 300, time.monotonic() + 25.0)
            result = lp_optimize(p, per_dl, rng, K_init=30, verbose=False, label="prof")
            sc = compute_score(result)
            record(result, sc)
            if i < 10 and sc < best_score + 0.02:
                print(f"  Profile {i}: score={sc:.10f}")

    print(f"After Phase 1 profiles: best={best_score:.10f}")

    # ===== Phase 2: Try n=1024 =====
    n2 = 1024
    if time.monotonic() < deadline - 250 and best_seq is not None:
        p2 = interpolate_profile(best_seq, n2)
        sc2 = compute_score(p2)
        record(p2, sc2)
        print(f"Phase 2: n={n2}, start score={sc2:.10f}")
        lp_dl2 = min(deadline - 200, time.monotonic() + 250.0)
        result = lp_optimize(p2, lp_dl2, rng, K_init=50, verbose=True, label="n1024")
        sc = compute_score(result)
        record(result, sc)
        print(f"After LP n=1024: score={sc:.10f}")

    # ===== Phase 3: Gradient polishing =====
    if time.monotonic() < deadline - 60 and best_seq is not None:
        npol = best_seq.size
        print(f"Phase 3: polishing n={npol}, score={best_score:.10f}")
        for lr_val in [0.06, 0.035, 0.08, 0.025]:
            if time.monotonic() > deadline - 40:
                break
            per_dl = min(deadline - 40, time.monotonic() + 55.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.003 * rng.normal(size=npol)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_adam_polish(p0, per_dl, rng, lr=lr_val)
            record(result, sc)
            print(f"  Adam lr={lr_val}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ===== Phase 4: Try n=768 and n=1536 =====
    if time.monotonic() < deadline - 120 and best_score > 1.503:
        for n_alt in [768, 1536]:
            if time.monotonic() > deadline - 80:
                break
            p_alt = interpolate_profile(best_seq, n_alt)
            sc_alt = compute_score(p_alt)
            record(p_alt, sc_alt)
            print(f"  Trying n={n_alt}: score={sc_alt:.10f}")
            lp_dl_alt = min(deadline - 60, time.monotonic() + 60.0)
            result = lp_optimize(p_alt, lp_dl_alt, rng, K_init=50, verbose=False, label=f"n{n_alt}")
            sc = compute_score(result)
            record(result, sc)
            print(f"  After LP n={n_alt}: score={sc:.10f}")

    # ===== Phase 5: Final fine polish =====
    if time.monotonic() < deadline - 3 and best_seq is not None:
        per = min(deadline - 3, time.monotonic() + 25.0)
        nfin = best_seq.size
        p0 = best_seq + 0.001 * rng.normal(size=nfin)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = logit_adam_polish(p0, per, rng, lr=0.015)
        record(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # Return best sequence
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        final = np.minimum(final, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
    else:
        t_arr = np.linspace(0, 1, 256)
        final = np.maximum(1.0 + 1.5 * (t_arr - 0.5), 1e-10)
        final = final / final.sum()

    final_score = compute_score(final)
    print(f"FINAL: n={final.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Eval check failed: {e}")

    return [float(x) for x in final]
