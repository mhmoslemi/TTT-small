def compute_gradient(a, w):
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n-1):
            # a[k-j] if 0 <= k-j < n
            if 0 <= k-j < n:
                s += w[k] * a[k-j]
            # a[j-k] if 0 <= j-k < n
            if 0 <= j-k < n:
                s += w[k] * a[j-k]
        grad[j] = s
    return grad

= 2 * sum_{m=0}^{n-1} w[m+j] * a[m] (where m = k-j, so k = m+j, and we need 0 <= m+j <= 2n-2)

For this to be valid, we need 0 <= m+j <= 2n-2, i.e., -j <= m <= 2n-2-j. Since m is in [0, n-1], the constraint is m <= 2n-2-j, which is always true for j >= 0 and m <= n-1 (since 2n-2-j >= n-1 when j <= n-1).

So: grad[j] = 2 * sum_{m=0}^{n-1} w[m+j] * a[m]

This is simply: grad = 2 * (w shifted and multiplied element-wise, summed)
= 2 * (w[n-1:2n-1+j] ... ) 

Actually, this is just a sliding dot product:
grad[j] = 2 * dot(w[j:j+n], a)

Where w has length 2n-1, and w[j:j+n] is the slice from j to j+n-1.

So the gradient computation is:
- For each j in [0, n-1]: grad[j] = 2 * dot(w[j:j+n], a)

This can be computed as a sliding dot product, which is equivalent to a correlation!

grad[j] = 2 * corr(w, a)[j] where corr is the cross-correlation.

In FFT terms:
Let N be a power of 2 >= (2n-1) + n - 1 = 3n-2.
W = FFT(w, N), A = FFT(a, N)
corr = IFFT(W * conj(A), N)
grad = 2 * corr[0:n]

Let me verify with the example:
a = [a0, a1, a2], w = [w0, w1, w2, w3, w4]
corr[j] = sum_k w[k] * a[k-j] (with a[k-j] = 0 if out of bounds)

corr[0] = w[0]*a[0] + w[1]*a[1] + w[2]*a[2] (k=0: a[0], k=1: a[1], k=2: a[2], k=3: a[3]=0, k=4: a[4]=0)
= w0*a0 + w1*a1 + w2*a2

corr[1] = w[0]*a[-1] + w[1]*a[0] + w[2]*a[1] + w[3]*a[2] + w[4]*a[3]
= 0 + w1*a0 + w2*a1 + w3*a2 + 0
= w1*a0 + w2*a1 + w3*a2

corr[2] = w[0]*a[-2] + w[1]*a[-1] + w[2]*a[0] + w[3]*a[1] + w[4]*a[2]
= 0 + 0 + w2*a0 + w3*a1 + w4*a2
= w2*a0 + w3*a1 + w4*a2

So grad = 2 * [w0*a0 + w1*a1 + w2*a2, w1*a0 + w2*a1 + w3*a2, w2*a0 + w3*a1 + w4*a2]

From the matrix:
grad[0] = 2a0*w0 + 2a1*w1 + 2a2*w2 = 2*(w0*a0 + w1*a1 + w2*a2) ✓
grad[1] = 2a0*w1 + 2a1*w2 + 2a2*w3 = 2*(w1*a0 + w2*a1 + w3*a2) ✓
grad[2] = 2a0*w2 + 2a1*w3 + 2a2*w4 = 2*(w2*a0 + w3*a1 + w4*a2) ✓

So the gradient is: grad = 2 * corr(w, a)[:n]

where corr(w, a)[j] = sum_k w[k] * a[k-j] (linear cross-correlation).

This can be computed as:
N = next_power_of_2(len(w) + len(a) - 1) = next_power_of_2(3n-2)
corr_full = ifft(fft(w, N) * conj(fft(a, N)))
grad = 2 * corr_full[:n]

This is O(N log N) which is very fast.

"""
GS (Gerchberg-Saxton) + Adam gradient optimization for minimizing
2n * max(conv(a,a)) / sum(a)^2 for non-negative sequences.

Key ideas:
1. GS algorithm for fast frequency-domain flattening
2. Smooth-max gradient with Adam for fine-tuning
3. Multiple lengths and initializations
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _gs_step(a, alpha=0.7):
    """One step of Gerchberg-Saxton for flattening autocorrelation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    
    # Pad to length N
    a_pad = np.zeros(N)
    a_pad[:n] = a
    
    # FFT
    A = np.fft.rfft(a_pad)
    
    # Compute target magnitude
    # Use geometric mean of non-DC magnitudes as target
    mags = np.abs(A)
    dc = mags[0]
    non_dc = mags[1:]
    C = float(np.mean(non_dc)) if len(non_dc) > 0 and np.any(non_dc > 0) else dc / np.sqrt(N - 1)
    
    if C < 1e-15:
        return a
    
    # Apply soft magnitude constraint
    for f in range(1, len(A)):
        if mags[f] > 0:
            A[f] = A[f] / mags[f] * (mags[f] ** (1-alpha) * C**alpha)
            # This is: A[f] *= (C/|A[f]|)^alpha
        # Also apply to the symmetric frequency
    # Handle the Nyquist frequency carefully
    # For rfft, the last element is the Nyquist frequency (no symmetric partner)
    # For other elements, we need to handle conjugate symmetry
    
    # Actually, since we're using rfft, we only have non-negative frequencies.
    # The constraint should be applied to both A[f] and A[N-f] (its conjugate).
    # But rfft only stores A[0:N/2], so we need to be careful.
    
    # For simplicity, let's work with full FFT
    # Actually, let's just use the rfft result and apply the constraint
    # The key is that |A[f]| and |A[N-f]| are equal (for real signals)
    # So modifying |A[f]| automatically handles the conjugate
    
    # Let me redo with full FFT for clarity
    A_full = np.fft.fft(a_pad)
    mags_full = np.abs(A_full)
    C = float(np.mean(mags_full[1:])) if np.any(mags_full[1:] > 0) else A_full[0] / np.sqrt(N - 1)
    
    if C < 1e-15:
        return a
    
    for f in range(1, N):
        if mags_full[f] > 1e-15:
            A_full[f] = A_full[f] / mags_full[f] * (mags_full[f] ** (1-alpha) * C**alpha)
    
    # Inverse FFT
    a_new = np.fft.ifft(A_full).real
    
    # Keep only first n elements
    a_new = a_new[:n]
    
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    
    return a_new


def _gs_optimize(a, deadline, max_iter=50000, alpha=0.7):
    """Run GS optimization until deadline or convergence."""
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):

        a_new = _gs_step(a, alpha)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break
        
        a = a_new
    
    return best_a, best_score


def _smooth_max_and_weights(c, beta):
    """Compute smooth max and weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # For numerical stability, shift by max
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):
    """Compute gradient of smooth-max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max_and_weights(c, beta)
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is 2 * cross_correlation(w, a)[j]
    N = _next_pow2(2 * n - 1 + n - 1)  # >= 3n-2
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(W * np.conj(A)).real
    grad = 2.0 * corr[:n]
    
    return grad, val


def _adam_optimize(a, deadline, max_iter=50000, lr=0.01, beta_init=None):
    """Adam optimization on the simplex (sum=1, non-negative) using log parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Work in log space
    z = np.log(a)
    z -= float(np.mean(z))  # Center for numerical stability
    
    if beta_init is None:
        c = _fft_convolve(a)
        beta_init = 0.1 * float(c.max())
    beta = beta_init
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    base_lr = lr
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient
        grad, sm_val = _gradient_smooth_max(a, beta)
        
        # Convert to log-space gradient
        # a = softmax(z) (approximately, since we're on the simplex)
        # d(log a_j)/d(a_j) = 1/a_j
        # d(score)/dz_j = d(score)/da_j * da_j/dz_j = grad_j * a_j
        # But we also need the constraint sum(a) = 1, which means
        # dz should be orthogonal to ones (in the softmax parameterization)
        gz = grad * a
        gz -= float(np.mean(gz))  # Remove mean for normalization
        
        if not np.all(np.isfinite(gz)):
            z = np.log(best_a)
            z -= float(np.mean(z))
            m, v = np.zeros(n), np.zeros(n)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        
        # Convert back
        e = np.exp(z)
        a = e / e.sum()
        
        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                # Restart
                z = np.log(best_a) + 0.5 * np.random.default_rng(it + 42).normal(size=n)
                z -= float(np.mean(z))
                e = np.exp(np.clip(z, -50, 50))
                a = e / e.sum()
                m, v = np.zeros(n), np.zeros(n)
                t = 0
                lr = base_lr
                beta = 0.1 * float(_fft_convolve(a).max())
                no_imp = 0
        
        # Decay beta
        if it % 300 == 0 and it > 0:
            beta = max(1e-10, beta * 0.7)
    
    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian_profile(n, width=0.5):
    """Gaussian-like profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    a = a / a.sum()
    return a


def _make_two_level(n, fraction=0.5):
    """Two-level sequence: alternating blocks."""
    a = np.ones(n)
    a[n//2:] = fraction
    return a / a.sum()


def _make_cosine_profile(n, k=1, amplitude=0.5):
    """Cosine profile."""
    t = np.arange(n)
    a = 1.0 + amplitude * np.cos(2 * np.pi * k * t / n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

"""
GS (Gerchberg-Saxton) + Adam gradient optimization for minimizing
2n * max(conv(a,a)) / sum(a)^2 for non-negative sequences.

Key ideas:
1. GS algorithm for fast frequency-domain autocorrelation flattening
2. Smooth-max gradient with Adam for fine-tuning in log-space
3. Multiple lengths and diverse initializations
4. Hybrid Input-Output (HIO) variant for faster GS convergence
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _gs_step(a, alpha=0.7, use_hio=False):
    """One step of Gerchberg-Saxton (or HIO) for flattening autocorrelation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)

    a_pad = np.zeros(N)
    a_pad[:n] = a

    A = np.fft.fft(a_pad)
    mags = np.abs(A)

    non_dc = mags[1:]
    C = float(np.mean(non_dc))
    if C < 1e-15:
        return a

    # Apply soft magnitude constraint for all non-DC frequencies
    A_new = A.copy()
    for f in range(1, N):
        if mags[f] > 1e-15:
            A_new[f] = A[f] / mags[f] * (mags[f] ** (1 - alpha) * C ** alpha)

    a_freq = np.fft.ifft(A_new).real

    if use_hio:
        # HIO: compute residual in the "forbidden" region
        a_projected = np.maximum(a_freq[:n], 0.0)
        s_proj = a_projected.sum()
        if s_proj > 1e-10:
            a_projected = a_projected / s_proj
        else:
            return a
        residual = a_freq[:n] - a_projected
        a_out = a_freq[:n] - residual
        a_new = np.maximum(a_out, 0.0)
    else:
        a_new = np.maximum(a_freq[:n], 0.0)

    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new


def _gs_optimize(a, deadline, max_iter=50000, alpha=0.7, use_hio=False):
    """Run GS/HIO optimization until deadline or convergence."""
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    if s > 0:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_new = _gs_step(a, alpha, use_hio)
        sc = _score(a_new)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break
        a = a_new

    return best_a, best_score


def _smooth_max_weights(c, beta):
    """Compute smooth max weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _gradient_smooth_max(a, beta):
    """Compute gradient of smooth-max of conv(a,a) w.r.t. a.
    
    Returns grad where grad[j] = d(smooth_max)/da[j].
    Formula: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j in [0,n-1])
    This equals 2 * cross_correlation(w, a)[j].
    """
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weights(c, beta)

    # Cross-correlation: corr[j] = sum_k w[k] * a[k-j]
    N = _next_pow2(3 * n)
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(W * np.conj(A)).real
    grad = 2.0 * corr[:n]

    return grad


def _adam_optimize(a, deadline, max_iter=50000, lr=0.05, beta_init=None):
    """Adam optimization in log-space for the simplex constraint."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    if beta_init is None:
        c = _fft_convolve(a)
        beta_init = max(1e-8, 0.05 * float(c.max()))
    beta = beta_init

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    base_lr = lr

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad = _gradient_smooth_max(a, beta)

        # Log-space gradient: gz[k] = a[k] * (grad[k] - dot(grad, a))
        dot_ga = float(np.dot(grad, a))
        gz = a * (grad - dot_ga)

        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_a, 1e-30))
            z -= float(np.mean(z))
            m, v = np.zeros(n), np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))

        e = np.exp(z - z.max())
        a = e / e.sum()

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(3000, 4 * n):
                # Diversify: perturb best and restart
                rng_local = np.random.default_rng(42 + it)
                z_pert = np.log(np.maximum(best_a, 1e-15)) + 0.3 * rng_local.normal(size=n)
                z_pert -= float(np.mean(z_pert))
                z_pert = np.clip(z_pert, -40, 40)
                e_p = np.exp(z_pert - z_pert.max())
                a = e_p / e_p.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                m, v = np.zeros(n), np.zeros(n)
                t = 0
                lr = base_lr
                c_tmp = _fft_convolve(a)
                beta = max(1e-8, 0.1 * float(c_tmp.max()))
                no_imp = 0

        if it % 500 == 0 and it > 0:
            beta = max(1e-10, beta * 0.65)

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profiles(n, rng):
    """Generate diverse initial profiles of length n."""
    profiles = []
    profiles.append(np.ones(n) / n)

    # Gaussian profiles
    for w in [0.2, 0.3, 0.4, 0.5]:
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * w * w))
        profiles.append(a / a.sum())

    # Cosine profiles
    for k in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            t = np.arange(n)
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t / n)
            a = np.maximum(a, 1e-15)
            profiles.append(a / a.sum())

    # Linear ramp
    t = np.linspace(0, 1, n)
    a = 0.5 + t
    profiles.append(a / a.sum())

    # Two-level
    a = np.ones(n)
    a[:n // 3] = 0.5
    a[2 * n // 3:] = 0.5
    profiles.append(a / a.sum())

    # Random
    a = np.abs(rng.normal(size=n)) + 0.1
    profiles.append(a / a.sum())

    # Sparse-ish peaks
    a = np.ones(n) * 0.1
    for _ in range(max(3, n // 32)):
        idx = rng.integers(0, n)
        a[max(0, idx - 2):idx + 3] += 1.0
    profiles.append(a / a.sum())

    return profiles

        a_best, sc_best = min(cand_list, key=lambda x: x[1])
        all_candidates.append((n_t, a_best, sc_best))

    all_candidates.sort(key=lambda x: x[2])

    for idx, (n_t, a0, sc0) in enumerate(all_candidates[:4]):
        if time.monotonic() > deadline - 120:
            break

        print(f"  Polishing candidate {idx}: n={n_t}, score={sc0:.10f}")
        ad_dl = min(time.monotonic() + 250, deadline - 120)
        a_ad, sc_ad = _adam_optimize(a0, ad_dl, max_iter=40000, lr=0.05)

        if sc_ad < best_score:
            best_score = sc_ad
            best_seq = a_ad.copy()
            print(f"    Improved to {sc_ad:.10f}")
        else:
            print(f"    No improvement ({sc_ad:.10f})")

        # Update candidate list
        all_candidates[idx] = (n_t, a_ad, sc_ad)

    # --- Phase 3: GS from polished results (cycle) ---
    print("\n=== Phase 3: GS cycle on polished results ===")
    if time.monotonic() < deadline - 100:
        for (n_t, a0, sc0) in all_candidates[:3]:
            if time.monotonic() > deadline - 100:
                break
            gs_dl = min(time.monotonic() + 60, deadline - 100)
            a_gs, sc_gs = _gs_optimize(a0, gs_dl, max_iter=10000, alpha=0.6, use_hio=True)
            if sc_gs < sc0:
                all_candidates = [(n_t, a_gs, sc_gs)] + [c for c in all_candidates if c != (n_t, a0, sc0)]
            if sc_gs < best_score:
                best_score = sc_gs
                best_seq = a_gs.copy()
                print(f"  n={n_t}: GS cycle improved to {sc_gs:.10f}")

    # --- Phase 4: Final Adam polish ---
    print("\n=== Phase 4: Final Adam polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        fp_dl = deadline - 10
        a_fp, sc_fp = _adam_optimize(best_seq, fp_dl, max_iter=30000, lr=0.02)
        if sc_fp < best_score:
            best_score = sc_fp
            best_seq = a_fp.copy()
            print(f"  Final polish: {sc_fp:.10f}")
        else:
            print(f"  Final polish: no improvement ({sc_fp:.10f})")

        # One more GS step
        if time.monotonic() < deadline - 5:
            gs_dl2 = deadline - 2
            a_gs_f, sc_gs_f = _gs_optimize(best_seq, gs_dl2, max_iter=2000, alpha=0.5, use_hio=False)
            if sc_gs_f < best_score:
                best_score = sc_gs_f
                best_seq = a_gs_f.copy()
                print(f"  Final GS: {sc_gs_f:.10f}")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256

    # Authoritative evaluation
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
