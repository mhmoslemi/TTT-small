def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_gradient(s, beta):
    """Compute gradient of log-sum-exp smoothed max autocorrelation."""
    n = len(s)
    conv = np.convolve(s, s)
    # Softmax weights
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -100, 100)
    exp_v = np.exp(shifted)
    p = exp_v / exp_v.sum()
    
    # grad[i] = 2 * sum_{k=0}^{n-1} p[i+k] * s[k] (with p[m]=0 for m >= 2n-1)
    # = 2 * (p * s_rev)[n-1 + i] where s_rev = s[::-1]
    n_fft = 1
    while n_fft < 3 * n + 1:
        n_fft *= 2
    
    s_rev = s[::-1]
    S = np.fft.rfft(s_rev, n_fft)
    P = np.fft.rfft(p, n_fft)
    conv_result = np.fft.irfft(S * P, n_fft)
    grad = 2.0 * conv_result[n-1:2*n-1]
    
    return grad, conv, p

def peak_deflation(s, n_reductions=5):
    """Directly reduce the peak of the autocorrelation."""
    n = len(s)
    s = s.copy()
    for _ in range(n_reductions):
        conv = np.convolve(s, s)
        m_star = np.argmax(conv)
        # Positions contributing to conv[m_star]: pairs (j, m_star-j)
        # Reduce s[j] for j that contribute most
        contributions = np.zeros(n)
        for j in range(max(0, m_star - n + 1), min(n, m_star + 1)):
            other = m_star - j
            if 0 <= other < n:
                contributions[j] += s[j] * s[other]
        
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break
        
        # Reduce proportionally to contribution
        reduction = 0.1 * s * (contributions / total_contrib)
        s -= reduction
        s = np.maximum(s, 0.0)
        s = s / s.sum()
    
    return s

        s[i] -= delta
        s[j] += delta
        s = np.maximum(s, 0.0) / s.sum()

            else:
                s[i] += delta
                s[j] -= delta
                s = np.maximum(s, 0.0) / s.sum()
        
        T *= cooling
    
    return best_s, best_score

import time

"""
Multi-stage optimizer minimizing 2n*max(conv(s,s))/(sum(s))^2 over non-negative
sequences. Uses log-sum-exp smoothed gradient with Nesterov momentum on the
probability simplex, peak deflation, coordinate-descent-style local moves,
simulated annealing polishing, diverse Fourier-based initialization, and
multiple sequence lengths with adaptive time allocation.
"""

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = float(np.sum(s))
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * np.max(conv) / (total * total))

def autograd_sum1(s, beta):
    """
    Compute gradient of LSE-smoothed max autocorrelation for sum(s)=1.
    Objective: (1/beta) * log(sum_k exp(beta * conv[k]))
    Returns (gradient, conv, peak_lag).
    """
    n = len(s)
    conv = np.convolve(s, s)  # length 2n-1
    mx = conv.max()
    # Numerically stable softmax
    log_w = beta * (conv - mx)
    log_w = np.clip(log_w, -700.0, 700.0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All weights underflow; fall back to one-hot at argmax
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    p = w / w_sum  # probability over lags

    # grad[i] = sum_m p[m] * d(conv[m])/d(s[i])
    # d(conv[m])/d(s[i]) = s[m-i] + s[i-m] (symmetric, with out-of-range = 0)
    #                     = 2*s[m-i] if 0<=m-i<n, else 0  (since m-i and i-m give same)
    # Wait: d(conv[m])/d(s[i]):
    #   conv[m] = sum_j s[j]*s[m-j]
    #   Term with s[i] as first factor: j=i -> s[i]*s[m-i], deriv = s[m-i] (if 0<=m-i<n)
    #   Term with s[i] as second factor: m-j=i -> j=m-i -> s[m-i]*s[i], deriv = s[m-i] (if 0<=m-i<n)
    #   If m-i == i (m=2i): only one term s[i]^2, deriv = 2*s[i]
    #   In all cases: d(conv[m])/d(s[i]) = s[m-i] + s[m-i] = 2*s[m-i] if 0<=m-i<n
    #   (When m-i=i, this gives 2*s[i] which is correct for d(s[i]^2)/d(s[i])=2*s[i])
    #
    # So: grad[i] = 2 * sum_{m: 0<=m-i<n} p[m] * s[m-i]
    #              = 2 * sum_{k=0}^{n-1} p[i+k] * s[k]  (where k=m-i, and i+k <= 2n-2)

    # Compute using FFT: grad[i] = 2 * cross_corr(p, s)[i]
    # where cross_corr(a, b)[i] = sum_k a[i+k] * b[k]
    # = sum_k a[i+k] * b[k] = (a * flip(b))[i + len(b) - 1]  (linear convolution)
    
    n_fft = 1
    while n_fft < 3 * n + 2:
        n_fft *= 2
    
    s_flip = s[::-1]
    A = np.fft.rfft(p, n_fft)
    B = np.fft.rfft(s_flip, n_fft)
    prod = A * B
    conv_p_s = np.fft.irfft(prod, n_fft)
    # cross_corr[i] = conv_p_s[i + n - 1] for i = 0..n-1
    grad = 2.0 * conv_p_s[n-1 : 2*n-1]
    
    peak_lag = int(np.argmax(conv))
    return grad, conv, peak_lag

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = float(np.sum(s))
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * np.max(conv) / (total * total))

def autograd_sum1(s, beta):
    """
    Compute gradient of LSE-smoothed max autocorrelation for sum(s)=1.
    Objective: (1/beta)*log(sum_k exp(beta * conv[k]))
    Returns (gradient, conv, peak_lag).
    """
    n = len(s)
    conv = np.convolve(s, s)  # length 2n-1
    mx = conv.max()
    log_w = beta * (conv - mx)
    log_w = np.clip(log_w, -700.0, 700.0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    p = w / w_sum

    # grad[i] = 2 * sum_{k=0}^{n-1} p[i+k] * s[k]
    # = 2 * cross_corr(p, s)[i] = 2 * (p * s_flip)[i + n - 1]
    n_fft = 1
    while n_fft < 3 * n + 2:

            s = np.maximum(s, 0.01)
            candidates.append(s / s.sum())

    tri = 1.0 - np.abs(2.0 * x - 1.0)

    # Plateau
    for frac in [0.3, 0.5, 0.7]:
