                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _adam_polish(a_init, deadline, rng, lr=0.05, beta_frac=0.01, n_max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-10, beta * 0.85)

        if t % 5000 == 0:
            if best_score < 1.51:
                lr = max(0.005, lr * 0.95)

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase L-BFGS with continuation for minimizing autocorrelation peak.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a0 = parent.copy()
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 512
        a0 = _make_linear_profile(n, 1.5)
        print(f"No parent, using linear profile n={n}")

    # --- Phase 0: Evaluate initial candidates ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
        update_best(parent, sc_parent, "parent")

    # Try a few deterministic profiles
    for sn in [256, 512, 1024]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_linear_profile(sn, 1.0),
            _make_linear_profile(sn, 1.5),
            _make_linear_profile(sn, 2.0),
            _make_quadratic_profile(sn, 1.0, 0.3),
            _make_quadratic_profile(sn, 1.5, -0.2),
            _make_cubic_profile(sn, 1.0, -0.5),
        ]
        for idx, p in enumerate(profiles):
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"prof{idx}_n{sn}"))
            update_best(p, sc, f"prof{idx}_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best={best_score:.10f}")

    # --- Phase 1: L-BFGS continuation on top candidates ---
    phase1_end = min(deadline, t_start + 400.0)
    count = 0
    seen = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 6:
            break
        if sc > best_score * 1.05:
            break
        key = (pn, label[:8])
        if key in seen:
            continue
        seen.add(key)

        per_time = min(60.0, (phase1_end - time.monotonic()))
        p_best, sc_best = _run_lbfgs_continuation(p, t_start + per_time + time.monotonic() - time.monotonic() + per_time, rng, max_wall_time=per_time)
        # Fix: compute deadline properly
        level_deadline = min(phase1_end, time.monotonic() + per_time)
        p_best, sc_best = _run_lbfgs_continuation(p, level_deadline, rng, max_wall_time=per_time)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 2: Multi-restart L-BFGS on best ---
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(80.0, (phase2_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase2_end - 10:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.1 + 0.05 * i
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase2_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _run_lbfgs_continuation(p_start, per_dl, rng, max_wall_time=per_restart)
            update_best(p_best, sc_best, f"restart{i}_n{n_best}")

        print(f"Phase 2: best={best_score:.10f}")

    # --- Phase 3: Adam polish on best ---
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 5, time.monotonic() + 200.0)
        p_best, sc_best = _adam_polish(best_seq, per_dl, rng, lr=0.03, beta_frac=0.005, n_max_iter=30000)
        update_best(p_best, sc_best, "adam_polish")
        print(f"Phase 3: best={best_score:.10f}")

    # --- Phase 4: Try larger length ---
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr < 2048:
            target_n = min(2048, n_curr * 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            per_dl = min(phase4_end - 20, time.monotonic() + 180.0)
            p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl, rng, max_wall_time=180.0)
            update_best(p_best, sc_best, f"lbfgs_n{target_n}")

            if time.monotonic() < phase4_end - 30:
                per_dl2 = min(phase4_end - 10, time.monotonic() + 100.0)
                p_best2, sc_best2 = _adam_polish(best_seq, per_dl2, rng, lr=0.02, beta_frac=0.003, n_max_iter=20000)
                update_best(p_best2, sc_best2, f"adam_n{target_n}")

        print(f"Phase 4: best={best_score:.10f}")

    # --- Phase 5: Final fine polish ---
    phase5_end = deadline
    if time.monotonic() < phase5_end and best_seq is not None:
        per_dl = min(phase5_end - 5, time.monotonic() + 60.0)
        p_best, sc_best = _run_lbfgs_continuation(best_seq, per_dl, rng, max_wall_time=55.0)
        update_best(p_best, sc_best, "final_lbfgs")

        if time.monotonic() < phase5_end - 5:
            per_dl2 = min(phase5_end - 3, time.monotonic() + 30.0)
            p_best2, sc_best2 = _adam_polish(best_seq, per_dl2, rng, lr=0.01, beta_frac=0.002, n_max_iter=10000)
            update_best(p_best2, sc_best2, "final_adam")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Try evaluate_sequence for authoritative score
        if len(final_seq) <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
