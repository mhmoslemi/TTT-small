import numpy as np
import time
from scipy.optimize import linprog

"""
Multi-scale autocorrelation flattening with Nesterov momentum,
FFT-based pressure computation, diverse arch-shaped initializations,
and LP-guided refinement for minimizing 2n*max(conv(s,s))/(sum(s))^2.
"""

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def compute_pressure_fft(s, excess):
    n = len(s)
    m = len(excess)
    N = 1
    while N < m + n:
        N *= 2
    S = np.fft.fft(s, N)
    E = np.fft.fft(excess, N)
    term1 = np.fft.ifft(np.conj(E) * S, N).real[:n]
    term2 = np.fft.ifft(E * S, N).real[:n]
    return np.maximum(term1 + term2, 0.0)

def generate_arch_initializations(n, rng):
    results = []
    
    # Parabolic arch: high at edges
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = 1.0 + a * (2.0 * x - 1.0) ** 2
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())
    
    # Cosine arch
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = 1.0 + 0.7 * np.cos(freq * np.pi * (2.0 * x - 1.0)) ** 2
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())
    
    # Power-law arch: s[i] proportional to (i+1)^(-p) * (n-1-i+1)^(-p)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5]:
        x = np.arange(n) + 1
        s = x ** (-p) * (n + 1 - x) ** (-p)
        s = s / s.sum()
        results.append(s.copy())
    
    # Uniform
    s = np.ones(n) / n
    results.append(s.copy())
    
    # Two-hump: high in middle, low at edges (opposite of arch)
    for width in [0.3, 0.5, 0.7]:
        x = np.arange(n) / max(n - 1, 1)
        s = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())
    
    # Random with arch bias
    for _ in range(5):
        x = np.arange(n) / max(n - 1, 1)
        arch = 1.0 + rng.uniform(0.5, 2.0) * (2.0 * x - 1.0) ** 2
        s = arch * rng.random(n)
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())
    
    # Pure random
    for _ in range(3):
        s = rng.random(n) + 0.1
        s = s / s.sum()
        results.append(s.copy())
    
    return results

def lp_refine(s, time_limit=3.0):
    """LP: find direction b that minimizes max conv(s,b)."""
    n = len(s)
    conv_s = np.convolve(s, s)
    M = conv_s.max()
    
    # Build constraint matrix (sparse would be better but let's try dense for small n)
    if n > 200:
        return None
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b / b.sum()
    except Exception:
        pass
    return None

def optimize_sequence(s0, deadline, rng, n, use_lp=True):

    total = s.sum()
    if total < 1e-15:
        s = np.ones(n) / n
    else:
        s = s / total
    
    best_s = s.copy()
    best_score = fast_score(s)
    
    eta = 0.4
    gamma = 1.3
    momentum = 0.5
    v = np.zeros(n)
    it = 0
    no_improve = 0
    patience = 80
    
    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(s, s)
        max_conv = conv.max()
        mean_conv = (s.sum() ** 2) / (2 * n - 1)  # This is the theoretical mean
        
        # Target: above mean by gamma factor
        target = gamma * mean_conv
        excess = np.maximum(conv - target, 0.0)
        
        if excess.sum() < 1e-18:

        else:
            pressure = compute_pressure_fft(s, excess)
            max_p = pressure.max()
            if max_p < 1e-18:

            # Nesterov step
            v = momentum * v - eta * pressure / max_p
            s_new = s + v
            s_new = np.maximum(s_new, 1e-10)

            score_new = fast_score(s_new)

            elif score_new < 2.0:  # Accept if better than uniform

            else:

import numpy as np
import time
from scipy.optimize import linprog

"""
Multi-scale autocorrelation flattening with Nesterov momentum, FFT-based pressure
computation, diverse arch-shaped initializations, and LP-guided refinement.
Minimizes 2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def compute_pressure_fft(s, excess):
    """
    Compute the pressure on each element of s from excess autocorrelation.
    pressure[j] = sum_k excess[k]*(s[k-j] + s[j-k]) over valid indices.
    Uses FFT for O(n log n) computation.
    """
    n = len(s)
    m = len(excess)
    N = 1
    while N < m + n:
        N *= 2
    S = np.fft.fft(s, N)
    E = np.fft.fft(excess, N)
    # Cross-correlation: term1[j] = sum_k excess[k] * s[k-j]
    term1 = np.fft.ifft(np.conj(E) * S, N).real[:n]
    # Convolution: term2[j] = sum_k excess[k] * s[j-k]
    term2 = np.fft.ifft(E * S, N).real[:n]
    return np.maximum(term1 + term2, 0.0)


def lp_refine(s, time_limit=3.0):
    """Solve LP to find direction b maximizing sum(b) s.t. conv(s,b) <= M, b>=0."""
    n = len(s)
    if n > 200:
        return None
    conv_s = np.convolve(s, s)
    M = conv_s.max()
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-12:
                return b / b.sum()
    except Exception:
        pass
    return None


def generate_initializations(n, rng, hseq=None):
    """Generate diverse initial sequences with various shapes."""
    results = []
    x = np.arange(n) / max(n - 1, 1)

    # Parabolic arch: high at edges, low in middle
    for a in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]:
        s = 1.0 + a * (2.0 * x - 1.0) ** 2
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())

    # Cosine-based arches
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        for amp in [0.3, 0.7, 1.0]:
            s = 1.0 + amp * np.cos(freq * np.pi * (2.0 * x - 1.0)) ** 2
            s = np.maximum(s, 0.01)
            s = s / s.sum()
            results.append(s.copy())

    # Power-law arch (Bell-shaped at edges)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        xn = np.arange(n, 1, -1, dtype=np.float64)
        xp = np.arange(1, n + 1, dtype=np.float64)
        s = (xp ** (-p)) * (xn ** (-p))
        s = s / s.sum()
        results.append(s.copy())

    # Uniform baseline
    s = np.ones(n) / n
    results.append(s.copy())

    # Hyperbolic secant shape
    for width in [0.2, 0.4, 0.6, 0.8]:
        s = 1.0 / np.cosh(3.0 * (2.0 * x - 1.0) / width)
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())

    # Two-hump (high in middle) - sometimes helps as perturbation
    for width in [0.2, 0.4, 0.6]:
        s = np.exp(-((x - 0.5) ** 2) / (2.0 * width ** 2))
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())

    # Random with arch bias
    for _ in range(5):
        arch = 1.0 + rng.uniform(0.5, 2.5) * (2.0 * x - 1.0) ** 2
        s = arch * (rng.random(n) + 0.1)
        s = s / s.sum()
        results.append(s.copy())

    # Pure random
    for _ in range(3):
        s = rng.random(n) + 0.1
        s = s / s.sum()
        results.append(s.copy())

    # Resized height_sequence_1
    if hseq is not None:
        x_old = np.linspace(0.0, 1.0, hseq.size)
        x_new = np.linspace(0.0, 1.0, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 0.0)
        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            results.insert(0, h_r)
        if hseq.size == n:
            results.insert(0, hseq / hseq.sum())

    return results


def optimize_sequence(s0, deadline, rng, use_lp=True):
    """
    Optimize a single sequence using FFT-based pressure with Nesterov momentum.
    Returns (best_s, best_score).
    """
    n = len(s0)
    s = np.maximum(s0, 0).astype(np.float64)
    total = s.sum()
    if total < 1e-15:
        s = np.ones(n) / n
    else:
        s = s / total

    best_s = s.copy()
    best_score = fast_score(s)

    eta = 0.4
    gamma = 1.0
    momentum = 0.4
    v = np.zeros(n)
    it = 0
    no_improve = 0
    patience = 60

    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(s, s)
        max_conv = conv.max()
        mean_conv = (s.sum() ** 2) / (2.0 * n - 1.0)

        # Compute excess above target level
        target = gamma * mean_conv
        excess = np.maximum(conv - target, 0.0)

        if excess.sum() < 1e-18:
            # Already flat enough at this target; be more aggressive
            if gamma > 0.3:
                gamma *= 0.9
            no_improve += 1
        else:
            pressure = compute_pressure_fft(s, excess)
            max_p = pressure.max()
            if max_p < 1e-18:
                break

            # Nesterov momentum step
            v = momentum * v - eta * pressure / max_p
            s_new = s + v
            s_new = np.maximum(s_new, 1e-10)
            s_new_sum = s_new.sum()
            if s_new_sum < 1e-15:
                s_new = s.copy()
            else:
                s_new = s_new / s_new_sum

            score_new = fast_score(s_new)

            if score_new < best_score - 1e-12:
                best_score = score_new
                best_s = s_new.copy()
                no_improve = 0
                s = s_new
            elif score_new < 1.8:
                s = s_new
                no_improve += 1
            else:
                # Rejected step
                momentum = max(momentum * 0.95, 0.1)
                no_improve += 1
                v *= 0.5

        # Adaptive adjustments when stuck
        if no_improve >= patience:
            if eta > 0.05:
                eta *= 0.65
            elif gamma < 0.3:
                # Switch to peak-only subgradient mode
                k_star = np.argmax(conv)
                # Direct subgradient on peak
                grad = np.zeros(n)
                for j in range(n):
                    if 0 <= k_star - j < n:
                        grad[j] += s[k_star - j]
                    if 0 <= j - k_star < n:
                        grad[j] += s[j - k_star]
                grad = np.maximum(grad, 0)
                if grad.max() > 0:
                    s = s - 0.1 * grad / grad.max()
                    s = np.maximum(s, 1e-10)
                    s = s / s.sum()
                gamma = 0.5
                eta = 0.3
            else:
                # Full restart with perturbation around best
                s = best_s + 0.03 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s = s / s.sum()
                v = np.zeros(n)
                eta = 0.4
                momentum = 0.4
                gamma = 1.0
            no_improve = 0

        # Periodic LP refinement for small n
        if use_lp and n <= 200 and it % 400 == 0:
            b = lp_refine(s, time_limit=2.0)
            if b is not None:
                best_t = 0.0
                best_ts = fast_score(s)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    cand = (1.0 - t) * s + t * b
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    cand = cand / cs
                    ts = fast_score(cand)
                    if ts < best_ts:
                        best_ts = ts
                        best_t = t
                if best_t > 0 and best_ts < best_score - 1e-12:
                    best_score = best_ts
                    best_s = ((1.0 - best_t) * s + best_t * b)
                    best_s = np.maximum(best_s, 0) / best_s.sum()
                    s = best_s.copy()
                    v = np.zeros(n)
                    no_improve = 0

        # Slowly decrease gamma to be more aggressive over time
        if it % 1000 == 0:
            gamma = max(gamma * 0.95, 0.2)

    return best_s, best_score
