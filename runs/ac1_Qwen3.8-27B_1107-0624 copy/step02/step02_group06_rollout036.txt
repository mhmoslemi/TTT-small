def lp_solve_min_peak(f):
    """
    Find b >= 0, sum(b) = 1, minimizing max_k (f*b)_k.
    Returns (b, M_opt).
    """
    n = len(f)
    A = build_conv_matrix(f)  # (2n-1) x n
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Constraints:
    # (f*b)_k - M <= 0 for k = 0,...,2n-2  =>  A@b - M <= 0
    A_ext = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub_ext = np.zeros(2*n-1)
    
    # sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ext, b_ub=b_ub_ext, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[-1]

1. Given f (sum=1), solve the LP to get b* and M*.
2. Check: is max(b**2) < M*? If yes, b* is better than f.
3. Update: f = (1-α)f + αb*.
4. Repeat.

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal "best response" directions,
combined with multi-scale search and diverse initialization.
"""
import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A where (A @ b)_k = (f * b)_k for full linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_find_best_response(f, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k.
    Returns optimal b or None if infeasible.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_min_peak(f):
    """
    Solve LP: min M s.t. b >= 0, sum(b) = 1, (f*b)_k <= M for all k.
    Returns (b, M_opt) or (None, None).
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ b - M <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[-1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    m = c.max()
    if not np.isfinite(m):
        return np.inf
    return float(2 * n * m / (s * s))


def lp_iteration(f0, alpha, deadline, max_iters=5000):
    """
    Run LP-based iteration to minimize the score.
    f0: initial sequence (will be normalized)
    alpha: mixing parameter
    deadline: time limit (monotonic)
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f0, compute_score(f0)
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score(f)
    no_imp = 0
    M = float(np.convolve(f, f).max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP to find best response
        A = build_conv_matrix(f)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.status != 0:
            # Try with slightly larger M
            M *= 1.01
            continue
        
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb < 1e-10:
            break
        b = b / sb
        
        # Try direct assignment and damped update
        improved = False
        for a_try in [1.0, alpha, 0.5, 0.3, 0.1]:
            f_try = (1.0 - a_try) * f + a_try * b
            f_try = np.maximum(f_try, 0.0)
            st = f_try.sum()
            if st < 1e-10:
                continue
            f_try = f_try / st
            sc = compute_score(f_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f_try.copy()
                f = f_try
                M = float(np.convolve(f, f).max())
                no_imp = 0
                improved = True
                break

            no_imp += 1
            if no_imp > 150:
                break
    
    return best_f, best_score

def diverse_inits(n, rng, count=20):
    """Generate diverse initial sequences of length n."""
    t = np.linspace(0, 1, n)
    inits = []
    
    # Exponential ramps
    for eta in [1, 2, 3, 5, 8, -1, -2, -3, -5]:
        a = np.exp(eta * t)
        inits.append(a / a.sum())
    
    # Power laws
    for p in [0.3, 0.5, 1, 2, 3]:
        a = t**p + 0.001
        inits.append(a / a.sum())
        a = (1-t)**p + 0.001
        inits.append(a / a.sum())
    
    # Sine-based
    for p in [1, 2, 3, 4]:
        a = np.sin(np.pi * t)**p + 0.001
        inits.append(a / a.sum())
    
    # Gaussian-like (asymmetric)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - c) / sigma)**2) + 0.001
            inits.append(a / a.sum())
    
    # Random smooth
    for _ in range(count):
        coeffs = rng.uniform(-4, 4, 5)
        a = np.exp(coeffs[0] + coeffs[1]*t + coeffs[2]*t**2 + coeffs[3]*t**3 + coeffs[4]*t**4)
        inits.append(a / a.sum())
    
    return inits


def interpolate_to(a, n_new):
    """Interpolate sequence a to length n_new."""
    n = len(a)
    if n == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def coordinate_polish(a, deadline, rng, max_sweeps=50):
    """Simple coordinate descent polish."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    best = a.copy()
    best_sc = compute_score(a)
    no_imp = 0
    
    for sweep in range(max_sweps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            for factor in [0.9, 0.95, 1.05, 1.1, 0.8, 1.2]:
                a[i] = orig * factor
                if a[i] < 0:
                    a[i] = orig
                    continue
                s = a.sum()
                if s < 0.01:
                    a[i] = orig
                    continue
                sc = compute_score(a / s)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best = (a / s).copy()
                    a = best.copy()
                    improved = True
                    break
            a[i] = best[i] if improved else orig
            a = a / a.sum()
        if not improved:
            no_imp += 1
            if no_imp > 5:
                break
        else:
            no_imp = 0
    
    return best, best_sc

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses linear programming to find optimal "best response" directions at multiple scales,
combined with diverse initialization and coordinate descent polish.
"""
import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build dense matrix A where (A @ b)_k = (f * b)_k for full linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def lp_iteration(f0, alpha, deadline, max_iters=5000):
    """
    LP-based iteration: at each step, solve LP to find best response b to current f,
    then update f toward b. Minimizes max(conv(f,f))/sum(f)^2.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = float(f.sum())
    if s < 1e-10:
        return f0, compute_score(f0)
    f = f / s

    best_f = f.copy()
    best_score = compute_score(f)
    no_imp = 0
    M = float(np.convolve(f, f).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = build_conv_matrix(f)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if res.status != 0:
            M *= 1.05
            no_imp += 50
            if no_imp > 300:
                break
            continue

        b = np.maximum(res.x, 0.0)
        sb = float(b.sum())
        if sb < 1e-10:
            break
        b = b / sb

        improved = False
        for a_try in [1.0, alpha, 0.5, 0.3, 0.15, 0.05]:
            f_try = (1.0 - a_try) * f + a_try * b
            f_try = np.maximum(f_try, 0.0)
            st = float(f_try.sum())
            if st < 1e-10:
                continue
            f_try = f_try / st
            sc = compute_score(f_try)
            if sc < best_score - 1e-13:
                best_score = sc
                best_f = f_try.copy()
                f = f_try
                M = float(np.convolve(f, f).max())
                no_imp = 0
                improved = True
                break

        if not improved:
            no_imp += 1
            if no_imp > 200:
                break

    return best_f, best_score


def lp_iteration_minpeak(f0, deadline, max_iters=3000):
    """
    LP iteration using min-peak formulation:
    min M s.t. b>=0, sum(b)=1, (f*b)_k <= M for all k.
    Then update f toward optimal b.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = float(f.sum())
    if s < 1e-10:
        return f0, compute_score(f0)
    f = f / s

    best_f = f.copy()
    best_score = compute_score(f)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = build_conv_matrix(f)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=np.float64)])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break

        b = np.maximum(res.x[:n], 0.0)
        sb = float(b.sum())
        if sb < 1e-10:
            break
        b = b / sb

        sc_b = compute_score(b)
        if sc_b < best_score - 1e-13:
            best_score = sc_b
            best_f = b.copy()
            f = b
            no_imp = 0
            continue

        improved = False
        for a_try in [0.5, 0.3, 0.15, 0.1]:
            f_try = (1.0 - a_try) * f + a_try * b
            f_try = np.maximum(f_try, 0.0)
            st = float(f_try.sum())
            if st < 1e-10:
                continue
            f_try = f_try / st
            sc = compute_score(f_try)
            if sc < best_score - 1e-13:
                best_score = sc
                best_f = f_try.copy()
                f = f_try
                no_imp = 0
                improved = True
                break

        if not improved:
            no_imp += 1
            if no_imp > 150:
                break

    return best_f, best_score


def diverse_inits(n, rng, count=20):
    """Generate diverse initial sequences of length n."""
    t = np.linspace(0, 1, n)
    inits = []

    # Exponential ramps (both directions)
    for eta in [1, 2, 3, 5, 8, 12, -1, -2, -3, -5, -8]:
        a = np.exp(eta * t)
        inits.append(a / a.sum())

    # Power laws
    for p in [0.3, 0.5, 0.7, 1, 1.5, 2, 3, 4]:
        a = t**p + 0.001
        inits.append(a / a.sum())
        a = (1 - t)**p + 0.001
        inits.append(a / a.sum())

    # Sine-based
    for p in [1, 2, 3, 4, 6]:
        a = np.sin(np.pi * t)**p + 0.001
        inits.append(a / a.sum())

    # Asymmetric Gaussians
    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.08, 0.15, 0.25, 0.4]:
            a = np.exp(-((t - c) / sigma)**2) + 0.001
            inits.append(a / a.sum())

    # Product of exponentials (skewed shapes)
    for a1 in [2, 4, 6]:
        for a2 in [1, 2, 3]:
            a = np.exp(a1 * t - a2 * t**2 * 5)
            a = np.maximum(a, 1e-6)
            inits.append(a / a.sum())

    # Random smooth polynomials
    for _ in range(count):
        k = rng.integers(3, 7)
        coeffs = rng.uniform(-4, 4, k)
        a = np.exp(np.sum(coeffs * t**np.arange(k)))
        a = np.maximum(a, 1e-6)
        inits.append(a / a.sum())

    return inits


def interpolate_to(a, n_new):
    """Interpolate sequence a to length n_new and normalize."""
    n = len(a)
    if n == n_new:
        s = float(a.sum())
        if s < 1e-10:
            return a.copy()
        return a / s
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n_new)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-10:
        return out
    return out / s


def coordinate_polish(a, deadline, rng, max_sweeps=100):
    """Coordinate descent polish on the best coordinates."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a /= a.sum()
    best = a.copy()
    best_sc = compute_score(a)
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            best_local = orig
            best_local_sc = best_sc
            for factor in [0.9, 0.95, 1.05, 1.1, 0.85, 1.15, 0.8, 1.2]:
                a[i] = orig * factor
                if a[i] < 0:
                    a[i] = orig
                    continue
                s = float(a.sum())
                if s < 0.01:
                    a[i] = orig
                    continue
                sc = compute_score(a / s)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local = orig * factor
            a[i] = best_local
            a = a / a.sum()
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best = a.copy()
                improved = True
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 8:
                break

    return best, best_sc
