Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated
soft-max gradient descent with spectral flattening and L-BFGS polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_and_grad(c, beta):
    """Compute soft max value and gradient weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w

    w_rev_conv_a = np.fft.irfft(fwr * fa, N)
    
    # Extract the relevant parts
    # (w * a) has length 2n-1, we want positions n-1 to 2n-2 for the "correlation" part
    # Actually, let me think about this more carefully.
    
    # c_k for k = 0, ..., 2n-2 corresponds to lag k - (n-1)
    # c_k = sum_j a_j * a_{j - (k - (n-1))} (with out-of-bounds = 0)
    
    # dc_k / da_i = a_{i - (k-(n-1))} + a_{i - (k-(n-1))} (by symmetry of the product)
    # Wait: c_k = sum_j a_j a_{j-k+(n-1)}
    # dc_k/da_i: appears when j=i (giving a_{i-k+(n-1)}) and when j-k+(n-1)=i i.e. j=i+k-(n-1) (giving a_i... no)
    
    # Let me be more careful. c_k = sum_j a_j * a_{j-k+(n-1)}
    # dc_k/da_i = a_{i-k+(n-1)} [from j=i term] 
    #            + a_{i} * [coefficient of a_i in the second factor]
    # The second factor is a_{j-k+(n-1)}, which equals a_i when j-k+(n-1)=i, i.e., j=i+k-(n-1)
    # So dc_k/da_i = a_{i-k+(n-1)} + (if i+k-(n-1) is valid) a_{i+k-(n-1)}
    
    # Let lag_l = k - (n-1), so lag ranges from -(n-1) to (n-1)
    # c_{n-1+lag} = sum_j a_j * a_{j-lag}
    # dc/da_i = a_{i-lag} + a_{i+lag} (with out-of-bounds = 0)
    
    # So grad_i = sum_lag w_{n-1+lag} * (a_{i-lag} + a_{i+lag})
    # = sum_lag w_{n-1+lag} * a_{i-lag} + sum_lag w_{n-1+lag} * a_{i+lag}
    # The first sum is the correlation of w (centered) and a at position i
    # The second sum is also a correlation (with w flipped)
    
    # Since autocorrelation is symmetric (c_k = c_{2(n-1)-k}), w is symmetric:
    # w_{n-1+lag} = w_{n-1-lag}
    # So both sums are equal, and grad = 2 * sum_lag w_{n-1+lag} * a_{i-lag}
    # = 2 * correlation(w_centered, a)[i]
    
    # correlation(w_centered, a)[i] = sum_lag w_{n-1+lag} * a_{i-lag}
    # This is the cross-correlation, which can be computed as:
    # (w_rev * a)[n-1 + i] where w_rev is w reversed
    # Or equivalently: (w * a_rev)[n-1+i]
    
    # Let's use: corr[i] = sum_j w[j] * a[i-j+n-1] (with out-of-bounds = 0)
    # This is (w * a)[i + n - 1] from full convolution
    
    # Actually, the full convolution (w * a)[m] = sum_j w[j] a[m-j]
    # We want sum_j w[j] a[i - j + n - 1] = sum_j w[j] a[(i+n-1) - j] = (w * a)[i + n - 1]
    
    # So corr[i] = (w * a)[i + n - 1] for i = 0, ..., n-1
    # And grad = 2 * corr = 2 * (w * a)[n-1 : 2n-1]
    
    grad = 2.0 * w_conv_a[n-1 : 2*n-1]
    
    # Verify: if w is a delta at the center (n-1), then grad should be 2*a
    # (w * a)[n-1 + i] = a[i], so grad = 2*a. Correct!
    
    return grad


def soft_max_adam_nesterov(p0, deadline, rng, lr=None, reg=1e-8, beta_init_frac=0.15, use_nesterov=True):
    """Nesterov or Adam soft-max gradient descent in logit space."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    if lr is None:
        lr = 0.05

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    velocity = np.zeros(n, dtype=np.float64)  # For Nesterov
    adam_t = 0
    nest_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_max_and_grad(c, beta)
        gp = gradient_autocorr_peak(p, w, None)
        # gp is gradient of soft_max w.r.t. p (unconstrained)
        # For logit parameterization: grad_z = p * (gp - sum(p * gp))
        # (This projects the gradient onto the tangent space of the simplex)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            velocity.fill(0.0)
            adam_t = 0
            nest_t = 0
            continue

        if use_nesterov:
            nest_t += 1
            # Nesterov: y = z + momentum * velocity
            momentum = 0.9
            y = z + momentum * velocity
            # Compute gradient at y (lookahead)
            p_y = softmax_from_z(y)
            c_y = full_convolve_fft(p_y)
            L_y, w_y = soft_max_and_grad(c_y, beta)
            gp_y = gradient_autocorr_peak(p_y, w_y, None)
            vp_y = gp_y + reg * p_y
            dot_y = float(np.dot(p_y, vp_y))
            gz_y = p_y * (vp_y - dot_y)
            gz_y -= float(np.mean(gz_y))
            
            if not np.all(np.isfinite(gz_y)):
                gz_y = gz
            
            velocity = momentum * velocity - lr * gz_y
            z = z + velocity
        else:
            adam_t += 1
            m = b1 * m + (1.0 - b1) * gz
            v = b2 * v + (1.0 - b2) * (gz * gz)
            mh = m / (1.0 - b1 ** adam_t)
            vh = v / (1.0 - b2 ** adam_t)
            z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            velocity.fill(0.0)
            adam_t = 0
            nest_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.01, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            velocity.fill(0.0)
            adam_t = 0
            nest_t = 0
            continue

        if nest_t % 100 == 0:
            beta = max(1e-14, beta * 0.75)

        if no_imp > max(400, 5 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            velocity.fill(0.0)
            adam_t = 0
            nest_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def spectral_flatten(a, gamma=0.3, n_iters=5):
    """
    Apply spectral flattening: compress the power spectrum of a.
    This pushes the autocorrelation towards being flatter.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0:
        return a
    a = a / s  # normalize to sum 1
    
    n = a.size
    N = next_pow2(2 * n - 1)
    
    for _ in range(n_iters):
        fa = np.fft.rfft(a, N)
        power = fa.real**2 + fa.imag**2
        mean_power = float(np.mean(power[power > 0])) if np.any(power > 0) else 1.0
        if mean_power <= 0:
            break
        # Compress: new_amplitude = amplitude / (1 + gamma * power/mean_power)^0.5
        # This reduces peaks in the spectrum
        scale = np.sqrt(1.0 + gamma * power / mean_power)
        fa_new = fa / scale
        a = np.fft.irfft(fa_new, N)[:n]
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s <= 1e-30:
            break
        a = a / s
    
    return a


def lfgsb_polish(p0, deadline, max_iter=200):
    """
    Polish using scipy L-BFGS-B on the logit parameterization.
    Objective: 2n * soft_max(conv(a,a), beta) with a = softmax(z)
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z0 = np.log(p0)
    z0 -= float(np.mean(z0))
    
    beta = 0.05 * float(full_convolve_fft(p0).max())
    
    best_score = internal_score(p0)
    best_p = p0.copy()
    
    def objective_and_grad(z):
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        mval = float(c.max())
        beta_local = max(1e-12, 0.05 * mval)
        L, w = soft_max_and_grad(c, beta_local)
        val = 2.0 * n * L  # score (up to normalization which is fixed by softmax)
        gp = gradient_autocorr_peak(p, w, None)
        vp = gp + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        return 2.0 * n * val / 4.0, 2.0 * n * gz / 4.0
        # Wait, the score is 2n * max_c / sum_a^2. Since sum_a = 1 (softmax), it's 2n * max_c.
        # Soft max approximates max_c, so objective = 2n * soft_max(c, beta)
        
    def obj_grad(z):
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        mval = float(c.max())
        L, w = soft_max_and_grad(c, beta)
        val = 2.0 * n * L
        gp = gradient_autocorr_peak(p, w, None)
        vp = gp + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        return val, gz
    
    if time.monotonic() > deadline:
        return best_p, best_score
    
    t_start = time.monotonic()
    time_per_iter = max(0.01, (deadline - t_start) / max_iter)
    
    try:
        result = minimize(
            obj_grad,
            z0,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter,
                'maxfun': max_iter,
                'ftol': 1e-14,
                'gtol': 1e-8,
                'maxls': 20,
            }
        )
        
        if result.success and result.x is not None:
            p_new = softmax_from_z(result.x)
            sc = internal_score(p_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p_new
    except Exception as e:
        pass
    
    return best_p, best_score


def make_deterministic_profiles(n, count=10):
    """Generate diverse deterministic profiles."""
    profs = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear profiles
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    
    # Exponential profiles
    for eta in (1.0, 2.0, 4.0, 8.0):
        a = np.exp(eta * (t - 0.5))
        profs.append(a / a.sum())
    
    # Power profiles
    for p in (0.5, 1.0, 2.0, 4.0, 8.0):
        a = t ** p + 1e-8
        profs.append(a / a.sum())
    
    # Sigmoid profiles
    for eta in (3.0, 6.0, 12.0, 25.0):
        for center in (0.3, 0.5, 0.7):
            a = 1.0 / (1.0 + np.exp(-eta * (t - center)))
            profs.append(a / a.sum())
    
    # Bell-shaped
    for width in (0.1, 0.2, 0.3, 0.5):
        for center in (0.3, 0.5, 0.7):
            a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
            profs.append(a / a.sum())
    
    # Two-hump
    for gap in (0.1, 0.2, 0.3):
        a = np.exp(-((t - 0.5 + gap/2)**2) / 0.02) + np.exp(-((t - 0.5 - gap/2)**2) / 0.02)
        profs.append(a / a.sum())
    
    # Cosine-modulated
    for freq in (1, 2, 3, 5):
        for amp in (0.3, 0.5, 0.7):
            a = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 1e-12)
            profs.append(a / a.sum())
    
    # Triangle / sawtooth
    a = 1.0 - np.abs(2 * t - 1)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    
    return profs[:count]


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def coordinate_descent_polish(p, deadline, rng, step_frac=0.05):
    """Coordinate descent on individual elements."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0
    sweep = 0
    
    while time.monotonic() < deadline:
        if no_imp > 3 * n:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            best_val = orig
            best_sc = best_score
            for factor in (1.15, 1.08, 0.92, 0.85, 1.25, 0.75):
                new_val = orig * factor
                if new_val < 1e-15 or new_val > 1000.0:
                    continue
                p[i] = new_val
                s = float(p.sum())
                if s < 1e-30:
                    p[i] = orig
                    continue
                p = p / s
                sc = internal_score(p)
                p[i] = orig * factor * (s / float(p.sum()))  # recompute
                p = p / p.sum()
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_sc < best_score - 1e-15:
                best_score = best_sc
                best_p = p.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        sweep += 1
        if not improved and sweep > 2:
            step_frac = max(0.02, step_frac * 0.8)

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: Nesterov soft-max gradient + spectral flattening + L-BFGS polish + coordinate descent.
    Targets n=512 primarily, with attempts at n=1024.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} ({label})")
        if len(candidates) < 30:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            if float(sc) < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        pass

    # Phase 0: Initialize
    print("Phase 0: Initialization")
    if parent is not None:
        sc = internal_score(parent)
        local_update(parent, sc, "parent")
        print(f"  Parent score: {sc:.10f}")

    n_main = 512
    
    # Generate diverse initial profiles
    init_profiles = []
    if parent is not None and parent.size == n_main:
        init_profiles.append(parent)
    
    for p0 in make_deterministic_profiles(n_main, 15):
        init_profiles.append(p0)
    
    # Add random smooth profiles
    for _ in range(10):
        t = np.linspace(0, 1, n_main)
        # Random low-frequency signal
        coeffs = rng.normal(size=8)
        a = np.ones(n_main)
        for k in range(1, 8):
            a += coeffs[k] / (k + 1) * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        init_profiles.append(a / a.sum())
    
    print(f"  {len(init_profiles)} initial profiles")
    
    # Phase 1: Quick evaluation and spectral flattening
    print("Phase 1: Spectral flattening of initial profiles")
    phase1_end = min(deadline, t_start + 60.0)
    
    for i, p0 in enumerate(init_profiles):
        if time.monotonic() > phase1_end:
            break
        if p0.size != n_main:
            p0 = interpolate_profile(p0, n_main)
        if p0 is None:
            continue
        
        sc0 = internal_score(p0)
        local_update(p0, sc0, f"init_{i}")
        
        # Spectral flattening
        for gamma in (0.1, 0.3, 0.5):
            p_flat = spectral_flatten(p0, gamma=gamma, n_iters=3)
            sc_flat = internal_score(p_flat)
            local_update(p_flat, sc_flat, f"flat_{i}_g{gamma}")
    
    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: Nesterov gradient descent on top candidates
    print("Phase 2: Nesterov gradient descent")
    phase2_end = min(deadline, t_start + 800.0)
    
    top_k = min(12, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n_main:
            p_c = interpolate_profile(p_c, n_main)
        if p_c is None:
            continue
        
        # Run Nesterov for a budget
        per_budget = min(phase2_end - time.monotonic(), 30.0)
        if per_budget < 5.0:
            continue
        
        p_best, sc = soft_max_adam_nesterov(p_c, time.monotonic() + per_budget, rng, lr=0.04, use_nesterov=True)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"nest_{i}")
        
        # Also try Adam variant
        per_budget2 = min(phase2_end - time.monotonic(), 20.0)
        if per_budget2 < 5.0:
            continue
        p_best2, sc2 = soft_max_adam_nesterov(p_c, time.monotonic() + per_budget2, rng, lr=0.08, use_nesterov=False)
        sc_full2 = internal_score(p_best2)
        local_update(p_best2, sc_full2, f"adam_{i}")
    
    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: Spectral flattening + re-optimization
    print("Phase 3: Spectral flatten + re-optimize")
    phase3_end = min(deadline, t_start + 1200.0)
    
    for i in range(min(8, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size != n_main:
            p_c = interpolate_profile(p_c, n_main)
        if p_c is None:
            continue
        
        for gamma in (0.2, 0.4, 0.8, 1.2):
            p_flat = spectral_flatten(p_c, gamma=gamma, n_iters=5)
            sc_flat = internal_score(p_flat)
            local_update(p_flat, sc_flat, f"sflat3_{i}_g{gamma}")
            
            # Re-optimize the flattened version
            per = min(phase3_end - time.monotonic(), 15.0)
            if per < 3.0:

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_and_grad(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def gradient_autocorr_peak(a, w):

def spectral_flatten(a, gamma=0.3, n_iters=5):
    """Compress the power spectrum to flatten autocorrelation."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0:
        return a
    a = a / s
    n = a.size
    N = next_pow2(2 * n - 1)
    for _ in range(n_iters):
        fa = np.fft.rfft(a, N)
        power = fa.real ** 2 + fa.imag ** 2
        mask = power > 1e-20
        if not np.any(mask):
            break
        mean_power = float(np.mean(power[mask]))
        if mean_power <= 0:
            break
        scale = np.sqrt(1.0 + gamma * power / mean_power)
        scale = np.maximum(scale, 1e-10)
        fa_new = fa / scale
        a = np.fft.irfft(fa_new, N)[:n]
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s <= 1e-30:
            break
        a = a / s
    return a
