def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    return None

"""
Active-set LP + Nesterov subgradient + coordinate descent for minimizing
2n * max(conv(a,a)) / sum(a)^2.

Key innovations over previous approach:
1. Active-set LP: only enforce top-m convolution constraints, making LP much smaller

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _subgradient_at(a, k_star):
    """Compute exact subgradient of max conv at lag k_star.
    grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    for i in range(i_min, i_max + 1):
        grad[i] = 2.0 * a[k_star - i]
    return grad


def _smooth_max_grad(a, beta):
    """Compute smooth-max gradient using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    return grad, mval


def _active_set_lp(a, m=40, inflate=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= conv(a,a)[k]*inflate for top-m k, g>=0."""
    n = len(a)
    c = _fft_convolve(a)
    top_m_idx = np.argsort(c)[::-1][:min(m, 2 * n - 1)]
    A = np.zeros((len(top_m_idx), n))
    b = np.zeros(len(top_m_idx))
    for idx, k in enumerate(top_m_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[idx, j] = a[k - j]
        b[idx] = c[k] * inflate
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine_active(a, deadline, m=40, max_iter=100000, inflate=1.0005, blend=0.25):
    """LP refinement using active-set (top-m) constraints only."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    m_cur = m

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        g = _active_set_lp(a, m_cur, inflate_cur)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.15)
                m_cur = min(2 * n - 1, int(m_cur * 1.4) + 5)
            if no_imp > 300:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                blend_cur = max(0.03, blend_cur * 0.85)
            if no_imp > max(200, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                m_cur = m
                no_imp = 0

        a = a_new

    return best_a, best_score


def _nesterov_subgradient(a, deadline, max_iter=100000):
    """Nesterov-accelerated subgradient method on the probability simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    y = a.copy()
    momentum = 0.85
    lr = 0.03
    base_lr = lr
    no_imp = 0
    v = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(y)
        k_star = int(np.argmax(c))

        grad = _subgradient_at(y, k_star)
        proj_grad = grad - np.mean(grad)

        v = momentum * v + proj_grad
        a_new = _project_simplex(a - lr * v)

        # Decay learning rate
        lr = base_lr / (1.0 + 0.005 * it)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.001, base_lr * 0.92)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                y = best_a.copy()
                v = np.zeros(n)
                base_lr = 0.03
                lr = base_lr
                no_imp = 0

        a = a_new
        y = a

    return best_a, best_score


def _smooth_max_polish(a, deadline, max_iter=100000, beta_init=None):
    """Smooth-max gradient descent with Nesterov momentum on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    c0 = _fft_convolve(a)
    beta = beta_init if beta_init else 0.05 * float(c0.max())

    lr = 0.04
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, mval = _smooth_max_grad(a, beta)
        proj_grad = grad - np.mean(grad)

        if not np.all(np.isfinite(proj_grad)):
            beta = max(1e-10, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v + proj_grad
        a_new = _project_simplex(a - lr * v)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                base_lr = max(0.002, base_lr * 0.93)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.05 * float(_fft_convolve(a).max())
                base_lr = 0.04
                no_imp = 0

        a = a_new
        lr = base_lr / (1.0 + 0.003 * it)

        if it % 300 == 0 and it > 0:
            beta = max(1e-12, beta * 0.6)

    return best_a, best_score


def _logit_adam_polish(a, deadline, max_iter=100000, lr=0.08):
    """Adam optimization in logit space for the probability vector."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    beta = 0.05 * float(_fft_convolve(a).max())
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        vp = 2.0 * h_full[n - 1:2 * n - 1]
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m_arr = b1 * m_arr + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m_arr / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c_new = _fft_convolve(a)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.94)

        if it % 200 == 0 and it > 0:
            beta = max(1e-14, beta * 0.5)

        if no_imp > max(4000, 6 * n):
            z = 0.5 * best_z + 0.5 * np.random.default_rng(42 + it).normal(size=n) * 0.3
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = 0.05 * float(_fft_convolve(a).max())
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _coordinate_descent(a, deadline, sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    rng = np.random.default_rng(12345)

    for sweep in range(sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            a_old = a[i]
            # Try several candidate values for a[i]
            candidates = [0.0, a_old * 0.5, a_old * 0.8, a_old * 1.2, a_old * 1.5,
                          a_old + 0.01 / n, a_old - 0.01 / n, 2.0 / n]
            candidates = [max(0.0, c) for c in candidates]

            best_val = a_old
            best_sc = _score(a)

            for t in candidates:
                a[i] = t
                s = a.sum()
                if s < 1e-10:
                    a[i] = a_old
                    continue
                sc = float(2.0 * n * _fft_convolve(a).max() / (s * s))
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_val = t
            a[i] = best_val

            if best_val != a_old:
                improved = True

            sc = _score(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()

        if not improved and sweep > 5:
            break

    return best_a, best_score


def _structured_restart(a, rng, strength=0.3):
    """Apply structured perturbation: sinusoidal tilt + random noise."""
    n = len(a)
    # Random sinusoidal tilt
    freq = rng.integers(1, max(2, n // 4))
    phase = rng.uniform(0, 2 * np.pi)
    t = np.linspace(0, 1, n)
    tilt = 1.0 + strength * np.sin(2 * np.pi * freq * t + phase)
    a_tilted = a * tilt
    a_tilted = np.maximum(a_tilted, 0)
    # Add small random noise
    a_tilted = a_tilted * (1.0 + 0.05 * strength * rng.normal(size=n))
    a_tilted = np.maximum(a_tilted, 0)
    s = a_tilted.sum()
    if s > 1e-10:
        a_tilted = a_tilted / s
    else:
        a_tilted = np.ones(n) / n
    return a_tilted


def _make_initial(n, rng, style='structured'):
    """Create an initial probability vector."""
    t = np.linspace(0, 1, n)
    if style == 'structured':
        # Low-frequency structured initialization
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 3) * t + rng.uniform(0, 6.28))
        a = a * (1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(2, 5) * t + rng.uniform(0, 6.28)))
    elif style == 'linear':
        a = 1.0 + rng.uniform(0.5, 2.0) * (t - 0.5) * 2
    elif style == 'bimodal':
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        a = 1.0 + 2.0 * np.exp(-((t - center) ** 2) / (2 * width ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        s = a.sum()
        if s > 1e-10:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Active-set LP refinement from parent
    Phase 2: Nesterov subgradient polish
    Phase 3: Smooth-max gradient polish
    Phase 4: Logit Adam polish
    Phase 5: Structured restarts + re-optimization
    Phase 6: Coordinate descent final polish
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = float(p.sum())
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Target length
    n_target = 512
    if parent is not None:
        n_target = parent.size

    # Prepare initial sequence
    if parent is not None:
        a0 = _interpolate_to_length(parent, n_target)
    else:
        a0 = _make_initial(n_target, rng, 'structured')

    sc0 = _score(a0)
    print(f"Initial: n={n_target}, score={sc0:.10f}")
    if np.isfinite(sc0) and sc0 < best_score:
        best_score = sc0
        best_seq = a0.copy()

    # === Phase 1: Active-set LP refinement ===
    phase1_deadline = t_start + min(400.0, (deadline - t_start) * 0.3)
    print(f"Phase 1: Active-set LP, budget={phase1_deadline - t_start:.0f}s")
    a1, sc1 = _lp_refine_active(a0, phase1_deadline, m=40, max_iter=200000,
                                  inflate=1.0005, blend=0.25)
    sc1_real = _score(a1)
    print(f"  Phase 1 result: score={sc1_real:.10f}")
    if sc1_real < best_score - 1e-14:
        best_score = sc1_real
        best_seq = a1.copy()

    # === Phase 2: Nesterov subgradient ===
    phase2_deadline = t_start + min(700.0, (deadline - t_start) * 0.45)
    print(f"Phase 2: Nesterov subgradient, budget={phase2_deadline - t_start:.0f}s")
    a2, sc2 = _nesterov_subgradient(a1, phase2_deadline, max_iter=500000)
    sc2_real = _score(a2)
    print(f"  Phase 2 result: score={sc2_real:.10f}")
    if sc2_real < best_score - 1e-14:
        best_score = sc2_real
        best_seq = a2.copy()

    # === Phase 3: Smooth-max gradient ===
    if time.monotonic() < deadline - 100:
        phase3_deadline = t_start + min(1000.0, (deadline - t_start) * 0.6)
        print(f"Phase 3: Smooth-max gradient, budget={phase3_deadline - t_start:.0f}s")
        a3, sc3 = _smooth_max_polish(best_seq, phase3_deadline, max_iter=500000)
        sc3_real = _score(a3)
        print(f"  Phase 3 result: score={sc3_real:.10f}")
        if sc3_real < best_score - 1e-14:
            best_score = sc3_real
            best_seq = a3.copy()

    # === Phase 4: Logit Adam ===
    if time.monotonic() < deadline - 150:
        phase4_deadline = t_start + min(1300.0, (deadline - t_start) * 0.75)
        print(f"Phase 4: Logit Adam, budget={phase4_deadline - t_start:.0f}s")
        a4, sc4 = _logit_adam_polish(best_seq, phase4_deadline, max_iter=200000, lr=0.06)
        sc4_real = _score(a4)
        print(f"  Phase 4 result: score={sc4_real:.10f}")
        if sc4_real < best_score - 1e-14:
            best_score = sc4_real
            best_seq = a4.copy()

    # === Phase 5: Structured restarts + re-optimization ===
    if time.monotonic() < deadline - 120:
        print(f"Phase 5: Structured restarts, budget={deadline - time.monotonic() - 120:.0f}s")
        for restart_idx in range(10):
            if time.monotonic() > deadline - 120:
                break
            a_restart = _structured_restart(best_seq, rng, strength=0.2 + 0.1 * restart_idx)
            sc_restart = _score(a_restart)

            # Quick LP on restarted sequence
            lp_dead = min(time.monotonic() + 30.0, deadline - 120.0)
            a_r, _ = _lp_refine_active(a_restart, lp_dead, m=30, max_iter=50000,
                                         inflate=1.001, blend=0.3)
            sc_r = _score(a_r)
            print(f"  Restart {restart_idx}: {sc_restart:.10f} -> {sc_r:.10f}")

            if sc_r < best_score - 1e-14:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 6: Final smooth-max polish ===
    if time.monotonic() < deadline - 60:
        phase6_deadline = deadline - 30.0
        print(f"Phase 6: Final smooth-max polish, budget={phase6_deadline - time.monotonic():.0f}s")
        a6, _ = _smooth_max_polish(best_seq, phase6_deadline, max_iter=300000)
        sc6 = _score(a6)
        print(f"  Phase 6 result: score={sc6:.10f}")
        if sc6 < best_score - 1e-14:
            best_score = sc6
            best_seq = a6.copy()

    # === Phase 7: Coordinate descent final polish ===
    if time.monotonic() < deadline - 15:
        cd_deadline = deadline - 10.0
        print(f"Phase 7: Coordinate descent, budget={cd_deadline - time.monotonic():.0f}s")
        a7, _ = _coordinate_descent(best_seq, cd_deadline, sweeps=20)
        sc7 = _score(a7)
        print(f"  Phase 7 result: score={sc7:.10f}")
        if sc7 < best_score - 1e-14:
            best_score = sc7
            best_seq = a7.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = _make_initial(512, rng, 'structured')

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_initial(512, rng, 'structured')

    # Evaluate with authoritative function
    if final_seq.size <= 6000:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    # Also try a few scaling variants to make sure we're not at a boundary
    for scale in [0.9, 1.1, 1.5, 2.0]:
        scaled = final_seq * scale
        scaled = np.clip(scaled, 0.0, 1000.0)
        if scaled.size <= 6000:
            try:
                sc_s = float(evaluate_sequence(scaled.tolist()))
                if np.isfinite(sc_s) and sc_s < best_score:
                    best_score = sc_s
                    final_seq = scaled.copy()
                    print(f"  Scaled by {scale}: {sc_s:.10f}")
            except Exception:
                pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
