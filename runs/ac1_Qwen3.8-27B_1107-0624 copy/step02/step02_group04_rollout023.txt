Let me think about the projected gradient more carefully.

The problem (normalized, sum a = 1):
minimize f(a) = 2n * max_k c[k] where c = conv(a,a)

Projecting gradient:
- Compute subgradient g of max_k c[k] (using soft-max or top-k)
- Update: a_new = a - lr * g
- Project a_new onto simplex {a >= 0, sum a = 1}

The simplex projection is:
- Sort a_new in decreasing order: a_new[1] >= a_new[2] >= ... >= a_new[n]
- Find the largest t such that sum of (a_new[i] - t)_+ = 1
- a_projected[i] = max(0, a_new[i] - t)

This is O(n log n) per iteration, which is fine.

For the step size, I'll use backtracking:
- Start with lr = lr_init
- While f(a - lr*g) > f(a) - c1 * lr * ||g||^2: lr /= 2
- (This is the Armijo condition)

"""
Optimized step function search using adaptive logit-space gradient descent
with diverse initialization, top-k max approximation, beta oscillation,
and multi-scale escalation.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = c - c.min()
    return c

def _softmax_from_z(z, floor=1e-12):
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + floor
    s = p.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(e) / e.size
    return p / s

def _correlation(w, p):
    """Compute cross-correlation: result[i] = sum_j w[j] * p[j-i] (valid range)"""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _score_normalized(p):
    """Score assuming sum(p) = 1: 2*n*max(conv(p,p))"""
    n = p.size
    c = _convolve_fft(p)
    return 2.0 * n * float(c.max())

def _top_k_weights(c, k, beta=0.0):
    """Get weights for top-k approximation of max."""
    mval = float(c.max())
    if k >= c.size:
        k = c.size
    # Get top-k indices
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    if beta <= 0:
        w[idx] = 1.0 / k
    else:
        vals = c[idx]
        x = (vals - vals.max()) / beta
        ex = np.exp(x)
        w[idx] = ex / ex.sum()
    return mval, w

def _make_profile_exp(n, alpha):
    """Exponential decay profile."""
    t = np.arange(n) / max(n - 1, 1)
    a = np.exp(-alpha * t)
    s = a.sum()
    return a / s

def _make_profile_power(n, p):
    """Power law profile."""
    t = np.arange(n) / max(n - 1, 1)
    a = (1.0 - t + 1e-10) ** p
    s = a.sum()
    return a / s

def _make_profile_sigmoid(n, sharpness, center=0.3):
    """Sigmoid-like profile."""
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 / (1.0 + np.exp(-sharpness * (t - center)))
    s = a.sum()
    return a / s

def _make_profile_u(n, depth):
    """U-shaped profile (high at edges, low in middle)."""
    t = np.arange(n) / max(n - 1, 1)
    u = 2.0 * t - 1.0
    a = 1.0 + depth * u * u
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def _make_profile_inv_u(n, depth):
    """Inverted U (high in middle)."""
    t = np.arange(n) / max(n - 1, 1)
    u = 2.0 * t - 1.0
    a = 1.0 - depth * u * u
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

    """Linear ramp."""
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-10)
    s = a.sum()

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logit-space Adam with diverse asymmetric initialization, adaptive top-k

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c -= mn
    return c


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / e.size
    return p / s

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return 2.0 * h[n - 1:2 * n - 1]
