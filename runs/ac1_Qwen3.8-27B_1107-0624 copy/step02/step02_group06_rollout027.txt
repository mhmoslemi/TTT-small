```python
import numpy as np
import time
from scipy.optimize import linprog

def lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) subject to (f*g)_k <= M for all k."""
    n = len(f)
    # (f*g)_k = sum_i f_i g_{k-i} for valid i
    # This is a linear constraint in g
    # Build constraint matrix A such that A @ g <= M * ones
    # A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    # But we need to be careful: (f*g)_k = sum_{i} f_i g_{k-i}
    # For this to be linear in g: (f*g)_k = sum_j g_j f_{k-j}
    
    N = 2 * n - 1
    A = np.zeros((N, n))
    for k in range(N):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # Constraints: A @ g <= M (vector of M's), g >= 0
    res = linprog(c, A_ub=A, b_ub=np.full(N, M), bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None
```

But `a_{k-j} = a_j` when `k = 2j`, so the second term is `a_j = a_{k-j}`, giving total `2 * a_{k-j}` when `k = 2j`.

In general: `d(a*a)_k / da_j = a_{k-j} + delta_{k,2j} * a_{k-j}` where `delta_{k,2j}` is 1 if `k=2j` and 0 otherwise.

Or more simply: the gradient vector is `2 * (shift of a)` with a factor of 2 at the center.

For practical purposes (n=512), the factor of 2 at the center (only 1 position out of 512) is negligible. So we can approximate:
`grad_k ≈ 2 * (a shifted by k)` (truncated to length n)

In the previous code, `corr_weights(w, p)` computes exactly this (it's the cross-correlation of weights with p, which is equivalent to the convolution of weights with reversed p).

```python
def solve_lp_direction(f, M, max_constraints=None):
    """
    Solve LP: maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.
    Returns the optimal g, or None if LP fails.
    """
    from scipy.optimize import linprog
    n = len(f)
    N = 2 * n - 1
    # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    if max_constraints is not None and max_constraints < N:
        # Only use a subset of constraints (most important ones)
        # We'll use all for now since n=512 is manageable
        pass
    A = np.zeros((N, n), dtype=np.float64)
    for k in range(N):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            i = k - j
            A[k, j] = f[i]
    c = -np.ones(n)
    b_ub = np.full(N, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def gradient_optimize(p0, deadline, rng, lr=0.10, beta_init_frac=0.2,
                      use_lp=False, lp_interval=200):
    """
    Soft-max gradient optimization with optional LP-based direction finding.
    Works in logit space with Adam optimizer.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    it = 0

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        it += 1

        # Periodic beta annealing
        if it % 60 == 0:
            beta = max(1e-14, beta * 0.75)

        # LP-based direction finding when stuck
        if use_lp and no_imp > 100 and (no_imp % lp_interval == 0):
            M_current = maxc * 1.001
            g = solve_lp_direction(p, M_current)
            if g is not None and g.sum() > 1e-10:
                g_norm = g / g.sum()
                # Blend towards g
                t_blend = min(0.5, 100.0 / float(no_imp))
                p_new = (1.0 - t_blend) * p + t_blend * g_norm
                p_new = np.maximum(p_new, 1e-12)
                p_new = p_new / p_new.sum()
                c_new = full_convolve_fft(p_new)
                maxc_new = float(c_new.max())
                score_new = float(2.0 * n * maxc_new)
                if np.isfinite(score_new) and score_new < best_score:
                    best_score = score_new
                    best_p = p_new.copy()
                    z = np.log(np.maximum(p_new, 1e-15))
                    z -= float(np.mean(z))
                    best_z = z.copy()
                    p = p_new
                    c = c_new
                    maxc = maxc_new
                no_imp = 0
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0

        # Restart if very stuck
        if no_imp > max(400, 3 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def active_set_equalize(p, top_k=5, n_iters=500, lr=0.05, deadline=None):
    """
    Try to equalize the top-k convolution peaks by adjusting p.
    Uses a penalty method: minimize sum_{top k} (conv_k - mean_topk)^2.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = internal_score(p)

    t_end = deadline if deadline else time.monotonic() + 30.0
    it = 0

    while it < n_iters and time.monotonic() < t_end:
        c = full_convolve_fft(p)
        # Find top k indices
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        mean_top = float(np.mean(top_vals))

        # Gradient of penalty = sum_{k in top} 2*(c_k - mean_top) * d(c_k)/da
        # d(c_k)/da = 2 * (shifted a)
        penalty_grad = np.zeros(n)
        for k in top_idx:
            weight = 2.0 * (c[k] - mean_top)
            # d(c_k)/da_j = a_{k-j} if valid
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                penalty_grad[j] += weight * a_safe(c, k, j, p)
        it += 1

        # Simplified: use the subgradient approach
        # Direction to reduce top peaks
        w = np.zeros_like(c)
        for k in top_idx:
            w[k] = c[k] - mean_top + 1e-10
        w = np.maximum(w, 0)
        if w.sum() < 1e-20:
            break
        w = w / w.sum()
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        # Adam step
        if not np.all(np.isfinite(gz)):
            break
        z = np.log(np.maximum(p, 1e-15))
        z -= lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)

        sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()

    return best_p, best_score


def a_safe(c, k, j, p):
    """Safe access for gradient computation."""
    idx = k - j
    if 0 <= idx < len(p):
        return p[idx]
    return 0.0


def lp_refinement_cycle(p, deadline, rng, max_cycles=20):
    """
    Repeatedly solve LP to find a good direction and blend.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = internal_score(p)

    for cycle in range(max_cycles):
        if time.monotonic() > deadline:
            break
        c = full_convolve_fft(p)
        maxc = float(c.max())
        M = maxc * 1.001

        g = solve_lp_direction(p, M)
        if g is None or g.sum() < 1e-10:
            break

        g_norm = g / g.sum()
        # Try several blend factors
        best_local = best_score
        best_local_p = best_p.copy()
        for t_blend in (0.1, 0.2, 0.3, 0.5):
            p_new = (1.0 - t_blend) * p + t_blend * g_norm
            p_new = np.maximum(p_new, 1e-12)
            p_new = p_new / p_new.sum()
            sc = internal_score(p_new)
            if np.isfinite(sc) and sc < best_local:
                best_local = sc
                best_local_p = p_new.copy()

        if best_local < best_score:
            best_score = best_local
            best_p = best_local_p.copy()
            p = best_local_p.copy()
        else:
            # No improvement; try a small perturbation
            p = p + 0.01 * rng.normal(size=n)
            p = np.maximum(p, 1e-12)
            p = p / p.sum()

    return best_p, best_score


def coordinate_descent_polish(p, deadline, rng, step_frac=0.05):
    """Coordinate descent with geometric step sizes."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    best_p = p.copy()
    best_score = internal_score(p)
    no_imp = 0

    while time.monotonic() < deadline:
        idx = rng.permutation(n)
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = p[i]
            if orig < 1e-14:
                continue
            best_val = orig
            best_loc_sc = best_score
            for factor in (1.0 + step_frac, 1.0 - step_frac, 1.0 + 0.5*step_frac, 1.0 - 0.5*step_frac):
                new_val = orig * factor
                if new_val < 1e-15:
                    continue
                old_sum = p.sum()
                p[i] = new_val
                new_sum = old_sum - orig + new_val
                if new_sum < 0.01:
                    p[i] = orig
                    continue
                p = p / new_sum
                sc = internal_score(p)
                if np.isfinite(sc) and sc < best_loc_sc:
                    best_loc_sc = sc
                    best_val = new_val
            p[i] = best_val
            p = p / p.sum()
            if best_loc_sc < best_score:
                best_score = best_loc_sc
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        if no_imp > 3 * n:
            step_frac = max(0.01, step_frac * 0.8)
            if step_frac < 0.01:
                break
            no_imp = 0

    return best_p, best_score


def try_scipy_lbfgs(p0, deadline, max_iter=5000):
    """
    Use scipy L-BFGS-B to optimize in log-space.
    Objective: 2*n*max(conv(exp(z), exp(z))) (scale invariant, z shifted).
    """
    from scipy.optimize import minimize
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z0 = np.log(p0)
    z0 -= float(np.mean(z0))

    def objective(z):
        z = z - float(np.max(z))
        p = np.exp(z)
        p = p / p.sum()
        c = full_convolve_fft(p)
        mval = float(c.max())
        return 2.0 * n * mval

    def gradient(z):
        z = z - float(np.max(z))
        p = np.exp(z)
        s = p.sum()
        p = p / s
        c = full_convolve_fft(p)
        # Use soft max with very small beta for gradient
        beta = 1e-6 * max(1e-10, c.max())
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot) / s
        gz -= float(np.dot(p, gz) / s)
        return gz

    try:
        res = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': 5000,
                                'ftol': 1e-14, 'gtol': 1e-10,
                                'maxls': 20})
        if res.success or res.fun < objective(z0):
            z_opt = res.x
            z_opt = z_opt - float(np.max(z_opt))
            p_opt = np.exp(z_opt)
            p_opt = p_opt / p_opt.sum()
            return p_opt, float(2.0 * n * full_convolve_fft(p_opt).max())
    except Exception:
        pass
    return p0, internal_score(p0)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: gradient optimization, LP-based refinement,
    multi-length search, and coordinate descent polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            if label:
                print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} {label}")
        if len(candidates) < 30:
            candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            if float(sc) < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])

    # Phase 0: Load parent and initial setup
    print("Phase 0: Initialization")
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = normalize_nonneg(parent)
    except Exception:
        parent = None

    n_parent = 512
    if parent is not None:
        sc = internal_score(parent)
        local_update(parent, sc, "parent")
        # Try different lengths
        for tn in (256, 1024):
            if parent.size != tn:
                p_interp = interpolate_profile(parent, tn)
                if p_interp is not None:
                    sc_i = internal_score(p_interp)
                    local_update(p_interp, sc_i, f"parent_interp_n{tn}")
    else:
        parent = make_linear_profile(n_parent, 1.5)
        sc = internal_score(parent)
        local_update(parent, sc, "default_parent")

    # Quick diverse profiles
    for n_init in (256, 512):
        for p0 in make_deterministic_profiles(n_init, 6):
            sc = internal_score(p0)
            local_update(p0, sc, f"init_n{n_init}")

    print(f"  After init: best={best_score:.10f}")

    # Phase 1: Gradient optimization on best candidates
    print("Phase 1: Gradient optimization")
    phase1_end = min(deadline, t_start + 500.0)
    n_opt = best_seq.size if best_seq is not None else 512

    # Optimize top 6 candidates
    top_k = min(6, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase1_end:
            break
        sc_c, p_c = candidates[i]
        per = min(phase1_end, time.monotonic() + 60.0)
        p_best, sc = gradient_optimize(p_c, per, rng, lr=0.12, beta_init_frac=0.25)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"p1_grad_{i}")

    print(f"  After phase 1: best={best_score:.10f}")

    # Phase 2: LP-based refinement
    print("Phase 2: LP-based refinement")
    phase2_end = min(deadline, t_start + 900.0)
    top_k = min(4, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 1024:
            continue
        per = min(phase2_end, time.monotonic() + 80.0)
        p_best, sc = lp_refinement_cycle(p_c, per, rng, max_cycles=10)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"p2_lp_{i}")

    # Also try gradient with LP on best
    if best_seq is not None and time.monotonic() < phase2_end:
        per = min(phase2_end, time.monotonic() + 100.0)
        p_best, sc = gradient_optimize(best_seq, per, rng, lr=0.08,
                                       beta_init_frac=0.15, use_lp=True, lp_interval=150)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p2_grad_lp")

    print(f"  After phase 2: best={best_score:.10f}")

    # Phase 3: Try scipy L-BFGS-B on best candidates
    print("Phase 3: L-BFGS-B optimization")
    phase3_end = min(deadline, t_start + 1200.0)
    top_k = min(3, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 512:
            continue
        per = min(phase3_end, time.monotonic() + 60.0)
        p_best, sc = try_scipy_lbfgs(p_c, per, max_iter=3000)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"p3_lbfgs_{i}")

    print(f"  After phase 3: best={best_score:.10f}")

    # Phase 4: Multi-length exploration
    print("Phase 4: Multi-length search")
    phase4_end = min(deadline, t_start + 1500.0)
    for tn in (256, 1024):
        if time.monotonic() > phase4_end:
            break
        # Interpolate best to this length
        if best_seq is not None:
            p0 = interpolate_profile(best_seq, tn)
            if p0 is not None:
                per = min(phase4_end, time.monotonic() + 60.0)
                p_best, sc = gradient_optimize(p0, per, rng, lr=0.10)
                sc_full = internal_score(p_best)
                local_update(p_best, sc_full, f"p4_n{tn}_best")
        # Deterministic profiles
        for p0 in make_deterministic_profiles(tn, 4):
            if time.monotonic() > phase4_end:
                break
            per = min(phase4_end, time.monotonic() + 30.0)
            p_best, sc = gradient_optimize(p0, per, rng, lr=0.10)
            sc_full = internal_score(p_best)
            local_update(p_best, sc_full, f"p4_n{tn}_det")

    print(f"  After phase 4: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    print("Phase 5: Coordinate descent polish")
    phase5_end = min(deadline, t_start + 1750.0)
    top_k = min(4, len(candidates))
    for i in range(top_k):
        if time.monotonic() > phase5_end:
            break
        sc_c, p_c = candidates[i]
        if p_c.size > 512:
            continue
        per = min(phase5_end, time.monotonic() + 40.0)
        p_best, sc = coordinate_descent_polish(p_c, per, rng, step_frac=0.04)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, f"p5_cd_{i}")

    # Extra: gradient polish on the absolute best
    if best_seq is not None and time.monotonic() < phase5_end:
        per = min(phase5_end, time.monotonic() + 50.0)
        p_best, sc = gradient_optimize(best_seq, per, rng, lr=0.04, beta_init_frac=0.05)
        sc_full = internal_score(p_best)
        local_update(p_best, sc_full, "p5_final_grad")

    print(f"  After phase 5: best={best_score:.10f}")

    # Phase 6: Final exact scoring and selection
    print("Phase 6: Final selection")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        # Exact score using np.convolve
        sc = true_score(final_seq)
        if np.isfinite(sc):
            final_score = sc

        # Check all candidates exactly
        for sc_c, p_c in candidates:
            if p_c.size <= 2048:
                sc_cx = true_score(p_c)
                if np.isfinite(sc_cx) and sc_cx < final_score:
                    final_score = sc_cx
                    final_seq = p_c

        # Verify with evaluate_sequence
        if final_seq is not None and final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval):
                    final_score = min(final_score, sc_eval)
            except Exception:
                pass

        # Final normalization
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"  FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_linear_profile(512, 1.5)
    print(f"  FALLBACK: score={internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
