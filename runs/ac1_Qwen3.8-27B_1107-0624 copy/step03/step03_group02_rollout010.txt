def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, mval
    return ex / s, mval

def _grad_smooth_max(a, w):

    conv_result = np.fft.irfft(fw * fa, N)

        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Bell
    for width in [0.2, 0.3, 0.5, 0.7]:
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Two-bump
    for sep in [0.2, 0.3, 0.4]:
        a = np.exp(-((t - 0.5 + sep) ** 2) / 0.02) + np.exp(-((t - 0.5 - sep) ** 2) / 0.02)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Polynomial (Legendre-like)
    for a3 in [-3.0, -2.0, -1.0, 1.0, 2.0, 3.0]:
        p = 1.0 + a3 * (4.0 * u ** 3 - 3.0 * u) / 3.0
        a = np.maximum(p, 1e-12)
        profiles.append(a / a.sum())
    
    return profiles

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _lp_refine(a, deadline, max_iter=2000, inflate=1.0005, blend=0.3):

        M = float(c.max()) * inflate

        b = np.full(2 * n - 1, M)

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)

        if s_new < 1e-10:

def _penalty_method(a, deadline, max_iter=20000):
    """Minimize sum(exp(beta * conv)) via projected gradient."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    beta = 0.05
    lr = 0.01
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        score_obj = float(ex.sum())
        
        # Gradient of sum(exp(beta*c[k])) wrt a
        # d/da[i] sum_k exp(beta*c[k]) = sum_k beta * exp(beta*c[k]) * dc[k]/da[i]
        # = beta * sum_k ex[k] * (a[k-i] + a[i-(k-(n-1))]) ... 
        # Use the smooth max gradient
        w = ex / ex.sum()
        grad = beta * _grad_smooth_max(a, w)
        
        # Project to tangent
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot
        
        a_new = a - lr * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        if it % 100 == 0:
            beta *= 0.9
            lr *= 0.95
        
        if no_imp > max(3000, 5 * n):
            break
    
    return best_a, best_score

"""
Multi-strategy optimization for step function coefficient search.
Uses Nesterov accelerated gradient with smooth-max penalty, adaptive
beta annealing, multi-peak targeting, pairwise local search, and LP
polishing. Explores multiple sequence lengths and diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a). Vectorized via FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _nesterov_opt(a, deadline, max_iter=40000, lr=0.02, beta_factor=0.04,
                  momentum=0.9, restart_threshold=6000):
    """Nesterov accelerated gradient in log-space for smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    lr_cur = lr
    v = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        z_look = z - momentum * v
        e = np.exp(np.clip(z_look, -60.0, 60.0))
        a_look = e / e.sum()

        c_curr = _fft_convolve(a_look)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a_look, w)

        # Project to tangent space
        dot = float(np.dot(a_look, grad_a))
        grad_z = a_look * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            break

        # Nesterov update
        v_new = momentum * v + lr_cur * grad_z
        z_new = z - v_new
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

def _pairwise_swap(a, deadline, max_iter=8000, step_frac=0.03):

                break

        a = a_new

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    # Linear profiles
    for eta in [0.8, 1.2, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + eta * u
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + alpha * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # U-shaped (cosine)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 - alpha * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Bell
    for width in [0.25, 0.4, 0.6]:
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cubic (Legendre P3-like)
    for c3 in [-2.0, -1.0, 1.0, 2.0]:
        p = 1.0 + c3 * (4.0 * u ** 3 - 3.0 * u) / 5.0
        a = np.maximum(p, 1e-12)
        profiles.append(a / a.sum())

    # Sine modulation
    for freq in [1, 2]:
        for amp in [0.3, 0.6]:
            a = 1.0 + amp * np.sin(freq * np.pi * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    return profiles


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _full_optimize(a0, deadline, n):
    """Run a full optimization pipeline on a single initialization."""
    best_a = a0.copy()
    best_score = _score(a0)

    # Phase 1: Nesterov (coarse)
    t1_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.45)
    a1, s1 = _nesterov_opt(a0, t1_end, max_iter=25000, lr=0.025,
                           beta_factor=0.05, momentum=0.9)
    if s1 < best_score:
        best_a, best_score = a1, s1

    # Phase 2: Multi-peak penalty
    if time.monotonic() < deadline - 30:
        t2_end = min(deadline - 30, time.monotonic() + (deadline - time.monotonic()) * 0.25)
        a2, s2 = _penalty_multi_peak(best_a, t2_end, max_iter=12000, top_k=5, lr=0.012)
        if s2 < best_score:
            best_a, best_score = a2, s2

    # Phase 3: Nesterov (fine)
    if time.monotonic() < deadline - 20:
        t3_end = min(deadline - 20, time.monotonic() + (deadline - time.monotonic()) * 0.2)
        a3, s3 = _nesterov_opt(best_a, t3_end, max_iter=15000, lr=0.008,
                               beta_factor=0.015, momentum=0.9, restart_threshold=4000)
        if s3 < best_score:
            best_a, best_score = a3, s3

    # Phase 4: Pairwise swap
    if time.monotonic() < deadline - 10:
        t4_end = min(deadline - 10, time.monotonic() + 20)
        a4, s4 = _pairwise_swap(best_a, t4_end, max_iter=4000, step_frac=0.02)
        if s4 < best_score:
            best_a, best_score = a4, s4

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining Nesterov accelerated gradient,
    multi-peak penalty, pairwise local search, and LP polishing.
    Explores sequence lengths 512-2048 with diverse profile initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define lengths to try, prioritizing those near the parent length
    lengths = [512, 768, 1024, 1280, 1536, 2048]
    total_time = deadline - t_start - 100  # reserve 100s for final polish
    n_lengths = len(lengths)

    # Allocate time: weight toward 1024 and 1536
    weights = [0.8, 1.0, 1.5, 1.0, 1.5, 1.0]
    w_sum = sum(weights)
    time_allocs = [total_time * w / w_sum for w in weights]

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 120:
            break

        d_len_end = t_start + sum(time_allocs[:li + 1]) + 50
        d_len_end = min(d_len_end, deadline - 120)

        if time.monotonic() > d_len_end:
            continue

        print(f"\n=== n={n} ===")

        # Prepare initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n))
            # Also a perturbed version
            rng_p = np.random.default_rng(seed + n + 1)
            pp = _interpolate_profile(parent, n)
            pp_pert = pp * (1.0 + 0.1 * rng_p.normal(size=n))
            pp_pert = np.maximum(pp_pert, 0.0)
            inits.append(pp_pert / pp_pert.sum())

        profiles = _make_profiles(n)
        # Take a diverse subset
        inits.extend(profiles[::2])

        # Limit number of inits based on available time
        avail_time = d_len_end - time.monotonic()
        max_inits = max(2, int(avail_time / 25))
        inits = inits[:max_inits]

        best_for_n = None
        best_score_n = np.inf

        for ii, a0 in enumerate(inits):
            if time.monotonic() > d_len_end - 15:
                break

            per_time = (d_len_end - 15 - time.monotonic()) / max(1, len(inits) - ii)
            d_init = min(d_len_end - 15, time.monotonic() + per_time)

            if per_time < 20:
                # Quick optimization only
                a_r, s_r = _nesterov_opt(a0, d_init, max_iter=5000,
                                         lr=0.02, beta_factor=0.05)
            else:
                a_r, s_r = _full_optimize(a0, d_init, n)

            if s_r < best_score_n:
                best_score_n = s_r
                best_for_n = a_r.copy()

            if ii < 10:
                print(f"  n={n} init={ii}: {s_r:.10f}")

        if best_for_n is not None and best_score_n < best_score:
            best_score = best_score_n
            best_seq = best_for_n.copy()
            print(f"  n={n} BEST: {best_score:.10f}")

    # === Final polishing on global best ===
    if best_seq is not None and time.monotonic() < deadline:
        print(f"\n=== Final polishing (current best={best_score:.10f}) ===")
        n_best = best_seq.size

        # LP refine
        if time.monotonic() < deadline - 30:
            d_lp = min(deadline - 30, time.monotonic() + 60)
            a_lp, s_lp = _lp_refine(best_seq, d_lp, max_iter=2000,
                                    inflate=1.0002, blend=0.2)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.10f}")

        # Fine Nesterov
        if time.monotonic() < deadline - 20:
            d_nl = min(deadline - 20, time.monotonic() + 50)
            a_nl, s_nl = _nesterov_opt(best_seq, d_nl, max_iter=15000,
                                       lr=0.005, beta_factor=0.008,
                                       momentum=0.9, restart_threshold=5000)
            if s_nl < best_score:
                best_score = s_nl
                best_seq = a_nl.copy()
            print(f"  After Fine Nesterov: {best_score:.10f}")

        # Multi-peak
        if time.monotonic() < deadline - 15:
            d_mp = min(deadline - 15, time.monotonic() + 30)
            a_mp, s_mp = _penalty_multi_peak(best_seq, d_mp, max_iter=8000,
                                             top_k=7, lr=0.008)
            if s_mp < best_score:
                best_score = s_mp
                best_seq = a_mp.copy()
            print(f"  After Multi-peak: {best_score:.10f}")

        # Pairwise swap (final)
        if time.monotonic() < deadline - 5:
            d_ps = min(deadline - 5, time.monotonic() + 20)
            a_ps, s_ps = _pairwise_swap(best_seq, d_ps, max_iter=5000, step_frac=0.01)
            if s_ps < best_score:
                best_score = s_ps
                best_seq = a_ps.copy()
            print(f"  After Pairwise: {best_score:.10f}")

        # Ultra-fine Nesterov
        if time.monotonic() < deadline:
            a_uf, s_uf = _nesterov_opt(best_seq, deadline, max_iter=8000,
                                       lr=0.003, beta_factor=0.004,
                                       momentum=0.9, restart_threshold=3000)
            if s_uf < best_score:
                best_score = s_uf
                best_seq = a_uf.copy()
            print(f"  After Ultra-fine: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profiles(256)[0]
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profiles(256)[0]

    # Validate
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence confirms: {sc_eval:.10f}")
        except Exception:
            pass

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
