**Practical algorithm: Subgradient descent with projection onto the simplex**

1. Start with a (on the simplex, a>=0, sum=1)
2. Compute conv = a * a (full)
3. Find the set S of indices where conv[k] is within epsilon of the max
4. Compute subgradient g = (1/|S|) * sum_{k in S} grad conv[k]
5. Update: a <- project_simplex(a - step * g)
6. Repeat

The projection onto the simplex (a>=0, sum=1) is well-known and efficient.

The step size can be:
- Fixed (decreasing)
- Polyak: step = (f(a) - f*) / ||g||^2 (requires knowing f*)
- Adaptive: step = c / sqrt(t)

"""
Cyclic Descent Method (CDM) with gradient refinement for minimizing
2n * max(conv(a,a)) / sum(a)^2 over non-negative sequences.
Uses exact 1D coordinate optimization followed by subgradient polishing.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def cdm_sweep(a, n_ternary=12, max_val_bound=None):
    """
    One CDM sweep: for each coordinate j, find the optimal value of a[j]
    given all other coordinates fixed. Uses ternary search.
    
    Returns updated a and convolution.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    # Normalize to sum=1
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    conv = np.convolve(a, a)
    conv = np.maximum(conv, 0.0)
    
    for j in range(n):
        old_x = a[j]
        
        # Build linear coefficient vector
        # For k in [j, j+n-1] (clipped to [0, 2n-2]):
        #   if k-j != j: lin[k] = 2*a[k-j]
        #   if k-j == j (k=2j): no linear term (handled as quadratic)
        lin = np.zeros(2*n-1)
        lo_k = j
        hi_k = min(2*n-1, j+n)  # exclusive upper bound
        if hi_k > lo_k:
            indices = np.arange(lo_k, hi_k)
            c_vals = indices - j  # = k - j, ranges from 0 to n-1
            # Exclude the diagonal
            mask = c_vals != j
            lin[indices[mask]] = 2.0 * a[c_vals[mask]]
        
        diag_pos = 2*j if 2*j < 2*n-1 else -1
        
        # Ternary search for optimal x in [0, upper]
        upper = 10.0 * max(old_x, 1.0/n)
        if max_val_bound is not None:
            upper = max(upper, max_val_bound)
        
        lo, hi = 0.0, upper
        best_x = old_x
        best_val = conv.max()
        
        for _ in range(n_ternary):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            
            # Evaluate at m1
            delta1 = m1 - old_x
            conv_m1 = conv + delta1 * lin
            if diag_pos >= 0:
                conv_m1 = conv_m1.copy()
                conv_m1[diag_pos] += m1*m1 - old_x*old_x
            val1 = float(conv_m1.max())
            
            # Evaluate at m2
            delta2 = m2 - old_x
            conv_m2 = conv + delta2 * lin
            if diag_pos >= 0:
                conv_m2 = conv_m2.copy()
                conv_m2[diag_pos] += m2*m2 - old_x*old_x
            val2 = float(conv_m2.max())
            
            if val1 < best_val:
                best_val = val1
                best_x = m1
            if val2 < best_val:
                best_val = val2
                best_x = m2
            
            if val1 < val2:
                hi = m2
            else:
                lo = m1
        
        # Update a[j] and conv
        a[j] = best_x
        delta = best_x - old_x
        if abs(delta) > 1e-15:
            conv = conv + delta * lin
            if diag_pos >= 0:
                conv[diag_pos] += best_x*best_x - old_x*old_x
            conv = np.maximum(conv, 0.0)
    
    return a, conv


def cdm_optimize(a0, n_sweeps, n_ternary=12, deadline=None):
    """Run multiple CDM sweeps."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    prev_score = best_score
    stall_count = 0
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a, conv = cdm_sweep(a, n_ternary=n_ternary)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # If stalled for many sweeps, break
        if stall_count > 10:
            break
        
        if sweep % 20 == 0:
            print(f"    CDM sweep {sweep}: score={score:.10f}")
    
    return best_a, best_score

So (A)_{j,k} = f[k] * 1_{0<=j-k<n} = f[k] * 1_{j-n+1<=k<=j} * 1_{0<=k<n}.

This is a banded matrix (each row has at most n nonzeros, each column has at most n nonzeros).

The LP is:
max 1^T g
s.t. A g <= M * 1  (2n-1 constraints)
     g >= 0  (n constraints)

Using scipy.optimize.linprog:

from scipy.optimize import linprog

A_ub = A  # (2n-1) x n
b_ub = M * np.ones(2n-1)

res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
g_0 = res.x

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2:
1. LP-based direction finding (solves dual LP for optimal perturbation direction)
2. Cyclic Descent Method (exact 1D coordinate optimization)
3. Subgradient polishing
Multi-length search with interpolation and diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (A*g)[j] = (f*g)[j]."""
    A = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        # (f*g)[j] = sum_k f[k]*g[j-k] for 0<=j-k<n
        k_min = max(0, j - n + 1)
        k_max = min(j, n - 1)
        if k_min <= k_max:
            A[j, k_min:k_max+1] = f[k_min:k_max+1]
    return A


def lp_find_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[j] <= M for all j, g >= 0.
    Returns the optimal direction g (normalized to sum=1) or None if infeasible/trivial.
    """
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(a0, deadline, max_iters=50, lr_range=(0.1, 0.5)):
    """Iteratively find LP directions and take steps."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve_fft(a)
        M = float(c.max())
        
        g = lp_find_direction(a, M, n)
        if g is None:
            break
        
        # Line search for step size
        best_t = 0.0
        best_sc = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = compute_score(a_try)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.05]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
