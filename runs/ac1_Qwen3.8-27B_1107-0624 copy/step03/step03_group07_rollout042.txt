```python
def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)
```

```python
"""
Nesterov-accelerated subgradient optimization with flat-spectrum initialization,
pairwise coordinate descent polishing, and multi-scale search for minimizing
2n * max(convolve(a,a)) / sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    k = 1
    while k < 2 * n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    return np.maximum(c, 0.0)

def convolve_subgradient(m, a):
    """Subgradient of c[m] w.r.t. a at the point a.
    dc[m]/da[k] = a[m-k] + a[k-m] (zero-padding outside [0, n-1])
    """
    n = len(a)
    grad = np.zeros(n)
    # First term: a[m-k] for k where 0 <= m-k <= n-1
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] += a[m - indices]
    # Second term: a[k-m] for k where 0 <= k-m <= n-1
    lo2 = max(0, m)
    hi2 = min(n - 1, m + n - 1)
    if lo2 <= hi2:
        indices = np.arange(lo2, hi2 + 1)
        grad[indices] += a[indices - m]
    return grad

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(ind[cond][-1])
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def flat_spectrum_init(n, rng):
    """Initialize with approximately flat amplitude spectrum."""
    phi = np.zeros(n)
    if n % 2 == 0:
        phi_mid = n // 2
        phi[1:phi_mid] = rng.uniform(-np.pi, np.pi, phi_mid - 1)
        phi[phi_mid+1:] = -phi[phi_mid-1:0:-1]
    else:
        phi_mid = (n - 1) // 2
        phi[1:phi_mid+1] = rng.uniform(-np.pi, np.pi, phi_mid)
        phi[phi_mid+1:] = -phi[phi_mid:0:-1]
    
    A = np.exp(1j * phi)
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def make_u_shape(n, alpha=2.0):
    """U-shaped profile."""
    t = np.linspace(0, 1, n)
    a = t**alpha + (1-t)**alpha + 0.1
    return a / a.sum()

def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def nesterov_subgrad_optimize(a0, deadline, rng, top_k=3, base_lr=0.005,
                               decay=0.9999, f_star=1.0, restart_period=300):
    """
    Nesterov-accelerated subgradient method with Polyak step size.
    Minimizes 2n * max(conv(a,a)) s.t. a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    n = a.size
    
    # Nesterov state
    x_prev = a.copy()
    x_curr = a.copy()
    y = a.copy()  # lookahead point
    
    c_y = full_convolve_fft(y)
    best_a = a.copy()
    best_score = float(2.0 * n * c_y.max())
    
    no_improve = 0
    t = 0
    current_lr = base_lr
    
    while time.monotonic() < deadline:
        t += 1
        
        # Nesterov momentum
        if t > 1:
            beta = 2.0 / (t + 3.0)
            y = x_curr + beta * (x_curr - x_prev)
            y = project_simplex(y)
        
        # Compute subgradient at y
        c_y = full_convolve_fft(y)
        max_val = c_y.max()
        
        if top_k >= len(c_y):
            top_indices = np.argsort(c_y)[-1:]
        else:
            top_indices = np.argsort(c_y)[-top_k:]
        
        g = np.zeros(n)
        for idx in top_indices:
            g += convolve_subgradient(int(idx), y)
        g /= float(len(top_indices))
        
        # Scale gradient to objective
        # Objective: f(a) = 2n * max(c(a)), so subgrad = 2n * (subgrad of max(c))
        g = 2.0 * n * g
        
        # Polyak step size
        f_curr = 2.0 * n * max_val
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq > 1e-20:
            polyak_lr = (f_curr - f_star) / g_norm_sq
            # Blend with current_lr for stability
            current_lr = 0.5 * current_lr + 0.5 * min(polyak_lr, 0.1)
        else:
            current_lr *= decay
        
        # Update
        x_new = project_simplex(y - current_lr * g)
        
        x_prev = x_curr
        x_curr = x_new
        
        # Evaluate at x_new (not y)
        c_new = full_convolve_fft(x_new)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart
        if no_improve > restart_period:
            x_curr = best_a + 0.005 * rng.normal(size=n)
            x_curr = project_simplex(x_curr)
            x_prev = x_curr.copy()
            y = x_curr.copy()
            current_lr = base_lr * (0.5 + rng.random())
            no_improve = 0
    
    return best_a, best_score

def softmax_subgrad_optimize(a0, deadline, rng, top_k=5, base_lr=0.01,
                              decay=0.9998, restart_period=400, f_star=1.0):
    """
    Subgradient method with weighted top-k (soft but exact-ish).
    Uses a concentration parameter that adapts over time.
    """
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    c = full_convolve_fft(a)
    best_score = float(2.0 * n * c.max())
    
    no_improve = 0
    t = 0
    current_lr = base_lr
    concentration = top_k  # starts broad, narrows over time
    
    # Adam-like state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    while time.monotonic() < deadline:
        t += 1
        c = full_convolve_fft(a)
        max_val = c.max()
        
        # Weighted subgradient: weight by exp(-lambda * (max - c))
        # This gives more weight to near-max entries
        lambda_param = max(1.0 / concentration, 0.1)
        weights = np.exp(-lambda_param * (max_val - c))
        weights_sum = weights.sum()
        if weights_sum < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = weights / weights_sum
        
        # Gradient is weighted average of subgradients
        # But computing all 2n-1 subgradients is expensive.
        # Instead, use the fact that the weighted sum of subgradients can be computed via FFT.
        # grad[k] = 2n * sum_m w[m] * (a[m-k] + a[k-m])
        #          = 2n * (conv(w, a)[k] + conv(w, a_flipped)[k + something])
        # Actually: sum_m w[m] * a[m-k] = (w conv a)[k] (with appropriate indexing)
        # And: sum_m w[m] * a[k-m] = (w_flipped conv a)[k] (with appropriate indexing)
        
        # Let's compute via the relation:
        # grad[k] = 2n * (sum_m w[m] * a[m-k] + sum_m w[m] * a[k-m])
        # First sum: for fixed k, sum over m where a[m-k] is valid
        # This is essentially a cross-correlation of w and a.
        # sum_m w[m] * a[m-k] = (w cross-correlate a)[k] = sum_m w[m] * a[m-k]
        # Which is the same as sum_j w[j+k] * a[j] = (shifted_w conv a_flipped)[k]
        
        # Let me just use FFT:
        # h1[k] = sum_m w[m] * a[m-k] = cross-correlation(w, a) at lag k
        # h2[k] = sum_m w[m] * a[k-m] = sum_m w[m] * a[m-k] with a replaced by a_flipped
        # Actually h2[k] = sum_m w[m] * a[k-m]
        # Let j = k-m, then m = k-j: sum_j w[k-j] * a[j] = (w conv a)[k] (valid range)
        
        # h1[k] = sum_m w[m] * a[m-k] = sum_j w[j+k] * a[j] 
        # Let's use the full convolution: (w * a_flipped)[n-1+k] where a_flipped = a[::-1]
        # (w * a_flipped)[m] = sum_j w[j] * a_flipped[m-j] = sum_j w[j] * a[n-1-m+j]
        # Hmm this is getting confusing. Let me just use the direct formula with FFT.
        
        # Actually, the simplest approach:
        # grad_a[k] = 2n * (T1[k] + T2[k])
        # T1[k] = sum_m w[m] * a[m-k] (valid when 0<=m-k<n)
        # T2[k] = sum_m w[m] * a[k-m] (valid when 0<=k-m<n)
        
        # T1[k] = sum_j a[j] * w[j+k] where j is in [0, n-1] and j+k is in [0, len(w)-1]
        # This is the cross-correlation of a and w, evaluated at lag k.
        # cross_corr(a, w)[k] = sum_j a[j] * w[j+k]
        
        # T2[k] = sum_j a[j] * w[k-j] where j is in [0, n-1] and k-j is in [0, len(w)-1]
        # This is the convolution of a and w, evaluated at k.
        # (a * w)[k] = sum_j a[j] * w[k-j]
        
        # So grad_a = 2n * (cross_corr(a, w) + (a * w))
        
        # Both can be computed via FFT.
        N = 1
        while N < n + len(w) - 1:
            N <<= 1
        
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w, N)
        
        # (a * w)[k] = ifft(fa * fw)[k]
        conv_aw = np.fft.irfft(fa * fw, N)[:n]
        
        # cross_corr(a, w)[k] = sum_j a[j] * w[j+k]
        # = (a_flipped * w)[n-1+k] where a_flipped = a[::-1]
        # Or: cross_corr(a, w)[k] = sum_j a[j+k] * w[j] = (a * w_flipped)[k] hmm
        # Actually: cross_corr(a, w)[k] = sum_j a[j] w[j+k]
        # Let's use: = ifft(conj(fa) * fw) ... no that's for the case where both are same length.
        
        # Simpler: cross_corr(a, w)[k] = sum_j a[j] w[j+k]
        # This is the same as the k-th coefficient of the "reversed convolution"
        # = (a * w_rev)[n-1-k + n-1] ... ugh.
        
        # Let me just use matmul for the cross-correlation when n is small enough,
        # or use the FFT trick properly.
        
        # cross_corr(a, w)[k] = sum_{j=0}^{n-1} a[j] * w[j+k]
        # Define a_rev = a[::-1], then:
        # (a_rev * w)[n-1+k] = sum_j a_rev[j] * w[n-1+k-j] = sum_j a[n-1-j] * w[n-1+k-j]
        # Let i = n-1-j: = sum_i a[i] * w[k+i] = cross_corr(a, w)[k] ✓
        
        fa_rev = np.fft.rfft(a[::-1], N)
        cross_conv = np.fft.irfft(fa_rev * fw, N)
        # cross_corr(a, w)[k] = cross_conv[n-1+k] for k = 0, ..., n-1
        t1 = cross_conv[n-1:2*n-1]
        t2 = conv_aw[:n]  # (a*w)[k] for k = 0, ..., n-1
        
        grad_a = 2.0 * n * (t1 + t2)
        
        # Project to tangent space (zero sum direction)
        grad_a = grad_a - grad_a.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * (grad_a * grad_a)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        step = current_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Apply step with projection
        a_new = project_simplex(a - step)
        
        # Evaluate
        c_new = full_convolve_fft(a_new)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt
        if t % 100 == 0:
            concentration = max(1.0, concentration * 0.98)
        
        if no_improve > 200:
            current_lr = max(0.001, current_lr * 0.95)
        
        if no_improve > restart_period:
            a = best_a + 0.01 * rng.normal(size=n)
            a = project_simplex(a)
            m = np.zeros(n)
            v = np.zeros(n)
            current_lr = base_lr
            concentration = top_k
            no_improve = 0
    
    return best_a, best_score

def pairwise_coordinate_descent(a0, deadline, rng, n_passes=10):
    """
    Pairwise coordinate descent: pick pairs (i,j) and redistribute mass.
    """
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())
    
    for passage in range(n_passes):
        if time.monotonic() > deadline:
            break
        # Create a random permutation of indices
        perm = rng.permutation(n)
        for pass_i in range(n // 2):
            if time.monotonic() > deadline - 1:
                break
            i = int(perm[2 * pass_i % n])
            j = int(perm[(2 * pass_i + 1) % n])
            if i == j:
                continue
            
            s = a[i] + a[j]
            if s < 1e-15:
                continue
            
            # Try several splits of s between i and j
            # Current split: a[i], a[j]
            # Try: 0, s/4, s/2, 3s/4, s (for a[i])
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
            best_split = s * a[i] / s if s > 0 else 0  # current
            best_local = compute_score(a)
            
            for frac in candidates:
                a_trial = a.copy()
                a_trial[i] = frac * s
                a_trial[j] = (1 - frac) * s
                sc = float(2.0 * n * full_convolve_fft(a_trial).max())
                if sc < best_local:
                    best_local = sc
                    best_split = frac * s
            
            a[i] = best_split
            a[j] = s - best_split
        
        # Check improvement
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def scipy_polish(a0, deadline, rng):
    """Use scipy L-BFGS-B to polish a smooth approximation."""
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())
    
    # Work in log-space for scipy
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    
    def objective(z):
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        c = full_convolve_fft(a_tmp)
        return 2.0 * n * c.max()
    
    def gradient(z):
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        c = full_convolve_fft(a_tmp)
        mval = c.max()
        
        # Soft-max with very small beta for differentiability
        beta = 0.01 * mval
        x = (c - mval) / beta
        x = np.clip(x, -50, 0)
        ex = np.exp(x)
        w = ex / ex.sum()
        
        # Gradient computation (same as before)
        N = 1
        while N < n + len(c) - 1:
            N <<= 1
```

```python
def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:

    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    k = 1
    while k < 2 * n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    return np.maximum(c, 0.0)


def convolve_subgradient(m, a):

    grad = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] += a[m - indices]
    lo2 = max(0, m)
    hi2 = min(n - 1, m + n - 1)
    if lo2 <= hi2:
        indices = np.arange(lo2, hi2 + 1)
        grad[indices] += a[indices - m]
    return grad


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(ind[cond][-1])
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def compute_weighted_gradient(a, w):
    """Compute sum_m w[m] * dc[m]/da for all components of a via FFT.
    w has length 2n-1 (the full convolution length).
    Returns grad of length n where grad[k] = 2n * sum_m w[m]*(a[m-k]+a[k-m]).
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fa_flip = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    # t1[k] = sum_j a[j]*w[j+k] = (a_flip * w)[n-1+k]
    conv_flip = np.fft.irfft(fa_flip * fw, N)
    t1 = conv_flip[n - 1:2 * n - 1]
    # t2[k] = sum_j a[j]*w[k-j] = (a * w)[k]
    conv_fwd = np.fft.irfft(fa * fw, N)
    t2 = conv_fwd[:n]
    return 2.0 * n * (t1 + t2)


def flat_spectrum_init(n, rng):
    phi = np.zeros(n)
    if n % 2 == 0:
        pm = n // 2
        if pm > 1:
            phi[1:pm] = rng.uniform(-np.pi, np.pi, pm - 1)
            phi[pm + 1:] = -phi[pm - 1:0:-1]
    else:
        pm = (n - 1) // 2
        if pm > 0:
            phi[1:pm + 1] = rng.uniform(-np.pi, np.pi, pm)
            phi[pm + 1:] = -phi[pm:0:-1]
    A = np.exp(1j * phi)
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def make_u_shape(n, alpha=2.0):
    t = np.linspace(0, 1, n)
    a = t ** alpha + (1 - t) ** alpha + 0.05
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_random_smooth(n, rng, n_harmonics=6):
    coeffs = rng.uniform(-0.4, 0.4, n_harmonics)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, cv in enumerate(coeffs, 1):
        a = a + cv * np.cos(2 * np.pi * i * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def soft_weighted_adam_optimize(a0, deadline, rng, base_lr=0.01,
                                 concentration_init=5.0, concentration_decay=0.995,
                                 restart_period=350, concentration_min=1.0):
    """
    Soft-weighted subgradient with Adam optimizer in logit space.
    The concentration parameter controls how "peaked" the weighting is.
    """
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    n = a.size

    z = np.log(a)
    z = z - z.mean()

    c = full_convolve_fft(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    no_improve = 0
    current_lr = base_lr
    concentration = concentration_init

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        max_val = float(c.max())

        # Compute weights: soft-max over convolution values
        beta = max(max_val / concentration, 1e-10)
        x = (c - max_val) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Compute gradient via FFT
        grad_a = compute_weighted_gradient(a, w)

        # Convert to logit-space gradient
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Stabilize
        if not np.all(np.isfinite(z)):
            z = np.log(best_a)
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a = np.exp(z - z.max())
        s_a = a.sum()
        if s_a < 1e-15:
            a = best_a.copy()
            z = np.log(a)
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue
        a = a / s_a

        c_new = full_convolve_fft(a)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt concentration (sharpen over time)
        if t_adam % 50 == 0:
            concentration = max(concentration_min, concentration * concentration_decay)

        # Adapt learning rate
        if no_improve > 200:
            current_lr = max(0.002, current_lr * 0.97)

        # Restart
        if no_improve > restart_period:
            # Try different restart strategies
            strategy = rng.random()
            if strategy < 0.4:
                # Small perturbation
                z = np.log(best_a)
                z = z - z.mean()
                z += 0.05 * rng.normal(size=n)
            elif strategy < 0.7:
                # Medium perturbation with structure
                z = np.log(best_a)
                z = z - z.mean()
                noise = rng.normal(size=n)
                # Low-frequency noise
                nz = np.fft.rfft(noise)
                cutoff = max(2, n // 8)
                nz[cutoff:] *= 0.1
                z += 0.15 * np.fft.irfft(nz, n)
            else:
                # Larger perturbation
                z = np.log(best_a)
                z = z - z.mean()
                z += 0.3 * rng.normal(size=n)

            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            current_lr = base_lr * (0.5 + 0.5 * rng.random())
            concentration = concentration_init
            no_improve = 0

    return best_a, best_score


def nesterov_exact_subgrad(a0, deadline, rng, base_lr=0.002, top_k=5,
                            polyak_blend=0.3, restart_period=250):
    """
    Nesterov-accelerated exact subgradient method with Polyak step size.
    """
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    n = a.size

    x_curr = a.copy()
    x_prev = a.copy()
    y = a.copy()

    c_y = full_convolve_fft(y)
    best_a = a.copy()
    best_score = float(2.0 * n * c_y.max())

    no_improve = 0
    t = 0
    current_lr = base_lr
    f_star = 1.0  # theoretical lower bound

    while time.monotonic() < deadline:
        t += 1

        # Nesterov lookahead
        if t > 1:
            beta = 2.0 / (t + 3.0)
            y = x_curr + beta * (x_curr - x_prev)
            y = project_simplex(y)

        c_y = full_convolve_fft(y)
        max_val = float(c_y.max())

        if top_k >= len(c_y):
            top_indices = np.argsort(c_y)[-1:]
        else:
            top_indices = np.argsort(c_y)[-top_k:]

        g = np.zeros(n)
        for idx in top_indices:
            g += convolve_subgradient(int(idx), y)
        g /= float(len(top_indices))
        g *= 2.0 * n  # scale to objective

        # Polyak step size
        f_curr = 2.0 * n * max_val
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq > 1e-20:
            polyak_lr = (f_curr - f_star) / g_norm_sq
            polyak_lr = min(polyak_lr, 0.05)  # cap for stability
            current_lr = (1.0 - polyak_blend) * current_lr * 0.9999 + polyak_blend * polyak_lr
        else:
            current_lr *= 0.99

        # Update with projection
        x_new = project_simplex(y - current_lr * g)

        x_prev = x_curr
        x_curr = x_new

        c_new = full_convolve_fft(x_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > restart_period:
            x_curr = best_a + 0.01 * rng.normal(size=n)
            x_curr = project_simplex(x_curr)
            x_prev = x_curr.copy()
            y = x_curr.copy()
            current_lr = base_lr * (0.5 + rng.random())
            no_improve = 0

    return best_a, best_score


def pairwise_cd(a0, deadline, rng, n_passes=5):
    """Pairwise coordinate descent for polishing."""
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())

    splits = [0.0, 0.2, 0.35, 0.5, 0.65, 0.8, 1.0]

    for passage in range(n_passes):
        if time.monotonic() > deadline:
            break
        perm = rng.permutation(2 * n)
        for pi in range(n):
            if time.monotonic() > deadline - 1:
                break
            i = int(perm[2 * pi] % n)
            j = int(perm[2 * pi + 1] % n)
            if i == j:
                continue
            s = a[i] + a[j]
            if s < 1e-15:
                continue

            best_local = float(2.0 * n * full_convolve_fft(a).max())
            best_frac = a[i] / s if s > 0 else 0.5

            for frac in splits:
                new_val = frac * s
                if abs(new_val - a[i]) < 1e-15:
                    continue
                a[i] = new_val
                a[j] = s - new_val
                sc = float(2.0 * n * full_convolve_fft(a).max())
                if sc < best_local:
                    best_local = sc
                    best_frac = frac

            a[i] = best_frac * s
            a[j] = s * (1.0 - best_frac)

        sc = float(2.0 * n * full_convolve_fft(a).max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def scipy_smoothed_polish(a0, deadline, rng):
    """Use scipy L-BFGS-B on a very smooth approximation."""
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())

    z0 = np.log(a).copy()
    z0 = z0 - z0.mean()

    beta = 0.02  # smoothing parameter

    def objective(z):
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        c = full_convolve_fft(a_tmp)
        mval = c.max()
        x = (c - mval) / beta
        x = np.clip(x, -50, 0)
        ex = np.exp(x)
        return 2.0 * n * (mval + beta * np.log(ex.sum()))

    def gradient(z):
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        c = full_convolve_fft(a_tmp)
        mval = c.max()
        x = (c - mval) / beta
        x = np.clip(x, -60, 0)
        ex = np.exp(x)
        s_ex = ex.sum()
        w = ex / s_ex
        grad_a = compute_weighted_gradient(a_tmp, w)
        dot = float(np.dot(a_tmp, grad_a))
        return a_tmp * (grad_a - dot)

    try:
        result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': 300, 'gtol': 1e-10, 'maxfun': 800,
                                  'ftol': 1e-12})
        if result.x is not None:
            z_best = result.x
            a_best = np.exp(z_best - z_best.max())
            a_best = a_best / a_best.sum()
            sc = float(2.0 * n * full_convolve_fft(a_best).max())
            if sc < best_score:
                best_score = sc
                best_a = a_best.copy()
    except Exception:
        pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining soft-weighted Adam, Nesterov subgradient,
    pairwise coordinate descent, and scipy polishing. Uses diverse initializations
    including flat-spectrum starts and multi-scale refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_global = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score_global
        if p is not None and np.isfinite(sc) and sc < best_score_global:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score_global = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting multi-phase search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial (height_sequence_1): n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick multi-n exploration (0 - 120s) =====
    phase1_end = t_start + 120.0
    print("\n--- Phase 1: Exploration ---")

    n_values_phase1 = [256, 384, 512, 768]
    best_p1 = None
    best_s1 = np.inf

    for n_val in n_values_phase1:
        if time.monotonic() > phase1_end - 5:
            break
        inits = [
            make_u_shape(n_val, 2.0),
            make_u_shape(n_val, 3.0),
            make_cosine_profile(n_val, 1, 0.6),
            make_cosine_profile(n_val, 2, 0.4),
            make_linear_profile(n_val, 2.0),
            flat_spectrum_init(n_val, rng),
            flat_spectrum_init(n_val, rng),
            make_random_smooth(n_val, rng),
        ]
        if init_seq is not None:
            inits.insert(0, interpolate_profile(init_seq, n_val))

        for init in inits:
            sc = compute_score(init)
            if sc < best_s1:
                best_s1 = sc
                best_p1 = init.copy()
            update(init, sc)

    print(f"  Best exploration: n={best_p1.size if best_p1 is not None else '?'}, score={best_s1:.10f}")

    # ===== Phase 2: Main optimization at n=512-1024 (120s - 600s) =====
    phase2_end = t_start + 600.0
    print("\n--- Phase 2: Main optimization ---")

    # Start from best found so far
    if init_seq is not None and init_seq.size in [512, 1024]:
        p2 = init_seq.copy()
    elif best_p1 is not None:
        p2 = interpolate_profile(best_p1, 512)
    else:
        p2 = make_u_shape(512, 2.0)

    n2 = p2.size
    print(f"  Optimizing at n={n2}")

    # Soft-weighted Adam
    dl2a = min(phase2_end, time.monotonic() + 200.0)
    r2a, sc2a = soft_weighted_adam_optimize(p2, dl2a, rng, base_lr=0.012,
                                             concentration_init=8.0, concentration_decay=0.996,
                                             restart_period=300)
    sc2a_final = compute_score(r2a)
    update(r2a, sc2a_final)
    print(f"  Soft-weighted Adam (n={n2}): {sc2a_final:.10f}")

    # Nesterov exact subgradient
    if time.monotonic() < phase2_end - 60:
        dl2b = min(phase2_end, time.monotonic() + 120.0)
        r2b, sc2b = nesterov_exact_subgrad(r2a, dl2b, rng, base_lr=0.002,
                                            top_k=3, restart_period=200)
        sc2b_final = compute_score(r2b)
        update(r2b, sc2b_final)
        print(f"  Nesterov subgrad (n={n2}): {sc2b_final:.10f}")
        if sc2b_final < sc2a_final:
            r2a = r2b

    # ===== Phase 3: Scale to larger n (600s - 1200s) =====
    phase3_end = t_start + 1200.0
    print("\n--- Phase 3: Scale up ---")

    # Try n=1024
    if time.monotonic() < phase3_end - 100:
        p3 = interpolate_profile(best_seq if best_seq is not None else r2a, 1024)
        sc3_init = compute_score(p3)
        update(p3, sc3_init)
        print(f"  Interpolated to n=1024: {sc3_init:.10f}")

        dl3 = min(phase3_end, time.monotonic() + 250.0)
        r3, sc3 = soft_weighted_adam_optimize(p3, dl3, rng, base_lr=0.008,
                                               concentration_init=6.0, concentration_decay=0.997,
                                               restart_period=400)
        sc3_final = compute_score(r3)
        update(r3, sc3_final)
        print(f"  Soft-weighted Adam (n=1024): {sc3_final:.10f}")

    # Try n=2048
    n4 = 2048
    if time.monotonic() < phase3_end - 100:
        print(f"  Trying n={n4}...")
        p4 = interpolate_profile(best_seq if best_seq is not None else p3, n4)
        sc4_init = compute_score(p4)
        update(p4, sc4_init)
        print(f"  Interpolated to n={n4}: {sc4_init:.10f}")

        if time.monotonic() < phase3_end - 60:
            dl4 = min(phase3_end, time.monotonic() + 150.0)
            r4, sc4 = soft_weighted_adam_optimize(p4, dl4, rng, base_lr=0.006,
                                                   concentration_init=5.0, concentration_decay=0.998,
                                                   restart_period=400)
            sc4_final = compute_score(r4)
            update(r4, sc4_final)
            print(f"  Soft-weighted Adam (n={n4}): {sc4_final:.10f}")

    # Try n=768 (non-power-of-2)
    if time.monotonic() < phase3_end - 50:
        n_alt = 768
        p_alt = interpolate_profile(best_seq if best_seq is not None else r2a, n_alt)
        sc_alt = compute_score(p_alt)
        update(p_alt, sc_alt)
        if sc_alt < best_score_global + 0.01:
            dl_alt = min(phase3_end, time.monotonic() + 60.0)
            r_alt, sc_alt2 = soft_weighted_adam_optimize(p_alt, dl_alt, rng, base_lr=0.008,
                                                          concentration_init=5.0,
                                                          concentration_decay=0.996)
            sc_alt3 = compute_score(r_alt)
            update(r_alt, sc_alt3)
            print(f"  n={n_alt} optimization: {sc_alt3:.10f}")

    # ===== Phase 4: Polish at best n (1200s - 1650s) =====
    phase4_end = t_start + 1650.0
    print("\n--- Phase 4: Polishing ---")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        n_polish = best_seq.size
        print(f"  Polishing at n={n_polish}, current best={best_score_global:.10f}")

        # Multiple rounds of soft-weighted Adam with decreasing lr
        for rnd, (lr_val, conc) in enumerate([(0.006, 3.0), (0.004, 2.0), (0.002, 1.5)]):
            if time.monotonic() > phase4_end - 30:
                break
            p_rnd = best_seq + 0.002 * rng.normal(size=n_polish)
            p_rnd = np.maximum(p_rnd, 1e-12)
            p_rnd = p_rnd / p_rnd.sum()
            dl_rnd = min(phase4_end, time.monotonic() + 80.0)
            r_rnd, _ = soft_weighted_adam_optimize(p_rnd, dl_rnd, rng,
                                                     base_lr=lr_val,
                                                     concentration_init=conc,
                                                     concentration_decay=0.998,
                                                     restart_period=200)
            sc_rnd = compute_score(r_rnd)
            update(r_rnd, sc_rnd)
            print(f"  Round {rnd} (lr={lr_val}, conc={conc}): {sc_rnd:.10f}")

        # Pairwise coordinate descent
        if time.monotonic() < phase4_end - 20:
            dl_cd = min(phase4_end, time.monotonic() + 40.0)
            if best_seq.size <= 1024:
                r_cd, _ = pairwise_cd(best_seq, dl_cd, rng, n_passes=3)
                sc_cd = compute_score(r_cd)
                update(r_cd, sc_cd)
                print(f"  Pairwise CD: {sc_cd:.10f}")

        # Scipy polish
        if time.monotonic() < phase4_end - 10:
            dl_sp = min(phase4_end, time.monotonic() + 20.0)
            r_sp, _ = scipy_smoothed_polish(best_seq, dl_sp, rng)
            sc_sp = compute_score(r_sp)
            update(r_sp, sc_sp)
            print(f"  Scipy polish: {sc_sp:.10f}")

    # ===== Phase 5: Final attempts (1650s - deadline) =====
    print("\n--- Phase 5: Final ---")

    # Try a fresh start with a different strategy at the best n
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = best_seq.size
        # Try with a blended initialization
        if init_seq is not None:
            blend = 0.5 * interpolate_profile(init_seq, n_final) + 0.5 * best_seq
            blend = blend / blend.sum()
        else:
            blend = best_seq.copy()

        dl_fin = min(deadline, time.monotonic() + 30.0)
        r_fin, _ = soft_weighted_adam_optimize(blend, dl_fin, rng, base_lr=0.005,
                                                 concentration_init=4.0,
                                                 concentration_decay=0.999)
        sc_fin = compute_score(r_fin)
        update(r_fin, sc_fin)
        print(f"  Final attempt: {sc_fin:.10f}")

    print(f"\n  FINAL BEST SCORE: {best_score_global:.10f}")

    # Final validation and return
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_u_shape(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_u_shape(256, 2.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
```
