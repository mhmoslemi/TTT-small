"""
Improved hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: multi-length search (768, 1024, 1536), refined LP with
adaptive inflation, coordinate-descent peak pressure method, L-BFGS-B polish,
and better beta annealing schedule.

import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_shape(n, gamma=2.0):
    """U-shaped profile: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    dist = np.abs(t - 0.5) * 2.0  # 0 at center, 1 at edges
    a = 1.0 + gamma * (dist ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_shape(n, gamma=1.5):
    """W-shaped profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + gamma * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 1e-12)

def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_symmetric_bell(n, width=0.4, edge_boost=0.5):

    dist = np.abs(t - 0.5)
    a = np.exp(-(dist / width) ** 2) + edge_boost * np.exp(-dist**2 / 0.02)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _logit_adam_optimize(a, deadline, max_iter=50000, lr=0.05, beta_init=None):
    """Adam optimization in logit space with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.05 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            continue

        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_candidate = dot_ss / max(abs(dot_sg), 1e-20)
                bb_candidate = np.clip(bb_candidate, 0.1, 10.0)
                bb_scale = 0.7 * bb_scale + 0.3 * bb_candidate
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.003, base_lr * 0.95)

        # Adaptive beta with smoother schedule
        if it % 50 == 0 and it > 50:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif peak_gap > beta * 8.0:
                    beta = min(0.5 * c_max_new, beta * 1.3)

        # Annealing restart
        if no_imp > max(5000, 10 * n):
            noise_scale = 0.2 * best_score
            rng_r = np.random.default_rng(it * 7919 + 1)
            z_restart = best_z + noise_scale * rng_r.standard_normal(n)
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

def _lp_refine(a, deadline, max_iter=20000, inflate=1.0006, blend=0.25):
    """LP-based spreading refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.003, inflate_cur * 1.04)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Gradually tighten inflation as we improve
            inflate_cur = max(1.0001, inflate_cur * 0.999)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.95)
                inflate_cur = min(1.002, inflate_cur * 0.97)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _peak_pressure_descent(a, deadline, max_iter=10000, n_peaks=8, strength=0.04):
    """Coordinate-descent peak pressure method."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / wsum

        # Compute pressure using FFT for efficiency
        # pressure[i] = sum_k w_k * sum_j a[j] * a[k-j] where k-j=i or j=i-k
        # More precisely: pressure[i] = sum_k w_k * (a convolved with delta at k)[i]
        # = sum_k w_k * sum_j a[j] * delta[k - j - i] ... 
        # Actually pressure[i] = sum_k w_k * sum_{j: k-j valid} a[j]*a[k-j] 
        # The contribution to a[i] from peak k is: sum of a[j]*a[k-j] where i is involved
        # = a[i] * (sum of a[k-i]) + (sum of a[i-j]) * a[k-i] ... 
        # Let's use the direct formula: d(conv)_k / d(a[i]) = a[k-i] + a[i-k] (if valid)
        # So pressure[i] = sum_k w_k * (a[max(0,k-n+1):k+1] involvement)
        # Simpler: pressure[i] = sum_k w_k * (a[(k-i) mod stuff])
        
        # Use the gradient approach: for peak at position k,
        # d c[k] / d a[i] = a[k-i] if 0<=k-i<n else 0  (plus symmetric term)
        # Actually for autocorrelation: c[k] = sum_j a[j]*a[j+k]
        # d c[k] / d a[i] = a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
        # For the full convolution: c[k] = sum_j a[j]*a[k-j]
        # d c[k] / d a[i] = a[k-i] (if 0 <= k-i < n)
        
        # Compute pressure efficiently using convolution with weights
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_idx] = weights
        # pressure[i] = sum_k w_vec[k] * a[k-i] if valid
        # This is a cross-correlation of w_vec with a
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w_vec, N)
        fa = np.fft.rfft(a, N)
        cross = np.fft.irfft(fw * fa, N)
        pressure = cross[n-1:2*n-1]  # extract valid n components
        
        # Alternative: direct computation for small n
        if n <= 2048:
            pressure = np.zeros(n)
            for k, w in zip(top_idx, weights):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    j = k - i
                    pressure[i] += w * a[j]

        pmax = pressure.max()
        if pmax < 1e-15:
            no_imp += 1
            if no_imp > 500:
                break
            continue

        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.15, str_cur * 1.005)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.01, str_cur * 0.9)
            if no_imp > max(800, n):
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _lbfgs_polish(a, deadline, max_iter=500):
    """L-BFGS-B polish in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    beta = max(1e-6, 0.02 * float(_fft_convolve(a).max()))

    def objective(z):
        e = np.exp(np.clip(z, -60, 60))
        aa = e / e.sum()
        c = _fft_convolve(aa)
        _, w = _smooth_max_val(c, beta)
        val = float(c.max()) + beta * float(np.log(np.exp((c - c.max()) / beta).sum()))
        return val

    def gradient(z):
        e = np.exp(np.clip(z, -60, 60))
        aa = e / e.sum()
        c = _fft_convolve(aa)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(aa[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(aa, vp))
        gz = aa * (vp - dot)
        gz -= float(np.mean(gz))
        return gz

    z0 = np.log(a) - np.mean(np.log(a))
    
    result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                               'maxfun': max_iter * 20})
    
    if result.success or result.fun < 0:
        e = np.exp(np.clip(result.x, -60, 60))
        aa = e / e.sum()
        return aa
    
    return a


def _gradient_polish_fine(a, deadline, max_iter=30000):
    """Fine gradient polish with very small beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.005 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.0005, lr * 0.9)

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.85)

        if no_imp > max(2000, 4 * n):
            break

    return best_a


def _multi_start_refine(a, deadline, n_starts=6, rng=None):
    """Try multiple perturbed versions and refine the best."""
    if rng is None:
        rng = np.random.default_rng(99999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.03 * (i + 1)
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.3:
            continue

        sub_deadline = min(deadline, time.monotonic() + 20.0)
        a_ref = _logit_adam_optimize(a_pert, sub_deadline, max_iter=3000, lr=0.03)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-length optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    1. Diversified initialization across multiple lengths and profiles
    2. Logit-Adam optimization with adaptive beta and BB step
    3. LP-based spreading refinement
    4. Peak pressure coordinate descent
    5. L-BFGS-B polish
    6. Fine gradient polish
    7. Multi-start local refinement
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try (focus on promising ones)
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [768, 1024, 1536]:
        if ln not in lengths:
            lengths.append(ln)
    # Sort by how close to parent size (prefer close sizes for interpolation quality)
    if parent is not None:
        lengths.sort(key=lambda x: abs(x - parent.size))
    else:
        lengths.sort()

    # Phase 1: Generate diverse initializations
    print(f"Phase 1: Initialization, {len(lengths)} lengths: {lengths}", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent_interp', n_t, _interpolate_profile(parent, n_t)))

        for eta in [1.0, 2.0, 3.0]:
            init_pool.append((f'linear_{eta}', n_t, _make_linear_profile(n_t, eta)))
        
        init_pool.append(('bell', n_t, _make_bell_profile(n_t, 0.3)))
        init_pool.append(('u_shape', n_t, _make_u_shape(n_t, 2.0)))
        init_pool.append(('u_shape_strong', n_t, _make_u_shape(n_t, 4.0)))
        init_pool.append(('w_shape', n_t, _make_w_shape(n_t, 1.5)))
        init_pool.append(('sym_bell', n_t, _make_symmetric_bell(n_t, 0.4, 0.3)))
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum_init(n_t, rng)))

    # Quick evaluation
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 8 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:8]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Main optimization on top candidates
    # Allocate time based on number of candidates
    n_top = min(5, len(init_scores))
    top_inits = init_scores[:n_top]
    
    time_per_candidate = min(350.0, (deadline - time.monotonic() - 300.0) / max(1, n_top))
    
    for idx, (sc0, name, n_t, a0) in enumerate(top_inits):
        if time.monotonic() > deadline - 200:
            break
        if sc0 >= best_score * 1.01 and idx > 0:
            continue

        sub_budget = time_per_candidate
        opt_deadline = min(deadline - 150, time.monotonic() + sub_budget * 0.45)
        print(f"Phase 2[{idx}]: Logit-Adam on {name} (n={n_t}), start={sc0:.8f}, budget={opt_deadline - time.monotonic():.0f}s", flush=True)
        a_opt = _logit_adam_optimize(a0, opt_deadline, max_iter=50000, lr=0.05)
        sc_opt = _score(a_opt)
        print(f"  After Logit-Adam: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < deadline - 120:
            lp_deadline = min(deadline - 100, time.monotonic() + sub_budget * 0.30)
            print(f"  LP refine, budget={lp_deadline - time.monotonic():.0f}s", flush=True)
            a_lp = _lp_refine(a_opt, lp_deadline, max_iter=50000, inflate=1.0005, blend=0.2)
            sc_lp = _score(a_lp)
            print(f"  After LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Peak pressure
        if time.monotonic() < deadline - 80:
            pp_deadline = min(deadline - 60, time.monotonic() + sub_budget * 0.15)
            a_pp = _peak_pressure_descent(a_opt, pp_deadline, max_iter=5000, n_peaks=8, strength=0.03)
            sc_pp = _score(a_pp)
            print(f"  After PeakPressure: {sc_pp:.10f}", flush=True)
            if sc_pp < best_score:
                best_score = sc_pp
                best_seq = a_pp.copy()
            a_opt = a_pp

    # Phase 3: L-BFGS polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        lb_deadline = min(deadline - 40, time.monotonic() + 60.0)
        print(f"Phase 3: L-BFGS polish, budget={lb_deadline - time.monotonic():.0f}s", flush=True)
        a_lb = _lbfgs_polish(best_seq, lb_deadline, max_iter=200)
        sc_lb = _score(a_lb)
        print(f"  After L-BFGS: {sc_lb:.10f}", flush=True)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()

    # Phase 4: Fine gradient polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        gp_deadline = min(deadline - 20, time.monotonic() + 80.0)
        print(f"Phase 4: Fine polish, budget={gp_deadline - time.monotonic():.0f}s", flush=True)
        a_fine = _gradient_polish_fine(best_seq, gp_deadline, max_iter=30000)
        sc_fine = _score(a_fine)
        print(f"  After Fine polish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Phase 5: Multi-start
    if best_seq is not None and time.monotonic() < deadline - 15:
        ms_deadline = deadline - 10
        a_ms = _multi_start_refine(best_seq, ms_deadline, n_starts=4, rng=rng)
        sc_ms = _score(a_ms)
        print(f"  After Multi-start: {sc_ms:.10f}", flush=True)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()

    # Phase 6: Final LP pass
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_deadline2 = deadline - 3
        if lp_deadline2 - time.monotonic() > 15:
            a_last = _lp_refine(best_seq, lp_deadline2, max_iter=10000, inflate=1.0003, blend=0.1)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LP: {sc_last:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(768, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
