def _next_pow2(n):

    while k < n:

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):

def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. convolve(f, g)[k] <= M for all k, g >= 0."""
    # convolve(f, g)[k] = sum_j f[j] * g[k-j]
    # For k = 0, 1, ..., 2n-2:
    #   sum_j f[j] * g[k-j] <= M
    # where g is defined on [0, n-1] and f on [0, n-1]
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # i.e., A[k, j] = f[j] if max(0, k-(n-1)) <= j <= min(k, n-1)
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = f[j] for k in [j, j+n-1]
        A[j:j + n, j] = f
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth-max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    """Golden section search for optimal t in [0,1] for (1-t)*a + t*g."""

        return _score(a_new)

        lo = max(0.0, best_t - span)
        hi = min(1.0, best_t + span)

    invphi = (np.sqrt(5.0) - 1.0) / 2.0

            total_iters += 1
            # Tighten inflation on success
            inflate = max(inflate_min, inflate * 0.96)
        else:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate = min(0.05, inflate * 1.25)
            if no_imp > max(150, n // 2):
                # Reset and try again with initial inflate
                a = best_a.copy()
                inflate = max(inflate_start, inflate_min * 5)
                no_imp = 0
                if total_iters > 200 and inflate > 1.005:
                    break

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient descent with smooth-max and adaptive parameters."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-10, 0.005 * float(c.max()))
    lr = 0.015
    base_lr = lr
    momentum = 0.92
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_trial = a + momentum * v
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            v = np.zeros(n)
            continue
        a_trial = a_trial / s_trial

        grad, maxc = _smooth_max_grad(a_trial, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.94)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.005 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 40 == 39:
            beta = max(1e-12, beta * 0.82)

    return best_a


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_expo(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_quad(n, c1=3.0, c2=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c1 * t + c2 * t * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_bell(n, center=0.7, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2) + 1e-8
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _perturb(a, rng, scale=0.01):
    """Add small random perturbation and renormalize."""
    n = len(a)
    noise = rng.uniform(-scale, scale, n)
    a_new = np.maximum(a + noise, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale LP + gradient optimization with golden-section line search,
    adaptive inflation, combined cycling, and perturbation restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define initializations at different lengths
    configs = []
    target_lengths = [512, 768]

    for n_t in target_lengths:
        if parent is not None:
            configs.append(('parent', n_t, _interpolate(parent, n_t)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp15', n_t, _make_ramp(n_t, 1.5)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('expo3', n_t, _make_expo(n_t, 3.0)))
        configs.append(('expo5', n_t, _make_expo(n_t, 5.0)))
        configs.append(('quad3', n_t, _make_quad(n_t, 3.0, 1.0)))
        configs.append(('quad5', n_t, _make_quad(n_t, 5.0, 2.0)))
        configs.append(('bell', n_t, _make_bell(n_t, 0.7, 0.3)))

    # Score and sort
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    print(f"Top 8 initializations:")
    for sc0, name, n_t, _ in scored_configs[:8]:
        print(f"  {name} n={n_t}: {sc0:.8f}")

    # Phase 1: LP refinement on promising initializations
    for i, (sc0, name, n_t, a0) in enumerate(scored_configs):
        if time.monotonic() > deadline - 200:
            break
        if sc0 >= best_score + 0.03:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = deadline - time.monotonic() - 150
        if remaining < 20:
            break

        # Allocate more time to better initializations
        if sc0 < best_score + 0.01:
            lp_time = min(350.0, remaining * 0.4)
        elif sc0 < best_score + 0.03:
            lp_time = min(200.0, remaining * 0.25)
        else:
            lp_time = min(80.0, remaining * 0.1)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"[Phase1 {i}] LP: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=50000, inflate_start=1.0015)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP improved: {sc_lp:.10f}")
        else:
            print(f"  -> LP: {sc_lp:.8f} (best={best_score:.8f})")

    # Phase 2: Combined LP + gradient cycling on best solution
    if best_seq is not None and time.monotonic() < deadline - 60:
        for cycle in range(4):
            if time.monotonic() > deadline - 40:
                break

            # Gradient step
            if time.monotonic() < deadline - 80:
                gp_budget = min(100.0, deadline - time.monotonic() - 60)
                if gp_budget > 10:
                    gp_end = time.monotonic() + gp_budget
                    a_gp = _gradient_polish(best_seq, gp_end, max_iter=40000)
                    sc_gp = _score(a_gp)
                    if np.isfinite(sc_gp) and sc_gp < best_score:
                        best_score = sc_gp
                        best_seq = a_gp.copy()
                        print(f"  Cycle{cycle} grad: {sc_gp:.10f} (improved)")
                    else:
                        print(f"  Cycle{cycle} grad: {sc_gp:.10f}")

            # LP step with tight inflation
            if time.monotonic() < deadline - 40:
                lp_budget = min(120.0, deadline - time.monotonic() - 30)
                if lp_budget > 10:
                    lp_end = time.monotonic() + lp_budget
                    a_lp = _lp_refine(best_seq, lp_end, max_iter=40000,
                                     inflate_start=1.001, inflate_min=1.00005)
                    sc_lp = _score(a_lp)
                    if np.isfinite(sc_lp) and sc_lp < best_score:
                        best_score = sc_lp
                        best_seq = a_lp.copy()
                        print(f"  Cycle{cycle} LP: {sc_lp:.10f} (improved)")
                    else:
                        print(f"  Cycle{cycle} LP: {sc_lp:.10f}")

    # Phase 3: Perturbation restart - try to escape local minimum
    if best_seq is not None and time.monotonic() < deadline - 60:
        for attempt in range(5):
            if time.monotonic() > deadline - 40:
                break
            perturbed = _perturb(best_seq, rng, scale=0.005)
            sc_p = _score(perturbed)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = perturbed.copy()
            # Quick LP on perturbed
            if time.monotonic() < deadline - 40:
                lp_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 30)
                a_lp = _lp_refine(perturbed, lp_end, max_iter=20000, inflate_start=1.002)
                sc_lp = _score(a_lp)
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"  Perturb{attempt}: {sc_lp:.10f} (improved!)")

    # Phase 4: Final tight LP pass
    if best_seq is not None and time.monotonic() < deadline - 20:
        lp_time = min(80.0, deadline - time.monotonic() - 15)
        if lp_time > 5:
            lp_end = time.monotonic() + lp_time
            a_final = _lp_refine(best_seq, lp_end, max_iter=30000,
                                inflate_start=1.0008, inflate_min=1.00005)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"Final LP: {sc_final:.10f}")

    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        gp_time = min(40.0, deadline - time.monotonic() - 8)
        if gp_time > 5:
            gp_end = time.monotonic() + gp_time
            a_g = _gradient_polish(best_seq, gp_end, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"Final grad: {sc_g:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)

    # Verify with evaluate_sequence if possible
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval check: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
