        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a.copy()
            no_imp = 0
            inflate = max(1.0003, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                inflate = min(1.015, inflate * 1.02)
            if no_imp > 1500:
                a = best_a.copy()
                no_imp = 0
                inflate = 1.001
                warm = a.copy()

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, float(c.max()), c


def _adam_polish(a, deadline, max_iter=30000, lr=0.04):
    """Adam-based polish in log-domain (softmax parameterization)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        sw = float(ex.sum())
        if not np.isfinite(sw) or sw <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            sw = 1.0
        w = ex / sw

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

        if no_imp > max(3000, 4 * n):
            rng = np.random.default_rng(it * 7919 + n)
            z = 0.8 * best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _gradient_polish(a, deadline, max_iter=20000):
    """Simple gradient descent with momentum on simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc, c_arr = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.96)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _make_hann(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def _make_blackman(n):
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / n) + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / n)
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_raised_cosine(n, beta=2.0):
    t = np.linspace(0.0, 1.0, n)
    w = 1.0 + beta * np.cos(np.pi * (2.0 * t - 1.0))
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _optimize_single(a_init, n, deadline, phase_budget):
    """Run LP then gradient polish on a single initialization."""
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_uniform(n)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    print(f"    Init: score={best_score:.10f}")

    # LP phase
    lp_budget = min(phase_budget * 0.6, deadline - time.monotonic() - 60.0)
    if lp_budget > 10:
        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_optimize(a, lp_end, max_iter=15000)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"    LP: {sc:.10f}")

    # Gradient polish
    gp_budget = min(phase_budget * 0.2, deadline - time.monotonic() - 30.0)
    if gp_budget > 5:
        gp_end = time.monotonic() + gp_budget
        a_gp = _gradient_polish(best_a, gp_end, max_iter=10000)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"    Grad: {sc:.10f}")

    # Adam polish
    ad_budget = min(phase_budget * 0.2, deadline - time.monotonic() - 15.0)
    if ad_budget > 5:
        ad_end = time.monotonic() + ad_budget
        a_ad = _adam_polish(best_a, ad_end, max_iter=15000, lr=0.03)
        sc = _score(a_ad)
        if sc < best_score:
            best_score = sc
            best_a = a_ad.copy()
        print(f"    Adam: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based spreading with adaptive inflation,
    diversified initializations, multi-length search, and gradient/Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    parent_n = 768
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                parent_n = parent.size
                sc = _score(parent)
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent: n={parent_n}, score={sc:.10f}")
    except Exception:
        pass

    if parent is None:
        parent = _make_hann(parent_n)
        parent_n = 768
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent, using Hann: score={best_score:.10f}")

    # Determine which lengths to try
    lengths = [parent_n]
    for ln in [512, 640, 1024, 384, 896]:
        if ln not in lengths:
            lengths.append(ln)

    print(f"\nLengths to try: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Main LP optimization on parent length with parent as init
    n_main = parent_n
    phase1_budget = min((deadline - time.monotonic() - 300.0) * 0.5, 600.0)
    phase1_budget = max(60.0, phase1_budget)
    print(f"\n=== Phase 1: Main LP at n={n_main}, budget={phase1_budget:.0f}s ===")
    p1_end = time.monotonic() + phase1_budget
    a_res, sc_res = _optimize_single(parent, n_main, deadline, phase1_budget)
    if sc_res < best_score:
        best_score = sc_res
        best_seq = a_res.copy()
        print(f"  ** New best: {sc_res:.10f}")

    # Phase 2: Diverse initializations at best length
    if time.monotonic() < deadline - 120:
        n_best = best_seq.size
        phase2_budget = min((deadline - time.monotonic() - 100.0) * 0.35, 400.0)
        print(f"\n=== Phase 2: Diverse inits at n={n_best}, budget={phase2_budget:.0f}s ===")

        inits = [
            _make_hann(n_best),
            _make_blackman(n_best),
            _make_uniform(n_best),
            _make_raised_cosine(n_best, 1.5),
            _make_raised_cosine(n_best, 3.0),
        ]
        # Perturbed best
        for scale in [0.1, 0.2, 0.3]:
            pert = best_seq * (1.0 + scale * rng.normal(size=n_best))
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            inits.append(pert)
        # Smooth random
        for _ in range(3):
            raw = np.maximum(rng.normal(1.0, 0.3, n_best), 0.1)
            # Smooth with moving average
            k = max(3, n_best // 100)
            kernel = np.ones(k) / k
            raw = np.convolve(raw, kernel, mode='same')
            raw = np.maximum(raw, 0.0)
            inits.append(raw / raw.sum())

        # Interpolate parent to n_best (might be different)
        inits.append(_interpolate_profile(parent, n_best))

        # Time per init
        per_init = max(20.0, phase2_budget / len(inits))
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            # Only optimize if promising or if we have time
            if per_init > 25:
                sub_budget = min(per_init, deadline - time.monotonic() - 60.0)
                if sub_budget > 20:
                    a_r, sc_r = _optimize_single(a0, n_best, deadline, sub_budget)
                    if sc_r < best_score:
                        best_score = sc_r
                        best_seq = a_r.copy()
                        print(f"  ** Init {i}: new best {sc_r:.10f}")

    # Phase 3: Try other lengths
    if time.monotonic() < deadline - 80:
        print(f"\n=== Phase 3: Other lengths ===")
        for n_t in lengths:
            if n_t == parent_n:
                continue
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic() - 40.0
            if rem < 30:
                break
            sub_budget = min(120.0, rem * 0.4)
            if sub_budget < 20:
                break
            a_init = _interpolate_profile(parent, n_t)
            print(f"  n={n_t}, budget={sub_budget:.0f}s")
            a_r, sc_r = _optimize_single(a_init, n_t, deadline, sub_budget)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  ** n={n_t}: new best {sc_r:.10f}")

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_fin = best_seq.size
        final_budget = deadline - time.monotonic() - 10.0
        print(f"\n=== Phase 4: Final polish, n={n_fin}, budget={final_budget:.0f}s ===")

        # LP with tight inflation
        lp_b = min(final_budget * 0.3, 100.0)
        if lp_b > 10:
            a_lp = _lp_optimize(best_seq, time.monotonic() + lp_b, max_iter=20000)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}")

        # Adam
        ad_b = min(final_budget * 0.3, 80.0)
        if ad_b > 10:
            a_ad = _adam_polish(best_seq, time.monotonic() + ad_b, max_iter=20000, lr=0.02)
            sc = _score(a_ad)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"  Final Adam: {sc:.10f}")

        # Gradient
        gp_b = min(final_budget * 0.2, 50.0)
        if gp_b > 5:
            a_gp = _gradient_polish(best_seq, time.monotonic() + gp_b, max_iter=10000)
            sc = _score(a_gp)
            if sc < best_score:
                best_score = sc
                best_seq = a_gp.copy()
            print(f"  Final Grad: {sc:.10f}")

        # One more tight LP
        lp_b2 = min(final_budget * 0.15, 30.0)
        if lp_b2 > 5:
            a_lp2 = _lp_optimize(best_seq, time.monotonic() + lp_b2, max_iter=5000)
            sc = _score(a_lp2)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp2.copy()
            print(f"  Final LP2: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_hann(768)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * np.sqrt(2.0 * final_seq.size)
    else:
        final_seq = _make_hann(768)

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            # Fallback: use sum=1 normalization
            final_seq = best_seq / best_seq.sum()
            sc_eval2 = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval2):
                print(f"evaluate_sequence (fallback): {sc_eval2:.10f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
        final_seq = best_seq / best_seq.sum()

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, sum={final_seq.sum():.4f}")
    return [float(x) for x in final_seq]
