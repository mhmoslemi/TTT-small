def _next_pow2(n):

def _minimax_lp(a, M_init=None):
    """Solve min t s.t. conv(a,g)[k] <= t, sum(g)=1, g>=0. Returns (g*, t*)."""
    n = len(a)
    # Variables: g[0..n-1], t -> total n+1 variables
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: conv(a,g)[k] - t <= 0 for k = 0..2n-2
    # conv(a,g)[k] = sum_j a[j]*g[k-j] (with bounds)
    # So: sum_j (coeff_{k,j} * g[j]) - t <= 0
    # where coeff_{k,j} = a[j] if 0 <= k-j <= n-1, else 0
    # Wait, conv(a,g)[k] = sum_{j'} a[j']*g[k-j'] where 0<=j'<=n-1 and 0<=k-j'<=n-1
    # = sum_{j'} a[j']*g[k-j']
    # So the coefficient of g[j] in conv(a,g)[k] is a[k-j] if 0 <= k-j <= n-1, else 0.
    
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for j in range(n):
            if 0 <= k - j <= n - 1:
                A[k, j] = a[k - j]
        A[k, -1] = -1.0  # coefficient of t
    
    b_ub = np.zeros(2 * n - 1)
    
    # sum(g) = 1 -> as two inequalities: sum(g) <= 1 and -sum(g) <= -1
    # Or use A_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        t_val = res.x[-1]
        return g, float(t_val)
    return None, None

def _interp(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n0 = a.size
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _random_phase_init(n, seed=42):

    rng = np.random.default_rng(seed)

    mags = np.ones(N//2 + 1)
    mags[0] = n  # DC = n means sum = n (will normalize later)
    
    A = mags * np.exp(1j * phases)
    # Make it Hermitian-symmetric for real output
    A_full = np.zeros(N, dtype=complex)
    A_full[:N//2+1] = A
    for k in range(1, N//2):
        A_full[N-k] = np.conj(A[k])
    if N % 2 == 0:
        A_full[N//2] = float(A[N//2])  # Nyquist must be real
        A_full[N//2] = abs(A_full[N//2])
    
    a = np.fft.ifft(A_full).real[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return _make_profile(n, 'linear', eta=1.5)
    return a / s

"""
Step function optimization using adaptive smooth-max Adam in logit-space,
minimax LP direction finding with line search, polynomial/ramp initializations,
and spectral analysis. Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key innovations: banded LP with precomputed structure, adaptive beta annealing,
polynomial family initialization, and aggressive fine-tuning of the best candidate.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c -= c.min()
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_raw(a):
    """Score assuming sum(a)=1."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = np.clip((c - mval) / beta, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_conv(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    h = np.fft.irfft(fw * fa, N)
    return 2.0 * h[n - 1:2 * n - 1]


def _adam_opt(a, deadline, max_iter=30000, lr=0.02, beta_factor=0.03, name=""):
    """Adam optimizer in logit-space with adaptive beta for smooth max of convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(z.mean())

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * c.max(), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        c_max = float(c_curr.max())
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_conv(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(grad_z.mean())

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m, v, t_count = np.zeros(n), np.zeros(n), 0
            continue

        t_count += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(z_new.mean())

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:

                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m, v, t_count = np.zeros(n), np.zeros(n), 0

                no_imp = 0

        z, a = z_new, a_new
        if it % 150 == 0:
            beta *= 0.82

def _minimax_lp(a):
    """Solve min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g_star, t_star) or (None, None)."""
    n = len(a)
    # Build constraint matrix: A_ub has 2n-1 rows, n+1 cols
    # Row k: sum_j a[k-j]*g[j] - t <= 0 (where 0<=k-j<=n-1)
    # Column n is for t (coefficient -1)
    A = np.zeros((2 * n - 1, n + 1))
    # Use the structure: entry (k, j) = a[k-j] if 0 <= k-j <= n-1
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
        A[k, -1] = -1.0

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], float(res.x[-1])
    return None, None


def _lp_refine(a, deadline, max_iter=300):
    """Iteratively apply minimax LP with line search for blend factor."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_sc = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g, t_star = _minimax_lp(a)
        if g is None:
            break

        c_curr = _fft_convolve(a)
        curr_max = float(c_curr.max())

        # If LP can't improve, stop
        if t_star >= curr_max - 1e-9:
            no_imp += 1
            if no_imp > 10:
                break
            continue

        # Line search for blend factor
        best_blend_sc = best_sc
        best_blend = 0.0
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_trial = (1 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0.0)
            s = float(a_trial.sum())
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc_trial = float(2.0 * n * _fft_convolve(a_trial).max())
            if sc_trial < best_blend_sc:
                best_blend_sc = sc_trial
                best_blend = alpha

        if best_blend == 0.0:
            no_imp += 1
            if no_imp > 20:
                break
            continue

        a = (1 - best_blend) * a + best_blend * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        if sc < best_sc - 1e-11:
            best_sc = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

    return best_a, best_sc


def _subgrad_opt(a, deadline, max_iter=10000, top_k=3):
    """Multi-peak subgradient method."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_sc = float(2.0 * n * c.max())
    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]

        # Weighted subgradient
        weights = np.array([1.0 + 0.3 * i for i in range(k)])
        weights /= weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            idx = int(idx)
            # dc[idx]/da[j] = a[idx-j] + a[j-idx] (with bounds)
            for j in range(n):
                v = 0.0
                k1 = idx - j
                if 0 <= k1 <= n - 1:
                    v += a[k1]
                k2 = j - idx
                if 0 <= k2 <= n - 1:
                    v += a[k2]
                if j == idx:
                    # Counted twice above if idx-j = j-idx = 0, so subtract one
                    v -= a[0]
                sg[j] += weights[i] * v

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_sc - 1e-11:
            best_sc = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 3 * n):
                break

        a = a_new

    return best_a, best_sc


def _subgrad_fast(a, deadline, max_iter=10000, top_k=3):
    """Faster subgradient using FFT for the gradient computation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_sc = float(2.0 * n * c.max())
    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:

k]
        weights = np.array([1.0 + 0.3 * i for i in range(k)])
        weights /= weights.sum()

        # Build combined weight vector for the gradient
        w = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_idx):
            w[int(idx)] += weights[i]

        grad_a = _grad_conv(a, w)

        a_new = a - base_lr / np.sqrt(it + 1) * grad_a
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_sc - 1e-11:
            best_sc = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 3 * n):
                break

        a = a_new

    return best_a, best_sc


def _make_profile(n, shape='linear', **kwargs):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if shape == 'linear':
        eta = kwargs.get('eta', 1.5)
        a = 1.0 + eta * (t - 0.5) * 2
    elif shape == 'quad':
        c1 = kwargs.get('c1', 1.0)
        c2 = kwargs.get('c2', -0.3)
        a = 1.0 + c1 * u + c2 * u * u
    elif shape == 'cubic':
        c1 = kwargs.get('c1', 1.2)
        c3 = kwargs.get('c3', -0.4)
        a = 1.0 + c1 * u + c3 * u ** 3
    elif shape == 'exp':
        alpha = kwargs.get('alpha', 2.0)
        a = np.exp(alpha * (t - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _random_phase_init(n, seed=42):
    """Sequence with random phases in frequency domain."""
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)
    A = np.zeros(N, dtype=complex)
    A[0] = n
    for k in range(1, N // 2 + 1):
        mag = 1.0 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = mag * np.exp(1j * phase)
    for k in range(1, N // 2):
        A[N - k] = A[k]
    if N % 2 == 0:
        A[N // 2] = float(A[N // 2].real)
    a = np.fft.ifft(A).real[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return _make_profile(n, 'linear', eta=1.5)
    return a / s


def _pipeline(a0, deadline, label=""):
    """Full optimization pipeline: Adam -> LP -> Adam -> LP -> subgrad."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} start: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0
    n = a0.size

    # Phase 1: Adam (coarse)
    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 180)
        a1, s1 = _adam_opt(a0, d1, max_iter=15000, lr=0.025, beta_factor=0.04, name="coarse")
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 2: LP refinement
    if time.monotonic() < deadline - 20 and n <= 2048:
        d2 = min(deadline - 20, time.monotonic() + 80)
        a2, s2 = _lp_refine(best_a, d2, max_iter=100)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} LP1: {best_sc:.10f}")

    # Phase 3: Adam (fine)
    if time.monotonic() < deadline - 15:
        d3 = min(deadline - 15, time.monotonic() + 120)
        a3, s3 = _adam_opt(best_a, d3, max_iter=12000, lr=0.008, beta_factor=0.015, name="fine")
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} Adam2: {best_sc:.10f}")

    # Phase 4: LP again
    if time.monotonic() < deadline - 10 and n <= 2048:
        d4 = min(deadline - 10, time.monotonic() + 60)
        a4, s4 = _lp_refine(best_a, d4, max_iter=80)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} LP2: {best_sc:.10f}")

    # Phase 5: Subgradient polish
    if time.monotonic() < deadline - 5:
        d5 = deadline - 5
        a5, s5 = _subgrad_fast(best_a, d5, max_iter=5000, top_k=3)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} Subgrad: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Priority: optimize n=1024 first (matches previous best), then 1536, 768, 2048
    lengths_priority = [1024, 1536, 768, 2048]

    for n in lengths_priority:
        if time.monotonic() > deadline - 80:
            break

        print(f"\n{'='*50}")
        print(f"  Optimizing n={n}")
        print(f"{'='*50}")

        # Time budget for this length
        remaining = deadline - time.monotonic()
        idx = lengths_priority.index(n)
        frac = [0.35, 0.25, 0.2, 0.2][idx]
        time_budget = min(remaining * 0.9, remaining * frac + 100)
        d_len = min(time.monotonic() + time_budget, deadline - 80)

        # Create and evaluate initializations
        inits = []
        if parent is not None:
            inits.append(('parent', _interp(parent, n)))
        inits.append(('lin12', _make_profile(n, 'linear', eta=1.2)))
        inits.append(('lin15', _make_profile(n, 'linear', eta=1.5)))
        inits.append(('lin18', _make_profile(n, 'linear', eta=1.8)))
        inits.append(('lin20', _make_profile(n, eta=2.0) if False else _make_profile(n, 'linear', eta=2.0)))
        inits.append(('quad', _make_profile(n, 'quad', c1=1.5, c2=-0.3)))
        inits.append(('cubic', _make_profile(n, 'cubic', c1=1.3, c3=-0.5)))
        inits.append(('exp', _make_profile(n, 'exp', alpha=1.5)))
        inits.append(('rand1', _random_phase_init(n, seed=seed)))
        inits.append(('rand2', _random_phase_init(n, seed=seed + 777)))

        # Evaluate all inits
        init_scores = []
        for name, a0 in inits:
            sc = _score(a0)
            init_scores.append((sc, name, a0))
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()

        init_scores.sort()
        print(f"  Top inits: " + ", ".join(
            f"{nm}={sc:.6f}" for sc, nm, _ in init_scores[:5]))

        # Pick top 2-3 initializations to optimize
        n_opt = min(3, len(init_scores))
        per_time = (d_len - time.monotonic()) / max(1, n_opt)

        for i in range(n_opt):
            if time.monotonic() > deadline - 60:
                break
            sc, name, a0 = init_scores[i]
            d_i = min(time.monotonic() + per_time, deadline - 60)
            if d_i - time.monotonic() < 30:
                break

            a_opt, sc_opt = _pipeline(a0, d_i, f"{name}@{n}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {best_score:.10f} ({name}@{n})")

    # Final polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*50}")
        print(f"  FINAL POLISH (best={best_score:.10f}, n={best_seq.size})")
        print(f"{'='*50}")

        # Adam with very small lr
        d_f1 = min(deadline - 5, time.monotonic() + 60)
        a_f1, s_f1 = _adam_opt(best_seq, d_f1, max_iter=8000, lr=0.003, beta_factor=0.005)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Fine Adam: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 5 and best_seq.size <= 2048:
            d_f2 = min(deadline - 3, time.monotonic() + 30)
            a_f2, s_f2 = _lp_refine(best_seq, d_f2, max_iter=40)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final LP: {best_score:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 2:
            a_f3, s_f3 = _adam_opt(best_seq, deadline - 2, max_iter=5000, lr=0.001, beta_factor=0.002)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Ultra Adam: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profile(512, 'linear', eta=1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
