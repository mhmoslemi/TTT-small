"""
Improved multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)  # numerical safety
    return c


def _score(p):
    """Score = 2*n*max(conv(p,p)) assuming sum(p)=1"""
    n = len(p)
    c = _full_convolve(p)
    return 2.0 * n * float(c.max())


def _smooth_max_weights(c, beta):
    """Returns (smooth_max_value, gradient_weights)"""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_conv(w, p):
    """Compute cross-correlation: gradient of smooth max w.r.t. p"""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s


def _lp_solve_banded(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0.
    If active_indices given, only enforce constraints for those k."""
    n = len(a)
    num_constraints = 2 * n - 1
    
    if active_indices is not None:
        rows_idx = active_indices
    else:
        rows_idx = np.arange(num_constraints)
    
    rows = []
    cols = []
    vals = []
    for k in rows_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([int(k)] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    # For non-active constraints, we still need them in the matrix but with large bound
    b_ub = np.full(num_constraints, 1e10)
    b_ub[rows_idx] = M
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine_iterative(a, deadline, max_iters=50, verbose=True):
    """Iterative LP refinement with adaptive step size."""
    n = len(a)
    a = a.copy()
    current_score = _score(a)
    best_a = a.copy()
    best_score = current_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        # Find near-active constraints
        threshold = M * 0.85
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        b = _lp_solve_banded(a, M, active)
        if b is None or b.sum() < 1e-10:
            break
        
        b = b / b.sum()
        
        # Adaptive step size: try several values
        best_t = None
        best_new_score = current_score
        for t in [0.5, 0.3, 0.15, 0.08, 0.04]:
            a_try = (1.0 - t) * a + t * b
            a_try = _normalize(a_try)
            if a_try is None:
                continue
            sc = _score(a_try)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_t is not None:
            a = (1.0 - best_t) * a + best_t * b
            a = _normalize(a)
            current_score = best_new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP iter {it}: score={best_score:.10f}, t={best_t}, active={len(active)}")
        else:
            # No improvement from LP direction, break
            if verbose:
                print(f"    LP iter {it}: no improvement, M={M:.8f}, active={len(active)}, b_sum={b.sum():.4f}")
            break
    
    return best_a, best_score


def _adam_smooth_max(p0, deadline, rng, lr=0.05, beta_frac=0.08, 
                     beta_decay=0.93, no_imp_limit=200, verbose=False):

    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z -= float(np.mean(z))

    v = np.zeros(n)
    t_adam = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr
    iter_count = 0

        iter_count += 1
        _, w = _smooth_max_weights(c, beta)
        g = _cross_conv(w, p)
        
        # Gradient in logit space

        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1

        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
                if verbose and iter_count % 100 == 0:
                    print(f"    Adam iter {iter_count}: score={best_score:.10f}")
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

            p = _softmax_from_z(z)

            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)

def _make_profile_uniform(n):

def _make_profile_linear(n, eta):

def _make_profile_exponential(n, lam):

    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_hann(n):
    t = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_hamming(n):
    t = np.arange(n)
    a = 0.54 + 0.46 * np.cos(2 * np.pi * t / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))

"""
Improved multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: (1) diverse profile initialization + parent interpolation, (2) Adam smooth-max
in logit space, (3) aggressive LP refinement with near-active constraint selection and
adaptive step size, (4) alternating Adam+LP scheme, (5) multi-scale escalation,
(6) final fine-tuning with very small LP steps.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return 2.0 * n * float(c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_conv(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s


def _lp_solve(a, M, active_indices):
    n = len(a)
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([ki] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[ki - idx].tolist())
    b_ub = np.full(num_constraints, 1e12)
    for k in active_indices:
        b_ub[int(k)] = M
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, max_iters=80, verbose=False):

    a = a.copy()
    current_score = _score(a)
    best_a = a.copy()

            break

def _profile_uniform(n):
    return np.ones(n) / n


def _profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_hann(n):
    t = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_hamming(n):
    t = np.arange(n)
    a = 0.54 + 0.46 * np.cos(2.0 * np.pi * t / max(1, n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    state = {'best_seq': None, 'best_score': np.inf}

    def _check_update(p, sc, label):
        if p is not None and np.isfinite(sc) and sc < state['best_score']:
            state['best_seq'] = np.asarray(p, dtype=np.float64).copy()
            state['best_score'] = float(sc)
            print(f"  BEST: {state['best_score']:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent: n={parent_raw.size}, score={_score(parent):.10f}")
    except Exception:
        parent = None

    # Phase 1: Initial candidates
    print("Phase 1: Initial candidates")
    candidates = []
    n_values = [256, 512, 1024, 2048]

    profile_list = [
        ('uniform', 256, _profile_uniform, (256,)),
        ('lin1.0', 256, _profile_linear, (256, 1.0)),
        ('lin1.5', 256, _profile_linear, (256, 1.5)),
        ('lin2.0', 256, _profile_linear, (256, 2.0)),
        ('lin2.5', 256, _profile_linear, (256, 2.5)),
        ('exp0.5', 256, _profile_exponential, (256, 0.5)),
        ('exp1.0', 256, _profile_exponential, (256, 1.0)),
        ('exp1.5', 256, _profile_exponential, (256, 1.5)),
        ('exp2.0', 256, _profile_exponential, (256, 2.0)),
        ('exp3.0', 256, _profile_exponential, (256, 3.0)),
        ('cos1_05', 256, _profile_cosine, (256, 1, 0.5)),
        ('cos1_08', 256, _profile_cosine, (256, 1, 0.8)),
        ('cos2_04', 256, _profile_cosine, (256, 2, 0.4)),
        ('hann', 256, _profile_hann, (256,)),
        ('hamming', 256, _profile_hamming, (256,)),
        ('twob1', 256, _profile_two_bump, (256, 0.3, 0.7, 0.15)),
        ('twob2', 256, _profile_two_bump, (256, 0.2, 0.8, 0.2)),
        ('cubic1', 256, _profile_cubic, (256, 1.5, -1.0)),
        ('cubic2', 256, _profile_cubic, (256, 2.0, -1.5)),
    ]

    for n in n_values:
        for label, _, fn, args in profile_list:
            p = fn(*args)
            sc = _score(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        pn = parent.size
        if 0 < pn <= 4096:
            sc = _score(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))
        for n in n_values:
            if n != pn:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top 5:")
    for sc, p, n, label in candidates[:5]:
        print(f"    {sc:.10f} n={n} [{label}]")

    _check_update(candidates[0][1], candidates[0][0], 'init')

    # Phase 2: Adam polish top candidates
    print(f"\nPhase 2: Adam polish (deadline t={400:.0f}s)")
    phase2_end = t_start + 400.0
    polished = []
    top_k = 8
    for i in range(min(top_k, len(candidates))):
        sc, p, n, label = candidates[i]
        if time.monotonic() > phase2_end:
            break
        per = min(45.0, phase2_end - time.monotonic() - 5)
        if per < 15:
            break
        p_b, sc_b = _adam_optimize(p, time.monotonic() + per, rng,
                                   lr=0.06, beta_frac=0.1,
                                   beta_decay=0.92, no_imp_limit=200)
        _check_update(p_b, sc_b, f'adam_{label}_n{n}')
        polished.append((sc_b, p_b, n, label))

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={state['best_score']:.10f}")

    # Phase 3: LP refinement on best candidates
    print(f"\nPhase 3: LP refinement (deadline t={900:.0f}s)")
    phase3_end = t_start + 900.0
    lp_results = []
    for sc, p, n, label in polished[:4]:
        if time.monotonic() > phase3_end:
            break
        if n > 2048:
            continue
        per = min(150.0, phase3_end - time.monotonic() - 10)
        if per < 20:
            break
        p_lp, sc_lp = _lp_refine(p, time.monotonic() + per, max_iters=80, verbose=True)
        _check_update(p_lp, sc_lp, f'LP_{label}_n{n}')
        lp_results.append((sc_lp, p_lp, n, label))

    lp_results.sort(key=lambda x: x[0])
    print(f"  After Phase 3: best={state['best_score']:.10f}")

    # Phase 4: Alternating Adam+LP
    print(f"\nPhase 4: Alternating Adam+LP (deadline t={1300:.0f}s)")
    phase4_end = t_start + 1300.0
    if state['best_seq'] is not None:
        n_best = state['best_seq'].size
        for round_idx in range(6):
            if time.monotonic() > phase4_end:
                break
            # Short Adam
            adam_time = min(30.0, (phase4_end - time.monotonic()) / 2.0 - 5)
            if adam_time > 10:
                p_a, sc_a = _adam_optimize(state['best_seq'], time.monotonic() + adam_time, rng,
                                           lr=0.04, beta_frac=0.06,
                                           beta_decay=0.94, no_imp_limit=100)
                _check_update(p_a, sc_a, f'alt_adam_r{round_idx}')
            # LP
            lp_time = min(60.0, (phase4_end - time.monotonic()) / 2.0 - 5)
            if lp_time > 15 and n_best <= 2048:
                p_l, sc_l = _lp_refine(state['best_seq'], time.monotonic() + lp_time,
                                       max_iters=40, verbose=False)
                _check_update(p_l, sc_l, f'alt_LP_r{round_idx}')

    print(f"  After Phase 4: best={state['best_score']:.10f}")

    # Phase 5: Multi-scale escalation
    print(f"\nPhase 5: Multi-scale escalation (deadline t={budget_s-60:.0f}s)")
    phase5_end = t_start + float(budget_s) - 60.0
    if state['best_seq'] is not None and time.monotonic() < phase5_end:
        n_cur = state['best_seq'].size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]
        for tn in target_ns:
            if time.monotonic() > phase5_end - 30:
                break
            p_int = _interp_profile(state['best_seq'], tn)
            if p_int is None:
                continue
            sc_int = _score(p_int)
            _check_update(p_int, sc_int, f'interp_n{tn}')
            per = min(40.0, phase5_end - time.monotonic() - 10)
            if per > 15:
                p_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                           lr=0.04, beta_frac=0.07,
                                           beta_decay=0.93, no_imp_limit=150)
                _check_update(p_e, sc_e, f'escalate_n{tn}')

    # Phase 6: Final fine-tuning
    print(f"\nPhase 6: Final fine-tuning")
    if time.monotonic() < deadline - 10 and state['best_seq'] is not None:
        n_b = state['best_seq'].size
        # One more LP with very small steps
        if n_b <= 2048:
            per = min(60.0, (deadline - time.monotonic() - 10) * 0.5)
            if per > 10:
                p_f, sc_f = _lp_refine(state['best_seq'], time.monotonic() + per,
                                       max_iters=30, verbose=False)
                _check_update(p_f, sc_f, 'final_LP')
        # Final Adam
        if time.monotonic() < deadline - 5:
            per2 = deadline - time.monotonic() - 5
            if per2 > 10:
                p_f2, sc_f2 = _adam_optimize(state['best_seq'], time.monotonic() + per2, rng,
                                             lr=0.01, beta_frac=0.03,
                                             beta_decay=0.96, no_imp_limit=100)
                _check_update(p_f2, sc_f2, 'final_adam')

    # Final validation
    print(f"\n=== FINAL: score={state['best_score']:.10f} n={state['best_seq'].size if state['best_seq'] is not None else 0} ===")

    if state['best_seq'] is not None:
        final = np.clip(state['best_seq'], 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        exact_sc = _score(final)
        print(f"Verified score: {exact_sc:.10f}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"evaluate_sequence: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
        return [float(x) for x in final]

    fb = _profile_linear(512, 1.5)
    print("FALLBACK: n=512 linear")
    return [float(x) for x in fb]
