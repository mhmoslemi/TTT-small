"""
Multi-length mirror descent with adaptive log-sum-exp smoothing,
integrated LP direction finding, and coordinate descent refinement
for minimizing 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _conv_ops(a, w):
    """Compute sum_k w_k * a_{k-i} for all i using FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]


def _make_conv_matrix(a, M, top_k=80):
    """Reduced constraint matrix for LP using near-active lags."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b


def _lp_direction(a, M, top_k=80):
    """LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    n = len(a)
    if n > 512:
        A, b = _make_conv_matrix(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _line_search(a, direction, max_iter=25):
    """Golden section line search for (1-t)*a + t*direction."""
    best_t = 0.0
    best_sc = _score_norm(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (direction - a), 0))
    fd = _score_norm(np.maximum(a + d * (direction - a), 0))
    if not np.isfinite(fc):
        fc = best_sc + 1.0
    if not np.isfinite(fd):
        fd = best_sc + 1.0
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (direction - a), 0))
            if not np.isfinite(fc):
                fc = best_sc + 1.0
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (direction - a), 0))
            if not np.isfinite(fd):
                fd = best_sc + 1.0
    best_t = (lo + hi) / 2.0
    return best_t


def _mirror_descent(a, deadline, T_start=0.03, T_end=0.001,
                    lr_start=0.5, max_iter=200000,
                    lp_every=200, lp_inflate=1.001):
    """Mirror descent with adaptive log-sum-exp and periodic LP."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    lr = lr_start
    no_imp = 0
    base_lr = lr_start

    # Compute decay ratio
    if max_iter > 10:
        T_ratio = (T_end / T_start) ** (1.0 / max_iter)
    else:
        T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Adaptive: if very few active constraints, increase T slightly
        threshold = c_max - 0.1 * T
        n_active = int(np.sum(c >= threshold))
        if n_active <= 2 and T > T_end:
            T = max(T_end, T * 0.9)

        # Compute log-sum-exp weights
        x = (c - c_max) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Gradient: 2 * sum_k w_k * a_{k-i}
        grad = 2.0 * _conv_ops(a, w)

        # Center the gradient (tangent to simplex)
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot

        # Mirror descent update: a_i *= exp(-lr * grad_tan_i)
        step = lr * grad_tan
        step = np.clip(step, -20, 20)
        a_new = a * np.exp(-step)
        s_new = float(a_new.sum())
        if s_new < 1e-15 or not np.isfinite(s_new):
            a_new = a.copy()
            lr = max(0.01, lr * 0.8)
            continue
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc_new = _score_norm(a_new)
        if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            a = a_new  # still move, but track best separately
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.05, base_lr * (0.95 ** (no_imp // 200)))
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                lr = base_lr
                no_imp = 0
                # Add small perturbation
                rng_p = np.random.default_rng(12345 + it)
                a = a * (1.0 + 0.03 * rng_p.normal(size=n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()

        # Periodic LP refinement
        if lp_every > 0 and it % lp_every == lp_every - 1 and no_imp > 50:
            M_lp = c_max * lp_inflate
            g = _lp_direction(a, M_lp)
            if g is not None:
                sum_g = float(g.sum())
                if sum_g > 1.0 + 1e-6:
                    g = g / sum_g
                    t_opt = _line_search(a, g, max_iter=20)
                    if t_opt > 0.02:
                        a_lp = (1.0 - t_opt) * a + t_opt * g
                        a_lp = np.maximum(a_lp, 1e-15)
                        s_lp = float(a_lp.sum())
                        if s_lp > 1e-15:
                            a_lp = a_lp / s_lp
                            sc_lp = _score_norm(a_lp)
                            if np.isfinite(sc_lp) and sc_lp < best_score - 1e-10:
                                best_score = sc_lp
                                best_a = a_lp.copy()
                                a = a_lp
                                no_imp = 0

        T *= T_ratio
        T = max(T, T_end)

    return best_a


def _coordinate_descent(a, deadline, max_iter=50000, lr=0.01):
    """Coordinate descent on pairs (i, k*-i) for the active lag."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Gradient at this lag: 2*a_{k_star - i}
        rev_a = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                rev_a[i] = 2.0 * a[j]

        # Also consider second max
        c_sorted = np.sort(c)[::-1]
        k_second = int(np.argsort(c)[::-1][1]) if n > 1 else k_star
        rev_a2 = np.zeros(n)
        for i in range(n):
            j = k_second - i
            if 0 <= j < n:
                rev_a2[i] = 2.0 * a[j]

        # Blend gradients based on how close they are
        gap = c_max - c[k_second]
        weight2 = min(0.5, gap / (0.1 * c_max + 1e-15))
        grad = rev_a + weight2 * rev_a2

        # Projected gradient step on simplex
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot
        a_new = a - lr * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc_new = _score_norm(a_new)
        if np.isfinite(sc_new) and sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, base_lr * 0.9)
            if no_imp > max(5000, 5 * n):
                a = best_a.copy()
                lr = base_lr
                no_imp = 0

    return best_a

def _interpolate(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profile(n, kind='gaussian', **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'gaussian':
        sigma = kwargs.get('sigma', 0.3)
        a = np.exp(-0.5 * ((t - 0.5) / (sigma * 0.5)) ** 2)
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == 'hann':
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * t))
    elif kind == 'flat_edge':
        ew = kwargs.get('ew', 0.3)
        a = 1.0 + ew * np.cos(2.0 * np.pi * t)
    elif kind == 'linear':
        eta = kwargs.get('eta', 1.0)
        a = 1.0 + eta * (t - 0.5)
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'triangular':
        a = 1.0 - 2.0 * np.abs(t - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

                    best_seq = parent.copy()
    except Exception:
        pass

    # Lengths to try
    lengths = [1024]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    for ln in [512, 768, 1536]:
        if ln not in lengths:
            lengths.append(ln)

    # Sort by likely priority (parent length first, then 1024, then others)
    lengths.sort(key=lambda x: (x != 1024, x != (parent.size if parent is not None else -1), x))

    print(f"Trying lengths: {lengths}")

    best_per_length = {}

    # Phase 1: Quick initialization and scoring
    print("\n=== Phase 1: Initialization ===")
    init_candidates = {}
    for n_t in lengths:
        if time.monotonic() > deadline - 600:
            break
        cands = []
        if parent is not None:
            cands.append((_interpolate(parent, n_t), 'parent'))
            # Perturbed parent
            p_pert = _interpolate(parent, n_t) * (1.0 + 0.05 * rng.normal(size=n_t))
            p_pert = np.maximum(p_pert, 0)
            p_pert = p_pert / p_pert.sum()
            cands.append((p_pert, 'parent_pert'))
        cands.append((_make_profile(n_t, 'raised_cosine'), 'cos'))
        cands.append((_make_profile(n_t, 'hann'), 'hann'))
        cands.append((_make_profile(n_t, 'flat_edge', ew=0.4), 'edge'))
        cands.append((_make_profile(n_t, 'gaussian', sigma=0.4), 'gauss'))
        cands.append((_make_profile(n_t, 'triangular'), 'tri'))

        best_here = None
        best_here_sc = np.inf
        for a0, name in cands:
            sc0 = _score_norm(a0)
            if np.isfinite(sc0):
                full_sc = _score(a0)
                if full_sc < best_score:
                    best_score = full_sc
                    best_seq = a0.copy()
                if sc0 < best_here_sc:
                    best_here_sc = sc0
                    best_here = a0
                print(f"  n={n_t}, {name}: {sc0:.8f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            init_candidates[n_t] = best_here

    # Phase 2: Mirror descent on best candidates per length
    print("\n=== Phase 2: Mirror descent ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 120:
            break

        a0 = best_per_length[n_t]
        sc0 = _score_norm(a0)
        print(f"  MD start n={n_t}: {sc0:.10f}")

        md_time = min(300.0, deadline - time.monotonic() - 120.0)
        if md_time < 30:
            break
        md_end = time.monotonic() + md_time

        # Multiple T settings
        best_md = a0
        best_md_sc = sc0

        for T_val in [0.02, 0.03, 0.05]:
            if time.monotonic() > md_end:
                break
            a_md = _mirror_descent(a0, md_end, T_start=T_val, T_end=0.002,
                                   lr_start=0.3, max_iter=100000,
                                   lp_every=300, lp_inflate=1.002)
            sc_md = _score_norm(a_md)
            if sc_md < best_md_sc:
                best_md_sc = sc_md
                best_md = a_md

        sc_md_full = _score(best_md)
        if np.isfinite(sc_md_full) and sc_md_full < best_score:
            best_score = sc_md_full
            best_seq = best_md.copy()
        best_per_length[n_t] = best_md
        print(f"  MD result n={n_t}: {best_md_sc:.10f} (full: {sc_md_full:.10f})")

    # Phase 3: Coordinate descent refinement
    print("\n=== Phase 3: Coordinate descent ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 90:
            break

        a0 = best_per_length[n_t]
        cd_time = min(120.0, deadline - time.monotonic() - 90.0)
        if cd_time < 20:
            break
        cd_end = time.monotonic() + cd_time

        a_cd = _coordinate_descent(a0, cd_end, max_iter=50000, lr=0.005)
        sc_cd = _score_norm(a_cd)
        sc_cd_full = _score(a_cd)
        if np.isfinite(sc_cd_full) and sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
        best_per_length[n_t] = a_cd
        print(f"  CD result n={n_t}: {sc_cd:.10f} (full: {sc_cd_full:.10f})")

    # Phase 4: LP polish on best
    print("\n=== Phase 4: LP polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        lp_time = min(180.0, deadline - time.monotonic() - 60.0)
        if lp_time > 20:
            lp_end = time.monotonic() + lp_time
            a_lp = _lp_polish(best_seq, lp_end, max_iter=3000,
                              inflate=1.0008, blend=0.12)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  LP polish: {sc_lp:.10f}")

    # Phase 5: Final mirror descent with small T on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        md_time = min(90.0, deadline - time.monotonic() - 30.0)
        if md_time > 15:
            md_end = time.monotonic() + md_time
            a_final = _mirror_descent(best_seq, md_end, T_start=0.01, T_end=0.001,
                                      lr_start=0.15, max_iter=50000,
                                      lp_every=500, lp_inflate=1.001)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final MD: {sc_final:.10f}")

    # Phase 6: Very fine LP
    if best_seq is not None and time.monotonic() < deadline - 15:
        lp_time = min(45.0, deadline - time.monotonic() - 15.0)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            a_fine = _lp_polish(best_seq, lp_end, max_iter=1500,
                                inflate=1.0002, blend=0.05)
            sc_fine = _score(a_fine)
            if np.isfinite(sc_fine) and sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()
            print(f"  Fine LP: {sc_fine:.10f}")

    # Final: try coordinate descent on the very best
    if best_seq is not None and time.monotonic() < deadline - 10:
        cd_time = min(25.0, deadline - time.monotonic() - 10.0)
        if cd_time > 5:
            cd_end = time.monotonic() + cd_time
            a_cd2 = _coordinate_descent(best_seq, cd_end, max_iter=20000, lr=0.002)
            sc_cd2 = _score(a_cd2)
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
            print(f"  Final CD: {sc_cd2:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        # Keep as-is (scale invariant) but ensure sum >= 0.01
        pass
    else:
        final_seq = final_seq * 1.0 / s if s > 0 else np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
