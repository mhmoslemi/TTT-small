from scipy.optimize import minimize

def compute_conv(a, b):
    n, m = len(a), len(b)
    from scipy.fft import fft, ifft
    N = 1
    while N < n + m - 1:
        N *= 2
    A = fft(a, N)
    B = fft(b, N)
    return np.real(ifft(A * B, N)[:n+m-1])

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = compute_conv(a, a)
    return 2.0 * n * c.max() / (s * s)

def lp_improve(a, M, n_iter=5, t=0.5):
    """Iteratively improve a using LP."""
    n = len(a)
    a = a.copy()
    for _ in range(n_iter):
        M = max(compute_conv(a, a))
        g = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(g))
        # Build constraints: conv(a, g)[k] <= M for all k

                g_idx_hi = k - j_lo

                g_slice = g[g_idx_lo:g_idx_hi+1]

                # a[j] for j from j_lo to j_hi

                # j = j_lo+1: a[j_lo+1] * g[k - j_lo - 1] = a[j_lo+1] * g[g_idx_hi - 1]
                # ...
                # j = j_hi: a[j_hi] * g[k - j_hi] = a[j_hi] * g[g_idx_lo]
                # So it's a[j_lo:j_hi+1] @ g[g_idx_lo:g_idx_hi+1][::-1]
                # Or equivalently: a[j_lo:j_hi+1][::-1] @ g[g_idx_lo:g_idx_hi+1]
                constraints.append(a[j_lo:j_hi+1][::-1] @ g[g_idx_lo:g_idx_hi+1] <= M)
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.GLOP)  # or cp.GLPK
        
        if g.value is not None and g.value.sum() > 0:

            g_val = g_val / g_val.sum()
            a = (1 - t) * a / a.sum() + t * g_val
            a = np.maximum(a, 0)
            a = a / a.sum()
    return a

And we want `T1[j] = sum_m w[m+j] * a[m]` (with appropriate range restrictions).

So `T1 = np.correlate(w, a, 'full')` and `T1[j + len(a) - 1]`... no wait.

`np.correlate(w, a, 'full')` has length `len(w) + len(a) - 1`. The output at index `n` is `sum_m w[m+n] * a[m]` where `m` ranges over valid indices.

So `T1[j] = np.correlate(w, a, 'full')[j + len(a) - 1]`? No...

Let me just be concrete. If `w` has length `2n-1` and `a` has length `n`:
`np.correlate(w, a, 'full')` has length `3n-2`.
Output[0] = w[0]*a[0] (only overlapping term)
Output[1] = w[0]*a[1] + w[1]*a[0]
...
Output[k] = sum_{m} w[m+k]*a[m]

We want `T1[j] = sum_k w[k]*a[k-j]`. Let `i = k-j`, `k = i+j`:
`T1[j] = sum_i w[i+j]*a[i]`

This is `Output[j]` from `np.correlate(w, a, 'full')`! Because `Output[j] = sum_m w[m+j]*a[m]`.

Wait, but we need `m` to be in `[0, n-1]` (valid index for `a`) and `m+j` to be in `[0, 2n-2]` (valid index for `w`). The `np.correlate` handles this automatically.

So `T1 = np.correlate(w, a, 'full')` and `T1[j]` gives us what we want (for `j` from 0 to `2n-2`, but we only need `j` from 0 to `n-1`).

Similarly, `T2[j] = sum_k w[k]*a[j-k]`. Let `i = j-k`, `k = j-i`:
`T2[j] = sum_i w[j-i]*a[i]`

This is `np.correlate(a, w, 'full')[j + len(w) - 1]`... let me check.
`np.correlate(a, w, 'full')[n] = sum_m a[m+n]*w[m]`

We want `sum_i w[j-i]*a[i]`. Let `m = j-i`, `i = j-m`: `= sum_m w[m]*a[j-m]`.
This is `np.correlate(a, w, 'full')[j]`? Let's check: `Output[n] = sum_m a[m+n]*w[m]`.
`Output[j] = sum_m a[m+j]*w[m]`. Hmm, that's `sum_m a[m+j]*w[m]`, not `sum_m w[m]*a[j-m]`.

Let me try `np.correlate(w, a, 'full')` again but with different indexing:
`Output[n] = sum_m w[m+n]*a[m]`

For `T2[j] = sum_k w[k]*a[j-k]`: let `m = j-k`, so `k = j-m`:
`= sum_m w[j-m]*a[m]`

This is `sum_m w[(j)-m]*a[m]`. Compare with `Output[n] = sum_m w[m+n]*a[m]`. If I set `n = -j`, I get `sum_m w[m-j]*a[m]`. That's not quite right either.

OK let me just use FFT for both:
`T1[j] = sum_k w[k]*a[k-j]` → this is the cross-correlation
`T2[j] = sum_k w[k]*a[j-k]` → this is also a cross-correlation (with roles swapped)

Using the fact that `T2[j] = T1[j]` when `w` is symmetric (which it is, since `w` depends on `conv(a,a)` which is... wait, is `conv(a,a)` symmetric?

`conv(a,a)[k] = sum_j a[j]*a[k-j]`
`conv(a,a)[2n-2-k] = sum_j a[j]*a[2n-2-k-j]`

Let `i = n-1-j`: `= sum_i a[n-1-i]*a[n-1-k+i]`

This is NOT the same as `conv(a,a)[k]` in general (unless `a` is symmetric).

So `w` is NOT symmetric in general. We need to compute both `T1` and `T2` separately.

For efficiency, let me use FFT:
- `T1[j] = (w ⋆ a)[j]` (cross-correlation of w and a)
  = `ifft(fft(w, L) * conj(fft(a, L)))[:n]` where L is the FFT size

- `T2[j] = (a ⋆ w)[j]` (cross-correlation of a and w, or equivalently w and a with sign flip)
  
  Actually, `T2[j] = sum_k w[k]*a[j-k]`. If I define `a_rev[k] = a[n-1-k]`, then:
  `T2[j] = sum_k w[k]*a[j-k] = sum_k w[k]*a_rev[n-1-(j-k)] = sum_k w[k]*a_rev[n-1-j+k]`
  
  This is the cross-correlation of `w` and `a_rev` at some lag... 

You know what, let me just use the direct formula for the gradient when `w` is concentrated (which it will be for small beta). For the general case, I'll use a simple O(n^2) computation or FFT.

Let me step back and think about the overall algorithm design:

**Algorithm: LP-based iterative improvement + gradient polish**

Phase 1: Get a good initial sequence (from parent or profiles)
Phase 2: Apply LP-based improvement (the SOTA approach) for a few iterations
Phase 3: Apply gradient-based polish (subgradient or smooth)
Phase 4: Try different lengths and parameters

The key difference from the previous code:
- Use LP for the main improvement (more principled, avoids the issues with log-sum-exp approximation)
- Use subgradient method for polishing (simpler, more robust)
- Try a wider range of initial conditions

Let me now write the code. I'll focus on:
1. Efficient convolution using FFT
2. LP-based improvement using scipy.linprog or cvxpy
3. Subgradient descent for polishing
4. Multi-restart with different initial conditions and lengths

For the LP, I'll use `scipy.optimize.linprog` with the HiGHS solver (available in recent scipy versions) or cvxpy with GLOP/GLPK.

Actually, let me check: the constraint matrix for the LP is (2n-1) x n. For n=2048, that's 4095 x 2048. With HiGHS, this should solve in well under a second.

"""
LP-based iterative refinement with subgradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses the active-constraint LP from the SOTA
combined with subgradient descent and multi-restart strategies.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize
import time

def full_convolve(a, b):
    """Full convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = fft(a, N)
    fb = fft(b, N)
    return np.real(ifft(fa * fb, N)[:out_n])

def score_fn(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def score_norm(a):
    """Score with sum(a)=1 assumed: 2*n*max(conv(a,a))."""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def lp_step(a, M=None, method='highs'):
    """
    One LP step: given a, find g >= 0 maximizing sum(g)
    subject to conv(a,g)[k] <= M for all k.
    Returns g (normalized to sum 1).
    """
    n = len(a)
    if M is None:
        c = full_convolve(a, a)
        M = float(c.max())
    
    # Build constraint matrix: (2n-1) x n
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A_ub @ g <= M
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        # a[j] appears in constraints k = j, j+1, ..., j+n-1
        # A_ub[j+m, j] = a[j] for m = 0, ..., n-1 ... no wait
        # conv(a,g)[k] = sum_j a[j]*g[k-j]
        # Coefficient of g[i] in constraint k: a[k-i] if 0 <= k-i < n
        # So A_ub[k, i] = a[k-i] if 0 <= k-i < n else 0
        # Equivalently: for row k, column i gets a[k-i] if 0<=k-i<n
        # For column i: a[k-i] is nonzero when 0<=k-i<n, i.e., i<=k<i+n
        # So A_ub[i:i+n, i] = a[n-1:n-1:-1] ... let me think again
        # A_ub[k,i] = a[k-i] for max(0,i) <= k <= min(2n-2, i+n-1)
        # For a fixed column i, the nonzeros are in rows i to i+n-1 (clipped to [0, 2n-2])
        lo = max(0, i)
        hi = min(2*n-2, i+n-1)
        A_ub[lo:hi+1, i] = a[(hi-i):(lo-i + (hi-lo)) + 1]  # a[k-i] for k=lo..hi => a[lo-i..hi-i]
    
    # Actually let me just do it more carefully:
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A_ub[k, i] = a[k-i] if 0 <= k-i < n else 0
        # k-i in [0, n-1] => i in [k-n+1, k] and i in [0, n-1]
        i_lo = max(0, k-n+1)
        i_hi = min(k, n-1)
        if i_lo <= i_hi:
            A_ub[k, i_lo:i_hi+1] = a[k-i_hi:k-i_lo+1][::-1]
            # a[k-i_hi], a[k-i_hi+1], ..., a[k-i_lo]
            # which is a[k-i_hi : k-i_lo+1] reversed
    
    # LP: maximize sum(g) subject to A_ub @ g <= M, g >= 0
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method=method)
    
    if res.success:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def lp_improve(a, n_iter=10, t=0.5):
    """Apply LP-based improvement for n_iter steps."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best = a.copy()
    best_score = score_norm(a)
    
    for it in range(n_iter):
        c = full_convolve(a, a)
        M = float(c.max())
        g = lp_step(a, M)
        if g is None:
            break
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
        a = a_new
    
    return best

def subgradient_polish(a, n_iter=5000, lr=0.01, beta=0.01):
    """Subgradient method for polishing."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best = a.copy()
    best_score = score_norm(a)
    
    for it in range(n_iter):
        c = full_convolve(a, a)
        cm = float(c.max())
        
        # Smooth weights
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient: grad[j] = 4*n*a[j]*(h[j] - W)
        # where h[j] = sum_k w[k]*a[k-j] (with zero padding)
        # and W = sum_k w[k]*c[k]
        W = float(np.dot(w, c))
        
        # Compute h[j] = sum_k w[k] * a[k-j] for j=0..n-1
        # This is cross-correlation of w and a
        # h[j] = sum_k w[k] * a[k-j]
        # = (w convolved with reversed a) at appropriate position
        a_rev = a[::-1]
        w_conv_a = full_convolve(w, a_rev)
        # w_conv_a[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-m+k]
        # h[j] = sum_k w[k]*a[k-j]
        # Let i = k-j: h[j] = sum_i w[i+j]*a[i]
        # w_conv_a[m] = sum_k w[k]*a[n-1-(m-k)] = sum_k w[k]*a[n-1-m+k]
        # Set n-1-m+k = k-j => n-1-m = -j => m = n-1+j
        # So h[j] = w_conv_a[n-1+j] ... let me verify
        # w_conv_a[n-1+j] = sum_k w[k]*a[n-1-(n-1+j)+k] = sum_k w[k]*a[k-j] ✓
        
        h = w_conv_a[n-1:n-1+n]
        
        grad = 4.0 * n * a * (h - W)
        
        # Update
        a = a - lr * grad / n  # normalize step size
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        sc = score_norm(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
        
        # Anneal
        if it % 200 == 0:
            beta *= 0.9
            lr *= 0.995
    
    return best, best_score

def make_ramp(n, offset=0.0):
    """Create a ramp profile with optional offset."""
    t = np.linspace(0, 1, n)
    a = offset + t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_quadratic(n, a1, a2):
    """Create a quadratic profile."""
    t = np.linspace(0, 1, n)
    u = 2*t - 1
    a = 1.0 + a1*u + a2*u**2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cubic(n, a1, a3):
    """Create a cubic profile."""
    t = np.linspace(0, 1, n)
    u = 2*t - 1
    a = 1.0 + a1*u + a3*u**3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
