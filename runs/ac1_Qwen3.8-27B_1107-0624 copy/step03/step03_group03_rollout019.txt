Combines aggressive LP-based spreading, log-sum-exp gradient, and subgradient
polishing with Nemirovski-Yudin averaging. Uses sparse LP for speed and

import numpy as np
import time
import scipy.optimize
from scipy.sparse import lil_matrix, csc_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_mat_sparse(a):
    """Build sparse matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[j])
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = data[i]
    return A.tocsr()


def _lp_spread(a, M, top_frac=1.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Uses sparse matrix and HiGHS solver."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_mat_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    
    # Optionally reduce constraints: only keep top fraction of tight constraints
    if top_frac < 1.0:
        c_aa = _fft_convolve(a)
        threshold = np.percentile(c_aa, (1 - top_frac) * 100)
        keep = c_aa >= threshold
        A_red = A[keep, :]
        b_red = b_ub[keep]
    else:
        A_red = A
        b_red = b_ub
    
    res = scipy.optimize.linprog(c_obj, A_ub=A_red, b_ub=b_red,
                                  bounds=[(0, None)] * n, method='highs',
                                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def _line_search(a, g, max_iter=30):
    """Find best t in [0,1.2] for blend (1-t)*a + t*g using golden section."""
    g = np.asarray(g, dtype=np.float64)
    
    def f(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-10:
            return np.inf
        x = x / s
        return _score_norm(x)
    
    # First find a good range
    f0 = f(0.0)
    f_half = f(0.5)
    f_one = f(1.0)
    
    if f_one < f_half < f0:
        lo, hi = 0.5, 1.2
    elif f_half < f0:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = 0.0, 0.5
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    best_t = (lo + hi) / 2.0
    fb = f(best_t)
    if fb > f0:
        best_t = 0.0
    return best_t

def _lse_step(a, z, state, T, lr):
    """One Adam step in logit space using log-sum-exp smoothed max."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -40, 40)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient computation via FFT
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    # Project to tangent space
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val

            no_imp += 1
            if no_imp % 5000 == 0:
                lr = max(0.01, lr * 0.9)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(77 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        
        T = max(T_end, T * T_ratio)
    
    return best_a


def _subgrad_nx(a, deadline, max_iter=50000, eta0=0.05):
    """Subgradient method with Nemirovski-Yudin averaging."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    # NX running average
    xbar = a.copy()
    total_weight = 1.0
    eta = eta0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if it > 0 and it % 10000 == 0:
            print(f"    SubGrad iter {it}: score={best_score:.10f}, eta={eta:.4f}")
        
        c = _fft_convolve(a)
        m_val = float(c.max())
        
        # Subgradient at all max positions
        max_mask = c >= m_val - 1e-10
        n_max = int(max_mask.sum())
        
        # Compute gradient vector g[i] = sum_{k: c[k]=max} d c[k]/d a[i]
        # d c[k]/d a[i] = a[k-i] (if valid) + a[i-k] (if valid)
        g = np.zeros(n)
        for k in np.where(max_mask)[0]:
            # a[k-i] for valid i
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            for i in range(lo1, hi1 + 1):
                g[i] += a[k - i]
            # a[i-k] for valid i
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            for i in range(lo2, hi2 + 1):
                g[i] += a[i - k]
        
        # Average over max positions
        g /= n_max
        
        # Project to tangent space
        g = g - float(np.mean(g))
        
        # Step
        a_old = a.copy()
        a = a - eta * g
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Score
        c_new = _fft_convolve(a)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # NX update
        w = 1.0 / ((it + 1) * eta)
        xbar = (total_weight * xbar + w * (a - a_old)) / (total_weight + w)
        total_weight += w
        xbar = np.maximum(xbar, 1e-15)
        xbar = xbar / xbar.sum()
        
        # Decay eta
        eta *= 0.9995
        eta = max(eta, 1e-4)
    
    # Compare best point with NX average
    c_avg = _fft_convolve(xbar)
    avg_score = float(2.0 * n * c_avg.max())
    if avg_score < best_score:
        best_a = xbar.copy()
        best_score = avg_score
    
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Gaussian
    g = np.exp(-8.0 * (t - 0.5) ** 2)
    g = np.maximum(g, 1e-10)
    profiles.append(g / g.sum())
    
    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    rc = np.maximum(rc, 1e-10)
    profiles.append(rc / rc.sum())
    
    # Flat
    profiles.append(np.ones(n) / n)
    
    # Edge-weighted
    ew = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
    ew = np.maximum(ew, 1e-10)
    profiles.append(ew / ew.sum())
    
    # SuperGaussian
    sg = np.exp(-3.0 * (t - 0.5) ** 4)
    sg = np.maximum(sg, 1e-10)
    profiles.append(sg / sg.sum())
    
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LSE + LP + subgradient with NX averaging.
    Explores multiple lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    print(f"Starting optimization, budget={budget_s}s")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Determine lengths to try, prioritizing the parent length
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    # Add promising other lengths
    for ln in [1024, 2048, 512]:
        if ln not in lengths:
            lengths.append(ln)
    
    # Sort by expected quality (parent first, then by size)
    lengths.sort(key=lambda x: (0 if parent is not None and x == parent.size else 1, -x))
    
    print(f"Lengths to try: {lengths}")
    
    # Allocate time per length
    n_lengths = min(len(lengths), 3)  # try at most 3 lengths
    lengths = lengths[:n_lengths]
    
    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 30:
            break
        
        remaining = deadline - time.monotonic()
        if li == 0:
            time_alloc = remaining * 0.5
        elif li == 1:
            time_alloc = remaining * 0.3
        else:
            time_alloc = remaining * 0.2
        
        print(f"\n=== Length n={n}, time budget={time_alloc:.0f}s ===")
        phase_deadline = time.monotonic() + time_alloc
        
        # Generate candidates
        candidates = []
        if parent is not None:
            p_interp = _interpolate_profile(parent, n)
            candidates.append(p_interp)
            # Perturbed versions
            for pi in range(3):
                pp = p_interp * (1.0 + 0.06 * rng.normal(size=n))
                pp = np.maximum(pp, 0.0)
                pp = pp / pp.sum()
                candidates.append(pp)
        
        profiles = _make_profiles(n)
        candidates.extend(profiles)
        
        # Score all candidates
        best_cand = None
        best_cand_score = np.inf
        for ci, c in enumerate(candidates):
            sc = _score(c)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = c.copy()
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = c
            print(f"  Candidate {ci}: score={sc:.8f}")
        
        if best_cand is None:
            best_cand = np.ones(n) / n
            best_cand_score = _score(best_cand)
        
        # Phase 1: LSE optimization on best candidate
        lse_time = min(60.0, time_alloc * 0.2)
        if lse_time > 10:
            lse_end = time.monotonic() + lse_time
            a_lse = _lse_optimize(best_cand, lse_end, max_iter=80000,
                                   T_start=0.03, T_end=1e-6, lr=0.08)
            sc_lse = _score(a_lse)
            print(f"  LSE: score={sc_lse:.10f}")
            if sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            best_cand = a_lse if sc_lse < best_cand_score else best_cand
            best_cand_score = min(best_cand_score, sc_lse)
        
        # Phase 2: Main LP iteration
        lp_time = min(120.0, time_alloc * 0.5)
        if lp_time > 15 and time.monotonic() < phase_deadline - 20:
            lp_end = time.monotonic() + lp_time
            a_lp = _lp_iteration(best_cand, lp_end, max_iter=5000,
                                  inflate_start=1.0005, inflate_end=1.00001)
            sc_lp = _score(a_lp)
            print(f"  LP: score={sc_lp:.10f}")
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            best_cand = a_lp if sc_lp < best_cand_score else best_cand
            best_cand_score = min(best_cand_score, sc_lp)
        
        # Phase 3: Subgradient with NX averaging
        sp_time = min(40.0, time_alloc * 0.15)
        if sp_time > 10 and time.monotonic() < phase_deadline - 15:
            sp_end = time.monotonic() + sp_time
            a_sp = _subgrad_nx(best_cand, sp_end, max_iter=30000, eta0=0.03)
            sc_sp = _score(a_sp)
            print(f"  SubGrad: score={sc_sp:.10f}")
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            best_cand = a_sp if sc_sp < best_cand_score else best_cand
            best_cand_score = min(best_cand_score, sc_sp)
        
        # Phase 4: Final fine LP polish
        fp_time = min(30.0, time_alloc * 0.1)
        if fp_time > 10 and time.monotonic() < phase_deadline - 10:
            fp_end = time.monotonic() + fp_time
            a_fp = _lp_iteration(best_cand, fp_end, max_iter=3000,
                                  inflate_start=1.0001, inflate_end=1.000005)
            sc_fp = _score(a_fp)
            print(f"  Final LP: score={sc_fp:.10f}")
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
        
        # Phase 5: One more LSE pass with very small T
        if time.monotonic() < phase_deadline - 10:
            flse_time = min(20.0, phase_deadline - time.monotonic() - 10)
            if flse_time > 5:
                flse_end = time.monotonic() + flse_time
                a_flse = _lse_optimize(best_cand, flse_end, max_iter=30000,
                                        T_start=0.01, T_end=1e-7, lr=0.03)
                sc_flse = _score(a_flse)
                print(f"  Final LSE: score={sc_flse:.10f}")
                if sc_flse < best_score:
                    best_score = sc_flse
                    best_seq = a_flse.copy()
    
    # Final: one more LP polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline:
        final_time = min(30.0, deadline - time.monotonic())
        if final_time > 10:
            final_end = time.monotonic() + final_time
            a_final = _lp_iteration(best_seq, final_end, max_iter=2000,
                                     inflate_start=1.0001, inflate_end=1.000005)
            sc_final = _score(a_final)
            print(f"\nAbsolute final LP: score={sc_final:.10f}")
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
    
    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(512) / 512.0
    
    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")
    
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
