Multi-scale L-BFGS-B optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax parameterization, adaptive smooth-max temperature, sparse LP refinement,
and multi-length search up to n=2048. Novel peak-equalization step directly targets
balancing the top convolution lags.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy import sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _obj_and_grad(z, n, beta):
    """Compute smooth-max objective and gradient w.r.t. z (softmax space)."""
    a = _softmax(z)
    c = _fft_convolve(a)
    val, w = _smooth_max_val(c, beta)
    # Gradient of smooth max w.r.t. a
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    # Chain rule for softmax
    dot = float(np.dot(a, grad_a))
    gz = a * (grad_a - dot)
    return 2.0 * n * val, gz


def _lbfgsb_optimize(a, deadline, max_time=120.0, beta_init=None, max_iter=2000):
    """L-BFGS-B optimization in softmax space with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.08 * c_max)

    z = np.log(a + 1e-30)
    z -= np.mean(z)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    t_start = time.monotonic()
    no_imp = 0

    for round_num in range(20):
        if time.monotonic() > deadline:
            break

        # Define objective and gradient for this round's beta
        def fun_g(z_val, _beta=beta):
            val, grad = _obj_and_grad(z_val, n, _beta)
            return val, grad

        # Run L-BFGS-B
        res = minimize(fun_g, z, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                'gtol': 1e-8, 'maxls': 50})

        z_new = res.x
        if not np.all(np.isfinite(z_new)):
            break
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new -= np.mean(z_new)

        a_new = _softmax(z_new)
        c_new = _fft_convolve(a_new)
        c_max_new = float(c_new.max())
        score_new = float(2.0 * n * c_max_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Adapt beta: decrease for sharper max
        if no_imp == 0:
            beta = max(1e-10, beta * 0.7)
        else:
            beta = max(beta * 0.85, 1e-10)

        z = z_new

        # Check time
        elapsed = time.monotonic() - t_start
        if elapsed > max_time:
            break
        if no_imp > 5:
            break

    # Final exact evaluation
    c_final = _fft_convolve(best_a)
    final_score = float(2.0 * len(best_a) * c_final.max())
    return best_a, final_score


def _make_sparse_conv_matrix(a):
    """Create sparse matrix for conv(a, g) constraints."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(a[idx])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_spread(a, M):
    """LP: maximize sum(g) s.t. conv(a,g)_k <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_sparse_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=500, inflate=1.001, blend=0.25):
    """LP-based spreading refinement with sparse matrix."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    best_a = a.copy()

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 50:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.003, inflate_cur * 0.97)
            if no_imp > max(30, n // 10):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _peak_equalize(a, deadline, max_iter=2000, n_peaks=5, strength=0.08):
    """
    Direct peak equalization: identify top convolution lags and
    apply targeted pressure to reduce the dominant ones.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]

        # Weights: how much each peak exceeds the median
        med = np.median(c)
        weights = np.maximum(top_vals - med, 1e-10)
        weights = weights / weights.sum()

        # Compute pressure on each coordinate from top peaks
        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]

        # Apply exponential tilting
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        a_new = a * np.exp(-str_cur * pressure / p_max)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.25, str_cur * 1.02)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.02, str_cur * 0.85)
            if no_imp > 500:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _fine_polish(a, deadline, max_iter=10000):
    """Very fine gradient polish with tiny beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-9, 0.005 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a + 1e-30)
    z -= np.mean(z)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        try:
            val, gz = _obj_and_grad(z, n, beta)
        except Exception:
            break

        if not np.all(np.isfinite(gz)):
            break

        # Simple gradient step with adaptive learning rate
        gnorm = np.linalg.norm(gz)
        if gnorm < 1e-12:
            break
        lr = 0.05 / (gnorm + 1e-8)
        lr = min(lr, 0.1)

        z = z - lr * gz
        z = np.clip(z, -80.0, 80.0)
        z -= np.mean(z)

        a = _softmax(z)
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                beta = max(1e-12, beta * 0.7)

        if it % 500 == 0 and it > 0:
            beta = max(1e-12, beta * 0.8)

        if no_imp > 5000:
            break

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_bell_profile(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_edge_boost_profile(n, boost=0.3):
    """Profile with slight edge boost - helps reduce central convolution."""
    t = np.linspace(0.0, 1.0, n)
    center = np.exp(-((t - 0.5) / 0.4) ** 2)
    edges = np.exp(-t**2 / 0.05) + np.exp(-(1 - t)**2 / 0.05)
    a = center + boost * edges
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _multi_start_refine(a, deadline, rng, n_starts=5):
    """Try perturbed versions and keep the best."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        scales = [0.03, 0.05, 0.08, 0.12, 0.18]
        noise_scale = scales[i % len(scales)]
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.3:
            continue

        # Quick polish
        sub_deadline = min(deadline, time.monotonic() + 30.0)
        a_ref, sc_ref = _lbfgsb_optimize(a_pert, sub_deadline, max_time=25.0)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Fast L-BFGS-B exploration across multiple lengths
    Phase 2: Focused optimization (L-BFGS-B + LP) on the best
    Phase 3: Peak equalization + fine polish
    Phase 4: Multi-start + final LP
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to search
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [1024, 1536, 2048]:
        if ln not in lengths:
            lengths.append(ln)
    lengths.sort()

    print(f"Search lengths: {lengths}", flush=True)

    # === Phase 1: Fast exploration (allocate ~35% of time) ===
    phase1_deadline = t_start + (deadline - t_start) * 0.35
    print(f"Phase 1: Fast exploration, deadline in {phase1_deadline - time.monotonic():.0f}s", flush=True)

    for n_t in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        # Generate initializations
        inits = []
        if parent is not None:
            inits.append(('parent', _interpolate_profile(parent, n_t)))
        inits.append(('bell', _make_bell_profile(n_t, 0.35)))
        inits.append(('edge', _make_edge_boost_profile(n_t, 0.3)))
        inits.append(('flat_spec', _make_flat_spectrum_init(n_t, rng)))

        # Quick L-BFGS-B on each (short time)
        t_per_init = max(15.0, (phase1_deadline - time.monotonic()) / (len(inits) + 1))
        for name, a0 in inits:
            if time.monotonic() > phase1_deadline - 20:
                break
            sub_deadline = min(phase1_deadline - 10, time.monotonic() + t_per_init)
            a_opt, sc_opt = _lbfgsb_optimize(a0, sub_deadline, max_time=t_per_init - 5)
            sc_actual = _score(a_opt)
            print(f"  n={n_t}, {name}: score={sc_actual:.10f}", flush=True)
            if sc_actual < best_score:
                best_score = sc_actual
                best_seq = a_opt.copy()

    print(f"Phase 1 best: score={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}", flush=True)

    if best_seq is None:
        best_seq = _make_bell_profile(1024, 0.35)
        best_score = _score(best_seq)

    # === Phase 2: Focused optimization (allocate ~35% of time) ===
    phase2_deadline = t_start + (deadline - t_start) * 0.70
    n_best = best_seq.size
    print(f"Phase 2: Focused optimization on n={n_best}, deadline in {phase2_deadline - time.monotonic():.0f}s", flush=True)

    # Longer L-BFGS-B
    t_lbfgs = min(180.0, (phase2_deadline - time.monotonic()) * 0.45)
    a_opt, sc_opt = _lbfgsb_optimize(best_seq, min(phase2_deadline - 60, time.monotonic() + t_lbfgs),
                                     max_time=t_lbfgs)
    sc_actual = _score(a_opt)
    print(f"  L-BFGS-B: score={sc_actual:.10f}", flush=True)
    if sc_actual < best_score:
        best_score = sc_actual
        best_seq = a_opt.copy()

    # LP refinement
    if time.monotonic() < phase2_deadline - 30:
        n_lp = best_seq.size
        lp_max_iter = max(20, min(200, int(120.0 / max(0.5, n_lp / 512.0))))
        lp_deadline = min(phase2_deadline - 20, time.monotonic() + 150.0)
        print(f"  LP refine (n={n_lp}, max_iter={lp_max_iter}), budget={lp_deadline - time.monotonic():.0f}s", flush=True)
        a_lp = _lp_refine(best_seq, lp_deadline, max_iter=lp_max_iter, inflate=1.0008, blend=0.2)
        sc_lp = _score(a_lp)
        print(f"  LP: score={sc_lp:.10f}", flush=True)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

    # === Phase 3: Peak equalization + fine polish ===
    phase3_deadline = t_start + (deadline - t_start) * 0.88
    print(f"Phase 3: Peak equalize + polish, deadline in {phase3_deadline - time.monotonic():.0f}s", flush=True)

    # Peak equalization
    if time.monotonic() < phase3_deadline - 30:
        ps_deadline = min(phase3_deadline - 30, time.monotonic() + 60.0)
        if ps_deadline - time.monotonic() > 10:
            a_ps = _peak_equalize(best_seq, ps_deadline, max_iter=2000, n_peaks=5, strength=0.06)
            sc_ps = _score(a_ps)
            print(f"  Peak equalize: score={sc_ps:.10f}", flush=True)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()

    # Fine polish (L-BFGS-B with tiny beta)
    if time.monotonic() < phase3_deadline - 10:
        fp_deadline = min(phase3_deadline, time.monotonic() + 60.0)
        if fp_deadline - time.monotonic() > 10:
            a_fp, sc_fp = _lbfgsb_optimize(best_seq, fp_deadline, max_time=fp_deadline - time.monotonic() - 5,
                                           beta_init=1e-5)
            sc_fp = _score(a_fp)
            print(f"  Fine polish: score={sc_fp:.10f}", flush=True)
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()

    # === Phase 4: Multi-start + final LP ===
    print(f"Phase 4: Multi-start + final LP, deadline in {deadline - time.monotonic():.0f}s", flush=True)

    # Multi-start
    if time.monotonic() < deadline - 60:
        ms_deadline = min(deadline - 60, time.monotonic() + 120.0)
        a_ms = _multi_start_refine(best_seq, ms_deadline, rng, n_starts=5)
        sc_ms = _score(a_ms)
        print(f"  Multi-start: score={sc_ms:.10f}", flush=True)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()

    # Final LP pass
    if time.monotonic() < deadline - 10:
        lp_final_deadline = deadline - 5
        n_lp2 = best_seq.size
        lp_iter2 = max(5, min(50, int(60.0 / max(0.5, n_lp2 / 512.0))))
        if lp_final_deadline - time.monotonic() > 10:
            a_final = _lp_refine(best_seq, lp_final_deadline, max_iter=lp_iter2,
                                 inflate=1.0005, blend=0.15)
            sc_final = _score(a_final)
            print(f"  Final LP: score={sc_final:.10f}", flush=True)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = _make_bell_profile(1024, 0.35)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s <= 0.01:
        final_seq = _make_bell_profile(1024, 0.35)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence score: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in final_seq]
