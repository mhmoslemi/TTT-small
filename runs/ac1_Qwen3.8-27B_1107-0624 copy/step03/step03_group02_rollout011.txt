def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_direct(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_weighted(w, a):
    """Compute 2*conv(w, a_reversed) truncated to length len(a), via FFT."""
    n = len(a)
    N = _next_pow2(2 * n)
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    result = np.fft.irfft(fw * fa, N)
    h = result[n - 1:2 * n - 1]
    return 2.0 * h


def _smooth_max_grad(a, beta):
    """Gradient of smooth-approximation to max(conv(a,a)) using FFT."""
    c = _fft_convolve(a)
    c_shifted = c - c.max()
    c_shifted = np.clip(c_shifted / beta, -50.0, 50.0)
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    grad = _convolve_weighted(w, a)
    return grad, c.max(), w

def _coordinate_descent(a_init, deadline, max_rounds=50, verbose=False):
    """Coordinate descent: optimize one element at a time to reduce max convolution."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c0 = _fft_convolve(a)
    best_score = float(2.0 * n * c0.max())

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try to find best value for a[i]
            # conv(a,a) as function of a[i]:
            # c[k] changes: for k where i and k-i are both valid
            # d c[k]/d a[i] = a[k-i] (if valid) + a[i-k] (if valid) [but i-k < 0 always if k < i]
            # Actually: c[k] = sum_j a[j]*a[k-j]. 
            # d c[k]/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            # For fixed other elements, c[k] = a[i]*a[k-i] + a[i-k]*a[i] + rest
            #   = a[i]*(a[k-i] + a[i-k]) + rest  (where rest doesn't involve a[i])
            # So c[k] is linear in a[i] for k != 2i, and quadratic (a[i]^2 + rest) for k = 2i.
            
            # Simpler: just do a few candidate values
            old_val = a[i]
            if old_val < 1e-12:
                # Try small positive values
                candidates = [old_val, 0.01 / n, 0.001]
            else:
                candidates = [old_val, old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5, 0.0]
            
            best_local = best_score
            best_candidate = a.copy()
            for cv in candidates:
                a[i] = max(0.0, cv)
                s_curr = a.sum()
                if s_curr < 0.01:
                    a[i] = old_val
                    continue
                c_curr = _fft_convolve(a)
                sc_curr = float(2.0 * n * c_curr.max() / (s_curr * s_curr))
                if sc_curr < best_local - 1e-12:
                    best_local = sc_curr
                    best_candidate = a.copy()
            
            a = best_candidate
            a[i] = best_candidate[i]
            if best_candidate is not a:
                improved = True
        
        # Normalize
        s = a.sum()
        if s > 0.01:
            a = a / s
        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max() / (a.sum() ** 2))
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            improved = True
        
        if not improved and rd > 5:
            break

    return best_a


def _lp_spread(a, M):
    """Find g >= 0 maximizing sum(g) subject to conv(a,g) <= M elementwise."""
    n = len(a)
    c_obj = -np.ones(n)
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
    # This is a matrix A @ g <= M where A[k,j] = a[j] if 0 <= k-j < n else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _make_profile(n, type="bell", **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if type == "uniform":
        a = np.ones(n)
    elif type == "bell":
        width = kwargs.get("width", 0.4)
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    elif type == "cosh":
        alpha = kwargs.get("alpha", 2.0)
        a = np.cosh(alpha * (t - 0.5))
    elif type == "poly":
        deg = kwargs.get("deg", 2)
        u = 2.0 * t - 1.0
        a = np.ones(n)
        for d in range(2, deg + 1, 2):
            a += 0.3 * u ** d
    elif type == "sin":
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
    elif type == "rect":
        frac = kwargs.get("frac", 0.5)
        a = np.where((t > 0.5 - frac / 2) & (t < 0.5 + frac / 2), 1.0, 0.01)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    # Define list of (length, profile_type, kwargs) to explore
    if parent is not None:
        lengths_to_try = [parent.size]
        if parent.size != 1024:
            lengths_to_try.append(1024)
        if parent.size not in [512, 1024, 1536, 2048]:
            lengths_to_try.append(1024)
    else:
        lengths_to_try = [512, 1024]

    # Deduplicate and sort by priority (parent length first)
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - (parent.size if parent is not None else 1024)))

    # --- Phase A: Optimize parent at its native length ---
    if parent is not None and time.monotonic() < deadline - 300:
        n_parent = parent.size
        print(f"\n=== Phase A: Optimizing parent @ n={n_parent} ===")
        
        # A1: Adam smooth-max (generous budget)
        d_a1 = min(deadline - 250, t_start + 400)
        a1, s1 = _adam_optimize(parent, d_a1, max_iter=30000, lr=0.025,
                                beta_init=0.03, beta_decay=0.95,
                                switch_to_exact_iter=15000, verbose=True)
        s1_full = _score_direct(a1)
        if s1_full < best_score:
            best_score = s1_full
            best_seq = a1.copy()
        print(f"  A1 Adam: {s1_full:.10f}")

        # A2: Multi-peak subgradient
        if time.monotonic() < deadline - 200:
            d_a2 = min(deadline - 180, time.monotonic() + 100)
            a2, s2 = _multi_peak_subgrad(best_seq, d_a2, max_iter=5000, top_k=5)
            s2_full = _score_direct(a2)
            if s2_full < best_score:
                best_score = s2_full
                best_seq = a2.copy()
            print(f"  A2 Subgrad: {s2_full:.10f}")

        # A3: LP spread
        if time.monotonic() < deadline - 150:
            d_a3 = min(deadline - 130, time.monotonic() + 120)
            a3 = _lp_iterate(best_seq, d_a3, max_iter=1000, inflate=1.003, blend=0.25)
            s3_full = _score_direct(a3)
            if s3_full < best_score:
                best_score = s3_full
                best_seq = a3.copy()
            print(f"  A3 LP: {s3_full:.10f}")

        # A4: Fine Adam
        if time.monotonic() < deadline - 100:
            d_a4 = min(deadline - 80, time.monotonic() + 100)
            a4, s4 = _adam_optimize(best_seq, d_a4, max_iter=20000, lr=0.008,
                                    beta_init=0.01, beta_decay=0.97,
                                    switch_to_exact_iter=10000)
            s4_full = _score_direct(a4)
            if s4_full < best_score:
                best_score = s4_full
                best_seq = a4.copy()
            print(f"  A4 Fine Adam: {s4_full:.10f}")

        # A5: Peak shave
        if time.monotonic() < deadline - 60:
            d_a5 = min(deadline - 50, time.monotonic() + 50)
            a5 = _peak_shave(best_seq, d_a5, max_iter=1000)
            s5_full = _score_direct(a5)
            if s5_full < best_score:
                best_score = s5_full
                best_seq = a5.copy()
            print(f"  A5 PeakShave: {s5_full:.10f}")

    # --- Phase B: Try different lengths ---
    for n_test in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        if parent is not None and n_test == parent.size:
            continue  # Already done

        print(f"\n=== Phase B: Trying n={n_test} ===")
        
        # Get initial profile at this length
        if parent is not None:
            init = _interpolate_profile(parent, n_test)
        else:
            init = _make_profile(n_test, "cosh", alpha=2.5)

        # B1: Quick Adam
        d_b1 = min(deadline - 100, time.monotonic() + 150)
        if time.monotonic() < d_b1 - 10:
            a_b1, s_b1 = _adam_optimize(init, d_b1, max_iter=20000, lr=0.02,
                                        beta_init=0.04, beta_decay=0.95,
                                        switch_to_exact_iter=10000)
            s_b1_full = _score_direct(a_b1)
            if s_b1_full < best_score:
                best_score = s_b1_full
                best_seq = a_b1.copy()
            print(f"  B1 Adam n={n_test}: {s_b1_full:.10f}")

        # B2: Subgradient
        if time.monotonic() < deadline - 80:
            d_b2 = min(deadline - 70, time.monotonic() + 60)
            a_b2, s_b2 = _multi_peak_subgrad(a_b1 if 'a_b1' in dir() else init, d_b2,
                                             max_iter=3000, top_k=5)
            s_b2_full = _score_direct(a_b2)
            if s_b2_full < best_score:
                best_score = s_b2_full
                best_seq = a_b2.copy()
            print(f"  B2 Subgrad n={n_test}: {s_b2_full:.10f}")

        # B3: LP
        if time.monotonic() < deadline - 60:
            d_b3 = min(deadline - 50, time.monotonic() + 60)
            a_b3 = _lp_iterate(a_b2 if 'a_b2' in dir() else init, d_b3,
                               max_iter=800, inflate=1.002, blend=0.2)
            s_b3_full = _score_direct(a_b3)
            if s_b3_full < best_score:
                best_score = s_b3_full
                best_seq = a_b3.copy()
            print(f"  B3 LP n={n_test}: {s_b3_full:.10f}")

    # --- Phase C: Diverse initializations at best length ---
    if time.monotonic() < deadline - 100:
        n_best = best_seq.size if best_seq is not None else 1024
        print(f"\n=== Phase C: Diverse inits @ n={n_best} ===")
        
        inits = [
            _make_profile(n_best, "cosh", alpha=1.5),
            _make_profile(n_best, "cosh", alpha=3.0),
            _make_profile(n_best, "bell", width=0.3),
            _make_profile(n_best, "bell", width=0.6),
            _make_profile(n_best, "poly", deg=4),
            _make_profile(n_best, "sin"),
        ]
        # Perturbed best
        inits.append(_random_perturb(best_seq, rng, 0.15))
        inits.append(_random_perturb(best_seq, rng, 0.3))

        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break
            per_budget = min(30.0, (deadline - 80 - time.monotonic()) / max(1, len(inits) - i))
            if per_budget < 8:
                break
            d_ci = time.monotonic() + per_budget
            a_ci, s_ci = _adam_optimize(init, d_ci, max_iter=8000, lr=0.02,
                                        beta_init=0.05, beta_decay=0.94,
                                        switch_to_exact_iter=4000)
            s_ci_full = _score_direct(a_ci)
            if s_ci_full < best_score:
                best_score = s_ci_full
                best_seq = a_ci.copy()
            print(f"  C-{i}: {s_ci_full:.10f}")

    # --- Phase D: Final intensive optimization on best ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase D: Final focus (best={best_score:.10f}) ===")
        remaining = deadline - time.monotonic()
        
        # D1: Long Adam run
        if remaining > 50:
            d_d1 = min(deadline - 20, time.monotonic() + remaining * 0.4)
            a_d1, s_d1 = _adam_optimize(best_seq, d_d1, max_iter=50000, lr=0.005,
                                        beta_init=0.008, beta_decay=0.98,
                                        switch_to_exact_iter=25000)
            s_d1_full = _score_direct(a_d1)
            if s_d1_full < best_score:
                best_score = s_d1_full
                best_seq = a_d1.copy()
            print(f"  D1: {s_d1_full:.10f}")

        # D2: LP polish
        if time.monotonic() < deadline - 15:
            d_d2 = min(deadline - 10, time.monotonic() + 40)
            a_d2 = _lp_iterate(best_seq, d_d2, max_iter=1000, inflate=1.001, blend=0.15)
            s_d2_full = _score_direct(a_d2)
            if s_d2_full < best_score:
                best_score = s_d2_full
                best_seq = a_d2.copy()
            print(f"  D2 LP: {s_d2_full:.10f}")

        # D3: Ultra-fine Adam
        if time.monotonic() < deadline - 8:
            d_d3 = deadline - 5
            a_d3, s_d3 = _adam_optimize(best_seq, d_d3, max_iter=30000, lr=0.002,
                                        beta_init=0.003, beta_decay=0.99,
                                        switch_to_exact_iter=15000)
            s_d3_full = _score_direct(a_d3)
            if s_d3_full < best_score:
                best_score = s_d3_full
                best_seq = a_d3.copy()
            print(f"  D3 Ultra: {s_d3_full:.10f}")

        # D4: Peak shave
        if time.monotonic() < deadline - 3:
            a_d4 = _peak_shave(best_seq, deadline - 2, max_iter=500)
            s_d4_full = _score_direct(a_d4)
            if s_d4_full < best_score:
                best_score = s_d4_full
                best_seq = a_d4.copy()
            print(f"  D4 Shave: {s_d4_full:.10f}")

    # Final validation and formatting
    if best_seq is None:
        best_seq = _make_profile(512, "cosh", alpha=2.0)
        best_score = _score_direct(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    # Validate with evaluate_sequence if feasible
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval  # Should be the same
            else:
                # Invalid, use fallback
                print("evaluate_sequence returned inf, using fallback")
                final_seq = _make_profile(512, "cosh", alpha=2.0)
                best_score = _score_direct(final_seq)
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, elapsed={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
