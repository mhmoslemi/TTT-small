"""
Optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated Adam in log-space, aggressive LP spreading,
and coordinate-wise peak shaving with adaptive smooth-max temperature.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 80.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _grad_smooth_max(a, beta):
    """Gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_and_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.abs(np.real(np.fft.ifft(X))[:n])
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _bell_profile(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _u_profile(n, gamma=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + gamma * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_to_length(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _nesterov_adam_optimize(a, deadline, max_iter=100000, lr=0.03, beta_init=None,
                             nesterov_mu=0.9, adapt_beta=True, aggressive_cooldown=True):
    """
    Nesterov-accelerated Adam optimization in log-space.
    Key features:
    - Nesterov lookahead for faster convergence
    - Adaptive smooth-max temperature (beta)
    - Aggressive cooling schedule
    - Periodic restarts with noise
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))
    z_prev = z.copy()

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    mu = nesterov_mu

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        z_look = z + mu * (z - z_prev)

        # Convert to probability
        e = np.exp(z_look)
        p = e / e.sum()

        # Compute gradient of smooth max
        c = _fft_convolve(p)
        _, w = _smooth_max_and_weights(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p  # small regularization
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)  # gradient in log-space
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            z_prev = z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        # Adam update
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        # Evaluate
        e = np.exp(z_new)
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:

def _coordinate_descent_polish(a, deadline, max_iter=20000, n_top=3, step=0.02):
    """
    Coordinate-wise descent targeting the top peaks of the convolution.
    For each peak, reduce the coordinates that contribute most to it.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    step_cur = step

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Find top peaks (excluding k=0 which is always a_0^2)
        # Actually include all peaks
        top_idx = np.argsort(c)[::-1][:n_top]

        # For each peak, compute which coordinates to reduce
        # (a*a)_k = sum_i a_i * a_{k-i}
        # To reduce (a*a)_k, we should reduce the a_i that appear with large a_{k-i}
        improvement = 0.0
        for k in top_idx:
            # Get the contributing pairs
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # pressure[i] = a_{k-i} for i in [lo, hi]
            indices = np.arange(lo, hi + 1)
            partners = k - indices
            pressure = a[partners] * a[indices]
            total_p = pressure.sum()
            if total_p < 1e-20:
                continue
            # Reduce coordinates proportional to their contribution
            # Only reduce if it helps (i.e., the coordinate isn't needed for other peaks)
            frac = pressure / total_p
            # Apply small reduction
            reduce_amt = step_cur * frac * total_p
            a[indices] -= reduce_amt

        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c_new = _fft_convolve(a)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a.copy()
            no_imp = 0
        else:
            # Revert and try smaller step
            a = best_a.copy()
            no_imp += 1
            if no_imp % 50 == 0:
                step_cur = max(0.001, step_cur * 0.8)
            if no_imp > 500:
                break

    return best_a


def _peak_shave_smooth(a, deadline, max_iter=10000, n_peaks=5, strength=0.05):

    a = np.maximum(a, 1e-15)

    n = len(a)

"""
LP-centric optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Sparse LP with bisection on inflation factor for tight convergence
2. LP-dominant time allocation (80% of budget on LP iterations)
3. Nesterov-accelerated gradient for quick resets between LP phases
4. Adaptive blend factor that decreases as convergence improves
5. Multi-phase: warmup -> LP loop -> fine polish -> validation
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 80.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _make_conv_matrix_sparse(a):
    """Create sparse convolution matrix where A @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        jslice = slice(j, j + n)
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(a.tolist())
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_spread_sparse(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0, using sparse matrix."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix_sparse(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _nesterov_gradient(a, deadline, max_iter=50000, lr=0.03, beta_init=None):
    """Nesterov-accelerated gradient in log-space with adaptive smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))
    z_prev = z.copy()

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    mu = 0.9

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        z_look = z + mu * (z - z_prev)
        e = np.exp(z_look)
        p = e / e.sum()

        c = _fft_convolve(p)
        _, w = _smooth_max_and_weights(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            z_prev = z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.95)

        if it % 50 == 0:
            top3 = np.sort(c_new)[::-1][:min(3, len(c_new))]
            if len(top3) >= 2:
                gap = top3[0] - top3[-1]
                if gap < beta * 0.3:
                    beta = max(1e-12, beta * 0.5)
                elif gap > beta * 10.0:
                    beta = min(0.3 * c_new.max(), beta * 1.5)
            if it > 1000:
                beta = max(1e-12, beta * 0.99)

        if no_imp > max(2000, 4 * n):

            r = np.random.default_rng(it * 31 + 7)
            z_restart = best_z + scale * r.standard_normal(n)
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            z = z_restart
            z_prev = z_restart.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            beta = max(1e-8, 0.03 * float(_fft_convolve(best_a).max()))
            no_imp = 0

        z_prev = z
        z = z_new

    return best_a


def _lp_centric_optimize(a, deadline, max_iter=100000):

    blend = 0.08
    no_imp = 0
    it = 0

        it += 1
        c = _fft_convolve(a)
        M = float(c.max())

def _peak_shave(a, deadline, max_iter=5000, n_peaks=5):
    """Exponential-tilting peak shaving."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    strength = 0.04

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]

        pressure = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            jdx = k - idx
            pressure[idx] += a[jdx]

        max_p = pressure.max()
        if max_p < 1e-20:
            break
        a_new = a * np.exp(-strength * pressure / max_p)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            strength = min(0.2, strength * 1.01)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                strength = max(0.005, strength * 0.85)
            if no_imp > 300:
                a = best_a.copy()
                strength = 0.04
                no_imp = 0

        a = a_new

    return best_a


def _flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.abs(np.real(np.fft.ifft(X))[:n])
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _bell_profile(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_to_length(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
