"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based alternating optimization, L-BFGS with smooth max,
and multi-scale search to approach the theoretical lower bound.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.signal import fftconvolve
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute the objective score for a sequence."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1 (for speed during optimization)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind='linear', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'linear':
        a = 1.0 + param * (t - 0.5)
    elif kind == 'quad':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u ** 2
    elif kind == 'cubic':
        u = 2.0 * t - 1.0
        a = 1.0 + param * u ** 3
    elif kind == 'cosine':
        a = 1.0 + param * np.cos(np.pi * t)
    elif kind == 'trapezoid':
        # Trapezoidal shape
        w = param  # width of flat top (fraction)
        a = np.zeros(n)
        for i in range(n):
            if t[i] < (1-w)/2:
                a[i] = 2*t[i] / ((1-w)/2)
            elif t[i] < (1+w)/2:
                a[i] = 1.0
            else:
                a[i] = 2*(1-t[i]) / ((1-w)/2)
        a = np.maximum(a, 1e-12)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _lp_alternate(f, max_iters=50, step_size=0.3, time_limit=60.0):
    """
    LP-based alternating optimization.
    Given f (sum=1), repeatedly solve LP to find best g with conv(f,g) <= max(conv(f,f)),
    then update f.
    """
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_score = _score_norm(f)
    t_start = time.monotonic()
    
    M = float(_full_convolve(f, f).max())
    
    # Build constraint matrix A where A[k,j] = f[j] if 0 <= k-j < n
    # Shape: (2n-1, n)
    # Use sparse representation for efficiency
    # Actually, let's just build it as a dense array for simplicity
    # For n=2048, it's 4095 x 2048 = ~8M floats = ~64MB, which is OK
    # But let's be smarter - we can use the banded structure
    
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n), dtype=np.float64)
    for k in range(n_cons):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = f[j_min:j_max+1]
    
    b = M * np.ones(n_cons)
    c_obj = -np.ones(n)  # minimize negative = maximize
    bounds = [(0, None)] * n
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        if M < 1e-15:
            break
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                # Try fallback
                break
            g = res.x
            g_sum = float(g.sum())
            if g_sum < 1e-15:
                break
            g = g / g_sum
        except Exception:
            break
        
        # Update f
        f_new = (1.0 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        sc = _score_norm(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        
        f = f_new
        M = float(_full_convolve(f, f).max())
        
        # Update constraint matrix (f changed)
        for k in range(n_cons):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            A[k, j_min:j_max+1] = f[j_min:j_max+1]
        b = M * np.ones(n_cons)
        
        # Adaptive step size
        if it % 10 == 9:
            # Check if we're making progress
            pass
    
    return best_f, best_score

def _lbfgs_smooth_max(f_init, time_limit=60.0, beta_init=None):
    """
    L-BFGS optimization using smooth max (log-sum-exp) approximation.
    Optimizes in log-space with softmax parameterization.
    """
    n = len(f_init)
    f = f_init.copy()
    z = np.log(np.maximum(f, 1e-15))
    z = z - z.mean()
    
    best_f = f.copy()
    best_score = _score_norm(f)
    t_start = time.monotonic()
    
    # Determine initial beta
    c0 = _full_convolve(f, f)
    max_c0 = float(c0.max())
    if beta_init is None:
        beta_init = 0.01 * max_c0
    
    # Beta schedule (continuation)
    betas = [beta_init * (0.5 ** i) for i in range(8)]
    # [0.01, 0.005, 0.0025, 0.00125, 0.000625, 0.0003125, 0.00015625, 0.000078]
    betas = [b * max_c0 for b in [0.01, 0.005, 0.0025, 0.00125, 0.000625, 0.0003, 0.00015, 0.00008]]
    
    time_per_beta = time_limit / len(betas)
    
    for beta in betas:
        if time.monotonic() - t_start > time_limit - 2:
            break
        
        def objective(z_vec):
            a = _softmax(z_vec)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            exp_vals = np.exp(c_shifted)
            log_sum = float(np.log(np.sum(exp_vals)))
            return 2.0 * n * (cm + beta * log_sum)
        
        def gradient(z_vec):
            a = _softmax(z_vec)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                idx = int(np.argmax(c))
                w = np.zeros_like(c)
                w[idx] = 1.0
            else:
                w = e / s
            W = float(np.dot(w, c))
            a_rev = a[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                return np.zeros_like(z_vec)
            return grad
        
        deadline_for_min = time.monotonic() + time_per_beta
        try:
            result = minimize(
                objective, z, jac=gradient,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 1000,
                         'iprint': -1}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_f = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_f, best_score

def _adam_polish(f_init, time_limit=60.0, lr=0.01, beta_frac=0.005):
    """Adam optimizer in log-space with smooth max."""
    n = len(f_init)
    z = np.log(np.maximum(f_init, 1e-15))
    z = z - z.mean()
    
    best_f = f_init.copy()
    best_score = _score_norm(f_init)
    t_start = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(f_init, f_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)
    
    while time.monotonic() - t_start < time_limit:
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        grad = 4.0 * n * a * (h - W)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax(z)
        
        if t % 10 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = a.copy()
            beta = max(1e-12, beta * 0.95)
    
    return best_f, best_score

def _interp_to_length(a, n_new):
    """Interpolate a profile to a new length."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a / a.sum()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

Let me also think about one more idea: **coordinate descent**.

For each coordinate `a[i]`, we can optimize it while keeping all other coordinates fixed. The objective as a function of `a[i]` (with `sum(a)/n = 1` constraint) is:
- `c[k]` is quadratic in `a[i]` for each k that involves `a[i]`
- Specifically, `c[k]` involves `a[i]` when `0 <= k-i < n`, i.e., `i <= k <= i+n-1`
- So changing `a[i]` affects `c[i], c[i+1], ..., c[i+n-1]` (the lags that involve position i)

The sub-problem for coordinate `a[i]`:
- Minimize `max_{k in {i, ..., i+n-1}} c[k](a[i])` subject to `a[i] >= 0` and the sum constraint

This is a 1D problem (we're only changing one coordinate), and the function is the max of quadratics, which is convex in `a[i]`. We can solve it with a golden section search or by finding where the quadratics cross.

Actually, the sum constraint couples all coordinates, so we can't do pure coordinate descent without handling the constraint. We could do:
- Fix `sum(a) \ a[i] = S - a[i]`
- Optimize `a[i]` to minimize `max_k c[k]` subject to `a[i] >= 0`
- Then renormalize (or use the constraint directly)

This is similar to the "multiplicative weight" approach but in a more principled way.
