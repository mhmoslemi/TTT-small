Coordinate descent + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses 1D coordinate minimization (scipy bounded) combined with logit-space Adam,
multi-scale interpolation, and diverse restarts.

import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def coordinate_descent_pass(a, conv, n, max_evals=40):
    """One full pass of coordinate descent. Returns updated a and conv."""
    old_score = compute_score(a)  # for tracking
    S = float(a.sum())
    
    for i in range(n):
        old = a[i]
        if old < 1e-15:
            continue
        
        # Compute influence vector: influence[k] = a[i+k] + a[i-k] (0 if out of range)
        # influence has length 2n-1 (indices 0 to 2n-2)
        # influence[k] for k in [0, 2n-2]
        # From j=i: a[i]*a[i+k], valid when 0 <= i+k < n, i.e., k in [0, n-1-i]
        # From j=i-k: a[i-k]*a[i], valid when 0 <= i-k < n, i.e., k in [i-(n-1), i]
     
        # Let's compute influence more carefully
        # conv_new[k] = conv[k] + (x-old)*(a[i+k] if 0<=i+k<n else 0 + a[i-k] if 0<=i-k<n else 0)
        #              + (x^2 - old^2) * [k==0]
        
        # influence[k] = a[i+k] (if valid) + a[i-k] (if valid)
        # For k in [0, 2n-2]:
        #   a[i+k] is valid when 0 <= i+k <= n-1, i.e., -i <= k <= n-1-i
        #   a[i-k] is valid when 0 <= i-k <= n-1, i.e., i-(n-1) <= k <= i
        
        # Let me just compute it directly
        # The range of k where influence is nonzero:
        # k_min = max(0, i - (n-1))  [from a[i-k] constraint: i-k >= 0 → k <= i, and i-k <= n-1 → k >= i-n+1]
        # k_max = min(2n-2, i + (n-1))  [from a[i+k] constraint: i+k <= n-1 → k <= n-1-i, and i+k >= 0 → k >= -i]
        
        # Actually, let me just compute it as:
        # idx_plus = i + np.arange(0, 2*n-1)  → a[idx_plus] where 0 <= idx_plus < n
        # idx_minus = i - np.arange(0, 2*n-1) → a[idx_minus] where 0 <= idx_minus < n
        
        k_range = np.arange(2 * n - 1)
        idx_plus = i + k_range  # for a[i+k]
        idx_minus = i - k_range  # for a[i-k]
        
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        
        influence = np.zeros(2 * n - 1)
        if np.any(mask_plus):
            influence[mask_plus] += a[idx_plus[mask_plus]]
        if np.any(mask_minus):
            influence[mask_minus] += a[idx_minus[mask_minus]]
        
        # Now minimize f(x) = 2n * max_k (conv[k] + (x-old)*influence[k] + (x^2-old^2)*[k==0]) / (S-old+x)^2
        # over x in [0, upper]
        
        upper = 3.0 * max(old, 2.0 / n)
        if upper < 1e-10:
            continue
        
        def obj_1d(x):
            delta = x - old
            conv_new = conv + delta * influence
            if abs(delta) > 1e-15:
                conv_new[0] += delta * (x + old)  # x^2 - old^2 = (x-old)(x+old)
            s_new = S - old + x
            if s_new <= 0:
                return 1e10
            return 2.0 * n * float(conv_new.max()) / (s_new * s_new)
        
        result = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                                  options={'maxiter': max_evals})
        
        if result.fun < old_score:  # Only update if it improves
            x_new = result.x
            delta = x_new - old
            conv += delta * influence
            conv[0] += delta * (x_new + old)  # x^2 - old^2
            a[i] = x_new
            S = S - old + x_new
            old_score = result.fun
    
    return a, conv

"""
Coordinate descent + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses 1D bounded minimization per coordinate (Gauss-Seidel), logit-space Adam gradient,
multi-scale interpolation, and periodic noise injection to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize_scalar


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length, preserving normalization."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_linear_profile(n, eta=1.5):
    """Create a linear ramp profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_profile(n, rng):
    """Create a smooth random profile using low-frequency Fourier terms."""
    coeffs = rng.uniform(-0.3, 0.3, 6)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, c in enumerate(coeffs, 1):
        a = a + c * np.cos(2.0 * np.pi * i * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def soft_max_weights(c, beta):
    """Compute softmax weights for approximating the max."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h_fft(w, a):
    """h[i] = sum_m a[m]*w[i+m] via FFT (cross-correlation style)."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def gradient_step_adam(a, c, z, m, v, t_adam, beta, lr, rng):
    """One Adam gradient step in logit space. Returns updated z, a, m, v, new beta."""
    n = len(a)
    _, w = soft_max_weights(c, beta)
    h = compute_h_fft(w, a)
    grad_a = 2.0 * h
    dot = float(np.dot(a, grad_a))
    dz = a * (grad_a - dot)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * dz
    v = b2 * v + (1.0 - b2) * (dz * dz)
    mh = m / (1.0 - b1 ** t_adam)
    vh = v / (1.0 - b2 ** t_adam)
    z = z - lr * mh / (np.sqrt(vh) + eps)

    if not np.all(np.isfinite(z)):
        return None, a, m, v, beta  # Signal failure

    z = np.clip(z, -50.0, 50.0)
    z = z - z.mean()
    a_new = np.exp(z - z.max())
    s = a_new.sum()
    if s < 1e-15:
        return None, a, m, v, beta
    a_new = a_new / s
    return z, a_new, m, v, beta


def gradient_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.04,
                      beta_decay=0.95, beta_min=1e-12, restart_period=800):

    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-12:

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        t_adam += 1
        result = gradient_step_adam(a0, c, z, m, v, t_adam, beta, current_lr, rng)
        if result[0] is None:
            a0 = best_a.copy()
            z = np.log(a0)
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

def cd_objective_eval(x, conv, influence, conv0, S, old, n):
    """Evaluate the 1D objective for coordinate descent at value x."""
    s_new = S - old + x
    if s_new <= 1e-15:
        return 1e10
    val0 = conv0 + x * x - old * old
    delta = x - old
    vals = conv + delta * influence
    vals[0] = val0
    m = float(vals.max())
    return 2.0 * n * m / (s_new * s_new)


def coordinate_descent_pass(a, conv, n, max_evals=25, order=None):
    """
    One full pass of coordinate descent (Gauss-Seidel).
    Modifies a and conv in-place. Returns final score.
    """
    S = float(a.sum())
    n_conv = 2 * n - 1
    k_range = np.arange(1, n_conv)

    if order is None:
        order = np.arange(n)

    for idx in order:
        i = int(idx)
        old = a[i]
        if old < 1e-15:
            old = 1e-15

        # Compute influence for k = 1, ..., 2n-2
        idx_plus = i + k_range
        idx_minus = i - k_range

        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        mask_minus = (idx_minus >= 0) & (idx_minus < n)

        influence = np.zeros(n_conv)
        if np.any(mask_plus):
            influence[1:][mask_plus] += a[idx_plus[mask_plus]]
        if np.any(mask_minus):
            influence[1:][mask_minus] += a[idx_minus[mask_minus]]

        conv0 = conv[0]
        upper = max(4.0 * old, 6.0 / n)
        upper = min(upper, 1.0 / n * 20.0)
        if upper < 1e-12:
            upper = 1e-6

        result = minimize_scalar(
            cd_objective_eval,
            args=(conv, influence, conv0, S, old, n),
            bounds=(0.0, upper),
            method='bounded',
            options={'maxiter': max_evals, 'xatol': 1e-10}
        )

        if result.success and result.fun < 1e9:
            x_new = float(result.x)
            if x_new >= 0:
                delta = x_new - old
                if abs(delta) > 1e-15:
                    conv[1:] += delta * influence[1:]
                    conv[0] = conv0 + x_new * x_new - old * old
                    a[i] = x_new
                    S += delta

    return a, conv

            noise_scale = 0.002 * max(1.0, best_cd_score - 1.4)
            a = best_cd_a + noise_scale * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            conv = np.convolve(a, a)
            best_cd_conv = conv.copy()
            no_improve_cd = 0
            pass_count += 1  # Count as an extra pass

    final_sc = compute_score(a)
    try_update(a, final_sc)
    print(f"  Phase 2 done: {pass_count} passes, score={final_sc:.10f}")

    # === Phase 3: Gradient polish (150s) ===
    if time.monotonic() < deadline - 30:
        print(f"\n--- Phase 3: Gradient polish ---")
        phase3_dl = min(deadline, time.monotonic() + 150.0)
        a2, _ = gradient_optimize(a, phase3_dl, rng, lr=0.012, beta_factor=0.015,
                                  beta_decay=0.97, beta_min=1e-14, restart_period=500)
        sc3 = compute_score(a2)
        try_update(a2, sc3)
        print(f"  After polish: {sc3:.10f}")
        a = a2

    # === Phase 4: Second CD round (time remaining - 100s) ===
    phase4_dl = min(deadline - 80.0, time.monotonic() + 300.0)
    if time.monotonic() < phase4_dl:
        print(f"\n--- Phase 4: Second CD round ---")
        a = a / a.sum()
        conv = np.convolve(a, a)
        pass_count_4 = 0
        no_improve_4 = 0
        best_4 = 2.0 * n * float(conv.max()) / (float(a.sum()) ** 2)
        best_4_a = a.copy()

        while time.monotonic() < phase4_dl:
            order = rng.permutation(n)
            a, conv = coordinate_descent_pass(a, conv, n, max_evals=20, order=order)
            pass_count_4 += 1
            S = float(a.sum())
            cur = 2.0 * n * float(conv.max()) / (S * S) if S > 1e-10 else float('inf')
            if cur < best_4 - 1e-13:
                best_4 = cur
                best_4_a = a.copy()
                no_improve_4 = 0
            else:
                no_improve_4 += 1
            if no_improve_4 > 60:
                a = best_4_a + 0.001 * rng.normal(size=n)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                conv = np.convolve(a, a)
                no_improve_4 = 0

        sc4 = compute_score(a)
        try_update(a, sc4)
        print(f"  Phase 4 done: {pass_count_4} passes, score={sc4:.10f}")

    # === Phase 5: Try larger n (last 80s) ===
    if time.monotonic() < deadline - 20 and best_score > 1.503:
        n_large = min(4096, 2 * n)
        print(f"\n--- Phase 5: Trying n={n_large} ---")
        a_large = interpolate_profile(best_seq, n_large)
        sc_large = compute_score(a_large)
        try_update(a_large, sc_large)
        print(f"  Interpolated: {sc_large:.10f}")

        dl_large = min(deadline, time.monotonic() + 50.0)
        a_large, _ = gradient_optimize(a_large, dl_large, rng, lr=0.02,
                                       beta_factor=0.03, beta_decay=0.95)
        sc_large2 = compute_score(a_large)
        try_update(a_large, sc_large2)
        print(f"  After gradient: {sc_large2:.10f}")

        # Quick CD on large
        if time.monotonic() < deadline - 15:
            a_large = a_large / a_large.sum()
            conv_large = np.convolve(a_large, a_large)
            cd_dl = min(deadline, time.monotonic() + 30.0)
            n_large_cd = a_large.size
            while time.monotonic() < cd_dl:
                order = rng.permutation(n_large_cd)
                a_large, conv_large = coordinate_descent_pass(
                    a_large, conv_large, n_large_cd, max_evals=15, order=order)
                S_l = float(a_large.sum())
                cur_l = 2.0 * n_large_cd * float(conv_large.max()) / (S_l * S_l)
                try_update(a_large, cur_l)
            print(f"  After large CD: {compute_score(a_large):.10f}")

    # === Final ===
    print(f"\n  FINAL best score: {best_score:.10f}")

    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = make_linear_profile(256, 1.5)
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
