def _fft_convolve(a):

"""
Optimizes 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Adaptive smooth-max (log-sum-exp) gradient descent with aggressive beta scheduling
2. Exact peak subgradient for hard-peaked regimes

5. Occasional LP-based spreading for local minima escape
6. Multi-scale progressive length increase
All convolution via FFT for O(n log n) per iteration.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-10:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_convolve_w(a, w):
    """Gradient of sum_k w[k]*c[k] where c = convolve(a,a). Result has length n."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    return 2.0 * h_full[n - 1:2 * n - 1]


def _peak_subgradient(a, c, top_k=3, epsilon=0.05):
    """Compute weighted subgradient of top-k peaks of c."""
    n = len(a)
    cmax = float(c.max())
    if cmax < 1e-30:
        return np.zeros(n)
    threshold = cmax * (1.0 - epsilon)
    peak_idxs = np.where(c >= threshold)[0]
    if len(peak_idxs) == 0:
        peak_idxs = np.array([int(np.argmax(c))])
    if len(peak_idxs) > top_k:
        peak_idxs = peak_idxs[np.argsort(c[peak_idxs])[::-1][:top_k]]
    weights = c[peak_idxs]
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for k, w in zip(peak_idxs, weights):
        k = int(k)
        # subgrad[j] = 2*a[k-j] if 0 <= k-j < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * w * a[k - j]
    return grad


def _gradient_smooth(a, beta):
    c = _fft_convolve(a)
    w = _smooth_max_weights(c, beta)
    grad = _grad_convolve_w(a, w)
    return grad, float(c.max()), c


def _make_ramp(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_symmetric_ramp(n, eta=1.0):

    d = np.abs(t - 0.5) * 2.0
    a = 1.0 + eta * d
    a = np.maximum(a, 1e-12)

def _make_gaussian_mixture(n, centers, sigmas, weights):

    a = np.zeros(n)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _grad_descent_main(a, deadline, max_iter=50000):
    """Main gradient descent with adaptive beta and momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.08 * float(c.max()))

    lr = 0.04
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    peak_mode = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Switch to peak subgradient mode when beta becomes very small
        use_peak = beta < 0.002 * float(c.max()) and it % 3 == 0

        if use_peak:
            c_cur = _fft_convolve(a)
            grad = _peak_subgradient(a, c_cur, top_k=5, epsilon=0.03)
        else:
            grad, maxc, c = _gradient_smooth(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue

        # Normalize gradient for stability
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            grad = grad / gnorm

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-11:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.97)
            if no_imp > max(800, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.08 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        # Adaptive beta: decrease when making progress, increase when stuck
        if no_imp == 0:
            beta = max(1e-10, beta * 0.92)
        elif no_imp % 200 == 0:
            beta = min(0.5 * float(c.max()), beta * 1.3)
            momentum = min(0.95, momentum * 1.02)

    return best_a, best_score


def _logit_adam(a, deadline, max_iter=30000, lr=0.05):
    """Adam optimizer in logit space for better handling of non-negativity."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.06 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weights(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m, v_arr, t_adam = np.zeros(n), np.zeros(n), 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m, v_arr, t_adam = np.zeros(n), np.zeros(n), 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-11:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.85)

        if no_imp > max(1500, 3 * n):
            rng_r = np.random.default_rng(hash((it, n, 7)) % (2**32))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n) * 0.5
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.06 * float(_fft_convolve(a).max()))
            m, v_arr, t_adam = np.zeros(n), np.zeros(n), 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _freq_domain_polish(a, deadline, max_iter=5000):
    """Gerchberg-Saxton style frequency domain iteration."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        c = _fft_convolve(a)
        cmax = float(c.max())

        # Find which frequency bins contribute most to the peak
        # c = IFFT(|A|^2), so c[m] = (1/N) sum_k |A[k]|^2 * exp(2*pi*i*k*m/N)
        # To reduce c[m_peak], reduce |A[k]| where Re(exp(2*pi*i*k*m_peak/N)) > 0
        m_peak = int(np.argmax(c))

        # Compute contribution of each bin to the peak
        # Use a simpler heuristic: reduce |A| where it's large (energy concentration)
        mag = np.abs(A)
        # Soft threshold: reduce large magnitudes
        threshold = cmax * 0.5 + np.median(mag)
        reduction = np.where(mag > threshold, (mag - threshold) / mag * 0.1, 0.0)
        mag_new = mag * (1.0 - reduction)

        # Reconstruct
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        # Keep DC component fixed (to preserve sum)
        A_new[0] = A[0]

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-11:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break

    return best_a, best_score

            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-11:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
            inflate = max(1.0005, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.05, blend * 0.9)
                inflate = min(1.015, inflate * 1.03)
            if no_imp > 200:
                break

    return best_a, best_score


def _coordinate_search(a, deadline, max_iter=10000):
    """Random coordinate moves to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(999)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Random perturbation: move mass from one position to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.uniform(0.05, 0.3)
        if delta > a[i] * 0.5:
            delta = a[i] * 0.5

        a_try = a.copy()
        a_try[i] -= delta
        a_try[j] += delta
        sc = _score(a_try)
        if sc < best_score - 1e-11:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a, best_score


def _multi_peak_gradient(a, deadline, max_iter=30000):
    """Focus on reducing both the peak and the second-highest peak simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.03
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        # Weight top 5 peaks by their values
        sorted_idx = np.argsort(c)[::-1]
        top5 = sorted_idx[:5]
        w = np.zeros(len(c))
        w[top5] = c[top5]
        w = w / w.sum()

        grad = _grad_convolve_w(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            grad = grad / gnorm

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-11:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            momentum = min(0.95, momentum * 1.01)
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.002, lr * 0.96)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                momentum = 0.8
                no_imp = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses gradient descent with adaptive smoothing, frequency-domain polishing,
    LP spreading, and multi-start diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine which lengths to try
    target_lengths = []
    if parent is not None:
        target_lengths.append(parent.size)
    for ln in [512, 640, 768, 1024, 1280]:
        if ln not in target_lengths:
            target_lengths.append(ln)
    target_lengths = list(dict.fromkeys(target_lengths))

    print(f"Target lengths: {target_lengths}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Optimize each length with multiple initializations
    n_lengths = len(target_lengths)
    per_length = (deadline - t_start - 300.0) / n_lengths
    per_length = max(60.0, per_length)

    for n_t in target_lengths:
        if time.monotonic() > deadline - 200:
            break

        elapsed_frac = (time.monotonic() - t_start) / budget_s
        remaining_frac = max(0.15, 1.0 - elapsed_frac)
        phase_budget = per_length * remaining_frac
        phase_budget = max(40.0, min(phase_budget, deadline - time.monotonic() - 100.0))

        if phase_budget < 30:
            continue

        print(f"\n=== Length n={n_t}, budget={phase_budget:.0f}s ===")

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_ramp(n_t, 1.0))
        inits.append(_make_ramp(n_t, 2.0))
        inits.append(_make_ramp(n_t, 0.5))
        inits.append(_make_symmetric_ramp(n_t, 1.5))
        inits.append(_make_symmetric_ramp(n_t, 3.0))
        inits.append(_make_u_profile(n_t, 1.5))
        inits.append(_make_u_profile(n_t, 3.0))
        inits.append(_make_gaussian_mixture(n_t, [0.15, 0.85], [0.2, 0.2], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_t, [0.05, 0.95], [0.15, 0.15], [1.0, 1.0]))

        # Filter to those that haven't been tried and have good initial score
        for a0 in inits:
            if time.monotonic() > deadline - 100:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

        # Sort inits by initial score and take top ones
        init_scores = [(_score(ai), i, ai) for i, ai in enumerate(inits)]
        init_scores.sort(key=lambda x: x[0])
        top_inits = [ai for _, _, ai in init_scores[:5]]

        # Run optimization on top initializations
        sub_budget = phase_budget / max(1, len(top_inits))
        sub_budget = max(15.0, sub_budget)

        for idx, a0 in enumerate(top_inits):
            if time.monotonic() > deadline - 80:
                break
            rem = deadline - time.monotonic() - 60.0
            if rem < 10:
                break
            alloc = min(sub_budget, rem)

            # Main gradient descent
            d1 = time.monotonic() + alloc * 0.45
            a1, s1 = _grad_descent_main(a0, d1)
            if s1 < best_score:
                best_score = s1
                best_seq = a1.copy()
            print(f"  Init {idx} grad: {s1:.10f}")

            # Adam polish
            if time.monotonic() < deadline - 60:
                d2 = time.monotonic() + min(alloc * 0.25, deadline - time.monotonic() - 50)
                a2, s2 = _logit_adam(a1, d2)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2.copy()
                print(f"  Init {idx} adam: {s2:.10f}")

            # Multi-peak gradient
            if time.monotonic() < deadline - 60:
                d3 = time.monotonic() + min(alloc * 0.2, deadline - time.monotonic() - 40)
                a3, s3 = _multi_peak_gradient(a2, d3)
                if s3 < best_score:
                    best_score = s3
                    best_seq = a3.copy()
                print(f"  Init {idx} multi-peak: {s3:.10f}")

            # Occasional LP
            if time.monotonic() < deadline - 50 and n_t <= 768:
                d4 = time.monotonic() + min(alloc * 0.1, deadline - time.monotonic() - 30)
                a4, s4 = _lp_refine_quick(a3, d4)
                if s4 < best_score:
                    best_score = s4
                    best_seq = a4.copy()
                print(f"  Init {idx} LP: {s4:.10f}")

    # Phase 2: Intensive polish on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 15.0
        print(f"\n=== Final polish, n={n_best}, budget={final_budget:.0f}s ===")

        # 2a: Strong gradient
        d_a = time.monotonic() + final_budget * 0.3
        a_fa, s_fa = _grad_descent_main(best_seq, d_a, max_iter=80000)
        if s_fa < best_score:
            best_score = s_fa
            best_seq = a_fa.copy()
        print(f"  Final grad: {s_fa:.10f}")

        # 2b: Adam
        d_b = time.monotonic() + final_budget * 0.2
        a_fb, s_fb = _logit_adam(best_seq, d_b, max_iter=30000, lr=0.03)
        if s_fb < best_score:
            best_score = s_fb
            best_seq = a_fb.copy()
        print(f"  Final adam: {s_fb:.10f}")

        # 2c: Multi-peak
        d_c = time.monotonic() + final_budget * 0.15
        a_fc, s_fc = _multi_peak_gradient(best_seq, d_c, max_iter=20000)
        if s_fc < best_score:
            best_score = s_fc
            best_seq = a_fc.copy()
        print(f"  Final multi-peak: {s_fc:.10f}")

        # 2d: Frequency domain
        d_d = time.monotonic() + final_budget * 0.1
        a_fd, s_fd = _freq_domain_polish(best_seq, d_d)
        if s_fd < best_score:
            best_score = s_fd
            best_seq = a_fd.copy()
        print(f"  Final freq: {s_fd:.10f}")

        # 2e: Coordinate search
        d_e = time.monotonic() + final_budget * 0.1
        a_fe, s_fe = _coordinate_search(best_seq, d_e)
        if s_fe < best_score:
            best_score = s_fe
            best_seq = a_fe.copy()
        print(f"  Final coord: {s_fe:.10f}")

        # 2f: One more gradient round
        if time.monotonic() < deadline - 15:
            d_f = deadline - 10
            a_ff, s_ff = _grad_descent_main(best_seq, d_f, max_iter=50000)
            if s_ff < best_score:
                best_score = s_ff
                best_seq = a_ff.copy()
            print(f"  Final grad2: {s_ff:.10f}")

        # 2g: LP if time and size allow
        if time.monotonic() < deadline - 15 and n_best <= 768:
            d_g = deadline - 10
            a_fg, s_fg = _lp_refine_quick(best_seq, d_g)
            if s_fg < best_score:
                best_score = s_fg
                best_seq = a_fg.copy()
            print(f"  Final LP: {s_fg:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.5)
        best_score = _score(best_seq)

    # Final normalization and validation
    final_seq = np.asarray(best_seq, dtype=np.float64)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.5)

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
        else:
            print(f"\nevaluate_sequence: invalid!")
    except Exception as e:
        print(f"\nevaluate_sequence error: {e}")

    print(f"FINAL: local_score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
