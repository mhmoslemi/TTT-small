def _spectral_flatten(a, n_iter=200, alpha=0.5):
    """
    Iterative spectral flattening: modify FFT magnitude to be flatter,
    then project back to non-negative. This gives a good initialization
    with approximately flat autocorrelation.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    N = _next_pow2(2 * n)
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: constant magnitude
        target_mag = np.mean(mag[1:-1] if N > 2 else mag)  # exclude DC and Nyquist
        # Blend: keep some of original, push toward flat
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, N)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

    return a

    b2 = 0.999

    lr_cur = lr

            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())

def _lp_spread(a, M):
    """LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: (a*b)[k] = sum_j a[k-j]*b[j] <= M
    # For efficiency with large n, only include "active" constraints
    # Find top constraints
    # For now, include all (works for n up to ~2048 with HiGHS)
    if n > 2048:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _subgradient_refine(a, deadline, max_iter=10000, top_k=3):

        c_shifted = c_curr - c_curr[0] + 1e-10
        w = np.zeros_like(c_curr)
        for i, idx in enumerate(top_indices):
            w[idx] = weights[i] / max(weights.sum(), 1e-10)

        grad_a = _grad_smooth_max_obj(a, w)

        # Projected gradient step onto simplex
        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * grad_a
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 5 * n):
                break

        a = a_new

    return best_a, best_score


def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_random_profile(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 2.0, size=n)
    return a / a.sum()


def _make_blue_noise_profile(n, seed=0):
    """Create a profile with blue-noise-like characteristics (low autocorrelation)."""
    rng = np.random.default_rng(seed)
    # Start with random, apply spectral whitening
    a = rng.standard_normal(n)
    A = np.fft.rfft(a, _next_pow2(2 * n))
    mag = np.abs(A)
    target = np.mean(mag[1:-1]) if len(mag) > 2 else 1.0
    new_mag = 0.5 * mag + 0.5 * target
    new_A = new_mag * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(new_A, _next_pow2(2 * n))[:n]
    a_new = np.maximum(a_new, 0.1)
    return a_new / a_new.sum()


def _make_polynomial_profile(n, deg=4):
    """Create a polynomial profile that has low peak autocorrelation."""
    t = np.linspace(0.0, 1.0, n)
    # Use a Legendre polynomial approach - alternate signs reduce peaks
    a = np.ones(n)
    for d in range(1, deg + 1):
        a += 0.3 * np.cos(np.pi * d * t) * np.cos(np.pi * d * t * 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _grow_candidate(base_a, target_n, deadline, label=""):
    """
    Grow a small candidate to target_n via iterative upscaling and optimization.
    """
    a = base_a.copy()
    current_n = len(a)
    print(f"  {label} Starting: n={current_n}, score={_score(a):.10f}")

    # Upscale in stages
    stages = []
    cn = current_n
    while cn < target_n:
        cn = min(cn * 2, target_n)
        stages.append(cn)

    for stage_n in stages:
        if time.monotonic() > deadline:
            break

        budget_remaining = deadline - time.monotonic()
        if budget_remaining < 30:
            break

        a = _interpolate(a, stage_n)
        print(f"  {label} Upscaled to n={stage_n}, score={_score(a):.10f}")

        # Optimize at this scale
        stage_budget = min(budget_remaining * 0.6, 300.0)
        d_stage = time.monotonic() + stage_budget

        a, s1 = _adam_optimize(a, d_stage, max_iter=20000, lr=0.02, beta_factor=0.04)
        print(f"  {label} n={stage_n} after Adam: score={s1:.10f}")

        if time.monotonic() < deadline - 20:
            a2, s2 = _subgradient_refine(a, time.monotonic() + 60, max_iter=5000, top_k=3)
            if s2 < s1:
                a, s1 = a2, s2
                print(f"  {label} n={stage_n} after Subgrad: score={s1:.10f}")

    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Spectral flattening initialization at multiple scales
    2. Adam optimization with hybrid smooth-max + top-K
    3. LP polishing
    4. Subgradient refinement
    5. Final fine-tuning
    Uses multi-scale warm starts and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # === Phase 1: Generate diverse initializations via spectral flattening ===
    print("=== Phase 1: Spectral initialization ===")
    n_base = 256
    inits = []

    # From parent (interpolated to n_base)
    if parent is not None:
        p_base = _interpolate(parent, n_base)
        p_flat = _spectral_flatten(p_base, n_iter=300, alpha=0.4)
        inits.append(("parent_spectral", p_flat))
        inits.append(("parent_raw", p_base))

    # Random + spectral
    for i in range(3):
        a_rand = _make_random_profile(n_base, seed=100 + i)
        a_flat = _spectral_flatten(a_rand, n_iter=400, alpha=0.35)
        inits.append((f"rand_{i}_spectral", a_flat))

    # Blue noise + spectral
    a_blue = _make_blue_noise_profile(n_base, seed=42)
    a_blue_flat = _spectral_flatten(a_blue, n_iter=300, alpha=0.4)
    inits.append(("blue_noise_spectral", a_blue_flat))

    # Polynomial profile + spectral
    a_poly = _make_polynomial_profile(n_base, deg=6)
    a_poly_flat = _spectral_flatten(a_poly, n_iter=300, alpha=0.3)
    inits.append(("poly_spectral", a_poly_flat))

    # Perturbed parent
    if parent is not None:
        p_pert = _interpolate(parent, n_base) * (1.0 + 0.2 * rng.normal(size=n_base))
        p_pert = np.maximum(p_pert, 0.01)
        p_pert_flat = _spectral_flatten(p_pert, n_iter=200, alpha=0.3)
        inits.append(("perturbed_spectral", p_pert_flat))

    # Evaluate all initializations
    for name, a_init in inits:
        sc = _score(a_init)
        print(f"  Init '{name}': score={sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_init.copy()

    # === Phase 2: Main optimization on best initializations ===
    print("=== Phase 2: Main optimization ===")

    # Sort initializations by score
    scored_inits = []
    for name, a_init in inits:
        sc = _score(a_init)
        scored_inits.append((sc, name, a_init))
    scored_inits.sort(key=lambda x: x[0])

    # Also include parent at 512 and 1024
    if parent is not None:
        p512 = _interpolate(parent, 512)
        p1024 = _interpolate(parent, 1024)
        scored_inits.append((_score(p512), "parent_512", p512))
        scored_inits.append((_score(p1024), "parent_1024", p1024))
        # Spectral flatten parent at 512
        p512_flat = _spectral_flatten(p512, n_iter=200, alpha=0.3)
        scored_inits.append((_score(p512_flat), "parent_512_flat", p512_flat))

    scored_inits.sort(key=lambda x: x[0])

    # Optimize top 3 initializations (by current score)
    n_to_optimize = min(3, len(scored_inits))
    time_per_opt = (deadline - time.monotonic() - 200) / n_to_optimize

    for idx in range(n_to_optimize):
        if time.monotonic() > deadline - 200:
            break

        sc_init, name, a_init = scored_inits[idx]
        n_cur = len(a_init)
        print(f"  Optimizing '{name}' (n={n_cur}, init_score={sc_init:.10f})")

        # Determine target length
        if n_cur <= 512:
            target_scaleup = 512
        else:
            target_scaleup = n_cur

        # Scale up if needed
        if n_cur < target_scaleup:
            a_opt = _interpolate(a_init, target_scaleup)
        else:
            a_opt = a_init.copy()

        # Adam optimization
        d_adam = time.monotonic() + min(time_per_opt * 0.5, 400)
        a_opt, s_adam = _adam_optimize(a_opt, d_adam, max_iter=30000,
                                        lr=0.02, beta_factor=0.04,
                                        topk_blend=0.3, topk_K=5)
        print(f"    After Adam: score={s_adam:.10f}")

        if s_adam < best_score:
            best_score = s_adam
            best_seq = a_opt.copy()

        # Subgradient refinement
        if time.monotonic() < deadline - 150:
            d_sg = time.monotonic() + min(time_per_opt * 0.2, 100)
            a_sg, s_sg = _subgradient_refine(a_opt, d_sg, max_iter=8000, top_k=3)
            if s_sg < best_score:
                best_score = s_sg
                best_seq = a_sg.copy()
                print(f"    After Subgrad: score={best_score:.10f}")
                a_opt = a_sg

    # === Phase 3: Try growing the best to larger n ===
    print(f"=== Phase 3: Growth (current best={best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        best_n = best_seq.size
        if best_n < 1024:
            # Grow to 1024
            d_grow = deadline - 150
            a_grown = _grow_candidate(best_seq, 1024, d_grow, "grow1024")
            s_grown = _score(a_grown)
            print(f"  Grown to 1024: score={s_grown:.10f}")
            if s_grown < best_score:
                best_score = s_grown
                best_seq = a_grown.copy()
            else:
                # Try Adam on the grown version
                d_adam_g = time.monotonic() + min(200, deadline - time.monotonic() - 100)
                if time.monotonic() < deadline - 100:
                    a_grown2, s_grown2 = _adam_optimize(a_grown, d_adam_g,
                                                         max_iter=15000, lr=0.01,
                                                         beta_factor=0.02,
                                                         topk_blend=0.4, topk_K=5)
                    if s_grown2 < best_score:
                        best_score = s_grown2
                        best_seq = a_grown2.copy()
                        print(f"  Grown+Adam: score={best_score:.10f}")

    # === Phase 4: LP polishing ===
    print(f"=== Phase 4: LP polishing (current best={best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        best_n = best_seq.size
        if best_n <= 2048:
            d_lp = time.monotonic() + min(200, deadline - time.monotonic() - 30)
            a_lp = _lp_refine(best_seq, d_lp, max_iter=3000, inflate=1.005, blend=0.4)
            s_lp = _score(a_lp)
            print(f"  After LP: score={s_lp:.10f}")
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp.copy()

    # === Phase 5: Final fine-tuning ===
    print(f"=== Phase 5: Final tuning (current best={best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very fine Adam
        d_final = deadline - 15
        a_fin, s_fin = _adam_optimize(best_seq, d_final, max_iter=20000,
                                       lr=0.005, beta_factor=0.01,
                                       topk_blend=0.5, topk_K=10)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
        print(f"  After final Adam: score={best_score:.10f}")

        # One more subgradient pass
        if time.monotonic() < deadline - 8:
            d_sg_final = deadline - 8
            a_sg_f, s_sg_f = _subgradient_refine(best_seq, d_sg_final,
                                                  max_iter=5000, top_k=5)
            if s_sg_f < best_score:
                best_score = s_sg_f
                best_seq = a_sg_f.copy()
            print(f"  After final Subgrad: score={best_score:.10f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = _make_random_profile(256, seed=0)
        best_score = _score(best_seq)

    # Ensure valid output
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * np.sqrt(2 * len(final_seq))
    else:
        final_seq = np.ones(256) / 256

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval > best_score:
                    # The evaluate function clips to [0,1000], so rescale
                    print("  Warning: evaluate score differs, using it")
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
