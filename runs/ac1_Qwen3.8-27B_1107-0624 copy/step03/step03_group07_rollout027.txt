```python
            val += a_other[idx2]
        b[k] = val
    
    # sum_new = sum_other + x where sum_other = sum(a_other)
    sum_other = a_other.sum()
    
    # Objective: f(x) = 2n * max_k (c_base[k] + x * b[k]) / (sum_other + x)^2
    # We want to minimize this over x >= 0
    
    # The max is the upper envelope of lines c_base[k] + x*b[k]
    # For the top K values of c_base, find the x that minimizes the ratio
    
    # Simpler approach: binary search or golden section on x
    # f(x) is the max of piecewise-rational functions, each of which is
    # (c_base[k] + x*b[k]) / (sum_other + x)^2
    # Each such function has a maximum at x = -c_base[k]/b[k] (if b[k] > 0 and c_base[k] < 0)
    # or is monotonically decreasing (if b[k] <= 0)
    # or is monotonically increasing (if b[k] > 0 and c_base[k] >= 0)
    
    # Since c_base[k] >= 0 (it's a convolution of non-negative values) and b[k] >= 0,
    # each (c_base[k] + x*b[k]) / (sum_other + x)^2 is either:
    # - increasing then decreasing (peak at x = c_base[k]/b[k] if b[k] > 0)
    # - monotonically decreasing (if b[k] = 0, value is c_base[k]/(sum_other+x)^2)
    
    # So f(x) is the max of such functions, which is unimodal (increases then decreases)
    # We can use ternary search!
    
    # Upper bound for x: when x is very large, f(x) ~ 2n * x * max(b) / x^2 = 2n * max(b) / x -> 0
    # So there's an optimal x. Let's find it via ternary search.
    
    # Range for x: [0, x_max] where x_max is where f starts decreasing
    # A rough upper bound: x_max = n * sum_other (i.e., a[i] can be at most as large as the rest)
    # Actually, let's use a wider range.
    
    x_lo = 0.0
    x_hi = 2.0 * sum_other  # generous upper bound
    
    # Ternary search
    for _ in range(60):
        x1 = x_lo + (x_hi - x_lo) / 3
        x2 = x_hi - (x_hi - x_lo) / 3
        
        # Compute f(x1) and f(x2)
        c1 = c_base + x1 * b
        max1 = c1.max()
        f1 = 2.0 * n * max1 / (sum_other + x1)**2 if (sum_other + x1) > 0 else np.inf
        
        c2 = c_base + x2 * b
        max2 = c2.max()
        f2 = 2.0 * n * max2 / (sum_other + x2)**2 if (sum_other + x2) > 0 else np.inf
        
        if f1 < f2:
            x_hi = x2
        else:
            x_lo = x1
    
    x_opt = (x_lo + x_hi) / 2
    a[i] = x_opt
    return a
```

Wait, this is O(n) per evaluation of f(x) due to the `c_base + x * b` computation, and we do 60 evaluations per coordinate, and n coordinates. So one full pass is O(n^2 * 60). For n=1024, that's about 62 million operations per pass. With vectorized numpy, this should be fast (maybe 1-2 seconds per pass). We could do hundreds of passes.

But there's a problem: the upper bound `x_hi = 2 * sum_other` might not be large enough or might be too large. Also, the ternary search assumes unimodality, which I argued above but need to verify.

Let me verify: `f(x) = 2n * max_k (c_k + x*b_k) / (S + x)^2` where `c_k >= 0`, `b_k >= 0`, `S > 0`.

Each individual term `g_k(x) = (c_k + x*b_k) / (S+x)^2` has derivative:
`g_k'(x) = [b_k(S+x) - 2(c_k + x*b_k)] / (S+x)^3 = [b_k*S - 2*c_k - x*b_k] / (S+x)^3`

Setting to zero: `b_k*S - 2*c_k - x*b_k = 0` => `x = S - 2*c_k/b_k` (if `b_k > 0`)

So each `g_k` is unimodal (increases then decreases) with peak at `x = S - 2*c_k/b_k` (which could be negative, in which case it's monotonically decreasing on `[0, inf)`).

The max of unimodal functions is NOT necessarily unimodal! This is a problem.

Example: `g_1` peaks at x=1, `g_2` peaks at x=5. Their max could have a "valley" between x=1 and x=5.

So ternary search is not guaranteed to work. We need a different approach.

Alternative: **Binary search on the objective value.**

We want to find the minimum of `f(x)`. We know `f(0) = 2n * max(c_base) / S^2` and `f(x) -> 0` as `x -> inf`.

Binary search idea: for a target value `T`, check if `f(x) <= T` for some `x >= 0`.
`f(x) <= T` iff for all k: `(c_k + x*b_k) / (S+x)^2 <= T / (2n)`
iff for all k: `c_k + x*b_k <= (T/(2n)) * (S+x)^2`
iff for all k: `(T/(2n))*x^2 + (2S*T/(2n) - b_k)*x + (S^2*T/(2n) - c_k) >= 0`

This is a system of quadratic inequalities in x. We need to check if there exists x >= 0 satisfying all of them. Each quadratic is either always positive (discriminant < 0 or opens upward with negative roots) or positive outside an interval. The intersection of all "positive regions" gives the feasible set.

This is feasible but complex. Let me think of a simpler approach.

**Simpler: Grid search + refinement.**

1. Evaluate f(x) at several points on a log-spaced grid: x = [0, 0.01*S, 0.05*S, 0.1*S, 0.5*S, 1*S, 2*S, 5*S, 10*S]
2. Find the best point
3. Refine with ternary search in a neighborhood (even if not strictly unimodal, it often works in practice)

Or even simpler: since the number of "active" constraints (k values that achieve the max) is small, we can:
1. Identify the top K values of c_base
2. For each pair of active constraints, find the x where they're equal (kink point)
3. Also check x=0
4. Evaluate f(x) at all these candidate points
5. Pick the best

This is O(K^2 * n) per coordinate which is very fast for small K.

Let me implement this approach:

```python
def optimal_coordinate_v2(a, i, topK=15):
    """Find optimal value of a[i] using active set analysis."""
    n = len(a)
    old = a[i]
    a[i] = 0.0
    c_base = np.convolve(a, a)
    a[i] = old
    
    a_other = a.copy()
    a_other[i] = 0.0
    sum_other = float(a_other.sum())
    
    # b[k] = a_other[k-i] + a_other[i-k] (with bounds, excluding i)
    b = np.zeros(2*n-1)
    # Vectorized computation
    for k in range(2*n-1):
        idx1 = k - i
        idx2 = i - k
        val = 0.0
        if 0 <= idx1 < n and idx1 != i:
            val += a_other[idx1]
        if 0 <= idx2 < n and idx2 != i:
            val += a_other[idx2]
        b[k] = val
    
    # Find top K indices of c_base
    top_idx = np.argsort(c_base)[-topK:]
    
    # Candidate x values:
    # 1. x = 0
    # 2. For each pair (k1, k2) in top_idx: x where c_base[k1] + x*b[k1] = c_base[k2] + x*b[k2]
    #    => x = (c_base[k2] - c_base[k1]) / (b[k1] - b[k2]) if b[k1] != b[k2]
    # 3. x where each individual g_k(x) peaks: x = sum_other - 2*c_base[k]/b[k] (if b[k] > 0 and result > 0)
    
    candidates = [0.0]
    
    for idx_a in range(len(top_idx)):
        for idx_b in range(idx_a + 1, len(top_idx)):
            k1, k2 = top_idx[idx_a], top_idx[idx_b]
            db = b[k1] - b[k2]
            if abs(db) > 1e-15:
                x_cand = (c_base[k2] - c_base[k1]) / db
                if x_cand > 0:
                    candidates.append(x_cand)
        # Peak of individual g_k
        if b[top_idx[idx_a]] > 1e-15:
            x_peak = sum_other - 2.0 * c_base[top_idx[idx_a]] / b[top_idx[idx_a]]
            if x_peak > 0:
                candidates.append(x_peak)
    
    # Also add a few heuristic candidates
    candidates.extend([0.01 * sum_other, 0.1 * sum_other, 0.5 * sum_other, sum_other])
    
    # Evaluate f(x) at all candidates
    best_x = 0.0
    best_f = np.inf
    for x in candidates:
        c_new = c_base + x * b
        max_c = c_new.max()
        s = sum_other + x
        if s > 0:
            f = 2.0 * n * max_c / (s * s)
        else:
            f = np.inf
        if f < best_f:
            best_f = f
            best_x = x
    
    a[i] = best_x
    return a
```

This is O(K^2 + K + n) per coordinate (the c_base + x*b is O(n) but we do it for each candidate, so it's O(n * (K^2 + K + 4))). For n=1024, K=15, that's about 1024 * 250 = 256K operations per coordinate. With n coordinates, one pass is about 260M operations. That might be a bit slow.

Let me optimize: instead of computing `c_base + x*b` for each candidate, I can just evaluate the max over the top few indices (since the max of the full convolution is likely achieved at one of the top K base values or a nearby one).

Actually, a simpler optimization: just evaluate `max(c_base + x*b)` by taking the max over all indices, but do it in a vectorized way for all candidates at once:

```python
# c_candidates[j, k] = c_base[k] + candidates[j] * b[k]
# Then take max along axis 1
c_mat = c_base[None, :] + np.array(candidates)[:, None] * b[None, :]
max_vals = c_mat.max(axis=1)
f_vals = 2.0 * n * max_vals / (sum_other + np.array(candidates))**2
```

This is O(len(candidates) * (2n-1)) which for 250 candidates and 2047 length is about 500K operations. That's fast.

Now, one key insight: **the shape of the optimal solution**. 

For a "flat" convolution, we want `|FFT(a)|^2` to be as flat as possible (by Parseval's theorem, the sum of squared convolution values equals the sum of squared FFT values). But we need the LINEAR convolution to be flat, not the circular one.

Actually, let me reconsider. The relevant quantity is the aperiodic autocorrelation. We want the aperiodic autocorrelation to be as "flat" as possible (all sidelobes equal to the minimum possible).

This is related to the theory of "low-autocorrelation sequences" but in the continuous domain. For continuous sequences, the optimal is related to prolate spheroidal wave functions or similar.

Actually, for the continuous case with the constraint `a >= 0`, the optimal shape is likely a "raised cosine" or "sinc-like" function. Let me try different parametric families.

```python
"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines coordinate descent (exact per-coordinate optimization),
logit-space Adam with soft-max surrogates, and multi-scale search.
"""
import numpy as np
import time
import sys

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def full_convolve_fft(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def compute_h(w, a):
    """h[i] = sum_{m} a[m]*w[i+m] for i=0,...,n-1."""
    n = len(a)
    af = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1: 2 * n - 1]

def coordinate_descent_pass(a, topK=15):
    """One pass of coordinate descent (randomized order)."""
    n = len(a)
    for i in range(n):
        old = a[i]
        a[i] = 0.0
        c_base = np.convolve(a, a)
        a[i] = old
        
        a_other = a.copy()
        a_other[i] = 0.0
        sum_other = float(a_other.sum())
        
        if sum_other < 1e-15:
            continue
            
        # Compute b vector
        b = np.zeros(2 * n - 1)
        # b[k] = a_other[k-i] (if valid and != i) + a_other[i-k] (if valid and != i)
        # Vectorize:
        idx1 = np.arange(2 * n - 1) - i  # k - i for k in range(2n-1)
        idx2 = i - np.arange(2 * n - 1)  # i - k for k in range(2n-1)
        
        mask1 = (idx1 >= 0) & (idx1 < n) & (idx1 != i)
        mask2 = (idx2 >= 0) & (idx2 < n) & (idx2 != i)
        
        b[mask1] += a_other[idx1[mask1]]
        b[mask2] += a_other[idx2[mask2]]
        
        # Find top K
        top_idx = np.argsort(c_base)[-topK:]
        
        # Candidates
        candidates = [0.0, old]
        
        # Kink points between pairs
        for ia in range(len(top_idx)):
            for ib in range(ia + 1, len(top_idx)):
                k1, k2 = top_idx[ia], top_idx[ib]
                db = b[k1] - b[k2]
                if abs(db) > 1e-15:
                    xc = (c_base[k2] - c_base[k1]) / db
                    if xc > 0 and xc < 10 * sum_other:
                        candidates.append(xc)
        
        # Individual peaks
        for k in top_idx:
            if b[k] > 1e-15:
                xp = sum_other - 2.0 * c_base[k] / b[k]
                if xp > 0 and xp < 10 * sum_other:
                    candidates.append(xp)
        
        # Heuristic
        candidates.extend([0.01 * sum_other, 0.05 * sum_other, 0.1 * sum_other,
                          0.5 * sum_other, sum_other])
        
        candidates = np.array(candidates)
        candidates = np.maximum(candidates, 0.0)
        
        # Evaluate
        c_mat = c_base[None, :] + candidates[:, None] * b[None, :]
        max_vals = c_mat.max(axis=1)
        s_vals = sum_other + candidates
        f_vals = 2.0 * n * max_vals / (s_vals * s_vals)
        
        best_idx = int(np.argmin(f_vals))
        a[i] = candidates[best_idx]
    
    return a

def coordinate_descent(a0, deadline, topK=15, passes_max=50):
    """Run coordinate descent until deadline."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for p in range(passes_max):
        if time.monotonic() > deadline:
            break
        a = coordinate_descent_pass(a, topK=topK)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if p % 10 == 0:
            print(f"    CD pass {p}: score={sc:.10f}")
    
    return best_a, best_score


def logit_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.08,
                        beta_decay=0.92, beta_min=1e-10, restart_period=1500):
    """Logit-space Adam with soft-max."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.15 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result
```

```python
"""
Hybrid optimization combining exact coordinate descent (with incremental
convolution updates), logit-space Adam with adaptive soft-max, and
multi-scale search to minimize 2n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve_fft(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    n = len(a)
    af = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1: 2 * n - 1]


def coordinate_descent(a0, deadline, topK=20, evalK=50, passes_max=200, rng=None):
    """
    Coordinate descent with incremental convolution updates.
    For each coordinate, finds the exact optimal value given all others fixed.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    S = float(a.sum())

    c = np.convolve(a, a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max() / (S * S))

    order = np.arange(n)

    for p in range(passes_max):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        for i in order:
            old = a[i]
            if old < 1e-18:
                continue

            # Find optimal x for a[i]
            # c_new[k] = c[k] + 2*(x-old)*a[k-i] for k != 2i
            #           c[k] + x^2 - old^2 for k = 2i
            # S_new = S - old + x

            # Compute the "shifted a" vector: a_shifted[k] = a[k-i] if 0<=k-i<n else 0
            # We need this for k in range(2n-1)
            k_range = np.arange(2 * n - 1)
            idx = k_range - i
            a_shifted = np.zeros(2 * n - 1)
            valid = (idx >= 0) & (idx < n)
            a_shifted[valid] = a[idx[valid]]

            # But at k=2i, the formula is different: c[2i] + x^2 - old^2
            # vs the general formula: c[2i] + 2*(x-old)*a[2i-i] = c[2i] + 2*(x-old)*a[i] = c[2i] + 2*(x-old)*old
            # The correct formula for k=2i: c_new[2i] = c[2i] - old^2 + x^2
            # The "general" formula gives: c[2i] + 2*(x-old)*old = c[2i] + 2*x*old - 2*old^2
            # Difference: (x^2 - old^2) - (2*x*old - 2*old^2) = x^2 - 2*x*old + old^2 = (x-old)^2
            # So: c_new[2i] = c[2i] + 2*(x-old)*a_shifted[2i] + (x-old)^2
            # Since a_shifted[2i] = a[2i-i] = a[i] = old:
            # c_new[2i] = c[2i] + 2*(x-old)*old + (x-old)^2 = c[2i] + (x-old)*(2*old + x - old) = c[2i] + (x-old)*(x+old) = c[2i] + x^2 - old^2 ✓

            # So the general formula is: c_new[k] = c[k] + 2*(x-old)*a_shifted[k] + (x-old)^2*delta[k-2i, 0]
            # Let d = x - old, then: c_new[k] = c[k] + 2*d*a_shifted[k] + d^2*delta[k,2i]

            # For the top evalK positions of c, evaluate the objective
            top_eval = np.argsort(c)[-evalK:]

            # Generate candidate x values
            candidates = [0.0, old]

            # Kink points where two top positions are tied
            top_c = top_eval[:min(topK, evalK)]
            for ia in range(len(top_c)):
                for ib in range(ia + 1, len(top_c)):
                    k1, k2 = top_c[ia], top_c[ib]
                    # c[k1] + 2*x*a_s[k1] = c[k2] + 2*x*a_s[k2] (ignoring the d^2 term for kink finding)
                    # 2*x*(a_s[k1] - a_s[k2]) = c[k2] - c[k1]
                    da = 2.0 * (a_shifted[k1] - a_shifted[k2])
                    if abs(da) > 1e-15:
                        xc = (c[k2] - c[k1]) / da
                        if xc > 0:
                            candidates.append(xc)

            # Where individual term peaks (derivative of (c[k]+2*x*a_s[k]+x^2*delta)/(S-x+new... )^2 = 0)
            # Simplified: where c[k] + 2*x*a_s[k] is maximized relative to (S+x-old)^2
            # d/dx [(c[k]+2*x*a_s[k])/(S+x-old)^2] = 0
            # (2*a_s[k]*(S+x-old)^2 - 2*(c[k]+2*x*a_s[k])*(S+x-old)) = 0
            # 2*a_s[k]*(S+x-old) - 2*(c[k]+2*x*a_s[k]) = 0
            # a_s[k]*(S+x-old) = c[k]+2*x*a_s[k]
            # a_s[k]*S + a_s[k]*x - a_s[k]*old = c[k] + 2*x*a_s[k]
            # a_s[k]*S - a_s[k]*old - c[k] = x*a_s[k]
            # x = (a_s[k]*S - a_s[k]*old - c[k]) / a_s[k] = S - old - c[k]/a_s[k]
            for k in top_c:
                if a_shifted[k] > 1e-15:
                    xp = S - old - c[k] / a_shifted[k]
                    if xp > 0:
                        candidates.append(xp)

            # Heuristic candidates
            candidates.extend([old * 0.5, old * 1.5, old * 0.1, old * 10.0,
                               S / n, S / (2 * n)])

            candidates = np.array(candidates)
            candidates = np.clip(candidates, 0.0, 100.0 * S)
            # Remove duplicates
            candidates = np.unique(np.round(candidates, 10))

            if len(candidates) == 0:
                continue

            # Evaluate: for each candidate x, compute max over top_eval positions
            # c_new[k] = c[k] + 2*(x-old)*a_shifted[k] + (x-old)^2 * (1 if k==2i else 0)
            d = candidates - old  # shape (C,)
            # c_new for top_eval: shape (C, E)
            c_new_mat = c[top_eval][None, :] + 2.0 * d[:, None] * a_shifted[top_eval][None, :]
            # Add the (x-old)^2 correction for k==2i
            if 2 * i < len(c):
                idx_2i = np.where(top_eval == 2 * i)[0]
                if len(idx_2i) > 0:
                    c_new_mat[:, idx_2i] += d ** 2
            max_new = c_new_mat.max(axis=1)
            S_new = S - old + candidates
            f_vals = 2.0 * n * max_new / (S_new * S_new)

            # Also check the full max (not just top_eval) for the best candidate
            best_local_idx = int(np.argmin(f_vals))
            best_x = candidates[best_local_idx]

            # Verify with full convolution for the best candidate
            # (skip for speed, trust the top_eval approximation)

            # Update
            if best_x >= 0:
                d_val = best_x - old
                if abs(d_val) > 1e-18:
                    c += 2.0 * d_val * a_shifted
                    if 2 * i < len(c):
                        c[2 * i] += d_val * d_val  # correction for the (x-old)^2 term
                    c = np.maximum(c, 0.0)
                    a[i] = best_x
                    S = S - old + best_x

        # End of pass
        sc = float(2.0 * n * c.max() / (S * S))
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
        if p % 20 == 0:
            print(f"      CD pass {p}: score={sc:.10f}")

    return best_a, best_score


def logit_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.08,
                        beta_decay=0.92, beta_min=1e-10, restart_period=1500):
    """Logit-space Adam with adaptive soft-max."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.15 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def peak_targeting_refine(a0, deadline, rng, n_peaks=5, lr=0.03):
    """Refine by targeting top peaks in convolution."""
    n = len(a0)
    a0 = np.asarray(a0, dtype=np.float64).ravel()

    a0 = a0 / a0.sum()

    best_a = a0.copy()

    best_score = float(2.0 * n * c.max())

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        peak_indices = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        for idx in peak_indices:
            w[idx] = 1.0 / n_peaks

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score:

    return best_a, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_spectral_profile(n, rng, nfreq=8):
    """Generate a profile by designing the spectrum for flat autocorrelation."""
    t = np.linspace(0.0, 1.0, n)
    # Try to make |FFT(a)|^2 as flat as possible
    # This means a should be like a random phase sequence (but we need a >= 0)
    # Compromise: use a raised-cosine window modulated by a low-frequency signal
    k = rng.integers(1, nfreq + 1)
    amp = rng.uniform(0.1, 0.5)
    phase = rng.uniform(0, 2 * np.pi)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)  # base raised cosine
    a = a + amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting hybrid search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Coordinate descent on initial sequence =====
    if init_seq is not None:
        n_cd = init_seq.size
        print(f"\n--- Phase 1: Coordinate descent n={n_cd} ---")
        cd_deadline = min(deadline, t_start + 300.0)
        result_cd, sc_cd = coordinate_descent(init_seq, cd_deadline, topK=20, evalK=60,
                                              passes_max=100, rng=rng)
        sc_final = compute_score(result_cd)
        update(result_cd, sc_final)
        print(f"  After CD: {sc_final:.10f}")
    else:
        result_cd = None

    # ===== Phase 2: Logit Adam for global exploration =====
    n_opt = 1024
    print(f"\n--- Phase 2: Logit Adam n={n_opt} ---")

    # Start from best available
    if best_seq is not None:
        p0 = interpolate_profile(best_seq, n_opt)
    else:
        p0 = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n_opt))
        p0 = p0 / p0.sum()

    dl2 = min(deadline, t_start + 600.0)
    result2, sc2 = logit_adam_optimize(p0, dl2, rng, lr=0.03,
                                        beta_factor=0.06, beta_decay=0.93)
    sc2_final = compute_score(result2)
    update(result2, sc2_final)
    print(f"  After Adam: {sc2_final:.10f}")

    # Also try from the CD result
    if result_cd is not None:
        p0b = interpolate_profile(result_cd, n_opt)
        dl2b = min(deadline, t_start + 700.0)
        if time.monotonic() < dl2b:
            result2b, sc2b = logit_adam_optimize(p0b, dl2b, rng, lr=0.02,
                                                  beta_factor=0.04, beta_decay=0.94)
            sc2b_final = compute_score(result2b)
            update(result2b, sc2b_final)
            print(f"  After Adam (from CD): {sc2b_final:.10f}")

    # ===== Phase 3: Coordinate descent refinement =====
    if best_seq is not None:
        n_cd3 = best_seq.size
        print(f"\n--- Phase 3: CD refinement n={n_cd3} ---")
        cd3_deadline = min(deadline, t_start + 1000.0)
        result_cd3, sc_cd3 = coordinate_descent(best_seq, cd3_deadline, topK=25, evalK=80,
                                                passes_max=200, rng=rng)
        sc_cd3_final = compute_score(result_cd3)
        update(result_cd3, sc_cd3_final)
        print(f"  After CD refinement: {sc_cd3_final:.10f}")

    # ===== Phase 4: Peak targeting =====
    if best_seq is not None:
        n_pt = best_seq.size
        print(f"\n--- Phase 4: Peak targeting n={n_pt} ---")
        pt_deadline = min(deadline, t_start + 1300.0)
        if time.monotonic() < pt_deadline:
            dl_pt = min(pt_deadline, time.monotonic() + 100.0)
            r_pt, sc_pt = peak_targeting_refine(best_seq, dl_pt, rng, n_peaks=3, lr=0.015)
            sc_pt_final = compute_score(r_pt)
            update(r_pt, sc_pt_final)
            print(f"  After peak targeting: {sc_pt_final:.10f}")

    # ===== Phase 5: Try larger n =====
    if best_seq is not None and best_score > 1.503:
        for n_big in [2048, 4096]:
            if time.monotonic() > deadline - 60:
                break
            print(f"\n--- Trying n={n_big} ---")
            p_big = interpolate_profile(best_seq, n_big)
            sc_big_init = compute_score(p_big)
            update(p_big, sc_big_init)
            print(f"  Interpolated: {sc_big_init:.10f}")

            # CD at large n
            dl_cd = min(deadline - 30, time.monotonic() + 120.0)
            if time.monotonic() < dl_cd:
                r_big, sc_big = coordinate_descent(p_big, dl_cd, topK=20, evalK=60,
                                                   passes_max=50, rng=rng)
                sc_big_final = compute_score(r_big)
                update(r_big, sc_big_final)
                print(f"  After CD: {sc_big_final:.10f}")

    # ===== Phase 6: Final CD polish =====
    if best_seq is not None:
        n_final = best_seq.size
        print(f"\n--- Phase 6: Final CD polish n={n_final} ---")
        final_cd_deadline = deadline - 15.0
        if time.monotonic() < final_cd_deadline:
            r_final, sc_final = coordinate_descent(best_seq, final_cd_deadline,
                                                   topK=30, evalK=100,
                                                   passes_max=500, rng=rng)
            sc_final2 = compute_score(r_final)
            update(r_final, sc_final2)
            print(f"  Final CD: {sc_final2:.10f}")

    # ===== Phase 7: Try smaller n if best is large =====
    if best_seq is not None and best_seq.size > 1024:
        for n_small in [1024, 512, 256]:
            if time.monotonic() > deadline - 10:
                break
            p_small = interpolate_profile(best_seq, n_small)
            sc_s = compute_score(p_small)
            update(p_small, sc_s)
            if sc_s < best_score + 0.01:
                dl_s = min(deadline - 5, time.monotonic() + 30.0)
                r_s, _ = coordinate_descent(p_small, dl_s, topK=20, evalK=60,
                                            passes_max=100, rng=rng)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  n={n_small}: {sc_s2:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        t = np.linspace(0, 1, 512)
        final_seq = (0.5 + 0.5 * np.cos(2 * np.pi * t))
        final_seq = final_seq / final_seq.sum()

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        t = np.linspace(0, 1, 256)
        final_seq = (0.5 + 0.5 * np.cos(2 * np.pi * t))
        final_seq = final_seq / final_seq.sum()

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
```
