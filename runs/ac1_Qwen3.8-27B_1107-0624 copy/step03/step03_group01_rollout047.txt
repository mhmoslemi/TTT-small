"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_max_info(a):

    cm = float(c.max())
    idx_max = int(np.argmax(c))

def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):

    c = _full_convolve(a, a)
    cm = float(c.max())

    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0

    return grad

    if not np.any(active):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:

def _make_profiles(n):

        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Exponential decay
    for lam in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-lam * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Power law decay
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = (1.0 - t + 1e-6) ** p
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Hann window (symmetric)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())

    # Asymmetric: rising then falling
    for peak_pos in [0.3, 0.4, 0.5, 0.6, 0.7]:
        a = np.exp(-5.0 * (t - peak_pos) ** 2)

    # Step-like

        a = np.ones(n)
        a[int(n * frac):] *= 0.3
        a = np.maximum(a, 1e-10)

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        key = (pn, label[:10])

            continue
        seen.add(key)

        per_time = min(60.0, phase1_end - time.monotonic() - 10)
        if per_time < 5:
            break

        p_best, sc_best = _run_lbfgs_continuation(
            p, time.monotonic() + per_time, rng, max_wall_time=per_time
        )
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Directed descent on best (200s) ---
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        per_time = min(80.0, phase2_end - time.monotonic() - 5)
        if per_time > 5:
            p_best, sc_best = _directed_descent(
                best_seq, time.monotonic() + per_time,
                lr_init=0.005, n_steps=10000
            )
            update_best(p_best, sc_best, "directed")

            # Repeat with different lr
            per_time2 = min(60.0, phase2_end - time.monotonic() - 5)
            if per_time2 > 5:
                p_best2, sc_best2 = _directed_descent(
                    best_seq, time.monotonic() + per_time2,
                    lr_init=0.001, n_steps=10000
                )
                update_best(p_best2, sc_best2, "directed_fine")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam optimization (300s) ---
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_time = min(120.0, phase3_end - time.monotonic() - 10)
        if per_time > 10:
            p_best, sc_best = _adam_optimize(
                best_seq, time.monotonic() + per_time,
                lr=0.01, beta_frac=0.003, n_max_iter=50000
            )
            update_best(p_best, sc_best, "adam1")

            # Second Adam pass with different params
            per_time2 = min(100.0, phase3_end - time.monotonic() - 5)
            if per_time2 > 10:
                p_best2, sc_best2 = _adam_optimize(
                    best_seq, time.monotonic() + per_time2,
                    lr=0.005, beta_frac=0.001, n_max_iter=30000
                )
                update_best(p_best2, sc_best2, "adam2")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Structural mutations + L-BFGS on mutated (200s) ---
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_time = min(40.0, phase4_end - time.monotonic() - 10)
        if per_time > 5:
            p_best, sc_best = _structural_mutations(
                best_seq, time.monotonic() + per_time, rng
            )
            update_best(p_best, sc_best, "structural")

        # L-BFGS on the structurally mutated result
        per_time2 = min(80.0, phase4_end - time.monotonic() - 10)
        if per_time2 > 10:
            p_best2, sc_best2 = _run_lbfgs_continuation(
                best_seq, time.monotonic() + per_time2, rng, max_wall_time=per_time2
            )
            update_best(p_best2, sc_best2, "lbfgs_post_struct")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Length adaptation (200s) ---
    phase5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)

        # Try doubling the length
        if n_curr < 4096:
            target_n = min(4096, n_curr * 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            per_time = min(80.0, phase5_end - time.monotonic() - 20)
            if per_time > 10:
                p_best, sc_best = _run_lbfgs_continuation(
                    p_interp, time.monotonic() + per_time, rng, max_wall_time=per_time
                )
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

            # Adam on the interpolated
            per_time2 = min(60.0, phase5_end - time.monotonic() - 10)
            if per_time2 > 10:
                p_best2, sc_best2 = _adam_optimize(
                    p_interp, time.monotonic() + per_time2,
                    lr=0.005, beta_frac=0.002, n_max_iter=20000
                )
                update_best(p_best2, sc_best2, f"adam_n{target_n}")

        # Try halving the length (if currently large)
        if n_curr > 512:
            target_n = max(256, n_curr // 2)
            p_interp2 = _interp_profile(best_seq, target_n)
            sc_interp2 = _score_norm_fn(p_interp2)
            update_best(p_interp2, sc_interp2, f"interp_n{target_n}")

            per_time3 = min(40.0, phase5_end - time.monotonic() - 10)
            if per_time3 > 5:
                p_best3, sc_best3 = _run_lbfgs_continuation(
                    p_interp2, time.monotonic() + per_time3, rng, max_wall_time=per_time3
                )
                update_best(p_best3, sc_best3, f"lbfgs_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine polishing (remaining time) ---
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        # Final L-BFGS with very small beta
        per_time = min(60.0, phase6_end - time.monotonic() - 15)
        if per_time > 5:
            p_best, sc_best = _run_lbfgs_continuation(
                best_seq, time.monotonic() + per_time, rng, max_wall_time=per_time
            )
            update_best(p_best, sc_best, "final_lbfgs")

        # Local search
        per_time2 = min(30.0, phase6_end - time.monotonic() - 10)
        if per_time2 > 5:
            p_best2, sc_best2 = _local_search(
                best_seq, time.monotonic() + per_time2, n_trials=500
            )
            update_best(p_best2, sc_best2, "local_search")

        # Another directed descent
        per_time3 = min(20.0, phase6_end - time.monotonic() - 5)
        if per_time3 > 3:
            p_best3, sc_best3 = _directed_descent(
                best_seq, time.monotonic() + per_time3,
                lr_init=0.0005, n_steps=5000
            )
            update_best(p_best3, sc_best3, "final_directed")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Multi-restart: try different noise levels on best ---
    if time.monotonic() < deadline - 30 and best_seq is not None:
        n_best = len(best_seq)
        for noise_level in [0.05, 0.02, 0.01]:
            if time.monotonic() > deadline - 10:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_time = min(30.0, deadline - time.monotonic() - 5)
            if per_time < 5:
                break
            p_best, sc_best = _run_lbfgs_continuation(
                p_start, time.monotonic() + per_time, rng, max_wall_time=per_time
            )
            update_best(p_best, sc_best, f"restart_n{noise_level}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(512)[0]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence if possible
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence check: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(512)[2]
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
