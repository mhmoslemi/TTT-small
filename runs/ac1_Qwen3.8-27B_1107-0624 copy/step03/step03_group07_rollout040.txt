import numpy as np
import time

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * max(0, c.max()) / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def convolve_fft(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    # Algorithm from Duchi et al. (2011)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_exact(a, c, eps=0):
    """Compute subgradient of max(c) targeting peaks within eps of max."""
    n = len(a)
    c_max = c.max()
    peaks = np.where(c >= c_max - eps)[0]
    # For each peak k, the gradient wrt a[m] is 2*a[k-m] (if 0 <= k-m < n)
    g = np.zeros(n)
    for k in peaks:
        # k-m is valid when 0 <= k-m < n, i.e., max(0, k-n+1) <= m <= min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi+1)
            g[indices] += 2.0 * a[k - indices]
    # Normalize by number of peaks for stability
    g /= len(peaks)
    return g

def optimize_main(a0, deadline, rng, method="combined"):
    """Main optimization loop."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    c = convolve_fft(a)
    best_a = a.copy()
    best_score = 2.0 * n * c.max()
    
    # Parameters
    lr = 0.1
    beta = 0.01 * c.max()  # softmax temperature
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() < deadline:
        c = convolve_fft(a)
        c_max = c.max()
        score = 2.0 * n * c_max

"""
Optimized step function search using multi-peak subgradient descent,
softmax smoothing, and parametric initialization. Combines exact
subgradient methods with adaptive step sizes to minimize 2n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * max(0.0, c.max()) / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_multi(a, c, rel_eps=0.05):
    """Compute averaged subgradient targeting peaks within rel_eps of max."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    threshold = c_max * (1.0 - rel_eps)
    peaks = np.where(c >= threshold)[0]
    g = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            g[indices] += 2.0 * a[k - indices]
    g /= len(peaks)
    return g


def softmax_weights(c, beta):

    c_max = float(c.max())

        cnt = int(mask.sum())

        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - c_max) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def grad_from_weights(a, w):
    """Compute gradient from weights: g[m] = 2 * sum_k w[k]*a[k-m]."""
    n = len(a)

    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1]
