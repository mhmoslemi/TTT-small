"""
Optimized search for minimum-peak-autocorrelation sequences.
Uses LP spreading, Nesterov-accelerated subgradient, and coordinate descent.
Explores multiple sequence lengths (512-3072) to find the best trade-off.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_conv_matrix(a):
    n = len(a)
    # Build the (2n-1) x n convolution matrix
    # Row k of the matrix, when multiplied by a, gives conv(a,a)_k
    # Actually, for the LP we need: conv(a_fixed, g)_k <= M
    # conv(a_fixed, g)_k = sum_i a_fixed_i * g_{k-i}
    # So the matrix A has A[k, j] = a_fixed_i where i = k-j (if valid)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g_j contributes to conv(a, g)_k for k = j, j+1, ..., j+n-1
        # with coefficient a_{k-j} = a_0, a_1, ..., a_{n-1}
        A[j:j+n, j] = a
    return A

def _lp_spread(a, M, max_time=10.0):
    """LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0"""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    
    t0 = time.monotonic()

                  bounds=[(0, None)] * n, method='highs',

def _interpolate_profile(p0, n_new):

    if n0 == n_new:
        s = p0.sum()

        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _conv_grad(a, w):

    = 2 * conv_rev(w, a) truncated to length n

    # = (conv(w, flip(a)))_j shifted...
    # Actually, let's just compute it directly via FFT:
    # sum_k w_k * a_{k-j} for each j
    # Let b = a (length n), w (length 2n-1)
    # We want c_j = sum_k w_k * b_{k-j} = (corr)(w, b)_j
    # Using FFT: corr(w,b) = IFFT(FFT(w) * conj(FFT(b)))
    # But we need to be careful about lengths.
    
    # Simpler: use the fact that sum_k w_k * a_{k-j} = (conv(w, rev_a))_{(2n-2) - j}

    # Actually, let me just use the direct formula via FFT:

    # We want c_j = corr[j] for j = 0, ..., n-1

    b2 = 0.999

    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    base_lr = lr

    # Nesterov momentum
    z_prev = z.copy()
    momentum = 0.9

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_grad(c, beta)
        grad = _conv_grad(p, w)
        
        # Add small regularization to prevent collapse
        grad += 1e-8 * p

        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        
        # Nesterov: use extrapolated position
        z_nest = z + momentum * (z - z_prev)
        z_new = z_nest - base_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            z_new = z - base_lr * mh / (np.sqrt(vh) + eps)

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                base_lr = max(0.003, base_lr * 0.9)

        # Adaptive beta
        if it % 500 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.8)
                elif gap > beta * 10.0:
                    beta = min(0.5 * c_max_new, beta * 1.5)

        # Restart
        max_no_imp = max(8000, 10 * n)
        if no_imp > max_no_imp:
            rng = np.random.default_rng(it * 104729 + 7)
            z_restart = best_z + 0.2 * best_score * rng.standard_normal(n)
            z_restart = np.clip(z_restart, -60.0, 60.0)

            z = z_restart

            m = np.zeros(n)

            momentum = 0.9
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
            no_imp = 0

        z_prev = z.copy()
        z = z_new

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):

    n = a.size

        return np.array([])

def _cross_corr(w, a):
    """Compute cross-correlation: result[j] = sum_i w[i] * a[i-j] for j=0..n-1
    Actually: result[j] = sum_i w[i+j] * a[i] (for Hankel gradient)
    We want: grad_j = sum_i w_{j+i} * a_i

    g_j = sum_i b_i * w_{i+j}

    Using FFT: g = IFFT(conj(FFT(w)) * FFT(b)) at appropriate indices.
    Actually: cross_corr(b, w)[j] = sum_i b[i] * conj(w[i-j]) = sum_i b[i] * w[i-j] (for real)
    We want sum_i b[i] * w[i+j] = cross_corr(w, b)[-j] = cross_corr(b, w)[j]... 
    
    Let me just do it via convolution:
    g_j = sum_i w_{i+j} * a_i

    g_j = sum_i w_{i+j} * a_i = sum_i w_{i+j} * ar_{n-1-i}

    # g_j = conv_full[n-1+j] for j = 0, ..., n-1
    g = conv_full[n - 1: 2 * n - 1]
    return g


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # First project onto simplex sum=1, then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return w * (target_sum / s)


def _make_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _interpolate_tapered(p0, n_new, taper_strength=0.3):

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size

        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    # Apply partial Hann taper
    hann = 0.5 * (1.0 + np.cos(np.pi * x1))
    p = p * (1.0 - taper_strength + taper_strength * hann)
    p = np.maximum(p, 1e-15)
    s = p.sum()
    return p / s


def _smooth_max_weights(c, beta):
    """Compute soft-max weights over convolution values."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return np.zeros_like(c)
    return ex / s


def _active_set_qp_optimize(a, deadline, max_iter=30000, beta_init=None, lr_init=0.01):
    """
    Active-set QP optimization using projected gradient on:
    minimize sum_k w_k * conv(a_new, a_new)_k
    subject to a_new >= 0, sum(a_new) = 1
    
    where w_k are smooth-max weights from current convolution.
    The gradient is computed via FFT exploiting Hankel structure.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    best_score = float(2.0 * n * c_max)
    best_a = a.copy()

    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-8, 0.02 * c_max)

    lr = lr_init
    no_imp = 0
    prev_obj = None
    prev_a = None

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weights(c, beta)
        
        # Compute gradient: grad_j = 2 * sum_i w_{i+j} * a_i
        grad = 2.0 * _cross_corr(w, a)
        
        # Add small regularization
        grad += 1e-10 * a

        # Projected gradient step with backtracking
        # First, project gradient onto simplex tangent:
        # We minimize f(a) s.t. a >= 0, sum(a) = 1
        # Projected gradient: a_new = project_simplex(a - lr * grad)
        
        # Backtracking line search
        alpha = lr
        f_curr = float(np.dot(w, c))  # approximate objective
        max_inner = 20
        improved = False
        for inner in range(max_inner):
            a_try = _project_simplex(a - alpha * grad)
            c_try = _fft_convolve(a_try)
            f_try = float(np.dot(w, c_try))
            c_try_max = float(c_try.max())
            score_try = float(2.0 * n * c_try_max)
            
            # Armijo condition (approximate)
            if f_try <= f_curr + 1e-4 * alpha * float(np.dot(grad, a - a_try)) + 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Take the step anyway with very small alpha
            a = _project_simplex(a - 1e-6 * grad)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        c = _fft_convolve(a)
        c_max = float(c.max())
        score = float(2.0 * n * c_max)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            # Slightly increase lr if improving
            lr = min(lr * 1.02, 0.05)
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(1e-5, lr * 0.9)

        # Adaptive beta
        if it % 300 == 0 and it > 50:
            sorted_c = np.sort(c)[::-1]
            top_k = min(10, len(sorted_c))
            gap = sorted_c[0] - sorted_c[top_k - 1]
            if gap < beta * 0.3:
                beta = max(1e-12, beta * 0.85)
            elif gap > beta * 20.0:
                beta = min(0.5 * float(c.max()), beta * 1.3)

        # Restage if stuck
        max_no_imp = max(5000, 8 * n)
        if no_imp > max_no_imp:
            rng = np.random.default_rng(it * 7919 + 3)
            noise = 0.1 * best_score * rng.standard_normal(n)
            a = best_a * (1.0 + noise / (best_a.sum() + 1e-15))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            lr = lr_init
            beta = max(1e-8, 0.02 * float(_fft_convolve(a).max()))
            no_imp = 0

    return best_a


def _logit_adam_optimize(a, deadline, max_iter=40000, lr=0.04):
    """Adam in logit space (proven effective, kept as backup)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weights(c, beta)
        grad = 2.0 * _cross_corr(w, p)
        grad += 1e-8 * p

        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                base_lr = max(0.003, base_lr * 0.92)

        if it % 500 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.8)
                elif gap > beta * 10.0:
                    beta = min(0.5 * c_max_new, beta * 1.4)

        max_no_imp = max(8000, 10 * n)
        if no_imp > max_no_imp:
            rng = np.random.default_rng(it * 104729 + 7)
            z_restart = best_z + 0.2 * best_score * rng.standard_normal(n)
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _lp_refine(a, deadline, max_iter=5000, inflate_init=1.0005, blend_init=0.3):
    """LP-based spreading with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    inflate = inflate_init
    blend = blend_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        # Time limit per LP call based on remaining budget
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break
        max_time_lp = min(15.0, max(2.0, remaining * 0.05))

        g = _lp_spread(a, M_lp, max_time=max_time_lp)

        if g is None:
            inflate = min(1.01, inflate * 1.5)
            if inflate > 1.005:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate = min(1.01, inflate * 1.1)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            # Tighten inflation
            inflate = max(1.0002, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend = max(0.05, blend * 0.9)
                inflate = min(1.005, inflate * 1.02)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend = blend_init
                inflate = inflate_init
                no_imp = 0

        a = a_new

    return best_a


def _coordinate_descent(a, deadline, max_iter=100000):
    """Coordinate descent: adjust one coordinate at a time.
    For coordinate j, the objective as a function of a_j is:
    f(a_j) = 2n * max_k (c_k + 2*a_j*a_{k-j} + a_j^2 * delta_{k, 2j}... )
    This is complex. Instead, use a simple approach:
    Try moving each coordinate up or down by small amounts.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    step_size = 0.01 / n

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Select coordinate to modify: the one that most reduces the max
        c = _fft_convolve(a)
        c_max = float(c.max())
        peak_idx = np.argmax(c)
        
        # The peak at position peak_idx is: sum_i a_i * a_{peak_idx - i}
        # To reduce it, we should reduce a at positions that contribute most
        # Contribution of a_j to c[peak_idx] is a_{peak_idx - j} (if valid)
        contrib = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            i = peak_idx - j
            if 0 <= i < n:
                contrib[j] = a[i]

        # Pick the coordinate with highest contribution
        j_max = int(np.argmax(contrib))
        
        # Try reducing a[j_max]
        delta = step_size * a[j_max]
        a_try = a.copy()
        a_try[j_max] = max(1e-15, a_try[j_max] - delta)
        a_try = a_try / a_try.sum()
        c_try = _fft_convolve(a_try)
        score_try = float(2.0 * n * c_try.max())
        
        if score_try < best_score - 1e-12:
            best_score = score_try
            best_a = a_try.copy()
            a = a_try
            no_imp = 0
            # Slightly increase step
            step_size = min(0.05 / n, step_size * 1.01)
        else:
            # Try increasing (might help by red distribution)
            a_try = a.copy()
            a_try[j_max] += delta
            a_try = a_try / a_try.sum()
            c_try = _fft_convolve(a_try)
            score_try = float(2.0 * n * c_try.max())
            
            if score_try < best_score - 1e-12:
                best_score = score_try
                best_a = a_try.copy()
                a = a_try
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    step_size = max(1e-6 / n, step_size * 0.9)
                if no_imp > 5000:
                    break

    return best_a


def _make_hann_tapered(n, center_boost=1.0, edge_decay=0.5):
    """Create a Hann-tapered initialization with optional center boost."""
    t = np.linspace(0.0, 1.0, n)
    hann = 0.5 * (1.0 + np.cos(np.pi * t))
    a = hann + center_boost * np.exp(-((t - 0.5) / 0.2) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_tapered(n, slope=1.0):
    """Linear profile with taper."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    hann = 0.5 * (1.0 + np.cos(np.pi * t))
    a = a * (0.5 + 0.5 * hann)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
