```python
def _subgradient_step(a, lr):
    """One subgradient step with simplex projection."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    # Find active set
    tol = 1e-10
    active = np.where(c >= m_val - tol)[0]
    w = np.zeros(2*n-1)
    w[active] = 1.0 / len(active)
    
    # Compute subgradient using FFT
    N = _next_pow2(2*n-1)
    Af = np.fft.fft(a, N)
    Wf = np.fft.fft(w, N)
    grad = 2.0 * np.fft.ifft(Wf * np.conj(Af)).real[:n]
    
    # Projected gradient step on simplex
    a_new = a - lr * grad
    a_new = _project_simplex(a_new)
    return a_new, m_val

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
- LSE-smoothed gradient descent (Adam in logit space)
- FFT-accelerated exact subgradient with simplex projection
- LP-based spreading with line search

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_fft(a, w, N):
    n = len(a)
    Af = np.fft.fft(a, N)
    Wf = np.fft.fft(w, N)
    grad = 2.0 * np.fft.ifft(Wf * np.conj(Af)).real[:n]
    return grad


def _lse_adam_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.05, T_end=1e-5, lr=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, m_val = _lse_adam_step(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1500 == 0:
                lr = max(0.01, lr * 0.92)
            if no_imp > max(4000, 8 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.06 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

                no_imp = 0

    return best_a


def _subgradient_optimize(a, deadline, max_iter=50000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        tol = m_val * 1e-6 + 1e-14
        active = np.where(c >= m_val - tol)[0]
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        grad = _subgrad_fft(a, w, N)
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = _project_simplex(a_new)

        score = _score_normalized(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.0005, lr * 0.95)
                momentum = min(0.95, momentum * 1.005)
            if no_imp > max(3000, 6 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                momentum = 0.9
                no_imp = 0

    return best_a


def _make_conv_matrix_reduced(a, M, top_k=200):
    n = len(a)
    c_aa = _fft_convolve(a)
    threshold = M * 0.95
    active = set(np.where(c_aa >= threshold)[0].tolist())
    remaining = top_k - len(active)
    if remaining > 0:
        idx = np.argsort(c_aa)[::-1][:remaining]
        active.update(idx.tolist())
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    return A, b


def _lp_spread(a, M, top_k=200):
    n = len(a)
    c_obj = -np.ones(n)
    if n > 128:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, n_evals=30):
    n = len(a)
    best_t = 0.0
    best_score = _score_normalized(a)
    ts = np.linspace(0.0, 1.0, n_evals)
    for t in ts:
        a_new = np.maximum((1.0 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    # Refine
    if best_t > 0:
        lo = max(0, best_t - 1.0 / n_evals)
        hi = min(1.0, best_t + 1.0 / n_evals)
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            a1 = np.maximum((1.0 - m1) * a + m1 * g, 0)
            a1 = a1 / a1.sum()
            a2 = np.maximum((1.0 - m2) * a + m2 * g, 0)
            a2 = a2 / a2.sum()
            s1 = _score_normalized(a1)
            s2 = _score_normalized(a2)
            if s1 < s2:
                hi = m2
            else:
                lo = m1
        best_t = (lo + hi) / 2.0
    return best_t

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):

    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))

def _make_bell(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - (2.0 * t - 1.0) ** 2) ** alpha
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            sc_lse = _score_normalized(a_lse)
            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse
            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: Subgradient optimization
    print("\n=== Phase 2: FFT subgradient ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 350:
            break
        a_in = best_per_length[n_t]
        sg_end = min(time.monotonic() + 200.0, deadline - 350.0)
        if time.monotonic() > sg_end - 10:
            continue
        a_sg = _subgradient_optimize(a_in, sg_end, max_iter=30000)
        sc_sg = _score_normalized(a_sg)
        if np.isfinite(sc_sg) and sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
        best_per_length[n_t] = a_sg
        print(f"  Subgradient n={n_t}: score={sc_sg:.10f}")

    # Phase 3: LP refinement
    print("\n=== Phase 3: LP refinement ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 200:
            break
        a_in = best_per_length[n_t]
        lp_end = min(time.monotonic() + 200.0, deadline - 200.0)
        if time.monotonic() > lp_end - 10:
            continue
        a_lp = _lp_refine(a_in, lp_end, max_iter=2000, inflate=1.0003, top_k=150)
        sc_lp = _score_normalized(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP n={n_t}: score={sc_lp:.10f}")

    # Phase 4: Local search (pairwise mass transfer)
    print("\n=== Phase 4: Local search ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        ls_end = min(time.monotonic() + 250.0, deadline - 60.0)
        if time.monotonic() < ls_end - 10:
            a_ls = _local_search(best_seq, ls_end, max_iters=300000)
            sc_ls = _score_normalized(a_ls)
            if np.isfinite(sc_ls) and sc_ls < best_score:
                best_score = sc_ls
                best_seq = a_ls.copy()
            print(f"  Local search: score={sc_ls:.10f}")

    # Phase 5: Final LP + subgradient cycle
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\n=== Phase 5: Final polish ===")
        fp_end = min(time.monotonic() + 80.0, deadline - 30.0)
        if time.monotonic() < fp_end - 5:
            a_fp = _lp_refine(best_seq, fp_end, max_iter=1000, inflate=1.0002, top_k=100)
            sc_fp = _score_normalized(a_fp)
            if np.isfinite(sc_fp) and sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
            print(f"  Final LP: score={sc_fp:.10f}")

    # Phase 6: One more local search pass
    if best_seq is not None and time.monotonic() < deadline - 15:
        ls2_end = min(time.monotonic() + 40.0, deadline - 15.0)
        if time.monotonic() < ls2_end - 5:
            a_ls2 = _local_search(best_seq, ls2_end, max_iters=100000)
            sc_ls2 = _score_normalized(a_ls2)
            if np.isfinite(sc_ls2) and sc_ls2 < best_score:
                best_score = sc_ls2
                best_seq = a_ls2.copy()
            print(f"  Final local: score={sc_ls2:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score_normalized(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
