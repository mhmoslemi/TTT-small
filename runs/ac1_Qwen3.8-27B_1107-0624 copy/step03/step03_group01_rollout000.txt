"""
LP-guided iterative refinement with L-BFGS continuation for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses linear programming to find improvement
directions (from the state-of-art approach), combined with smooth gradient
optimization for fine polishing. Multi-phase strategy with adaptive continuation.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

try:
    import cvxpy as cp

except ImportError:

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a) / (s * s)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-14)
    return p / p.sum()


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        a_idx = k - np.arange(j_start, j_end)
        A[k, j_start:j_end] = a[a_idx]
    return A


def _solve_lp_improve(a, deadline):
    n = len(a)
    M = float(_full_convolve(a, a).max())
    if M <= 0:
        return a.copy()
    A = _build_conv_matrix(a)
    b_opt = a.copy()
    best_sum_b = 1.0

    for solver_try in range(2):
        try:
            if HAS_CVXPY:
                b_var = cp.Variable(n)
                prob = cp.Problem(
                    cp.Maximize(cp.sum(b_var)),
                    [A @ b_var <= M, b_var >= 0]
                )
                solver = cp.ECOS
                try:
                    prob.solve(solver=solver, verbose=False)
                except Exception:
                    prob.solve(solver=cp.SCS, verbose=False)
                if b_var.value is not None and np.all(np.isfinite(b_var.value)):
                    b_opt = np.maximum(b_var.value, 0.0)
                    best_sum_b = float(b_opt.sum())
            else:
                from scipy.optimize import linprog
                c_neg = -np.ones(n)
                A_ub = A
                b_ub = np.full(2 * n - 1, M)
                bounds = [(0, None)] * n
                res = linprog(c_neg, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
                if res.success and res.x is not None:
                    b_opt = np.maximum(res.x, 0.0)
                    best_sum_b = float(b_opt.sum())
            break
        except Exception as e:
            if solver_try < 1:
                continue
            return a.copy()

    if best_sum_b <= 1.001 or not np.isfinite(best_sum_b):
        return a.copy()

    b_norm = b_opt / best_sum_b
    return b_norm


def _lp_refinement(a_init, deadline, max_iters=40):
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    c = _full_convolve(a, a)
    current_max = float(c.max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        start_t = time.monotonic()
        b_new = _solve_lp_improve(a, deadline)

        if np.array_equal(b_new, a):
            break

        b_norm = b_new / b_new.sum()

        # Line search: try several mixing ratios
        best_t = 0.5
        best_sc = best_score
        for t_val in [1.0, 0.7, 0.5, 0.3, 0.15, 0.1]:
            a_try = (1.0 - t_val) * a + t_val * b_norm
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val

        if best_t == 0.5 and best_score <= best_sc:
            # No improvement found
            break

        a = (1.0 - best_t) * a + best_t * b_norm
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        elapsed = time.monotonic() - start_t
        if it % 5 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t={elapsed:.1f}s")

    return best_a, best_score


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    # Clamp for numerical stability
    c_shifted = np.clip(c_shifted, -50, 50)
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    c_shifted = np.clip(c_shifted, -50, 50)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_continuation(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max_wall_time / len(betas)
    else:
        per_level = 60.0

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 30, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)) and a_new.sum() > 1e-10:
                    a_new = a_new / a_new.sum()
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, beta_frac=0.008, n_max_iter=40000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        c_shifted = np.clip(c_shifted, -50, 50)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            beta = max(1e-14, beta * 0.8)

    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_sweeps=10):
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    n = len(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize order
        order = np.random.default_rng(42 + sweep).permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            # Try perturbed values
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                a[i] = old_val * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = old_val  # revert if didn't improve in place

        if not improved and sweep > 2:
            break

    return best_a, best_score
