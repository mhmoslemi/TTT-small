import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _lp_solve(a, M):
    """Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns optimal b and dual variables.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        b = np.maximum(res.x, 0.0)
        # Extract dual variables
        y = np.zeros(2 * n - 1)
        if hasattr(res, 'ineqlin_duals') and res.ineqlin_duals is not None:
            y = np.maximum(-np.asarray(res.ineqlin_duals).ravel(), 0.0)
        return b, y
    return None, None


def _lp_refine_with_linesearch(a, deadline, max_iters=50):
    """LP refinement with line search on step size."""
    n = len(a)
    current_best = a.copy()
    current_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        b, y = _lp_solve(a, M)
        if b is None:
            break
        
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        
        b_norm = b / s_b
        
        # Line search: try multiple step sizes
        best_t = 0.3
        best_sc = current_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_sc < current_score - 1e-14:
            a = (1.0 - best_t) * a + best_t * b_norm
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            current_score = best_sc
            current_best = a.copy()
        else:
            break
    
    return current_best, current_score

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_init_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _grad_smoothmax(w, p)
        
        # Project gradient to tangent space of simplex
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        # Adaptive beta based on gap
        if t_adam % 20 == 0:
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                target_beta = max(1e-5, gap * 0.5)
                beta = max(1e-5, min(beta * beta_decay, target_beta))
            else:
                beta = max(1e-5, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > no_imp_limit:
            # Structured perturbation based on binding constraints
            _, w_bind = _smooth_max_and_weights(c, max(1e-6, beta * 0.1))
            # Find positions where (a*a) is near max
            threshold = c.max() * 0.99
            binding = c > threshold
            
            # Perturb in a way that targets binding positions
            ns = 0.05 + 0.05 * (no_imp // 100)
            z = best_z + ns * rng.normal(size=n)
            
            # Extra perturbation at positions related to binding constraints
            if binding.any():
                bind_idx = np.where(binding)[0]
                # Perturb positions that contribute to binding convolutions
                for bi in bind_idx[:10]:  # Limit to top 10 binding positions
                    j_start = max(0, bi - n + 1)
                    j_end = min(n - 1, bi)
                    if j_start <= j_end:
                        z[j_start:j_end+1] += 0.1 * rng.normal(size=j_end-j_start+1)
            
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score, best_z


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _make_profile(name, n):
    """Create various initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    if name == 'uniform':
        a = np.ones(n)
    elif name == 'lin15':
        a = 1.0 + 1.5 * (t - 0.5)
    elif name == 'lin20':
        a = 1.0 + 2.0 * (t - 0.5)
    elif name == 'cubic':
        u = 2.0 * t - 1.0
        a = 1.0 + 1.5 * u - 1.0 * u**3
    elif name == 'cubic2':
        u = 2.0 * t - 1.0
        a = 1.0 + 2.0 * u - 1.5 * u**3
    elif name == 'exp05':
        a = np.exp(-0.5 * t)
    elif name == 'exp10':
        a = np.exp(-1.0 * t)
    elif name == 'exp20':
        a = np.exp(-2.0 * t)
    elif name == 'twobump':
        a = np.exp(-0.5 * ((t - 0.3) / 0.15)**2) + np.exp(-0.5 * ((t - 0.7) / 0.15)**2)
    elif name == 'cos1':
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * t)
    elif name == 'cos2':
        a = 1.0 + 0.6 * np.cos(4.0 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
