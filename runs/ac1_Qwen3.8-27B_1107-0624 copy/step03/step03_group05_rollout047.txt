def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    n = len(a)
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:2*n-1]
    return np.maximum(c, 0)

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0*n*c.max()/(s*s)

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u*np.arange(1,n+1) > (css-1.0))[0]

    w = np.maximum(v-theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n)/n
    return w/s

And I compute dc[k]/da[i] = 2*a[k-i] when both i and k-i are valid indices (in [0,n-1]), with the caveat about edge effects.

Actually, you know what, let me just verify the FFT formula numerically in the code and use whichever is correct.

The formula I've seen in practice (and which the previous code uses) is:
N = next_pow2(2*n-1)
fw = rfft(w, N)
fa_rev = rfft(a[::-1], N)
h_full = irfft(fw * fa_rev, N)
h = h_full[n-1 : 2*n-1]
grad = 2 * h

Let me verify this with my example.
a = [1, 2, 3], n=3, a[::-1] = [3, 2, 1]
w = [0.5, 0.3, 0.2, 0.3, 0.7] (length 5 = 2n-1)

conv(w, a[::-1]) we computed above: [1.5, 1.9, 1.7, 1.6, 2.9, 1.7, 0.7]
Taking [n-1:2n-1] = [2:5] = [1.7, 1.6, 2.9]
grad = 2 * [1.7, 1.6, 2.9] = [3.4, 3.2, 5.8]

dc[1]/da[0] = 2*a[1] = 4 (from: dc[1]/da[i] = 2*a[1-i] for i in {0,1})
