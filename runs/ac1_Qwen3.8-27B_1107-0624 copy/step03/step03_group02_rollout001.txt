"""
Improved optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive log-sum-exp smooth-max with Adam in log-space, LP spreading,
multi-resolution search, and careful polishing from the parent sequence.

LP calls, multi-resolution pipeline, and finer final polishing.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _adam_smooth_max(a, deadline, max_iter=30000, lr=0.025, beta_init=0.08,
                     beta_min=0.003, beta_decay=0.97, warmup=500,
                     restart_interval=3000, restart_perturb=0.25,

    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = beta_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        # Learning rate schedule: warmup then decay

            lr_cur = lr * (adam_t / warmup)
        else:

            lr_cur = lr * (1.0 - 0.5 * progress)

        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        e_sum = e.sum()
        if e_sum < 1e-30 or not np.isfinite(e_sum):
            z = best_z.copy()
            continue
        a_new = e / e_sum

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        z = z_new
        a = a_new

        # Restart mechanism

            z = best_z + restart_perturb * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(beta_min * 2, beta)
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp = 0

        if verbose and it % 5000 == 0:
            print(f"      Adam it={it}: best={best_score:.10f}, beta={beta:.4f}, no_imp={no_imp}")

def _subgradient_polyak(a, deadline, max_iter=20000, verbose=False):
    """Subgradient method with Polyak step size targeting max coefficient."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    # Polyak step: need a lower bound on optimal value
    # For uniform: score = 2, so max(c) = 1/n. Our target is ~0.75/n
    c_opt_lower = 0.7 / n  # conservative lower bound for max(c)
    c_current = float(c.max())

    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))

        sg = _subgradient_at_max(a, max_idx)
        sg_norm_sq = float(np.dot(sg, sg))
        if sg_norm_sq < 1e-30:
            break

        # Polyak step: minimize t*(max(c) - c_lower) / ||sg||^2
        gap = max_val - c_opt_lower
        if gap < 1e-15:
            break
        lr = gap / sg_norm_sq
        lr = min(lr, 0.5 / np.sqrt(it + 1))  # cap step size

        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5000:
                break

        a = a_new

        if verbose and it % 5000 == 0:
            print(f"      PolyakSG it={it}: best={best_score:.10f}")

    return best_a, best_score


def _lp_spread(a, M):

    n = len(a)
    if n > 2048:
        return None  # Too large for LP
    c_obj = -np.ones(n)
    # Build constraint matrix: (a * b)(k) <= M for all k
    # (a*b)(k) = sum_j a(j) * b(k-j) for valid j
    # This is a (2n-1) x n matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b_vec = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 10, 'dual_feasibility_tolerance': 1e-7})
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.002, blend=0.25,
               verbose=False):
    """LP-based refinement to flatten autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        g = _lp_spread(a, M_lp)

        if g is None:
            if verbose:
                print(f"      LP failed at iter {it}")
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
                blend_cur = min(0.5, blend_cur * 1.1)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            if verbose and it % 100 == 0:
                print(f"      LP it={it}: score={best_score:.10f}")
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = max(1.0005, inflate_cur * 0.95)
            if no_imp > 200:
                a = best_a.copy()
                no_imp = 0
                blend_cur = blend
                inflate_cur = inflate

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _fine_polish(a, deadline, max_iter=10000):
    """Very fine polishing with tiny Adam steps and subgradient."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    # Phase A: Very low lr Adam
    d1 = min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    a1, s1 = _adam_smooth_max(a, d1, max_iter=max_iter // 2, lr=0.005,
                              beta_init=0.01, beta_min=0.002, beta_decay=0.99,
                              warmup=100, restart_interval=2000, restart_perturb=0.1)
    if s1 < best_score:
        best_score = s1
        best_a = a1

    # Phase B: Subgradient with tiny steps
    if time.monotonic() < deadline - 3:
        a2, s2 = _subgradient_polyak(best_a, deadline - 3, max_iter=max_iter // 2)
        if s2 < best_score:
            best_score = s2
            best_a = a2

    return best_a, best_score


def _multi_peak_subgradient(a, deadline, max_iter=10000, top_k=5):
    """Subgradient targeting top-k peaks simultaneously with annealed weights."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Annealed weights: more focus on top peak as iterations progress
        progress = min(1.0, it / max_iter)
        focus = 1.0 + 3.0 * progress  # increases focus on top peak
        weights = np.array([focus ** i for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: parent interpolation -> Adam smooth-max at multiple
    resolutions -> LP spreading -> fine polishing. Uses adaptive temperature,
    Polyak subgradient, and careful time management.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Get parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception as e:
        print(f"Parent not available: {e}")

    if parent is None:
        # Fallback initializations
        n_fallback = 512
        parent = np.ones(n_fallback) / n_fallback
        # Add a linear tilt
        t = np.linspace(0, 1, n_fallback)
        parent = (1.0 + 1.5 * (t - 0.5))
        parent = np.maximum(parent, 1e-6)
        parent = parent / parent.sum()
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"Fallback init: score={best_score:.10f}")

    # --- Phase 1: Optimize at n=512 (quick warm-up) ---
    n1 = 512
    p1 = _interpolate_profile(parent, n1)
    print(f"\n=== Phase 1: n={n1} ===")
    sc0 = _score(p1)
    print(f"  Initial: {sc0:.10f}")

    d_p1 = min(deadline - 600, t_start + 300)
    a_p1, s_p1 = _adam_smooth_max(p1, d_p1, max_iter=20000, lr=0.03,
                                  beta_init=0.1, beta_min=0.005, beta_decay=0.97,
                                  warmup=300, restart_interval=2500)
    print(f"  After Adam: {s_p1:.10f}")
    if s_p1 < best_score:
        best_score = s_p1
        best_seq = a_p1.copy()

    # LP on n=512 result
    if time.monotonic() < deadline - 500:
        d_lp1 = min(deadline - 500, time.monotonic() + 120)
        a_lp1 = _lp_refine(a_p1, d_lp1, max_iter=1500, inflate=1.002, blend=0.3)
        s_lp1 = _score(a_lp1)
        print(f"  After LP: {s_lp1:.10f}")
        if s_lp1 < best_score:
            best_score = s_lp1
            best_seq = a_lp1.copy()

    # --- Phase 2: Interpolate to n=1024 and optimize ---
    n2 = 1024
    if time.monotonic() < deadline - 350:
        p2 = _interpolate_profile(best_seq, n2)
        print(f"\n=== Phase 2: n={n2} ===")
        sc2 = _score(p2)
        print(f"  Interpolated: {sc2:.10f}")

        if sc2 < best_score:
            best_score = sc2
            best_seq = p2.copy()

        # Adam on n=1024
        d_p2a = min(deadline - 250, time.monotonic() + 350)
        a_p2, s_p2 = _adam_smooth_max(p2, d_p2a, max_iter=25000, lr=0.02,
                                      beta_init=0.06, beta_min=0.003, beta_decay=0.975,
                                      warmup=500, restart_interval=3000, restart_perturb=0.2)
        print(f"  After Adam: {s_p2:.10f}")
        if s_p2 < best_score:
            best_score = s_p2
            best_seq = a_p2.copy()

        # Multi-peak subgradient
        if time.monotonic() < deadline - 150:
            d_p2b = min(deadline - 150, time.monotonic() + 120)
            a_p2b, s_p2b = _multi_peak_subgradient(best_seq, d_p2b, max_iter=8000, top_k=5)
            print(f"  After MultiSG: {s_p2b:.10f}")
            if s_p2b < best_score:
                best_score = s_p2b
                best_seq = a_p2b.copy()

        # LP refinement on n=1024
        if time.monotonic() < deadline - 100:
            d_p2c = min(deadline - 100, time.monotonic() + 150)
            a_p2c = _lp_refine(best_seq, d_p2c, max_iter=2000, inflate=1.0015, blend=0.25)
            s_p2c = _score(a_p2c)
            print(f"  After LP: {s_p2c:.10f}")
            if s_p2c < best_score:
                best_score = s_p2c
                best_seq = a_p2c.copy()

    # --- Phase 3: Try n=1280 if time permits ---
    if time.monotonic() < deadline - 200:
        n3 = 1280
        print(f"\n=== Phase 3: n={n3} ===")
        p3 = _interpolate_profile(best_seq, n3)
        sc3 = _score(p3)
        print(f"  Interpolated: {sc3:.10f}")

        d_p3 = min(deadline - 80, time.monotonic() + 150)
        a_p3, s_p3 = _adam_smooth_max(p3, d_p3, max_iter=15000, lr=0.015,
                                      beta_init=0.04, beta_min=0.003, beta_decay=0.98,
                                      warmup=300, restart_interval=2500)
        print(f"  After Adam: {s_p3:.10f}")
        if s_p3 < best_score:
            best_score = s_p3
            best_seq = a_p3.copy()

    # --- Phase 4: Fine polishing on best ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Fine polishing (current best={best_score:.10f}, n={best_seq.size}) ===")
        d_polish = deadline - 5
        a_pol, s_pol = _fine_polish(best_seq, d_polish, max_iter=15000)
        print(f"  After polish: {s_pol:.10f}")
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol.copy()

    # --- Phase 5: Try a few random perturbations of the best ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_best = best_seq.size
        for trial in range(3):
            if time.monotonic() > deadline - 5:
                break
            perturb_scale = 0.05 * (trial + 1)
            a_pert = best_seq * (1.0 + perturb_scale * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert
            d_trial = min(deadline - 3, time.monotonic() + 15)
            a_tr, s_tr = _adam_smooth_max(a_pert, d_trial, max_iter=3000, lr=0.01,
                                          beta_init=0.02, beta_min=0.002, beta_decay=0.98,
                                          warmup=100, restart_interval=2000)
            if s_tr < best_score:
                best_score = s_tr
                best_seq = a_tr.copy()
                print(f"  Perturb {trial}: {s_tr:.10f} (improved!)")
            else:
                print(f"  Perturb {trial}: {s_tr:.10f} (no improvement)")

    # --- Final: also try parent at native length with quick polish ---
    if parent is not None and time.monotonic() < deadline - 3:
        p_native = parent.copy()
        sc_native = _score(p_native)
        if sc_native < best_score:
            best_score = sc_native
            best_seq = p_native.copy()
            print(f"  Parent native: {sc_native:.10f} (new best)")

    # --- Final validation and formatting ---
    if best_seq is None:
        best_seq = np.ones(256) / 256

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final < 0.01:
        final_seq = final_seq * (1.0 / s_final)
        s_final = 1.0

    # Verify with evaluate_sequence if possible
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
        else:
            print(f"\nevaluate_sequence returned inf, using internal score")
    except Exception as e:
        print(f"eval error: {e}")

    print(f"\nFINAL: internal_score={best_score:.10f}, n={final_seq.size}, sum={s_final:.6f}")
    return [float(x) for x in final_seq]
