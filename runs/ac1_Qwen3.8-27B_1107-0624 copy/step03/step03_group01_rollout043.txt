Coordinate descent + L-BFGS hybrid for minimizing 2*n*max(conv(a,a))/sum(a)^2.

# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def score_norm_fn(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def make_profile(n, type='linear', params=None):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if type == 'linear':
        slope = params if params is not None else 1.5
        a = 1.0 + slope * (t - 0.5)
    elif type == 'quadratic':
        a1, a2 = (params if params is not None else (1.0, 0.3))
        a = 1.0 + a1 * u + a2 * u**2
    elif type == 'cubic':
        a1, a3 = (params if params is not None else (1.0, -0.5))
        a = 1.0 + a1 * u + a3 * u**3
    elif type == 'sinusoidal':
        freq, amp = (params if params is not None else (1.0, 0.3))
        a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
    else:

    a = np.maximum(a, 1e-10)
    return a / a.sum()

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def lbfgs_objective(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def lbfgs_gradient(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

                    best_a = a_new.copy()
                    z = result.x.copy()

def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time=60.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.10, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004]
    betas = [b * max_c0 for b in beta_starts]
    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 25, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _coordinate_descent(a_init, max_cycles=500, deadline=None, verbose=True, perturb_every=200):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = _full_convolve(a, a)
    conv_len = 2 * n - 1
    best_conv_max = float(conv.max())
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for cycle in range(max_cycles):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Periodic perturbation to escape local minima
        if perturb_every > 0 and cycle > 0 and cycle % perturb_every == 0:
            noise = 0.02 * rng.normal(size=n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
                cp = _full_convolve(a_pert, a_pert)
                if cp.max() < best_conv_max * 1.05:
                    a = a_pert
                    conv = cp

        for m in range(n):
            am = a[m]
            if am < 1e-14:
                # Try a small positive value
                x_opt = 1e-8 / n
            else:
                lo = m
                hi = min(m + n, conv_len)
                if hi <= lo:
                    x_opt = am
                    continue

                sub_len = hi - lo
                sub_a = a[:sub_len].copy()

                # Base values (conv with a[m] contribution removed)
                base = conv[lo:hi].copy() - 2.0 * am * sub_a

                # Quadratic term at k=2m
                quad_local = 2 * m - lo
                has_quad = 0 <= quad_local < sub_len
                if has_quad:
                    base[quad_local] = conv[2 * m] - am * am

                # Slopes: 2*a[k-m] for k != 2m, 0 for k=2m
                slope = 2.0 * sub_a.copy()
                if has_quad:
                    slope[quad_local] = 0.0

                quad_base = base[quad_local] if has_quad else None

                # Evaluate f(0)
                f0_vals = base.copy()
                if has_quad:
                    f0_vals[quad_local] = quad_base
                f0 = float(f0_vals.max())

                # Check subgradient at 0
                active = f0_vals >= f0 - 1e-14
                if active.any():
                    min_slope = float(slope[active].min())
                    if min_slope >= -1e-14:
                        x_opt = 0.0

                        for _ in range(45):

                            m2 = b_hi - (b_hi - a_lo) / 3.0
                            f1 = float((base + slope * m1).max())
                            if has_quad:
                                f1 = max(f1, quad_base + m1 * m1)
                            f2 = float((base + slope * m2).max())
                            if has_quad:
                                f2 = max(f2, quad_base + m2 * m2)
                            if f1 < f2:
                                b_hi = m2
                            else:
                                a_lo = m1
                        x_opt = (a_lo + b_hi) / 2.0
                else:
                    x_opt = am

            # Update a[m] and conv
            delta = x_opt - am
            if abs(delta) > 1e-16:
                a[m] = x_opt
                lo = m
                hi = min(m + n, conv_len)
                sub_a = a[:hi - lo].copy()
                sub_a[m % (hi - lo) if m < hi - lo else 0] = 0  # avoid using new a[m]
                # Actually sub_a should be the OLD a[0:sub_len], but a[m] is already updated
                # Let me fix: use the original sub_a computed before the update
                # Actually I computed sub_a before the update. Let me restructure.
                
                # The issue: I already updated a[m], so sub_a = a[:sub_len] now has the new value
                # I need the old sub_a. Let me handle this differently.
                # Actually, looking back at the code, sub_a was computed BEFORE updating a[m].
                # But I'm using it AFTER the update. Let me fix by saving it.
                
                # Hmm, this is a bug. Let me restructure the loop.
                pass

        # This approach has a bug with the sub_a. Let me rewrite the coordinate descent
        # more carefully.
        
        # Re-normalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
            conv = _full_convolve(a, a)

        conv_max = float(conv.max())
        if conv_max < best_conv_max:
            best_conv_max = conv_max
            best_a = a.copy()

        if verbose and cycle % 100 == 0:
            print(f"    CD cycle {cycle}: conv_max={conv_max:.12f}, score={2*n*conv_max:.10f}")

    return best_a, 2.0 * n * best_conv_max


def _coord_descent_v2(a_init, max_cycles=500, deadline=None, verbose=True):
    """Coordinate descent with correct conv updates."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = _full_convolve(a, a)
    conv_len = 2 * n - 1
    best_conv_max = float(conv.max())
    best_a = a.copy()

    for cycle in range(max_cycles):
        if deadline is not None and time.monotonic() > deadline:
            break

        for m in range(n):
            am = a[m]

            # Determine affected range: k in [m, min(m+n-1, 2n-2)]
            lo = m
            hi = min(m + n, conv_len)
            sub_len = hi - lo
            if sub_len <= 0:
                continue

            # Save the relevant portion of a BEFORE updating a[m]
            # sub_a[j] = a[j] for j=0..sub_len-1, but we need a[k-m] for k in [lo,hi)
            # which is a[k-m] where k-m ranges from 0 to sub_len-1
            # So sub_a = a[0:sub_len]
            sub_a = a[0:sub_len].copy()

            # base[k] = conv[k] - contribution of a[m]
            # For k in [lo, hi): contribution = 2*am*a[k-m] except at k=2m where it's am^2
            base = conv[lo:hi].copy() - 2.0 * am * sub_a

            quad_local = 2 * m - lo
            has_quad = (0 <= quad_local < sub_len)
            if has_quad:
                base[quad_local] = conv[2 * m] - am * am

            slope = 2.0 * sub_a.copy()
            if has_quad:
                slope[quad_local] = 0.0
            quad_base = base[quad_local] if has_quad else None

            # 1D minimization
            f0_vals = base.copy()
            if has_quad:
                f0_vals[quad_local] = quad_base
            f0 = float(f0_vals.max())

            active = f0_vals >= f0 - 1e-14
            x_opt = am  # default: keep same

            if active.any():
                min_slope = float(np.min(slope[active]))
                if min_slope >= -1e-14:
                    x_opt = 0.0
                else:
                    x_max = max(2.0 * am + 1.0 / n, 20.0 / n)
                    x_max = min(x_max, 1.0)

                    a_lo = 0.0
                    b_hi = x_max
                    for _ in range(40):
                        m1 = a_lo + (b_hi - a_lo) / 3.0
                        m2 = b_hi - (b_hi - a_lo) / 3.0
                        f1 = float(np.max(base + slope * m1))
                        if has_quad:
                            f1 = max(f1, quad_base + m1 * m1)
                        f2 = float(np.max(base + slope * m2))
                        if has_quad:
                            f2 = max(f2, quad_base + m2 * m2)
                        if f1 < f2:
                            b_hi = m2
                        else:
                            a_lo = m1
                    x_opt = (a_lo + b_hi) / 2.0

            # Update conv using OLD sub_a
            delta = x_opt - am
            if abs(delta) > 1e-16:
                conv[lo:hi] += 2.0 * delta * sub_a
                if has_quad:
                    # Extra term: delta^2 (since the quadratic gives delta*(x+am) not delta*2*am)
                    conv[2 * m] += delta * delta
                a[m] = x_opt

        # Re-normalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
            conv = _full_convolve(a, a)
        else:
            a = np.ones(n) / n
            conv = _full_convolve(a, a)

        conv_max = float(conv.max())
        if conv_max < best_conv_max:
            best_conv_max = conv_max
            best_a = a.copy()

        if verbose and cycle % 100 == 0:
            elapsed = time.monotonic()
            print(f"    CD2 cycle {cycle}: score={2*n*conv_max:.10f}")

    return best_a, 2.0 * n * best_conv_max


def _adam_polish(a_init, deadline, lr=0.03, beta_frac=0.005, n_max_iter=20000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    a_cur = a_init.copy()
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a_cur[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a_cur * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_cur = _softmax_fn(z)

        sc = _score_norm_fn(a_cur)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        if t % 300 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS (logsumexp) for global search,
    then coordinate descent for precise local minimization of max convolution peak.
    Explores multiple sequence lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score, best_n
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = len(best_seq)
            print(f"  *** NEW BEST: {best_score:.10f} (n={best_n}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # --- Phase 0: Initial candidates (60s) ---
    print(f"Phase 0: Generating initial candidates...")
    phase0_end = min(deadline, t_start + 60.0)

    candidates = []

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
        update_best(parent, sc_parent, "parent")
        print(f"  Parent: n={len(parent)}, score={sc_parent:.10f}")

    # Try various profiles at different lengths
    lengths = [512, 1024, 2048, 3072]
    for sn in lengths:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_profile(sn, 'linear', 1.0),
            _make_profile(sn, 'linear', 1.5),
            _make_profile(sn, 'linear', 2.0),
            _make_profile(sn, 'linear', 2.5),
            _make_profile(sn, 'quadratic', (1.0, 0.3)),
            _make_profile(sn, 'quadratic', (1.5, -0.2)),
            _make_profile(sn, 'cubic', (1.0, -0.5)),
            _make_profile(sn, 'cubic', (1.5, -0.8)),
            _make_profile(sn, 'asymmetric_ramp', None),
            _make_profile(sn, 'sqrt_ramp', None),
            _make_profile(sn, 'sinusoidal', (1.0, 0.2)),
        ]
        for idx, p in enumerate(profiles):
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"prof{idx}_n{sn}"))
            update_best(p, sc, f"prof{idx}_n{sn}")

    # Interpolate parent to other lengths
    if parent is not None:
        for sn in lengths:
            if sn != len(parent) and time.monotonic() < phase0_end:
                p_interp = _interp_profile(parent, sn)
                sc = _score_norm_fn(p_interp)
                candidates.append((sc, p_interp, sn, f"parent_interp_n{sn}"))
                update_best(p_interp, sc, f"parent_interp_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f} (n={best_n})")

    # --- Phase 1: L-BFGS on top candidates (400s) ---
    print(f"\nPhase 1: L-BFGS optimization...")
    phase1_end = min(deadline, t_start + 460.0)
    count = 0
    seen_keys = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end or count >= 8:
            break
        if sc > best_score * 1.08:
            break
        key = (pn, label[:10])
        if key in seen_keys:
            continue
        seen_keys.add(key)

        per_time = min(50.0, (phase1_end - time.monotonic()) / max(1, 8 - count))
        level_deadline = time.monotonic() + per_time
        p_best, sc_best = _run_lbfgs(p, level_deadline, max_wall_time=per_time)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f} (n={best_n})")

    # --- Phase 2: Coordinate descent polishing (700s) ---
    print(f"\nPhase 2: Coordinate descent...")
    phase2_end = min(deadline, t_start + 1160.0)

    if best_seq is not None:
        per_cd = min(120.0, (phase2_end - time.monotonic()) / 3.0)
        for i in range(3):
            if time.monotonic() > phase2_end:
                break
            dl = time.monotonic() + per_cd
            p_best, sc_best = _coord_descent_v2(best_seq, max_cycles=2000, deadline=dl, verbose=False)
            update_best(p_best, sc_best, f"cd2_pass{i}")

    # Also try coordinate descent on a different length if time allows
    if time.monotonic() < phase2_end - 60 and best_seq is not None:
        n_curr = best_n
        for target_n in [n_curr * 2] if n_curr * 2 <= 4096 else []:
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            dl = min(phase2_end - 10, time.monotonic() + 100.0)
            p_best, sc_best = _coord_descent_v2(p_interp, max_cycles=1500, deadline=dl, verbose=False)
            update_best(p_best, sc_best, f"cd2_n{target_n}")

    print(f"  Phase 2 done: best={best_score:.10f} (n={best_n})")

    # --- Phase 3: Adam polish + more CD (300s) ---
    print(f"\nPhase 3: Adam polish + fine tuning...")
    phase3_end = min(deadline, t_start + 1460.0)

    if best_seq is not None:
        dl = min(phase3_end - 30, time.monotonic() + 80.0)
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.02, beta_frac=0.003, n_max_iter=15000)
        update_best(p_best, sc_best, "adam_polish")

        # More coordinate descent rounds
        for i in range(4):
            if time.monotonic() > phase3_end - 10:
                break
            dl = time.monotonic() + min(50.0, phase3_end - time.monotonic() - 5)
            p_best, sc_best = _coord_descent_v2(best_seq, max_cycles=3000, deadline=dl, verbose=False)
            update_best(p_best, sc_best, f"cd_fine{i}")

    # Try a multi-restart approach: perturb best and re-optimize
    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        for i in range(3):
            if time.monotonic() > phase3_end - 10:
                break
            n_b = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_level = 0.05 + 0.03 * i
            z = z + noise_level * rng.normal(size=n_b)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase3_end - 10, time.monotonic() + 40.0)
            p_best, sc_best = _run_lbfgs(p_start, dl, max_wall_time=min(30.0, (phase3_end - time.monotonic())))
            if sc_best < best_score:
                update_best(p_best, sc_best, f"restart{i}")
                # Follow up with CD
                if time.monotonic() < phase3_end - 20:
                    dl2 = min(phase3_end - 10, time.monotonic() + 40.0)
                    p_best2, sc_best2 = _coord_descent_v2(p_best, max_cycles=1000, deadline=dl2, verbose=False)
                    update_best(p_best2, sc_best2, f"restart_cd{i}")

    print(f"  Phase 3 done: best={best_score:.10f} (n={best_n})")

    # --- Phase 4: Final fine polishing (remaining time) ---
    print(f"\nPhase 4: Final polishing...")
    if best_seq is not None:
        # Multiple short CD passes
        remaining = deadline - time.monotonic()
        n_passes = max(1, int(remaining / 40))
        for i in range(min(n_passes, 15)):
            if time.monotonic() > deadline - 5:
                break
            dl = time.monotonic() + min(35.0, deadline - time.monotonic() - 3)
            p_best, sc_best = _coord_descent_v2(best_seq, max_cycles=5000, deadline=dl, verbose=False)
            update_best(p_best, sc_best, f"final_cd{i}")

        # Very fine Adam
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 5, time.monotonic() + 20.0)
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.005, beta_frac=0.001, n_max_iter=5000)
            update_best(p_best, sc_best, "final_adam")

    print(f"  Phase 4 done: best={best_score:.10f} (n={best_n})")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(512, 'linear', 1.5)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(512, 'linear', 1.5)
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
