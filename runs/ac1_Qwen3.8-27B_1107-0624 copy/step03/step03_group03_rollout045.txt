import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return _score_normalized(a / s)


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / a.size
    return a / s


def _gradient_lse(a, T):
    """Compute gradient of LSE-smoothed objective 2n*sum(exp(c/T))/(sum a)^2.
    Returns (gradient, max_val)."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient of max(c) ≈ sum(w * dc/da)
    # dc[k]/da[i] = a[k-i] + a[i-k] (for valid indices)
    # So grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # This is 2 * conv(w, a)[n-1+n-1-i]... let me think.

    # Let's use: sum_k w[k] * a[k-i] = (w * a_reversed)[i] where * is correlation.
    # In FFT: this is ifft(fft(w) * fft(a_reversed)) ...

    grad = 2.0 * h  # factor of 2 because dc/dai = a[k-i] + a[k-i] by symmetry
    
    return grad, m_val, w
