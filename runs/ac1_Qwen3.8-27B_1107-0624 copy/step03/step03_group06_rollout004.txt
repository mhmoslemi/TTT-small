Improved LP + Gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 200:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()


def _make_ramp2(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 1e-6
    return a / a.sum()


def _make_lin(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_quad(n, curvature=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + curvature * t + curvature * t * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_raised_cos(n, harmonic=1):

    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * harmonic * t)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_gaussian(n, sigma=0.3):

    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    return a / a.sum()


def _make_bell(n, width=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + ((t - 0.5) / width) ** 4)
    return a / a.sum()


def _build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_maxsum(f, M, n):
    """Solve: max sum(g) s.t. f*g <= M, g >= 0."""
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _solve_lp_minpeak(f, n):
    """Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    A_conv = _build_conv_matrix(f, n)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Constraints: (f*g)[k] - t <= 0  =>  [A_conv[k,:] | -1] @ x <= 0
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(g) = 1
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        t_opt = res.x[n]
        return g, t_opt
    return None, None

    # Coarse grid
    ts = np.linspace(0.03, 1.0, n_coarse)
    coarse_best_t = 0.0
    coarse_best_sc = best_sc
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < coarse_best_sc:
            coarse_best_sc = sc
            coarse_best_t = t

    if coarse_best_t <= 0:
        return 0.0, best_sc

    # Golden section refinement around coarse best
    lo = max(0.0, coarse_best_t - 1.0 / n_coarse)
    hi = min(1.0, coarse_best_t + 1.0 / n_coarse)
    if hi - lo < 1e-6:
        lo, hi = max(0.0, coarse_best_t - 0.05), min(1.0, coarse_best_t + 0.05)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_val = hi - gr * (hi - lo)
    d_val = lo + gr * (hi - lo)

    for _ in range(n_fine):
        if time.monotonic() > _ls_deadline[0]:
            break
        fc = _eval_line(a, g, c_val)
        fd = _eval_line(a, g, d_val)
        if fc < fd:
            hi = d_val
            d_val = c_val
            fd = fc
            c_val = hi - gr * (hi - lo)
        else:
            lo = c_val
            c_val = d_val
            fc = fd
            d_val = lo + gr * (hi - lo)

    mid_t = (lo + hi) / 2.0
    mid_sc = _eval_line(a, g, mid_t)
    if mid_sc < coarse_best_sc:
        return mid_t, mid_sc
    return coarse_best_t, coarse_best_sc


_ls_deadline = [np.inf]


def _eval_line(a, g, t):
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.inf
    a_new = a_new / s
    return _score(a_new)


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.004, inflate_min=1.0003):
    """LP refinement using both max-sum and min-peak LPs."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Alternate between LP variants
        use_minpeak = (it % 3 == 2)

        if use_minpeak:
            g, t_opt = _solve_lp_minpeak(a, n)
            if g is None:
                if no_imp % 100 == 99:
                    inflate = min(0.03, inflate * 1.1)
                no_imp += 1
                if no_imp > 300:
                    break
                continue
            # Check if this direction actually helps
            # t_opt < M means the peak can be reduced
            if t_opt >= M * (1.0 - 1e-6):
                no_imp += 1
                if no_imp > 200:
                    break
                continue
        else:
            M_lp = M * inflate
            g = _solve_lp_maxsum(a, M_lp, n)
            if g is None:
                break
            sum_g = float(g.sum())
            if sum_g <= 1.0 + 1e-7:
                no_imp += 1
                if no_imp % 30 == 29:
                    inflate = min(0.03, inflate * 1.2)
                if no_imp > 300:
                    break
                continue
            g = g / sum_g

        # Line search
        _ls_deadline[0] = deadline - 0.5
        best_t, best_sc = _line_search_golden(a, g)

        if best_t > 0.01 and best_sc < best_score - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Gradually reduce inflation
            inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 40 == 39:
                inflate = min(0.03, inflate * 1.1)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate < 1.001:
                    break

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Adam-based gradient polish with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-7, 0.03 * float(c.max()))
    lr = 0.02
    base_lr = lr
    m = np.zeros(n)
    v_hat = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    no_imp = 0
    reinit_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_adam += 1
        c = _fft_convolve(a)
        maxc = float(c.max())
        beta = max(1e-8, 0.03 * maxc)

        grad, _ = _smooth_max_grad(a, beta)
        if not np.all(np.isfinite(grad)):
            m = np.zeros(n)
            v_hat = np.zeros(n)
            continue

        m = beta1 * m + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat_corrected = v_hat / (1 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat_corrected) + eps)

        a_new = a - step
        a = _project_simplex(a_new)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 299:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(5000, 5 * n):
                a = best_a.copy()
                m = np.zeros(n)
                v_hat = np.zeros(n)
                t_adam = 0
                lr = base_lr
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                no_imp = 0
                reinit_count += 1
                if reinit_count > 5:
                    break

        if it % 200 == 199:
            beta = max(1e-9, beta * 0.85)

    return best_a


def _coord_descent(a0, deadline, max_rounds=200):
    """Coordinate descent: for each coordinate, find optimal value given others fixed."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    no_imp = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        np.random.seed(rnd * 7919)
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try geometric grid of values
            candidates = np.concatenate([
                [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5],
                np.linspace(0.0, 1.0 / n, 5)
            ])
            candidates = np.maximum(candidates, 0.0)
            best_local = old_val
            best_local_sc = _score(a)
            for cv in candidates:
                a[i] = cv
                sc = _score(a)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local = cv
            a[i] = best_local
            if abs(best_local - old_val) > 1e-10:
                improved = True

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initializations, dual-LP refinement,
    Adam gradient polish, and coordinate descent for final improvement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate diverse initializations at multiple lengths
    configs = []
    lengths = [512, 640, 768, 896]

    for n_t in lengths:
        if parent is not None:
            configs.append((f'parent_{n_t}', n_t, _interpolate(parent, n_t)))
            # Symmetrized parent
            sym = (parent_interp := _interpolate(parent, n_t))
            sym = (sym + sym[::-1]) / 2.0
            sym = sym / sym.sum()
            configs.append((f'sym_parent_{n_t}', n_t, sym))
        configs.append((f'ramp1_{n_t}', n_t, _make_ramp(n_t, 1.0)))
        configs.append((f'ramp2_{n_t}', n_t, _make_ramp(n_t, 2.0)))
        configs.append((f'ramp3_{n_t}', n_t, _make_ramp(n_t, 3.0)))
        configs.append((f'lin2_{n_t}', n_t, _make_lin(n_t, 2.0)))
        configs.append((f'lin4_{n_t}', n_t, _make_lin(n_t, 4.0)))
        configs.append((f'quad3_{n_t}', n_t, _make_quad(n_t, 3.0)))
        configs.append((f'rcos_{n_t}', n_t, _make_raised_cos(n_t, 1)))
        configs.append((f'gauss_{n_t}', n_t, _make_gaussian(n_t, 0.35)))
        configs.append((f'bell_{n_t}', n_t, _make_bell(n_t, 0.45)))

    # Score and sort
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    # Print top initializations
    print("Top 10 initializations:")
    for sc, name, n_t, _ in scored_configs[:10]:
        print(f"  {name}: {sc:.8f}")

    # Phase 1: LP refinement on top initializations
    # Allocate time proportionally
    top_k = 6
    for idx in range(min(top_k, len(scored_configs))):
        if time.monotonic() > deadline - 200:
            break
        sc0, name, n_t, a0 = scored_configs[idx]

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time: more for better initializations
        remaining = deadline - time.monotonic() - 150
        if remaining < 20:
            break

        if idx == 0:
            lp_time = min(400.0, remaining * 0.4)
        elif idx == 1:
            lp_time = min(300.0, remaining * 0.3)
        else:
            lp_time = min(150.0, remaining * 0.15)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"\nPhase 1 [{idx}]: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=50000, inflate_start=1.004)
        sc_lp = _score(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  LP result: {sc_lp:.10f} (improved!)")
        else:
            print(f"  LP result: {sc_lp:.10f}")

    # Phase 2: Gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 100:
        gp_time = min(200.0, deadline - time.monotonic() - 80)
        if gp_time > 20:
            gp_end = time.monotonic() + gp_time
            print(f"\nPhase 2: Gradient polish, budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=50000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  Gradient result: {sc_gp:.10f}")

    # Phase 3: LP refinement again (tighter)
    if best_seq is not None and time.monotonic() < deadline - 80:
        lp_time = min(180.0, deadline - time.monotonic() - 60)
        if lp_time > 15:
            lp_end = time.monotonic() + lp_time
            print(f"\nPhase 3: Tight LP, budget={lp_time:.0f}s")
            a_lp2 = _lp_refine(best_seq, lp_end, max_iter=30000,
                               inflate_start=1.002, inflate_min=1.0002)
            sc_lp2 = _score(a_lp2)
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()
            print(f"  Tight LP result: {sc_lp2:.10f}")

    # Phase 4: Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 60:
        cd_time = min(120.0, deadline - time.monotonic() - 40)
        if cd_time > 15:
            cd_end = time.monotonic() + cd_time
            print(f"\nPhase 4: Coordinate descent, budget={cd_time:.0f}s")
            a_cd = _coord_descent(best_seq, cd_end, max_rounds=100)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  Coord descent result: {sc_cd:.10f}")

    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        gp_time = min(80.0, deadline - time.monotonic() - 15)
        if gp_time > 10:
            gp_end = time.monotonic() + gp_time
            print(f"\nPhase 5: Final gradient, budget={gp_time:.0f}s")
            a_g = _gradient_polish(best_seq, gp_end, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  Final gradient result: {sc_g:.10f}")

    # Phase 6: Try extending to longer sequence and re-optimize
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Try a longer version
        n_extend = min(1200, best_seq.size + 200)
        a_ext = _interpolate(best_seq, n_extend)
        sc_ext = _score(a_ext)
        if sc_ext < best_score:
            print(f"\nPhase 6: Extended to n={n_extend}, score={sc_ext:.10f}")
            # Quick LP on extended
            ext_time = min(60.0, deadline - time.monotonic() - 10)
            if ext_time > 10:
                ext_end = time.monotonic() + ext_time
                a_ext_lp = _lp_refine(a_ext, ext_end, max_iter=20000,
                                      inflate_start=1.003)
                sc_ext_lp = _score(a_ext_lp)
                if sc_ext_lp < best_score:
                    best_score = sc_ext_lp
                    best_seq = a_ext_lp.copy()
                print(f"  Extended LP result: {sc_ext_lp:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final normalization and validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
