If supp(a) = {0, 1, ..., n-1} (full support), then {k*-i : i ∈ supp(a)} = {k*-(n-1), ..., k*}, intersected with {0,...,n-1}.

For k* = n-1 (the center): {0, 1, ..., n-1}, so a[j] = γ for all j, meaning a is constant. This gives score = 2.

For other k* values (asymmetric peaks), the set {k*-i : i=0,...,n-1} ∩ {0,...,n-1} is a proper subset. For example, if k* = 0: {0, -1, -2, ..., -(n-1)} ∩ {0,...,n-1} = {0}. So only a[0] is constrained to be γ.

This suggests that at the optimum with a single active peak not at the center, the sequence has a special structure: it's constant on the "reflected support" and free elsewhere.

For k* near the center (say k* = n-1-δ for small δ), the reflected support is {δ, δ+1, ..., n-1} ∩ {0,...,n-1} = {δ, δ+1, ..., n-1}. So a[j] = γ for j = δ, δ+1, ..., n-1, and a is free for j = 0, 1, ..., δ-1.

This is a very specific structure! The sequence is constant on most of its length, with a "free" part at the beginning.

But wait, this analysis assumes a single active peak. In practice, there might be multiple active peaks (multiple positions where c[k] = t). If there are two or more active peaks, the condition becomes a weighted sum, which allows more flexibility.

"""
Focused refinement for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses KKT-informed profile design, aggressive soft-max gradient in logit space

import numpy as np
import time


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def compute_h(w, a):
    """h[i] = sum_k w[k]*a[k-i] for valid indices, via FFT cross-correlation."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fab = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fab * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval, w


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def kkt_profile(n, delta_frac=0.1, gamma=1.0):
    """KKT-inspired: constant over most of range, linear taper at start."""
    a = np.ones(n)
    n_delta = max(1, int(delta_frac * n))
    for i in range(n_delta):
        a[i] = gamma + (1.0 - gamma) * (i / max(1, n_delta - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def cosine_taper_profile(n, taper_frac=0.3):
    """Constant with cosine taper at one end."""
    a = np.ones(n)
    n_tap = max(1, int(taper_frac * n))
    for i in range(n_tap):
        t = i / max(1, n_tap)
        a[i] = 0.3 + 0.7 * 0.5 * (1 - np.cos(np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def power_taper_profile(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.2 + 0.8 * (1.0 - t) ** (1.0 / p)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def random_smooth_profile(n, rng, n_freq=4):
    t = np.linspace(0.0, 1.0, n)
    coeffs = rng.uniform(-0.3, 0.3, n_freq)
    a = np.ones(n)
    for i, cval in enumerate(coeffs, 1):
        a += cval * np.cos(2 * np.pi * i * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def logit_adam_optimize(a0, deadline, rng, lr=0.03, beta_frac=0.04,
                        beta_decay=0.92, beta_min=1e-10,
                        restart_period=800, momentum_boost=1.0):
    """
    Adam optimizer in logit space with soft-max gradient.
    Returns (best_a, best_score).
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = max(1e-6, beta_frac * max_c)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * momentum_boost * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
            if no_improve > 150:
                current_lr = max(0.002, current_lr * 0.98)

        if no_improve > restart_period:
            noise = 0.12 * (1.0 + 0.5 * min(no_improve / restart_period, 2.0))
            z = best_z + noise * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, beta_frac * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def peak_targeting_optimize(a0, deadline, rng, lr=0.02, n_peaks=3):
    """Optimize by targeting top peaks in convolution directly."""
    a0 = np.asarray(a0, dtype=np.float64).raval().copy() if False else np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = 2.0 * n * float(c.max())

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        top_k = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        for idx in top_k:
            w[idx] = 1.0 / n_peaks

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()

    return best_a, best_score


def direct_gradient_step(a0, lr=0.01, n_steps=100):
    """Simple projected gradient with exact max (no soft-max)."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = full_convolve_fft(a)
    best_score = 2.0 * n * float(c.max())

    for _ in range(n_steps):
        c = full_convolve_fft(a)
        max_c = float(c.max())
        peak_idx = int(np.argmax(c))

        # Exact gradient at the peak
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] += a[j]

        # Projected gradient step: a - lr*grad, then normalize
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = full_convolve_fft(a_new)
        max_c_new = float(c_new.max())
        score_new = 2.0 * n * max_c_new

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.8

    return best_a, best_score

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
