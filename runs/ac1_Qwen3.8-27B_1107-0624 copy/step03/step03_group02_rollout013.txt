"""
Optimized step-function search using FISTA accelerated smooth-max,
reduced-constraint LP refinement, and multi-length exploration.
Combines convex optimization insights with practical heuristics
to minimize 2*n*max(autocorr)/(sum)^2 for non-negative sequences.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _fista_smooth_max(a0, deadline, max_iter=30000, beta_factor=0.05, verbose=False):
    """FISTA (Nesterov accelerated gradient) on smooth max of autocorrelation, projected to simplex."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(beta_factor * float(c.max()), 1e-10)
    t = 1.0
    t_old = 1.0
    y = a.copy()
    no_imp = 0
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(y)
        w = _smooth_max_weights(c_curr, beta)
        grad = _grad_smooth_max(y, w)

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = (t - 1.0) / t_new
        y_new = a + mu * (y - a)

        a_new = y_new - lr * grad
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 3000:
            # Reset momentum
            beta = max(beta * 1.5, 1e-10)
            lr = base_lr * (1.0 + 0.1 * (it % 1000) / 1000.0)
            no_imp = 0

        y = a_new
        a = a_new
        t_old = t
        t = t_new

        if it % 200 == 0:
            beta = max(beta * 0.85, 1e-12)
            lr = base_lr / (1.0 + 0.001 * it)

    return best_a, best_score


def _adam_smooth_max(a0, deadline, max_iter=30000, lr=0.03, beta_factor=0.05, verbose=False):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.95)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(1000 + it)
                z = best_z + 0.5 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 100 == 0:
            beta = max(1e-12, beta * 0.8)

    return best_a, best_score


def _lp_spread_reduced(a, M, top_k=None):
    """LP to find spreading direction, using only top constraints for speed."""
    n = len(a)
    c_curr = _fft_convolve(a)

    if top_k is not None and top_k < 2 * n - 1:
        # Only use the most binding constraints
        idx = np.argsort(c_curr)[::-1][:top_k]
        M_vals = c_curr[idx] * (1.0 + 0.001) + 1e-6
        A = np.zeros((top_k, n))
        for i, k in enumerate(idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            A[i, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
        b = M_vals
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M * 1.001 + 1e-6)

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3, top_k=100):
    """LP-based refinement using reduced constraints."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_spread_reduced(a, M, top_k=top_k)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:

            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
            if no_imp > max(100, n // 2):
                a = best_a.copy()
                blend_cur = blend
                no_imp = 0

        a = a_new

    return best_a


def _subgradient_opt(a, deadline, max_iter=10000):
    """Subgradient method targeting the max autocorrelation coefficient."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.5 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_idx = int(np.argmax(c_curr))

        # Subgradient at max
        sg = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.8):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_bump_profile(n, pos=0.3, width=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - pos) ** 2) / (2.0 * width ** 2)) + \
        np.exp(-((t - (1.0 - pos)) ** 2) / (2.0 * width ** 2)) + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _pipeline(a0, deadline, label=""):
    """Full optimization pipeline: FISTA -> LP -> Adam -> LP -> Subgrad."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.8f}", flush=True)

    best_a = a0.copy()
    best_sc = sc0
    n = len(a0)

    # Phase A: FISTA (30% of available time)
    if time.monotonic() < deadline - 60:
        d = min(deadline - 60, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        a_f, s_f = _fista_smooth_max(a0, d, max_iter=20000, beta_factor=0.04)
        if s_f < best_sc:
            best_a, best_sc = a_f, s_f
        print(f"  {label} FISTA: {best_sc:.8f}", flush=True)

    # Phase B: LP refinement
    if time.monotonic() < deadline - 40:
        d = min(deadline - 40, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        a_lp = _lp_refine(best_a, d, max_iter=500, inflate=1.001, blend=0.35, top_k=80)
        s_lp = _score(a_lp)
        if s_lp < best_sc:
            best_a, best_sc = a_lp, s_lp
        print(f"  {label} LP1: {best_sc:.8f}", flush=True)

    # Phase C: Adam (30%)
    if time.monotonic() < deadline - 30:
        d = min(deadline - 30, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        a_ad, s_ad = _adam_smooth_max(best_a, d, max_iter=20000, lr=0.02, beta_factor=0.03)
        if s_ad < best_sc:
            best_a, best_sc = a_ad, s_ad
        print(f"  {label} Adam: {best_sc:.8f}", flush=True)

    # Phase D: Final LP
    if time.monotonic() < deadline - 15:
        d = min(deadline - 15, time.monotonic() + 0.1 * (deadline - time.monotonic()))
        a_lp2 = _lp_refine(best_a, d, max_iter=300, inflate=1.0008, blend=0.2, top_k=60)
        s_lp2 = _score(a_lp2)
        if s_lp2 < best_sc:
            best_a, best_sc = a_lp2, s_lp2
        print(f"  {label} LP2: {best_sc:.8f}", flush=True)

    # Phase E: Subgradient polish
    if time.monotonic() < deadline - 5:
        d = deadline - 5
        a_sg, s_sg = _subgradient_opt(best_a, d, max_iter=5000)
        if s_sg < best_sc:
            best_a, best_sc = a_sg, s_sg
        print(f"  {label} SG: {best_sc:.8f}", flush=True)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: FISTA + LP + Adam + Subgradient.
    Explores n=512 and n=1024 with diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
    else:
        p512 = _make_linear_profile(512, 1.5)
        p1024 = _make_linear_profile(1024, 1.5)

    # === Phase 1: Quick diverse exploration at n=512 (fast) ===
    # Try several initializations quickly to find a good basin
    print("=== Phase 1: Diverse exploration n=512 ===", flush=True)
    d_p1 = min(deadline - 600, t_start + 180)

    candidates_512 = [
        p512,
        _make_linear_profile(512, 2.0),
        _make_linear_profile(512, 0.8),
        _make_cubic_profile(512, 2.0, -1.5),
        _make_cosine_profile(512, 0.7),
        _make_bell_profile(512, 0.6),
        _make_two_bump_profile(512, 0.3, 0.25),
    ]

    # Add perturbed parent
    pp = p512 * (1.0 + 0.2 * rng.normal(size=512))
    pp = np.maximum(pp, 0.0)
    candidates_512.append(pp / pp.sum())

    per_cand_time = (d_p1 - time.monotonic()) / len(candidates_512) * 0.8
    best_512 = p512.copy()
    best_512_sc = _score(p512)

    for i, a0 in enumerate(candidates_512):
        if time.monotonic() > d_p1 - 10:
            break
        d_i = min(d_p1 - 5, time.monotonic() + max(5.0, per_cand_time))
        # Quick Adam for each candidate
        a_q, s_q = _adam_smooth_max(a0, d_i, max_iter=3000, lr=0.04, beta_factor=0.06)
        if s_q < best_512_sc:
            best_512_sc = s_q
            best_512 = a_q.copy()
        if s_q < best_score:
            best_score = s_q
            best_seq = a_q.copy()
        print(f"  P1-{i}: {s_q:.8f}", flush=True)

    # Quick FISTA on best 512
    if time.monotonic() < d_p1 - 10:
        a_f1, s_f1 = _fista_smooth_max(best_512, d_p1 - 5, max_iter=8000, beta_factor=0.04)
        if s_f1 < best_512_sc:
            best_512_sc = s_f1
            best_512 = a_f1.copy()
        if best_512_sc < best_score:
            best_score = best_512_sc
            best_seq = best_512.copy()
        print(f"  P1 FISTA: {best_512_sc:.8f}", flush=True)

    # === Phase 2: Full pipeline on best n=512 ===
    print("=== Phase 2: Full pipeline n=512 ===", flush=True)
    d_p2 = min(deadline - 400, t_start + 500)
    a_best512, s_best512 = _pipeline(best_512, d_p2, "P2-512")
    if s_best512 < best_score:
        best_score = s_best512
        best_seq = a_best512.copy()

    # === Phase 3: Full pipeline on n=1024 ===
    if time.monotonic() < deadline - 300:
        print("=== Phase 3: Full pipeline n=1024 ===", flush=True)
        # Start from interpolated best 512 or parent 1024
        start_1024 = _interpolate_profile(a_best512, 1024)
        d_p3 = min(deadline - 300, t_start + 900)
        a_best1024, s_best1024 = _pipeline(start_1024, d_p3, "P3-1024")
        if s_best1024 < best_score:
            best_score = s_best1024
            best_seq = a_best1024.copy()

    # === Phase 4: Try n=1024 from parent directly ===
    if time.monotonic() < deadline - 200 and parent is not None:
        print("=== Phase 4: Pipeline parent@1024 ===", flush=True)
        d_p4 = min(deadline - 200, t_start + 1200)
        a_p4, s_p4 = _pipeline(p1024, d_p4, "P4-1024")
        if s_p4 < best_score:
            best_score = s_p4
            best_seq = a_p4.copy()

    # === Phase 5: Final refinement on overall best ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"=== Phase 5: Final refinement (best={best_score:.8f}, n={best_seq.size}) ===", flush=True)
        d_p5 = deadline - 20

        # Fine Adam
        d5a = min(d_p5, time.monotonic() + 0.4 * (d_p5 - time.monotonic()))
        a_f5, s_f5 = _adam_smooth_max(best_seq, d5a, max_iter=15000, lr=0.008, beta_factor=0.01)
        if s_f5 < best_score:
            best_score = s_f5
            best_seq = a_f5.copy()
        print(f"  P5 Adam: {best_score:.8f}", flush=True)

        # Fine FISTA
        if time.monotonic() < d_p5 - 30:
            d5b = min(d_p5 - 30, time.monotonic() + 0.3 * (d_p5 - time.monotonic()))
            a_f5b, s_f5b = _fista_smooth_max(best_seq, d5b, max_iter=10000, beta_factor=0.01)
            if s_f5b < best_score:
                best_score = s_f5b
                best_seq = a_f5b.copy()
            print(f"  P5 FISTA: {best_score:.8f}", flush=True)

        # Final LP
        if time.monotonic() < d_p5 - 10:
            d5c = d_p5 - 10
            a_f5c = _lp_refine(best_seq, d5c, max_iter=500, inflate=1.0005, blend=0.15, top_k=50)
            s_f5c = _score(a_f5c)
            if s_f5c < best_score:
                best_score = s_f5c
                best_seq = a_f5c.copy()
            print(f"  P5 LP: {best_score:.8f}", flush=True)

    # === Final validation ===
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}", flush=True)
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}", flush=True)
    return [float(x) for x in final_seq]
