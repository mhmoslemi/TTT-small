import numpy as np

from scipy.optimize import linprog


def _next_pow2(x):
    k = 1
    while k < x:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        x = np.asarray(res.x, dtype=np.float64)
        if np.all(x >= -1e-10):
            return np.maximum(x, 0.0)
    return None


def _smooth_max_weights(c, beta):
    c_max = float(c.max())
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / s
    return c_max, w


def _gradient_direction(a, beta):
    """Compute gradient of smooth max of (a*a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    c_max, w = _smooth_max_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    return grad, c_max


def _logit_adam_optimize(a, deadline, max_iter=50000, lr=0.03):

        vp = grad + 1e-8 * a

        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)

        e = np.exp(z)
        a = e / e.sum()

def _lp_tight_loop(a, deadline, initial_epsilon=0.002, min_epsilon=0.0001,
                   line_search_ks=None, verbose=True):
    """
    Tight LP refinement with adaptive epsilon and line search.
    
    At each iteration:
    1. Compute M = max(a*a)
    2. Solve LP: max sum(g) s.t. (a*g)_k <= M*(1+eps), g >= 0
    3. Line search over blend factors
    4. Update a, adapt epsilon
    """
    if line_search_ks is None:
        line_search_ks = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    epsilon = initial_epsilon
    no_imp = 0
    total_iters = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * (1.0 + epsilon)

        t0 = time.monotonic()
        g = _lp_spread(a, M_lp)
        lp_time = time.monotonic() - t0

        if g is None:
            epsilon = min(0.01, epsilon * 2.0)
            no_imp += 1
            if no_imp > 100:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 5 == 0:
                epsilon = min(0.01, epsilon * 1.4)
            if no_imp > 150:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in line_search_ks:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        total_iters += 1
        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            remaining = deadline - time.monotonic()
            print(f"    LP iter {total_iters}: score={best_score:.10f}, "

                  f"no_imp={no_imp}, t={elapsed:.0f}s, rem={remaining:.0f}s",
                  flush=True)

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_a = a.copy()
                no_imp = 0
                epsilon = max(min_epsilon, epsilon * 0.95)
            else:
                no_imp += 1
        else:

            if no_imp > 300:
                a = best_a.copy()
                epsilon = min(0.01, max(min_epsilon, epsilon * 3))
                no_imp = 0

def _spectral_interpolate(a, n_new):
    """Interpolate sequence to new length via FFT spectrum."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        s = a.sum()
        return (a / s) if s > 1e-30 else np.ones(n_new) / n_new

    N_old = _next_pow2(n_old)
    N_new = _next_pow2(n_new)

    fa = np.fft.rfft(a, N_old)
    k_old = np.arange(N_old // 2 + 1)
    k_new = np.arange(N_new // 2 + 1)

    fa_real = np.interp(k_new, k_old, fa.real)
    fa_imag = np.interp(k_new, k_old, fa.imag)
    fa_new = fa_real + 1j * fa_imag

    a_new = np.fft.irfft(fa_new, N_new)[:n_new]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        x0 = np.linspace(0, 1, n_old)
        x1 = np.linspace(0, 1, n_new)
        a_new = np.maximum(np.interp(x1, x0, a), 0.0)
        s = a_new.sum()
        if s < 1e-30:
            return np.ones(n_new) / n_new
    return a_new / s

            best_seq = a_p2.copy()

    # === Phase 3: Tighter LP ===
    p3_deadline = t_start + 1050.0
    if p3_deadline > deadline:
        p3_deadline = deadline - 250.0
    if time.monotonic() < p3_deadline:
        print(f"\n=== Phase 3: Tight LP (eps_start=0.001, min=0.0001), budget={p3_deadline - time.monotonic():.0f}s ===", flush=True)
        a_p3 = _lp_tight_loop(best_seq, p3_deadline,
                              initial_epsilon=0.001, min_epsilon=0.0001,
                              line_search_ks=[0.02, 0.05, 0.08, 0.1, 0.15, 0.2])
        sc = _score(a_p3)
        print(f"Phase 3 result: {sc:.10f}", flush=True)
        if sc < best_score:
            best_score = sc
            best_seq = a_p3.copy()

    # === Phase 4: Multi-length exploration ===
    p4_deadline = min(t_start + 1500.0, deadline - 150.0)
    if time.monotonic() < p4_deadline and best_n < 1024:
        print(f"\n=== Phase 4: Multi-length exploration, budget={p4_deadline - time.monotonic():.0f}s ===", flush=True)

        # Try n=1024
        n_try = 1024
        a_try = _spectral_interpolate(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  Spectral interp to n={n_try}: {sc_try:.10f}", flush=True)

        # Also try linear interpolation
        a_try_lin = _linear_interpolate(best_seq, n_try)
        sc_try_lin = _score(a_try_lin)
        print(f"  Linear interp to n={n_try}: {sc_try_lin:.10f}", flush=True)

        # Use the better one
        if sc_try < sc_try_lin:
            a_4 = a_try
        else:
            a_4 = a_try_lin

        # LP refinement on n=1024
        sub_deadline = min(p4_deadline, time.monotonic() + 350.0)
        a_4 = _lp_tight_loop(a_4, sub_deadline,
                             initial_epsilon=0.003, min_epsilon=0.0005,
                             line_search_ks=[0.05, 0.1, 0.15, 0.2, 0.3])
        sc_4 = _score(a_4)
        print(f"  Phase 4 LP (n={n_try}): {sc_4:.10f}", flush=True)
        if sc_4 < best_score:
            best_score = sc_4
            best_seq = a_4.copy()
            best_n = n_try
            print(f"  -> New best! n={n_try}", flush=True)

        # Try n=896 if not already at best_n
        if best_n != 896 and time.monotonic() < p4_deadline - 60:
            n_try2 = 896
            a_try2 = _spectral_interpolate(best_seq, n_try2)
            sc_try2 = _score(a_try2)
            print(f"  Spectral interp to n={n_try2}: {sc_try2:.10f}", flush=True)

            if sc_try2 < best_score * 1.01:
                sub_dl2 = min(p4_deadline, time.monotonic() + 120.0)
                a_try2 = _lp_tight_loop(a_try2, sub_dl2,
                                        initial_epsilon=0.003, min_epsilon=0.0005,
                                        line_search_ks=[0.05, 0.1, 0.2, 0.3])
                sc_896 = _score(a_try2)
                print(f"  Phase 4 LP (n={n_try2}): {sc_896:.10f}", flush=True)
                if sc_896 < best_score:
                    best_score = sc_896
                    best_seq = a_try2.copy()
                    best_n = n_try2
                    print(f"  -> New best! n={n_try2}", flush=True)

    # === Phase 5: Final polish ===
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 5: Final polish, budget={deadline - 15 - time.monotonic():.0f}s ===", flush=True)

        # Fine gradient
        a_fine = _fine_polish(best_seq, min(deadline - 40, time.monotonic() + 120.0))
        sc = _score(a_fine)
        print(f"  Fine polish: {sc:.10f}", flush=True)
        if sc < best_score:
            best_score = sc
            best_seq = a_fine.copy()

        # Final LP
        if time.monotonic() < deadline - 15:
            a_final_lp = _lp_tight_loop(best_seq, deadline - 10,
                                        initial_epsilon=0.0005, min_epsilon=0.0001,
                                        line_search_ks=[0.02, 0.05, 0.1],
                                        verbose=False)
            sc = _score(a_final_lp)
            print(f"  Final LP: {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_final_lp.copy()

        # Perturbation search
        if time.monotonic() < deadline - 10:
            a_pert = _perturb_and_refine(best_seq, deadline - 5, rng, n_perturbs=3)
            sc = _score(a_pert)
            print(f"  Perturbation: {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_pert.copy()

    # Final validation
    final_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(768) / 768.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic() - t_start:.1f}s", flush=True)

    return [float(x) for x in final_seq]
