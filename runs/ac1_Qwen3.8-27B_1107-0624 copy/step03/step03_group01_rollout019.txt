The key insight: the LP ensures that (a*g)[k] <= M for all k. For small t:

For small t, the change is approximately:

= (a*a)[k] + 2t[(a*g)[k] - (a*a)[k]]

For k where (a*a)[k] = M (the active set), the change is 2t[(a*g)[k] - M] <= 0 (by LP constraint). So the active constraints are not violated to first order!

For k where (a*a)[k] < M, the change could be positive or negative, but since there's slack, small t won't cause violation.

So for sufficiently small t, a_t will have max(conv) <= M (the active set is not violated to first order, and inactive sets have slack). This means we can always take a small step!

The question is: does this step actually REDUCE the max, or just keep it the same? If the LP is "tight" (the constraint (a*g)[k] = M is active for k in the active set), then the first-order change at those k is zero, and we need second-order analysis.

In practice, the LP will often be tight at the active set, and we'll get very small improvements per iteration. But over many iterations, we can make progress.

```python
from scipy.optimize import linprog
from scipy import sparse

def build_constraint_matrix(a):
    """Build the constraint matrix A for the LP.
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    # A is (2n-1) x n
    # For row k, non-zero entries are at columns i where 0 <= k-i < n
    # i.e., k-(n-1) <= i <= k, intersected with [0, n-1]
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_lo = max(0, k - (n-1))
        i_hi = min(k, n-1)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
"""
LP-guided optimization for minimizing autocorrelation peak of non-negative sequences.
Uses iterative LP to find descent directions, combined with L-BFGS smoothing and
multi-restart for robustness.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _max_conv_fn(a):
    """Returns max of conv(a,a), assuming sum(a)=1."""
    c = _full_convolve(a, a)
    return float(c.max())

def _build_lp_matrix(a):
    """Build the constraint matrix for the LP.
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_lo = max(0, k - (n-1))
        i_hi = min(k, n-1)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _solve_lp_direction(a, M):
    """Solve max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None if LP fails.
    """
    n = len(a)
    A = _build_lp_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            g = np.maximum(0, result.x)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def _lp_guided_descent(a, deadline, max_iters=500, reopt_every=1):
    """LP-guided descent. a is normalized (sum=1). Returns best a and score."""
    n = len(a)
    best_a = a.copy()
    best_M = _max_conv_fn(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = _max_conv_fn(a)
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
        
        # Solve LP
        g = _solve_lp_direction(a, M)
        if g is None:
            break
        
        # Line search along g
        ca = _full_convolve(a, a)
        cg = _full_convolve(a, g)
        gg = _full_convolve(g, g)
        
        # Find best t in (0, 1]
        best_t = 0.0
        best_val = M

def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _max_conv_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    betas = [b * max_c0 for b in beta_starts]
    
    if max_wall_time is not None:
        per_level = max_wall_time / len(betas)
    else:
        per_level = 30.0
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _max_conv_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new

        except Exception:

    return best_a, best_score

def _adam_polish(a_init, deadline, lr=0.03, beta_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _max_conv_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)
    
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)

        sc = _max_conv_fn(a)

            best_score = sc
            best_a = a.copy()
        
        if t % 200 == 0:
            beta = max(1e-12, beta * 0.8)
    
    return best_a, best_score

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
```
