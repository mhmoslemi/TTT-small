def _smooth_max_grad(a, beta):

    Returns (gradient, current max value, convolution array).

    c = _fft_convolve(a)

    # = 2 * (w_cross_correlate_a)[j]

    # This is the cross-correlation of w and a (reversed).
    # Let's use the FFT approach:
    # Define h[j] = sum_k w_k * a_{k-j} for j = 0, ..., n-1
    # This is (w * a_rev)[n-1 + j] where a_rev = a[::-1]

    # Let i = k-j, so k = i+j.
    # h[j] = sum_i w[i+j] * a[i] where 0 <= i < n and 0 <= i+j < 2n-1
    # Since i < n and j >= 0, i+j < 2n-1 is automatic (max is (n-1)+(n-1) = 2n-2).
    # So h[j] = sum_{i=0}^{n-1} a[i] * w[i+j]

    # = (a * w_reversed)[n-1+j] where * is convolution...

    # Hmm, that's not quite right.
    
    # Let me use a different approach:
    # h[j] = sum_{i=0}^{n-1} a[i] * w[i+j]
    # Let's define w_ext of length 2n-1 (same as c) and a of length n.
    # This is the "valid" part of the correlation.
    
    # Actually, the simplest way: use the fact that
    # h = np.correlate(w, a, mode='same') ... no, that's not right either.
    
    # Let me just compute it directly for small n, or use FFT:
    # h[j] = sum_i a[i] * w[i+j] for j = 0,...,n-1
    # This is the "valid" cross-correlation of w and a.
    # In numpy: np.correlate(w, a, mode='valid') gives the valid part.
    # np.correlate(w, a, 'valid')[j] = sum_i w[i+j] * a[i] ... let me check.
    # np.correlate(v, w, 'valid')[k] = sum_n v[n+k] * conj(w[n])

    # h[j] = sum_i a[i] * w[i+j]
    # = sum_i a[i] * w[i+j]

def _bb_step(grad_prev, grad_curr, x_prev, x_curr):
    """Barzilai-Borwein step size computation."""
    s = x_curr - x_prev
    y = grad_curr - grad_prev
    sy = float(np.dot(s, y))
    yy = float(np.dot(y, y))
    ss = float(np.dot(s, s))
    if yy < 1e-16:
        return 0.01
    if sy > 1e-16:
        return ss / sy
    elif sy < -1e-16:
        return -ss / sy
    else:
        return 0.01

    lr = 0.01

            lr = np.clip(lr_bb, 0.001, 0.1)

        # Momentum update

def _spectral_shuffle(a, rng, strength=0.3):
    """Apply random phase modulation in Fourier domain to create diversity."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    
    fa = np.fft.rfft(a)
    # Add random phase
    phase = rng.uniform(0, 2*np.pi, size=fa.size)
    # Blend: keep some of original phase, add random
    amp = np.abs(fa)
    orig_phase = np.angle(fa)
    new_phase = (1-strength) * orig_phase + strength * phase
    fa_new = amp * np.exp(1j * new_phase)
    # Ensure DC component is real and positive
    fa_new[0] = np.abs(fa_new[0])
    
    a_new = np.fft.irfft(fa_new, n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new

def _lp_spread(a, M, n_constraints=None):
    """LP-based spreading: maximize sum(b) s.t. (a*b)(k) <= M, b >= 0."""
    from scipy.optimize import linprog
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix: (a*b)(k) = sum_j a_j * b_{k-j} for valid indices
    # For k = 0,...,2n-2:
    # (a*b)(k) = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} a_{k-j} * b_j
    # So the k-th row of A has A[k,j] = a_{k-j} if 0<=k-j<n
    
    if n_constraints is None:
        n_constraints = 2*n-1
    
    # For large n, this matrix is (2n-1) x n which is ~1535 x 768 = 1.18M entries
    # This might be slow for linprog. Let's only use the top constraints.
    
    # Compute current convolution to find important constraints
    c = _fft_convolve(a * 1.0)  # a*a
    # Find top constraints
    top_k = min(n_constraints, max(100, n))
    sorted_idx = np.argsort(c)[::-1][:top_k]
    
    A = np.zeros((top_k, n))
    for idx, k in enumerate(sorted_idx):
        for j in range(n):
            if 0 <= k-j < n:
                A[idx, j] = a[k-j]
    
    b_ub = np.full(top_k, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'maxiter': 5000})
    if res.success:
        return res.x
    return None

def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2

def _make_linear_profile(n, eta=1.5):

"""
Multi-objective optimization combining L2 convolution norm minimization with
L-infinity peak shaving, BB adaptive step sizes, spectral phase modulation,
and targeted coordinate descent for peak reduction.

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast linear convolution using FFT with zero padding."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        k = 1
        while k < 2 * n - 1:
            k <<= 1
        fa = np.fft.rfft(a, k)
        c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    """Compute softmax weights over convolution entries."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, mval
    return ex / s, mval


def _grad_smooth_max(a, beta):
    """Gradient of smooth max of convolution w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    w, mval = _smooth_max_weights(c, beta)
    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * (w cross-correlate a)[j]
    h = np.correlate(w, a, mode='valid')[:n]
    grad = 2.0 * h
    return grad, mval, c


def _grad_l2_conv(a):
    """Gradient of sum_k ((a*a)(k))^2 w.r.t. a (L2 norm of convolution)."""
    n = len(a)
    c = _fft_convolve(a)
    # grad[j] = 4 * sum_k c[k] * a[k-j] = 4 * (c cross-correlate a)[j]
    h = np.correlate(c, a, mode='valid')[:n]
    grad = 4.0 * h
    l2_val = float(np.sum(c * c))
    return grad, l2_val, c


def _bb_step(s, y):
    """Barzilai-Borwein step size. Returns avg of BB+ and BB- for stability."""
    sy = float(np.dot(s, y))
    yy = float(np.dot(y, y))
    ss = float(np.dot(s, s))
    if yy < 1e-30:
        return 0.01
    if sy > 1e-30:
        return ss / sy
    elif sy < -1e-30:
        return abs(ss / sy)
    return 0.01


def _hybrid_optimize(a, deadline, max_iter=100000, beta_init=None,
                     l2_weight=0.3, use_l2_interval=5):
    """
    Hybrid optimizer alternating between L2-norm and L-infinity gradient steps.
    Uses BB adaptive step size and momentum.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    if beta_init is None:
        c = _fft_convolve(a)
        beta_init = 0.03 * float(c.max())

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    beta = beta_init
    lr = 0.01
    v = np.zeros(n)
    momentum = 0.8
    no_imp = 0
    base_lr = 0.01

    prev_grad = None
    prev_x = None

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Choose which gradient to use
        use_l2 = (it % use_l2_interval < l2_weight * use_l2_interval)

        if use_l2:
            grad, _, c = _grad_l2_conv(a)
            # Scale L2 gradient to be comparable to L-inf gradient
            scale = float(c.max()) / max(1e-10, np.sqrt(float(np.sum(c*c))) + 1e-10)
            grad = grad * scale * 0.5
        else:
            grad, mval, c = _grad_smooth_max(a, beta)

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            beta = max(1e-10, beta * 2.0)
            continue

        # BB step size
        if prev_grad is not None and prev_x is not None:
            s_vec = a - prev_x
            y_vec = grad - prev_grad
            lr_bb = _bb_step(s_vec, y_vec)
            lr = float(np.clip(lr_bb, 0.002, 0.05))

        prev_grad = grad.copy()
        prev_x = a.copy()

        # Momentum update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        # Evaluate
        c_new = _fft_convolve(a)
        cur_score = float(2.0 * n * c_new.max())

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, base_lr * 0.94)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = beta_init
                lr = base_lr
                no_imp = 0
                # Perturb to escape
                rng_local = np.random.default_rng(hash((it, n)) % (2**32))
                a = a * (1.0 + 0.08 * rng_local.normal(size=n))
                a = _project_simplex(a)

        # Adapt beta (for L-inf component)
        if it % 300 == 0:
            beta = max(1e-8, beta * 0.82)

        # Dampen momentum periodically
        if it % 2000 == 0:
            v *= 0.5

    return best_a, best_score


def _peak_shave(a, deadline, max_iter=5000):
    """
    Directly target the peak of the convolution by adjusting coordinates
    that contribute most to it.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    rng = np.random.default_rng(999)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        peak_k = int(np.argmax(c))
        M = float(c[peak_k])

        # Find which coordinates contribute most to the peak
        # (a*a)[peak_k] = sum_i a[i] * a[peak_k - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contributions[i] = a[i]

        # Find the index with maximum contribution
        top_idx = int(np.argmax(contributions))
        # Find a lower-contributing index to move mass to
        low_indices = np.argsort(contributions)[:max(10, n // 10)]
        target_idx = int(low_indices[rng.integers(0, len(low_indices))])

        if top_idx == target_idx:
            continue

        # Try transferring a small fraction of mass
        found = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
            delta = frac * a[top_idx]
            if delta >= a[top_idx]:
                continue
            a_try = a.copy()
            a_try[top_idx] -= delta
            a_try[target_idx] += delta
            a_try = _project_simplex(a_try)
            sc_try = float(2.0 * n * _fft_convolve(a_try).max())
            if sc_try < best_score - 1e-14:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                no_imp = 0
                found = True
                break

        if not found:
            no_imp += 1
            if no_imp > 2000:
                break

    return best_a, best_score


def _spectral_phase_shuffle(a, rng, strength=0.2):
    """Random phase modulation in Fourier domain for diversity."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    fa = np.fft.rfft(a)
    amp = np.abs(fa)
    orig_phase = np.angle(fa)
    rand_phase = rng.uniform(0, 2 * np.pi, size=fa.size)
    new_phase = (1.0 - strength) * orig_phase + strength * rand_phase
    fa_new = amp * np.exp(1j * new_phase)
    fa_new[0] = np.abs(fa_new[0])  # Keep DC real and positive
    a_new = np.fft.irfft(fa_new, n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def _lp_spread(a, inflate=1.001):
    """LP-based spreading: find b>=0 maximizing sum(b) s.t. (a*b)(k) <= M*inflate."""
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max()) * inflate

    # Use top constraints only (for speed)
    top_k = min(2 * n - 1, max(200, n))
    sorted_idx = np.argsort(c)[::-1][:top_k]

    A = np.zeros((top_k, n))
    for idx, k in enumerate(sorted_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = a[k - j]

    b_ub = np.full(top_k, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'maxiter': 3000, 'dual_feasibility_tol': 1e-8})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.25):
    """Iterative LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = _lp_spread(a, inflate_cur)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = _project_simplex(a_new)

        sc_new = float(2.0 * n * _fft_convolve(a_new).max())
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.95)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.008, inflate_cur * 0.97)
            if no_imp > 300:
                a = best_a.copy()
                inflate_cur = inflate
                blend_cur = blend
                no_imp = 0

        a = a_new

    return best_a, best_score


def _make_u_profile(n, alpha=2.0):
    """U-shaped profile: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    """Linear (monotone) profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, width=0.12, height=3.0):
    """Bimodal: two peaks at the edges."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + height * (np.exp(-0.5 * ((t - 0.0) / width) ** 2) +
                        np.exp(-0.5 * ((t - 1.0) / width) ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quartic_profile(n, alpha=1.0):
    """Quartic U-shape (sharper than quadratic)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u ** 4
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _coordinate_refine(a, deadline, max_iter=20000):
    """
    Coordinate-wise mass transfer optimization.
    For each peak location, try moving mass from high-contributing to
    low-contributing coordinates.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        # Consider top few peaks
        top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for pk in top_peaks:
            pk = int(pk)
            # Find contributing coordinates
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            if lo >= hi:
                continue

            # Contribution of each coordinate i to this peak: a[i] * a[pk-i]
            # We want to reduce the peak by moving mass from high a[i]*a[pk-i]
            # to elsewhere.
            # Simply: find the i in [lo,hi] with max a[i], and move some to
            # a random coordinate outside [lo,hi].
            i_max = lo + int(np.argmax(a[lo:hi+1]))

            # Choose target outside the peak's support
            outside = list(range(0, lo)) + list(range(hi + 1, n))
            if not outside:
                continue
            j_target = int(outside[rng.integers(0, len(outside))])

            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = frac * a[i_max]
                if delta >= a[i_max] * 0.999:
                    continue
                a_try = a.copy()
                a_try[i_max] -= delta
                a_try[j_target] += delta
                a_try = _project_simplex(a_try)
                sc = float(2.0 * n * _fft_convolve(a_try).max())
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    no_imp = 0
                    improved = True
                    break

        if not improved:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a, best_score
