```python
"""
Optimized step function search using active-set sub-gradient method with LP refinement.

Key ideas:
1. Active-set sub-gradient: minimize max(a*a) using sub-gradient of the max of quadratic forms
2. Adaptive active set: dynamically include lags that are close to the maximum
3. LP spreading: periodically solve LP to find a "spread" direction
4. Multiple sequence lengths: explore n=768, 1024, 1536
5. Simplex projection: maintain non-negativity and sum=1 constraint
6. Barzilai-Borwein step size for fast convergence
"""

import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    """Compute a*a via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = 1 << (2*n-2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    if c.min() < 0:
        c -= c.min()
    return c

def _score(a):
    """Compute 2n*max(a*a)/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _subgradient(c, a, active_indices):
    """Compute sub-gradient of max(a*a) at a using active set."""
    n = len(a)
    g = np.zeros(n)
    m = len(active_indices)
    if m == 0:
        return g
    # For each active lag k, gradient is 2*a_{k-j} (reflected)
    # Collect all needed values
    for k in active_indices:
        # g_j += (2/m) * a_{k-j} for valid j
        # a_{k-j} is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a_{k-j} for j in [lo, hi]
            # When j=lo, k-j = k-lo = k-max(0,k-n+1)
            # When j=hi, k-j = k-hi
            indices = np.arange(lo, hi+1)
            sources = k - indices
            g[lo:hi+1] += (2.0/m) * a[sources]
    return g

def _subgradient_fast(c, a, active_indices, n):
    """Faster sub-gradient using numpy vectorization."""
    g = np.zeros(n)
    m = len(active_indices)
    if m == 0:
        return g
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            sources = k - np.arange(lo, hi+1)
            g[lo:hi+1] += (2.0/m) * a[sources]
    return g
```
