"""
LP-centric optimization with adaptive constraint tightening and hybrid refinement.
Combines iterative LP spreading with Riemian gradient descent and coordinate descent
to minimize the max linear autocorrelation of non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    # Next power of 2 >= 2n-1
    N = 1
    while N < 2*n-1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(c.max()) / (s*s)

def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0"""
    n = len(a)
    c_obj = -np.ones(n)  # minimize -sum(b) = max sum(b)
    # Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0
    # A has shape (2n-1, n)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For column j, A[k,j] = a[k-j] if 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(j+n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-j]
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

"""
LP-centric optimization with adaptive constraint tightening and hybrid gradient refinement.
Minimizes the maximum linear autocorrelation of non-negative sequences using iterative LP
spreading as the primary engine, supplemented by smooth-max Adam and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(c.max()) / (s * s)


def _build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix for a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_spread_sparse(a, M):
    """Solve LP using sparse matrix for larger n."""
    from scipy.sparse import csr_matrix
    n = len(a)
    # Build sparse convolution matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            rows.append(j + i)
            cols.append(j)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _subgradient_at_max(a, k):
    """Subgradient of max autocorrelation at lag k."""
    n = len(a)
    sg = np.zeros(n)
    # dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n
    m_start = max(0, k - n + 1)
    m_end = min(k, n - 1)
    if m_start <= m_end:
        sg[m_start:m_end + 1] = 2.0 * a[m_end: m_start - 1: -1]
    return sg


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.03, beta_factor=0.05):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        mval = float(c_curr.max())
        x = (c_curr - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c_curr)
            ex[int(np.argmax(c_curr))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        # Gradient: dc/a weighted by w
        # grad[m] = 2 * sum_k w[k] * a[k-m] (if valid)
        # This is 2 * (w * a_rev)[m-1] (linear convolution, appropriately sliced)
        # w has length 2n-1, a has length n
        # (w_rev * a)[k] = sum_j w_rev[j] * a[k-j]
        # We want sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j] for j in [0, n-1]
        # = (a * w_shifted)[m] where w_shifted[k] = w[k+m]
        # Actually, let's use the formula: grad[m] = 2 * sum_k w[k] * a[k-m] * I(0<=k-m<n)
        # = 2 * sum_j w[m+j] * a[j] (for j in [0,n-1], m+j < 2n-1)
        # This is like a cross-correlation.
        
        # Efficient computation via FFT:
        # grad[m] = 2 * sum_j w[m+j] * a[j] for j=0..n-1
        # Let's define w_pad of length N >= 3n-1
        # grad = 2 * correlate(w, a) where (w * a_rev)[n-1 + m] = sum_j w[j] * a[n-1-(n-1+m-j)] ... 
        
        # Simpler: use the convolution matrix approach
        # grad[m] = 2 * (A^T w)[m] where A is the conv matrix of a
        # A[k,j] = a[k-j] if 0<=k-j<n
        # (A^T w)[m] = sum_k A[k,m] * w[k] = sum_k a[k-m] * w[k] * I(0<=k-m<n)
        # = sum_j a[j] * w[j+m] (with j = k-m, valid when 0<=j<n and 0<=j+m<2n-1)
        
        # Use FFT: pad w and a to length N, compute convolution
        N = _next_pow2(3 * n)
        a_pad = np.zeros(N)
        a_pad[:n] = a
        w_pad = np.zeros(N)
        w_pad[:2*n-1] = w
        # We want: grad[m]/2 = sum_j a[j] * w[m+j]
        # This is the cross-correlation of a and w, at lag m
        # corr[m] = sum_j a[j] * w[m+j]
        # = conv(a_rev, w)[n-1+m] where a_rev = a[::-1]
        a_rev = a[::-1].copy()
        conv_res = np.fft.irfft(np.fft.rfft(a_rev, N) * np.fft.rfft(w_pad, N), N)
        # conv_res[k] = sum_j a_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
        # Let i = n-1-j: = sum_i a[i] * w[k-n+1+i]
        # We want sum_i a[i] * w[m+i], so we need k-n+1 = m, i.e., k = n-1+m
        grad_a = 2.0 * conv_res[n - 1:2 * n - 1]

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.001, lr_cur * 0.93)
            if no_imp > max(4000, 6 * n):
                rng_r = np.random.default_rng(1000 + it)
                z = best_z + 0.3 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.78)

    return best_a, best_score


def _lp_refine_intensive(a, deadline, max_iter=5000, eps_start=0.01, eps_min=0.0001):
    """Intensive LP-based refinement with adaptive tightening."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    eps = eps_start
    no_imp = 0
    consecutive_fails = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Try current eps
        b_opt = _lp_spread(a, M * (1.0 + eps))

        if b_opt is None:
            consecutive_fails += 1
            eps = max(eps_min, eps * 0.7)
            if consecutive_fails > 20:
                break
            continue

        sum_b = float(b_opt.sum())
        if sum_b > 1.0 + 1e-6:
            # Normalize and check improvement
            b_norm = b_opt / sum_b
            c_new = _fft_convolve(b_norm)
            score_new = float(2.0 * n * c_new.max())

            if score_new < best_score - 1e-10:
                best_score = score_new
                best_a = b_norm.copy()
                a = b_norm.copy()
                no_imp = 0
                consecutive_fails = 0
                # Slightly decrease eps for next iteration
                if it % 10 == 0:
                    eps = max(eps_min, eps * 0.95)
            else:
                # LP found a solution but no actual improvement
                no_imp += 1
                consecutive_fails += 1
                if consecutive_fails > 30:
                    # Try a different strategy: blend with LP solution
                    blend = min(0.5, 0.1 + 0.01 * consecutive_fails)
                    a_new = (1.0 - blend) * a + blend * b_norm
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()

def _coordinate_descent(a, deadline, max_iter=20000):
    """Coordinate descent: transfer mass between pairs to reduce max autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_max = float(c.max())
    best_score = float(2.0 * n * best_max)

    rng = np.random.default_rng(12345)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_val = float(c.max())
        max_idx = int(np.argmax(c))

        # Find the subgradient (which peaks indicate which coordinates to adjust)
        sg = _subgradient_at_max(a, max_idx)
        
        # Find coordinate with highest sg (most "responsible" for the peak)
        if sg.max() <= 0:
            break
        
        # Pick the source (high sg) and a random target (low sg or zero)
        src = int(np.argmax(sg))
        
        # Find a target: prefer coordinates with low value
        # Use a score: high value + high sg = bad, want to move from there
        target = int(rng.integers(0, n))
        if target == src:
            target = (src + 1) % n
        
        # Transfer mass: reduce src, increase target
        # How much to transfer?
        amount = min(a[src], 0.01 / n) * rng.uniform(0.1, 0.5)
        
        if amount <= 0:
            no_imp += 1
            if no_imp > 200:
                break
            continue
        
        a_new = a.copy()
        a_new[src] -= amount
        a_new[target] += amount
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        new_max = float(c_new.max())
        
        if new_max < max_val - 1e-12:
            a = a_new
            if new_max < best_max:
                best_max = new_max
                best_score = float(2.0 * n * best_max)
                best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                # Try a different strategy: direct subgradient step
                sg_norm = float(np.linalg.norm(sg))
                if sg_norm > 0:
                    lr = 0.1 / np.sqrt(it + 1) / n
                    a_direct = a - lr * sg / sg_norm
                    a_direct = np.maximum(a_direct, 0.0)
                    s_d = a_direct.sum()
                    if s_d > 1e-10:
                        a_direct = a_direct / s_d
                        c_d = _fft_convolve(a_direct)
                        if float(c_d.max()) < max_val:
                            a = a_direct
                            no_imp = 0
                        else:
                            no_imp = 200  # encourage restart
                if no_imp > 1000:
                    break

    return best_a


def _make_profile(n, profile_type='cosine', **kwargs):
    """Generate various profile shapes."""
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'cosine':
        alpha = kwargs.get('alpha', 1.0)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * t)
    elif profile_type == 'linear':
        eta = kwargs.get('eta', 1.5)
        a = 1.0 + eta * (t - 0.5)
    elif profile_type == 'bell':
        width = kwargs.get('width', 0.3)
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    elif profile_type == 'cubic':
        a1 = kwargs.get('a1', 1.5)
        a3 = kwargs.get('a3', -1.0)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a3 * u ** 3
    elif profile_type == 'uniform':
        a = np.ones(n)
    elif profile_type == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

   = (a_0*a_0)[k] + 2*(a_0*da)[k] (by symmetry)
   
   So the linearized constraint is: (a_0*a_0)[k] + 2*(a_0*da)[k] <= t
   
   With a = a_0 + da, sum(a) = 1:
   sum(da) = 1 - sum(a_0)
   
   This gives a QCQP that's approximately LP at each iteration. But it's still not a standard LP.
