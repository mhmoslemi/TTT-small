def compute_score(a):

    a = np.asarray(a, dtype=np.float64).ravel()

    s = float(a.sum())
    if s < 0.01:
        return np.inf

        N *= 2
    fa = np.fft.rfft(a, N)

def build_conv_matrix(f):
    """Build matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    i_idx = np.arange(n)[None, :]

"""
LP-guided optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal improvement directions at each step,
combined with gradient-based refinement, multi-scale search, and diverse restarts.
The LP at each iteration finds the direction g maximizing sum(g) subject to
||f*g||_inf <= ||f*f||_inf, which gives the most efficient improvement direction.
"""
import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def build_conv_matrix(f):
    """Build matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0. Shape: (2n-1, n)."""
    n = len(f)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    i_idx = np.arange(n)[None, :]
    diff = k_idx - i_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((m, n))
    A[mask] = f[diff[mask]]
    return A


def solve_lp_direction(f, M, time_limit=8.0):
    """
    Solve: max sum(b) s.t. b >= 0, A(f) @ b <= M * ones.
    Returns normalized b (sum=1) or f if LP fails.
    """
    n = len(f)
    A = build_conv_matrix(f)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception as e:
        print(f"    LP error: {e}")
    return f.copy()


def lp_iterate(f, deadline, max_iters=10000, verbose=True, t_init=0.1):
    """
    Perform LP-based iterations to minimize the objective.
    At each step: solve LP to find direction g, then line search for step size.
    """
    f = np.asarray(f, dtype=np.float64).ravel().copy()
    n = f.size
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 0.01:
        f = np.ones(n) / n
    else:
        f = f / s

    current_score = compute_score(f)
    best_f = f.copy()
    best_score = current_score
    no_improve = 0
    iters = 0

    while time.monotonic() < deadline and iters < max_iters:
        iters += 1
        c = full_convolve_fft(f)
        M = float(c.max())

        g = solve_lp_direction(f, M, time_limit=6.0)

        if np.allclose(g, f, atol=1e-12):
            no_improve += 5
            if no_improve > 30:
                break
            continue

        improved = False
        t = t_init
        for _ in range(10):
            a_new = (1.0 - t) * f + t * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 0.01:
                break
            a_new = a_new / sn
            score_new = compute_score(a_new)
            if score_new < current_score - 1e-13:
                f = a_new
                current_score = score_new
                improved = True
                if score_new < best_score:
                    best_score = score_new
                    best_f = f.copy()
                no_improve = 0
                break
            t *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 40:
                break

        if verbose and iters % 200 == 0:
            el = time.monotonic() - t_start_global
            print(f"    LP iter {iters}: score={current_score:.10f} "
                  f"no_improve={no_improve} elapsed={el:.0f}s")

    if verbose:
        print(f"    LP done: {iters} iters, final_score={best_score:.10f}")
    return best_f, best_score


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT (cross-correlation)."""
    n = len(a)
    af = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        w = np.zeros_like(c)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def gradient_optimize(a0, deadline, lr=0.03, beta_factor=0.06,
                      beta_decay=0.93, max_iter=100000):
    """Logit-space Adam gradient optimization with soft-max surrogate."""
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    beta = beta_factor * max_c
    if beta < 1e-10:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    iters = 0
    while time.monotonic() < deadline and iters < max_iter:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.clip(z, -50.0, 50.0)
        else:
            z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            break
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        iters += 1

        if iters % 60 == 0:
            beta = max(1e-10, beta * beta_decay)

        if no_improve > 800:
            current_lr = max(0.003, current_lr * 0.95)
            if no_improve > 3000:
                break

    return best_a, best_score


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 0.0)
    s = result.sum()
    return result / s if s > 1e-10 else np.ones(n_new) / n_new


def make_linear_profile(n, eta=1.5):
    """Linear ramp profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_bump(n, width=0.5, offset=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_ramp(n, p=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(make_linear_profile(n, eta))
    for k in [1, 2, 3]:
        for amp in [0.3, 0.5]:

    for width in [0.3, 0.5, 0.7]:
        for offset in [0.2, 0.4, 0.5]:
            profiles.append(make_smooth_bump(n, width, offset))
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_power_ramp(n, p))

        profiles.append(a / a.sum())

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    # Validate with evaluate_sequence if available
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
            if eval_score != final_score:
                print(f"  WARNING: mismatch! internal={final_score}, eval={eval_score}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check failed: {e}")

    return [float(x) for x in final_seq]
