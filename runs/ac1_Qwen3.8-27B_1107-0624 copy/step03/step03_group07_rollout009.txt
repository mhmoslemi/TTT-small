"""
Hybrid exact-subgradient and soft-max optimization for minimum peak convolution sequences.
Uses direct subgradient computation (no soft-max approximation) for precision,
combined with peak-averaging for stability, simplex projection, and multi-scale search.
"""
import numpy as np
import time

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_grad_at_peak(a, k):
    """Compute d(conv(a,a)[k])/da. Returns vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    # d/d(a[j]) conv(a,a)[k] = 2*a[k-j] if 0 <= k-j < n
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k - j]
    return grad

def compute_grad_at_peak_fast(a, k):
    """Faster version using the relationship between convolution and correlation."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    # grad[j] = 2 * a[k-j] for j in [lo, hi]
    # a[k-j] for j in [lo, hi] = a[hi] ... a[lo] reversed = a[lo:hi+1][::-1] ... 
    # Actually: for j = lo, k-j = k-lo; for j = hi, k-j = k-hi
    # So grad[lo:hi+1] = 2 * a[k-hi : k-lo+1][::-1]
    # k-hi to k-lo: that's a slice of a
    a_start = k - hi
    a_end = k - lo + 1
    grad[lo:hi+1] = 2.0 * a[a_start:a_end][::-1]
    return grad

def subgradient_multi_peak(a, c, top_k=1):
    """Compute subgradient as average over top-k peaks."""
    n = len(a)
    idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        grad += compute_grad_at_peak_fast(a, int(k))
    grad /= top_k
    return grad

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    # Simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logit_project(v):
    """Convert logit space to simplex."""
    v = v - v.max()
    a = np.exp(v)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return a / s

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

"""

Key differences from previous approach: uses exact (non-approximated) subgradient of the max
convolution value, adaptive active-peak set that narrows over time, multi-n exploration,
and a hybrid Adam/projected-gradient optimizer with backtracking line search for final polish.

import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_h_from_weights(w, a):
    """
    Compute h[i] = sum_j a[j] * w[i+j] for i=0..n-1 via FFT.
    This gives d(sum_k w[k]*conv(a,a)[k])/da[i] = 2*h[i].
    """
    n = len(a)
    a_rev = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa_rev * fw, N)
    return conv[n - 1:2 * n - 1]


def make_initial_profiles(n, rng):

    profiles = []

    profiles.append(np.ones(n) / n)

    tri = 1.0 - np.abs(2 * t - 1)
    tri = np.maximum(tri, 1e-10)
    profiles.append(tri / tri.sum())

    # Cosine bell
    cos_bell = 0.5 * (1 + np.cos(2 * np.pi * t))
    cos_bell = np.maximum(cos_bell, 1e-10)
    profiles.append(cos_bell / cos_bell.sum())

    # Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    profiles.append(ramp / ramp.sum())

    # Sine arch
    sine = np.sin(np.pi * t)
    sine = np.maximum(sine, 1e-10)
    profiles.append(sine / sine.sum())

    # Tukey windows
    for r in [0.3, 0.5, 0.7]:
        tukey = np.ones(n)
        m = int(r * n / 2)
        for i in range(m):
            val = 0.5 * (1 + np.cos(np.pi * (2.0 * i / n - r) / r))
            tukey[i] = max(val, 1e-10)
            tukey[n - 1 - i] = tukey[i]
        tukey = np.maximum(tukey, 1e-10)
        profiles.append(tukey / tukey.sum())

    # Gaussian
    gauss = np.exp(-15.0 * (t - 0.5) ** 2)

            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    return profiles


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def logit_adam_exact(a0, deadline, rng, lr=0.01, top_k_init=50, top_k_min=1,
                     top_k_decay=0.9995, restart_period=400, blend_decay=0.999):
    """
    Logit-space Adam optimization using EXACT subgradient (no soft-max).
    The active peak set adaptively shrinks from top_k_init to top_k_min.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    c_max_init = float(c.max())

    z = np.log(a0)
    z = z - z.mean()

    best_a = a0.copy()
    best_s = z.copy()
    best_score = float(2.0 * n * c.max())
    no_improve = 0

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    current_lr = lr
    base_lr = lr

    top_k = float(top_k_init)
    # Blend between top-k uniform and soft-max (for initial smoothness)
    blend = 0.3  # fraction of soft-max

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_s = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute weights: blend of top-k uniform and soft-max
        k = max(1, min(2 * n - 1, int(top_k)))
        idx = np.argsort(c)[-k:]
        w_hard = np.zeros(2 * n - 1)
        w_hard[idx] = 1.0 / k

        # Soft-max component
        beta = 0.02 * c_max
        x = (c - c_max) / beta
        x = np.clip(x, -50.0, 0.0)
        ex = np.exp(x)
        sw = float(ex.sum())
        w_soft = ex / sw

        w = blend * w_soft + (1.0 - blend) * w_hard

        # Gradient: d(sum_k w[k]*conv(a,a)[k])/da[i] = 2*h[i]
        h = compute_h_from_weights(w, a0)

        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Stabilize

            z = best_s.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_s.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        # Adapt parameters
        top_k = max(float(top_k_min), top_k * top_k_decay)
        blend = max(0.0, blend * blend_decay)

        if no_improve > 200:
            current_lr = max(0.002, current_lr * 0.98)

        if no_improve > restart_period:
            noise_scale = 0.1
            z = best_s + noise_scale * rng.normal(size=n)

            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            top_k = float(top_k_init)
            blend = 0.3
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def exact_projected_gradient(a0, deadline, rng, lr_init=0.005, top_k=5,
                             backtracking=True):
    """
    Direct projected gradient on the simplex with exact subgradient.
    Used for final polish when Adam has converged.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s

    c = full_convolve_fft(a0)
    best_a = a0.copy()
    best_score = float(2.0 * n * c.max())
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score - 1e-15:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Exact subgradient: average over top-k peaks
        k = min(top_k, 2 * n - 1)
        idx = np.argsort(c)[-k:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k

        h = compute_h_from_weights(w, a0)
        grad = 2.0 * h

        # Projected gradient step
        a_new = a0 - lr * grad

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        # Evaluate
        c_new = full_convolve_fft(a_new)
        score_new = 2.0 * n * c_new.max()

        if score_new <= score + 1e-12:
            a0 = a_new
        else:
            lr *= 0.5
            if lr < 1e-14:
                lr = lr_init * 0.1
                a0 = a_new

        if no_improve > 200:
            break

    return best_a, best_score

        a0 = a0 / a0.sum()

        # Slowly decrease learning rate
        lr = max(1e-6, lr * 0.9999)

    return best_a, best_score


def diverse_structural_perturbation(a0, rng, n_pert=50):
    """Apply structural (smooth) perturbation to escape local minima."""
    n = len(a0)
    t = np.linspace(0, 1, n)
    perturbation = np.zeros(n)
    for _ in range(3):
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.1)
        perturbation += amp * np.cos(2 * np.pi * freq * t + phase)
    a_new = a0 + n_pert * perturbation
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase exact-subgradient search for minimum peak convolution sequences.
    Phase 1: Explore multiple n values with diverse initializations.
    Phase 2: Main optimization with exact subgradient + Adam in logit space.
    Phase 3: Fine-tuning with projected gradient and peak shaving.
    Phase 4: Final validation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
                update(init_seq, sc_init)
    except Exception:
        pass

    print(f"Starting exact-subgradient search. Budget={budget_s}s")

    # ===== Phase 1: Multi-n exploration (0 - 400s) =====
    phase1_deadline = min(deadline, t_start + 400.0)
    print(f"\n--- Phase 1: Multi-n exploration ---")

    n_candidates = [512, 768, 1024, 1536, 2048, 3072, 4096]

    # If we have an initial sequence, include its size
    if init_seq is not None and init_seq.size not in n_candidates:
        n_candidates.append(init_seq.size)
        n_candidates.sort()

    phase1_best = None
    phase1_best_score = np.inf
    phase1_best_n = 0

    for n_test in n_candidates:
        if time.monotonic() > phase1_deadline - 20:
            break

        # Build initial profiles for this n
        if init_seq is not None:
            p_interp = interpolate_profile(init_seq, n_test)
            sc_interp = compute_score(p_interp)
            print(f"  n={n_test}: interpolated score={sc_interp:.8f}")
            if sc_interp < phase1_best_score:
                phase1_best_score = sc_interp
                phase1_best = p_interp.copy()
                phase1_best_n = n_test
            update(p_interp, sc_interp)

            # Quick optimization from interpolated
            if sc_interp < phase1_best_score + 0.05:
                dl = min(phase1_deadline, time.monotonic() + 30.0)
                r, sc_r = logit_adam_exact(p_interp, dl, rng, lr=0.02,
                                            top_k_init=30, top_k_min=2,
                                            top_k_decay=0.999, restart_period=200)
                sc_f = compute_score(r)
                update(r, sc_f)
                if sc_f < phase1_best_score:
                    phase1_best_score = sc_f
                    phase1_best = r.copy()
                    phase1_best_n = n_test
                print(f"  n={n_test}: after quick opt={sc_f:.8f}")
        else:
            # No init, try a few profiles
            profiles = make_initial_profiles(n_test, rng)[:4]
            for p in profiles:
                sc_p = compute_score(p)
                if sc_p < phase1_best_score + 0.5:
                    dl = min(phase1_deadline, time.monotonic() + 10.0)
                    r, sc_r = logit_adam_exact(p, dl, rng, lr=0.03,
                                                top_k_init=50, top_k_min=5)
                    sc_f = compute_score(r)
                    update(r, sc_f)
                    if sc_f < phase1_best_score:
                        phase1_best_score = sc_f
                        phase1_best = r.copy()
                        phase1_best_n = n_test

    print(f"  Phase 1 best: n={phase1_best_n}, score={phase1_best_score:.10f}")

    # ===== Phase 2: Main optimization (400s - 1300s) =====
    phase2_deadline = min(deadline, t_start + 1300.0)
    print(f"\n--- Phase 2: Main optimization ---")

    # Determine best n from phase 1
    if phase1_best is not None:
        main_n = phase1_best_n
        main_start = phase1_best.copy()
    elif init_seq is not None:
        main_n = init_seq.size
        main_start = init_seq.copy()
    else:
        main_n = 2048
        main_start = make_initial_profiles(main_n, rng)[0]

    print(f"  Main optimization at n={main_n}")

    # Long Adam run with exact subgradient
    dl2a = min(phase2_deadline, time.monotonic() + 500.0)
    r2, sc2 = logit_adam_exact(main_start, dl2a, rng, lr=0.012,
                                top_k_init=50, top_k_min=1,
                                top_k_decay=0.9997, restart_period=500)
    sc2f = compute_score(r2)
    update(r2, sc2f)
    print(f"  After main Adam: n={main_n}, score={sc2f:.10f}")

    # Try a few more n values by interpolation and quick optimization
    for n_try in [main_n // 2, main_n * 2]:
        if n_try < 128 or n_try > 8192:
            continue
        if time.monotonic() > phase2_deadline - 30:
            break
        p_try = interpolate_profile(r2, n_try)
        sc_try = compute_score(p_try)
        update(p_try, sc_try)
        if sc_try < best_score + 0.02:
            dl = min(phase2_deadline, time.monotonic() + 60.0)
            r_try, sc_try_r = logit_adam_exact(p_try, dl, rng, lr=0.01,
                                                top_k_init=30, top_k_min=2,
                                                top_k_decay=0.9995)
            sc_try_f = compute_score(r_try)
            update(r_try, sc_try_f)
            print(f"  Try n={n_try}: score={sc_try_f:.10f}")

    # Second Adam run with different hyperparams
    if time.monotonic() < phase2_deadline - 60:
        p2b = best_seq.copy() + 0.01 * rng.normal(size=best_seq.size)
        p2b = np.maximum(p2b, 1e-12)
        p2b = p2b / p2b.sum()
        dl2b = min(phase2_deadline, time.monotonic() + 150.0)
        r2b, sc2b = logit_adam_exact(p2b, dl2b, rng, lr=0.008,
                                      top_k_init=20, top_k_min=1,
                                      top_k_decay=0.9998, restart_period=300,
                                      )
        sc2bf = compute_score(r2b)
        update(r2b, sc2bf)
        print(f"  After 2nd Adam (different params): score={sc2bf:.10f}")

    print(f"  Phase 2 best: score={best_score:.10f}")

    # ===== Phase 3: Fine-tuning (1300s - 1700s) =====
    phase3_deadline = min(deadline, t_start + 1700.0)
    print(f"\n--- Phase 3: Fine-tuning ---")

    if best_seq is not None:
        n_polish = best_seq.size

        # Projected gradient polish
        if time.monotonic() < phase3_deadline - 60:
            dl3a = min(phase3_deadline, time.monotonic() + 100.0)
            r3a, sc3a = exact_projected_gradient(best_seq, dl3a, rng,
                                                  lr_init=0.002, top_k=3)
            sc3af = compute_score(r3a)
            update(r3a, sc3af)
            print(f"  After projected gradient: score={sc3af:.10f}")

        # Peak shaving
        if time.monotonic() < phase3_deadline - 40:
            p3b = best_seq + 0.001 * rng.normal(size=n_polish)
            p3b = np.maximum(p3b, 1e-12)
            p3b = p3b / p3b.sum()
            dl3b = min(phase3_deadline, time.monotonic() + 80.0)
            r3b, sc3b = peak_shaving(p3b, dl3b, rng, n_peaks=3, lr=0.002)
            sc3bf = compute_score(r3b)
            update(r3b, sc3bf)
            print(f"  After peak shaving: score={sc3bf:.10f}")

        # Another Adam round with very small lr
        if time.monotonic() < phase3_deadline - 30:
            p3c = best_seq.copy()
            dl3c = min(phase3_deadline, time.monotonic() + 60.0)
            r3c, sc3c = logit_adam_exact(p3c, dl3c, rng, lr=0.004,
                                          top_k_init=10, top_k_min=1,
                                          top_k_decay=0.9999, restart_period=300)
            sc3cf = compute_score(r3c)
            update(r3c, sc3cf)
            print(f"  After final Adam: score={sc3cf:.10f}")

    # Try adjacent n values
    if best_seq is not None and time.monotonic() < phase3_deadline - 10:
        for delta in [-128, -64, 64, 128, 256]:
            n_adj = best_seq.size + delta
            if n_adj < 128 or n_adj > 8192:
                continue
            p_adj = interpolate_profile(best_seq, n_adj)
            sc_adj = compute_score(p_adj)
            update(p_adj, sc_adj)
            if sc_adj < best_score + 0.005:
                dl_adj = min(phase3_deadline, time.monotonic() + 20.0)
                r_adj, _ = logit_adam_exact(p_adj, dl_adj, rng, lr=0.005,
                                             top_k_init=10, top_k_min=1,
                                             top_k_decay=0.9999)
                sc_adj_f = compute_score(r_adj)
                update(r_adj, sc_adj_f)
                print(f"  Adjacent n={n_adj}: score={sc_adj_f:.10f}")

    print(f"  Phase 3 best: score={best_score:.10f}")

    # ===== Phase 4: Final validation =====
    print(f"\n--- Phase 4: Final validation ---")

    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_initial_profiles(512, rng)[0]

    # Ensure validity
    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"evaluate_sequence returned: {eval_score}")
    except NameError:
        print("evaluate_sequence not available for validation")
    except Exception as e:
        print(f"Evaluation check error: {e}")

    return [float(x) for x in final_seq]
