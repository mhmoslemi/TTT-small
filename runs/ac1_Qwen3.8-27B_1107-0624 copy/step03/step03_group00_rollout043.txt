"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) diverse initialization, (2) smooth-max Adam with adaptive beta,
(3) LP refinement with golden-section line search, (4) coordinate descent
on peak-contributing pairs, (5) multi-restart with annealing, (6) final polish.
Key improvements over previous: line search after LP, adaptive beta scheduling,
coordinate descent phase, and more aggressive multi-n exploration.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _line_search_lp(a, b_norm, n_iter=40):
    """Golden-section line search for LP direction."""
    n = len(a)
    c_a = _full_convolve(a)
    c_b = _full_convolve(b_norm)
    # Cross terms: c_ad[k] = sum_i a[i]*b[k-i]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b_norm, N)
    c_ab = np.fft.irfft(fa * fb[::-1], N)[:2 * n - 1]
    # c[k](t) = (1-t)^2 * c_a[k] + 2*t*(1-t)*c_ab[k] + t^2 * c_b[k]
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    d = hi - lo
    x1 = lo + d * (1 - gr)
    x2 = lo + d * gr

    def f_of_t(t):
        val = (1 - t) ** 2 * c_a + 2 * t * (1 - t) * c_ab + t ** 2 * c_b
        return float(val.max())

    f1 = f_of_t(x1)
    f2 = f_of_t(x2)
    for _ in range(n_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            d = hi - lo
            x1 = lo + d * (1 - gr)
            f1 = f_of_t(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            d = hi - lo
            x2 = lo + d * gr
            f2 = f_of_t(x2)
    t_opt = (lo + hi) / 2.0
    return t_opt, f_of_t(t_opt)


def _lp_refine(a, deadline, max_solves=50):
    """LP refinement with line search."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    stale = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        # Line search for best step
        t_opt, f_opt = _line_search_lp(a, b_norm, n_iter=30)
        a_new = (1.0 - t_opt) * a + t_opt * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score - 1e-12:
            current_score = sc_new
            current_best = a_new.copy()
            stale = 0
        else:
            stale += 1
            if stale > 5:
                # Try with a reduced M to allow more aggressive moves
                M = M * 0.95
                stale = 0
                n_solves = 0
                if n_solves > 200:
                    break
                continue
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                no_imp_limit=300, lp_every=800, base_lr=None):
    """Smooth-max Adam in logit space with periodic LP."""
    if base_lr is None:
        base_lr = lr
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive beta
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.4)
                elif gap > beta * 2.0:
                    beta = min(maxc * 0.3, beta * 1.5)

        # Periodic LP step
        if lp_every > 0 and t_adam % lp_every == 0 and t_adam > 0:
            lp_deadline = min(deadline, time.monotonic() + 30.0)
            if lp_deadline > time.monotonic() + 5:
                p_lp, sc_lp = _lp_refine(p, lp_deadline, max_solves=5)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_p = p_lp.copy()
                    z = np.log(np.maximum(p_lp, 1e-12))
                    z -= z.mean()
                    best_z = z.copy()
                    p = p_lp.copy()
                    c = _full_convolve(p)
                    maxc = float(c.max())
                    beta = max(1e-3, beta_frac * maxc)
                    m.fill(0.0)
                    v.fill(0.0)
                    t_adam = 0
                    lr = base_lr
                    no_imp = 0

        if no_imp > no_imp_limit:
            # Restart with noise
            if best_score < 1.51:
                ns = 0.06
            elif best_score < 1.53:
                ns = 0.12
            else:
                ns = 0.2
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _coordinate_descent(a, deadline, rng, max_rounds=50):
    """Coordinate descent: optimize pairs of coordinates to reduce peak."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    round_num = 0

    while time.monotonic() < deadline and round_num < max_rounds:
        round_num += 1
        c = _full_convolve(current_best)
        maxc = float(c.max())
        # Find top peak positions
        peak_idx = int(np.argmax(c))
        # Coordinates that contribute most to the peak
        # c[peak_idx] = sum_i a[i]*a[peak_idx - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = current_best[i] * current_best[j]
        # Pick the two largest contributors (different indices)
        top_idx = np.argsort(contribs)[::-1][:10]
        improved = False
        for i in range(min(5, len(top_idx))):
            ii = top_idx[i]
            # Find best partner jj to swap with
            best_jj = -1
            best_jj_val = current_score
            # Try a few random partners plus the second largest contributor
            candidates = [int(top_idx[j]) for j in range(len(top_idx)) if top_idx[j] != ii][:3]
            if len(candidates) < 3:
                candidates.extend(rng.integers(0, n, size=3 - len(candidates)).tolist())
            for jj in candidates:
                if jj == ii:
                    continue
                # Optimize the pair (ii, jj) keeping sum constant
                s = current_best[ii] + current_best[jj]
                if s < 1e-15:
                    continue
                # Binary search / golden section on alpha = a[ii]
                lo_a = 0.0
                hi_a = s
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                best_alpha = current_best[ii]
                best_f = best_jj_val
                for _ in range(25):
                    d = hi_a - lo_a
                    x1 = lo_a + d * (1 - gr)
                    x2 = lo_a + d * gr
                    # Evaluate f(x1)
                    a_test1 = current_best.copy()
                    a_test1[ii] = x1
                    a_test1[jj] = s - x1
                    f1 = float(_full_convolve(a_test1).max())
                    # Evaluate f(x2)
                    a_test2 = current_best.copy()
                    a_test2[ii] = x2
                    a_test2[jj] = s - x2
                    f2 = float(_full_convolve(a_test2).max())
                    if f1 < f2:
                        hi_a = x2
                        best_f = min(best_f, f1)
                        best_alpha = x1
                    else:
                        lo_a = x1
                        best_f = min(best_f, f2)
                        best_alpha = x2
                if best_f < best_jj_val - 1e-12:
                    best_jj_val = best_f
                    best_jj = jj
                    best_alpha_val = best_alpha
            if best_jj >= 0:
                current_best[ii] = best_alpha_val
                current_best[jj] = s - best_alpha_val
                current_best = np.maximum(current_best, 0.0)
                current_best = current_best / current_best.sum()
                new_score = _score_norm(current_best)
                if new_score < current_score:
                    current_score = new_score
                    improved = True

        if not improved:
            # Try random pairs
            for _ in range(10):
                ii = int(rng.integers(0, n))
                jj = int(rng.integers(0, n))
                if ii == jj:
                    continue
                s = current_best[ii] + current_best[jj]
                if s < 1e-15:
                    continue
                lo_a = 0.0
                hi_a = s
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                best_alpha = current_best[ii]
                best_f = current_score
                for _ in range(20):
                    d = hi_a - lo_a
                    x1 = lo_a + d * (1 - gr)
                    x2 = lo_a + d * gr
                    a_t1 = current_best.copy()
                    a_t1[ii] = x1
                    a_t1[jj] = s - x1
                    f1 = float(_full_convolve(a_t1).max())
                    a_t2 = current_best.copy()
                    a_t2[ii] = x2
                    a_t2[jj] = s - x2
                    f2 = float(_full_convolve(a_t2).max())
                    if f1 < f2:
                        hi_a = x2
                        best_f = min(best_f, f1)
                        best_alpha = x1
                    else:
                        lo_a = x1
                        best_f = min(best_f, f2)
                        best_alpha = x2
                if best_f < current_score - 1e-12:
                    current_best[ii] = best_alpha
                    current_best[jj] = s - best_alpha
                    current_best = np.maximum(current_best, 0.0)
                    current_best = current_best / current_best.sum()
                    current_score = _score_norm(current_best)
                    improved = True
        if not improved:
            break
    return current_best, current_score


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_r_cos(n, alpha):
    """Raised cosine: a[i] = 1 + alpha*cos(pi*(i/n - 0.5))"""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_parab(n, alpha):
    """Parabolic: a[i] = 1 + alpha*(1 - 4*(t-0.5)^2)"""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (1.0 - 4.0 * (t - 0.5) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

            ('twobump2', _make_profile_two_bump(n, 0.25, 0.75, 0.2)),
            ('cos1', _make_profile_cosine(n, 1, 0.8)),
            ('cos2', _make_profile_cosine(n, 2, 0.6)),
            ('cos3', _make_profile_cosine(n, 3, 0.5)),
            ('rcos0.5', _make_profile_r_cos(n, 0.5)),
            ('rcos0.8', _make_profile_r_cos(n, 0.8)),
            ('parab0.5', _make_profile_parab(n, 0.5)),
            ('parab1.0', _make_profile_parab(n, 1.0)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in n_list:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Polish top candidates with Adam ---
    phase2_end = t_start + 600.0
    polished = []
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end - 30 or count >= 8:
            break
        per_end = min(phase2_end - 10, time.monotonic() + 55.0)
        if per_end - time.monotonic() < 15:
            break
        p_b, sc_b = _adam_logit(p, per_end, rng, lr=0.05, beta_frac=0.1,
                                beta_decay=0.93, no_imp_limit=250, lp_every=1500)
        _update(p_b, sc_b, f'adam_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_pol'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 3: LP refinement on best ---
    phase3_end = t_start + 1000.0
    if best_seq is not None and best_seq.size <= 2048:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")
        p_lp, sc_lp = _lp_refine(best_seq, min(phase3_end - 20, time.monotonic() + 200),
                                  max_solves=40)
        _update(p_lp, sc_lp, 'LP_refine')

        # Adam polish after LP
        if time.monotonic() < phase3_end - 30:
            per_end = min(phase3_end - 10, time.monotonic() + 80.0)
            p_ap, sc_ap = _adam_logit(p_lp, per_end, rng, lr=0.03, beta_frac=0.06,
                                       beta_decay=0.94, no_imp_limit=300, lp_every=0)
            _update(p_ap, sc_ap, 'post_LP_polish')

    # --- Phase 4: Coordinate descent ---
    phase4_end = t_start + 1350.0
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"  Coord descent, score={best_score:.10f}")
        p_cd, sc_cd = _coordinate_descent(best_seq, min(phase4_end - 10, time.monotonic() + 200),
                                           rng, max_rounds=30)
        _update(p_cd, sc_cd, 'coord_desc')

        # Follow with short Adam
        if time.monotonic() < phase4_end - 20:
            per_end = min(phase4_end - 5, time.monotonic() + 60.0)
            p_a2, sc_a2 = _adam_logit(p_cd, per_end, rng, lr=0.03, beta_frac=0.05,
                                       beta_decay=0.95, no_imp_limit=200, lp_every=0)
            _update(p_a2, sc_a2, 'post_cd_polish')

    # --- Phase 5: Escalate n if beneficial ---
    phase5_end = t_start + 1550.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase5_end - 40:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            per = min(50.0, phase5_end - time.monotonic() - 20)
            if per > 15:
                p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                         lr=0.04, beta_frac=0.07, beta_decay=0.93,
                                         no_imp_limit=200, lp_every=0)
                _update(p_e, sc_e, f'escalate_n{tn}')

    # --- Phase 6: Multi-restart ---
    phase6_end = deadline - 40.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 5:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.05 + 0.05 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per, rng, lr=0.04,
                                     beta_frac=0.06, beta_decay=0.94, no_imp_limit=
                                     int(0.25 * n_best), lp_every=0)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 7: Final LP + polish ---
    if time.monotonic() < deadline - 10 and best_seq is not None:
        if best_seq.size <= 2048:
            per = min(60.0, deadline - time.monotonic() - 5)
            if per > 20:
                p_fl, sc_fl = _lp_refine(best_seq, time.monotonic() + per * 0.5,
                                          max_solves=15)
                _update(p_fl, sc_fl, 'final_LP')
                if time.monotonic() < deadline - 5:
                    per2 = min(30.0, deadline - time.monotonic() - 3)
                    if per2 > 10:
                        p_fp, sc_fp = _adam_logit(p_fl, time.monotonic() + per2, rng,
                                                   lr=0.015, beta_frac=0.04,
                                                   beta_decay=0.96, no_imp_limit=100,
                                                   lp_every=0)
                        _update(p_fp, sc_fp, 'final_polish')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512")
    return [float(x) for x in fb]
