import numpy as np
import time
from scipy.optimize import linprog, minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project vector v onto {x : sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

    #      sum(b) = 1
    #      b >= 0
    #      (t is free, but we can bound it)

def _next_pow2(n):
    k = 1

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()

                    else:
                        lo = mid1

                x_opt = (lo + hi) / 2.0

            a[i] = x_opt

        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a


def _lp_refine_combined(a, deadline, max_iter=2000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    blend = 0.25
    inflate = 1.0003

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        if it % 3 == 0:
            b_dir, _ = _lp_minmax_direction(a, top_k=120)
        else:
            b_dir = _lp_spread_direction(a, M * inflate, top_k=80)

        if b_dir is None:
            no_imp += 3
            if no_imp % 20 == 0:
                inflate = min(1.01, inflate * 1.15)
            if no_imp > 300:
                break
            continue

        b_dir = np.maximum(b_dir, 0)
        s_b = b_dir.sum()
        if s_b < 1e-10:
            no_imp += 1
            continue
        b_dir = b_dir / s_b

        a_new = (1.0 - blend) * a + blend * b_dir
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.05, blend * 0.9)
                inflate = min(1.008, inflate * 0.95)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend = 0.25
                inflate = 1.0003
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_twisted_cosine(n, freq=2.0, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_fourier_profile(n, coeffs=None, rng=None):
    if coeffs is None:
        if rng is None:
            rng = np.random.default_rng(0)
        coeffs = {0: 1.0, 1: 0.2, 2: 0.1, 3: -0.05}
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for k, c in coeffs.items():
        a += c * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    def get_candidates(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for _ in range(2):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.1 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                if p_pert.sum() > 1e-10:
                    cands.append(p_pert / p_pert.sum())
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_twisted_cosine(n_t, 2.0, 0.3))
        cands.append(_make_twisted_cosine(n_t, 3.0, 0.25))
        cands.append(_make_fourier_profile(n_t, {0: 1.0, 1: 0.3, 2: -0.15, 4: 0.08}, rng))
        return cands

    best_per_length = {}

    print("\n=== Phase 1: Nesterov LSE + Proximal Gradient ===")
    for n_t in sorted(lengths, key=lambda x: -x):
        if time.monotonic() > deadline - 300:
            break

        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            remaining = deadline - time.monotonic() - 300
            if remaining < 30:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            nesterov_budget = min(180.0, remaining * 0.4)
            if nesterov_budget > 15:
                ne_end = time.monotonic() + nesterov_budget
                a_ne = _nesterov_lse_optimize(a0, ne_end, max_iter=50000,
                                              T_start=0.04, T_end=1e-5, lr=0.08)
                sc_ne = _score(a_ne)
                if np.isfinite(sc_ne) and sc_ne < best_score:
                    best_score = sc_ne
                    best_seq = a_ne.copy()
                if sc_ne < best_here_score:
                    best_here_score = sc_ne
                    best_here = a_ne
                print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> Nesterov LSE={sc_ne:.10f}")

            else:
                best_here_score = sc0
                best_here = a0

            prox_budget = min(120.0, deadline - time.monotonic() - 300)
            if prox_budget > 15 and best_here is not None:
                pr_end = time.monotonic() + prox_budget
                a_pr = _proximal_gradient_optimize(best_here, pr_end, max_iter=20000,
                                                   T_start=0.06, T_end=0.005, lr_start=0.12)
                sc_pr = _score(a_pr)
                if np.isfinite(sc_pr) and sc_pr < best_score:
                    best_score = sc_pr
                    best_seq = a_pr.copy()
                if sc_pr < best_here_score:
                    best_here_score = sc_pr
                    best_here = a_pr
                print(f"    -> Proximal: {sc_pr:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t} after Phase 1: {best_here_score:.10f}")

    print("\n=== Phase 2: LP Refinement (min-max + spread) ===")
    for n_t in sorted(best_per_length.keys(), key=lambda x: -x):
        if time.monotonic() > deadline - 200:
            break

        a_in = best_per_length[n_t]
        lp_budget = min(200.0, deadline - time.monotonic() - 200)
        if lp_budget < 20:
            continue

        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_refine_combined(a_in, lp_end, max_iter=2000)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        if sc_lp < _score(best_per_length.get(n_t, a_in)):
            best_per_length[n_t] = a_lp
        print(f"  LP n={n_t}: score={sc_lp:.10f}")

    print("\n=== Phase 3: Coordinate Descent Polish ===")
    if best_seq is not None:
        cd_budget = min(200.0, deadline - time.monotonic() - 100)
        if cd_budget > 20:
            cd_end = time.monotonic() + cd_budget
            a_cd = _coordinate_descent(best_seq, cd_end, max_sweeps=150)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc_cd:.10f}")

    print("\n=== Phase 4: Second round Nesterov LSE on best ===")
    if best_seq is not None:
        ne2_budget = min(150.0, deadline - time.monotonic() - 60)
        if ne2_budget > 15:
            ne2_end = time.monotonic() + ne2_budget
            a_ne2 = _nesterov_lse_optimize(best_seq, ne2_end, max_iter=40000,
                                           T_start=0.02, T_end=1e-6, lr=0.05)
            sc_ne2 = _score(a_ne2)
            if np.isfinite(sc_ne2) and sc_ne2 < best_score:
                best_score = sc_ne2
                best_seq = a_ne2.copy()
            print(f"  Nesterov LSE round 2: score={sc_ne2:.10f}")

    print("\n=== Phase 5: Final LP polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp2_budget = min(80.0, deadline - time.monotonic() - 30)
        if lp2_budget > 15:
            lp2_end = time.monotonic() + lp2_budget
            a_lp2 = _lp_refine_combined(best_seq, lp2_end, max_iter=1500)
            sc_lp2 = _score(a_lp2)
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()
            print(f"  Final LP: score={sc_lp2:.10f}")

    print("\n=== Phase 6: Final coordinate polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        cd2_budget = min(60.0, deadline - time.monotonic() - 15)
        if cd2_budget > 10:
            cd2_end = time.monotonic() + cd2_budget
            a_cd2 = _coordinate_descent(best_seq, cd2_end, max_sweeps=80)
            sc_cd2 = _score(a_cd2)
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
            print(f"  Final coord descent: score={sc_cd2:.10f}")

    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
