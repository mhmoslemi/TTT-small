def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)

def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. convolve(f, g) <= M, g >= 0"""
    # Build constraint matrix A where A[k, j] = f[k-j] if valid
    # convolve(f, g)[k] = sum_j f[k-j] * g[j] (with valid indices)
    # Actually: (f*g)[k] = sum_j f[j] * g[k-j] = sum_j f[k-j] * g[j]
    # For np.convolve(f, g): result[k] = sum_i f[i] * g[k-i]
    # = sum_i g[k-i] * f[i]
    # So for fixed k, result[k] = sum_j g[j] * f[k-j] where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # But also 0 <= j < n.
    
    # Let's use a sparse-like approach: for each j (variable), the coefficient in row k is f[k-j] if 0 <= k-j < n
    # Row k has non-zero entries for j in [max(0, k-n+1), min(k, n-1)]
    
    # For efficiency with large n, use the full matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in rows k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        # In those rows, the coefficient is f[k-j]
        A[j:j+n, j] = f
    
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of a*a using log-sum-exp."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_direct(a):
    """Computes max of convolution assuming sum=1, returns 2n*max."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    return a / a.sum()


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):

    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval

            if no_imp > max(300, 4 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                consecutive_no_imp = 0

    return best_a


def _gradient_polish(a0, deadline, max_iter=60000, seed=99):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-7, 0.01 * float(c.max()))
    lr = 0.08
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    no_imp = 0
    rng = np.random.default_rng(seed)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, maxc = _smooth_max_grad(a, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-7, beta * 2.0)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = _project_simplex(a_new)
        a = a_new

        score = float(2.0 * n * _fft_convolve(a).max())

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.01, lr * 0.85)
            if no_imp > max(8000, 15 * n):
                # Restart
                n_best = len(best_a)
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.08
                beta = max(1e-9, 0.01 * float(_fft_convolve(a).max()))
                no_imp = 0
                break

        if it % 500 == 499:
            beta = max(1e-10, beta * 0.8)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: LP refinement from multiple initializations and lengths.
    Phase 2: Adam gradient polish.
    Phase 3: Final tight LP pass.
    Phase 4: Final gradient polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Build initializations
    configs = []
    lengths = [512, 768, 1024, 1536]

    for n_t in lengths:
        if parent is not None:
            configs.append(('parent_interp', n_t, _interpolate(parent, n_t)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('cosine', n_t, _make_cosine(n_t)))
        configs.append(('sym', n_t, _make_symmetric(n_t, 1.5)))

    if parent is not None:
        configs.append(('parent_orig', parent.size, parent.copy()))

    # Score and sort
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Top 5 initializations:")
    for i, (sc, name, n_t, _) in enumerate(scored[:5]):
        print(f"  {i}: {name} n={n_t} score={sc:.8f}")

    # Phase 1: LP on best configs
    phase1_fraction = 0.50
    phase1_end = t_start + budget_s * phase1_fraction

    for i, (sc0, name, n_t, a0) in enumerate(scored):
        if time.monotonic() > phase1_end:
            break
        if sc0 >= best_score + 0.03:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break

        if i == 0:
            lp_time = min(remaining * 0.45, 350.0)
        elif i == 1:
            lp_time = min(remaining * 0.25, 200.0)
        elif i == 2:
            lp_time = min(remaining * 0.12, 100.0)
        else:
            lp_time = min(remaining * 0.08, 60.0)

        if lp_time < 15:
            continue

        lp_end = time.monotonic() + lp_time
        print(f"\nLP phase1: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")
        t0 = time.monotonic()

        a_lp = _lp_refine(a0, lp_end, max_iter=50000,
                          inflate_start=1.002, inflate_min=1.0004,
                          seed=100 + i)
        sc_lp = _score(a_lp)
        dt = time.monotonic() - t0
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> {sc_lp:.10f} (improved, {dt:.0f}s)")
        else:
            print(f"  -> {sc_lp:.8f} (no improvement, {dt:.0f}s)")

    # Phase 2: Gradient polish on current best
    if best_seq is not None:
        phase2_end = t_start + budget_s * 0.72
        if time.monotonic() < phase2_end - 10:
            print(f"\nPhase 2: Gradient polish, budget={phase2_end - time.monotonic():.0f}s")
            t0 = time.monotonic()
            a_gp = _gradient_polish(best_seq, phase2_end - 10, max_iter=60000, seed=77)
            sc_gp = _score(a_gp)
            dt = time.monotonic() - t0
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> {sc_gp:.10f} ({dt:.0f}s)")

    # Phase 3: Final LP with very tight inflate
    if best_seq is not None:
        phase3_end = t_start + budget_s * 0.88
        if time.monotonic() < phase3_end - 10:
            print(f"\nPhase 3: Final tight LP, budget={phase3_end - time.monotonic():.0f}s")
            t0 = time.monotonic()
            a_final = _lp_refine(best_seq, phase3_end - 10, max_iter=40000,
                                 inflate_start=1.001, inflate_min=1.0002, seed=200)
            sc_final = _score(a_final)
            dt = time.monotonic() - t0
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  -> {sc_final:.10f} ({dt:.0f}s)")

    # Phase 4: Final gradient
    if best_seq is not None and time.monotonic() < deadline - 5:
        gp_end = deadline - 3
        if gp_end - time.monotonic() > 10:
            print(f"\nPhase 4: Final gradient, budget={gp_end - time.monotonic():.0f}s")
            t0 = time.monotonic()
            a_g = _gradient_polish(best_seq, gp_end, max_iter=30000, seed=300)
            sc_g = _score(a_g)
            dt = time.monotonic() - t0
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  -> {sc_g:.10f} ({dt:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)
        best_score = _score(final_seq)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
