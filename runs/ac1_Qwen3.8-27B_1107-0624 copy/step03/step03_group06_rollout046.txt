def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x, res
        return None, None
    except Exception:

def _smooth_max_gradient(a, beta):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / max(beta, 1e-12)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1].astype(np.float64), N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval, w


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

            break

        phase_end = time.monotonic() + alloc_time
        print(f"Optimizing: {name} n={n_t} init={sc0:.8f} time={alloc_time:.0f}s")

        # Run combined optimization
        a_opt = _combined_optimize(a0, phase_end, phase_time=45.0)
        sc_opt = _score(a_opt)

        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  -> New best: {sc_opt:.10f} (from {name} n={n_t})")
        else:
            print(f"  -> score={sc_opt:.8f} (no improvement)")

    # Final polish on the best solution
    if best_seq is not None and time.monotonic() < deadline - 15:
        final_time = min(120.0, deadline - time.monotonic() - 10)
        if final_time > 10:
            print(f"Final polish: budget={final_time:.0f}s, n={best_seq.size}")
            a_final = _gradient_optimize(best_seq, time.monotonic() + final_time * 0.6, max_iter=20000)
            sc_f1 = _score(a_final)
            if sc_f1 < best_score:
                best_score = sc_f1
                best_seq = a_final.copy()
                print(f"  Gradient polish: {sc_f1:.10f}")

            if time.monotonic() < deadline - 10:
                lp_time = deadline - time.monotonic() - 8
                if lp_time > 10:
                    a_final2 = _lp_refine_v2(best_seq, time.monotonic() + lp_time, max_iter=5000)
                    sc_f2 = _score(a_final2)
                    if sc_f2 < best_score:
                        best_score = sc_f2
                        best_seq = a_final2.copy()
                        print(f"  LP polish: {sc_f2:.10f}")

    # Try a few random restarts from the best if time remains
    if best_seq is not None and time.monotonic() < deadline - 20:
        for _ in range(3):
            if time.monotonic() > deadline - 15:
                break
            a_pert = _perturb(best_seq, rng, amount=0.03)
            restart_time = min(60.0, deadline - time.monotonic() - 10)
            if restart_time < 10:
                break
            a_r = _gradient_optimize(a_pert, time.monotonic() + restart_time, max_iter=5000)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart improvement: {sc_r:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profile(768, 'ramp1')
        best_score = _score(best_seq)

    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profile(768, 'ramp1')

    # Validate with evaluate_sequence
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
