import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_max_sum(a, M):
    """Maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if res.success:
        return res.x, res.slack
    return None, None


def _lp_max_min(a):
    """Maximize t s.t. conv(a,b)_k >= t for all k, sum(b)=1, b>=0."""
    n = len(a)
    c_line = np.convolve(a, a)
    M = float(c_line.max())
    # Variables: [b_0, ..., b_{n-1}, t]
    # Minimize -t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    # A*b - t <= 0 for each k  =>  [A, -I_col] @ x <= 0
    A_mat = _make_conv_matrix(a)
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_constraints)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0.0, M)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def _smooth_max_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(n), mval
    w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    val = mval + beta * float(np.log(s))
    return val, grad, mval


def _adam_optimize(a, deadline, max_iter=30000, lr=0.04):
    """Adam optimization in logit space with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.05 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            continue

        cur_lr = base_lr
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.93)

        if it % 200 == 0 and it > 100:
            top3 = np.sort(c)[::-1][:3]
            if len(top3) > 1:
                gap = top3[0] - top3[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 3.0:
                    beta = min(0.3 * c_max_new, beta * 1.3)

        if no_imp > max(3000, 6 * n):
            rng_r = np.random.default_rng(it * 104729 + 7)
            noise = 0.2 * best_score * rng_r.standard_normal(n)
            z_restart = best_z + noise
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _dual_guided_update(a, deadline, max_iter=10000, strength=0.02):
    """Use LP dual to guide targeted coordinate reductions."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * 1.0001

        A = _make_conv_matrix(a)
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M_lp),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})

        if not res.success:
            break

        # Get dual variables (shadow prices) for each constraint
        try:
            duals = res.ineqlin_slack
            # Use slack: constraints with zero slack are binding
            binding = duals < 1e-8
        except Exception:
            binding = c >= M * 0.999

        if not np.any(binding):
            no_imp += 1
            if no_imp > 100:
                break
            continue

        # Compute pressure from binding constraints
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            if binding[k]:
                # c_k = sum_j a[j] * a[k-j] (valid indices)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        pressure[i] += a[j]

        if pressure.max() < 1e-15:
            break

        # Reduce high-pressure coordinates
        a_new = a * np.exp(-strength * pressure / (pressure.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score_new = _score(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                strength = max(0.002, strength * 0.8)
            if no_imp > 500:
                break

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    1. Adam refinement on parent + alternatives
    2. Aggressive LP (max-sum) with adaptive tightening
    3. Max-min LP for flatness
    4. Dual-guided targeted updates
    5. Fine gradient polish
    6. Final LP pass
    7. Try larger length if time permits
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine primary length
    if parent is not None:
        primary_n = parent.size
    else:
        primary_n = 768
        best_seq = _make_linear_profile(primary_n, 1.5)
        best_score = _score(best_seq)
        print(f"No parent, using linear profile n={primary_n}, score={best_score:.10f}", flush=True)

    # Phase 1: Adam refinement
    phase1_deadline = t_start + 180.0
    print(f"\nPhase 1: Adam refinement (budget {phase1_deadline - t_start:.0f}s)", flush=True)

    # Start from parent
    a_work = best_seq.copy() if best_seq is not None else _make_linear_profile(primary_n, 1.5)
    a_opt = _adam_optimize(a_work, phase1_deadline, max_iter=30000, lr=0.04)
    sc = _score(a_opt)
    print(f"  Adam result: {sc:.10f}", flush=True)
    if sc < best_score:
        best_score = sc
        best_seq = a_opt.copy()
    a_opt = best_seq.copy()

    # Also try a flat-spectrum init quickly
    a_flat = _make_flat_spectrum_init(primary_n, rng)
    sc_flat = _score(a_flat)
    print(f"  Flat spectrum init: {sc_flat:.10f}", flush=True)
    if sc_flat < best_score * 1.05:
        phase1b_deadline = min(deadline - 900.0, t_start + 300.0)
        a_flat_opt = _adam_optimize(a_flat, phase1b_deadline, max_iter=15000, lr=0.04)
        sc_flat_opt = _score(a_flat_opt)
        print(f"  Flat spectrum after Adam: {sc_flat_opt:.10f}", flush=True)
        if sc_flat_opt < best_score:
            best_score = sc_flat_opt
            best_seq = a_flat_opt.copy()
            a_opt = a_flat_opt.copy()

    # Phase 2: Main LP refinement (most important phase)
    phase2_deadline = min(deadline - 400.0, t_start + 900.0)
    print(f"\nPhase 2: LP max-sum refinement (budget {phase2_deadline - time.monotonic():.0f}s)", flush=True)

    a_lp = _lp_refine_max_sum(a_opt, phase2_deadline, max_iter=50000,
                              inflate_init=1.0003, blend_init=0.12)
    sc_lp = _score(a_lp)
    print(f"  LP max-sum result: {sc_lp:.10f}", flush=True)
    if sc_lp < best_score:
        best_score = sc_lp
        best_seq = a_lp.copy()
    a_opt = best_seq.copy()

    # Phase 3: Max-min LP for flatness
    phase3_deadline = min(deadline - 300.0, time.monotonic() + 200.0)
    if time.monotonic() < phase3_deadline:
        print(f"\nPhase 3: LP max-min (budget {phase3_deadline - time.monotonic():.0f}s)", flush=True)
        a_mm = _lp_refine_max_min(a_opt, phase3_deadline, max_iter=2000, blend=0.15)
        sc_mm = _score(a_mm)
        print(f"  LP max-min result: {sc_mm:.10f}", flush=True)
        if sc_mm < best_score:
            best_score = sc_mm
            best_seq = a_mm.copy()
        a_opt = best_seq.copy()

    # Phase 4: Dual-guided updates
    phase4_deadline = min(deadline - 200.0, time.monotonic() + 150.0)
    if time.monotonic() < phase4_deadline:
        print(f"\nPhase 4: Dual-guided updates (budget {phase4_deadline - time.monotonic():.0f}s)", flush=True)
        a_dg = _dual_guided_update(a_opt, phase4_deadline, max_iter=5000, strength=0.015)
        sc_dg = _score(a_dg)
        print(f"  Dual-guided result: {sc_dg:.10f}", flush=True)
        if sc_dg < best_score:
            best_score = sc_dg
            best_seq = a_dg.copy()
        a_opt = best_seq.copy()

    # Phase 5: Fine gradient polish
    phase5_deadline = min(deadline - 100.0, time.monotonic() + 120.0)
    if time.monotonic() < phase5_deadline:
        print(f"\nPhase 5: Fine gradient polish (budget {phase5_deadline - time.monotonic():.0f}s)", flush=True)
        a_fine = _gradient_polish(a_opt, phase5_deadline, max_iter=20000)
        sc_fine = _score(a_fine)
        print(f"  Fine polish result: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
        a_opt = best_seq.copy()

    # Phase 6: Final LP pass (tight parameters)
    phase6_deadline = min(deadline - 30.0, time.monotonic() + 200.0)
    if time.monotonic() < phase6_deadline:
        print(f"\nPhase 6: Final LP (budget {phase6_deadline - time.monotonic():.0f}s)", flush=True)
        a_final = _lp_refine_max_sum(a_opt, phase6_deadline, max_iter=30000,
                                     inflate_init=1.0001, blend_init=0.08)
        sc_final = _score(a_final)
        print(f"  Final LP result: {sc_final:.10f}", flush=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Phase 7: Try a second round of LP+polish cycle if time allows
    if time.monotonic() < deadline - 50:
        remaining = deadline - 20 - time.monotonic()
        if remaining > 60:
            print(f"\nPhase 7: Second LP cycle (budget {remaining:.0f}s)", flush=True)
            half = remaining / 2
            a_s2 = _lp_refine_max_sum(best_seq, time.monotonic() + half,
                                      max_iter=20000, inflate_init=1.00005, blend_init=0.05)
            sc_s2 = _score(a_s2)
            if sc_s2 < best_score:
                best_score = sc_s2
                best_seq = a_s2.copy()
            a_s2 = _gradient_polish(best_seq, time.monotonic() + remaining * 0.3, max_iter=10000)
            sc_s2 = _score(a_s2)
            if sc_s2 < best_score:
                best_score = sc_s2
                best_seq = a_s2.copy()
            print(f"  Second cycle: {best_score:.10f}", flush=True)

    # Try larger length if time permits
    if time.monotonic() < deadline - 80:
        n_large = primary_n + 256
        print(f"\nTrying larger length n={n_large}...", flush=True)
        a_large = _interpolate_profile(best_seq, n_large)
        sc_large = _score(a_large)
        print(f"  Interpolated to n={n_large}: {sc_large:.10f}", flush=True)
        if sc_large < best_score * 1.02:
            large_deadline = deadline - 20
            a_large_opt = _adam_optimize(a_large, min(large_deadline, time.monotonic() + 60),
                                         max_iter=5000, lr=0.03)
            sc_large_opt = _score(a_large_opt)
            if sc_large_opt < best_score:
                a_large_lp = _lp_refine_max_sum(a_large_opt, large_deadline,
                                                max_iter=10000, inflate_init=1.0002, blend_init=0.1)
                sc_large_lp = _score(a_large_lp)
                if sc_large_lp < best_score:
                    best_score = sc_large_lp
                    best_seq = a_large_lp.copy()
                print(f"  Large n result: {sc_large_lp:.10f}", flush=True)

    # Final safety check
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final < 0.01:
        final_seq = _make_linear_profile(768, 1.5)

    # Validate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
