"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS continuation in log-softmax space.
Key insight: linear ramp (1,2,...,n) gives score ~4/3, much better than previous 1.5066.
Optimizes from this starting point with adaptive beta continuation and Adam polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Score assuming sum(a) = 1 (for comparison during optimization)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad(a, beta, n):
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(a), c, cm
    return grad, c, cm


def _objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _gradient(z, beta, n):
    a = _softmax_fn(z)
    grad, _, _ = _compute_grad(a, beta, n)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _linear_ramp(n):
    """Generate a linear ramp sequence (1, 2, ..., n) normalized."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _run_lbfgs_continuation(a_init, deadline, max_wall_time):
    """Run L-BFGS with beta continuation starting from a_init."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Beta schedule: start large, decrease
    beta_starts = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        level_start = time.monotonic()
        level_deadline = min(deadline, level_start + per_level)
        
        if time.monotonic() >= deadline:
            break

        try:
            result = minimize(
                _objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_gradient,
                options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception as e:
            pass

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.02, beta_init_frac=0.01, decay=0.9, n_max_iter=100000):
    """Adam optimization in log-softmax space with adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    beta = beta_init_frac * float(c.max())

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        grad, c, cm = _compute_grad(a, beta, n)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 100 == 0:
            beta = max(1e-12, beta * decay)

        # Adaptive learning rate
        if t % 5000 == 0 and best_score < 1.35:
            lr = max(0.002, lr * 0.9)

    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS continuation in log-softmax space.
Key insight: linear ramp (1,2,...,n) gives score ~4/3 ≈ 1.334, far better than previous 1.5066.
Optimizes from this starting point with adaptive beta continuation, multi-restart, and Adam polishing.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad(a, beta, n):
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(a)
    return grad


def _objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if beta < 1e-12:
        return 2.0 * n * cm
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _gradient(z, beta, n):
    a = _softmax_fn(z)
    grad = _compute_grad(a, beta, n)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _linear_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()
