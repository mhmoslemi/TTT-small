def spectral_flatting(a, n_iters=1000):
    n = len(a)
    a = a / a.sum()
    for i in range(n_iters):
        F = np.fft.rfft(a)
        P = np.abs(F)**2
        # Flatten: scale P to be more uniform
        # But we need to be careful about DC component (sum)
        # Target: P_flat = mean(P) * ones
        # Update: a_new = IFFT(sqrt(P_flat) * exp(i*angle(F)))
        # But this might give negative values
        
        # Alternative: threshold the power spectrum
        P_target = np.maximum(P, 0.5 * P.mean())  # compress dynamic range
        P_target = P_target / P_target.sum() * P.sum()  # preserve total energy
        
        # Reconstruct with same phase
        F_new = np.sqrt(P_target) * np.exp(1j * np.angle(F))
        a_new = np.fft.irfft(F_new, n)
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            a = a_new
    return a

2. Solve the resulting QP
3. Take a step in the solution direction
4. Iterate

"""
Polynomial profile optimization with iterative peak reduction.
Uses analytic family of polynomial profiles (degree 2-6) with grid search for
initial parameters, then fine-tunes using subgradient with line search and
smooth-max Adam. Key insight: the optimal non-negative sequence for minimizing
peak autocorrelation is close to a low-degree polynomial in the normalized
index, and this can be found efficiently via parameter search + local optimization.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a). Vectorized via FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    """Compute dc[k]/da vectorized."""
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_poly_profile(n, coeffs):
    """Create a polynomial profile: a[i] = p(u_i) / S where u_i in [0,1].
    coeffs: list of polynomial coefficients [c0, c1, c2, ...] for p(u) = c0 + c1*u + c2*u^2 + ...
    """
    u = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for j, c in enumerate(coeffs):
        a += c * (u ** j)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def _adam_optimize(a, deadline, max_iter=50000, lr=0.02, beta_factor=0.03):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr_cur = max(0.001, lr_cur * 0.95)
            if no_imp > max(5000, 8 * n):
                rng_r = np.random.default_rng(7000 + it)
                z = best_z + 0.2 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.8)

    return best_a, best_score


def _multi_peak_subgradient(a, deadline, max_iter=20000, top_k=5):
    """Subgradient targeting top-k peaks with adaptive weights."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.5 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Weight by proximity to max
        wts = np.exp(-0.5 * (c_curr[top_indices] - max_val) / (max_val * 0.01 + 1e-10))
        wts = wts / wts.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += wts[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)

        # Line search
        found = False
        for mult in [1.0, 0.5, 0.25, 0.125]:
            a_new = a - lr * mult * sg / sg_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_score - 1e-10:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                no_imp = 0
                found = True
                break
        if not found:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _spectral_flatten(a, n_iters=500, compress=0.3):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score(a)

    for it in range(n_iters):
        F = np.fft.rfft(a, N)
        P = np.abs(F)**2
        phase = np.angle(F)

        # Compress dynamic range of power spectrum
        P_floor = compress * (P.mean() + 1e-20)
        P_new = np.maximum(P, P_floor)

        # Preserve total energy (Parseval)
        P_new = P_new * (P.sum() / P_new.sum())

        # Reconstruct
        F_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, N)[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization using polynomial profile initialization,
    iterative peak reduction, spectral flattening, and Adam refinement.
    Key improvement: starts from analytically informed polynomial profiles
    (degree 2-5) that are much closer to optimal than generic initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # === Phase 1: Grid search for best polynomial profile at n=2048 ===
    print("=== Phase 1: Polynomial profile grid search (n=2048) ===")
    n_main = 2048
    d1 = min(deadline - 600, t_start + 120)
    profile_2048, prof_score = _grid_search_profile(n_main, d1)
    if profile_2048 is not None:
        print(f"  Grid search best: {prof_score:.10f}")
        if prof_score < best_score:
            best_score = prof_score
            best_seq = profile_2048.copy()

    if profile_2048 is None:
        # Fallback: linear profile
        t = np.linspace(0, 1, n_main)
        profile_2048 = (1.0 + 2.0 * t) / (1.0 + 2.0 * t).sum()

    # Also try at n=1024
    d1b = min(deadline - 500, t_start + 180)
    profile_1024, prof_score_1024 = _grid_search_profile(1024, d1b)
    if profile_1024 is not None:
        print(f"  Grid search 1024 best: {prof_score_1024:.10f}")
        if prof_score_1024 < best_score:
            best_score = prof_score_1024
            best_seq = profile_1024.copy()

    if profile_1024 is None and parent is not None:
        profile_1024 = _interpolate_profile(parent, 1024)
    elif profile_1024 is None:
        t = np.linspace(0, 1, 1024)
        profile_1024 = (1.0 + 2.0 * t) / (1.0 + 2.0 * t).sum()

    # If parent is better than our profiles, use it
    if parent is not None:
        parent_2048 = _interpolate_profile(parent, 2048)
        sc_p = _score(parent_2048)
        if sc_p < best_score:
            best_score = sc_p
            best_seq = parent_2048.copy()
        # Also blend parent with best profile
        blend = 0.5 * parent_2048 + 0.5 * profile_2048
        blend = np.maximum(blend, 0)
        blend = blend / blend.sum()
        sc_blend = _score(blend)
        if sc_blend < best_score:
            best_score = sc_blend
            best_seq = blend.copy()
            print(f"  Blend score: {sc_blend:.10f}")

    print(f"  After Phase 1: best={best_score:.10f}")

    # === Phase 2: Subgradient refinement on best profile ===
    print("=== Phase 2: Subgradient refinement ===")
    d2 = min(deadline - 400, t_start + 350)
    a_ref, sc_ref = _refine_profile_params(best_seq, d2, max_iter=30000)
    if sc_ref < best_score:
        best_score = sc_ref
        best_seq = a_ref.copy()
    print(f"  After Phase 2: best={best_score:.10f}")

    # === Phase 3: Spectral flattening ===
    print("=== Phase 3: Spectral flattening ===")
    d3 = min(deadline - 300, t_start + 450)
    a_spec = _spectral_flatten(best_seq, n_iters=200, compress=0.2)
    sc_spec = _score(a_spec)
    if sc_spec < best_score:
        best_score = sc_spec
        best_seq = a_spec.copy()
    print(f"  After Phase 3: best={best_score:.10f}")

    # === Phase 4: Adam optimization (coarse) ===
    print("=== Phase 4: Adam coarse ===")
    d4 = min(deadline - 250, t_start + 650)
    a_adam, sc_adam = _adam_optimize(best_seq, d4, max_iter=40000, lr=0.03, beta_factor=0.04)
    if sc_adam < best_score:
        best_score = sc_adam
        best_seq = a_adam.copy()
    print(f"  After Phase 4: best={best_score:.10f}")

    # === Phase 5: Multi-peak subgradient ===
    print("=== Phase 5: Multi-peak subgradient ===")
    d5 = min(deadline - 200, t_start + 800)
    a_mp, sc_mp = _multi_peak_subgradient(best_seq, d5, max_iter=20000, top_k=5)
    if sc_mp < best_score:
        best_score = sc_mp
        best_seq = a_mp.copy()
    print(f"  After Phase 5: best={best_score:.10f}")

    # === Phase 6: Also optimize the 1024 version ===
    print("=== Phase 6: Optimize 1024 version ===")
    if time.monotonic() < deadline - 150:
        # Interpolate best to 1024 and optimize
        best_1024 = _interpolate_profile(best_seq, 1024)
        sc_1024 = _score(best_1024)
        print(f"  Interpolated 1024: {sc_1024:.10f}")

        d6 = min(deadline - 150, t_start + 1000)
        a_1024, sc_1024_opt = _adam_optimize(best_1024, d6, max_iter=30000, lr=0.02, beta_factor=0.03)
        if sc_1024_opt < best_score:
            best_score = sc_1024_opt
            best_seq = a_1024.copy()
        print(f"  After 1024 Adam: best={best_score:.10f}")

        d6b = min(deadline - 100, t_start + 1100)
        a_1024b, sc_1024b = _multi_peak_subgradient(a_1024, d6b, max_iter=15000, top_k=5)
        if sc_1024b < best_score:
            best_score = sc_1024b
            best_seq = a_1024b.copy()
        print(f"  After 1024 MultiSG: best={best_score:.10f}")

    # === Phase 7: Fine Adam on the best ===
    print("=== Phase 7: Fine Adam ===")
    if time.monotonic() < deadline - 50:
        d7 = min(deadline - 50, t_start + 1300)
        a_fine, sc_fine = _adam_optimize(best_seq, d7, max_iter=30000, lr=0.008, beta_factor=0.01)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
        print(f"  After fine Adam: best={best_score:.10f}")

    # === Phase 8: Ultra-fine subgradient ===
    print("=== Phase 8: Ultra-fine subgradient ===")
    if time.monotonic() < deadline - 20:
        d8 = deadline - 20
        a_ultra, sc_ultra = _refine_profile_params(best_seq, d8, max_iter=20000)
        if sc_ultra < best_score:
            best_score = sc_ultra
            best_seq = a_ultra.copy()
        print(f"  After ultra subgrad: best={best_score:.10f}")

    # === Phase 9: Spectral flatten + re-optimize cycle ===
    if time.monotonic() < deadline - 15:
        a_spec2 = _spectral_flatten(best_seq, n_iters=100, compress=0.15)
        sc_spec2 = _score(a_spec2)
        if sc_spec2 < best_score:
            best_score = sc_spec2
            best_seq = a_spec2.copy()
        # Quick Adam on result
        if time.monotonic() < deadline - 10:
            a_final, sc_final = _adam_optimize(best_seq, deadline - 10, max_iter=10000, lr=0.005, beta_factor=0.008)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
        print(f"  After final cycle: best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        t = np.linspace(0, 1, 512)
        best_seq = (1.0 + 2.0 * t) / (1.0 + 2.0 * t).sum()
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        t = np.linspace(0, 1, 256)
        final_seq = (1.0 + 2.0 * t) / (1.0 + 2.0 * t).sum()

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
