import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _gs_flat_spectrum(a, deadline, max_iter=5000, flatten_power=1.0):
    """
    GS-style iteration: alternate between projecting to flat magnitude spectrum
    and projecting to non-negative simplex. flatten_power controls how aggressive
    the flattening is (1.0 = full flattening, <1.0 = partial).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)

    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target magnitude: interpolate between current and flat
        non_dc = mag[1:]
        if non_dc.sum() > 0:
            flat_val = non_dc.mean()
        else:
            flat_val = mag[0] / (len(mag) - 1)

        # Blended target
        target = np.empty_like(mag)
        target[0] = mag[0]  # Keep DC
        target[1:] = (1.0 - flatten_power) * non_dc + flatten_power * flat_val

        # Adjust to preserve energy
        current_energy = np.sum(mag ** 2)
        target_energy = np.sum(target ** 2)
        if target_energy > 0 and current_energy > 0:
            scale = np.sqrt(current_energy / target_energy)
            target[1:] *= scale

        A_new = target * np.exp(1j * phase)
        A_new[0] = A[0]

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def _smooth_max_gradient_step(a, beta, lr, momentum_v):
    """One gradient step using smooth-max of the convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s_ex = float(ex.sum())
    if not np.isfinite(s_ex) or s_ex <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s_ex = 1.0
    w = ex / s_ex

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, mval, c


def _gradient_descent(a, deadline, max_iter=20000, beta_init=0.03, lr_init=0.03, momentum=0.9):
    """Smooth-max gradient descent with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()

    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score(a)
    beta = max(1e-8, beta_init * float(_fft_convolve(a).max()))
    lr = lr_init
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + v
        if np.any(a_nest < 0):
            a_nest = np.maximum(a_nest, 0)
            s = a_nest.sum()
            if s > 1e-10:
                a_nest = a_nest / s
            else:
                a_nest = a.copy()

        grad, maxc, c_arr = _smooth_max_gradient_step(a_nest, beta, lr, v)

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            beta = max(1e-8, beta * 3.0)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:

                lr = base_lr
                no_imp = 0

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
