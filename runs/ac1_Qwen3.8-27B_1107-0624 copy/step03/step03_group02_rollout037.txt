"""
Optimization of non-negative step function to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses improved Adam in logit-space, pairwise coordinate descent, variance-penalized
smooth-max, and LP refinement. Explores multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def _next_pow2(n):
    if n <= 1:
        return 2
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -40.0, 40.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_weighted_conv(a, w):
    """Gradient of sum_k w[k]*c[k] where c = conv(a,a)."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _grad_variance(a, c):
    """Gradient of var(c) = mean(c^2) - mean(c)^2, where sum(c)=(sum(a))^2."""
    n = len(a)
    m = 2 * n - 1
    # d(mean(c^2))/da[i] = (2/m) * sum_k c[k] * 2*a[k-i]
    # = (4/m) * (c * a_rev)[i]
    N = _next_pow2(3 * n)
    c_pad = np.zeros(N)
    c_pad[:2 * n - 1] = c
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fc = np.fft.rfft(c_pad, N)
    fa = np.fft.rfft(a_pad, N)
    cross = np.fft.irfft(fc * fa, N)
    h = cross[n - 1:2 * n - 1]
    return (4.0 / m) * h


def _adam_smooth_max(a_init, deadline, max_iter=30000, lr=0.03, beta_factor=0.05,
                     var_alpha=0.0, verbose=False, label=""):
    """Adam in logit-space minimizing smooth_max + var_alpha * var(c)."""
    a = np.asarray(a_init, dtype=np.float64).copy().ravel()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weights(c, beta)

        if var_alpha > 0:
            grad_a = _grad_weighted_conv(a, w) + var_alpha * _grad_variance(a, c)
        else:
            grad_a = _grad_weighted_conv(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue

        t += 1
        m_arr = b1 * m_arr + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m_arr / (1.0 - b1 ** t)
        vh = v_arr / (1.0 - b2 ** t)

        lr_cur = base_lr * 0.5 * (1.0 + np.cos(np.pi * min(it, max_iter) / max_iter))
        lr_cur = max(lr_cur, base_lr * 0.01)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.001, base_lr * 0.92)
            if no_imp > max(3000, 4 * n):
                rng_r = np.random.default_rng(7777 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m_arr = np.zeros(n)
                v_arr = np.zeros(n)
                t = 0
                base_lr = lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.75)

        if verbose and it % 5000 == 0:
            print(f"    {label} Adam it={it}: best={best_score:.10f}, beta={beta:.2e}")

    return best_a, best_score


def _pairwise_cd(a_init, deadline, max_sweeps=5000, n_evals=20, verbose=False, label=""):
    """Pairwise coordinate descent: for each pair (i, i+1), find optimal split."""
    a = np.asarray(a_init, dtype=np.float64).copy().ravel()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved_this_sweep = False

        for i in range(n):
            j = (i + 1) % n
            s = a[i] + a[j]
            if s < 1e-14:
                continue

            # Precompute the "base" convolution at affected indices
            # Affected k range: [min(i,j), max(i,j)+n-1] intersected with [0, 2n-2]
            k_lo = max(0, min(i, j))
            k_hi = min(2 * n - 2, max(i, j) + n - 1)
            k_range = np.arange(k_lo, k_hi + 1)

            # For each d in [-a[i], a[j]], compute new max
            # change[k] = 2d*(a[k-i] - a[k-j]) + d^2*([k==2i] - 2[k==i+j] + [k==2j])
            
            # Precompute a[k-i] and a[k-j] for k in k_range
            ki = k_range - i
            kj = k_range - j
            ai = np.where((ki >= 0) & (ki < n), a[ki.clip(0, n-1)], 0.0)
            ai[ki.clip(0, n-1)[(ki < 0) | (ki >= n)]] = 0.0
            # Fix: just compute properly
            ai = np.zeros_like(k_range, dtype=np.float64)
            valid_i = (ki >= 0) & (ki < n)
            ai[valid_i] = a[ki[valid_i]]
            
            aj = np.zeros_like(k_range, dtype=np.float64)
            valid_j = (kj >= 0) & (kj < n)
            aj[valid_j] = a[kj[valid_j]]
            
            # Quadratic coefficients
            quad_coeff = np.zeros_like(k_range, dtype=np.float64)
            quad_coeff[k_range == 2 * i] += 1.0
            if i + j <= 2 * n - 2:
                quad_coeff[k_range == (i + j)] -= 2.0
            quad_coeff[k_range == 2 * j] += 1.0

            base_conv = c[k_range].copy()
            linear_coeff = 2.0 * (ai - aj)

            # Evaluate at n_evals points
            d_vals = np.linspace(-a[i], a[j], n_evals)
            
            best_d = 0.0
            best_val = float(base_conv.max())  # d=0 is current

            for d in d_vals:
                if abs(d) < 1e-15:
                    continue
                new_conv = base_conv + d * linear_coeff + d * d * quad_coeff
                val = float(new_conv.max())
                if val < best_val:
                    best_val = val
                    best_d = d

            if abs(best_d) > 1e-12:
                a[i] = a[i] + best_d
                a[j] = a[j] - best_d
                # Update convolution
                d = best_d
                new_conv = base_conv + d * linear_coeff + d * d * quad_coeff
                c[k_range] = new_conv
                improved_this_sweep = True

        if not improved_this_sweep:
            no_imp += 1
            if no_imp > 200:
                break
        else:
            no_imp = 0

        if sweep % 500 == 0:
            cur_score = float(2.0 * n * c.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    c = _fft_convolve(a)
    final_score = float(2.0 * n * c.max())
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def _subgradient_opt(a_init, deadline, max_iter=10000, top_k=3, verbose=False):
    """Subgradient method targeting top-k peaks."""
    a = np.asarray(a_init, dtype=np.float64).copy().ravel()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    def _subgradient_at_max(a, k):
        sg = np.zeros(len(a))
        j1_s = max(0, k - n + 1)
        j1_e = min(k, n - 1)
        if j1_s <= j1_e:
            idx = np.arange(j1_s, j1_e + 1)
            sg[idx] += a[k - idx]
        j2_s = max(k, 0)
        j2_e = min(k + n - 1, n - 1)
        if j2_s <= j2_e:
            idx = np.arange(j2_s, j2_e + 1)
            sg[idx] += a[idx - k]
        return sg

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]

        weights = np.array([1.0 + 0.3 * i for i in range(k)])
        weights /= weights.sum()

        sg = np.zeros(n)
        for i_idx, idx in enumerate(top_idx):
            sg += weights[i_idx] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _lp_refine(a_init, deadline, max_iter=1000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a_init, dtype=np.float64).copy().ravel()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for jj in range(n):
            A[jj:jj + n, jj] = a
        b_ub = np.full(2 * n - 1, M_lp)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.03)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 80:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_structured(n, kind=0):
    t = np.linspace(0.0, 1.0, n)
    if kind == 0:
        a = 1.0 + 1.5 * (t - 0.5)
    elif kind == 1:
        u = 2.0 * t - 1.0
        a = 1.0 + 1.8 * u - 1.2 * u ** 3
    elif kind == 2:
        a = 1.0 + 1.0 * np.cos(np.pi * t)
    elif kind == 3:
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * 0.4 ** 2))
    elif kind == 4:
        a = 1.0 + 0.8 * np.sin(2 * np.pi * t)
    elif kind == 5:
        a = (t ** 0.5 + (1 - t) ** 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _full_pipeline(a0, deadline, label=""):
    """Run multi-phase optimization pipeline."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Adam with variance penalty
    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 120)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=20000, lr=0.03,
                                   beta_factor=0.05, var_alpha=0.3, label=f"{label}-P1")
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam+var: {best_sc:.10f}")

    # Phase 2: Pairwise CD
    if time.monotonic() < deadline - 25:
        d2 = min(deadline - 25, time.monotonic() + 80)
        a2, s2 = _pairwise_cd(best_a, d2, max_sweeps=3000, n_evals=20, label=f"{label}-P2")
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} PairCD: {best_sc:.10f}")

    # Phase 3: Adam fine (lower var_alpha)
    if time.monotonic() < deadline - 20:
        d3 = min(deadline - 20, time.monotonic() + 80)
        a3, s3 = _adam_smooth_max(best_a, d3, max_iter=15000, lr=0.01,
                                   beta_factor=0.02, var_alpha=0.1, label=f"{label}-P3")
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} Adam fine: {best_sc:.10f}")

    # Phase 4: Pairwise CD again
    if time.monotonic() < deadline - 15:
        d4 = min(deadline - 15, time.monotonic() + 50)
        a4, s4 = _pairwise_cd(best_a, d4, max_sweeps=2000, n_evals=25, label=f"{label}-P4")
        if s4 < best_sc:
            best_a, best_sc = a4, s2 if False else s4
        print(f"  {label} PairCD2: {best_sc:.10f}")

    # Phase 5: Subgradient
    if time.monotonic() < deadline - 10:
        d5 = min(deadline - 10, time.monotonic() + 40)
        a5, s5 = _subgradient_opt(best_a, d5, max_iter=5000, top_k=3)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} SubGrad: {best_sc:.10f}")

    # Phase 6: LP polish
    if time.monotonic() < deadline - 5:
        d6 = min(deadline - 5, time.monotonic() + 30)
        a6 = _lp_refine(best_a, d6, max_iter=500, inflate=1.001, blend=0.25)
        s6 = _score(a6)
        if s6 < best_sc:
            best_a, best_sc = a6, s6
        print(f"  {label} LP: {best_sc:.10f}")

    # Phase 7: Final Adam ultra-fine
    if time.monotonic() < deadline - 2:
        a7, s7 = _adam_smooth_max(best_a, deadline - 2, max_iter=8000, lr=0.005,
                                   beta_factor=0.005, var_alpha=0.0, label=f"{label}-P7")
        if s7 < best_sc:
            best_a, best_sc = a7, s7
        print(f"  {label} Final: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining Adam smooth-max with variance penalty,
    pairwise coordinate descent, subgradient, and LP refinement.
    Explores sequence lengths 512, 1024, and 2048.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = _make_structured(256, 0)
        print("No parent, using structured initialization")

    # Define strategies: (n, a0, time_budget)
    strategies = []

    # Strategy 1: Parent at n=1024 (highest priority - closest to previous best)
    p1024 = _interpolate_profile(parent, 1024)
    strategies.append((1024, p1024, 500, "S1"))

    # Strategy 2: Parent at n=2048 (more degrees of freedom)
    p2048 = _interpolate_profile(parent, 2048)
    strategies.append((2048, p2048, 400, "S2"))

    # Strategy 3: Diverse initializations at n=1024
    diverse_inits = []
    for kind in range(6):
        diverse_inits.append(_make_structured(1024, kind))
    # Perturbed parent
    p_pert = p1024 * (1.0 + 0.2 * rng.normal(size=1024))
    p_pert = np.maximum(p_pert, 0)
    diverse_inits.append(p_pert / p_pert.sum())
    strategies.append((1024, diverse_inits[0], 60, "S3a"))
    strategies.append((1024, diverse_inits[2], 60, "S3b"))
    strategies.append((1024, diverse_inits[3], 60, "S3c"))
    strategies.append((1024, p_pert, 60, "S3d"))

    # Strategy 4: Parent at n=512 (faster, for quick improvements)
    p512 = _interpolate_profile(parent, 512)
    strategies.append((512, p512, 200, "S4"))

    # Execute strategies in order, with time management
    for (n, a0, budget, label) in strategies:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        actual_budget = min(budget, remaining - 20)
        if actual_budget < 15:
            break

        print(f"\n=== {label}: n={n}, budget={actual_budget:.0f}s ===")
        d_end = time.monotonic() + actual_budget

        a_best, s_best = _full_pipeline(a0, d_end, label)

        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  {label} NEW BEST: {best_score:.10f}")
        else:
            print(f"  {label} final: {s_best:.10f} (not better than {best_score:.10f})")

    # Final focused optimization on the best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final Focus: current best={best_score:.10f} ===")
        best_n = best_seq.size

        # If best is at 2048, also try at 1024
        if best_n == 2048 and time.monotonic() < deadline - 40:
            best_1024 = _interpolate_profile(best_seq, 1024)
            d_ff = min(deadline - 30, time.monotonic() + 60)
            a_ff, s_ff = _full_pipeline(best_1024, d_ff, "FF-1024")
            if s_ff < best_score:
                best_score = s_ff
                best_seq = a_ff.copy()
                print(f"  Final Focus 1024: {best_score:.10f}")

        # Fine-tune the best
        if time.monotonic() < deadline - 15:
            d_fin = deadline - 10
            # Pairwise CD for fine polish
            a_pcd, s_pcd = _pairwise_cd(best_seq, min(d_fin, time.monotonic() + 40),
                                         max_sweeps=3000, n_evals=30, label="FinPCD")
            if s_pcd < best_score:
                best_score = s_pcd
                best_seq = a_pcd.copy()
            print(f"  Final PairCD: {best_score:.10f}")

        if time.monotonic() < deadline - 8:
            d_fin2 = deadline - 5
            a_fin, s_fin = _adam_smooth_max(best_seq, d_fin2, max_iter=5000,
                                             lr=0.003, beta_factor=0.003, var_alpha=0.0)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()
            print(f"  Final Adam: {best_score:.10f}")

        if time.monotonic() < deadline - 5:
            a_sub, s_sub = _subgradient_opt(best_seq, deadline - 3, max_iter=3000, top_k=2)
            if s_sub < best_score:
                best_score = s_sub
                best_seq = a_sub.copy()
            print(f"  Final SubGrad: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_structured(256, 0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_structured(256, 0)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluate error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
