def _lp_spread_fast(a, M, max_constraints=None):
    """
    LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0.
    Only use top constraints for speed.
    """
    n = len(a)
    c = _fft_convolve(a)
    
    # Build constraint matrix (sparse banded)
    # A[k, j] = a[j] if 0 <= k-j < n
    # For speed, only include rows where conv is near the max
    if max_constraints is not None:
        # Include only top max_constraints rows
        top_idx = np.argsort(c)[::-1][:max_constraints]
    else:
        top_idx = np.arange(2 * n - 1)
    
    # Build A matrix
    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = a[lo:hi+1]
    
    c_obj = -np.ones(n)
    b = np.full(len(top_idx), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _interpolate_profile(p0, n_new):

def _next_pow2(n):

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

        return np.array([])

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:

    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)

        top_idx = np.arange(n_rows)

    # Build A matrix: A[k, j] = a[j] if 0 <= k - j < n
    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[i, lo:hi + 1] = a[lo:hi + 1]

    c_obj = -np.ones(n)
    b = np.full(len(top_idx), M)

    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=3000, inflate=1.002, blend=0.2, max_rows=None):
    """LP-based peak spreading with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        g = _lp_spread(a, M_lp, max_rows=max_rows)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.04)
                blend_cur = max(0.05, blend_cur * 0.92)
            if no_imp > 300:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            blend_cur = min(0.4, blend_cur * 1.03)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.95)
                inflate_cur = min(1.008, inflate_cur * 0.97)
            if no_imp > max(150, n // 3):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _peak_shave_direct(a, deadline, max_iter=5000, n_peaks=3, strength=0.05):
    """
    Direct peak shaving: for each top peak, reduce the coordinates
    that contribute most to it, redistributing to lower-peak contributors.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute pressure on each coordinate from top peaks
        pressure = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Contribution of a[j] to peak k is a[j] * a[k-j]
            # We want to reduce a[j] where it contributes most
            for j in range(lo, hi + 1):
                other = k - j
                if 0 <= other < n:
                    pressure[j] += a[j] * a[other]

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Exponentially reduce high-pressure coordinates
        a_new = a * np.exp(-str_cur * pressure)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.3, str_cur * 1.01)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.01, str_cur * 0.92)
            if no_imp > 500:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _nesterov_subgrad(a, deadline, max_iter=50000, top_k=1):
    """Nesterov-accelerated subgradient on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    y = a.copy()
    x_prev = a.copy()
    L = 2.5
    t_count = 0
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(y)
        c_max = float(c.max())

        if top_k == 1:
            m = int(np.argmax(c))
            g = _subgradient_at(y, m)
        else:
            sort_idx = np.argsort(c)[::-1][:top_k]
            vals = c[sort_idx]
            spread = vals[0] - vals[-1] + 1e-15
            w = np.exp((vals - vals[0]) / (spread * 0.5))
            w = w / w.sum()
            g = np.zeros(n)
            for idx, wi in zip(sort_idx, w):
                g += wi * _subgradient_at(y, int(idx))

        t_count += 1
        step = 1.0 / (L * np.sqrt(t_count))

        x_new = _project_simplex(y - step * g)

        alpha = (t_count - 1) / (t_count + 1) if t_count > 1 else 0.0
        y = (1 - alpha) * x_new + alpha * (x_new - x_prev)
        y = _project_simplex(y)
        x_prev = x_new

        if it % 50 == 0:
            sc = float(2.0 * n * _fft_convolve(x_new).max())
            if sc < best_score:
                best_score = sc
                best_a = x_new.copy()
                no_imp = 0
            else:
                no_imp += 1

        if no_imp > 20000:
            break

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_flat_spectrum_init(n, rng):
    """Initialize from flat power spectrum (random phases)."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_chirp_init(n, rng):
    """Chirp-like initialization with random modulation."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.abs(np.sin(freq * t + phase)) + 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_init(n, eta=2.0):
    """Tapered (raised cosine) initialization."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1 + np.cos(np.pi * (2 * t - 1))) ** eta
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization:
    Phase 1: Diversified initialization + quick Adam
    Phase 2: Nesterov subgradient for global exploration
    Phase 3: Aggressive LP refinement
    Phase 4: Peak shaving
    Phase 5: Final fine Adam polish
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Lengths to try (prioritize parent length and larger)
    n_parent = parent.size if parent is not None else 768
    lengths = []
    if n_parent in [512, 768, 1024, 1536]:
        lengths.append(n_parent)
    for ln in [768, 1024, 1536, 512]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = lengths[:3]  # Max 3 lengths

    print(f"Lengths: {lengths}, budget={budget_s}s", flush=True)

    for n_t in lengths:
        if time.monotonic() > deadline - 150:
            break

        # Generate candidates
        candidates = []
        if parent is not None:
            candidates.append(_interpolate_profile(parent, n_t))
        candidates.append(_make_flat_spectrum_init(n_t, rng))
        candidates.append(_make_flat_spectrum_init(n_t, rng))
        candidates.append(_make_tapered_init(n_t))
        candidates.append(_make_chirp_init(n_t, rng))

        # Quick score all
        scored = []
        for c in candidates:
            sc = _score(c)
            if np.isfinite(sc):
                scored.append((sc, c))
        scored.sort(key=lambda x: x[0])
        top_cands = scored[:2]

        print(f"\n  n={n_t}: best inits={['%.6f' % s for s, _ in top_cands]}", flush=True)

        for sc0, a0 in top_cands:
            if time.monotonic() > deadline - 80:
                break
            # Skip if clearly worse and not parent length
            if sc0 > best_score * 1.05 and n_t != (n_parent if parent is not None else 768):
                continue

            # Phase A: Adam in log-space (main workhorse, ~40% of remaining)
            remaining = deadline - time.monotonic()
            adam_budget = min(remaining * 0.35, 300.0)
            adam_deadline = time.monotonic() + adam_budget
            print(f"    Adam ({adam_budget:.0f}s):", end=" ", flush=True)
            a_adam = _adam_logspace(a0, adam_deadline, max_iter=200000, lr=0.04)
            sc_adam = _score(a_adam)
            print(f"{sc_adam:.10f}", flush=True)
            if sc_adam < best_score:
                best_score = sc_adam
                best_seq = a_adam.copy()

            # Phase B: Nesterov subgradient (global exploration, ~15%)
            if time.monotonic() < deadline - 60:
                nesterov_budget = min((deadline - time.monotonic()) * 0.15, 120.0)
                if nesterov_budget > 20:
                    n_deadline = time.monotonic() + nesterov_budget
                    a_ne = _nesterov_subgrad(a_adam, n_deadline, max_iter=100000, top_k=3)
                    sc_ne = _score(a_ne)
                    print(f"    Nesterov ({nesterov_budget:.0f}s): {sc_ne:.10f}", flush=True)
                    if sc_ne < best_score:
                        best_score = sc_ne
                        best_seq = a_ne.copy()
                    # Also try from a_adam directly
                    a_ne2 = _nesterov_subgrad(a_adam, n_deadline, max_iter=50000, top_k=1)
                    sc_ne2 = _score(a_ne2)
                    if sc_ne2 < best_score:
                        best_score = sc_ne2
                        best_seq = a_ne2.copy()
                        a_ne = a_ne2

            # Phase C: LP refinement (~25%)
            if time.monotonic() < deadline - 30:
                lp_budget = min((deadline - time.monotonic()) * 0.25, 180.0)
                if lp_budget > 15:
                    lp_deadline = time.monotonic() + lp_budget
                    max_rows = min(4 * n_t, 2 * n_t - 1)  # All rows if small
                    a_lp = _lp_refine(a_adam, lp_deadline, max_iter=5000,
                                       inflate=1.003, blend=0.2, max_rows=max_rows)
                    sc_lp = _score(a_lp)
                    print(f"    LP ({lp_budget:.0f}s): {sc_lp:.10f}", flush=True)
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_seq = a_lp.copy()

            # Phase D: Peak shaving (~10%)
            if time.monotonic() < deadline - 15:
                ps_budget = min((deadline - time.monotonic()) * 0.1, 60.0)
                if ps_budget > 10:
                    ps_deadline = time.monotonic() + ps_budget
                    a_ps = _peak_shave_direct(a_adam, ps_deadline, max_iter=3000,
                                               n_peaks=3, strength=0.04)
                    sc_ps = _score(a_ps)
                    print(f"    PeakShave ({ps_budget:.0f}s): {sc_ps:.10f}", flush=True)
                    if sc_ps < best_score:
                        best_score = sc_ps
                        best_seq = a_ps.copy()

    # Final intensive polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n  Final polish on best (n={best_seq.size}, score={best_score:.10f})", flush=True)

        # Fine Adam
        if time.monotonic() < deadline - 5:
            fine_deadline = deadline - 3
            a_fine = _adam_logspace(best_seq, fine_deadline, max_iter=50000, lr=0.01)
            sc_fine = _score(a_fine)
            print(f"  Fine Adam: {sc_fine:.10f}", flush=True)
            if sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()

        # Final LP
        if time.monotonic() < deadline - 2:
            lp_deadline2 = deadline - 1
            a_final_lp = _lp_refine(best_seq, lp_deadline2, max_iter=2000,
                                    inflate=1.001, blend=0.1, max_rows=3 * best_seq.size)
            sc_fl = _score(a_final_lp)
            print(f"  Final LP: {sc_fl:.10f}", flush=True)
            if sc_fl < best_score:
                best_score = sc_fl
                best_seq = a_final_lp.copy()

    # Fallback
    if best_seq is None:
        best_seq = _make_tapered_init(768)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s <= 0.01:
        final_seq = _make_tapered_init(512)

    # Validate
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nevaluate_sequence: {sc_eval:.10f}", flush=True)
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
