"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with logsumexp smoothing, projected subgradient descent,
and randomized coordinate descent with FFT-based convolution.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score assuming sum(a) = 1: returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[int(rho_hat) - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def _subgrad_peak(a, k):
    """
    Compute the gradient of conv(a,a)[k] with respect to a.
    conv(a,a)[k] = sum_i a[i] * a[k-i]
    d/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    # First term: a[k-j] for j where 0 <= k-j < n, i.e., k-n < j <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        grad[idx] += a[k - idx]
    # Second term: a[j-k] for j where 0 <= j-k < n, i.e., k <= j < k+n
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad[idx] += a[idx - k]
    return grad


def _lbfgs_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_phase(a_init, deadline, rng, max_time):
    """Run L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    levels = len(betas)
    per_level = max_time / levels

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        this_budget = min(per_level, remaining)
        if this_budget < 2.0:
            break

        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_subgradient(a_init, deadline, lr_schedule='adaptive'):

        if step % 500 == 0:

def _run_coord_descent(a_init, deadline, rng):
    """
    Randomized coordinate descent on the simplex.
    For each coordinate, try increasing/decreasing it while adjusting
    another coordinate to maintain sum=1.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    step = 0
    noise_scale = 0.01
    
    while time.monotonic() < deadline:
        step += 1
        
        # Pick two random coordinates to swap mass between
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        
        # Try different amounts of transfer
        for delta in [noise_scale, noise_scale * 0.5, noise_scale * 0.1, 
                      -noise_scale, -noise_scale * 0.5, -noise_scale * 0.1]:
            if a[i] + delta < 0 or a[j] - delta < 0:
                continue
            a_try = a.copy()
            a_try[i] += delta

def _make_uniform(n):
    return np.ones(n) / n


def _make_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _adam_polish(a_init, deadline, rng, lr=0.05, beta_frac=0.01, n_max_iter=50000):
    """Adam optimizer in log-softmax space with logsumexp smoothing."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a, best_score

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with logsumexp smoothing, projected subgradient descent on the simplex,
Adam polish in log-softmax space, and randomized coordinate descent.
Uses FFT-based convolution for efficiency. Supports multiple sequence lengths.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = float(np.max(rho[cond]))
    idx = int(rho_hat) - 1
    theta = (cssv[idx] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def _subgrad_peak(a, k):
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        grad[idx] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad[idx] += a[idx - k]
    return grad


def _lbfgs_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_phase(a_init, deadline, max_time):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003, 0.0001, 0.00004]
    betas = [b * max_c0 for b in beta_starts]

    levels = len(betas)
    per_level = max_time / levels

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        this_budget = min(per_level, remaining)
        if this_budget < 3.0:
            break

        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_subgradient(a_init, deadline):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = 0.05
    patience = 0
    max_patience = 300
    decay_counter = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())
        score = 2.0 * n * cm

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            lr *= 0.5
            patience = 0
            decay_counter += 1
            if lr < 1e-10:
                break

        peak_thresh = cm * 0.99
        peak_mask = c >= peak_thresh
        if not np.any(peak_mask):
            peak_mask = np.zeros_like(c, dtype=bool)
            peak_mask[int(np.argmax(c))] = True

        peak_indices = np.where(peak_mask)[0]
        weights = c[peak_mask] - cm + peak_thresh + 1e-8
        weights = weights / weights.sum()

        grad = np.zeros(n, dtype=np.float64)
        for j in range(len(peak_indices)):
            k = int(peak_indices[j])
            grad += weights[j] * _subgrad_peak(a, k)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        grad_normalized = grad / grad_norm

        a_new = a - lr * grad_normalized
        a_new = _project_simplex(a_new)

        c_new = _full_convolve(a_new, a_new)
        cm_new = float(c_new.max())
        score_new = 2.0 * n * cm_new

        if score_new <= score + 1e-10:
            a = a_new
        else:
            a_new2 = a - 0.3 * lr * grad_normalized
            a_new2 = _project_simplex(a_new2)
            c_new2 = _full_convolve(a_new2, a_new2)
            if 2.0 * n * float(c_new2.max()) <= score:
                a = a_new2
            else:
                lr *= 0.7

        if best_score < 2.01 and lr > 0.01:
            lr = max(0.01, lr * 0.95)

    return best_a, best_score


def _run_coord_descent(a_init, deadline, rng):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    noise_scale = 0.02
    step = 0

    while time.monotonic() < deadline:
        step += 1

        if step % 10000 == 0:
            noise_scale *= 0.8
            if noise_scale < 1e-7:
                break

        for attempt in range(3):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if j == i:
                continue

            delta = noise_scale * (1.0 if rng.random() < 0.5 else -1.0)
            if a[i] + delta < 0 or a[j] - delta < 0:
                continue

            a_try = a.copy()
            a_try[i] += delta
            a_try[j] -= delta

            c_try = _full_convolve(a_try, a_try)
            score_try = 2.0 * n * float(c_try.max())

            if score_try < best_score:
                best_score = score_try
                best_a = a_try.copy()
                a = a_try
                break

        if step % 5000 == 0:
            print(f"    CoordDescent step {step}: score={best_score:.10f}, noise={noise_scale:.8f}")

    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _make_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p):
    t = np.linspace(0.0, 1.0, n)
    t = np.maximum(t, 1e-6)
    t2 = np.linspace(1.0, 0.0, n)
    t2 = np.maximum(t2, 1e-6)
    a = t ** p + t2 ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _adam_polish(a_init, deadline, rng, lr=0.05, beta_frac=0.01, n_max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-10, beta * 0.85)

        if t % 5000 == 0 and best_score < 2.01:
            lr = max(0.005, lr * 0.9)

    return best_a, best_score


def _score_key(item):
    return item[0]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: L-BFGS + subgradient + Adam + coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_full_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
    else:
        n_parent = 512
        parent = _make_uniform(n_parent)
        print(f"No parent, using uniform n={n_parent}")

    # --- Phase 1: Quick evaluation of diverse starting points (30s) ---
    phase1_end = min(deadline, t_start + 30.0)
    candidates = []

    pc = _score_norm_fn(parent)
    candidates.append((pc, parent.copy(), "parent"))
    if pc < best_score:
        best_score = pc
        best_seq = parent.copy()
        print(f"  INIT BEST: {best_score:.10f} (n={len(parent)})")

    profile_configs = []
    for sn in [256, 512, 1024]:
        profile_configs.append((_make_uniform(sn), f"uni_n{sn}"))
        profile_configs.append((_make_linear(sn, 1.0), f"lin1_n{sn}"))
        profile_configs.append((_make_linear(sn, 2.0), f"lin2_n{sn}"))
        profile_configs.append((_make_linear(sn, -1.5), f"linm15_n{sn}"))
        profile_configs.append((_make_quadratic(sn, 1.0, 0.3), f"q1_n{sn}"))
        profile_configs.append((_make_quadratic(sn, 0.5, -0.4), f"q2_n{sn}"))
        profile_configs.append((_make_cubic(sn, 1.5, -0.8), f"c1_n{sn}"))
        profile_configs.append((_make_bell(sn, 0.15), f"bell15_n{sn}"))
        profile_configs.append((_make_bell(sn, 0.25), f"bell25_n{sn}"))
        profile_configs.append((_make_sinusoidal(sn, 2, 0.4), f"sin2_n{sn}"))
        profile_configs.append((_make_sinusoidal(sn, 4, 0.2), f"sin4_n{sn}"))
        profile_configs.append((_make_power(sn, 2.0), f"pow2_n{sn}"))
        profile_configs.append((_make_power(sn, 3.0), f"pow3_n{sn}"))

    for p, label in profile_configs:
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, label))
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()

    candidates.sort(key=_score_key)
    print(f"Phase 1: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 2: L-BFGS continuation on top candidates (550s) ---
    phase2_end = min(deadline, t_start + 580.0)
    count = 0
    seen = set()

    for sc, p, label in candidates:
        if time.monotonic() > phase2_end:
            break
        if count >= 10:
            break
        if sc > best_score * 1.01 + 1e-6:
            break
        key = (len(p), label)
        if key in seen:
            continue
        seen.add(key)

        remaining = phase2_end - time.monotonic()
        per_time = min(60.0, remaining / max(1, 10 - count))
        if per_time < 10.0:
            break

        p_best, sc_best = _run_lbfgs_phase(p, time.monotonic() + per_time, per_time)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_best)}) [lbfgs_{label}]")
        count += 1

    print(f"Phase 2: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 3: Subgradient on best (350s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 5, time.monotonic() + 350.0)
        p_best, sc_best = _run_subgradient(best_seq, per_dl)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")
    print(f"Phase 3: best={best_score:.10f}")

    # --- Phase 4: Adam polish (300s) ---
    phase4_end = min(deadline, t_start + 1230.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 5, time.monotonic() + 300.0)
        p_best, sc_best = _adam_polish(best_seq, per_dl, rng, lr=0.03, beta_frac=0.005, n_max_iter=80000)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print(f"  NEW BEST: {best_score:.10f} [adam]")
    print(f"Phase 4: best={best_score:.10f}")

    # --- Phase 5: L-BFGS refinement (150s) ---
    phase5_end = min(deadline, t_start + 1380.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        remaining = phase5_end - time.monotonic()
        if remaining > 30:
            p_best, sc_best = _run_lbfgs_phase(best_seq, phase5_end - 5, remaining - 5)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  NEW BEST: {best_score:.10f} [lbfgs2]")
    print(f"Phase 5: best={best_score:.10f}")

    # --- Phase 6: Multi-restart with noise (120s) ---
    phase6_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = (phase6_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_scale = 0.05 + 0.03 * i
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase6_end - 10, time.monotonic() + per_restart * 0.7)
            p_best, sc_best = _run_lbfgs_phase(p_start, per_dl, min(40.0, per_restart * 0.5))
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  NEW BEST: {best_score:.10f} [restart{i}]")

    print(f"Phase 6: best={best_score:.10f}")

    # --- Phase 7: Coordinate descent + subgradient final (180s) ---
    phase7_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase7_end and best_seq is not None:
        remaining = phase7_end - time.monotonic()
        half = remaining / 2.0

        if half > 15:
            p_best, sc_best = _run_subgradient(best_seq, time.monotonic() + half)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  NEW BEST: {best_score:.10f} [subgrad2]")

        if time.monotonic() < phase7_end - 10:
            remaining2 = phase7_end - time.monotonic() - 5
            if remaining2 > 10:
                p_best, sc_best = _run_coord_descent(best_seq, time.monotonic() + remaining2, rng)
                if np.isfinite(sc_best) and sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best.copy()
                    print(f"  NEW BEST: {best_score:.10f} [coord]")

    print(f"Phase 7: best={best_score:.10f}")

    # --- Phase 8: Final fine polish (120s) ---
    phase8_end = deadline
    if time.monotonic() < phase8_end and best_seq is not None:
        remaining = phase8_end - time.monotonic()
        if remaining > 20:
            p_best, sc_best = _adam_polish(best_seq, min(phase8_end - 3, time.monotonic() + remaining * 0.6),
                                           rng, lr=0.008, beta_frac=0.001, n_max_iter=30000)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = p_best.copy()
                print(f"  NEW BEST: {best_score:.10f} [final_adam]")

            if time.monotonic() < phase8_end - 5:
                p_best, sc_best = _run_subgradient(best_seq, min(phase8_end - 3, time.monotonic() + 20))
                if np.isfinite(sc_best) and sc_best < best_score:
                    best_score = sc_best
                    best_seq = p_best.copy()
                    print(f"  NEW BEST: {best_score:.10f} [final_subgrad]")

    print(f"Phase 8: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_uniform(256)

        sc_final = _score_full_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_uniform(256)
    print(f"FALLBACK: n=256, score={_score_full_fn(fallback):.10f}")
    return [float(x) for x in fallback]
