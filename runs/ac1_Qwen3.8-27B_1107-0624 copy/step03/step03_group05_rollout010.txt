"""
Multi-scale LP iteration with adaptive inflation, diverse initializations,
gradient polish with log-sum-exp smoothing, and block perturbation restarts
to minimize 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key improvements over previous: sparse LP for larger n, multi-scale LP
direction selection, aggressive symmetry-breaking perturbations, and
adaptive time allocation focusing on most promising lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _make_conv_matrix_dense(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_solve(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    if n <= 1024:
        A = _make_conv_matrix_dense(a)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    else:
        A = _make_conv_matrix_sparse(a)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_iterate(a, deadline, max_iter=10000, init_inflate=1.002, blend=0.35):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_M = float(c.max())

    no_imp = 0
    inflate = init_inflate
    blend_cur = blend
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Try multi-scale: solve LP at different inflation levels
        best_g = None
        best_g_score = np.inf

        for inf_factor in [inflate, inflate * 1.5, inflate * 3]:
            M_try = M * inf_factor
            g = _lp_solve(a, M_try)
            if g is None:
                continue
            sum_g = float(g.sum())
            if sum_g <= 1.0 + 1e-8:
                continue
            g_norm = g / sum_g
            a_trial = (1.0 - blend_cur) * a + blend_cur * g_norm
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            c_trial = _fft_convolve(a_trial)
            sc_trial = float(2.0 * n * c_trial.max())
            if sc_trial < best_g_score:
                best_g_score = sc_trial
                best_g = a_trial.copy()

        if best_g is None:
            no_imp += 1
            if no_imp > 100:
                inflate = min(1.05, inflate * 1.1)
            if no_imp > 500:
                break
            continue

        a = best_g
        sc = best_g_score

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
            stall_count = 0
            # Tighten on success

        else:
            no_imp += 1
            stall_count += 1
            if no_imp % 30 == 0:

                inflate = min(1.03, inflate * 1.03)
            if no_imp > max(200, 2 * n // 4):
                a = best_a.copy()
                blend_cur = blend
                inflate = init_inflate
                no_imp = 0
                stall_count = 0
            if stall_count > 20:
                break

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad


def _gradient_polish(a, deadline, max_iter=20000, lr_init=0.03):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(a.size) / a.size
    a = a / s
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.04 * float(c.max()))
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0)
        s_nest = a_nest.sum()
        if s_nest < 1e-10:
            a_nest = a
        else:
            a_nest = a_nest / s_nest

        grad = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.04 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 50 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a


def _adam_polish(a, deadline, max_iter=15000, lr=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.04 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

        if no_imp > max(1500, 3 * n):
            rng_r = np.random.default_rng(it * 7 + n)
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.04 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _perturb_restart(a, rng, magnitude=0.2):
    n = len(a)
    a_pert = a * (1.0 + magnitude * rng.standard_normal(n))
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def _block_perturb(a, rng):
    n = len(a)
    a_new = a.copy()
    for _ in range(3):
        block_size = max(2, n // rng.integers(16, 64))
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue
        a_new[i:i + block_size], a_new[j:j + block_size] = \
            a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _make_ramp(n, direction=1):
    t = np.linspace(0, 1, n)
    if direction == 1:
        a = 0.5 + t
    else:
        a = 0.5 + (1 - t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exp(n, rate=2.0):
    t = np.linspace(0, 1, n)
    a = np.exp(-rate * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, center_width=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.1) / (center_width * 0.3)) ** 2) + \
        np.exp(-0.5 * ((t - 0.9) / (center_width * 0.3)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic_asym(n, shift=0.3):
    t = np.linspace(0, 1, n)
    u = t - 0.5 - shift
    a = 1.0 + 2.0 * u + 1.5 * u ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

            a_res, sc_res = _optimize_one(a0, ln, init_deadline,
                                         max_lp_iter=3000)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best (n={ln}, init {idx}): {sc_res:.10f}")

    # Final intensive polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = best_seq.size
        remaining = deadline - time.monotonic() - 20
        print(f"\n=== Final polish: n={n_best}, time={remaining:.0f}s ===")

        # Intensive LP
        lp_end = time.monotonic() + remaining * 0.4
        a_f = _lp_iterate(best_seq, lp_end, max_iter=20000,
                         init_inflate=1.0008, blend=0.25)
        sc_f = _score(a_f)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
        print(f"  Final LP: {sc_f:.10f}")

        # Gradient
        gp_end = time.monotonic() + (deadline - time.monotonic() - 20) * 0.35
        if gp_end > time.monotonic() + 10:
            a_f = _gradient_polish(best_seq, gp_end, max_iter=20000, lr_init=0.02)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  Final Grad: {sc_f:.10f}")

        # Adam
        ad_end = time.monotonic() + (deadline - time.monotonic() - 20) * 0.5
        if ad_end > time.monotonic() + 10:
            a_f = _adam_polish(best_seq, ad_end, max_iter=20000, lr=0.03)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  Final Adam: {sc_f:.10f}")

        # One more LP
        lp2_end = deadline - 10
        if lp2_end > time.monotonic() + 10:
            a_f = _lp_iterate(best_seq, lp2_end, max_iter=10000,
                             init_inflate=1.0005, blend=0.15)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  Final LP2: {sc_f:.10f}")

    # If we still have time, try perturbation restarts on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        print(f"\n=== Perturbation restarts: n={n_best} ===")
        for trial in range(5):
            if time.monotonic() > deadline - 15:
                break
            a_pert = _perturb_restart(best_seq, rng, magnitude=0.15)
            sub_dl = min(deadline - 5, time.monotonic() + (deadline - time.monotonic() - 15) / (5 - trial))
            a_res, sc_res = _optimize_one(a_pert, n_best, sub_dl, max_lp_iter=2000)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  Perturb {trial}: {sc_res:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(768, 1)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = final_seq * (1.0 / s) if s > 0 else _make_ramp(768, 1)
        s = float(final_seq.sum())

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
