    w = np.exp((c_top - c_top[0]) / beta)
    w = w / w.sum()
    
    g = np.zeros(n)
    for idx, ki in enumerate(top):
        k = int(ki)
        wi = w[idx]
        # subgradient of c_k: g[j] += 2*w_i*p[k-j] if 0<=k-j<n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            g[indices] += wi * 2.0 * p[k - indices]
    
    return g


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.1, t * 0.9)
    return current_best, current_score


def _spectral_flatten(p, target_flatness=0.3, max_iter=10):
    """Try to flatten the power spectrum of p while keeping p non-negative.
    Uses iterative amplitude modification in frequency domain.
    """
    n = len(p)
    N = _next_pow2(n)
    p_work = p.copy()
    
    for _ in range(max_iter):
        F = np.fft.rfft(p_work, N)
        amp = np.abs(F)
        phase = np.angle(F)
        
        # Target: make amplitude as flat as possible
        # Blend current amplitude with its mean
        flat = np.full_like(amp, amp.mean())
        amp_new = (1 - target_flatness) * amp + target_flatness * flat
        amp_new[0] = amp[0]  # Keep DC component
        
        # Reconstruct
        F_new = amp_new * np.exp(1j * phase)
        p_new = np.fft.irfft(F_new, N)[:n]
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-10:
            break
        p_new = p_new / s
        
        if np.allclose(p_new, p_work, atol=1e-10):
            break
        p_work = p_new
    
    return p_work


def _nesterov_subgrad(p0, deadline, rng, lr=0.02, momentum=0.9, 
                       top_k=5, beta_lag=0.1, lp_every=500,
                       flat_every=2000, no_imp_limit=500):
    """Nesterov-accelerated exact subgradient in logit space."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    p = p0.copy()
    c = _full_convolve(p)
    score = float(2.0 * n * c.max())
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = score
    no_imp = 0
    step = 0
    z_prev = z.copy()
    
    # Nesterov momentum
    v = np.zeros(n)
    mu = momentum
    
    while time.monotonic() < deadline:
        step += 1
        
        # Compute multi-lag subgradient
        g = _subgrad_multi_lag(c, p, top_k=top_k, beta=beta_lag * c.max())
        
        # Convert to logit-space gradient
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))  # Center
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            v.fill(0)
            no_imp = no_imp_limit // 2
            continue
        
        # Nesterov update
        v = mu * v - lr * gz
        z_new = z + v
        
        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            v.fill(0)
            lr = max(0.001, lr * 0.5)
            continue
        
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        p = _softmax_from_z(z_new)
        
        c = _full_convolve(p)
        score = float(2.0 * n * c.max())
        
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        z = z_new
        
        # Adaptive step size
        if no_imp > 100 and no_imp % 100 == 0:
            lr = max(0.001, lr * 0.9)
            mu = max(0.5, mu * 0.99)
        
        # Periodic LP refinement
        if lp_every > 0 and step % lp_every == 0 and time.monotonic() < deadline - 60:
            p_lp, sc_lp = _lp_refine(p, min(deadline - 60, time.monotonic() + 30), 
                                       t=0.3, max_solves=5)
            if sc_lp < best_score:
                best_score = sc_lp
                best_p = p_lp.copy()
                best_z = np.log(np.maximum(p_lp, 1e-15))
                best_z -= best_z.mean()
                no_imp = 0
            p = _normalize_nonneg((1-0.5)*p + 0.5*p_lp) if p_lp is not None else p
            c = _full_convolve(p)
            z = np.log(np.maximum(p, 1e-15))
            z -= z.mean()
            v.fill(0)
        
        # Periodic spectral flattening
        if flat_every > 0 and step % flat_every == 0:
            p_flat = _spectral_flatten(p, target_flatness=0.2, max_iter=5)
            sc_flat = _score_norm(p_flat)
            if sc_flat < best_score:
                best_score = sc_flat
                best_p = p_flat.copy()
                best_z = np.log(np.maximum(p_flat, 1e-15))
                best_z -= best_z.mean()
                no_imp = 0
                p = p_flat.copy()
                c = _full_convolve(p)
                z = np.log(p)
                z -= z.mean()
                v.fill(0)
        
        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.1 if best_score < 1.51 else 0.2
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            v.fill(0)
            lr = 0.02
            mu = 0.9
            no_imp = 0
        
    return best_p, best_score


def _adam_logit(p0, deadline, rng, lr=0.06, beta_frac=0.1, beta_decay=0.92,
                no_imp_limit=400, noise_scale=None):
    """Adam smooth-max optimization (from previous approach, kept as fallback)."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Smooth max weights
        x = (c - maxc) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s
        
        # Cross correlation for gradient
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        g = h[n - 1:2 * n - 1]
        
        gz = p * (g - float(np.dot(p, g)))
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
        
        if no_imp > no_imp_limit:
            if noise_scale is None:
                ns = 0.08 if best_score < 1.51 else 0.15
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_fourier_flat(n, n_harmonics=10, decay=0.5):
    """Create a profile with flat power spectrum (in frequency domain).
    This gives a signal whose autocorrelation is as close to delta as possible.
    """
    # In frequency domain, flat amplitude means the time-domain signal
    # has low autocorrelation (like a Barker sequence)
    # We construct: p = sum_k cos(2*pi*k*t/n + phi_k) + bias
    # with random phases to avoid periodicity
    t = np.arange(n)
    a = np.ones(n) * n_harmonics  # DC bias to ensure non-negativity
    for k in range(1, n_harmonics + 1):
        phi = 2.0 * np.pi * k * t / n
        amp = 1.0 / (k ** decay)
        a += amp * np.cos(phi)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Combines Nesterov subgradient, Adam smooth-max,
    LP refinement, spectral flattening, and multi-scale escalation."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")
    
    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")
    
    # --- Phase 1: Generate diverse initial candidates ---
    phase1_end = t_start + 180.0
    candidates = []
    
    for n in [512, 1024, 2048]:
        profs = []
        profs.append(('lin1.5', _make_profile_linear(n, 1.5)))
        profs.append(('lin2.5', _make_profile_linear(n, 2.5)))
        profs.append(('exp1.0', _make_profile_exponential(n, 1.0)))
        profs.append(('exp2.0', _make_profile_exponential(n, 2.0)))
        profs.append(('twobump1', _make_profile_two_bump(n, 0.25, 0.75, 0.2)))
        profs.append(('twobump2', _make_profile_two_bump(n, 0.3, 0.7, 0.15)))
        profs.append(('cos1', _make_profile_cosine(n, 1, 0.8)))
        profs.append(('cos2', _make_profile_cosine(n, 2, 0.6)))
        profs.append(('cos3', _make_profile_cosine(n, 3, 0.5)))
        profs.append(('fflat5', _make_profile_fourier_flat(n, 5, 0.5)))
        profs.append(('fflat10', _make_profile_fourier_flat(n, 10, 0.3)))
        profs.append(('fflat20', _make_profile_fourier_flat(n, 20, 0.2)))
        
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))
    
    if parent is not None:
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))
        for n in [512, 1024, 2048]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')
    
    # --- Phase 2: Nesterov subgradient on top candidates ---
    # Allocate time: 60% to Nesterov, 20% to Adam, 10% to LP, 10% to misc
    nest_end = t_start + 800.0
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > nest_end or count >= 8:
            break
        per_end = min(nest_end, time.monotonic() + 90.0)
        p_b, sc_b = _nesterov_subgrad(p, per_end, rng, lr=0.015, momentum=0.92,
                                       top_k=8, beta_lag=0.15,
                                       lp_every=300, flat_every=1500,
                                       no_imp_limit=400)
        _update(p_b, sc_b, f'nest_{label}_n{n}')
        count += 1
    
    print(f"After Nesterov: best={best_score:.10f}")
    
    # --- Phase 3: Adam smooth-max polish on best ---
    adam_end = t_start + 1200.0
    if best_seq is not None and time.monotonic() < adam_end:
        per = min(adam_end - time.monotonic() - 10, 120.0)
        if per > 20:
            p_a, sc_a = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.05, beta_frac=0.08, beta_decay=0.93,
                                     no_imp_limit=400)
            _update(p_a, sc_a, 'adam_polish')
    
    # --- Phase 4: LP refinement on best ---
    lp_end = t_start + 1400.0
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < lp_end:
        p_lp, sc_lp = _lp_refine(best_seq, min(lp_end, time.monotonic() + 120),
                                   t=0.35, max_solves=25)
        _update(p_lp, sc_lp, 'LP_refine')
        
        # Adam after LP
        if time.monotonic() < lp_end - 10:
            per = min(lp_end - time.monotonic() - 5, 60.0)
            if per > 15:
                p_ap, sc_ap = _adam_logit(p_lp, time.monotonic() + per, rng,
                                            lr=0.04, beta_frac=0.06, beta_decay=0.94,
                                            no_imp_limit=300)
                _update(p_ap, sc_ap, 'post_LP_adam')
    
    # --- Phase 5: Multi-scale escalation ---
    scale_end = t_start + 1600.0
    if time.monotonic() < scale_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        else:
            target_ns = []
        
        for tn in target_ns:
            if time.monotonic() > scale_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            _update(p_int, sc_int, f'interp_n{tn}')
            
            per = min(90.0, scale_end - time.monotonic() - 30)
            if per > 20:
                p_e, sc_e = _nesterov_subgrad(p_int, time.monotonic() + per, rng,
                                                lr=0.012, momentum=0.9,
                                                top_k=5, beta_lag=0.1,
                                                lp_every=400, flat_every=2000,
                                                no_imp_limit=300)
                _update(p_e, sc_e, f'escalate_n{tn}')
    
    # --- Phase 6: Multi-restart at best length ---
    restart_end = deadline - 30.0
    if time.monotonic() < restart_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 3
        per_r = max(20.0, (restart_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > restart_end:
                break
            per = min(per_r, restart_end - time.monotonic() - 5)
            if per < 15:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.08 + 0.04 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            
            # Use Nesterov for restarts
            p_r, sc_r = _nesterov_subgrad(p_s, time.monotonic() + per, rng,
                                            lr=0.015, momentum=0.9,
                                            top_k=5, beta_lag=0.1,
                                            lp_every=500, flat_every=0,
                                            no_imp_limit=300)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')
    
    # --- Phase 7: Final combined polish ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 20:
            # Try both methods and keep the best
            half = per / 2
            p_f1, sc_f1 = _nesterov_subgrad(best_seq, time.monotonic() + half * 0.8, rng,
                                              lr=0.01, momentum=0.95,
                                              top_k=10, beta_lag=0.2,
                                              lp_every=0, flat_every=0,
                                              no_imp_limit=200)
            _update(p_f1, sc_f1, 'final_nest')
            
            if time.monotonic() < deadline - 5:
                p_f2, sc_f2 = _adam_logit(best_seq, deadline - 5, rng,
                                            lr=0.02, beta_frac=0.04, beta_decay=0.96,
                                            no_imp_limit=200)
                _update(p_f2, sc_f2, 'final_adam')
    
    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
