"""
Spectral flattening + L-BFGS-B optimization for minimizing autocorrelation peak.

Key ideas:
1. Spectral flattening iteration: alternate between time-domain and frequency-domain
   to push |A(ω)|^2 toward a flat shape
2. L-BFGS-B on smooth max objective for refinement
3. Direct peak shaving with Newton-like steps
4. Multi-length, multi-start search

import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _spectral_flatten_init(n, rng, n_iter=200):
    """Initialize by iteratively flattening the power spectrum."""
    a = np.ones(n) / n
    for _ in range(n_iter):
        # Compute power spectrum
        N = _next_pow2(n)
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        # Apply flattening: multiply by 1/sqrt(P + eps) in frequency domain
        # This is like "water filling" in the frequency domain
        P_smooth = P + 1e-6
        # We want the new power spectrum to be closer to flat
        # New A' = A / sqrt(|A|^2 + eps) approximately gives flat spectrum
        # But we need to maintain non-negativity in time domain
        target = 1.0 / np.sqrt(P_smooth)
        A_new = A * target * np.sqrt(P_smooth) / np.mean(target * P_smooth)
        # Actually let's just do: A_new = A * (1/sqrt(P) normalized)
        # This gives |A_new|^2 = P / (P + eps) * const ≈ const for large P
        A_new = A / np.sqrt(P_smooth)
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
    return a

def _spectral_flatten_with_phases(n, rng, n_iter=100):
    """Spectral flattening starting from random phases."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = 1.0
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * phases[k])
    
    a = np.real(np.fft.irfft(A, N))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    for _ in range(n_iter):
        A_full = np.fft.rfft(a, N)
        P = np.abs(A_full)**2 + 1e-8
        A_flat = A_full / np.sqrt(P)
        a_new = np.real(np.fft.irfft(A_flat, N))[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
    return a

def _lbfgsb_optimize(a, deadline, max_iter=5000, beta_init=None):
    """L-BFGS-B optimization on log-space parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.03 * c_max)
    
    def objective(z):
        e = np.exp(z - np.exp(z).max())  # for numerical stability
        p = e / e.sum()
        c = _fft_convolve(p)
        val, _ = _smooth_max(c, beta)
        return 2.0 * n * val
    
    def gradient(z):
        e = np.exp(z - np.exp(z).max())
        p = e / e.sum()
        c = _fft_convolve(p)
        val, w = _smooth_max(c, beta)
        
        # Gradient of smooth max w.r.t. p
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        p_rev = p[::-1]
        fp_rev = np.fft.rfft(p_rev, N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        
        # Gradient w.r.t. z: d/dz [f(p(z))] where p = softmax(z)
        # dp_i/dz_j = p_i (delta_ij - p_j)
        # df/dz_j = sum_i (df/dp_i)(dp_i/dz_j) = p_j (sum_i df/dp_i * p_i - df/dp_j)
        # Wait, let me be more careful
        # f(p) = 2n * smooth_max(p*p)
        # df/dp_j = 2 * sum_k w_k * (p*p)_k / p_j contribution
        # Actually (p*p)_k = sum_i p_i * p_{i+k}, so d(p*p)_k/dp_j = sum_i [delta_{i,j} p_{i+k} + p_i delta_{i+k,j}]
        # = p_{k-j} + p_{j-k} (with appropriate bounds)
        # So df/dp_j = 2 * sum_k w_k * (p_{|k-j|} adjusted for bounds)
        # Which is exactly 2 * h as computed above (cross-correlation of w with p)
        
        # Now df/dz_j = sum_i (df/dp_i)(dp_i/dz_j)
        # dp_i/dz_j = p_i(δ_ij - p_j)
        # df/dz_j = (df/dp_j) * p_j - p_j * sum_i (df/dp_i) * p_i
        dot = float(np.dot(vp, p))
        gz = p * (vp - dot)
        gz = 2.0 * n * gz  # chain rule for the 2n factor
        return gz
    
    z0 = np.log(a)
    z0 = z0 - z0.max()
    
    result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                               'max_fun': max_iter})
    
    # Convert back
    z_best = result.x
    e = np.exp(z_best - z_best.max())
    a_opt = e / e.sum()
    return a_opt

def _direct_peak_reduce(a, deadline, n_peaks=10, max_iter=2000):
    """Directly reduce the top autocorrelation peaks using small Newton-like steps."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    step_size = 0.1
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        
        # Weight by excess
        threshold = np.median(c)
        weights = np.maximum(top_vals - threshold, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights = weights / wsum
        
        # For each peak at lag k, the "pressure" on coordinate i is:
        # pressure_i = sum over peaks k of w_k * a_{k-i} (for valid i)
        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]
        
        # Reduce coordinates with high pressure
        pmax = pressure.max()
        if pmax < 1e-15:
            break
        
        a_new = a * np.exp(-step_size * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            step_size = min(0.5, step_size * 1.005)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                step_size = max(0.01, step_size * 0.9)
            if no_imp > 500:
                a = best_a.copy()
                step_size = 0.1
                no_imp = 0
    
    return best_a

def _alternating_projection(a, deadline, max_iter=5000):
    """
    Alternating: 
    1. In frequency domain, flatten |A(ω)|^2
    2. In time domain, clip to non-negative and normalize
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(n)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    # Blend factor between flat spectrum and current
    alpha = 0.5
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2 + 1e-10
        
        # Target: flat power spectrum
        # New A = A * (flat_target / P)^gamma for some gamma in (0, 1)
        # gamma=1 gives perfectly flat, gamma=0 gives no change
        gamma = min(1.0, 0.3 + 0.1 * (it / max(1, max_iter//10)))
        P_target = np.full_like(P, P.mean())
        ratio = (P_target / P) ** gamma
        A_new = A * ratio
        
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 1e-15)

            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            # Try with smaller blend
            A_blend = A * (ratio ** 0.5)
            a_blend = np.real(np.fft.irfft(A_blend, N))[:n]
            a_blend = np.abs(a_blend)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _custom_smooth_optimize(a, deadline, beta_start=None, max_iter=100000):

                bb_cand = np.clip(bb_cand, 0.1, 10.0)

        prev_grad = gz.copy()
        
        cur_lr = lr * bb

        z_new = np.clip(z_new, -60.0, 60.0)

        else:

                lr = max(0.003, lr * 0.95)
        
        # Adaptive beta

            top5 = np.sort(c)[::-1][:min(5, len(c))]

                    beta = min(0.5 * c_max, beta * 1.3)
        
        if no_imp > max(5000, 10 * n):
            # Restart with noise

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key strategies:
1. Spectral flattening initialization (iterative frequency-domain equalization)
2. Multi-scale smooth max optimization (beta annealing)
3. L-BFGS-B with careful line search for refinement
4. Direct peak shaving with adaptive step sizes
5. Multi-length, multi-start search with diverse initializations
6. Alternating time-frequency domain projections

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _softmax(z):
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _gradient_smooth_max(a, beta):
    """Compute gradient of smooth max autocorrelation w.r.t. log-space params."""
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    p_rev = a[::-1]
    fp_rev = np.fft.rfft(p_rev, N)
    h_full = np.fft.irfft(fw * fp_rev, N)
    h = h_full[n - 1:2 * n - 1]
    vp = 2.0 * h
    dot = float(np.dot(vp, a))
    gz = a * (vp - dot)
    return 2.0 * n * gz, val, c


def _spectral_flatten(n, rng, n_iter=200, gamma_start=0.5, gamma_end=1.0):
    """Iteratively flatten the power spectrum in frequency domain."""
    a = np.ones(n) / n
    N = _next_pow2(n)
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2 + 1e-12
        P_target = np.full_like(P, P.mean())
        gamma = gamma_start + (gamma_end - gamma_start) * (it / max(1, n_iter - 1))
        ratio = (P_target / P) ** gamma
        A_new = A * ratio
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
    return a


def _random_phase_flatten(n, rng, n_iter=100):
    """Start from random phases, then flatten spectrum."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = 1.0
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * phases[k])
    a = np.real(np.fft.irfft(A, N))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    for it in range(n_iter):
        A_full = np.fft.rfft(a, N)
        P = np.abs(A_full) ** 2 + 1e-10
        P_target = np.full_like(P, P.mean())
        ratio = (P_target / P) ** 0.7
        A_new = A_full * ratio
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
    return a


def _make_tapered(n, alpha=0.5):
    """Create a tapered (raised-cosine-like) profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5))) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell(n, width=0.35):

    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _multiscale_smooth_optimize(a, deadline, beta_max=None, beta_min=1e-10,
                                 max_iter=200000, lr=0.05):
    """
    Multi-scale optimization: start with large beta (smooth), gradually reduce.
    Uses Barzilai-Borwein adaptive step size.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_max is None:
        beta_max = 0.1 * c_max
    beta_min = max(beta_min, 1e-8 * c_max)

    z = np.log(a)
    z = z - z.max()

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    cur_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb = 1.0
    it = 0

    while time.monotonic() < deadline:
        gz, val, c = _gradient_smooth_max(a, beta)
        if not np.all(np.isfinite(gz)):
            break

        # BB step size
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20 and abs(dot_sg) > 1e-20:
                bb_cand = dot_ss / abs(dot_sg)
                bb_cand = np.clip(bb_cand, 0.05, 20.0)
                bb = 0.6 * bb + 0.4 * bb_cand
        prev_z = z.copy()
        prev_grad = gz.copy()

        step = cur_lr * bb
        z_new = z - step * gz
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.max()
        a_new = _softmax(z_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                cur_lr = max(0.002, cur_lr * 0.93)

        # Anneal beta
        if it % 500 == 0 and it > 500:
            top3 = np.sort(c_new)[::-1][:min(3, len(c_new))]
            if len(top3) > 1:
                gap = top3[0] - top3[-1]
                if gap < beta * 0.2:
                    beta = max(beta_min, beta * 0.6)
                elif gap > beta * 20:
                    beta = min(beta_max, beta * 1.5)

        # Restart if stuck
        if no_imp > max(5000, 15 * n):

            z = best_z + 0.15 * rng_r.standard_normal(n)
            z = z - z.max()
            a = _softmax(z)
            cur_lr = lr
            bb = 1.0
            prev_grad = None
            no_imp = 0

        z = z_new
        a = a_new
        it += 1

    return best_a


def _lbfgsb_refine(a, deadline, max_iter=3000, beta=None):
    """L-BFGS-B refinement with fixed (small) beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta is None:
        beta = max(1e-8, 0.01 * c_max)

    def obj(z):
        p = _softmax(z)
        c_loc = _fft_convolve(p)
        val, _ = _smooth_max(c_loc, beta)
        return 2.0 * n * val

    def grad(z):
        p = _softmax(z)
        c_loc = _fft_convolve(p)
        val, w = _smooth_max(c_loc, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        p_rev = p[::-1]
        fp_rev = np.fft.rfft(p_rev, N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(vp, p))
        gz = 2.0 * n * p * (vp - dot)
        return gz

    z0 = np.log(a)
    z0 = z0 - z0.max()

    try:
        result = minimize(obj, z0, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 50})
        z_best = result.x
        a_opt = _softmax(z_best)
        return a_opt
    except Exception:
        return a

    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / wsum

        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]

        pmax = pressure.max()
        if pmax < 1e-15:
            break

        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score_new = _score(a_new)
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            str_cur = min(0.3, str_cur * 1.003)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                str_cur = max(0.01, str_cur * 0.92)
            if no_imp > 1000:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

    return best_a


def _freq_time_alternate(a, deadline, max_iter=3000):
    """
    Alternate between:
    - Frequency domain: flatten |A(w)|^2
    - Time domain: enforce non-negativity and normalize
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(n)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    gamma = 0.4

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2 + 1e-12
        P_mean = P.mean()
        ratio = (P_mean / P) ** gamma
        A_new = A * ratio
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            gamma = min(1.0, gamma * 1.01)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                gamma = max(0.1, gamma * 0.95)
            if no_imp > 500:
                break

    return best_a


def _lp_spread(a, M):
    """LP: maximize sum(g) subject to conv(a, g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.2):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    inf_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inf_cur
        g = _lp_spread(a, M)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inf_cur = min(1.01, inf_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.95)
            if no_imp > 300:
                a = best_a.copy()
                blend_cur = blend
                inf_cur = inflate
                no_imp = 0

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining spectral methods, gradient descent,
    L-BFGS-B, LP refinement, and peak shaving to minimize autocorrelation peak.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = [512, 768, 1024]
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths.insert(0, pn)
    lengths = sorted(set(lengths))

    print(f"Lengths to try: {lengths}", flush=True)
    print(f"Total budget: {deadline - t_start:.0f}s", flush=True)

    # Phase 1: Generate diverse initializations
    print("\n=== Phase 1: Initialization ===", flush=True)
    init_pool = []

    for n_t in lengths:
        # Parent interpolation
        if parent is not None:
            a_p = _interpolate_profile(parent, n_t)
            init_pool.append((f"parent_{n_t}", n_t, a_p))

        # Spectral flattening from uniform
        a_sf = _spectral_flatten(n_t, rng, n_iter=300, gamma_start=0.3, gamma_end=1.0)
        init_pool.append((f"flatten_{n_t}", n_t, a_sf))

        # Random phase + flatten (multiple seeds)
        for ph_iter in [50, 150, 300]:
            a_rp = _random_phase_flatten(n_t, rng, n_iter=ph_iter)
            init_pool.append((f"phase{ph_iter}_{n_t}", n_t, a_rp))

        # Tapered profiles
        for alpha in [0.5, 1.0, 2.0]:
            a_tp = _make_tapered(n_t, alpha)
            init_pool.append((f"taper{alpha}_{n_t}", n_t, a_tp))

        # Bell
        a_bl = _make_bell(n_t, 0.35)
        init_pool.append((f"bell_{n_t}", n_t, a_bl))

    # Evaluate and sort
    scored_inits = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        scored_inits.append((sc, name, n_t, a0))
    scored_inits.sort(key=lambda x: x[0])

    print("Top 10 initializations:", flush=True)
    for sc, name, n_t, _ in scored_inits[:10]:
        print(f"  {name}: {sc:.8f}", flush=True)

    if scored_inits[0][0] < best_score:
        best_score = scored_inits[0][0]
        best_seq = scored_inits[0][3].copy()

    # Phase 2: Optimize top candidates
    print("\n=== Phase 2: Optimization ===", flush=True)
    top_k = min(5, len(scored_inits))
    for idx in range(top_k):
        sc0, name, n_t, a0 = scored_inits[idx]
        if time.monotonic() > deadline - 120:
            break
        if sc0 > best_score * 1.1 and idx > 0:
            continue

        print(f"\n  [{idx}] {name} (n={n_t}), init={sc0:.8f}", flush=True)
        a_cur = a0.copy()

        # Step A: Multi-scale smooth optimization (main workhorse)
        if time.monotonic() < deadline - 100:
            d_a = min(deadline - 80, time.monotonic() + 250.0)
            print(f"    Multi-scale smooth (budget {d_a - time.monotonic():.0f}s)...", flush=True)
            a_opt = _multiscale_smooth_optimize(a_cur, d_a, beta_max=0.08 * float(_fft_convolve(a_cur).max()))
            sc = _score(a_opt)
            print(f"    -> {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
            a_cur = a_opt if sc < _score(a_cur) else a_cur

        # Step B: Frequency-time alternation
        if time.monotonic() < deadline - 80:
            d_b = min(deadline - 60, time.monotonic() + 80.0)
            print(f"    Freq-time alt (budget {d_b - time.monotonic():.0f}s)...", flush=True)
            a_ft = _freq_time_alternate(a_cur, d_b, max_iter=2000)
            sc = _score(a_ft)
            print(f"    -> {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_ft.copy()
            a_cur = a_ft if sc < _score(a_cur) else a_cur

        # Step C: Peak shaving
        if time.monotonic() < deadline - 60:
            d_c = min(deadline - 50, time.monotonic() + 50.0)
            a_pk = _peak_shave(a_cur, d_c, n_peaks=8, strength=0.06)
            sc = _score(a_pk)
            print(f"    Peak shave -> {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_pk.copy()
            a_cur = a_pk if sc < _score(a_cur) else a_cur

        # Step D: L-BFGS-B refinement
        if time.monotonic() < deadline - 40:
            d_d = min(deadline - 30, time.monotonic() + 40.0)
            a_lb = _lbfgsb_refine(a_cur, d_d, max_iter=2000, beta=0.005 * float(_fft_convolve(a_cur).max()))
            sc = _score(a_lb)
            print(f"    L-BFGS-B -> {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_lb.copy()

    # Phase 3: Intensive final optimization on the best
    print("\n=== Phase 3: Final intensive optimization ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_final = best_seq.copy()
        n_f = len(a_final)

        for rnd in range(4):
            if time.monotonic() > deadline - 20:
                break

            # Fine multiscale
            d = min(deadline - 20, time.monotonic() + 50.0)
            c_f = _fft_convolve(a_final)
            a_final = _multiscale_smooth_optimize(a_final, d, beta_max=0.02 * float(c_f.max()), beta_min=1e-9)
            sc = _score(a_final)
            print(f"  Round {rnd} fine smooth: {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()

            # Peak shave with very small strength
            if time.monotonic() < deadline - 20:
                d = min(deadline - 20, time.monotonic() + 25.0)
                a_final = _peak_shave(a_final, d, n_peaks=5, strength=0.03)
                sc = _score(a_final)
                print(f"  Round {rnd} fine peak: {sc:.10f}", flush=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_final.copy()

            # LP refine
            if time.monotonic() < deadline - 20 and n_f <= 1024:
                d = min(deadline - 20, time.monotonic() + 20.0)
                a_final = _lp_refine(a_final, d, max_iter=2000, inflate=1.0005, blend=0.1)
                sc = _score(a_final)
                print(f"  Round {rnd} LP: {sc:.10f}", flush=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_final.copy()

    # Phase 4: Try a few random restarts from best
    print("\n=== Phase 4: Random restarts ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_b = len(best_seq)
        for r in range(5):
            if time.monotonic() > deadline - 15:
                break
            noise = 0.05 * (r + 1)
            a_pert = best_seq * (1.0 + noise * rng.standard_normal(n_b))
            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / a_pert.sum()
            sc_pert = _score(a_pert)
            if sc_pert > best_score * 1.1:
                continue
            d = min(deadline - 15, time.monotonic() + 20.0)
            a_opt = _multiscale_smooth_optimize(a_pert, d, beta_max=0.02 * float(_fft_convolve(a_pert).max()))
            sc = _score(a_opt)
            print(f"  Restart {r}: {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final < 0.01:
        final_seq = np.ones(512) / 512.0

    # Authoritative evaluation
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            print(f"\nevaluate_sequence: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: best_score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic() - t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
