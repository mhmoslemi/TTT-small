    n = len(a_fixed)
    b = a_fixed.copy()
    a = a_fixed.copy()
    
    best_score = _score_normalized(a)
    best_a = a.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # min t s.t. conv(a, b)[k] <= t, a >= 0, sum(a) = 1

        # Objective: minimize t (index n)

            break

        res2 = linprog(c_obj2, A_ub=A_ub2, b_ub=np.zeros(2*n-1),
                       A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res2.success:

            new_b = np.maximum(new_b, 0)
            sb = new_b.sum()
            if sb > 1e-10:
                new_b = new_b / sb
                b = new_b

            avg = 0.5 * (a + b)
            sa2 = avg.sum()
            if sa2 > 1e-10:
                avg = avg / sa2
            score_a = _score_normalized(a)

                best_a = avg.copy()
            
            a = avg
            b = avg.copy()
    
    return best_a

"""
Hybrid optimizer combining biconvex LP (cutting plane), projected subgradient
with Polyak step, LSE gradient with Nesterov momentum, and spectral GS for
minimizing 2n * max(conv(a,a)) / sum(a)^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _fft_cross(a, b):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-14:
        return np.ones_like(v) / len(v) * target_sum
    return v * (target_sum / s)


def _biconvex_lp_cutting_plane(a_init, b_init, deadline, max_iter=300, top_k=40):
    """Biconvex LP with cutting plane (reduced constraints)."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    b = np.asarray(b_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    b = np.maximum(b, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    if b.sum() < 1e-10:
        b = np.ones(n) / n
    a = a / a.sum()
    b = b / b.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    active_a = list(range(2 * n - 1))
    active_b = list(range(2 * n - 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # --- Solve for a (b fixed) using reduced LP ---
        # Find active constraints
        cross_ab = _fft_cross(a, b)
        M_a = float(cross_ab.max())
        idx_a = np.argsort(cross_ab)[::-1][:top_k]
        active_a = sorted(set(int(i) for i in idx_a))

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((len(active_a), n + 1))
        b_ub = np.full(len(active_a), M_a * 1.0001)
        for row, k in enumerate(active_a):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    A_ub[row, i] = b[j]
            A_ub[row, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

        bounds = [(0, None)] * (n + 1)

        if res.success:

        # --- Solve for b (a fixed) using reduced LP ---

        idx_b = np.argsort(cross_ba)[::-1][:top_k]
        active_b = sorted(set(int(i) for i in idx_b))

        b_ub2 = np.full(len(active_b), M_b * 1.0001)

        if s_avg > 1e-10:

def _subgradient_polyak(a_init, deadline, max_iter=50000, momentum=0.7):
    """Projected subgradient with Polyak step size and momentum."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(c.max())
    lower_bound = 1.0 / (2 * n - 1)

    v = np.zeros(n)
    no_imp = 0
    momentum_cur = momentum

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        gap = m_val - lower_bound
        if gap < 1e-10:
            break

        # Find active lags (within 1% of max)
        threshold = m_val * 0.99
        active = np.where(c >= threshold)[0]
        w = np.zeros(2 * n - 1)
        # Weight by how close to max
        weights = c[active]
        weights = weights / weights.sum()
        w[active] = weights

        # Gradient: 2 * conv(w_padded_shifted, a_reversed)
        # conv(a,a)[k] = sum_i a[i]*a[k-i], grad wrt a[j] = 2*sum_k w[k]*a[k-j]
        # = 2 * (w * a_rev)[n-1-j] where * is convolution
        # Simpler: grad[j] = 2 * sum_k w[k] * a[k-j] (with bounds checking)
        # This is 2 * conv(w, a_rev)[n-1-j]... let me use FFT
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        # h_full[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)]
        # We want grad[j] = 2 * sum_k w[k] * a[k-j]
        # a[k-j] = a_rev[n-1-k+j]
        # sum_k w[k] * a_rev[n-1-k+j] = conv(w, a_rev)[n-1+j]... 
        # Actually let me just compute it directly with the FFT trick
        # grad[j] = 2 * sum_k w[k] * a[k-j] for 0<=k-j<n
        # = 2 * (w * a)[j] where (w*a) is cross-correlation
        # Cross-correlation: (w ⋆ a)[j] = sum_k w[k] a[k+j]... no
        # Let's use: grad[j] = 2 * sum_k w[k] * a[k-j]
        # Define a_ext of size N: a_ext[i] = a[i] for i < n, 0 otherwise
        # Then sum_k w[k]*a[k-j] = (conv(w, a_rev))[n-1-j]... 
        # I'll just use the full convolution approach
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Polyak step
        grad_centered = grad - np.dot(a, grad) * a
        g_norm_sq = float(np.dot(grad_centered, grad_centered))
        if g_norm_sq < 1e-14:
            break
        step = gap / g_norm_sq
        step = min(step, 0.5)

        v = momentum_cur * v - step * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        sc_new = float(c_new.max())

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                momentum_cur = max(0.3, momentum_cur * 0.95)
            if no_imp > max(2000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                momentum_cur = momentum
                no_imp = 0

    return best_a


def _lse_optimize(a_init, deadline, max_iter=50000, T_start=0.03, T_end=1e-6, lr=0.08):
    """LSE gradient with Nesterov momentum in logit space."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2 = 0.9, 0.999

    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr

    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            continue

        t_count += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)

        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -30, 30)
        z -= z.mean()
        e = np.exp(z)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.92)
            if no_imp > max(3000, 8 * n):
                rng_r = np.random.default_rng(123 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                lr = base_lr
                no_imp = 0

        T = max(T * T_ratio, T_end)

    return best_a


def _spectral_gs(a_init, deadline, max_iter=500, flat_frac=0.5):
    """Gerchberg-Saxton style spectral flattening."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score_normalized(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        fa = np.fft.rfft(a, N)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Target: make |fa| more uniform
        target_mag = np.full_like(mag, mag.mean())
        # Blend: partially flatten
        blend = 0.3
        new_mag = (1 - blend) * mag + blend * target_mag

        fa_new = new_mag * np.exp(1j * phase)
        a_recovered = np.fft.irfft(fa_new, N)[:n]

        a_new = np.maximum(a_recovered, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(eta * (t - 0.5))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.2) / width) ** 2) + np.exp(-((t - 0.8) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer: biconvex LP (cutting plane), projected subgradient
    with Polyak step, LSE gradient with Adam, and spectral GS.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    def get_candidates(n_t, rng_local):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for _ in range(2):
                p = _interpolate_profile(parent, n_t)
                pp = p * (1.0 + 0.1 * rng_local.normal(size=n_t))
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    cands.append(pp / pp.sum())
        cands.append(_make_gaussian(n_t, 0.3))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_asymmetric(n_t, 2.0))
        cands.append(_make_bimodal(n_t, 0.15))
        cands.append(_make_flat_edges(n_t, 0.5))
        return cands

    # Phase 1: Biconvex LP on diverse candidates
    print("\n=== Phase 1: Biconvex LP (cutting plane) ===")
    best_per_length = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 300:
            break

        cands = get_candidates(n_t, rng)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 300:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            lp_budget = min(150.0, deadline - time.monotonic() - 300.0)
            if lp_budget < 20:
                break
            lp_end = time.monotonic() + lp_budget

            a_lp = _biconvex_lp_cutting_plane(a0, a0, lp_end, max_iter=200, top_k=40)
            sc_lp = _score(a_lp)

            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            if sc_lp < best_here_score:
                best_here_score = sc_lp
                best_here = a_lp

            print(f"  n={n_t} cand {ci}: init={sc0:.8f} -> BC-LP={sc_lp:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best BC-LP n={n_t}: {best_here_score:.10f}")

    # Phase 2: LSE gradient on best per length
    print("\n=== Phase 2: LSE gradient optimization ===")
    for n_t in list(best_per_length.keys()):
        if time.monotonic() > deadline - 200:
            break

        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        lse_budget = min(200.0, deadline - time.monotonic() - 200.0)
        if lse_budget < 20:
            continue
        lse_end = time.monotonic() + lse_budget

        a_lse = _lse_optimize(a_in, lse_end, max_iter=80000,
                              T_start=0.03, T_end=1e-6, lr=0.08)
        sc_lse = _score(a_lse)

        if np.isfinite(sc_lse) and sc_lse < best_score:
            best_score = sc_lse
            best_seq = a_lse.copy()
        if sc_lse < _score(best_per_length[n_t]):
            best_per_length[n_t] = a_lse
            print(f"  LSE n={n_t}: {sc_in:.10f} -> {sc_lse:.10f}")
        else:
            print(f"  LSE n={n_t}: {sc_in:.10f} -> {sc_lse:.10f} (no improvement)")

    # Phase 3: Projected subgradient with Polyak step
    print("\n=== Phase 3: Projected subgradient (Polyak) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        sp_budget = min(200.0, deadline - time.monotonic() - 150.0)
        if sp_budget > 20:
            sp_end = time.monotonic() + sp_budget
            a_sp = _subgradient_polyak(best_seq, sp_end, max_iter=50000, momentum=0.7)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
                best_per_length[len(best_seq)] = a_sp
            print(f"  Polyak subgrad: score={sc_sp:.10f}")

    # Phase 4: Spectral GS on best
    print("\n=== Phase 4: Spectral GS ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        gs_budget = min(100.0, deadline - time.monotonic() - 100.0)
        if gs_budget > 10:
            gs_end = time.monotonic() + gs_budget
            a_gs = _spectral_gs(best_seq, gs_end, max_iter=300)
            sc_gs = _score(a_gs)
            if np.isfinite(sc_gs) and sc_gs < best_score:
                best_score = sc_gs
                best_seq = a_gs.copy()
            print(f"  Spectral GS: score={sc_gs:.10f}")

    # Phase 5: Final LSE polish on global best
    print("\n=== Phase 5: Final LSE polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        lse_budget = min(80.0, deadline - time.monotonic() - 60.0)
        if lse_budget > 10:
            lse_end = time.monotonic() + lse_budget
            a_final = _lse_optimize(best_seq, lse_end, max_iter=30000,
                                    T_start=0.01, T_end=1e-7, lr=0.04)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LSE: score={sc_final:.10f}")

    # Phase 6: One more biconvex LP on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp_budget = min(60.0, deadline - time.monotonic() - 30.0)
        if lp_budget > 10:
            lp_end = time.monotonic() + lp_budget
            a_last = _biconvex_lp_cutting_plane(best_seq, best_seq, lp_end,
                                                max_iter=200, top_k=50)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final BC-LP: score={sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\n[Evaluate] score={sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
