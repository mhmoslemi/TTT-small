    """

"""

spectral flattening, and multi-length exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2n*max(conv)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    return 2.0 * n * (cm + beta * float(np.log(np.sum(np.exp(shifted)))))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _fft_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _subgradient_step(z, n, top_k=5, lr=0.01):
    """Single subgradient step using top-k peaks. Returns new z."""
    a = _softmax(z)
    c = _fft_convolve(a, a)
    # Get top-k peak positions
    top_idx = np.argsort(c)[-top_k:]
    # Normalize weights
    w = np.zeros_like(c)
    w[top_idx] = 1.0 / top_k
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _fft_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_z = 4.0 * n * a * (h - W)
    # Adaptive step size
    gnorm = np.linalg.norm(grad_z)
    if gnorm < 1e-15:
        return z
    z_new = z - lr * grad_z / (gnorm + 1e-15)
    z_new = np.clip(z_new, -50, 50)
    z_new = z_new - z_new.mean()
    return z_new


def _spectral_flatten(a, alpha=0.5, n_iter=50):
    """Flatten the power spectrum of a while maintaining non-negativity and sum=1."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break
        # Compress dynamic range
        P_flat = P ** alpha
        # Reconstruct
        phase = A / (np.abs(A) + 1e-300)
        A_new = np.sqrt(P_flat) * phase
        a_raw = np.fft.irfft(A_new, N)[:n]
        # Project onto non-negative simplex
        a_proj = _project_simplex(a_raw)
        a = a_proj
    return a


def _project_simplex(v):
    """Project v onto the standard simplex: sum x = 1, x >= 0."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Binary search for threshold
    lo = v.min() - 1e-10
    hi = v.max() + 1e-10
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    result = result / result.sum()
    return result


def _adam_subgradient(a_init, deadline, lr=0.05, top_k=5, beta_frac=0.01):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _fft_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:

        a_rev = a[::-1]
        w_conv = _fft_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        v = b2 * v + (1 - b2) * (grad * grad)

def _lbfgs_continuation(a_init, deadline, max_wall_time=None):

    n = len(a_init)

    best_score = _score_norm(a_init)

    beta_fracs = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    best_score = _score_norm(a_init)
    stall_count = 0

    while time.monotonic() < deadline:
        a = _softmax(z)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            lr = min(0.1, lr * 1.05)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                lr = lr * 0.8
                stall_count = 0
                if lr < 0.001:
                    break

        z = _subgradient_step(z, n, top_k=top_k, lr=lr)

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    # Linear
    for slope in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Quadratic
    for a2 in [-0.5, -0.3, 0.0, 0.3, 0.5]:
        a = 1.0 + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cubic
    for a3 in [-0.3, -0.5, -0.8, 0.3, 0.5]:
        a = 1.0 + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cosine bump
    for freq in [1.0, 2.0, 3.0]:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * (t - 0.5))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Two Gaussian bumps
    for center in [0.3, 0.4]:
        a = 0.5 + np.exp(-20 * (t - center) ** 2) + np.exp(-20 * (t - (1 - center)) ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _spectral_polish(a, deadline, alpha_init=0.7, n_flatten_iters=30):
    """Alternating spectral flattening and simplex projection."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    alpha = alpha_init

    while time.monotonic() < deadline:
        a_flat = _spectral_flatten(a, alpha=alpha, n_iter=n_flatten_iters)
        sc = _score_norm(a_flat)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_flat.copy()
            alpha = max(0.3, alpha * 0.95)
        else:
            alpha = min(0.95, alpha * 1.05)
            if alpha > 0.99:
                break
        a = best_a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization: L-BFGS continuation + subgradient polish + spectral flattening
    + multi-length exploration to minimize autocorrelation peak.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:.0f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profiles(512)[5]  # linear slope=1.5
        n_parent = 512
        print(f"No parent, using default n={n_parent}")

    # === Phase 1: L-BFGS continuation from parent (300s) ===
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\n--- Phase 1: L-BFGS continuation (n={n_parent}) ---")
    p1, s1 = _lbfgs_continuation(parent, phase1_end, max_wall_time=295.0)
    update_best(p1, s1, "lbfgs_parent")

    # === Phase 2: Subgradient + spectral polish (300s) ===
    phase2_end = min(deadline, t_start + 600.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"\n--- Phase 2: Subgradient + spectral polish ---")
        p2, s2 = _pure_subgradient_polish(best_seq, min(phase2_end - 100, time.monotonic() + 100),
                                          lr_init=0.03, top_k=3)
        update_best(p2, s2, "subgrad")

        if time.monotonic() < phase2_end - 50:
            p2b, s2b = _spectral_polish(best_seq, min(phase2_end - 20, time.monotonic() + 150),
                                        alpha_init=0.6, n_flatten_iters=40)
            update_best(p2b, s2b, "spectral")

            if time.monotonic() < phase2_end - 30:
                p2c, s2c = _adam_subgradient(best_seq, min(phase2_end - 10, time.monotonic() + 90),
                                             lr=0.03, top_k=5, beta_frac=0.005)
                update_best(p2c, s2c, "adam_subgrad")

    # === Phase 3: Multi-restart L-BFGS (400s) ===
    phase3_end = min(deadline, t_start + 1000.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"\n--- Phase 3: Multi-restart L-BFGS ---")
        n_best = len(best_seq)
        time_per_restart = min(80.0, (phase3_end - time.monotonic()) / 5.0)
        n_restarts = 5

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break
            # Different perturbation strategies
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            if i == 0:
                noise = 0.05 * rng.normal(size=n_best)
            elif i == 1:
                noise = 0.1 * rng.normal(size=n_best)
            elif i == 2:
                # Structured: add a cosine perturbation
                t = np.linspace(0, 2 * np.pi * (3 + i), n_best)
                noise = 0.08 * np.sin(t)
            elif i == 3:
                # Random mask perturbation
                mask = rng.random(n_best)
                noise = 0.1 * (mask - 0.5)
            else:
                noise = 0.15 * rng.normal(size=n_best)
            z = z + noise
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax(z)

            per_dl = min(phase3_end - 10, time.monotonic() + time_per_restart)
            p_r, s_r = _lbfgs_continuation(p_start, per_dl, max_wall_time=time_per_restart)
            update_best(p_r, s_r, f"restart{i}")

    # === Phase 4: Try different lengths (300s) ===
    phase4_end = min(deadline, t_start + 1300.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"\n--- Phase 4: Multi-length exploration ---")
        n_curr = len(best_seq)
        lengths_to_try = []
        for L in [512, 1024, 2048, 4096, 8192]:
            if L != n_curr and L >= 256:
                lengths_to_try.append(L)
        lengths_to_try.sort(key=lambda x: abs(x - n_curr))

        for L in lengths_to_try[:3]:  # Try up to 3 different lengths
            if time.monotonic() > phase4_end - 60:
                break
            p_interp = _interp_profile(best_seq, L)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{L}")

            time_left = phase4_end - time.monotonic() - 30
            if time_left < 30:
                break
            per_dl = min(phase4_end - 20, time.monotonic() + min(120.0, time_left * 0.6))
            p_N, s_N = _lbfgs_continuation(p_interp, per_dl, max_wall_time=min(110.0, time_left * 0.6))
            update_best(p_N, s_N, f"lbfgs_n{L}")

            if time.monotonic() < phase4_end - 40:
                per_dl2 = min(phase4_end - 10, time.monotonic() + min(60.0, (phase4_end - time.monotonic()) * 0.5))
                p_S, s_S = _adam_subgradient(p_N if s_N < sc_interp else p_interp,
                                             per_dl2, lr=0.02, top_k=3, beta_frac=0.003)
                update_best(p_S, s_S, f"adam_n{L}")

    # === Phase 5: Final intensive polish (350s) ===
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"\n--- Phase 5: Final polish ---")
        n_final = len(best_seq)

        # L-BFGS final
        per_dl = min(phase5_end - 100, time.monotonic() + 100)
        p_f1, s_f1 = _lbfgs_continuation(best_seq, per_dl, max_wall_time=min(90.0, per_dl - time.monotonic()))
        update_best(p_f1, s_f1, "final_lbfgs")

        if time.monotonic() < phase5_end - 80:
            # Adam subgradient
            per_dl2 = min(phase5_end - 60, time.monotonic() + 80)
            p_f2, s_f2 = _adam_subgradient(best_seq, per_dl2, lr=0.01, top_k=3, beta_frac=0.002)
            update_best(p_f2, s_f2, "final_adam")

        if time.monotonic() < phase5_end - 50:
            # Spectral
            per_dl3 = min(phase5_end - 30, time.monotonic() + 50)
            p_f3, s_f3 = _spectral_polish(best_seq, per_dl3, alpha_init=0.5, n_flatten_iters=50)
            update_best(p_f3, s_f3, "final_spectral")

        if time.monotonic() < phase5_end - 30:
            # One more round of subgradient
            per_dl4 = min(phase5_end - 10, time.monotonic() + 30)
            p_f4, s_f4 = _pure_subgradient_polish(best_seq, per_dl4, lr_init=0.01, top_k=2)
            update_best(p_f4, s_f4, "final_subgrad")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            t = np.linspace(0, 1, 256)
            final_seq = np.maximum(1.0 + 1.5 * (t - 0.5), 1e-12)
            final_seq = final_seq / final_seq.sum()

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    t = np.linspace(0, 1, 512)
    fallback = np.maximum(1.0 + 1.5 * (t - 0.5), 1e-12)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
