"""
Optimized step function search using Adam smooth-max, multi-peak subgradient,
coordinate-wise refinement, and LP polishing. Targets score < 1.5027.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Remove tiny negative values from numerical error
    mn = c.min()
    if mn < 0 and mn > -1e-12:
        c = c - mn
    return c

def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = a.size
    c = _fft_convolve(a)
    return 2.0 * n * float(c.max())

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(_fft_convolve(a).max()) / (s * s)

def _log_sum_exp(c, beta):
    """Compute log-sum-exp of c/beta, centered for numerical stability."""
    c = np.asarray(c, dtype=np.float64)
    m = c.max()
    x = (c - m) / beta
    x = np.clip(x, -100, 100)
    return m / beta + np.log(np.exp(x).sum())

def _softmax_weights(c, beta):
    """Get softmax weights for smooth max gradient."""
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s

def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*c[k] where c = conv(a,a)."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h

def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg

def _multi_subgradient(a, deadline, max_iter=10000, top_k=5):
    """Subgradient targeting top-k peaks simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    base_lr = 0.2 / np.sqrt(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        
        weights = np.zeros(k)
        for i in range(k):
            weights[i] = c_curr[top_indices[i]]
        weights = weights / weights.sum()
        
        sg = np.zeros(n)
        for i in range(k):
            sg += weights[i] * _subgradient_at_max(a, int(top_indices[i]))
        
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break
        
        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break
        
        a = a_new
    
    return best_a, best_score

def _coordinate_refinement(a, deadline, max_sweeps=20):

    n = len(a)
    s = a.sum()
    a = a / s

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Use random order for diversity
        order = np.random.default_rng(sweep * 7 + 1).permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try several perturbation sizes
            best_local = old_val
            best_local_score = float(_fft_convolve(a).max())
            
            for frac in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0]:
                a[i] = old_val * frac
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                c_new = _fft_convolve(a_norm)
                sc = float(2.0 * n * c_new.max())
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_local = old_val * frac

                a = a / a.sum()
                improved = True

            sc_curr = float(2.0 * n * c_curr.max())
            if sc_curr < best_score - 1e-14:

def _lp_refine(a, deadline, max_iter=500, inflate=1.0005, blend=0.3):
    """LP-based refinement targeting the max peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, 2.0 * n * float(_fft_convolve(a / max(s, 1e-10)).max())
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        # Build LP: maximize sum(g) subject to conv(a, g) <= M_lp, g >= 0
        # conv(a, g)[k] = sum_j a[j] * g[k-j]
        # This is a linear constraint in g for fixed a
        
        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M_lp)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            break
        
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

Optimized step function search: Adam smooth-max with aggressive cooling,
multi-peak subgradient, coordinate-wise local search, and LP polishing.
Targets score < 1.5027 (lower is better).
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-10:
        c = c - mn
    return c


def _score_normalized(a):
    """Score assuming sum(a)=1, i.e., 2*n*max(conv(a,a))."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * _fft_convolve(a).max()) / (s * s)


def _softmax_weights(c, beta):
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg

    b2 = 0.999

    m = np.zeros(n)
    v_arr = np.zeros(n)

def _multi_subgradient(a, deadline, max_iter=12000, top_k=5, weight_power=2.0):
    """Subgradient targeting top-k peaks with power weighting."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        weights = np.zeros(k)
        for i in range(k):
            weights[i] = c_curr[top_indices[i]] ** weight_power
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum

        sg = np.zeros(n)
        for i in range(k):
            sg += weights[i] * _subgradient_at_max(a, int(top_indices[i]))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _coordinate_refinement(a, deadline, max_sweeps=30, n_trials=4):
    """Coordinate-wise local search with multiple trial values."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    # Precompute trial fractions
    trials = [0.5, 0.7, 1.3, 1.6, 0.3, 2.0, 0.8, 1.2]
    trials = trials[:n_trials * 2]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        rng_s = np.random.default_rng(sweep * 101 + 13)
        order = rng_s.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            old_val = a[i]
            if old_val < 1e-14:
                continue

            current_max = float(_fft_convolve(a).max())

            best_local = old_val
            best_local_max = current_max

            for frac in trials:
                a[i] = old_val * frac
                s_new = float(a.sum())
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                c_new = _fft_convolve(a_norm)
                new_max = float(c_new.max())
                if new_max < best_local_max - 1e-15:
                    best_local_max = new_max
                    best_local = old_val * frac

            if best_local != old_val:
                a[i] = best_local
                a = a / a.sum()
                improved = True

            c_curr = _fft_convolve(a)
            sc_curr = float(2.0 * n * c_curr.max())
            if sc_curr < best_score - 1e-14:
                best_score = sc_curr
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def _lp_refine(a, deadline, max_iter=300, inflate=1.0005, blend=0.3):
    """LP-based refinement targeting the max peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, float(_score(a))
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M_lp)

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.08)
                blend_cur = max(0.1, blend_cur * 0.85)
            if no_imp > 80:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation=0.5, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5 + separation / 2.0) ** 2) / (2.0 * width ** 2)) + \
        np.exp(-((t - 0.5 - separation / 2.0) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _adam_multi_restart(a0, deadline, n_restarts=3, lr=0.025, beta_factor=0.06):
    """Run Adam with multiple restarts from perturbed best."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)

    best_a = a.copy()
    best_score = _score_normalized(a)

    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break
        sub_deadline = min(deadline, time.monotonic() + max(60.0, (deadline - time.monotonic()) / (n_restarts - restart)))
        a_r, s_r = _adam_smooth_max(best_a, sub_deadline, max_iter=20000, lr=lr, beta_factor=beta_factor)
        if s_r < best_score:
            best_score = s_r
            best_a = a_r.copy()
        print(f"    Adam restart {restart}: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining Adam smooth-max with aggressive cooling,
    multi-peak subgradient, coordinate-wise local search, and LP polishing.
    Explores parent sequence and multiple initializations at various lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare initializations from parent
    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p768 = _interpolate_profile(parent, 768)
        p1024 = _interpolate_profile(parent, 1024) if parent.size != 1024 else parent.copy()
        p1280 = _interpolate_profile(parent, 1280)
        parent_n = parent.size
    else:
        p512 = _make_linear_profile(512, 1.5)
        p768 = _make_linear_profile(768, 1.5)
        p1024 = _make_linear_profile(1024, 1.5)
        p1280 = _make_linear_profile(1280, 1.5)
        parent_n = 0

    # === Phase 1: Adam multi-restart on n=1024 (main workhorse) ===
    print("=== Phase 1: Adam multi-restart @ n=1024 ===")
    d1 = min(deadline - 700, t_start + 650)
    a1, s1 = _adam_multi_restart(p1024, d1, n_restarts=3, lr=0.025, beta_factor=0.06)
    if s1 < best_score:
        best_score = s1
        best_seq = a1.copy()
    print(f"  Phase 1 done: {best_score:.10f}")

    # === Phase 2: Multi-subgradient refinement ===
    if time.monotonic() < deadline - 450:
        print("=== Phase 2: Multi-subgradient (top-5, power=2) ===")
        d2 = min(deadline - 450, time.monotonic() + 200)
        a2, s2 = _multi_subgradient(best_seq, d2, max_iter=12000, top_k=5, weight_power=2.0)
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()
        print(f"  Phase 2 done: {best_score:.10f}")

    # === Phase 3: Coordinate-wise local search ===
    if time.monotonic() < deadline - 300:
        print("=== Phase 3: Coordinate refinement ===")
        d3 = min(deadline - 300, time.monotonic() + 200)
        a3, s3 = _coordinate_refinement(best_seq, d3, max_sweeps=25, n_trials=4)
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()
        print(f"  Phase 3 done: {best_score:.10f}")

    # === Phase 4: LP refinement ===
    if time.monotonic() < deadline - 180:
        print("=== Phase 4: LP refinement ===")
        d4 = min(deadline - 180, time.monotonic() + 100)
        a4, s4 = _lp_refine(best_seq, d4, max_iter=300, inflate=1.0003, blend=0.25)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
        print(f"  Phase 4 done: {best_score:.10f}")

    # === Phase 5: Adam polish (fine-grained) ===
    if time.monotonic() < deadline - 80:
        print("=== Phase 5: Fine Adam polish ===")
        d5 = min(deadline - 80, time.monotonic() + 70)
        a5, s5 = _adam_smooth_max(best_seq, d5, max_iter=15000, lr=0.008, beta_factor=0.02, beta_decay=0.85)
        if s5 < best_score:
            best_score = s5
            best_seq = a5.copy()
        print(f"  Phase 5 done: {best_score:.10f}")

    # === Phase 6: Try alternative lengths ===
    if time.monotonic() < deadline - 60:
        print("=== Phase 6: Alternative lengths ===")
        for n_alt in [512, 768, 1280]:
            if time.monotonic() > deadline - 60:
                break
            p_alt = _interpolate_profile(best_seq, n_alt)
            d6 = min(deadline - 60, time.monotonic() + 70)
            a6, s6 = _adam_smooth_max(p_alt, d6, max_iter=10000, lr=0.02, beta_factor=0.05, beta_decay=0.82)
            if s6 < best_score:
                best_score = s6
                best_seq = a6.copy()
            print(f"  Alt n={n_alt}: best={best_score:.10f}")

    # === Phase 7: Final coordinate refinement ===
    if time.monotonic() < deadline - 30:
        print("=== Phase 7: Final coordinate refinement ===")
        d7 = min(deadline - 30, time.monotonic() + 30)
        a7, s7 = _coordinate_refinement(best_seq, d7, max_sweeps=10, n_trials=3)
        if s7 < best_score:
            best_score = s7
            best_seq = a7.copy()
        print(f"  Phase 7 done: {best_score:.10f}")

    # === Phase 8: Quick subgradient on best ===
    if time.monotonic() < deadline - 15:
        d8 = min(deadline - 15, time.monotonic() + 15)
        a8, s8 = _multi_subgradient(best_seq, d8, max_iter=3000, top_k=3, weight_power=1.0)
        if s8 < best_score:
            best_score = s8
            best_seq = a8.copy()
        print(f"  Phase 8: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    sc_eval = float(evaluate_sequence(final_seq.tolist()))
    if np.isfinite(sc_eval):
        print(f"Evaluate_sequence: {sc_eval:.10f}")
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        final_seq = _make_linear_profile(256, 1.5)
        sc_eval = _score(final_seq)

    print(f"FINAL: score={best_score:.10f}, eval={sc_eval:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
