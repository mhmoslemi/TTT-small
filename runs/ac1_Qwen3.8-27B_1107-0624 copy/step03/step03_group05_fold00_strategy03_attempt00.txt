def get_sequence_a(seq_length):
    # Start from a linear parent with the given length
    t = np.linspace(0, 1, seq_length)
    base = 1.0 + 1.0*t*t - 1.0*t*t*t*t  # Quadratic decay shape
    base = base/ base.sum()
    return base

def create_initial_candidates(seq_length=768):
    # Create a set of candidates with possible zero padding and summation checks
    cand = get_sequence_a(seq_length)

    n = len(candidate)
