"""
LP-based alternating optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions, combined with gradient-based polishing.
Key idea: Given current a, solve LP max(1^T b) s.t. (a*b)_k <= M, b>=0.
The optimal value S_b > 1 indicates improvement potential. Mix to update.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve(a):
    """Full linear convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def build_conv_matrix(a):
    """Build sparse convolution matrix A where (A @ b)_k = (a*b)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    # For each column j, non-zero at rows k=j...j+n-1 with values a[0]...a[n-1]
    # But clipped to valid range k in [0, 2n-2]
    j_max = n - 1
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_start < k_end:
            idx = np.arange(k_start, k_end)
            rows.extend(idx)
            cols.extend([j] * len(idx))
            vals.extend(a[0:k_end - k_start])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(a, M):
    """
    Solve: maximize 1^T b subject to (a*b)_k <= M for all k, b >= 0.
    Returns (optimal_b, optimal_value).
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x, -res.fun
        return None, 0.0
    except Exception:
        return None, 0.0


def lp_optimize(a0, deadline, max_iters=500, verbose=True):
    """
    LP-based alternating optimization.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = full_convolve(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M

    t_mix = 0.1
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_star, S_b = lp_solve(a, M)

        if b_star is None:
            break

        if S_b <= 1.0 + 1e-8:
            stuck_count += 1
            if stuck_count > 20:
                break
            # Try with slightly relaxed M
            M_relaxed = M * 1.001
            b_star2, S_b2 = lp_solve(a, M_relaxed)
            if b_star2 is not None and S_b2 > 1.0 + 1e-6:
                g = b_star2 / S_b2
                a_new = (1 - t_mix) * a + t_mix * g
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                c_new = full_convolve(a_new)
                M_new = float(c_new.max())
                if M_new < M:
                    a = a_new
                    M = M_new
                    stuck_count = 0
                else:
                    t_mix = max(0.01, t_mix * 0.9)
            continue
        else:
            stuck_count = 0

        g = b_star / S_b
        g = np.maximum(g, 0.0)
        s_g = g.sum()
        if s_g < 1e-15:
            break
        g = g / s_g

        # Mix

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = full_convolve(a_new)
        M_new = float(c_new.max())

            t_mix = max(0.01, t_mix * 0.8)

        score = 2.0 * n * M

            best_a = a.copy()
            best_M = M

def soft_max_weights(c, beta):

    if beta < 1e-14:

        w[int(np.argmax(c))] = 1.0
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval, w


def compute_h(w, a):
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def gradient_polish(a0, deadline, lr=0.01, beta_init=0.02, beta_decay=0.95):
    """Logit-space gradient polish."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()

    c = full_convolve(a)
    beta = beta_init * float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * float(c.max())

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    no_improve = 0

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = z.copy()
            z[~np.isfinite(z)] = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a = np.exp(z - z.max())
        s = a.sum()
        a = a / s

        c = full_convolve(a)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(1e-10, beta * beta_decay)

        if no_improve > 200:
            break

    return best_a, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_random_smooth(n, rng, n_freq=5, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i in range(1, n_freq + 1):
        coeff = rng.uniform(-amp, amp)
        a = a + coeff * np.cos(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def peak_targeting(a0, deadline, n_peaks=5, lr=0.01):
    """Target the top peaks specifically."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()

    best_a = a.copy()
    c = full_convolve(a)
    best_score = 2.0 * n * float(c.max())

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        c = full_convolve(a)
        peak_indices = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        for idx in peak_indices:
            w[idx] = 1.0 / n_peaks

        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = full_convolve(a)
        score = 2.0 * n * float(c.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    LP-based alternating optimization + gradient polish.
    Uses LP to find descent directions (maximize sum(b) s.t. (a*b)_k <= M, b>=0),
    then gradient-based refinement for final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                update(init_seq, sc_init)
                print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
    except Exception:
        pass

    # ===== Phase 1: LP optimization at n=512 =====
    n1 = 512
    print(f"\n--- Phase 1: LP at n={n1} ---")
    phase1_dl = min(deadline, t_start + 300.0)

    if init_seq is not None:
        p1 = interpolate_profile(init_seq, n1)
    else:
        p1 = make_linear_profile(n1, 1.5)

    sc1_init = compute_score(p1)
    update(p1, sc1_init)
    print(f"  Start: score={sc1_init:.10f}")

    result1, sc1 = lp_optimize(p1, phase1_dl, max_iters=300)
    sc1_final = compute_score(result1)
    update(result1, sc1_final)
    print(f"  LP done: score={sc1_final:.10f}")

    # Quick gradient polish at n=512
    if time.monotonic() < phase1_dl - 20:
        gp_dl = min(phase1_dl, time.monotonic() + 60.0)
        r_gp, sc_gp = gradient_polish(result1, gp_dl, lr=0.02, beta_init=0.03)
        sc_gp_f = compute_score(r_gp)
        update(r_gp, sc_gp_f)
        print(f"  After grad polish: score={sc_gp_f:.10f}")
        if sc_gp_f < sc1_final:
            result1 = r_gp
            sc1_final = sc_gp_f

    # ===== Phase 2: LP optimization at n=1024 =====
    n2 = 1024
    print(f"\n--- Phase 2: LP at n={n2} ---")
    phase2_dl = min(deadline, t_start + 800.0)

    p2 = interpolate_profile(result1, n2)
    sc2_init = compute_score(p2)
    update(p2, sc2_init)
    print(f"  Interpolated: score={sc2_init:.10f}")

    result2, sc2 = lp_optimize(p2, phase2_dl, max_iters=300)
    sc2_final = compute_score(result2)
    update(result2, sc2_final)
    print(f"  LP done: score={sc2_final:.10f}")

    # Gradient polish at n=1024
    if time.monotonic() < phase2_dl - 30:
        gp2_dl = min(phase2_dl, time.monotonic() + 100.0)
        r_gp2, sc_gp2 = gradient_polish(result2, gp2_dl, lr=0.015, beta_init=0.02)
        sc_gp2_f = compute_score(r_gp2)
        update(r_gp2, sc_gp2_f)
        print(f"  After grad polish: score={sc_gp2_f:.10f}")
        if sc_gp2_f < sc2_final:
            result2 = r_gp2
            sc2_final = sc_gp2_f

    # Peak targeting at n=1024
    if time.monotonic() < phase2_dl - 20:
        pt_dl = min(phase2_dl, time.monotonic() + 60.0)
        r_pt, sc_pt = peak_targeting(result2, pt_dl, n_peaks=5, lr=0.008)
        sc_pt_f = compute_score(r_pt)
        update(r_pt, sc_pt_f)
        print(f"  After peak targeting: score={sc_pt_f:.10f}")
        if sc_pt_f < sc2_final:
            result2 = r_pt

    # ===== Phase 3: LP optimization at n=2048 =====
    n3 = 2048
    print(f"\n--- Phase 3: LP at n={n3} ---")
    phase3_dl = min(deadline, t_start + 1400.0)

    p3 = interpolate_profile(result2, n3)
    sc3_init = compute_score(p3)
    update(p3, sc3_init)
    print(f"  Interpolated: score={sc3_init:.10f}")

    result3, sc3 = lp_optimize(p3, phase3_dl, max_iters=200)
    sc3_final = compute_score(result3)
    update(result3, sc3_final)
    print(f"  LP done: score={sc3_final:.10f}")

    # ===== Phase 4: Final polish =====
    print(f"\n--- Phase 4: Final polish ---")
    phase4_dl = deadline

    if best_seq is not None:
        n_best = best_seq.size

        # Gradient polish
        if time.monotonic() < phase4_dl - 30:
            gp3_dl = min(phase4_dl, time.monotonic() + 120.0)
            p0 = best_seq + 0.0003 * rng.normal(size=n_best)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            r_gp3, sc_gp3 = gradient_polish(p0, gp3_dl, lr=0.008, beta_init=0.015, beta_decay=0.96)
            sc_gp3_f = compute_score(r_gp3)
            update(r_gp3, sc_gp3_f)
            print(f"  Grad polish: score={sc_gp3_f:.10f}")

        # Peak targeting
        if time.monotonic() < phase4_dl - 30:
            pt3_dl = min(phase4_dl, time.monotonic() + 60.0)
            r_pt3, sc_pt3 = peak_targeting(best_seq, pt3_dl, n_peaks=3, lr=0.005)
            sc_pt3_f = compute_score(r_pt3)
            update(r_pt3, sc_pt3_f)
            print(f"  Peak targeting: score={sc_pt3_f:.10f}")

        # One more LP round at best size
        if time.monotonic() < phase4_dl - 60:
            lp4_dl = min(phase4_dl, time.monotonic() + 80.0)
            r_lp4, sc_lp4 = lp_optimize(best_seq, lp4_dl, max_iters=50)
            sc_lp4_f = compute_score(r_lp4)
            update(r_lp4, sc_lp4_f)
            print(f"  LP polish: score={sc_lp4_f:.10f}")

    # ===== Phase 5: Try alternative starting points =====
    if time.monotonic() < deadline - 60 and best_score > 1.503:
        print(f"\n--- Phase 5: Alternative starts ---")
        for trial in range(3):
            if time.monotonic() > deadline - 30:
                break
            n_try = 1024
            p_alt = make_random_smooth(n_try, rng, n_freq=4, amp=0.25)
            sc_alt_init = compute_score(p_alt)
            if sc_alt_init > best_score + 0.2:
                continue
            # Quick LP optimization
            alt_dl = min(deadline - 30, time.monotonic() + 80.0)
            r_alt, sc_alt = lp_optimize(p_alt, alt_dl, max_iters=50)
            sc_alt_f = compute_score(r_alt)
            update(r_alt, sc_alt_f)
            print(f"  Alt trial {trial}: score={sc_alt_f:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"WARNING: evaluate_sequence returned {eval_score}")
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
