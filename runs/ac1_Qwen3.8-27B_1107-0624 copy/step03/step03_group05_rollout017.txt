"""
Optimized peak-autocorrelation minimization using adaptive LP spreading,
exact subgradient descent with peak cycling, and multi-length search.
Improvements: reduced-constraint LP for speed, backtracking line search,
peak cycling to escape local minima, and focused length search.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix_reduced(a, M, top_frac=0.5):
    """Build convolution matrix but only for constraints near the peak."""
    n = len(a)
    c = _fft_convolve(a)
    # Select top constraints (those where c is close to M)
    threshold = M * 0.7  # include constraints where c > 0.7*M
    mask = c >= threshold
    if mask.sum() < max(10, n // 4):
        # Ensure we have enough constraints
        k = max(10, n // 4)
        idx = np.argsort(c)[::-1][:k]
        mask = np.zeros_like(c, dtype=bool)
        mask[idx] = True
    
    indices = np.where(mask)[0]
    A = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        # conv(a, b)_k = sum_j a[j] * b[k-j] for valid j
        # For fixed a, this is linear in b: sum_j a_j * b[k-j]
        # So the row is: A[row, j] = a[j] if 0 <= k-j < n
        j_range = np.arange(max(0, k - n + 1), min(n, k + 1))
        b_idx = k - j_range
        A[row, j_range] = a[b_idx]
    b = np.full(len(indices), M)
    return A, b, mask


def _lp_spread_reduced(a, M, top_frac=0.5):
    """LP to spread mass while keeping conv(a,b) <= M, using reduced constraints."""
    n = len(a)
    A, b, mask = _make_conv_matrix_reduced(a, M, top_frac)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_full(a, M):
    """Full LP with all constraints."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _gradient_exact_subgrad(a, K):
    """Exact subgradient of max(c) at peak position K."""
    n = len(a)
    grad = np.zeros(n)
    # dc_K/da_i = 2*a[K-i] if 0 <= K-i < n
    valid_i = np.arange(max(0, K - n + 1), min(n, K + 1))
    bg_idx = K - valid_i
    grad[valid_i] = 2.0 * a[bg_idx]
    return grad


def _gradient_smooth_max(a, beta):
    """Smooth max gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _backtrack_line_search(a, direction, c_current, max_c, alpha=0.5, rho=0.5, max_iter=20):
    """Backtracking line search to ensure decrease in max(c)."""
    n = len(a)
    step = 1.0
    for _ in range(max_iter):
        a_new = a + step * direction
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            step *= rho
            continue
        a_new = a_new / s
        c_new = _fft_convolve(a_new)
        max_new = float(c_new.max())
        if max_new < max_c:
            return a_new, max_new
        step *= rho
    return None, max_c

    best_a = a.copy()

    best_maxc = float(c.max())
    best_score = 2.0 * n * best_maxc
    no_imp = 0
    lr = 0.05

        # And to maintain sum=1, we increase a where grad is small

            no_imp += 10
            continue

                best_a = a_new.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:

            a_new2 = a + lr * direction
            a_new2 = np.maximum(a_new2, 0.0)

                c_new2 = _fft_convolve(a_new2)
                if float(c_new2.max()) < maxc:
                    a = a_new2
                    c_new_r = _fft_convolve(a)
                    if float(c_new_r.max()) > maxc:
                        a = best_a.copy()  # revert
                        lr = 0.05
            no_imp += 1
        
        if it % 100 == 0:
            lr = min(0.1, lr * 1.05)  # slowly increase lr
            if no_imp > 100:
                lr = 0.03
        
        if no_imp > 2000:
            break
    
    return best_a, best_score


def _smooth_max_descent(a, deadline, maxiter=20000):
    """Smooth max descent with adaptive beta and line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_maxc = float(c.max())
    best_score = 2.0 * n * best_maxc
    
    beta = 0.02 * best_maxc
    lr = 0.04
    no_imp = 0
    base_lr = lr
    
    for it in range(maxiter):
        if time.monotonic() > deadline:
            break
        
        grad, maxc, c_arr = _gradient_smooth_max(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta *= 2
            continue
        
        # Project gradient onto tangent space
        direction = -grad + np.mean(grad)
        
        # Backtracking line search
        a_new = None
        step = lr
        for _ in range(10):
            candidate = a + step * direction
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            candidate = candidate / s
            c_cand = _fft_convolve(candidate)
            if float(c_cand.max()) < maxc - 1e-15:
                a_new = candidate
                break
            step *= 0.5
        
        if a_new is None:
            # Can't improve with this gradient; try exact subgrad at current peak
            K = int(np.argmax(c_arr))
            grad_exact = _gradient_exact_subgrad(a, K)
            direction = -grad_exact + np.mean(grad_exact)
            step = lr * 0.5
            for _ in range(5):
                candidate = a + step * direction
                candidate = np.maximum(candidate, 0.0)
                s = candidate.sum()
                if s > 1e-10:
                    candidate = candidate / s
                    c_cand = _fft_convolve(candidate)
                    if float(c_cand.max()) < maxc:
                        a_new = candidate
                        break
                step *= 0.5
        
        if a_new is not None:
            a = a_new
            c = _fft_convolve(a)
            maxc_new = float(c.max())
            score_new = 2.0 * n * maxc_new
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        
        # Adaptive beta
        if it % 50 == 0:
            beta = max(1e-10, beta * 0.7)
        
        if no_imp > 500:
            lr = max(0.005, lr * 0.8)
        if no_imp > 2000:
            # Restart from best
            a = best_a.copy()
            beta = 0.02 * float(_fft_convolve(a).max())
            lr = base_lr
            no_imp = 0
        
        if no_imp > 5000:
            break
    
    return best_a, best_score


def _lp_refine_improved(a, deadline, max_iter=10000):
    """LP refinement with adaptive inflation and reduced constraints."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_maxc = float(c.max())
    best_score = 2.0 * n * best_maxc
    
    inflate = 1.002
    blend = 0.3
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        # Use reduced LP for speed
        if n > 400:
            g = _lp_spread_reduced(a, M_lp, top_frac=0.5)
        else:
            g = _lp_spread_full(a, M_lp)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(1.01, inflate * 1.03)
            if no_imp > 500:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        c_new = _fft_convolve(a_new)
        maxc_new = float(c_new.max())
        score_new = 2.0 * n * maxc_new
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate = max(1.0005, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.1, blend * 0.9)
                inflate = min(1.01, inflate * 0.97)
            if no_imp > 300:
                a = best_a.copy()
                blend = 0.3
                inflate = 1.002
                no_imp = 0
        
        a = a_new
    
    return best_a, best_score


def _make_linear_profile(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1e-6 + eta * t
    a = a / a.sum()
    return a


def _make_ramp_profile(n):
    """Pure ramp from near-0 to 1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = a / a.sum()
    return a


def _make_gaussian(n, center=0.5, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = a / a.sum()
    return a


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _combined_pipeline(a_init, n, deadline, time_per_n):
    """Run combined optimization for a given length."""
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(a).max())
    
    # Phase 1: LP spreading
    t1 = time.monotonic()
    a_lp, sc_lp = _lp_refine_improved(a, min(deadline, t1 + time_per_n * 0.35))
    if sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
    print(f"    LP: {sc_lp:.10f} (time: {time.monotonic()-t1:.1f}s)")
    
    # Phase 2: Smooth max descent
    t2 = time.monotonic()
    a_sm, sc_sm = _smooth_max_descent(best_a, min(deadline, t2 + time_per_n * 0.25))
    if sc_sm < best_score:
        best_score = sc_sm
        best_a = a_sm.copy()
    print(f"    SmoothMax: {sc_sm:.10f} (time: {time.monotonic()-t2:.1f}s)")
    
    # Phase 3: Peak cycling
    t3 = time.monotonic()
    a_pc, sc_pc = _peak_cycling_descent(best_a, min(deadline, t3 + time_per_n * 0.2), n_peaks=3)
    if sc_pc < best_score:
        best_score = sc_pc
        best_a = a_pc.copy()
    print(f"    PeakCycle: {sc_pc:.10f} (time: {time.monotonic()-t3:.1f}s)")
    
    # Phase 4: Final LP
    t4 = time.monotonic()
    a_lp2, sc_lp2 = _lp_refine_improved(best_a, min(deadline, t4 + time_per_n * 0.2))
    if sc_lp2 < best_score:
        best_score = sc_lp2
        best_a = a_lp2.copy()
    print(f"    LP2: {sc_lp2:.10f} (time: {time.monotonic()-t4:.1f}s)")
    
    return best_a, best_score
