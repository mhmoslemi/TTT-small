import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _smooth_max_weights(c, beta):

        ex[int(np.argmax(c))] = 1.0
        s = 1.0

    return w


def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a). Vectorized via FFT."""
    n = len(a)

    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]

def _subgradient_at_max(a, k):
    """Compute dc[k]/da vectorized."""
    n = len(a)
    sg = np.zeros(n)
    # First term: a[k-j] for j in [max(0,k-n+1), min(k,n-1)]

    j1_end = min(k, n - 1)

        idx = np.arange(j1_start, j1_end + 1)

    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _multi_subgradient(a, deadline, max_iter=10000, top_k=3, verbose=False):
    """Subgradient targeting top-k peaks simultaneously."""

    c = _fft_convolve(a)

    base_lr = 0.2 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        # Get top-k peaks
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Weighted subgradient
        weights = np.array([(1.0 + 0.3 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
