"""
Spectral-flattening + smooth-max Adam optimization for step function coefficient minimization.
Two-phase approach: first flatten autocorrelation via L2 spectral minimization, then reduce peaks.
Uses n=2048 with FFT-based gradients, IAA initialization, and adaptive learning rates.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _l2_autocorr(a):
    """Compute sum(c^2) where c = a*a using FFT."""
    n = len(a)
    N = _next_pow2(2 * 2 * n - 1)
    fa = np.fft.rfft(a, N)
    power = fa * np.conj(fa)
    # FFT of c = IFFT(|power|^2)... actually sum(c^2) = (1/N)*sum|FFT(c)|^2
    # c = IFFT(|A|^2) padded to N, so FFT(c) = |A|^2 (circular)
    # sum(c^2) = (1/N) * sum(|A|^4)
    return float(np.sum(power * power) / N)


def _grad_l2_autocorr(a):
    """Gradient of sum(c^2) where c = linear conv(a,a), using circular approx."""
    n = len(a)
    N = _next_pow2(2 * 2 * n - 1)
    fa = np.fft.rfft(a, N)
    power = fa * np.conj(fa)
    # grad = 4 * Re(IFFT(|A|^2 * conj(A))) restricted to first n
    d = power * np.conj(fa)
    # Convert to full complex for irfft
    d_full = np.zeros(N, dtype=np.complex128)
    d_full[:len(power)] = d
    # Mirror for proper IFFT
    half = N // 2
    if N % 2 == 0:
        d_full[half + 1:] = np.conj(d_full[1:half][::-1])
    else:
        d_full[(half + 1):] = np.conj(d_full[1:half + 1][::-1])
    grad_full = np.fft.irfft(d_full, N)
    return 4.0 * grad_full[:n]


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg

    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

                m = np.zeros(n)

        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v_arr = b2 * v_arr + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)

        z_new = z - lr_eff * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                pass  # cosine schedule handles decay
            if no_imp > max(3000, 4 * n):
                # Perturb and restart
                rng_r = np.random.default_rng(7777 + it)
                z = best_z
                # Targeted perturbation at peak locations
                c_b = _fft_convolve(best_a)
                top_peaks = np.argsort(c_b)[::-1][:5]
                for pk in top_peaks:
                    # Perturb elements that contribute to this peak
                    j_start = max(0, pk - n + 1)
                    j_end = min(pk, n - 1)
                    if j_start <= j_end:
                        idx = np.arange(j_start, j_end + 1)
                        z[idx] += 0.1 * rng_r.normal(size=len(idx))
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()) * 0.5)
                m = np.zeros(n)
                v_arr = np.zeros(n)
                t = 0
                no_imp = 0
                continue

        z = z_new
        a = a_new

        # Adapt beta: decrease to sharpen the max approximation
        if it % 200 == 0:
            beta = max(1e-10, beta * 0.82)

    return best_a, best_score


def _peak_shaving(a, deadline, max_iter=5000):
    """Targeted subgradient steps on the top peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        
        # Get top 3 peaks
        k = min(3, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]
        weights = np.array([1.0, 0.5, 0.25])[:k]
        weights /= weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _iaa_init(n, iterations=100):
    """Iterative Amplitude Adjustment: alternate between flat spectrum and non-negativity."""
    N = _next_pow2(2 * n - 1)
    # Start with flat magnitude spectrum
    A = np.ones(N, dtype=np.complex128)
    # Add some structure: slightly emphasized low frequencies
    k = np.arange(N)
    A *= np.exp(-0.3 * (k / N) ** 2)

    a = np.zeros(n)
    for _ in range(iterations):
        # Time domain: take IFFT, keep real part, project to non-negative
        a_full = np.fft.ifft(A).real[:n]
        a = np.maximum(a_full, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            break
        a = a / s

        # Frequency domain: flatten the magnitude
        A_new = np.fft.fft(a, N)
        mag = np.abs(A_new)
        tgt_mag = np.ones(N)
        # Smooth transition
        tgt_mag = 0.7 * np.ones(N) + 0.3 * mag / (mag.max() + 1e-10)
        phase = np.angle(A_new)
        A = tgt_mag * np.exp(1j * phase)

    return a


def _make_linear_profile(n, eta=1.8):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_quad_profile(n, a1=1.5, a2=0.8):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_cubic_profile(n, a1=2.0, a3=-1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _optimize_candidate(a0, deadline):
    """Run full two-phase optimization on a candidate."""
    n = len(a0)
    best_a = a0.copy()
    best_sc = _score(a0)

    # Phase 1: L2 flattening (30% of time)
    t_remaining = deadline - time.monotonic()
    if t_remaining > 30:
        d1 = time.monotonic() + 0.3 * (t_remaining - 5)
        a1, s1 = _adam_l2_flatten(a0, d1, max_iter=10000, lr=0.04)
        if s1 < best_sc:
            best_a, best_sc = a1, s1

    # Phase 2: Smooth-max fine tuning (55% of time)
    t_remaining = deadline - time.monotonic()
    if t_remaining > 20:
        d2 = time.monotonic() + 0.55 * (t_remaining - 5)
        a2, s2 = _adam_smooth_max_v2(best_a, d2, max_iter=20000, lr=0.015, beta_factor=0.03)
        if s2 < best_sc:
            best_a, best_sc = a2, s2

    # Phase 3: Peak shaving (10% of time)
    t_remaining = deadline - time.monotonic()
    if t_remaining > 15:
        d3 = time.monotonic() + 0.1 * (t_remaining - 3)
        a3, s3 = _peak_shaving(best_a, d3, max_iter=3000)
        if s3 < best_sc:
            best_a, best_sc = a3, s3

    # Phase 4: Final fine Adam (5% of time)
    t_remaining = deadline - time.monotonic()
    if t_remaining > 10:
        a4, s4 = _adam_smooth_max_v2(best_a, time.monotonic() + t_remaining * 0.05,
                                     max_iter=5000, lr=0.005, beta_factor=0.01)
        if s4 < best_sc:
            best_a, best_sc = a4, s4

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Two-phase spectral optimization: flatten autocorrelation via L2, then reduce peaks via smooth-max.
    Uses n=2048, IAA initialization, multiple profiles, and adaptive Adam.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    best_seq = None
    best_score = np.inf
    n_target = 2048

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations for n=2048
    inits = []

    if parent is not None:
        p2048 = _interpolate_profile(parent, n_target)
        inits.append(("parent", p2048))
        # Perturbed parent
        rng = np.random.default_rng(seed)
        pp = p2048 * (1.0 + 0.1 * rng.normal(size=n_target))
        pp = np.maximum(pp, 0.0)
        inits.append(("parent_perturb", pp / pp.sum()))

    inits.append(("linear", _make_linear_profile(n_target, 1.8)))
    inits.append(("linear2", _make_linear_profile(n_target, 2.5)))
    inits.append(("quad", _make_quad_profile(n_target, 1.5, 0.8)))
    inits.append(("cubic", _make_cubic_profile(n_target, 2.0, -1.5)))

    # IAA initialization
    try:
        iaa = _iaa_init(n_target, iterations=150)
        inits.append(("iaa", iaa))
    except Exception:
        pass

    print(f"Number of initializations: {len(inits)}")
    for name, a in inits:
        sc = _score(a)
        print(f"  Init '{name}': score={sc:.10f}")

    # Allocate time: spend more on the best initializations
    # Sort by initial score
    scored = [(name, a, _score(a)) for name, a in inits]
    scored.sort(key=lambda x: x[2])

    # Strategy: give 40% of time to the top 2 inits, 30% to next 2, 30% to rest
    total_time = deadline - time.monotonic() - 60  # reserve 60s for final polish
    n_inits = len(scored)

    # Time allocation
    if n_inits >= 4:
        times = [0.25, 0.20, 0.15, 0.15, 0.10, 0.10, 0.05]
    elif n_inits >= 2:
        times = [0.40, 0.30, 0.30][:n_inits]
    else:
        times = [1.0]

    for i, (name, a0, sc0) in enumerate(scored):
        if time.monotonic() > deadline - 65:
            break
        alloc = total_time * times[min(i, len(times) - 1)]
        if alloc < 20:
            alloc = 20
        d_i = min(deadline - 65 - i * 2, time.monotonic() + alloc)
        if d_i <= time.monotonic() + 15:
            break

        print(f"\n=== Optimizing '{name}' (init score={sc0:.10f}, budget={alloc:.1f}s) ===")
        a_best, s_best = _optimize_candidate(a0, d_i)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  New best: {s_best:.10f}")
        else:
            print(f"  No improvement (best still {best_score:.10f})")

    # Final polish phase
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish (current best={best_score:.10f}) ===")
        remaining = deadline - time.monotonic() - 5

        # Interpolate to 2048 if needed
        if best_seq.size != n_target:
            best_2048 = _interpolate_profile(best_seq, n_target)
        else:
            best_2048 = best_seq.copy()

        # Fine Adam polish
        d_pol = time.monotonic() + remaining * 0.5
        a_pol, s_pol = _adam_smooth_max_v2(best_2048, d_pol, max_iter=10000, lr=0.003, beta_factor=0.008)
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol.copy()
            print(f"  Polish Adam: {s_pol:.10f}")

        # Peak shaving
        if time.monotonic() < deadline - 8:
            d_ps = time.monotonic() + (deadline - time.monotonic() - 8) * 0.4
            a_ps, s_ps = _peak_shaving(best_seq, d_ps, max_iter=2000)
            if s_ps < best_score:
                best_score = s_ps
                best_seq = a_ps.copy()
                print(f"  Polish Shave: {s_ps:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 5:
            a_uf, s_uf = _adam_smooth_max_v2(best_seq, deadline - 5, max_iter=5000, lr=0.001, beta_factor=0.003)
            if s_uf < best_score:
                best_score = s_uf
                best_seq = a_uf.copy()
                print(f"  Ultra-fine: {s_uf:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.8)
        best_score = _score(best_seq)

    # Final: ensure validity
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Validate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
