"""
Optimization of step function heights minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding with golden section line search,
symmetry exploitation, and coordinate descent refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _symmetrize(a):
    return (a + a[::-1]) / 2.0

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()

def _make_sym_ramp(n, alpha=1.0):
    # Symmetric: high in middle, low at edges
    t = np.linspace(-1.0, 1.0, n)
    a = (1.0 - np.abs(t)) ** alpha + 1e-6
    return a / a.sum()

def _make_cosine(n, alpha=1.0):
    t = np.linspace(0.0, np.pi, n)
    a = np.cos(t / 2) ** (2 * alpha) + 1e-6
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def _solve_lp(f, M, n):
    # max sum(g) s.t. A@ g <= M, g >= 0
    # A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # i.e., A[k,j] = f[k-j] if j <= k < j+n else 0
    # For efficiency, use the banded structure
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _golden_search(a, g, best_score):
    # Find best t in [0, 1] minimizing score((1-t)*a + t*g)
    a = np.maximum(a, 0)
    n = len(a)
    
    def f(t):
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return _score(a_new)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    a_lo, a_hi = 0.0, 1.0
    c = a_lo + (1 - gr) * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(40):  # ~40 iterations gives precision ~gr^40 ~ 1e-9
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_lo + (1 - gr) * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    
    t_best = (a_lo + a_hi) / 2
    sc = f(t_best)
    return t_best, sc

"""
LP-based optimization with golden section line search, symmetry exploitation,
coordinate descent, and perturbation-based restarts for minimizing
2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: tighter inflation, golden section search,
symmetrization, coordinate descent polish, and structured restarts.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return _normalize(a)


def _make_sym_tri(n, alpha=1.0):
    t = np.linspace(-1.0, 1.0, n)
    a = (1.0 - np.abs(t)) ** alpha + 1e-6
    return _normalize(a)


def _make_cosine(n, power=1.0):
    t = np.linspace(0.0, np.pi, n)
    a = np.cos(t / 2.0) ** (2.0 * power) + 1e-6
    a = np.maximum(a, 1e-6)
    return _normalize(a)


def _make_lin(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return _normalize(a)


def _make_quad(n, c=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c * t + c * t * t
    a = np.maximum(a, 1e-8)
    return _normalize(a)


def _make_gaussian(n, width=0.4):
    t = np.linspace(-1.0, 1.0, n)
    a = np.exp(-t * t / (2 * width * width))
    return _normalize(a)


def _make_bump(n, center=0.5, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    return _normalize(a)


def _build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, M, n, A=None):
    if A is None:
        A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _golden_section_search(a, g, n_iter=35):
    """Find best t in [0,1] for (1-t)*a + t*g using golden section search."""
    a_base = np.maximum(a, 0.0)
    g_base = np.maximum(g, 0.0)

    def f_of_t(t):
        a_new = (1.0 - t) * a_base + t * g_base
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return _score(a_new)

    # Include t=0 as a candidate
    f0 = f_of_t(0.0)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = lo + (1.0 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - gr) * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)

    t_mid = (lo + hi) / 2.0
    f_mid = f_of_t(t_mid)

    # Compare with t=0
    if f0 <= f_mid:
        return 0.0, f0
    return t_mid, f_mid


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.002,

    """LP-based refinement with golden section line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    if use_sym:
        a = _symmetrize(a)

    c0 = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c0.max())

    no_imp = 0
    inflate = inflate_start
    stuck_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        A = _build_conv_matrix(a, n)
        g = _solve_lp(a, M_lp, n, A=A)
        if g is None:

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:

            if no_imp > 200:

                if stuck_count > 5:

                a = best_a.copy()
                # Small random perturbation

                pert = rng.normal(0, 0.05, n)

                if use_sym:
                    a = _symmetrize(a)
                no_imp = 0
                inflate = inflate_start
                continue
            continue

        g = g / sum_g
        if use_sym:
            g = _symmetrize(g)
            g = np.maximum(g, 0.0)
            g = g / g.sum()

        t_best, sc_best = _golden_section_search(a, g, n_iter=30)

        if t_best > 0.0 and sc_best < best_score - 1e-14:

            a_new = np.maximum(a_new, 0.0)

        a = _symmetrize(a)

        if time.monotonic() > deadline:
            break

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            if use_sym and i >= n // 2:

                if j >= i:
                    continue
                # Handle pair (i, j) together
                old_i = a[i]
                old_j = a[j]
                best_local = best_score
                best_ii, best_jj = old_i, old_j

                    a[j] = old_j * factor
                    sc = _score(a)
                    if sc < best_local - 1e-15:

                a[i] = best_ii
                a[j] = best_jj
                if best_local < best_score - 1e-14:
                    best_score = best_local
                    best_a = a.copy()
                    improved = True
            else:
                old_val = a[i]
                best_local = best_score
                best_new = old_val
                for factor in [0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                    a[i] = old_val * factor

                    sc = _score(a)
                    if sc < best_local - 1e-15:
                        best_local = sc

                if best_local < best_score - 1e-14:

        if not improved:
            # Try smaller steps
            improved2 = False
            for i in range(n):
                if time.monotonic() > deadline:
                    break
                old_val = a[i]
                if old_val < 1e-15:
                    continue
                for factor in [0.9, 0.95, 1.05, 1.1]:
                    a[i] = old_val * factor
                    if use_sym:
                        a[n - 1 - i] = a[i]
                    sc = _score(a)
                    if sc < best_score - 1e-15:
                        best_score = sc
                        best_a = a.copy()
                        improved2 = True
                        break
                    a[i] = old_val
                    if use_sym:
                        a[n - 1 - i] = a[i]
                if improved2:
                    improved = True
            if not improved:
                break

    return best_a


def _gradient_polish(a0, deadline, max_iter=20000):
    """Smooth-max gradient descent with momentum."""
    a = np.asarray(a0, dtype=np.float64).copy()

    best_a = a.copy()
