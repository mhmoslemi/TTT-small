"""
Optimized step-function search minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses improved LP-based spreading, adaptive subgradient, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses sparse LP spreading, adaptive subgradient with line search,
symmetrization, peak-targeted moves, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix_sparse(a):
    """Build sparse convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    # For column j, non-zero at rows j..j+n-1 with values a[0]..a[n-1]
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        r = np.arange(j, j + n)
        rows_list.append(r)
        cols_list.append(np.full(n, j))
        vals_list.append(a)
    
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    """Solve max sum(b) s.t. a*b <= M, b >= 0. Returns b or None."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=50000, inflate_init=1.0001, blend_init=0.5):
    """LP-based iterative refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_conv(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflate = inflate_init
    blend = blend_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        M = float(c.max())
        
        # Solve LP with slightly inflated bound
        M_lp = M * inflate
        g = _lp_solve(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 100:
                # Try with larger inflation
                inflate = min(1.01, inflate * 1.5)
                no_imp = 0
                if inflate >= 1.01:
                    break
                continue
            continue

        # Normalize g
        g = g / sum_g

        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Evaluate
        c_new = _fft_conv(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Adapt: try larger blend, smaller inflate

                a = best_a.copy()
                blend = blend_init
                inflate = inflate_init

def _subgradient_polish(a, deadline, max_iter=50000):
    """Subgradient method with adaptive step size and momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_conv(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.05
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        M = float(c.max())
        
        # Find argmax positions (use top-k for robustness)
        k_top = min(5, 2 * n - 1)

            hi = min(n - 1, k)
            if lo <= hi:

        # grad[j] = sum_k w_k * 2 * a[k-j] (if 0 <= k-j < n)
        # For each k: a[k-j] is valid when 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
        # So: j ranges from max(0, k-n+1) to min(n-1, k)
        # And a[k-j] for j in that range: when j = max(0,k-n+1), k-j = min(k, n-1)
        #                                     when j = min(n-1,k), k-j = max(0, k-n+1)
        # So a[k-j] goes from a[min(k,n-1)] down to a[max(0,k-n+1)]
        
        # Let me just use the FFT-based approach for the gradient
        # The gradient of (a*a)_k w.r.t. a is the cross-correlation of a with delta_k
        # which is just a shifted/reversed a
        
        # Actually, the cleanest way: 
        # d/d(a_j) (a*a)_k = 2 * a[k-j] (with a indexed 0..n-1, a[i]=0 for i out of range)
        # So grad[j] = 2 * sum_k w_k * a[k-j]
        # This is 2 * (w * a_reversed) evaluated at the right positions
        # Specifically, if we define r = a[::-1] (reversed), then
        # sum_k w_k * a[k-j] = sum_k w_k * r[n-1-(k-j)] = sum_k w_k * r[n-1-k+j]
        # = (w * r)[n-1+j] (circular convolution)
        # Hmm, this is getting complicated. Let me just use the direct formula.
        
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            # For position j: contribution is w[i] * 2 * a[k-j] if 0 <= k-j < n
            # j ranges from max(0, k-n+1) to min(n-1, k)
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                # a[k-j] for j = j_lo..j_hi gives a[k-j_lo]..a[k-j_hi] = a[min(k,n-1)]..a[max(0,k-n+1)]
                # which is a reversed slice
                idx = k - np.arange(j_lo, j_hi + 1)  # k-j_lo, k-j_lo-1, ..., k-j_hi
                grad[j_lo:j_hi+1] += w[i] * 2.0 * a[idx]
        
        if not np.all(np.isfinite(grad)):
            grad = 2.0 * a[::-1]  # fallback: gradient at k=n-1 (the DC-like peak)
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_conv(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:

            best_a = a.copy()

            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > 2000:

                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

def _make_linear_profile(n, eta=1.5):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _peak_shave(a, deadline, max_iter=20000):
    """Peak shaving: directly reduce the peak by moving mass from high-contribution positions."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(a)
        M = float(c.max())
        k_peak = int(np.argmax(c))
        
        # Find positions j that contribute most to the peak
        # Contribution of a_j to (a*a)_{k_peak} is a_j * a[k_peak - j]
        contrib = np.zeros(n)
        j_lo = max(0, k_peak - (n - 1))
        j_hi = min(n - 1, k_peak)
        if j_lo <= j_hi:
            idx = k_peak - np.arange(j_lo, j_hi + 1)
            contrib[j_lo:j_hi+1] = a[j_lo:j_hi+1] * a[idx]
        
        # Find the position with the largest contribution
        j_max = j_lo + int(np.argmax(contrib[j_lo:j_hi+1]))
        
        # Find a position j2 with small "peak contribution" to move mass to
        # We want a position where a[j2] is small and a[k_peak - j2] is small
        peak_contrib_all = np.zeros(n)
        for j in range(n):
            if 0 <= k_peak - j < n:
                peak_contrib_all[j] = a[j] * a[k_peak - j]
            else:
                peak_contrib_all[j] = 0  # this position doesn't contribute to this peak
        
        # Move mass from j_max to the position with smallest peak contribution
        j2 = int(np.argmin(peak_contrib_all))
        
        # Try several fractions
        improved = False
        for frac in [0.1, 0.5, 0.1, 0.05, 0.02]:
            delta = frac * a[j_max]
            if delta <= 0:
                break
            a_try = a.copy()
            a_try[j_max] -= delta
            a_try[j2] += delta
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                break
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if not improved:
            no_imp += 1
            if no_imp > 2000:
                break
        else:
            no_imp = 0

    return best_a


def _symmetrize(a):
    """Symmetrize: (a + a_reversed) / 2"""
    n = len(a)

    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-30:
        return a
    return a_sym / s
