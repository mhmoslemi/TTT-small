import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score for sum(a)=1: 2n*max(conv)"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _build_lp_matrix(a):
    """Build constraint matrix for LP: (a*g)_k for all k"""
    n = len(a)
    # (a*g)_k = sum_j a_j * g_{k-j}
    # For each k, the constraint is: sum_j a_j * g_{k-j} <= M
    # The coefficient of g_j in constraint k is a_{k-j} (if 0 <= k-j < n)
    # So A[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    return A

def _lp_spread_full(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0"""
    n = len(a)
    A = _build_lp_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

                     lr=0.1, fast_frac=0.6):

    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    base_lr = lr
    
    # Estimate total steps
    total_steps = 100000  # will be limited by deadline
    fast_steps = int(total_steps * fast_frac)
    
    # Phase A cooling ratio
    ratio_a = (T_mid / T_start) ** (1.0 / fast_steps)
    # Phase B cooling ratio
    ratio_b = (T_end / T_mid) ** (1.0 / (total_steps - fast_steps))
    
    T = T_start
    it = 0
    
    while time.monotonic() < deadline:
        # Determine current temperature

        h = h_full[n-1:2*n-1]
        grad = 2.0 * h

"""
Optimized step function coefficient search using two-phase LSE gradient,
full LP-based spreading with adaptive M, swap local search, and multi-length exploration.
Improvements over previous: faster cooling schedule, full LP (not reduced),
binary-search on LP bound, pairwise swap local search, and n=2048 exploration.

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _build_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A


def _lp_spread(a, M):
    n = len(a)
    A = _build_lp_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_reduced(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[row, j] = a[k - j]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, n_evals=12):
    best_t = 0.0
    best_s = _score_norm(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t


def _lse_optimize_v2(a, deadline, T_start=0.06, T_mid=0.004, T_end=1e-5,
                     lr=0.08, fast_frac=0.55, max_iter=200000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    base_lr = lr
    N = _next_pow2(2 * n - 1)

    total_est = max_iter
    fast_steps = int(total_est * fast_frac)
    ratio_a = (T_mid / T_start) ** (1.0 / max(fast_steps, 1))
    ratio_b = (T_end / T_mid) ** (1.0 / max(total_est - fast_steps, 1))

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        if it < fast_steps:
            T = T_start * (ratio_a ** it)
        else:
            T = T_mid * (ratio_b ** (it - fast_steps))
        T = max(T, T_end)

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            s = 1.0
        else:
            w = ex / s

        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            it += 1
            continue

        state_t += 1
        b1, b2 = 0.9, 0.999
        state_m = b1 * state_m + (1.0 - b1) * gz
        state_v = b2 * state_v + (1.0 - b2) * (gz * gz)
        mh = state_m / (1.0 - b1 ** state_t)
        vh = state_v / (1.0 - b2 ** state_t)

        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()

        if not np.all(np.isfinite(a_new)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            it += 1
            continue

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 3000 == 0:
                lr = max(0.005, lr * 0.85)
            if no_imp > max(8000, 12 * n):
                rng_r = np.random.default_rng(1000 + it)

                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state_t = 0
                state_m = np.zeros(n)
                state_v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        a = a_new
        it += 1

    return best_a

def _swap_local_search(a, deadline, max_rounds=500):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        for trial in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            amt = 0.01 * a[i]
            if amt < 1e-12:
                continue

            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            if a_trial[i] < 0:
                a_trial[i] = 0.0
                a_trial[j] += amt

            sc = _score_norm(a_trial)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True

        if not improved and rnd % 50 == 0:
            break

    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.01):

            break

        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-10)[0]

        w[max_idx] = 1.0 / len(max_idx)

        fw = np.fft.rfft(w, N)

        h = h_full[n - 1:2 * n - 1]

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        a_new = np.maximum(a_new, 0.0)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cos(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_w=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_w * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_skewed(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_skew(n, center=0.4, width=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
