LP-direction + Adam hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Use LP to find optimal descent direction (minimize cross-correlation max),
then polish with Adam in logit space. Multi-scale escalation for larger n.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _lp_direction(p):
    """Find b that minimizes max cross-correlation (p*b) subject to sum(b)=1, b>=0.
    Returns (b, M) where b is optimal direction and M is the min achievable max cross-corr."""
    n = len(p)
    # Build constraint matrix: A[k, j] = p[k-j] if 0<=k-j<n else 0
    # for k=0..2n-2, j=0..n-1
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(float(p[k - j]))
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Variables: b[0..n-1], M
    # Objective: minimize M -> c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    # Constraints:
    # A @ b - M <= 0  =>  [A, -I] @ [b; M] <= 0
    # -sum(b) <= -1  =>  [-1, -1, ..., -1, 0] @ [b; M] <= -1
    # sum(b) >= 1     =>  [1, 1, ..., 1, 0] @ [b; M] >= 1  (or -sum(b) <= -1 and sum(b) >= 1)

    b_ub_full = np.zeros(2 * n - 1)
    
    # sum(b) = 1: [1...1 0] x = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]

"""
Hybrid LP-direction + adaptive Adam optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key ideas: (1) diverse asymmetric profile initialization, (2) adaptive-temperature smooth-max
Adam in logit space, (3) periodic LP-based direction finding for escaping local minima,
(4) multi-scale escalation (1024 -> 2048 -> 4096), (5) multi-restart with noise.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _lp_direction(p, max_time=5.0):
    """Find b minimizing max cross-correlation (p*b) subject to sum(b)=1, b>=0."""
    t0 = time.monotonic()
    n = len(p)
    if n > 4096:
        return None, None
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend([float(p[k - j]) for j in idx])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    # Full constraint: [A | -I_{2n-1}] @ [b; M] <= 0
    # Plus equality: sum(b) = 1
    A_ub = np.hstack([A.toarray(), -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={"time_limit": max(1.0, max_time - (time.monotonic() - t0))})
    if res.success:
        b = np.maximum(res.x[:n], 0.0)
        M = float(res.x[n])
        return b, M
    return None, None


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                   no_imp_limit=500, max_noise=0.2):
    """Adam optimization in logit space with adaptive smooth-max temperature."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_cross_corr(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-15:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)

        limit = max(100, no_imp_limit)
        if no_imp > limit:
            ns = min(max_noise, 0.05 + 0.02 * (no_imp // limit))
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_refine(a, deadline, t=0.4, max_solves=30, min_time_per_solve=0.1):
    """LP-based iterative refinement."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        remaining = deadline - time.monotonic()
        if remaining < min_time_per_solve:
            break
        b, _ = _lp_direction(a, max_time=min(remaining * 0.8, 5.0))
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        else:
            a_new = (1.0 - t * 0.5) * a + t * 0.5 * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if sc_new < current_score:
                current_score = sc_new
                current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.15, t * 0.92)
        n_solves += 1
    return current_best, current_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    return _normalize_nonneg(result)

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, deadline in {deadline-t_start:.0f}s")

    # === Phase 1: Generate diverse candidates ===
    phase1_end = t_start + 60.0
    candidates = []

    n_values = [512, 1024, 2048]
    profiles_per_n = {
        512: [
            ('lin2', _make_profile_linear(512, 2.0)),
            ('lin3', _make_profile_linear(512, 3.0)),
            ('quad1', _make_profile_quadratic(512, 1.5, 0.5)),
            ('quad2', _make_profile_quadratic(512, -1.5, 0.8)),
            ('exp1', _make_profile_exponential(512, 1.0)),
            ('exp2', _make_profile_exponential(512, 2.0)),
            ('pow1', _make_profile_power(512, 1.0)),
            ('pow2', _make_profile_power(512, 2.0)),
            ('cos_h1', _make_profile_cosine_half(512, 1.0)),
            ('cos_h2', _make_profile_cosine_half(512, 2.0)),
            ('cos1', _make_profile_cosine(512, 1, 0.7)),
            ('twob', _make_profile_two_bump(512, 0.2, 0.8, 0.2)),
        ],
        1024: [
            ('lin2', _make_profile_linear(1024, 2.0)),
            ('lin3', _make_profile_linear(1024, 3.0)),
            ('lin4', _make_profile_linear(1024, 4.0)),
            ('quad1', _make_profile_quadratic(1024, 1.5, 0.5)),
            ('quad2', _make_profile_quadratic(1024, -2.0, 1.0)),
            ('quad3', _make_profile_quadratic(1024, 2.0, -0.5)),
            ('exp1', _make_profile_exponential(1024, 1.0)),
            ('exp2', _make_profile_exponential(1024, 2.0)),
            ('exp3', _make_profile_exponential(1024, 3.0)),
            ('pow1', _make_profile_power(1024, 1.0)),
            ('pow15', _make_profile_power(1024, 1.5)),
            ('pow2', _make_profile_power(1024, 2.0)),
            ('cos_h1', _make_profile_cosine_half(1024, 1.0)),
            ('cos_h15', _make_profile_cosine_half(1024, 1.5)),
            ('cos_h2', _make_profile_cosine_half(1024, 2.0)),
            ('cos_h3', _make_profile_cosine_half(1024, 3.0)),
            ('cos1', _make_profile_cosine(1024, 1, 0.8)),
            ('cos2', _make_profile_cosine(1024, 2, 0.5)),
            ('twob', _make_profile_two_bump(1024, 0.2, 0.8, 0.25)),
            ('twob2', _make_profile_two_bump(1024, 0.1, 0.9, 0.3)),
        ],
        2048: [
            ('lin2', _make_profile_linear(2048, 2.0)),
            ('lin3', _make_profile_linear(2048, 3.0)),
            ('quad1', _make_profile_quadratic(2048, 1.5, 0.5)),
            ('exp2', _make_profile_exponential(2048, 2.0)),
            ('pow15', _make_profile_power(2048, 1.5)),
            ('cos_h2', _make_profile_cosine_half(2048, 2.0)),
            ('cos_h3', _make_profile_cosine_half(2048, 3.0)),
        ],
    }

    for n in n_values:
        for label, p in profiles_per_n.get(n, []):
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None and 0 < pn <= 4096:
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))
        for n in [1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # === Phase 2: Polish top candidates with Adam ===
    phase2_end = t_start + 500.0
    top_k = 6
    for i in range(min(top_k, len(candidates))):
        if time.monotonic() > phase2_end - 30:
            break
        sc, p, n, label = candidates[i]
        per_end = min(phase2_end - 10, time.monotonic() + 70.0)
        if per_end <= time.monotonic() + 10:
            break
        p_b, sc_b = _adam_optimize(p, per_end, rng, lr=0.06, beta_frac=0.1,
                                   beta_decay=0.93, no_imp_limit=300, max_noise=0.2)
        _update(p_b, sc_b, f'adam_{label}_n{n}')

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP refinement on best ===
    phase3_end = t_start + 750.0
    if best_seq is not None and best_seq.size <= 2048:
        p_lp, sc_lp = _lp_refine(best_seq, phase3_end, t=0.4, max_solves=25)
        _update(p_lp, sc_lp, 'LP')
        if time.monotonic() < phase3_end - 20:
            p_ad, sc_ad = _adam_optimize(p_lp, phase3_end, rng, lr=0.04,
                                         beta_frac=0.07, beta_decay=0.94,
                                         no_imp_limit=300, max_noise=0.1)
            _update(p_ad, sc_ad, 'post_LP')

    # === Phase 4: Scale escalation ===
    phase4_end = t_start + 1200.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_cur = best_seq.size
        if n_cur <= 1024:
            target_n = 2048
        elif n_cur <= 2048:
            target_n = 4096
        else:
            target_n = 0

        if target_n > 0 and time.monotonic() < phase4_end - 30:
            p_int = _interp_profile(best_seq, target_n)
            if p_int is not None:
                sc_int = _score_norm(p_int)
                print(f"  Escalate to n={target_n}: {sc_int:.10f}")
                _update(p_int, sc_int, f'interp_n{target_n}')

                # Adam on the escalated sequence
                per = min(phase4_end - time.monotonic() - 20, 200.0)
                if per > 30:
                    p_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                               lr=0.04, beta_frac=0.07,
                                               beta_decay=0.94, no_imp_limit=500,
                                               max_noise=0.12)
                    _update(p_e, sc_e, f'escalate_n{target_n}')

                # LP if n <= 2048
                if target_n <= 2048 and time.monotonic() < phase4_end - 15:
                    best_for_lp = best_seq if best_seq.size <= 2048 else p_int
                    p_lp2, sc_lp2 = _lp_refine(best_for_lp, min(phase4_end - 5,
                                                                time.monotonic() + 120),
                                               t=0.3, max_solves=15)
                    _update(p_lp2, sc_lp2, f'LP_n{best_for_lp.size}')

    # === Phase 5: Multi-restart with noise ===
    phase5_end = deadline - 40.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        remaining = phase5_end - time.monotonic()
        n_restarts = max(2, int(remaining // 80))
        n_restarts = min(n_restarts, 5)
        per_r = remaining / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 20:
                break
            noise_scale = 0.08 + 0.04 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_optimize(p_s, time.monotonic() + per, rng,
                                       lr=0.05, beta_frac=0.07, beta_decay=0.94,
                                       no_imp_limit=300, max_noise=0.15)
            _update(p_r, sc_r, f'restart{i}_ns{noise_scale:.2f}')

    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            p_f, sc_f = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                       lr=0.015, beta_frac=0.04, beta_decay=0.96,
                                       no_imp_limit=200, max_noise=0.05)
            _update(p_f, sc_f, 'final_polish')

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 2.0)
        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 2.0)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
