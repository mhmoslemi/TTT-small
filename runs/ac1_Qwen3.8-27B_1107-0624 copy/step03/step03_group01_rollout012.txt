        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.8)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _active_set_polish(a_init, deadline, n_iter=5000):
    """Subgradient method focusing on active (near-max) lags."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.005
    momentum = np.zeros(n)
    mom_coeff = 0.9

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: lags within 1% of max
        active = c >= cm * 0.999
        if not np.any(active):
            active[int(np.argmax(c))] = True

        # Average subgradient over active lags
        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        # Momentum step
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if it % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                # Revert if got worse
                lr = lr * 0.9

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation -> Nesterov with aggressive
    beta schedule -> Adam polishing -> active-set subgradient -> fine restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 2048
        parent = _make_linear_profile(n, 1.5)
        print(f"No parent, using linear profile n={n}")

    # --- Phase 0: Quick evaluation of starting points (20s) ---
    phase0_end = min(deadline, t_start + 20.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
    update_best(parent, sc_parent, "parent")

    for sn in [1024, 2048]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_linear_profile(sn, 1.5),
            _make_quadratic_profile(sn, 1.0, 0.2),
            _make_cubic_profile(sn, 0.8, -0.4),
        ]
        for idx, p in enumerate(profiles):
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"prof{idx}_n{sn}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 1: L-BFGS continuation on best candidate (250s) ---
    phase1_end = min(deadline, t_start + 270.0)
    if best_seq is not None:
        per_time = min(200.0, phase1_end - time.monotonic())
        if per_time > 10:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_phase1")

    # Also try L-BFGS on second-best candidate if time allows
    if time.monotonic() < phase1_end - 30:
        for sc, p, pn, label in candidates[1:3]:
            if time.monotonic() > phase1_end - 10:
                break
            if len(p) != len(best_seq):
                continue
            per_time = min(40.0, phase1_end - time.monotonic())
            if per_time > 5:
                p_best, sc_best = _run_lbfgs(p, phase1_end, max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with aggressive continuation (350s) ---
    phase2_end = min(deadline, t_start + 620.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs, lr_init=0.015, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov_phase2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polishing (300s) ---
    phase3_end = min(deadline, t_start + 920.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(250.0, phase3_end - time.monotonic())
        p_best, sc_best = _adam_optimize(best_seq, t_start + t_start + per_time - t_start + time.monotonic() + per_time,
                                         lr=0.03, beta_frac=0.005)
        # Fix deadline
        p_best, sc_best = _adam_optimize(best_seq, min(phase3_end, time.monotonic() + per_time),
                                         lr=0.03, beta_frac=0.005)
        update_best(p_best, sc_best, "adam_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Nesterov restarts (250s) ---
    phase4_end = min(deadline, t_start + 1170.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(60.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            noise_level = 0.05 + 0.03 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short, lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine polishing with very small beta (250s) ---
    phase5_end = min(deadline, t_start + 1420.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        # Nesterov with tiny beta
        per_time = min(100.0, phase5_end - time.monotonic() - 100)
        if per_time > 10:
            beta_fracs_tiny = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7]
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase5_end - 80, time.monotonic() + per_time),
                                                  beta_fracs_tiny, lr_init=0.005, mom_init=0.95)
            update_best(p_best, sc_best, "fine_nesterov")

        # Adam with tiny beta
        if time.monotonic() < phase5_end - 20:
            per_time2 = min(100.0, phase5_end - time.monotonic() - 10)
            if per_time2 > 10:
                p_best, sc_best = _adam_optimize(best_seq, min(phase5_end - 5, time.monotonic() + per_time2),
                                                  lr=0.01, beta_frac=0.001)
                update_best(p_best, sc_best, "fine_adam")

        # Active set polish
        if time.monotonic() < phase5_end - 10:
            per_time3 = min(80.0, phase5_end - time.monotonic() - 5)
            if per_time3 > 10:
                p_best, sc_best = _active_set_polish(best_seq, min(phase5_end - 2, time.monotonic() + per_time3))
                update_best(p_best, sc_best, "active_set")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different length (200s) ---
    phase6_end = min(deadline, t_start + 1620.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        # Try doubling
        if n_curr <= 2048:
            target_n = min(4096, n_curr * 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase6_end - 30:
                per_time = min(120.0, phase6_end - time.monotonic() - 20)
                if per_time > 20:
                    p_best, sc_best = _run_lbfgs(p_interp, min(phase6_end - 10, time.monotonic() + per_time),
                                                  max_wall_time=per_time)
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}")

        # Try halving (in case parent was too long)
        if n_curr >= 2048:
            target_n = max(512, n_curr // 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
            elif time.monotonic() < phase6_end - 50:
                per_time = min(60.0, phase6_end - time.monotonic() - 20)
                if per_time > 10:
                    p_best, sc_best = _run_lbfgs(p_interp, min(phase6_end - 10, time.monotonic() + per_time),
                                                  max_wall_time=per_time)
                    update_best(p_best, sc_best, f"lbfgs_n{target_n}_short")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 30:
            # Final Nesterov
            per_time = min(60.0, remaining - 20)
            beta_fracs_final = [1e-6, 5e-7, 2e-7, 1e-7]
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase7_end - 20, time.monotonic() + per_time),
                                                  beta_fracs_final, lr_init=0.003, mom_init=0.97)
            update_best(p_best, sc_best, "final_nesterov")

        if time.monotonic() < phase7_end - 10:
            per_time = min(40.0, phase7_end - time.monotonic() - 5)
            if per_time > 5:
                p_best, sc_best = _adam_optimize(best_seq, min(phase7_end - 2, time.monotonic() + per_time),
                                                  lr=0.005, beta_frac=0.0005)
                update_best(p_best, sc_best, "final_adam")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Try evaluate_sequence for authoritative score
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal score: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(2048, 1.5)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
