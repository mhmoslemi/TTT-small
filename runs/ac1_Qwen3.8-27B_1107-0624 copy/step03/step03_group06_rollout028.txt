def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    
    # Compute softmax weights
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient: grad_j = 2 * sum_k w_k * a_{k-j} (for valid indices)
    # This is 2 * (w * a_rev) truncated
    a_rev = a[::-1]
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a_rev, N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _nesterov_polish(a0, deadline, max_iter=50000):
    """Nesterov-accelerated gradient descent on smooth max of conv(a,a)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(c.max())
    
    beta = max(1e-6, 0.02 * float(c.max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.75
    v = np.zeros(n)
    no_imp = 0
    y = a.copy()  # Nesterov lookahead
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy
        
        grad, maxc = _smooth_max_grad(y, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 2.0)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        # Evaluate
        c = _fft_convolve(a)
        score = float(c.max())
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.93)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.02 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
                y = a.copy()
        
        if it % 50 == 49:
            beta = max(1e-8, beta * 0.85)
    
    return best_a

def _random_perturb(a, sigma=0.01):
    """Add random perturbation and re-normalize."""
    n = len(a)
    noise = np.random.default_rng().normal(0, sigma, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses active-set LP with adaptive constraints, Nesterov-accelerated smooth-max
gradient descent, multiple sequence lengths, and perturbation restarts.
Key innovations: adaptive active-set size, vectorized LP matrix construction,
and tighter convergence criteria.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_profiles(n):
    t = np.linspace(0.0, 1.0, n)
    profiles = {}
    for alpha in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0]:
        a = t**alpha + 1e-8
        profiles[f'pw{alpha}'] = a / a.sum()
    for beta in [1.0, 1.5, 2.0, 3.0]:
        a = np.cos(np.pi * t * beta / 2)**2 + 1e-8
        prof_name = f'cs{beta}'
        profiles[prof_name] = a / a.sum()
    center = n // 2
    for sig_frac in [0.2, 0.3, 0.4, 0.5]:
        sigma = n * sig_frac
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - center) / sigma)**2) + 1e-8
        profiles[f'ga{sig_frac}'] = a / a.sum()
    for slope in [1.0, 2.0, 3.0, 5.0, 8.0]:
        a = 1.0 + slope * t
        a = np.maximum(a, 1e-8)
        profiles[f'ln{slope}'] = a / a.sum()
    for edge_w in [2.0, 4.0, 6.0]:
        a = np.exp(-edge_w * t) + 1e-8
        profiles[f'el{edge_w}'] = a / a.sum()
        a = np.exp(-edge_w * (1 - t)) + 1e-8
        profiles[f'er{edge_w}'] = a / a.sum()
    a = np.exp(-4.0 * np.abs(2 * t - 1)) + 1e-8
    profiles['edge'] = a / a.sum()
    return profiles


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _build_lp_matrix(a, active_indices):
    n = len(a)
    n_act = len(active_indices)
    A = np.zeros((n_act, n), dtype=np.float64)
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            A[idx, j_min:j_max + 1] = a[k - j_min:k - j_max - 1:-1]
    return A


def _solve_lp(a, M, active_indices):
    n = len(a)
    if len(active_indices) == 0:
        return np.ones(n) / n
    A = _build_lp_matrix(a, active_indices)
    b = np.full(len(active_indices), M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, best_score, n_grid=12):
    best_t = 0.0
    best_sc = best_score
    ts = np.linspace(0.05, 1.0, n_grid)
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        c = _fft_convolve(a_new)
        sc = float(c.max())
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=80000, inflate_start=1.002,
               inflate_min=1.0001, top_frac_start=0.10, top_frac_end=0.05):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(c.max())

    no_imp = 0
    inflate = inflate_start
    total_iters = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        total_iters += 1

        c = _fft_convolve(a)
        M = float(c.max())

        progress = min(1.0, total_iters / 5000.0)
        top_frac = top_frac_start + (top_frac_end - top_frac_start) * progress
        n_active = max(16, int(n * top_frac))
        n_active = min(n_active, 2 * n - 1)

        if n_active < 2 * n - 1:
            top_idx = np.argpartition(c, -n_active)[-n_active:]
        else:
            top_idx = np.arange(2 * n - 1)

        M_lp = M * inflate
        g = _solve_lp(a, M_lp, top_idx)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.1, inflate * 1.15)
            if no_imp > 500:
                break
            continue

        g = g / sum_g

        best_t, best_sc = _line_search(a, g, best_score)

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 300 == 299:
                inflate = max(inflate_min, inflate * 0.93)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.1, inflate * 1.08)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate > 0.05:
                    break

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    a_rev = a[::-1]
    N = _next_pow2(3 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _nesterov_polish(a0, deadline, max_iter=80000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(c.max())

    beta = max(1e-7, 0.03 * float(c.max()))
    lr = 0.04
    base_lr = lr
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    y = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        y = a + momentum * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy

        grad, maxc = _smooth_max_grad(y, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-7, beta * 3.0)
            v = np.zeros(n)
            y = a.copy()
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.003, lr * 0.92)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                y = a.copy()
                beta = max(1e-7, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 80 == 79:
            beta = max(1e-9, beta * 0.82)

    return best_a


def _perturb_and_optimize(a, rng, deadline):
    n = len(a)
    a_pert = a + rng.normal(0, 0.003, n)
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a
    a_pert = a_pert / s
    quick_end = time.monotonic() + min(30.0, max(5.0, deadline - time.monotonic() - 10))
    return _lp_refine(a_pert, quick_end, max_iter=3000,
                      inflate_start=1.005, top_frac_start=0.15, top_frac_end=0.10)
