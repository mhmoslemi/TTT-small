def _cutting_plane_lp(a, M, max_rounds=50):
    n = len(a)
    c_obj = -np.ones(n)
    # Start with no constraints (or a few)
    active_constraints = []  # list of lag indices
    
    # Add the most likely active constraints initially
    c_aa = _fft_convolve(a)
    top_lags = np.argsort(c_aa)[::-1][:30]
    active_constraints = list(top_lags)
    
    for round in range(max_rounds):
        # Build constraint matrix
        A_ub = np.zeros((len(active_constraints), n))
        b_ub = np.full(len(active_constraints), M)
        for i, k in enumerate(active_constraints):
            for j in range(max(0, k-n+1), min(n, k+1)):
                A_ub[i, j] = a[j]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0, None)]*n, method='highs')
        if not res.success:
            return None
        
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-8:
            return b
        
        # Check for violated constraints
        ab = _fft_convolve_poly(a, b)  # need to compute a*b efficiently
        violations = np.where(ab > M + 1e-10)[0]
        
        if len(violations) == 0:
            return b
        
        # Add violated constraints (top 10)
        new_lags = violations[np.argsort(ab[violations])[::-1]][:10]
        for lag in new_lags:
            if lag not in active_constraints:
                active_constraints.append(lag)
    
    return None

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using cutting-plane LP, LSE gradient,
and multi-start with diverse initializations. Targets improvement over 1.5067.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, N=None):
    """Forward real FFT."""
    a = np.asarray(a, dtype=np.float64)
    if N is None:
        N = _next_pow2(a.size)
    return np.fft.rfft(a, N)


def _fft_convolve(a):
    """Linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0)  # remove tiny negative artifacts
    return c


def _fft_cross_conv(a, b):
    """Linear cross-correlation a*b via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n1, n2 = a.size, b.size
    N = _next_pow2(n1 + n2 - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n1 + n2 - 1]
    return c


def _score(a):
    """Full score: 2n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _build_lp_matrix(a, lags):
    """Build constraint matrix for LP: (a*b)_k = sum_j A[k,j] * b[j] for k in lags."""
    n = len(a)
    rows = len(lags)
    A = np.zeros((rows, n))
    for i, k in enumerate(lags):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[i, j] = a[k - j]
    return A


def _cutting_plane_lp(a, M, time_budget, max_rounds=200, init_topk=50):
    """
    Solve: max sum(b) s.t. (a*b)_k <= M for active k, b >= 0.
    Uses cutting plane (constraint generation).
    Returns optimal b or None.
    """
    n = len(a)
    c_obj = -np.ones(n)
    
    # Initial active set: top lags of a*a
    c_aa = _fft_convolve(a)
    top_idx = np.argsort(c_aa)[::-1][:init_topk]
    active_lags = set(int(x) for x in top_idx)
    
    best_b = None
    best_sum = 0.0
    
    t0 = time.monotonic()
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_budget:
            break
        
        lags_sorted = sorted(active_lags)
        A_ub = _build_lp_matrix(a, lags_sorted)
        b_ub = np.full(len(lags_sorted), M)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        
        if not res.success:
            # Try with slightly inflated M
            continue
        
        b = res.x
        sum_b = float(b.sum())
        
        if sum_b > best_sum:
            best_sum = sum_b
            best_b = b.copy()
        
        if sum_b <= 1.0 + 1e-7:
            break
        
        # Check for violated constraints
        ab = _fft_cross_conv(a, b)
        violations = np.where(ab > M + 1e-9)[0]
        
        if len(violations) == 0:
            break
        
        # Add top violators
        worst = violations[np.argsort(ab[violations])[::-1]][:20]
        for v in worst:
            active_lags.add(int(v))
    
    return best_b


def _lp_refine(a, deadline, max_iter=2000, inflate_start=1.001, inflate_min=1.0001,
               blend_start=0.3, blend_min=0.02):
    """
    LP-based refinement: iteratively find b that spreads the convolution,
    then blend a with b.
    """
    a = _normalize(a)
    n = len(a)
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    
    no_imp = 0
    inflate = inflate_start
    blend = blend_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        t_budget = min(5.0, (deadline - time.monotonic()) / max(max_iter - it, 1))
        b = _cutting_plane_lp(a, M_lp, time_budget, max_rounds=50, init_topk=30)
        
        if b is None:
            inflate = min(1.01, inflate * 1.5)
            no_imp += 5
            if no_imp > 200:
                break
            continue
        
        sum_b = float(b.sum())
        if sum_b <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.01, inflate * 1.1)
            if no_imp > 300:
                break
            continue
        
        b = b / sum_b
        
        # Line search on blend factor
        best_t = blend
        best_sc = _score_norm(a)
        
        # Try a few blend factors
        for t in [blend, blend * 0.5, blend * 2.0, 0.1, 0.5]:
            t = min(max(t, 0.001), 0.99)
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())
        
        if sc_new < best_score - 1e-11:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            inflate = max(inflate_min, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend = max(blend_min, blend * 0.85)
                inflate = max(inflate_min, inflate * 0.95)
            if no_imp > max(150, n):
                a = best_a.copy()
                inflate = inflate_start
                blend = blend_start
                no_imp = 0
        
        a = a_new
    
    return best_a


def _lse_step(a, z, T, lr, state):
    """One LSE gradient step with adaptive temperature."""
    n = len(a)
    c = _fft_convolve(a)
    m = float(c.max())
    x = (c - m) / T
    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient in logit space: d/dz_i [sum_k softmax((c-m)/T)_k * c_k]
    # = (2/T) * (conv(w, a_rev) - <a, grad>) * a_i ... simplified
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_flip = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_flip, N)[n-1:2*n-1]
    grad = 2.0 * h / T
    
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= gz.mean()
    
    if not np.all(np.isfinite(gz)):
        return a, z, state, m
    
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    state['m'] = b1 * state['m'] + (1 - b1) * gz
    state['v'] = b2 * state['v'] + (1 - b2) * gz * gz
    mh = state['m'] / (1 - b1**t)
    vh = state['v'] / (1 - b2**t)
    
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50, 50)
    z -= z.mean()
    
    e = np.exp(z)
    a_new = e / e.sum()
    
    state['t'] = t
    return a_new, z, state, m


def _lse_optimize(a, deadline, T_start=0.03, T_end=0.001, lr=0.05, max_iter=50000):
    """LSE gradient optimization with annealing."""
    a = _normalize(a)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    T = T_start
    no_imp = 0
    base_lr = lr
    
    # Precompute decay ratio
    total_steps = max(max_iter, 10)
    T_ratio = (T_end / T_start) ** (1.0 / total_steps)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a, z, state, m = _lse_step(a, z, T, lr, state)
        
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-30))
            z -= z.mean()
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        
        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.01, lr * 0.92)
            if no_imp > max(5000, 5 * n):
                rng = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng.normal(size=n))
                a = _normalize(a)
                z = np.log(np.maximum(a, 1e-30))
                z -= z.mean()
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        
        T = max(T * T_ratio, T_end)
    
    return best_a


def _subgrad_polish(a, deadline, lr_init=0.01, max_iter=30000):
    """Polish with subgradient at the max."""
    a = _normalize(a)
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    v = np.zeros(n)
    mu = 0.9
    lr = lr_init
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_flip = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_flip, N)[n-1:2*n-1]
        grad = 2.0 * h
        
        dot = float(np.dot(a, grad))
        g_tan = grad - dot * a
        
        v = mu * v - lr * g_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        sc = float(2.0 * n * c_new.max())
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.001, lr * 0.95)
                mu = max(0.5, mu * 0.99)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                mu = 0.9
                no_imp = 0
    
    return best_a


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_flat_edges(n, w=0.3):
    t = np.linspace(0, 1, n)
    a = 1.0 + w * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_linear(n, eta=1.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interpolate(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == new_n:
        return _normalize(a)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_n)
    p = np.interp(x1, x0, a)
    return _normalize(p)

Actually let me just think about it differently. The constraint `(a*b)_k <= M` means `sum_j a_j * b_{k-j} <= M` where the sum is over j such that 0<=j<n and 0<=k-j<n.

For fixed k, the coefficient of b_j in this constraint is a_{k-j} (when j is in the valid range).

So the constraint matrix A has:
A[k,j] = a_{k-j} if 0<=k-j<n, else 0

This is a "hankel-like" matrix (not exactly Hankel because of the boundary effects).

For the dual, the constraint for variable j is:
sum_k A[k,j] * y_k >= 1  (since the primal objective is max 1^T b, the dual constraint RHS is 1)

sum_k a_{k-j} * y_k >= 1  (summing over valid k)

This says: for each position j, the "weighted overlap" with the dual variable y (using a as weights) must be at least 1.

The dual objective is min M * sum_k y_k.

At the optimum, complementary slackness gives:
- b_j > 0 implies sum_k a_{k-j} y_k = 1 (dual constraint tight)
- y_k > 0 implies (a*b)_k = M (primal constraint tight)

So the support of y tells us which lags are at the peak in the optimal solution.

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using cutting-plane LP with adaptive inflation,
LSE gradient with annealing, and subgradient polishing. Multi-start with diverse profiles.
Key improvements over prior: constraint-generation LP, zoom-in inflation schedule,

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _fft_cross_conv(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n1, n2 = a.size, b.size
    N = _next_pow2(n1 + n2 - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n1 + n2 - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _build_full_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def _build_lp_matrix_lags(a, lags):
    n = len(a)
    rows = len(lags)
    A = np.zeros((rows, n))
    for i, k in enumerate(lags):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[i, j] = a[k - j]
    return A


def _solve_lp(a, M, use_cutting_plane=False, init_topk=40, time_budget=10.0, max_rounds=80):
    """Solve max sum(b) s.t. (a*b)_k <= M, b>=0. Returns b or None."""
    n = len(a)
    c_obj = -np.ones(n)

    if not use_cutting_plane or n <= 1500:
        A_ub = _build_full_lp_matrix(a)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_budget})
        if res.success and res.x is not None:
            return res.x
        return None
    else:
        c_aa = _fft_convolve(a)
        top_idx = np.argsort(c_aa)[::-1][:init_topk]
        active = set(int(x) for x in top_idx)
        best_b = None
        best_sum = 0.0
        t0 = time.monotonic()
        for _ in range(max_rounds):
            if time.monotonic() - t0 > time_budget:
                break
            lags_sorted = sorted(active)
            A_ub = _build_lp_matrix_lags(a, lags_sorted)
            b_ub = np.full(len(lags_sorted), M)
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True})
            if not res.success:
                break
            b = res.x
            sum_b = float(b.sum())
            if sum_b > best_sum:
                best_sum = sum_b
                best_b = b.copy()
            if sum_b <= 1.0 + 1e-7:
                break
            ab = _fft_cross_conv(a, b)
            viol = np.where(ab > M + 1e-9)[0]
            if len(viol) == 0:
                break
            worst = viol[np.argsort(ab[viol])[::-1]][:20]
            for v in worst:
                active.add(int(v))
        return best_b


def _lp_refine(a, deadline, max_iter=3000, inflate_start=0.005, inflate_end=0.00005,
               blend_start=0.3, blend_end=0.02):
    """LP refinement with adaptive zoom-in."""
    a = _normalize(a)
    n = len(a)
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    no_imp = 0
    total_iter = max(max_iter, 100)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        frac = it / total_iter
        inflate = inflate_start * (inflate_end / inflate_start) ** frac
        blend = blend_start * (blend_end / blend_start) ** frac
        inflate = max(inflate, 0.00001)

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * (1.0 + inflate)

        use_cp = n > 1500
        t_budget = min(3.0, max(0.5, (deadline - time.monotonic()) / max(max_iter - it, 1)))
        b = _solve_lp(a, M_lp, use_cutting_plane=use_cp, init_topk=30,
                      time_budget=t_budget, max_rounds=40)

        if b is None:
            no_imp += 3
            if no_imp > 100:
                break
            continue

        sum_b = float(b.sum())
        if sum_b <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 150:
                break
            continue

        b = b / sum_b

        # Line search: try multiple blend factors
        best_t = blend
        best_sc = _score_norm(a)
        for t in [blend, blend * 0.5, blend * 2.0, 0.05, 0.15, 0.5]:
            t = min(max(t, 0.005), 0.95)
            a_try = (1.0 - t) * a + t * b
            a_try = _normalize(a_try)
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * b
        a_new = _normalize(a_new)
        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(80, n // 4):
                a = best_a.copy()
                no_imp = 0

        a = a_new

    return best_a


def _lse_step(a, z, T, lr, state):
    n = len(a)
    c = _fft_convolve(a)
    m = float(c.max())
    x = (c - m) / T
    x = np.clip(x, -80, 80)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_flip = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_flip, N)[n - 1:2 * n - 1]
    grad = 2.0 * h / T

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= gz.mean()

    if not np.all(np.isfinite(gz)):
        return a, z, state, m

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    state['m'] = b1 * state['m'] + (1.0 - b1) * gz
    state['v'] = b2 * state['v'] + (1.0 - b2) * gz * gz
    mh = state['m'] / (1.0 - b1 ** t)
    vh = state['v'] / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50, 50)
    z -= z.mean()

    e = np.exp(z)
    a_new = e / e.sum()
    state['t'] = t
    return a_new, z, state, m


def _lse_optimize(a, deadline, T_start=0.03, T_end=0.001, lr=0.06, max_iter=50000):

    z = np.log(np.maximum(a, 1e-30))

    T = T_start
    no_imp = 0
    base_lr = lr

    T_ratio = (T_end / T_start) ** (1.0 / total)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, m = _lse_step(a, z, T, lr, state)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-30))
            z -= z.mean()
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1500 == 0:

                z -= z.mean()
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

def _subgrad_polish(a, deadline, lr_init=0.006, max_iter=40000):
    a = _normalize(a)
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    mu = 0.85
    lr = lr_init
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_flip = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_flip, N)[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        g_tan = grad - dot * a
        v = mu * v - lr * g_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        sc = float(2.0 * n * c_new.max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.0005, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                mu = 0.85
                no_imp = 0

    return best_a


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_flat_edges(n, w=0.4):
    t = np.linspace(0, 1, n)
    a = 1.0 + w * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_flat_edges2(n, w=0.2, freq=2):
    t = np.linspace(0, 1, n)
    a = 1.0 + w * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interpolate(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == new_n:
        return _normalize(a)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_n)
    p = np.interp(x1, x0, a)
    return _normalize(p)

        sc_eval = evaluate_sequence(final.tolist())
        print(f"\nevaluate_sequence: {sc_eval:.10f}")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final.size}")
    return [float(x) for x in final]
