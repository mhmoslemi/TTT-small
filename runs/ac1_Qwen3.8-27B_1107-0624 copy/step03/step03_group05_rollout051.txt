def _refine_with_perturbation(a_best, deadline, n_pert=5, scale=0.1):
    """Try small perturbations of the best solution."""
    a_best = np.asarray(a_best, dtype=np.float64).ravel()
    n = a_best.size
    best_score = _score(a_best)
    best_a = a_best.copy()
    rng = np.random.default_rng(999)

    for i in range(n_pert):
        if time.monotonic() > deadline - 30:
            break
        # Structured perturbation
        t = np.linspace(0, 1, n)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = scale / (i + 1)
        pert = a_best * (1.0 + amp * np.sin(freq * t * np.pi + phase))
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s < 1e-10:
            continue
        pert = pert / s
        sc = _score(pert)
        if sc < best_score:
            best_score = sc
            best_a = pert.copy()
        # Refine
        a_res, sc_res = _continuation_optimize(pert, deadline, beta_start=0.01 * _fft_convolve(pert).max(), n_sub=4)
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"    Perturb {i}: new best {sc_res:.10f}")

    return best_a, best_score


def _multi_scale_search(n_target, deadline, rng):
    """Coarse-to-fine: optimize short sequence, upsample, refine."""
    # Start with very short sequences
    coarse_lengths = [16, 32, 64, 128]
    best_a = None
    best_score = np.inf

    for n_c in coarse_lengths:
        if time.monotonic() > deadline - 60:
            break
        # Try several initializations at this length
        inits = [
            _make_poly_prof(n_c, (1, 0), (-1, 1)),
            _make_poly_prof(n_c, (1, 0), (-2, 1), (1, 2)),
            _make_shape(n_c, lambda t: t * (1 - t) + 0.3),
            _make_shape(n_c, lambda t: np.sin(np.pi * t) ** 0.5 + 0.1),
            _make_poly_prof(n_c, (1, 0), (-1.5, 1), (0.8, 2)),
        ]
        # Also try random
        a_rand = np.maximum(rng.normal(1, 0.3, n_c), 0.01)
        inits.append(a_rand / a_rand.sum())

        for a0 in inits:
            if time.monotonic() > deadline - 60:
                break
            a_res, sc = _continuation_optimize(a0, deadline, beta_start=None, n_sub=5)
            if sc < best_score:
                best_score = sc
                best_a = a_res.copy()
                print(f"    Coarse n={n_c}: score={sc:.10f}")

    if best_a is None:
        best_a = np.ones(n_target) / n_target
        best_score = _score(best_a)

    # Upsample progressively
    for n_mid in [256, 384, 512]:
        if n_mid >= n_target or time.monotonic() > deadline - 60:
            continue
        a_mid = _interpolate_to(best_a, n_mid)
        a_mid, sc = _continuation_optimize(a_mid, deadline, beta_start=0.01 * _fft_convolve(a_mid).max(), n_sub=4)
        if sc < best_score:
            best_score = sc
            best_a = a_mid.copy()
            print(f"    Mid n={n_mid}: score={sc:.10f}")

    # Final upsample to target
    a_final = _interpolate_to(best_a, n_target)
    a_final, sc = _continuation_optimize(a_final, deadline, beta_start=0.01 * _fft_convolve(a_final).max(), n_sub=5)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
        print(f"    Fine n={n_target}: score={sc:.10f}")

    return best_a, best_score


def _spectral_flat_init(n, rng):
    """Initialize by trying to flatten the power spectrum."""
    # Start with random phase, compute amplitude
    z = rng.normal(size=n)
    a = _softmax(z)
    # In frequency domain, try to make |FFT(a)|^2 flat
    # This is a heuristic initialization
    t = np.linspace(0, 1, n)
    # Try various shapes
    candidates = [
        _make_poly_prof(n, (1, 0), (-1, 1)),
        _make_poly_prof(n, (1, 0), (-1, 1), (0.5, 2)),
        _make_shape(n, lambda tt: tt**0.5 * (1-tt)**0.3 + 0.2),
        _make_shape(n, lambda tt: (1-tt)**0.7 * tt**0.2 + 0.1),
        _make_poly_prof(n, (1, 0), (-0.8, 1), (-0.3, 2)),
        _make_shape(n, lambda tt: 1.0 + 0.5*np.cos(2*np.pi*tt) + 0.3*np.cos(4*np.pi*tt)),
    ]
    best_c = None
    best_sc = np.inf
    for c in candidates:
        sc = _score(c)
        if sc < best_sc:
            best_sc = sc
            best_c = c
    return best_c, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining L-BFGS-B continuation,
    multi-scale search, spectral initialization, and perturbation refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_a = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    n_parent = parent.size if parent is not None else 768

    # === Phase 1: Refine parent with L-BFGS-B continuation ===
    if parent is not None and time.monotonic() < deadline - 100:
        print("\n=== Phase 1: Refine parent ===")
        phase1_deadline = min(deadline - 200, t_start + budget_s * 0.25)
        a_ref, sc_ref = _continuation_optimize(parent, phase1_deadline, n_sub=10)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()
            print(f"  Phase 1 result: {sc_ref:.10f}")

    # === Phase 2: Multi-scale search at parent length ===
    if time.monotonic() < deadline - 150:
        print(f"\n=== Phase 2: Multi-scale at n={n_parent} ===")
        phase2_deadline = min(deadline - 150, t_start + budget_s * 0.45)
        a_ms, sc_ms = _multi_scale_search(n_parent, phase2_deadline, rng)
        if sc_ms < best_score:
            best_score = sc_ms
            best_a = a_ms.copy()
            print(f"  Multi-scale result: {sc_ms:.10f}")

    # === Phase 3: Spectral-flat initialization and optimize ===
    if time.monotonic() < deadline - 120:
        print(f"\n=== Phase 3: Spectral init at n={n_parent} ===")
        phase3_deadline = min(deadline - 120, t_start + budget_s * 0.60)
        a_spec, sc_spec = _spectral_flat_init(n_parent, rng)
        a_spec2, sc_spec2 = _continuation_optimize(a_spec, phase3_deadline, n_sub=8)
        if sc_spec2 < best_score:
            best_score = sc_spec2
            best_a = a_spec2.copy()
            print(f"  Spectral result: {sc_spec2:.10f}")

    # === Phase 4: Try alternative lengths ===
    alt_lengths = [n_parent - 16, n_parent + 16, n_parent - 32, n_parent + 32]
    alt_lengths = [l for l in alt_lengths if 128 <= l <= 2048]
    # Deduplicate
    seen = set()
    alt_lengths_unique = []
    for l in alt_lengths:
        if l not in seen and l != n_parent:
            seen.add(l)
            alt_lengths_unique.append(l)

    for n_alt in alt_lengths_unique:
        if time.monotonic() > deadline - 100:
            break
        print(f"\n=== Alt length n={n_alt} ===")
        a_init = _interpolate_to(best_a, n_alt) if best_a is not None else np.ones(n_alt) / n_alt
        alt_deadline = min(deadline - 80, t_start + budget_s * 0.80)
        a_alt, sc_alt = _continuation_optimize(a_init, alt_deadline, n_sub=6)
        if sc_alt < best_score:
            best_score = sc_alt
            best_a = a_alt.copy()
            print(f"  Alt n={n_alt}: {sc_alt:.10f}")

    # === Phase 5: Perturbation refinement ===
    if best_a is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 5: Perturbation refinement ===")
        pert_deadline = min(deadline - 40, t_start + budget_s * 0.90)
        a_pert, sc_pert = _refine_with_perturbation(best_a, pert_deadline, n_pert=4, scale=0.08)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  Perturb result: {sc_pert:.10f}")

    # === Phase 6: Final intensive polish ===
    if best_a is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 6: Final polish ===")
        final_deadline = deadline - 10
        # Very tight continuation
        a_final, sc_final = _continuation_optimize(best_a, final_deadline,
                                                   beta_start=0.005 * _fft_convolve(best_a).max(),
                                                   beta_min=1e-8, n_sub=6)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"  Final polish: {sc_final:.10f}")

    # Ensure valid output
    if best_a is None:
        best_a = _make_poly_prof(n_parent, (1, 0), (-1, 1))
        best_score = _score(best_a)

    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if s < 0.01:
        best_a = _make_poly_prof(768, (1, 0), (-1, 1))
        best_score = _score(best_a)
    else:
        best_a = best_a / s  # normalize to sum=1

    # Final evaluation
    final_list = [float(x) for x in best_a]
    try:
        sc_eval = float(evaluate_sequence(final_list))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={len(final_list)}")
    return final_list
