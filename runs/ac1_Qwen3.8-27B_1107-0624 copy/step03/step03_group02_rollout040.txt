    v = np.asarray(v, dtype=np.float64).copy()

    v = np.zeros(n)

        z = np.clip(z, -50, 50)

def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

The problem is: minimize `max_k b[k]` subject to `sum(a) = 1, a >= 0`.

The Lagrangian dual: for each k, introduce a multiplier `lambda_k >= 0` for the constraint `b[k] <= t`. The dual function is:

One more idea: **use the "mirror descent" framework with a KL divergence penalty.**

The mirror descent update is:
`a_new = argmin_{a >= 0, sum(a) = 1} [sum_j a[j] * grad[j] + (1/eta) * KL(a || a_old)]`

where `KL(a || a_old) = sum_j a[j] * log(a[j] / a_old[j])`.

The solution is:
`a_new[j] = a_old[j] * exp(-eta * grad[j]) / Z`

where `Z = sum_j a_old[j] * exp(-eta * grad[j])`.

This is the "exponentiated gradient" or "Hedge" algorithm. It's different from Adam in that it uses multiplicative updates rather than additive ones. This might be better for the simplex constraint.

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive Adam in logit-space, LP spreading, targeted simulated annealing,
mirror descent, and multi-scale search at n=512 and n=1024.

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def _score_from_conv(a, c):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _subgrad_at(a, k):
    n = len(a)
    sg = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        sg[idx] = 2.0 * a[k - idx]
    return sg


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None
