Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines sparse LP refinement, L-BFGS smooth-max on log-simplex,
and a novel coordinate descent that directly optimizes individual
elements to minimize the peak autocorrelation.
Key novelty: coordinate descent with analytical 1D minimax per coordinate,
plus adaptive multi-phase strategy with sparse LP for efficiency.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 192:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _build_sparse_conv_matrix(a):
    """Build sparse matrix A where row k of A corresponds to (a*b)[k] = A[k,:] @ b."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_spread(a, M):
    """LP: maximize sum(b) subject to (a*b)[k] <= M, b >= 0."""
    n = len(a)
    A = _build_sparse_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate_init=1.002, blend_init=0.35):
    """Iterative LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend_init
    inflate_cur = inflate_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate_cur = min(1.015, inflate_cur * 1.04)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.97)
            blend_cur = min(0.5, blend_cur * 1.02)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = max(1.0003, inflate_cur * 0.95)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend_cur = blend_init
                inflate_cur = inflate_init
                no_imp = 0

        a = a_new

    return best_a


def _smooth_max_loss_and_grad(a, beta):
    """Compute smooth-max loss and its gradient w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    # Gradient: d/d a_j [sum_k w_k * (a*a)[k]]
    # (a*a)[k] = sum_i a_i * a_{k-i}
    # d/d a_j = sum_k w_k * (a_{k-j} + a_{k-j}) for k != 2j, and w_{2j} * 2*a_j for k=2j
    # More precisely: d/d a_j = 2 * sum_k w_k * a_{k-j} (with appropriate bounds)
    # Using FFT: convolve w with reversed a
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    loss = mval + beta * np.log(s)
    return loss, grad, c


def _logit_lbfgs_polish(a, deadline, max_iter=500, lr_scale=1.0):
    """L-BFGS-B on log-simplex parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))

    def objective(z_vec):
        e = np.exp(np.clip(z_vec, -50, 50))
        a_tmp = e / e.sum()
        c = _fft_convolve(a_tmp)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s = float(ex.sum())
        if s <= 0:
            return mval, None
        loss = mval + beta * np.log(s)
        # Gradient
        w = ex / s
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_tmp[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(a_tmp, vp))
        gz = a_tmp * (vp - dot)
        gz -= float(np.mean(gz))
        return loss, gz

    for round_idx in range(8):
        if time.monotonic() > deadline:
            break
        # Anneal beta
        result = minimize(objective, z, method='L-BFGS-B',
                          options={'maxiter': max_iter // 8, 'gtol': 1e-8})
        if result.success or result.fun < np.inf:
            z = result.x
            e = np.exp(np.clip(z, -50, 50))
            a = e / e.sum()
            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())
            if score < best_score:
                best_score = score
                best_a = a.copy()
        beta = max(1e-10, beta * 0.6)

    return best_a


def _coordinate_descent(a, deadline, max_sweeps=20):
    """Coordinate descent: for each coordinate, find optimal value minimizing max autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        improved_this_sweep = False

        for j in range(n):
            if time.monotonic() > deadline:
                break

            a_j_old = a[j]
            if a_j_old < 1e-12:
                continue

            # Determine which lags are affected
            k_lo = j
            k_hi = j + n - 1

            # Precompute coefficients: for k != 2j, coefficient is 2*a_{k-j}
            # For k = 2j, it's quadratic: old_c[2j] + (x^2 - a_j_old^2)
            k_2j = 2 * j
            in_range = k_2j < 2 * n - 1

            # Current max
            current_max = float(c.max())

            # Try a few candidate values
            candidates = [a_j_old]  # current
            # Try reducing
            if a_j_old > 0:
                candidates.append(0.5 * a_j_old)
                candidates.append(0.25 * a_j_old)
                candidates.append(0.0)
            # Try increasing slightly
            candidates.append(1.5 * a_j_old)
            candidates.append(2.0 * a_j_old)

            best_delta = 0.0
            best_local_max = current_max

            for x_try in candidates:
                if x_try < 0:
                    continue
                delta = x_try - a_j_old
                if delta == 0:
                    continue

                # Compute new max
                new_max = current_max  # start with values not affected
                # Check affected lags
                for k in range(k_lo, min(k_hi + 1, 2 * n - 1)):
                    if k == k_2j:
                        new_val = c[k] + (x_try * x_try - a_j_old * a_j_old)
                    else:
                        coeff = k - j
                        if 0 <= coeff < n:
                            new_val = c[k] + 2.0 * delta * a[coeff]
                        else:
                            new_val = c[k]
                    if new_val > new_max:
                        new_max = new_val

                if new_max < best_local_max - 1e-14:
                    best_local_max = new_max
                    best_delta = delta

            if best_delta != 0.0:
                a[j] = max(0.0, a_j_old + best_delta)
                improved_this_sweep = True

        # Normalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if not improved_this_sweep:
            break
        if no_imp > 5:
            break

    return best_a


def _coordinate_descent_fast(a, deadline, max_sweeps=15):
    """Faster coordinate descent using vectorized evaluation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved_this_sweep = False

        for j in range(n):
            if time.monotonic() > deadline:
                break

            a_j_old = a[j]
            if a_j_old < 1e-10:
                continue

            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            k_2j = 2 * j
            in_range_2j = k_2j < 2 * n - 1

            # Get affected coefficients
            ks = np.arange(k_lo, k_hi + 1)
            coeffs = ks - j  # these are valid indices since k_lo=j gives coeff 0

            # Current values at affected positions
            c_aff = c[k_lo:k_hi + 1].copy()

            # For non-quadratic terms: new_val = c_aff[i] + 2*delta*a_coeffs[i]
            # For quadratic (if k_2j in range): new_val = c[k_2j] + delta^2 + 2*a_j_old*delta

            a_coeffs = a[coeffs]

            # Try fractional changes
            fracs = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.8])
            x_tries = fracs * a_j_old
            x_tries = x_tries[x_tries >= 0]

            best_x = a_j_old
            best_max = float(c.max())

            for x_try in x_tries:
                delta = x_try - a_j_old
                if abs(delta) < 1e-15:
                    continue

                # Linear part
                new_vals = c_aff + 2.0 * delta * a_coeffs

                # Fix quadratic term at k_2j
                if in_range_2j:
                    idx_2j = k_2j - k_lo
                    new_vals[idx_2j] = c[k_2j] + (x_try * x_try - a_j_old * a_j_old)

                local_max = float(new_vals.max())
                # Also need to consider unaffected values
                if local_max < best_max:
                    best_max = local_max
                    best_x = x_try

            if best_x != a_j_old and best_max < float(c.max()) - 1e-13:
                a[j] = best_x
                improved_this_sweep = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 4 or not improved_this_sweep:
            break

    return best_a


def _gradient_polish(a, deadline, max_iter=10000):
    """Momentum-based gradient descent on simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.04 * float(c.max()))
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, _, _ = _smooth_max_loss_and_grad(a, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.04 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 50 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a


def _adam_logit_polish(a, deadline, max_iter=15000):
    """Adam on log-simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.04 * float(c.max()))
    no_imp = 0
    lr = 0.04
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_exp = float(ex.sum())
        if s_exp <= 0:
            continue
        w = ex / s_exp

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(1500, 3 * n):
            rng_r = np.random.default_rng(it * 7919 + 13)
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n) * 0.3
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.04 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _block_swap_optimize(a, deadline, max_iter=3000):
    """Random block swaps."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        block_size = int(rng.integers(max(2, n // 64), max(3, n // 16)))
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        tmp = a_new[i:i + block_size].copy()
        a_new[i:i + block_size] = a_new[j:j + block_size]
        a_new[j:j + block_size] = tmp
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 1500:
                break

    return best_a


def _full_pipeline(a_init, n, deadline, phase_budget):
    """Run all optimization phases in sequence."""
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    remaining = deadline - time.monotonic()

    # Phase 1: LP refinement
    lp_budget = min(phase_budget * 0.35, remaining * 0.35)
    if lp_budget > 10:
        a_lp = _lp_refine(a, time.monotonic() + lp_budget)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"    LP: {sc:.10f}")

    # Phase 2: Gradient polish
    gp_budget = min(phase_budget * 0.2, remaining * 0.2)
    if gp_budget > 5:
        a_gp = _gradient_polish(best_a, time.monotonic() + gp_budget)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"    Grad: {sc:.10f}")

    # Phase 3: Coordinate descent (NEW)
    cd_budget = min(phase_budget * 0.15, remaining * 0.15)
    if cd_budget > 5:
        a_cd = _coordinate_descent_fast(best_a, time.monotonic() + cd_budget)
        sc = _score(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
        print(f"    CoordDesc: {sc:.10f}")
        # Second pass of coordinate descent
        if time.monotonic() < deadline - 30:
            a_cd2 = _coordinate_descent_fast(best_a, time.monotonic() + min(cd_budget, 30))
            sc = _score(a_cd2)
            if sc < best_score:
                best_score = sc
                best_a = a_cd2.copy()
            print(f"    CoordDesc2: {sc:.10f}")

    # Phase 4: Adam logit
    ad_budget = min(phase_budget * 0.2, remaining * 0.2)
    if ad_budget > 5:
        a_ad = _adam_logit_polish(best_a, time.monotonic() + ad_budget)
        sc = _score(a_ad)
        if sc < best_score:
            best_score = sc
            best_a = a_ad.copy()
        print(f"    Adam: {sc:.10f}")

    # Phase 5: Block swap
    bs_budget = min(phase_budget * 0.05, remaining * 0.05)
    if bs_budget > 3:
        a_bs = _block_swap_optimize(best_a, time.monotonic() + bs_budget, max_iter=2000)
        sc = _score(a_bs)
        if sc < best_score:
            best_score = sc
            best_a = a_bs.copy()
        print(f"    BlockSwap: {sc:.10f}")

    # Phase 6: Final LP
    lp2_budget = min(phase_budget * 0.1, remaining * 0.1)
    if lp2_budget > 5:
        a_lp2 = _lp_refine(best_a, time.monotonic() + lp2_budget,
                           inflate_init=1.001, blend_init=0.2)
        sc = _score(a_lp2)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2.copy()
        print(f"    LP2: {sc:.10f}")

    return best_a, best_score
