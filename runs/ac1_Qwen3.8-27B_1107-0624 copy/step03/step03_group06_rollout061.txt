def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 200:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):

def _normalize(a):

def _solve_lp_conv(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    # Build constraint matrix: A has shape (2n-1, n)
    # A[j, i] = a[i] if j-i is in [0, n-1], else 0

    # Using FFT trick: conv(a, g)[j] = sum_i a[i]*g[j-i]
    # This is the same as (a_rev * g)[n-1+j] in circular conv, but for linear:
    # A is a banded Toeplitz-like matrix
    # For efficiency, build it implicitly via FFT in the LP? No, linprog needs explicit matrix.
    # For n <= 2048, explicit matrix is fine (2n-1 rows, n cols)
    
    # Build A using the structure: row j has a[0:min(j+1,n)] in first positions
    # and a[max(0,j-n+1):n] shifted
    A = np.zeros((2 * n - 1, n))

        hi = min(j + 1, n)
        A[j, lo:hi] = a[hi - 1 - (j - lo) if j >= n else np.arange(hi - lo)][::-1] if j >= n else a[:hi - lo]
    # Actually let me just do it clearly:

    # So A[j, i] = a[i] if max(0,j-n+1) <= i <= min(j,n-1) and j-i is in [0,n-1]
    # Since i is in [0,n-1] and j-i is in [0,n-1], we need i in [max(0,j-n+1), min(j, n-1)]

    # So A[j, k] = a[j-k] if j-k is in [0,n-1], i.e., k in [j-n+1, j] and k in [0,n-1]
    
    A = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):

        k_hi = min(j, n - 1)

def _solve_lp_fast(a, M, n):
    """Faster LP using the special structure - only non-zero entries."""
    # A[j, k] = a[j-k] for max(0,j-n+1) <= k <= min(j, n-1)
    # Total non-zeros: sum_{j=0}^{2n-2} (min(j,n-1) - max(0,j-n+1) + 1)
    # = n + (n-1) + ... + 1 + 1 + ... + (n-1) + n = n^2
    # So the matrix is n^2 non-zeros in a (2n-1) x n matrix
    
    # Use sparse matrix for speed
    from scipy.sparse import csc_matrix
    
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        for k in range(k_lo, k_hi + 1):
            rows.append(j)
            cols.append(k)
            vals.append(a[j - k])
    
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

    #                                      0 otherwise
    
    # Efficient computation via FFT:
    # grad[k] = sum_j w[j] * 2*a[j-k] (for valid j-k)
    # = 2 * sum_j w[j] * a[j-k]
    # This is 2 * (w * a_rev)[... ] in convolution
    # Let a_rev[i] = a[n-1-i]
    # sum_j w[j]*a[j-k] = (w convolve a)[k + n - 1] ... 
    
    # Actually: sum_j w[j] * a[j-k] for j-k in [0,n-1]
    # = sum_m w[m+k] * a[m] for m in [0, n-1] and m+k in [0, 2n-2]
    # This is the cross-correlation of w and a at lag k
    # = (IFFT(FFT(w) * conj(FFT(a))))[k + offset]
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa = np.fft.rfft(a, N_pad)
    # Cross correlation: ifft(fw * conj(fa))
    corr = np.fft.irfft(fw * np.conj(fa), N_pad)
    # corr[k] = sum_j w[j+k]*a[j] ... not quite what we want
    # We want sum_j w[j]*a[j-k] = sum_m w[m+k]*a[m] = corr[k] where corr is (w * a_rev)
    
    # Let me just use: sum_j w[j]*a[j-k] = (w convolve a_rev)[n-1+k]
    # where a_rev[i] = a[n-1-i]
    # conv(w, a_rev)[n-1+k] = sum_i w[i]*a_rev[n-1+k-i] = sum_i w[i]*a[i-(n-1+k-i)]... 
    
    # Simpler: just compute directly
    # grad[k] = 2 * sum_{j: 0<=j-k<2n-1} w[j] * a[j-k] (with a indexed 0 to n-1)
    # For k in [0, n-1]: sum_{j=k}^{k+n-1} w[j]*a[j-k] (j-k ranges 0 to n-1)
    # This is the correlation of w (length 2n-1) and a (length n) at lag k
    
    # Using FFT: pad a to length N_pad, compute circular cross-correlation
    # corr[k] = ifft(FFT(w) * conj(FFT(a)))[k] = sum_j w[(j+k) mod N]*a[j]
    # We want sum_j w[j]*a[j-k] = sum_j w[(j+k)]*a[j] (by substitution)
    # So it's corr[k] from the circular cross-correlation (with proper indexing)
    
    grad = np.zeros(n)
    # corr_full[k] = sum_j w[j+k]*a[j] for j in valid range
    # Using the FFT result:
    # ifft(FFT(w) * conj(FFT(a)))[k] = sum_j w[(j+k) mod N] * a[j]
    # For k >= 0 and j+k < 2n-1 (no wraparound): this is sum_j w[j+k]*a[j]
    # which is what we want: sum_m w[m]*a[m-k] (letting m=j+k)
    
    # So grad[k] ≈ 2 * corr[k] (ignoring the j=2k special case)
    grad = 2.0 * corr[:n]
    
    # Fix the diagonal: for j=2k, the derivative is a[k] not 2*a[k]
    # The error term is: for each k, subtract w[2k]*a[k] (if 2k < 2n-1)
    for k in range(n):
        j_diag = 2 * k
        if j_diag < 2 * n - 1:
            grad[k] -= w[j_diag] * a[k]
    
    return grad, mval


def _project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient polish using smooth max approximation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * best_M)
    
    beta = max(1e-8, 0.03 * best_M)
    lr = 0.05
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_mom = a + momentum * v
        grad, mval = _smooth_max_grad(a_mom, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_nonneg_sum1(a + v)
        a = a_new
        
        c = _fft_convolve(a)
        M_new = float(c.max())
        score = float(2.0 * n * M_new)
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            best_score = score
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.03 * best_M)
                lr = base_lr
                no_imp = 0
        
        if it % 200 == 199:
            beta = max(1e-10, beta * 0.85)
    
    return best_a, best_score


def _multi_stage_optimize(a0, deadline):
    """Run multi-stage optimization: LP with decreasing inflation + gradient."""
    n = len(a0)
    a = a0.copy()
    
    # Stage 1: LP with moderate inflation
    stage1_deadline = min(deadline - 60, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    a, sc1 = _lp_refine(a, stage1_deadline, max_iter=20000)
    print(f"      Stage 1 (LP): score={sc1:.10f}")
    
    # Stage 2: Gradient polish
    if time.monotonic() < deadline - 30:
        gp_deadline = min(deadline - 10, time.mon
otonic() + (deadline - time.monotonic()) * 0.4)
        a, sc2 = _gradient_polish(a, gp_deadline, max_iter=20000)
        print(f"      Stage 2 (Grad): score={sc2:.10f}")
    
    # Stage 3: Final tight LP
    if time.monotonic() < deadline - 15:
        a, sc3 = _lp_refine(a, deadline - 5, max_iter=10000)
        print(f"      Stage 3 (Tight LP): score={sc3:.10f}")
    
    # Stage 4: Final micro-gradient
    if time.monotonic() < deadline - 5:
        a, sc4 = _gradient_polish(a, deadline - 1, max_iter=5000)
        print(f"      Stage 4 (Micro Grad): score={sc4:.10f}")
    
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization combining LP refinement with tight inflation,
    gradient polish with smooth max, and adaptive length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Define lengths to try
    lengths = [512, 768, 1024, 1536]
    
    # Create candidate initializations
    candidates = []  # (score, name, sequence)
    
    for n_t in lengths:
        # Parent interpolated
        if parent is not None:
            a_p = _interpolate(parent, n_t)
            sc = _score(a_p)
            candidates.append((sc, f'parent_n{n_t}', a_p))
        
        # Best profiles
        profiles = _make_profiles(n_t)
        for name, a_p in profiles.items():
            sc = _score(a_p)
            candidates.append((sc, f'{name}_n{n_t}', a_p))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    
    print(f"Generated {len(candidates)} candidates, best initial: {candidates[0][0]:.8f} ({candidates[0][1]})")
    print(f"Top 10 candidates:")
    for i, (sc, name, a) in enumerate(candidates[:10]):
        print(f"  {i}: {name}: {sc:.8f}")
    
    # Select top candidates to optimize (limited by time)
    # Allocate time based on number of candidates
    n_to_try = 0
    time_per_candidate = 200  # seconds per candidate
    
    for sc, name, a0 in candidates:
        if time.monotonic() > deadline - 100:
            break
        if sc >= best_score + 0.05:
            continue
        
        n_to_try += 1
        remaining = deadline - time.monotonic() - 60
        if remaining < time_per_candidate * 0.5:
            break
        
        # Allocate time for this candidate
        alloc = min(time_per_candidate, remaining * 0.3)
        if remaining < 300:
            alloc = min(alloc, remaining * 0.4)
        
        cand_deadline = time.monotonic() + alloc
        n_t = len(a0)
        print(f"\nOptimizing {name} (n={n_t}, init_score={sc:.8f}, alloc={alloc:.0f}s)")
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        
        try:
            a_result = _multi_stage_optimize(a0, cand_deadline)
            sc_result = _score(a_result)
            if np.isfinite(sc_result) and sc_result < best_score:
                best_score = sc_result
                best_seq = a_result.copy()
                print(f"  ** NEW BEST: {sc_result:.10f}")
            else:
                print(f"  Score: {sc_result:.10f} (no improvement)")
        except Exception as e:
            print(f"  Error: {e}")
    
    # Final refinement on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        final_deadline = deadline
        print(f"\nFinal refinement on best (score={best_score:.10f}, n={len(best_seq)})")
        
        # One more LP pass with very tight inflation
        a_final, sc_final = _lp_refine(best_seq, final_deadline - 20, max_iter=30000)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  Tight LP: {sc_final:.10f}")
        
        # Final gradient
        if time.monotonic() < final_deadline - 10:
            a_gp, sc_gp = _gradient_polish(best_seq, final_deadline - 2, max_iter=20000)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  Final Grad: {sc_gp:.10f}")
    
    # Validate and return
    if best_seq is None:
        # Fallback
        best_seq = _make_profiles(512)['ramp1.0']
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profiles(512)['ramp1.0']
    
    # Validate with evaluate_sequence if possible
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")
    
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
