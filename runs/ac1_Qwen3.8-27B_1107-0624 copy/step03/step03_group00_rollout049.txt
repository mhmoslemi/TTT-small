"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) diverse multi-scale initialization, (2) Adam with adaptive smooth-max
in logit space, (3) reduced-constraint LP refinement exploiting active-set
structure, (4) scale escalation, (5) multi-restart with diverse perturbations.
Key improvements over baseline: larger n (up to 8192), reduced-LP for speed,
active-set-aware gradient focusing, and better time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        return mval
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        return mval
    return mval + beta * float(np.log(s))


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _reduced_lp_solve(a, M, top_k=200):
    n = len(a)
    conv_aa = _full_convolve(a)
    # Select top_k lags closest to M (most binding)
    idx = np.argsort(conv_aa)[::-1][:top_k]
    rows = []
    cols = []
    vals = []
    b_ub_list = []
    for k in idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            row_id = len(b_ub_list)
            rows.extend([row_id] * len(j_idx))
            cols.extend(j_idx.tolist())
            vals.extend(a[k - j_idx].tolist())
            b_ub_list.append(M)
    if not rows:
        return None
    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(b_ub_list), n))
    c_obj = -np.ones(n)
    b_ub_arr = np.array(b_ub_list)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                      options={'time_limit': 60, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _full_lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(j_idx))
            cols.extend(j_idx.tolist())
            vals.extend(a[k - j_idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                      options={'time_limit': 120, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_init=0.4, max_solves=15, use_reduced=True, top_k=200):
    n = len(a)
    current_best = a.copy()
    c = _full_convolve(a)
    M = float(c.max())
    current_score = float(2.0 * n * M)
    n_solves = 0
    t = t_init

    while time.monotonic() < deadline and n_solves < max_solves:
        if use_reduced or n > 1500:
            b = _reduced_lp_solve(a, M, top_k=top_k)
        else:
            b = _full_lp_solve(a, M)
        if b is None:
            n_solves += 1
            if n_solves > max_solves:
                break
            continue

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _full_convolve(a_new)
        M_new = float(c_new.max())
        sc_new = float(2.0 * n * M_new)

        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()

        a = a_new
        c = c_new
        M = M_new
        n_solves += 1
        t = max(0.08, t * 0.88)

    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=350, focus_top_k=20):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)

        # Focus: zero out weights for lags far from max
        if focus_top_k > 0 and c.size > focus_top_k:
            threshold = np.sort(c)[::-1][min(focus_top_k, c.size) - 1]
            w[c < threshold * 0.999] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum

        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.15:
                    beta = max(1e-5, beta * 0.4)

        if no_imp > no_imp_limit:
            ns = 0.06 if best_score < 1.51 else 0.12
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _adam_logit_v2(p0, deadline, rng, lr=0.04, beta_frac=0.06, beta_decay=0.94,

def _make_profile_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * (2 * t - 1) ** alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_gaussian(n, w=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / w) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam=1.0):
    t = np.arange(n) / n
    a = np.exp(-lam * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq=1, amp=0.7):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1=0.3, c2=0.7, s=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_sine(n, periods=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.sin(2.0 * np.pi * periods * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    return _normalize_nonneg(p_new)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Generate diverse candidates ---
    candidates = []

    target_ns = [1024, 2048, 4096]
    if parent is not None and pn > 512:
        target_ns.append(pn)

    profiles_fn = [
        ('raised_cos1.0', lambda n: _make_profile_raised_cosine(n, 1.0)),
        ('raised_cos1.5', lambda n: _make_profile_raised_cosine(n, 1.5)),
        ('raised_cos2.0', lambda n: _make_profile_raised_cosine(n, 2.0)),
        ('gauss0.35', lambda n: _make_profile_gaussian(n, 0.35)),
        ('gauss0.45', lambda n: _make_profile_gaussian(n, 0.45)),
        ('lin1.5', lambda n: _make_profile_linear(n, 1.5)),
        ('lin2.5', lambda n: _make_profile_linear(n, 2.5)),
        ('exp0.8', lambda n: _make_profile_exponential(n, 0.8)),
        ('exp1.5', lambda n: _make_profile_exponential(n, 1.5)),
        ('cos1_0.7', lambda n: _make_profile_cosine(n, 1, 0.7)),
        ('cos2_0.5', lambda n: _make_profile_cosine(n, 2, 0.5)),
        ('twobump', lambda n: _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
        ('twobump2', lambda n: _make_profile_two_bump(n, 0.25, 0.75, 0.2)),
        ('sine1', lambda n: _make_profile_sine(n, 1)),
    ]

    for n in target_ns:
        for label, fn in profiles_fn:
            p = fn(n)
            if p is not None:
                sc = _score_norm(p)
                candidates.append((sc, p, n, label))

    # Interpolated parent
    if parent is not None:
        for n in [1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
        if 0 < pn <= 8192:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    # Noisy parent variants
    if parent is not None:
        for n in [1024, 2048]:
            for noise_scale in [0.05, 0.1, 0.2]:
                p_base = _interp_profile(parent, n) if pn != n else parent.copy()
                z = np.log(np.maximum(p_base, 1e-12))
                z -= z.mean()
                z += noise_scale * rng.normal(size=n)
                z = np.clip(z, -60, 60)
                z -= z.mean()
                p = _softmax_from_z(z)
                sc = _score_norm(p)
                candidates.append((sc, p, n, f'parent_noise{noise_scale}'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f} n={candidates[0][2]} [{candidates[0][3]}]")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Adam optimization on top candidates ---
    phase2_end = t_start + 500.0
    polished = []
    count = 0
    max_polish = 8

    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= max_polish:
            break
        if sc > best_score * 1.02:
            break
        per_end = min(phase2_end, time.monotonic() + 40.0)
        if per_end - time.monotonic() < 5:
            break
        p_b, sc_b = _adam_logit(p, per_end, rng, lr=0.05, beta_frac=0.08,
                                beta_decay=0.92, no_imp_limit=250, focus_top_k=30)
        _update(p_b, sc_b, f'adam1_{label}_n{n}')
        polished.append((sc_b, p_b, n, label))
        count += 1

    # Second round of Adam with different parameters on top polished
    polished.sort(key=lambda x: x[0])
    phase2b_end = t_start + 700.0
    for sc, p, n, label in polished[:4]:
        if time.monotonic() > phase2b_end:
            break
        per_end = min(phase2b_end, time.monotonic() + 45.0)
        if per_end - time.monotonic() < 5:
            break
        p_b, sc_b = _adam_logit_v2(p, per_end, rng, lr=0.03, beta_frac=0.05,
                                    beta_decay=0.94, no_imp_limit=400, focus_top_k=50,
                                    huber_delta=0.3)
        _update(p_b, sc_b, f'adam2_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_v2'))

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: LP refinement on best ---
    phase3_end = t_start + 1100.0
    if best_seq is not None:
        n_lp = best_seq.size
        use_reduced = n_lp > 1500
        top_k_val = 150 if n_lp > 2000 else 250

        print(f"  LP refine n={n_lp}, score={best_score:.10f}, reduced={use_reduced}, top_k={top_k_val}")
        per_lp = min(phase3_end - time.monotonic(), 180.0)
        if per_lp > 10:
            p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + per_lp, t_init=0.35,
                                      max_solves=20, use_reduced=use_reduced, top_k=top_k_val)
            _update(p_lp, sc_lp, 'LP_refine')

        # Adam polish after LP
        if best_seq is not None and time.monotonic() < phase3_end - 20:
            per = min(60.0, phase3_end - time.monotonic() - 10)
            if per > 10:
                p_ap, sc_ap = _adam_logit(best_seq, time.monotonic() + per, rng,
                                           lr=0.03, beta_frac=0.05, beta_decay=0.94,
                                           no_imp_limit=300, focus_top_k=30)
                _update(p_ap, sc_ap, 'post_LP_polish')

        # Second LP round
        if best_seq is not None and time.monotonic() < phase3_end - 15:
            n_lp2 = best_seq.size
            use_reduced2 = n_lp2 > 1500
            per2 = min(120.0, phase3_end - time.monotonic() - 5)
            if per2 > 10:
                p_lp2, sc_lp2 = _lp_refine(best_seq, time.monotonic() + per2, t_init=0.25,
                                            max_solves=15, use_reduced=use_reduced2, top_k=top_k_val)
                _update(p_lp2, sc_lp2, 'LP_refine2')

    # --- Phase 4: Scale escalation ---
    phase4_end = t_start + 1450.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns_esc = []
        if n_cur <= 1024:
            target_ns_esc = [2048, 4096]
        elif n_cur <= 2048:
            target_ns_esc = [4096]
        elif n_cur <= 4096:
            target_ns_esc = [8192]

        for tn in target_ns_esc:
            if time.monotonic() > phase4_end - 40:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            # Adam at new scale
            per = min(80.0, phase4_end - time.monotonic() - 20)
            if per > 15:
                p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                         lr=0.04, beta_frac=0.07, beta_decay=0.93,
                                         no_imp_limit=300, focus_top_k=40)
                _update(p_e, sc_e, f'escalate_n{tn}')

            # LP at new scale (only if feasible)
            if best_seq is not None and best_seq.size <= 4096:
                if time.monotonic() < phase4_end - 15:
                    n_esc = best_seq.size
                    use_red = n_esc > 1500
                    per_lp_esc = min(90.0, phase4_end - time.monotonic() - 5)
                    if per_lp_esc > 10:
                        p_lp_e, sc_lp_e = _lp_refine(best_seq, time.monotonic() + per_lp_esc,
                                                      t_init=0.3, max_solves=12,
                                                      use_reduced=use_red,
                                                      top_k=200 if n_esc > 2048 else 300)
                        _update(p_lp_e, f'LP_n{n_esc}_esc')

    # --- Phase 5: Multi-restart at best length ---
    phase5_end = deadline - 30.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 3
        per_r = max(15.0, (phase5_end - time.monotonic()) / max(1, n_restarts + 1))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 10:
                break
            # Different perturbation strategies
            if i == 0:
                # Small Gaussian noise
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z += 0.04 * rng.normal(size=n_best)
            elif i == 1:
                # Frequency-domain perturbation
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                fa = np.fft.rfft(z)
                mask = np.ones_like(fa)
                mask[1:min(5, len(fa))] = 1.0 + 0.3 * rng.standard_normal(max(1, min(4, len(fa)-1)))
                z = np.fft.irfft(fa * mask, n_best)
            else:
                # Larger noise
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z += 0.15 * rng.normal(size=n_best)

            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per, rng, lr=0.04,
                                     beta_frac=0.06, beta_decay=0.93, no_imp_limit=
                                     int(0.4 * n_best), focus_top_k=30)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 6: Final fine-tune ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 8:
            p_f, sc_f = _adam_logit_v2(best_seq, time.monotonic() + per, rng,
                                        lr=0.015, beta_frac=0.03, beta_decay=0.96,
                                        no_imp_limit=200, focus_top_k=20,
                                        huber_delta=0.2)
            _update(p_f, sc_f, 'final_polish')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_raised_cosine(1024, 1.5)

        exact_sc = _score_norm(final)
        elapsed = time.monotonic() - t_start
        print(f"FINAL: score={exact_sc:.10f} n={final.size} [{elapsed:.1f}s]")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"EVAL error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_raised_cosine(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
