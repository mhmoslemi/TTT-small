    # = 2*a_{j-k} when j=2k (since j-k=k), and a_{j-k} when j≠2k.
    # Hmm wait, that's not right either. Let me be more careful.
    # conv_j = sum_i a_i * a_{j-i}
    # d(conv_j)/d(a_k) = sum_i [delta(i,k)*a_{j-i} + a_i * delta(j-i,k)]
    # = a_{j-k} + a_{j-k} = 2*a_{j-k} (when both i=k and j-i=k are in range)
    # = a_{j-k} (when only one is in range, e.g., j-k < 0 or j-k >= n)
    # = 2*a_{j-k} when 0 <= j-k < n (both terms contribute)
    # Actually if 0 <= j-k < n, then i=k is valid AND j-i=k means i=j-k which is also valid.
    # So the derivative is 2*a_{j-k} when 0 <= j-k < n, and 0 otherwise.
    # Wait no: if j-k < 0 or j-k >= n, then a_{j-k} = 0, so the derivative is 0.
    # If 0 <= j-k < n, then both i=k (valid) and i=j-k (valid, since 0<=j-k<n) contribute.
    # So d(conv_j)/d(a_k) = 2*a_{j-k} for 0<=j-k<n, else 0.
    
    # So grad[k] = sum_j w[j] * 2*a_{j-k} (for 0<=j-k<n)
    # = 2 * sum_j w[j] * a_{j-k}
    # = 2 * (w cross-correlate a)[k]
    
    # Cross-correlation: (w ⋆ a)[k] = sum_j w[j+k]*a[j] = sum_m w[m]*a[m-k] (sub m=j+k)
    # Hmm, let me use the standard definition: R[k] = sum_j w[j]*a[j+k]
    # Then we want sum_j w[j]*a[j-k] = R[-k]
    
    # Using FFT: R = IFFT(conj(FFT(w_pad)) * FFT(a_pad))
    # where w_pad and a_pad are zero-padded to length >= 2n-1.
    
    W = np.fft.rfft(w, N_pad)
    A = np.fft.rfft(a, N_pad)
    R = np.fft.irfft(np.conj(W) * A, N_pad)
    # R[k] = sum_j w[j]*a[j+k] (for the circular correlation, but with N_pad >= 2n-1 it's linear)
    # We want grad[k] = 2 * R[-k] = 2 * R[N_pad - k]
    # But R is of length N_pad, and R[N_pad-k] for k in [0, n-1]...
    # Actually R[k] for k in [-(n-1), n-1] is the linear cross-correlation.
    # R[-k] = R[N_pad - k] (by circularity), but since N_pad >= 2n-1, there's no wrap-around.
    # R[N_pad - k] for k in [1, n-1] gives us R[-k].
    # R[0] gives R[0].
    
    # So grad[k] = 2 * R[(N_pad - k) % N_pad]
    idx = (N_pad - np.arange(n)) % N_pad
    grad = 2.0 * R[idx]
    
    return grad, smax


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _solve_lp(f, M, n):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.success or res.status == 1:
        return res.x
    return None


def _golden_section_search(a, g, f0, f1, n_eval=12):
    """Golden section search for best t in [f0, f1] blending a and g."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = f0
    hi = f1
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    
    def eval_blend(t):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return _score(a_new)
    
    fc = eval_blend(c)
    fd = eval_blend(d)
    
    for _ in range(n_eval):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = eval_blend(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = eval_blend(d)
    
    t_best = (lo + hi) / 2.0
    a_new = (1.0 - t_best) * a + t_best * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return 0.0, np.inf
    a_new = a_new / s
    return t_best, _score(a_new)


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.01, inflate_min=1.001):
    """LP-based refinement with adaptive inflation and golden section line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            # Increase inflation to allow more improvement
            inflate = min(0.05, inflate * 1.2)
            if no_imp > 200:
                break
            continue

        g = g / sum_g

        # Golden section line search
        best_t, best_sc = _golden_section_search(a, g, 0.01, 1.0, n_eval=10)

        if best_t > 0.001 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Decrease inflation as we make progress
            inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.1)
            if no_imp > max(100, n):
                # Reset to best and restart with higher inflation
                a = best_a.copy()
                inflate = min(0.05, inflate_start * 1.5)
                no_imp = 0
                if inflate > 0.02:
                    break

    return best_a


def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient polishing with adaptive log-sum-exp smoothing."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.03 * float(c.max()))
    lr = 0.1
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        if it >= max_iter:
            break

        # Compute gradient of smoothed max
        grad, smax = _gradient_smooth_max(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        # Nesterov-style momentum
        v = momentum * v - lr * grad
        a_candidate = a + v
        a_new = _project_simplex(a_candidate)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.01, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        # Anneal beta
        if it % 50 == 49:
            beta = max(1e-8, beta * 0.9)

        it += 1

    return best_a


def _random_kick(a, sigma, rng):
    """Add random noise and project to simplex."""
    noise = rng.normal(0, sigma, size=len(a))
    a_noisy = a + noise
    a_noisy = np.maximum(a_noisy, 0)
    s = a_noisy.sum()
    if s < 1e-10:
        return a.copy()
    return a_noisy / s


def _combined_optimize(a0, deadline, n, rng):
    """Combined LP + gradient optimization with random restarts."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Phase 1: LP refinement
    t1 = time.monotonic()
    remaining = deadline - t1
    lp_time = min(remaining * 0.5, 400.0)
    
    if lp_time > 10:
        lp_end = t1 + lp_time
        a_lp = _lp_refine(a, lp_end, max_iter=30000)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
    
    # Phase 2: Gradient polish
    t2 = time.monotonic()
    remaining = deadline - t2
    gp_time = min(remaining * 0.3, 200.0)
    
    if gp_time > 10:
        gp_end = t2 + gp_time
        a_gp = _gradient_polish(best_a, gp_end)
        sc_gp = _score(a_gp)
        if sc_gp < best_score:
            best_score = sc_gp
            best_a = a_gp.copy()
    
    # Phase 3: Random kicks + short local optimization
    t3 = time.monotonic()
    remaining = deadline - t3
    if remaining > 30:
        n_kicks = int(remaining / 15)
        for k in range(n_kicks):
            if time.monotonic() > deadline - 10:
                break
            sigma = 0.05 / (1.0 + 0.1 * k)
            a_kick = _random_kick(best_a, sigma, rng)
            
            # Short LP refinement
            kick_end = min(time.monotonic() + 12, deadline - 10)
            a_kick_opt = _lp_refine(a_kick, kick_end, max_iter=5000,
                                     inflate_start=1.02, inflate_min=1.005)
            sc_kick = _score(a_kick_opt)
            if sc_kick < best_score:
                best_score = sc_kick
                best_a = a_kick_opt.copy()
    
    # Phase 4: Final gradient polish
    if time.monotonic() < deadline - 10:
        final_end = deadline - 5
        a_final = _gradient_polish(best_a, final_end, max_iter=10000)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: LP direction finding + gradient polish + random restarts.
    Tries multiple initializations and lengths to find the global optimum.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations at various lengths
    lengths = [512, 768, 1024]
    
    all_configs = []
    
    for n_t in lengths:
        configs_n = []
        
        if parent is not None:
            configs_n.append(('parent_interp', _interpolate(parent, n_t)))
        
        configs_n.append(('ramp1', _make_ramp(n_t, 1.0)))
        configs_n.append(('ramp2', _make_ramp(n_t, 2.0)))
        configs_n.append(('ramp3', _make_ramp(n_t, 3.0)))
        configs_n.append(('bell', _make_bell(n_t, 0.6)))
        configs_n.append(('bell_wide', _make_bell(n_t, 0.9)))
        configs_n.append(('arcsine', _make_arcsine(n_t)))
        configs_n.append(('cosine', _make_cosine(n_t)))
        configs_n.append(('three_part', _make_three_part(n_t)))
        configs_n.append(('uniform', _make_uniform(n_t)))
        
        # Add some random initializations
        for _ in range(3):
            a_rand = rng.uniform(0.1, 1.0, n_t)
            configs_n.append(('rand', a_rand / a_rand.sum()))
        
        all_configs.extend([(n_t, name, a) for name, a in configs_n])
    
    # Score and sort all configs
    scored = []
    for n_t, name, a0 in all_configs:
        sc = _score(a0)
        if np.isfinite(sc):
            scored.append((sc, n_t, name, a0))
    
    scored.sort(key=lambda x: x[0])
    
    print(f"Total configs: {len(scored)}, best initial: {scored[0][0]:.8f} ({scored[0][2]}, n={scored[0][1]})")
    
    # Update best with best initialization
    for sc0, n_t, name, a0 in scored[:5]:
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
    
    # Allocate time to top configs
    # Give more time to the best configs
    time_per_config = 120.0  # base time per config
    top_configs = scored[:12]  # Try top 12 configs
    
    print(f"Trying {len(top_configs)} top configs with ~{time_per_config}s each")
    
    for idx, (sc0, n_t, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 60:
            break
        
        # Skip if clearly worse than current best
        if sc0 > best_score + 0.05 and idx > 2:
            continue
        
        # Allocate time
        remaining = deadline - time.monotonic() - 30
        if remaining < 20:
            break
        
        # Better configs get more time
        if idx < 3:
            alloc_time = min(remaining * 0.25, 300.0)
        elif idx < 8:
            alloc_time = min(remaining * 0.15, 200.0)
        else:
            alloc_time = min(remaining * 0.1, 120.0)
        
        if alloc_time < 15:
            break
        
        config_end = time.monotonic() + alloc_time
        print(f"\nConfig {idx}: {name} n={n_t} init={sc0:.8f} budget={alloc_time:.0f}s")
        
        t_cfg_start = time.monotonic()
        a_opt, sc_opt = _combined_optimize(a0, config_end, n_t, rng)
        elapsed = time.monotonic() - t_cfg_start
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: score={sc_opt:.10f} (elapsed={elapsed:.0f}s)")
        else:
            print(f"  score={sc_opt:.8f} (no improvement, elapsed={elapsed:.0f}s)")
    
    # Final intensive polish on the best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nFinal polish: budget={deadline - time.monotonic():.0f}s, current best={best_score:.10f}")
        final_end = deadline - 10
        
        # Extended gradient polish
        a_fp = _gradient_polish(best_seq, final_end, max_iter=100000)
        sc_fp = _score(a_fp)
        if sc_fp < best_score:
            best_score = sc_fp
            best_seq = a_fp.copy()
            print(f"  Final polish: score={sc_fp:.10f}")
        
        # One more LP pass
        if time.monotonic() < deadline - 15:
            a_lpf = _lp_refine(best_seq, deadline - 10, max_iter=20000,
                              inflate_start=1.005, inflate_min=1.0005)
            sc_lpf = _score(a_lpf)
            if sc_lpf < best_score:
                best_score = sc_lpf
                best_seq = a_lpf.copy()
                print(f"  Final LP: score={sc_lpf:.10f}")
    
    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)
    
    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={total_time:.0f}s")
    return [float(x) for x in final_seq]
