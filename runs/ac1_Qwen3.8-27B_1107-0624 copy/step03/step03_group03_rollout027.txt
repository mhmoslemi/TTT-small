"""
Optimized approach for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses symmetry-enforced LP, log-sum-exp gradient with Nesterov acceleration,
and adaptive multi-phase optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    """Compute full convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Remove tiny negative values from FFT
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_conv(a)
    return float(2.0 * n * c.max())

def _make_sym_conv_matrix(a, M):
    """
    Create constraint matrix for the LP, exploiting symmetry.
    a is the full sequence (length n), but we only optimize the first n//2 elements.
    Returns A_ub, b_ub where A_ub @ b_half <= b_ub.
    """
    n = len(a)
    half = n // 2
    # The full b is symmetric: b[i] = b[n-1-i]
    # conv(a, b)[k] = sum_i a[i] * b[k-i]
    # With symmetric b: b[k-i] = b[n-1-(k-i)] = b[n-1-k+i]
    
    # We need conv(a, b)[k] <= M for all k
    # By symmetry of a (which we enforce), conv(a,b) is symmetric in k
    # So we only need k = 0, ..., n-1
    
    num_constraints = n  # We check lags 0 to n-1 (by symmetry, n+1 to 2n-2 are the same)
    A = np.zeros((num_constraints, half))
    b_vec = np.full(num_constraints, M)
    
    for k in range(num_constraints):
        for j in range(half):
            # b[j] appears in conv(a,b)[k] as a[k-j]*b[j]
            # b[n-1-j] = b[j] appears as a[k-(n-1-j)]*b[j]
            coeff = 0.0
            if 0 <= k - j < n:
                coeff += a[k - j]
            # From the symmetric part: b[n-1-j]
            idx2 = k - (n - 1 - j)
            if 0 <= idx2 < n:
                coeff += a[idx2]
            A[k, j] = coeff
    
    return A, b_vec

def _lp_symmetric(a, M):
    """
    Solve LP: max sum(b) subject to conv(a,b) <= M, b >= 0, b symmetric.
    Returns the full symmetric b vector.
    """
    n = len(a)
    half = n // 2
    
    A, b_vec = _make_sym_conv_matrix(a, M)
    
    c_obj = -np.ones(half)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * half, method='highs')
    
    if res.success:
        b_half = res.x
        b_full = np.empty(n)
        b_full[:half] = b_half
        b_full[half:] = b_half[::-1]
        # Handle odd n (middle element)
        if n % 2 == 1:
            b_full[half] = b_half[-1]  # This won't happen for even n
        return b_full
    return None

def _lp_dense(a, M, n_top=200):
    """
    LP with only the top n_top constraints (near-active lags).
    For large n, this is much faster than the full LP.
    """
    n = len(a)
    c_aa = _fft_conv(a)
    
    # Get the top n_top lags
    idx = np.argsort(c_aa)[::-1][:n_top]
    
    # Build constraint matrix for these lags only
    num_c = len(idx)
    A = np.zeros((num_c, n))
    b_vec = np.full(num_c, M)
    
    for row, k in enumerate(idx):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

def _lse_adam(a, z, state, T, lr):
    """One Adam step in logit space using log-sum-exp."""
    n = len(a)
    c = _fft_conv(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())

        w[int(np.argmax(c))] = 1.0

        w = ex / s

    N = _next_pow2(2 * n - 1)

    fa_rev = np.fft.rfft(a[::-1], N)

    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    
    state['t'] = t
    state['m'] = m
    state['v'] = v
    
    return a_new, z, state, m_val

def _nesterov_subgrad(a, deadline, max_iter=50000, lr=0.01):

    avg_a = a.copy()

        if time.monotonic() > deadline:

        m_val = float(c_y.max())

        w[max_idx] = 1.0 / len(max_idx)

        fw = np.fft.rfft(w, N)
        fy_rev = np.fft.rfft(y[::-1], N)
        h_full = np.fft.irfft(fw * fy_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Project to tangent space
        dot = float(np.dot(y, grad))
        grad_tan = grad - dot * y
        
        # Nesterov update
        t_it = it + 1
        beta_it = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * t_it)) / 2.0
        # Simplified: use constant momentum
        v = beta * v - base_lr * grad_tan
        y_new = y + v
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()
        
        # Miracle step: running average
        avg_a = (t_it * avg_a + y_new) / (t_it + 1)
        
        # Check both y and avg
        score_y = _score_norm(y_new)
        score_avg = _score_norm(avg_a)
        
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y_new.copy()
            no_imp = 0

                v = np.zeros(n)
                avg_a = best_a.copy()

def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_tukey(n, r=0.3):
    """Tukey (tapered cosine) window."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    half = int(r * n / 2)
    if half > 0:
        t_left = np.linspace(0, np.pi, half)
        taper = 0.5 * (1 + np.cos(t_left))
        a[:half] = taper
        a[-half:] = taper[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_optimal_profile(n):
    """
    Theoretical profile that minimizes peak autocorrelation.
    Based on the idea that the optimal non-negative sequence 
    has a specific smooth profile related to the inverse of the 
    triangular function.
    """
    # For large n, the optimal profile is approximately:
    # a[i] proportional to 1/sqrt(i*(n-1-i)) near the edges
    # and roughly constant in the middle
    # This is derived from the requirement that conv(a,a) is flat.
    t = np.linspace(1e-6, 1 - 1e-6, n)
    a = 1.0 / np.sqrt(t * (1 - t) + 0.1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_bistable(n, center_frac=0.5):
    """Two-peaked profile."""
    t = np.linspace(0.0, 1.0, n)
    sigma = 0.15
    a = np.exp(-0.5 * ((t - center_frac * 0.3) / sigma) ** 2) + \
        np.exp(-0.5 * ((t - center_frac * 1.7) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_conv(a)
    return float(2.0 * n * c.max())


def _lp_topk(a, M, k=300):
    """LP: max sum(b) s.t. conv(a,b)[top-k lags] <= M, b>=0. Vectorized."""
    n = len(a)
    c_aa = _fft_conv(a)
    idx = np.argsort(c_aa)[::-1][:k]
    idx = np.sort(idx)
    
    A = np.zeros((len(idx), n))
    for row, lag in enumerate(idx):
        lo = max(0, lag - n + 1)
        hi = min(n, lag + 1)
        A[row, lo:hi] = a[lag - lo:lag - lo + (hi - lo)]
    
    b_vec = np.full(len(idx), M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_full(a, M):
    """Full LP for smaller n."""
    n = len(a)
    c_aa = _fft_conv(a)
    A = np.zeros((2 * n - 1, n))
    for lag in range(2 * n - 1):
        lo = max(0, lag - n + 1)
        hi = min(n, lag + 1)
        A[lag, lo:hi] = a[lag - lo:lag - lo + (hi - lo)]
    
    b_vec = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lse_adam_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_conv(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a0, deadline, max_iter=100000, T_start=0.05, T_end=1e-6, lr=0.1):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    
    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, m_val = _lse_adam_step(a, z, state, T, lr)
        
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        
        c = _fft_conv(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > max(5000, 10 * n):
                rng = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.06 * rng.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        
        T = max(T * T_ratio, T_end)
    
    return best_a


def _nesterov_polish(a0, deadline, max_iter=80000, lr=0.005):
    """Nesterov-accelerated subgradient with running average."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    base_lr = lr
    no_imp = 0
    avg_a = a.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_y = _fft_conv(y)
        m_val = float(c_y.max())
        max_idx = np.where(c_y >= m_val - 1e-10)[0]
        w = np.zeros(2 * n - 1)
        w[max_idx] = 1.0 / len(max_idx)
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fy_rev = np.fft.rfft(y[::-1], N)
        h_full = np.fft.irfft(fw * fy_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        dot = float(np.dot(y, grad))
        grad_tan = grad - dot * y
        
        v = beta * v - base_lr * grad_tan
        y_new = y + v
        y_new = np.maximum(y_new, 1e-15)
        s = y_new.sum()
        if s < 1e-10:
            y_new = y
        else:
            y_new = y_new / s
        
        t_it = it + 1
        avg_a = (t_it * avg_a + y_new) / (t_it + 1)
        
        score_y = _score_norm(y_new)
        score_avg = _score_norm(avg_a)
        
        improved = False
        if score_y < best_score - 1e-13:
            best_score = score_y
            best_a = y_new.copy()
            improved = True
        if score_avg < best_score - 1e-13:
            best_score = score_avg
            best_a = avg_a.copy()
            improved = True
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                base_lr = max(0.0005, base_lr * 0.95)
            if no_imp > max(8000, 12 * n):
                y = best_a.copy()
                v = np.zeros(n)
                avg_a = best_a.copy()
                base_lr = lr
                no_imp = 0
        
        y = y_new
    
    return best_a


def _lp_refine(a0, deadline, max_iter=8000, inflate=1.0001, blend=0.12, k=300):
    """Tight LP refinement with adaptive parameters."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend
    
    use_full = n <= 256
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        
        if use_full:
            g = _lp_full(a, M_lp)
        else:
            g = _lp_topk(a, M_lp, k=k)
        
        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.4)
            no_imp += 3
            if no_imp > 300:
                break
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.03)
            if no_imp > 400:
                break
            continue
        
        g = g / sum_g
        
        # Multi-point line search
        best_t = blend_cur
        best_sc = _score_norm(np.maximum((1.0 - blend_cur) * a + blend_cur * g, 0) /
                              np.maximum((1.0 - blend_cur) * a + blend_cur * g, 0).sum())
        
        for t_try in [blend_cur * 0.25, blend_cur * 0.5, blend_cur, blend_cur * 1.5, blend_cur * 2.0]:
            a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t_try
        
        a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        score_new = _score_norm(a_new)
        
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 25 == 0:
                blend_cur = max(0.01, blend_cur * 0.85)
                inflate_cur = min(1.003, inflate_cur * 0.92)
            if no_imp > max(150, n // 3):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tukey(n, r=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    half = max(1, int(r * n / 2))
    if half < n // 2:
        taper = 0.5 * (1 + np.cos(np.linspace(0, np.pi, half)))
        a[:half] = taper
        a[-half:] = taper[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_inverse_sqrt(n):
    """Profile ~ 1/sqrt(x*(1-x)) + constant, related to optimal autocorrelation flattening."""
    t = np.linspace(0.01, 0.99, n)
    a = 1.0 / (np.sqrt(t * (1.0 - t)) + 0.3)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bell_edge(n, edge_frac=0.15):
    """Slightly elevated edges."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.exp(-((t - 0) / edge_frac) ** 2) + 0.3 * np.exp(-((t - 1) / edge_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LSE gradient -> tight LP -> Nesterov polish -> final LP.
    Uses diverse initializations, adaptive parameters, and top-k LP for efficiency.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Determine lengths
    lengths = [1024]
    if parent is not None and parent.size != 1024:
        lengths = [parent.size, 1024]
    lengths.append(512)
    lengths.append(2048)
    
    # Deduplicate
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    lengths = unique_lengths
    print(f"Lengths: {lengths}")
    
    def get_candidates(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            p = _interpolate_profile(parent, n_t)
            p2 = p * (1.0 + 0.04 * rng.normal(size=n_t))
            p2 = np.maximum(p2, 0)
            if p2.sum() > 1e-10:
                cands.append(p2 / p2.sum())
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_tukey(n_t, 0.25))
        cands.append(_make_inverse_sqrt(n_t))
        cands.append(_make_bell_edge(n_t))
        return cands
    
    # Phase 1: LSE optimization
    print("\n=== Phase 1: LSE gradient ===")
    best_per_len = {}
    
    for n_t in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf
        
        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 400:
                break
            
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            remaining = deadline - time.monotonic() - 400
            lse_budget = min(120.0, remaining)
            if lse_budget < 15:
                break
            lse_end = time.monotonic() + lse_budget
            
            a_res = _lse_optimize(a0, lse_end, max_iter=80000,
                                   T_start=0.04, T_end=1e-6, lr=0.08)
            sc = _score(a_res)
            
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_res.copy()
            if sc < best_here_score:
                best_here_score = sc
                best_here = a_res
            print(f"  n={n_t} c{ci}: {sc0:.8f} -> {sc:.10f}")
        
        if best_here is not None:
            best_per_len[n_t] = best_here
            print(f"  Best n={n_t}: {best_here_score:.10f}")
    
    # Phase 2: LP refinement (tight)
    print("\n=== Phase 2: Tight LP ===")
    for n_t in lengths:
        if n_t not in best_per_len:
            continue
        if time.monotonic() > deadline - 250:
            break
        
        a_in = best_per_len[n_t]
        sc_in = _score(a_in)
        remaining = deadline - time.monotonic() - 250
        lp_budget = min(250.0, remaining)
        if lp_budget < 20:
            continue
        lp_end = time.monotonic() + lp_budget
        
        k = 400 if n_t <= 1024 else 600
        a_lp = _lp_refine(a_in, lp_end, max_iter=8000,
                          inflate=1.0001, blend=0.1, k=k)
        sc_lp = _score(a_lp)
        
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_len[n_t] = a_lp
        print(f"  LP n={n_t}: {sc_in:.10f} -> {sc_lp:.10f}")
    
    # Phase 3: Nesterov polish
    print("\n=== Phase 3: Nesterov subgradient ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        remaining = deadline - time.monotonic() - 120
        ns_budget = min(150.0, remaining)
        if ns_budget > 20:
            ns_end = time.monotonic() + ns_budget
            a_ns = _nesterov_polish(best_seq, ns_end, max_iter=100000, lr=0.004)
            sc_ns = _score(a_ns)
            if np.isfinite(sc_ns) and sc_ns < best_score:
                best_score = sc_ns
                best_seq = a_ns.copy()
            print(f"  Nesterov: {sc_ns:.10f}")
    
    # Phase 4: Final very-tight LP
    print("\n=== Phase 4: Final tight LP ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic() - 50
        lp_budget = min(80.0, remaining)
        if lp_budget > 15:
            lp_end = time.monotonic() + lp_budget
            n_t = len(best_seq)
            k = 500 if n_t <= 1024 else 800
            a_fin = _lp_refine(best_seq, lp_end, max_iter=5000,
                               inflate=1.00005, blend=0.05, k=k)
            sc_fin = _score(a_fin)
            if np.isfinite(sc_fin) and sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
            print(f"  Final LP: {sc_fin:.10f}")
    
    # Phase 5: Final LSE micro-polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 15
        lse_budget = min(30.0, remaining)
        if lse_budget > 5:
            lse_end = time.monotonic() + lse_budget
            a_last = _lse_optimize(best_seq, lse_end, max_iter=20000,
                                    T_start=0.01, T_end=1e-7, lr=0.02)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LSE: {sc_last:.10f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0
    
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            best_score = min(best_score, sc_eval)
    except Exception as e:
        print(f"Eval error: {e}")
    
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
