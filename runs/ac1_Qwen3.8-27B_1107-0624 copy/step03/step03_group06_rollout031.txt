"""
LP-based optimization with adaptive inflation, golden-section line search,
diverse initializations (cosine, Chebyshev, ramp), symmetrization, and
gradient polish for minimizing 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _make_cosine(n, k=1):
    """Cosine profile: p_j = (1 + cos(2*pi*k*(j+0.5)/n)) normalized."""
    j = np.arange(n)
    p = 1.0 + np.cos(2.0 * np.pi * k * (j + 0.5) / n)
    p = np.maximum(p, 1e-10)
    return _normalize(p)


def _make_chebyshev(n, order=2):
    """Chebyshev polynomial profile (ensured non-negative)."""
    x = np.linspace(-1, 1, n)
    p = np.polynomial.chebyshev.chebval(x, [1]* (order+1))  # T_order(x)
    # T_2(x) = 2x^2 - 1, range [-1, 2]
    # Shift and clip to make non-negative
    p = p - p.min() + 1e-8
    return _normalize(p)


def _make_chebyshev2(n, order=3):
    """Higher-order Chebyshev, shifted to be non-negative."""
    x = np.linspace(-1, 1, n)
    p = np.polynomial.chebyshev.chebval(x, np.zeros(order+1))
    p[order] = 1.0  # T_order
    p = p - p.min() + 1e-8
    return _normalize(p)


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    p = t**alpha + 1e-8
    return _normalize(p)

    """Gaussian-like bell profile."""
    t = np.linspace(-1, 1, n)

def _interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _build_conv_matrix(n):
    """Build the (2n-1) x n convolution matrix for auto-convolution."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = 1.0  # Wait, this isn't right
    # Actually, for convolve(a, a), the k-th element is sum_j a_j * a_{k-j}
    # The matrix A such that (A @ a)_k = sum_j A[k,j] * a_j gives us conv(a, something)
    # For the LP: we want A @ g such that (A @ g)_k = sum_j a_j * g_{k-j}
    # This is the convolution of a with g, so A[k,j] = a_{k-j} (when valid)
    return A


def _build_conv_matrix_with_a(a):
    """Build the (2n-1) x n matrix M such that M @ g = convolve(a, g)."""
    n = len(a)
    M = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g_j contributes to positions j, j+1, ..., j+n-1 in the convolution
        # (a * g)_k = sum_j a_{k-j} * g_j
        # So M[k, j] = a_{k-j} when 0 <= k-j < n
        start = j
        end = min(2 * n - 1, j + n)
        # M[j:j+n, j] = a
        M[start:end, j] = a[:end - start]
    return M


def _solve_lp(a, M_val, n):
    """Solve: max sum(g) s.t. conv(a,g) <= M_val, g >= 0."""
    M_matrix = _build_conv_matrix_with_a(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M_val)
    res = linprog(c_obj, A_ub=M_matrix, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, res.fun
    return None, np.inf


def _golden_section_search(a, g, n_eval_max=20):
    """Find t in [0,1] that minimizes score((1-t)*a + t*g)."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate ≈ 0.618
    
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    
    a_c = np.maximum((1.0 - c) * a + c * g, 0.0)
    a_c = _normalize(a_c)
    sc_c = _score(a_c)
    
    a_d = np.maximum((1.0 - d) * a + d * g, 0.0)
    a_d = _normalize(a_d)
    sc_d = _score(a_d)
    
    n_eval = 2
    for _ in range(n_eval_max - 2):
        if sc_c < sc_d:
            hi = d
            d = c
            sc_d = sc_c
            c = hi - phi * (hi - lo)
            a_c = np.maximum((1.0 - c) * a + c * g, 0.0)
            a_c = _normalize(a_c)
            sc_c = _score(a_c)
        else:
            lo = c
            c = d
            sc_c = sc_d
            d = lo + phi * (hi - lo)
            a_d = np.maximum((1.0 - d) * a + d * g, 0.0)
            a_d = _normalize(a_d)
            sc_d = _score(a_d)
        n_eval += 1
    
    t = (lo + hi) / 2.0
    a_best = np.maximum((1.0 - t) * a + t * g, 0.0)
    a_best = _normalize(a_best)
    sc_best = _score(a_best)
    return t, a_best, sc_best


def _lp_refine(a0, deadline, max_iter=50000):
    """Main LP refinement loop with adaptive inflation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    # Adaptive inflation
    inflate = 0.005
    min_inflate = 0.0005
    no_imp = 0
    stall_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * (1.0 + inflate)
        
        g, lp_val = _solve_lp(a, M_lp, n)
        if g is None:
            break
        
        sum_g = float(g.sum())
        
        if sum_g <= 1.0 + 1e-6:
            # No improvement possible with this M
            no_imp += 1
            if no_imp % 10 == 0:
                inflate *= 1.5
                if inflate > 0.05:
                    inflate = 0.05
            if no_imp > 200:
                # Try a kick
                rng = np.random.default_rng(it)
                kick = rng.random(n) * 0.1
                a_kick = _normalize(a + kick)
                sc_kick = _score(a_kick)
                if sc_kick < best_score:
                    a = a_kick
                no_imp = 0
                inflate = 0.005
                if stall_count > 5:
                    break
                stall_count += 1
            continue
        
        # Normalize g
        g = g / sum_g
        
        # Golden section line search
        t, a_new, sc_new = _golden_section_search(a, g, n_eval_max=15)
        
        # Also try symmetrized version
        a_new_sym = _normalize(_symmetrize(a_new))
        sc_new_sym = _score(a_new_sym)
        if sc_new_sym < sc_new:
            a_new = a_new_sym
            sc_new = sc_new_sym
        
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            stall_count = 0
            # Narrow inflation on success
            if it % 50 == 49:
                inflate = max(min_inflate, inflate * 0.9)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.3)
            if no_imp > 150:
                stall_count += 1
                no_imp = 0
                inflate = 0.003
                # Kick
                rng = np.random.default_rng(12345 + it)
                kick = rng.standard_normal(n) * 0.05
                a_kick = np.maximum(a + kick, 0.0)
                a_kick = _normalize(a_kick)
                sc_kick = _score(a_kick)
                if sc_kick < best_score:
                    a = a_kick
                else:
                    a = best_a.copy()
                if stall_count > 8:
                    break
    
    return best_a, best_score


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip for numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient of sum_k w_k * (a*a)_k w.r.t. a
    # d/d a_j [sum_k w_k * sum_l a_l * a_{k-l}] = sum_k w_k * (a_{k-j} + a_{k-j}) ... 
    # Actually: (a*a)_k = sum_j a_j * a_{k-j}
    # d/d a_j (a*a)_k = a_{k-j} + a_{k-j} = 2 * a_{k-j} (if k-j != j) or a_k (if k-j = j)
    # Wait, more carefully: d/d a_j [sum_l a_l * a_{k-l}] = a_{k-j} + (if j = k-j then a_j... no)
    # sum_l a_l * a_{k-l}: derivative w.r.t. a_j is a_{k-j} + a_{k-j} if we count both terms
    # Actually no: it's just a_{k-j} (from the first factor) + a_{k-j} (from the second factor when l = k-j, i.e., when the term is a_{k-j} * a_j)
    # Hmm, let me be more careful.
    # (a*a)_k = sum_{l: 0<=l<n, 0<=k-l<n} a_l * a_{k-l}
    # d/d a_j: for the term where l=j: a_{k-j}; for the term where k-l=j (i.e., l=k-j): a_j
    # So d/d a_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (when j != k-j, i.e., 2j != k)
    # When 2j = k (if k is even and j = k/2): it's just a_j (the term is a_j * a_j, derivative is 2*a_j, same thing!)
    # Wait: d/d a_j (a_j * a_j) = 2*a_j. And 2*a_{k-j} = 2*a_j when k=2j. So the formula 2*a_{k-j} works in all cases.
    # 
    # So grad_j = sum_k w_k * 2 * a_{k-j} (where a_{k-j} is 0 if k-j not in [0,n-1])
    # This is 2 * (w * a_reversed)[j + n - 1] ... let me use FFT.
    
    # Actually, the gradient is: grad = 2 * convolve(w, a_reversed)[n-1:2n-1]
    # Where a_reversed = a[::-1]
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient descent with smooth max and momentum."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-8, 0.02 * float(c.max()))
    lr = 0.02
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_trial = a + momentum * v
        a_trial = np.maximum(a_trial, 0.0)
        if a_trial.sum() < 1e-10:
            a_trial = a
        a_trial = _normalize(a_trial)
        
        grad, maxc = _smooth_max_grad(a_trial, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > 2000:
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.02 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
        
        if it % 200 == 199:
            beta = max(1e-10, beta * 0.7)
    
    return best_a, best_score

LP-based optimization with adaptive inflation, golden-section line search,
diverse initializations (cosine, Chebyshev, bell), symmetrization, and

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _make_cosine(n, k=1):
    j = np.arange(n)
    p = 1.0 + np.cos(2.0 * np.pi * k * (j + 0.5) / n)
    p = np.maximum(p, 1e-10)
    return _normalize(p)


def _make_cosine_asym(n, k=1, shift=0.3):
    j = np.arange(n)
    t = (j + shift) / n
    p = 1.0 + np.cos(2.0 * np.pi * k * t)
    p = np.maximum(p, 1e-10)
    return _normalize(p)


def _make_chebyshev(n, order=2):
    x = np.linspace(-1, 1, n)
    coeffs = np.zeros(order + 1)
    coeffs[order] = 1.0
    p = np.polynomial.chebyshev.chebval(x, coeffs)
    p = p - p.min() + 1e-8
    return _normalize(p)


def _make_chebyshev_mix(n, o1=1, o2=3, w=0.5):
    x = np.linspace(-1, 1, n)
    c1 = np.zeros(o1 + 1)
    c1[o1] = 1.0
    c2 = np.zeros(o2 + 1)
    c2[o2] = 1.0
    p1 = np.polynomial.chebyshev.chebval(x, c1)
    p2 = np.polynomial.chebyshev.chebval(x, c2)
    p = w * p1 + (1 - w) * p2
    p = p - p.min() + 1e-8
    return _normalize(p)


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    p = t ** alpha + 1e-8
    return _normalize(p)


def _make_bell(n, width=0.5):
    t = np.linspace(-1, 1, n)
    p = np.exp(-(t / width) ** 2)
    return _normalize(p)


def _make_uniform(n):
    return np.ones(n) / n


def _make_flat_top(n, frac=0.7):
    p = np.zeros(n)
    start = int(n * (1 - frac) / 2)
    end = n - start
    p[start:end] = 1.0
    # Smooth edges
    edge = max(1, n // 32)
    for i in range(edge):
        w = (i + 1) / (edge + 1)
        if start - 1 - i >= 0:
            p[start - 1 - i] = w
        if end + i < n:
            p[end + i] = w
    return _normalize(p)


def _interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _build_conv_matrix(a):
    n = len(a)
    M = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        M[j:j + n, j] = a
    return M


def _solve_lp(a, M_val, n, reduced_k=None):
    if reduced_k is not None and reduced_k < 2 * n - 1:
        c = _fft_convolve(a)
        threshold = np.sort(c)[::-1][min(reduced_k, len(c) - 1)]
        active = np.where(c >= threshold)[0]
    else:
        active = np.arange(2 * n - 1)

    M_matrix = _build_conv_matrix(a)
    M_sub = M_matrix[active, :]

    c_obj = -np.ones(n)
    b = np.full(len(active), M_val)
    try:
        res = linprog(c_obj, A_ub=M_sub, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, np.inf
    if res.success:
        return res.x, res.fun
    return None, np.inf


def _golden_section_search(a, g, n_eval=15):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)

    a_c = _normalize(np.maximum((1.0 - c) * a + c * g, 0.0))
    sc_c = _score(a_c)
    a_d = _normalize(np.maximum((1.0 - d) * a + d * g, 0.0))
    sc_d = _score(a_d)

    for _ in range(max(0, n_eval - 2)):
        if sc_c < sc_d:
            hi = d
            d = c
            sc_d = sc_c
            c = hi - phi * (hi - lo)
            a_c = _normalize(np.maximum((1.0 - c) * a + c * g, 0.0))
            sc_c = _score(a_c)
        else:
            lo = c
            c = d
            sc_c = sc_d
            d = lo + phi * (hi - lo)
            a_d = _normalize(np.maximum((1.0 - d) * a + d * g, 0.0))
            sc_d = _score(a_d)

    t = (lo + hi) / 2.0
    a_best = _normalize(np.maximum((1.0 - t) * a + t * g, 0.0))
    sc_best = _score(a_best)
    return t, a_best, sc_best

def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.015 * float(c.max()))
    lr = 0.015
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_trial = _normalize(np.maximum(a + momentum * v, 0.0))
        grad, maxc = _smooth_max_grad(a_trial, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a = _project_simplex(a + v)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 80 == 0:
                lr = max(0.002, lr * 0.94)
            if no_imp > 1500:
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, 0.015 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 150 == 149:
            beta = max(1e-10, beta * 0.75)

    return best_a, best_score

                print(f"\nevaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
