```python
def lp_step(a, deadline, rng):
    """
    Given current a (on simplex), solve the LP to find g >= 0
    maximizing sum(g) subject to conv(a, g) <= M where M = max(conv(a,a)).
    Then interpolate: a_new = (1-t)*a + t*g_normalized
    """
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    
    # Build constraint matrix A where A[k, j] = a[j] if 0 <= k-j < n, else 0
    # A has shape (2n-1, n)
    # Total nonzeros: n^2
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A_sparse = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) s.t. A@g <= M, g >= 0
    # In linprog: minimize -sum(g) s.t. A@g <= M, g >= 0
    from scipy.optimize import linprog
    result = linprog(
        c=-np.ones(n),
        A_ub=A_sparse,
        b_ub=M * np.ones(2*n-1),
        bounds=[(0, None)] * n,
        method='highs'
    )
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 0:
            g_norm = g / g_sum  # put on simplex
            # Interpolate
            t = 0.1  # step size
            a_new = (1-t) * a + t * g_norm
            return a_new / a_new.sum()
    return None
```

```python
"""
LP-based and gradient-based hybrid optimization for minimizing 
2n*max(conv(a,a))/sum(a)^2. Uses sparse linear programming to find 
optimal descent directions, combined with soft-max gradient methods 
for polishing. Multi-scale exploration with diverse restarts.
"""
import numpy as np
import scipy.sparse
import scipy.optimize
import time

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def build_lp_matrix(a):
    """Build the sparse constraint matrix for the LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            if a[j] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_descent_step(a, max_t=1.0, method='highs'):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    Return the normalized g direction.
    """
    n = len(a)
    c = full_convolve_fft(a)
    M = float(c.max())
    
    if M < 1e-15:
        return None
    
    A = build_lp_matrix(a)
    
    # Maximize sum(g) s.t. A@g <= M, g >= 0
    # linprog minimizes, so negate objective
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    result = scipy.optimize.linprog(
        c=c_obj,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method=method,
        options={'presolve': True, 'time_limit': 60}
    )
    
    if result.success and result.x is not None:
        g = result.x
        g_sum = float(g.sum())
        if g_sum > 1e-10:
            return g / g_sum
    return None

def lp_with_line_search(a, deadline, rng, max_lp_calls=50):
    """
    Perform LP-based descent with line search on step size.
    """
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    lp_calls = 0
    
    while time.monotonic() < deadline and lp_calls < max_lp_calls:
        g = lp_descent_step(a, method='highs')
        lp_calls += 1
        
        if g is None:
            break
        
        # Line search: find best t
        best_t = 0.0
        best_sc = compute_score(a)
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc = compute_score(a_try)
                if sc < best_score:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    break
            else:
                break
    
    return best_a, best_score

def softmax_gradient_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.05,
                               beta_decay=0.93, beta_min=1e-12, 
                               restart_period=500, use_nesterov=False):
    """
    Soft-max gradient optimization in logit space.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6
    
    # Use Nesterov momentum if requested
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    y = z.copy()  # Nesterov lookahead
    
    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        # Compute gradient at (possibly Nesterov) point
        if use_nesterov:
            y = z + b1 * (z - (z - m / (np.sqrt(v) + eps) * 0))  # simplified
            # Actually let's just use standard Adam for reliability
            a_eval = np.exp(y - y.max())
            a_eval = a_eval / a_eval.sum()
        else:
            a_eval = a0
        
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a_eval)
        grad_a = 2.0 * h
        dot = float(np.dot(a_eval, grad_a))
        dz = a_eval * (grad_a - dot)
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_a.__class__(np.log(np.maximum(best_a, 1e-15)))
            z = z - z.mean()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            a0 = best_a.copy()
            continue
        a0 = a0 / s
        
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
        if no_improve > 200:

            v.fill(0.0)

def multi_beta_gradient(a0, deadline, rng, betas=None, lr=0.02):
    """
    Gradient optimization using multiple beta values simultaneously.
    Combines gradients from different soft-max sharpness levels.
    """
    if betas is None:
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        betas = [max_c * f for f in [0.01, 0.05, 0.1, 0.2, 0.5]]
    
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    z = np.log(a0)
    z = z - z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a0.copy()
    best_score = compute_score(a0)
    best_score_inner = float(2.0 * n * full_convolve_fft(a0).max())
    no_improve = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        
        # Combine gradients from multiple betas
        grad_total = np.zeros(n)
        for beta in betas:
            if beta < 1e-14:
                beta = 1e-14
            _, w = soft_max_weights(c, beta)
            h = compute_h(w, a0)
            grad_total += 2.0 * h
        
        dot = float(np.dot(a0, grad_total))
        dz = a0 * (grad_total - dot)
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            a0 = best_a.copy()
            continue
        a0 = a0 / s
        
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score_inner = float(2.0 * n * max_c)
        
        if score_inner < best_score_inner - 1e-14:
            best_score_inner = score_inner
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if t_adam % 100 == 0:
            current_lr = max(0.002, current_lr * 0.99)
        
        if no_improve > 800:
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            z = z + 0.08 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0
    
    return best_a, compute_score(best_a)

def cosine_profile(n, k=1, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result
```

As a matrix: `(A_k)_{ij} = 1` if `i + j = k`, else 0. Wait, that's not quite right either.

`conv(k) = sum_{j: valid} a[j] * a[k-j]`. For this to be written as `sum_{i,j} a[i] * (A_k)_{ij} * a[j]`, we need `(A_k)_{ij} = 1/2` if `i+j = k` (for `i ≠ j`) and `= 1` if `i = j` and `2i = k`. Actually, the standard way: `a^T A a = sum_{i,j} a[i] A[i,j] a[j]`. We want this to equal `sum_j a[j] a[k-j]`. So `A[i,j] = 1` if `j = k-i` (i.e., `i+j = k`), else 0. But then `a^T A a = sum_i a[i] a[k-i]`, which is exactly `conv(k)`.

Is `A_k` positive semi-definite? The eigenvalues of `A_k` (where `A_k[i,j] = 1` if `i+j=k`)... This is a permutation-like matrix (anti-diagonal). For a square matrix, the eigenvalues of an anti-diagonal matrix with 1s are `±1`. So `A_k` is NOT positive semi-definite!

This means `conv(k)` is NOT convex in `a`! It's an indefinite quadratic form. So the soft-max of these is also not convex.

This is a problem for using convex optimization methods. The landscape is non-convex, and we need to be careful about local minima.

Given this, the gradient-based approaches (Adam, L-BFGS) are still appropriate, but we need to be aware that they can get stuck in local minima. The LP-based approach and simulated annealing can help escape.
