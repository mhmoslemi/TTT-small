    """Make the spectrum flatter."""
    A = np.fft.rfft(a)
    mag = np.abs(A)
    phase = np.angle(A)
    # Blend magnitude with uniform

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

    n = len(a)
    vals = _lp_fill_values(a, rows, cols, n)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_init=0.5, max_solves=200, perturb_every=40,
               rng=None):
    """LP-based duality refinement with adaptive step and spectral perturbation."""
    n = len(a)
    rows, cols = _build_lp_pattern(n)
    current_best = a.copy()
    current_score = _score_norm(a)
    a = a.copy()
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    no_imp = 0
    n_solves = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M, rows, cols)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b

        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < current_score - 1e-14:
            current_score = sc_new
            current_best = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())

        # Adaptive step size
        diff = np.linalg.norm(a - b_norm) / max(1.0, np.linalg.norm(a))
        if diff < 0.01:
            t = max(0.1, t * 0.8)
        elif diff > 0.3:
            t = min(0.8, t * 1.1)
        else:
            t = max(0.1, t * 0.95)

        # Spectral perturbation to escape plateaus
        if no_imp >= perturb_every and rng is not None:
            # Apply spectral shaping
            A = np.fft.rfft(a)
            mag = np.abs(A)
            phase = np.angle(A)
            # Make spectrum slightly flatter
            mag_new = mag ** 0.95
            A_new = mag_new * np.exp(1j * phase)
            a_pert = np.fft.irfft(A_new, n)
            a_pert = np.maximum(a_pert, 1e-15)
            # Add small random noise in log space
            log_a = np.log(a_pert)
            log_a += 0.05 * rng.normal(size=n)
            a_pert = np.exp(log_a - log_a.max()) + 1e-15
            a_pert = a_pert / a_pert.sum()
            sc_pert = _score_norm(a_pert)
            if sc_pert < current_score * 1.05:
                a = a_pert
                c = _full_convolve(a)
                M = float(c.max())
            else:
                a = current_best.copy()
                c = _full_convolve(a)
                M = float(c.max())
            t = t_init * 0.7
            no_imp = 0

        # Also try pure gradient step every few iterations
        if n_solves % 10 == 0 and time.monotonic() < deadline:
            beta = max(1e-3, 0.05 * M)
            _, w = _smooth_max_and_weights(c, beta)
            g = _conv_cross_corr(w, a)
            vp = g
            dot = float(np.dot(a, vp))
            gz = a * (vp - dot)
            gz -= float(np.mean(gz))
            a_grad = a - 0.02 * gz
            a_grad = np.maximum(a_grad, 1e-15)
            a_grad = a_grad / a_grad.sum()
            sc_grad = _score_norm(a_grad)
            if sc_grad < current_score - 1e-12:
                current_score = sc_grad
                current_best = a_grad.copy()
                a = a_grad
                c = _full_convolve(a)
                M = float(c.max())

    return current_best, current_score, n_solves


def _adam_logit(p0, deadline, rng, lr=0.04, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=500):
    """Adam optimizer in logit space with smooth-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)

        if no_imp > no_imp_limit:
            z = best_z + 0.08 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _spectral_shape(a, alpha=0.9):
    """Make spectrum flatter via power-law shaping."""
    n = len(a)
    A = np.fft.rfft(a)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_new = mag ** alpha
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_risen_cosine(n, beta_param):
    """Raised cosine window: a[i] = 0.5*(1+cos(2*pi*beta*i/n)) shape."""
    t = np.arange(n)
    a = 0.5 * (1.0 + beta_param * np.cos(2.0 * np.pi * t / n))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _nelder_mead_1d(f, x0, dx, max_iter=50):
    """Simple 1D Nelder-Mead (golden section) minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = x0 - dx, x0 + dx
    for _ in range(max_iter):
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc, fd = f(c), f(d)
        if fc < fd:
            b = d
        else:
            a = c
    return (a + b) / 2.0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Initial candidates (60s) ---
    phase1_end = t_start + 60.0
    candidates = []

    # Spectral-shaped variants of parent
    if parent is not None:
        for alpha in [0.7, 0.8, 0.9, 1.1, 1.3]:
            ps = _spectral_shape(parent, alpha)
            sc = _score_norm(ps)
            candidates.append((sc, ps, pn, f'spec_a{alpha}'))
        candidates.append((_score_norm(parent), parent.copy(), pn, 'parent'))

        # Interpolated to different lengths
        for n in [512, 1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, f'parent_n{n}'))
                    # Also spectral shape the interpolated
                    ps = _spectral_shape(pi, 0.85)
                    sc2 = _score_norm(ps)
                    candidates.append((sc2, ps, n, f'parent_spec_n{n}'))

    # Diverse profiles
    for n in [512, 1024, 2048]:
        profiles = [
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin1.5', _make_profile_linear(n, 1.5)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('cubic1', _make_profile_cubic(n, 1.5, -1.0)),
            ('cubic2', _make_profile_cubic(n, 2.0, -1.5)),
            ('cos1', _make_profile_cosine(n, 1, 0.8)),
            ('cos2', _make_profile_cosine(n, 2, 0.6)),
            ('rc05', _make_profile_risen_cosine(n, 0.5)),
            ('rc08', _make_profile_risen_cosine(n, 0.8)),
        ]
        for label, p in profiles:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: LP refinement on top candidates (900s) ---
    phase2_end = t_start + 960.0
    n_cands_to_polish = min(6, len(candidates))
    for i in range(n_cands_to_polish):
        if time.monotonic() > phase2_end:
            break
        sc, p, n, label = candidates[i]
        if sc > best_score * 1.1:
            break
        per = min(150.0, phase2_end - time.monotonic() - 20)
        if per < 30:
            break
        print(f"  LP refine [{label}] n={n} score={sc:.10f} budget={per:.0f}s")
        p_b, sc_b, nsolv = _lp_refine(p, time.monotonic() + per, t_init=0.5,
                                        max_solves=150, perturb_every=30, rng=rng)
        _update(p_b, sc_b, f'LP_{label}_n{n}_{nsolv}s')

    print(f"After Phase 2: best={best_score:.10f}")

    # --- Phase 3: Gradient polish on best (300s) ---
    phase3_end = t_start + 1260.0
    if best_seq is not None and time.monotonic() < phase3_end:
        per = min(300.0, phase3_end - time.monotonic() - 10)
        if per > 20:
            print(f"  Adam polish n={best_seq.size} budget={per:.0f}s")
            p_ap, sc_ap = _adam_logit(best_seq, time.monotonic() + per, rng,
                                       lr=0.04, beta_frac=0.06, beta_decay=0.94,
                                       no_imp_limit=400)
            _update(p_ap, sc_ap, 'adam_polish')

            # Second LP pass after gradient
            if time.monotonic() < phase3_end - 30 and best_seq.size <= 2048:
                per2 = min(100.0, phase3_end - time.monotonic() - 10)
                if per2 > 20:
                    p_lp2, sc_lp2, _ = _lp_refine(best_seq, time.monotonic() + per2,
                                                    t_init=0.3, max_solves=80,
                                                    perturb_every=25, rng=rng)
                    _update(p_lp2, sc_lp2, 'LP_after_adam')

    # --- Phase 4: Multi-scale escalation (200s) ---
    phase4_end = t_start + 1460.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            # LP at new scale
            per = min(120.0, phase4_end - time.monotonic() - 30)
            if per > 30:
                p_e, sc_e, _ = _lp_refine(p_int, time.monotonic() + per, t_init=0.4,
                                           max_solves=80, perturb_every=30, rng=rng)
                _update(p_e, sc_e, f'escalate_n{tn}')

                # Gradient at new scale
                if time.monotonic() < phase4_end - 20:
                    per_g = min(60.0, phase4_end - time.monotonic() - 10)
                    if per_g > 15:
                        p_g, sc_g = _adam_logit(p_e, time.monotonic() + per_g, rng,
                                                 lr=0.03, beta_frac=0.05,
                                                 beta_decay=0.94, no_imp_limit=200)
                        _update(p_g, sc_g, f'adam_n{tn}')

    # --- Phase 5: Multi-restart with spectral diversity (150s) ---
    phase5_end = deadline - 30.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 3
        per_r = max(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts + 1))

        # Spectral perturbations
        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 10)
            if per < 15:
                break

            # Create diverse starting point via spectral shaping + noise
            alpha = 0.8 + 0.15 * i
            p_div = _spectral_shape(best_seq, alpha)
            # Add log-space noise
            z = np.log(p_div)
            z += 0.06 * rng.normal(size=n_best)
            p_s = _softmax_from_z(z)

            # Quick LP
            p_r, sc_r, _ = _lp_refine(p_s, time.monotonic() + per * 0.6,
                                       t_init=0.4, max_solves=50,
                                       perturb_every=20, rng=rng)
            _update(p_r, sc_r, f'restart_spec{i}_n{n_best}')

            # Gradient finish
            if time.monotonic() < phase5_end - 5:
                per_g = min(20.0, phase5_end - time.monotonic() - 5)
                if per_g > 10:
                    p_rg, sc_rg = _adam_logit(p_r, time.monotonic() + per_g, rng,
                                               lr=0.03, beta_frac=0.05,
                                               beta_decay=0.94, no_imp_limit=100)
                    _update(p_rg, sc_rg, f'restart_adam{i}')

    # --- Phase 6: Final intensive LP + polish (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        per = deadline - time.monotonic() - 5
        if per > 30 and best_seq.size <= 2048:
            p_f, sc_f, _ = _lp_refine(best_seq, time.monotonic() + per * 0.6,
                                       t_init=0.3, max_solves=100,
                                       perturb_every=25, rng=rng)
            _update(p_f, sc_f, 'final_LP')

        if time.monotonic() < deadline - 5:
            per_g = deadline - time.monotonic() - 5
            if per_g > 10:
                p_fg, sc_fg = _adam_logit(best_seq, time.monotonic() + per_g, rng,
                                           lr=0.015, beta_frac=0.04,
                                           beta_decay=0.95, no_imp_limit=100)
                _update(p_fg, sc_fg, 'final_adam')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512")
    return [float(x) for x in fb]
