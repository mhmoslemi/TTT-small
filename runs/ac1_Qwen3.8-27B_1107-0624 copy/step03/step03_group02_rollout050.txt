import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_opt(a, deadline, max_iter=50000, lr=0.04, beta_factor=0.06,
              beta_decay=0.75, beta_iter=100, momentum=0.9):
    """Adam in logit space with aggressive beta annealing and restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    momentum_vec = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            momentum_vec = np.zeros(n)
            adam_t = 0
            no_imp += 100
            continue

        # Nesterov-like momentum
        momentum_vec = momentum * momentum_vec + (1.0 - momentum) * grad_z

        adam_t += 1
        m = b1 * m + (1.0 - b1) * momentum_vec
        v_arr = b2 * v_arr + (1.0 - b2) * (momentum_vec * momentum_vec)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        z = z_new
        a = a_new

        # Beta annealing
        if it % beta_iter == 0:
            beta = max(1e-12, beta * beta_decay)

        # Learning rate decay
        if no_imp % 200 == 0 and no_imp > 0:
            lr_cur = max(0.002, lr_cur * 0.92)

        # Aggressive restart
        restart_thresh = max(3000, 4 * n)
        if no_imp > restart_thresh:
            rng_r = np.random.default_rng(50000 + it * 7919)
            pert_scale = min(0.5, 0.1 + 0.1 * (no_imp // restart_thresh))
            z = best_z + pert_scale * rng_r.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            momentum_vec = np.zeros(n)
            adam_t = 0
            lr_cur = base_lr
            no_imp = 0

    return best_a, best_score


def _coord_descent_polish(a, deadline, max_sweeps=500):
    """Coordinate descent: for each coordinate, try scaling it up/down."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    scales = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize order
        idx = np.random.default_rng(sweep * 1000).permutation(n)
        for i in idx:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-15:
                # Try adding a small amount
                a[i] = 1e-6
                s2 = a.sum()
                a2 = a / s2
                sc2 = float(2.0 * n * _fft_convolve(a2).max())
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a2.copy()
                    a = a2.copy()
                    improved = True
                else:
                    a[i] = orig
                continue

            for scale in scales:
                new_val = orig * scale
                a[i] = new_val
                s2 = a.sum()
                a2 = a / s2
                sc2 = float(2.0 * n * _fft_convolve(a2).max())
                if sc2 < best_score - 1e-10:
                    best_score = sc2
                    best_a = a2.copy()
                    a = a2.copy()
                    improved = True
                    break
            a[i] = orig  # restore for next scale try if not improved

        if not improved and sweep > 10:
            break

    return best_a


def _subgradient_opt(a, deadline, max_iter=10000, top_k=3):
    """Multi-peak subgradient method."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.25 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        weights = np.array([1.0 / (1.0 + 0.5 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            idx = int(idx)
            # Subgradient at max index: 2*a reversed appropriately
            for j in range(n):
                diff = idx - j
                if 0 <= diff < n:
                    sg[j] += 2.0 * weights[i] * a[diff]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _make_hann(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_blackman(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabolic(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * (1.0 - u * u)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, sigma=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * sigma ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p=1.5):
    """Symmetric power-law profile: a_i ~ (min(i, n-1-i) + 1)^(-p)"""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1.0
    a = d ** (-p)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

            p_interp = _interpolate_profile(parent, n)
            all_inits[n].insert(0, p_interp)
            # Perturbed parent
            p_pert = p_interp * (1.0 + 0.1 * rng.normal(size=n))
            p_pert = np.maximum(p_pert, 0.0)
            all_inits[n].append(p_pert / p_pert.sum())

    # Process each length
    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break

        n_budget = time_alloc.get(n, 200)
        n_deadline = min(deadline - 60, time.monotonic() + n_budget)

        inits = all_inits[n]
        # Limit to at most 3 inits per length if time is tight
        max_inits = 3 if n >= 2048 else 4
        inits = inits[:max_inits]

        per_init_time = n_budget / len(inits)

        print(f"=== n={n}, {len(inits)} inits, budget={n_budget:.0f}s ===")

        for ii, a0 in enumerate(inits):
            if time.monotonic() > n_deadline - 30:
                break
            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            sc0 = float(2.0 * n * _fft_convolve(a0).max())
            print(f"  n={n} init {ii}: score={sc0:.10f}")

            a_res, s_res = _optimize_single(a0, init_deadline, n)
            if s_res < best_score:
                best_score = s_res
                best_seq = a_res.copy()
                print(f"  n={n} init {ii} -> {s_res:.10f} (NEW BEST)")
            else:
                print(f"  n={n} init {ii} -> {s_res:.10f}")

    # Final polish on the best solution
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        print(f"=== Final polish: n={n_best}, current best={best_score:.10f} ===")

        # Adam fine-tune
        d_final = min(deadline - 15, time.monotonic() + (deadline - 15 - time.monotonic()) * 0.5)
        a_f1, s_f1 = _adam_opt(best_seq, d_final, max_iter=20000, lr=0.008,
                                beta_factor=0.015, beta_decay=0.88, beta_iter=150)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
            print(f"  Final Adam: {best_score:.10f}")

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            d_cd = min(deadline - 10, time.monotonic() + (deadline - 10 - time.monotonic()) * 0.5)
            a_cd = _coord_descent_polish(best_seq, d_cd, max_sweeps=300)
            s_cd = float(2.0 * n_best * _fft_convolve(a_cd).max())
            if s_cd < best_score:
                best_score = s_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.10f}")

        # One more tiny Adam
        if time.monotonic() < deadline - 5:
            a_f2, s_f2 = _adam_opt(best_seq, deadline - 5, max_iter=10000, lr=0.003,
                                    beta_factor=0.005, beta_decay=0.92, beta_iter=300)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
                print(f"  Ultra-final Adam: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_hann(512)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_hann(256)

    # Validate with evaluator
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
