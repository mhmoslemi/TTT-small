import numpy as np
import time
from scipy.optimize import linprog, minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-12:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

    c_obj[-1] = 1.0
    
    A = np.zeros((2 * n, n + 1))
    b_ub = np.zeros(2 * n)
    
    conv_mat = _make_conv_matrix(a)
    A[:2*n-1, :n] = conv_mat
    A[:2*n-1, n] = -1.0
    
    A[2*n-1, :n] = -1.0
    b_ub[2*n-1] = -1.0

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        M_opt = res.x[n]

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    if mn < -1e-10:

    return c

def _lp_min_peak(a):
    """Solve: min M s.t. conv(a,b)[m] <= M for all m, sum(b) >= 1, b >= 0.
    Returns (b_opt, M_opt) or (None, None) on failure."""
    n = len(a)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A = np.zeros((2 * n, n + 1))
    b_ub = np.zeros(2 * n)

    conv_mat = _make_conv_matrix(a)
    A[:2*n-1, :n] = conv_mat
    A[:2*n-1, n] = -1.0

    A[2*n-1, :n] = -1.0
    b_ub[2*n-1] = -1.0

    bounds = [(0, None)] * n + [(0, 100.0)]

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None


def _lp_min_peak_symmetric(a):
    """Same as _lp_min_peak but with warm-start-like bounds."""
    n = len(a)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A = np.zeros((2 * n, n + 1))
    b_ub = np.zeros(2 * n)

    conv_mat = _make_conv_matrix(a)
    A[:2*n-1, :n] = conv_mat
    A[:2*n-1, n] = -1.0

    A[2*n-1, :n] = -1.0
    b_ub[2*n-1] = -1.0

    current_max = float(_fft_conv(a).max())
    bounds = [(0, None)] * n + [(0, current_max * 2 + 1)]

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None


def _compute_softmax_grad(a, beta):
    """Compute value and gradient of soft-max objective in log-space.
    Returns (value, gradient_z) where gradient is w.r.t. z (log-params)."""
    n = len(a)
    c = _fft_conv(a)
    c_max = float(c.max())
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        return c_max, np.zeros(n)
    val = c_max + beta * float(np.log(s))
    w = ex / s

    # h[j] = sum_m w[m] * a[m-j], for j = 0,...,n-1
    # h = conv(w, a_rev)[n-1 : 2n-1] where a_rev = a[::-1]
    N = _next_pow2(3 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]

    wc = float(np.dot(w, c))
    # df/da[j] = 2*h[j]
    # df/dz[j] = a[j] * (df/da[j] - sum_k a[k]*df/da[k])
    # sum_k a[k]*df/da[k] = 2*sum_k a[k]*h[k] = 2*wc
    gz = 2.0 * a * (h - wc)
    return val, gz


def _adam_logit_optimize(a, deadline, max_iter=100000, lr=0.03, beta_init=None):
    """Adam optimization in log-space with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_conv(a)
    c_max = float(c.max())
    beta = beta_init if beta_init is not None else max(1e-6, 0.1 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, gz = _compute_softmax_grad(a, beta)

        if not np.all(np.isfinite(gz)):
            break

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - base_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_conv(a)
        c_max = float(c.max())
        score = float(2.0 * n * c_max)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.95)

        if it % 200 == 0:
            top = np.sort(c)[::-1][:5]
            if len(top) > 1:
                gap = top[0] - top[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 10:
                    beta = min(0.5 * c_max, beta * 1.3)

        if no_imp > max(5000, 10 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            z = z + 0.2 * rng.standard_normal(n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            e = np.exp(z)
            a = e / e.sum()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            no_imp = 0
            beta = max(1e-6, 0.1 * float(_fft_conv(a).max()))

        if it % 2000 == 0 and np.log(best_score) == np.log(float(2.0 * n * _fft_conv(a).max())):
            if base_lr < 0.003:
                break

    return best_a


def _lp_alternating(a, deadline, max_iter=5000):
    """LP alternating method: iterate a -> LP(a) -> LP(LP(a)) -> ...
    At each step, solve min M s.t. conv(a, b) <= M, sum(b)>=1, b>=0,
    then do line search along the direction."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_conv(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    blend = 0.5

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Solve LP with current a as left factor
        b_star, M_opt = _lp_min_peak(a)
        if b_star is None:
            break

        sum_b = float(b_star.sum())
        if sum_b < 1e-10:
            break
        b_star = b_star / sum_b

        # Line search along a -> b_star
        best_t = 0.0
        best_local_score = best_score
        for t_cand in [blend, 0.7, 0.4, 0.2, 0.1, 0.3, 0.6, 0.8, 0.5]:
            a_trial = (1.0 - t_cand) * a + t_cand * b_star
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc = _score(a_trial)
            if sc < best_local_score:
                best_local_score = sc
                best_t = t_cand

        if best_t == 0.0 or best_local_score >= best_score - 1e-14:
            no_imp += 1
            if no_imp % 20 == 0:
                blend = max(0.1, blend * 0.9)
            if no_imp > 200:
                # Try symmetric LP
                b_star2, M_opt2 = _lp_min_peak(b_star)
                if b_star2 is not None:
                    sum_b2 = float(b_star2.sum())
                    if sum_b2 > 1e-10:
                        b_star2 = b_star2 / sum_b2
                        for t_cand in [0.3, 0.5, 0.7]:
                            a_trial = (1.0 - t_cand) * a + t_cand * b_star2
                            a_trial = np.maximum(a_trial, 0.0)
                            s = a_trial.sum()
                            if s > 1e-10:
                                a_trial = a_trial / s
                                sc = _score(a_trial)
                                if sc < best_score:
                                    best_score = sc
                                    best_a = a_trial.copy()
                                    a = a_trial.copy()
                                    no_imp = 0
                                    break
                if no_imp > 500:
                    break
            continue

        a = (1.0 - best_t) * best_a + best_t * b_star
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        best_score = best_local_score
        best_a = a.copy()
        no_imp = 0
        blend = min(0.8, blend * 1.05)

    return best_a


def _gradient_polish(a, deadline, max_iter=50000):
    """Fine gradient polish with very small beta and L-BFGS-like steps."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_conv(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.02 * c_max)

    z = np.log(a)
    z -= np.mean(z)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    lr = 0.02
    no_imp = 0
    prev_grad = None
    prev_z = z.copy()
    bb_scale = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, gz = _compute_softmax_grad(a, beta)
        if not np.all(np.isfinite(gz)):
            break

        # BB step size
        if prev_grad is not None:
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20 and abs(dot_sg) > 1e-20:
                bb_cand = dot_ss / abs(dot_sg)
                bb_cand = np.clip(bb_cand, 0.2, 5.0)
                bb_scale = 0.6 * bb_scale + 0.4 * bb_cand
        prev_z = z.copy()
        prev_grad = gz.copy()

        z = z - lr * bb_scale * gz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_conv(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.93)

        if it % 500 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(3000, 5 * n):
            break

    return best_a


def _peak_targeted_polish(a, deadline, max_iter=10000):
    """Target the specific peak location for direct optimization."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_conv(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Gradient of c[peak_idx] w.r.t. a
        # dc[m]/da[j] = 2*a[m-j] (if 0 <= m-j < n)
        m = peak_idx
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        grad = np.zeros(n)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[m - j]

        # Project gradient to simplex tangent: remove mean component weighted by a
        correction = float(np.dot(a, grad))
        grad_tangent = grad - correction * np.ones(n)

        # Line search
        a_new = a - 0.01 * grad_tangent
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            # Try smaller step
            a_new = a - 0.003 * grad_tangent
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_score - 1e-14:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 500:
                    break

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp_init(n, curvature=1.0):
    """Create a ramp initialization with adjustable curvature."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.1 + 0.9 * t ** curvature
    return a / a.sum()


def _make_flat_spectrum_init(n, rng):
    """Initialize from flat power spectrum."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = abs(X[0])
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.abs(np.real(np.fft.ifft(X)))[:n]
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _multi_start_refine(a, deadline, n_starts=6, rng=None):
    """Try multiple perturbations and keep the best."""
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.03 * (1 + i * 0.5)
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.3:
            continue

        sub_deadline = min(deadline, time.monotonic() + 40.0)
        a_ref = _adam_logit_optimize(a_pert, sub_deadline, max_iter=3000, lr=0.02)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    1. Diversified initialization
    2. LP alternating (main workhorse - direct peak minimization)
    3. Gradient polish (Adam in log-space)
    4. Peak-targeted direct optimization
    5. Multi-start refinement
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine working length
    if parent is not None:
        n_work = parent.size
    else:
        n_work = 768

    # Phase 1: Generate diverse initializations
    print(f"Phase 1: Initialization (n={n_work})", flush=True)
    init_pool = []

    if parent is not None:
        init_pool.append(('parent', _interpolate_profile(parent, n_work)))

    # Ramp initializations with different curvatures
    for curv in [0.7, 1.0, 1.3, 1.5, 2.0]:
        init_pool.append((f'ramp_{curv}', _make_ramp_init(n_work, curv)))

    # Flat spectrum
    for _ in range(3):
        init_pool.append(('flat_spec', _make_flat_spectrum_init(n_work, rng)))

    # Evaluate all
    init_best = None
    init_best_score = np.inf
    for name, a0 in init_pool:
        a0 = np.maximum(a0, 1e-15)
        a0 = a0 / a0.sum()
        sc = _score(a0)
        if sc < init_best_score:
            init_best_score = sc
            init_best = a0.copy()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a0.copy()

    print(f"  Best init: {init_best_score:.8f}", flush=True)

    if init_best is None:
        init_best = _make_ramp_init(n_work, 1.0)

    # Phase 2: LP alternating (main workhorse)
    lp_deadline = min(deadline - 300, t_start + 600)
    print(f"Phase 2: LP alternating, budget={lp_deadline - time.monotonic():.0f}s", flush=True)
    a_lp = _lp_alternating(init_best, lp_deadline, max_iter=10000)
    sc_lp = _score(a_lp)
    print(f"  LP result: {sc_lp:.10f}", flush=True)
    if sc_lp < best_score:
        best_score = sc_lp
        best_seq = a_lp.copy()

    # Phase 3: Gradient polish on the LP result
    gp_deadline = min(deadline - 200, time.monotonic() + 400)
    print(f"Phase 3: Gradient polish, budget={gp_deadline - time.monotonic():.0f}s", flush=True)
    a_grad = _adam_logit_optimize(a_lp, gp_deadline, max_iter=80000, lr=0.025)
    sc_grad = _score(a_grad)
    print(f"  Gradient result: {sc_grad:.10f}", flush=True)
    if sc_grad < best_score:
        best_score = sc_grad
        best_seq = a_grad.copy()

    # Phase 4: Second LP pass on gradient result
    if time.monotonic() < deadline - 150:
        lp2_deadline = min(deadline - 100, time.monotonic() + 300)
        print(f"Phase 4: LP pass 2, budget={lp2_deadline - time.monotonic():.0f}s", flush=True)
        a_lp2 = _lp_alternating(best_seq, lp2_deadline, max_iter=8000)
        sc_lp2 = _score(a_lp2)
        print(f"  LP2 result: {sc_lp2:.10f}", flush=True)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()

    # Phase 5: Peak-targeted polish
    if time.monotonic() < deadline - 80:
        pt_deadline = min(deadline - 60, time.monotonic() + 150)
        print(f"Phase 5: Peak-targeted polish, budget={pt_deadline - time.monotonic():.0f}s", flush=True)
        a_pt = _peak_targeted_polish(best_seq, pt_deadline, max_iter=10000)
        sc_pt = _score(a_pt)
        print(f"  Peak-targeted: {sc_pt:.10f}", flush=True)
        if sc_pt < best_score:
            best_score = sc_pt
            best_seq = a_pt.copy()

    # Phase 6: Fine gradient with tiny beta
    if time.monotonic() < deadline - 40:
        fine_deadline = min(deadline - 25, time.monotonic() + 100)
        print(f"Phase 6: Fine polish, budget={fine_deadline - time.monotonic():.0f}s", flush=True)
        a_fine = _gradient_polish(best_seq, fine_deadline, max_iter=30000)
        sc_fine = _score(a_fine)
        print(f"  Fine polish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Phase 7: Multi-start
    if time.monotonic() < deadline - 20:
        ms_deadline = deadline - 15
        print(f"Phase 7: Multi-start, budget={ms_deadline - time.monotonic():.0f}s", flush=True)
        a_ms = _multi_start_refine(best_seq, ms_deadline, n_starts=4, rng=rng)
        sc_ms = _score(a_ms)
