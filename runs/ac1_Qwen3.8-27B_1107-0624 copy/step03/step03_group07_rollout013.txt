The outer problem is max_w (result of inner QP), which is a concave maximization (since the value function of a QP with respect to the objective coefficients is concave).

So we could use a subgradient method on the outer problem:
1. Given w, solve the inner QP: min_a a^T (sum_k w[k] Q_k) a s.t. a >= 0, sum(a) = 1

This is essentially what the soft-max approach does (the soft-max gives a smooth approximation to the max, and the weights w are the soft-max probabilities).

                beta = max(beta * 0.95, beta_min)
            else:
                beta = max(beta * beta_decay, beta_min)

        # Learning rate adaptation
        if no_improve > 150:
            current_lr = max(0.004, current_lr * 0.98)

        # Restart
        if no_improve > restart_period:
            noise_scale = 0.08 + 0.05 * min(1.0, no_improve / (2.0 * restart_period))
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_factor * 0.5 * max_c, 1e-8)
            m.fill(0.0)
            v.fill(0.0)
            vel.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0
            peak_memory = np.zeros(2 * n - 1)
            peak_memory[int(np.argmax(c))] = 1.0

    return best_a, best_score


def exact_subgradient_step(a0, z, n_peaks=1):
    """Take one step using exact subgradient at the top peaks."""
    c = full_convolve_fft(a0)
    k = c.size
    # Get top peaks
    top_idx = np.argsort(c)[-n_peaks:]
    w = np.zeros(k)
    for idx in top_idx:
        w[idx] = 1.0 / n_peaks
    h = compute_h(w, a0)
    grad_a = 2.0 * h
    dot = float(np.dot(a0, grad_a))
    dz = a0 * (grad_a - dot)
    # Small step
    step_size = 0.02
    z = z - step_size * dz
    z = np.clip(z, -50.0, 50.0)
    z = z - z.mean()
    a0 = np.exp(z - z.max())
    a0 = a0 / a0.sum()
    return a0, z


def lowdim_sweep(a_base, n, rng, budget_s):
    """
    Try low-dimensional parameterizations centered on the base sequence.
    a[i] = a_base[i] * (1 + sum_k c_k * phi_k(i/n))
    where phi_k are sine/cosine basis functions.
    """
    t_start = time.monotonic()
    a_base = np.asarray(a_base, dtype=np.float64).ravel()
    a_base = np.maximum(a_base, 1e-15)
    a_base = a_base / a_base.sum()

    best_a = a_base.copy()
    best_score = compute_score(a_base)

    t = np.linspace(0.0, 1.0, n)
    # Basis functions
    basis = []
    for k in [1, 2, 3, 4, 5]:
        basis.append(np.cos(2 * np.pi * k * t - np.pi / 2))
        basis.append(np.sin(2 * np.pi * k * t - np.pi / 2))
    basis = np.array(basis)  # shape (n_basis, n)
    n_basis = basis.shape[0]

    # Coordinate descent on coefficients
    coeffs = np.zeros(n_basis)
    for _ in range(20):  # Outer iterations
        for j in range(n_basis):
            if time.monotonic() > t_start + budget_s:
                return best_a, best_score
            # Try a few values for coefficient j
            best_cj = coeffs[j]
            best_sc = best_score
            for cj_trial in np.linspace(-0.15, 0.15, 7):
                if j == int(best_cj * 100 + 3):
                    continue
                coeffs[j] = cj_trial
                a_trial = a_base * (1.0 + basis @ coeffs)
                a_trial = np.maximum(a_trial, 1e-12)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_cj = cj_trial
            coeffs[j] = best_cj

        # Update best
        a_trial = a_base * (1.0 + basis @ coeffs)
        a_trial = np.maximum(a_trial, 1e-12)
        a_trial = a_trial / a_trial.sum()
        sc = compute_score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()

    return best_a, best_score


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine profiles
    for k in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Quadratic
    for shift in [0.3, 0.5, 0.7]:
        a = 1.0 + 2.0 * (t - shift) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Random smooth
    for _ in range(8):
        coeffs = rng.uniform(-0.3, 0.3, 6)
        a = np.ones(n)
        for i, c_val in enumerate(coeffs, 1):
            a = a + c_val * np.cos(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search combining adaptive gradient optimization,
    exact subgradient steps, low-dimensional sweeps, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    n_init = init_seq.size if init_seq is not None else 2048

    # ===== Phase 1: Quick refinement at initial n (200s) =====
    phase1_deadline = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: Quick refinement at n={n_init} ---")

    if init_seq is not None:
        a1 = init_seq.copy()
    else:
        a1 = diverse_profiles(n_init, rng)[2]  # Moderate ramp

    # Run adaptive beta optimization
    dl1 = min(phase1_deadline, time.monotonic() + 180.0)
    r1, sc1 = adaptive_beta_optimize(a1, dl1, rng, lr=0.04, beta_factor=0.08,
                                     beta_decay=0.96, restart_period=500,
                                     momentum_type="adam")
    sc1f = compute_score(r1)
    update(r1, sc1f)
    print(f"  After Adam: {sc1f:.10f}")

    # Exact subgradient steps
    if time.monotonic() < phase1_deadline - 5:
        z1 = np.log(np.maximum(r1, 1e-15))
        z1 = z1 - z1.mean()
        for pt_n in [1, 2, 3, 5]:
            if time.monotonic() > phase1_deadline - 2:
                break
            for _ in range(20):
                r1, z1 = exact_subgradient_step(r1, z1, n_peaks=pt_n)
            sc_pt = compute_score(r1)
            update(r1, sc_pt)
            if pt_n == 1:
                print(f"  After exact subgrad (1 peak, 20 steps): {sc_pt:.10f}")

    # Nesterov variant
    if time.monotonic() < phase1_deadline - 10:
        a1b = best_seq.copy() if best_seq is not None else r1.copy()
        a1b = a1b + 0.01 * rng.normal(size=n_init)
        a1b = np.maximum(a1b, 1e-12)
        a1b = a1b / a1b.sum()
        dl1b = min(phase1_deadline, time.monotonic() + 15.0)
        r1b, sc1b = adaptive_beta_optimize(a1b, dl1b, rng, lr=0.03,
                                           beta_factor=0.06, beta_decay=0.95,
                                           momentum_type="nesterov")
        sc1bf = compute_score(r1b)
        update(r1b, sc1bf)
        print(f"  After Nesterov: {sc1bf:.10f}")

    print(f"  Phase 1 best: {best_score:.10f}")

    # ===== Phase 2: Try different n values (150s) =====
    phase2_deadline = min(deadline, t_start + 350.0)
    print(f"\n--- Phase 2: Try different n values ---")

    n_candidates = [n_init]
    # Add nearby values
    for dn in [-256, -128, 128, 256, 512]:
        n_try = n_init + dn
        if 128 <= n_try <= 8192 and n_try != n_init:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > phase2_deadline - 5:
            break
        if n_try == n_init:
            p_try = best_seq.copy()
        else:
            p_try = interpolate_profile(best_seq, n_try)

        sc_try = compute_score(p_try)
        update(p_try, sc_try)

        if sc_try > best_score + 0.02:
            continue

        # Short optimization
        dl = min(phase2_deadline, time.monotonic() + 30.0)
        r_try, _ = adaptive_beta_optimize(p_try, dl, rng, lr=0.03,
                                          beta_factor=0.06, beta_decay=0.95,
                                          restart_period=300)
        sc_try_f = compute_score(r_try)
        update(r_try, sc_try_f)
        print(f"  n={n_try}: {sc_try_f:.10f}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Main long optimization (1100s) =====
    n_main = best_seq.size
    phase3_deadline = min(deadline, t_start + 1450.0)
    print(f"\n--- Phase 3: Main optimization at n={n_main} ---")

    # Run multiple optimization rounds with different hyperparameters
    configs = [
        {"lr": 0.035, "beta_factor": 0.07, "beta_decay": 0.965, "momentum_type": "adam", "restart_period": 800},
        {"lr": 0.025, "beta_factor": 0.05, "beta_decay": 0.97, "momentum_type": "adam", "restart_period": 1000},
        {"lr": 0.04, "beta_factor": 0.09, "beta_decay": 0.955, "momentum_type": "nesterov", "restart_period": 600},
        {"lr": 0.02, "beta_factor": 0.04, "beta_decay": 0.975, "momentum_type": "adam", "restart_period": 1200},
    ]

    time_per_config = 250.0
    for cfg_idx, cfg in enumerate(configs):
        if time.monotonic() > phase3_deadline - 30:
            break

        # Perturb from best
        a_start = best_seq.copy()
        noise = 0.005 * rng.normal(size=n_main)
        a_start = a_start + noise
        a_start = np.maximum(a_start, 1e-12)
        a_start = a_start / a_start.sum()

        dl = min(phase3_deadline, time.monotonic() + time_per_config)
        r, sc_r = adaptive_beta_optimize(a_start, dl, rng, **cfg)
        sc_r = compute_score(r)
        update(r, sc_r)
        print(f"  Config {cfg_idx} ({cfg['momentum_type']}, lr={cfg['lr']}): {sc_r:.10f}")

    # Low-dimensional sweep on best
    if time.monotonic() < phase3_deadline - 20:
        dl_ld = min(phase3_deadline, time.monotonic() + 40.0)
        r_ld, sc_ld = lowdim_sweep(best_seq, n_main, rng, dl_ld - time.monotonic())
        sc_ld = compute_score(r_ld)
        update(r_ld, sc_ld)
        print(f"  After lowdim sweep: {sc_ld:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Intensive polish (300s) =====
    phase4_deadline = deadline
    n_polish = best_seq.size
    print(f"\n--- Phase 4: Intensive polish at n={n_polish} ---")

    # Very fine-grained optimization
    for round_idx, (lr, bf, bd) in enumerate([
        (0.015, 0.03, 0.98),
        (0.01, 0.025, 0.985),
        (0.008, 0.02, 0.99),
        (0.012, 0.035, 0.975),
        (0.006, 0.015, 0.99),
    ]):
        if time.monotonic() > phase4_deadline - 5:
            break
        a_p = best_seq.copy()
        a_p = a_p + 0.002 * rng.normal(size=n_polish)
        a_p = np.maximum(a_p, 1e-12)
        a_p = a_p / a_p.sum()

        dl = min(phase4_deadline, time.monotonic() + 50.0)
        r_p, _ = adaptive_beta_optimize(a_p, dl, rng, lr=lr, beta_factor=bf,
                                        beta_decay=bd, restart_period=400,
                                        momentum_type="adam")
        sc_p = compute_score(r_p)
        update(r_p, sc_p)
        print(f"  Polish {round_idx} (lr={lr}): {sc_p:.10f}")

    # Final exact subgradient steps
    if time.monotonic() < phase4_deadline - 10:
        a_f = best_seq.copy()
        z_f = np.log(np.maximum(a_f, 1e-15))
        z_f = z_f - z_f.mean()
        for pt_n in [1, 2, 3]:
            if time.monotonic() > phase4_deadline - 3:
                break
            for _ in range(50):
                a_f, z_f = exact_subgradient_step(a_f, z_f, n_peaks=pt_n)
            sc_f = compute_score(a_f)
            update(a_f, sc_f)
        print(f"  After final exact subgrad: {best_score:.10f}")

    # ===== Phase 5: Quick try of alternative n =====
    if time.monotonic() < deadline - 30:
        for n_alt in [n_polish + 256, n_polish - 256, n_polish + 512]:
            if n_alt < 128 or n_alt > 8192:
                continue
            if time.monotonic() > deadline - 10:
                break
            p_alt = interpolate_profile(best_seq, n_alt)
            sc_alt = compute_score(p_alt)
            if sc_alt < best_score + 0.005:
                dl_alt = min(deadline, time.monotonic() + 20.0)
                r_alt, _ = adaptive_beta_optimize(p_alt, dl_alt, rng, lr=0.02,
                                                  beta_factor=0.05, beta_decay=0.97)
                sc_alt_f = compute_score(r_alt)
                update(r_alt, sc_alt_f)
                print(f"  Alt n={n_alt}: {sc_alt_f:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    # ===== Final validation and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        # Fallback: generate a reasonable sequence
        n_fb = 512
        t_fb = np.linspace(0, 1, n_fb)
        final_seq = 1.0 + 1.5 * (0.5 - t_fb)
        final_seq = np.maximum(final_seq, 1e-12)
        final_seq = final_seq / final_seq.sum()

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        n_fb = 256
        t_fb = np.linspace(0, 1, n_fb)
        final_seq = (1.0 + 1.5 * (0.5 - t_fb))
        final_seq = np.maximum(final_seq, 1e-12)
        final_seq = final_seq / final_seq.sum()

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
