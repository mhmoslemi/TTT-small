import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-10:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_sparse_conv_matrix(a):
    """Build sparse convolution matrix for LP: A[g] = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if a[j] > 1e-15:
            idx = np.arange(j, j + n)
            rows.extend(idx)
            cols.extend([j] * n)
            vals.extend([a[j]] * n)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_sparse_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success:
        return res.x
    return None


def _gradient_topk(a, beta, top_k=8):
    """Compute gradient with top-k peak focusing plus smooth background."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    # Smooth weights for background
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w_bg = np.zeros_like(c)
        w_bg[int(np.argmax(c))] = 1.0
    else:
        w_bg = ex / s
    # Top-k peak weights
    sorted_idx = np.argsort(c)[::-1]
    k = min(top_k, len(c))
    w_peak = np.zeros_like(c)
    peak_vals = c[sorted_idx[:k]]
    pv_sum = float(peak_vals.sum())
    if pv_sum > 0:

    # Blend

    w = w / w.sum()
    # Gradient: grad_j = 2 * sum_k w_k * a_reversed contribution

    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c

    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta_init = max(1e-8, 0.03 * float(c.max()))
    beta_final = max(1e-10, 0.001 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        # Cosine annealing on beta
        frac = it / max(1, max_iter - 1)
        beta = beta_final + 0.5 * (beta_init - beta_final) * (1 + np.cos(np.pi * frac))

        grad, maxc, c_arr = _gradient_topk(p, beta, top_k=top_k + it // 2000)
        # Add small regularization to prevent collapse
        vp = grad + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z - z.max())
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, base_lr * (0.97 ** (no_imp // 500)))

        if no_imp > max(3000, 5 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n) * 0.3
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z - z.max())
            a = e / e.sum()
            c = _fft_convolve(a)
            beta_init = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _peak_peel(a, deadline, max_iter=5000):
    """Specifically target individual peaks by solving smaller LPs."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find the lag that achieves the peak
        # conv(a,a)[peak_idx] = sum_{i=max(0,peak_idx-n+1)}^{min(n-1,peak_idx)} a[i]*a[peak_idx-i]
        # We want to reduce this specific peak
        # Solve: max sum(g) s.t. conv(a,g)[peak_idx] <= M*(1-eps), conv(a,g) <= M*(1+delta), g >= 0

        eps = 0.01
        delta = 0.05
        M_target = M * (1.0 - eps)
        M_other = M * (1.0 + delta)

        # Build constraint matrix
        n = len(a)
        c_obj = -np.ones(n)
        A_rows = []
        b_vec = []

        # Constraint for the peak lag
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)
        row = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            row[i] = a[peak_idx - i]
        A_rows.append(row)
        b_vec.append(M_target)

        # Add a few other top peaks as constraints
        sorted_peaks = np.argsort(c)[::-1]
        for k in range(1, min(5, len(c))):
            pi = int(sorted_peaks[k])
            if pi == peak_idx:
                continue
            i_lo2 = max(0, pi - n + 1)
            i_hi2 = min(n - 1, pi)
            row2 = np.zeros(n)
            for i in range(i_lo2, i_hi2 + 1):
                row2[i] = a[pi - i]
            A_rows.append(row2)
            b_vec.append(M_other)

        if A_rows:
            A_mat = np.array(A_rows)
            res = linprog(c_obj, A_ub=A_mat, b_ub=np.array(b_vec),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 5})
            if res.success:
                g = res.x
                sum_g = float(g.sum())
                if sum_g > 1.0 + 1e-6:
                    g = g / sum_g
                    for t_blend in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1 - t_blend) * a + t_blend * g
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        sc = _score(a_try)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_a = a_try.copy()
                            a = a_try.copy()
                            no_imp = 0
                            break
                    else:
                        no_imp += 1
                else:
                    no_imp += 1
            else:
                no_imp += 1
        else:
            break

        if no_imp > 500:
            break

    return best_a


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, center_sep=0.3, width=0.1):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5 + center_sep) / width) ** 2)
    a += np.exp(-0.5 * ((t - 0.5 - center_sep) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _multi_start_optimize(a_init, n, deadline, methods=('lp', 'adam', 'peel')):
    """Run multiple optimization methods sequentially, keeping the best."""
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    elapsed_at_start = time.monotonic()
    total_budget = deadline - elapsed_at_start

    # Phase 1: LP
    if 'lp' in methods:
        sub_deadline = elapsed_at_start + total_budget * 0.35
        a_res = _lp_optimize(best_a, sub_deadline)
        sc = _score(a_res)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"    LP: {sc:.10f}")

    # Phase 2: Adam
    if 'adam' in methods:
        sub_deadline = deadline - (deadline - time.monotonic()) * 0.4
        sub_deadline = min(sub_deadline, time.monotonic() + total_budget * 0.3)
        sub_deadline = max(sub_deadline, time.monotonic() + 10)
        a_res = _adam_logspace(best_a, sub_deadline, lr=0.04, top_k=8)
        sc = _score(a_res)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"    Adam: {sc:.10f}")

    # Phase 3: Peak peel
    if 'peel' in methods and time.monotonic() < deadline - 30:
        sub_deadline = deadline - (deadline - time.monotonic()) * 0.3
        sub_deadline = max(sub_deadline, time.monotonic() + 10)
        a_res = _peak_peel(best_a, sub_deadline)
        sc = _score(a_res)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"    Peel: {sc:.10f}")

    # Phase 4: Second LP pass
    if time.monotonic() < deadline - 50:
        sub_deadline = deadline - 40
        a_res = _lp_optimize(best_a, sub_deadline, max_iter=5000,
                             top_k_constraints=50)
        sc = _score(a_res)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"    LP2: {sc:.10f}")

    # Phase 5: Final Adam
    if time.monotonic() < deadline - 30:
        sub_deadline = deadline - 20
        a_res = _adam_logspace(best_a, sub_deadline, lr=0.02, top_k=12)
        sc = _score(a_res)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"    Adam2: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: LP spreading, log-space Adam with top-k
    peak focusing, peak-peeling LPs, and multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 896, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))

    print(f"Trying lengths: {lengths}")
    print(f"Time: {budget_s}s")

    # Phase 1: Multi-length search
    num_lengths = len(lengths)
    # Give more budget to lengths near the parent
    base_budget = (deadline - time.monotonic() - 200.0) / num_lengths
    per_length = {}
    for n_t in lengths:
        if parent is not None:
            dist = abs(n_t - parent.size) / max(1, parent.size)
            weight = 1.0 / (1.0 + 2.0 * dist)
        else:
            weight = 1.0
        per_length[n_t] = base_budget * weight

    # Normalize budgets
    total_w = sum(per_length.values())
    total_avail = deadline - time.monotonic() - 200.0
    for n_t in per_length:
        per_length[n_t] = per_length[n_t] / total_w * total_avail

    for n_t in lengths:
        if time.monotonic() > deadline - 180:
            break

        phase_budget = min(per_length[n_t], deadline - time.monotonic() - 150.0)
        phase_budget = max(30.0, phase_budget)

        print(f"\n=== Length {n_t}: budget={phase_budget:.0f}s ===")

        # Primary init: parent or linear
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_linear_profile(n_t, 1.5)

        sc_init = _score(a_init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()
        print(f"  Init: {sc_init:.10f}")

        a_res, sc_res = _multi_start_optimize(a_init, n_t, time.monotonic() + phase_budget)
        if sc_res < best_score:
            best_score = sc_res
            best_seq = a_res.copy()
            print(f"  ** NEW BEST n={n_t}: {sc_res:.10f}")

    # Phase 2: Diverse initializations at best length
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===")

        inits = []
        # U-shaped
        for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Bimodal
        for sep in [0.15, 0.25, 0.35, 0.45]:
            inits.append(_make_bimodal_profile(n_best, sep, 0.12))
        # Linear with various slopes
        for eta in [0.5, 1.0, 2.0, 3.0, 5.0]:
            inits.append(_make_linear_profile(n_best, eta))
        # Random structured
        for _ in range(4):
            a_r = np.maximum(rng.normal(1.0, 0.6, n_best), 0.0)
            a_r = a_r / a_r.sum()
            inits.append(a_r)
        # Perturbed best
        for _ in range(3):
            a_p = best_seq * (1.0 + 0.2 * rng.normal(size=n_best))
            a_p = np.maximum(a_p, 0.0)
            a_p = a_p / a_p.sum()
            inits.append(a_p)

        rem_bud = deadline - time.monotonic() - 60.0
        per_init = max(15.0, rem_bud / len(inits))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 50:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
                print(f"  Init {idx}: {sc0:.10f} (NEW BEST)")
                continue

            sub_deadline = time.monotonic() + min(per_init, deadline - time.monotonic() - 40.0)
            if sub_deadline - time.monotonic() < 10:
                continue

            # Quick optimization
            a_res, sc_res = _multi_start_optimize(a0, n_best, sub_deadline,
                                                   methods=('lp', 'adam'))
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  Init {idx}: {sc_res:.10f} (NEW BEST)")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = best_seq.size
        rem = deadline - time.monotonic() - 25.0
        print(f"\n=== Phase 3: Final polish, rem={rem:.0f}s ===")

        # LP
        if rem > 60:
            a_res = _lp_optimize(best_seq, time.monotonic() + rem * 0.3,
                                 max_iter=20000)
            sc = _score(a_res)
            if sc < best_score:
                best_score = sc
                best_seq = a_res.copy()
            print(f"  Final LP: {sc:.10f}")

        # Adam
        if time.monotonic() < deadline - 30:
            a_res = _adam_logspace(best_seq, deadline - 30, lr=0.015, top_k=12)
            sc = _score(a_res)
            if sc < best_score:
                best_score = sc
            best_seq = a_res if sc < best_score else best_seq
            print(f"  Final Adam: {sc:.10f}")

        # Peak peel
        if time.monotonic() < deadline - 20:
            a_res = _peak_peel(best_seq, deadline - 15)
            sc = _score(a_res)
            if sc < best_score:
                best_score = sc
                best_seq = a_res.copy()
            print(f"  Final Peel: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        # Keep normalized (score is invariant to scaling)
        pass

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
