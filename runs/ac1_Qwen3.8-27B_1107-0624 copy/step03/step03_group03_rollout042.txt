def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return _score_normalized(a / s)


def _build_conv_matrix(a):
    n = len(a)
    A_conv = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_conv[j:j + n, j] = a
    return A_conv


def _lp_linearized(a, M_target=None, max_lp_time=10.0):
    n = len(a)
    c = _fft_convolve(a)
    n_c = 2 * n - 1
    A_conv = _build_conv_matrix(a)

    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

    A_top = np.column_stack([-np.ones(n_c), 2.0 * A_conv])
    if M_target is not None:
        b_top = M_target - c
    else:
        b_top = -c.copy()

    A_bot = np.zeros((n, n + 1))
    for i in range(n):
        A_bot[i, i + 1] = -1.0
    b_bot = a.copy()

    A_ub = np.vstack([A_top, A_bot])
    b_ub = np.concatenate([b_top, b_bot])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])

    bounds = [(None, None)] + [(-a[i], None) for i in range(n)]

    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if time.monotonic() - t0 > max_lp_time * 3:
        return None, None

    if not res.success:
        return None, None

    delta = res.x[1:]
    t_val = float(res.x[0])
    if not np.all(np.isfinite(delta)):
        return None, None
    return delta, t_val


def _line_search_alpha(a, d, max_evals=25):
    n = len(a)
    neg = d < 0
    if np.any(neg):
        alpha_max = float(np.min(-a[neg] / d[neg]))
    else:
        alpha_max = 1.0
    alpha_max = min(alpha_max, 2.0)
    if alpha_max <= 0:
        return 0.0
    best_alpha = 0.0
    best_score = _score_normalized(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = _score_normalized(np.maximum(a + c_pt * d, 0))
    fd = _score_normalized(np.maximum(a + d_pt * d, 0))
    for _ in range(max_evals):
        if not np.isfinite(fc):
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = _score_normalized(np.maximum(a + c_pt * d, 0))
            continue
        if not np.isfinite(fd):
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = _score_normalized(np.maximum(a + d_pt * d, 0))
            continue
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = _score_normalized(np.maximum(a + c_pt * d, 0))
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = _score_normalized(np.maximum(a + d_pt * d, 0))
    best_alpha = (lo + hi) / 2.0
    best_score = _score_normalized(np.maximum(a + best_alpha * d, 0))
    if not np.isfinite(best_score) or best_score >= _score_normalized(a):
        return 0.0
    return best_alpha


def _lp_descent_loop(a, deadline, max_iter=2000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        delta, t_val = _lp_linearized(a, max_lp_time=8.0)
        if delta is None:
            no_imp += 10
            if no_imp > 200:
                break
            continue
        alpha = _line_search_alpha(a, delta, max_evals=20)
        if alpha < 1e-6:
            no_imp += 5
            if no_imp % 50 == 0:
                # Try with a target M
                M_cur = _fft_convolve(a).max()
                delta2, _ = _lp_linearized(a, M_target=M_cur * 0.995, max_lp_time=8.0)
                if delta2 is not None and np.any(np.abs(delta2) > 1e-10):
                    alpha2 = _line_search_alpha(a, delta2, max_evals=20)
                    if alpha2 > 1e-6:
                        a = np.maximum(a + alpha2 * delta2, 0)
                        ss = a.sum()
                        if ss > 1e-10:
                            a = a / ss
                        sc = _score_normalized(a)
                        if sc < best_score - 1e-10:
                            best_score = sc
                            best_a = a.copy()
                            no_imp = 0
                        continue
            if no_imp > 300:
                break
            continue
        a_new = a + alpha * delta
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-10:
            a = best_a.copy()
            no_imp += 10
            continue
        a_new = a_new / ss
        sc = _score_normalized(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            a = a_new
    return best_a


def _lse_score_and_grad(a, T):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return m_val, grad, w


def _lse_adam_step(a, z, state, T, lr):
    n = len(a)
    m_val, grad, w = _lse_score_and_grad(a, T)
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.05, T_end=1e-5, lr=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, m_val = _lse_adam_step(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.01, lr * 0.9)
            if no_imp > max(10000, 15 * n):
                rng_r = np.random.default_rng(42 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        T *= T_ratio
        T = max(T, T_end)
    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    v = np.zeros(n)
    momentum = 0.8
    no_imp = 0
    base_lr = lr
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        grad_tan -= float(np.mean(grad_tan))
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.002, lr * 0.92)
            if no_imp > max(8000, 10 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bell(n, width=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + ((t - 0.5) / width) ** 4)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_smooth(n, rng, n_knots=20):
    knots = np.sort(rng.uniform(0, 1, n_knots))
    vals = rng.uniform(0.5, 1.5, n_knots)
    t = np.linspace(0.0, 1.0, n)
    a = np.interp(t, knots, vals)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _combined_optimize(a, deadline, n_candidates=4):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = _score_normalized(a)

    used = 0.0
    remaining = lambda: max(0.0, deadline - time.monotonic())

    # Phase A: LP descent
    if remaining() > 30:
        lp_deadline = time.monotonic() + min(remaining() * 0.4, 400.0)
        a_lp = _lp_descent_loop(a, lp_deadline, max_iter=1500)
        sc_lp = _score_normalized(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
        used += time.monotonic() - (lp_deadline - min(remaining(), 400.0))
        a = a_lp

    # Phase B: LSE gradient
    if remaining() > 20:
        lse_deadline = time.monotonic() + min(remaining() * 0.35, 300.0)
        a_lse = _lse_optimize(a, lse_deadline, max_iter=80000,
                              T_start=0.03, T_end=1e-5, lr=0.08)
        sc_lse = _score_normalized(a_lse)
        if sc_lse < best_score:
            best_score = sc_lse
            best_a = a_lse.copy()
        a = a_lse

    # Phase C: LP again on refined
    if remaining() > 20:
        lp2_deadline = time.monotonic() + min(remaining() * 0.3, 250.0)
        a_lp2 = _lp_descent_loop(a, lp2_deadline, max_iter=1000)
        sc_lp2 = _score_normalized(a_lp2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()
        a = a_lp2

    # Phase D: Subgradient polish
    if remaining() > 15:
        sp_deadline = time.monotonic() + min(remaining() * 0.25, 200.0)
        a_sp = _subgradient_polish(a, sp_deadline, max_iter=60000, lr=0.015)
        sc_sp = _score_normalized(a_sp)
        if sc_sp < best_score:
            best_score = sc_sp
            best_a = a_sp.copy()

    # Phase E: Final LSE
    if remaining() > 10:
        lse2_deadline = time.monotonic() + min(remaining() * 0.8, 150.0)
        a_lse2 = _lse_optimize(best_a, lse2_deadline, max_iter=50000,
                               T_start=0.01, T_end=1e-6, lr=0.04)
        sc_lse2 = _score_normalized(a_lse2)
        if sc_lse2 < best_score:
            best_score = sc_lse2
            best_a = a_lse2.copy()

    return best_a
