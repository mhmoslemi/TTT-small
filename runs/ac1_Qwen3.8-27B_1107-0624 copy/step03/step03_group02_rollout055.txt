"""
Optimized step function search using Nesterov-accelerated smooth-max Adam,
sparse LP spreading, and multi-scale optimization at large sequence lengths (2048-4096).
Key improvements: Nesterov momentum, better LP with sparse matrices, adaptive beta scheduling,
and aggressive exploitation of larger n for more degrees of freedom.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c += -c.min()
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_nesterov(a, deadline, max_iter=50000, lr=0.02, beta_factor=0.04,
                   nesterov_mu=0.9, verbose=False):
    """Adam with Nesterov momentum in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    z_lookahead = z.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        # Nesterov lookahead
        z_lookahead = z - lr_cur * nesterov_mu * mh / (np.sqrt(vh) + eps)
        z_lookahead = np.clip(z_lookahead, -60.0, 60.0)

        e_la = np.exp(z_lookahead)
        a_la = e_la / e_la.sum()

        c_la = _fft_convolve(a_la)
        w_la = _smooth_max_weights(c_la, beta)
        grad_a_la = _grad_smooth_max(a_la, w_la)
        dot_la = float(np.dot(a_la, grad_a_la))
        grad_z_la = a_la * (grad_a_la - dot_la)
        grad_z_la -= float(np.mean(grad_z_la))

        z_new = z - lr_cur * grad_z_la / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.0005, lr_cur * 0.95)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(2000 + it)
                z = best_z + 0.2 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                z_lookahead = z.copy()
                continue

        z = z_new
        a = a_new
        z_lookahead = z.copy()

        if it % 200 == 0:
            beta = max(1e-14, beta * 0.82)

        if verbose and it % 5000 == 0:
            print(f"      Nesterov it={it}: best={best_score:.10f}, beta={beta:.2e}")

    # Also try the reverse
    a_rev = best_a[::-1].copy()
    s_rev = float(2.0 * n * _fft_convolve(a_rev).max())
    if s_rev < best_score:
        best_a = a_rev
        best_score = s_rev

    return best_a, best_score

    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.0005, lr_cur * 0.95)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(3000 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-14, beta * 0.82)

    a_rev = best_a[::-1].copy()
    s_rev = float(2.0 * n * _fft_convolve(a_rev).max())
    if s_rev < best_score:
        best_a = a_rev
        best_score = s_rev

    return best_a, best_score


def _lp_spread_sparse(a, M):
    """LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0. Uses sparse matrix."""
    n = len(a)
    c_obj = -np.ones(n)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    A = A.tocsr()
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3, verbose=False):
    """LP-based refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    lp_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        t0 = time.monotonic()
        g = _lp_spread_sparse(a, M_lp)
        lp_time = time.monotonic() - t0

        if g is None:
            break

        lp_count += 1
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.08)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            # Decrease blend as we converge
            if lp_count % 10 == 0:
                blend_cur = max(0.05, blend_cur * 0.95)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.01, inflate_cur * 0.92)
            if no_imp > max(100, n // 4):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

        if verbose and it % 50 == 0:
            print(f"      LP it={it}: best={best_score:.10f}, lp_time={lp_time:.2f}s")

    return best_a


def _subgradient_multi(a, deadline, max_iter=8000, top_k=5, verbose=False):
    """Multi-peak subgradient with adaptive top-k."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Weight peaks by their value relative to max
        peak_vals = np.array([float(c_curr[idx]) for idx in top_indices])
        weights = peak_vals / peak_vals.sum()

        # Use smooth-max gradient instead of subgradient for better behavior
        beta = 0.01 * max_val
        w = _smooth_max_weights(c_curr, beta)
        sg = _grad_smooth_max(a, w)

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        # Polyak step size
        gap = max_val / n  # rough gap estimate
        lr = min(base_lr, gap / (sg_norm * sg_norm + 1e-10))
        lr = lr / np.sqrt(1 + it * 0.01)

        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break

        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp_profile(n, skew=1.0):
    """Create a ramp profile with variable skew."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + skew * (1.0 - 2.0 * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_cosine_profile(n, alpha=0.8, phase=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * (t + phase))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_quadratic_profile(n, c2=-1.0, c1=0.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c2 * u ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _optimize_pipeline(a0, deadline, label="", use_nesterov=True):
    """Run a full optimization pipeline on a0."""
    a0 = np.asarray(a0, dtype=np.float64).copy()
    n = len(a0)
    sc0 = _score(a0)
    print(f"  {label} init: {sc0:.10f}, n={n}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Main optimization (Nesterov or standard Adam)
    t1_deadline = min(deadline - 120, time.monotonic() + 350)
    if use_nesterov:
        a1, s1 = _adam_nesterov(a0, t1_deadline, max_iter=30000, lr=0.02, beta_factor=0.04)
    else:
        a1, s1 = _adam_standard(a0, t1_deadline, max_iter=30000, lr=0.02, beta_factor=0.04)
    if s1 < best_sc:
        best_a, best_sc = a1, s1
    print(f"  {label} Phase1: {best_sc:.10f}")

    # Phase 2: LP refinement
    if time.monotonic() < deadline - 80:
        d2 = min(deadline - 80, time.monotonic() + 180)
        a2 = _lp_refine(best_a, d2, max_iter=1500, inflate=1.001, blend=0.3)
        s2 = _score(a2)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} Phase2 (LP): {best_sc:.10f}")

    # Phase 3: Fine Adam
    if time.monotonic() < deadline - 40:
        d3 = min(deadline - 40, time.monotonic() + 150)
        a3, s3 = _adam_nesterov(best_a, d3, max_iter=20000, lr=0.008, beta_factor=0.015)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} Phase3: {best_sc:.10f}")

    # Phase 4: Final LP polish
    if time.monotonic() < deadline - 15:
        d4 = min(deadline - 15, time.monotonic() + 60)
        a4 = _lp_refine(best_a, d4, max_iter=800, inflate=1.0005, blend=0.15)
        s4 = _score(a4)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} Phase4 (LP): {best_sc:.10f}")

    # Phase 5: Ultra-fine Adam
    if time.monotonic() < deadline - 5:
        a5, s5 = _adam_nesterov(best_a, deadline - 5, max_iter=10000, lr=0.003, beta_factor=0.008)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} Phase5: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization at large sequence lengths (2048, 4096) using
    Nesterov-accelerated Adam, LP spreading, and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare initializations at different scales
    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
        p2048 = _interpolate_profile(parent, 2048)
    else:
        p512 = _make_ramp_profile(512, 1.0)
        p1024 = _make_ramp_profile(1024, 1.0)
        p2048 = _make_ramp_profile(2048, 1.0)

    # === Main strategy: Optimize at n=2048 (most time) ===
    print("=== Main: n=2048 ===")
    d_main = min(deadline - 400, t_start + 900)
    a_main, s_main = _optimize_pipeline(p2048, d_main, "MAIN-2048", use_nesterov=True)
    if s_main < best_score:
        best_score = s_main
        best_seq = a_main.copy()

    # === Strategy: Try n=4096 if time permits ===
    if best_seq is not None and time.monotonic() < deadline - 500:
        print("=== Try n=4096 ===")
        p4096 = _interpolate_profile(best_seq, 4096)
        d_4096 = min(deadline - 200, time.monotonic() + 400)
        a_4096, s_4096 = _optimize_pipeline(p4096, d_4096, "N4096", use_nesterov=True)
        if s_4096 < best_score:
            best_score = s_4096
            best_seq = a_4096.copy()

    # === Strategy: Diverse initializations at n=2048 ===
    if time.monotonic() < deadline - 200:
        print("=== Diverse inits @ n=2048 ===")
        n = 2048
        candidates = [
            _make_ramp_profile(n, 0.5),
            _make_ramp_profile(n, 1.5),
            _make_ramp_profile(n, 2.0),
            _make_cosine_profile(n, 0.5, 0.0),
            _make_cosine_profile(n, 1.0, 0.1),
            _make_quadratic_profile(n, -0.5, 0.3),
            _make_quadratic_profile(n, -1.5, 0.8),
        ]
        # Perturbed best
        if best_seq is not None:
            bp = _interpolate_profile(best_seq, n)
            bp_pert = bp * (1.0 + 0.1 * rng.normal(size=n))
            bp_pert = np.maximum(bp_pert, 0.0)
            candidates.append(bp_pert / bp_pert.sum())
            # Reversed
            candidates.append(bp[::-1].copy())

        remaining = deadline - 200 - time.monotonic()
        per_cand = remaining / len(candidates)

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 200:
                break
            if per_cand < 30:
                break
            d_i = min(deadline - 200, time.monotonic() + per_cand)
            a_c, s_c = _adam_nesterov(a0, d_i, max_iter=8000, lr=0.015, beta_factor=0.04)
            if s_c < best_score:
                best_score = s_c
                best_seq = a_c.copy()
            print(f"  Div-{i}: {s_c:.10f} (best={best_score:.10f})")

    # === Final refinement on the best found ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"=== Final refinement (best={best_score:.10f}) ===")
        n_best = best_seq.size

        # Fine Adam
        d_f1 = min(deadline - 15, time.monotonic() + 80)
        a_f1, s_f1 = _adam_nesterov(best_seq, d_f1, max_iter=10000, lr=0.004, beta_factor=0.008)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Fine Adam: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 10:
            d_f2 = min(deadline - 10, time.monotonic() + 50)
            a_f2 = _lp_refine(best_seq, d_f2, max_iter=500, inflate=1.0003, blend=0.1)
            s_f2 = _score(a_f2)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  LP polish: {best_score:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 5:
            a_f3, s_f3 = _adam_nesterov(best_seq, deadline - 5, max_iter=5000, lr=0.002, beta_factor=0.005)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Ultra-fine: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp_profile(256, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp_profile(256, 1.0)

    # Verify with evaluate_sequence
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
