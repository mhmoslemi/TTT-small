def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a, active_lags=None):
    n = len(a)
    c_len = 2 * n - 1
    if active_lags is not None:
        m = len(active_lags)
    else:
        active_lags = list(range(c_len))
        m = c_len
    A = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[idx, i] = a[k - i]
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_smooth_max(a, beta, top_k=0):
    n = len(a)
    c = _fft_convolve(a)
    if top_k > 0:
        sorted_idx = np.argsort(c)[::-1]
        w_smooth = _smooth_max_weight(c, beta)
        w_peak = np.zeros_like(c)
        top_indices = sorted_idx[:top_k]
        peak_vals = c[top_indices]
        w_peak[top_indices] = peak_vals / peak_vals.sum()
        w = 0.6 * w_smooth + 0.4 * w_peak
        w = w / w.sum()
    else:
        w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _lp_refine(a, deadline, max_iter=20000, inflate_start=1.001, blend_start=0.4):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    inflate_cur = inflate_start
    blend_cur = blend_start
    prev_best = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Find active lags (top portion)
        thr = M * 0.7
        active = np.where(c >= thr)[0]
        if len(active) < 5:
            active = np.argsort(c)[::-1][:10]
        if len(active) > 60:
            active = np.argsort(c)[::-1][:60]

        b_ub = np.full(len(active), M_lp)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:

            if no_imp > 2000:
                break
            continue

        g = g / sum_g

        score_new = float(2.0 * n * M_new)

            best_M = M_new

def _adam_peak_polish(a, deadline, max_iter=30000, lr=0.04):

        w = 0.5 * w + 0.5 * w_peak

            continue

        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)

            no_imp = 0

def _block_operations(a, deadline, max_iter=8000):
    """Block swaps, reversals, and rotations to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        op = rng.integers(0, 3)
        a_new = a.copy()

        if op == 0:
            # Block swap
            bs = rng.integers(max(2, n // 32), max(3, n // 8))
            i = rng.integers(0, n - bs)
            j = rng.integers(0, n - bs)
            if abs(i - j) < bs:
                continue
            a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
        elif op == 1:
            # Block reversal
            bs = rng.integers(max(2, n // 16), max(4, n // 6))
            i = rng.integers(0, n - bs)
            a_new[i:i+bs] = a_new[i:i+bs][::-1].copy()
        else:
            # Cyclic shift of a block
            bs = rng.integers(max(3, n // 16), max(5, n // 6))
            i = rng.integers(0, n - bs)
            shift = rng.integers(1, bs)
            block = a_new[i:i+bs].copy()
            a_new[i:i+bs] = np.roll(block, shift)

        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-11:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_v(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gauss(n, c, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_v(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        final_tb = deadline - time.monotonic() - 30
        print(f"\n=== Final polish at n={n_best} (budget={final_tb:.0f}s) ===")

        # Sub-phase: LP with very tight constraints
        lp_t = min(final_tb * 0.3, 300.0)
        if lp_t > 15:
            a_r = _lp_refine(best_seq, time.monotonic() + lp_t, max_iter=30000,
                            inflate_start=1.0005, blend_start=0.25)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final LP: {sc:.10f}")

        # Sub-phase: Long Adam
        ad_t = min(final_tb * 0.3, 250.0)
        if ad_t > 15:
            a_r = _adam_peak_polish(best_seq, time.monotonic() + ad_t, max_iter=30000, lr=0.03)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final Adam: {sc:.10f}")

        # Sub-phase: Block operations
        bo_t = min(final_tb * 0.2, 150.0)
        if bo_t > 10:
            a_r = _block_operations(best_seq, time.monotonic() + bo_t, max_iter=8000)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final Blocks: {sc:.10f}")

        # Sub-phase: One more LP
        lp2_t = min(final_tb * 0.15, 100.0)
        if lp2_t > 10:
            a_r = _lp_refine(best_seq, time.monotonic() + lp2_t, max_iter=20000,
                            inflate_start=1.0003, blend_start=0.15)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final LP2: {sc:.10f}")

        # Sub-phase: Final Adam
        ad2_t = min(final_tb * 0.1, 80.0)
        if ad2_t > 8:
            a_r = _adam_peak_polish(best_seq, time.monotonic() + ad2_t, max_iter=20000, lr=0.015)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final Adam2: {sc:.10f}")

    # Phase 3: Try a different length with the best shape
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        alt_lengths = [l for l in [n_best - 64, n_best + 64, n_best - 128, n_best + 128]
                      if 256 <= l <= 2048 and l != n_best]
        for alt_n in alt_lengths[:3]:
            if time.monotonic() > deadline - 40:
                break
            alt_a = _interpolate_v(best_seq, alt_n)
            tb_alt = min(60.0, deadline - time.monotonic() - 30)
            if tb_alt > 20:
                a_r, sc_r = _full_optimize(alt_a, deadline, tb_alt)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  ** New best at alt n={alt_n}: {sc_r:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear(768, 1.5)

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
