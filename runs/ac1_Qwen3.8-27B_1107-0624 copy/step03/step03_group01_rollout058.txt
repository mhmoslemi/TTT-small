from scipy.optimize import minimize
import time

# Helper functions
def full_convolve(a, b):
    """FFT-based full convolution"""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score_fn(a):
    """Compute the score"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def score_norm_fn(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

    t_start = time.monotonic()
    
    for beta in betas:
        if time.monotonic() - t_start >= max_time:
            break
        try:
            result = minimize(
                lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=lbfgs_gradient,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9,
                         'maxls': 30, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax_fn(result.x)
                sc = score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except:
            pass
    return best_a, best_score

def adam_subgradient(a_init, max_time, rng, lr=0.01, n_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    t_start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t_start >= max_time:
            break
        t += 1
        a = softmax_fn(z)
        c = full_convolve(a, a)
        cm = c.max()
        
        # Soft subgradient: use softmax over top values
        # Temperature: start high, decrease
        temp = max(0.001, 0.1 * c.std())
        c_shifted = (c - cm) / temp
        e = np.exp(c_shifted)
        s = e.sum()
        w = e / s
        
        W = np.dot(w, c)
        a_rev = a[::-1]
        w_conv = full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        grad = 4.0 * n * a * (h - W)
        
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        a_new = softmax_fn(z)
        sc = score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            
        if t % 500 == 0:
            lr = max(0.001, lr * 0.98)
    
    return best_a, best_score
