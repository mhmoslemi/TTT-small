"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logit-space Adam, LP spreading, peak shaving, and coordinate descent.
Explores multiple lengths and initializations within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _compute_grad(a, beta):
    """Compute smooth max gradient w.r.t. a (on the simplex)."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_val(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    vp = 2.0 * h + 1e-8 * a
    dot = float(np.dot(a, vp))
    gz = a * (vp - dot)
    gz -= float(np.mean(gz))
    return gz, float(c.max())


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_bell_profile(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _logit_adam_optimize(a, deadline, max_iter=50000, lr=0.05, beta_init=None):
    """Adam in logit space with adaptive smooth-max temperature."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        gz, c_max_cur = _compute_grad(a, beta)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        # BB step size
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_candidate = dot_ss / max(abs(dot_sg), 1e-20)
                bb_candidate = np.clip(bb_candidate, 0.1, 10.0)
                bb_scale = 0.7 * bb_scale + 0.3 * bb_candidate
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        e_sum = e.sum()
        if e_sum < 1e-30:
            z = best_z.copy()
            continue
        a = e / e_sum

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.93)

        # Adaptive beta annealing
        if it % 200 == 0 and it > 200:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.3:
                    beta = max(1e-12, beta * 0.5)
                elif peak_gap > beta * 10.0:
                    beta = min(0.3 * c_max_new, beta * 1.5)

        # Annealing restart
        if no_imp > max(4000, 6 * n):
            rng_r = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.5 * max(base_lr, 0.01)
            z_restart = best_z + noise_scale * rng_r.standard_normal(n)
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a, best_score


def _lp_refine(a, deadline, max_iter=20000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.003, inflate_cur * 0.97)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _peak_shave(a, deadline, max_iter=3000, n_peaks=3, strength=0.04):
    """Targeted peak shaving."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        med = float(np.median(c))
        weights = np.maximum(top_vals - med, 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / wsum

        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]

        pmax = pressure.max()
        if pmax < 1e-15:
            break
        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.15, str_cur * 1.005)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.01, str_cur * 0.85)
            if no_imp > 800:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a, best_score


def _coord_descent(a, deadline, max_rounds=200, top_k=50):
    """Coordinate descent on the top-k most important coordinates."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a)
    c_max = float(c.max())

    # Find most "active" coordinates (those contributing most to peaks)
    pressure = np.zeros(n)
    top_idx = np.argsort(c)[::-1][:5]
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                pressure[i] += a[j]

    # Top k coordinates to optimize
    coord_idx = np.argsort(pressure)[::-1][:min(top_k, n)]

    no_imp = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in coord_idx:
            if time.monotonic() > deadline:
                break
            # Try scaling coordinate i
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                a_try = a.copy()
                a_try[i] = a[i] * factor
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                sc = _score(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a, best_score


def _multi_start_refine(a, deadline, n_starts=5, rng=None):
    """Try multiple perturbed versions and refine the best."""
    if rng is None:
        rng = np.random.default_rng(99999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.03 * (i + 1)
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.3:
            continue

        sub_deadline = min(deadline, time.monotonic() + 60.0)
        a_ref, sc_ref = _logit_adam_optimize(a_pert, sub_deadline, max_iter=3000, lr=0.03)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a, best_score


def _fine_polish(a, deadline, max_iter=20000):
    """Very fine gradient polish with tiny beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-10, 0.005 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        gz, _ = _compute_grad(a, beta)
        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        e_sum = e.sum()
        if e_sum < 1e-30:
            break
        a = e / e_sum

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.0005, lr * 0.9)

        if it % 500 == 0:
            beta = max(1e-15, beta * 0.7)

        if no_imp > max(2000, 4 * n):
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Combines logit-Adam, LP spreading, peak shaving, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine candidate lengths
    lengths = set()
    if parent is not None:
        lengths.add(parent.size)
    for ln in [256, 384, 512, 768]:
        lengths.add(ln)
    lengths = sorted(lengths)

    # Phase 1: Generate diverse initializations
    print(f"Phase 1: Initialization, {deadline - time.monotonic():.0f}s remaining", flush=True)
    init_pool = []

    for n_t in lengths:
        # Uniform (theoretical optimum)
        init_pool.append(('uniform', n_t, np.ones(n_t) / n_t))

        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))

        # Linear profile
        for eta in [1.0, 2.0]:
            init_pool.append((f'lin_{eta}', n_t, _make_linear_profile(n_t, eta)))

        # Bell
        init_pool.append(('bell', n_t, _make_bell_profile(n_t, 0.3)))

        # Flat spectrum
        init_pool.append(('flatspec', n_t, _make_flat_spectrum_init(n_t, rng)))

    # Quick evaluation
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 6 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:6]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Optimize top candidates
    # Focus on best few, prioritizing the parent length and 512
    top_inits = init_scores[:5]
    # Ensure parent-length is included
    if parent is not None:
        parent_len = parent.size
        found = any(n_t == parent_len for _, _, n_t, _ in top_inits)
        if not found:
            for item in init_scores:
                if item[2] == parent_len:
                    top_inits.append(item)
                    break

    for sc0, name, n_t, a0 in top_inits:
        if time.monotonic() > deadline - 60:
            break
        # Skip if clearly worse than best and not the main parent length
        if sc0 > best_score * 1.1 and n_t != (parent.size if parent is not None else 512):
            continue

        remaining = deadline - time.monotonic()
        # Allocate time proportionally
        alloc = min(remaining * 0.4, 500.0)

        # Logit Adam
        opt_deadline = min(deadline - 100, time.monotonic() + alloc)
        print(f"Phase 2: Adam on {name} (n={n_t}), start={sc0:.8f}, t={opt_deadline - time.monotonic():.0f}s", flush=True)
        a_opt, sc_opt = _logit_adam_optimize(a0, opt_deadline, max_iter=50000, lr=0.05)
        print(f"  Adam: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < deadline - 80:
            lp_dl = min(deadline - 60, time.monotonic() + min(200, remaining * 0.2))
            a_lp, sc_lp = _lp_refine(a_opt, lp_dl, max_iter=15000, inflate=1.0008, blend=0.25)
            print(f"  LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Peak shaving
        if time.monotonic() < deadline - 40:
            ps_dl = min(deadline - 20, time.monotonic() + 40.0)
            a_ps, sc_ps = _peak_shave(a_opt, ps_dl, max_iter=3000, n_peaks=3, strength=0.03)
            print(f"  PeakShave: {sc_ps:.10f}", flush=True)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            a_opt = a_ps

    # Phase 3: Fine polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 30:
        gp_dl = min(deadline - 20, time.monotonic() + 180.0)
        print(f"Phase 3: Fine polish, t={gp_dl - time.monotonic():.0f}s", flush=True)
        a_fine, sc_fine = _fine_polish(best_seq, gp_dl, max_iter=20000)
        print(f"  Fine: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Phase 4: Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 20:
        cd_dl = min(deadline - 10, time.monotonic() + 120.0)
        print(f"Phase 4: Coord descent, t={cd_dl - time.monotonic():.0f}s", flush=True)
        a_cd, sc_cd = _coord_descent(best_seq, cd_dl, max_rounds=100, top_k=30)
        print(f"  CoordDesc: {sc_cd:.10f}", flush=True)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()

    # Phase 5: Multi-start
    if best_seq is not None and time.monotonic() < deadline - 10:
        ms_dl = min(deadline - 5, time.monotonic() + 60.0)
        print(f"Phase 5: Multi-start, t={ms_dl - time.monotonic():.0f}s", flush=True)
        a_ms, sc_ms = _multi_start_refine(best_seq, ms_dl, n_starts=4, rng=rng)
        print(f"  MultiStart: {sc_ms:.10f}", flush=True)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()

    # Phase 6: Final LP pass
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_dl2 = deadline - 3
        if lp_dl2 - time.monotonic() > 15:
            a_last, sc_last = _lp_refine(best_seq, lp_dl2, max_iter=5000, inflate=1.0005, blend=0.15)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LP: {sc_last:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(512) / 512
        best_score = _score(final_seq)

    # Final validation with authoritative function
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
