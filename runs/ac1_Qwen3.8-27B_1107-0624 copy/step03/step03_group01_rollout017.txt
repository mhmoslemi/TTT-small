"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with logsumexp smoothing, Adam with top-k active sets,
and KKT-based balancing for final polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_smooth_grad(a, beta, n, top_k=0):
    """Compute softmax-weights and gradient for the logsumexp-smoothed max."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return w, h, W, c, cm


def _z_gradient(z, beta, n, top_k=0):
    """Gradient of the smoothed objective with respect to z (logit space)."""
    a = _softmax_fn(z)
    w, h, W, c, cm = _compute_smooth_grad(a, beta, n, top_k)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective(z, beta, n, top_k=0):
    """Smoothed objective value in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0):
    """Run L-BFGS with beta continuation on a normalized sequence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=50000):
    """Adam optimizer with adaptive beta for the smoothed max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    # Adaptive top_k: start small, grow over time
    if top_k == 0:
        adaptive_k = True
        cur_k = 3
    else:
        adaptive_k = False
        cur_k = top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1

        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 100 == 0:
            beta = max(1e-14, beta * 0.88)
            if adaptive_k and t % 500 == 0:
                cur_k = min(cur_k + 2, 30)

        if t % 2000 == 0:
            if best_score < 1.51:
                lr = max(0.003, lr * 0.97)

    return best_a, best_score


def _kkt_balance(a_init, deadline, max_iter=5000):
    """KKT-based balancing: at optimum, weighted sum of active gradients is constant."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: indices where c is within 1% of max
        threshold = cm * 0.995
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break

        # Compute the "balanced" gradient direction
        # For each active k, the gradient of c(k) w.r.t. a is 2*a_rev_shifted
        # We want to find a direction d such that for all active k:
        #   c(k + d) < c(k) (all active peaks decrease)
        # Approximate: move a in direction that equalizes the active c values

        # Compute h_i = (1/|A|) * sum_{k in A} a[(k-i) mod n] (circular shift)
        # Actually: h_i = (1/|A|) * sum_{k in A} a[k-i] if 0<=k-i<n
        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            # a[k_val - i] for i = 0..n-1, where valid means 0 <= k_val-i <= n-1
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi+1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)

        W = float(np.dot(h, a))  # should be cm/|A| approximately

        # Update direction: reduce a where h > W, increase where h < W
        # But maintain non-negativity via logit space
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        # Step size: adaptive based on the "spread" of active values
        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = 0.01 * active_spread / (cm + 1e-15)
        step = min(step, 0.05)

        z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step and retry
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        if it > 100 and (best_score - sc if np.isfinite(sc) else 0) < 1e-10:
            break

    return best_a, best_score


def _make_exp_profile(n, alpha):
    """Exponential decay profile: a_i = (1-alpha)^i, normalized."""
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    """Power profile: a_i = (i+1)^p, normalized."""
    a = (np.arange(n) + 1.0) ** p
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS continuation + Adam + KKT balancing.
    Minimizes 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Gather initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 20.0)
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent available: {e}")

    n_search = 2048

    # Generate diverse profiles at n_search
    profiles = []
    # Exponential profiles with different rates
    for alpha in [0.0005, 0.001, 0.002, 0.003, 0.005, 0.008, 0.01]:
        profiles.append(_make_exp_profile(n_search, alpha))
    # Linear profiles
    for slope in [-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0]:
        profiles.append(_make_linear_profile(n_search, slope))
    # Quadratic profiles
    for a1, a2 in [(-1.0, 0.5), (-1.5, 0.3), (1.0, -0.3), (1.5, -0.5), (-0.5, 1.0)]:
        profiles.append(_make_quadratic_profile(n_search, a1, a2))
    # Power profiles
    for p in [0.5, 1.0, 1.5, 2.0, -0.5]:
        profiles.append(_make_power_profile(n_search, p))

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_search, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Also try parent length if different
    if parent is not None and len(parent) != n_search:
        # Interpolate parent to n_search
        p_interp = _interp_profile(parent, n_search)
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, n_search, "parent_interp"))
        update_best(p_interp, sc, "parent_interp")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: L-BFGS continuation on top candidates ---
    print("=== Phase 1: L-BFGS continuation ===")
    phase1_end = min(deadline, t_start + 500.0)
    count = 0
    seen_keys = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 4:
            break
        if sc > best_score * 1.02:
            break
        key = f"{label}_n{pn}"
        if key in seen_keys:
            continue
        seen_keys.add(key)

        remaining = phase1_end - time.monotonic()
        per_time = max(5.0, min(120.0, remaining / max(1, 4 - count)))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS on {label} (n={pn}), time_budget={per_time:.1f}s")

        p_best, sc_best = _run_lbfgs(p, dl, max_wall_time=per_time, top_k=0)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam polish on best ---
    print("=== Phase 2: Adam polish ===")
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + remaining)
        print(f"  Adam on best (n={len(best_seq)}), time_budget={remaining:.1f}s")
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.04, top_k=0, n_max_iter=80000)
        update_best(p_best, sc_best, "adam1")

        # Second adam pass with smaller lr
        if time.monotonic() < phase2_end - 10:
            dl2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.015, top_k=5, n_max_iter=40000)
            update_best(p_best2, sc_best2, "adam2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: KKT balancing ---
    print("=== Phase 3: KKT balancing ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining)
        print(f"  KKT balance on best, time_budget={remaining:.1f}s")
        p_best, sc_best = _kkt_balance(best_seq, dl, max_iter=10000)
        update_best(p_best, sc_best, "kkt")

        # Follow up with short L-BFGS
        if time.monotonic() < phase3_end - 5:
            dl2 = min(phase3_end - 5, time.monotonic() + 60.0)
            p_best2, sc_best2 = _run_lbfgs(best_seq, dl2, max_wall_time=55.0, top_k=5)
            update_best(p_best2, sc_best2, "lbfgs_post_kkt")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart ---
    print("=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(60.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            noise_levels = [0.05, 0.1, 0.15, 0.2]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase4_end - 5, time.monotonic() + per_restart)
            # Quick L-BFGS then Adam
            per_lbfgs = per_restart * 0.6
            per_adam = per_restart * 0.35
            p_m, sc_m = _run_lbfgs(p_start, min(phase4_end - 5, time.monotonic() + per_lbfgs),
                                    max_wall_time=per_lbfgs, top_k=3)
            update_best(p_m, sc_m, f"r{i}_lbfgs")
            if time.monotonic() < phase4_end - 5:
                p_a, sc_a = _adam_polish(p_m, min(phase4_end - 5, time.monotonic() + per_adam),
                                         lr=0.03, top_k=5, n_max_iter=20000)
                update_best(p_a, sc_a, f"r{i}_adam")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try different lengths ---
    print("=== Phase 5: Different lengths ===")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 1024:
            target_ns.append(1024)
        if n_curr != 4096:
            target_ns.append(4096)

        for tn in target_ns:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase5_end - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(120.0, remaining * 0.6)
            dl = min(phase5_end - 10, time.monotonic() + per_opt)
            p_best, sc_best = _run_lbfgs(p_interp, dl, max_wall_time=per_opt, top_k=0)
            update_best(p_best, sc_best, f"lbfgs_n{tn}")

            if time.monotonic() < phase5_end - 20:
                dl2 = min(phase5_end - 10, time.monotonic() + min(60.0, remaining * 0.3))
                p_best2, sc_best2 = _adam_polish(p_best, dl2, lr=0.03, top_k=5, n_max_iter=20000)
                update_best(p_best2, sc_best2, f"adam_n{tn}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish ---
    print("=== Phase 6: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.5))
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.01, top_k=10, n_max_iter=30000)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + min(40.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _run_lbfgs(best_seq, dl2, max_wall_time=max(5.0, dl2 - time.monotonic()), top_k=10)
            update_best(p_best2, sc_best2, "final_lbfgs")

    # --- Final validation and return ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(2048, 0.003)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence if feasible
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 8192:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(2048, 0.003)
    print(f"  FALLBACK: n=2048, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
