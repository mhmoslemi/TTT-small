"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines projected gradient with Nesterov momentum, continuation on smoothness,
frequency-domain initialization, LP-based direction search, and coordinate descent polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, beta):
    """Compute gradient of 2*n*smooth_max(conv(a,a), beta) w.r.t. a."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * h


def _nesterov_projected_grad(a_init, deadline, beta_schedule, lr_init=0.01, mom_init=0.9):
    """
    Projected gradient with Nesterov momentum and adaptive learning rate.
    a_init: initial point on the simplex
    beta_schedule: list of beta values to use (decreasing)
    """
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = _score_norm(a)
    
    beta_idx = 0
    beta = beta_schedule[0]
    lr = lr_init
    mom = mom_init
    t = 0
    
    while time.monotonic() < deadline:
        # Compute gradient
        grad = _compute_grad(a, beta)
        
        # Nesterov: look-ahead
        a_look = a - mom * v
        a_look = _proj_simplex(a_look)
        grad_look = _compute_grad(a_look, beta)
        
        # Update
        v = mom * v + lr * grad_look
        a_new = _proj_simplex(a - v)
        
        # Check improvement
        score_new = _score_norm(a_new)
        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        t += 1
        
        # Adaptive learning rate
        if t % 50 == 0:
            # Reduce lr if not improving
            if t % 200 == 0:
                lr = max(1e-5, lr * 0.9)
            
        # Advance beta
        if t % 100 == 0 and beta_idx < len(beta_schedule) - 1:
            beta_idx += 1
            beta = beta_schedule[beta_idx]
            v = np.zeros(n)  # Reset momentum on beta change
            mom = max(0.85, mom * 1.02)  # Increase momentum slightly
        
        if t % 500 == 0:
            print(f"    Nesterov: t={t}, beta={beta:.6f}, lr={lr:.6f}, score={best_score:.8f}")
    
    return best_a, best_score

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses projected gradient with Nesterov momentum on the simplex, adaptive beta continuation,
diverse initializations (analytical + frequency-domain guided), and multi-phase polishing.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target_sum / n)
    rho = idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * h


def _nesterov_optimize(a_init, deadline, beta_list, lr_init, mom_init):
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)
    best_a = a.copy()

    beta_idx = 0
    beta = beta_list[0]
    lr = lr_init
    mom = mom_init
    t = 0
    stale = 0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 0.5:
            break
        grad = _compute_grad(a, beta)
        a_look = _proj_simplex(a - mom * v)
        grad_look = _compute_grad(a_look, beta)
        v = mom * v + lr * grad_look
        a_new = _proj_simplex(a - v)
        score_new = _score_norm(a_new)
        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stale = 0
        else:
            stale += 1
        a = a_new
        t += 1
        if t % 200 == 0:
            lr = max(1e-6, lr * 0.92)
        if stale > 500:
            lr = max(1e-6, lr * 0.7)

        _update(p_best, sc_best, "fine_polish")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Final micro-optimization (remaining time)
    if time.monotonic() < deadline and best_seq is not None:
        remain = deadline - time.monotonic() - 3.0
        if remain > 30:
            c_max = _full_convolve(best_seq, best_seq).max()
            beta_list_final = [b * c_max for b in [0.0005, 0.0003, 0.0001, 0.00005, 0.00002]]
            dl = time.monotonic() + remain
            p_best, sc_best = _nesterov_optimize(best_seq, dl, beta_list_final, lr_init=0.005, mom_init=0.8)
            _update(p_best, sc_best, "final_nesterov")

        if time.monotonic() < deadline - 5 and remain > 80:
            c_max2 = _full_convolve(best_seq, best_seq).max()
            dl2 = min(deadline - 3, time.monotonic() + (deadline - time.monotonic()) * 0.4)
            p_best2, sc_best2 = _fine_polish(best_seq, dl2, beta_start=0.0003 * c_max2)
            _update(p_best2, sc_best2, "final_polish")

    print(f"Phase 6 done: best={best_score:.10f}")

    # Final: try evaluate_sequence for authoritative check
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_cosine(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL internal score: {sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"FINAL evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_cosine(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
