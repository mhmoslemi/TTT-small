"""
Optimization of step function sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam smooth-max in logit space, coordinate descent, and simulated annealing.
Key insight: For non-negative sequences, the uniform sequence gives score=2 (theoretical min).
We verify this empirically and run optimizers to confirm or find improvements.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.03, beta_factor=0.05):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:

                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.8)

    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=200, step_size=0.01):
    """Coordinate descent: try adjusting each element to reduce max autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_idx = int(np.argmax(c_curr))
        max_val = float(c_curr[max_idx])

        # Randomly shuffle the order of coordinates
        order = rng.permutation(n)

        improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try increasing and decreasing
            for delta_frac in [step_size, -step_size, 2*step_size, -2*step_size, 0.5*step_size, -0.5*step_size]:
                new_val = old_val + delta_frac * a.sum() / n
                if new_val < 0:
                    continue
                a[i] = new_val
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                c_new = _fft_convolve(a)
                sc = float(2.0 * n * c_new.max() / (s_new * s_new))
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                a[i] = old_val
            a[i] = old_val

        if not improved:
            step_size *= 0.7
            if step_size < 1e-8:
                break

    return best_a


def _simulated_annealing(a, deadline, max_iter=20000, T0=0.1, T1=0.001):
    """Simulated annealing in logit space."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    best_z = z.copy()
    best_score = _score(a)
    curr_score = best_score
    rng = np.random.default_rng(67890)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iter)

        # Random perturbation in logit space
        z_new = z + T * rng.normal(size=n)
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()
        sc_new = _score(a_new)

        if sc_new < curr_score or rng.random() < np.exp(-(sc_new - curr_score) / max(T, 1e-10)):
            z = z_new
            curr_score = sc_new
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = a_new.copy()
                best_z = z_new.copy()
        else:
            # Some chance to reset to best
            if rng.random() < 0.01:
                z = best_z.copy()
                e = np.exp(z)
                curr_score = best_score
                a = e / e.sum()

    return best_a, best_score


def _subgradient_opt(a, deadline, max_iter=15000):
    """Subgradient method targeting the max coefficient directly."""
    a = np.asarray(a, dtype=np.float64).copy()

    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)

    base_lr = 0.3 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_idx = int(np.argmax(c_curr))

        # Subgradient at max
        sg = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 4 * n):
                break

        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):

        # conv(a,g) has length 2n-1, and (a*g)[k] = sum_j a[j]*g[k-j]
        # This is a matrix-vector product where A[k,j] = a[j] if k-j in [0,n-1], i.e., j in [k-n+1, k]
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):

            A[start:end, j] = a
        b = np.full(2 * n - 1, M_lp)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,

        except Exception:
            break

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:

        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
        print(f"  Final CoordDescent: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_uniform(512)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        pass  # Keep as is (score is scale-invariant)
    else:
        final_seq = final_seq * 100

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

    return [float(x) for x in final_seq]
