"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with logsumexp continuation, top-K smoothed gradient descent,

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_gradient(a, k_top=5):
    """Compute gradient using top-K lag averaging for smoother optimization."""
    n = len(a)
    c = _full_convolve(a, a)
    # Get top K indices
    k_actual = min(k_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    # Average gradient over top K lags
    grad = np.zeros(n)
    for k_val in top_idx:
        k_val = int(k_val)
        # d(a*a)_{k_val}/da_j = 2*a_{k_val - j} if 0 <= k_val-j <= n-1
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
    grad /= k_actual
    grad *= 4.0 * n
    return grad, float(c.max())


def _topk_smooth_objective(z, n, k_top, alpha):
    """Smooth objective using interpolation between top-K average and logsumexp."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    # Top-K average
    k_actual = min(k_top, len(c))
    top_vals = np.sort(c)[-k_actual:]
    topk_avg = float(top_vals.mean())
    # Logsumexp
    c_shifted = c / alpha
    c_shifted = c_shifted - np.max(c_shifted)
    ls = float(np.log(np.sum(np.exp(c_shifted)))) + np.max(c_shifted)
    # Blend
    obj = 2.0 * n * (0.5 * (topk_avg + cm) + 0.5 * ls)
    return obj


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=None, beta_scale=1.0):
    """L-BFGS with logsumexp continuation, optimized for this problem."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012, 0.0006, 0.0003, 0.00015]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(5.0, max_wall_time / len(betas))
    else:
        per_level = 60.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_topk(a_init, deadline, rng, lr=0.02, k_top=5, decay=0.9995):
    """Adam optimization with top-K smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    for it in range(200000):
        if time.monotonic() > deadline - 1.0:
            break

        t += 1
        grad, _ = _topk_gradient(a_init if t == 1 else _softmax_fn(z), k_top)
        # Recompute with current z
        a_curr = _softmax_fn(z)
        grad, _ = _topk_gradient(a_curr, k_top)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 1000 == 0:
            lr = max(0.001, lr * decay)

    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=5000):
    """Local search: randomly perturb small blocks and keep improvements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        # Choose a block to perturb
        block_size = rng.integers(3, min(20, n // 4))
        start = rng.integers(0, n - block_size)
        
        a_trial = a.copy()
        # Multiplicative perturbation
        perturbation = rng.uniform(0.85, 1.15, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            # Occasionally accept worse solutions (simulated annealing style)
            if rng.random() < 0.01 and sc < best_score * 1.01:
                a = a_trial.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(2048, 1.5)
            s = float(final_seq.sum())
        final_seq = final_seq / s  # Normalize to sum=1

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator if feasible
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(2048, 1.5)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=2048, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
