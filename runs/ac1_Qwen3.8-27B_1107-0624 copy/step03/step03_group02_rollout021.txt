            i_vals2 = np.arange(j - (k_end2 - 1), j - k_start2 + 1)
            grad[j] += np.dot(w[k_start2:k_end2], a[i_vals2])
    
    return mval + beta * np.log(s) if s > 0 else mval, 2.0 * grad if s > 0 else np.zeros(n)


def _adam_optimize(a_init, deadline, max_iter=30000, lr=0.02, beta_init=None, 
                   beta_min=1e-4, beta_decay=0.95, restart_every=3000):
    """Adam optimizer in log-space for smooth max of autocorrelation."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    if beta_init is None:
        c0 = _fft_convolve(a)
        beta_init = 0.05 * c0.max()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    beta = beta_init
    best_a = a.copy()
    best_score = 2.0 * n * _fft_convolve(a).max()
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        mval = c_curr.max()
        score_curr = 2.0 * n * mval
        
        if score_curr < best_score - 1e-14:
            best_score = score_curr
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Compute smooth max gradient
        x = (c_curr - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = ex.sum()
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c_curr)
            w[int(np.argmax(c_curr))] = 1.0
        else:
            w = ex / s_ex
        
        # Gradient via FFT
        N = _next_pow2(3 * n)
        w_pad = np.zeros(N)
        w_pad[:2*n-1] = w
        a_flipped = a[::-1].copy()
        a_pad = np.zeros(N)
        a_pad[:n] = a_flipped
        
        fw = np.fft.rfft(w_pad)
        fa = np.fft.rfft(a_pad)
        conv_result = np.fft.irfft(fw * fa, N)
        
        # Extract gradient: grad[j] = 2 * conv_result[n-1+j] ... 
        # Let me verify: conv_result[m] = sum_k w_pad[k] * a_pad[m-k]
        # a_pad[m-k] = a_flipped[m-k] = a[n-1-(m-k)] (for 0 <= m-k < n)
        # So conv_result[m] = sum_k w[k] * a[n-1-m+k] (for valid k and valid a index)
        # Let i = n-1-m+k, so k = i+m-n+1
        # conv_result[m] = sum_i w[i+m-n+1] * a[i]
        # For i = j: conv_result[m] includes w[j+m-n+1] * a[j]
        # We want grad[j] = 2 * sum_k w[k] * a[k-j] (approximately, the first term)
        # sum_k w[k]*a[k-j]: let i=k-j, = sum_i w[i+j]*a[i]
        # This equals conv_result[m] where i+m-n+1 = i+j, so m = j+n-1
        # So the first term for all j is in conv_result[n-1 : 2n-1]
        
        grad_a = 2.0 * conv_result[n-1:2*n-1]
        
        # Project to tangent space of simplex
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        grad_z -= grad_z.mean()
        
        if not np.all(np.isfinite(grad_z)):
            # Reset
            z = np.log(best_a)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_imp = 0
            continue
        
        t += 1
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * grad_z**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= z_new.mean()
        
        e = np.exp(z_new)
        a_new = e / e.sum()
        
        z = z_new
        a = a_new
        
        # Decay beta
        if it % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
    
    return best_a, best_score


def _multi_peak_subgrad(a_init, deadline, max_iter=10000, top_k=5):
    """Subgradient method targeting top-k peaks."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * _fft_convolve(a).max()
    
    base_lr = 0.5 / np.sqrt(n)
    no_imp = 0
    prev_score = best_score
    prev_grad = None
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        max_val = c_curr.max()
        score_curr = 2.0 * n * max_val
        
        if score_curr < best_score - 1e-14:
            best_score = score_curr
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 5*n):
                break
        
        # Find top-k peaks
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]
        
        # Weighted subgradient
        weights = np.exp(-0.5 * (c_curr[top_idx] - c_curr[top_idx[0]]))
        weights = weights / weights.sum()
        
        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            idx = int(idx)
            # dc[idx]/da[j] = a[idx-j] (valid) + a[j-idx] (valid)
            # First term: a[idx-j] for j in [max(0,idx-n+1), min(idx, n-1)]
            j1s = max(0, idx - n + 1)
            j1e = min(idx, n - 1)
            if j1s <= j1e:
                sg[j1s:j1e+1] += weights[i] * a[idx - np.arange(j1s, j1e+1)]
            # Second term: a[j-idx] for j in [max(idx,0), min(idx+n-1, n-1)]
            j2s = max(idx, 0)
            j2e = min(idx + n - 1, n - 1)
            if j2s <= j2e:
                sg[j2s:j2e+1] += weights[i] * a[np.arange(j2s, j2e+1) - idx]
        
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-20:
            break
        
        # Barzilai-Borwein step size
        if prev_grad is not None:
            yg = a - (a + base_lr / np.sqrt(it) * sg / sg_norm)  # approximate
            # Skip BB for simplicity, use diminishing step
            lr = base_lr / np.sqrt(it + 1)
        else:
            lr = base_lr
        
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        prev_grad = sg / sg_norm
        a = a_new
    
    return best_a, best_score


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _make_init(n, seed=0, parent=None):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    inits = []
    
    if parent is not None:
        inits.append(_interpolate(parent, n))
        # Perturbed parent
        p = parent
        p_n = _interpolate(p, n)
        for amp in [0.1, 0.3, 0.5]:
            pert = p_n * (1 + amp * rng.normal(size=n))
            pert = np.maximum(pert, 1e-10)
            inits.append(pert / pert.sum())
    
    # Linear ramp
    for eta in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # U-shaped
    for alpha in [1.0, 2.0, 5.0]:
        a = 1.0 + alpha * (2*t - 1)**2
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Cosine
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.7, 1.0]:
            a = 1.0 + amp * np.cos(2*np.pi*freq*t)
            a = np.maximum(a, 1e-10)
            inits.append(a / a.sum())
    
    # Random
    for _ in range(3):
        a = np.abs(rng.normal(size=n)) + 0.1
        inits.append(a / a.sum())
    
    return inits
