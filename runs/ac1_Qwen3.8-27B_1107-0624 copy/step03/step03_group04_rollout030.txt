def _next_pow2(n):

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

    return c

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_symmetric_init(n, rng, mode='bell'):
    if mode == 'bell':
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.5) / 0.3) ** 2)
    elif mode == 'edge':
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.0) / 0.15) ** 2) + np.exp(-((t - 1.0) / 0.15) ** 2) + 0.3
    elif mode == 'flat':
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
        a = np.maximum(a, 0.1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:

    return p / s

    b2 = 0.999
    eps = 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0

    no_imp = 0
    base_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Find top peaks
        N_peaks = min(n_peaks, len(c))
        peak_idx = np.argsort(c)[::-1][:N_peaks]
        peak_vals = c[peak_idx]
        peak_weights = peak_vals / peak_vals.sum()

        # Compute combined gradient
        vp = _grad_conv_combined(a, peak_idx.tolist(), peak_weights.tolist())
        # Add small regularization
        vp += 1e-8 * a

        # Project gradient to simplex tangent space
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            adam_t = 0
            continue

        # BB step size
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20 and abs(dot_sg) > 1e-20:
                bb_candidate = dot_ss / abs(dot_sg)
                bb_candidate = np.clip(bb_candidate, 0.1, 10.0)

        prev_z = z.copy()
        prev_grad = gz.copy()

        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)

        vh = v_adam / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.93)

        # Annealing restart
        if no_imp > max(3000, 5 * n):

            z = z_restart
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0

def _greedy_redistribute(a, deadline, max_iter=20000, n_peaks=3, strength=0.02):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Find top peaks
        Np = min(n_peaks, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:Np]

        # For each peak, compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in peak_idx:
            w = c[k] / c_max  # normalized weight
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1) + 1
            for j in range(j_start, j_end):
                pressure[j] += w * a[k - j]

        pmax = pressure.max()
        if pmax < 1e-15:
            break

        # Exponential tilting: reduce high-pressure coordinates
        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.15, str_cur * 1.005)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.005, str_cur * 0.92)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a

    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    lr = lr_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        grad = _grad_conv_k_fast(a, k_star)

        # Project to simplex tangent
        dot = float(np.dot(a, grad))
        dz = a * (grad - dot)
        dz -= float(np.mean(dz))

        if not np.all(np.isfinite(dz)):
            break

        z = np.log(a)
        z = z - lr * dz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(2000, 4 * n):
                break

    return best_a


def _coordinate_descent(a, deadline, max_iter=5000):
    """Try small coordinate moves to reduce the peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    rng = np.random.default_rng(999)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))

        # Find position with highest contribution to peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1) + 1
        if j_start >= j_end:
            break

        # Contribution at each j in range
        contribs = np.zeros(n)
        for j in range(j_start, j_end):
            contribs[j] = a[j] * a[k_star - j]

        # Source: position with highest contribution in the peak range
        src_candidates = range(j_start, j_end)
        src = max(src_candidates, key=lambda j: contribs[j])

        # Destination: random position that minimizes impact
        # Try a few random targets
        best_target = None
        best_target_score = best_score
        for _ in range(3):
            tgt = rng.integers(0, n)
            if tgt == src:
                continue
            # Move a small fraction of mass
            frac = 0.05
            delta = a[src] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[tgt] += delta
            if a_trial[src] < 0:
                continue
            s = a_trial.sum()
            a_trial = a_trial / s
            c_t = _fft_convolve(a_trial)
            sc_t = float(2.0 * n * c_t.max())
            if sc_t < best_target_score:
                best_target_score = sc_t
                a_trial_best = a_trial.copy()
                best_target = (src, tgt, frac)

        if best_target is not None and best_target_score < best_score - 1e-12:
            a = a_trial_best.copy()
            best_score = best_target_score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                a = best_a.copy()
                no_imp = 0

    return best_a


def _multi_phase_optimize(a, deadline):
    """Run multiple optimization phases in sequence on a single sequence."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    # Phase A: Peak-cycling gradient
    t_a = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.35)
    a1 = _peak_cycling_optimize(a, t_a, max_iter=30000, lr=0.04, n_peaks=5)
    sc1 = float(2.0 * n * _fft_convolve(a1).max())
    if sc1 < best_score:
        best_score = sc1
        best_a = a1.copy()

    # Phase B: Greedy redistribution
    if time.monotonic() < deadline - 10:
        t_b = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.25)
        a2 = _greedy_redistribute(best_a, t_b, max_iter=10000, n_peaks=4, strength=0.03)
        sc2 = float(2.0 * n * _fft_convolve(a2).max())
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()

    # Phase C: LP refinement
    if time.monotonic() < deadline - 5:
        t_c = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.25)
        a3 = _lp_refine(best_a, t_c, max_iter=5000, inflate=1.0008, blend=0.2, n_top_frac=0.4)
        sc3 = float(2.0 * n * _fft_convolve(a3).max())
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    # Phase D: Subgradient polish
    if time.monotonic() < deadline - 3:
        t_d = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.15)
        a4 = _subgradient_polish(best_a, t_d, max_iter=15000, lr_init=0.015)
        sc4 = float(2.0 * n * _fft_convolve(a4).max())
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()

    # Phase E: Coordinate descent
    if time.monotonic() < deadline - 2:
        t_e = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.1)
        a5 = _coordinate_descent(best_a, t_e, max_iter=2000)
        sc5 = float(2.0 * n * _fft_convolve(a5).max())
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-start optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Combines peak-cycling gradients, greedy redistribution, pruned LP, and
    subgradient polish across multiple initializations and lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 896, 1024]:
        if ln not in lengths:
            lengths.append(ln)
    # Sort by n (shorter first for quick wins, then longer)
    lengths = [l for l in lengths if l <= 1100]

    print(f"Trying lengths: {lengths}", flush=True)

    # Phase 1: Generate and evaluate initializations
    print(f"Phase 1: Initialization", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))

        # Flat spectrum
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum_init(n_t, rng)))

        # Symmetric variants
        init_pool.append(('sym_bell', n_t, _make_symmetric_init(n_t, rng, 'bell')))
        init_pool.append(('sym_edge', n_t, _make_symmetric_init(n_t, rng, 'edge')))

        # Uniform with noise
        a_noise = np.ones(n_t) * (1.0 + 0.2 * rng.standard_normal(n_t))
        a_noise = np.maximum(a_noise, 0.05)
        init_pool.append(('uni_noise', n_t, a_noise / a_noise.sum()))

    # Quick score all inits
    init_scores = []
    for name, n_t, a0 in init_pool:
        c = _fft_convolve(a0)
        sc = float(2.0 * n_t * c.max())
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 8 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:8]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Optimize top candidates with multi-phase pipeline
    # Allocate time based on number of candidates
    n_candidates = min(5, len(init_scores))
    time_per_cand = (deadline - time.monotonic() - 120.0) / n_candidates

    for i in range(n_candidates):
        if time.monotonic() > deadline - 120:
            break
        sc0, name, n_t, a0 = init_scores[i]
        cand_deadline = min(deadline - 120, time.monotonic() + time_per_cand)
        if cand_deadline - time.monotonic() < 30:
            continue

        # Skip if already clearly worse and not parent
        if sc0 > best_score * 1.3 and name != 'parent' and n_t != 768:
            continue

        print(f"Phase 2: Optimizing {name} (n={n_t}), init_score={sc0:.8f}, "
              f"budget={cand_deadline - time.monotonic():.0f}s", flush=True)

        a_opt, sc_opt = _multi_phase_optimize(a0, cand_deadline)
        sc_final = _score(a_opt)
        print(f"  Result: {sc_final:.10f}", flush=True)

        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_opt.copy()

    # Phase 3: Additional refinement on the overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"Phase 3: Additional refinement on best (score={best_score:.10f})", flush=True)

        # Try another round of LP + subgradient on the best
        refine_deadline = min(deadline - 10, time.monotonic() + 120.0)
        a_r1 = _lp_refine(best_seq, refine_deadline, max_iter=3000,
                          inflate=1.0006, blend=0.15, n_top_frac=0.3)
        sc_r1 = _score(a_r1)
        if sc_r1 < best_score:
            best_score = sc_r1
            best_seq = a_r1.copy()
            print(f"  LP refine: {sc_r1:.10f}", flush=True)

        # Subgradient polish
        if time.monotonic() < deadline - 5:
            t_sp = min(deadline - 5, time.monotonic() + 60.0)
            a_r2 = _subgradient_polish(best_seq, t_sp, max_iter=10000, lr_init=0.01)
            sc_r2 = _score(a_r2)
            if sc_r2 < best_score:
                best_score = sc_r2
                best_seq = a_r2.copy()
                print(f"  Subgrad polish: {sc_r2:.10f}", flush=True)

    # Phase 4: Multi-start perturbation
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"Phase 4: Multi-start perturbation", flush=True)
        n = best_seq.size
        best_score_ms = best_score
        best_seq_ms = best_seq.copy()

        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            n_starts = min(4, int(remaining // 40))
            for i in range(n_starts):
                if time.monotonic() > deadline - 5:
                    break
                noise_scale = 0.03 * (i + 1)
                rng_ms = np.random.default_rng(seed * 1000 + i)
                a_pert = best_seq_ms * (1.0 + noise_scale * rng_ms.standard_normal(n))
                a_pert = np.maximum(a_pert, 1e-15)
                a_pert = a_pert / a_pert.sum()

                sc_pert = _score(a_pert)
                if sc_pert > best_score_ms * 1.5:
                    continue

                # Quick polish
                sub_dl = min(deadline - 5, time.monotonic() + min(60.0, remaining / max(1, n_starts - i)))
                a_pol, sc_pol = _multi_phase_optimize(a_pert, sub_dl)
                sc_pol = _score(a_pol)
                if sc_pol < best_score_ms:
                    best_score_ms = sc_pol
                    best_seq_ms = a_pol.copy()
                    print(f"  Pert {i}: {sc_pol:.10f}", flush=True)

        if best_score_ms < best_score:
            best_score = best_score_ms
            best_seq = best_seq_ms.copy()

    # Phase 5: Final micro-optimization
    if best_seq is not None and time.monotonic() < deadline - 3:
        t_f = deadline - 3
        a_f = _subgradient_polish(best_seq, t_f, max_iter=5000, lr_init=0.005)
        sc_f = _score(a_f)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"Final micro: {sc_f:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_symmetric_init(768, rng, 'flat')
        best_score = _score(best_seq)

    # Final cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(512) / 512

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
