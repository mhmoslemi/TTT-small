def _next_pow2(n):

            a_test = a_test / s_test
            sc_test = _score(a_test)
            if sc_test < best_g_improvement:
                best_g_improvement = sc_test
                best_g = g_candidate

        if best_g is None:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > 100:
                if inflate > 0.01:
                    inflate = 0.01
                    continue
                break
            continue

        # Line search with the best direction
        best_t, best_sc = _line_search_fine(a, best_g, best_score)

        if best_t > 0 and best_sc < best_score - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * best_g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Slowly decrease inflation
            inflate = max(inflate_min, inflate * 0.995)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.05, inflate * 1.1)
            if no_imp > max(150, 2 * n):
                break

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _adam_polish(a0, deadline, max_iter=50000):
    """Adam optimizer with smooth-max gradient."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.03 * float(c.max()))
    lr = 0.02
    base_lr = lr
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    no_imp = 0
    t_step = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_step += 1

        grad, maxc = _smooth_max_grad(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            m = np.zeros(n)
            v = np.zeros(n)
            continue

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 200 == 199:
            beta = max(1e-8, beta * 0.7)

    return best_a


def _perturb_and_refine(a0, n, rng, deadline, perturb_scale=0.05):
    """Add random perturbation and briefly re-optimize."""
    a = a0.copy()
    perturbation = rng.uniform(-perturb_scale, perturb_scale, n)
    a_pert = a + perturbation * a.mean()
    a_pert = np.maximum(a_pert, 0.0)
    a_pert = _normalize(a_pert)

    # Quick LP refine
    lp_deadline = min(deadline, time.monotonic() + 30)
    if time.monotonic() < lp_deadline:
        a_new = _lp_refine(a_pert, lp_deadline, max_iter=500,
                           inflate_start=1.002, inflate_min=1.0005)
    else:
        a_new = a_pert

    sc = _score(a_new)
    return a_new, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP refinement with multi-level inflation,
    Adam gradient polish, perturbation restarts, and multiple lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define initializations at various lengths
    lengths_to_try = [640, 768, 896, 1024]

    configs = []
    for n_t in lengths_to_try:
        if parent is not None:
            configs.append(('parent', n_t, _interpolate(parent, n_t)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp_mid', n_t, _make_ramp_mid(n_t, 1.5)))
        configs.append(('bowl', n_t, _make_bowl(n_t, 0.4)))
        configs.append(('bowl2', n_t, _make_bowl(n_t, 0.6)))

    # Score and sort initializations
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    print(f"Initializations sorted by score:")
    for sc, name, n_t, _ in scored_configs[:8]:
        print(f"  {name} n={n_t}: {sc:.8f}")

    # Phase 1: LP refinement on best initializations
    for sc0, name, n_t, a0 in scored_configs:
        if time.monotonic() > deadline - 300:
            break
        if sc0 >= best_score + 0.3:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = deadline - time.monotonic() - 200
        if remaining < 20:
            break

        # Allocate more time to better initializations
        if sc0 <= best_score + 0.01:
            lp_time = min(400.0, remaining * 0.45)
        elif sc0 <= best_score + 0.1:
            lp_time = min(200.0, remaining * 0.25)
        else:
            lp_time = min(80.0, remaining * 0.12)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"Phase1 LP: {name} n={n_t} score={sc0:.8f} time={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=50000,
                          inflate_start=1.002, inflate_min=1.0003)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP score={sc_lp:.10f}")
        else:
            print(f"  -> LP score={sc_lp:.8f} (no improvement)")

    # Phase 2: Adam gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 150:
        gp_time = min(250.0, deadline - time.monotonic() - 100)
        if gp_time > 15:
            gp_end = time.monotonic() + gp_time
            print(f"Phase2 Adam: budget={gp_time:.0f}s, n={best_seq.size}")
            a_gp = _adam_polish(best_seq, gp_end, max_iter=50000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> Adam score={sc_gp:.10f}")

    # Phase 3: LP refinement again (tighter)
    if best_seq is not None and time.monotonic() < deadline - 100:
        lp_time = min(200.0, deadline - time.monotonic() - 60)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            print(f"Phase3 LP tight: budget={lp_time:.0f}s")
            a_final = _lp_refine(best_seq, lp_end, max_iter=50000,
                                 inflate_start=1.001, inflate_min=1.0002)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  -> LP tight score={sc_final:.10f}")

    # Phase 4: Perturbation restarts to escape local minima
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_pert = best_seq.size
        print(f"Phase4 Perturbation: n={n_pert}")
        for attempt in range(5):
            if time.monotonic() > deadline - 30:
                break
            a_pert, sc_pert = _perturb_and_refine(
                best_seq, n_pert, rng, deadline - 30,
                perturb_scale=0.03 * (1 + attempt * 0.3))
            if np.isfinite(sc_pert) and sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {attempt}: score={sc_pert:.10f} (improved!)")
            else:
                print(f"  Perturb {attempt}: score={sc_pert:.8f}")

    # Phase 5: Final short Adam polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        gp_time = min(60.0, deadline - time.monotonic() - 10)
        if gp_time > 5:
            gp_end = time.monotonic() + gp_time
            a_g = _adam_polish(best_seq, gp_end, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  Final Adam: score={sc_g:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(768, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(768, 1.0)
        s = final_seq.sum()
        final_seq = final_seq / s

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print(f"evaluate_sequence returned inf, trying scaled version")
                # Try different scaling
                for scale in [0.5, 1.0, 2.0, 5.0, 10.0]:
                    trial = final_seq * scale
                    if trial.max() > 1000:
                        continue
                    sc_t = float(evaluate_sequence(trial.tolist()))
                    if np.isfinite(sc_t) and sc_t < best_score:
                        best_score = sc_t
                        final_seq = trial
                        break
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
