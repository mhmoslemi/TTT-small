"""
Multi-phase optimizer for min 2n*max(conv(a,a))/sum(a)^2.
Phase 1: L-BFGS with log-sum-exp continuation (softmax parameterization)
Phase 2: Projected subgradient on simplex (exact max)
Phase 3: Multi-restart L-BFGS
Phase 4: Length interpolation and re-optimization
"""
import numpy as np
from scipy.optimize import minimize
import time

# (no closures, all top-level)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _max_conv(a):
    return float(_full_convolve(a, a).max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    zm = np.max(z)
    e = np.exp(z - zm)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(a, target=1.0):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rhe = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (cssv - target) / rhe) > 0
    if not np.any(cond):
        return np.full(n, target / n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(a - theta, 0.0)

def _grad_smooth_max(a, beta):
    """Gradient and value of smooth max of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = cm
    else:
        w = e / s
        val = cm + beta * float(np.log(s))
    a_rev = a[::-1]
    g = 2.0 * _full_convolve(w, a_rev)[n-1:2*n-1]
    return g, val

def _lbfgs_obj(z, beta, n):
    a = _softmax_fn(z)
    _, val = _grad_smooth_max(a, beta)
    return 2.0 * n * val

def _lbfgs_grad(z, beta, n):
    a = _softmax_fn(z)
    g, val = _grad_smooth_max(a, beta)
    # Chain rule: d/dz = (I - outer(a,a)) * (2n * g)
    # For softmax: d a_i / d z_j = a_i * (delta_ij - a_j)
    # So d f / d z_j = sum_i (d f / d a_i) * (d a_i / d z_j)
    # = sum_i 2n*g_i * a_i * (delta_ij - a_j)
    # = 2n * a_j * (sum_i g_i * a_i - sum_i g_i * a_i * a_j)
    # = 2n * a_j * (g·a - (g·a)*a_j)  ... wait that's not right
    
    # Let me redo: d f / d z_j = sum_i (d f / d a_i) * (d a_i / d z_j)
    # d a_i / d z_j = a_i * (delta_{ij} - a_j)
    # d f / d a_i = 2n * g_i  (where g_i is the gradient of smooth max w.r.t. a)
    # d f / d z_j = 2n * sum_i g_i * a_i * (delta_{ij} - a_j)
    # = 2n * (g_j * a_j - a_j * sum_i g_i * a_i)
    # = 2n * a_j * (g_j - dot(g, a))
    
    dot_ga = float(np.dot(g, a))
    return 2.0 * n * a * (g - dot_ga)

def _run_lbfgs(a_init, deadline, beta_start, nv_levels=8):
    """Run L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_mc = _max_conv(a_init)
    c0 = float(_full_convolve(a_init, a_init).max())
    beta0 = beta_start * c0
    
    for level in range(nv_levels):
        beta = beta0 * (0.5 ** level)
        level_deadline = min(deadline, time.monotonic() + max(10.0, (deadline - time.monotonic()) / (nv_levels - level)))
        try:
            res = minimize(_lbfgs_obj, z, args=(beta, n),
                          method='L-BFGS-B', jac=_lbfgs_grad,
                          options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12,
                                   'maxls': 20, 'maxfun': 1000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x
                a_new = _softmax_fn(z)
                mc = _max_conv(a_new)
                if mc < best_mc:
                    best_mc = mc
                    best_a = a_new.copy()
        except Exception:
            pass
        if time.monotonic() >= deadline:
            break
    return best_a, best_mc

    best_a = a.copy()

        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            g[i] = 2.0
