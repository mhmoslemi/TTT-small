    return c


def _fft_cross_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    mn = float(c.min())
    if mn < -1e-15:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix_reduced(a, M, frac=0.7, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    threshold = M * frac
    active = np.where(c_aa >= threshold)[0]
    top_idx = np.argsort(c_aa)[::-1][:top_k]
    active = np.unique(np.concatenate([active, top_idx]))
    active = np.sort(active)
    m = len(active)
    A = np.zeros((m, n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[k - j]
    b = np.full(m, M)
    return A, b


def _make_conv_matrix_full(a, M):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for row in range(2 * n - 1):
        lo = max(0, row - n + 1)
        hi = min(n, row + 1)
        for j in range(lo, hi):
            A[row, j] = a[row - j]
    b = np.full(2 * n - 1, M)
    return A, b


def _lp_spread(a, M, use_reduced=True, frac=0.7, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if use_reduced and n > 128:
        A, b = _make_conv_matrix_reduced(a, M, frac, top_k)
    else:
        A, b = _make_conv_matrix_full(a, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _quadratic_line_search(a, g, max_iter=50):
    """Exact quadratic-model line search: minimize max_k q_k(t) over t in [0,1]."""
    c_a = _fft_convolve(a)
    c_ag = _fft_cross_convolve(a, g)
    c_g = _fft_convolve(g)
    # Make lengths match (pad if needed)
    L = max(len(c_a), len(c_ag), len(c_g))
    c_a = np.concatenate([c_a, np.zeros(L - len(c_a))])
    c_ag = np.concatenate([c_ag, np.zeros(L - len(c_ag))])
    c_g = np.concatenate([c_g, np.zeros(L - len(c_g))])

    A_q = c_g - 2.0 * c_ag + c_a
    B_q = 2.0 * (c_ag - c_a)
    C_q = c_a.copy()

    # Golden section on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(A_q * c * c + B_q * c + C_q))
    fd = float(np.max(A_q * d * d + B_q * d + C_q))
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(A_q * c * c + B_q * c + C_q))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(A_q * d * d + B_q * d + C_q))
    return (lo + hi) / 2.0


def _lp_refine(a, deadline, max_iter=5000, inflate=1.0003, blend=0.2, use_line_search=True):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    infl_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * infl_cur
        g = _lp_spread(a, M_lp)
        if g is None:
            infl_cur = min(1.01, infl_cur * 1.15)
            no_imp += 5
            if no_imp > 400:
                break
            continue
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                infl_cur = min(1.01, infl_cur * 1.1)
            if no_imp > 500:
                break
            continue
        g = g / sum_g
        if use_line_search:
            t_opt = _quadratic_line_search(a, g, max_iter=40)
            if t_opt < 0.005:
                t_opt = blend_cur * 0.3
        else:
            t_opt = blend_cur
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        score_new = _score_norm(a_new)
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.02, blend_cur * 0.92)
                infl_cur = max(1.0001, infl_cur * 0.95)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                infl_cur = inflate
                no_imp = 0
        a = a_new
    return best_a


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.05, T_end=1e-6, lr=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        else:
            w = ex / s_ex
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            continue
        state_t += 1
        b1, b2 = 0.9, 0.999
        state_m = b1 * state_m + (1.0 - b1) * gz
        state_v = b2 * state_v + (1.0 - b2) * (gz * gz)
        mh = state_m / (1.0 - b1 ** state_t)
        vh = state_v / (1.0 - b2 ** state_t)
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()
        a = a_new
        c_new = _fft_convolve(a)
        score = float(2.0 * n * c_new.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.01, lr * 0.9)
            if no_imp > max(8000, 12 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state_t = 0
                state_m = np.zeros(n)
                state_v = np.zeros(n)
                lr = base_lr
                no_imp = 0
        T *= T_ratio
        T = max(T, T_end)
    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.01):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr
    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-14)[0]
        w = np.zeros(len(c))
        w[max_idx] = 1.0 / len(max_idx)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 800 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(6000, 10 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _peaking_correction(a, deadline, max_iters=500):
    """Targeted correction: reduce the dominant lag iteratively."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    N = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = int(np.argmax(c))
        # Gradient of conv(a,a)[max_idx] w.r.t. a is 2 * a[max_idx - j] (reversed)
        # Decrease a where it contributes most to the peak lag
        # a contributes to lag k via a[j]*a[k-j]
        # For the peak lag, reduce a[j] where a[j]*a[max_idx-j] is large
        contrib = np.zeros(n)
        lo_j = max(0, max_idx - n + 1)
        hi_j = min(n, max_idx + 1)
        for j in range(lo_j, hi_j):
            contrib[j] = a[max_idx - j]
        # Also account for the symmetric contribution
        contrib2 = np.zeros(n)
        for j in range(lo_j, hi_j):
            contrib2[max_idx - j] = a[j]
        grad_peak = 2.0 * (contrib + contrib2)
        # Subtract a small step in the gradient direction
        step = 0.01 * m_val / (float(np.max(grad_peak)) + 1e-15)
        a_new = a - step * grad_peak
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        score = _score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LSE gradient, LP with quadratic
    line search, subgradient polish, and peaking corrections.
    Explores multiple sequence lengths for best result.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)
    # Sort by expected difficulty (smaller first for quick wins)
    lengths.sort()

    print(f"Trying lengths: {lengths}")

    # Phase 1: LSE optimization on parent length
    n_main = lengths[0] if len(lengths) > 0 else 1024
    print(f"\n=== Phase 1: LSE on n={n_main} ===")
    a0 = None
    if parent is not None:
        a0 = _interpolate_profile(parent, n_main)
    else:
        a0 = _make_gaussian(n_main, 0.35)

    sc0 = _score(a0)
    print(f"  Initial: {sc0:.10f}")
    if sc0 < best_score:
        best_score = sc0
        best_seq = a0.copy()

    lse_end = min(t_start + 250.0, deadline - 600.0)
    if time.monotonic() < lse_end:
        a_lse = _lse_optimize(a0, lse_end, max_iter=120000, T_start=0.06, T_end=1e-6, lr=0.1)
        sc_lse = _score(a_lse)
        print(f"  After LSE: {sc_lse:.10f}")
        if sc_lse < best_score:
            best_score = sc_lse
            best_seq = a_lse.copy()
    else:
        a_lse = a0

    # Phase 2: LP refinement with quadratic line search
    print(f"\n=== Phase 2: LP refinement n={n_main} ===")
    lp_end = min(t_start + 500.0, deadline - 400.0)
    if time.monotonic() < lp_end:
        a_lp = _lp_refine(a_lse, lp_end, max_iter=8000, inflate=1.0003, blend=0.15, use_line_search=True)
        sc_lp = _score(a_lp)
        print(f"  After LP: {sc_lp:.10f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
    else:
        a_lp = a_lse

    # Phase 3: LSE polish (low temperature)
    print(f"\n=== Phase 3: LSE polish n={n_main} ===")
    lse2_end = min(t_start + 700.0, deadline - 300.0)
    if time.monotonic() < lse2_end:
        a_lse2 = _lse_optimize(a_lp, lse2_end, max_iter=80000, T_start=0.02, T_end=1e-7, lr=0.05)
        sc_lse2 = _score(a_lse2)
        print(f"  After LSE2: {sc_lse2:.10f}")
        if sc_lse2 < best_score:
            best_score = sc_lse2
            best_seq = a_lse2.copy()
    else:
        a_lse2 = a_lp

    # Phase 4: Peaking correction
    print(f"\n=== Phase 4: Peaking correction n={n_main} ===")
    peak_end = min(t_start + 850.0, deadline - 200.0)
    if time.monotonic() < peak_end:
        a_peak = _peaking_correction(a_lse2, peak_end, max_iters=2000)
        sc_peak = _score(a_peak)
        print(f"  After peaking: {sc_peak:.10f}")
        if sc_peak < best_score:
            best_score = sc_peak
            best_seq = a_peak.copy()
    else:
        a_peak = a_lse2

    # Phase 5: Final LP polish
    print(f"\n=== Phase 5: Final LP n={n_main} ===")
    lp2_end = min(t_start + 1100.0, deadline - 100.0)
    if time.monotonic() < lp2_end:
        a_lp2 = _lp_refine(a_peak, lp2_end, max_iter=6000, inflate=1.0001, blend=0.08, use_line_search=True)
        sc_lp2 = _score(a_lp2)
        print(f"  After LP2: {sc_lp2:.10f}")
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()

    # Phase 6: Try a different length if time permits
    if len(lengths) > 1 and time.monotonic() < deadline - 150:
        n_alt = lengths[1]
        print(f"\n=== Phase 6: Try n={n_alt} ===")
        if parent is not None:
            a_alt = _interpolate_profile(parent, n_alt)
        else:
            a_alt = _make_raised_cosine(n_alt)
        sc_alt = _score(a_alt)
        print(f"  Initial alt: {sc_alt:.10f}")
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()

        alt_end = deadline - 50.0
        remaining = alt_end - time.monotonic()
        if remaining > 60:
            # Quick LSE then LP
            lse_alt_end = time.monotonic() + min(120.0, remaining * 0.5)
            a_alt_lse = _lse_optimize(a_alt, lse_alt_end, max_iter=60000, T_start=0.04, T_end=1e-5, lr=0.08)
            sc_alt_lse = _score(a_alt_lse)
            print(f"  Alt LSE: {sc_alt_lse:.10f}")
            if sc_alt_lse < best_score:
                best_score = sc_alt_lse
                best_seq = a_alt_lse.copy()

            lp_alt_end = min(alt_end, time.monotonic() + min(150.0, remaining * 0.4))
            if time.monotonic() < lp_alt_end:
                a_alt_lp = _lp_refine(a_alt_lse, lp_alt_end, max_iter=4000, inflate=1.0002, blend=0.12, use_line_search=True)
                sc_alt_lp = _score(a_alt_lp)
                print(f"  Alt LP: {sc_alt_lp:.10f}")
                if sc_alt_lp < best_score:
                    best_score = sc_alt_lp
                    best_seq = a_alt_lp.copy()

    # Phase 7: Final subgradient polish on overall best
    print(f"\n=== Phase 7: Final subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        sp_end = deadline - 5.0
        remaining = sp_end - time.monotonic()
        if remaining > 20:
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=50000, lr=0.005)
            sc_sp = _score(a_sp)
            print(f"  Subgradient: {sc_sp:.10f}")
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0
        best_score = _score(final_seq)

    # Validate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    elapsed = time.monotonic() - t_start
    print(f"Elapsed: {elapsed:.1f}s")
    return [float(x) for x in final_seq]
