2. **Multi-resolution approach**: Optimize at n=512 first, then use that as initialization for n=1024, then n=2048. The smaller problem is easier to solve globally.

3. **Peak-targeted coordinate descent**: After each gradient step, identify the top few peaks in the convolution and perform coordinate updates specifically to reduce them.

4. **Better smoothing schedule**: Use a more aggressive annealing schedule for beta, with restarts.

5. **Ensemble approach**: Run multiple independent optimizations with different initializations and keep the best.

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)

        return np.ones_like(z) / len(z)

    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1)).sum()
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho-1] - 1) / rho
    result = np.maximum(v - theta, 0)
    return result

def _lbfgs_smooth(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _lbfgs_smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _run_lbfgs(a_init, deadline, rng, betas=None):
    """Run L-BFGS with continuation on beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if betas is None:
        betas = [max_c0 * f for f in [0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        try:
            result = minimize(
                _lbfgs_smooth, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_smooth_grad,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,

                z = result.x
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass

def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    n = len(a)
    if n == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new

def _make_bell(n, width):
    """Create a bell-shaped profile."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = a / a.sum()
    return a

def _make_cosine(n):
    """Cosine window."""
    t = np.linspace(0, 1, n)
    a = np.cos(np.pi * (t - 0.5)) ** 2
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a

def _peak_shave(a, n_iter=1000, deadline=None):
    """Iteratively reduce the peak of the convolution by adjusting coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    rng_local = np.random.default_rng(42)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        k_peak = int(np.argmax(c))
        
        # Find the coordinate that most reduces c[k_peak]
        # dc[k]/da[i] = 2*a[k-i] (for 0<=k-i<n)
        # To reduce c[k_peak], we should decrease a[i] where a[k_peak-i] is large
        # And increase a[i] where a[k_peak-i] is small
        
        # Simple approach: find the i that maximizes a[k_peak-i] (if in range)
        # and the j that minimizes a[k_peak-j] (if in range), then move mass from i to j
        if 0 <= k_peak < n:
            # k_peak - i in [0,n-1] means i in [k_peak-(n-1), k_peak]
            # which is always valid for i in [0,n-1] when k_peak in [0,n-1]
            idx = np.arange(n)
            a_shifted = a[(k_peak - idx) % n]  # This is wrong for the boundary
            # Correct: a[k_peak - i] is valid only when 0 <= k_peak-i < n
            valid = (k_peak - idx >= 0) & (k_peak - idx < n)
            a_shifted_valid = np.where(valid, a[k_peak - idx[valid]], 0)
            
            # Move mass from high a_shifted to low a_shifted
            i_src = int(np.argmax(a_shifted_valid))
            low_candidates = a_shifted_valid < np.median(a_shifted_valid)
            i_dst = int(np.random.choice(np.where(low_candidates)[0]))
        elif n <= k_peak < 2*n-1:
            # k_peak - i in [0,n-1] means i in [k_peak-(n-1), k_peak]
            # k_peak >= n, so i >= k_peak-(n-1) >= 1
            # i <= min(n-1, k_peak) = n-1
            i_min = max(0, k_peak - (n-1))
            i_max = min(n-1, k_peak)
            idx = np.arange(i_min, i_max+1)
            a_shifted_valid = a[k_peak - idx]
            
            i_src = i_min + int(np.argmax(a_shifted_valid))
            low_mask = a_shifted_valid < np.median(a_shifted_valid)
            if low_mask.sum() > 0:
                i_dst = int(np.random.choice(np.where(low_mask)[0])) + i_min
            else:
                i_dst = i_min + int(np.argmin(a_shifted_valid))
        else:
            continue
        
        # Move a small amount of mass
        delta = 0.01 * a[i_src]
        a[i_src] -= delta
        a[i_dst] += delta
        a = _project_simplex(a)
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # Revert
            a[i_src] += delta
            a[i_dst] -= delta
            a = _project_simplex(a)
    
    return best_a, best_score
