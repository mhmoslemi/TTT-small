"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Power-law and parent-sequence initialization
2. L-BFGS with logsumexp continuation
3. Top-k gradient polishing for final refinement
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _power_profile(n, p):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i**p
    return a / a.sum()

def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))

def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _topk_grad(z, n, k=3):
    """Gradient using top-k averaging instead of logsumexp."""
    a = _softmax(z)
    c = _full_convolve(a)
    # Get top-k indices
    top_k = np.argsort(c)[-k:]
    w = np.zeros_like(c)
    w[top_k] = 1.0 / k
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _adam_optimize(a_init, deadline, lr=0.1, use_topk=False, k=3):
    """Adam optimization in log-softmax space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    patience = 5000
    no_improve = 0
    
    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        
        if use_topk:
            grad = _topk_grad(z, n, k)
        else:
            c = _full_convolve(a)
            cm = c.max()
            beta = max(1e-4 * cm, 1e-8)
            shifted = (c - cm) / beta
            e = np.exp(shifted)
            s = e.sum()
            w = e / s if s > 1e-300 else np.zeros_like(c)
            if s <= 1e-300:
                w[int(np.argmax(c))] = 1.0
            W = float(np.dot(w, c))
            a_rev = a[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            grad = 4.0 * n * a * (h - W)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax(z)
        
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            lr = max(0.001, lr * 0.5)
            no_improve = 0
    
    return best_a, best_score

def _lbfgs_run(z, beta_schedule, n, deadline):
    """Run L-BFGS with continuation over beta schedule."""
    best_z = z.copy()
    best_score = _score_norm(_softmax(z))
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:

                z = z_new.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using power-law initialization with
theoretically-motivated exponents (p ~ log2(n)/2), L-BFGS with logsumexp
continuation, multi-restart exploration, and top-k gradient polishing.
Key insight: optimal non-negative profiles are steep power laws, not linear/quadratic.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _power_profile(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    return a / a.sum()


def _exp_profile(n, alpha):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * i / n)
    return a / a.sum()


def _gauss_tail_profile(n, center_frac, width):
    i = np.arange(n, dtype=np.float64)
    t = i / n
    a = np.exp(-((t - center_frac) ** 2) / (2 * width * width))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _lbfgs_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _lbfgs_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_continuation(z_init, beta_schedule, n, deadline):

    best_z = z.copy()

    best_score = _score_norm(best_a)

    p0 = np.asarray(p0, dtype=np.float64).ravel()

    return p / p.sum()

                cm = float(c.max())
                beta_schedule = [f * cm for f in [0.03, 0.01, 0.005, 0.001, 0.0005, 0.0002]]
                dl = min(phase5_end - 10, time.monotonic() + 150.0)
                a_best, sc_best, z_best = _run_lbfgs_continuation(z, beta_schedule, target_n, dl)
                update(a_best, f"lbfgs_interp_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Top-k gradient final polish
    print("=== Phase 6: Top-k final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        phase6_end = min(deadline, t_start + 1700.0)
        n = len(best_seq)

        # Top-3 polish
        dl1 = min(phase6_end - 30, time.monotonic() + 100.0)
        a_best, sc_best = _adam_topk(best_seq, dl1, lr=0.03, k=3)
        update(a_best, "topk3_polish")

        # Top-2 polish
        if time.monotonic() < phase6_end - 20:
            dl2 = min(phase6_end - 10, time.monotonic() + 60.0)
            a_best2, sc_best2 = _adam_topk(best_seq, dl2, lr=0.01, k=2)
            update(a_best2, "topk2_polish")

    print(f"Phase 6 done: best={best_score:.10f}")

    # Phase 7: Final ultra-fine L-BFGS
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        c = _full_convolve(best_seq)
        cm = float(c.max())
        beta_schedule = [f * cm for f in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
        dl = min(deadline - 15, time.monotonic() + 50.0)
        a_best, sc_best, z_best = _run_lbfgs_continuation(z, beta_schedule, n, dl)
        update(a_best, "final_lbfgs")

    # Finalize and validate
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _power_profile(512, 3.0)

        sc_final = _score_fn(final)
        print(f"FINAL score_fn: {sc_final:.10f}, n={len(final)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _power_profile(512, 3.0)
    sc_fb = _score_norm(fallback)
    print(f"FALLBACK: n=512, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
