"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses verified gradient of log-sum-exp smoothed max, L-BFGS continuation,
subgradient recovery, Adam polishing, and multi-length exploration.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_h_correlation(a, w, n):
    h = np.zeros(n)
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    Wf = np.fft.rfft(w, N)
    H = np.fft.irfft(np.conj(A) * Wf, N)
    return H[:n].copy()


def _smooth_max_val(c, beta):
    cm = float(c.max())
    c_shifted = np.minimum((c - cm) / beta, 50.0)
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return cm + beta * log_sum


def _smooth_max_weights(c, beta):
    cm = float(c.max())
    c_shifted = np.minimum((c - cm) / beta, 50.0)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _obj_smooth(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _smooth_max_val(c, beta)


def _grad_smooth(z, beta, n):

    c = _full_convolve(a, a)
    w = _smooth_max_weights(c, beta)
    W = float(np.dot(w, c))
    h = _compute_h_correlation(a, w, n)
    grad_z = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad_z)):

    return grad_z


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):

    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def _subgradient_step(a, eta):

    c = _full_convolve(a, a)
    max_c = float(c.max())
    tol = 1e-6 * max(max_c, 1e-15)

def _interp_profile(p0, n_new):

    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _run_lbfgs(a_init, deadline, beta_start_frac, n_beta_levels, per_level_max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_list = []
    for i in range(n_beta_levels):
        beta = beta_start_frac * max_c0 * (0.5 ** i)
        beta_list.append(max(beta, 1e-6 * max_c0))

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / max(1, n_beta_levels))
        try:
            result = minimize(
                _obj_smooth, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_smooth,
                options={'maxiter': per_level_max_iter, 'ftol': 1e-16,
                         'gtol': 1e-10, 'maxls': 40, 'maxfun': per_level_max_iter * 4}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score

        W = float(np.dot(w, c))
        h = _compute_h_correlation(a, w, n)
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t_step)
        vh = v / (1.0 - b2 ** t_step)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -55.0, 55.0)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t_step % 500 == 0:
            beta = max(1e-8, beta * 0.9)

        if t_step % 1000 == 0:
            if best_score < 1.51:
                lr = max(0.005, lr * 0.95)

    return best_a, best_score


def _subgradient_recovery(a_init, deadline, n_iter, eta_scale=0.5):
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    n = len(a)
    c0 = _full_convolve(a, a)
    max_c0 = float(c0.max())

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        eta = eta_scale * max_c0 / np.sqrt(it + 1)
        a = _subgradient_step(a, eta)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 1024
        parent = _make_hann_profile(n)
        print(f"No parent, using Hann profile n={n}")

    # Phase 0: Evaluate initial candidates
    print("Phase 0: Initial candidates...")
    phase0_end = min(deadline, t_start + 20.0)
    candidates = []

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        candidates.append((sc_p, parent.copy(), len(parent), "parent"))
        update_best(parent, sc_p, "parent")

    for sn in [512, 1024, 2048]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            ("lin1.0", _make_linear_profile(sn, 1.0)),
            ("lin1.5", _make_linear_profile(sn, 1.5)),
            ("lin2.0", _make_linear_profile(sn, 2.0)),
            ("hann", _make_hann_profile(sn)),
            ("blackman", _make_blackman_profile(sn)),
            ("gauss0.25", _make_gaussian_profile(sn, 0.25)),
            ("gauss0.35", _make_gaussian_profile(sn, 0.35)),
            ("quad1.0_0.2", _make_quadratic_profile(sn, 1.0, 0.2)),
            ("quad1.5_-0.3", _make_quadratic_profile(sn, 1.5, -0.3)),
            ("cubic1.0_-0.4", _make_cubic_profile(sn, 1.0, -0.4)),
            ("cubic1.5_-0.6", _make_cubic_profile(sn, 1.5, -0.6)),
        ]
        for name, prof in profiles:
            sc = _score_norm_fn(prof)
            candidates.append((sc, prof, sn, f"{name}_n{sn}"))
            update_best(prof, sc, f"{name}_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # Phase 1: L-BFGS continuation on top candidates
    print("Phase 1: L-BFGS continuation...")
    phase1_end = min(deadline, t_start + 500.0)
    count = 0
    seen = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 5:
            break
        if sc > best_score * 1.08:
            break
        key = (pn, label[:10])
        if key in seen:
            continue
        seen.add(key)

        per_time = min(100.0, max(20.0, (phase1_end - time.monotonic())))
        p_best, sc_best = _run_lbfgs(
            p, time.monotonic() + per_time,
            beta_start_frac=0.1, n_beta_levels=8, per_level_max_iter=300
        )
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # Phase 2: Multi-restart L-BFGS
    print("Phase 2: Multi-restart L-BFGS...")
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = max(10.0, (phase2_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase2_end - 10:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.05 + 0.03 * i
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase2_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _run_lbfgs(
                p_start, per_dl,
                beta_start_frac=0.05, n_beta_levels=6, per_level_max_iter=200
            )
            update_best(p_best, sc_best, f"restart{i}_n{n_best}")

        print(f"Phase 2 done: best={best_score:.10f}")

    # Phase 3: Subgradient recovery + L-BFGS
    print("Phase 3: Subgradient recovery...")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 30, time.monotonic() + 80.0)
        p_sub, sc_sub = _subgradient_recovery(best_seq, per_dl, n_iter=2000, eta_scale=0.3)
        update_best(p_sub, sc_sub, "subgrad")

        if time.monotonic() < phase3_end - 10:
            p_best, sc_best = _run_lbfgs(
                best_seq, min(phase3_end - 5, time.monotonic() + 60.0),
                beta_start_frac=0.03, n_beta_levels=5, per_level_max_iter=300
            )
            update_best(p_best, sc_best, "lbfgs_phase3")

        print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Adam polish
    print("Phase 4: Adam polish...")
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 10, time.monotonic() + 250.0)
        p_best, sc_best = _adam_polish(
            best_seq, per_dl, lr=0.03, beta_frac=0.005, n_max_iter=50000
        )
        update_best(p_best, sc_best, "adam_polish")
        print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Multi-length exploration
    print("Phase 5: Multi-length exploration...")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        target_lengths = []
        if n_curr < 512:
            target_lengths = [512, 1024]
        elif n_curr < 1024:
            target_lengths = [1024, 2048]
        elif n_curr < 2048:
            target_lengths = [2048, 4096]
        else:
            target_lengths = [2048]

        for tn in target_lengths:
            if time.monotonic() > phase5_end - 60:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            per_dl = min(phase5_end - 30, time.monotonic() + 120.0)
            p_best, sc_best = _run_lbfgs(
                p_interp, per_dl,
                beta_start_frac=0.05, n_beta_levels=6, per_level_max_iter=250
            )
            update_best(p_best, sc_best, f"lbfgs_n{tn}")

            if time.monotonic() < phase5_end - 20:
                per_dl2 = min(phase5_end - 10, time.monotonic() + 60.0)
                p_best2, sc_best2 = _adam_polish(
                    best_seq if len(best_seq) == tn else p_best,
                    per_dl2, lr=0.02, beta_frac=0.003, n_max_iter=20000
                )
                update_best(p_best2, sc_best2, f"adam_n{tn}")

        print(f"Phase 5 done: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # Phase 6: Final fine polish
    print("Phase 6: Final polish...")
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        remaining = phase6_end - time.monotonic()
        if remaining > 30:
            per_dl = min(phase6_end - 15, time.monotonic() + min(60.0, remaining * 0.5))
            p_best, sc_best = _run_lbfgs(
                best_seq, per_dl,
                beta_start_frac=0.01, n_beta_levels=4, per_level_max_iter=400
            )
            update_best(p_best, sc_best, "final_lbfgs")

        if time.monotonic() < phase6_end - 10:
            per_dl2 = min(phase6_end - 5, time.monotonic() + min(40.0, (phase6_end - time.monotonic()) * 0.7))
            p_best2, sc_best2 = _adam_polish(
                best_seq, per_dl2, lr=0.01, beta_frac=0.001, n_max_iter=30000
            )
            update_best(p_best2, sc_best2, "final_adam")

        # One more subgradient + L-BFGS combo
        if time.monotonic() < phase6_end - 5:
            per_dl3 = min(phase6_end - 2, time.monotonic() + 15.0)
            p_sub2, sc_sub2 = _subgradient_recovery(best_seq, per_dl3, n_iter=500, eta_scale=0.2)
            if sc_sub2 < best_score:
                update_best(p_sub2, sc_sub2, "final_subgrad")
                # Quick L-BFGS after subgrad
                if time.monotonic() < phase6_end - 2:
                    p_f, sc_f = _run_lbfgs(
                        best_seq, phase6_end,
                        beta_start_frac=0.005, n_beta_levels=3, per_level_max_iter=200
                    )
                    update_best(p_f, sc_f, "final_lbfgs2")

        print(f"Phase 6 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_hann_profile(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_hann_profile(512)
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
