def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max() / (p.sum()**2))

def _score_normalized(p):
    """Score assuming p is normalized to sum=1."""
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _topk_smooth_max(c, k, beta):
    """Compute smooth max using top-k elements with softmax."""
    c = np.asarray(c, dtype=np.float64)
    if k >= c.size:
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -k)[-k:]
        idx = idx[np.argsort(c[idx])[::-1]]
    top_k = c[idx]
    m = top_k.max()
    x = (top_k - m) / beta
    ex = np.exp(x)
    s = ex.sum()
    val = m + beta * np.log(s)
    w_full = np.zeros_like(c)
    w_full[idx] = ex / s
    return val, w_full

def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _adam_optimize(p0, deadline, rng, lr=0.05, top_k=10, beta_init=0.05,
                   beta_decay=0.95, beta_min=0.005, no_imp_patience=200,
                   noise_scale=0.1):

    p = np.asarray(p0, dtype=np.float64).ravel()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    
    z = np.log(p)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    c = _full_convolve(p)
    maxc = c.max()
    beta = max(beta_min, beta_init * maxc)

    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _topk_smooth_max(c, top_k, beta)
        g = _conv_cross_corr(w, p)
        
        # Gradient in z-space (mirror descent)
        dot = np.dot(p, g)
        gz = p * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0

        z -= z.mean()
        p = np.exp(z)
        p = p / p.sum()
        
        c = _full_convolve(p)
        maxc = c.max()
        score = 2.0 * n * maxc
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive beta
        if t % 20 == 0:
            beta = max(beta_min, beta * beta_decay)
            # Adjust based on gap
            sorted_c = np.sort(c)[::-1]
            if len(sorted_c) > top_k:
                gap = sorted_c[top_k-1] - sorted_c[top_k]
                if gap < beta * 0.5:

            no_imp = 0
        
        # Learning rate decay

    return best_p, best_score

def _lp_refine(a, deadline, t_blend=0.3, max_iter=30):
    """LP-based refinement: find dual sequence and blend."""
    n = len(a)
    a = a / a.sum()
    c = _full_convolve(a)
    M = c.max()
    best = a.copy()
    best_sc = 2.0 * n * M
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # LP: maximize sum(b) s.t. (a*b)_k <= M, b >= 0
        # Build constraint matrix (sparse)
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = np.arange(j_lo, j_hi + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2*n-1)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': 5000})
        
        if not res.success:
            break
        
        b = np.maximum(res.x, 0)
        s_b = b.sum()
        if s_b < 1e-10:
            break
        
        b_norm = b / s_b
        blend = (1 - t_blend) * best + t_blend * b_norm
        blend = np.maximum(blend, 1e-12)
        blend = blend / blend.sum()
        
        c_new = _full_convolve(blend)
        sc_new = 2.0 * n * c_new.max()
        
        if sc_new < best_sc:
            best_sc = sc_new
            best = blend.copy()
        
        a = blend
        c = _full_convolve(a)
        M = c.max()
        t_blend = max(0.1, t_blend * 0.92)
    
    return best, best_sc
