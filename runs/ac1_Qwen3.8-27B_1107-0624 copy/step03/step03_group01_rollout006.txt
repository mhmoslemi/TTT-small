def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) is normalized."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):

def _lbfgs_optimize(z_init, beta, n, maxiter=300):
    """Run L-BFGS-B on the smooth objective for a given beta."""
    
    def objective(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if beta < 1e-15:
            return 2.0 * n * cm
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)
    
    def gradient(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if beta < 1e-15:
            idx = np.argmax(c)
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                idx = np.argmax(c)
                w = np.zeros_like(c)
                w[idx] = 1.0
            else:
                w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            return np.zeros_like(z)
        return grad
    
    result = minimize(
        objective, z_init, method='L-BFGS-B', jac=gradient,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                 'maxls': 50, 'maxfun': 5000}
    )
    return result.x if result.x is not None else z_init


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: adaptive Nesterov descent, L-BFGS continuation,
    and Adam polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 2048
        parent = None
        print(f"No parent, will generate profiles at n={n}")
    
    # === Phase 0: Generate and evaluate initial candidates (30s) ===
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []
    
    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        candidates.append((sc_parent, parent.copy(), "parent"))
        update_best(parent, sc_parent, "parent")
    
    # Generate diverse profiles
    profiles = _make_profiles(n)
    for idx, p in enumerate(profiles):
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    # Add some random profiles
    for idx in range(5):
        p = np.abs(rng.normal(1.0, 0.5, size=n))
        p = p / p.sum()
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"rand{idx}"))
        update_best(p, sc, f"rand{idx}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best={best_score:.10f}, top-5 scores: {[f'{s:.6f}' for s,_,_ in candidates[:5]]}")
    
    if time.monotonic() > phase0_end:
        pass
    
    # === Phase 1: Nesterov smooth descent on top candidates (400s) ===
    phase1_end = min(deadline, t_start + 430.0)
    top_k = min(5, len(candidates))
    n_trials = 0
    
    beta_base = 0.05
    for trial_idx in range(top_k):
        if time.monotonic() > phase1_end:
            break
        
        sc_init, p_init, label = candidates[trial_idx]
        if sc_init > best_score * 1.02:
            break
        
        per_time = min(80.0, (phase1_end - time.monotonic()) / max(1, top_k - trial_idx))
        z_init = np.log(np.maximum(p_init, 1e-15))
        z_init = z_init - z_init.mean()
        
        # Adaptive beta schedule
        c_init = _full_convolve(p_init, p_init)
        max_c_init = float(c_init.max())
        
        # Create adaptive beta schedule: start large, end small
        n_betas = 12
        betas = [beta_base * max_c_init * (0.5 ** (i * 3.0 / n_betas)) for i in range(n_betas)]
        # Ensure betas are decreasing
        for i in range(1, len(betas)):
            betas[i] = min(betas[i], betas[i-1] * 0.6)
        betas = [max(b, 1e-8) for b in betas]
        
        p_best, sc_best = _nesterov_smooth_descent(z_init, betas, t_start + per_time + (time.monotonic() - t_start) + per_time * 0, momentum=0.85, lr_init=0.08)
        # Fix: proper deadline
        p_best, sc_best = _nesterov_smooth_descent(z_init, betas, time.monotonic() + per_time, momentum=0.85, lr_init=0.08)
        
        update_best(p_best, sc_best, f"nesterov_{label}")
        n_trials += 1
    
    print(f"Phase 1: best={best_score:.10f}")
    
    # === Phase 2: L-BFGS continuation on best (400s) ===
    phase2_end = min(deadline, t_start + 830.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        n_best = len(best_seq)
        z_best = np.log(np.maximum(best_seq, 1e-15))
        z_best = z_best - z_best.mean()
        
        c_best = _full_convolve(best_seq, best_seq)
        max_c_best = float(c_best.max())
        
        # L-BFGS with continuation
        beta_starts = [0.10, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002]
        betas = [b * max_c_best for b in beta_starts]
        
        per_level = (phase2_end - time.monotonic()) / len(betas)
        z_curr = z_best.copy()
        
        for i, beta in enumerate(betas):
            if time.monotonic() > phase2_end - 5:
                break
            level_dl = min(phase2_end - 5, time.monotonic() + max(5.0, per_level))
            
            try:
                z_new = _lbfgs_optimize(z_curr, beta, n_best, maxiter=200)
                if np.all(np.isfinite(z_new)):
                    a_new = _softmax_fn(z_new)
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        update_best(a_new, sc_new, f"lbfgs_beta{i}")
                        z_curr = z_new.copy()
                    else:
                        z_curr = z_new.copy()  # Continue from here anyway
            except Exception as ex:
                print(f"  L-BFGS error at beta {i}: {ex}")
                break
        
        print(f"Phase 2: best={best_score:.10f}")
    
    # === Phase 3: Multi-restart Nesterov (300s) ===
    phase3_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        n_best = len(best_seq)
        n_restarts = 8
        
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break
            
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            
            # Different perturbation strategies
            if i < 2:
                noise = 0.05 + 0.02 * i
                z = z + noise * rng.normal(size=n_best)
            elif i < 4:
                # Perturb a random subset
                mask = rng.random(n_best) < 0.3
                z[mask] += 0.2 * rng.normal(size=int(mask.sum()))
            elif i < 6:
                # Smooth perturbation (low-freq)
                freqs = np.fft.rfft(np.zeros(n_best))
                freqs[1:5] = rng.normal(size=4)
                pert = np.fft.irfft(freqs, n_best)
                z = z + 0.1 * pert
            else:
                # Symmetry-breaking perturbation
                z = z + 0.1 * rng.normal(size=n_best)
                z_mirror = z[::-1]
                z = 0.7 * z + 0.3 * z_mirror
            
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            c_start = _full_convolve(p_start, p_start)
            max_c_start = float(c_start.max())
            
            betas = [0.05 * max_c_start * (0.5 ** (j * 3.0 / 10)) for j in range(10)]
            for j in range(1, len(betas)):
                betas[j] = min(betas[j], betas[j-1] * 0.6)
            betas = [max(b, 1e-8) for b in betas]
            
            per_dl = time.monotonic() + min(30.0, (phase3_end - time.monotonic()) / max(1, n_restarts - i))
            p_best, sc_best = _nesterov_smooth_descent(z, betas, per_dl, momentum=0.85, lr_init=0.06)
            update_best(p_best, sc_best, f"restart{i}")
        
        print(f"Phase 3: best={best_score:.10f}")
    
    # === Phase 4: Adam polishing (250s) ===
    phase4_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        z_best = np.log(np.maximum(best_seq, 1e-15))
        z_best = z_best - z_best.mean()
        
        # Run Adam with different hyperparameters
        configs = [
            {'lr': 0.05, 'beta_frac': 0.01, 'max_iter': 50000},
            {'lr': 0.03, 'beta_frac': 0.005, 'max_iter': 50000},
            {'lr': 0.02, 'beta_frac': 0.003, 'max_iter': 50000},
        ]
        
        for cfg_idx, cfg in enumerate(configs):
            if time.monotonic() > phase4_end - 10:
                break
            per_dl = min(phase4_end - 5, time.monotonic() + 75.0)
            p_best, sc_best = _adam_descent(z_best, per_dl, **cfg)
            update_best(p_best, sc_best, f"adam_cfg{cfg_idx}")
            if sc_best < best_score:
                z_best = np.log(np.maximum(p_best, 1e-15))
                z_best = z_best - z_best.mean()
        
        print(f"Phase 4: best={best_score:.10f}")
    
    # === Phase 5: Try larger length (150s) ===
    phase5_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase5_end and len(best_seq) < 4096:
        target_n = min(4096, len(best_seq) * 2)
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{target_n}")
        
        if sc_interp < best_score * 1.01:
            z_interp = np.log(np.maximum(p_interp, 1e-15))
            z_interp = z_interp - z_interp.mean()
            c_interp = _full_convolve(p_interp, p_interp)
            max_c_interp = float(c_interp.max())
            betas = [0.05 * max_c_interp * (0.5 ** (j * 3.0 / 8)) for j in range(8)]
            for j in range(1, len(betas)):
                betas[j] = min(betas[j], betas[j-1] * 0.6)
            betas = [max(b, 1e-8) for b in betas]
            
            per_dl = min(phase5_end - 10, time.monotonic() + 100.0)
            p_best, sc_best = _nesterov_smooth_descent(z_interp, betas, per_dl, momentum=0.85, lr_init=0.06)
            update_best(p_best, sc_best, f"nesterov_n{target_n}")
        
        print(f"Phase 5: best={best_score:.10f}")
