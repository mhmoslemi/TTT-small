```python
def _project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-10:
        c = c - mn
    return c

    if s < 1e-10:
        return np.inf

    return float(2.0 * n * c.max() / (s * s))

    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n-1:2*n-1]
    
    # Now convert to score gradient: score = 2n * LSE / S^2
    # d(score)/d(a_i) = 2n * (grad_i * S^2 - LSE * 2 * S) / S^4
    #                  = 2n * (grad_i * S - 2 * LSE) / S^3
    score_grad = 2.0 * n * (grad * S - 2.0 * LSE) / (S ** 3)
    return float(2.0 * n * LSE_val / (S * S)), score_grad

def _project_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _make_conv_matrix(a, M, top_k=100):
    """Create constraint matrix for LP: conv(a, g) <= M for active lags."""
    n = len(a)
    conv_aa = _fft_convolve(a)
    # Get top_k lags
    idx = np.argsort(conv_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b

def _lp_shave(a, M, max_iter=500):
    """
    LP-based peak shaving: iteratively find a direction that reduces the peak.
    At each step, solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then blend: a <- a + t*(g - a) and renormalize.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_score = 2.0 * n * _fft_convolve(a).max()
    
    for it in range(max_iter):
        conv = _fft_convolve(a)
        M_cur = float(conv.max())
        
        # Solve LP: max sum(g) s.t. conv(a,g) <= M_cur * inflate, g >= 0
        inflate = 1.0 + 0.01 / (1 + it * 0.1)
        M_lp = M_cur * inflate
        
        A, b = _make_conv_matrix(a, M_lp, top_k=80)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        
        if sum_g <= 1.0 + 1e-8:
            break
        
        g = g / sum_g
        
        # Line search for blend factor
        best_t = 0.0
        best_sc = float(2.0 * n * _fft_convolve(a).max())
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = np.maximum(a + t * (g - a), 0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            sc = float(2.0 * n * _fft_convolve(a_trial).max())
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            break
        
        a = np.maximum(a + best_t * (g - a), 0)
        a = a / a.sum()
        
        sc = float(2.0 * n * _fft_convolve(a).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
    
    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new

def _subgradient_step(a, step_size):
    """One subgradient step for the non-smooth objective (sum=1, a>=0)."""
    n = len(a)
    conv = _fft_convolve(a)
    m_val = conv.max()
    # Find lags at or near the max
    threshold = m_val - 1e-10
    max_idx = np.where(conv >= threshold)[0]
    w = np.zeros_like(conv)
    w[max_idx] = 1.0 / len(max_idx)
    
    # Gradient of conv_k w.r.t. a: 2 * a reversed/shifted
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n-1:2*n-1]
    
    # Project gradient onto tangent space of simplex
    grad_tan = grad - np.dot(a, grad)
    
    a_new = a - step_size * grad_tan
    a_new = _project_simplex(a_new)
    return a_new
```

```python
"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B on LSE-smoothed objective, momentum subgradient, and iterative
LP peak shaving. Explores n=1024 and n=2048 with diverse initializations.
"""

import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-10:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _lse_obj(z, n, T):
    """LSE-smoothed objective in logit space."""
    ez = np.exp(z - z.max())
    a = ez / ez.sum()
    conv = _fft_convolve(a)
    x = conv / T
    x = x - x.max()
    val = T * np.log(np.exp(x).sum())
    return 2.0 * n * val


def _lse_obj_grad(z, n, T):
    """Gradient of LSE-smoothed objective in logit space."""
    ez = np.exp(z - z.max())
    a = ez / ez.sum()
    conv = _fft_convolve(a)
    x = conv / T
    x = x - x.max()
    ex = np.exp(x)
    s_ex = ex.sum()
    w = ex / s_ex

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad_a = 2.0 * h_full[n - 1:2 * n - 1]

    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    return 2.0 * n * grad_z


def _subgradient_step(a, step_size):
    """One subgradient step for non-smooth objective on simplex."""
    n = len(a)
    conv = _fft_convolve(a)
    m_val = conv.max()
    threshold = m_val - 1e-10
    max_idx = np.where(conv >= threshold)[0]
    w = np.zeros_like(conv)
    w[max_idx] = 1.0 / len(max_idx)

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]

    grad_tan = grad - float(np.dot(a, grad))
    a_new = a - step_size * grad_tan
    return _project_simplex(a_new)


def _make_conv_matrix(a, M, top_k=80):
    """Create constraint matrix for LP: conv(a, g) <= M for active lags."""
    n = len(a)
    conv_aa = _fft_convolve(a)
    idx = np.argsort(conv_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)

    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b


def _lp_shave(a, max_iter=300, top_k=80):
    """LP-based iterative peak shaving."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0

    for it in range(max_iter):
        conv = _fft_convolve(a)
        M_cur = float(conv.max())

        inflate = 1.0 + 0.02 / (1.0 + 0.3 * it)
        M_lp = M_cur * inflate

        A, b = _make_conv_matrix(a, M_lp, top_k=top_k)
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success:
            no_imp += 5
            if no_imp > 100:
                break
            continue

        g = res.x
        sum_g = float(g.sum())

        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 100:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = _score_normalized(a)
        for t_val in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = np.maximum(a + t_val * (g - a), 0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            sc = _score_normalized(a_trial)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = t_val

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 150:
                break
            continue

        a = np.maximum(a + best_t * (g - a), 0)
        a = a / a.sum()

        sc = _score_normalized(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _run_lbfgsb(a0, n, T, maxiter=5000, gtol=1e-8):
    """Run L-BFGS-B on LSE-smoothed objective."""
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 -= z0.mean()

    try:
        res = opt.minimize(
            _lse_obj, z0,
            args=(n, T),
            jac=_lse_obj_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 50}
        )
        z_opt = res.x
        ez = np.exp(z_opt - z_opt.max())
        return ez / ez.sum()
    except Exception:
        return a0.copy()


def _subgradient_optimize(a, deadline, lr_init=0.008, max_iter=80000):
    """Momentum subgradient optimization on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    momentum = 0.75
    lr = lr_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_sg = _subgradient_step(a, lr)
        v = momentum * v - lr * (a_sg - a)
        a_inertial = _project_simplex(a + 2.0 * v)

        sc1 = _score_normalized(a_sg)
        sc2 = _score_normalized(a_inertial)

        if sc1 <= sc2:
            a_try, sc_try = a_sg, sc1
        else:
            a_try, sc_try = a_inertial, sc2

        if sc_try < best_score - 1e-11:
            best_score = sc_try
            best_a = a_try.copy()
            a = a_try
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.0005, lr * 0.95)
            if no_imp > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_init
                no_imp = 0

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS-B, momentum subgradient,
    and iterative LP peak shaving. Explores multiple lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception as e:
        print(f"Parent load error: {e}")

    # Lengths to explore
    lengths = [1024, 2048]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)

    # Generate initial candidates
    def _get_cands(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            rng = np.random.default_rng(seed + n_t)
            for _ in range(3):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.12 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                if p_pert.sum() > 1e-10:
                    cands.append(p_pert / p_pert.sum())
        t = np.linspace(0.0, 1.0, n_t)
        # Gaussian
        g = np.exp(-2.5 * (t - 0.5) ** 2)
        cands.append(np.maximum(g, 1e-10) / np.maximum(g, 1e-10).sum())
        # Raised cosine
        rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        cands.append(np.maximum(rc, 1e-10) / np.maximum(rc, 1e-10).sum())
        # Flat with edge boost
        fe = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        cands.append(np.maximum(fe, 1e-10) / np.maximum(fe, 1e-10).sum())
        # Linear
        ln = np.linspace(0.2, 1.0, n_t)
        cands.append(ln / ln.sum())
        # Bimodal
        bm = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2)
        cands.append(np.maximum(bm, 1e-10) / np.maximum(bm, 1e-10).sum())
        return cands

    # Phase 1: L-BFGS-B at moderate T
    print("\n=== Phase 1: L-BFGS-B (T=0.04) ===")
    results = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 700:
            break

        cands = _get_cands(n_t)
        best_here = None
        best_here_sc = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 700:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            a_opt = _run_lbfgsb(a0, n_t, T=0.04, maxiter=3000, gtol=1e-7)
            sc_opt = _score(a_opt)

            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
            if sc_opt < best_here_sc:
                best_here_sc = sc_opt
                best_here = a_opt

            print(f"  n={n_t}, cand {ci}: {sc0:.8f} -> {sc_opt:.10f}")

        if best_here is not None:
            results[n_t] = best_here
            print(f"  Best n={n_t} P1: {best_here_sc:.10f}")

    # Phase 2: Subgradient refinement
    print("\n=== Phase 2: Subgradient refinement ===")
    for n_t in lengths:
        if n_t not in results:
            continue
        if time.monotonic() > deadline - 400:
            break

        a_in = results[n_t]
        sg_deadline = min(time.monotonic() + 120.0, deadline - 400.0)
        a_sg = _subgradient_optimize(a_in, sg_deadline, lr_init=0.006)
        sc_sg = _score(a_sg)

        if np.isfinite(sc_sg) and sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()

        if sc_sg < _score(results[n_t]):
            results[n_t] = a_sg

        print(f"  n={n_t}: subgrad={sc_sg:.10f}")

    # Phase 3: LP peak shaving (aggressive)
    print("\n=== Phase 3: LP peak shaving ===")
    for n_t in lengths:
        if n_t not in results:
            continue
        if time.monotonic() > deadline - 250:
            break

        a_in = results[n_t]
        sc_in = _score(a_in)
        print(f"  LP start n={n_t}: {sc_in:.10f}")

        # Multiple LP passes with varying parameters
        for top_k in [60, 100, 50]:
            if time.monotonic() > deadline - 250:
                break
            a_lp = _lp_shave(a_in, max_iter=200, top_k=top_k)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            if sc_lp < _score(a_in):
                a_in = a_lp
            print(f"    top_k={top_k}: {sc_lp:.10f}")

        results[n_t] = a_in
        print(f"  Best n={n_t} P3: {_score(a_in):.10f}")

    # Phase 4: Fine L-BFGS at small T
    print("\n=== Phase 4: Fine L-BFGS (T=0.01) ===")
    for n_t in lengths:
        if n_t not in results:
            continue
        if time.monotonic() > deadline - 120:
            break

        a_in = results[n_t]
        a_fine = _run_lbfgsb(a_in, n_t, T=0.01, maxiter=3000, gtol=1e-9)
        sc_fine = _score(a_fine)

        if np.isfinite(sc_fine) and sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
        if sc_fine < _score(a_in):
            results[n_t] = a_fine

        print(f"  n={n_t}: fine={sc_fine:.10f}")

    # Phase 5: Another LP pass
    print("\n=== Phase 5: Final LP polish ===")
    for n_t in lengths:
        if n_t not in results:
            continue
        if time.monotonic() > deadline - 50:
            break

        a_in = results[n_t]
        a_lp = _lp_shave(a_in, max_iter=150, top_k=80)
        sc_lp = _score(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        results[n_t] = a_lp
        print(f"  n={n_t}: final LP={sc_lp:.10f}")

    # Phase 6: Very fine T L-BFGS on the overall best
    print("\n=== Phase 6: Ultra-fine L-BFGS (T=0.005) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        a_ultra = _run_lbfgsb(best_seq, n_best, T=0.005, maxiter=2000, gtol=1e-10)
        sc_ultra = _score(a_ultra)
        if np.isfinite(sc_ultra) and sc_ultra < best_score:
            best_score = sc_ultra
            best_seq = a_ultra.copy()
        print(f"  Ultra-fine: {sc_ultra:.10f}")

    # Final LP on absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_final_lp = _lp_shave(best_seq, max_iter=80, top_k=60)
        sc_flp = _score(a_final_lp)
        if np.isfinite(sc_flp) and sc_flp < best_score:
            best_score = sc_flp
            best_seq = a_final_lp.copy()
        print(f"  Final LP on best: {sc_flp:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
```
