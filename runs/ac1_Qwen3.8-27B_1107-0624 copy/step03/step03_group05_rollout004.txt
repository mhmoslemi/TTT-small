"""
Iterative LP optimization with adaptive parameters and diverse multi-length search.
Minimizes 2n*max(conv(a,a))/sum(a)^2 using LP-based fixed-point iteration,
gradient polish, and simulated annealing restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _build_conv_matrix(a):
    """Build the convolution constraint matrix A where (A@g)_k = (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    options = {'presolve': True}
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs', options=options)
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=100000, verbose=False):
    """Iterative LP refinement with adaptive inflation and blending."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflation = 1.0003
    blend = 0.25
    no_imp = 0
    stagnant = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflation

        g = _solve_lp(a, M)
        if g is None:
            inflation = min(1.05, inflation * 1.15)
            stagnant += 1
            if stagnant > 100:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp % 50 == 0:
                inflation = max(1.0001, inflation * 0.98)
                blend = max(0.03, blend * 0.95)
            if no_imp > 2000:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7 + n)
                a = best_a * (1.0 + 0.05 * rng.normal(size=n))
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a / s
                no_imp = 0
                inflation = 1.0003
                blend = 0.25
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            stagnant = 0
            if verbose and it % 1000 == 0:
                print(f"    LP iter {it}: score={score_new:.10f}")
            # Tighten inflation on progress

                blend = max(0.03, blend * 0.93)

def _gradient_polish(a, deadline, max_iter=50000):
    """Smooth-max gradient descent with simplex projection."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.03 * float(c.max()))
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        c_nest = _fft_convolve(a_nest)
        mval = float(c_nest.max())
        x = (c_nest - mval) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c_nest)
            ex[int(np.argmax(c_nest))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _adam_polish(a, deadline, max_iter=50000, lr=0.04):
    """Logit-parameterized Adam optimization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.04 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c_arr = _fft_convolve(p)
        mval = float(c_arr.max())
        x = (c_arr - mval) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s_ex = float(ex.sum())
        w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v_arr / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(3000, 4 * n):
            rng = np.random.default_rng((it * 31 + n) % (2**32))
            z = 0.6 * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.04 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, centers=(0.25, 0.75), sigma=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((t - c) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_random(n, seed=0):
    rng = np.random.default_rng(seed)
    a = np.maximum(rng.normal(1.0, 0.5, n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
