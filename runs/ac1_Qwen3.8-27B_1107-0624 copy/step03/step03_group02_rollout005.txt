and multi-length exploration (1024, 2048) with careful time management.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _adam_optimize(a, deadline, max_iter=30000, lr=0.03, beta_factor=0.05,
                   restart_every=5000, pert_scale=0.25):
    """Adam optimizer in logit-space with adaptive beta and structured restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-12)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.0005, lr_cur * 0.95)
            if no_imp > restart_every:
                # Structured restart: perturb with sinusoidal + random
                z = best_z.copy()
                freqs = rng.integers(1, 8, size=3)
                phases = rng.uniform(0, 2 * np.pi, size=3)
                t = np.arange(n)
                pert = sum(pert_scale / (f + 1) * np.sin(2 * np.pi * f * t / n + ph)
                           for f, ph in zip(freqs, phases))
                pert += 0.1 * rng.normal(size=n)
                z = z + pert
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-12, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 100 == 0:
            beta = max(1e-14, beta * 0.8)

    return best_a, best_score


def _lp_tight_refine(a, deadline, max_iter=2000, inflate_start=1.0005, blend=0.25):
    """LP refinement with very tight inflation for precise peak reduction."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate_cur = inflate_start
    blend_cur = blend
    max_k = 2 * n - 1

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Build LP: maximize sum(g) subject to (a*g)[k] <= M_lp, g >= 0
        # (a*g)[k] = sum_j a[j] * g[k-j]
        c_obj = -np.ones(n)
        A = np.zeros((max_k, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(max_k, M_lp)

        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_infeasibility_tolerance': 1e-9})
        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            # Tighten as we improve
            inflate_cur = max(1.0002, inflate_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.005, inflate_cur * 1.02)
            if no_imp > 300:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate_start
                no_imp = 0

        a = a_new

    return best_a


def _directed_peak_reduction(a, deadline, max_iter=5000, step_size=0.002):
    """Directly target the peak lag by adjusting contributing pairs."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a)
    peak_k = int(np.argmax(c))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        peak_val = float(c[peak_k])

        # Compute contributions to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Find top contributors
        top_idx = np.argsort(contribs)[::-1][:min(10, n)]
        # Find bottom contributors (positions that don't contribute much)
        bot_idx = np.argsort(contribs)[:min(10, n)]

        # Transfer mass from top to bottom contributors
        made_change = False
        for i in top_idx:
            for j in bot_idx:
                if i == j or (peak_k - i) == j or (peak_k - j) == i:
                    continue
                delta = step_size * a[i]
                if a[i] > 2 * delta and a[j] < 1.0:
                    # Check if this would help
                    new_val = peak_val - delta * a[peak_k - i] - delta * a[peak_k - j]
                    if new_val < peak_val - 1e-10:
                        a[i] -= delta
                        a[j] += delta
                        made_change = True
                        break
            if made_change:
                break

        if not made_change:
            no_imp += 1
            if no_imp > 500:
                break
            # Re-evaluate peak location
            peak_k = int(np.argmax(c))
            step_size *= 0.95
            continue

        no_imp = 0
        s_curr = a.sum()
        if s_curr < 1e-10:
            break
        a = a / s_curr

        c_new = _fft_convolve(a)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a.copy()

        if it % 100 == 0:
            peak_k = int(np.argmax(c_new))

    return best_a


def _multi_peak_subgradient(a, deadline, max_iter=10000, top_k=3):
    """Subgradient targeting top-k peaks with adaptive weights."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.2 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        # Exponential weighting based on peak values
        peak_vals = c_curr[top_indices]
        wts = np.exp(5.0 * (peak_vals - peak_vals[0]))
        wts = wts / wts.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += wts[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _full_refinement_pipeline(a0, deadline):
    """Multi-phase refinement: Adam -> LP -> Directed -> Fine Adam -> Final LP."""
    best_a = a0.copy()
    best_sc = _score(a0)

    # Phase 1: Adam (main)
    t1 = min(deadline - 120, time.monotonic() + 280)
    a1, s1 = _adam_optimize(best_a, t1, max_iter=25000, lr=0.025, beta_factor=0.04)
    if s1 < best_sc:
        best_a, best_sc = a1, s1
    print(f"    After Adam: {best_sc:.10f}")

    # Phase 2: Tight LP
    if time.monotonic() < deadline - 80:
        t2 = min(deadline - 80, time.monotonic() + 180)
        a2 = _lp_tight_refine(best_a, t2, max_iter=1500, inflate_start=1.0005, blend=0.2)
        s2 = _score(a2)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"    After LP1: {best_sc:.10f}")

    # Phase 3: Directed peak reduction
    if time.monotonic() < deadline - 60:
        t3 = min(deadline - 60, time.monotonic() + 80)
        a3 = _directed_peak_reduction(best_a, t3, max_iter=3000, step_size=0.001)
        s3 = _score(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"    After Directed: {best_sc:.10f}")

    # Phase 4: Fine Adam
    if time.monotonic() < deadline - 30:
        t4 = min(deadline - 30, time.monotonic() + 100)
        a4, s4 = _adam_optimize(best_a, t4, max_iter=15000, lr=0.008, beta_factor=0.015)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"    After Fine Adam: {best_sc:.10f}")

    # Phase 5: Final tight LP
    if time.monotonic() < deadline - 10:
        t5 = min(deadline - 10, time.monotonic() + 60)
        a5 = _lp_tight_refine(best_a, t5, max_iter=800, inflate_start=1.0003, blend=0.15)
        s5 = _score(a5)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"    After Final LP: {best_sc:.10f}")

    # Phase 6: Ultra-fine Adam burst
    if time.monotonic() < deadline - 5:
        t6 = deadline - 5
        a6, s6 = _adam_optimize(best_a, t6, max_iter=8000, lr=0.003, beta_factor=0.005)
        if s6 < best_sc:
            best_a, best_sc = a6, s6
        print(f"    After Ultra Adam: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: adaptive Adam + tight LP + directed peak reduction
    + multi-length exploration. Targets n=1024 primarily, n=2048 as secondary.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare working sequences at different lengths
    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
        p2048 = _interpolate_profile(parent, 2048)
    else:
        # Fallback: create good initial profiles
        n_tmp = 512
        t = np.linspace(0.0, 1.0, n_tmp)
        base = 1.0 + 1.5 * (t - 0.5)
        base = np.maximum(base, 0.01)
        p512 = base / base.sum()
        p1024 = _interpolate_profile(p512, 1024)
        p2048 = _interpolate_profile(p512, 2048)
        p512 = p512 / p512.sum()
        p1024 = p1024 / p1024.sum()
        p2048 = p2048 / p2048.sum()

    # === PRIMARY: n=1024 (most time) ===
    print("=== Primary: n=1024 ===")
    d_primary = min(deadline - 400, time.monotonic() + 900)
    a1024, s1024 = _full_refinement_pipeline(p1024, d_primary)
    if s1024 < best_score:
        best_score = s1024
        best_seq = a1024.copy()
    print(f"  n=1024 best: {best_score:.10f}")

    # === SECONDARY: n=2048 ===
    if time.monotonic() < deadline - 150:
        print("=== Secondary: n=2048 ===")
        # Start from the best 1024 result, interpolated
        p2048_interp = _interpolate_profile(best_seq, 2048) if best_seq is not None else p2048
        d_2048 = min(deadline - 100, time.monotonic() + 350)
        a2048, s2048 = _full_refinement_pipeline(p2048_interp, d_2048)
        if s2048 < best_score:
            best_score = s2048
            best_seq = a2048.copy()
        print(f"  n=2048 best: {best_score:.10f}")

    # === TERTIARY: diverse init at n=1024 ===
    if time.monotonic() < deadline - 100:
        print("=== Tertiary: diverse inits n=1024 ===")
        n = 1024
        t = np.linspace(0.0, 1.0, n)
        candidates = [
            (1.0 + 1.0 * (t - 0.5) + 0.3 * np.cos(2 * np.pi * t)) ,
            (1.0 + 2.0 * (t - 0.5) - 0.5 * np.cos(4 * np.pi * t)),
            (1.0 + 1.5 * np.sin(np.pi * t) ** 2),
        ]
        # Perturbed parent
        if parent is not None:
            pp = p1024 * (1.0 + 0.1 * rng.normal(size=n))
            pp = np.maximum(pp, 0.0)
            candidates.append(pp)

        remaining = deadline - 100 - time.monotonic()
        per_cand = remaining / len(candidates)
        for i, c in enumerate(candidates):
            if time.monotonic() > deadline - 100 + i * per_cand:
                break
            c = np.maximum(c, 0.01)
            c = c / c.sum()
            d_c = min(deadline - 100 + i * per_cand, time.monotonic() + per_cand - 5)
            # Quick: just Adam + LP
            ac, sc = _adam_optimize(c, d_c, max_iter=8000, lr=0.02, beta_factor=0.04)
            if sc < best_score:
                best_score = sc
                best_seq = ac.copy()
            print(f"  Diverse-{i}: {sc:.10f}")

    # === FINAL POLISH ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"=== Final Polish (score={best_score:.10f}) ===")
        # Very fine Adam
        d_f1 = min(deadline - 8, time.monotonic() + 80)
        af1, sf1 = _adam_optimize(best_seq, d_f1, max_iter=10000, lr=0.004, beta_factor=0.004)
        if sf1 < best_score:
            best_score = sf1
            best_seq = af1.copy()
        print(f"  Fine Adam: {best_score:.10f}")

        # Very tight LP
        if time.monotonic() < deadline - 5:
            d_f2 = min(deadline - 5, time.monotonic() + 50)
            af2 = _lp_tight_refine(best_seq, d_f2, max_iter=500, inflate_start=1.0002, blend=0.1)
            sf2 = _score(af2)
            if sf2 < best_score:
                best_score = sf2
                best_seq = af2.copy()
            print(f"  Tight LP: {best_score:.10f}")

        # Directed peak reduction
        if time.monotonic() < deadline - 3:
            d_f3 = min(deadline - 3, time.monotonic() + 30)
            af3 = _directed_peak_reduction(best_seq, d_f3, max_iter=2000, step_size=0.0005)
            sf3 = _score(af3)
            if sf3 < best_score:
                best_score = sf3
                best_seq = af3.copy()
            print(f"  Directed: {best_score:.10f}")

    # Final validation and output
    if best_seq is None:
        best_seq = p1024
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = p1024
    else:
        final_seq = final_seq / s  # Normalize to sum=1 for cleanliness

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence confirms: {sc_eval:.10f}")
            else:
                print(f"WARNING: evaluate_sequence returned {sc_eval}")
        except Exception as e:
            print(f"Error in evaluate_sequence: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
