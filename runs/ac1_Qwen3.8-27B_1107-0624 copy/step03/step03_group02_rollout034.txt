Step function optimization: minimizes 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z, temp=1.0):
    z = z - float(np.max(z))
    z = z / temp
    e = np.exp(z)
    s = e.sum()
    return e / s

def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c_obj = -np.ones(n)
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = a
    b = np.full(m, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _spectral_flatten(a, deadline, n_iters=2000, strength=0.1):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute power spectrum
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        
        # Compute autocorrelation
        c = np.fft.ifft(P).real[:2*n-1]
        c = np.maximum(c, 0)
        peak_val = float(c.max())
        peak_idx = int(np.argmax(c))
        
        # Target: reduce frequencies that contribute to the peak
        # c[peak_idx] = sum_m P[m] * exp(2πi*m*peak_idx/N) / N
        # To reduce c[peak_idx], reduce P[m] where exp(2πi*m*peak_idx/N) has positive real part
        phase = 2.0 * np.pi * np.arange(N) * peak_idx / N
        # Contribution weight (real part of exp)
        contrib = np.cos(phase)
        
        # Reduce P where contrib is positive, slightly increase where negative
        # This is a spectral "notch" at the peak frequency
        P_new = P.copy()
        P_new = P_new * (1.0 - strength * np.maximum(contrib, 0))
        P_new = np.maximum(P_new, 0)
        
        # Reconstruct a from P_new (keeping the phase)
        phase_A = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase_A)
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        # Decay strength
        if it % 100 == 0:
            strength = max(0.001, strength * 0.9)
    
    return best_a, best_score


def _interpolate_to_length(a, target_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == target_n:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, target_n)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(target_n) / target_n
    return p / s


def _make_gaussian_weighted(n, center=0.5, width=0.3):
    j = np.arange(n)
    t = j / (n - 1) if n > 1 else np.array([0.5])
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_shape(n, power=2.0):

    a = t ** power + (1 - t) ** power + 0.1
    a = np.maximum(a, 1e-12)

"""
Multi-strategy step function optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses smooth-max Adam with exact FFT gradient, Fourier spectral peak deflation,
LP-based spreading, and multi-length multi-start exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_safe(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.max(z))
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-30:
        return np.ones_like(z) / z.size
    return e / s


def _grad_convolution_weighted(a, w):
    """Exact gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses FFT for the cross-correlation part."""
    n = len(a)
    N = _next_pow2(2 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1]
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_res = np.fft.irfft(fw * fa, N)
    h = conv_res[n - 1:2 * n - 1]
    # grad_a[i] = h[i] + a[i] * sum(w[i:i+n])
    csum_w = np.concatenate([[0.0], np.cumsum(w)])
    starts = np.arange(n)
    ends = np.minimum(starts + n, len(w))
    w_shift = csum_w[ends] - csum_w[starts]
    return h + a * w_shift


def _adam_smooth_max(a0, deadline, max_iter=30000, lr=0.02, beta_start=0.06,
                     beta_min=0.001, beta_decay=0.88, restart_patience=2000,
                     restart_noise=0.15, seed_offset=0):
    """Adam optimizer in log-space minimizing smooth max of autocorrelation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = beta_start
    no_imp = 0
    lr_cur = lr
    rng = np.random.default_rng(12345 + seed_offset)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        mval = float(c_curr.max())

        x = (c_curr - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c_curr)
            ex[int(np.argmax(c_curr))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        grad_a = _grad_convolution_weighted(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        a_new = _softmax_safe(z_new)
        a_new = np.maximum(a_new, 1e-14)

        score_new = float(2.0 * n * c_new.max())

            best_z = z_new.copy()

            a = _softmax_safe(z)
            a = np.maximum(a, 1e-14)
            a = a / a.sum()
            beta = max(beta_min, beta_start * 0.4)
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr_cur = lr

def _spectral_deflate(a0, deadline, n_iters=1500, strength=0.08):
    """Fourier-domain peak deflation: reduce spectral content aligned with autocorr peak."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    N = _next_pow2(2 * n - 1)
    strength_cur = strength

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        phase_A = np.angle(A)

        c_full = np.fft.ifft(P).real[:2 * n - 1]
        c_safe = np.maximum(c_full, 0.0)
        peak_idx = int(np.argmax(c_safe))

        if peak_idx == 0 or peak_idx == 2 * n - 2:
            # Peak at edge - try to reduce low frequencies
            freqs = np.arange(N)
            weight = freqf = freqs / float(N)
            P_new = P * (1.0 - strength_cur * weight)
            P_new = np.maximum(P_new, 0.0)
        else:
            # Reduce frequencies aligned with the peak
            m = np.arange(N)
            phase = 2.0 * np.pi * m * peak_idx / N
            contrib = np.cos(phase)
            P_new = P * (1.0 - strength_cur * np.maximum(contrib, 0.0))
            # Slightly boost anti-aligned frequencies
            P_new = P_new * (1.0 + 0.3 * strength_cur * np.maximum(-contrib, 0.0))
            P_new = np.maximum(P_new, 0.0)

        A_new = np.sqrt(P_new) * np.exp(1j * phase_A)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s < 1e-10:
            break
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        if it % 50 == 0:
            strength_cur = max(0.001, strength_cur * 0.92)

    return best_a, best_score


def _lp_spread_solve(a, M):
    n = len(a)
    m = 2 * n - 1
    c_obj = -np.ones(n)
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(m, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': 5000})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a0, deadline, max_iter=1000, inflate=1.001, blend=0.3):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_spread_solve(a, M * inflate_cur)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 80:
                break
            if no_imp % 15 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.04)
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new

        else:

            if no_imp > max(80, n // 2):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _interpolate_to_length(a, target_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == target_n:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, target_n)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(target_n) / target_n
    return p / s


def _init_gaussian(n, width=0.35):
    j = np.arange(n, dtype=np.float64)
    t = j / max(n - 1, 1)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_u_shape(n, power=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = t ** power + (1.0 - t) ** power + 0.15
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_v_shape(n, power=1.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.abs(t - 0.5) ** power + 0.15
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_symmetric_power(n, p=0.8):
    j = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    if mid < 1:
        return np.ones(n) / n
    d = np.abs(j - mid) / mid
    a = (1.0 + d) ** p + 0.08
    return a / a.sum()


def _init_linear(n, slope=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_random_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    # Generate a smooth random signal via low-pass filtering
    raw = rng.standard_normal(n)
    # Simple moving average as low-pass
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    smoothed = np.convolve(raw, kernel, mode='same')
    a = np.maximum(smoothed + 1.0, 0.01)
    return a / a.sum()


def _init_two_bumps(n, pos1=0.3, pos2=0.7, width=0.15):
    j = np.arange(n, dtype=np.float64)
    t = j / max(n - 1, 1)
    a = np.exp(-((t - pos1) ** 2) / (2.0 * width ** 2)) + \
        np.exp(-((t - pos2) ** 2) / (2.0 * width ** 2)) + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _hill_climb(a0, deadline, n_trials=5000, step_size=0.02, seed=42):
    """Stochastic hill climbing: random coordinate perturbations."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    rng = np.random.default_rng(seed)
    step = step_size

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Perturb a random subset of coordinates
        n_perturb = max(1, n // 20)
        idx = rng.integers(0, n, size=n_perturb)
        a_try = a.copy()
        a_try[idx] *= (1.0 + step * rng.standard_normal(n_perturb))
        a_try = np.maximum(a_try, 0.0)
        s = float(a_try.sum())
        if s < 1e-10:
            continue
        a_try = a_try / s

        c_try = _fft_convolve(a_try)
        score_try = float(2.0 * n * c_try.max())

        if score_try < best_score - 1e-14:
            best_score = score_try
            best_a = a_try.copy()
            a = a_try
        elif score_try < _score(a) - 1e-14:
            a = a_try
        else:
            # Occasionally accept (simulated annealing)
            if rng.random() < 0.05 and score_try < best_score * 1.001:
                a = a_try

        if it % 500 == 0 and it > 0:
            step = max(0.001, step * 0.85)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search combining smooth-max Adam, spectral deflation,
    LP refinement, and hill climbing across multiple lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Lengths to explore
    lengths = [512, 768, 1024, 1536]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)
    # Cap at reasonable sizes
    lengths = [ln for ln in lengths if 256 <= ln <= 2048]

    print(f"Exploring lengths: {lengths}")
    print(f"Time budget: {budget_s - 30:.0f}s")

    # Main multi-start search
    total_length_budget = deadline - t_start - 400.0  # Reserve 400s for final
    total_length_budget = max(total_length_budget, 100.0)
    time_per_length = total_length_budget / len(lengths)

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 400:
            break

        length_start = time.monotonic()
        length_deadline = min(deadline - 400, length_start + time_per_length)
        print(f"\n--- n={n}, budget={time_per_length:.0f}s ---")

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(('parent', _interpolate_to_length(parent, n)))
        if best_seq is not None and (parent is None or best_seq.size != n):
            inits.append(('best', _interpolate_to_length(best_seq, n)))
        inits.append(('gauss35', _init_gaussian(n, 0.35)))
        inits.append(('gauss25', _init_gaussian(n, 0.25)))
        inits.append(('u_shape', _init_u_shape(n, 1.5)))
        inits.append(('v_shape', _init_v_shape(n, 1.2)))
        inits.append(('sym_power', _init_symmetric_power(n, 0.8)))
        inits.append(('linear', _init_linear(n, 2.0)))
        inits.append(('two_bumps', _init_two_bumps(n)))
        inits.append(('rand_smooth', _init_random_smooth(n, seed=100 + n)))

        # Limit to max 6 initializations to save time
        max_inits = min(len(inits), max(3, int(time_per_length / 60)))
        inits = inits[:max_inits]

        time_per_init = max(15.0, time_per_length / len(inits))

        for ii, (name, a0) in enumerate(inits):
            if time.monotonic() > length_deadline:
                break

            init_deadline = min(length_deadline, time.monotonic() + time_per_init)
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Phase 1: Adam smooth-max (primary optimizer)
            adam_d = min(init_deadline - 10, time.monotonic() + time_per_init * 0.55)
            a1, s1 = _adam_smooth_max(
                a0, adam_d,
                max_iter=25000, lr=0.025,
                beta_start=0.07, beta_min=0.002,
                beta_decay=0.87, restart_patience=max(1500, n),
                restart_noise=0.2, seed_offset=li * 100 + ii
            )
            if s1 < best_score:
                best_score = s1
                best_seq = a1.copy()
                print(f"    {name}@{n}: Adam -> {s1:.10f}")
            a0 = a1

            # Phase 2: Spectral deflation
            if time.monotonic() < init_deadline - 5:
                spec_d = min(init_deadline - 5, time.monotonic() + time_per_init * 0.2)
                a2, s2 = _spectral_deflate(a0, spec_d, n_iters=800, strength=0.06)
                if s2 < best_score:
                    best_score = s2
                    best_seq = a2.copy()
                    print(f"    {name}@{n}: Spectral -> {s2:.10f}")
                a0 = a2

            # Phase 3: LP refinement (only for n <= 1024 for speed)
            if n <= 1024 and time.monotonic() < init_deadline - 3:
                lp_d = min(init_deadline - 3, time.monotonic() + min(40.0, time_per_init * 0.2))
                a3, s3 = _lp_refine(a0, lp_d, max_iter=300, inflate=1.001, blend=0.3)
                if s3 < best_score:
                    best_score = s3
                    best_seq = a3.copy()
                    print(f"    {name}@{n}: LP -> {s3:.10f}")
                a0 = a3

            # Phase 4: Quick Adam polish
            if time.monotonic() < init_deadline:
                a4, s4 = _adam_smooth_max(
                    a0, init_deadline,
                    max_iter=8000, lr=0.008,
                    beta_start=0.02, beta_min=0.001,
                    beta_decay=0.92, restart_patience=3000,
                    restart_noise=0.08, seed_offset=1000 + li * 100 + ii
                )
                if s4 < best_score:
                    best_score = s4
                    best_seq = a4.copy()
                    print(f"    {name}@{n}: Polish -> {s4:.10f}")

    print(f"\nAfter main search: best_score={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # === Final intensive phase ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        final_budget = deadline - 20 - time.monotonic()
        print(f"\n=== Final intensive phase ({final_budget:.0f}s) ===")

        # Try at best's native length and at 1024
        final_lengths = [best_seq.size]
        if 1024 not in final_lengths:
            final_lengths.append(1024)
        if 512 not in final_lengths and best_seq.size > 512:
            final_lengths.append(512)

        for fl in final_lengths:
            if time.monotonic() > deadline - 15:
                break

            fc = _interpolate_to_length(best_seq, fl)
            sub_budget = max(30.0, final_budget / len(final_lengths))
            sub_deadline = min(deadline - 15, time.monotonic() + sub_budget)

            # Long Adam run with very fine settings
            a_f, s_f = _adam_smooth_max(
                fc, sub_deadline - 30 if sub_budget > 60 else sub_deadline,
                max_iter=50000, lr=0.004,
                beta_start=0.008, beta_min=0.0005,
                beta_decay=0.96, restart_patience=8000,
                restart_noise=0.04, seed_offset=9999
            )
            if s_f < best_score:
                best_score = s_f
                best_seq = a_f.copy()
                print(f"  Final Adam n={fl}: {s_f:.10f}")

            # Spectral deflation
            if time.monotonic() < sub_deadline - 10:
                a_sf, s_sf = _spectral_deflate(
                    best_seq, min(sub_deadline - 10, time.monotonic() + 30),
                    n_iters=2000, strength=0.03
                )
                if s_sf < best_score:
                    best_score = s_sf
                    best_seq = a_sf.copy()
                    print(f"  Final Spectral n={fl}: {s_sf:.10f}")

            # Hill climbing
            if time.monotonic() < sub_deadline - 5:
                a_hc, s_hc = _hill_climb(
                    best_seq, min(sub_deadline - 5, time.monotonic() + 40),
                    n_trials=10000, step_size=0.01, seed=777
                )
                if s_hc < best_score:
                    best_score = s_hc
                    best_seq = a_hc.copy()
                    print(f"  Hill climb n={fl}: {s_hc:.10f}")

            # LP if small enough
            if fl <= 1024 and time.monotonic() < sub_deadline - 3:
                a_lp, s_lp = _lp_refine(
                    best_seq, min(sub_deadline - 3, time.monotonic() + 25),
                    max_iter=500, inflate=1.0005, blend=0.15
                )
                if s_lp < best_score:
                    best_score = s_lp
                    best_seq = a_lp.copy()
                    print(f"  Final LP n={fl}: {s_lp:.10f}")

    # Validation and output
    if best_seq is None:
        best_seq = _init_gaussian(256, 0.35)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _init_gaussian(256, 0.35)

    # Authoritative evaluation
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            print(f"evaluate_sequence returned non-finite: {sc_eval}")
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
