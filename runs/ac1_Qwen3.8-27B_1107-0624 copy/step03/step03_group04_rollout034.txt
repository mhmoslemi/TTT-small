Step function optimization minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _smooth_max_grad_a(a, beta):
    """Gradient of smooth max of conv(a,a) w.r.t. a (NOT log-space)."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, float(c.max())


def _make_conv_matrix(a):
    n = len(a)
    # A[k, j] = a[k - j] if 0 <= k-j < n, else 0
    # conv(a, g)[k] = sum_j A[k, j] * g[j]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M, max_time=30.0):
    """Solve LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A_mat = _make_conv_matrix(a)
    b_vec = np.full(2 * n - 1, M)
    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': max_time})
    if res.success:
        return res.x
    return None

        # grad w.r.t. a: d/d a_i [conv(a,a)[k*]] = 2 * sum_j a[j] * δ(i+j, k*)

def _project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _multi_length_escale(parent, target_n):
    """Interpolate parent to target length."""
    p = np.asarray(parent, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == target_n:
        return p.copy() / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, target_n)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(target_n) / target_n
    return p_new / s

            if sc < best_score:
                best_score = sc

        phase5_deadline = min(deadline - 30, time.monotonic() + 60)

    # Phase 6: Fine polish

        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # Phase 7: Final LP
    if time.monotonic() < deadline - 5:
        final_lp_deadline = deadline - 3
        if final_lp_deadline - time.monotonic() > 10:

                best_score = sc
                best_seq = a.copy()

    # Final validation
    if best_seq is None:
        best_seq = parent.copy()

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())

    # Evaluate with authoritative function

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
