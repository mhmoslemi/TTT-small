Uses power-mean loss annealing, logit-space Adam, and coordinate descent polish.

import numpy as np
import time

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0..n-1"""
    n = len(a)
    N = next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n-1:2*n-1]

def power_mean_weights(c, p):
    """Weights for power-mean gradient: proportional to c^(p-1)."""
    c = np.maximum(c, 0.0)
    if p <= 1:
        return np.ones_like(c) / len(c)
    w = c ** (p - 1)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = w / sw
    return w

def softmax_weights(c, beta):
    """Weights for log-sum-exp gradient."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = ex / s
    return w

    a0 = a0 / a0.sum()
    
    z = np.log(a0)
    z = z - z.mean()

    v = np.zeros(n)
    t_adam = 0

    best_z = z.copy()
    best_score = compute_score(a0)
    no_improve = 0

        dz = a0 * (grad_a - dot)
        
        # Adam

        mh = m / (1.0 - b1**t_adam)
        vh = v / (1.0 - b2**t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

            m.fill(0.0)

            a0 = best_a.copy()

            continue

            best_z = z.copy()

            a0 = np.exp(z - z.max())

            m.fill(0.0)
            v.fill(0.0)

            current_lr = base_lr
            no_improve = 0
    
    return best_a, best_score

"""
Multi-resolution optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses power-mean loss annealing, chirp-based initialization, logit-space Adam,
and coordinate descent polish with gradient-guided mass transport.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0..n-1 via FFT."""
    n = len(a)
    N = next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n - 1:2 * n - 1]


def power_mean_weights(c, p):
    """Weights proportional to c^(p-1), normalized to sum to 1."""
    c = np.maximum(c, 0.0)
    if p <= 1:
        return np.ones_like(c) / len(c)
    w = c ** (p - 1)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = w / sw
    return w


def logsumexp_weights(c, beta):
    """Softmax weights for log-sum-exp approximation of max."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = ex / s
    return w


def make_chirp_profile(n, K=3, power=1.0):
    """Quadratic chirp: a_i ∝ (1 + cos(2*pi*K*i^2/n))^power."""
    i = np.arange(n)
    phase = 2.0 * np.pi * K * (i * 1.0 / n) ** 2
    a = (1.0 + np.cos(phase)) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_asymmetric_ramp(n, alpha=2.0, beta=0.5):
    """Asymmetric beta-distribution shape: a_i ∝ (i/n)^alpha * ((n-1-i)/n)^beta."""
    t = np.arange(n) / max(n - 1, 1)
    a = (t + 1e-6) ** alpha * ((1.0 - t) + 1e-6) ** beta
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_asymmetric(n, k=1, phase=0.3):
    """Asymmetric cosine: a_i ∝ 1 + cos(2*pi*k*t + phase)."""
    t = np.arange(n) / max(n - 1, 1)
    a = 1.0 + 0.8 * np.cos(2.0 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_dual_peak(n, pos1=0.25, pos2=0.75, width=0.15):
    """Two peaks at different positions."""
    t = np.arange(n) / max(n - 1, 1)
    a = np.exp(-((t - pos1) / width) ** 2) + 0.7 * np.exp(-((t - pos2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def logit_adam_pm(a0, deadline, rng, p_start=4.0, p_end=40.0, lr=0.04,
                  restart_period=1500, beta_fallback=0.0):
    """
    Logit-space Adam with power-mean loss (annealing p from p_start to p_end).
    If beta_fallback > 0, switches to logsumexp when p is very large.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = compute_score(a0)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    iters = 0
    t_start = time.monotonic()
    total_time_est = max(deadline - t_start, 1.0)
    expected_iters = int(total_time_est / 0.0005)
    expected_iters = max(expected_iters, 1000)

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)

        frac = min(1.0, iters / expected_iters)
        p = p_start + (p_end - p_start) * frac

        if p > 30 and beta_fallback > 0:
            beta = beta_fallback * float(c.max())
            w = logsumexp_weights(c, beta)
        else:
            w = power_mean_weights(c, p)

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        score = compute_score(a0)
        iters += 1

        if score < best_score - 1e-12:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            current_lr = max(0.003, current_lr * 0.97)

        if no_improve > restart_period:
            noise = 0.12 * rng.normal(size=n)
            z = best_z + noise
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def coord_descent_polish(a0, deadline, rng, p=10, batch_size=5000,
                         step_frac=0.05):
    """
    Coordinate descent: transfer mass from high-gradient to low-gradient positions.
    Uses power-mean loss for gradient computation.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s > 0:
        a0 = a0 / s

    best_a = a0.copy()
    best_score = compute_score(a0)
    no_improve = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        w = power_mean_weights(c, p)
        h = compute_h(w, a0)

        # Sort indices by h value for more effective transport
        order = np.argsort(h)
        low_half = order[:n // 2]
        high_half = order[n // 2:]

        for batch_start in range(0, batch_size, 500):
            batch_end = min(batch_start + 500, batch_size)
            for _ in range(batch_end - batch_start):
                hi = rng.integers(0, len(high_half))
                lo = rng.integers(0, len(low_half))
                i = high_half[hi]
                j = low_half[lo]

                if a0[i] < 1e-15:
                    continue

                # Transfer small amount from i (high pressure) to j (low pressure)
                delta = step_frac * a0[i]
                if delta < 1e-15:
                    continue

                a0[i] -= delta
                a0[j] += delta

            # Periodic verification every 500 updates
            score = compute_score(a0)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a0.copy()
                no_improve = 0
            elif score > best_score + 0.01:
                # Revert if we got much worse
                a0 = best_a.copy()
                no_improve += batch_size

        no_improve += 1
        if no_improve > 50:
            break

    return best_a, best_score


def block_optimize(a0, deadline, rng, block_size=8, p=10, inner_iters=50):
    """
    Block coordinate optimization: optimize blocks of consecutive coordinates.
    Within each block, use projected gradient on the simplex for that block.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s > 0:
        a0 = a0 / s

    best_a = a0.copy()
    best_score = compute_score(a0)

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        w = power_mean_weights(c, p)
        h = compute_h(w, a0)

        nr_blocks = n // block_size
        for _ in range(nr_blocks):
            block_idx = rng.integers(0, nr_blocks)
            start = block_idx * block_size
            end = start + block_size

            block = a0[start:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-15:
                continue

            # Gradient within the block
            grad_block = h[start:end]
            # Projected gradient: move on the simplex for the block
            for _ in range(inner_iters):
                # Normalize gradient within block
                grad_mean = grad_block.mean()
                grad_centered = grad_block - grad_mean

                # Step
                lr_block = 0.1 * block_sum / (np.max(np.abs(grad_centered)) + 1e-10)
                block = block - lr_block * grad_centered
                block = np.maximum(block, 0.0)
                bs = block.sum()
                if bs < 1e-15:
                    block = np.ones(block_size) / block_size
                else:
                    # Rescale to maintain block_sum
                    block = block * (block_sum / bs)

            a0[start:end] = block

        score = compute_score(a0)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a0.copy()

    return best_a, best_score

        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
