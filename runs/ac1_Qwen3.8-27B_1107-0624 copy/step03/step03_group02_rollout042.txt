"""
Optimization of step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on a smooth surrogate (log-sum-exp) in log-space, combined with
projected subgradient, LP-based spreading, and coordinate descent.
Explores multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_corr_via_conv(w, a):
    """Compute (w cross-correlate a)[i] = sum_j w[j+i]*a[j] for i=0..n-1.
    Returns array of length n."""
    n = a.size
    a_rev = a[::-1].copy()
    conv_w = _fft_convolve_padded(w, a_rev)
    # (w * a_rev)[n-1+i] gives cross-correlation at lag i
    return conv_w[n - 1:2 * n - 1]


def _fft_convolve_padded(a, b):
    """Linear convolution of a (len 2n-1) and b (len n) using FFT."""
    na = a.size
    nb = b.size
    N = _next_pow2(na + nb - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:na + nb - 1]
    return c


def _smooth_max_and_grad(a, beta):
    """Return (smooth_max_value, gradient_wrt_a) for smooth max of conv(a,a)."""
    n = a.size
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    sm_val = mval + beta * np.log(s)
    # Gradient: 2 * (w cross-correlate a)
    a_rev = a[::-1].copy()
    N = _next_pow2((2 * n - 1) + n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_rev, N)
    conv_w = np.fft.irfft(fw * fa, N)[:2 * n - 1 + n - 1]
    grad_a = 2.0 * conv_w[n - 1:2 * n - 1]
    return sm_val, grad_a


def _lbfgs_smooth(a0, deadline, beta_init=0.1, beta_min=0.001, beta_decay=0.9997,
                  max_iter=5000, verbose=False):
    """L-BFGS optimization in log-space with smooth max surrogate."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    z0 = z.copy()

    best_a = a.copy()
    best_score = _score(a)

    beta = beta_init
    beta = max(beta, 0.02 * float(_fft_convolve(a).max()))

    def obj_grad(zv):
        ev = np.exp(np.clip(zv - np.max(zv), -50, 50))
        av = ev / ev.sum()
        sm, grad_a = _smooth_max_and_grad(av, beta)
        obj = 2.0 * n * sm
        dot = float(np.dot(av, grad_a))
        grad_z = av * (grad_a - dot)
        return obj, grad_z

    # Run L-BFGS with adaptive beta
    z_cur = z0.copy()
    no_improve = 0

    for phase in range(8):
        if time.monotonic() > deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / (8 - phase))

        def obj_grad_phase(zv):
            ev = np.exp(np.clip(zv - np.max(zv), -50, 50))
            av = ev / ev.sum()
            sm, grad_a = _smooth_max_and_grad(av, beta)
            obj = 2.0 * n * sm
            dot = float(np.dot(av, grad_a))
            grad_z = av * (grad_a - dot)
            if not np.all(np.isfinite(grad_z)):
                grad_z = np.zeros_like(zv)
            return obj, grad_z

        res = minimize(obj_grad_phase, z_cur, jac=True, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': 1e-6, 'ftol': 1e-10})

        if res.x is not None:
            ev = np.exp(np.clip(res.x - np.max(res.x), -50, 50))
            a_new = ev / ev.sum()
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z_cur = res.x.copy()
                z_cur -= np.mean(z_cur)
                no_improve = 0
            else:
                no_improve += 1

        # Decay beta
        beta = max(beta_min, beta * beta_decay * 0.85)

        if verbose and no_improve > 0:
            print(f"    L-BFGS phase {phase}: beta={beta:.4f}, best={best_score:.10f}")

    return best_a, best_score


def _project_simplex(v):
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        sg[idx] += a[max_idx - idx]
    j2_start = max(max_idx, 0)
    j2_end = min(max_idx + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - max_idx]
    # a_new = project(a - lr * sg)
    a_new = _project_simplex(a - lr * sg)
    return a_new


def _projected_subgradient(a0, deadline, max_iter=20000):
    """Projected subgradient with backtracking."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)

    base_lr = 1.0 / np.sqrt(n)
    it = 0
    no_imp = 0

    while it < max_iter and time.monotonic() < deadline:
        c = _fft_convolve(a)
        max_idx = int(np.argmax(c))
        max_val = float(c[max_idx])

        # Subgradient
        sg = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = _project_simplex(a - lr * sg / sg_norm)

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break
        a = a_new
        it += 1

    return best_a, best_score


def _lp_spread(a, M):
    """LP: maximize sum(b) subject to conv(a*b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=1000, inflate=1.002, blend=0.4):
    """LP-based refinement."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        if g is None:
            break
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 100:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 100:
                a = best_a.copy()
                no_imp = 0
                inflate_cur = inflate * 1.1
                if inflate_cur > 1.01:
                    break
        a = a_new

    return best_a


def _coordinate_descent(a0, deadline, max_iter=50000):
    """Greedy coordinate descent on the simplex."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    c = _fft_convolve(a)
    max_val = float(c.max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if it % 50 == 0:
            c = _fft_convolve(a)
            max_val = float(c.max())

        # Find the coordinate to increase and the one to decrease
        # Increasing a[i] by delta changes conv[k] by delta*(a[k-i] + a[i-k]) for valid k
        # We want to decrease the peak, so find i where the "peak contribution" is lowest
        # and j where it's highest, then move mass from j to i

        # Simplified: find i that maximizes the "descent direction"
        # grad at max position tells us which coords contribute most to the peak
        max_idx = int(np.argmax(c))
        sg = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        # Source: where sg is highest (contributes most to peak)
        # Destination: where sg is lowest
        src = int(np.argmax(sg))
        dst = int(np.argmin(sg))
        if src == dst:
            break

        # Try to move mass from src to dst
        # Find max delta such that a[src] - delta >= 0
        delta = min(a[src] * 0.5, 0.1 * a.mean())
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                break

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cosine(n, alpha=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic(n, gamma=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + gamma * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _full_optimize(a0, deadline, verbose=True):
    """Run full optimization pipeline on a single candidate."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    if verbose:
        print(f"    Init: {best_sc:.10f} (n={n})")

    # Phase 1: L-BFGS smooth (main workhorse)
    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 120)
        a1, s1 = _lbfgs_smooth(a, d1, beta_init=0.08, beta_min=0.002,
                                beta_decay=0.999, max_iter=3000)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        if verbose:
            print(f"    L-BFGS: {best_sc:.10f}")

    # Phase 2: Projected subgradient
    if time.monotonic() < deadline - 20:
        d2 = min(deadline - 20, time.monotonic() + 60)
        a2, s2 = _projected_subgradient(best_a, d2, max_iter=10000)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        if verbose:
            print(f"    Subgrad: {best_sc:.10f}")

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 15:
        d3 = min(deadline - 15, time.monotonic() + 80)
        a3 = _lp_refine(best_a, d3, max_iter=500, inflate=1.001, blend=0.35)
        s3 = _score(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        if verbose:
            print(f"    LP: {best_sc:.10f}")

    # Phase 4: Fine L-BFGS
    if time.monotonic() < deadline - 10:
        d4 = min(deadline - 10, time.monotonic() + 60)
        a4, s4 = _lbfgs_smooth(best_a, d4, beta_init=0.02, beta_min=0.001,
                                beta_decay=0.9995, max_iter=3000)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        if verbose:
            print(f"    L-BFGS2: {best_sc:.10f}")

    # Phase 5: Coordinate descent
    if time.monotonic() < deadline - 5:
        d5 = min(deadline - 5, time.monotonic() + 30)
        a5, s5 = _coordinate_descent(best_a, d5, max_iter=10000)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        if verbose:
            print(f"    CoordDesc: {best_sc:.10f}")

    # Phase 6: Final LP polish
    if time.monotonic() < deadline - 2:
        d6 = min(deadline - 2, time.monotonic() + 20)
        a6 = _lp_refine(best_a, d6, max_iter=200, inflate=1.0005, blend=0.15)
        s6 = _score(a6)
        if s6 < best_sc:
            best_a, best_sc = a6, s6
        if verbose:
            print(f"    FinalLP: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: L-BFGS smooth max + projected subgradient + LP spreading.
    Explores multiple sequence lengths (512, 1024, 2048, 4096) and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
        p2048 = _interpolate_profile(parent, 2048)
        p4096 = _interpolate_profile(parent, 4096)
    else:
        p512 = _make_ramp(512, 1.5)
        p1024 = _make_ramp(1024, 1.5)
        p2048 = _make_ramp(2048, 1.5)
        p4096 = _make_ramp(4096, 1.5)

    # === Priority 1: Optimize parent at n=1024 (most promising) ===
    print("=== Phase A: Parent @ n=1024 ===")
    t_mid = min(deadline - 600, t_start + 400)
    a_best, s_best = _full_optimize(p1024, t_mid, verbose=True)
    if s_best < best_score:
        best_score = s_best
        best_seq = a_best.copy()
    print(f"  After Phase A: {best_score:.10f}")

    # === Priority 2: Optimize parent at n=512 ===
    if time.monotonic() < deadline - 400:
        print("=== Phase B: Parent @ n=512 ===")
        t_mid = min(deadline - 400, time.monotonic() + 250)
        a_best, s_best = _full_optimize(p512, t_mid, verbose=True)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
        print(f"  After Phase B: {best_score:.10f}")

    # === Priority 3: Optimize at n=2048 ===
    if time.monotonic() < deadline - 300:
        print("=== Phase C: Parent @ n=2048 ===")
        t_mid = min(deadline - 300, time.monotonic() + 200)
        a_best, s_best = _full_optimize(p2048, t_mid, verbose=True)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
        print(f"  After Phase C: {best_score:.10f}")

    # === Priority 4: Diverse initializations at n=1024 ===
    if time.monotonic() < deadline - 200:
        print("=== Phase D: Diverse inits @ n=1024 ===")
        n = 1024
        candidates = [
            _make_ramp(n, 0.8),
            _make_ramp(n, 1.2),
            _make_ramp(n, 2.0),
            _make_cosine(n, 0.3),
            _make_cosine(n, 0.8),
            _make_quadratic(n, 0.5),
            _make_quadratic(n, 1.5),
        ]
        # Random perturbations of best
        if best_seq is not None:
            b1024 = _interpolate_profile(best_seq, n)
            for _ in range(3):
                pert = b1024 * (1.0 + 0.2 * rng.normal(size=n))
                pert = np.maximum(pert, 0)
                if pert.sum() > 0:
                    candidates.append(pert / pert.sum())

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 200:
                break
            per_budget = (deadline - 200 - time.monotonic()) / max(1, len(candidates) - i)
            if per_budget < 15:
                break
            d_i = time.monotonic() + min(per_budget, 40.0)
            # Quick L-BFGS only
            a_q, s_q = _lbfgs_smooth(a0, d_i, beta_init=0.06, beta_min=0.005,
                                      beta_decay=0.999, max_iter=1500)
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"  D-{i}: {s_q:.10f}")

    # === Priority 5: Try n=4096 ===
    if time.monotonic() < deadline - 150:
        print("=== Phase E: Parent @ n=4096 ===")
        t_mid = min(deadline - 150, time.monotonic() + 120)
        a_best, s_best = _full_optimize(p4096, t_mid, verbose=True)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
        print(f"  After Phase E: {best_score:.10f}")

    # === Priority 6: Final intensive optimization on best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"=== Phase F: Final focus (best={best_score:.10f}) ===")
        best_n = best_seq.size

        # Focus at the current best length
        d_f1 = min(deadline - 8, time.monotonic() + 80)
        a_f1, s_f1 = _lbfgs_smooth(best_seq, d_f1, beta_init=0.015, beta_min=0.0005,
                                    beta_decay=0.9997, max_iter=5000)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Final L-BFGS: {best_score:.10f}")

        # Quick subgradient
        if time.monotonic() < deadline - 5:
            d_f2 = min(deadline - 5, time.monotonic() + 20)
            a_f2, s_f2 = _projected_subgradient(best_seq, d_f2, max_iter=5000)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final Subgrad: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 3:
            d_f3 = min(deadline - 3, time.monotonic() + 15)
            a_f3 = _lp_refine(best_seq, d_f3, max_iter=200, inflate=1.0003, blend=0.1)
            s_f3 = _score(a_f3)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Final LP: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(256, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                # Use this as the authoritative score
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
