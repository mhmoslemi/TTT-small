def _smooth_max_val(c, beta):
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        return m
    return m + beta * np.log(s)


def _smooth_max_weights(c, beta):
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _grad_conv_w(a, w):
    """Gradient of sum_k w[k]*conv_k wrt a, computed via FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1]
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    h = np.fft.irfft(fw * fa, N)[n - 1:2 * n - 1]
    return 2.0 * h


def _lbfgs_smooth_max(a0, deadline, max_iter_per_phase=200, beta_start=0.06, beta_end=0.004, n_phases=6):
    """L-BFGS-B optimization with multi-phase beta annealing."""
    n = len(a0)
    a0 = np.asarray(a0, dtype=np.float64)
    a0 = np.maximum(a0, 1e-12)
    a0 = a0 / a0.sum()

    z0 = np.log(a0)
    z0 -= z0.mean()

    best_a = a0.copy()
    best_score = float(2.0 * n * _fft_convolve(a0).max())

    betas = np.geomspace(beta_start, beta_end, n_phases)
    z = z0.copy()

    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        beta = betas[phase]

        def obj(z_vec):
            z_shifted = z_vec - z_vec.max()
            a = np.exp(z_shifted)
            a = a / a.sum()
            c = _fft_convolve(a)
            return 2.0 * n * _smooth_max_val(c, beta)

        def grad(z_vec):
            z_shifted = z_vec - z_vec.max()
            a = np.exp(z_shifted)
            a = a / a.sum()
            c = _fft_convolve(a)
            w = _smooth_max_weights(c, beta)
            grad_a = _grad_conv_w(a, w)
            dot = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - dot)
            grad_z -= grad_z.mean()
            return grad_z

        try:
            res = minimize(obj, z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_phase, 'ftol': 1e-14,
                                   'gtol': 1e-9, 'maxfun': 5000, 'maxls': 50})
            z = res.x
        except Exception:
            break

        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        score_new = float(2.0 * n * _fft_convolve(a_new).max())
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

    return best_a, best_score


def _adam_smooth_max(a0, deadline, max_iter=30000, lr=0.02, beta_factor=0.03):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    beta = max(beta_factor * float(c.max()), 1e-10)

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weights(c, beta)
        grad_a = _grad_conv_w(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = best_a * 0 + np.log(np.maximum(best_a, 1e-12))
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.001, lr_cur * 0.94)
            if no_imp > max(4000, 5 * n):
                rng = np.random.default_rng(1000 + it)
                z = np.log(np.maximum(best_a, 1e-12)) + 0.25 * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= z.mean()
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr_cur = lr
                no_imp = 0
                continue

        if it % 100 == 0:
            beta *= 0.82

    return best_a, best_score


def _topk_subgradient(a, deadline, top_k=5, max_iter=8000):
    """Subgradient method targeting top-k peaks simultaneously."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    base_lr = 0.4 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        weights = c[top_idx]
        w_sum = float(weights.sum())
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        w = np.zeros(len(c))
        for i, idx in enumerate(top_idx):
            w[int(idx)] = weights[i]

        sg = _grad_conv_w(a, w)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score_new = float(2.0 * n * _fft_convolve(a_new).max())
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break
        a = a_new

    return best_a, best_score


def _lp_refine(a, deadline, max_iter=800):
    """LP-based spreading refinement."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    blend = 0.3
    inflate = 1.001

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate

        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.01, inflate * 1.06)
            if no_imp > 120:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score_new = float(2.0 * n * _fft_convolve(a_new).max())
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.9)
            if no_imp > 150:
                break
        a = a_new

    return best_a


def _interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64)
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        if s > 1e-30:
            return p / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _make_diverse_init(n, seed):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    candidates = []

    # Linear slope
    a = 1.0 + 1.5 * (t - 0.5)
    candidates.append(np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum())

    # Steeper linear
    a = 1.0 + 2.5 * (t - 0.5)
    candidates.append(np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum())

    # Cosine
    a = 1.0 + 0.8 * np.cos(2 * np.pi * t)
    candidates.append(np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum())

    # Two-hump
    a = 1.0 + 1.2 * np.cos(4 * np.pi * t)
    candidates.append(np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum())

    # Random smooth
    coeffs = rng.normal(size=6)
    a = 1.0
    for k in range(1, 6):
        a += coeffs[k] * np.cos(2 * np.pi * k * t) * (0.3 / (k + 1))
    candidates.append(np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum())

    # Quadratic
    a = 1.0 + 0.5 * (2 * t - 1.0) ** 2
    candidates.append(a / a.sum())

    return candidates


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining L-BFGS-B smooth-max (with beta annealing),
    Adam refinement, multi-peak subgradient, LP polishing, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = parent.copy()
        p256 = _interpolate_profile(parent, 256)
    else:
        p512 = _make_diverse_init(512, 0)[0]
        p1024 = _interpolate_profile(p512, 1024)
        p256 = _interpolate_profile(p512, 256)

    # === Strategy 1: L-BFGS-B multi-phase on n=512 (coarse, fast) ===
    print("=== S1: L-BFGS-B @ n=512 ===")
    d1 = min(deadline - 700, t_start + 280)
    a1, s1 = _lbfgs_smooth_max(p512, d1, max_iter_per_phase=250,
                                beta_start=0.06, beta_end=0.004, n_phases=6)
    if s1 < best_score:
        best_score, best_seq = s1, a1.copy()
    print(f"  S1: {best_score:.10f}")

    # === Strategy 2: L-BFGS-B on n=1024 (the main target length) ===
    if time.monotonic() < deadline - 500:
        print("=== S2: L-BFGS-B @ n=1024 ===")
        best_1024 = _interpolate_profile(best_seq, 1024)
        d2 = min(deadline - 500, time.monotonic() + 250)
        a2, s2 = _lbfgs_smooth_max(best_1024, d2, max_iter_per_phase=200,
                                    beta_start=0.05, beta_end=0.003, n_phases=6)
        if s2 < best_score:
            best_score, best_seq = s2, a2.copy()
        print(f"  S2: {best_score:.10f}")

    # === Strategy 3: Adam refinement on current best ===
    if time.monotonic() < deadline - 350:
        print("=== S3: Adam refinement ===")
        d3 = min(deadline - 350, time.monotonic() + 200)
        a3, s3 = _adam_smooth_max(best_seq, d3, max_iter=20000, lr=0.015, beta_factor=0.02)
        if s3 < best_score:
            best_score, best_seq = s3, a3.copy()
        print(f"  S3: {best_score:.10f}")

    # === Strategy 4: Multi-peak subgradient ===
    if time.monotonic() < deadline - 200:
        print("=== S4: Multi-peak subgradient (k=5) ===")
        d4 = min(deadline - 200, time.monotonic() + 100)
        a4, s4 = _topk_subgradient(best_seq, d4, top_k=5, max_iter=8000)
        if s4 < best_score:
            best_score, best_seq = s4, a4.copy()
        print(f"  S4: {best_score:.10f}")

        # Also try k=3
        if time.monotonic() < deadline - 150:
            d4b = min(deadline - 150, time.monotonic() + 60)
            a4b, s4b = _topk_subgradient(best_seq, d4b, top_k=3, max_iter=5000)
            if s4b < best_score:
                best_score, best_seq = s4b, a4b.copy()
            print(f"  S4b (k=3): {best_score:.10f}")

    # === Strategy 5: LP refinement ===
    if time.monotonic() < deadline - 120:
        print("=== S5: LP refinement ===")
        d5 = min(deadline - 120, time.monotonic() + 80)
        a5 = _lp_refine(best_seq, d5, max_iter=600)
        s5 = _score(a5)
        if s5 < best_score:
            best_score, best_seq = s5, a5.copy()
        print(f"  S5: {best_score:.10f}")

    # === Strategy 6: Try nearby lengths ===
    if time.monotonic() < deadline - 100:
        print("=== S6: Multi-length exploration ===")
        for n_try in [960, 1152, 1280]:
            if time.monotonic() > deadline - 100:
                break
            p_try = _interpolate_profile(best_seq, n_try)
            d6i = min(deadline - 100, time.monotonic() + 35)
            a6, s6 = _lbfgs_smooth_max(p_try, d6i, max_iter_per_phase=80,
                                        beta_start=0.05, beta_end=0.01, n_phases=4)
            if s6 < best_score:
                best_score, best_seq = s6, a6.copy()
                print(f"  S6 n={n_try}: {best_score:.10f}")

    # === Strategy 7: Diverse initializations at best length ===
    if time.monotonic() < deadline - 80:
        print("=== S7: Diverse inits ===")
        n_best = best_seq.size
        d7_total = deadline - 80 - time.monotonic()
        if d7_total > 30:
            inits = _make_diverse_init(n_best, seed + 7)
            # Also add perturbed best
            rng = np.random.default_rng(seed + 100)
            p_pert = best_seq * (1.0 + 0.1 * rng.normal(size=n_best))
            p_pert = np.maximum(p_pert, 0.0)
            inits.append(p_pert / p_pert.sum())

            per_time = d7_total / len(inits)
            for i, a0 in enumerate(inits):
                if time.monotonic() > deadline - 80:
                    break
                d7i = min(deadline - 80, time.monotonic() + per_time)
                a7, s7 = _lbfgs_smooth_max(a0, d7i, max_iter_per_phase=60,
                                            beta_start=0.05, beta_end=0.01, n_phases=3)
                if s7 < best_score:
                    best_score, best_seq = s7, a7.copy()
                    print(f"  S7-{i}: {best_score:.10f}")

    # === Strategy 8: Final intensive polish ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"=== S8: Final polish (score={best_score:.10f}) ===")
        # Short Adam
        d8a = min(deadline - 3, time.monotonic() + 30)
        a8, s8 = _adam_smooth_max(best_seq, d8a, max_iter=8000, lr=0.005, beta_factor=0.008)
        if s8 < best_score:
            best_score, best_seq = s8, a8.copy()

        # Short subgradient
        if time.monotonic() < deadline - 2:
            a8b, s8b = _topk_subgradient(best_seq, deadline - 2, top_k=3, max_iter=2000)
            if s8b < best_score:
                best_score, best_seq = s8b, a8b.copy()

        # Short L-BFGS
        if time.monotonic() < deadline - 1:
            a8c, s8c = _lbfgs_smooth_max(best_seq, deadline - 1, max_iter_per_phase=50,
                                          beta_start=0.02, beta_end=0.002, n_phases=3)
            if s8c < best_score:
                best_score, best_seq = s8c, a8c.copy()

        print(f"  S8: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = p512
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    # Authoritative evaluation
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
