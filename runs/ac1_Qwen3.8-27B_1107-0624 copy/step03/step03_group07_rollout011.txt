"""
LP-guided hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP direction finding (maximize sum(g) s.t. a*g <= M), soft-max
gradient descent in logit space, coordinate descent, and multi-scale search.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-10 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p)
    s = result.sum()
    return result / s if s > 1e-10 else np.ones(n_new) / n_new


def soft_max_weights(c, beta):
    """Compute softmax weights for soft-max approximation."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = rfft(af, N)
    fw = rfft(w, N)
    conv = irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def lp_direction(a, n_top=150, slack_factor=1.001):
    """
    Find g maximizing sum(g) subject to (a*g)[k] <= M*slack for top constraints, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    c_full = full_convolve_fft(a)
    M = float(c_full.max())
    
    if n_top < len(c_full):
        top_idx = np.argsort(c_full)[-n_top:]
    else:
        top_idx = np.arange(len(c_full))
    
    num_c = len(top_idx)
    A = np.zeros((num_c, n))
    
    for i, k in enumerate(top_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx_start = k - j_end
            idx_end = k - j_start
            A[i, j_start:j_end + 1] = a[idx_start:idx_end + 1][::-1]
    
    b = np.full(num_c, M * slack_factor + 1e-12)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status != 0:
            return None
        g = res.x
        s = g.sum()
        if s < 1e-10:
            return None
        return g / s
    except Exception:
        return None


def gradient_step(z, a, c, beta, lr, b1, b2, m, v, t_adam):
    """One Adam gradient step in logit space. Returns updated z, m, v."""
    _, w = soft_max_weights(c, beta)
    h = compute_h(w, a)
    grad_a = 2.0 * h
    dot = float(np.dot(a, grad_a))
    dz = a * (grad_a - dot)
    
    t_adam += 1
    m = b1 * m + (1.0 - b1) * dz
    v = b2 * v + (1.0 - b2) * (dz * dz)
    mh = m / (1.0 - b1 ** t_adam)
    vh = v / (1.0 - b2 ** t_adam)
    z_new = z - lr * mh / (np.sqrt(vh) + 1e-8)
    return z_new, m, v, t_adam


def logit_to_a(z):
    """Convert logit vector to normalized non-negative array."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(z) / len(z)
    return a / s


def optimize_gradient(a0, deadline, rng, lr=0.03, beta_factor=0.06,
                      beta_decay=0.95, max_iters=None):
    """Pure gradient descent in logit space."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6
    
    b1, b2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        if max_iters is not None and t_adam >= max_iters:
            break
        z, m, v, t_adam = gradient_step(z, a0, c, beta, current_lr, b1, b2, m, v, t_adam)
        
        if not np.all(np.isfinite(z)):
            z = best_z = np.log(best_a)
            z = z - z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = logit_to_a(z)
        
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if t_adam % 50 == 0:
            beta = max(1e-10, beta * beta_decay)
        if no_improve > 500:
            current_lr = max(0.003, current_lr * 0.9)
        
        # best_z for restart
        best_z = np.log(best_a) - np.log(best_a).mean()
    
    return best_a, best_score

    b1, b2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr_grad
    step_count = 0
    
    print(f"  Starting hybrid: n={n}, initial score={best_score:.10f}")
    
    while time.monotonic() < deadline:
        phase = step_count % (lp_every + coord_every)
        
        if phase < lp_every:
            # Gradient step
            z, m, v, t_adam = gradient_step(z, a, c, beta, current_lr, b1, b2, m, v, t_adam)
            if not np.all(np.isfinite(z)):
                z = np.log(best_a) - np.log(best_a).mean()
                m.fill(0)
                v.fill(0)
                t_adam = 0
                step_count += 1
                continue
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a = logit_to_a(z)
            
        elif phase < lp_every + 3:
            # LP step
            g = lp_direction(a, n_top=n_top)
            if g is not None:
                t_blend = lp_blend * (1.0 + 0.5 * np.exp(-no_improve / 100.0))
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                c_new = full_convolve_fft(a_new)
                score_new = 2.0 * n * c_new.max()
                if score_new < best_score + 0.01:
                    a = a_new
                    c = c_new
                    max_c = float(c.max())
                    z = np.log(a) - np.log(a).mean()
                
        else:
            # Coordinate descent on random coordinates
            coords = rng.choice(n, size=min(n_coord, n), replace=False)
            for i in coords:
                # Optimize a[i] in 1D
                # Affected c[k] for k in [i, i+n-1]
                # c[k] = 2*a[i]*a[k-i] + rest (for k != 2i)
                # c[2i] = a[i]^2 + rest (for k = 2i)
                
                # Find optimal a[i] via ternary search on [0, upper]
                upper = max(a[i] * 3, 2 * a.max())
                
                def obj_1d(x):
                    # Compute max of affected conv values
                    k_min = i
                    k_max = min(i + n - 1, 2 * n - 2)
                    if k_min > k_max:
                        return c.max()
                    
                    # Affected terms
                    # For k in [k_min, k_max]:
                    # c[k] has term involving a[i]: 2*a[i]*a[k-i] if k-i != i
                    #                                    a[i]^2 if k-i == i (i.e., k=2i)
                    
                    # Get baseline (current c minus the a[i] contribution)
                    base = c.copy()
                    # Remove a[i] contribution from base
                    for k in range(max(0, i), min(2 * n - 1, i + n)):
                        j = k - i
                        if 0 <= j < n:
                            if k == 2 * i:
                                base[k] -= a[i] * a[i]
                            else:
                                base[k] -= 2 * a[i] * a[j]
                    
                    # Add new x contribution
                    max_val = base.max()
                    for k in range(max(0, i), min(2 * n - 1, i + n)):
                        j = k - i
                        if 0 <= j < n:
                            if k == 2 * i:
                                val = base[k] + x * x
                            else:
                                val = base[k] + 2 * x * a[j]
                            if val > max_val:
                                max_val = val
                    return max_val
                
                # Ternary search
                lo, hi = 0.0, upper
                for _ in range(30):
                    m1 = lo + (hi - lo) / 3
                    m2 = hi - (hi - lo) / 3
                    if obj_1d(m1) < obj_1d(m2):
                        hi = m2
                    else:
                        lo = m1
                
                a[i] = (lo + hi) / 2.0
            
            # Renormalize
            s = a.sum()
            if s > 1e-15:
                a = a / s
            c = full_convolve_fft(a)
        
        # Evaluate
        max_c = float(c.max())
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive
        if step_count % 30 == 0:
            beta = max(1e-10, beta * 0.96)
        if no_improve > 300:
            current_lr = max(0.003, current_lr * 0.92)
        if no_improve > 800:
            # Restart with noise
            z = np.log(best_a) + 0.1 * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            a = logit_to_a(z)
            c = full_convolve_fft(a)
            max_c = float(c.max())
            beta = max(1e-8, beta_factor * max_c)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            current_lr = lr_grad
            no_improve = 0
            step_count = 0
        
        step_count += 1
    
    return best_a, best_score


def make_ramp(n, p=1.0, floor=0.05):
    """Create a ramp profile: high at start, low at end."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + floor
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_skewed(n, alpha=2.0):
    """Create a Beta-like skewed distribution."""
    t = np.linspace(0.01, 0.99, n)
    a = t ** (alpha - 1) * (1 - t) ** 0.5
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_two_bump(n, w1=0.3, w2=0.2, o1=0.2, o2=0.7, h1=1.0, h2=0.6):
    """Two-bump profile."""
    t = np.linspace(0.0, 1.0, n)
    a = h1 * np.exp(-((t - o1) / w1) ** 2) + h2 * np.exp(-((t - o2) / w2) ** 2) + 0.01
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def diverse_starts(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(make_ramp(n, p, 0.01))
    for alpha in [1.5, 2.0, 3.0, 4.0]:
        profiles.append(make_skewed(n, alpha))
    for w1 in [0.2, 0.3, 0.4]:
        for o1 in [0.1, 0.3]:
            profiles.append(make_two_bump(n, w1, 0.2, o1, 0.7, 1.0, 0.5))
    # Low-frequency random
    for _ in range(5):
        coeffs = rng.uniform(-0.3, 0.3, 4)
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a += cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining LP-guided optimization, gradient descent,
    and coordinate descent. Tries multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass
    
    print(f"=== LP-Hybrid Search. Budget={budget_s}s ===")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)
    
    # ===== Phase 1: Quick exploration at n=512 =====
    n1 = 512
    phase1_end = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: n={n1} (exploration, {200}s) ---")
    
    starts = diverse_starts(n1, rng)
    if init_seq is not None:
        starts.insert(0, interp_profile(init_seq, n1))
    
    best_p1 = None
    best_s1 = np.inf
    for idx, p in enumerate(starts):
        if time.monotonic() > phase1_end - 5:
            break
        sc = compute_score(p)
        if sc >= best_s1 + 0.15 and idx > 0:
            continue
        dl = min(phase1_end, time.monotonic() + 15.0)
        result, _ = optimize_gradient(p, dl, rng, lr=0.04, beta_factor=0.08)
        sc_f = compute_score(result)
        if sc_f < best_s1:
            best_s1 = sc_f
            best_p1 = result.copy()
        update(result, sc_f)
        if idx < 8:
            print(f"  Start {idx}: {sc_f:.8f}")
    
    if best_p1 is not None:
        print(f"  Phase 1 best: {best_s1:.10f}")
    update(best_p1, best_s1)
    
    # ===== Phase 2: Hybrid LP+gradient at n=512 and n=1024 =====
    print(f"\n--- Phase 2: Hybrid optimization ---")
    
    # Try n=512 first
    if best_p1 is not None:
        p2 = best_p1
    elif init_seq is not None:
        p2 = interp_profile(init_seq, 512)
    else:
        p2 = make_ramp(512, 1.5)
    
    dl2a = min(deadline, t_start + 600.0)
    if time.monotonic() < dl2a:
        print(f"  Hybrid at n=512...")
        result, sc = hybrid_lp_gradient(p2, dl2a, rng, lr_grad=0.02,
                                         beta_factor=0.04, lp_every=10,
                                         lp_blend=0.08, n_top=80,
                                         coord_every=30, n_coord=3)
        sc_f = compute_score(result)
        update(result, sc_f)
        print(f"  n=512 hybrid result: {sc_f:.10f}")
    
    # Try n=1024
    if time.monotonic() < deadline - 200:
        n3 = 1024
        if best_seq is not None:
            p3 = interp_profile(best_seq, n3)
        else:
            p3 = interp_profile(p2, n3)
        
        sc3_init = compute_score(p3)
        update(p3, sc3_init)
        print(f"  Interpolated to n={n3}: {sc3_init:.10f}")
        
        dl2b = min(deadline, t_start + 1200.0)
        if time.monotonic() < dl2b:
            print(f"  Hybrid at n={n3}...")
            result3, sc3 = hybrid_lp_gradient(p3, dl2b, rng, lr_grad=0.015,
                                               beta_factor=0.03, lp_every=12,
                                               lp_blend=0.06, n_top=100,
                                               coord_every=40, n_coord=3)
            sc3_f = compute_score(result3)
            update(result3, sc3_f)
            print(f"  n={n3} hybrid result: {sc3_f:.10f}")
    
    # ===== Phase 3: Try n=2048 if time permits =====
    if time.monotonic() < deadline - 300 and best_score > 1.503:
        n4 = 2048
        if best_seq is not None:
            p4 = interp_profile(best_seq, n4)
        else:
            p4 = interp_profile(p2, n4)
        
        sc4_init = compute_score(p4)
        update(p4, sc4_init)
        print(f"\n  Trying n={n4}: init={sc4_init:.10f}")
        
        dl3 = min(deadline, t_start + 1500.0)
        if time.monotonic() < dl3:
            result4, sc4 = hybrid_lp_gradient(p4, dl3, rng, lr_grad=0.01,
                                               beta_factor=0.02, lp_every=15,
                                               lp_blend=0.05, n_top=120,
                                               coord_every=50, n_coord=3)
            sc4_f = compute_score(result4)
            update(result4, sc4_f)
            print(f"  n={n4} hybrid result: {sc4_f:.10f}")
    
    # ===== Phase 4: Final polish =====
    print(f"\n--- Phase 4: Final polish ---")
    if best_seq is not None:
        n_polish = best_seq.size
        print(f"  Polishing n={n_polish}, current best={best_score:.10f}")
        
        # Multiple short gradient refinements
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            p0 = best_seq.copy()
            p0 = p0 + 0.001 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            
            dl = min(deadline, time.monotonic() + 60.0)
            r, _ = optimize_gradient(p0, dl, rng, lr=0.008 * (0.8 ** rnd),
                                      beta_factor=0.02)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Polish {rnd}: {sc_f:.10f}")
        
        # One more LP-guided pass
        if time.monotonic() < deadline - 30:
            dl = min(deadline, time.monotonic() + 50.0)
            r, sc = hybrid_lp_gradient(best_seq, dl, rng, lr_grad=0.008,
                                        beta_factor=0.02, lp_every=8,
                                        lp_blend=0.04, n_top=60,
                                        coord_every=20, n_coord=2)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Final LP pass: {sc_f:.10f}")
    
    # ===== Check smaller n values =====
    if best_seq is not None and best_seq.size > 512:
        for n_try in [256, 384, 512, 768]:
            if time.monotonic() > deadline - 10:
                break
            p_s = interp_profile(best_seq, n_try)
            sc_s = compute_score(p_s)
            update(p_s, sc_s)
            if sc_s < best_score + 0.02 and time.monotonic() < deadline - 20:
                dl_s = min(deadline, time.monotonic() + 20.0)
                r_s, _ = optimize_gradient(p_s, dl_s, rng, lr=0.015, beta_factor=0.03)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Small n={n_try}: {sc_s2:.10f}")
    
    print(f"\n=== FINAL: n={best_seq.size if best_seq is not None else '?'}, "
          f"score={best_score:.10f} ===")
    
    if best_seq is not None:
        final = best_seq
    else:
        final = make_ramp(512, 1.5)
    
    final = np.maximum(final, 0.0)
    final = np.minimum(final, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = make_ramp(256, 1.5)
    
    # Validate
    try:
        ev = float(evaluate_sequence([float(x) for x in final]))
        print(f"evaluate_sequence: {ev:.10f}")
        if not np.isfinite(ev) or ev > best_score:
            print("WARNING: evaluate_sequence disagrees, using raw score")
    except NameError:
        pass
    except Exception as e:
        print(f"Eval check: {e}")
    
    return [float(x) for x in final]
