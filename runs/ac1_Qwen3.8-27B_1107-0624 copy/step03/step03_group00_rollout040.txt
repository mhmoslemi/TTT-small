def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

    return a / s

def _autocorr_grad_at_k(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _autocorr_subgrad_topm(a, c, m=5):
    """Subgradient of max(c) using average of top-m gradients."""
    n = len(a)
    m = min(m, c.size)
    top_m = np.argsort(c)[::-1][:m]
    grad = np.zeros(n)
    for i in range(m):
        k = int(top_m[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += 2.0 * a[k - idx]
    grad /= m
    return grad


def _to_z_grad(g_a, a):
    """Convert gradient in a-space to z-space (logit space)."""
    # g_z[j] = a[j] * (g_a[j] - dot(a, g_a))
    return a * (g_a - float(np.dot(a, g_a)))

    best_p = p0.copy()
    best_z = z.copy()
    best_score = score
    no_imp = 0
    m = m_start
    
    while time.monotonic() < deadline:

        z_new = z - lr * (g_z / g_norm) * float(np.linalg.norm(p - p0))  # scaled step

        z_new = np.clip(z_new, -50.0, 50.0)

            # Accept
            z = z_new
            p = p_new
            c = c_new
            maxc = maxc_new
            score = score_new
            
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

                m = min(m_end, m + 1)
        else:

                no_imp = 0

def _lp_max_sum(a, M):
    """LP: max sum(b) s.t. (a*b)(k) <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_max_min(a, M):
    """LP: max t s.t. (a*b)(k) >= t, (a*b)(k) <= M, b >= 0, sum(b)=1.
    This directly targets flatness of a*b."""
    n = len(a)
    # Variables: b[0..n-1], t
    # Constraints: (a*b)(k) - t <= 0 for all k (lower bound)
    #              (a*b)(k) <= M for all k (upper bound)
    #              sum(b) = 1
    #              b >= 0, t free (but bounded by M)
    num_vars = n + 1  # b_0..b_{n-1}, t
    rows_ub = []
    cols_ub = []
    vals_ub = []
    b_ub_list = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            # (a*b)(k) - t <= 0
            rows_ub.append(len(b_ub_list))
            for i, j in enumerate(idx):
                cols_ub.append(j)
                vals_ub.append(float(a[k - j]))
            cols_ub.append(n)  # t variable
            vals_ub.append(-1.0)
            b_ub_list.append(0.0)
            # (a*b)(k) <= M
            r2 = len(b_ub_list)
            for i, j in enumerate(idx):
                rows_ub.append(r2)
                cols_ub.append(j)
                vals_ub.append(float(a[k - j]))
            b_ub_list.append(M)

    A_ub = csr_matrix((vals_ub, (rows_ub, cols_ub)), shape=(len(b_ub_list), num_vars))
    # Objective: maximize t -> minimize -t
    c_obj = np.zeros(num_vars)
    c_obj[n] = -1.0
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, M)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub_list),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None


def _lp_refine(a, deadline, t=0.35, max_solves=25, use_max_min=False):
    """Iterative LP-based refinement."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    cur_t = t

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        if use_max_min:
            b = _lp_max_min(a, M)
        else:
            b = _lp_max_sum(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - cur_t) * a + cur_t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        cur_t = max(0.1, cur_t * 0.92)

    return current_best, current_score


def _make_profile_hann(n):
    if n <= 1:
        return np.array([1.0])
    t = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_hamming(n, alpha=0.54):
    if n <= 1:
        return np.array([1.0])
    t = np.arange(n)
    a = alpha + (1.0 - alpha) * np.cos(2.0 * np.pi * t / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_poly(n, coeffs):
    """Polynomial profile: a_i = prod/sum of poly(i/(n-1))."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for c in coeffs:
        a = a * (1.0 + c * (2.0 * t - 1.0))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_gaussian(n, sigma):
    t = np.linspace(-1.0, 1.0, n)
    a = np.exp(-0.5 * (t / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function: finds best sequence minimizing 2*n*max(conv(a,a))/sum(a)^2."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Generate diverse candidates ---
    phase1_end = t_start + 60.0
    candidates = []

    profiles_func = [
        ('hann', lambda n: _make_profile_hann(n)),
        ('hamm0.5', lambda n: _make_profile_hamming(n, 0.5)),
        ('hamm0.54', lambda n: _make_profile_hamming(n, 0.54)),
        ('lin1.0', lambda n: _make_profile_linear(n, 1.0)),
        ('lin1.5', lambda n: _make_profile_linear(n, 1.5)),
        ('lin2.0', lambda n: _make_profile_linear(n, 2.0)),
        ('exp0.5', lambda n: _make_profile_exponential(n, 0.5)),
        ('exp1.0', lambda n: _make_profile_exponential(n, 1.0)),
        ('exp2.0', lambda n: _make_profile_exponential(n, 2.0)),
        ('gauss0.7', lambda n: _make_profile_gaussian(n, 0.7)),
        ('gauss0.8', lambda n: _make_profile_gaussian(n, 0.8)),
        ('gauss1.0', lambda n: _make_profile_gaussian(n, 1.0)),
        ('twobump', lambda n: _make_profile_two_bump(n, 0.3, 0.7, 0.2)),
        ('twobump2', lambda n: _make_profile_two_bump(n, 0.25, 0.75, 0.15)),
        ('cos1_0.8', lambda n: _make_profile_cosine(n, 1, 0.8)),
        ('cos1_0.5', lambda n: _make_profile_cosine(n, 1, 0.5)),
        ('cos2_0.6', lambda n: _make_profile_cosine(n, 2, 0.6)),
        ('polyn1', lambda n: _make_profile_poly(n, [1.5])),
        ('polyn2', lambda n: _make_profile_poly(n, [2.0, -1.0])),
    ]

    for n in [256, 512, 1024, 2048]:
        for label, fn in profiles_func:
            try:
                p = fn(n)
                if p is not None:
                    sc = _score_norm(p)
                    if np.isfinite(sc):
                        candidates.append((sc, p, n, label))
            except Exception:
                pass

    # Add parent at various scales
    if parent is not None:
        for n in [256, 512, 1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    if np.isfinite(sc):
                        candidates.append((sc, pi, n, 'parent_interp'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            if np.isfinite(sc):
                candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Polish top candidates with Adam ---
    phase2_end = t_start + 700.0
    polished = []
    count = 0
    top_k = 8

    for sc, p, n, label in candidates[:top_k]:
        if time.monotonic() > phase2_end or count >= top_k:
            break
        per_end = min(phase2_end, time.monotonic() + 60.0)
        if per_end - time.monotonic() < 10:
            break
        p_b, sc_b = _adam_topm(p, per_end, rng, lr=0.06, m_start=3, m_end=30,
                               beta_frac=0.08, beta_decay=0.94,
                               no_imp_limit=200, noise_base=0.12)
        _update(p_b, sc_b, f'adam_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_pol'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2: best={best_score:.10f}")

    # --- Phase 3: LP refinement on best ---
    phase3_end = t_start + 1100.0
    if best_seq is not None and best_seq.size <= 2048:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")

        # Standard LP (max sum)
        p_lp, sc_lp = _lp_refine(best_seq, min(phase3_end - 60, time.monotonic() + 120),
                                  t=0.4, max_solves=20, use_max_min=False)
        _update(p_lp, sc_lp, 'LP_maxsum')

        # Max-min LP
        if time.monotonic() < phase3_end - 30:
            p_lp2, sc_lp2 = _lp_refine(best_seq, min(phase3_end - 10, time.monotonic() + 90),
                                        t=0.3, max_solves=15, use_max_min=True)
            _update(p_lp2, sc_lp2, 'LP_maxmin')

        # Adam polish after LP
        if time.monotonic() < phase3_end - 10:
            per = min(80.0, phase3_end - time.monotonic() - 5)
            if per > 15:
                p_ap, sc_ap = _adam_topm(best_seq, time.monotonic() + per, rng,
                                          lr=0.04, m_start=5, m_end=40,
                                          beta_frac=0.06, beta_decay=0.95,
                                          no_imp_limit=300, noise_base=0.08)
                _update(p_ap, sc_ap, 'post_LP_adam')

    # --- Phase 4: Multi-scale escalation ---
    phase4_end = t_start + 1450.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            per = min(60.0, phase4_end - time.monotonic() - 30)
            if per > 15:
                p_e, sc_e = _adam_topm(p_int, time.monotonic() + per, rng,
                                        lr=0.05, m_start=3, m_end=30,
                                        beta_frac=0.08, beta_decay=0.94,
                                        no_imp_limit=200, noise_base=0.1)
                _update(p_e, sc_e, f'escalate_n{tn}')

    # --- Phase 5: Multi-restart at best length ---
    phase5_end = deadline - 60.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 10:
                break
            # Different noise strategies
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            if i % 3 == 0:
                ns = 0.08 + 0.04 * i
                z = z + ns * rng.normal(size=n_best)
            elif i % 3 == 1:
                # Structured perturbation
                block = max(1, n_best // 16)
                noise = rng.normal(size=n_best // block)
                z = z + 0.1 * np.repeat(noise, block)[:n_best]
            else:
                # Sparse perturbation
                n_pert = max(5, n_best // 32)
                idx = rng.choice(n_best, size=n_pert, replace=False)
                z_pert = z.copy()
                z_pert[idx] += 0.3 * rng.normal(size=n_pert)
                z = z_pert

            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_topm(p_s, time.monotonic() + per, rng, lr=0.05,
                                    m_start=3, m_end=30, beta_frac=0.07,
                                    beta_decay=0.94, no_imp_limit=200, noise_base=0.1)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

            # Quick LP on result
            if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase5_end - 10:
                p_lpr, sc_lpr = _lp_refine(best_seq, time.monotonic() + 30,
                                            t=0.3, max_solves=8, use_max_min=False)
                _update(p_lpr, sc_lpr, f'LP_restart{i}')

    # --- Phase 6: Final polish ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 10:
            p_f, sc_f = _adam_topm(best_seq, time.monotonic() + per, rng,
                                    lr=0.02, m_start=10, m_end=50,
                                    beta_frac=0.04, beta_decay=0.96,
                                    no_imp_limit=150, noise_base=0.05)
            _update(p_f, sc_f, 'final_polish')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_hann(512)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_hann(512)
    print(f"FALLBACK n=512, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
