    return c

        # So D conv(a,a)[d] = conv(d, a) + conv(a, d) = 2 * conv(d, a)

        # w[i+j] appears when m-n+1+k = i+j, i.e., k = i+j-m+n-1
        # This is getting too complicated. Let me just use the formula directly.
        
        # Simple approach: grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (where w is zero-padded)
        # This is a cross-correlation. We can compute it via FFT.

"""
Optimized step function for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. Aggressive cooling schedule for smooth max (faster convergence to true max)
2. Exact top-k subgradient computation using FFT-based cross-correlation
3. Fast LP refinement with reduced constraint set (top-5 peaks only)
4. Structured perturbations for simulated annealing restarts
5. Adaptive phase allocation based on progress
6. Multi-length exploration (512, 768, 1024)

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _compute_grad_from_weights(a, w):
    """Compute gradient of smooth-max objective using precomputed weights w.
    grad_i = 2 * sum_j w[i+j] * a[j] (cross-correlation of w with a)
    """
    n = len(a)
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa), N)
    grad = 2.0 * cross[:n]
    return grad


def _compute_grad_topk(a, c, k):
    """Compute exact top-k subgradient using FFT.
    grad_i = (1/k) * sum_{j in top_k} 2*a[k_j - i] (for valid indices)
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    indicator = np.zeros(2 * n - 1)
    indicator[top_idx] = 1.0 / k
    N = _next_pow2(3 * n - 2)
    fi = np.fft.rfft(indicator, N)
    fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(fi * np.conj(fa), N)
    grad = 2.0 * cross[:n]
    return grad


def _logit_project(z):
    """Project to simplex via logit parameterization."""
    z = np.clip(z, -60.0, 60.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    return e / e.sum()


def _main_optimize(a, deadline, max_iter=80000, lr=0.06, use_topk_switch=True):
    """
    Main optimization: smooth max gradient descent with adaptive cooling,
    switching to exact top-k subgradient when beta is small enough.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.1 * c_max)
    beta_min = max(1e-10, c_max * 0.001)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    topk = 1
    last_c = c

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient
        if use_topk_switch and beta < c_max * 0.01:
            grad = _compute_grad_topk(a, last_c, topk)
        else:
            _, w = _smooth_max_weights(last_c, beta)
            grad = _compute_grad_from_weights(a, w)

        # Regularization
        grad += 1e-7 * a

        # Logit-space gradient
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            continue

        # Barzilai-Borwein
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_candidate = dot_ss / max(abs(dot_sg), 1e-20)
                bb_candidate = np.clip(bb_candidate, 0.1, 10.0)
                bb_scale = 0.6 * bb_scale + 0.4 * bb_candidate
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale

        Adam step
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        a = _logit_project(z_new)

        # Evaluate
        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)
        last_c = c

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.003, base_lr * 0.93)

        # Adaptive beta cooling - more aggressive
        if it % 50 == 0 and it > 50:
            # Cool beta based on progress
            if no_imp < 100:
                beta = max(beta_min, beta * 0.85)
            elif no_imp < 500:
                beta = max(beta_min, beta * 0.95)
            else:
                beta = max(beta_min, beta * 0.7)

            # Increase topk as we get more precise
            if use_topk_switch and beta < c_max * 0.005:
                topk = min(5, topk + 1)

        # Annealing restart
        if no_imp > max(4000, 6 * n):
            # Structured perturbation: shift mass from high-pressure to low-pressure
            c_sorted_idx = np.argsort(c)
            n_half = n // 2
            # Reduce top half, increase bottom half
            perturbation = np.zeros(n)
            perturbation[c_sorted_idx[:n_half]] = -0.5
            perturbation[c_sorted_idx[-n_half:]] = 0.5
            noise = 0.2 * np.random.default_rng(it * 104729 + 7).standard_normal(n)
            z_restart = best_z + 0.3 * perturbation + noise
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            a = _logit_project(z_restart)
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            c = _fft_convolve(a)
            beta = max(1e-6, 0.05 * float(c.max()))
            topk = 1
            no_imp = 0
            last_c = c

        z = z_new

    return best_a


def _lp_refine_fast(a, deadline, max_iter=5000, n_top=5, inflate=1.0005, blend=0.2):
    """FAST LP refinement: only use top-n_top constraints."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Get top constraint indices
        top_idx = np.argsort(c)[::-1][:n_top]

        # Build LP: max sum(g) s.t. conv(a,g)_k <= M_lp for k in top_idx, g >= 0
        # conv(a,g)_k = sum_i a[i]*g[k-i] for valid i
        m_cons = len(top_idx)
        A = np.zeros((m_cons, n))
        for row, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                A[row, i] = a[k - i]

        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(m_cons, M_lp),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.93)
                inflate_cur = min(1.001, inflate_cur * 0.97)
            if no_imp > 150:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _structured_perturb(a, rng, strength=0.3):
    """Create a structured perturbation that shifts mass from peaks to valleys."""
    n = len(a)
    c = _fft_convolve(a)
    # Find high-pressure and low-pressure regions
    top_half = np.argsort(c)[::-1][:n // 2]
    bot_half = np.argsort(c)[:n // 2]
    pert = np.zeros(n)
    for idx in top_half:
        k = idx
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            pert[i] -= a[k - i] * 0.5
    for idx in bot_half:
        k = idx
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            pert[i] += a[k - i] * 0.5
    # Add some random noise
    pert += strength * 0.1 * rng.standard_normal(n)
    return pert

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = [768]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    for ln in [512, 1024]:
        if ln not in lengths:
            lengths.append(ln)

    print(f"Trying lengths: {lengths}", flush=True)

    # Phase 1: Generate initializations and quick-evaluate
    print(f"Phase 1: Initialization, time_left={deadline - time.monotonic():.0f}s", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))
        for eta in [1.0, 2.0, 3.0]:
            init_pool.append((f'linear_{eta}', n_t, _make_linear_profile(n_t, eta)))
        for width in [0.2, 0.3, 0.5]:
            init_pool.append((f'bell_{width}', n_t, _make_bell_profile(n_t, width)))
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum_init(n_t, rng)))
        init_pool.append(('uniform', n_t, np.ones(n_t) / n_t))

    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 5 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:5]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Main optimization on top candidates
    # Allocate time based on number of candidates
    top_inits = init_scores[:4]
    n_candidates = len(top_inits)
    time_per_candidate = max(100.0, (deadline - time.monotonic() - 300) / n_candidates)

    for ci, (sc0, name, n_t, a0) in enumerate(top_inits):
        if time.monotonic() > deadline - 300:
            break

        # Skip obviously worse candidates for non-primary lengths
        if sc0 >= best_score * 1.1 and name != 'parent' and n_t != 768:
            continue

        # Main gradient optimization
        opt_deadline = min(deadline - 250, time.monotonic() + time_per_candidate * 0.6)
        print(f"Phase 2[{ci}]: Optimize {name} (n={n_t}), start={sc0:.8f}, budget={opt_deadline - time.monotonic():.0f}s", flush=True)
        a_opt = _main_optimize(a0, opt_deadline, max_iter=80000, lr=0.06)
        sc_opt = _score(a_opt)
        print(f"  Main opt: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < deadline - 150:
            lp_dl = min(deadline - 120, time.monotonic() + time_per_candidate * 0.25)
            print(f"  LP refine ({lp_dl - time.monotonic():.0f}s budget)", flush=True)
            a_lp = _lp_refine_fast(a_opt, lp_dl, max_iter=3000, n_top=5, inflate=1.0004, blend=0.15)
            sc_lp = _score(a_lp)
            print(f"  LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Exact top-k polish
        if time.monotonic() < deadline - 100:
            ep_dl = min(deadline - 80, time.monotonic() + 60.0)
            a_ep = _exact_topk_polish(a_opt, ep_dl, max_iter=10000, k=1, lr_init=0.01)
            sc_ep = _score(a_ep)
            print(f"  Topk polish: {sc_ep:.10f}", flush=True)
            if sc_ep < best_score:
                best_score = sc_ep
                best_seq = a_ep.copy()
            a_opt = a_ep

    # Phase 3: Final polish on overall best with remaining time
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 15
        print(f"Phase 3: Final polish, budget={remaining:.0f}s, current={best_score:.10f}", flush=True)

        # LP refinement
        if remaining > 60:
            lp_dl = min(deadline - 15, time.monotonic() + remaining * 0.4)
            a_lp = _lp_refine_fast(best_seq, lp_dl, max_iter=5000, n_top=3, inflate=1.0003, blend=0.1)
            sc_lp = _score(a_lp)
            print(f"  Final LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()

        # Exact polish
        if time.monotonic() < deadline - 25:
            ep_dl = min(deadline - 15, time.monotonic() + 60.0)
            a_ep = _exact_topk_polish(best_seq, ep_dl, max_iter=5000, k=1, lr_init=0.005)
            sc_ep = _score(a_ep)
            print(f"  Final topk: {sc_ep:.10f}", flush=True)
            if sc_ep < best_score:
                best_score = sc_ep
                best_seq = a_ep.copy()

        # One more LP pass if time allows
        if time.monotonic() < deadline - 10:
            lp_dl2 = deadline - 5
            if lp_dl2 - time.monotonic() > 15:
                a_last = _lp_refine_fast(best_seq, lp_dl2, max_iter=2000, n_top=3, inflate=1.0002, blend=0.05)
                sc_last = _score(a_last)
                if sc_last < best_score:
                    best_score = sc_last
                    best_seq = a_last.copy()
                print(f"  Last LP: {sc_last:.10f}", flush=True)

    # Phase 4: Try a few structured restarts from the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"Phase 4: Structured restarts", flush=True)
        for ri in range(3):
            if time.monotonic() > deadline - 20:
                break
            pert = _structured_perturb(best_seq, rng, strength=0.2 * (ri + 1))
            z_best = np.log(np.maximum(best_seq, 1e-15))
            z_best -= np.mean(z_best)
            z_pert = z_best + 0.5 * pert
            a_pert = _logit_project(z_pert)
            sc_pert = _score(a_pert)
            if sc_pert < best_score * 1.01:
                sub_dl = min(deadline - 20, time.monotonic() + 40.0)
                a_sub = _main_optimize(a_pert, sub_dl, max_iter=10000, lr=0.04)
                sc_sub = _score(a_sub)
                if sc_sub < best_score:
                    best_score = sc_sub
                    best_seq = a_sub.copy()
                    print(f"  Restart {ri}: improved to {sc_sub:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(768, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
