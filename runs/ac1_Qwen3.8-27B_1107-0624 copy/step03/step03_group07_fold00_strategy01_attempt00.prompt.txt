        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_{m=0}^{n-1} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def logit_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.08,
                        beta_decay=0.92, beta_min=1e-10, restart_period=1500):
    """
    Optimize in logit space using Adam with soft-max approximation of the max.
    Objective: minimize 2n * max(conv(a,a)) subject to a on simplex.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())

    # Initialize logits
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    iters = 0
    while time.monotonic() < deadline:
        # Soft max and weights
        _, w = soft_max_weights(c, beta)

        # Gradient: d(soft_max(conv(a,a)))/da[i] = 2*h[i]
        h = compute_h(w, a0)
        grad_a = 2.0 * h  # dL_soft/da

        # In logit space: dz = a * (grad_a - dot(a, grad_a))
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Stabilize
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # New a via softmax
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        iters += 1

        # Adapt beta (sharpen over time)
        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Adapt learning rate
        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        # Random restart

            noise_scale = 0.15
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def logit_lambdad_optimize(a0, deadline, rng, lr=0.02, beta_factor=0.05,
                           beta_decay=0.90, beta_min=1e-10, restart_period=1000):
    """
    Logit-space optimization with LAMBDA-style adaptive step (no first moment).
    Simpler and sometimes more stable for this problem.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = z - z.mean()

        beta = 1e-6

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # RMSprop-style (no momentum)
        rms = np.sqrt(np.mean(dz * dz)) + 1e-10
        step = current_lr * dz / rms
        step = np.clip(step, -0.1, 0.1)
        z = z + step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 60 == 0:
            beta = max(beta_min, beta * beta_decay)
            current_lr = max(0.003, current_lr * 0.99)

        if no_improve > restart_period:
            z = best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.06 * max_c)

def peak_targeting_refine(a0, deadline, rng, n_peaks=5, lr=0.03):
    """
    Refine by specifically targeting the top peaks in the convolution.
    Creates correction vectors that push down the identified peaks.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()
    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = float(2.0 * n * c.max())

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        # Identify top peaks
        peak_indices = np.argsort(c)[-n_peaks:]

        # Create targeted weights: uniform over top peaks
        w = np.zeros_like(c)
        for idx in peak_indices:
            w[idx] = 1.0 / n_peaks

        # Gradient
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score:
            best_score = score
            best_a = a0.copy()

    return best_a, best_score


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_linear_profile(n, eta))
    # Cosine
    for k in [1, 2, 3]:
        profiles.append(make_cosine_profile(n, k, 0.4))
    # Smooth bumps
    for width in [0.3, 0.5, 0.7]:
        for offset in [0.2, 0.4, 0.6]:
            profiles.append(make_smooth_bump(n, width, offset))
    # Power ramps
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(make_ramp_profile(n, p))
    # Random smooth (low-frequency)
    for _ in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 4)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, c_val in enumerate(coeffs, 1):
            a = a + c_val * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles
