"""
LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal update directions (following the
state-of-the-art max-min approach), combined with gradient-based polishing
and multi-scale search.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_ramp_profile(n, p=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def lp_update(f, M, max_iters=50):
    """
    Solve the LP: maximize sum(g) s.t. f*g[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized) or None if infeasible.
    """
    n = len(f)
    # f*g has length 2n-1
    # constraint A @ g <= M * ones(2n-1)
    # A[k, i] = f[k - i] if 0 <= k - i < n, else 0
    # i.e., A is a Toeplitz-like matrix
    
    # Build A matrix (2n-1) x n
    # A[k, i] = f[max(0, k-i)] if k-i >= 0 and k-i < n... 
    # Actually: (f*g)[k] = sum_i f[i] * g[k-i] where 0<=i<n and 0<=k-i<n
    # So (f*g)[k] = sum_i f[i] * g[k-i]
    # Let j = k-i, so i = k-j: sum_j f[k-j] * g[j] where 0<=j<n and 0<=k-j<n
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency with large n, use sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
            return None
    except Exception:
        pass
    return None

def lp_iteration(f, t_blend=0.3, max_lp_calls=100):
    """
    Iterate: compute M = ||f*f||_inf, solve LP for g, blend.
    """
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(max_lp_calls):
        c = full_convolve_fft(f)
        M = float(c.max())
        
        g = lp_update(f, M)
        if g is None:
            break
        
        # Blend
        f_new = (1 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score

def logit_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.08,
                        beta_decay=0.92, beta_min=1e-10, restart_period=1500):
    """Logit-space Adam optimization with soft-max."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_period:
            noise_scale = 0.15
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score

    mval = float(c.max())
    if beta < 1e-14:

    rows, cols = sparse_indices
    # Values: f[k-j] for each (k,j) pair
    # We need to compute the values
    vals = np.zeros(len(rows))
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        i = k - j

            s = g.sum()

def subgradient_optimize(a0, deadline, rng, lr=0.01, decay=0.999):
    """Exact subgradient descent on max(conv(a,a))."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15) / np.maximum(a0).sum()
    
    best_a = a0.copy()
    best_score = compute_score(a0)
    current_lr = lr
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        k_star = int(np.argmax(c))
        # Subgradient: 2 * a[k_star - i] for valid i
        # i.e., the gradient of conv(a,a)[k_star] w.r.t. a[i] is 2*a[k_star-i]
        # We want to minimize, so move in direction -subgrad
        # Projected gradient: a_new[i] = a[i] - lr * 2*a[k_star-i]
        # But we need to maintain sum(a)=1 and a>=0
        
        # In logit space: dz[i] = -a[i] * subgrad[i] + a[i] * mean(a*subgrad)
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = 2.0 * a0[j]
        
        # Projected gradient in logit space
        z = np.log(a0)
        z = z - z.mean()
        dot = float(np.dot(a0, subgrad))
        dz = a0 * (subgrad - dot)
        z = z - current_lr * dz
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        a0 = a0 / a0.sum()
        
        score = compute_score(a0)
        if score < best_score:
            best_score = score
            best_a = a0.copy()
        
        current_lr *= decay
    
    return best_a, best_score

    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_ramp_profile(n, p=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_bump(n, width=0.6, offset=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, c=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def precompute_lp_structure(n):
    """Precompute sparse structure for the LP constraint matrix."""
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        nnz += (j_max - j_min + 1)
    
    indptr = np.zeros(2 * n, dtype=np.int32)
    indices = np.zeros(nnz, dtype=np.int32)
    col_map = np.zeros(nnz, dtype=np.int32)
    
    pos = 0
    for k in range(2 * n - 1):
        indptr[k] = pos
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            indices[pos] = j
            col_map[pos] = k - j  # index into f
            pos += 1
    indptr[2 * n - 1] = pos
    
    return indptr, indices, col_map


def lp_solve(f, M, precomp):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    indptr, indices, col_map = precomp
    n = len(f)
    
    data = f[col_map].astype(np.float64)
    A = csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(f0, deadline, t_blend=0.5, max_iters=5000):
    """
    Iterate LP updates until deadline.
    f0: initial sequence (normalized)
    Returns best sequence and score found.
    """
    n = len(f0)
    f = f0.copy()
    f = f / f.sum()
    
    precomp = precompute_lp_structure(n)
    
    best_f = f.copy()
    c = full_convolve_fft(f)
    best_score = float(2.0 * n * c.max())
    
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        c = full_convolve_fft(f)
        M = float(c.max())
        
        g = lp_solve(f, M, precomp)
        if g is None:
            break
        
        # Adaptive blending: start aggressive, reduce as we converge
        score_before = float(2.0 * n * M)
        
        f_new = (1.0 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        c_new = full_convolve_fft(f_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        
        f = f_new
        it += 1
        
        # Adapt blend factor: reduce as we converge
        if it % 100 == 0:
            t_blend = max(0.1, t_blend * 0.95)
    
    return best_f, best_score


def logit_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.08,
                        beta_decay=0.92, beta_min=1e-10, restart_period=1500):
    """Logit-space Adam with soft-max gradient."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_period:
            noise_scale = 0.15
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def peak_targeting_refine(a0, deadline, rng, n_peaks=5, lr=0.03):
    """Refine by targeting top peaks in convolution."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15) / a0.sum()
    z = np.log(a0)
    z = z - z.mean()
    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = float(2.0 * n * c.max())
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        peak_indices = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        for idx in peak_indices:
            w[idx] = 1.0 / n_peaks
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        if score < best_score:
            best_score = score
            best_a = a0.copy()

    return best_a, best_score


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(make_linear_profile(n, eta))
    for k in [1, 2, 3]:
        profiles.append(make_cosine_profile(n, k, 0.4))
    for width in [0.3, 0.5]:
        for offset in [0.2, 0.5]:
            profiles.append(make_smooth_bump(n, width, offset))
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(make_ramp_profile(n, p))
    profiles.append(make_quadratic_profile(n, 2.0))
    for _ in range(4):
        coeffs = rng.uniform(-0.4, 0.4, 4)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, c_val in enumerate(coeffs, 1):
            a = a + c_val * np.cos(2.0 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles

    best_seq = None
    best_score = np.inf

    def update(p, sc):

        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()

    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting LP-driven search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick exploration at n=256 (100s) =====
    n1 = 256

            if sc_s < best_score + 0.005:
                dl_s = min(deadline, time.monotonic() + 20.0)
                r_s, _ = logit_adam_optimize(p_small, dl_s, rng, lr=0.02,
                                              beta_factor=0.03, beta_decay=0.93)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Smaller n={n_try}: {sc_s2:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    # ===== Final validation and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 2.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
