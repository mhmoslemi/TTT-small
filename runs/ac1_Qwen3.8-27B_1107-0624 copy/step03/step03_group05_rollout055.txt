def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_max_and_grad(a):

    # If k-i == i (k = 2i): d c[k]/d a[i] = 2*a[i]

def _top_k_peaks_subgradient(a, top_k=3):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    c = _fft_convolve(a)
    # Get top-k peak positions
    sorted_idx = np.argsort(c)[::-1][:top_k]
    # Weight by peak value (normalized)
    peak_vals = c[sorted_idx]
    weights = peak_vals / peak_vals.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, sorted_idx):
        k_minus_i = int(k) - np.arange(n)
        mask = (k_minus_i >= 0) & (k_minus_i < n)
        grad[mask] += w * 2.0 * a[k_minus_i[mask]]
    
    return grad, float(c.max()), c


def _project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _nesterov_subgradient_polish(a, deadline, top_k=3, max_iter=50000):
    """Nesterov-accelerated subgradient method targeting the top-k peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    
    # Nesterov: y = a + momentum * v, then gradient at y
    momentum = 0.7
    v = np.zeros(n)
    a_curr = a.copy()
    
    # Adaptive step size
    lr = 0.1
    lr_min = 0.001
    lr_decay = 0.995
    
    no_imp = 0
    best_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step: evaluate gradient at y = a + momentum * v
        y = a_curr + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a_curr.copy()
        else:
            y = y / s_y
        
        # Compute subgradient at y
        grad, M_y, c_y = _top_k_peaks_subgradient(y, top_k=top_k)
        
        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            continue
        
        # Update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a_curr + v)
        a_curr = a_new
        
        # Evaluate
        c = _fft_convolve(a_curr)
        M = float(c.max())
        score = float(2.0 * n * M)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_curr.copy()
            no_imp = 0
            best_lr = lr
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(lr_min, lr * lr_decay)
            if no_imp > max(2000, 5 * n):
                # Restart
                a_curr = best_a.copy()
                v = np.zeros(n)
                lr = best_lr
                no_imp = 0
        
        # Periodically increase momentum (Nesterov schedule)
        if it % 500 == 0:
            momentum = min(0.9, 0.6 + 0.02 * (it // 500))
    
    return best_a, best_score


def _peak_redistribution(a, deadline, max_iter=20000):
    """
    Heuristic: find the peak, identify contributing positions,
    and move mass from them to non-contributing positions.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    rng = np.random.default_rng(42)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        
        # Find contributing positions: j where a[j]*a[k_star-j] > 0
        j_range = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        contributions = a[j_range] * a[k_star - j_range]
        
        # Find the top contributing positions
        if len(contributions) == 0:
            break
        top_contrib_idx = j_range[np.argsort(contributions)[::-1][:5]]
        top_contrib_vals = contributions[np.argsort(contributions)[::-1][:5]]
        
        # Find "cold" positions: positions m where a[k_star-m] is small (or 0)
        # These are positions that won't increase c[k_star] much
        k_minus_all = k_star - np.arange(n)
        valid = (k_minus_all >= 0) & (k_minus_all < n)
        # a[k_star - m] for valid m
        contrib_if_added = np.zeros(n)
        contrib_if_added[valid] = a[k_minus_all[valid]]
        
        # Cold positions: low contrib_if_added AND not in top_contrib
        contrib_if_added[top_contrib_idx] = np.inf  # Exclude hot positions
        cold_positions = np.argsort(contrib_if_added)[:10]
        
        # Try moving mass from each contributing position to cold positions
        improved = False
        for j in top_contrib_idx:
            partner = k_star - j
            if partner < 0 or partner >= n:
                continue
            # Move mass from max(a[j], a[partner]) to a cold position
            if a[j] > a[partner]:
                src = j
                amount = 0.1 * a[j]
            else:
                src = partner
                amount = 0.1 * a[partner]
            
            if amount < 1e-12:
                continue
            
            for m in cold_positions[:3]:
                if m == src or (0 <= partner < n and m == partner):
                    continue
                a_try = a.copy()
                a_try[src] -= amount
                a_try[m] += amount
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                sc = float(2.0 * n * _fft_convolve(a_try).max())
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        
        if not improved:
            # Try random perturbation
            if rng.random() < 0.1:
                pert = 0.05 * rng.normal(size=n)
                a_try = a + pert
                a_try = _project_simplex(a_try)
                sc = float(2.0 * n * _fft_convolve(a_try).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                # Accept with some probability (simulated annealing)
                elif rng.random() < 0.01:
                    a = a_try.copy()
            else:
                # Try a different approach: uniform small perturbation
                pert = 0.02 * rng.normal(size=n)
                a_try = _project_simplex(a + pert)
                sc = float(2.0 * n * _fft_convolve(a_try).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
    
    return best_a, best_score


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

        a_new = (1.0 - blend_cur) * a + blend_cur * g

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _adam_smooth_max_polish(a, deadline, max_iter=30000, lr=0.03):
    """Adam optimizer on log-parameterization with smooth-max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    beta = max(1e-10, 0.02 * float(_fft_convolve(a).max()))
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.97)
        
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.8)
        
        if no_imp > max(2000, 3 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.8 * best_z + 0.2 * rng_r.normal(size=n) * 0.5
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-10, 0.03 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score


def _parametric_init(n, family='poly2'):
    """Generate a parametric initialization."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0  # [-1, 1]
    
    if family == 'linear':
        a = 1.0 + 0.5 * u
    elif family == 'quad':
        a = 1.0 + 0.3 * u + 0.2 * u**2
    elif family == 'poly3':
        a = 1.0 + 0.4 * u + 0.1 * u**2 - 0.2 * u**3
    elif family == 'exp_ramp':
        a = np.exp(0.5 * u)
    elif family == 'bimodal':
        a = np.exp(-4*(u-0.5)**2) + 0.5*np.exp(-4*(u+0.5)**2) + 0.3
    elif family == 'cosine':
        a = 1.0 + 0.3*np.cos(np.pi*u)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _full_polish_pipeline(a, deadline, n):

    a = np.asarray(a, dtype=np.float64).copy()

    # Phase 1: LP spreading (30% of remaining time)

    if sc1 < best_score:

    # Phase 2: Nesterov subgradient (25%)

    # Phase 3: Peak redistribution (15%)

    if sc3 < best_score:

    # Phase 4: Adam smooth-max (20%)
    t4 = time.monotonic() + remaining * 0.2
