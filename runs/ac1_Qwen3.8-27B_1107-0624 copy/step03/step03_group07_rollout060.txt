```python
import numpy as np
import time

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c  # Don't clip to 0, as small negative values are OK numerically

def full_convolve_fft_clipped(a):
    c = full_convolve_fft(a)
    return np.maximum(c, 0.0)

def compute_h(w, a):
    """h[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1"""
    n = len(a)
    a_rev = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n-1:2*n-1]

def convolve_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]
```

```python
def lp_refine_step(a, M):
    """
    One step of LP-based refinement.
    Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0
    Return the optimal b (rescaled to have sum = sum(a)).
    """
    import cvxpy as cp
    
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Constraint: conv(a, b) <= M
    # conv(a,b) is a linear function of b
    # We can use cp.multiply or explicit formulation
    # For efficiency, let's use the fact that conv(a,b)[k] = sum_i a[i]*b[k-i]
    # This is a Toeplitz matrix times b
    
    # Build the Toeplitz matrix (sparse for efficiency)
    # Actually, for n=1024, a dense 2047x1024 matrix is about 16MB, manageable
    # But let's use a more efficient approach
    
    # The constraint conv(a,b) <= M can be written as T @ b <= M where T is the
    # Toeplitz matrix with first column a and first row [a[0], 0, ..., 0]
    
    # Actually, let's just use numpy to compute the constraint matrix
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is the same as: T @ b = conv(a, b) (with appropriate indexing)
    
    # For CVXPY, we can use:
    # cp.multiply doesn't work directly for convolution
    # Let's use an explicit matrix
    
    # Alternative: use the FFT representation
    # But CVXPY doesn't support FFT natively
    
    # Let's build the sparse matrix
    # T is (2n-1) x n
    # T[k, j] = a[k-j] if 0 <= k-j <= n-1 else 0
    
    from scipy.sparse import lil_matrix, csr_matrix
    
    T = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            T[k, j] = a[k-j]
    T = csr_matrix(T)
    
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            prob.solve()
    
    if b_var.value is None:
        return a.copy()
    
    b_opt = np.maximum(b_var.value, 0)
    s = b_opt.sum()
    if s < 1e-10:
        return a.copy()
    
    # Rescale to match sum(a)
    target_sum = a.sum()
    b_opt = b_opt * (target_sum / s)
    
    return b_opt
```

```python
def mw_lp_approx(a, M, b_init, n_iters=100, eta=0.1):
    """
    Multiplicative weights approximation to:
    max sum(b) s.t. conv(a,b) <= M, b >= 0
    """
    n = len(a)
    b = b_init.copy()
    b = np.maximum(b, 1e-15)
    b = b / b.sum()
    
    # Maintain weights on constraints
    w = np.ones(2*n-1) / (2*n-1)
    
    for it in range(n_iters):
        # Compute conv(a, b)
        c = convolve_ab(a, b)
        c = np.maximum(c, 0)
        
        # Update weights on violated/near-violated constraints
        violation = np.maximum(c - M, 0) / (M + 1e-10)
        w = w * np.exp(eta * (c / (M + 1e-10) - 1))
        w = w / w.sum()
        
        # Update b: reduce where it contributes to tight constraints
        # grad[i] = sum_k w[k] * a[|k-i|] (approximately)
        # More precisely: d(conv(a,b)[k])/db[i] = a[|k-i|] if |k-i| < n
        # So the "penalty" on b[i] is sum_k w[k] * a[|k-i|] (with appropriate indexing)
        
        # Let's compute: penalty[i] = sum_k w[k] * (d conv(a,b)[k] / db[i])
        # d conv(a,b)[k] / db[i] = a[k-i] if 0 <= k-i <= n-1, else 0
        # penalty[i] = sum_k w[k] * a[k-i] (for valid k-i)
        # = sum_{j} a[j] * w[i+j] (where j = k-i, k = i+j)
        # This is exactly compute_h(w, a)[i] (by definition of compute_h)!
        
        penalty = compute_h(w, a)  # penalty[i] = sum_j a[j] * w[i+j]
        
        # Multiplicative update: b[i] *= exp(-eta * penalty[i] / max(penalty))
        pen_norm = penalty / (penalty.max() + 1e-10)
        b = b * np.exp(-eta * pen_norm)
        b = np.maximum(b, 1e-15)
        b = b / b.sum()
    
    return b
```

This is essentially a two-player game:
- Player 1 (row player) chooses a distribution w over the convolution positions
- Player 2 (column player) chooses b to minimize the expected violation
- The value of the game approximates the LP optimum

This converges at rate O(1/sqrt(T)) to the LP optimum, which is good enough.

```python
"""
Combined optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses: (1) Nesterov-accelerated soft-max gradient, (2) exact subgradient with top-k,
(3) multiplicative-weights LP approximation, and (4) alternating dual-ascent.
Multi-resolution search with diverse restarts.

import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def full_convolve_fft_pos(a):
    """Full linear convolution, clipped to non-negative."""
    c = full_convolve_fft(a)
    return np.maximum(c, 0.0)


def compute_h(w, a):
    """h[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n - 1:2 * n - 1]


def convolve_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def interpolate_profile(p0, n_new):
    """Interpolate a profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_simple_start(n, rng):

    eta = rng.uniform(1.0, 2.5)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)

def mw_lp_refine(a0, deadline, rng, mw_iters=300, eta=0.3, t_blend=0.15):
    """
    Multiplicative-weights LP approximation for refinement.
    Solves approximately: max sum(b) s.t. conv(a,b) <= M, b >= 0
    Then does convex combination update.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = full_convolve_fft_pos(a)
    M = float(c.max())

    best_a = a.copy()
    best_score = float(2.0 * n * M)

    b = a.copy()
    p = np.ones(2 * n - 1) / (2 * n - 1)

    outer_it = 0
    stagnation = 0
    current_t = t_blend
    current_eta = eta

    while time.monotonic() < deadline:
        # Inner MW iterations
        for mw_it in range(mw_iters):
            if time.monotonic() > deadline:
                break

            # Cost for each b_i: h[i] = sum_j a[j] * p[i+j]
            h = compute_h(p, a)
            h = np.maximum(h, 1e-15)
            h_max = h.max()
            if h_max < 1e-10:
                b = np.ones(n) / n
            else:
                b = b * np.exp(-current_eta * h / h_max)
                b = np.maximum(b, 1e-15)
                b = b / b.sum()

            # Violation update for p
            c_ab = convolve_ab(a, b)
            c_ab = np.maximum(c_ab, 0.0)
            violation = np.maximum(c_ab - M, 0.0)
            v_max = violation.max()
            if v_max > 1e-12:
                p = p * np.exp(current_eta * violation / v_max)
                p = p / p.sum()

        # Rescale b to match sum(a)
        s_b = b.sum()
        b_scaled = b * (a.sum() / s_b) if s_b > 1e-10 else a.copy()

        # Convex combination
        a_new = (1.0 - current_t) * a + current_t * b_scaled
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = full_convolve_fft_pos(a_new)
        M_new = float(c_new.max())
        score = float(2.0 * n * M_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            c = c_new.copy()
            M = M_new
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 30:
                current_t = max(0.02, current_t * 0.8)
                current_eta = max(0.05, current_eta * 0.9)
                # Reset MW state
                b = a.copy()
                p = np.ones(2 * n - 1) / (2 * n - 1)
                stagnation = 0

        outer_it += 1
        if outer_it % 20 == 0:
            # Periodic reset of p to uniform to avoid getting stuck
            p = 0.5 * p + 0.5 * np.ones(2 * n - 1) / (2 * n - 1)

    return best_a, best_score


def dual_ascent_refine(a0, deadline, rng, n_inner=50, k=5):
    """
    Alternating dual-ascent: alternate between optimizing a for fixed dual weights
    and updating dual weights to target current peaks.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = full_convolve_fft_pos(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    # Dual variable: distribution over convolution positions
    p = np.ones(2 * n - 1) / (2 * n - 1)

    z = np.log(a)
    z = z - z.mean()

    # Adam state for a optimization
    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = 0.03

    outer_it = 0
    while time.monotonic() < deadline:
        # Update dual weights: target top-k peaks of current c
        c = full_convolve_fft_pos(a)
        peak_idx = np.argsort(c)[-k:]
        p_target = np.zeros(2 * n - 1)
        for idx in peak_idx:
            p_target[idx] = 1.0 / k
        # Smooth update
        p = 0.7 * p + 0.3 * p_target
        p = p / p.sum()

        # Inner: optimize a to minimize p^T conv(a,a) = a^T Q_p a
        for inner_it in range(n_inner):
            if time.monotonic() > deadline:
                break

            h = compute_h(p, a)
            grad_a = 2.0 * h
            dot = float(np.dot(a, grad_a))
            dz = a * (grad_a - dot)

            t_adam += 1
            m_adam = b1 * m_adam + (1 - b1) * dz
            v_adam = b2 * v_adam + (1 - b2) * (dz * dz)
            mh = m_adam / (1 - b1 ** t_adam)
            vh = v_adam / (1 - b2 ** t_adam)
            z = z - lr * mh / (np.sqrt(vh) + eps_adam)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            a = np.exp(z - z.max())
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                z = np.log(a)
                z = z - z.mean()
                break
            a = a / s

        c = full_convolve_fft_pos(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        outer_it += 1
        if outer_it % 20 == 0:
            lr = max(0.005, lr * 0.97)

    return best_a, best_score


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cosine
    for k in [1, 2, 3]:
        for amp in [0.3, 0.5]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    # Gaussian bumps
    for width in [0.2, 0.4, 0.6]:
        for offset in [0.2, 0.5, 0.8]:
            a = np.exp(-((t - offset) / width) ** 2) + 0.05
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    # Power ramps
    for p in [0.5, 1.0, 2.0]:
        a = (1.0 - t) ** p + 0.1
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Random smooth
    for _ in range(4):
        coeffs = rng.uniform(-0.3, 0.3, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method search combining soft-max gradient, exact subgradient,
    MW-LP refinement, and dual-ascent. Multi-resolution with diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc = compute_score(init_seq)
                update(init_seq, sc)
                print(f"Initial: n={init_seq.size}, score={sc:.10f}")
    except Exception:
        pass

    # ===== Phase 1: Optimize at n=2048 from initial =====
    if init_seq is not None:
        n1 = 2048
        p1 = interpolate_profile(init_seq, n1) if init_seq.size != n1 else init_seq.copy()
        sc1 = compute_score(p1)
        update(p1, sc1)
        print(f"\nPhase 1: n={n1}, start score={sc1:.10f}")

        # Method A: Soft-max gradient with Nesterov
        dl = min(deadline, t_start + 400.0)
        r, sc_r = softmax_grad_nesterov(p1, dl, rng, lr=0.025,
                                        beta_factor=0.05, beta_decay=0.92)
        sc_r = compute_score(r)
        update(r, sc_r)
        print(f"  Soft-max Nesterov: {sc_r:.10f}")

        # Method B: Exact subgradient
        if time.monotonic() < deadline - 200:
            dl = min(deadline, time.monotonic() + 300.0)
            r, sc_r = exact_subgrad_nesterov(r, dl, rng, lr=0.015, k=3)
            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  Exact subgrad: {sc_r:.10f}")

        # Method C: MW-LP refinement
        if time.monotonic() < deadline - 200:
            dl = min(deadline, time.monotonic() + 250.0)
            r, sc_r = mw_lp_refine(r, dl, rng, mw_iters=200, eta=0.25, t_blend=0.12)
            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  MW-LP: {sc_r:.10f}")

        # Method D: Dual ascent
        if time.monotonic() < deadline - 150:
            dl = min(deadline, time.monotonic() + 200.0)
            r, sc_r = dual_ascent_refine(r, dl, rng, n_inner=30, k=5)
            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  Dual ascent: {sc_r:.10f}")

    # ===== Phase 2: Try n=1024 =====
    n2 = 1024
    if time.monotonic() < deadline - 200:
        if best_seq is not None:
            p2 = interpolate_profile(best_seq, n2)
        elif init_seq is not None:
            p2 = interpolate_profile(init_seq, n2)
        else:
            p2 = make_simple_start(n2, rng)

        sc2 = compute_score(p2)
        update(p2, sc2)
        print(f"\nPhase 2: n={n2}, start score={sc2:.10f}")

        dl = min(deadline, time.monotonic() + 300.0)
        r, sc_r = softmax_grad_nesterov(p2, dl, rng, lr=0.03,
                                        beta_factor=0.06, beta_decay=0.91)
        sc_r = compute_score(r)
        update(r, sc_r)
        print(f"  Soft-max Nesterov: {sc_r:.10f}")

        if time.monotonic() < deadline - 150:
            dl = min(deadline, time.monotonic() + 150.0)
            r2, sc_r2 = mw_lp_refine(r, dl, rng, mw_iters=300, eta=0.3, t_blend=0.15)
            sc_r2 = compute_score(r2)
            update(r2, sc_r2)
            print(f"  MW-LP: {sc_r2:.10f}")
            if sc_r2 < sc_r:
                r = r2

        # Small perturbations and re-optimize
        if time.monotonic() < deadline - 100:
            for _ in range(3):
                p_pert = r + 0.002 * rng.normal(size=n2)
                p_pert = np.maximum(p_pert, 0)
                p_pert = p_pert / p_pert.sum()
                dl = min(deadline, time.monotonic() + 40.0)
                r_p, sc_p = softmax_grad_nesterov(p_pert, dl, rng, lr=0.01,
                                                  beta_factor=0.02, beta_decay=0.93)
                sc_p = compute_score(r_p)
                update(r_p, sc_p)
                if sc_p < best_score:
                    print(f"  Perturbation improved: {sc_p:.10f}")

    # ===== Phase 3: Try n=512 =====
    n3 = 512
    if time.monotonic() < deadline - 150:
        if best_seq is not None:
            p3 = interpolate_profile(best_seq, n3)
        else:
            p3 = make_simple_start(n3, rng)

        sc3 = compute_score(p3)
        update(p3, sc3)
        print(f"\nPhase 3: n={n3}, start score={sc3:.10f}")

        dl = min(deadline, time.monotonic() + 200.0)
        r, sc_r = softmax_grad_nesterov(p3, dl, rng, lr=0.035,
                                        beta_factor=0.07, beta_decay=0.90)
        sc_r = compute_score(r)
        update(r, sc_r)
        print(f"  Soft-max Nesterov: {sc_r:.10f}")

        if time.monotonic() < deadline - 100:
            dl = min(deadline, time.monotonic() + 100.0)
            r2, sc_r2 = mw_lp_refine(r, dl, rng, mw_iters=400, eta=0.35, t_blend=0.18)
            sc_r2 = compute_score(r2)
            update(r2, sc_r2)
            print(f"  MW-LP: {sc_r2:.10f}")

    # ===== Phase 4: Final polish at best n =====
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_polish = best_seq.size
        print(f"\nPhase 4: Polish at n={n_polish}, current best={best_score:.10f}")

        # Multiple polish rounds
        for rnd in range(4):
            if time.monotonic() > deadline - 10:
                break
            p0 = best_seq + 0.0003 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 0)
            p0 = p0 / p0.sum()

            dl = min(deadline, time.monotonic() + 80.0)
            if rnd % 2 == 0:
                r, sc_r = softmax_grad_nesterov(p0, dl, rng, lr=0.008,
                                                beta_factor=0.015, beta_decay=0.94)
            else:
                r, sc_r = exact_subgrad_nesterov(p0, dl, rng, lr=0.005, k=2)
            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  Polish {rnd}: {sc_r:.10f}")

        # Final MW-LP polish
        if time.monotonic() < deadline - 20:
            dl = min(deadline, time.monotonic() + 60.0)
            r, sc_r = mw_lp_refine(best_seq, dl, rng, mw_iters=500,
                                   eta=0.1, t_blend=0.05)
            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  Final MW-LP: {sc_r:.10f}")

    # ===== Phase 5: Try diverse profiles at best n if time allows =====
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_try = best_seq.size
        profiles = diverse_profiles(n_try, rng)
        for i, prof in enumerate(profiles[:4]):
            if time.monotonic() > deadline - 30:
                break
            sc_p = compute_score(prof)
            if sc_p < best_score + 0.05:
                dl = min(deadline, time.monotonic() + 30.0)
                r, sc_r = softmax_grad_nesterov(prof, dl, rng, lr=0.02,
                                                beta_factor=0.04)
                sc_r = compute_score(r)
                update(r, sc_r)
                if sc_r < best_score:
                    print(f"  Diverse profile {i}: {sc_r:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        n = 1024
        t = np.linspace(0, 1, n)
        final_seq = 1.0 + 1.5 * (0.5 - t)
        final_seq = np.maximum(final_seq, 0)
        final_seq = final_seq / final_seq.sum()

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * np.sqrt(2.0 * final_seq.size)

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
```
