def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n=a.size
    if n==0: return np.array([], dtype=np.float64)
    if n <= 128:
        c = np.convolve(a,a)
        c = np.maximum(c,0.0)
        return c
    N=_next_pow2(2*n-1)
    fa=np.fft.rfft(a,N)
    c=np.fft.irfft(fa*fa,N)[:2*n-1]
    c=np.maximum(c,0.0)
    return c

"""
Alternating bi-convex pressure balancing for minimizing

    2*n*max(conv(a,a)) / sum(a)^2

over nonnegative sequences a.

The core idea is that near a local optimum there should exist nonnegative
dual weights w on the active convolution peaks such that the "pressure"
    h_i = sum_k w_k * a_{k-i}
is nearly constant on the support of a.  The optimizer alternates:

1. choose weights on the currently hottest convolution coefficients;
2. solve a small convex QP for those weights so that the pressure is flat;
3. solve another convex QP for the sequence, keeping the sequence in the
   simplex and close to the previous iterate;
4. accept only changes that improve (or very occasionally barely worsen)
   the actual score, then repeat with updated active peaks.

A final projected-gradient / multiplicative polish and a small LP-based
spreading stage are used on the best incumbent.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _next_pow2(n):
    n = int(n)
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-30:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(c, n):
    if c.size == 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    if cm < 0.0:
        cm = 0.0
    return float(2.0 * float(n) * cm)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        n = a.size
        if n == 0:
            return a
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    rho_idx = int(rho[-1])
    theta = (cssv[rho_idx] - 1.0) / float(rho_idx + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if not np.isfinite(s) or s < 1e-30:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_bell_profile(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-(((t - 0.5) / width) ** 2))
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_flat_spectrum_init(n, rng, dc=0.0):
    N = _next_pow2(max(2, n))
    X = np.zeros(N, dtype=np.complex128)
    X[0] = float(dc)
    half = N // 2
    if half > 1:
        phases = rng.uniform(0.0, 2.0 * np.pi, size=half - 1)
        X[1:half] = np.exp(1j * phases)
        X[N - 1:N - half:-1] = np.exp(-1j * phases)
    if N > 1:
        X[half] = 1.0 if rng.random() < 0.5 else -1.0
    y = np.fft.ifft(X)
    p = np.abs(y[:n])
    p = np.maximum(p, 1e-10)
    return _normalize(p)


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n_new == n0:
        return _normalize(p0)
    if n0 == 0:
        return np.ones(n_new, dtype=np.float64) / float(max(1, n_new))
    if n0 == 1:
        return np.ones(n_new, dtype=np.float64) / float(max(1, n_new))
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return _normalize(p)


def _mk_conv_defs(idx, n):
    defs = []
    for kk in idx:
        k = int(kk)
        if k < 0 or k >= 2 * n - 1:
            continue
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if lo > hi:
            continue
        js = k - lo
        je = k - hi
        defs.append((lo, hi, js, je))
    return defs


def _kv_vec(v, defs, W):
    n = len(v)
    out = np.zeros(n, dtype=np.float64)
    for tup, w in zip(defs, W):
        if w == 0.0:
            continue
        lo, hi, js, je = tup
        if je > 0:
            out[lo:hi + 1] += w * v[js:je - 1:-1]
        else:
            out[lo:hi + 1] += w * v[js::-1]
    return out


def _build_A(a, defs):
    n = len(a)
    r = len(defs)
    A = np.zeros((n, r), dtype=np.float64)
    for col, tup in enumerate(defs):
        lo, hi, js, je = tup
        if je > 0:
            A[lo:hi + 1, col] = a[js:je - 1:-1]
        else:
            A[lo:hi + 1, col] = a[js::-1]
    return A


def _weight_obj(A, W, s, ATA, W0, eta):
    n = A.shape[0]
    AtAW = ATA @ W
    val = float(W @ AtAW) - float((W @ s) ** 2) / float(n)
    val += float(eta) * float(np.dot(W - W0, W - W0))
    if val < 0.0:
        val = 0.0
    return val


def _optimize_weights(A, W0, max_iter=30, eta=0.1):
    r = A.shape[1]
    if r <= 1:
        return np.ones(max(1, r), dtype=np.float64)
    W0 = np.asarray(W0, dtype=np.float64).ravel()
    if W0.size != r:
        W0 = np.ones(r, dtype=np.float64)
    W0 = _normalize(W0)
    W = _normalize(W0)
    s = A.sum(axis=0).astype(np.float64)
    ATA = (A.T @ A).astype(np.float64)
    f = _weight_obj(A, W, s, ATA, W0, eta)
    step = 1.0
    for _ in range(max_iter):
        Aw = A @ W
        mu = float(W @ s) / float(A.shape[0])
        grad = 2.0 * (ATA @ W - mu * s) + 2.0 * eta * (W - W0)
        grad -= float(np.mean(grad))
        if not np.all(np.isfinite(grad)):
            break
        if float(np.linalg.norm(grad)) < 1e-13:
            break
        st = step
        accepted = False
        for __ in range(6):
            W_try = _project_simplex(W - st * grad)
            f_try = _weight_obj(A, W_try, s, ATA, W0, eta)
            if f_try < f - 1e-20:
                W = W_try
                f = f_try
                accepted = True
                break
            st *= 0.5
        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            step = min(20.0, step * 1.1)
    return _normalize(W)


def _pressure_p_obj(p, defs, W, lam, anchor):
    h = _kv_vec(p, defs, W)
    mu = float(h.mean())
    d = h - mu
    val = float(np.dot(d, d)) + float(lam) * float(np.dot(p - anchor, p - anchor))
    if val < 0.0:
        val = 0.0
    return val


def _pressure_p_qp(a, defs, W, inner_iters, lam, deadline=None):
    n = len(a)
    if inner_iters <= 0 or len(defs) == 0:
        return a.copy()
    p = a.copy()
    anchor = a.copy()
    ones = np.ones(n, dtype=np.float64)
    K1 = _kv_vec(ones, defs, W)
    f = _pressure_p_obj(p, defs, W, lam, anchor)
    step = 0.5
    for _ in range(inner_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        h = _kv_vec(p, defs, W)
        mu = float(h.mean())
        kh = _kv_vec(h, defs, W)
        grad = 2.0 * (kh - mu * K1) + 2.0 * lam * (p - anchor)
        grad -= float(np.mean(grad))
        if not np.all(np.isfinite(grad)):
            break
        if float(np.linalg.norm(grad)) < 1e-14:
            break
        st = step
        accepted = False
        for __ in range(8):
            p_try = _project_simplex(p - st * grad)
            f_try = _pressure_p_obj(p_try, defs, W, lam, anchor)
            if f_try < f - 1e-20:
                p = p_try
                f = f_try
                accepted = True
                break
            st *= 0.5
        if not accepted:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            step = min(2.0, step * 1.05)
    return p


def _log_noise_restart(a_base, rng, scale=0.05):
    a_base = np.asarray(a_base, dtype=np.float64).ravel()
    n = len(a_base)
    if n == 0:
        return a_base.copy()
    base = np.maximum(a_base, 1e-12)
    z = np.log(base)
    z += scale * rng.standard_normal(n)
    z -= float(np.mean(z))
    e = np.exp(np.clip(z, -50.0, 50.0))
    return _normalize(e)


def _lp_polish(a, deadline, max_iters=6, inflate=1.001, blend=0.15):
    if linprog is None:
        return a
    a = _normalize(a)
    n = len(a)
    if n == 0 or n > 1200:
        return a
    if max_iters <= 0:
        return a
    best = a.copy()
    best_score = _score_fast(a)
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        p = a
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = p
        c = _fft_convolve(p)
        if c.size == 0:
            break
        M = float(c.max())
        M_lp = M * inflate
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M_lp, dtype=np.float64)
        bounds = [(0.0, 50.0)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            break
        if not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if not np.isfinite(gs) or gs <= (1.0 + 1e-7):
            if it == 0:
                inflate = min(1.005, inflate * 1.5)
                continue
            break
        g = g / gs
        improved = False
        for bnd in (blend, 0.5 * blend, 0.25 * blend, 0.1 * blend):
            p_try = _normalize((1.0 - bnd) * p + bnd * g)
            sc_try = _score_fast(p_try)
            if sc_try < best_score - 1e-14:
                best_score = sc_try
                best = p_try.copy()
                a = best.copy()
                improved = True
                break
        if not improved:
            inflate = min(1.005, inflate * 1.2)
            blend = max(0.02, blend * 0.8)
            if it > 3:
                break
        else:
            inflate = max(1.0005, inflate * 0.95)
        if it % 2 == 1 or it == max_iters - 1:
            print(f"  LP polish it={it} score={best_score:.12f}", flush=True)
    return _normalize(best)


def _refine_alt(a, deadline, rng, tag="", max_r=0):

    a = _normalize(a)
    n = len(a)
    if n == 0 or deadline <= t_enter:
        return a
    best = a.copy()
    best_score = _score_fast(a)
    Nc = 2 * n - 1

        inner_start = 40
    elif n <= 1536:

        inner_start = 25

    else:
        r = min(max_r, target_r)
    r = max(min_r, r)

    inner = inner_start
    lam = 0.2
    no_imp = 0
    outer = 0

    print(f"[{tag}] start n={n} r={r} score={best_score:.12f}", flush=True)

    while time.monotonic() < deadline - 2.0:
        c = _fft_convolve(a)
        cur_score = _score_norm(c, n)
        if not np.isfinite(cur_score):

                sc_try = _score_norm(c_try, n)
                if sc_try < cur_score - 1e-14:
                    a = p_try
                    cur_score = sc_try
                    direct_improved = True
                    if sc_try < best_score - 1e-15:

                        best = a.copy()
                    break

        explore_accepted = False
        if not direct_improved:

            hard_explore_limit = best_score * 1.03
            for b in (0.3, 0.15):

            no_imp = 0
            cur_cap = auto_max_r if max_r <= 0 else max_r
            r = max(min_r, min(cur_cap, int(r * 1.3) + 4))

                  flush=True)
        outer += 1

    return best


def _pair_score(pair):

def _select_candidates(pool, priority_lengths, max_count):
    selected = []
    used_ids = []
    if pool:
        selected.append(pool[0][1])
        used_ids.append(id(pool[0][1]))
    for L in priority_lengths:
        if len(selected) >= max_count:
            break
        for _sc, a in pool:
            if len(a) == L and id(a) not in used_ids:
                selected.append(a)
                used_ids.append(id(a))
                break
    for _sc, a in pool:
        if len(selected) >= max_count:
            break
        if id(a) not in used_ids:
            selected.append(a)
            used_ids.append(id(a))
    return selected

            a3 = _make_bell_profile(n, 0.4)
            pool.append((_score_fast(a3), a3))

    if not pool:
        a = _make_linear_profile(512, 1.5)
        pool.append((_score_fast(a), a))

    good_pool = []
    for sc, a in pool:
        if np.isfinite(sc):
            good_pool.append((sc, a))

    if not good_pool:
        a = _make_linear_profile(512, 1.5)
        good_pool.append((_score_fast(a), a))

    good_pool.sort(key=_pair_score)
    best_score = float(good_pool[0][0])
    best_seq = np.asarray(good_pool[0][1], dtype=np.float64).copy()

    print(f"Initial best: {best_score:.12f} n={best_seq.size}", flush=True)

    priority = [int(best_seq.size)]
    for L in [768, int(best_seq.size), 1024, 1536, 512]:
        if L not in priority:
            priority.append(L)

    selected = _select_candidates(good_pool, priority, 4)

    main_limit = hard_stop - 90.0
    if main_limit < t_start + 60.0:
        main_limit = hard_stop - 20.0

    total_work = max(30.0, main_limit - time.monotonic())
    fracs = [0.40, 0.20, 0.25, 0.15]

    for i, a in enumerate(selected):
        if time.monotonic() >= main_limit - 5.0:
            break
        frac = fracs[i] if i < len(fracs) else 0.1
        chunk = total_work * frac
        if chunk < 40.0:
            chunk = 40.0
        sub_deadline = min(main_limit - 5.0, time.monotonic() + chunk)
        tag = f"c{i}-n{len(a)}"
        a = _refine_alt(a, sub_deadline, rng, tag, max_r=0)
        sc = _score_fast(a)
        if sc < best_score:
            best_score = sc
            best_seq = np.asarray(a, dtype=np.float64).copy()
        print(f"After {tag}: best={best_score:.12f} n={len(best_seq)}", flush=True)

    if time.monotonic() < main_limit - 30.0:
        best_seq = _refine_alt(best_seq, main_limit, rng, tag="extra", max_r=30)
        sc = _score_fast(best_seq)
        if sc < best_score:
            best_score = sc

    if best_seq.size <= 1100 and hard_stop - time.monotonic() > 70.0:
        lp_deadline = min(hard_stop - 25.0, time.monotonic() + 60.0)
        max_lp = 6 if best_seq.size <= 800 else 3
        best_seq = _lp_polish(best_seq, lp_deadline, max_iters=max_lp)
        sc = _score_fast(best_seq)
        if sc < best_score:
            best_score = sc
        print(f"After LP: best={best_score:.12f}", flush=True)

    if hard_stop - time.monotonic() > 25.0:
        best_seq = _refine_alt(best_seq, hard_stop - 12.0, rng, tag="final", max_r=20)
        sc = _score_fast(best_seq)
        if sc < best_score:
            best_score = sc

    final_np = _normalize(best_seq)
    final_seq = [float(x) for x in final_np]

    try:
        ev_sc = float(evaluate_sequence(final_seq))
        if np.isfinite(ev_sc):
            print(f"FINAL local={best_score:.12f} eval={ev_sc:.12f} "
                  f"n={len(final_seq)} time={time.monotonic() - t_start:.1f}s", flush=True)
        else:
            print(f"FINAL local={best_score:.12f} eval=inf "
                  f"n={len(final_seq)} time={time.monotonic() - t_start:.1f}s", flush=True)
    except Exception as e:
        print(f"FINAL local={best_score:.12f} eval_error={e} "
              f"n={len(final_seq)} time={time.monotonic() - t_start:.1f}s", flush=True)

    return final_seq
