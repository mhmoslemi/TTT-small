def _project_simplex(v):
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):

    s = float(a.sum())

def _grad_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    N_pad = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N_pad)
    fw = np.fft.rfft(w, N_pad)
    cross = np.fft.irfft(fa * np.conj(fw), N_pad)[:n]
    
    grad = 2.0 * cross
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _subgradient_method(a0, deadline, max_iter=50000):
    """Subgradient method with Polyak step size."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    f_current = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * f_current)
    
    # Lower bound: 1/(2n-1) (from L2 bound)
    f_lower = 1.0 / (2 * n - 1)
    # Better lower bound: use the current best as pseudo-bound
    f_lb = best_score / (2.0 * n)
    
    lr_scale = 1.0
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mval = float(c.max())
        
        # Subgradient: use the active constraint (max convolution index)
        k_max = int(np.argmax(c))
        # grad_j = 2 * a[k_max - j] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_max - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Polyak step size
        f_gap = mval - f_lb
        if f_gap < 1e-10:
            break
        lr = lr_scale * f_gap / (gnorm * gnorm)
        lr = min(lr, 10.0 / n)  # Cap step size
        
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        a = a_new
        
        c = _fft_convolve(a)
        f_new = float(c.max())
        score = float(2.0 * n * f_new)
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            f_lb = f_new - 0.1 * (f_current - f_new)  # conservative update
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr_scale *= 0.8
        
        if no_imp > 5000:
            a = best_a.copy()
            lr_scale = 1.0
            no_imp = 0
            f_lb = best_score / (2.0 * n) * 0.99
    
    return best_a, best_score

sum(a) = 1

Each constraint a^T T_k a <= t is a non-convex constraint (it's the interior of an ellipsoid, which is non-convex). Wait, no: a^T T_k a <= t where T_k is PSD. The set {a : a^T T_k a <= t} is an ellipsoid, which is convex. So we have:

"""
Hybrid accelerated optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines adaptive FISTA (smooth-max), LP direction finding with warm-starting,
and away-step LP for escaping local minima. Uses FFT for fast convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 200:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _grad_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -40.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N_pad)
    fw = np.fft.rfft(w, N_pad)
    cross = np.fft.irfft(fa * np.conj(fw), N_pad)[:n]
    grad = 2.0 * cross
    return grad, mval


def _grad_multi_active(a, n_active=5):
    """Gradient from average of top active constraints."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    # Get top n_active indices
    indices = np.argsort(c)[::-1][:n_active]
    threshold = mval - 0.01 * mval
    active = [k for k in indices if c[k] >= threshold]
    if not active:
        active = [int(indices[0])]
    
    N_pad = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N_pad)
    grad = np.zeros(n)
    for k_idx in active:
        # grad_j = 2 * a[k_idx - j] for valid j
        offset = k_idx
        for j in range(n):
            idx = offset - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(active)
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _fista_adaptive(a0, deadline, max_iter=200000, beta_start_frac=0.08, beta_min=1e-8, beta_decay=0.85, decay_every=150):
    """FISTA with adaptive beta annealing and momentum reset."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    mval0 = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * mval0)

    beta = beta_start_frac * mval0
    y = a.copy()
    t_k = 1.0
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, _ = _grad_smooth_max(y, beta)

        # Step size: L ~ 2 * max(a) * (2n-1) / beta for smooth max
        a_max = float(y.max())
        L_est = 2.0 * a_max * (2 * n - 1) / beta + 1e-10
        lr = 1.0 / L_est

        y_new = y - lr * grad
        a_new = _project_simplex(y_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        y = a_new + alpha * (a_new - a)
        t_k = t_new
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Beta annealing
        if it % decay_every == decay_every - 1:
            beta = max(beta_min, beta * beta_decay)
            y = a.copy()
            t_k = 1.0
            no_imp = 0

        # Stuck detection
        if no_imp > 5000:
            a = best_a.copy()
            y = a.copy()
            t_k = 1.0
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
            no_imp = 0

    return best_a, best_score


def _build_lp_matrix(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_solve(a, M_lp, deadline, warm_start=None, away_eps=0.0):
    """Solve the LP: max sum(b) s.t. A@b <= M_lp, -away_eps <= b."""
    n = len(a)
    A = _build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_lp)

    if away_eps > 0:
        bounds = [(-away_eps, None)] * n
    else:
        bounds = [(0.0, None)] * n

    try:
        rem = deadline - time.monotonic()
        tl = max(0.5, min(10.0, rem * 0.03))
        kwargs = {'time_limit': tl}
        if warm_start is not None:
            kwargs['x0'] = warm_start
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options=kwargs)
    except Exception:
        return None, None

    if res.success:
        return res.x, res
    return None, None


def _lp_refine_v2(a0, deadline, max_iter=50000, inflate_start=1.004, inflate_min=1.0003, use_away=False):

    a = np.asarray(a0, dtype=np.float64).copy()

    c = _fft_convolve(a)
    best_a = a.copy()

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):

        M = float(c.max())

        t_lp_start = time.monotonic()

        total_lp_time += time.monotonic() - t_lp_start

        if g is None:
            no_imp += 10
            if no_imp > 50:
                break
            continue

        # Clamp to non-negative for blending
        g_clamped = np.maximum(g, 0.0)
        sum_g = float(g_clamped.sum())

        if sum_g > 1.0 + 1e-5:
            g_norm = g_clamped / sum_g
            warm_start = g_clamped  # warm start for next LP

            # Line search with golden section refinement
            best_t = 0.0
            best_sc = best_score
            ts_coarse = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            for t in ts_coarse:
                a_new = (1.0 - t) * a + t * g_norm
                a_new = np.maximum(a_new, 0.0)

            if best_t > 0 and best_sc < best_score - 1e-13:

                a_new = np.maximum(a_new, 0.0)

                if no_imp > 80:

                    inflate = inflate_start
                    no_imp = 0
                    warm_start = None
                    if it > 300:
                        break
        else:
            no_imp += 1
            if no_imp > 60:
                break

    return best_a, best_score


def _lp_homotopy(a0, deadline, max_iter=20000):
    """LP with adaptive inflate (homotopy): start loose, tighten gradually."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    # Start with a relatively loose constraint
    inflate = 1.02
    no_imp = 0
    warm_start = None

    while time.monotonic() < deadline and no_imp < 200:
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g, _ = _lp_solve(a, M_lp, deadline, warm_start=warm_start)
        if g is None:
            no_imp += 5
            continue

        g_clamped = np.maximum(g, 0.0)
        sum_g = float(g_clamped.sum())

        if sum_g > 1.0 + 1e-4:
            g_norm = g_clamped / sum_g
            warm_start = g_clamped

            best_t = 0.0
            best_sc = best_score
            for t in [0.1, 0.3, 0.5, 0.8, 1.0]:
                a_new = (1.0 - t) * a + t * g_norm
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                sc = _score_norm(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t

            if best_t > 0 and best_sc < best_score - 1e-13:
                a_new = (1.0 - best_t) * a + best_t * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                best_score = best_sc
                best_a = a_new.copy()
                a = a_new
                no_imp = 0
                # Tighten
                inflate = max(1.0001, inflate * 0.9)
            else:
                no_imp += 1
                inflate = min(0.05, inflate * 1.05)
                if no_imp > 50:
                    break
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a, best_score


def _multi_start_lp(a0, deadline, n_starts=3, away_scale=0.005):
    """Try LP from slightly perturbed starting points."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    per_start_time = (deadline - time.monotonic()) / n_starts
    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        if per_start_time < 5:
            break

        # Create perturbed start
        if i == 0:
            a_try = a.copy()
        else:
            noise = rng.standard_normal(n) * away_scale
            a_try = a + noise
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try / a_try.sum()

        sc_try = _score_norm(a_try)
        # Only proceed if not much worse
        if sc_try > best_score * 1.05:
            continue

        t_end = time.monotonic() + per_start_time
        a_opt, sc_opt = _lp_refine_v2(a_try, t_end, max_iter=5000,
                                       inflate_start=1.008, inflate_min=1.001,
                                       use_away=(i > 0))
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  multi-start {i}: score={sc_opt:.10f}")

    return best_a, best_score


def _interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _round_sequence(a):
    """Round to reduce numerical noise while preserving score."""
    a = np.asarray(a, dtype=np.float64).copy()
    # Scale so the sequence has a nice sum
    s = a.sum()
    a_scaled = a * 10000.0 / s
    a_rounded = np.round(a_scaled, 4)
    a_rounded = np.maximum(a_rounded, 0.0)
    if a_rounded.sum() < 0.01:
        return a
    a_rounded = a_rounded / a_rounded.sum()
    # Check if rounding improved or didn't hurt
    if _score_norm(a_rounded) <= _score_norm(a) + 1e-10:
        return a_rounded
    return a
