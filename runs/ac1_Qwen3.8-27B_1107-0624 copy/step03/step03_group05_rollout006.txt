        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c -= c.min()
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] contributes to conv positions j, j+1, ..., j+n-1
        # i.e., A[k, j] = a[j] for k in [j, j+n-1]
        A[j:j + n, j] = a[j]
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _make_profile(n, kind='u', **kwargs):
    """Generate various profile shapes."""
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        eta = kwargs.get('eta', 1.5)
        a = 1.0 + eta * (t - 0.5)
    elif kind == 'u_shape':
        alpha = kwargs.get('alpha', 2.0)
        a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    elif kind == 'u_shape_cubic':
        alpha = kwargs.get('alpha', 2.0)
        u = 2.0 * t - 1.0
        a = 1.0 + alpha * u**2 - 0.5 * alpha * u**4
    elif kind == 'bimodal':
        sep = kwargs.get('sep', 0.2)
        width = kwargs.get('width', 0.08)
        a = np.exp(-0.5 * ((t - sep) / width) ** 2) + np.exp(-0.5 * ((t - (1-sep)) / width) ** 2)
    elif kind == 'cosine':
        a = 1.0 + kwargs.get('amp', 0.5) * np.cos(2 * np.pi * t)
    elif kind == 'v_shape':
        alpha = kwargs.get('alpha', 2.0)
        a = 1.0 + alpha * np.abs(2.0 * t - 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _downsample(a, factor):
    """Downsample by averaging consecutive groups."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    new_n = n // factor
    if new_n == 0:
        return a[:1]
    # Reshape and average
    trimmed = a[:new_n * factor]
    return trimmed.reshape(new_n, factor).mean(axis=1) / trimmed.reshape(new_n, factor).mean(axis=1).sum() * new_n


def _adaptive_lp_refine(a, deadline, max_iter=50000):
    """
    Core LP refinement with adaptive inflation and blending.
    This is the main workhorse.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    # Adaptive parameters
    inflate = 0.003  # Start with 0.3% inflation
    blend = 0.5  # Start with aggressive blending
    no_imp = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = _fft_convolve(a)
        M = float(c.max())

        # Try with current M * (1 + inflate)
        M_lp = M * (1.0 + inflate)
        b = _lp_spread(a, M_lp)

        if b is None:
            # Try with larger inflation
            inflate = min(0.1, inflate * 1.5)
            continue

        sum_b = float(b.sum())
        if sum_b <= 1.0 + 1e-6:
            # LP can't improve much; increase inflation
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > 500:
                break
            continue

        # Normalize b
        b = b / sum_b

        # Blend
        a_new = (1.0 - blend) * a + blend * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Evaluate
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # On improvement, can reduce inflation slightly
            inflate = max(0.0005, inflate * 0.95)
            # Keep blend high for a while
            if no_imp == 0 and blend < 0.5:
                blend = min(0.5, blend * 1.1)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.1, blend * 0.95)
                inflate = min(0.05, inflate * 1.05)
            if no_imp > max(200, n):
                # Reset to best and try different params
                a = best_a.copy()
                inflate = 0.01
                blend = 0.4
                no_imp = 0
                if time.monotonic() > deadline - 30:
                    break

        a = a_new

    return best_a


def _gradient_polish(a, deadline, max_iter=30000):
    """
    Gradient-based polish using smooth-max surrogate.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.03 * float(c.max()))
    lr = 0.05
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1

        # Compute smooth-max gradient
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        # Gradient of smooth_max(conv(a,a)) w.r.t. a
        # d/da_i [sum_k w_k * conv(a,a)_k] = 2 * sum_k w_k * (conv(a, e_i))_k
        # = 2 * (w * a reversed) convolved appropriately
        # Actually: d(conv(a,a)/da_i) is the vector with (conv(a,a)/da_i)_k = a_{k-i} + a_{i-k} (for valid indices)
        # The gradient component for a_i is: sum_k w_k * (a_{k-i} + a_{k-(n-1-i)})... 
        # Let me use the FFT approach:
        # grad_i = 2 * sum_k w_k * (a * flip(a))_k evaluated at position i
        # Actually, the gradient of sum_k w_k * c_k where c_k = sum_j a_j a_{k-j}
        # d/dc: dc_k/da_i = a_{k-i} + a_{k-i} ... no
        # c_k = sum_j a_j a_{k-j}, so dc_k/da_i = a_{k-i} + a_i * (d a_{k-j}/da_i) 
        # Actually dc_k/da_i = a_{k-i} (if 0<=k-i<n) + a_i * delta... 
        # Wait: c_k = sum_j a_j * a_{k-j}. If we differentiate w.r.t. a_i:
        # dc_k/da_i = 1 * a_{k-i} (from the first factor, when j=i) + a_i * (delta_{k-j, i} = delta_{j, k-i})
        # = a_{k-i} + a_{k-i} = 2 * a_{k-i} (if k-i is in [0, n-1])
        # Wait no: c_k = sum_j a_j * a_{k-j}. 
        # dc_k/da_i: the term where j=i gives a_{k-i}. The term where k-j=i, i.e., j=k-i, gives a_{k-i} * (da_{k-(k-i)}/da_i) = a_{k-i} * 1.
        # But wait, we need j and k-j to both be in [0, n-1].
        # So dc_k/da_i = a_{k-i} (if 0 <= k-i < n) + a_{k-i} (if 0 <= k-i < n)... 
        # Hmm, actually if j=i, we get a_{k-i}. If j=k-i (different from i unless k=2i), we get a_i * (da_{k-j}/da_i) = a_i * 1 = a_i... 
        # No wait. Let me be more careful.
        # c_k = sum_j a_j * a_{k-j}
        # dc_k/da_i = sum_j [delta_{j,i} * a_{k-j} + a_j * delta_{k-j,i}]
        # = a_{k-i} + a_{k-i} (if 0 <= k-i < n) [first term: j=i, second term: k-j=i means j=k-i, so a_{k-i}*1]
        # Wait, second term: sum_j a_j * delta_{k-j, i} = a_{k-i} (if 0 <= k-i < n)
        # So dc_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (if 0 <= k-i < n), else 0? 
        # No! First term: delta_{j,i} * a_{k-j} = a_{k-i} (only if k-i is in [0,n-1])
        # Second term: a_j * delta_{k-j,i} = a_{k-i} (only if k-i is in [0,n-1])
        # So dc_k/da_i = 2 * a_{k-i} if 0 <= k-i < n, else 0? 
        # Hmm, that doesn't seem right either. Let me try with a specific example.
        # a = [a0, a1, a2], n=3
        # c_0 = a0^2
        # c_1 = a0*a1 + a1*a0 = 2*a0*a1
        # c_2 = a0*a2 + a1*a1 + a2*a0 = 2*a0*a2 + a1^2
        # c_3 = a1*a2 + a2*a1 = 2*a1*a2
        # c_4 = a2^2
        # dc_0/da_0 = 2*a0 ✓ (2*a0 since k-i = 0)
        # dc_1/da_0 = 2*a1 ✓ (k-i = 1, a_1)
        # dc_2/da_0 = 2*a2 ✓ (k-i = 2, a_2)
        # dc_1/da_1 = 2*a0 ✓ (k-i = 0, a_0)
        # dc_2/da_1 = 2*a1 ✓ (k-i = 1, a_1)
        # dc_3/da_2 = 2*a1 ✓ (k-i = 1, a_1)
        # Great, so dc_k/da_i = 2 * a_{k-i} if 0 <= k-i < n, else 0.
        
        # So gradient_i = sum_k w_k * 2 * a_{k-i} (for valid k-i)
        # = 2 * sum_k w_k * a_{k-i}
        # This is 2 * (w * a) convolved, evaluated at position (n-1-i) or something...
        # Let me think: sum_k w_k * a_{k-i} where k ranges over [0, 2n-2] and k-i over [0, n-1]
        # = sum_{j=0}^{n-1} a_j * w_{j+i} (where j = k-i, so k = j+i)
        # This is (w * a)_reversed or something...
        # Actually it's the cross-correlation of w and a, evaluated at shift i.
        # cross_corr(w, a)[i] = sum_j w_{j+i} * a_j
        # Which is the same as conv(w, a)[i + n - 1] (for full convolution)... not exactly.
        # Let me just use FFT: 
        # We want g_i = sum_k w_k * a_{k-i} for i = 0, ..., n-1
        # This is (w conv a) evaluated at specific positions, or equivalently:
        # g = ifft(fft(w) * fft(a_reversed)) [n-1:2n-1] or similar.
        
        # Actually, the simplest way: use the fact that this is a convolution-like operation.
        # g_i = sum_{k} w_k * a_{k-i}
        # Let me define h = flip(a), then h_j = a_{n-1-j}
        # conv(w, h)_m = sum_k w_k * h_{m-k} = sum_k w_k * a_{n-1-m+k}
        # Hmm, this is getting complicated. Let me just use the direct formula with FFT.
        
        # Alternative: grad = 2 * ifft(fft(w, N) * conj(fft(a, N))) [n-1:2n-1]
        # Actually, let me use a simpler approach:
        # The gradient of f(a) = sum_k w_k * c_k where c = conv(a,a)
        # is: grad_i = 2 * sum_k w_k * (conv(a, e_i))_k
        # where e_i is the unit vector at position i.
        # conv(a, e_i)_k = a_{k-i} (if valid)
        # So grad_i = 2 * sum_k w_k * a_{k-i}
        # = 2 * <w, shift_i(a)>... 
        
        # The cleanest way with FFT:
        # Define the "correlation" of w with a: corr[i] = sum_k w[k] * a[k-i]
        # This is ifft(fft(w) * conj(fft(a)))... no, that's cross-correlation.
        # Actually: ifft(conj(fft(a)) * fft(w)) gives the cross-correlation of a and w.
        # Let me just compute it directly for small n, or use the full FFT trick.
        
        # For efficiency, I'll compute:
        # g = ifft(fft(w, N) * fft(a_flipped, N)) where a_flipped = a[n-1::-1]
        # Then take the appropriate slice.
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_flip = np.fft.rfft(a[::-1], N)
        # conv(w, a_flipped) = ifft(fw * fa_flip)
        # conv(w, a_flipped)[m] = sum_k w[k] * a_flipped[m-k] = sum_k w[k] * a[n-1-(m-k)]
        # = sum_k w[k] * a[n-1-m+k]
        # We want g[i] = sum_k w[k] * a[k-i]
        # Set n-1-m+k = k-i => n-1-m = -i => m = n-1+i
        # So g[i] = conv(w, a_flipped)[n-1+i]
        h_full = np.fft.irfft(fw * fa_flip, N)[:2*n-1]
        grad = 2.0 * h_full[n-1:2*n-1]
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        # Adam-like update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.05
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                no_imp = 0

        # Decay beta periodically
        if it % 200 == 0:
            beta = max(1e-10, beta * 0.7)

    return best_a


def _peak_targeted_polish(a, deadline, max_iter=20000):
    """
    Polish by directly targeting the top peaks.
    At each step, find the peak lag, compute gradient of that specific
    convolution value, and take a projected step.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _score(a)
    lr = 0.02
    no_imp = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = _fft_convolve(a)
        # Find top 3 peaks
        top_k = np.argsort(c)[::-1][:3]
        peak_vals = c[top_k]
        
        # Weight: emphasize the largest peak
        w = np.zeros_like(c)
        w[top_k] = peak_vals ** 2
        w = w / w.sum()
        
        # Compute gradient of weighted sum
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_flip = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_flip, N)[:2*n-1]
        grad = 2.0 * h_full[n-1:2*n-1]
        
        a_new = _project_simplex(a - lr * grad)
        
        score = _score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > 500:
                a = best_a.copy()
                lr = 0.02
                no_imp = 0

    return best_a


def _block_perturbation(a, rng, strength=0.1):
    """Create a perturbed version of a by swapping/reshuffling blocks."""
    a = a.copy()
    n = len(a)
    n_blocks = rng.integers(2, max(3, n // 16))
    block_size = n // n_blocks
    perm = rng.permutation(n_blocks)
    a_new = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        p_start = perm[i] * block_size
        p_end = p_start + block_size if perm[i] < n_blocks - 1 else n
        a_new[start:end] = a[p_start:p_end]
    # Reorder within blocks
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        idx = np.arange(start, end)
        rng.shuffle(idx)
        a_new[idx] = a[idx] * (1 + strength * rng.normal(size=len(idx)))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    return a_new / s if s > 1e-10 else a


def _full_optimize(a_init, n, deadline):
    """Run the full optimization pipeline on a starting sequence."""
    a = _interpolate_to_length(a_init, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = _make_profile(n, 'u_shape', alpha=2.0)
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    print(f"    Init: score={best_score:.10f}")

    # Phase 1: LP refinement (most of the time)
    lp_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.5
    a_lp = _adaptive_lp_refine(a, lp_deadline, max_iter=50000)
    sc = _score(a_lp)
    if sc < best_score:
        best_score = sc
        best_a = a_lp.copy()
    print(f"    LP: score={sc:.10f}")

    # Phase 2: Gradient polish
    if time.monotonic() < deadline - 20:
        gp_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.3
        a_gp = _gradient_polish(best_a, gp_deadline, max_iter=20000)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"    Grad: score={sc:.10f}")

    # Phase 3: Peak-targeted polish
    if time.monotonic() < deadline - 15:
        pt_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.1
        a_pt = _peak_targeted_polish(best_a, pt_deadline, max_iter=10000)
        sc = _score(a_pt)
        if sc < best_score:
            best_score = sc
            best_a = a_pt.copy()
        print(f"    Peak: score={sc:.10f}")

    # Phase 4: Final LP
    if time.monotonic() < deadline - 10:
        final_deadline = deadline - 5
        a_final = _adaptive_lp_refine(best_a, final_deadline, max_iter=20000)
        sc = _score(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
        print(f"    Final LP: score={sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):

    Uses aggressive adaptive LP as the core, combined with gradient polishing,

    t_start = time.monotonic()

    best_seq = None
    best_score = np.inf

    parent = None

            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc

    lengths_to_try = list(dict.fromkeys(lengths_to_try))

    print(f"Lengths to try: {lengths_to_try}")
    print(f"Time budget: {budget_s}s")

    # Allocate time: give most to parent length, rest to others
    # Try parent length first with most time
    main_deadline = t_start + 900  # 15 minutes for main
    other_deadline = deadline - 30

    # Main optimization at parent length
    if parent is not None:
