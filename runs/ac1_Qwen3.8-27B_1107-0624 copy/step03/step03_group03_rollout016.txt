def _lp_spread_max_sum(a, M, n_constraints=80):
    """LP: max sum(b) s.t. conv(a,b) <= M for top constraints, b >= 0."""
    n = len(a)
    c_aa = _fft_convolve(a)
    # Get most important constraints
    idx = np.argsort(c_aa)[::-1]
    # Include top constraints and their neighbors
    active = set()
    for i in idx[:n_constraints]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)

    n_ineq = len(active)
    A = np.zeros((n_ineq, n))
    b_ub = np.full(n_ineq, M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]

    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

    idx = np.argsort(c_aa)[::-1]
    active = set()

        for offset in range(-3, 4):

            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)

    n_active = len(active)
    # Variables: b_0, ..., b_{n-1}, t (total n+1 variables)
    # Objective: min t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq[0, :n] = 1.0

        if g2 is not None and g2.sum() > 1e-6:
            g2 = g2 / g2.sum()
            for t_blend in [0.3, 0.5, 0.7]:
                a_try = _project_simplex((1 - t_blend) * a + t_blend * g2)
                sc_try = float(_fft_convolve(a_try).max())
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    no_imp = 0
                    break

        if g is None and g2 is None:
            no_imp += 5
        elif no_imp > 0:
            no_imp += 1

        if no_imp > 2000:
            break

    return best_a


def _interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _annealed_subgrad(a, deadline, max_iter=50000, top_k=1, lr=0.005,
                      T_start=1.0, T_end=0.01, noise_scale=0.02):
    """Subgradient with simulated annealing restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = float(_fft_convolve(a).max())
    v = np.zeros(n)
    mu = 0.85
    rng = np.random.default_rng(12345)
    it = 0
    T = T_start
    T_decay = (T_end / T_start) ** (1.0 / max(max_iter, 1))

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        T = max(T * T_decay, T_end)

        c = _fft_convolve(a)
        g = _compute_subgrad(a, c, top_k)
        g = g - g.mean()

        v = mu * v + g
        lr_t = lr / (1.0 + it / 3000.0)
        a_new = _project_simplex(a - lr_t * v)

        # Annealing: sometimes add noise
        if T > 0.05 and rng.random() < 0.05:
            noise = rng.normal(size=n) * noise_scale * T
            a_new = _project_simplex(a_new + noise)

        c_new = _fft_convolve(a_new)
        sc_new = float(c_new.max())

        # Metropolis acceptance
        if sc_new < best_sc or rng.random() < np.exp(-(sc_new - best_sc) / (T * 1e-4)):
            a = a_new
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                v = np.zeros(n)
        else:
            # Rejected: keep current a

    return best_a


def _block_optimize(a, deadline, block_size=16, max_iter=5000):
    """Optimize blocks of coordinates sequentially."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = float(_fft_convolve(a).max())
    rng = np.random.default_rng(777)

    while time.monotonic() < deadline:
        # Pick a random block
        start = rng.integers(0, max(1, n - block_size))
        block_idx = np.arange(start, min(n, start + block_size))

        # Subgradient step focused on the block
        c = _fft_convolve(a)
        g = _compute_subgrad(a, c, top_k=3)
        g = g - g.mean()

        # Only update the block
        a_new = a.copy()
        a_new[block_idx] -= 0.01 * g[block_idx]
        a_new = _project_simplex(a_new)

        sc = float(_fft_convolve(a_new).max())
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new

    return best_a


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining momentum subgradient, Nesterov acceleration,
    LP spreading (both max-sum and min-peak), annealed restarts, and multi-length
    exploration to minimize 2n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    n_parent = parent.size if parent is not None else 1024

    # === Phase 1: Momentum subgradient on parent (200s) ===
    print("\n=== Phase 1: Momentum subgradient ===")
    phase1_end = min(t_start + 200.0, deadline - 300.0)
    if parent is not None:
        a1 = _momentum_subgrad(parent, phase1_end, max_iter=300000,
                               top_k=1, lr=0.004, mu=0.85)
        sc1 = _score(a1)
        print(f"  Momentum subgrad: {sc1:.10f}")
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()

        # Also try with top_k=3
        phase1b_end = min(t_start + 250.0, deadline - 250.0)
        a1b = _momentum_subgrad(parent, phase1b_end, max_iter=300000,
                                top_k=3, lr=0.003, mu=0.85)
        sc1b = _score(a1b)
        print(f"  Momentum subgrad (top_k=3): {sc1b:.10f}")
        if sc1b < best_score:
            best_score = sc1b
            best_seq = a1b.copy()

    # === Phase 2: LP refinement (300s) ===
    print("\n=== Phase 2: LP refinement ===")
    phase2_end = min(t_start + 550.0, deadline - 200.0)
    if best_seq is not None:
        a2 = _lp_refine_adaptive(best_seq, phase2_end, max_iter=2000,
                                 n_cons_start=40, n_cons_end=120,
                                 inflate_start=0.001, inflate_end=0.0001)
        sc2 = _score(a2)
        print(f"  LP refine: {sc2:.10f}")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()

    # === Phase 3: Nesterov subgradient (200s) ===
    print("\n=== Phase 3: Nesterov subgradient ===")
    phase3_end = min(t_start + 750.0, deadline - 200.0)
    if best_seq is not None:
        a3 = _nesterov_subgrad(best_seq, phase3_end, max_iter=300000,
                               top_k=1, lr=0.008)
        sc3 = _score(a3)
        print(f"  Nesterov: {sc3:.10f}")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()

    # === Phase 4: Multi-length exploration (200s) ===
    print("\n=== Phase 4: Multi-length exploration ===")
    phase4_end = min(t_start + 950.0, deadline - 150.0)
    candidate_lengths = [n_parent]
    for ln in [768, 1152, 1280, 1536]:
        if ln not in candidate_lengths and abs(ln - n_parent) > 64:
            candidate_lengths.append(ln)

    for n_t in candidate_lengths[:3]:
        if time.monotonic() > phase4_end:
            break
        if parent is not None:
            a_t = _interpolate_profile(parent, n_t)
        else:
            a_t = _make_raised_cosine(n_t)

        # Quick optimization
        quick_end = min(time.monotonic() + 40.0, phase4_end - 30.0)
        a_t = _momentum_subgrad(a_t, quick_end, max_iter=100000,
                                top_k=1, lr=0.004, mu=0.85)
        sc_t = _score(a_t)
        print(f"  n={n_t}: {sc_t:.10f}")
        if sc_t < best_score:
            best_score = sc_t
            best_seq = a_t.copy()

    # === Phase 5: Annealed subgradient (200s) ===
    print("\n=== Phase 5: Annealed subgradient ===")
    phase5_end = min(t_start + 1150.0, deadline - 150.0)
    if best_seq is not None:
        a5 = _annealed_subgrad(best_seq, phase5_end, max_iter=100000,
                               top_k=1, lr=0.005, T_start=1.0,
                               T_end=0.01, noise_scale=0.01)
        sc5 = _score(a5)
        print(f"  Annealed: {sc5:.10f}")
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()

    # === Phase 6: Final LP polish (200s) ===
    print("\n=== Phase 6: Final LP polish ===")

    # === Phase 7: Final subgradient polish (remaining) ===
    print("\n=== Phase 7: Final polish ===")
    final_end = deadline - 5.0
    if best_seq is not None and time.monotonic() < final_end:
        a7 = _momentum_subgrad(best_seq, final_end, max_iter=200000,
                               top_k=2, lr=0.002, mu=0.9)

        print(f"  Final subgrad: {sc7:.10f}")
        if sc7 < best_score:

            best_seq = a7.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
