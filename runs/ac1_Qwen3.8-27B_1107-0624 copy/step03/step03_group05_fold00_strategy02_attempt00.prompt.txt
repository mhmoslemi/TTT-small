    a = np.asarray(a, dtype=np.float64).ravel()

def _project_simplex(v):

    n = len(v)
    if n == 0:
        return v

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]

    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    return None

def _block_swap_optimize(a, deadline, max_iter=5000):
    """Try random block swaps to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(12345)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Random block swap
        block_size = rng.integers(max(2, n // 32), max(3, n // 8))
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        a_new[i:i + block_size], a_new[j:j + block_size] = \
            a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break

    return best_a


def _peak_aware_gradient(a, beta, top_k=5):
    """Gradient that emphasizes the top-k peaks more."""
    n = len(a)
    c = _fft_convolve(a)
    # Combine uniform weight with peak-focused weight
    w_uniform = _smooth_max_weight(c, beta)
    # Focus on top-k peaks
    sorted_idx = np.argsort(c)[::-1]
    w_peak = np.zeros_like(c)
    peak_vals = c[sorted_idx[:top_k]]
    w_peak[sorted_idx[:top_k]] = peak_vals / peak_vals.sum()
    w = 0.7 * w_uniform + 0.3 * w_peak
    w = w / w.sum()

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())
