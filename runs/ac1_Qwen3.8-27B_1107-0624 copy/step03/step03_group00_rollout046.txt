def _next_pow2(n):

    return max(2, k)


def _full_convolve(a):

        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _make_fourier_profile(n, n_harm, amps, phases, dc):
    t = np.arange(n)
    a = np.full(n, dc)
    for k in range(1, n_harm + 1):
        a += amps[k - 1] * np.cos(2.0 * np.pi * k * t / n + phases[k - 1])
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_poly_profile(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, c in enumerate(coeffs):
        a += c * t ** (i + 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_tapered_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_extra_bump(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.exp(-20.0 * (t - 0.5) ** 2) + 0.3 * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                no_imp_limit=300):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-5, beta * 0.5)

        if no_imp > no_imp_limit:
            ns = 0.05 + 0.05 * (1.0 if best_score < 1.51 else 2.0)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _coord_descend(a, n_passes=10, deadline=None, rng=None, verbose=False):
    """
    Coordinate descent with incremental convolution updates.
    For each element, try several candidate values and keep the best.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    best_score = _score_norm(a)
    current_score = best_score

    c = _full_convolve(a)
    S = float(a.sum())

    factors = [0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 5.0]

    for pass_num in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Randomize element order
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved = 0
        for i in order:
            old_val = a[i]
            S_others = S - old_val
            if S_others <= 1e-15:
                continue

            # Compute base convolution (with a_i = 0)
            # c_base[k] = c[k] - contribution of a_i
            # Contribution: for k in [i, i+n-1]:
            #   if k == 2*i: old_val^2
            #   else: 2 * old_val * a[k-i] (if 0 <= k-i < n)
            #   Only if 0 <= k-i < n

            best_x = old_val
            best_f = current_score

            # Candidate values
            candidates = [old_val * f for f in factors]
            # Also try to match the current max
            max_c = float(c.max())
            if old_val > 1e-15:
                # Estimate: if lag k is active, x that makes c_k = max_c * 0.99
                # c_k(x) = c_k_base + delta_k(x)
                # For the lag where c is max, try adjusting x
                k_max = int(np.argmax(c))
                if 0 <= k_max - i < n:
                    k_off = k_max - i
                    if k_max == 2 * i:
                        c_base_k = c[k_max] - old_val * old_val
                        target = max_c * 0.99
                        x_try = np.sqrt(max(0, target - c_base_k))
                        candidates.append(x_try)
                    else:
                        c_base_k = c[k_max] - 2.0 * old_val * a[k_off]
                        target = max_c * 0.99
                        coef = 2.0 * a[k_off]
                        if coef > 1e-15:
                            x_try = (target - c_base_k) / coef
                            if x_try > 0:
                                candidates.append(x_try)

            for x in candidates:
                x = max(0.0, x)
                S_new = S_others + x
                if S_new < 1e-10:
                    continue

                # Compute new max convolution value incrementally
                # Only lags k where 0 <= k-i < n are affected
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)

                # New c values for affected lags
                new_c_max = 0.0
                for k in range(k_start, k_end + 1):
                    k_off = k - i
                    if 0 <= k_off < n:
                        if k == 2 * i:
                            c_new_k = c[k] - old_val * old_val + x * x
                        else:
                            c_new_k = c[k] - 2.0 * old_val * a[k_off] + 2.0 * x * a[k_off]
                    else:
                        c_new_k = c[k]
                    if c_new_k > new_c_max:
                        new_c_max = c_new_k

                # Unaffected lags
                if k_start > 0:
                    unaffected_max = float(c[:k_start].max())
                else:
                    unaffected_max = 0.0
                if k_end < 2 * n - 2:
                    unaffected_tail = float(c[k_end + 1:].max())
                    if unaffected_tail > unaffected_max:
                        unaffected_max = unaffected_tail
                total_max = max(new_c_max, unaffected_max)

                f_val = 2.0 * n * total_max / (S_new * S_new)
                if f_val < best_f - 1e-14:
                    best_f = f_val
                    best_x = x

            if best_x != old_val:
                # Update a, c, S
                dx = best_x - old_val
                if abs(dx) > 1e-15:
                    a[i] = best_x
                    S = S + dx
                    # Update convolution
                    k_start = i
                    k_end = min(i + n - 1, 2 * n - 2)
                    for k in range(k_start, k_end + 1):
                        k_off = k - i
                        if 0 <= k_off < n:
                            if k == 2 * i:
                                c[k] += dx * (old_val + best_x)
                            else:
                                c[k] += 2.0 * dx * a[k_off]
                    c = np.maximum(c, 0.0)
                    current_score = best_f
                    if current_score < best_score:
                        best_score = current_score
                        improved += 1

        if verbose:
            print(f"    CoordDescend pass {pass_num}: score={best_score:.10f}, improved={improved}")
        if improved == 0:
            break

    return a, best_score


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        # Build LP: maximize sum(b) s.t. convolve(a,b)[k] <= M, b >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub_arr = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x, 0.0)
            s_b = float(b.sum())
            if s_b < 1e-10:
                break
            b_norm = b / s_b
            a_new = (1.0 - t) * a + t * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if sc_new < current_score:
                current_score = sc_new
                current_best = a_new.copy()
            a = a_new
            c = _full_convolve(a)
            M = float(c.max())
            t = max(0.1, t * 0.9)
        else:
            break
    return current_best, current_score


def _fourier_optimize(n, n_harm, deadline, rng):
    """Optimize Fourier coefficients to minimize the score."""
    best_score = np.inf
    best_p = None

    # Initial random Fourier coefficients
    for trial in range(5):
        if time.monotonic() > deadline:
            break
        amps = rng.uniform(0.1, 1.0, size=n_harm)
        phases = rng.uniform(0, 2 * np.pi, size=n_harm)
        dc = 1.0

        # Coordinate descent on Fourier parameters
        best_local = np.inf
        current_p = _make_fourier_profile(n, n_harm, amps, phases, dc)
        current_score = _score_norm(current_p)
        best_local = current_score
        best_p_local = current_p.copy()

        for feat in range(200):
            if time.monotonic() > deadline:
                break
            improved = False
            # Try perturbing each harmonic
            for k in range(n_harm):
                if time.monotonic() > deadline:
                    break
                for da in [-0.3, -0.1, 0.1, 0.3]:
                    new_amps = amps.copy()
                    new_amps[k] = max(0.01, new_amps[k] + da)
                    for dp in [-0.5, 0.5]:
                        new_phases = phases.copy()
                        new_phases[k] = new_phases[k] + dp
                        p_try = _make_fourier_profile(n, n_harm, new_amps, new_phases, dc)
                        sc_try = _score_norm(p_try)
                        if sc_try < best_local - 1e-10:
                            best_local = sc_try
                            best_p_local = p_try.copy()
                            amps = new_amps.copy()
                            phases = new_phases.copy()
                            improved = True
                        elif sc_try < current_score - 1e-10:
                            current_score = sc_try
                            amps = new_amps.copy()
                            phases = new_phases.copy()
                            improved = True
                for dd in [-0.2, 0.2]:
                    new_dc = max(0.1, dc + dd)
                    p_try = _make_fourier_profile(n, n_harm, amps, phases, new_dc)
                    sc_try = _score_norm(p_try)
                    if sc_try < best_local - 1e-10:
                        best_local = sc_try
                        best_p_local = p_try.copy()
                        dc = new_dc
                        improved = True
            if not improved:
                break

        if best_local < best_score:
            best_score = best_local
            best_p = best_p_local.copy()

    return best_p, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Generate initial candidates ---
    candidates = []

    # Various n values
    n_list = [512, 1024, 2048]

    for n in n_list:
        # Fourier profiles
        for n_harm in [3, 5, 8]:
            amps = np.concatenate([np.ones(n_harm) * np.array([0.8, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05][:n_harm])])
            phases = np.zeros(n_harm)
            p = _make_fourier_profile(n, n_harm, amps, phases, 1.0)
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'fourier_{n_harm}'))

        # Tapered profiles
        for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
            p = _make_tapered_profile(n, alpha)
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'taper_{alpha}'))

        # Extra bump
        p = _make_extra_bump(n)
        sc = _score_norm(p)
        candidates.append((sc, p, n, 'extra_bump'))

        # Polynomial profiles
        for coeffs in [[0.5, -1.0], [1.0, -0.5], [0.3, 0.3, -0.5], [-0.5, 1.0, -0.3]]:
            p = _make_poly_profile(n, coeffs)
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'poly_{len(coeffs)}'))

    # Parent interpolations
    if parent is not None:
        for n in n_list:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Quick Adam on top candidates ---
    phase2_end = t_start + 400.0
    for sc, p, n, label in candidates[:6]:
        if time.monotonic() > phase2_end:
            break
        per = min(60.0, phase2_end - time.monotonic())
        if per < 15:
            break
        p_b, sc_b = _adam_logit(p, time.monotonic() + per, rng, lr=0.05,
                                beta_frac=0.1, beta_decay=0.93, no_imp_limit=200)
        _update(p_b, sc_b, f'adam_{label}_n{n}')

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Coordinate descent (KEY INNOVATION) ---
    phase3_end = t_start + 900.0
    if best_seq is not None:
        n_cd = best_seq.size
        # Scale to a good size for CD (1024-2048 is sweet spot)
        cd_n = min(n_cd, 2048)
        if n_cd != cd_n:
            p_cd = _interp_profile(best_seq, cd_n)
        else:
            p_cd = best_seq.copy()
        sc_cd = _score_norm(p_cd)
        print(f"Phase 3: CoordDescend on n={cd_n}, score={sc_cd:.10f}")

        # Multiple rounds of CD with re-Adam in between
        for round_idx in range(3):
            if time.monotonic() > phase3_end - 60:
                break
            cd_deadline = min(phase3_end, time.monotonic() + 80.0)
            p_cd, sc_cd = _coord_descend(p_cd, n_passes=15, deadline=cd_deadline, rng=rng)
            _update(p_cd, sc_cd, f'cd_r{round_idx}_n{cd_n}')

            # Brief Adam re-optimization to escape CD local minima
            if time.monotonic() < phase3_end - 30:
                adam_deadline = min(phase3_end, time.monotonic() + 30.0)
                p_cd, sc_cd = _adam_logit(p_cd, adam_deadline, rng, lr=0.03,
                                           beta_frac=0.06, beta_decay=0.94, no_imp_limit=100)
                _update(p_cd, sc_cd, f'adam_post_cd_r{round_idx}')

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: LP refinement ---
    phase4_end = t_start + 1200.0
    if best_seq is not None and time.monotonic() < phase4_end:
        lp_n = best_seq.size
        if lp_n > 2048:
            lp_p = _interp_profile(best_seq, 2048)
        else:
            lp_p = best_seq.copy()
        print(f"Phase 4: LP refine n={lp_p.size}, score={_score_norm(lp_p):.10f}")
        p_lp, sc_lp = _lp_refine(lp_p, min(phase4_end, time.monotonic() + 120.0), t=0.3, max_solves=20)
        _update(p_lp, sc_lp, 'LP_refine')

        # CD after LP
        if time.monotonic() < phase4_end - 20:
            cd_deadline = min(phase4_end, time.monotonic() + 60.0)
            p_lp, sc_lp = _coord_descend(p_lp, n_passes=10, deadline=cd_deadline, rng=rng)
            _update(p_lp, sc_lp, 'cd_post_LP')

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart with structured perturbations ---
    phase5_end = deadline - 60.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 3
        per_r = max(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 10)
            if per < 20:
                break

            # Structured perturbation: block-noise instead of i.i.d.
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            block_size = max(4, n_best // 32)
            block_noise = rng.normal(size=n_best // block_size + 1) * 0.15
            z_noise = np.repeat(block_noise, block_size)[:n_best]
            z = z + z_noise
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            # Adam then CD
            adam_per = per * 0.6
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + adam_per, rng, lr=0.04,
                                     beta_frac=0.07, beta_decay=0.93, no_imp_limit=150)
            _update(p_r, sc_r, f'restart{i}_adam_n{n_best}')

            cd_per = min(30.0, per - adam_per - 5)
            if cd_per > 10:
                p_r, sc_r = _coord_descend(p_r, n_passes=8, deadline=time.monotonic() + cd_per, rng=rng)
                _update(p_r, sc_r, f'restart{i}_cd_n{n_best}')

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine-tuning CD ---
    if time.monotonic() < deadline - 10 and best_seq is not None:
        final_deadline = deadline - 10
        p_final = best_seq.copy()
        p_final, sc_final = _coord_descend(p_final, n_passes=20, deadline=final_deadline, rng=rng, verbose=True)
        _update(p_final, sc_final, 'final_cd')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_tapered_profile(1024, 1.0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_tapered_profile(1024, 1.0)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
