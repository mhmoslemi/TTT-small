"""
Optimized step function search using L-BFGS-B on smooth max of convolution,
with monotone parameterization, adaptive beta scheduling, and subgradient polishing.
Key improvements over previous approach: superlinear convergence via L-BFGS-B,
monotone manifold constraint, and multi-scale beta annealing.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_value(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval
    return mval + beta * np.log(s)


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _grad_conv_weighted(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] with respect to a."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return e / s


def _lbfgs_optimize(a0, deadline, beta_init=0.005, total_betas=8,
                    max_iter_per_beta=300, label=""):
    """Run L-BFGS-B with adaptive beta annealing in log-space."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    c = _fft_convolve(a)
    best_val = float(c.max())

    betas = [beta_init * (0.5 ** i) for i in range(total_betas)]
    betas = [max(b, 1e-6) for b in betas]

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        def obj_fn(z):
            a_s = _softmax(z)
            c_s = _fft_convolve(a_s)
            return _smooth_max_value(c_s, beta)

        def grad_fn(z):
            a_s = _softmax(z)
            c_s = _fft_convolve(a_s)
            w = _smooth_max_weights(c_s, beta)
            grad_a = _grad_conv_weighted(a_s, w)
            dot = float(np.dot(a_s, grad_a))
            grad_z = a_s * (grad_a - dot)
            return grad_z

        z_cur = z.copy()
        options = {
            'maxiter': max_iter_per_beta,
            'ftol': 1e-14,
            'gtol': 1e-10,
            'maxls': 50,
            'maxfun': max_iter_per_beta * 5,
        }

        try:
            res = minimize(obj_fn, grad_fn, z_cur, method='L-BFGS-B',
                           jac=True, options=options)
            if res.success or res.fun < obj_fn(z_cur):
                a_new = _softmax(res.x)
                c_new = _fft_convolve(a_new)
                if c_new.max() < best_val:
                    best_val = float(c_new.max())
                    best_a = a_new.copy()
                    z = res.x.copy()
                    z -= float(np.mean(z))
        except Exception:
            pass

        # Update z to current best for next beta
        a_check = _softmax(z)
        c_check = _fft_convolve(a_check)
        if c_check.max() < best_val:
            best_val = float(c_check.max())
            best_a = a_check.copy()

    return best_a, best_val


def _subgradient_polish(a0, deadline, max_iter=20000, label=""):
    """Subgradient method for final polishing."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_val = float(c.max())

    base_lr = 0.5 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))

        # Subgradient
        sg = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        val_new = float(c_new.max())

        if val_new < best_val - 1e-14:
            best_val = val_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 4 * n):
                break

        a = a_new

    return best_a, best_val


def _multi_peak_subgradient(a0, deadline, max_iter=15000, top_k=5):
    """Subgradient targeting top-k peaks."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_val = float(c.max())

    base_lr = 0.3 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        weights = np.array([(1.0 + 0.5 * i) ** 2 for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            idx = int(idx)
            j1_start = max(0, idx - n + 1)
            j1_end = min(idx, n - 1)
            if j1_start <= j1_end:
                ii = np.arange(j1_start, j1_end + 1)
                sg[ii] += weights[i] * a[idx - ii]
            j2_start = max(idx, 0)
            j2_end = min(idx + n - 1, n - 1)
            if j2_start <= j2_end:
                ii = np.arange(j2_start, j2_end + 1)
                sg[ii] += weights[i] * a[ii - idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        val_new = float(c_new.max())

        if val_new < best_val - 1e-14:
            best_val = val_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_val


def _interpolate_seq(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _make_linear(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad(n, c1=1.0, c2=-0.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, c1=2.0, c3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + c1 * u + c3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_decay(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _monotone_optimize(a0, deadline, beta_init=0.005, total_betas=6,
                       max_iter=200, decreasing=True):
    """Optimize with monotone constraint using cumulative parameterization."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    # Convert to monotone: a[i] = sum_{j=i}^{n-1} d[j] (decreasing)
    # or a[i] = sum_{j=0}^{i} d[j] (increasing)
    if decreasing:
        d = np.diff(np.concatenate([a, [0]]))
    else:
        d = np.diff(np.concatenate([[0], a]))
    d = np.maximum(d, 1e-15)
    d = d / d.sum()

    def d_to_a(d_vec):
        if decreasing:
            a_vec = np.cumsum(d_vec[::-1])[::-1]
        else:
            a_vec = np.cumsum(d_vec)
        s = a_vec.sum()
        return a_vec / s if s > 1e-30 else np.ones(n) / n

    def a_to_grad_d(grad_a):
        if decreasing:
            return np.cumsum(grad_a[::-1])[::-1]
        else:
            return np.cumsum(grad_a)

    best_a = d_to_a(d)
    c = _fft_convolve(best_a)
    best_val = float(c.max())

    betas = [max(beta_init * (0.5 ** i), 1e-7) for i in range(total_betas)]

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        def obj(d_z):
            d_pos = _softmax(d_z)
            a_s = d_to_a(d_pos)
            c_s = _fft_convolve(a_s)
            return _smooth_max_value(c_s, beta)

        def grad(d_z):
            d_pos = _softmax(d_z)
            a_s = d_to_a(d_pos)
            c_s = _fft_convolve(a_s)
            w = _smooth_max_weights(c_s, beta)
            grad_a = _grad_conv_weighted(a_s, w)
            grad_d = a_to_grad_d(grad_a)
            dot = float(np.dot(d_pos, grad_d))
            grad_z = d_pos * (grad_d - dot)
            return grad_z

        z_d = np.log(d)
        z_d -= float(np.mean(z_d))

        try:
            res = minimize(obj, grad, z_d, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxls': 30})
            if res.fun < obj(z_d):
                d_new = _softmax(res.x)
                a_new = d_to_a(d_new)
                c_new = _fft_convolve(a_new)
                if c_new.max() < best_val:
                    best_val = float(c_new.max())
                    best_a = a_new.copy()
                z_d = res.x
                z_d -= float(np.mean(z_d))
        except Exception:
            pass

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: L-BFGS-B with smooth max, monotone constraints,
    adaptive beta annealing, and subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_seq(parent, 512)
        p1024 = _interpolate_seq(parent, 1024)
        p2048 = _interpolate_seq(parent, 2048)
    else:
        p512 = _make_linear(512, 2.0)
        p1024 = _make_linear(1024, 2.0)
        p2048 = _make_linear(2048, 2.0)

    def try_optimize(a0, n, d_limit, label):
        nonlocal best_score, best_seq
        if time.monotonic() > d_limit:
            return
        sc0 = _score_normalized(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
            print(f"  {label} init: {sc0:.10f}")

        # L-BFGS-B free optimization
        a1, v1 = _lbfgs_optimize(a0, d_limit, beta_init=0.003,
                                  total_betas=10, max_iter_per_beta=200,
                                  label=label)
        sc1 = _score_normalized(a1)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  {label} LBFGS: {sc1:.10f}")

        # Monotone optimization (decreasing)
        if time.monotonic() < d_limit - 30:
            a2, v2 = _monotone_optimize(a1, d_limit, beta_init=0.002,
                                         total_betas=6, max_iter=150,
                                         decreasing=True)
            sc2 = _score_normalized(a2)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
                print(f"  {label} mono_dec: {sc2:.10f}")

        # Subgradient polish
        if time.monotonic() < d_limit - 10:
            a3, v3 = _subgradient_polish(best_seq, d_limit,
                                          max_iter=20000)
            sc3 = _score_normalized(a3)
            if sc3 < best_score:
                best_score = sc3
                best_seq = a3.copy()
                print(f"  {label} subgrad: {sc3:.10f}")

    # === Phase 1: n=512 (fast, good for exploration) ===
    print("=== Phase 1: n=512 ===")
    d1 = min(deadline - 900, t_start + 300)

    # Parent interpolation
    try_optimize(p512, 512, d1, "S512-P")

    # Linear profiles
    if time.monotonic() < d1:
        for eta in [1.5, 2.5, 3.0]:
            a0 = _make_linear(512, eta)
            try_optimize(a0, 512, d1, f"S512-L{eta}")

    # Exponential decay
    if time.monotonic() < d1:
        for rate in [2.0, 4.0, 6.0]:
            a0 = _make_exp_decay(512, rate)
            try_optimize(a0, 512, d1, f"S512-E{rate}")

    # === Phase 2: n=1024 (main target) ===
    print(f"=== Phase 2: n=1024 (best so far: {best_score:.10f}) ===")
    d2 = min(deadline - 400, t_start + 900)

    try_optimize(p1024, 1024, d2, "S1024-P")

    if time.monotonic() < d2:
        # Best from 512, interpolated
        a_best_512 = _interpolate_seq(best_seq, 1024) if best_seq.size != 1024 else best_seq.copy()
        try_optimize(a_best_512, 1024, d2, "S1024-B512")

    if time.monotonic() < d2:
        for eta in [1.5, 2.0, 3.0, 4.0]:
            a0 = _make_linear(1024, eta)
            try_optimize(a0, 1024, d2, f"S1024-L{eta}")

    if time.monotonic() < d2:
        for rate in [2.0, 3.0, 5.0, 8.0]:
            a0 = _make_exp_decay(1024, rate)
            try_optimize(a0, 1024, d2, f"S1024-E{rate}")

    if time.monotonic() < d2:
        for c1 in [1.5, 2.5, 3.5]:
            a0 = _make_cubic(1024, c1, -c1 * 0.4)
            try_optimize(a0, 1024, d2, f"S1024-C{c1}")

    # === Phase 3: n=2048 (explore) ===
    print(f"=== Phase 3: n=2048 (best so far: {best_score:.10f}) ===")
    d3 = min(deadline - 200, t_start + 1500)

    if time.monotonic() < d3:
        a2048 = _interpolate_seq(best_seq, 2048)
        try_optimize(a2048, 2048, d3, "S2048-B")

    if time.monotonic() < d3:
        a2048 = _interpolate_seq(p1024, 2048)
        try_optimize(a2048, 2048, d3, "S2048-P")

    if time.monotonic() < d3:
        for eta in [2.0, 3.0]:
            a0 = _make_linear(2048, eta)
            try_optimize(a0, 2048, d3, f"S2048-L{eta}")

    # === Phase 4: Final deep optimization on best ===
    print(f"=== Phase 4: Final deep opt (best={best_score:.10f}, n={best_seq.size}) ===")
    d4 = deadline - 20

    if best_seq is not None:
        n_best = best_seq.size

        # Fine L-BFGS with very small beta
        a_f1, v_f1 = _lbfgs_optimize(best_seq, d4, beta_init=0.0005,
                                       total_betas=15, max_iter_per_beta=150)
        sc_f1 = _score_normalized(a_f1)
        if sc_f1 < best_score:
            best_score = sc_f1
            best_seq = a_f1.copy()
            print(f"  Final LBFGS fine: {best_score:.10f}")

        # Multi-peak subgradient
        if time.monotonic() < d4 - 30:
            a_f2, v_f2 = _multi_peak_subgradient(best_seq, d4,
                                                  max_iter=30000, top_k=5)
            sc_f2 = _score_normalized(a_f2)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_seq = a_f2.copy()
                print(f"  Final MultiSG: {best_score:.10f}")

        # Ultra-fine L-BFGS
        if time.monotonic() < d4 - 10:
            a_f3, v_f3 = _lbfgs_optimize(best_seq, d4, beta_init=0.0001,
                                           total_betas=20, max_iter_per_beta=100)
            sc_f3 = _score_normalized(a_f3)
            if sc_f3 < best_score:
                best_score = sc_f3
                best_seq = a_f3.copy()
                print(f"  Final ULBFGS: {best_score:.10f}")

        # Final subgradient
        if time.monotonic() < d4 - 2:
            a_f4, v_f4 = _subgradient_polish(best_seq, d4, max_iter=50000)
            sc_f4 = _score_normalized(a_f4)
            if sc_f4 < best_score:
                best_score = sc_f4
                best_seq = a_f4.copy()
                print(f"  Final SG: {best_score:.10f}")

    # === Try different length of best answer ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        for target_n in [512, 1024, 2048]:
            if best_seq.size == target_n:
                continue
            if time.monotonic() > deadline - 5:
                break
            a_t = _interpolate_seq(best_seq, target_n)
            a_t, v_t = _lbfgs_optimize(a_t, deadline - 5, beta_init=0.001,
                                         total_betas=8, max_iter_per_beta=100)
            sc_t = _score_normalized(a_t)
            if sc_t < best_score:
                best_score = sc_t
                best_seq = a_t.copy()
                print(f"  Length {target_n}: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear(512, 2.0)
        best_score = _score_normalized(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear(256, 2.0)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
