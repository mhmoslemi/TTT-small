def solve_lp_subproblem(a, M):
    """
    Solve: maximize sum(b) subject to (a*b)_k <= M for all k, b >= 0.
    Returns the optimal b.
    """
    n = len(a)
    # (a*b)_k = sum_j a[k-j] * b_j for j where 0 <= k-j < n
    # = sum_j a[k-j] * b_j
    
    # c^T b (maximize) => minimize -c^T b where c = ones
    c = -np.ones(n)
    
    # Constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # A_ub @ b <= b_ub
    # Row k: sum_j A[k,j] * b_j <= M
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Build the constraint matrix (sparse would be better but let's try dense for now)
    # For n=2048, this is a 4095 x 2048 matrix, which is about 8M entries - manageable
    
    A_ub = np.zeros((2*n-1, n))

        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    
    b_ub = np.full(2*n-1, M)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=(0, None))

        return result.x
    else:
        return None

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding (SOTA), edge-emphasizing analytical profiles,
adaptive active set management, and gradient polishing in logit space.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def convolve_with(a, b):
    """Compute a * b (linear convolution) using FFT."""
    n = len(a)
    m = len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    c = np.maximum(c, 0.0)
    return c


def soft_max_weights(c, beta):
    """Compute softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def make_edge_profile(n, alpha=0.5):
    """a_i proportional to 1/sqrt((i+1)(n-i))^alpha"""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power((i + 1.0) * (n - i), alpha)
    # Avoid division by zero at edges (i=0 gives n, i=n-1 gives 1)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_cosine_edge_profile(n, power=2.0):
    """a_i proportional to (1 + cos(pi*i/(n-1)))^power / (2 - cos(pi*i/(n-1)))^power"""
    t = np.linspace(0.0, 1.0, n)
    a = np.power(1.0 + np.cos(np.pi * t), power) / np.power(2.0 - np.cos(np.pi * t), 0.5)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_inverse_sqrt_profile(n):
    """a_i proportional to 1/sqrt(i+1) * 1/sqrt(n-i)"""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_power_edge_profile(n, p=0.3):
    """a_i proportional to (i+1)^(-p) * (n-i)^(-p)"""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(i * (n - i + 1.0), -p)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_symmetric_edge_profile(n, sharpness=1.0):
    """Symmetric profile emphasizing edges: a_i = (1 + sharpness * |2i/(n-1) - 1|^2)"""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / max(n - 1, 1) - 1.0
    a = 1.0 + sharpness * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def solve_lp_direction(a, M, active_indices):
    """
    Solve: maximize sum(b) subject to (a*b)[k] <= M for k in active_indices, b >= 0.
    Returns optimal b or None.
    """
    n = len(a)
    # Build constraint matrix: row k has a[k-j] at column j (for valid j)
    # Only build rows for active_indices
    K = len(active_indices)
    if K == 0:
        return None
    
    # For each active index k, the constraint is sum_j a[k-j] * b_j <= M
    # where a[k-j] is defined only for 0 <= k-j < n
    
    A_rows = []
    for k in active_indices:
        row = np.zeros(n)
        # a[k-j] is non-zero when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            row[j_start:j_end+1] = a[j_end - (j_end - j_start):j_end+1][::-1]
            # Wait, let me be more careful
            # For j in [j_start, j_end]: a[k-j]
            # k-j ranges from k-j_start to k-j_end
            # k - j_start to k - j_end should be in [0, n-1]
            indices = k - np.arange(j_start, j_end + 1)
            row[j_start:j_end+1] = a[indices]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(K, M)
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                         bounds=[(0, None)] * n,
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_refine(a0, deadline, rng, max_lp_iters=500, active_frac=0.05,
              blend_rate=0.1, min_improvement=1e-8):
    """
    Refine using LP-based direction finding.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    c = full_convolve_fft(a)
    best_score = float(2.0 * n * c.max())  # Since sum(a)=1, score = 2n*max(c)
    
    no_improve = 0
    lp_count = 0
    grad_count = 0
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with noise
            a = best_a.copy()
            a = a + 0.001 * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            continue
        
        # Determine active set
        threshold = max_c * (1.0 - active_frac)
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]
        
        # If too many active, reduce
        max_active = max(10, min(2 * n, len(active_indices)))
        if len(active_indices) > max_active:
            # Keep only the largest
            top_vals = c[active_indices]
            order = np.argsort(top_vals)[::-1][:max_active]
            active_indices = active_indices[order]
        
        # Solve LP
        b = solve_lp_direction(a, max_c, active_indices)
        
        if b is not None and float(b.sum()) > 1.0 + min_improvement:
            b_norm = b / b.sum()
            # Blend
            a = (1.0 - blend_rate) * a + blend_rate * b_norm
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            lp_count += 1
        else:
            # Fall back to gradient step
            # Soft-max gradient
            beta = 0.01 * max_c
            _, w = soft_max_weights(c, beta)
            # h[i] = sum_k w[k] * a[k-i] (correlation of w with a)
            # This is conv(w, a_reversed) shifted
            # Actually: grad of soft_max(conv(a,a)) w.r.t. a[i] = 2 * sum_k w[k] * a[k-i]
            # where a[k-i] is defined as 0 if k-i not in [0,n-1]
            # This is the cross-correlation of w and a
            # h = correlate(w, a) = conv(w, a[::-1])[n-1:2n-1]
            n_w = len(w)
            N_fft = next_pow2(n_w + n - 1)
            fw = np.fft.rfft(w, N_fft)
            fa_rev = np.fft.rfft(a[::-1], N_fft)
            h = np.fft.irfft(fw * fa_rev, N_fft)[:n_w + n - 1]
            h = h[n - 1:2 * n - 1]  # correlation
            
            grad_a = 2.0 * h
            # Project to simplex (logit update)
            dot = float(np.dot(a, grad_a))
            step = a * (grad_a - dot)
            step = -0.01 * step / (np.max(np.abs(step)) + 1e-15)
            a = a + step
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            grad_count += 1
    
    return best_a, best_score / (1.0**2)  # score = 2n*max_c since sum=1


def logit_adam_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.05,
                        beta_decay=0.93, beta_min=1e-10, restart_period=1000):
    """
    Optimize in logit space using Adam with soft-max approximation.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())

    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)

        # Gradient in logit space
        n_w = len(w)
        N_fft = next_pow2(n_w + n - 1)
        fw = np.fft.rfft(w, N_fft)
        fa_rev = np.fft.rfft(a0[::-1], N_fft)
        h = np.fft.irfft(fw * fa_rev, N_fft)[:n_w + n - 1]
        h = h[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 200:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.05 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def diverse_profiles(n, rng):

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, edge-emphasizing analytical profiles,
adaptive active set management, and gradient polishing in logit space.
Key insight: edge-emphasized profiles have flatter autocorrelation than uniform.
"""
import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def cross_correlate_fft(w, a):
    """Compute h[i] = sum_k w[k] * a[k-i] for i=0..n-1 (circular-like cross-corr)."""
    n = len(a)
    n_w = len(w)
    N_fft = next_pow2(n_w + n - 1)
    fw = np.fft.rfft(w, N_fft)
    fa_rev = np.fft.rfft(a[::-1], N_fft)
    conv = np.fft.irfft(fw * fa_rev, N_fft)[:n_w + n - 1]
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def make_edge_profile(n, alpha=0.5):
    i = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(i * (n - i + 1.0), -alpha)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_symmetric_edge_profile(n, sharpness=1.0):
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / max(n - 1, 1) - 1.0
    a = 1.0 + sharpness * np.power(np.abs(t), 1.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_arcsin_profile(n, power=0.5):
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = 1.0 / np.power(np.sin(np.pi * t), power) + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bimodal_profile(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.1) / width)**2) + np.exp(-((t - 0.9) / width)**2) + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def solve_lp_direction(a, M, active_indices):
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return None
    A_rows = []
    for k in active_indices:
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            row[js] = a[k - js]
        A_rows.append(row)
    A_ub = np.array(A_rows)
    b_ub = np.full(K, M)
    c_obj = -np.ones(n)
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                         bounds=[(0, None)] * n,
                         options={'presolve': True, 'time_limit': 5.0})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_refine(a0, deadline, rng, active_frac=0.03, blend_rate=0.08,
              max_active_frac=0.1):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = full_convolve_fft(a)
    best_score = float(2.0 * n * c.max())
    no_improve = 0
    lp_fails = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            lp_fails = 0
        else:
            no_improve += 1

        if no_improve > 300:
            a = best_a + 0.002 * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            continue

        threshold = max_c * (1.0 - active_frac)
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]

        max_active = max(20, int(max_active_frac * (2 * n - 1)))
        if len(active_indices) > max_active:
            top_vals = c[active_indices]
            order = np.argsort(top_vals)[::-1][:max_active]
            active_indices = active_indices[order]

        b = solve_lp_direction(a, max_c, active_indices)

        if b is not None and float(b.sum()) > 1.0 + 1e-6:
            b_norm = b / b.sum()
            a = (1.0 - blend_rate) * a + blend_rate * b_norm
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            lp_fails += 1
            # Gradient fallback
            beta = max(0.005 * max_c, 1e-10)
            _, w = soft_max_weights(c, beta)
            h = cross_correlate_fft(w, a)
            grad_a = 2.0 * h
            dot = float(np.dot(a, grad_a))
            step = a * (grad_a - dot)
            step_size = 0.02 / (np.max(np.abs(step)) + 1e-15)
            a = a - step_size * step
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

            if lp_fails > 50:
                active_frac = max(0.005, active_frac * 0.7)
                blend_rate = min(0.3, blend_rate * 1.1)
                lp_fails = 0

    return best_a, best_score


def logit_adam_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.05,
                        beta_decay=0.93, beta_min=1e-10, restart_period=800):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = max(beta_factor * max_c, 1e-8)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = cross_correlate_fft(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)
        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.97)
        if no_improve > restart_period:
            z = best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.05 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score

            dl = min(phase3_dl, time.monotonic() + 120.0)
            r, sc_r = logit_adam_optimize(p0, dl, rng, lr=lr_v, beta_factor=bf, beta_decay=0.94)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Round {round_i} (lr={lr_v}): {sc_f:.10f}")

        # Peak targeting
        if time.monotonic() < phase3_dl - 30:
            dl_pt = min(phase3_dl, time.monotonic() + 80.0)
            r_pt, _ = peak_targeting_refine(best_seq, dl_pt, rng, n_peaks=5, lr=0.012)
            sc_pt = compute_score(r_pt)
            update(r_pt, sc_pt)
            print(f"  Peak targeting: {sc_pt:.10f}")

    # Try different n
    if best_seq is not None:
        for n_try in [best_seq.size // 2, best_seq.size * 2]:
            if time.monotonic() > phase3_dl - 20:
                break
            if n_try < 128 or n_try > 4096:
                continue
            p_small = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_small)
            update(p_small, sc_s)
            if sc_s < best_score + 0.02:
                dl_s = min(phase3_dl, time.monotonic() + 50.0)
                r_s, _ = logit_adam_optimize(p_small, dl_s, rng, lr=0.02, beta_factor=0.03)
                sc_sf = compute_score(r_s)
                update(r_s, sc_sf)
                print(f"  n={n_try}: {sc_sf:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Final polish (remaining time) ===
    print(f"\n--- Phase 4: Final Polish ---")

    if best_seq is not None:
        for attempt in range(3):
            if time.monotonic() > deadline - 10:
                break
            p0 = best_seq + 0.0005 * rng.normal(size=best_seq.size)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            dl = min(deadline, time.monotonic() + 80.0)
            r, _ = logit_adam_optimize(p0, dl, rng, lr=0.008, beta_factor=0.015, beta_decay=0.96)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Final attempt {attempt}: {sc_f:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
    else:
        final_seq = make_edge_profile(512, 0.5)

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
