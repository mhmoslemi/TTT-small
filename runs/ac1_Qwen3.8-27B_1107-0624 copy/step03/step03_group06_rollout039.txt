"""
Hybrid LP + Subgradient + Local Search optimizer for minimizing
2n * max(conv(a,a)) / sum(a)^2 over non-negative sequences.
Uses asymmetric initializations, LP-based refinement, subgradient polishing,
and pairwise local search to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 200:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_ramp(n, alpha=1.0, direction='right'):
    t = np.linspace(0.0, 1.0, n)
    if direction == 'right':
        a = t ** alpha + 1e-8
    else:
        a = (1.0 - t) ** alpha + 1e-8
    return a / a.sum()


def _make_bell(n, center=0.3, width=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2) + 1e-8
    return a / a.sum()


def _make_two_peak(n, c1=0.25, c2=0.75, w=0.12):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2) + 1e-8
    return a / a.sum()


def _make_strong_ramp(n, alpha=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_linear(n, slope=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_sinusoidal(n, freq=1.0, phase=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.8 * np.sin(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_random_smooth(n, rng, n_basis=20):
    x = np.linspace(0, 1, n_basis)
    y = rng.random(n_basis)
    y = np.maximum(y, 0.05)
    t = np.linspace(0.0, 1.0, n)
    a = np.interp(t, x, y)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _solve_lp(a, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _subgradient_polish(a0, deadline, max_iter=200000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    lr = 0.015
    base_lr = lr
    momentum = 0.75
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))

        # Compute subgradient: grad[j] = 2*a[k_star - j] for valid j
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        j_range = np.arange(lo, hi + 1)
        grad[j_range] = 2.0 * a[k_star - j_range]

        # Momentum update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        # Evaluate
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
                # Try a smaller momentum
                momentum = max(0.3, momentum * 0.9)

        # Periodically reduce learning rate
        if it % 5000 == 4999:
            lr = max(0.002, lr * 0.85)

    return best_a

    best_score = _score(a)
    rng = np.random.default_rng(777)

    step_mult = 1.0
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        j = int(rng.integers(n))
        jp = int(rng.integers(n - 1))
        if jp >= j:
            jp += 1

        base = max(1e-12, min(a[j], a[jp], 0.1))
        multipliers = step_mult * np.array([-1.0, -0.6, -0.3, -0.15, 0.15, 0.3, 0.6, 1.0])
        deltas = base * multipliers

        improved = False
        for d in deltas:
            new_j = a[j] + d
            new_jp = a[jp] - d
            if new_j < -1e-15 or new_jp < -1e-15:
                continue
            new_j = max(0.0, new_j)
            new_jp = max(0.0, new_jp)

            a[j] = new_j
            a[jp] = new_jp
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
                break
            a = a / s

            sc = _score(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                improved = True
            else:
                a = best_a.copy()

        if not improved:
            no_imp += 1
            if no_imp % 500 == 0:
                step_mult *= 0.75
                if step_mult < 0.05:
                    break
        else:
            no_imp = 0
            # Occasionally increase step size back
            if no_imp == 0 and it % 10000 == 9999:
                step_mult = min(1.0, step_mult * 1.2)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP + Subgradient + Local Search optimizer.
    Phase 1: Subgradient on diverse asymmetric initializations.
    Phase 2: LP refinement (main workhorse).
    Phase 3: Subgradient polish.
    Phase 4: Local pairwise search.
    Phase 5: Final LP + polish.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s_val = p.sum()
            if s_val > 0.01:
                parent = p / s_val
                sc_p = _score(parent)
                if np.isfinite(sc_p) and sc_p < best_score:
                    best_score = sc_p
                    best_seq = parent.copy()
                print(f"Parent: score={sc_p:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Generate diverse initializations
    inits = []
    lengths_to_try = [512, 640, 768, 896, 1024, 1152]

    for n_t in lengths_to_try:
        if parent is not None:
            inits.append(('parent', n_t, _interpolate_seq(parent, n_t)))
        inits.append(('ramp1', n_t, _make_ramp(n_t, 1.0, 'right')))
        inits.append(('ramp2', n_t, _make_ramp(n_t, 2.0, 'right')))
        inits.append(('ramp3', n_t, _make_ramp(n_t, 3.0, 'right')))
        inits.append(('rampL', n_t, _make_ramp(n_t, 2.0, 'left')))
        inits.append(('bell30', n_t, _make_bell(n_t, 0.3, 0.22)))
        inits.append(('bell70', n_t, _make_bell(n_t, 0.7, 0.22)))
        inits.append(('twopeak', n_t, _make_two_peak(n_t)))
        inits.append(('lin3', n_t, _make_linear(n_t, 3.0)))
        inits.append(('lin5', n_t, _make_linear(n_t, 5.0)))
        inits.append(('sin1', n_t, _make_sinusoidal(n_t, 1.0, 0.5)))
        inits.append(('sin2', n_t, _make_sinusoidal(n_t, 2.0, 1.0)))
        inits.append(('rand', n_t, _make_random_smooth(n_t, rng)))

    # Score all and sort
    scored_inits = []
    for name, n_t, a0 in inits:
        sc = _score(a0)
        if np.isfinite(sc):
            scored_inits.append((sc, name, n_t, a0))
    scored_inits.sort(key=lambda x: x[0])

    print(f"Generated {len(scored_inits)} initializations", flush=True)
    for sc, name, n_t, _ in scored_inits[:5]:
        print(f"  {name:8s} n={n_t:4d} score={sc:.8f}", flush=True)

    # Phase 1: Subgradient polish on top initializations (fast screening)
    # Budget: ~300s
    p1_end = t0 + 300.0
    top_k = min(10, len(scored_inits))
    for i in range(top_k):
        if time.monotonic() > p1_end:
            break
        sc0, name, n_t, a0 = scored_inits[i]
        if sc0 >= best_score:
            continue
        # Allocate time per initialization
        remaining = p1_end - time.monotonic()
        per_init = min(40.0, remaining / max(1, top_k - i))
        if per_init < 5:
            break
        end_i = time.monotonic() + per_init
        a_pg = _subgradient_polish(a0, end_i, max_iter=30000)
        sc_pg = _score(a_pg)
        if np.isfinite(sc_pg) and sc_pg < best_score:
            best_score = sc_pg
            best_seq = a_pg.copy()
            print(f"P1 {name} n={n_t}: {sc_pg:.10f}", flush=True)

    # Phase 2: LP Refinement (main workhorse) - Budget: ~600s
    p2_end = min(t0 + 900.0, deadline - 350.0)
    if best_seq is not None:
        print(f"P2 LP: starting score={best_score:.10f}, n={best_seq.size}", flush=True)
        a_lp = _lp_refine(best_seq, p2_end, max_iter=50000, inflate_start=1.004)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"P2 LP done: score={sc_lp:.10f}", flush=True)

    # Try LP on a second-best initialization if time allows
    p2b_end = min(t0 + 1100.0, deadline - 250.0)
    if time.monotonic() < p2b_end:
        # Find the best initialization that's different from current best length
        for sc0, name, n_t, a0 in scored_inits[:5]:
            if time.monotonic() > p2b_end:
                break
            if n_t == (best_seq.size if best_seq is not None else -1):
                continue
            per_try = min(60.0, p2b_end - time.monotonic())
            if per_try < 10:
                break
            a_lp2 = _lp_refine(a0, time.monotonic() + per_try, max_iter=20000, inflate_start=1.005)
            sc2 = _score(a_lp2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_lp2.copy()
                print(f"P2b {name} n={n_t}: {sc2:.10f}", flush=True)

    # Phase 3: Subgradient polish - Budget: ~200s
    p3_end = min(t0 + 1300.0, deadline - 180.0)
    if best_seq is not None and time.monotonic() < p3_end:
        per_time = min(150.0, p3_end - time.monotonic())
        if per_time > 5:
            a_sg = _subgradient_polish(best_seq, time.monotonic() + per_time, max_iter=100000)
            sc_sg = _score(a_sg)
            if np.isfinite(sc_sg) and sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
            print(f"P3 Subgrad: score={sc_sg:.10f}", flush=True)

    # Phase 4: Local pairwise search - Budget: ~200s
    p4_end = min(t0 + 1500.0, deadline - 100.0)
    if best_seq is not None and time.monotonic() < p4_end:
        per_time = min(180.0, p4_end - time.monotonic())
        if per_time > 5:
            a_ls = _local_search(best_seq, time.monotonic() + per_time, max_iters=100000)
            sc_ls = _score(a_ls)
            if np.isfinite(sc_ls) and sc_ls < best_score:
                best_score = sc_ls
                best_seq = a_ls.copy()
            print(f"P4 LocalSearch: score={sc_ls:.10f}", flush=True)

    # Phase 5: Final LP pass - Budget: ~80s
    p5_end = min(t0 + 1580.0, deadline - 30.0)
    if best_seq is not None and time.monotonic() < p5_end:
        per_time = min(70.0, p5_end - time.monotonic())
        if per_time > 5:
            a_final = _lp_refine(best_seq, time.monotonic() + per_time,
                                 max_iter=10000, inflate_start=1.002)
            sc_f = _score(a_final)
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_final.copy()
            print(f"P5 FinalLP: score={sc_f:.10f}", flush=True)

    # Phase 6: Quick final polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        per_time = min(30.0, deadline - time.monotonic() - 5)
        if per_time > 3:
            a_b = _local_search(best_seq, time.monotonic() + per_time, max_iters=30000)
            sc_b = _score(a_b)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_seq = a_b.copy()
            print(f"P6 Polish: score={sc_b:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(768, 1.5, 'right')
        best_score = _score(best_seq)

    # Final validation and cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final < 0.01:
        final_seq = _make_ramp(768, 1.5, 'right')

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}", flush=True)

    # Validate with evaluate_sequence if possible
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
    except Exception as e:
        print(f"Eval check: {e}", flush=True)

    return [float(x) for x in final_seq]
