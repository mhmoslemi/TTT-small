def _next_pow2(n):

"""
Focused gradient-based optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key strategies: (1) FFT-based fast convolution, (2) log-domain Adam with
adaptive smooth-max targeting, (3) spectral-domain flattening moves,
(4) targeted peak reduction, (5) multi-start with structured inits,
(6) no expensive LP calls - pure gradient/spectral methods.
"""

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-10:
        c = c - mn
    return c


def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_u_shape(n, alpha=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2 * t - 1) ** 2
    return a / a.sum()


def _make_bimodal(n, sigma=0.15, edge=0.05):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - edge) / sigma) ** 2) + np.exp(-0.5 * ((t - 1 + edge) / sigma) ** 2) + 0.05
    return a / a.sum()


def _make_linear_v(n, eta=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * np.abs(2 * t - 1)
    return a / a.sum()


def _make_parabola_inv(n, alpha=3.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (1.0 - (2 * t - 1) ** 2)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _compute_smooth_max_weight(c, beta):
    mval = c.max()
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return ex / s


def _gradient_smooth_max(a, c, w):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_log_optimize(a, deadline, max_iter=100000, beta_scale=0.04, lr_init=0.06,
                       decay_beta_every=80, beta_decay=0.82, restart_every=2500):
    """Adam optimizer in log-domain with adaptive smooth-max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _fft_conv(a)
    M = c.max()
    beta = max(1e-8, beta_scale * M)

    best_a = a.copy()
    best_score = 2.0 * n * M

    no_imp = 0
    lr = lr_init
    rng = np.random.default_rng(42)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        w = _compute_smooth_max_weight(c, beta)
        grad = _gradient_smooth_max(a, c, w)
        grad += 1e-9 * a

        dot = np.dot(a, grad)
        gz = a * (grad - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(a)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()

        e = np.exp(z)
        a = e / e.sum()

        c = _fft_conv(a)
        M = c.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.002, lr * 0.97)
            if no_imp > restart_every:
                # Perturb and restart
                scale = 0.2 + 0.1 * min(5, no_imp // restart_every)
                z_pert = z + scale * rng.normal(size=n)
                z_pert -= z_pert.mean()
                e_p = np.exp(np.clip(z_pert, -60, 60))
                a = e_p / e_p.sum()
                c = _fft_conv(a)
                M = c.max()
                sc = 2.0 * n * M
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                z = np.log(a)
                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                lr = lr_init
                no_imp = 0
                beta = max(1e-8, beta_scale * M)

        if it % decay_beta_every == 0:
            beta = max(1e-10, beta * beta_decay)

    return best_a, best_score


def _spectral_flatten(a, n_peaks=None, strength=0.4):
    """Spectral-domain flattening move."""
    n = len(a)
    if n_peaks is None:
        n_peaks = max(5, n // 10)

    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a_n = a / s

    c = _fft_conv(a_n)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    # Blend target with current (reduce peaks, boost valleys)
    c_blend = (1 - strength) * c + strength * target

    N = _next_pow2(2 * n - 1)
    fc = np.fft.rfft(c_blend, N)
    # Complex square root with Hermitian constraint
    fa = np.zeros(N // 2 + 1, dtype=np.complex128)
    for k in range(len(fc)):
        fa[k] = np.sqrt(fc[k] + 0j)
    # Enforce Hermitian: A[N-k] = conj(A[k])
    a_full = np.fft.irfft(fa, N)
    a_cand = a_full[:n]

    a_cand = np.maximum(a_cand, 0)
    sc = a_cand.sum()
    if sc < 1e-10:
        return a

    a_new = (1 - strength) * a_n + strength * (a_cand / sc)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-10:
        return a
    a_new = a_new / sn * s
    return a_new


def _targeted_peak_reduction(a, n_top=5, n_lags=3):
    """Directly reduce contributions to top peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a_n = a / s

    c = _fft_conv(a_n)
    top_lags = np.argsort(c)[::-1][:n_lags]

    # For each top lag, reduce the largest contributing pairs
    a_new = a_n.copy()
    for k in top_lags:
        # Contributions: a[i] * a[k-i] for valid i
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        contribs = []
        for i in range(i_min, i_max + 1):
            j = k - i
            contribs.append((a_new[i] * a_new[j], i, j))
        contribs.sort(reverse=True)
        # Reduce top contributors
        for _, i, j in contribs[:n_top]:
            reduction = 0.1 * a_new[i]
            a_new[i] = max(0, a_new[i] - reduction)

    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-10:
        return a
    a_new = a_new / sn * s
    return a_new


def _multi_perturb_optimize(a, deadline, n_perturbs=5, perturb_scale=0.15,

        if time.monotonic() > deadline - 30:
            break
        sub_dl = time.monotonic() + min(per_perturb_budget, deadline - time.monotonic() - 20)
        if sub_dl <= time.monotonic():
            break

        # Random perturbation
        a_pert = a * (1 + perturb_scale * rng.normal(size=n))
        a_pert = np.maximum(a_pert, 0)
        sp = a_pert.sum()
        if sp < 1e-10:
            continue
        a_pert = a_pert / sp * a.sum()

        sc_p = _score(a_pert)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_pert.copy()

        a_opt, sc_opt = _adam_log_optimize(a_pert, sub_dl, max_iter=5000, lr_init=0.05)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()

    return best_a, best_score

        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  Phase 4 done: {sc4:.10f}")

        # Also try structured inits
        inits = [
            _make_u_shape(n, 1.5),
            _make_u_shape(n, 3.0),
            _make_bimodal(n, 0.12),
            _make_parabola_inv(n, 4.0),
            _make_linear_v(n, 1.5),
        ]
        per_init = min(40.0, (phase4_dl - time.monotonic()) / (len(inits) + 1))
        for a0 in inits:
            if time.monotonic() > phase4_dl - 40:
                break
            sub_dl = time.monotonic() + per_init
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            a_i, sc_i = _adam_log_optimize(a0, sub_dl, max_iter=3000, lr_init=0.05)
            if sc_i < best_score:
                best_score = sc_i
                best_seq = a_i.copy()
                print(f"  Structured init -> {sc_i:.10f}")

    # Phase 5: Final intensive polish (15%)
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 5: Final polish")
        # Round 1: spectral + Adam
        a_f = _spectral_flatten(best_seq, n_peaks=15, strength=0.3)
        sc_f = _score(a_f)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()

        sub_dl = time.monotonic() + min(80.0, (deadline - time.monotonic()) * 0.4)
        a_f2, sc_f2 = _adam_log_optimize(best_seq, sub_dl, max_iter=20000,
                                          beta_scale=0.02, lr_init=0.03)
        if sc_f2 < best_score:
            best_score = sc_f2
            best_seq = a_f2.copy()
            print(f"  Final round 1: {sc_f2:.10f}")

        # Round 2: targeted + Adam
        if time.monotonic() < deadline - 40:
            a_t = _targeted_peak_reduction(best_seq, n_top=5, n_lags=3)
            sc_t = _score(a_t)
            if sc_t < best_score:
                best_score = sc_t
                best_seq = a_t.copy()

            sub_dl2 = time.monotonic() + min(60.0, (deadline - time.monotonic()) * 0.4)
            a_f3, sc_f3 = _adam_log_optimize(best_seq, sub_dl2, max_iter=15000,
                                              beta_scale=0.015, lr_init=0.025)
            if sc_f3 < best_score:
                best_score = sc_f3
                best_seq = a_f3.copy()
                print(f"  Final round 2: {sc_f3:.10f}")

        # Round 3: small Adam
        if time.monotonic() < deadline - 20:
            sub_dl3 = deadline - 10
            a_f4, sc_f4 = _adam_log_optimize(best_seq, sub_dl3, max_iter=10000,
                                              beta_scale=0.01, lr_init=0.02)
            if sc_f4 < best_score:
                best_score = sc_f4
                best_seq = a_f4.copy()
                print(f"  Final round 3: {sc_f4:.10f}")

    # Final validation
    final_seq = np.maximum(best_seq, 0.0).astype(np.float64)
    s = final_seq.sum()
    if s < 0.01:
        final_seq = _make_u_shape(768, 2.0)

    # Scale to ensure sum is in a good range (score is scale-invariant,
    # but evaluate_sequence clips to [0,1000])
    if s > 1000:
        final_seq = final_seq * (500.0 / s)

    sc_final = _score(final_seq)
    print(f"\nInternal final score: {sc_final:.10f}, n={final_seq.size}")

    # Validate with evaluate_sequence
    try:
        sc_eval = evaluate_sequence(final_seq.tolist())
        print(f"evaluate_sequence score: {sc_eval:.10f}")
        if not np.isfinite(sc_eval):
            print("WARNING: evaluate_sequence returned inf, falling back")
            final_seq = _make_u_shape(768, 2.0)
    except Exception as e:
        print(f"Evaluation error: {e}")

    return [float(x) for x in final_seq]
