"""
Improved step function optimization using:
1. L-BFGS on log-sum-exp smooth max with softmax parametrization
2. Multi-scale growth (256 -> 512 -> 1024 -> 2048)
3. Peak-aware subgradient with backtracking line search
4. LP-based mass spreading for local refinement
5. Multiple sequence lengths and diverse initializations
6. Deterministic annealing (temperature schedule for log-sum-exp)
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_power2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_power2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _lse_max(c, temp):
    m = c.max()
    if temp < 1e-12:
        return m, np.zeros_like(c), 1.0
    x = (c - m) / temp
    x = np.clip(x, -200.0, 200.0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return m, w, 0.0
    w = ex / s
    val = m + temp * np.log(s)
    return val, w, temp


def _grad_conv_weighted(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses FFT: grad = 2 * (a * w_rev)[n-1:2n-1][::-1]
    where w_rev = w reversed, and * is linear convolution.
    """
    n = len(a)
    conv_len = 2 * n - 1
    w_rev = w[::-1].copy()
    N = _next_power2(n + conv_len - 1)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:conv_len] = w_rev
    fa = np.fft.rfft(a_pad, N)
    fw = np.fft.rfft(w_pad, N)
    result = np.fft.irfft(fa * fw, N)
    h = result[n - 1:2 * n - 1][::-1]
    return 2.0 * h


def _score_and_grad_lse(a, temp):
    """Compute LSE-smoothed score and its gradient w.r.t. a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a_norm = a / s
    c = _fft_convolve(a_norm)
    val, w, t_eff = _lse_max(c, temp)
    grad_w = _grad_conv_weighted(a_norm, w)
    # Chain rule: d/d a of (2*n*val) where val depends on a_norm = a/s
    # d(a_norm)/d a = (I - a_norm * 1^T) / s
    # So d(val)/d a = grad_w^T (I - a_norm*1^T)/s + val_correction
    # But since we're optimizing with sum=1 constraint, we work in the tangent space
    # Project gradient to sum=0 subspace
    grad = grad_w - np.mean(grad_w)
    return 2.0 * n * val, grad


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _obj_lse(z, temp):
    """Objective in softmax space: 2*n*LSE_max(conv(softmax(z), softmax(z)))"""
    a = _softmax(z)
    c = _fft_convolve(a)
    val, _, _ = _lse_max(c, temp)
    n = len(a)
    return 2.0 * n * val


def _obj_grad_lse(z, temp):
    """Objective and gradient in softmax space."""
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a)
    val, w, _ = _lse_max(c, temp)
    grad_w = _grad_conv_weighted(a, w)
    # Gradient w.r.t. z: dL/dz = a * (dL/da - a^T dL/da)
    # where dL/da = 2*n*grad_w
    dL_da = 2.0 * n * grad_w
    dL_dz = a * (dL_da - np.dot(a, dL_da))
    return 2.0 * n * val, dL_dz


def _anneal_lse(z0, deadline, temp_init=0.5, temp_min=0.001, max_iter=50000):
    """Deterministic annealing with L-BFGS on log-sum-exp in softmax space."""
    n = len(z0)
    z = z0.copy()
    best_score = _score(_softmax(z))
    best_z = z.copy()
    temp = temp_init

    it = 0
    stall_count = 0
    last_best = best_score

    while time.monotonic() < deadline:
        if it >= max_iter:
            break

        def func(z_eval):
            return _obj_lse(z_eval, temp)

        def func_and_grad(z_eval):
            return _obj_grad_lse(z_eval, temp)

        try:
            res = minimize(func_and_grad, z, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'gtol': 1e-8,
                                             'maxfun': 300})
        except Exception:
            break

        z_new = res.x
        a_new = _softmax(z_new)
        s_new = _fft_convolve(a_new).max()
        score_new = 2.0 * n * s_new  # since sum=1

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_z = z_new.copy()
            stall_count = 0
            last_best = best_score
        else:
            stall_count += 1

        # Anneal temperature
        if stall_count > 50:
            temp = max(temp_min, temp * 0.7)
            # Reset from best
            z = best_z + 0.1 * np.random.default_rng(it).normal(size=n)
            stall_count = 0
        elif it % 100 == 0:
            temp = max(temp_min, temp * 0.95)

        z = z_new
        it += 1

        if it % 5000 == 0:
            print(f"        Anneal it={it}, temp={temp:.4f}, best={best_score:.8f}")

    # Final refinement at very low temp
    if time.monotonic() < deadline - 5:
        try:
            def func_final(z_eval):
                return _obj_lse(z_eval, temp_min)

            res = minimize(func_final, best_z, method='L-BFGS-B',
                           jac=True, options={'maxiter': 100, 'gtol': 1e-10})
            a_final = _softmax(res.x)
            score_final = 2.0 * n * _fft_convolve(a_final).max()
            if score_final < best_score:
                best_score = score_final
                best_z = res.x.copy()
        except Exception:
            pass

    return best_z, best_score


def _subgradient_peak_shave(a, deadline, max_iter=5000, top_k=5):
    """Peak-aware subgradient with adaptive step size and top-k peaks."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * _fft_convolve(a).max()
    no_imp = 0
    base_lr = 1.0 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = c.max()

        # Get top-k peaks
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        # Weights: emphasize the max more
        weights = np.exp(-0.5 * (c[top_idx] - c[top_idx[0]]) / max(c_max * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Combined gradient
        grad = np.zeros(n)
        for i, idx in enumerate(top_idx):
            grad += weights[i] * _grad_conv_weighted(a, _one_hot(idx, len(c)))

        # Project to sum=0 and normalize
        grad -= grad.mean()
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Backtracking line search
        lr = base_lr
        a_new = a
        c_max_new = c_max
        for _ls in range(10):
            a_try = a - lr * grad / g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                lr *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try)
            c_max_try = c_try.max()
            if c_max_try < c_max_new * 1.0001:
                a_new = a_try
                c_max_new = c_max_try
                break
            lr *= 0.5

        score_new = 2.0 * n * c_max_new
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(200, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _one_hot(idx, length):
    w = np.zeros(length)
    w[idx] = 1.0
    return w


def _lp_spread(a, M, n_constraints=None):
    """LP: maximize sum(g) s.t. conv(a, g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: A[j, i] = a[j-i] if valid
    # conv(a, g)[k] = sum_i a[i] * g[k-i] <= M
    # For each k: sum_i a[i] * g[k-i] <= M
    # Variable is g[j], so coefficient of g[j] in row k is a[k-j] (if 0<=k-j<n)
    n_cons = 2 * n - 1
    if n_constraints is not None and n_constraints < n_cons:

            no_imp += 1
            if no_imp % 15 == 0:
                inflate = min(1.01, inflate * 1.04)
            if no_imp > 80:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        sc_new = 2.0 * n * _fft_convolve(a_new).max()

        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend = max(0.08, blend * 0.92)
                inflate = max(1.001, inflate * 0.97)
            if no_imp > 80:
                a = best_a.copy()
                no_imp = 0
                blend = 0.3
                inflate = 1.003

        a = a_new

    return best_a, best_sc


def _interp(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _pipeline(a0, deadline, tag=""):
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = len(a0)
    a0 = a0 / a0.sum()
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 -= z0.mean()

    best_a = a0.copy()
    best_sc = 2.0 * n * _fft_convolve(a0).max()
    if np.isfinite(best_sc):
        print(f"      {tag} init={best_sc:.8f}")

    # Phase 1: Annealed L-BFGS
    if time.monotonic() < deadline - 20:
        d1 = min(deadline - 20, time.monotonic() + 0.45 * (deadline - time.monotonic()))
        z1, s1 = _anneal_lse(z0, d1, temp_init=max(0.1, 5.0 / n), temp_min=0.0008)
        a1 = _softmax(z1)
        if s1 < best_sc:
            best_sc, best_a = s1, a1.copy()
        print(f"      {tag} LSE={best_sc:.8f}")

    # Phase 2: Peak subgradient
    if time.monotonic() < deadline - 15:
        d2 = min(deadline - 15, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        a2, s2 = _peak_subgradient(best_a, d2, max_iter=2500, top_k=5)
        if s2 < best_sc:
            best_sc, best_a = s2, a2.copy()
        print(f"      {tag} PkSG={best_sc:.8f}")

    # Phase 3: LP
    if time.monotonic() < deadline - 12:
        d3 = min(deadline - 12, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        a3, s3 = _lp_refine(best_a, d3, max_iter=200, n_cons=35)
        if s3 < best_sc:
            best_sc, best_a = s3, a3.copy()
        print(f"      {tag} LP={best_sc:.8f}")

    # Phase 4: Final L-BFGS
    if time.monotonic() < deadline - 5:
        d4 = deadline - 5
        z4 = np.log(np.maximum(best_a, 1e-15))
        z4 -= z4.mean()
        z4f, s4 = _anneal_lse(z4, d4, temp_init=0.005, temp_min=0.0003, max_iter=8000)
        a4 = _softmax(z4f)
        if s4 < best_sc:
            best_sc, best_a = s4, a4.copy()
        print(f"      {tag} LSE2={best_sc:.8f}")

    # Phase 5: Final peak shave
    if time.monotonic() < deadline - 2:
        d5 = deadline - 2
        a5, s5 = _peak_subgradient(best_a, d5, max_iter=1500, top_k=3)
        if s5 < best_sc:
            best_sc, best_a = s5, a5.copy()
        print(f"      {tag} FinSG={best_sc:.8f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimization using L-BFGS on log-sum-exp in softmax space,
    peak subgradient, and LP refinement. Explores multiple sequence lengths
    starting from parent. Time-budgeted to complete within budget_s seconds.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 30.0
    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.8f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths
    if parent is not None:
        pn = parent.size
        candidates = [512, 768, 1024, 1536, 2048]
        if pn in candidates:
            candidates.remove(pn)
            candidates.insert(0, pn)
    else:
        candidates = [512, 1024, 2048]

    # Sort by priority: parent first, then by expected quality
    # Allocate time: larger n gets more time but we need to try multiple
    remaining = deadline - time.monotonic()
    n_cand = len(candidates)

    # Time allocation: first gets 40%, rest split
    times = []
    for i, ln in enumerate(candidates):
        if i == 0:
            times.append(0.35 * remaining)
        else:
            times.append(0.65 * remaining / (n_cand - 1))
    # Cap individual times
    for i in range(len(times)):
        times[i] = min(times[i], remaining * 0.45)

    rng = np.random.default_rng(seed % (2**31 - 1))

    for idx, n_t in enumerate(candidates):
        if time.monotonic() > deadline - 50:
            break

        t_n = time.monotonic()
        n_dead = min(deadline - 30, t_n + times[idx])
        if n_dead - t_n < 30:
            continue

        print(f"\n--- n={n_t} (budget={times[idx]:.0f}s) ---")

        # Create initializations
        inits = []
        if parent is not None:
            inits.append(_interp(parent, n_t))
            # Perturbed
            pp = _interp(parent, n_t)
            pp = pp * (1.0 + 0.08 * rng.normal(size=n_t))
            pp = np.maximum(pp, 0.0)
            inits.append(pp / pp.sum())
        else:
            inits.append(np.ones(n_t) / n_t)

        # Add structure
        t = np.linspace(0, 1, n_t)
        a_lin = 1.0 + 1.5 * (t - 0.5)
        a_lin = np.maximum(a_lin, 0.01)
        inits.append(a_lin / a_lin.sum())

        a_quad = 1.0 + 1.2 * (1.0 - 2.0 * t) ** 2
        inits.append(a_quad / a_quad.sum())

        # Limit number of inits based on time
        per_init = (n_dead - time.monotonic()) / max(1, len(inits))
        max_inits = max(1, int((n_dead - time.monotonic()) / 25.0))

        for ii in range(min(max_inits, len(inits))):
            if time.monotonic() > n_dead - 10:
                break
            d_init = min(n_dead - 5, time.monotonic() + per_init)
            if d_init - time.monotonic() < 15:
                break
            a_r, s_r = _pipeline(inits[ii], d_init, tag=f"i{ii}")
            if s_r < best_score:
                best_score = s_r
                best_seq = a_r.copy()
                print(f"    ** best={best_score:.8f} n={n_t}")

    # Final global focus
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n--- Final focus: score={best_score:.8f}, n={best_seq.size} ---")
        d_fin = deadline - 10
        a_f, s_f = _pipeline(best_seq, d_fin, tag="FIN")
        if s_f < best_score:
            best_score = s_f
            best_seq = a_f.copy()

        # Try 2x if small enough
        if best_seq.size <= 1024 and time.monotonic() < d_fin - 60:
            n2 = best_seq.size * 2
            a_2 = _interp(best_seq, n2)
            d_2 = min(d_fin - 10, time.monotonic() + 80)
            if d_2 - time.monotonic() > 20:
                a_2r, s_2r = _pipeline(a_2, d_2, tag=f"2x_{n2}")
                if s_2r < best_score:
                    best_score = s_2r
                    best_seq = a_2r.copy()

    # Prepare output
    if best_seq is None:
        best_seq = np.ones(512) / 512

    final = np.clip(best_seq, 0.0, 1000.0)
    s = final.sum()
    if s < 0.01:
        final = np.ones(512) / 512
    else:
        final = final / s

    try:
        sc = float(evaluate_sequence(final.tolist()))
        print(f"\nevaluate_sequence: {sc:.8f}")
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.8f}, n={final.size}")
    return [float(x) for x in final]
