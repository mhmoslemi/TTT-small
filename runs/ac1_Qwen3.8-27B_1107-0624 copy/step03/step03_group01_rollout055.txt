"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses palindromic symmetry, FFT-based convolution, log-sum-exp smoothing,
continuation method, multi-length exploration, and peak-directed refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score_of(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses palindromic symmetry for 2x speedup, FFT-based convolution,
log-sum-exp continuation, multi-length exploration (2048/3072/4096),
peak-directed gradient refinement, and Adam polishing in log-space.
"""
import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score_norm(a):
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def grad_smooth_max(a, beta, n):
    c = full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * a * (h - W)


def obj_full(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def grad_full(z, beta, n):
    a = softmax_fn(z)
    g = grad_smooth_max(a, beta, n)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g


def obj_palin(z, beta, n):
    m = (n + 1) // 2
    p = softmax_fn(z)
    a = np.zeros(n)
    a[:m] = p
    if n % 2 == 0:
        a[m:] = p[::-1]
    else:
        a[m:] = p[:-1][::-1]
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def grad_palin(z, beta, n):
    m = (n + 1) // 2
    p = softmax_fn(z)
    a = np.zeros(n)
    a[:m] = p
    if n % 2 == 0:
        a[m:] = p[::-1]
    else:
        a[m:] = p[:-1][::-1]
    s = a.sum()
    if s < 1e-15:
        return np.zeros(m)
    a = a / s
    g = grad_smooth_max(a, beta, n)
    if not np.all(np.isfinite(g)):
        return np.zeros(m)
    gp = np.zeros(m)
    if n % 2 == 0:
        gp[:] = g[:m] + g[m::-1]
    else:
        gp[:m] = g[:m]
        gp[m-1] = g[m-1]
        for j in range(m - 1):
            gp[j] += g[n - 1 - j]
    return gp


def run_continuation(z_init, n, deadline, rng, use_palin=False):
    m = (n + 1) // 2 if use_palin else n
    z = z_init.copy()
    best_a = softmax_fn(z)
    if use_palin:
        a_tmp = np.zeros(n)
        a_tmp[:m] = best_a
        if n % 2 == 0:
            a_tmp[m:] = best_a[::-1]
        else:
            a_tmp[m:] = best_a[:-1][::-1]
        s = a_tmp.sum()
        best_a = a_tmp / s if s > 1e-15 else a_tmp
    else:
        best_a = best_a / best_a.sum()
    best_score = score_norm(best_a)

    c0 = full_convolve(best_a, best_a)
    max_c0 = float(c0.max())
    beta_starts = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    remaining = deadline - time.monotonic()
    if remaining <= 0:
        return best_a, best_score
    per_level = remaining / len(betas)

    for li, beta in enumerate(betas):
        if time.monotonic() > deadline - 1:
            break
        level_dl = min(deadline, time.monotonic() + per_level * 1.5)
        if use_palin:
            f_obj = obj_palin
            f_grad = grad_palin
        else:
            f_obj = obj_full
            f_grad = grad_full
        try:
            res = minimize(f_obj, z, args=(beta, n), method='L-BFGS-B', jac=f_grad,
                           options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 40, 'maxfun': 3000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_new = res.x
                if use_palin:
                    p_new = softmax_fn(z_new)
                    a_new = np.zeros(n)
                    a_new[:m] = p_new
                    if n % 2 == 0:
                        a_new[m:] = p_new[::-1]
                    else:
                        a_new[m:] = p_new[:-1][::-1]
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a_new = a_new / s_new
                else:
                    a_new = softmax_fn(z_new)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a_new = a_new / s_new
                sc = score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    return best_a, best_score


def adam_polish(a_init, deadline, rng, lr=0.05, beta_frac=0.005, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    a = a_init.copy()
    c = full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max(max_c, 1e-10))

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        g = grad_smooth_max(a, beta, n)
        m_vec = b1 * m_vec + (1 - b1) * g
        v_vec = b2 * v_vec + (1 - b2) * (g * g)
        mh = m_vec / (1 - b1 ** t)
        vh = v_vec / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = softmax_fn(z)
        sc = score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 300 == 0:
            beta = max(1e-14, beta * 0.82)
        if t % 5000 == 0 and best_score < 1.51:
            lr = max(0.003, lr * 0.93)
    return best_a, best_score


def peak_directed_polish(a_init, deadline, rng, n_steps=20000, lr_init=0.5):
    n = len(a_init)
    m = (n + 1) // 2
    p = np.zeros(m)
    p[:m] = a_init[:m]
    if n % 2 == 0:
        p[:] = 0.5 * (a_init[:m] + a_init[m::-1])
    else:
        p[:m-1] = 0.5 * (a_init[:m-1] + a_init[n-1:0-1:-1])
        p[m-1] = a_init[m-1]
    p = np.maximum(p, 1e-12)
    p = p / p.sum()

    z = np.log(p)
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    lr = lr_init

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        p_cur = softmax_fn(z)
        a_cur = np.zeros(n)
        a_cur[:m] = p_cur
        if n % 2 == 0:
            a_cur[m:] = p_cur[::-1]
        else:
            a_cur[m:] = p_cur[:-1][::-1]
        s = a_cur.sum()
        if s < 1e-15:
            break
        a_cur = a_cur / s
        c = full_convolve(a_cur, a_cur)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        # Gradient of c[peak_idx] w.r.t. a
        g_a = 2.0 * a_cur  # simplified: dc_k/da_i = 2*a_{k-i}, evaluated at peak
        # More precisely: g_a[i] = 2*a[peak_idx - i] (valid indices)
        g_a = np.zeros(n)
        for i in range(n):
            k_i = peak_idx - i
            if 0 <= k_i < n:
                g_a[i] = 2.0 * a_cur[k_i]
            else:
                g_a[i] = 0.0
        # Normalize gradient by peak_val for scale invariance
        if peak_val > 1e-15:
            g_a = g_a / peak_val
        # Fold gradient to p-space
        gp = np.zeros(m)
        if n % 2 == 0:
            gp[:] = g_a[:m] + g_a[m::-1]
        else:
            gp[:m-1] = g_a[:m-1] + g_a[n-1:0-1:-1]
            gp[m-1] = g_a[m-1]
        z = z - lr * gp
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        # Score check every 100 steps
        if step % 100 == 0:
            p_chk = softmax_fn(z)
            a_chk = np.zeros(n)
            a_chk[:m] = p_chk
            if n % 2 == 0:
                a_chk[m:] = p_chk[::-1]
            else:
                a_chk[m:] = p_chk[:-1][::-1]
            sc = score_norm(a_chk / a_chk.sum())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk / a_chk.sum()
                lr = min(lr * 1.02, 0.3)
            elif step > 0:
                lr *= 0.98
    return best_a, best_score


def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        per_dl = min(phase5_end - 10, time.monotonic() + 220.0)
        a_p, sc_p = adam_polish(best_seq, per_dl, rng, lr=0.04, beta_frac=0.003, n_max_iter=40000)
        update_best(a_p, sc_p, "adam_polish")

    # Phase 6: Peak-directed polish (150s)
    phase6_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase6_end - 10 and best_seq is not None:
        per_dl = min(phase6_end - 5, time.monotonic() + 130.0)
        a_pd, sc_pd = peak_directed_polish(best_seq, per_dl, rng, n_steps=30000, lr_init=0.3)
        update_best(a_pd, sc_pd, "peak_directed")

    # Phase 7: Final fine polish (remaining time - 30s)
    phase7_end = deadline - 30.0
    if time.monotonic() < phase7_end and best_seq is not None:
        n_f = len(best_seq)
        per_dl1 = min(phase7_end - 20, time.monotonic() + 60.0)
        a_f, sc_f = run_continuation(
            np.log(np.maximum(best_seq, 1e-15)) - np.log(np.maximum(best_seq, 1e-15)).mean(),
            n_f, per_dl1, rng, use_palin=False)
        update_best(a_f, sc_f, "final_lbfgs")

        if time.monotonic() < phase7_end - 5:
            per_dl2 = min(phase7_end - 3, time.monotonic() + 30.0)
            a_f2, sc_f2 = adam_polish(best_seq, per_dl2, rng, lr=0.01, beta_frac=0.001, n_max_iter=10000)
            update_best(a_f2, sc_f2, "final_adam")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256) / 256
        sc_final = score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")
        return [float(x) for x in final_seq]

    # Fallback
    fb = np.ones(256) / 256
    print(f"FALLBACK: n=256, score={score_fn(fb):.10f}")
    return [float(x) for x in fb]
