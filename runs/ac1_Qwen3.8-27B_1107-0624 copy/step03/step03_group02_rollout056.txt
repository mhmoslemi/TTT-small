- If i = k-i, i.e., k=2i, these are the same term, so derivative = 2*a[i] (from d/dx(x^2) = 2x)
- If i ≠ k-i, they're different terms, so derivative = a[k-i] + a[k-i] = 2*a[k-i]

Wait, that can't be right either. Let's check: c[0] = a[0]*a[0]. dc[0]/da[0] should be 2*a[0]. Using the formula: k=0, i=0, k-i=0, which is in [0,n), so dc[0]/da[0] = 2*a[0]. ✓

And grad[i] = sum_k w[k] * 2 * a[k-i] * (0 <= k-i < n)

Let j = k-i: grad[i] = 2 * sum_j w[i+j] * a[j] * (0 <= i+j < 2n-1 and 0 <= j < n)

This is a cross-correlation! grad[i] = 2 * (a ⋆ w)[i] where (a ⋆ w)[i] = sum_j a[j] * w[i+j].

"""
Optimized step function coefficient search using FFT-based gradient methods.
Primary approach: Adam optimizer in logit-space with adaptive smooth-max,
run at n=2048 with diverse initializations and peak-targeting auxiliary steps.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s

def _grad_convolution(a, w):
    """Compute gradient of sum_k w[k]*c[k] where c = conv(a,a).
    grad[i] = 2 * sum_j a[j] * w[i+j]
    Computed as 2 * conv(a, w_rev)[n-1:2n-1][::-1]
    """
    n = len(a)
    n_w = len(w)  # should be 2n-1
    # Need conv(a, w_rev) of length n + n_w - 1 = 3n - 2
    N = _next_pow2(n + n_w - 1)
    w_rev = w[::-1].copy()
    a_fft = np.fft.rfft(a, N)
    w_rev_fft = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(a_fft * w_rev_fft, N)
    # grad[i] = 2 * conv[2*n-2-i] for i=0..n-1
    # So grad = 2 * conv[n-1:2n-1][::-1]
    idx_start = n - 1
    idx_end = 2 * n - 1  # exclusive
    grad = 2.0 * conv[idx_start:idx_end][::-1]
    return grad

def _adam_optimize(a, deadline, max_iter=50000, lr=0.05, beta_start=0.08, 
                   beta_decay=0.95, beta_min=0.001, restart_every=3000,
                   verbose=False):

    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    
    beta = max(beta_start * float(c.max()), 1e-12)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_convolution(a, w)
        
        # Project to tangent space (maintain sum=1)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m, v, t = np.zeros(n), np.zeros(n), 0
            a = best_a.copy()
            continue
        
        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * grad_z * grad_z
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        
        e = np.exp(z_new)
        a_new = e / e.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        z = z_new
        a = a_new
        
        # Decay beta

            beta = max(beta_min * float(c_curr.max()), beta * beta_decay)

            rng_r = np.random.default_rng(7919 + it)
            z = best_z + 0.5 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)

            e = np.exp(z)
            a = e / e.sum()
            beta = max(0.02, 0.05 * float(_fft_convolve(a).max()))
            m, v, t = np.zeros(n), np.zeros(n), 0
            no_imp = 0
        
        if verbose and it % 5000 == 0:
            print(f"    Adam it={it}: score={score_new:.10f}, best={best_score:.10f}, beta={beta:.2e}")
    
    return best_a, best_score

def _peak_targeting(a, deadline, max_iter=5000, lr_base=0.1, top_k=5):
    """Directly target the top-k peaks using subgradient descent."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        # Get top-k peak positions
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c_curr)[::-1][:k]
        
        # Weighted subgradient
        weights = np.exp(-0.5 * np.arange(k))
        weights /= weights.sum()
        
        # For each peak at position p, the subgradient direction is:
        # dc[p]/da[i] = 2*a[p-i] (if 0 <= p-i < n)
        # We want to DECREASE c[p], so we move in direction -dc[p]/da
        # But we need to maintain sum(a)=1, so project.
        
        sg = np.zeros(n)
        for wi, p in zip(weights, top_idx):
            p = int(p)
            i_start = max(0, p - n + 1)
            i_end = min(n - 1, p)
            if i_start <= i_end:
                sg[i_start:i_end+1] += wi * 2.0 * a[p - np.arange(i_start, i_end+1)]
        
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break
        
        lr = lr_base / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break
        
        a = a_new
    
    return best_a, best_score

def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a / a.sum()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return result / s

"""
FFT-based Adam optimization for step function coefficient minimization.
Uses n=2048 as primary length with correct convolution gradient, adaptive 
smooth-max, peak-targeting, and diverse initializations. Exploits the structure
of the autocorrelation to efficiently minimize the peak-to-sum ratio.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _grad_convolution(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a).
    grad[i] = 2 * sum_j a[j]*w[i+j], computed via FFT.
    """
    n = len(a)
    n_w = len(w)
    N = _next_pow2(n + n_w - 1)
    w_rev = w[::-1].copy()
    a_fft = np.fft.rfft(a, N)
    w_rev_fft = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(a_fft * w_rev_fft, N)
    idx_start = n - 1
    idx_end = 2 * n - 1
    grad = 2.0 * conv[idx_start:idx_end][::-1]
    return grad


def _adam_optimize(a, deadline, max_iter=50000, lr=0.05, beta_start=0.08,
                   beta_decay=0.96, beta_min=0.001, restart_patience=2500,
                   verbose=False):
    """Adam in logit-space minimizing smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_start * float(c.max()), 1e-12)
    no_imp = 0
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_convolution(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m, v, t = np.zeros(n), np.zeros(n), 0
            a = best_a.copy()
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * grad_z * grad_z
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0 and no_imp > 500:
                lr_cur = max(0.005, lr_cur * 0.92)

        z = z_new
        a = a_new

        if it % 80 == 0:
            beta = max(beta_min * float(c_curr.max()), beta * beta_decay)

        if no_imp > restart_patience:
            seed_r = 100000 + (it % 10000)
            rng_r = np.random.default_rng(seed_r)
            z = best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(0.03, 0.06 * float(_fft_convolve(a).max()))
            m, v, t = np.zeros(n), np.zeros(n), 0
            no_imp = 0
            lr_cur = lr

        if verbose and it % 5000 == 0:
            print(f"      Adam it={it}: cur={score_new:.10f}, best={best_score:.10f}, beta={beta:.2e}, lr={lr_cur:.4f}")

    return best_a, best_score


def _peak_targeting(a, deadline, max_iter=5000, lr_base=0.15, top_k=5):
    """Subgradient descent targeting top-k peaks simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c_curr)[::-1][:k]

        weights = np.exp(-0.4 * np.arange(k))
        weights /= weights.sum()

        sg = np.zeros(n)
        for wi, p in zip(weights, top_idx):
            p = int(p)
            i_start = max(0, p - n + 1)
            i_end = min(n - 1, p)
            if i_start <= i_end:
                indices = np.arange(i_start, i_end + 1)
                sg[i_start:i_end + 1] += wi * 2.0 * a[p - indices]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = lr_base / np.sqrt(float(it + 1))
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

        a = a_new

    return best_a, best_score


def _make_profile(n, mode):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if mode == 'flat':
        a = np.ones(n)
    elif mode == 'w_shape':
        a = 1.0 + 0.6 * np.cos(4.0 * np.pi * t)
    elif mode == 'm_shape':
        a = 1.0 + 0.4 * np.cos(6.0 * np.pi * t)
    elif mode == 'polynomial':
        a = 1.0 + 0.5 * u ** 2 - 0.4 * u ** 4
    elif mode == 'sine':
        a = 1.0 + 0.6 * np.sin(3.0 * np.pi * t)
    elif mode == 'triangle':
        a = 1.0 + 0.5 * (1.0 - 2.0 * np.abs(u))
    elif mode == 'edge':
        a = 1.0 + 0.8 * u ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return result / s


def _hybrid_step(a, deadline, max_iter=3000):
    """Alternate between Adam-like steps and peak targeting."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = 0.04 * float(c.max())
    no_imp = 0
    lr = 0.04

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        if it % 3 == 2:
            # Peak targeting step
            c_curr = _fft_convolve(a)
            top_idx = np.argsort(c_curr)[::-1][:3]
            sg = np.zeros(n)
            for p in top_idx:
                p = int(p)
                i_start = max(0, p - n + 1)
                i_end = min(n - 1, p)
                if i_start <= i_end:
                    indices = np.arange(i_start, i_end + 1)
                    sg[i_start:i_end + 1] += 2.0 * a[p - indices]
            sg_norm = float(np.linalg.norm(sg))
            if sg_norm > 1e-20:
                lr_pt = 0.05 / np.sqrt(float(it + 1))
                a_new = a - lr_pt * sg / sg_norm
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                a = a_new
                z = np.log(a)
                z -= float(np.mean(z))
        else:
            # Adam step
            c_curr = _fft_convolve(a)
            w = _smooth_max_weights(c_curr, beta)
            grad_a = _grad_convolution(a, w)
            dot = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - dot)
            grad_z -= float(np.mean(grad_z))

            if np.all(np.isfinite(grad_z)):
                t += 1
                m = b1 * m + (1.0 - b1) * grad_z
                v = b2 * v + (1.0 - b2) * grad_z * grad_z
                mh = m / (1.0 - b1 ** t)
                vh = v / (1.0 - b2 ** t)
                z_new = z - lr * mh / (np.sqrt(vh) + eps)
                z_new = np.clip(z_new, -50.0, 50.0)
                z_new -= float(np.mean(z_new))
                e = np.exp(z_new)
                a = e / e.sum()
                z = z_new

        c_curr = _fft_convolve(a)
        score_curr = float(2.0 * n * c_curr.max())

        if score_curr < best_score - 1e-14:
            best_score = score_curr
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 100 == 0:
            beta = max(0.002, beta * 0.97)

        if no_imp > 2000:
            rng_r = np.random.default_rng(50000 + it)
            z = best_z + 0.3 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(0.02, 0.05 * float(_fft_convolve(a).max()))
            m, v, t = np.zeros(n), np.zeros(n), 0
            no_imp = 0

    return best_a, best_score
