"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses logsumexp-smoothed gradient descent with adaptive beta continuation,
hybrid Adam/L-BFGS, diverse initializations, and multi-scale refinement.
Key insight: non-palindromic sequences can achieve score < 2 because
conv(a,a) peak need not equal sum(a_i^2) for asymmetric a.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_smooth_max_and_grad(z, beta, n):
    a = _softmax_fn(z)
    a_rev = a[::-1]
    c = _full_convolve(a, a)
    c_scaled = c / beta
    c_max = np.max(c_scaled)
    e = np.exp(c_scaled - c_max)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c_scaled))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    M = beta * (c_max + float(np.log(s)))
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * h - 2.0 * W
    grad_z = 2.0 * n * a * (grad_a - float(np.dot(grad_a, a)))
    return M, grad_z, a, c, w


def _lbfgs_objective(z, beta, n):
    M, _, _, _, _ = _compute_smooth_max_and_grad(z, beta, n)
    return 2.0 * n * M


def _lbfgs_gradient(z, beta, n):
    _, grad_z, _, _, _ = _compute_smooth_max_and_grad(z, beta, n)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _run_lbfgs_phase(a_init, deadline, beta_start_frac, n_beta_steps):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c0 = _full_convolve(a_init, a_init)
    best_score = float(2.0 * n * c0.max())
    max_c0 = float(c0.max())

    beta_factors = np.geomspace(beta_start_frac, 0.0001, n_beta_steps)

    for bf in beta_factors:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-10, bf * max_c0)
        level_time = max(5.0, min(80.0, (deadline - time.monotonic()) / max(1, n_beta_steps - int(np.searchsorted(-np.log(beta_factors), -np.log(bf), side='right')))))

        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01, momentum_b1=0.9, momentum_b2=0.999):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c0 = _full_convolve(a_init, a_init)
    best_score = float(2.0 * n * c0.max())
    max_c0 = float(c0.max())
    beta = max(1e-10, beta_frac * max_c0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    eps = 1e-8

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        a_rev = a[::-1]
        c = _full_convolve(a, a)
        c_scaled = (c - c.max()) / beta
        e = np.exp(c_scaled)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = momentum_b1 * m + (1 - momentum_b1) * grad
        v = momentum_b2 * v + (1 - momentum_b2) * (grad * grad)
        mh = m / (1 - momentum_b1 ** t)
        vh = v / (1 - momentum_b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            beta = max(1e-12, beta * 0.8)
        if t % 2000 == 0:
            lr = max(0.003, lr * 0.95)

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)

    a = np.maximum(a, 1e-12)

def _make_asymmetric_profile(n, shift_frac, width):

    center = 0.5 + shift_frac * 0.3
    a = np.exp(-(t - center) ** 2 / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)

def _make_periodic_profile(n, n_periods, asymmetric=False):

    else:
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * n_periods * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        update_best(p, sc, f"prof{idx}_n{n_base}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: Optimize top candidates with combined Adam+L-BFGS (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    top_k = min(5, len(candidates))
    per_candidate = (phase1_end - time.monotonic()) / max(1, top_k)

    for i in range(top_k):
        if time.monotonic() > phase1_end:
            break
        sc_init, p_init, label = candidates[i]
        if sc_init > best_score * 1.02:
            break
        per_dl = min(phase1_end - 5, time.monotonic() + per_candidate)
        a_best, sc_best = _combined_optimize(p_init, per_dl, rng)
        update_best(a_best, sc_best, f"combined_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Multi-restart from best (400s) ---
    phase2_end = min(deadline, t_start + 830.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 6
        per_restart = min(65.0, (phase2_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase2_end - 10:
                break
            if i < 2:
                p_start = _anti_palindromic_perturb(best_seq, rng, scale=0.15)
                label = f"anti_pal{i}"
            elif i < 4:
                p_start = _perturb_sequence(best_seq, rng, scale=0.2 + 0.1 * i)
                label = f"perturb{i}"
            else:
                p_start = _perturb_sequence(best_seq, rng, scale=0.4)
                label = f"big_perturb{i}"

            per_dl = min(phase2_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _combined_optimize(p_start, per_dl, rng)
            update_best(a_best, sc_best, label)

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Try larger n by interpolation (300s) ---
    phase3_end = min(deadline, t_start + 1130.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        n_curr = len(best_seq)
        for target_n in [n_curr * 2, n_curr * 2, n_curr * 3]:
            if time.monotonic() > phase3_end:
                break
            if target_n > 8192:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            per_dl = min(phase3_end - 30, time.monotonic() + 120.0)
            a_best, sc_best = _combined_optimize(p_interp, per_dl, rng)
            update_best(a_best, sc_best, f"opt_n{target_n}")
            if target_n == n_curr * 2:
                break

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Intensive L-BFGS polish on best (200s) ---
    phase4_end = min(deadline, t_start + 1330.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 5, time.monotonic() + 150.0)
        a_best, sc_best = _run_lbfgs_phase(best_seq, per_dl, beta_start_frac=0.001, n_beta_steps=12)
        update_best(a_best, sc_best, "polish_lbfgs")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final Adam fine-tune (remaining time) ---
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per_dl = min(deadline - 10, time.monotonic() + 120.0)
        a_best, sc_best = _adam_optimize(best_seq, per_dl, lr=0.01, beta_frac=0.002)
        update_best(a_best, sc_best, "final_adam")

    # --- Phase 6: One more L-BFGS pass with very small beta ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per_dl = min(deadline - 5, time.monotonic() + 60.0)
        a_best, sc_best = _run_lbfgs_phase(best_seq, per_dl, beta_start_frac=0.0003, n_beta_steps=6)
        update_best(a_best, sc_best, "final_lbfgs_tiny")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_linear_profile(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
