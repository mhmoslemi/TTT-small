def _smooth_max_grad(c, beta):
    """Compute softmax weights for smooth max gradient."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:

        ex[int(np.argmax(c))] = 1.0

def _gradient(a, beta, top_k=0, peak_weight=0.0):
    """Compute gradient of smooth max of conv(a,a) with optional peak awareness."""
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_grad(c, beta)
    
    if top_k > 0 and peak_weight > 0:
        sorted_idx = np.argsort(c)[::-1]
        w_peak = np.zeros_like(c)
        peak_vals = c[sorted_idx[:top_k]]
        ps = peak_vals.sum()
        if ps > 0:
            w_peak[sorted_idx[:top_k]] = peak_vals / ps
        w = (1 - peak_weight) * w + peak_weight * w_peak
        w = w / w.sum()
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c

def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def _logit_lbfgs_polish(a, deadline, max_iter=10000):
    """L-BFGS polish using log-parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z -= float(np.mean(z))

    best_score = float(2.0 * n * c.max())

        z_vec = np.asarray(z_vec, dtype=np.float64)
        z_vec = np.clip(z_vec, -50.0, 50.0)
        z_vec -= float(np.mean(z_vec))
        e = np.exp(z_vec)
        a_tmp = e / e.sum()
        if not np.all(np.isfinite(a_tmp)):
            return 1e10
        c_tmp = _fft_convolve(a_tmp)
        return float(c_tmp.max())
    
    def gradient(z_vec):
        z_vec = np.asarray(z_vec, dtype=np.float64)
        z_vec = np.clip(z_vec, -50.0, 50.0)
        z_vec -= float(np.mean(z_vec))
        e = np.exp(z_vec)
        a_tmp = e / e.sum()
        c_tmp = _fft_convolve(a_tmp)
        w = _smooth_max_grad(c_tmp, beta)

        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(a_tmp, vp))
        gz = a_tmp * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            return np.zeros(n)
        return gz
    
    # Run L-BFGS with periodic beta updates
    lbfgs_iter = 0
    no_imp = 0
    
    while lbfgs_iter < max_iter and time.monotonic() < deadline:
        res = minimize(objective, z, method='L-BFGS-B', jac=gradient,
                      options={'maxiter': min(200, max_iter - lbfgs_iter),
                               'ftol': 1e-12, 'gtol': 1e-8,
                               'maxls': 20})

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()

            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5:
                break
        
        lbfgs_iter += int(res.nit) if hasattr(res, 'nit') else 1
        beta = max(1e-10, beta * 0.8)
    
    return best_a

def _block_perturb(a, rng, max_iter=3000, deadline=None):
    """Random block perturbations to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Random perturbation: transfer mass between random positions
        n_transfers = rng.integers(1, max(2, n // 100) + 1)
        a_try = a.copy()
        for _ in range(n_transfers):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.2)
            delta = frac * a_try[i]
            if delta > a_try[i]:
                continue
            a_try[i] -= delta
            a_try[j] += delta
        
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        sc = _score(a_try)
        # Accept with some probability (simulated annealing)
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
            no_imp = 0
        else:
            # Accept with probability exp(-(sc - best_score) / T)
            T = 0.001
            if rng.random() < np.exp(-(sc - best_score) / T):
                a = a_try.copy()
            no_imp += 1
        
        if no_imp > 2000:
            break
    
    return best_a

    best_peak_w = peak_w

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0.0)
        sn = a_nest.sum()
        if sn < 1e-10:
            a_nest = a.copy()
        else:
            a_nest = a_nest / sn

        grad, maxc, c_arr = _compute_gradient(a_nest, beta, top_k=peak_k, peak_weight=peak_w)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.004, lr * 0.94)
                peak_w = max(0.05, peak_w * 0.97)
            if no_imp > max(800, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, beta_scale * float(_fft_convolve(a).max()))
                lr = base_lr
                peak_w = best_peak_w
                no_imp = 0

        if it % 250 == 0:
            beta = max(1e-10, beta * 0.7)

        if it % 50 == 0 and score < best_score - 1e-14:
            best_peak_w = peak_w

    return best_a


def _logit_lbfgs_polish(a, deadline, max_iter_total=200):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.06 * float(c.max()))
    total_iters = 0
    lbfgs_calls = 0

    while total_iters < max_iter_total and lbfgs_calls < 30 and time.monotonic() < deadline:
        def obj(zv):
            zv = np.asarray(zv, dtype=np.float64)
            zv = np.clip(zv, -40.0, 40.0)
            zv = np.delete(zv, np.argmax(zv))
            e = np.exp(zv)
            at = e / e.sum()
            if not np.all(np.isfinite(at)):
                return 1e10
            ct = _fft_convolve(at)
            return float(ct.max())

        def grad_fn(zv):
            zv = np.asarray(zv, dtype=np.float64)
            zv = np.clip(zv, -40.0, 40.0)
            zv = np.delete(zv, np.argmax(zv))
            e = np.exp(zv)
            at = e / e.sum()
            ct = _fft_convolve(at)
            w = _smooth_max_weights(ct, beta)
            N = _next_pow2(2 * n - 1)
            fw = np.fft.rfft(w, N)
            fa_rev = np.fft.rfft(at[::-1], N)
            h_full = np.fft.irfft(fw * fa_rev, N)
            h = h_full[n - 1:2 * n - 1]
            vp = 2.0 * h
            dot = float(np.dot(at, vp))
            gz = at * (vp - dot)
            gz -= float(np.mean(gz))
            if not np.all(np.isfinite(gz)):
                return np.zeros(n - 1)
            return gz

        # Work in (n-1) dimensional space (drop one coordinate for the simplex constraint)
        z_work = np.delete(z, np.argmax(z))

        res = minimize(obj, z_work, method='L-BFGS-B', jac=grad_fn,
                       options={'maxiter': min(50, max_iter_total - total_iters + 1),
                                'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30})

        z_work = res.x
        z_work = np.clip(z_work, -40.0, 40.0)
        z_new = np.insert(z_work, int(np.argmax(z)), float(np.max(z)))
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            z = z_new.copy()
        else:
            break

        total_iters += 50
        lbfgs_calls += 1
        beta = max(1e-10, beta * 0.7)

    return best_a


def _simulated_annealing(a, deadline, max_iter=5000, T_init=0.005, T_min=0.0001):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(999)

    best_a = a.copy()
    best_score = _score(a)
    T = T_init
    no_imp = 0
    decay = (T_min / T_init) ** (1.0 / max(1, max_iter))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_try = a.copy()
        n_moves = rng.integers(1, max(2, n // 64) + 1)
        for _ in range(n_moves):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.15)
            delta = frac * a_try[i]
            a_try[i] -= delta
            a_try[j] += delta

        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s

        sc = _score(a_try)
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
            no_imp = 0
        else:
            delta_sc = sc - best_score
            if rng.random() < np.exp(-delta_sc / max(T, 1e-10)):
                a = a_try.copy()
            no_imp += 1

        T = max(T_min, T * decay)

        if no_imp > 3000:
            break

    return best_a


def _full_refinement(a, deadline, phase_budget):
    """Run a multi-phase refinement on sequence a."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    # Phase 1: LP with line search
    lp_budget = phase_budget * 0.35
    lp_end = time.monotonic() + lp_budget
    if lp_end < deadline - 30:
        a_lp = _lp_line_search_optimize(best_a, lp_end, max_iter=10000, inflate_init=1.0012)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"    LP+LS: {sc:.10f}")

    # Phase 2: Gradient polish
    gp_budget = phase_budget * 0.25
    gp_end = time.monotonic() + gp_budget
    if gp_end < deadline - 20:
        a_gp = _gradient_polish(best_a, gp_end, max_iter=25000, beta_scale=0.035)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"    Grad: {sc:.10f}")

    # Phase 3: Simulated annealing
    sa_budget = phase_budget * 0.15
    sa_end = time.monotonic() + sa_budget
    if sa_end < deadline - 15:
        a_sa = _simulated_annealing(best_a, sa_end, max_iter=4000, T_init=0.004)
        sc = _score(a_sa)
        if sc < best_score:
            best_score = sc
            best_a = a_sa.copy()
        print(f"    Anneal: {sc:.10f}")

    # Phase 4: L-BFGS polish
    lb_budget = phase_budget * 0.15
    lb_end = time.monotonic() + lb_budget
    if lb_end < deadline - 10:
        a_lb = _logit_lbfgs_polish(best_a, lb_end, max_iter_total=100)
        sc = _score(a_lb)
        if sc < best_score:
            best_score = sc
            best_a = a_lb.copy()
        print(f"    L-BFGS: {sc:.10f}")

    # Phase 5: Final LP
    lp2_budget = phase_budget * 0.1
    lp2_end = time.monotonic() + lp2_budget
    if lp2_end < deadline - 5:
        a_lp2 = _lp_line_search_optimize(best_a, lp2_end, max_iter=5000, inflate_init=1.0008)
        sc = _score(a_lp2)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2.copy()
        print(f"    LP2: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LP+line search, peak-aware gradient,
    simulated annealing, and L-BFGS polish with multi-scale initialization.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 896, 1024]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))

    print(f"Trying lengths: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Multi-scale initialization
    # Try to build up a good initial sequence from small to large
    if parent is None and time.monotonic() < deadline - 120:
        print("\nPhase 1: Multi-scale initialization")
        small_len = 64
        a_small = _make_u_profile(small_len, 2.5)
        for sc_try in [_score(_make_u_profile(small_len, al)) for al in [1.5, 2.0, 2.5, 3.0, 4.0]]:
            if sc_try < _score(a_small):
                a_small = _make_u_profile(small_len, 2.5)
                break

        # Optimize at small scale
        a_small_opt = _gradient_polish(a_small, time.monotonic() + 15, max_iter=5000, beta_scale=0.05)
        print(f"  Small scale (n={small_len}): {_score(a_small_opt):.8f}")

        # Interpolate up to first target length
        target_n = lengths[0]
        a_init = _interpolate_profile(a_small_opt, target_n)
        a_init = _project_simplex(a_init)
    else:
        target_n = lengths[0]
        if parent is not None:
            a_init = _interpolate_profile(parent, target_n)
        else:
            a_init = _make_u_profile(target_n, 2.0)

    # Phase 2: Main optimization for each length
    num_lengths = len(lengths)
    # Allocate more time to the parent length
    base_budget = (deadline - time.monotonic() - 150.0) / num_lengths

    for idx, n_t in enumerate(lengths):
        if time.monotonic() > deadline - 120:
            break

        # Give more budget to parent length and smaller lengths
        if parent is not None and n_t == parent.size:
            phase_budget = base_budget * 1.5
        elif n_t < 768:
            phase_budget = base_budget * 0.7
        else:
            phase_budget = base_budget * 1.0
        phase_budget = max(30.0, min(phase_budget, deadline - time.monotonic() - 100.0))

        if phase_budget < 20:
            continue

        print(f"\nLength {n_t}: budget={phase_budget:.0f}s")

        # Initialization
        if parent is not None:
            a_current = _interpolate_profile(parent, n_t)
        elif idx == 0:
            a_current = a_init
        else:
            if best_seq is not None:
                a_current = _interpolate_profile(best_seq, n_t)
            else:
                a_current = _make_u_profile(n_t, 2.0)

        a_current = _project_simplex(a_current)
        sc_init = _score(a_current)
        print(f"  Init score: {sc_init:.8f}")
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_current.copy()

        # Run full refinement
        a_res, sc_res = _full_refinement(a_current, deadline, phase_budget)
        if sc_res < best_score:
            best_score = sc_res
            best_seq = a_res.copy()
            print(f"  ** New best at n={n_t}: {sc_res:.10f}")

    # Phase 3: Diverse restarts at the best length
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        print(f"\nPhase 3: Diverse restarts at n={n_best}")

        inits = []
        # U-shaped profiles with various parameters
        for alpha in [1.5, 2.5, 3.5, 5.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Gaussian mixtures
        inits.append(_make_gaussian_mixture(n_best, [0.1, 0.9], [0.12, 0.12], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_best, [0.05, 0.95], [0.08, 0.08], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_best, [0.2, 0.5, 0.8], [0.15, 0.1, 0.15], [1.0, 0.5, 1.0]))
        # Linear profiles
        for eta in [1.0, 2.0, 3.0]:
            inits.append(_make_linear_profile(n_best, eta))
        # Random structured
        for _ in range(3):
            a_rand = np.maximum(rng.normal(1.0, 0.6, n_best), 0.0)
            a_rand = a_rand / a_rand.sum()
            inits.append(a_rand)
        # Perturbed best
        for _ in range(4):
            a_pert = best_seq * (1.0 + 0.2 * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s > 1e-10:
                a_pert = a_pert / s
                inits.append(a_pert)
        # Frequency-domain inspired
        for freq in [2, 3, 5]:
            t = np.linspace(0, 1, n_best)
            a_freq = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
            a_freq = np.maximum(a_freq, 1e-12)
            inits.append(a_freq / a_freq.sum())

        # Limit based on time
        remaining = deadline - time.monotonic() - 50
        per_init = max(20, remaining / max(1, len(inits)))
        inits = inits[:min(len(inits), int(remaining / 40) + 1)]

        for a0 in inits:
            if time.monotonic() > deadline - 40:
                break
            a0 = _project_simplex(a0)
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            # Quick refine
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            sub_budget = min(50.0, rem * 0.6)
            a_res, sc_res = _full_refinement(a0, deadline, sub_budget)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best from diverse init: {sc_res:.10f}")

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 15
        print(f"\nPhase 4: Final polish, budget={final_budget:.0f}s, n={n_best}")

        # 4a: LP
        lp_b = min(final_budget * 0.35, 150.0)
        if lp_b > 10:
            a_lp = _lp_line_search_optimize(best_seq, time.monotonic() + lp_b,
                                           max_iter=20000, inflate_init=1.0006)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}")

        # 4b: Gradient
        gp_b = min(final_budget * 0.3, 120.0)
        if gp_b > 10:
            a_gp = _gradient_polish(best_seq, time.monotonic() + gp_b,
                                   max_iter=30000, beta_scale=0.025)
            sc = _score(a_gp)
            if sc < best_score:
                best_score = sc
                best_seq = a_gp.copy()
            print(f"  Final Grad: {sc:.10f}")

        # 4c: L-BFGS
        lb_b = min(final_budget * 0.2, 80.0)
        if lb_b > 8:
            a_lb = _logit_lbfgs_polish(best_seq, time.monotonic() + lb_b, max_iter_total=150)
            sc = _score(a_lb)
            if sc < best_score:
                best_score = sc
                best_seq = a_lb.copy()
            print(f"  Final L-BFGS: {sc:.10f}")

        # 4d: Short LP
        lp_b2 = min(final_budget * 0.15, 40.0)
        if lp_b2 > 5:
            a_lp2 = _lp_line_search_optimize(best_seq, time.monotonic() + lp_b2,
                                            max_iter=5000, inflate_init=1.0004)
            sc = _score(a_lp2)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp2.copy()
            print(f"  Final LP2: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(512, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(512, 2.0)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print("WARNING: evaluate_sequence returned inf, using fallback")
                final_seq = _make_u_profile(512, 2.0)
                best_score = _score(final_seq)
        except Exception as e:
            print(f"Evaluation error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
