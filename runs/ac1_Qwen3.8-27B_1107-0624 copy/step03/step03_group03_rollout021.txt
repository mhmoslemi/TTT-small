"""
Convex minimax optimization for sequence convolution flattening.
Uses fictitious play, LSE-Adam gradient, active-set LP, and subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score for sum(a)=1: 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _palindromize(a):
    return (a + a[::-1]) / 2.0

def _lse_adam(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = np.clip((c - m_val) / T, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    m = 0.9 * state['m'] + 0.1 * gz
    v = 0.999 * state['v'] + 0.001 * (gz * gz)
    mh = m / (1.0 - 0.9 ** t)
    vh = v / (1.0 - 0.999 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val

def _lse_optimize(a, deadline, max_iter=100000, T_start=0.05, T_end=1e-5, lr=0.1):
    a = _normalize(a.copy())
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a) - float(np.mean(np.log(a)))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, _ = _lse_adam(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12)) - float(np.mean(np.log(np.maximum(a, 1e-12))))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.01, lr * 0.9)
            if no_imp > max(5000, 10 * n):
                rng_r = np.random.default_rng(42 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = _normalize(np.maximum(a, 1e-12))
                z = np.log(a) - float(np.mean(np.log(a)))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        T = max(T * T_ratio, T_end)
    return best_a

def _make_conv_matrix_reduced(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b

def _lp_spread(a, M_inf, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if n > 64:
        A, b = _make_conv_matrix_reduced(a, M_inf, top_k)
    else:
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = a[j]
        b = np.full(2*n-1, M_inf)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_refine(a, deadline, max_iter=5000, inflate=1.0003, blend=0.2):
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        if g is None:
            inflate_cur = min(1.01, inflate_cur * 1.2)
            no_imp += 5
            if no_imp > 300:
                break
            continue
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 400:
                break
            continue
        g = g / sum_g
        # Line search
        best_t = 0.0
        best_sc = _score_norm(a)
        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = _score_norm(_normalize(np.maximum(a + c_pt * (g - a), 0)))
        fd = _score_norm(_normalize(np.maximum(a + d_pt * (g - a), 0)))
        for _2 in range(15):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = _score_norm(_normalize(np.maximum(a + c_pt * (g - a), 0)))
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = _score_norm(_normalize(np.maximum(a + d_pt * (g - a), 0)))
        best_t = (lo + hi) / 2.0
        if best_t < 0.01:
            best_t = blend_cur * 0.5

        a_new = _normalize(np.maximum((1.0 - best_t) * a + best_t * g, 0))
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new
    return best_a

def _fictitious_play(a, deadline, max_iter=50000, step_schedule='harmonic'):
    """Fictitious play for the convolution flattening game."""
    a = _normalize(a.copy())
    n = len(a)
    n_conv = 2 * n - 1
    
    # Initialize dual variable (distribution over k values)
    q = np.ones(n_conv) / n_conv
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    lr = 0.1
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Primal step: minimize a^T Q(a) a where Q depends on q
        # Q[i,j] = q[i+j]
        # Gradient of a^T Q a w.r.t. a: 2 * Q @ a (since Q is symmetric)
        # But Q is a Hankel matrix, so Q @ a is a convolution-like operation
        
        # Compute Q @ a efficiently:
        # (Q @ a)[i] = sum_j q[i+j] * a[j] for j = 0..n-1
        # This is a cross-correlation of a with q!
        # (Q @ a)[i] = sum_j a[j] * q[i+j]
        
        # We can compute this using FFT:
        # Let's extend q to length 2*n-1 and compute the cross-correlation
        N = _next_pow2(n_conv + n)
        fa = np.fft.rfft(a, N)
        fq = np.fft.rfft(q, N)
        
        # cross-correlation: sum_j a[j] * q[i+j] = (a * rev(q))[i + n - 1]... 
        # Actually, let's think about this more carefully.
        # sum_j a[j] * q[i+j] for j=0..n-1
        # Let b[j] = a[j], c[k] = q[k]
        # We want sum_j b[j] * c[i+j]
        # This is the cross-correlation (b ⊛ c)[i] = sum_j b[j] * c[i+j]
        # In FFT terms: IFFT(FFT(b) * conj(FFT(c))) gives sum_j b[j] * c[j-i] (correlation)
        # We need sum_j b[j] * c[i+j] which is a different shift
        
        # Actually: sum_j a[j] * q[i+j] = (a * q_rev)[i + n - 1] where q_rev is q reversed
        # Let me just compute it directly for now since n is at most a few thousand
        
        # Direct computation (O(n^2) but n is at most ~4096)
        # Actually, let's use the FFT approach properly
        # sum_j a[j] * q[i+j] where j ranges over 0..n-1 and i+j ranges over 0..2n-2
        # Let's define r[i] = sum_j a[j] * q[i+j]
        # This is the same as: for each i, take the inner product of a with q[i:i+n]
        # This is a "sliding dot product" which can be computed via cross-correlation
        
        # Using numpy correlate: np.correlate(q, a, mode='full')[n-1+i] = sum_j a[j] * q[i+j+n-1-j]... 
        # Hmm, let me just think about it differently.
        
        # r[i] = sum_{j=0}^{n-1} a[j] * q[i+j]
        # Let's define b[j] = a[j] for j=0..n-1, and c[k] = q[k] for k=0..2n-2
        # r[i] = sum_j b[j] * c[i+j]
        # This is the "anti-convolution" or cross-correlation
        
        # Using FFT: r = ifft(fft(b, N) * fft(c, N)) ... no that's convolution
        # Cross-correlation: r[i] = sum_j b[j] * c[i+j]
        # = sum_j b[j] * c[(i+j)]
        # Let's flip b: b_flip[j] = b[n-1-j]
        # Then r[i] = sum_j b_flip[n-1-j] * c[i+j]
        # = (b_flip * c)[i + n - 1] (convolution)
        
        N2 = _next_pow2(n + n_conv - 1)
        b_flip = a[::-1]
        fb = np.fft.rfft(b_flip, N2)
        fc = np.fft.rfft(q, N2)
        conv_bc = np.fft.irfft(fb * fc, N2)[:n + n_conv - 1]
        r = conv_bc[n-1:n-1+n]  # r[i] = sum_j a[j] * q[i+j]
        
        # Now the gradient of a^T Q a is 2*r (since Q is symmetric Hankel)
        # But we need to be careful: a^T Q a = sum_{i,j} a[i] a[j] q[i+j]
        # d/da[k] (a^T Q a) = 2 * sum_j a[j] q[k+j] = 2 * r[k]
        
        grad = 2.0 * r
        
        # Projected gradient step on the simplex
        lr_cur = lr / (1.0 + it * 0.01)  # decaying step size
        a_new = a - lr_cur * grad
        a_new = _normalize(a_new)
        
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 5000 == 0:
                lr = max(0.01, lr * 0.8)
            if no_imp > 20000:
                break
        
        a = a_new
        
        # Dual step: update q towards the argmax of conv
        k_star = int(np.argmax(c_new))
        if step_schedule == 'harmonic':
            alpha = 1.0 / (it + 1)
        else:
            alpha = 0.1
        q = (1.0 - alpha) * q + alpha * np.eye(n_conv)[k_star]
    
    return best_a

def _equalize_active_set(a, n_top=20):
    """Try to equalize the top n_top convolution values."""
    a = _normalize(a.copy())
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max())
    
    # Find the top k values
    top_idx = np.argsort(c)[::-1][:n_top]
    top_vals = c[top_idx]
    
    # We want to reduce the max by "pushing down" the active constraints
    # Use a weighted subgradient step
    w = np.zeros_like(c)
    for k in top_idx:
        w[k] = (c[k] - top_vals.min()) / (M - top_vals.min() + 1e-15) ** 2 + 0.1
    
    # Gradient contribution from these k values
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]
    grad = 2.0 * h
    
    # Projected gradient step
    dot = float(np.dot(a, grad))
    grad_tan = grad - dot * a
    a_new = a - 0.01 * grad_tan
    a_new = _normalize(a_new)
    
    return a_new

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_flat_with_edges(n, edge_weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0
    * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02):
    a = _normalize(np.maximum(a.copy(), 1e-12))
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    v = np.zeros(n)
    momentum = 0.8
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n-1:2*n-1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = _normalize(a + v)
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(5000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
    return best_a
