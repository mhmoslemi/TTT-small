   If we want to minimize `sum(a_n^2)` (which contributes to the peak), we should minimize the total energy, which means making `|A[k]|` for `k > 0` as small as possible. But then `a` is close to uniform, and the peak is `1/n`.

   Alternatively, if we allow larger `|A[k]|` for `k > 0`, we can potentially shape the phase to make the inverse DFT have lower peaks (at the cost of `a` having negative values, which we then project).

4. Apply the phase of the original `A` to the new magnitude: `A_new = |A_new| * exp(i*angle(A))`.
5. Compute `a_new = IFFT(A_new)`.
6. Project to non-negative: `a_new = max(a_new, 0)`.
7. Normalize: `a_new = a_new / sum(a_new)`.
8. Repeat.

This is essentially the Gerchberg-Saxton algorithm with a non-negativity constraint!

- Lower bound: `c[n-1] = a · rev(a) >= 0`, and by the constraint `sum(a) = 1`, we have some structural limits.

The theoretical minimum: if we could make `c[k]` perfectly flat at value `V` for all `k`, then:

- There are `2n-1` values, so `V = 1/(2n-1)`
- Score = `2n / (2n-1) ≈ 1`

"""
Spectral optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Gerchberg-Saxton style iterative spectral shaping combined with
LP refinement and L-BFGS polish.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gerchberg_saxton(a, deadline, max_iter=10000, target_flatness=1.0):
    """
    Gerchberg-Saxton style spectral optimization.
    Iteratively shapes the power spectrum to be flat while projecting
    back to non-negative time domain.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, N)
        
        # Target magnitude: flat spectrum
        # A[0] is fixed (DC component)
        # For k > 0, target |A[k]| to be as equal as possible
        # while preserving total energy
        
        # Current magnitude
        mag = np.abs(A)
        dc = mag[0]
        
        # Target: spread the non-DC energy equally
        # Total non-DC energy budget: we can choose this
        # To minimize peak, we want non-DC components to be present
        # but not too large (they contribute to a_n^2)
        
        # Strategy: set non-DC magnitude to a fraction of DC
        # gamma * dc for k > 0
        # This gives a "chirp-like" time domain signal
        
        # Actually, let's use a more principled approach:
        # We want the inverse DFT to be non-negative and have low peak autocorrelation.
        # The best target is a flat spectrum, but we need to preserve DC.
        
        # Target magnitude:
        # |A_target[0]| = dc
        # |A_target[k]| = target_val for k > 0
        # 
        # Choose target_val to balance:
        # - Too small: time domain is close to constant (uniform), peak = 1/n
        # - Too large: time domain has large oscillations, some will be negative
        #
        # Optimal target_val: approximately dc / sqrt(N-1) * some_factor
        # Let's parameterize and adapt.
        
        # Use adaptive target: start with flat, reduce if too many negatives
        target_val = dc / np.sqrt(N - 1)  # This gives approximately uniform time domain
        
        # Build target magnitude
        mag_target = np.full(N // 2 + 1, target_val)
        mag_target[0] = dc
        
        # Actually, for better performance, let's use a "soft" target:
        # Blend between current magnitude and flat target
        # This avoids the hard projection that causes oscillations
        
        alpha = 0.5  # Blending factor
        mag_new = alpha * mag_target + (1 - alpha) * mag
        
        # Preserve phase
        A_new = mag_new * np.exp(1j * np.angle(A))
        
        # Inverse DFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            # Reset
            a_new = np.ones(n) / n
            s = 1.0
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adapt alpha
        if no_imp > 50:
            alpha = max(0.1, alpha * 0.9)
        
        a = a_new
        
        if no_imp > 500:
            break
    
    return best_a


def _gs_pure(a, deadline, max_iter=20000):
    """
    Pure Gerchberg-Saxton: project between flat magnitude and non-negativity.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        dc = mag[0]
        
        # Set non-DC magnitude to a target level
        # Try several target levels and pick the best
        # For now, use: target = dc / sqrt(2*(N-1))
        # This is motivated by: for a "random" sequence with sum=1,
        # the typical |A[k]| for k>0 is about 1/sqrt(N), and |A[0]| = 1.
        # So the ratio is about sqrt(N). We want a bit less than that.
        
        # Actually, let's think about this differently.
        # If |A[0]| = dc and |A[k]| = t for k=1,...,N-1 (with conjugate symmetry),
        # then the time-domain signal has:
        # a[0] ≈ (dc + (N-1)*t*cos(phi)) / N
        # For the signal to be non-negative, we need the DC to dominate the oscillations.
        # The max oscillation amplitude is about (N-1)*t/N.
        # So we need dc > (N-1)*t/N, i.e., t < dc*N/(N-1) ≈ dc.
        
        # For a "good" signal (low peak autocorrelation), we want t to be as large
        # as possible (flat spectrum) while keeping a >= 0.
        # Let's try t = dc * 0.5 as a starting point.
        
        t = dc * 0.5
        
        mag_target = np.full(N // 2 + 1, t)
        mag_target[0] = dc
        
        # Apply target magnitude with original phase
        A_new = mag_target * np.exp(1j * phase)
        
        # Inverse DFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            # Try different target levels
            if no_imp % 100 == 0:
                t_mult = 0.3 + 0.4 * (no_imp % 5) / 5.0
                t = dc * t_mult
                mag_target = np.full(N // 2 + 1, t)
                mag_target[0] = dc
                A_try = mag_target * np.exp(1j * phase)
                a_try = np.fft.irfft(A_try, N)[:n]
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_score:
                        best_score = sc_try
                        best_a = a_try.copy()
                        no_imp = 0
                        t = dc * t_mult
        
        a = a_new
        
        if no_imp > 2000:
            break
    
    return best_a

            break

        h = h_full[n - 1:2 * n - 1]

def _lp_refine(a, deadline, max_iter=20000, inflate=1.0008, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.05)
            if no_imp > 300:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-13:

            best_M = M_new
            best_a = a_new.copy()

        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.1, blend_cur * 0.95)
                inflate_cur = min(1.002, inflate_cur * 0.98)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, alpha=2.0):

    a = 1.0 + alpha * (t - 0.5) ** 2 - 0.25 * alpha
    a = np.maximum(a, 1e-12)

def _make_chirp_profile(n, rate=1.0):

def _gs_multi_target(a, deadline, max_iter=10000):
    """
    Gerchberg-Saxton with multiple target spectral shapes.
    Tries different target magnitudes and picks the best.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Try different target magnitude profiles
    # Target 1: flat (equal energy at all frequencies)
    # Target 2: slightly peaked at low frequencies
    # Target 3: slightly peaked at high frequencies
    
    target_types = ['flat', 'low_boost', 'high_boost', 'sqrt_taper']
    
    for target in target_types:
        if time.monotonic() > deadline:
            break
        
        a_cur = best_a.copy()
        no_imp = 0
        
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            
            A = np.fft.rfft(a_cur, N)
            mag = np.abs(A)
            phase = np.angle(A)
            dc = mag[0]
            
            # Compute target magnitude
            if target == 'flat':
                # Uniform non-DC magnitude
                t = dc / np.sqrt(N - 1) * 1.5
                mag_t = np.full(N // 2 + 1, t)
                mag_t[0] = dc
            elif target == 'low_boost':
                freqs = np.arange(N // 2 + 1)
                weights = 1.0 / (1.0 + 0.5 * freqs)
                t = dc / np.sqrt(N - 1) * 1.5
                mag_t = t * weights / weights[0]
                mag_t[0] = dc
            elif target == 'high_boost':
                freqs = np.arange(N // 2 + 1)
                weights = np.sqrt(1.0 + 0.5 * freqs / (N // 2))
                t = dc / np.sqrt(N - 1) * 1.5
                mag_t = t * weights / weights.max()
                mag_t[0] = dc
            elif target == 'sqrt_taper':
                freqs = np.arange(N // 2 + 1)
                weights = np.sqrt(1.0 - freqs / (N // 2 + 1))
                t = dc / np.sqrt(N - 1) * 1.5
                mag_t = t * weights
                mag_t[0] = dc
            
            # Blend with current (soft projection)
            alpha = 0.3
            mag_new = alpha * mag_t + (1 - alpha) * mag
            mag_new[0] = dc  # Preserve DC exactly
            
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            
            sc = _score(a_new)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_new.copy()
                no_imp = 0
                a_cur = a_new.copy()
            else:
                no_imp += 1
                a_cur = a_new  # Still update for convergence
                if no_imp > 500:
                    break
        
        print(f"  GS target={target}: score={best_score:.10f}", flush=True)
    
    return best_a


def _coordinate_descent(a, deadline, max_iter=50000):
    """
    Coordinate descent: optimize one element at a time.
    For each position i, find the optimal a[i] given all other elements fixed.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    # Pre-compute the convolution contributions from all elements except a[i]
    # c[k] = a[i]*a[k-i] + rest[k]
    # To minimize max(c), for each i we want a[i] to be as small as possible
    # (since it only adds to the convolution)
    # But we need sum(a) = 1, so making a[i] smaller means making others larger.
    # This is the tension.
    
    # Actually, coordinate descent on this problem is tricky because
    # changing a[i] affects all convolution values.
    # Let's use a simpler approach: cyclic coordinate descent with line search.
    
    for outer in range(max_iter):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try reducing a[i]
            delta = a[i] * 0.05
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[i] = max(a[i] - delta, 1e-15)
            a_try[i+1 if i+1 < n else 0] += delta  # Redistribute to neighbor
            s = a_try.sum()
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                no_imp = 0
            else:
                no_imp += 1
        if not improved and no_imp > 5 * n:
            break
    
    return best_a


def _lbfgs_smooth(a, deadline, max_iter=10000):
    """L-BFGS with smooth max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.01 * c_max)
    
    def objective(z):
        e = np.exp(z - z.max())
        p = e / e.sum()
        c = _fft_convolve(p)
        val, _ = _smooth_max_val(c, beta)
        return 2.0 * n * val
    
    def gradient(z):
        e = np.exp(z - z.max())
        p = e / e.sum()
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N2 = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N2)
        fp_rev = np.fft.rfft(p[::-1], N2)
        h_full = np.fft.irfft(fw * fp_rev, N2)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        return 2.0 * n * gz
    
    z0 = np.log(a)
    z0 -= np.mean(z0)
    
    start_time = time.monotonic()
    try:
        res = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': 20000,
                                'gtol': 1e-8})
        z_opt = res.x
        e = np.exp(z_opt - z_opt.max())
        a_opt = e / e.sum()
        sc = _score(a_opt)
        if sc < _score(a):
            return a_opt
    except Exception as e:
        print(f"  L-BFGS error: {e}", flush=True)
    
    return a
