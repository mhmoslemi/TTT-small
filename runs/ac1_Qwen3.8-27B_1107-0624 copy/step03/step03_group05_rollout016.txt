"""
Improved step function optimization using:
1. Aggressive LP iteration with adaptive stepping
2. Dual-informed gradient polishing
3. Multi-scale initialization
4. Spectral reshaping for diversity
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming a is already normalized to sum=1."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:

    return w / s

def _spectral_perturb(a, rng, strength=0.3):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    # Perturb the phase
    phase_perturb = strength * rng.normal(size=len(fa)) * 0.1
    fa_perturbed = fa * np.exp(1j * phase_perturb)
    a_new = np.fft.irfft(fa_perturbed, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _coarse_to_fine(parent, n_target, deadline):
    """Optimize at coarse resolution then interpolate to fine."""
    n_coarse = max(64, n_target // 4)
    # Interpolate parent to coarse
    x0 = np.linspace(0, 1, parent.size)
    x1 = np.linspace(0, 1, n_coarse)
    a_coarse = np.interp(x1, x0, parent)
    a_coarse = np.maximum(a_coarse, 0)
    a_coarse = a_coarse / a_coarse.sum()

    # Quick LP at coarse
    coarse_budget = min(60.0, (deadline - time.monotonic()) * 0.1)
    if coarse_budget > 5:
        a_coarse = _lp_refine_v2(a_coarse, time.monotonic() + coarse_budget, max_iter=2000)

    # Interpolate back to fine
    x2 = np.linspace(0, 1, n_target)
    a_fine = np.interp(x2, x1, a_coarse)
    a_fine = np.maximum(a_fine, 0)
    a_fine = a_fine / a_fine.sum()
    return a_fine


def _pipeline(a_init, n, deadline, time_budget):
    """Run optimization pipeline for a given initial sequence."""
    a = np.asarray(a_init, dtype=np.float64).ravel()
    if len(a) != n:
        x0 = np.linspace(0, 1, len(a))
        x1 = np.linspace(0, 1, n)
        a = np.interp(x1, x0, a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(best_a)
    print(f"    Init: {best_score:.10f}")

    elapsed = lambda: time.monotonic() - t_start_global
    remaining = lambda: deadline - time.monotonic()

    # Phase A: LP refinement (40% of budget)
    lp_time = time_budget * 0.40
    if remaining() > lp_time + 10:
        end = time.monotonic() + lp_time
        a_lp = _lp_refine_v2(best_a, end, max_iter=15000)
        sc = _score_norm(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"    LP v2: {sc:.10f}")

    # Phase B: Symmetric LP (15% of budget)
    sym_time = time_budget * 0.15
    if remaining() > sym_time + 10:
        end = time.monotonic() + sym_time
        a_sym = _lp_symmetric(best_a, end, max_iter=3000)
        sc = _score_norm(a_sym)
        if sc < best_score:
            best_score = sc
            best_a = a_sym.copy()
        print(f"    SymLP: {sc:.10f}")

    # Phase C: Gradient polish (20% of budget)
    gp_time = time_budget * 0.20
    if remaining() > gp_time + 10:
        end = time.monotonic() + gp_time
        a_gp = _gradient_polish_v2(best_a, end, max_iter=15000)
        sc = _score_norm(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"    GradPolish: {sc:.10f}")

    # Phase D: Adam polish (15% of budget)
    ad_time = time_budget * 0.15
    if remaining() > ad_time + 10:
        end = time.monotonic() + ad_time
        a_ad = _adam_polish(best_a, end, max_iter=15000, lr=0.035)
        sc = _score_norm(a_ad)
        if sc < best_score:
            best_score = sc
            best_a = a_ad.copy()
        print(f"    Adam: {sc:.10f}")

    # Phase E: Final LP (10% of budget)
    lp2_time = time_budget * 0.10
    if remaining() > lp2_time + 5:
        end = time.monotonic() + lp2_time
        a_lp2 = _lp_refine_v2(best_a, end, max_iter=5000)
        sc = _score_norm(a_lp2)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2.copy()
        print(f"    LP2: {sc:.10f}")

    return best_a, best_score

    print(f"\n=== Phase 4: Final polish ===")
    final_budget = deadline - time.monotonic() - 30.0
    if final_budget > 30 and best_seq is not None:

    return [float(x) for x in final_seq]
