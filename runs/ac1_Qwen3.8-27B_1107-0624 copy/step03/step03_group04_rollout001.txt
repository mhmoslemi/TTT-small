import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _smooth_max_powermean(c, p):
    """Power mean smooth max: (sum c^p / K)^(1/p) for large p."""
    c = np.asarray(c, dtype=np.float64)
    K = len(c)
    c_pos = np.maximum(c, 1e-30)
    cp = c_pos ** p
    s = float(cp.sum())
    if s < 1e-30:
        mval = float(c.max())
        return mval, np.zeros_like(c)
    val = (s / K) ** (1.0 / p)
    w = cp / s
    return val, w

def _grad_smooth_max(a, c, w):
    """Gradient of smooth max wrt a, given weights w on c = a*a."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h

def _logit_grad_step(a, beta, lr, method='logsumexp'):
    """One gradient step in logit space."""
    n = len(a)
    c = _fft_convolve(a)
    if method == 'logsumexp':
        _, w = _smooth_max_logsumexp(c, beta)
    else:
        _, w = _smooth_max_powermean(c, p=200)
    
    grad_c = _grad_smooth_max(a, c, w)
    # Project gradient to simplex tangent space
    dot = float(np.dot(a, grad_c))
    gz = a * (grad_c - dot)
    gz -= float(np.mean(gz))
    
    z = np.log(np.maximum(a, 1e-30))
    z -= np.mean(z)
    z_new = z - lr * gz
    z_new = np.clip(z_new, -50.0, 50.0)
    z_new -= np.mean(z_new)
    e = np.exp(z_new)
    a_new = e / e.sum()
    return a_new

def _adam_logit_optimize(a, deadline, max_iter=30000, lr=0.05, beta_init=None, method='logsumexp'):
    """Adam optimization in logit space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = beta_init if beta_init is not None else max(1e-6, 0.05 * c_max)
    
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        if method == 'logsumexp':
            _, w = _smooth_max_logsumexp(c, beta)
        else:
            _, w = _smooth_max_powermean(c, p=200)
        
        grad_c = _grad_smooth_max(p, c, w)
        dot = float(np.dot(p, grad_c)) + 1e-7 * float(np.dot(p, p))
        gz = p * (grad_c - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            break
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z_new)):
            continue
        
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.95)
        
        # Adaptive beta
        if it % 200 == 0 and it > 200:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.7)
                elif gap > beta * 5.0:
                    beta = min(0.5 * c_max_new, beta * 1.3)
        
        # Restart
        if no_imp > max(3000, 5 * n):
            rng_r = np.random.default_rng(it * 7919 + 1)
            z_restart = best_z + 0.3 * rng_r.standard_normal(n)
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            no_imp = 0
            lr = 0.05
        
        z = z_new
    
    return best_a

def _lp_refine_aggressive(a, deadline, max_iter=10000):
    """Aggressive LP-based refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_a = a.copy()
    
    no_imp = 0
    epsilon = 0.01
    blend = 0.3
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_target = M * (1.0 + epsilon)
        
        b = _lp_spread(a, M_target)
        if b is None:
            epsilon = min(0.5, epsilon * 2)
            continue
        
        sum_b = float(b.sum())
        if sum_b <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                epsilon = min(0.5, epsilon * 1.5)
            if no_imp > 100:
                break
            continue
        
        b = b / sum_b
        
        # Try the update
        a_new = (1.0 - blend) * a + blend * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Decrease epsilon and blend for finer steps
            epsilon = max(0.001, epsilon * 0.95)
            blend = max(0.1, blend * 0.99)
        else:
            no_imp += 1
            # Try smaller blend
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.8)
                epsilon = min(0.5, epsilon * 1.2)
            if no_imp > 200:
                a = best_a.copy()
                epsilon = 0.01
                blend = 0.3
                no_imp = 0
        
        a = a_new
    
    return best_a

def _gradient_polish(a, deadline, max_iter=20000):

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _make_init_families(n, rng):
    """Generate diverse initialization families."""
    inits = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    
    # Bell
    for width in [0.2, 0.3, 0.4, 0.5]:
        a = np.exp(-((t - 0.5) / width) ** 2)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    
    # Lorentzian
    for gamma in [0.3, 0.5, 1.0]:
        a = 1.0 / (1.0 + gamma * ((t - 0.5) / 0.1) ** 2)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    
    # Sinusoidal
    for freq in [1, 2, 3]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    
    # Power
    for p in [0.5, 1.0, 2.0, 4.0]:
        a = np.abs(t - 0.5) ** p + 0.1
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    
    # Symmetric edge-boost
    for boost in [0.3, 0.5, 1.0]:
        dist = np.abs(t - 0.5)
        a = np.exp(-(dist / 0.4)**2) + boost * np.exp(-dist**2 / 0.01)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random smooth (via FFT)
    for _ in range(3):
        N = _next_pow2(n)
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        X = np.zeros(N, dtype=complex)
        X[0] = 1.0
        for k in range(1, N // 2 + 1):
            mag = rng.uniform(0.5, 1.0)
            X[k] = mag * np.exp(1j * phases[k])
            X[N - k] = mag * np.exp(-1j * phases[k])
        a = np.real(np.fft.ifft(X))[:n]
        a = np.abs(a)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    return inits
