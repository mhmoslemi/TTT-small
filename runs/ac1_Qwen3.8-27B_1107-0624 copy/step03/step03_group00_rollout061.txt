We want to minimize `f(a) = max_k conv(a,a)[k]` subject to `sum(a) = 1`, `a >= 0`.

The gradient of `conv(a,a)[k]` with respect to `a[m]` is `2 * a[k-m]` (with `a[j] = 0` for `j` outside `[0,n-1]`).

The subgradient of `max_k g_k(a)` at a point where the max is achieved at set S is:
`(1/|S|) * sum_{k in S} grad g_k(a)`

So the subgradient at position m is:
`(2/|S|) * sum_{k in S} a[k-m]`

This is what my `_subgradient` function computes (with the special case temperature=0).

3. If score(a_new) > score(a), reject and halve lr
4. Otherwise accept and maybe increase lr slightly

Let me also consider: since we're on the simplex, the projected gradient might not always decrease the objective (because the projection can increase it). In that case, we should use a "mirror descent" or "exponential weights" approach instead.

Actually, for the constraint `a >= 0, sum(a) = 1`, the natural parameterization is in logit space: `a = softmax(z)`. Then we can do unconstrained optimization on z.

The gradient of `f(softmax(z))` with respect to `z` is:
`dz = diag(a) @ (g - sum(a * g))`

where `g` is the gradient of `f` with respect to `a`.

This is because `da_i/dz_j = a_i * (delta_ij - a_j)`, so:
`df/dz_j = sum_i (df/da_i) * (da_i/dz_j) = sum_i g_i * a_i * (delta_ij - a_j) = a_j * (g_j - sum_i g_i * a_i)`

So: `dz_j = a_j * (g_j - dot(a, g))`

This is the "score function" gradient in the softmax parameterization. It automatically preserves the constraint (since softmax always gives a valid probability vector).

def _fft_convolve(a):

                    'cos1', 'cos2', 'twobump']
        for pk in profiles:
            if time.monotonic() > deadline - 1400:
                break
            p = _make_profile(n, pk)
            sc = _score(p)
            candidates.append((sc, p, n, f'{pk}_n{n}'))

    candidates.sort(key=lambda x: x[0])
    top_sc, top_p, top_n, top_label = candidates[0]
    print(f"  Best init: {top_sc:.10f} n={top_n} [{top_label}]")
    if top_p is not None and np.isfinite(top_sc) and top_sc < best_score:
        best_score = float(top_sc)
        best_seq = np.asarray(top_p, dtype=np.float64).copy()

    # Phase 2: Adam optimization on top candidates
    print("Phase 2: Logit-Adam optimization")
    phase2_end = t_start + 700.0
    polished = []

    for idx in range(min(8, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        sc, p, n, label = candidates[idx]
        remaining = phase2_end - time.monotonic()
        per = min(remaining, 100.0)
        if per < 15:
            break
        p_b, sc_b = _logit_adam(p, time.monotonic() + per, rng,
                                lr_init=0.06, temp_init=0.0,
                                temp_decay=0.95, no_imp_limit=300)
        if sc_b < best_score:
            best_score = float(sc_b)
            best_seq = np.asarray(p_b, dtype=np.float64).copy()
            print(f"  ADAM: {sc_b:.10f} n={p_b.size} [{label}]")
        polished.append((sc_b, p_b, n, label + '_ad'))

    polished.sort(key=lambda x: x[0])
    print(f"  Phase 2 done. Best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # Phase 3: LP refinement
    print("Phase 3: LP Refinement")
    phase3_end = t_start + 1100.0
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase3_end:
        n_lp = best_seq.size
        per_lp = min(phase3_end - time.monotonic(), 250.0)
        if per_lp > 20:
            p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + per_lp,
                                      t_init=0.35, max_solves=20)
            if sc_lp < best_score:
                best_score = float(sc_lp)
                best_seq = np.asarray(p_lp, dtype=np.float64).copy()
                print(f"  LP: {sc_lp:.10f} n={p_lp.size}")

            # Adam after LP
            if time.monotonic() < phase3_end - 15:
                per2 = min(phase3_end - time.monotonic() - 5, 80.0)
                if per2 > 10:
                    p_ap, sc_ap = _logit_adam(best_seq, time.monotonic() + per2, rng,
                                              lr_init=0.04, temp_init=0.0,
                                              temp_decay=0.95, no_imp_limit=200)
                    if sc_ap < best_score:
                        best_score = float(sc_ap)
                        best_seq = np.asarray(p_ap, dtype=np.float64).copy()
                        print(f"  Post-LP: {sc_ap:.10f} n={p_ap.size}")

    # Phase 4: Multi-scale escalation
    print("Phase 4: Multi-scale escalation")
    phase4_end = t_start + 1450.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_cur = best_seq.size
        escalations = [2048, 4096]
        for tn in escalations:
            if tn <= n_cur:
                continue
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            sc_int = _score(p_int)
            print(f"  Escalate n={tn}: {sc_int:.10f}")
            if sc_int < best_score:
                best_score = float(sc_int)
                best_seq = np.asarray(p_int, dtype=np.float64).copy()

            per_e = min(120.0, phase4_end - time.monotonic() - 30)
            if per_e > 15:
                p_e, sc_e = _logit_adam(p_int, time.monotonic() + per_e, rng,
                                         lr_init=0.05, temp_init=0.0,
                                         temp_decay=0.95, no_imp_limit=250)
                if sc_e < best_score:
                    best_score = float(sc_e)
                    best_seq = np.asarray(p_e, dtype=np.float64).copy()
                    print(f"  Escalated: {sc_e:.10f} n={p_e.size}")

            # Quick LP at new scale if feasible
            if best_seq.size <= 2048 and time.monotonic() < phase4_end - 30:
                per_el = min(60.0, phase4_end - time.monotonic() - 10)
                if per_el > 10:
                    p_el, sc_el = _lp_refine(best_seq, time.monotonic() + per_el,
                                              t_init=0.3, max_solves=8)
                    if sc_el < best_score:
                        best_score = float(sc_el)
                        best_seq = np.asarray(p_el, dtype=np.float64).copy()
                        print(f"  Escalate LP: {sc_el:.10f} n={p_el.size}")

    # Phase 5: Multi-restart
    print("Phase 5: Multi-restart")
    phase5_end = deadline - 60.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts + 1))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 10:
                break
            z0 = np.log(np.maximum(best_seq, 1e-12))
            z0 -= z0.mean()
            ns = 0.05 + 0.1 * i
            z_r = z0 + ns * rng.normal(size=n_best)
            p_s = _softmax_from_z(z_r)
            p_r, sc_r = _logit_adam(p_s, time.monotonic() + per, rng,
                                     lr_init=0.05, temp_init=0.0,
                                     temp_decay=0.95, no_imp_limit=
                                     max(100, int(0.2 * n_best)))
            if sc_r < best_score:
                best_score = float(sc_r)
                best_seq = np.asarray(p_r, dtype=np.float64).copy()
                print(f"  Restart {i}: {sc_r:.10f} n={p_r.size}")

    # Phase 6: Coordinate descent + final polish
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Coordinate descent
        per_cd = min(60.0, deadline - time.monotonic() - 10)
        if per_cd > 10:
            p_cd, sc_cd = _coord_descent(best_seq, time.monotonic() + per_cd, max_sweeps=50)
            if sc_cd < best_score:
                best_score = float(sc_cd)
                best_seq = np.asarray(p_cd, dtype=np.float64).copy()
                print(f"  CoordDesc: {sc_cd:.10f} n={p_cd.size}")

        # Final Adam
        per_f = deadline - time.monotonic() - 3
        if per_f > 8:
            p_f, sc_f = _logit_adam(best_seq, time.monotonic() + per_f, rng,
                                     lr_init=0.02, temp_init=0.0,
                                     temp_decay=0.97, no_imp_limit=100)
            if sc_f < best_score:
                best_score = float(sc_f)
                best_seq = np.asarray(p_f, dtype=np.float64).copy()
                print(f"  Final: {sc_f:.10f} n={p_f.size}")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(1024) / 1024

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = np.ones(1024) / 1024
    print("FALLBACK: uniform n=1024")
    return [float(x) for x in fb]
