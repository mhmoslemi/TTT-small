import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(n), c
    w = ex / s
    val = mval + beta * float(np.log(s))
    # Gradient via FFT
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, c


def _exact_subgrad(a):
    """Compute exact subgradient at the max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())

    threshold = c_max - 1e-10

        peak_idx = np.array([np.argmax(c)])
    # Random tie-breaking
    w = np.random.dirichlet(np.ones(len(peak_idx)))
    # Build weight vector
    full_w = np.zeros(2 * n - 1)
    for i, idx in enumerate(peak_idx):

    # Gradient
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(full_w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return c_max, grad, c


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):

    b = np.full(2 * n - 1, M)

    if res.success:

    return None


def _init_flat_spectrum(n, rng):
    """Initialize from flat power spectrum."""
    N = _next_pow2(2 * n - 1)
    # We want a signal of length n whose zero-padded FFT has flat magnitude
    # Start with random phases
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = np.sqrt(N)  # DC component
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k]) * (N / (N - 1)) ** 0.5
        X[N - k] = np.exp(-1j * phases[k]) * (N / (N - 1)) ** 0.5
    a_full = np.real(np.fft.ifft(X))
    a = a_full[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

def _nesterov_optimize(a, deadline, max_iter=100000, lr_init=0.03,

    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    lr = lr_init
    mu = 0.9  # momentum parameter
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Evaluate at Nesterov point
        z_eval = (1 + mu) * z_y - mu * z
        z_eval = np.clip(z_eval, -60.0, 60.0)
        z_eval -= float(np.mean(z_eval))
        e = np.exp(z_eval)
        a_eval = e / e.sum()

        gz = a_eval * (vp / s - dot / s)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z_y = best_z.copy()
            z = best_z.copy()
            continue

        z_new = z_y - mu * (z_y - z)

                    beta = max(1e-10, beta * 0.6)

                    beta = min(0.3 * c_max_new, beta * 1.3)

def _fourier_refine(a, deadline, max_iter=5000, lr=0.01):
    """
    Refine by working in Fourier domain.
    Minimize max |A_k|^2 by adjusting phases (keeping magnitudes roughly fixed).
    This is like a projected gradient in Fourier domain.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, N)
        # Gradient of max |A_k|^2 with respect to a
        # max |A_k|^2 -> subgradient at argmax
        mag2 = np.abs(A[:N//2+1])**2
        k_max = np.argmax(mag2)
        
        # d/d a_j |A_k|^2 = 2 * Re{ A_k * exp(-2*pi*i*k*j/N) }
        # For the linear convolution, it's slightly different due to zero-padding
        # But the gradient through the FFT is:
        # grad_a = 2 * ifft(conj(A) * e_k) where e_k has 1 at position k_max
        # Actually for real signal: grad = 2 * real(ifft(A_conj * delta_k))
        
        # Simpler: use the time-domain gradient
        c = _fft_convolve(a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))
        
        # Subgradient: 2 * a convolved with delta at peak_idx
        # (a * delta_k)(j) = a(j-k) if valid, else 0
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[peak_idx - j]
        
        # Project gradient to simplex tangent
        s = a.sum()
        dot = np.dot(a, grad)
        gz = a * (grad - dot / s)
        
        a_new = a - lr * gz / s
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        score = float(2.0 * n * _fft_convolve(a_new).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        if it % 1000 == 0:
            lr = max(0.001, lr * 0.95)
    
    return best_a

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_grad(a, beta, n_top=1):
    """Compute smooth max of conv(a,a) and gradient. Uses top-n peaks weighting."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(n), c
    w = ex / s
    val = mval + beta * float(np.log(s))
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, c


def _multi_peak_subgrad(a, n_peaks=3):
    """Compute subgradient using top n_peaks with Dirichlet weights."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    # Find top peaks within a small tolerance
    tol = c_max * 1e-6 + 1e-12
    peak_indices = np.where(c >= c_max - tol)[0]
    if len(peak_indices) == 0:
        peak_indices = np.array([int(np.argmax(c))])
    if len(peak_indices) > n_peaks:
        # Take the top n_peaks
        top_sorted = peak_indices[np.argsort(c[peak_indices])[::-1]][:n_peaks]
        peak_indices = top_sorted
    # Dirichlet weights
    w = np.random.dirichlet(np.ones(len(peak_indices)))
    full_w = np.zeros(2 * n - 1)
    for i, idx in enumerate(peak_indices):
        full_w[idx] = w[i]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(full_w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return c_max, grad, c


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _init_cosine_mix(n, rng):
    """Initialize as a mix of low-frequency cosines with random phases."""
    a = np.ones(n) / n
    n_harm = min(5, n // 4)
    for k in range(1, n_harm + 1):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4) / k
        t = np.arange(n)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _init_parent_interp(parent, n_target):
    """Interpolate parent sequence to target length."""
    n_src = len(parent)
    x0 = np.linspace(0.0, 1.0, n_src)
    x1 = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x1, x0, parent)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _nesterov_smooth(a, deadline, max_iter=200000, lr_init=0.04,
                      beta_init=None, n_top=1, use_multi_peak=False):
    """
    Nesterov-accelerated gradient descent in log-space with adaptive smooth max.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))
    z_y = z.copy()

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.05 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    lr = lr_init
    mu = 0.85
    rng = np.random.default_rng(42)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov evaluation point
        z_eval = (1.0 + mu) * z_y - mu * z
        z_eval = np.clip(z_eval, -60.0, 60.0)
        z_eval -= float(np.mean(z_eval))
        e = np.exp(z_eval)
        a_eval = e / e.sum()

        # Compute gradient
        if use_multi_peak and it > 500:
            _, grad_c, c = _multi_peak_subgrad(a_eval, n_peaks=2)
        else:
            _, grad_c, c = _smooth_max_grad(a_eval, beta, n_top)

        # Map to log-space gradient
        s = float(a_eval.sum())
        vp = grad_c + 1e-8 * a_eval
        dot = float(np.dot(a_eval, vp))
        gz = a_eval * (vp / s - dot / s)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            z_y = best_z.copy()
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            continue

        # Nesterov update
        z_y = z_y - lr * gz
        z_new = z_y - mu * (z_y - z)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        c_max_new = float(c_new.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 3000 == 0:
                lr = max(0.002, lr * 0.92)

        # Adaptive beta
        if it % 200 == 0 and it > 200:
            sorted_c = np.sort(c_new)[::-1]
            top5 = sorted_c[:min(5, len(sorted_c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 4.0:
                    beta = min(0.4 * c_max_new, beta * 1.2)

        # Annealing restart
        if no_imp > max(6000, 8 * n):
            noise = 0.15 * best_score * rng.standard_normal(n)
            z_restart = best_z + noise
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            z = z_restart.copy()
            z_y = z_restart.copy()
            lr = lr_init
            beta = max(1e-6, 0.05 * float(_fft_convolve(
                np.exp(z_restart) / np.exp(z_restart).sum()).max()))
            no_imp = 0

        z = z_new

    return best_a


def _adam_logit(a, deadline, max_iter=100000, lr_init=0.03, beta_init=None):
    """Adam optimizer in log-space with adaptive smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.04 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    lr = lr_init
    rng = np.random.default_rng(7)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        _, grad_c, c = _smooth_max_grad(p, beta)

        s = float(p.sum())
        vp = grad_c + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp / s - dot / s)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        c_max_new = float(c_new.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.001, lr * 0.93)

        # Adaptive beta
        if it % 300 == 0 and it > 300:
            sorted_c = np.sort(c_new)[::-1]
            top5 = sorted_c[:min(5, len(sorted_c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.65)
                elif gap > beta * 4.0:
                    beta = min(0.3 * c_max_new, beta * 1.15)

        # Restart
        if no_imp > max(5000, 6 * n):
            noise = 0.2 * best_score * rng.standard_normal(n)
            z_restart = best_z + noise
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            z = z_restart.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = lr_init
            beta = max(1e-6, 0.04 * float(_fft_convolve(
                np.exp(z_restart) / np.exp(z_restart).sum()).max()))
            no_imp = 0

        z = z_new

    return best_a


def _lp_refine(a, deadline, max_iter=3000, inflate_start=1.0005, blend_start=0.25):
    """LP-based spreading refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    blend = blend_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate = min(1.01, inflate * 1.08)
            if no_imp > 80:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 15 == 0:
                blend = max(0.05, blend * 0.92)
                inflate = max(1.0001, inflate * 0.97)
            if no_imp > max(80, n // 4):
                a = best_a.copy()
                blend = blend_start
                inflate = inflate_start
                no_imp = 0

        a = a_new

    return best_a


def _polyak_subgrad(a, deadline, max_iter=50000, lr_init=0.015):
    """Polyak-averaged subgradient method in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= np.mean(z)
    z_avg = z.copy()

    c = _fft_convolve(a)
    M_curr = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * M_curr)
    best_z = z.copy()
    lr = lr_init
    no_imp = 0
    rng = np.random.default_rng(99)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, grad_c, c = _multi_peak_subgrad(a, n_peaks=2)
        M_curr = float(c.max())

        s = float(a.sum())
        vp = grad_c + 1e-9 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp / s - dot / s)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        z_avg = (it * z_avg + z) / (it + 1)

        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        # Check averaged iterate
        e_avg = np.exp(z_avg - np.mean(z_avg))
        a_avg = e_avg / e_avg.sum()
        c_avg = _fft_convolve(a_avg)
        score_avg = float(2.0 * n * c_avg.max())

        if score_avg < best_score - 1e-14:
            best_score = score_avg
            best_a = a_avg.copy()
            best_z = z_avg.copy()
            no_imp = 0
        elif score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 5000 == 0:
                lr = max(0.001, lr * 0.95)

        if no_imp > max(8000, 10 * n):
            break

    return best_a


def _fine_grind(a, deadline, max_iter=30000, lr_init=0.005):
    """Very fine gradient grind with tiny beta for final polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.01 * c_max)

    z = np.log(a)
    z -= np.mean(z)
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    lr = lr_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, grad_c, c = _smooth_max_grad(a, beta)
        s = float(a.sum())
        vp = grad_c + 1e-9 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp / s - dot / s)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.0005, lr * 0.9)
                beta = max(1e-12, beta * 0.7)

        if it % 500 == 0:
            beta = max(1e-12, beta * 0.85)

        if no_imp > max(5000, 5 * n):
            break

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Combines Nesterov/Adam gradient methods, LP spreading, and subgradient averaging
    across multiple sequence lengths with spectral initialization.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths: focus on larger n for better bounds
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [1024, 1536, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    print(f"Lengths to try: {lengths}", flush=True)

    # Phase 1: Generate and evaluate initializations
    print(f"Phase 1: Initialization (budget={deadline - time.monotonic():.0f}s)", flush=True)
    candidates = []

    for n_t in lengths:
        # Flat spectrum
        a_spec = _init_flat_spectrum(n_t, rng)
        candidates.append((f"spec_{n_t}", n_t, a_spec))

        # Cosine mix
        a_cos = _init_cosine_mix(n_t, rng)
        candidates.append((f"cos_{n_t}", n_t, a_cos))

        # Random
        a_rand = rng.random(n_t)
        a_rand = a_rand / a_rand.sum()
        candidates.append((f"rand_{n_t}", n_t, a_rand))

        # Parent interpolation
        if parent is not None:
            a_par = _init_parent_interp(parent, n_t)
            candidates.append((f"par_{n_t}", n_t, a_par))

    # Quick scoring
    for i, (name, n_t, a0) in enumerate(candidates):
        sc = _score(a0)
        candidates[i] = (name, n_t, a0, sc)

    candidates.sort(key=lambda x: x[3])
    print("  Top initializations:", flush=True)
    for name, n_t, _, sc in candidates[:6]:
        print(f"    {name}: {sc:.8f}", flush=True)

    # Phase 2: Main optimization on top candidates
    # Allocate time: spend most time on the best 2-3 candidates
    n_top = min(3, len(candidates))
    # Reserved time for final phases
    reserved = 250.0
    avail = max(60.0, deadline - t_start - reserved)
    time_per = avail / n_top

    for idx in range(n_top):
        if time.monotonic() > deadline - reserved:
            break
        name, n_t, a0, sc0 = candidates[idx]
        if sc0 > best_score * 1.5:
            continue

        # Sub-deadline for this candidate
        sub_dead = min(deadline - reserved, time.monotonic() + time_per * 1.8)
        print(f"Phase 2: {name} (n={n_t}) start={sc0:.8f} budget={sub_dead - time.monotonic():.0f}s", flush=True)

        # Strategy A: Nesterov with smooth max
        a_opt = _nesterov_smooth(a0, sub_dead, max_iter=150000, lr_init=0.04)
        sc_opt = _score(a_opt)
        print(f"  Nesterov: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < deadline - 150:
            lp_dead = min(deadline - 130, time.monotonic() + 50)
            if lp_dead - time.monotonic() > 10:
                a_lp = _lp_refine(a_opt, lp_dead, max_iter=2000, inflate_start=1.0004, blend_start=0.2)
                sc_lp = _score(a_lp)
                print(f"  LP: {sc_lp:.10f}", flush=True)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                a_opt = a_lp

        # Strategy B: Adam on the LP result
        if time.monotonic() < deadline - 100:
            adam_dead = min(deadline - 80, time.monotonic() + 40)
            if adam_dead - time.monotonic() > 10:
                a_adam = _adam_logit(a_opt, adam_dead, max_iter=50000, lr_init=0.02)
                sc_adam = _score(a_adam)
                print(f"  Adam2: {sc_adam:.10f}", flush=True)
                if sc_adam < best_score:
                    best_score = sc_adam
                    best_seq = a_adam.copy()
                a_opt = a_adam

    # Phase 3: On the overall best, do iterative refinement
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"Phase 3: Iterative refinement (current best={best_score:.10f})", flush=True)

        for round_i in range(3):
            if time.monotonic() > deadline - 60:
                break

            # Nesterov with smaller lr
            sub_dead = min(deadline - 60, time.monotonic() + 50)
            if sub_dead - time.monotonic() > 10:
                a_r = _nesterov_smooth(best_seq, sub_dead, max_iter=30000, lr_init=0.015,
                                       use_multi_peak=True)
                sc_r = _score(a_r)
                print(f"  Round {round_i} Nesterov: {sc_r:.10f}", flush=True)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()

            # LP
            if time.monotonic() < deadline - 40:
                lp_dead = min(deadline - 30, time.monotonic() + 25)
                if lp_dead - time.monotonic() > 8:
                    a_lp = _lp_refine(best_seq, lp_dead, max_iter=1000,
                                      inflate_start=1.0002, blend_start=0.1)
                    sc_lp = _score(a_lp)
                    print(f"  Round {round_i} LP: {sc_lp:.10f}", flush=True)
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_seq = a_lp.copy()

            # Polyak subgradient
            if time.monotonic() < deadline - 30:
                sg_dead = min(deadline - 20, time.monotonic() + 20)
                if sg_dead - time.monotonic() > 8:
                    a_sg = _polyak_subgrad(best_seq, sg_dead, max_iter=20000, lr_init=0.008)
                    sc_sg = _score(a_sg)
                    print(f"  Round {round_i} Polyak: {sc_sg:.10f}", flush=True)
                    if sc_sg < best_score:
                        best_score = sc_sg
                        best_seq = a_sg.copy()

    # Phase 4: Final fine grind
    if best_seq is not None and time.monotonic() < deadline - 15:
        fg_dead = min(deadline - 10, time.monotonic() + 30)
        if fg_dead - time.monotonic() > 5:
            print(f"Phase 4: Fine grind (budget={fg_dead - time.monotonic():.0f}s)", flush=True)
            a_fg = _fine_grind(best_seq, fg_dead, max_iter=20000, lr_init=0.003)
            sc_fg = _score(a_fg)
            print(f"  Fine grind: {sc_fg:.10f}", flush=True)
            if sc_fg < best_score:
                best_score = sc_fg
                best_seq = a_fg.copy()

        # Final LP squeeze
        if time.monotonic() < deadline - 5:
            lp_final = min(deadline - 3, time.monotonic() + 10)
            if lp_final - time.monotonic() > 5:
                a_fl = _lp_refine(best_seq, lp_final, max_iter=500,
                                  inflate_start=1.0001, blend_start=0.05)
                sc_fl = _score(a_fl)
                print(f"  Final LP: {sc_fl:.10f}", flush=True)
                if sc_fl < best_score:
                    best_score = sc_fl
                    best_seq = a_fl.copy()

    # Fallback
    if best_seq is None:
        n_fb = 1024
        best_seq = _init_flat_spectrum(n_fb, rng)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final < 0.01:
        final_seq = np.ones(512)
        best_score = _score(final_seq)

    # Validate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
