"""
Multi-resolution optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LSE/subgradient with Barzilai-Borwein step sizes,
multi-resolution upsampling, structured perturbations, and LP polishing.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score with sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _gradient_at_peak(a):
    """Compute exact subgradient of max conv at the maximizing lag."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    # Find all lags at (or very near) the max
    tol = max(1e-10, m_val * 1e-8)
    active = np.where(c >= m_val - tol)[0]
    # Use uniform weight over active lags
    w = np.zeros_like(c)
    w[active] = 1.0 / len(active)
    # Gradient of conv at lag k: d/d a[j] conv[k] = 2*a[k-j] if 0<=k-j<n
    # For weighted average: grad[j] = 2 * sum_k w[k] * a[k-j] (if 0<=k-j<n)
    # This is 2 * (w * a_reversed)[centered]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # conv(w, a) gives sum_k w[k]*a[k-j] for each j
    # Actually we need: for each j, sum_k w[k] * a[k-j] = (reverse_convolve)(w, a)[j]
    # conv(w, a)[j] = sum_k w[k] * a[j-k], which is what we want (with a[j-k] = a[k'-j] where k'=k... )
    # Let me think: we want grad[j] = 2 * sum_k w[k] * a[k-j] (for 0<=k-j<n)
    # conv(w, a)[m] = sum_k w[k] * a[m-k]
    # Set m-k = k-j => k = (m+j)/2... that's not right.
    # Actually: grad[j] = 2 * sum_k w[k] * a[k-j] 
    # Let i = k-j, so k = i+j: grad[j] = 2 * sum_i w[i+j] * a[i]
    # This is (conv(w, a))[j] reversed? Let's just compute directly via FFT.
    # grad[j] = 2 * sum_k w[k]*a[k-j] = 2 * (w cross-correlate with a)[j]
    # cross-correlation w ⊛ a at lag j = sum_k w[k] * a[k-j]
    # In FFT: IFFT(FFT(w) * conj(FFT(a))) gives cross-correlation
    # But for real signals: IFFT(FFT(w) * FFT(a)) gives convolution, not cross-corr
    # cross-corr(w, a)[j] = conv(w, a_reversed)[j + n - 1]
    # Or: cross-corr(w,a)[j] = sum_k w[k] a[k-j]
    # Let's use: if we define b[k] = w[k], then cross-corr(b, a)[j] = sum_k b[k] a[k-j]
    # FFT of cross-corr = conj(FFT(b)) * FFT(a) ... or FFT(b) * conj(FFT(a))?
    # For real: cross-corr(x,y)[n] = sum_k x[k+n] y[k] = (x * y_flipped)[n]
    # Hmm let me just compute it directly for correctness.
    grad = np.zeros(n)
    for k in active:
        # d/d a[j] of conv[k] = 2*a[k-j] if 0 <= k-j < n
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx] / len(active)
    return grad, m_val, active


def _gradient_at_peak_fast(a):
    """Fast gradient using FFT. grad[j] = 2*sum_k w[k]*a[k-j] for 0<=k-j<n."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    tol = max(1e-10, m_val * 1e-8)
    active = np.where(c >= m_val - tol)[0]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    # grad[j] = 2 * sum_{k: valid} w[k] * a[k-j] where 0<=k-j<n and 0<=j<n
    # = 2 * sum_{k} w[k] * a[k-j] * [0<=k-j<n]
    # Let's compute h[j] = sum_k w[k]*a[k-j] for j=0..n-1
    # This is the cross-correlation of w and a: h[j] = sum_k w[k] a[k-j]
    # M = w ⊛ a (cross-corr)
    # FFT approach: M[j] = IFFT(FFT(w) * conj(FFT(a)))[j + n - 1] ... 
    # Actually for cross-corr: (x ⊛ y)[n] = sum_k x[k+n] y[k]
    # We want h[j] = sum_k w[k] a[k-j] = sum_k w[k] a[k-j]
    # Let m = -j: h[j] = sum_k w[k] a[k-j]
    # Define x = w, y = a. Then (x ⊛ y)[m] = sum_k x[k+m] y[k]
    # Set k+m = k' => m = k'-k: (x⊛y)[m] = sum_{k'} x[k'] y[k'-m]
    # So (x⊛y)[-j] = sum_k x[k] y[k+j]... that's not quite right either.
    # Let me just use the direct formula via FFT:
    # h[j] = sum_k w[k] * a[k-j] for 0<=k-j<n, 0<=j<n
    # = sum_{i=0}^{n-1} a[i] * w[i+j] (substituting k=i+j)
    # = (w convolved with a_flipped)[j + n - 1] ... 
    # OK let me just use the convolution directly:
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # conv(w, a)[m] = sum_k w[k] a[m-k]
    # We want h[j] = sum_k w[k] a[k-j]
    # conv(w, a)[?] ... set m-k = k-j => m = 2k-j, not constant.
    # Alternative: h[j] = sum_k w[k] a[k-j] = (w * reverse(a))[j + n - 1] (circular)
    # Let ra = reverse(a), ra[i] = a[n-1-i]
    # conv(w, ra)[m] = sum_k w[k] ra[m-k] = sum_k w[k] a[n-1-(m-k)] = sum_k w[k] a[n-1-m+k]
    # Set n-1-m+k = k-j => n-1-m = -j => m = n-1+j
    # So h[j] = conv(w, ra)[n-1+j] (truncated to valid range)
    ra = a[::-1]
    fra = np.fft.rfft(ra, N)
    h_full = np.fft.irfft(fw * fra, N)
    # h[j] = h_full[n-1+j] for j=0..n-1
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, m_val, active


def _project_simplex(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _barzilai_borwein(s, y, mu):
    """Barzilai-Borwein step size."""
    sy = float(np.dot(s, y))
    if abs(sy) < 1e-15:
        return mu
    ss = float(np.dot(s, s))
    if ss < 1e-15:
        return mu
    return max(1e-10, min(1.0, sy / ss))


def _lse_subgrad_optimize(a, deadline, max_iter=200000, T_start=0.03, T_end=1e-5,
                          lr_init=0.08, use_bb=True, alternate_period=50):
    """
    Combined LSE-smoothed and exact subgradient optimization with
    Barzilai-Borwein adaptive step size and alternating objectives.
    Works on simplex (sum=1, a>=0).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    # State for BB
    prev_a = a.copy()
    prev_grad = np.zeros(n)
    bb_step = lr_init

    # Adam state
    adam_m = np.zeros(n)
    adam_v = np.zeros(n)
    adam_t = 0

    no_imp = 0
    base_lr = lr_init
    T = T_start
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Choose mode: LSE-smoothed vs exact subgradient
        use_exact = (it % alternate_period >= alternate_period // 2)

        c = _fft_convolve(a)
        m_val = float(c.max())

        if use_exact:
            # Exact subgradient at peak
            grad, m_val2, active = _gradient_at_peak_fast(a)
        else:
            # LSE-smoothed gradient
            x = (c - m_val) / T
            x = np.clip(x, -50, 50)
            ex = np.exp(x)
            s_ex = float(ex.sum())
            if not np.isfinite(s_ex) or s_ex <= 0:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ex / s_ex
            N = _next_pow2(2 * n - 1)
            fw = np.fft.rfft(w, N)
            fa = np.fft.rfft(a, N)
            ra = a[::-1]
            fra = np.fft.rfft(ra, N)
            h_full = np.fft.irfft(fw * fra, N)
            h = h_full[n - 1:2 * n - 1]
            grad = 2.0 * h

        # Project gradient onto tangent space of simplex
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a  # Actually this is wrong for simplex
        # For simplex, the tangent space is {d: sum(d)=0}
        # Project: grad_tan = grad - mean(grad)
        grad_tan = grad - float(np.mean(grad))

        if not np.all(np.isfinite(grad_tan)):
            a = best_a.copy()
            continue

        # Barzilai-Borwein step size
        if use_bb and it > 0:
            s_vec = a - prev_a
            y_vec = grad_tan - prev_grad
            bb_step = _barzilai_borwein(s_vec, y_vec, bb_step)
            # Blend BB with base_lr for stability
            step = 0.5 * bb_step + 0.5 * base_lr
        else:
            step = base_lr

        prev_a = a.copy()
        prev_grad = grad_tan.copy()

        # Adam update
        adam_t += 1
        b1, b2 = 0.9, 0.999
        adam_m = b1 * adam_m + (1 - b1) * grad_tan
        adam_v = b2 * adam_v + (1 - b2) * (grad_tan * grad_tan)
        mh = adam_m / (1 - b1 ** adam_t)
        vh = adam_v / (1 - b2 ** adam_t)
        a_new = a - step * mh / (np.sqrt(vh) + 1e-8)

        # Project onto simplex
        a_new = _project_simplex(a_new)

        # Evaluate
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                base_lr = max(0.005, base_lr * 0.95)
            if no_imp > max(3000, 5 * n):
                # Restart with perturbation
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = _normalize(a)
                adam_m = np.zeros(n)
                adam_v = np.zeros(n)
                adam_t = 0
                base_lr = lr_init
                no_imp = 0

        T *= T_ratio
        T = max(T, T_end)

    return best_a


def _fw_refine(a, deadline, max_iter=5000):
    """
    Frank-Wolfe (conditional gradient) refinement.
    For the max of quadratic forms, use subgradient at active lag.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        tol = max(1e-12, m_val * 1e-6)
        active = np.where(c >= m_val - tol)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # For each active lag k, the gradient contribution at position j is 2*a[k-j]
        # Frank-Wolfe: find vertex e_i minimizing grad · e_i = grad[i]
        # grad[i] = sum_{k in active} (1/|active|) * 2*a[k-i] (if 0<=k-i<n)
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx] / len(active)

        # FW step: move towards e_{argmin grad}
        target = int(np.argmin(grad))

        # Gap: grad · a - min(grad)
        gap = float(np.dot(grad, a)) - float(np.min(grad))
        if gap < 1e-10:
            break

        # Step size: 2/(it+2) or based on gap
        t_step = 2.0 / (it + 2)
        a_new = (1 - t_step) * a + t_step * np.zeros(n)
        a_new[target] += t_step

        # Scale back to simplex (already on simplex since we blend with a vertex)
        # a is on simplex, e_target is a vertex, so blend is on simplex
        a_new = _normalize(a_new)

        score_new = _score_norm(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                break

    return best_a


def _make_conv_matrix_reduced(a, M, top_k=80):
    """Create reduced constraint matrix using near-active constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if n > 128:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j+n, j] = a
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=3000, inflate=1.0003, blend=0.15):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            inflate_cur = min(1.01, inflate_cur * 1.15)
            no_imp += 5
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.008, inflate_cur * 1.05)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = blend_cur
        best_sc = _score_norm(a)
        for t in [blend_cur * 0.5, blend_cur, blend_cur * 1.5, blend_cur * 2.0]:
            a_try = _normalize((1 - t) * a + t * g)
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        a_new = _normalize((1 - best_t) * a + best_t * g)
        score_new = _score_norm(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.02, blend_cur * 0.9)
                inflate_cur = min(1.006, inflate_cur * 0.9)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    # Use cubic-like interpolation via FFT upsampling
    x0 = np.linspace(0.0, 1.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 1.0, n_new, endpoint=False)
    p = np.interp(x1, x0, np.append(p0, p0[0]))
    p = np.maximum(p, 0.0)
    return _normalize(p)


def _make_profiles(n, rng):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    profiles = []

    # Gaussian
    a = np.exp(-0.5 * ((t - 0.5) / 0.25) ** 2)
    profiles.append(_normalize(a))

    # Raised cosine
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    profiles.append(_normalize(np.maximum(a, 1e-10)))

    # Flat
    profiles.append(np.ones(n) / n)

    # Slightly asymmetric
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.3 * np.sin(np.pi * t)
    profiles.append(_normalize(np.maximum(a, 1e-10)))

    # Random smooth
    k = max(2, n // 32)
    freq = np.fft.rfftfreq(n)
    mask = freq < 1.0 / k
    spec = np.where(mask, rng.normal(size=n // 2 + 1), 0.0)
    spec[0] = n  # DC dominant
    a = np.fft.irfft(spec * n, n)
    profiles.append(_normalize(np.maximum(a, 1e-10)))

    # Bump (concentrated in middle)
    center = n // 2
    a = np.exp(-0.5 * ((np.arange(n) - center) / (n * 0.15)) ** 2)
    profiles.append(_normalize(a))

    return profiles


def _multi_resolution_optimize(parent, deadline, rng):
    """
    Multi-resolution: optimize at small n, upsample, optimize at large n.
    """
    best_a = parent.copy() if parent is not None else None
    best_score = _score(best_a) if best_a is not None else np.inf
    n_parent = parent.size if parent is not None else 1024

    # Determine target lengths
    target_n = n_parent
    small_n = max(64, target_n // 8)

    if best_a is not None:
        # Downsample to small_n
        a_small = _interpolate_profile(best_a, small_n)
        sc_small = _score(a_small)
        print(f"  Downsampled to n={small_n}: score={sc_small:.8f}")

        # Optimize at small_n (fast)
        small_budget = min(60.0, (deadline - time.monotonic()) * 0.15)
        if small_budget > 10:
            small_deadline = time.monotonic() + small_budget
            a_small_opt = _lse_subgrad_optimize(
                a_small, small_deadline, max_iter=50000,
                T_start=0.02, T_end=1e-5, lr_init=0.06)
            sc_small_opt = _score(a_small_opt)
            print(f"  Small-n optimized: score={sc_small_opt:.8f}")

            # Upsample back
            a_up = _interpolate_profile(a_small_opt, target_n)
            sc_up = _score(a_up)
            print(f"  Upsampled to n={target_n}: score={sc_up:.8f}")

            if sc_up < best_score:
                best_score = sc_up
                best_a = a_up

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-resolution optimization combining:
    - LSE/subgradient with Barzilai-Borwein
    - Frank-Wolfe refinement
    - LP spreading
    - Multi-resolution upsampling
    - Structured perturbations
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent found, using uniform n=1024, score={best_score:.6f}")

    n_target = parent.size
    print(f"Target length: {n_target}")

    # === Phase 1: Multi-resolution ===
    print("\n=== Phase 1: Multi-resolution ===")
    if time.monotonic() < deadline - 600:
        best_seq, best_score = _multi_resolution_optimize(parent, deadline, rng)
        print(f"  After multi-res: score={best_score:.10f}")

    # === Phase 2: Main LSE/subgradient optimization ===
    print("\n=== Phase 2: LSE/subgradient optimization ===")
    a_cur = best_seq.copy()

    # Generate a few diverse init candidates
    candidates = [a_cur]
    profiles = _make_profiles(n_target, rng)
    for p in profiles[:3]:
        candidates.append(p)
        sc = _score(p)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()

    # Optimize the best candidate with extended budget
    main_deadline = min(deadline - 200, time.monotonic() + 500)
    a_opt = _lse_subgrad_optimize(a_cur, main_deadline, max_iter=300000,
                                  T_start=0.03, T_end=1e-5, lr_init=0.08)
    sc_opt = _score(a_opt)
    print(f"  LSE/subgrad: score={sc_opt:.10f}")
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()

    # Try a second diverse candidate if time allows
    if time.monotonic() < deadline - 350:
        a2 = _make_profiles(n_target, np.random.default_rng(seed + 1))[1]
        # Perturb the main result
        a2 = _normalize(0.7 * best_seq + 0.3 * a2)
        sc2_init = _score(a2)
        print(f"  Diverse candidate init: score={sc2_init:.10f}")
        if sc2_init < best_score:
            best_score = sc2_init
            best_seq = a2.copy()

        div_deadline = min(deadline - 300, time.monotonic() + 200)
        a2_opt = _lse_subgrad_optimize(a2, div_deadline, max_iter=100000,
                                       T_start=0.02, T_end=1e-5, lr_init=0.06)
        sc2 = _score(a2_opt)
        print(f"  Diverse optimized: score={sc2:.10f}")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2_opt.copy()

    # === Phase 3: Frank-Wolfe refinement ===
    print("\n=== Phase 3: Frank-Wolfe refinement ===")
    if time.monotonic() < deadline - 200:
        fw_deadline = min(deadline - 200, time.monotonic() + 80)
        a_fw = _fw_refine(best_seq, fw_deadline, max_iter=3000)
        sc_fw = _score(a_fw)
        print(f"  FW: score={sc_fw:.10f}")
        if sc_fw < best_score:
            best_score = sc_fw
            best_seq = a_fw.copy()

    # === Phase 4: LP refinement ===
    print("\n=== Phase 4: LP refinement ===")
    if time.monotonic() < deadline - 100:
        lp_deadline = min(deadline - 100, time.monotonic() + 150)
        a_lp = _lp_refine(best_seq, lp_deadline, max_iter=2000,
                          inflate=1.0003, blend=0.12)
        sc_lp = _score(a_lp)
        print(f"  LP: score={sc_lp:.10f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

    # === Phase 5: Second LSE pass on best ===
    print("\n=== Phase 5: Final LSE polish ===")
    if time.monotonic() < deadline - 50:
        lse_deadline = min(deadline - 40, time.monotonic() + 120)
        a_final_lse = _lse_subgrad_optimize(best_seq, lse_deadline, max_iter=100000,
                                            T_start=0.01, T_end=1e-6, lr_init=0.04)
        sc_final = _score(a_final_lse)
        print(f"  Final LSE: score={sc_final:.10f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final_lse.copy()

    # === Phase 6: Final LP micro-polish ===
    if time.monotonic() < deadline - 15:
        lp2_deadline = min(deadline - 10, time.monotonic() + 50)
        a_lp2 = _lp_refine(best_seq, lp2_deadline, max_iter=1000,
                           inflate=1.0001, blend=0.05)
        sc_lp2 = _score(a_lp2)
        print(f"  Micro LP: score={sc_lp2:.10f}")
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()

    # Final: try different length if time permits
    if time.monotonic() < deadline - 60 and n_target != 2048:
        print("\n=== Phase 7: Try n=2048 ===")
        n_alt = 2048
        a_alt = _interpolate_profile(best_seq, n_alt)
        sc_alt = _score(a_alt)
        print(f"  n={n_alt} init: score={sc_alt:.10f}")
        if sc_alt < best_score + 0.02:  # Only worth trying if competitive
            alt_deadline = min(deadline - 30, time.monotonic() + 80)
            a_alt_opt = _lse_subgrad_optimize(a_alt, alt_deadline, max_iter=50000,
                                              T_start=0.01, T_end=1e-5, lr_init=0.05)
            sc_alt_opt = _score(a_alt_opt)
            print(f"  n={n_alt} optimized: score={sc_alt_opt:.10f}")
            if sc_alt_opt < best_score:
                best_score = sc_alt_opt
                best_seq = a_alt_opt.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
                final_seq = np.clip(best_seq, 0.0, 1000.0)
                s2 = float(final_seq.sum())
                if s2 > 0.01:
                    final_seq = final_seq / s2
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    print(f"Time used: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final_seq]
