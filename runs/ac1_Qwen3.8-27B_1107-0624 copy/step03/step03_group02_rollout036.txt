    """One step of water-filling: push down elements contributing to high conv peaks."""
    n = len(a)
    c = _fft_convolve(a)
    target = a.sum()**2 / (2*n - 1)
    
    for _ in range(n_steps):
        peaks = np.where(c > target * 1.01)[0]
        if len(peaks) == 0:
            break
        # For each peak, compute the subgradient
        total_sg = np.zeros(n)
        for k in peaks:

        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        c = _fft_convolve(a_new)
        a = a_new
    return a

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):

    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

def _grad_smooth_max(a, w):

    a_pad = np.zeros(N)

def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)

        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]

    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx2 = np.arange(j2_start, j2_end + 1)
        sg[idx2] += a[idx2 - k]
    return sg

def _peak_suppression(a, deadline, max_iter=10000, verbose=False):
    """Targeted peak suppression: iteratively attack the top peaks."""
    a = np.asarray(a, dtype=np.float64).copy()

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.1 / np.sqrt(n)
    no_imp = 0
    top_k = 5

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        
        # Get top-k peaks, weighted by how far above the mean they are
        mean_c = float(c_curr.mean())
        if max_val <= mean_c * 1.001:
            break  # Already very flat

        k = min(top_k, len(c_curr))
        sorted_idx = np.argsort(c_curr)[::-1][:k]
        
        # Weight by excess over mean
        excess = np.maximum(c_curr[sorted_idx] - mean_c, 0)
        if excess.sum() < 1e-15:
            break
        weights = excess / excess.sum()
        
        # Combined subgradient
        sg = np.zeros(n)
        for i, idx in enumerate(sorted_idx):
            sg += weights[i] * _subgradient_at_max(a, int(idx))
        
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        # Adaptive learning rate
        lr = base_lr / np.sqrt(it + 1)
        
        # Line search: try a few step sizes
        best_a_new = a.copy()
        best_s_new = best_score
        for t in [1.0, 0.5, 0.25, 0.1]:
            a_try = a - lr * t * sg / sg_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try)
            sc_try = float(2.0 * n * c_try.max())
            if sc_try < best_s_new:
                best_s_new = sc_try
                best_a_new = a_try.copy()

        if best_s_new < best_score - 1e-14:
            best_score = best_s_new
            best_a = best_a_new.copy()
            a = best_a_new.copy()
            no_imp = 0
        else:
            a = best_a_new.copy()
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        if verbose and it % 5000 == 0:
            print(f"      PeakSupp it={it}: best={best_score:.10f}")

    return best_a

def _lp_spread(a, M):
    n = len(a)
    if n > 1024:
        return None  # Too large for LP
    c_obj = -np.ones(n)
    # Build constraint matrix: conv(a, g)[k] <= M for all k
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # This is a linear function of g
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] * g[k-j] contributes to position k = j + i for g[i]
        # So for each j, a[j] * g[i] contributes to c[j+i]
        A[j:j+n, :] = np.diag(a)  # This is wrong
    # Actually let me redo this
    # c[k] = sum_j a[j] * g[k-j] where 0 <= k-j < n, i.e., j <= k < j+n
    # So c[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * g[k-j]
    # Let i = k-j, so j = k-i, and 0 <= i < n, max(0,k-n+1) <= k-i <= min(k,n-1)
    # i ranges from max(0, k-(n-1)) to min(k, n-1)
    # c[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * g[i]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A[k, i] = a[k - i]
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None

def _make_profiles(n):
    """Generate diverse profile initializations."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Cosine
    for alpha in [0.3, 0.7, 1.0, 1.5, 2.0]:
        a = 1.0 + alpha * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Bell
    for w in [0.2, 0.35, 0.5, 0.7, 1.0]:
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * w ** 2))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # U-shape (high at edges, low in middle)
    for alpha in [0.5, 1.0, 2.0]:
        a = 1.0 + alpha * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # W-shape
    a = 1.0 + 1.5 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Random
    rng = np.random.default_rng(42)
    a = rng.exponential(size=n)
    profiles.append(a / a.sum())
    
    # Alternating
    a = np.ones(n)
    a[::2] = 1.0
    a[1::2] = 0.5
    profiles.append(a / a.sum())
    
    return profiles


def _interpolate_to(a0, n_new):
    p0 = np.asarray(a0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _spectral_flatness_optimize(a, deadline, max_iter=5000, lr=0.01, verbose=False):
    """Optimize for spectral flatness (minimize variance of |A|^2)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= np.mean(z)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute spectrum
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        spec = np.abs(A) ** 2
        spec = spec / spec.sum()  # Normalize
        
        # Objective: minimize variance of spec (encourage flatness)
        # Gradient of variance wrt a
        mean_spec = spec.mean()
        grad_spec = 2.0 * (spec - mean_spec)  # d/var/d/spec
        
        # d/spec/d/a: spec[k] = |A[k]|^2 = A[k] * conj(A[k])
        # d|A[k]|^2/da = 2 * Re(FFT^{-1}(conj(A[k]) * FFT(1)))... 
        # Actually, d|A[k]|^2/da = 2 * Re(ifft(conj(A[k]) * 1)) ... 
        # Let me think differently.
        # A[k] = sum_j a[j] * exp(-2pi i j k / N)
        # |A[k]|^2 = A[k] * conj(A[k])
        # d|A[k]|^2/da[m] = conj(A[k]) * exp(-2pi i m k / N) + A[k] * exp(2pi i m k / N)
        #                  = 2 * Re(conj(A[k]) * exp(-2pi i m k / N))
        #                  = 2 * Re(ifft(conj(A[k]) * exp(-2pi i m k / N)))... 
        # Hmm, this is getting complicated. Let me use a different approach.
        # 
        # Actually, |A|² = ifft(fft(a) * conj(fft(a)))... no that's circular.
        # 
        # Let me just use the chain rule with the convolution:
        # c = conv(a,a), score = 2n * max(c)
        # The gradient of the smooth max of c wrt a is what we already computed.
        # 
        # Instead, let me minimize a combination: alpha * smooth_max(c) + (1-alpha) * var(c)
        
        c_curr = _fft_convolve(a)
        var_c = float(np.var(c_curr))
        max_c = float(c_curr.max())
        
        # Combined smooth max with emphasis on peaks
        beta = 0.02 * max_c
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)
        
        # Add variance gradient
        # d(var(c))/da[m] = 2 * sum_k (c[k] - mean_c) * dc[k]/da[m]
        # dc[k]/da[m] = a[k-m] + a[m-k] (with appropriate bounds)
        # This is expensive to compute directly. Let's use FFT:
        # The gradient of sum_k (c[k]-mean)^2 wrt a is:
        # 2 * ifft(fft(c - mean) * fft(ifft of something))... 
        # Actually: var(c) = sum c[k]^2 - (sum c[k])^2/(2n-1)
        # d(sum c[k]^2)/da[m] = 2 * sum_k c[k] * dc[k]/da[m]
        # dc[k]/da[m] is the convolution of the delta function at m with a (reversed)
        # = a[k-m] (reversed index)
        # So sum_k c[k] * a[k-m] (with bounds) = (c * a_rev)[m] where * is convolution
        # This can be computed via FFT!
        
        var_grad = np.zeros(n)  # We'll add the variance term
        # d(sum c^2)/da = 2 * reverse_conv(c, a) ... 
        # Actually: sum_k c[k] * a[k-m] for valid k
        # Let's compute via FFT: fft(c_padded) * fft(a_reversed_padded) -> ifft
        N2 = _next_pow2(4 * n)
        c_pad = np.zeros(N2)
        c_pad[:2*n-1] = c_curr
        a_rev_pad = np.zeros(N2)
        a_rev_pad[:n] = a[::-1]
        cross = np.fft.irfft(np.fft.rfft(c_pad, N2) * np.fft.rfft(a_rev_pad, N2), N2)
        var_grad = 2.0 * cross[n-1:2*n-1]
        
        # Combine gradients
        combined = grad_a + 0.5 * var_grad / max(var_c + 1e-10, 1e-10)
        
        dot = float(np.dot(a, combined))
        grad_z = a * (combined - dot)
        grad_z -= np.mean(grad_z)
        
        gnorm = np.linalg.norm(grad_z)
        if gnorm > 10:
            grad_z = grad_z / gnorm * 10
        if gnorm < 1e-15:
            break
        
        t_adam += 1
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * grad_z**2
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= np.mean(z)
        
        e = np.exp(z)
        a = e / e.sum()
        
        c_new = _fft_convolve(a)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a.copy()
    
    return best_a

def _sca_step(a):
    """Successive convex approximation step: minimize max conv(a+d, a+d) ≈ max(c + 2*conv(a,d))."""
    n = len(a)
    c = _fft_convolve(a)  # Current convolution
    
    # Variables: d (length n) and t (scalar)
    # Objective: minimize t
    # Constraints:
    #   c[k] + 2 * sum_j a[j] * d[k-j] <= t  for k = 0..2n-2
    #   a[i] + d[i] >= 0  for i = 0..n-1
    #   sum(d) = 0 (maintain normalization, since sum(a)=1)
    
    # In std form for linprog:
    # minimize t
    # c_obj = [0, ..., 0, 1] (n zeros + 1 for t)
    # Variables: [d_0, ..., d_{n-1}, t]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # c[k] + 2 * conv(a,d)[k] - t <= 0  for each k
    # conv(a,d)[k] = sum_j a[j] * d[k-j] (with bounds)
    # So: 2 * sum_j a[j] * d[k-j] - t <= -c[k]
    
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        b_ub[k] = -c[k]
        A_ub[k, -1] = -1.0  # -t
        # conv(a,d)[k] = sum_{j} a[j] * d[k-j] where 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(k, n-1)
        # and d index is k-j
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, k - j] += 2.0 * a[j]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: a[i] + d[i] >= 0 => d[i] >= -a[i], d[i] <= inf
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        return d
    return None

"""
Multi-strategy optimization using Successive Convex Approximation (SCA) as the main
workhorse, combined with smooth-max Adam, targeted peak suppression, and LP spreading.
Key innovation: SCA directly linearizes the quadratic convolution objective and
solves an LP to find the best peak-reducing direction at each step.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx2 = np.arange(j2_start, j2_end + 1)
        sg[idx2] += a[idx2 - k]
    return sg


def _sca_lp_direction(a, time_limit=20):
    """
    SCA step: Find direction d that minimizes max_k [c[k] + 2*(conv(a,d))[k]]
    subject to a+d >= 0, sum(d) = 0.
    Returns d (the direction) or None if LP fails.
    """
    n = len(a)
    c = _fft_convolve(a)

    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize t
    # s.t. c[k] + 2*conv(a,d)[k] - t <= 0  for k=0..2n-2
    #      a[i] + d[i] >= 0  (i.e., d[i] >= -a[i])
    #      sum(d) = 0

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Build A_ub: (2n-1) x (n+1)
    # Row k: 2*conv(a,d)[k] - t <= -c[k]
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = -c.copy()

    for k in range(2 * n - 1):
        A_ub[k, -1] = -1.0
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, k - j] += 2.0 * a[j]

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Bounds
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _sca_optimize(a, deadline, max_iter=500, alpha_init=1.0, verbose=True):
    """
    SCA optimization: iteratively solve LP for peak-reducing direction,
    then do line search on step size.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    alpha = alpha_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        d = _sca_lp_direction(a, time_limit=max(5, min(30, (deadline - time.monotonic()) * 0.1)))
        if d is None:
            break

        # Line search: try several step sizes
        best_local_a = a.copy()
        best_local_score = best_score

        for alpha_try in [alpha, alpha * 0.5, alpha * 0.25, 0.1, 0.05]:
            a_try = a + alpha_try * d
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < best_local_score:
                best_local_score = sc_try
                best_local_a = a_try.copy()
                alpha = alpha_try  # Remember good step size

        if best_local_score < best_score - 1e-14:
            best_score = best_local_score
            best_a = best_local_a.copy()
            a = best_local_a.copy()
            no_imp = 0
            # Slightly increase alpha if improving
            alpha = min(1.0, alpha * 1.05)
        else:
            a = best_local_a.copy()
            no_imp += 1
            alpha = max(0.01, alpha * 0.7)
            if no_imp > 30:
                break

        if verbose and it % 50 == 0:
            print(f"      SCA it={it}: score={best_score:.10f}, alpha={alpha:.4f}, no_imp={no_imp}")

    return best_a, best_score


def _adam_smooth_max_v2(a, deadline, max_iter=50000, lr=0.03, beta_factor=0.05, verbose=False):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        gnorm = float(np.linalg.norm(grad_z))
        if gnorm > 10.0:
            grad_z = grad_z / gnorm * 10.0
        if gnorm < 1e-15:
            break

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.001, lr_cur * 0.95)
            if no_imp > max(3000, 5 * n):
                z = best_z + 0.4 * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 100 == 0:
            if no_imp < 100:
                beta = max(1e-12, beta * 0.82)
            else:
                beta = min(0.1 * float(c_curr.max()), beta * 1.08)

        if verbose and it % 10000 == 0:
            print(f"      Adam v2 it={it}: best={best_score:.10f}, beta={beta:.2e}")

    return best_a, best_score


def _peak_suppression(a, deadline, max_iter=10000, verbose=False):
    """Targeted peak suppression with line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)
    no_imp = 0
    top_k = 5

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        mean_c = float(c_curr.mean())

        if max_val <= mean_c * 1.002:
            break

        k = min(top_k, len(c_curr))
        sorted_idx = np.argsort(c_curr)[::-1][:k]

        excess = np.maximum(c_curr[sorted_idx] - mean_c, 0)
        if excess.sum() < 1e-15:
            break
        weights = excess / excess.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(sorted_idx):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)

        # Line search
        best_local_a = a.copy()
        best_local_score = best_score
        for t in [1.0, 0.5, 0.25, 0.1]:
            a_try = a - lr * t * sg / sg_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try)
            sc_try = float(2.0 * n * c_try.max())
            if sc_try < best_local_score:
                best_local_score = sc_try
                best_local_a = a_try.copy()

        if best_local_score < best_score - 1e-14:
            best_score = best_local_score
            best_a = best_local_a.copy()
            a = best_local_a.copy()
            no_imp = 0
        else:
            a = best_local_a.copy()
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

    return best_a


def _lp_spread(a, M, time_limit=15):
    n = len(a)
    if n > 600:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, k - j] = a[j]
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=1000, inflate=1.001, blend=0.3, verbose=False):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp, time_limit=max(3, min(20, (deadline - time.monotonic()) * 0.1)))

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.08)
            if no_imp > 80:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(100, n // 3):
                break

        a = a_new

    return best_a


def _interpolate_to(a0, n_new):
    p0 = np.asarray(a0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profiles(n):
    """Generate diverse profile initializations."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for alpha in [0.3, 0.7, 1.0, 1.5]:
        a = 1.0 + alpha * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for w in [0.2, 0.35, 0.5, 0.7]:
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * w ** 2))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for alpha in [0.5, 1.0, 2.0]:
        a = 1.0 + alpha * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization with SCA as main workhorse.
    
    Phases:
    1. SCA optimization at n=512 (main, ~700s)
    2. Smooth-max Adam refinement (~200s)
    3. Peak suppression + LP (~100s)
    4. Try n=1024 (~500s)
    5. Final polish (~100s)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare n=512 starting point
    if parent is not None:
        a512 = _interpolate_to(parent, 512)
    else:
        a512 = _make_profiles(512)[2]  # linear with eta=1.0

    sc_512 = _score(a512)
    print(f"Initial n=512: score={sc_512:.10f}")
    if sc_512 < best_score:
        best_score = sc_512
        best_seq = a512.copy()

    # === Phase 1: SCA optimization at n=512 ===
    print("=== Phase 1: SCA n=512 ===")
    d1 = min(deadline - 800, t_start + 700)
    a_sca, s_sca = _sca_optimize(a512, d1, max_iter=200, alpha_init=1.0, verbose=True)
    if s_sca < best_score:
        best_score = s_sca
        best_seq = (a_sca / a_sca.sum()).copy()
    print(f"  After SCA: {best_score:.10f}")

    # === Phase 2: Smooth-max Adam refinement ===
    print("=== Phase 2: Adam refinement n=512 ===")
    if time.monotonic() < deadline - 600:
        d2 = min(deadline - 600, time.monotonic() + 250)
        a_adam, s_adam = _adam_smooth_max_v2(best_seq, d2, max_iter=25000, lr=0.025, beta_factor=0.04)
        if s_adam < best_score:
            best_score = s_adam
            best_seq = a_adam.copy()
        print(f"  After Adam: {best_score:.10f}")

    # === Phase 3: Peak suppression + LP ===
    print("=== Phase 3: PeakSupp + LP n=512 ===")
    if time.monotonic() < deadline - 500:
        d3a = min(deadline - 500, time.monotonic() + 100)
        a_ps = _peak_suppression(best_seq, d3a, max_iter=8000)
        s_ps = _score(a_ps)
        if s_ps < best_score:
            best_score = s_ps
            best_seq = a_ps.copy()
        print(f"  After PeakSupp: {best_score:.10f}")

    if time.monotonic() < deadline - 400:
        d3b = min(deadline - 400, time.monotonic() + 100)
        a_lp = _lp_refine(best_seq, d3b, max_iter=500, inflate=1.0005, blend=0.25)
        s_lp = _score(a_lp)
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()
        print(f"  After LP: {best_score:.10f}")

    # More SCA rounds
    if time.monotonic() < deadline - 350:
        d3c = min(deadline - 350, time.monotonic() + 150)
        a_sca2, s_sca2 = _sca_optimize(best_seq, d3c, max_iter=100, alpha_init=0.8)
        if s_sca2 < best_score:
            best_score = s_sca2
            best_seq = (a_sca2 / a_sca2.sum()).copy()
        print(f"  After SCA2: {best_score:.10f}")

    # === Phase 4: Try n=1024 ===
    print("=== Phase 4: n=1024 ===")
    if time.monotonic() < deadline - 200:
        a1024 = _interpolate_to(best_seq, 1024)
        sc_1024 = _score(a1024)
        print(f"  Interpolated n=1024: {sc_1024:.10f}")

        d4a = min(deadline - 150, time.monotonic() + 250)
        a_1024_adam, s_1024_adam = _adam_smooth_max_v2(a1024, d4a, max_iter=20000, lr=0.015, beta_factor=0.03)
        if s_1024_adam < best_score:
            best_score = s_1024_adam
            best_seq = a_1024_adam.copy()
        print(f"  After Adam 1024: {best_score:.10f}")

        if time.monotonic() < deadline - 100:
            d4b = min(deadline - 100, time.monotonic() + 100)
            a_1024_ps = _peak_suppression(best_seq, d4b, max_iter=5000)
            s_1024_ps = _score(a_1024_ps)
            if s_1024_ps < best_score:
                best_score = s_1024_ps
                best_seq = a_1024_ps.copy()
            print(f"  After PeakSupp 1024: {best_score:.10f}")

        # Try SCA on 1024 if time (expensive LP)
        if best_seq.size <= 600 and time.monotonic() < deadline - 80:
            d4c = min(deadline - 80, time.monotonic() + 80)
            a_1024_sca, s_1024_sca = _sca_optimize(best_seq, d4c, max_iter=50, alpha_init=0.5)
            if s_1024_sca < best_score:
                best_score = s_1024_sca
                best_seq = (a_1024_sca / a_1024_sca.sum()).copy()
            print(f"  After SCA 1024: {best_score:.10f}")

    # === Phase 5: Final polish ===
    print(f"=== Phase 5: Final polish (best={best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Fine Adam
        d5a = min(deadline - 8, time.monotonic() + 60)
        a_f1, s_f1 = _adam_smooth_max_v2(best_seq, d5a, max_iter=8000, lr=0.008, beta_factor=0.01)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Final Adam: {best_score:.10f}")

        # Peak suppression
        if time.monotonic() < deadline - 5:
            d5b = min(deadline - 5, time.monotonic() + 30)
            a_f2 = _peak_suppression(best_seq, d5b, max_iter=3000)
            s_f2 = _score(a_f2)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final PeakSupp: {best_score:.10f}")

    # === Phase 6: Quick diverse search if time ===
    if time.monotonic() < deadline - 30:
        print("=== Phase 6: Diverse quick search ===")
        n_div = 512
        profiles = _make_profiles(n_div)
        # Add perturbed best
        if best_seq is not None:
            pert = _interpolate_to(best_seq, n_div)
            pert = pert * (1.0 + 0.15 * rng.normal(size=n_div))
            pert = np.maximum(pert, 0)
            if pert.sum() > 0:
                profiles.append(pert / pert.sum())

        for i, a0 in enumerate(profiles[:8]):
            if time.monotonic() > deadline - 30:
                break
            per_bud = min(10.0, (deadline - 30 - time.monotonic()) / max(1, 8 - i))
            if per_bud < 3:
                break
            d6 = time.monotonic() + per_bud
            a_d, s_d = _adam_smooth_max_v2(a0, d6, max_iter=2000, lr=0.03, beta_factor=0.06)
            if s_d < best_score:
                best_score = s_d
                best_seq = a_d.copy()
            print(f"  Div-{i}: {s_d:.10f} (best={best_score:.10f})")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
            best_score = min(best_score, sc_eval)
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
