- c[2i] - a[i]^2 + x^2 for k = 2i (quadratic in x)

- M_const = max of c[k] for k not in S1 and k ≠ 2i

- e_k = 2*a[k-i] for k in S1
- d_2i = c[2i] - a[i]^2

To minimize f(x) over x >= 0:
- f is convex (max of convex functions)
- f(0) = max(M_const, max(d_k), d_2i)
- f is non-decreasing for large x (since e_k >= 0 and the quadratic term grows)

The minimum is where f'(x) = 0 (if in the interior) or at the boundary x = 0.

f'(x) = max( max_{k in S1, active} e_k, 2x ) where "active" means c_k(x) = f(x).

We can find the minimum by:
1. Start with x = a[i] (current value).
2. Compute f(x) and the active set.
3. If f is decreasing at x, increase x; if increasing, decrease x.
4. Use bisection or Newton's method.

"""
Optimized step function heights minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Adam smooth-max, novel coordinate descent with exact 1D subproblem solving,
and multi-start diversification to find near-optimal flat-convolution sequences.

import numpy as np
import time
from scipy.optimize import minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    elif mn < 0:
        c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.03, beta_factor=0.05):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.001, lr_cur * 0.95)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(7000 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue
        z = z_new
        a = a_new
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.8)
    return best_a, best_score


def _coord_descent_step(c, a, i, n):
    """Solve the 1D subproblem for coordinate i given current c and a."""
    old_val = a[i]
    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    
    if k_min > k_max:
        return old_val
    
    # Affected indices in c
    affected_len = k_max - k_min + 1
    a_indices = np.arange(k_min, k_max + 1) - i
    a_vals = a[a_indices]
    c_affected = c[k_min:k_max + 1]
    
    # Determine if 2i is in the affected range
    two_i = 2 * i
    has_quad = (two_i >= k_min) and (two_i <= k_max)
    
    # Precompute constant part
    const_parts = []
    if k_min > 0:
        const_parts.append(c[:k_min].max())
    if k_max < 2 * n - 2:
        const_parts.append(c[k_max + 1:].max())
    M_const = max(const_parts) if const_parts else 0.0
    
    # For linear terms (excluding k=2i if present)
    if has_quad:
        quad_idx = two_i - k_min
        lin_mask = np.ones(affected_len, dtype=bool)
        lin_mask[quad_idx] = False
        a_lin = a_vals[lin_mask]
        c_lin = c_affected[lin_mask]
        d_lin = c_lin - 2.0 * a_lin * old_val
        e_lin = 2.0 * a_lin
        d_2i = c[two_i] - old_val * old_val
    else:
        a_lin = a_vals
        c_lin = c_affected
        d_lin = c_lin - 2.0 * a_lin * old_val
        e_lin = 2.0 * a_lin
        d_2i = -np.inf
        has_quad = False
    
    # Define f(x) = max(M_const, max(d_lin + e_lin*x), d_2i + x^2)
    def f_of_x(x):
        val_lin = np.max(d_lin + e_lin * x) if len(d_lin) > 0 else -np.inf
        val_quad = d_2i + x * x if has_quad else -np.inf
        return max(M_const, val_lin, val_quad)
    
    # Upper bound: try current value times 3
    x_hi = max(3.0 * old_val, 1e-8)
    # Check if f is still decreasing at x_hi
    f_hi = f_of_x(x_hi)
    f_mid = f_of_x(0.5 * x_hi)
    if f_hi < f_mid * (1 - 1e-10):
        x_hi *= 10.0
    
    # Use bounded scalar minimization
    res = minimize_scalar(f_of_x, bounds=(0.0, x_hi), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 50})
    if res.success and res.fun < f_of_x(old_val):
        return float(res.x)
    return old_val


def _coordinate_descent(a, deadline, max_sweeps=5000, order='random', seed=0):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    rng = np.random.default_rng(seed)
    no_imp_sweeps = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved_this_sweep = False
        if order == 'random':
            indices = rng.permutation(n)
        else:
            indices = np.arange(n)
        
        for i in indices:
            new_val = _coord_descent_step(c, a, i, n)
            if abs(new_val - a[i]) > 1e-14:
                # Update a[i] and update c incrementally
                delta = new_val - a[i]
                a[i] = new_val
                # Update c: c[k] changes by terms involving a[i]
                # For k in [i, i+n-1]: delta_c[k] = 2 * delta * a[k-i] (if k != 2i)
                # For k = 2i: delta_c[k] = 2*old*a[i]_new - old^2 - ... 
                # Simpler: recompute affected part
                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)
                a_indices = np.arange(k_min, k_max + 1) - i
                # New c for affected part
                # c_new[k] = sum_j a[j]*a[k-j] where we only changed a[i]
                # = old_c[k] + delta * a[k-i] + a[k-i] * delta (if k-i != i)
                # or old_c[k] + 2*delta*a[k-i] if k-i != i
                # or old_c[k] - old_val^2 + new_val^2 if k == 2i
                for k in range(k_min, k_max + 1):
                    ki = k - i
                    if ki != i:
                        c[k] += 2.0 * delta * a[ki]
                    else:
                        c[k] = c[k] - (a[i] - delta) ** 2 + a[i] ** 2
                # This incremental update is error-prone for many steps; 
                # recompute periodically
                if sweep % 50 == 49:
                    c = _fft_convolve(a)
            
        # Renormalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
        
        # Periodic full recompute to avoid drift
        if sweep % 20 == 19:
            c = _fft_convolve(a)
        
        # Check improvement
        c_check = _fft_convolve(a)
        score_check = float(2.0 * n * c_check.max())
        if score_check < best_score - 1e-12:
            best_score = score_check
            best_a = a.copy()
            no_imp_sweeps = 0
            improved_this_sweep = True
        else:
            no_imp_sweeps += 1
        
        if no_imp_sweeps > 200:
            # Perturb and restart
            a = best_a * (1.0 + 0.05 * rng.normal(size=n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = _fft_convolve(a)
            no_imp_sweeps = 0
    
    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_smooth_profile(n, alpha=2.0):
    """Create a smooth tapered profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_double_hump(n, pos=0.3, width=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - pos) / width) ** 2) + np.exp(-((t - (1 - pos)) / width) ** 2) + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()

Optimized step function heights minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Adam smooth-max, coordinate descent with exact 1D subproblem solving,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    elif mn < 0:
        c = np.maximum(c, 0.0)
    return c


def _score(a):

def _smooth_max_weights(c, beta):

    x = (c - mval) / beta

    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:

        ex[int(np.argmax(c))] = 1.0
        s = 1.0

def _grad_smooth_max(a, w):

    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]

def _coord_subproblem(c, a, i, n):

    k_max = min(2 * n - 2, i + n - 1)

    if k_min > k_max:

    a_indices = np.arange(k_min, k_max + 1) - i

    has_quad = (two_i >= k_min) and (two_i <= k_max)

    # Constant part (unaffected k values)
    const_parts = []
    if k_min > 0:
        const_parts.append(float(c[:k_min].max()))
    if k_max < 2 * n - 2:
        const_parts.append(float(c[k_max + 1:].max()))
    M_const = max(const_parts) if const_parts else 0.0

    # Linear terms
    if has_quad:
        quad_pos = two_i - k_min
        lin_mask = np.ones(affected_len, dtype=bool)
        lin_mask[quad_pos] = False
        a_lin = a_vals[lin_mask]
        c_lin = c_affected[lin_mask]
    else:
        a_lin = a_vals
        c_lin = c_affected

    if len(a_lin) == 0:
        d_lin = np.array([0.0])
        e_lin = np.array([0.0])
        d_2i = float(c_affected[quad_pos]) - old_val * old_val if has_quad else 0.0
    else:
        d_lin = c_lin - 2.0 * a_lin * old_val
        e_lin = 2.0 * a_lin
        d_2i = float(c[two_i]) - old_val * old_val if has_quad else -np.inf

            val_lin = float(np.max(d_lin + e_lin * x))

            val_lin = float(np.max(d_lin)) if len(d_lin) > 0 else -np.inf

            val_quad = d_2i + x * x
        else:
            val_quad = -np.inf

def _interpolate_profile(p0, n_new):

    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
