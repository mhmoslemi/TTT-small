For a non-negative sequence with flat spectrum, A(0) = sum a_i > 0, and we want |A(ω)| = A(0)/√(2n-1) for all other frequencies. This is the "ideal" case that gives objective ≈ 1.

In practice, we can't achieve perfectly flat spectrum with a non-negative sequence, but we can get close.

One approach: use a sequence whose DFT is close to flat. This is similar to "Golay sequences" or "Zadoff-Chu sequences" but adapted for non-negative values.

2. Use Adam with smooth-max for the main optimization
3. Periodically inject random perturbations to escape local minima
4. Use a "restart" strategy: if no improvement for X iterations, restart from best + large perturbation
5. Try multiple initializations and keep the best

"""
Multi-scale Adam smooth-max optimization for step function coefficient minimization.
Uses coarse-to-fine resolution scaling, diverse initializations, adaptive beta cooling,

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w

    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]

def _adam_optimize(a, deadline, max_iter=30000, lr=0.03, beta_factor=0.06,
                   restart_patience=3000, restart_scale=0.4, verbose=True):
    """Adam optimizer in logit-space with smooth-max objective and periodic restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.8
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    beta_min = 1e-6

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        e_sum = float(e.sum())
        if not np.isfinite(e_sum) or e_sum <= 0:
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        a_new = e / e_sum

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:

            if no_imp > restart_patience:

                z = np.clip(z, -50.0, 50.0)

                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-8, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0

        z = z_new
        a = a_new

        if it % 200 == 0:

        if verbose and it % 5000 == 0:
            print(f"        Adam it={it}: best={best_score:.10f}, beta={beta:.2e}, lr={lr_cur:.4f}")

    return best_a, best_score


def _adam_multiscale(a0, deadline, verbose=True):
    """Multi-phase Adam with decreasing beta and adaptive restarts."""
    a = a0.copy()
    n = len(a)

    phases = [
        (25000, 0.04, 0.08, 2500, 0.5),
        (20000, 0.025, 0.04, 3000, 0.4),
        (15000, 0.015, 0.02, 4000, 0.3),
        (10000, 0.008, 0.01, 5000, 0.25),
    ]

    best_a = a0.copy()
    c = _fft_convolve(a0)
    best_score = float(2.0 * n * c.max())

    for phase_idx, (max_it, lr, bf, patience, rscale) in enumerate(phases):
        if time.monotonic() > deadline:
            break
        a_phase, s_phase = _adam_optimize(
            a, deadline, max_iter=max_it, lr=lr, beta_factor=bf,

        )
        if s_phase < best_score:
            best_score = s_phase
            best_a = a_phase.copy()
        a = a_phase.copy()
        if verbose:
            print(f"      Phase {phase_idx}: score={best_score:.10f}")

    return best_a, best_score


def _lp_spread(a, M):
    """LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=500, inflate=1.0005, blend=0.3):
    """Iterative LP spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.08)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.9)
            if no_imp > 80:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _make_cosine_profile(n, alpha=0.8, freq=1):

    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    a = np.maximum(a, 1e-12)

def _make_linear_profile(n, eta=1.2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)

def _make_random_profile(n, rng, smoothness=0.3):
    z = rng.normal(size=n)
    # Smooth it a bit
    from numpy.fft import rfft, irfft
    Z = rfft(z)
    freqs = np.fft.rfftfreq(n)
    Z *= np.exp(-smoothness * (2 * np.pi * freqs) ** 2)
    z = irfft(Z, n)
    a = 1.0 + 0.5 * z / (np.std(z) + 1e-10)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = float(p.sum())
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # === Stage 1: Quick search at n=512 (~120s) ===
    n1 = 512
    d1 = min(deadline - 1400, t_start + 120)
    print(f"\n=== Stage 1: n={n1}, budget={d1 - time.monotonic():.0f}s ===")

    inits_1 = []
    if parent is not None:
        inits_1.append(_interpolate_profile(parent, n1))
    inits_1.append(_make_cosine_profile(n1, 0.7, 1))
    inits_1.append(_make_dual_cosine_profile(n1, 0.6, 1, 0.3, 2))
    inits_1.append(_make_random_profile(n1, rng, 0.5))
    inits_1.append(_make_linear_profile(n1, 1.5))
    inits_1.append(_make_cosine_profile(n1, 0.4, 2))

    for i, a0 in enumerate(inits_1):
        if time.monotonic() > d1:
            break
        per_time = min(18.0, (d1 - time.monotonic()))
        d1i = time.monotonic() + max(5.0, per_time)
        a_res, s_res = _adam_optimize(a0, d1i, max_iter=5000, lr=0.03,
                                       beta_factor=0.07, restart_patience=2000,
                                       restart_scale=0.4, verbose=False)
        if s_res < best_score:
            best_score = s_res
            best_seq = a_res.copy()
        print(f"  S1-{i}: {s_res:.10f}")

    print(f"  Stage 1 best: {best_score:.10f}")

    # === Stage 2: Main optimization at n=1024 (~500s) ===
    n2 = 1024
    if time.monotonic() < deadline - 800:
        d2 = min(deadline - 700, time.monotonic() + 500)
        print(f"\n=== Stage 2: n={n2}, budget={d2 - time.monotonic():.0f}s ===")

        # Start from best at 512, interpolated
        a2_init = _interpolate_profile(best_seq, n2)
        sc2_init = _score(a2_init)
        print(f"  Interpolated init: {sc2_init:.10f}")
        if sc2_init < best_score:
            best_score = sc2_init

        # Multi-phase Adam
        a2, s2 = _adam_multiscale(a2_init, d2, verbose=True)
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()
        print(f"  Stage 2 Adam: {best_score:.10f}")

        # LP refinement
        if time.monotonic() < d2:
            a2_lp = _lp_refine(best_seq, d2, max_iter=200, inflate=1.0003, blend=0.25)
            s2_lp = _score(a2_lp)
            if s2_lp < best_score:
                best_score = s2_lp
                best_seq = a2_lp.copy()
            print(f"  Stage 2 LP: {best_score:.10f}")

        # Final Adam polish at 1024
        if time.monotonic() < d2:
            a2_f, s2_f = _adam_optimize(best_seq, d2, max_iter=10000, lr=0.01,
                                         beta_factor=0.02, restart_patience=5000,
                                         restart_scale=0.2, verbose=False)
            if s2_f < best_score:
                best_score = s2_f
                best_seq = a2_f.copy()
            print(f"  Stage 2 Final: {best_score:.10f}")

    # === Stage 3: High-resolution optimization at n=2048 (~800s) ===
    n3 = 2048
    if time.monotonic() < deadline - 200:
        d3 = min(deadline - 150, time.monotonic() + 800)
        print(f"\n=== Stage 3: n={n3}, budget={d3 - time.monotonic():.0f}s ===")

        a3_init = _interpolate_profile(best_seq, n3)
        sc3_init = _score(a3_init)
        print(f"  Interpolated init: {sc3_init:.10f}")
        if sc3_init < best_score:
            best_score = sc3_init

        # Long multi-phase Adam at 2048
        a3, s3 = _adam_multiscale(a3_init, d3, verbose=True)
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()
        print(f"  Stage 3 Adam: {best_score:.10f}")

        # Also try a fresh random init at 2048
        if time.monotonic() < d3 - 60:
            a3_r = _make_random_profile(n3, np.random.default_rng(999), 0.2)
            a3_r2, s3_r2 = _adam_optimize(a3_r, min(d3, time.monotonic() + 200),
                                           max_iter=15000, lr=0.025, beta_factor=0.05,
                                           restart_patience=3000, restart_scale=0.35,
                                           verbose=False)
            if s3_r2 < best_score:
                best_score = s3_r2
                best_seq = a3_r2.copy()
            print(f"  Stage 3 Random: {best_score:.10f}")

        # LP at 2048 (limited time)
        if time.monotonic() < d3:
            a3_lp = _lp_refine(best_seq, d3, max_iter=50, inflate=1.0002, blend=0.2)
            s3_lp = _score(a3_lp)
            if s3_lp < best_score:
                best_score = s3_lp
                best_seq = a3_lp.copy()
            print(f"  Stage 3 LP: {best_score:.10f}")

    # === Stage 4: Final polish (~150s) ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        d4 = deadline - 10
        print(f"\n=== Stage 4: Final polish ===")

        # Fine Adam on best
        a4, s4 = _adam_optimize(best_seq, min(d4, time.monotonic() + 80),
                                 max_iter=10000, lr=0.005, beta_factor=0.01,
                                 restart_patience=8000, restart_scale=0.15, verbose=False)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
        print(f"  Final Adam: {best_score:.10f}")

        # Very fine Adam
        if time.monotonic() < d4 - 10:
            a4b, s4b = _adam_optimize(best_seq, d4,
                                       max_iter=8000, lr=0.003, beta_factor=0.005,
                                       restart_patience=10000, restart_scale=0.1,
                                       verbose=False)
            if s4b < best_score:
                best_score = s4b
                best_seq = a4b.copy()
            print(f"  Ultra-fine: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_cosine_profile(512, 0.7, 1)
        best_score = _score(best_seq)
        print(f"  FALLBACK used: {best_score:.10f}")

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_cosine_profile(256, 0.7, 1)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence: {sc_eval:.10f}")
            else:
                print("  WARNING: evaluate_sequence returned inf!")
        except Exception as e:
            print(f"  WARNING: evaluate_sequence error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
