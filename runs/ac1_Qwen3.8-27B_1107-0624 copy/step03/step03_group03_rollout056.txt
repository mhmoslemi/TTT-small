Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines mirror descent (exponential weights) with top-K subgradient,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_conv(a)
    return float(2.0 * n * c.max())

def _subgradient_topk(a, K=5):

    c = _fft_conv(a)
    m = float(c.max())
    # Get top-K lags
    if K >= len(c):
        idx = np.arange(len(c))
        weights = np.ones(len(c)) / len(c)
    else:
        idx = np.argsort(c)[::-1][:K]
        # Weight by value (higher value = more weight)
        w = c[idx]
        w = w / w.sum()
    
    # Gradient for each active lag k: grad_j = 2 * a[k-j] for valid j
    # equivalently: grad = 2 * (circulant matrix row k applied to a)
    # In FFT: grad = 2 * ifft(fft(a) * fft(delta_k)) but it's simpler to do directly
    n_conv = 2 * n - 1
    N = _next_pow2(n_conv)

    grad = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] < 1e-10:
            continue
        # grad_j += 2 * w[i] * a[k-j] for valid j (0 <= j < n, 0 <= k-j < n)
        j_min = max(0, k - n + 1)

        seg = a[k - j_max:k - j_min + 1][::-1]
        grad[j_min:j_max + 1] += 2.0 * w[i] * seg
    
    return grad, m

def _mirror_descent_step(a, grad, eta):
    """Exponential weights / mirror descent step on the simplex."""
    n = len(a)
    # a_new[i] = a[i] * exp(-eta * grad[i]) / sum(...)
    log_shift = -eta * grad
    log_shift -= log_shift.max()  # numerical stability
    exp_shift = np.exp(log_shift)
    a_new = a * exp_shift
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def _lp_spread(a, M, reduced=True, top_k=100):
    """LP: maximize sum(g) s.t. conv(a,g)_k <= M, g >= 0."""
    n = len(a)
    n_conv = 2 * n - 1
    c_obj = -np.ones(n)
    
    if reduced and n > 128:
        # Get current conv to identify active lags
        c_aa = _fft_conv(a)
        # Keep lags where c_aa is close to max
        threshold = M * 0.95
        active = np.where(c_aa >= threshold)[0]
        if len(active) < 10:
            active = np.argsort(c_aa)[::-1][:top_k]
        # Add neighbors
        active_ext = set()
        for k in active:
            for off in range(-1, 2):
                j = k + off
                if 0 <= j < n_conv:
                    active_ext.add(j)
        active = sorted(active_ext)
    else:
        active = list(range(n_conv))
    
    m = len(active)
    A = np.zeros((m, n))
    b = np.full(m, M)
    for row, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A[row, j_min:j_max + 1] = a[j_min:j_max + 1]
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

        return p / s if s > 0 else np.ones(n_new) / n_new
