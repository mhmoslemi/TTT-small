def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    n = a.size
    if n <= 128:
        return np.maximum(np.convolve(a, a), 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_h(w, a):
    """h[i] = sum_{m=0}^{n-1} a[m]*w[i+m] for i=0,...,n-1."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    """Compute soft-max weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w, cnt
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    entropy = -float(np.sum(w * np.log(np.maximum(w, 1e-300))))
    return mval, w, entropy

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size

    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def coordinate_polish(a0, deadline, rng, n_sweeps=5):
    """Polish via coordinate descent: for each coordinate, find optimal value."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()

    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    best_a = a0.copy()

    iters = 0

    while time.monotonic() < deadline and iters < n_sweeps * n:
        # Try moving each coordinate a small amount
        idx = iters % n
        current_val = a0[idx]
        # Try a few perturbations
        for frac in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3]:
            trial = a0.copy()
            old_sum = trial.sum()
            trial[idx] = current_val * frac
            new_sum = trial.sum()

                a0 = trial
                best_score = sc
                best_a = a0.copy()
                current_val = a0[idx]
                break
        iters += 1

    return best_a, best_score

                print(f"    After peak targeting: {sc_pt_f:.10f}")
                if sc_pt_f < sc_f:
                    result = r_pt

            # Coordinate polish
            if time.monotonic() < phase2_deadline - 20 and result.size <= 1024:
                dl_cp = min(phase2_deadline, time.monotonic() + 60.0)
                r_cp, sc_cp = coordinate_polish(result, dl_cp, rng, n_sweeps=3)
                sc_cp_f = compute_score(r_cp)
                try_update(r_cp, sc_cp_f)
                print(f"    After coord polish: {sc_cp_f:.10f}")

    print(f"  Phase 2 best score: {best_score:.10f}")

    # ===== Phase 3: Fine-tuning and alternative strategies =====
    print("\n--- Phase 3: Fine-tuning ---")

    if best_seq is not None:
        n_final = best_seq.size

        # Multiple refinement passes with different parameters
        for round_idx, (lr_val, beta_f, n_pk) in enumerate([
            (0.015, 0.03, 3),
            (0.01, 0.02, 5),
            (0.008, 0.015, 7),
            (0.005, 0.01, 3),
        ]):
            if time.monotonic() > deadline - 30:
                break
            p0 = best_seq + 0.0003 * rng.normal(size=n_final)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            dl = min(deadline - 10, time.monotonic() + 60.0)
            if time.monotonic() >= dl:
                break

            # Alternate between Adam and peak targeting
            if round_idx % 2 == 0:
                r, sc_r = adam_optimize(p0, dl, rng, lr=lr_val, beta_init_frac=beta_f)
            else:
                r, sc_r = peak_targeting(p0, dl, rng, n_peaks=n_pk, lr=lr_val)
            sc_f = compute_score(r)
            try_update(r, sc_f)
            print(f"  Round {round_idx} (lr={lr_val}, beta_f={beta_f}): {sc_f:.10f}")

        # Try interpolating to slightly larger n and back
        for scale in [1.25, 0.8, 1.5]:
            if time.monotonic() > deadline - 30:
                break
            n_new = int(n_final * scale)
            if n_new < 64 or n_new > 8192:
                continue
            p_interp = interpolate_profile(best_seq, n_new)
            sc_int = compute_score(p_interp)
            try_update(p_interp, sc_int)
            if sc_int < best_score + 0.01:
                dl = min(deadline - 10, time.monotonic() + 40.0)
                if time.monotonic() < dl:
                    r, _ = adam_optimize(p_interp, dl, rng, lr=0.01, beta_init_frac=0.02)
                    sc_r = compute_score(r)
                    try_update(r, sc_r)
                    print(f"  Scale {scale} (n={n_new}): {sc_r:.10f}")

    # ===== Phase 4: Final attempts =====
    print("\n--- Phase 4: Final attempts ---")

    if time.monotonic() < deadline - 20:
        # Try a completely different approach: geometric progression
        for ratio in [1.01, 1.02, 0.99, 0.98]:
            if time.monotonic() > deadline - 10:
                break
            n_try = int(best_seq.size) if best_seq is not None else 1024
            a_geo = np.array([ratio ** i for i in range(n_try)])
            a_geo = np.maximum(a_geo, 1e-12)
            a_geo = a_geo / a_geo.sum()
            sc_geo = compute_score(a_geo)
            try_update(a_geo, sc_geo)

            if sc_geo < best_score + 0.05:
                dl = min(deadline - 5, time.monotonic() + 30.0)
                if time.monotonic() < dl:
                    r, _ = adam_optimize(a_geo, dl, rng, lr=0.02, beta_init_frac=0.05)
                    sc_r = compute_score(r)
                    try_update(r, sc_r)
                    print(f"  Geometric (ratio={ratio}): {sc_r:.10f}")

    # Final validation
    print(f"\n  Final best score: {best_score:.10f}")

    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_uniform(1024)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_uniform(512)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
