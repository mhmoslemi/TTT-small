```python
"""
Optimization of step function autocorrelation peak using L-BFGS-B with
adaptive log-sum-exp smoothing, LP refinement, and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lse_obj_and_grad_z(z, T):
    """
    Objective and gradient in log-space (softmax parameterization).
    z: log-space variables (length n)
    T: temperature for LSE smoothing
    Returns: (objective_value, gradient_wrt_z)
    """
    n = len(z)
    # Softmax
    z_max = z.max()
    ez = np.exp(z - z_max)
    a = ez / ez.sum()
    
    # Compute convolution
    c = _fft_convolve(a)
    m = float(c.max())
    x = (c - m) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    w = ex / s
    
    # Objective: 2n * (m + T * ln(s))
    obj = 2.0 * n * (m + T * np.log(s))
    
    # Gradient w.r.t. a:
    # df/da[j] = 4n * sum_k w[k] * a[k-j] (with a out of bounds = 0)
    # Using FFT: h = ifft(fft(w) * conj(fft(a)))
    N = _next_pow2(3 * n - 2)
    fw = np.fft.fft(w, N)
    fa = np.fft.fft(a, N)
    h = np.fft.ifft(fw * np.conj(fa)).real[:n]
    
    grad_a = 4.0 * n * h
    
    # Gradient w.r.t. z: gz = grad_a - a * (a @ grad_a)
    dot = float(np.dot(a, grad_a))
    grad_z = grad_a - a * dot
    
    return obj, grad_z


def _lbfgsb_optimize(a0, deadline, T_start=0.1, T_end=0.005, max_iter=500):
    """
    L-BFGS-B optimization with adaptive temperature.
    a0: initial point on simplex (sum=1, a>=0)
    """
    n = len(a0)
    
    # Convert to log-space
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 -= z0.mean()  # center
    
    best_a = a0.copy()
    best_score = _score_normalized(a0)
    
    # Multi-temperature schedule
    n_temps = 5
    temps = np.geomspace(T_start, T_end, n_temps)
    
    z = z0.copy()
    for ti, T in enumerate(temps):
        if time.monotonic() > deadline:
            break
        
        # Define objective and gradient for L-BFGS-B
        def func_grad(z_var):
            obj, grad = _lse_obj_and_grad_z(z_var, T)
            return obj, grad
        
        try:
            res = minimize(func_grad, z, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter // n_temps,
                                            'gtol': 1e-8,
                                            'maxfun': 2000})
            z = res.x
        except Exception:
            # If L-BFGS-B fails, try a simpler approach
            pass
        
        # Convert back to simplex
        z_max = z.max()
        ez = np.exp(z - z_max)
        a_new = ez / ez.sum()
        
        score = _score_normalized(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        print(f"    T={T:.5f}: score={score:.10f}")
    
    return best_a


def _lp_spread(a, M, top_k=100):
    """LP to find a vector g with sum(g)>sum(a) and conv(a,g)[k] <= M for all k."""
    n = len(a)
    c_aa = _fft_convolve(a)
    
    # Use reduced constraints (near-active)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    
    # Build constraint matrix
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    
    c_obj = -np.ones(n)
    b = np.full(len(active), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.2):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    blend_cur = blend
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_spread(a, M_lp)
        
        if g is None:
            no_imp += 5
            if no_imp > 200:
                break
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = _score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                blend_cur = max(0.03, blend_cur * 0.85)
            if no_imp > 300:
                break
        
        a = a_new
    
    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02, momentum=0.7):
    """Subgradient polish at the actual max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        
        # Gradient: 4n * h where h[j] = sum_k w[k] * a[k-j]
        N = _next_pow2(3 * n - 2)
        fw = np.fft.fft(w, N)
        fa = np.fft.fft(a, N)
        h = np.fft.ifft(fw * np.conj(fa)).real[:n]
        grad = 4.0 * n * h
        
        # Project to tangent space of simplex
        dot = float(np.dot(a, grad))
        grad_tan = grad - a * dot
        
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        
        score = _score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
        
        a = a_new
    
    return best_a


def _make_gaussian(n, sigma_frac=0.3):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lse_obj_grad_z(z, T):

    z_max = z.max()
    ez = np.exp(z - z_max)

    fa = np.fft.fft(a, N)
    h = np.fft.ifft(fw * np.conj(fa)).real[:n]

    dot = float(np.dot(a, grad_a))

        if s < 1e-10:
            break
        a_new = a_new / s

        score = _score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.03, blend_cur * 0.9)
            if no_imp > max(200, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)

    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)

    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[row, j] = a[j]

    c_obj = -np.ones(n)
    b = np.full(len(active), M)

    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.015, momentum=0.7):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-14)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(3 * n - 2)
        fw = np.fft.fft(w, N)
        fa = np.fft.fft(a, N)
        h = np.fft.ifft(fw * np.conj(fa)).real[:n]
        grad = 4.0 * n * h

        mean_grad = float(np.mean(grad))
        grad_tan = grad - mean_grad

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s

        score = _score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(5000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        a = a_new

    return best_a


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_with_edges(n, edge_weight=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_hump(n, sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.3) / 0.15) ** 2) + np.exp(-0.5 * ((t - 0.7) / 0.15) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 45.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    lengths = [1024]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)

    def get_candidates(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for _ in range(2):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.08 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                if p_pert.sum() > 1e-10:
                    p_pert = p_pert / p_pert.sum()
                    cands.append(p_pert)
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_flat_with_edges(n_t, 0.4))
        cands.append(_make_two_hump(n_t))
        return cands

    # Phase 1: L-BFGS-B with adaptive temperature
    print("=== Phase 1: L-BFGS-B optimization ===")
    best_per_length = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 400:
            break

        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 400:
                break

            sc0 = _score_normalized(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            lb_end = min(time.monotonic() + 150, deadline - 400)
            if lb_end - time.monotonic() < 10:
                break

            a_opt = _lbfgsb_optimize(a0, lb_end, T_start=0.08, T_end=0.002,
                                     max_iter_total=600)
            sc_opt = _score_normalized(a_opt)

            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
            if sc_opt < best_here_score:
                best_here_score = sc_opt
                best_here = a_opt

            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LBFGS={sc_opt:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best n={n_t}: {best_here_score:.10f}")

    # Phase 2: Direct gradient descent polish
    print("\n=== Phase 2: Gradient descent polish ===")
    for n_t in list(best_per_length.keys()):
        if time.monotonic() > deadline - 250:
            break

        a_in = best_per_length[n_t]
        gd_end = min(time.monotonic() + 60, deadline - 250)
        if gd_end - time.monotonic() < 10:
            continue

        a_gd = _direct_grad_descent(a_in, gd_end, lr=0.008, max_iter=30000, momentum=0.5)
        sc_gd = _score_normalized(a_gd)

        if np.isfinite(sc_gd) and sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.copy()
        if sc_gd < _score_normalized(best_per_length[n_t]):
            best_per_length[n_t] = a_gd

        print(f"  GD n={n_t}: score={sc_gd:.10f}")

    # Phase 3: Subgradient polish
    print("\n=== Phase 3: Subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        sp_end = min(time.monotonic() + 80, deadline - 120)
        if sp_end - time.monotonic() > 10:
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=40000, lr=0.012, momentum=0.7)
            sc_sp = _score_normalized(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgradient: score={sc_sp:.10f}")

    # Phase 4: LP refinement
    print("\n=== Phase 4: LP refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        lp_end = min(time.monotonic() + 100, deadline - 80)
        if lp_end - time.monotonic() > 15:
            a_lp = _lp_refine(best_seq, lp_end, max_iter=2000, inflate=1.0005, blend=0.12)
            sc_lp = _score_normalized(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  LP: score={sc_lp:.10f}")

    # Phase 5: Final L-BFGS-B pass with very low temperature
    print("\n=== Phase 5: Final fine-tune L-BFGS-B ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        final_lb_end = deadline - 5
        if final_lb_end - time.monotonic() > 15:
            a_final = _lbfgsb_optimize(best_seq, final_lb_end, T_start=0.02,
                                       T_end=0.0005, max_iter_total=400)
            sc_final = _score_normalized(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LBFGS: score={sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
```
