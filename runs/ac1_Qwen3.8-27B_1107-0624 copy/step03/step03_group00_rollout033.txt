"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) diverse initialization at multiple scales, (2) Adam in logit space
with adaptive smooth-max temperature, (3) LP-based duality refinement,

(6) multi-restart with noise injection, (7) final ultra-fine polish.
Primary scale: n=2048 for more degrees of freedom.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_init=0.35, max_solves=25):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)
    return current_best, current_score

    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * lr_decay)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-4, beta * 0.6)

        limit = max(100, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _projected_gradient(p0, deadline, rng, lr_init=0.01, max_iter=5000):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 0.0)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _score_norm(p)
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(p)
        maxc = float(c.max())
        # Subgradient: use exact max (or top few)
        top_k = min(5, c.size)
        sidx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        w[sidx] = 1.0 / top_k

        g = _conv_cross_corr(w, p)
        # Projected gradient step
        p_new = p - lr * g
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s > 1e-15:
            p_new = p_new / s

        sc = _score_norm(p_new)
        if sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
        else:
            lr = max(1e-5, lr * 0.95)

        if it % 200 == 0:
            lr = lr_init * (0.98 ** (it // 200))

    return best_p, best_score


def _make_profile_exponential(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_gaussian(n, sigma):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_flat_top(n, width_frac):
    t = np.linspace(0.0, 1.0, n)
    half_w = width_frac / 2.0
    a = np.zeros(n)
    mask = (t >= 0.5 - half_w) & (t <= 0.5 + half_w)
    a[mask] = 1.0
    # Smooth edges
    edge = width_frac * 0.3
    a += 0.3 * np.exp(-0.5 * ((t - (0.5 - half_w)) / edge) ** 2)
    a += 0.3 * np.exp(-0.5 * ((t - (0.5 + half_w)) / edge) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={elapsed:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Initial candidates (0-100s) ---
    phase1_end = t_start + 100.0
    candidates = []

    # Diverse profiles at n=1024 and n=2048
    for n in [1024, 2048]:
        profs = [
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin1.5', _make_profile_linear(n, 1.5)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('cubic', _make_profile_cubic(n, 1.5, -1.0)),
            ('cubic2', _make_profile_cubic(n, 2.0, -1.5)),
            ('exp0.5', _make_profile_exponential(n, 0.5)),
            ('exp1.0', _make_profile_exponential(n, 1.0)),
            ('exp2.0', _make_profile_exponential(n, 2.0)),
            ('twobump', _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
            ('twobump2', _make_profile_two_bump(n, 0.25, 0.75, 0.2)),
            ('cos1', _make_profile_cosine(n, 1, 0.8)),
            ('cos2', _make_profile_cosine(n, 2, 0.6)),
            ('cos3', _make_profile_cosine(n, 3, 0.5)),
            ('cos5', _make_profile_cosine(n, 5, 0.4)),
            ('gauss0.3', _make_profile_gaussian(n, 0.3)),
            ('gauss0.4', _make_profile_gaussian(n, 0.4)),
            ('gauss0.5', _make_profile_gaussian(n, 0.5)),
            ('flat0.6', _make_profile_flat_top(n, 0.6)),
            ('flat0.8', _make_profile_flat_top(n, 0.8)),
        ]
        for label, p in profs:
            if time.monotonic() > phase1_end:
                break
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    # Parent and interpolated parent
    if parent is not None:
        for n in [1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Adam polish top candidates (100s-700s) ---
    phase2_end = t_start + 700.0
    polished = []
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= 8:
            break
        per_end = min(phase2_end, time.monotonic() + 65.0)
        if per_end - time.monotonic() < 15:
            break
        p_b, sc_b = _adam_logit(p, per_end, rng, lr=0.05, beta_frac=0.08,
                                beta_decay=0.93, no_imp_limit=300)
        _update(p_b, sc_b, f'polish_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_pol'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 3: LP refinement (700s-1050s) ---
    phase3_end = t_start + 1050.0
    if best_seq is not None and best_seq.size <= 2048:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")
        p_lp, sc_lp = _lp_refine(best_seq, min(phase3_end, time.monotonic() + 120),
                                  t_init=0.4, max_solves=20)
        _update(p_lp, sc_lp, 'LP_refine')

        # Adam polish after LP
        if time.monotonic() < phase3_end - 30:
            per = min(90.0, phase3_end - time.monotonic() - 10)
            if per > 20:
                p_ap, sc_ap = _adam_logit(p_lp, time.monotonic() + per, rng,
                                           lr=0.03, beta_frac=0.06,
                                           beta_decay=0.94, no_imp_limit=300)
                _update(p_ap, sc_ap, 'post_LP')

    # Also LP refine the second best if it's different scale
    if len(polished) > 1 and polished[1][2] != (best_seq.size if best_seq is not None else 0):
        if time.monotonic() < phase3_end - 30 and polished[1][2] <= 2048:
            per = min(60.0, phase3_end - time.monotonic() - 10)
            if per > 20:
                p_lp2, sc_lp2 = _lp_refine(polished[1][1], time.monotonic() + per,
                                            t_init=0.35, max_solves=10)
                _update(p_lp2, sc_lp2, f'LP_2nd_n{polished[1][2]}')

    # --- Phase 4: Projected gradient on best (1050s-1250s) ---
    phase4_end = t_start + 1250.0
    if best_seq is not None and time.monotonic() < phase4_end:
        per = min(80.0, phase4_end - time.monotonic() - 10)
        if per > 15:
            p_pg, sc_pg = _projected_gradient(best_seq, time.monotonic() + per, rng,
                                              lr_init=0.005)
            _update(p_pg, sc_pg, 'proj_grad')

            # Adam after projected gradient
            if time.monotonic() < phase4_end - 10:
                per2 = min(40.0, phase4_end - time.monotonic() - 5)
                if per2 > 10:
                    p_ag, sc_ag = _adam_logit(p_pg, time.monotonic() + per2, rng,
                                               lr=0.02, beta_frac=0.05,
                                               beta_decay=0.95, no_imp_limit=200)
                    _update(p_ag, sc_ag, 'adam_after_pg')

    # --- Phase 5: Multi-scale escalation (1250s-1550s) ---
    phase5_end = t_start + 1550.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            per = min(80.0, phase5_end - time.monotonic() - 30)
            if per > 20:
                p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                         lr=0.04, beta_frac=0.07,
                                         beta_decay=0.93, no_imp_limit=250)
                _update(p_e, sc_e, f'escalate_n{tn}')

    # --- Phase 6: Multi-restart at best scale (1550s-1720s) ---
    phase6_end = t_start + 1720.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.05 + 0.04 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per, rng, lr=0.04,
                                     beta_frac=0.06, beta_decay=0.94,
                                     no_imp_limit=int(0.3 * n_best))
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 7: Final ultra-fine polish (1720s-deadline) ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            p_f, sc_f = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.01, beta_frac=0.03, beta_decay=0.96,
                                     no_imp_limit=150)
            _update(p_f, sc_f, 'final_polish')

        # One more LP squeeze
        if time.monotonic() < deadline - 10 and best_seq.size <= 2048:
            per2 = deadline - time.monotonic() - 10
            if per2 > 20:
                p_fl, sc_fl = _lp_refine(best_seq, time.monotonic() + per2,
                                          t_init=0.2, max_solves=10)
                _update(p_fl, sc_fl, 'final_LP')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(256, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512")
    return [float(x) for x in fb]
