def next_pow2(n):

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n + 1) > (cumsum - 1))[0][-1]
    theta = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

"""
Exact subgradient mirror descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses direct subgradient computation (no soft-max smoothing) with adaptive step sizes,
asymmetric profile initializations, multi-scale refinement (512→4096), and
momentum-based logit-space updates for focused peak suppression.

import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def exact_subgradient(a, c, k_index):
    """Compute subgradient of c[k_index] = sum_j a[j]*a[k_index-j] w.r.t. a."""
    n = a.size
    ki = int(k_index)
    grad = np.zeros(n)
    lo = max(0, ki - (n - 1))
    hi = min(n - 1, ki)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * a[ki - indices]
    return grad


def multi_subgradient(a, c, top_k, sharpness=0.02):
    """Weighted combination of subgradients at top_k peaks."""
    n = a.size
    if top_k <= 1:
        k_max = int(np.argmax(c))
        return exact_subgradient(a, c, k_max)
    peak_indices = np.argsort(c)[-top_k:]
    c_max = float(c.max())
    diffs = c_max - c[peak_indices]
    scale = max(1e-30, c_max * sharpness)
    weights = np.exp(-diffs / scale)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(weights)
        weights[0] = 1.0
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        grad += w * exact_subgradient(a, c, k)
    return grad


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n + 1) > (cumsum - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def mirror_descent_optimize(a0, deadline, rng, lr=0.02, lr_decay=0.9995,
                            top_k=1, sharpness=0.02, momentum_b=0.85,
                            restart_period=5000, clip_rng=50.0):
    """
    Mirror descent in logit space with exact subgradient and momentum.
    Minimizes 2n * max(conv(a,a)) on the simplex.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)

    current_lr = lr
    no_improve = 0
    momentum = np.zeros(n)
    total_iters = 0

    while time.monotonic() < deadline:
        # Compute subgradient at current max
        grad = multi_subgradient(a0, c, top_k, sharpness)

        # Logit-space gradient: dz = a * (grad - dot(a, grad))
        dot = float(np.dot(a0, grad))
        dz = a0 * (grad - dot)

        # Momentum
        momentum = momentum_b * momentum + (1.0 - momentum_b) * dz

        # Update logits
        z = z - current_lr * momentum
        z = np.clip(z, -clip_rng, clip_rng)
        z = z - z.mean()

        # Softmax
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-300:
            z = best_z.copy()
            a0 = best_a.copy()
            momentum.fill(0.0)
            continue
        a0 = a0 / s

        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        total_iters += 1

        # Adapt learning rate
        if no_improve > 300:
            current_lr *= lr_decay
        if no_improve > 1500:
            momentum_b = max(0.5, momentum_b * 0.995)

        # Gradually increase top_k for stability near convergence
        eff_top_k = top_k
        if no_improve > 3000:
            eff_top_k = min(top_k + 2, 7)

        # Random restart
        if no_improve > restart_period:
            noise_scale = 0.08 + 0.04 * min(5, no_improve // restart_period)
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -clip_rng, clip_rng)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            current_lr = lr
            momentum_b = 0.85
            momentum.fill(0.0)
            no_improve = 0

    return best_a, best_score

def make_profiles(n, rng):
    """Generate diverse asymmetric starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear ramps (asymmetric)
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Exponential ramps
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(alpha * (0.5 - t))
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Power ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (1.0 - t + 0.1) ** p
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # V-shaped and U-shaped
    for alpha in [1.0, 2.0, 3.0]:
        a = np.abs(t - 0.3) ** alpha + 0.1
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
        a = np.abs(t - 0.7) ** alpha + 0.1
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Cosine (asymmetric phase)
    for k in [1, 2, 3]:
        for phase in [0, 0.3, 0.7]:
            a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t + phase)
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())

    # Random smooth (low frequency Fourier)
    for _ in range(10):
        coeffs = rng.uniform(-0.6, 0.6, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t + rng.uniform(0, 1))
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Bimodal asymmetric
    for _ in range(4):
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        w1 = rng.uniform(0.05, 0.2)
        w2 = rng.uniform(0.05, 0.2)
        a = np.exp(-((t - c1) / w1)**2) + 0.5 * np.exp(-((t - c2) / w2)**2)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    return profiles


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def backtracking_mirror_descent(a0, deadline, rng, lr_init=0.05, top_k=1,

    More robust step size selection.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)

    current_lr = lr_init
    no_improve = 0
    momentum = np.zeros(n)

    while time.monotonic() < deadline:
        grad = multi_subgradient(a0, c, top_k, sharpness)
        dot = float(np.dot(a0, grad))
        dz = a0 * (grad - dot)
        momentum = momentum_b * momentum + (1.0 - momentum_b) * dz

        # Try step with backtracking
        accepted = False
        lr_try = current_lr
        for bt in range(max_bt):
            z_try = z - lr_try * momentum
            z_try = np.clip(z_try, -50.0, 50.0)
            z_try = z_try - z_try.mean()
            a_try = np.exp(z_try - z_try.max())
            s_try = a_try.sum()
            if s_try < 1e-300:
                lr_try *= 0.5
                continue
            a_try = a_try / s_try
            c_try = full_convolve_fft(a_try)
            score_try = 2.0 * n * float(c_try.max())

            if score_try < best_score * (1.0 + 1e-8):
                z = z_try
                a0 = a_try
                c = c_try
                current_lr = lr_try
                accepted = True
                if score_try < best_score - 1e-14:
                    best_score = score_try
                    best_a = a0.copy()
                    best_z = z.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            else:
                lr_try *= 0.5

        if not accepted:
            no_improve += 1
            if no_improve > 100:
                z = best_z.copy()
                a0 = best_a.copy()
                c = full_convolve_fft(a0)
                current_lr = lr_init * 0.5
                momentum.fill(0.0)
                no_improve = 0

        if no_improve > 2000:
            momentum_b = max(0.3, momentum_b * 0.99)

        if no_improve > restart_period:
            noise_scale = 0.1
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            current_lr = lr_init
            momentum_b = 0.8
            momentum.fill(0.0)
            no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-scale exact subgradient optimization.
    Phase 1: Explore at n=512 with diverse profiles
    Phase 2: Refine at n=1024 with mirror descent
    Phase 3: Scale to n=2048
    Phase 4: Scale to n=4096 if beneficial
    Phase 5: Final polish with very small steps
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None:
            sc = float(sc)
            if np.isfinite(sc) and sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = sc

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"=== Subgradient Mirror Descent Search ===")
    print(f"Budget: {budget_s}s, Deadline: {deadline - t_start:.1f}s from start")

    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial sequence: n={init_seq.size}, score={sc_init:.10f}")
        try_update(init_seq, sc_init)

    # ===== Phase 1: Exploration at n=512 =====
    n1 = 512
    phase1_end = min(deadline, t_start + 180.0)
    print(f"\n--- Phase 1: n={n1} (exploration) ---")

    candidates = []
    if init_seq is not None:
        candidates.append(interpolate_profile(init_seq, n1))
    candidates.extend(make_profiles(n1, rng))

    phase1_best = None
    phase1_score = np.inf
    for idx, p in enumerate(candidates[:15]):
        if time.monotonic() > phase1_end - 5:
            break
        sc = compute_score(p)
        if sc >= phase1_score + 0.2:
            continue
        dl = min(phase1_end, time.monotonic() + 12.0)
        result, sc_r = mirror_descent_optimize(p, dl, rng, lr=0.04, top_k=3,
                                                sharpness=0.03, momentum_b=0.85)
        sc_final = compute_score(result)
        if sc_final < phase1_score:
            phase1_score = sc_final
            phase1_best = result.copy()
        try_update(result, sc_final)
        print(f"  Candidate {idx}: score={sc_final:.8f}")

    if phase1_best is not None:
        print(f"  Phase 1 best: {phase1_score:.10f}")
        try_update(phase1_best, phase1_score)

    # ===== Phase 2: Refinement at n=1024 =====
    n2 = 1024
    phase2_end = min(deadline, t_start + 650.0)
    print(f"\n--- Phase 2: n={n2} (refinement) ---")

    base_2 = interpolate_profile(phase1_best if phase1_best is not None else candidates[0], n2)
    sc2_init = compute_score(base_2)
    try_update(base_2, sc2_init)
    print(f"  Interpolated to n={n2}: {sc2_init:.10f}")

    # Main mirror descent
    dl2a = min(phase2_end, time.monotonic() + 280.0)
    r2, sc2 = mirror_descent_optimize(base_2, dl2a, rng, lr=0.02, top_k=3,
                                       sharpness=0.02, momentum_b=0.88,
                                       restart_period=4000)
    sc2 = compute_score(r2)
    try_update(r2, sc2)
    print(f"  After mirror descent: {sc2:.10f}")

    # Backtracking variant
    if time.monotonic() < phase2_end - 40:
        dl2b = min(phase2_end, time.monotonic() + 200.0)
        r2b, sc2b = backtracking_mirror_descent(r2, dl2b, rng, lr_init=0.03,
                                                 top_k=1, sharpness=0.01, momentum_b=0.75)
        sc2b = compute_score(r2b)
        try_update(r2b, sc2b)
        print(f"  After backtracking: {sc2b:.10f}")
        if sc2b < sc2:
            r2 = r2b
            sc2 = sc2b

    # Euclidean projected gradient
    if time.monotonic() < phase2_end - 30:
        dl2c = min(phase2_end, time.monotonic() + 100.0)
        r2c, sc2c = euclidean_projected_optimize(r2, dl2c, rng, lr=0.0004, top_k=5,
                                                  sharpness=0.05)
        sc2c = compute_score(r2c)
        try_update(r2c, sc2c)
        print(f"  After euclidean proj: {sc2c:.10f}")
        if sc2c < sc2:
            r2 = r2c
            sc2 = sc2c

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Scale to n=2048 =====
    n3 = 2048
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\n--- Phase 3: n={n3} ---")

    base_3 = interpolate_profile(best_seq, n3) if best_seq is not None else interpolate_profile(r2, n3)
    sc3_init = compute_score(base_3)
    try_update(base_3, sc3_init)
    print(f"  Interpolated to n={n3}: {sc3_init:.10f}")

    dl3a = min(phase3_end, time.monotonic() + 350.0)
    r3, sc3 = mirror_descent_optimize(base_3, dl3a, rng, lr=0.012, top_k=3,
                                       sharpness=0.015, momentum_b=0.9,
                                       restart_period=6000)
    sc3 = compute_score(r3)
    try_update(r3, sc3)
    print(f"  After mirror descent: {sc3:.10f}")

    # Refine with backtracking
    if time.monotonic() < phase3_end - 50:
        dl3b = min(phase3_end, time.monotonic() + 200.0)
        r3b, sc3b = backtracking_mirror_descent(r3, dl3b, rng, lr_init=0.015,
                                                 top_k=1, sharpness=0.01, momentum_b=0.8)
        sc3b = compute_score(r3b)
        try_update(r3b, sc3b)
        print(f"  After backtracking: {sc3b:.10f}")
        if sc3b < sc3:
            r3 = r3b
            sc3 = sc3b

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Scale to n=4096 =====
    if best_score > 1.504 and time.monotonic() < deadline - 250:
        n4 = 4096
        phase4_end = min(deadline, t_start + 1550.0)
        print(f"\n--- Phase 4: n={n4} ---")

        base_4 = interpolate_profile(best_seq, n4)
        sc4_init = compute_score(base_4)
        try_update(base_4, sc4_init)
        print(f"  Interpolated to n={n4}: {sc4_init:.10f}")

        dl4a = min(phase4_end, time.monotonic() + 250.0)
        r4, sc4 = mirror_descent_optimize(base_4, dl4a, rng, lr=0.008, top_k=3,
                                           sharpness=0.012, momentum_b=0.9,
                                           restart_period=5000)
        sc4 = compute_score(r4)
        try_update(r4, sc4)
        print(f"  After mirror descent: {sc4:.10f}")

        print(f"  Phase 4 best: {best_score:.10f}")

    # ===== Phase 5: Final polish =====
    print(f"\n--- Phase 5: Final polish ---")
    polish_end = deadline
    polish_idx = 0

    for round_id in range(8):
        if time.monotonic() > polish_end - 8:
            break
        npol = best_seq.size if best_seq is not None else 1024
        dl = min(polish_end, time.monotonic() + 60.0)

        # Alternate strategies
        if round_id % 3 == 0:
            # Very small step mirror descent, exact subgradient
            r, sc = mirror_descent_optimize(best_seq, dl, rng, lr=0.004, top_k=1,
                                             sharpness=0.005, momentum_b=0.92,
                                             restart_period=10000)
        elif round_id % 3 == 1:
            # Backtracking with single peak
            r, sc = backtracking_mirror_descent(best_seq, dl, rng, lr_init=0.006,
                                                 top_k=1, sharpness=0.005, momentum_b=0.85)
        else:
            # Euclidean projected
            r, sc = euclidean_projected_optimize(best_seq, dl, rng, lr=0.0001, top_k=3,
                                                  sharpness=0.03)

        sc = compute_score(r)
        try_update(r, sc)
        polish_idx += 1
        print(f"  Polish {polish_idx} (n={npol}): {sc:.10f}")
        if sc >= best_score:
            break

    # Try reducing dimension
    if best_seq is not None and best_seq.size > 1024:
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > polish_end - 10:
                break
            p_small = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_small)
            try_update(p_small, sc_s)
            if sc_s < best_score + 0.003:
                dl_s = min(polish_end, time.monotonic() + 25.0)
                r_s, _ = mirror_descent_optimize(p_small, dl_s, rng, lr=0.01, top_k=3)
                sc_s2 = compute_score(r_s)
                try_update(r_s, sc_s2)
                print(f"  Reduced n={n_try}: {sc_s2:.10f}")

    print(f"\n=== Final best score: {best_score:.10f} ===")

    # Final validation and return
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profiles(512, rng)[0]

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profiles(256, rng)[0]

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
