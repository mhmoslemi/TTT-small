def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_grad(a, beta):
    """Compute smooth max value and gradient w.r.t. a using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(2 * n - 1), mval
    w = ex / s
    val = mval + beta * float(np.log(s))
    # Gradient via FFT
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, mval


def _subgrad_max_grad(a):
    """Compute exact max value and subgradient (uniform weight on argmax set)."""
    n = len(a)
    c = _fft_convolve(a)

    active = c >= threshold
    n_active = active.sum()
    if n_active == 0:

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)

    return c_max, grad


def _make_flat_spectrum_init(n, rng, structured=False):
    """Initialize from flat power spectrum."""
    N = _next_pow2(n)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    if structured:
        # Chirp-like phase: quadratic in frequency
        k = np.arange(N // 2 + 1)
        phases = 2 * np.pi * (k ** 2) / N
        X[k] = np.exp(1j * phases)
        for i in range(1, N // 2 + 1):
            X[N - i] = np.exp(-1j * phases[i])
    else:
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        for k in range(1, N // 2 + 1):
            X[k] = np.exp(1j * phases[k])
            X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_u_shaped_init(n, alpha=2.0):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:

    return p / s


def _adam_nesterov_optimize(a, deadline, max_iter=80000, lr=0.04,

    a = a / a.sum()
    n = len(a)

    z -= float(np.mean(z))

    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    momentum = np.zeros(n) if use_nesterov else None
    mu = 0.9 if use_nesterov else 0.0
    adam_t = 0

    c = _fft_convolve(a)

            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)

            adam_t = 0
            beta = max(1e-6, beta_init_frac * float(_fft_convolve(a).max()))

            no_imp += 1
            if no_imp % 5 == 0:
                inflate = min(1.002, inflate * 1.03)
                blend = max(0.08, blend * 0.97)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Tighten on improvement
            if no_imp == 0:
                inflate = max(1.0002, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.95)
            if no_imp > max(100, n):
                a = best_a.copy()
                inflate = 1.0005
                blend = 0.2
                no_imp = 0

        a = a_new

    return best_a


def _peak_responsibility_update(a, deadline, max_iter=3000, n_top=8, strength=0.04):
    """
    Targeted peak reduction: compute responsibility of each coordinate
    for the top peaks, then exponentially suppress high-responsibility coords.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_top]
        top_vals = c[top_idx]

        # Weights: proportional to excess over the median of all c values
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights = weights / wsum

        # Compute responsibility (pressure) on each coordinate
        # For peak at index k, coordinate j is responsible if 0<=k-j<n or 0<=j-k<n
        pressure = np.zeros(n)
        N = _next_pow2(2 * n - 1)
        fw_full = np.zeros(N)
        for idx, w in zip(top_idx, weights):
            fw_full[idx] += w
        # pressure[j] = sum_k fw[k] * (a[k-j] if valid + a[j-k] if valid)
        # = cross-correlation of fw with a
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(fw_full, N)
        # cross_corr[j] = sum_k fw[k] * a[k-j] (circular)
        # We want linear: sum_k fw[k] * a[k-j] for 0<=k-j<n
        h1 = np.fft.irfft(fw * fa, N)  # circular conv fw * a
        # For linear cross-correlation, we need to handle the indices carefully
        # h1[m] = sum_k fw[k] * a[(m-k) mod N]
        # We want: for each j, sum_k fw[k] * a[k-j] (when 0<=k-j<n)
        # = sum_k fw[k] * a[k-j] = cross-correlation
        # cross_corr[j] = sum_k fw[k] * conj_a[k-j] where conj_a[i] = a[i] (real)
        # In FFT: cross_corr = ifft(fft(fw) * conj(fft(a)))... 
        # Actually for real signals: ifft(fft(fw) * conj(fft(a))) gives cross-correlation
        # But since a is real, conj(fft(a)) = fft(a[::-1]) (approximately)
        # Let's just use the direct formula: pressure[j] = sum over active k of w[k] * a[k-j] + w[k] * a[j-k]
        # This is: (fw * a_reversed)[j+n-1] + (fw * a)[j+n-1]... 
        # Simpler: just do it directly for the top peaks (n_top is small)
        pressure = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            # a[k-j] for 0<=k-j<n means j in [k-n+1, k] intersect [0, n-1]
            lo1 = max(0, idx - n + 1)
            hi1 = min(n - 1, idx)
            for j in range(lo1, hi1 + 1):
                pressure[j] += w * a[idx - j]
            # a[j-k] for 0<=j-k<n means j in [k, k+n-1] intersect [0, n-1]
            lo2 = max(0, idx)
            hi2 = min(n - 1, idx + n - 1)
            for j in range(lo2, hi2 + 1):
                if j != idx - j:  # avoid double counting when k-j = j-k (i.e., k=2j)
                    pressure[j] += w * a[j - idx]

        # Exponential suppression
        pmax = pressure.max()
        if pmax < 1e-15:
            break
        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score_new = _score(a_new)
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.15, str_cur * 1.005)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.01, str_cur * 0.92)
            if no_imp > 800:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _final_fine_polish(a, deadline, max_iter=20000):
    """Very fine subgradient polish with tiny steps."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_max, grad = _subgrad_max_grad(a)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)) or np.max(np.abs(gz)) < 1e-14:
            break

        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        score = _score(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.0005, lr * 0.95)
            if no_imp > 5000:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diversified initialization (flat-spectrum, U-shaped, parent)
    2. Adam+Nesterov smooth-max optimization with curriculum
    3. Exact subgradient refinement
    4. LP-based spreading with progressive tightening
    5. Peak responsibility-based reduction
    6. Final fine polish
    Explores multiple sequence lengths to find the global best.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Lengths to explore
    lengths = [768, 512, 1024]
    if parent is not None:
        p_len = parent.size
        if p_len not in lengths:
            lengths.insert(0, p_len)

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        print(f"\n=== Processing n={n}, time left: {deadline - time.monotonic():.0f}s ===", flush=True)

        # Generate initializations
        inits = []
        if parent is not None:
            inits.append(('parent', _interpolate_profile(parent, n)))
        inits.append(('flat_spec', _make_flat_spectrum_init(n, rng)))
        inits.append(('flat_struct', _make_flat_spectrum_init(n, rng, structured=True)))
        inits.append(('u_shape', _make_u_shaped_init(n, 2.0)))
        inits.append(('u_shape3', _make_u_shaped_init(n, 3.0)))
        inits.append(('w_shape', _make_w_shaped_init(n, 3.0)))
        inits.append(('gauss', _make_gaussian_window(n, 0.35)))
        inits.append(('gauss_wide', _make_gaussian_window(n, 0.5)))

        # Score all inits
        init_scores = []
        for name, a0 in inits:
            sc = _score(a0)
            init_scores.append((sc, name, a0))
        init_scores.sort(key=lambda x: x[0])

        print(f"  Top 4 inits for n={n}:", flush=True)
        for sc, name, _ in init_scores[:4]:
            print(f"    {name}: {sc:.8f}", flush=True)

        # Optimize top 3 initializations
        for sc0, name, a0 in init_scores[:3]:
            if time.monotonic() > deadline - 30:
                break
            if sc0 >= best_score * 1.3:
                continue  # Skip if clearly worse

            a_curr = a0.copy()

            # Phase 1: Adam+Nesterov with smooth max (main workhorse)
            alloc1 = min(120.0, max(30.0, (deadline - time.monotonic()) * 0.35))
            d1 = time.monotonic() + alloc1
            print(f"  [{name}] Adam+Nesterov, budget={alloc1:.0f}s", flush=True)
            a_opt = _adam_nesterov_optimize(a_curr, d1, max_iter=60000, lr=0.04,
                                            beta_init_frac=0.05, use_nesterov=True)
            sc = _score(a_opt)
            print(f"  [{name}] After Adam: {sc:.10f}", flush=True)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
            a_curr = a_opt

            # Phase 2: Exact subgradient refinement
            if time.monotonic() < deadline - 30:
                alloc2 = min(60.0, max(15.0, (deadline - time.monotonic()) * 0.2))
                d2 = time.monotonic() + alloc2
                print(f"  [{name}] Subgradient exact, budget={alloc2:.0f}s", flush=True)
                a_sg = _subgradient_exact_optimize(a_curr, d2, max_iter=30000, lr_init=0.015)
                sc = _score(a_sg)
                print(f"  [{name}] After subgrad: {sc:.10f}", flush=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_sg.copy()
                a_curr = a_sg

            # Phase 3: LP refinement with tightening
            if time.monotonic() < deadline - 20:
                alloc3 = min(90.0, max(20.0, (deadline - time.monotonic()) * 0.3))
                d3 = time.monotonic() + alloc3
                print(f"  [{name}] LP refine, budget={alloc3:.0f}s", flush=True)
                a_lp = _lp_refine_tighten(a_curr, d3, max_iter=10000)
                sc = _score(a_lp)
                print(f"  [{name}] After LP: {sc:.10f}", flush=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_lp.copy()
                a_curr = a_lp

            # Phase 4: Peak responsibility
            if time.monotonic() < deadline - 15:
                alloc4 = min(30.0, max(10.0, (deadline - time.monotonic()) * 0.15))
                d4 = time.monotonic() + alloc4
                a_pr = _peak_responsibility_update(a_curr, d4, max_iter=2000,
                                                   n_top=8, strength=0.03)
                sc = _score(a_pr)
                print(f"  [{name}] After peak resp: {sc:.10f}", flush=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_pr.copy()
                a_curr = a_pr

            # Phase 5: Final polish
            if time.monotonic() < deadline - 10:
                alloc5 = min(30.0, max(10.0, (deadline - time.monotonic()) * 0.15))
                d5 = time.monotonic() + alloc5
                a_fin = _final_fine_polish(a_curr, d5, max_iter=15000)
                sc = _score(a_fin)
                print(f"  [{name}] After polish: {sc:.10f}", flush=True)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_fin.copy()
                a_curr = a_fin

    # Final: one more LP pass on the overall best with remaining time
    if best_seq is not None and time.monotonic() < deadline - 5:
        d_final = deadline - 3
        if d_final - time.monotonic() > 15:
            print(f"\nFinal LP pass, budget={d_final - time.monotonic():.0f}s", flush=True)
            a_last = _lp_refine_tighten(best_seq, d_final, max_iter=8000)
            sc = _score(a_last)
            if sc < best_score:
                best_score = sc
                best_seq = a_last.copy()
            print(f"  Final LP: {sc:.10f}", flush=True)

    # Safety fallback
    if best_seq is None:
        best_seq = _make_u_shaped_init(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_u_shaped_init(768, 2.0)

    # Validate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
