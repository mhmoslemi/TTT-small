import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _conv_matrix_full(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def _conv_matrix_active(a, M, n_active=200):
    n = len(a)
    c_aa = _fft_convolve(a)
    # Get the most active constraints
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:n_active]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    return A


def _lp_direction(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    if n <= 256:
        A = _conv_matrix_full(a)
        b = np.full(2 * n - 1, M)
    else:
        A = _conv_matrix_active(a, M, n_active=min(300, 2 * n))
        b = np.full(len(A), M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _grad_optimize(a, deadline, max_iter=50000, T=0.01, lr=0.5):
    """Gradient optimization using log-sum-exp smoothing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()
    best_a = a.copy()
    best_score = _score_norm(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    no_imp = 0
    b1, b2 = 0.9, 0.999

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = c.max()
        x = (c - m_val) / T
        x = np.clip(x, -40, 40)
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s_ex

        # Gradient of smooth max
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h[n - 1:2 * n - 1]

        # Project to tangent space (sum=1 constraint)
        grad_tan = grad - a * (grad @ a)
        grad_tan -= grad_tan.mean()

        t_count += 1
        m = b1 * m + (1 - b1) * grad_tan
        v = b2 * v + (1 - b2) * (grad_tan ** 2)
        m_hat = m / (1 - b1 ** t_count)
        v_hat = v / (1 - b2 ** t_count)

        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z -= z.mean()
        z = np.clip(z, -30, 30)
        a_new = np.exp(z)
        a_new /= a_new.sum()

        sc = _score_norm(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0

def _interpolate_profile(p, n_new):
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _perturb(a, rng, scale=0.03):

    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-10:
        return a.copy()
    return a_p / s

"""
Precision LP-based optimization with adaptive inflation, periodic line search,
gradient-assisted restarts, and multi-length exploration to minimize
2n*max(conv(a,a))/sum(a)^2. Builds on parent sequence with fine-tuned LP
iterations and stochastic perturbations to break local minima.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _conv_matrix_full(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def _lp_direction(a, M, full=True):
    n = len(a)
    c_obj = -np.ones(n)
    if full or n <= 512:
        A = _conv_matrix_full(a)
        b = np.full(2 * n - 1, M)
    else:
        c_aa = _fft_convolve(a)
        idx = np.argsort(c_aa)[::-1]
        active = set()
        n_act = min(400, 2 * n - 1)
        for i in idx[:n_act]:
            for offset in range(-3, 4):
                j = i + offset
                if 0 <= j < 2 * n - 1:
                    active.add(j)
        active = sorted(active)
        A = np.zeros((len(active), n))
        for row, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            A[row, lo:hi] = a[lo:hi]
        b = np.full(len(active), M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_quick(a, g, n_evals=12):
    """Efficient line search with coarse-to-fine strategy."""
    best_t = 0.0
    best_sc = _score_norm(a)
    # Coarse grid
    candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.5]
    for t in candidates:
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        return 0.0
    # Fine around best
    lo = max(0.0, best_t - 0.03)
    hi = min(1.0, best_t + 0.03)
    for _ in range(6):
        mid1 = lo + (hi - lo) / 3
        mid2 = hi - (hi - lo) / 3
        a1 = np.maximum((1 - mid1) * a + mid1 * g, 0)
        a2 = np.maximum((1 - mid2) * a + mid2 * g, 0)
        s1 = a1.sum()
        s2 = a2.sum()
        sc1 = _score_norm(a1 / s1) if s1 > 1e-10 else np.inf
        sc2 = _score_norm(a2 / s2) if s2 > 1e-10 else np.inf
        if sc1 < sc2:
            hi = mid2
            if sc1 < best_sc:
                best_sc = sc1
                best_t = mid1
        else:
            lo = mid1
            if sc2 < best_sc:
                best_sc = sc2
                best_t = mid2
    return best_t


def _lp_optimize(a, deadline, max_iter=5000, inflate_init=1.0002, ls_every=5):
    """LP-based optimization with adaptive inflation and periodic line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    cur_inflate = inflate_init
    base_blend = 0.08

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_direction(a, M * cur_inflate, full=(n <= 2048))

            cur_inflate = min(1.005, cur_inflate * 1.5)
            no_imp += 5
            if no_imp > 400:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 30 == 0:
                cur_inflate = min(1.005, cur_inflate * 1.15)
            if no_imp > 400:
                break
            continue

        g = g / sum_g

        if it % ls_every == 0:
            t = _line_search_quick(a, g)
            if t < 0.005:
                t = base_blend * 0.5
        else:
            t = base_blend

        a_new = np.maximum((1 - t) * a + t * g, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

                a = best_a.copy()
                cur_inflate = inflate_init

                no_imp = 0

        a = a_new

    return best_a


def _grad_step_nesterov(a, z, m, v, t_count, T, lr):
    """One Nesterov-accelerated gradient step in logit space."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -40, 40)
    ex = np.exp(x)
    s_ex = float(ex.sum())
    if s_ex < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s_ex

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h[n - 1:2 * n - 1]

    grad_tan = grad - a * float(grad @ a)
    grad_tan -= float(grad_tan.mean())

    t_count += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1 - b1) * grad_tan
    v = b2 * v + (1 - b2) * (grad_tan ** 2)
    m_hat = m / (1 - b1 ** t_count)
    v_hat = v / (1 - b2 ** t_count)

    z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    z -= float(z.mean())
    z = np.clip(z, -30, 30)
    a_new = np.exp(z)
    a_new /= a_new.sum()
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new, z, m, v, t_count


def _grad_optimize(a, deadline, max_iter=30000, T_start=0.02, T_end=0.003, lr=0.4):
    """Adam-based gradient optimization with annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(z.mean())
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter, 1))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a, z, m, v, t_count = _grad_step_nesterov(a, z, m, v, t_count, T, lr)
        sc = _score_norm(a)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 8000:
                rng = np.random.default_rng(12345 + it)
                a = best_a * (1 + 0.04 * rng.normal(size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                z = np.log(a)
                z -= float(z.mean())
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                no_imp = 0
                lr = max(0.1, lr * 0.85)

        T = max(T_end, T * T_ratio)

    return best_a


def _interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _perturb(a, rng, scale):
    a_p = a * (1.0 + scale * rng.normal(size=len(a)))
    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-10:
        return a.copy()
    return a_p / s
