"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Combines Adam smooth-max in log-space, projected subgradient with line search,
LP-based spreading, simulated annealing restarts, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    a = a / s
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_values(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    sm = mval + beta * np.log(s)
    return sm, w


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _adam_logspace(a0, deadline, max_iter=30000, lr=0.03, beta_factor=0.05,

def _multi_peak_subgradient(a0, deadline, max_iter=8000, top_k=5, verbose=False):

        if s_new < 1e-10:

def _lp_spread(a, M):
    """LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        x = np.maximum(res.x, 0.0)
        s = x.sum()
        if s > 1e-10:
            return x
    return None


def _lp_refine(a0, deadline, max_iter=2000, inflate=1.002, blend=0.4, verbose=False):
    """Iterative LP-based refinement."""
    a = np.asarray(a0, dtype=np.float64).ravel().astype(np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        a_normalized = a / a.sum()
        M_norm = float(_fft_convolve(a_normalized).max())
        M_lp_norm = M_norm * inflate_cur
        g = _lp_spread(a_normalized, M_lp_norm)
        if g is None:
            break
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.08)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.1, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.9)
            if no_imp > 150:
                a = best_a.copy()
                no_imp = 0
                blend_cur = blend
                inflate_cur = inflate

        a = a_new

        if verbose and it % 500 == 0:
            print(f"        LP it={it}: best={best_score:.10f}")

    return best_a


def _simulated_annealing(a0, deadline, max_iter=5000, verbose=False):
    """Simulated annealing on the simplex using log-space perturbations."""
    a = np.asarray(a0, dtype=np.float64).ravel().astype(np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    curr_score = best_score
    curr_z = z.copy()
    curr_a = a.copy()

    T = 0.3
    T_min = 0.005
    cooling = 0.997

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        T = max(T_min, T * cooling)
        rng_s = np.random.default_rng(50000 + it)
        magnitude = T * 0.5

        z_pert = curr_z + magnitude * rng_s.normal(size=n)
        z_pert = np.clip(z_pert, -40.0, 40.0)
        z_pert -= float(np.mean(z_pert))

        e = np.exp(z_pert)
        a_pert = e / e.sum()

        c_pert = _fft_convolve(a_pert)
        score_pert = float(2.0 * n * c_pert.max())

        delta = score_pert - curr_score
        if delta < 0 or np.random.random() < np.exp(-delta / (T + 1e-10)):
            curr_z = z_pert
            curr_a = a_pert
            curr_score = score_pert
            if curr_score < best_score:
                best_score = curr_score
                best_a = curr_a.copy()
                best_z = curr_z.copy()
        else:
            # Revert
            pass

        if verbose and it % 2000 == 0:
            print(f"        SA it={it}: best={best_score:.10f}, T={T:.4f}")

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse profile initializations."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []

    # Linear
    for eta in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + eta * u
        a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    a = 1.0 + 1.0 * u
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    a = 1.0 + 2.0 * u
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    # Quadratic
    a = 1.0 + 0.5 * u + 0.3 * u ** 2
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    # Cubic
    a = 1.0 + 1.5 * u - 1.5 * u ** 3
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    # Cosine
    for alpha in [0.3, 0.7, 1.2]:
        a = 1.0 + alpha * np.cos(np.pi * t)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())

    # Bell / Gaussian
    for w in [0.3, 0.5, 0.7, 1.0]:
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * w ** 2))
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())

    # Sinc-like (windowed)
    a = np.sinc(3.0 * (t - 0.5))
    a = a * (1.0 - t ** 2)
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    # Hann window
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    # Raised cosine (smoothstep-like)
    a = t * t * (3.0 - 2.0 * t)
    a = a + 0.3 * (1.0 - a)
    a = np.maximum(a, 1e-15)
    profiles.append(a / a.sum())

    # Flat with edges
    a = np.ones(n)
    edge = int(n * 0.1)
    if edge > 0:
        a[:edge] = np.linspace(0.5, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.5, edge)
    profiles.append(a / a.sum())

    return profiles


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _refined_pipeline(a0, deadline, label=""):
    """Run a refined multi-phase optimization pipeline."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = float(2.0 * n * _fft_convolve(a).max())
    print(f"    {label} start: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a

    # Phase A: Adam (60% of time)
    d_A = time.monotonic() + 0.6 * remaining
    a_A, s_A = _adam_logspace(best_a, d_A, max_iter=20000, lr=0.025,
                              beta_factor=0.05, beta_decay=0.72,
                              beta_decay_every=150, restart_threshold=max(2000, 4 * n))
    if s_A < best_sc:
        best_a, best_sc = a_A, s_A
    print(f"    {label} Adam: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a

    # Phase B: Multi-peak subgradient (20% of time)
    d_B = time.monotonic() + 0.2 * remaining
    a_B, s_B = _multi_peak_subgradient(best_a, d_B, max_iter=5000, top_k=5)
    if s_B < best_sc:
        best_a, best_sc = a_B, s_B
    print(f"    {label} MultiSG: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a

    # Phase C: LP refinement (10% of time)
    d_C = time.monotonic() + 0.1 * remaining
    a_C = _lp_refine(best_a, d_C, max_iter=1000, inflate=1.002, blend=0.35)
    s_C = float(2.0 * n * _fft_convolve(a_C).max())
    if s_C < best_sc:
        best_a, best_sc = a_C, s_C
    print(f"    {label} LP: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_a

    # Phase D: Fine Adam (10% of time)
    d_D = time.monotonic() + 0.1 * remaining
    a_D, s_D = _adam_logspace(best_a, d_D, max_iter=10000, lr=0.008,
                              beta_factor=0.02, beta_decay=0.8,
                              beta_decay_every=100, restart_threshold=max(1000, 3 * n))
    if s_D < best_sc:
        best_a, best_sc = a_D, s_D
    print(f"    {label} FineAdam: {best_sc:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Adam smooth-max, multi-peak subgradient,
    LP-based spreading, and simulated annealing. Explores multiple sequence lengths
    and diverse initializations with careful time management.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent loaded: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        parent_n = parent.size
    else:
        parent_n = 512

    # Time allocation
    # Phase 1: Quick exploration at multiple lengths (40% of time)
    # Phase 2: Deep optimization on best (50% of time)
    # Phase 3: Final polish (10% of time)

    phase1_deadline = t_start + 0.4 * (budget_s - 20)
    phase2_deadline = t_start + 0.9 * (budget_s - 20)

    # === PHASE 1: Exploration ===
    print("=== PHASE 1: Exploration ===")

    lengths_to_try = []
    if parent is not None:
        lengths_to_try = sorted(set([256, 512, 1024, 2048]))
    else:
        lengths_to_try = [256, 512, 1024]

    per_length_time = (phase1_deadline - t_start) / len(lengths_to_try)

    for n_try in lengths_to_try:
        if time.monotonic() > phase1_deadline - 5:
            break
        n_deadline = min(phase1_deadline - 5, time.monotonic() + per_length_time)
        print(f"  --- n={n_try} ---")

        # Generate candidates
        candidates = []
        if parent is not None:
            p_interp = _interpolate_profile(parent, n_try)
            candidates.append(p_interp)
            # Perturbed parent
            p_pert = p_interp * (1.0 + 0.2 * rng.normal(size=n_try))
            p_pert = np.maximum(p_pert, 0.0)
            if p_pert.sum() > 1e-10:
                candidates.append(p_pert / p_pert.sum())

        # Add profile candidates
        profiles = _make_profiles(n_try)
        candidates.extend(profiles[:4])  # Limit to keep time manageable

        # Remove duplicates (by checking if very similar)
        unique_candidates = [candidates[0]]
        for c in candidates[1:]:
            if len(unique_candidates) < 6:
                unique_candidates.append(c)

        # Run quick optimization on each
        per_cand_time = max(5.0, (n_deadline - time.monotonic()) / len(unique_candidates))
        for i, c0 in enumerate(unique_candidates):
            if time.monotonic() > n_deadline - 2:
                break
            c_deadline = min(n_deadline - 2, time.monotonic() + per_cand_time)
            # Quick Adam
            a_q, s_q = _adam_logspace(c0, c_deadline, max_iter=8000, lr=0.03,
                                      beta_factor=0.05, beta_decay=0.7,
                                      beta_decay_every=100, restart_threshold=1500)
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"    n={n_try} cand{i}: {s_q:.10f}")

    print(f"  Phase 1 best: {best_score:.10f} (n={best_seq.size if best_seq is not None else '?'})")

    # === PHASE 2: Deep optimization ===
    print("=== PHASE 2: Deep Optimization ===")
    if best_seq is not None:
        best_n = best_seq.size
        # Try to optimize at the best length and also at nearby lengths
        deep_lengths = [best_n]
        if best_n <= 512:
            deep_lengths.append(1024)
        elif best_n <= 1024:
            deep_lengths.append(2048)
        if best_n >= 1024:
            deep_lengths.append(max(256, best_n // 2))

        deep_lengths = list(dict.fromkeys(deep_lengths))  # Remove duplicates, preserve order
        per_deep_time = (phase2_deadline - time.monotonic()) / len(deep_lengths)

        for n_deep in deep_lengths:
            if time.monotonic() > phase2_deadline - 30:
                break
            n_ddl = min(phase2_deadline - 30, time.monotonic() + per_deep_time)
            print(f"  --- Deep n={n_deep} ---")

            # Start from best interpolated to this length
            start_a = _interpolate_profile(best_seq, n_deep)

            # Run refined pipeline
            a_deep = _refined_pipeline(start_a, n_ddl, label=f"D{n_deep}")

            s_deep = float(2.0 * n_deep * _fft_convolve(a_deep).max())
            if s_deep < best_score:
                best_score = s_deep
                best_seq = a_deep.copy()

            # Also try SA from the result
            if time.monotonic() < n_ddl and (n_ddl - time.monotonic()) > 20:
                sa_ddl = min(n_ddl, time.monotonic() + 30)
                a_sa, s_sa = _simulated_annealing(a_deep, sa_ddl, max_iter=3000)
                if s_sa < best_score:
                    best_score = s_sa
                    best_seq = a_sa.copy()
                    print(f"    SA improved: {best_score:.10f}")

            print(f"    Deep n={n_deep} best: {best_score:.10f}")

    # === PHASE 3: Final polish ===
    print("=== PHASE 3: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        polish_ddl = deadline - 5
        best_n = best_seq.size

        # Fine Adam
        d_f1 = min(polish_ddl - 20, time.monotonic() + 0.5 * (polish_ddl - time.monotonic()))
        a_f1, s_f1 = _adam_logspace(best_seq, d_f1, max_iter=10000, lr=0.005,
                                    beta_factor=0.01, beta_decay=0.85,
                                    beta_decay_every=50, restart_threshold=2000)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Polish Adam: {best_score:.10f}")

        # Subgradient linesearch
        if time.monotonic() < polish_ddl - 10:
            d_f2 = min(polish_ddl - 10, time.monotonic() + 30)
            a_f2, s_f2 = _subgradient_linesearch(best_seq, d_f2, max_iter=3000, base_lr=0.3)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Polish SubGrad: {best_score:.10f}")

        # LP final
        if time.monotonic() < polish_ddl - 5:
            d_f3 = min(polish_ddl - 3, time.monotonic() + 25)
            a_f3 = _lp_refine(best_seq, d_f3, max_iter=500, inflate=1.001, blend=0.2)
            s_f3 = float(2.0 * best_seq.size * _fft_convolve(a_f3).max())
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Polish LP: {best_score:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < polish_ddl - 2:
            a_f4, s_f4 = _adam_logspace(best_seq, polish_ddl - 2, max_iter=5000, lr=0.002,
                                        beta_factor=0.005, beta_decay=0.9,
                                        beta_decay_every=30, restart_threshold=1000)
            if s_f4 < best_score:
                best_score = s_f4
                best_seq = a_f4.copy()
            print(f"  Ultra-final Adam: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_profiles(256)[1].copy()
        best_score = _score_normalized(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = final_seq / s if s > 0 else np.ones(256) / 256
    else:
        # Keep as is - the score is scale invariant
        # But ensure sum is reasonable
        if s > 100:
            final_seq = final_seq / s
            s = 1.0

    # Final evaluation
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                best_score = sc_eval
        except Exception as e:
            print(f"  Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
