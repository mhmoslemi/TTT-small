"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses: (1) Adam with top-k weighted smooth max, (2) L-BFGS polish,
(3) LP-based spreading refinement, (4) targeted mass transfer moves,
(5) multi-length search with adaptive time allocation.
Key innovation: top-k smooth max focuses gradient on binding peaks,
and mass transfer moves provide combinatorial escapes from local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_topk_val(c, beta, k):
    mval = float(c.max())
    n_c = len(c)
    kk = min(k, n_c)
    if kk >= n_c:
        top_idx = np.arange(n_c)
    else:
        top_idx = np.argpartition(c, -kk)[-kk:]
    x = (c[top_idx] - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        return mval, np.zeros_like(c)
    w = np.zeros_like(c)
    w[top_idx] = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _grad_smooth_max(a, beta, k):
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max_topk_val(c, beta, k)
    # grad[i] = 2 * sum_j w[j] * a[j-i] (for valid j-i in [0,n-1])
    # This equals 2 * np.correlate(w, a, mode='valid')[i]
    h = np.correlate(w, a, mode='valid')
    grad = 2.0 * h
    return val, grad, c

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_quadratic_profile(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5) ** 2 * 4.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_skewed_profile(n, gamma=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + gamma * np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _adam_optimize(a, deadline, max_iter=200000, lr=0.03, beta_init=None, topk=5,
                   lp_every=200, lp_inflate=1.0005, lp_blend=0.2,
                   mass_transfer_every=500):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        val, grad_a, c = _grad_smooth_max(p, beta, topk)
        dot = float(np.dot(p, grad_a))
        gz = p * (grad_a - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                base_lr = max(0.002, base_lr * 0.95)

        # Adaptive beta
        if it % 500 == 0 and it > 500:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.3:
                    beta = max(1e-12, beta * 0.6)
                elif peak_gap > beta * 10.0:
                    beta = min(0.2 * c_max_new, beta * 1.5)

        # Periodic LP refinement
        if lp_every > 0 and it % lp_every == 0 and it > 0:
            M_lp = c_max_new * lp_inflate
            g = _lp_spread(a, M_lp)
            if g is not None:
                sum_g = float(g.sum())
                if sum_g > 1.0 + 1e-6:
                    g = g / sum_g
                    a_blend = (1.0 - lp_blend) * a + lp_blend * g
                    a_blend = np.maximum(a_blend, 1e-15)
                    a_blend = a_blend / a_blend.sum()
                    c_bl = _fft_convolve(a_blend)
                    sc_bl = float(2.0 * n * c_bl.max())
                    if sc_bl < best_score:
                        best_score = sc_bl
                        best_a = a_blend.copy()
                        z = np.log(a_blend)
                        z -= np.mean(z)
                        a = a_blend
                        m = np.zeros(n)
                        v_arr = np.zeros(n)
                        adam_t = 0
                        no_imp = 0

        # Periodic mass transfer
        if mass_transfer_every > 0 and it % mass_transfer_every == 0 and it > 0:
            mt_a = _mass_transfer_step(best_a, n_steps=min(20, n // 10))
            mt_sc = _score(mt_a)
            if mt_sc < best_score:
                best_score = mt_sc
                best_a = mt_a.copy()
                a = mt_a.copy()
                z = np.log(a)
                z -= np.mean(z)
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                no_imp = 0

        # Annealing restart
        if no_imp > max(10000, 15 * n):
            noise_scale = 0.15
            rng_mt = np.random.default_rng(it * 31 + 7)
            z_restart = best_z + noise_scale * rng_mt.standard_normal(n)
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr * 0.8
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a

def _mass_transfer_step(a, n_steps=20):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    for step in range(n_steps):
        c = _fft_convolve(a)
        c_max = float(c.max())
        best_peak_idx = int(np.argmax(c))

        # Compute pressure on each coordinate from the top peak
        # c[best_peak_idx] = sum_j a[j]*a[best_peak_idx - j] for valid j
        lo = max(0, best_peak_idx - n + 1)
        hi = min(n - 1, best_peak_idx)
        pressure = np.zeros(n)
        for j in range(lo, hi + 1):
            k = best_peak_idx - j
            if 0 <= k < n:
                pressure[j] += a[k]
                if k != j:
                    pressure[k] += a[j]

        # Find donor (highest pressure) and recipient (lowest pressure, not contributing to peak)
        if pressure.max() < 1e-15:
            break

        donor = int(np.argmax(pressure))
        # Recipient: low pressure and not in the peak's support
        in_peak = np.zeros(n, dtype=bool)
        for j in range(lo, hi + 1):
            k = best_peak_idx - j
            if 0 <= k < n:
                in_peak[j] = True
                in_peak[k] = True

        candidates = np.where(~in_peak)[0]
        if len(candidates) == 0:
            candidates = np.where(pressure == pressure.min())[0]
        recipient = candidates[int(np.argmin(pressure[candidates]))]

        if donor == recipient:
            break

        # Move a small amount of mass
        delta = 0.01 * a[donor]
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        if c_new.max() < c_max - 1e-15:
            a = a_new
        else:
            # Try smaller delta
            delta = 0.003 * a[donor]
            a_new = a.copy()
            a_new[donor] -= delta
            a_new[recipient] += delta
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            c_new = _fft_convolve(a_new)
            if c_new.max() < c_max - 1e-15:
                a = a_new

    return a

def _lbfgs_polish(a, deadline, max_iter=50000):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z0 = np.log(a)
    z0 -= np.mean(z0)

    beta = max(1e-10, 0.01 * float(_fft_convolve(a).max()))

    def objective(z):
        e = np.exp(np.clip(z, -50, 50))
        aa = e / e.sum()
        val, _, _ = _grad_smooth_max(aa, beta, 5)
        return val

    def objective_grad(z):
        e = np.exp(np.clip(z, -50, 50))
        aa = e / e.sum()
        val, grad_a, _ = _grad_smooth_max(aa, beta, 5)
        dot = float(np.dot(aa, grad_a))
        gz = aa * (grad_a - dot)
        gz -= np.mean(gz)
        return val, gz

    best_a = a.copy()
    best_score = _score(a)

    res = minimize(objective, z0, jac=objective_grad, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                            'maxls': 20})

    if res.x is not None and np.all(np.isfinite(res.x)):
        e = np.exp(np.clip(res.x, -50, 50))
        a_new = e / e.sum()
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    # Check time and return
    if time.monotonic() > deadline:
        pass

    return best_a

def _lp_refine_long(a, deadline, max_iter=5000, inflate=1.0003, blend=0.15):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.002, inflate_cur * 1.03)
                blend_cur = max(0.05, blend_cur * 0.95)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.9)
                inflate_cur = min(1.001, inflate_cur * 0.97)
            if no_imp > 300:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

    return best_a

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [768, 1024, 512]:
        if ln not in lengths:
            lengths.append(ln)

    # Phase 1: Quick initialization and evaluation (120s)
    print(f"Phase 1: Initialization, budget={120:.0f}s", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))
        init_pool.append(('linear', n_t, _make_linear_profile(n_t, 1.5)))
        init_pool.append(('linear2', n_t, _make_linear_profile(n_t, 2.5)))
        init_pool.append(('quad', n_t, _make_quadratic_profile(n_t, 1.0)))
        init_pool.append(('skew', n_t, _make_skewed_profile(n_t, 0.5)))

    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print("  Top inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:5]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Main optimization on top candidates (majority of time)
    main_deadline = deadline - 180.0
    top_inits = init_scores[:3]

    for sc0, name, n_t, a0 in top_inits:
        if time.monotonic() > main_deadline:
            break
        remaining = main_deadline - time.monotonic()
        if remaining < 100:
            break

        print(f"\nPhase 2: Optimizing {name} (n={n_t}), start={sc0:.8f}, time_left={remaining:.0f}s", flush=True)

        # Split time between Adam and LP
        adam_deadline = time.monotonic() + remaining * 0.7
        a_opt = _adam_optimize(a0, adam_deadline, max_iter=200000, lr=0.03,
                               topk=5, lp_every=500, lp_inflate=1.0004,
                               lp_blend=0.15, mass_transfer_every=1000)
        sc_opt = _score(a_opt)
        print(f"  After Adam: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < main_deadline - 50:
            lp_dl = min(main_deadline - 30, time.monotonic() + 120.0)
            print(f"  LP refine, budget={lp_dl - time.monotonic():.0f}s", flush=True)
            a_lp = _lp_refine_long(a_opt, lp_dl, max_iter=3000, inflate=1.0003, blend=0.12)
            sc_lp = _score(a_lp)
            print(f"  After LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Mass transfer
        if time.monotonic() < main_deadline - 20:
            a_mt = _mass_transfer_step(a_opt, n_steps=50)
            sc_mt = _score(a_mt)
            print(f"  After mass transfer: {sc_mt:.10f}", flush=True)
            if sc_mt < best_score:
                best_score = sc_mt
                best_seq = a_mt.copy()
            a_opt = a_mt

        # L-BFGS polish
        if time.monotonic() < main_deadline - 10:
            lbfgs_dl = min(main_deadline - 5, time.monotonic() + 60.0)
            if lbfgs_dl - time.monotonic() > 5:
                a_lb = _lbfgs_polish(a_opt, lbfgs_dl, max_iter=10000)
                sc_lb = _score(a_lb)
                print(f"  After L-BFGS: {sc_lb:.10f}", flush=True)
                if sc_lb < best_score:
                    best_score = sc_lb
                    best_seq = a_lb.copy()
                a_opt = a_lb

    # Phase 3: Final polish on the overall best (150s)
    if best_seq is not None and time.monotonic() < deadline - 20:
        polish_deadline = deadline - 15
        polish_budget = polish_deadline - time.monotonic()
        if polish_budget > 30:
            print(f"\nPhase 3: Final polish, budget={polish_budget:.0f}s", flush=True)

            # Another round of careful Adam with smaller lr and topk=3
            adam_dl2 = time.monotonic() + polish_budget * 0.5
            a_p1 = _adam_optimize(best_seq, adam_dl2, max_iter=100000, lr=0.01,
                                   topk=3, lp_every=1000, lp_inflate=1.0002,
                                   lp_blend=0.08, mass_transfer_every=2000)
            sc_p1 = _score(a_p1)
            print(f"  Polish Adam: {sc_p1:.10f}", flush=True)
            if sc_p1 < best_score:
                best_score = sc_p1
                best_seq = a_p1.copy()

            # Final LP
            if time.monotonic() < polish_deadline - 10:
                lp_dl2 = polish_deadline - 5
                if lp_dl2 - time.monotonic() > 10:
                    a_p2 = _lp_refine_long(best_seq, lp_dl2, max_iter=2000,
                                           inflate=1.0002, blend=0.08)
                    sc_p2 = _score(a_p2)
                    print(f"  Polish LP: {sc_p2:.10f}", flush=True)
                    if sc_p2 < best_score:
                        best_score = sc_p2
                        best_seq = a_p2.copy()

            # Final mass transfer
            if time.monotonic() < polish_deadline - 5:
                a_p3 = _mass_transfer_step(best_seq, n_steps=100)
                sc_p3 = _score(a_p3)
                print(f"  Polish MT: {sc_p3:.10f}", flush=True)
                if sc_p3 < best_score:
                    best_score = sc_p3
                    best_seq = a_p3.copy()

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(512, 1.5)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
