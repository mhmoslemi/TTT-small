    """
    Perform a projected subgradient step on f(a) = max_k c[k].
    Uses the weighted subgradient from the top-k largest c[k] values.
    """
    n = len(a)
    c = full_convolve(a, a)
    # Get top-k indices
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[-k:]
    top_vals = c[top_indices]
    # Exponential weights (soft-max over top-k)
    weights = np.exp(top_vals / 0.01 - np.max(top_vals) / 0.01)
    weights = weights / weights.sum()
    # Compute weighted subgradient
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        ki = int(idx)
        if ki < n:
            # c[ki] = sum_j a[j]*a[ki-j] for j=0..ki

    s = a_new.sum()
    if s > 0:

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with adaptive logsumexp continuation, exact subgradient
refinement via very small beta, multi-restart with diverse profiles,
and length scaling. Uses FFT-based convolution for efficiency.
"""
import numpy as np
from scipy.optimize import minimize
import time as time_module


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _obj_fn(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    c_shifted = np.clip(c_shifted, -700, 700)
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _grad_fn(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    c_shifted = np.clip(c_shifted, -700, 700)

    grad = 4.0 * n * a * (h - W)

    return grad


def _run_lbfgs(a_init, deadline, beta_start, n_betas, beta_ratio, max_iter_per=800):
    """Run L-BFGS with continuation over multiple beta levels."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    for i in range(n_betas):
        if time_module.monotonic() >= deadline:
            break
        beta = beta_start * (beta_ratio ** i) * max_c0
        if beta < 1e-10:
            beta = 1e-10

        level_deadline = min(deadline, time_module.monotonic() + (deadline - time_module.monotonic()) / (n_betas - i))

        try:
            result = minimize(
                _obj_fn, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_fn,

                         'maxls': 40, 'maxfun': 3000, 'iprint': -1}

            if result.x is not None and np.all(np.isfinite(result.x)):

                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                z = result.x.copy()
        except Exception:
            pass

def _adam_refine(a_init, deadline, lr=0.02, beta_frac=1e-4, max_iter=50000):
    """Adam-based refinement with very small logsumexp beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    for it in range(max_iter):
        if time_module.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        c_shifted = np.clip(c_shifted, -700, 700)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            beta = max(1e-12, beta * 0.8)
            lr = max(0.001, lr * 0.97)

    return best_a, best_score


def _make_profile(n, kind, params):
    """Generate various profile types."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'linear':
        a = 1.0 + params[0] * u
    elif kind == 'quadratic':
        a = 1.0 + params[0] * u + params[1] * u * u
    elif kind == 'cubic':
        a = 1.0 + params[0] * u + params[1] * u ** 3
    elif kind == 'sinusoid':
        a = 1.0 + params[0] * np.sin(params[1] * np.pi * u + params[2])
    elif kind == 'cos2':
        a = 1.0 + params[0] * np.cos(2.0 * np.pi * t) + params[1] * np.cos(4.0 * np.pi * t)
    elif kind == 'tanh':
        a = 1.0 + params[0] * np.tanh(params[1] * u)
    elif kind == 'gauss':
        a = np.exp(-params[0] * (t - params[1]) ** 2) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        nonlocal best_seq, best_score

            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")

    phase1_end = min(deadline, t_start + 500.0)

            break
        if idx >= n_to_try:
            break
        if sc > best_score * 1.02:

                if sc_best < best_score + 0.001 and time_module.monotonic() < phase4_end - 20:

                                          beta_ratio=0.5, max_iter_per=1500)
