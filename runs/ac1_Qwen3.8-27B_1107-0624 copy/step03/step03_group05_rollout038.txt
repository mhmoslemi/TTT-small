def _simulated_annealing(a, deadline, max_iter=100000, T_init=0.01, T_final=1e-6):
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)
    
    best_a = a.copy()
    best_score = _score(a)
    current_score = best_score
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max_iter)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Random move: transfer mass from one position to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        amount = rng.uniform(0.001, 0.05) * a[i]
        if amount > a[i]:
            continue
        
        a_new = a.copy()
        a_new[i] -= amount
        a_new[j] += amount
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        sc_new = _score(a_new)
        delta_sc = sc_new - current_score
        
        if delta_sc < 0 or rng.random() < np.exp(-delta_sc / T):
            a = a_new
            current_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
        
        T *= cooling
    
    return best_a, best_score

"""
Optimized step function search using direct LP peak minimization,
adaptive gradient methods, simulated annealing, and two-scale
coarse-to-fine refinement. Minimizes 2n*max(conv(a,a))/sum(a)^2.
Key novelty: Linearized LP that directly minimizes the convolution peak
with adaptive step control and line search, combined with multi-peak
awareness and coarse-to-fine multi-resolution strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix_fast(a):
    """Build matrix T such that (T @ delta)[k] = conv(a, delta)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _lp_peak_reduce(a, deadline, max_iter=500, top_k_ratio=0.1):
    """
    Direct LP-based peak minimization using linearized convolution.
    At each step, solves an LP to find the best direction to reduce
    the maximum convolution value, then performs line search.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    step_max = 0.05

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Build convolution matrix
        T = _make_conv_matrix_fast(a)

        # Identify top peaks for focused constraint
        n_peaks = max(3, int(len(c) * top_k_ratio))
        sorted_idx = np.argsort(c)[::-1]
        peak_idx = sorted_idx[:n_peaks]

        # LP: minimize M_new
        # s.t. c[k] + 2*(T@delta)[k] <= M_new  for peak k
        #      c[k] + 2*(T@delta)[k] <= M*1.005 for non-peak k (relaxed)
        #      sum(delta) = 0
        #      -a[i] <= delta[i] <= step_max
        #      M_new >= 0

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = 2.0 * T
        A_ub[:, n] = -1.0

        b_ub = -c.copy()
        # Relax non-peak constraints slightly to make LP less tight
        for k in range(len(c)):
            if k not in set(peak_idx.tolist()):
                b_ub[k] = -(c[k] * 0.995)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = [(-a[i], step_max) for i in range(n)] + [(0, None)]

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            step_max *= 0.7
            if step_max < 1e-8:
                break
            continue

        delta = res.x[:n]

        # Check if delta is meaningful
        if np.max(np.abs(delta)) < 1e-12:
            no_imp += 1
            if no_imp > 30:
                step_max *= 0.5
                if step_max < 1e-8:
                    break
                continue

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]:
            a_try = a + t * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = float(2.0 * n * _fft_convolve(a_try).max())
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_t = t

        if best_t > 0:
            a = a + best_t * delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_a = a.copy()
                no_imp = 0
                # Slightly increase step on success
                step_max = min(0.1, step_max * 1.02)
            else:
                no_imp += 1
        else:
            no_imp += 1
            step_max *= 0.8

        if no_imp > 80:
            if step_max < 1e-6:
                break
            # Reset and try again with smaller step
            step_max = max(1e-6, step_max * 0.3)
            no_imp = 0

    return best_a, best_score


def _lp_refine_v2(a, deadline, max_iter=1000):
    """
    Improved LP refinement: alternates between peak-focusing LP and
    spreading LP to balance peak reduction with overall structure.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        T = _make_conv_matrix_fast(a)

        if it % 3 == 0:
            # Peak-focused LP
            n_peaks = max(2, int(len(c) * 0.05))
            sorted_idx = np.argsort(c)[::-1]
            peak_idx = sorted_idx[:n_peaks]

            c_obj = np.zeros(n + 1)
            c_obj[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = 2.0 * T
            A_ub[:, n] = -1.0
            b_ub = -c.copy()
            for k in range(len(c)):
                if k not in set(peak_idx.tolist()):
                    b_ub[k] = -c[k] * 0.99

            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.zeros(1)
            bounds = [(-a[i], 0.03) for i in range(n)] + [(0, None)]
        elif it % 3 == 1:
            # Spreading LP: maximize sum of delta (push mass toward zero-conv regions)
            # Minimize -sum(delta) subject to c + 2*T@delta <= M*1.001
            c_obj = -np.ones(n)
            c_obj[n + 0] = 0  # placeholder
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = 2.0 * T
            A_ub[:, n] = -1.0
            b_ub = -c * 1.001 - 0.0  # c + 2*T@delta - M <= -c*1.001 + M = c*(1-1.001) + M
            # Actually: c + 2*T@delta <= M*1.001 => 2*T@delta - M <= M*1.001 - c
            b_ub = M * 1.001 - c

            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.zeros(1)
            bounds = [(-a[i], 0.02) for i in range(n)] + [(M, None)]
        else:
            # Uniform constraint LP (balance)
            c_obj = np.zeros(n + 1)
            c_obj[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = 2.0 * T
            A_ub[:, n] = -1.0
            b_ub = -c * 0.999

            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.zeros(1)
            bounds = [(-a[i], 0.02) for i in range(n)] + [(0, None)]

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            no_imp += 1
            if no_imp > 50:
                break
            continue

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-12:
            no_imp += 1
            if no_imp > 50:
                break
            continue

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a + t * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = float(2.0 * n * _fft_convolve(a_try).max())
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_t = t

        if best_t > 0:
            a = a + best_t * delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 100:
            break

    return best_a, best_score


def _gradient_polish(a, deadline, max_iter=20000):
    """Smooth-max gradient descent with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.03 * float(c.max()))
    lr = 0.04
    momentum = 0.88
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        c_arr = _fft_convolve(a_nest)
        w = _smooth_max_weight(c_arr, beta)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h + 1e-8 * a_nest

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.95)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                lr = 0.04
                no_imp = 0

        if it % 80 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a, best_score


def _multi_peak_gradient(a, deadline, max_iter=15000, top_k=5):
    """Gradient that focuses on the top-k peaks simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.02 * float(c.max()))
    lr = 0.03
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        c_arr = _fft_convolve(a_nest)

        # Combine smooth max with peak-focused weight
        w_smooth = _smooth_max_weight(c_arr, beta)
        sorted_idx = np.argsort(c_arr)[::-1]
        w_peak = np.zeros_like(c_arr)
        pk = c_arr[sorted_idx[:top_k]]
        w_peak[sorted_idx[:top_k]] = pk / max(pk.sum(), 1e-30)
        w = (1.0 - 0.4) * w_smooth + 0.4 * w_peak
        w = w / w.sum()

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h + 1e-8 * a_nest

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a = _project_simplex(a + v)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.02 * float(_fft_convolve(a).max()))
                lr = 0.03
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a, best_score


def _simulated_annealing(a, deadline, max_iter=200000, seed=123):
    """Simulated annealing with mass-transfer moves."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = _score(a)
    current_score = best_score

    T_init = 0.005
    T_final = 1e-7
    cooling = (T_final / T_init) ** (1.0 / max(1, max_iter))
    T = T_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Single mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amount = rng.uniform(0.005, 0.08) * max(a[i], 1e-6)
            amount = min(amount, a[i])
            if amount < 1e-10:
                continue
            a_try = a.copy()
            a_try[i] -= amount
            a_try[j] += amount
        elif r < 0.85:
            # Pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            a_try = a.copy()
            a_try[i], a_try[j] = a_try[j], a_try[i]
        else:
            # Block rotation
            blen = rng.integers(2, max(3, n // 16))
            i = rng.integers(0, max(1, n - blen))
            a_try = a.copy()
            a_try[i:i + blen] = np.roll(a_try[i:i + blen], -1)

        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        delta_sc = sc_try - current_score

        if delta_sc < 0 or (T > 1e-9 and rng.random() < np.exp(-delta_sc / T)):
            a = a_try
            current_score = sc_try
            if sc_try < best_score - 1e-12:
                best_score = sc_try
                best_a = a.copy()

        T *= cooling

    return best_a, best_score


def _two_scale_optimize(a, deadline):
    """Coarse-to-fine multi-resolution optimization."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    # Coarsen by factor 2
    n_coarse = n // 2
    if n_coarse < 16:
        return a, _score(a)

    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a[2 * i] + a[2 * i + 1]
    a_coarse = a_coarse / a_coarse.sum()

    # Optimize coarse with LP
    coarse_budget = min(60.0, (deadline - time.monotonic()) * 0.4 - 10)
    if coarse_budget > 10:
        a_coarse, _ = _lp_peak_reduce(a_coarse, time.monotonic() + coarse_budget, max_iter=100)

    # Interpolate back (preserve original shape as much as possible)
    a_interp = np.zeros(n)
    for i in range(n_coarse):
        frac_left = a[2 * i] / max(a[2 * i] + a[2 * i + 1], 1e-12)
        frac_right = 1.0 - frac_left
        a_interp[2 * i] = a_coarse[i] * frac_left
        a_interp[2 * i + 1] = a_coarse[i] * frac_right
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s < 1e-10:
        return a, _score(a)
    a_interp = a_interp / s

    # Refine at full scale
    fine_budget = deadline - time.monotonic() - 10
    if fine_budget > 10:
        a_interp, _ = _lp_peak_reduce(a_interp, time.monotonic() + fine_budget, max_iter=100)

    sc = _score(a_interp)
    sc_orig = _score(a)
    if sc < sc_orig:
        return a_interp, sc
    return a, sc_orig


def _adam_logit_polish(a, deadline, max_iter=20000, lr=0.04):
    """Adam optimization in logit space for smooth updates."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.04 * float(c.max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)
        if it % 120 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(2000, 4 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**31))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n) * 0.5
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.04 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = 0.04
            no_imp = 0

    return best_a, best_score


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.1) / width) ** 2) + np.exp(-0.5 * ((t - 0.9) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _full_pipeline(a_init, deadline, phase_budget):
    """Run full optimization pipeline on an initial sequence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    n = len(a)

    # Phase 1: LP peak reduction (primary)
    lp_budget = min(phase_budget * 0.35, deadline - time.monotonic() - 60)
    if lp_budget > 10:
        a_lp, sc_lp = _lp_peak_reduce(a, time.monotonic() + lp_budget, max_iter=300)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
        print(f"    LP: {sc_lp:.10f}")

    # Phase 2: LP refine v2 (alternating)
    lp2_budget = min(phase_budget * 0.2, deadline - time.monotonic() - 50)
    if lp2_budget > 10:
        a_lp2, sc_lp2 = _lp_refine_v2(best_a, time.monotonic() + lp2_budget, max_iter=500)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()
        print(f"    LP2: {sc_lp2:.10f}")

    # Phase 3: Multi-peak gradient
    grad_budget = min(phase_budget * 0.2, deadline - time.monotonic() - 40)
    if grad_budget > 10:
        a_gp, sc_gp = _multi_peak_gradient(best_a, time.monotonic() + grad_budget, max_iter=10000)
        if sc_gp < best_score:
            best_score = sc_gp
            best_a = a_gp.copy()
        print(f"    Grad: {sc_gp:.10f}")

    # Phase 4: Adam logit
    ad_budget = min(phase_budget * 0.15, deadline - time.monotonic() - 30)
    if ad_budget > 10:
        a_ad, sc_ad = _adam_logit_polish(best_a, time.monotonic() + ad_budget, max_iter=10000)
        if sc_ad < best_score:
            best_score = sc_ad
            best_a = a_ad.copy()
        print(f"    Adam: {sc_ad:.10f}")

    # Phase 5: Simulated annealing (brief)
    sa_budget = min(phase_budget * 0.1, deadline - time.monotonic() - 20)
    if sa_budget > 5:
        a_sa, sc_sa = _simulated_annealing(best_a, time.monotonic() + sa_budget, max_iter=50000)
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa.copy()
        print(f"    SA: {sc_sa:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LP peak minimization,
    multi-peak gradient, Adam logit, simulated annealing, and
    two-scale coarse-to-fine search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 896, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))

    print(f"Trying lengths: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Main search across lengths
    num_lengths = len(lengths)
    per_length_budget = (deadline - time.monotonic() - 200.0) / num_lengths

    for n_t in lengths:
        if time.monotonic() > deadline - 150:
            break

        phase_budget = min(per_length_budget, deadline - time.monotonic() - 100)
        phase_budget = max(30.0, phase_budget)

        if phase_budget < 25:
            continue

        print(f"\nLength {n_t}: budget={phase_budget:.0f}s")

        # Init from parent or profile
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)

        sc_init = _score(a_init)
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        a_best_n, sc_best_n = _full_pipeline(a_init, deadline, phase_budget)
        if sc_best_n < best_score:
            best_score = sc_best_n
            best_seq = a_best_n.copy()
            print(f"  ** New best at n={n_t}: {sc_best_n:.10f}")

    # Phase 2: Diverse initializations at best length
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\nPhase 2: Diverse inits at n={n_best}")

        inits = []
        # U-shaped profiles with various curvatures
        for alpha in [0.5, 1.0, 1.5, 2.5, 4.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Bimodal
        for width in [0.08, 0.12, 0.18, 0.25]:
            inits.append(_make_bimodal_profile(n_best, width))
        # Linear
        for eta in [0.5, 1.0, 2.0, 3.0]:
            inits.append(_make_linear_profile(n_best, eta))
        # Perturbed best
        for sigma in [0.05, 0.1, 0.2]:
            a_pert = best_seq * (1.0 + sigma * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)
        # Random smooth
        for _ in range(2):
            a_rand = np.abs(rng.normal(1.0, 0.4, n_best))
            # Smooth it
            kernel = np.ones(5) / 5
            a_rand = np.convolve(a_rand, kernel, mode='same')
