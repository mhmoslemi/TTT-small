def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _make_conv_matrix_reduced(a, M, top_k=80):

def _lp_spread(a, M, use_reduced=False, top_k=80):

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _poly_penalty_step(a, p, lr):
    """One gradient step for polynomial penalty (sum c^p)^(1/p)."""
    n = len(a)
    c = _fft_convolve(a)

        return a

    A = np.fft.rfft(a, N)

    # We need the linear part: cross[j] for j=0..n-1 should be sum_k w[k]*a[k-j]

    # Actually with N >= len(w) + len(a) - 1, the linear cross-correlation
    # starts at index 0 and has length len(w)+len(a)-1

    # For linear (non-circular), with sufficient padding:

    # But w has length 2n-1, so k in [j, j+n-1] intersect [0, 2n-2]

    # So cross[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j]

    a_new = a - lr * grad

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_parzen(n):
    """Parzen window: (3 - 2|t|)^3 / 8 for |t| < 1.5, 0 otherwise, on [-1,1] mapped to [0,1]."""
    t = np.linspace(-1.0, 1.0, n)
    a = np.zeros(n)
    mask = np.abs(t) < 1.5
    a[mask] = (3.0 - 2.0 * np.abs(t[mask])) ** 3 / 8.0
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_hamming(n):

    a = 0.54 + 0.46 * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_smooth(n, rng):

        top_k = 10
        idx = np.argsort(c)[::-1][:top_k]
        w[idx] = 1.0 / top_k
        # Gradient: 2 * ifft(fft(w) * conj(fft(a)))
        W = np.fft.rfft(w, N)
        grad = 2.0 * np.fft.irfft(W * np.conj(A), N)[:n]
        # Update in log-frequency domain
        A_abs = np.abs(A) + 1e-10
        A_abs = A_abs * np.exp(-0.01 * np.abs(grad))
        a = np.fft.irfft(np.abs(A) * A_abs / np.abs(A) if np.all(np.abs(A) > 1e-10) else A_abs, N)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining polynomial penalty, LSE gradient,
    LP refinement, and coordinate descent. Uses diverse initializations and
    multiple sequence lengths to find near-optimal solutions.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Track best per length
    best_per_length = {}
    
    # Define candidate generators
    def get_candidates(n_t, start_idx=0):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for k in range(2):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.1 * rng.standard_normal(n_t))
                p_pert = np.maximum(p_pert, 1e-12)
                p_pert = p_pert / p_pert.sum()
                cands.append(p_pert)
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_parzen(n_t))
        cands.append(_make_hamming(n_t))
        cands.append(_make_random_smooth(n_t, rng))
        return cands[start_idx:]
    
    # === Phase 1: Polynomial penalty optimization ===
    print("\n=== Phase 1: Polynomial penalty optimization ===")
    
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [1024, 1536, 2048]:
        if ln not in lengths:
            lengths.append(ln)
    
    # Limit to at most 3 lengths for time management
    lengths = lengths[:3]
    
    phase1_time_per_length = min(250.0, (deadline - time.monotonic() - 300.0) / max(len(lengths), 1))
    
    for n_t in lengths:
        remaining = deadline - time.monotonic() - 300.0
        if remaining < 60:
            break
        
        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf
        
        time_per_cand = phase1_time_per_length / max(len(cands), 1)
        
        for ci, a0 in enumerate(cands):
            remaining = deadline - time.monotonic() - 300.0
            if remaining < 30:
                break
            
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            # Polynomial penalty optimization
            p1_end = time.monotonic() + min(time_per_cand, 120.0)
            a_poly = _poly_optimize(a0, p1_end, p_start=2, p_end=80,
                                    max_iter=30000, lr=0.008)
            sc_poly = _score(a_poly)
            
            if np.isfinite(sc_poly) and sc_poly < best_score:
                best_score = sc_poly
                best_seq = a_poly.copy()
            if sc_poly < best_here_score:
                best_here_score = sc_poly
                best_here = a_poly
            
            print(f"  n={n_t}, cand{ci}: init={sc0:.8f} -> poly={sc_poly:.10f}")
        
        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best poly n={n_t}: {best_here_score:.10f}")
    
    # === Phase 2: LSE optimization on best candidates ===
    print("\n=== Phase 2: LSE gradient optimization ===")
    
    phase2_time = min(500.0, deadline - time.monotonic() - 200.0)
    n_cands_phase2 = min(3, len(best_per_length))
    time_per_lse = phase2_time / max(n_cands_phase2, 1)
    
    sorted_lengths = sorted(best_per_length.keys(), key=lambda x: _score(best_per_length[x]))
    
    for n_t in sorted_lengths[:n_cands_phase2]:
        remaining = deadline - time.monotonic() - 200.0
        if remaining < 30:
            break
        
        a_in = best_per_length[n_t]
        lse_end = time.monotonic() + min(time_per_lse, 200.0)
        
        a_lse = _lse_optimize(a_in, lse_end, max_iter=50000,
                              T_start=0.03, T_end=1e-6, lr=0.08)
        sc_lse = _score(a_lse)
        
        if np.isfinite(sc_lse) and sc_lse < best_score:
            best_score = sc_lse
            best_seq = a_lse.copy()
        if sc_lse < _score(best_per_length[n_t]):
            best_per_length[n_t] = a_lse
        
        print(f"  LSE n={n_t}: {sc_lse:.10f}")
    
    # === Phase 3: LP refinement ===
    print("\n=== Phase 3: LP refinement ===")
    
    phase3_time = min(400.0, deadline - time.monotonic() - 100.0)
    n_lp = min(2, len(best_per_length))
    time_per_lp = phase3_time / max(n_lp, 1)
    
    for n_t in sorted_lengths[:n_lp]:
        remaining = deadline - time.monotonic() - 100.0
        if remaining < 30:
            break
        
        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"  LP start n={n_t}: score={sc_in:.10f}")
        
        lp_end = time.monotonic() + min(time_per_lp, 250.0)
        a_lp = _lp_refine(a_in, lp_end, max_iter=2000,
                          inflate=1.0003, blend=0.15)
        sc_lp = _score(a_lp)
        
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        if sc_lp < _score(best_per_length[n_t]):
            best_per_length[n_t] = a_lp
        
        print(f"  LP result n={n_t}: {sc_lp:.10f}")
    
    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate descent polish ===")
    
    if best_seq is not None and time.monotonic() < deadline - 50:
        cd_budget = min(150.0, deadline - time.monotonic() - 50.0)
        if cd_budget > 20:
            cd_end = time.monotonic() + cd_budget
            # Only do coord descent for moderate n
            if best_seq.size <= 1536:
                a_cd = _coord_descent_polish(best_seq, cd_end, max_sweeps=100)
                sc_cd = _score(a_cd)
                if np.isfinite(sc_cd) and sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()
                print(f"  Coord descent: {sc_cd:.10f}")
    
    # === Phase 5: Final combined pass ===
    print("\n=== Phase 5: Final refinement ===")
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 30.0
        
        # Quick LP
        if remaining > 40 and best_seq.size <= 2048:
            lp_end = time.monotonic() + min(40.0, remaining - 20.0)
            a_final = _lp_refine(best_seq, lp_end, max_iter=500,
                                 inflate=1.0002, blend=0.1)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP: {sc_final:.10f}")
        
        # Quick LSE
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            lse_end = time.monotonic() + min(30.0, remaining - 5.0)
            a_last = _lse_optimize(best_seq, lse_end, max_iter=20000,
                                   T_start=0.01, T_end=1e-7, lr=0.04)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LSE: {sc_last:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0
    
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")
    
    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
