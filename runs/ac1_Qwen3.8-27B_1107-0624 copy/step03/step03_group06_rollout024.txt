"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using combined LP-guided gradient optimization.
Features: diverse initializations, Nesterov momentum, backtracking line search,
adaptive smooth-max parameter, LP recentering, perturbation restarts, multi-length search.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) with respect to a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    # Avoid overflow: shift by max
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (with appropriate index handling)
    # This is 2 * convolve(w, a_reversed)[n-1:2n-1]
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval, c


def _make_uniform(n):
    return np.ones(n) / n


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_cosine(n, harmonics=1, offset=0.5):
    t = np.arange(n)
    a = offset
    for h in range(harmonics):
        a = a + np.cos(2.0 * np.pi * (h + 1) * t / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_smooth_noise(n, rng, smoothness=0.3):
    """Generate a smooth random non-negative sequence."""
    # Generate in frequency domain for smoothness
    N = _next_pow2(n)
    spec = np.zeros(N // 2 + 1, dtype=np.complex128)
    for k in range(1, N // 2 + 1):
        decay = np.exp(-smoothness * k / n)
        spec[k] = (rng.standard_normal() + 1j * rng.standard_normal()) * decay
    spec[0] = 0
    a = np.fft.irfft(spec, N)[:n]
    a = np.abs(a) + 0.1  # Make non-negative and add floor
    return a / a.sum()


def _make_poly(n, coeffs):
    """Polynomial profile: sum of coeffs[j] * t^j"""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for j, c in enumerate(coeffs):
        a = a + c * t**j
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _lp_solve(a, M_lp, n):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M_lp, g >= 0."""
    # Build constraint matrix (Toeplitz-like)
    # For constraint k: sum_i g[i] * a[k-i] <= M_lp for k=0..2n-2
    # a[k-i] is non-zero only when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # and 0 <= i < n
    
    # Use dense matrix for moderate n
    if n > 2048:
        return None  # Too large for dense LP
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] contributes to constraints k=j, j+1, ..., j+n-1
        # constraint k uses a[k-i], so a[j] appears when k-i=j, i=k-j
        # For fixed j: constraint k uses a[j] when i=k-j and 0<=k-j<n
        # So for k in [j, j+n-1]: A[k, k-j] += a[j]
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M_lp)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=5000, inflate_start=1.003, inflate_min=1.0005):
    """LP-based refinement with adaptive inflation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    inflate = inflate_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        g = _lp_solve(a, M_lp, n)
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.05, inflate * 1.5)
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        
        # Line search: find best blend
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.9)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > max(100, n):
                # Reset and try again with fresh inflate
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
    
    return best_a, best_score


def _gradient_polish(a0, deadline, lp_every=50, lp_time_limit=5.0):
    """Gradient descent with Nesterov momentum, backtracking, and periodic LP."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-6, 0.1 * float(c.max()))
    lr = 0.1
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        if momentum > 0:
            a_nest = np.maximum(a + momentum * v, 0.0)
            a_nest = _normalize(a_nest)
        else:
            a_nest = a
        
        grad, mval, c = _smooth_max_grad(a_nest, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = beta * 3.0
            v = np.zeros(n)
            continue
        
        # Update momentum
        v = momentum * v - lr * grad
        
        # Proposal
        a_prop = a + v
        a_prop = np.maximum(a_prop, 0.0)
        a_prop = _normalize(a_prop)
        
        # Backtracking line search
        sc_prop = _score(a_prop)
        if not np.isfinite(sc_prop) or sc_prop >= best_score:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                a_test = a + shrink * (a_prop - a)
                a_test = np.maximum(a_test, 0.0)
                a_test = _normalize(a_test)
                sc_test = _score(a_test)
                if np.isfinite(sc_test) and sc_test < best_score:
                    a_prop = a_test
                    sc_prop = sc_test
                    break
            else:
                # No improvement, reduce lr
                lr = max(0.001, lr * 0.9)
                no_imp += 1
                if no_imp > 500:
                    a = best_a.copy()
                    v = np.zeros(n)
                    lr = base_lr
                    beta = max(1e-6, 0.1 * float(_fft_convolve(a).max()))
                    no_imp = 0
                continue
        
        a = a_prop
        
        if sc_prop < best_score - 1e-14:
            best_score = sc_prop
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive beta
        if it % 200 == 199:
            beta = max(1e-8, beta * 0.7)
        
        # Adaptive lr
        if no_imp > 100:
            lr = max(0.001, lr * 0.95)
        
        # Periodic LP step
        if lp_every > 0 and it % lp_every == lp_every - 1:
            lp_end = min(deadline, time.monotonic() + lp_time_limit)
            if time.monotonic() < lp_end:
                a_lp, sc_lp = _lp_refine(a, lp_end, max_iter=20, 
                                          inflate_start=1.002, inflate_min=1.001)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()
                    v = np.zeros(n)
                    lr = base_lr
                    beta = max(1e-6, 0.1 * float(_fft_convolve(a).max()))
        
        if it % 1000 == 999:
            print(f"  GPD it={it} score={best_score:.10f} beta={beta:.2e} lr={lr:.4f}")
    
    return best_a, best_score


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using combined LP-guided gradient optimization.
Features: diverse initializations (ramp, cosine, smooth noise, polynomial), Nesterov momentum,
adaptive smooth-max parameter, periodic LP recentering, multi-length search, perturbation restarts.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval, c


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_cosine(n, num_harm=1, offset=0.5):
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, offset, dtype=np.float64)
    for h in range(num_harm):
        a = a + np.cos(2.0 * np.pi * (h + 1) * t / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_smooth_noise(n, rng, smoothness=0.3):
    N = _next_pow2(n)
    spec = np.zeros(N // 2 + 1, dtype=np.complex128)
    for k in range(1, N // 2 + 1):
        decay = np.exp(-smoothness * k / n)
        spec[k] = (rng.standard_normal() + 1j * rng.standard_normal()) * decay
    a = np.fft.irfft(spec, N)[:n]
    a = np.abs(a) + 0.05
    return a / a.sum()


def _make_poly(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for j, c in enumerate(coeffs):
        a = a + c * (t ** j)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _lp_solve(a, M_lp, n):
    if n > 2048:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M_lp)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=5000, inflate_start=1.003, inflate_min=1.0005):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_solve(a, M_lp, n)
        if g is None:
            break
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.05, inflate * 1.4)
            if no_imp > 200:
                break
            continue
        g = g / sum_g
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.92)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > max(100, n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
    return best_a, best_score


def _gradient_polish(a0, deadline, lp_every=80, lp_nudge=True):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.1 * float(c.max()))
    lr = 0.5
    base_lr = 0.5
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        if it % 2000 == 0 and it > 0:
            print(f"    GPD it={it} score={best_score:.10f} b={beta:.1e} lr={lr:.4f}")

        a_nest = np.maximum(a + momentum * v, 0.0)
        a_nest = _normalize(a_nest)
        grad, mval, _ = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = beta * 3.0
            v = np.zeros(n)
            it += 1
            continue

        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        a_new = _normalize(a_new)
        sc_new = _score(a_new)
        a = a_new

        if np.isfinite(sc_new) and sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 150:
            lr = max(0.01, lr * 0.92)
        if no_imp > 600:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
            beta = max(1e-6, 0.1 * float(_fft_convolve(a).max()))
            no_imp = 0

        if it % 150 == 149:
            beta = max(1e-8, beta * 0.75)

        if lp_nudge and lp_every > 0 and it % lp_every == lp_every - 1 and n <= 1536:
            c_cur = _fft_convolve(a)
            M_cur = float(c_cur.max())
            M_lp = M_cur * 1.002
            g = _lp_solve(a, M_lp, n)
            if g is not None:
                sum_g = float(g.sum())
                if sum_g > 1.0 + 1e-6:
                    g = g / sum_g
                    for t in [0.1, 0.3, 0.5, 0.8, 1.0]:
                        a_test = np.maximum((1.0 - t) * a + t * g, 0.0)
                        a_test = _normalize(a_test)
                        sc_test = _score(a_test)
                        if sc_test < best_score - 1e-14:
                            a = a_test
                            best_score = sc_test
                            best_a = a.copy()
                            v = np.zeros(n)
                            no_imp = 0
                            break

        it += 1

    return best_a, best_score


def _perturb_and_restart(a0, rng, strength=0.02):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a = a + noise
    a = np.maximum(a, 0.0)
    return _normalize(a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Combined LP + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses diverse initializations, Nesterov momentum, adaptive smooth-max, periodic LP nudge,
    perturbation restarts, and multi-length search.
    """
    t_start = time.monotonic()
    total_deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate candidate initializations at multiple lengths
    candidates = []
    lengths = [512, 768, 1024, 1536]

    for n_t in lengths:
        profiles = []
        profiles.append(('ramp1', _make_ramp(n_t, 1.0)))
        profiles.append(('ramp2', _make_ramp(n_t, 2.0)))
        profiles.append(('ramp3', _make_ramp(n_t, 3.0)))
        profiles.append(('cos1', _make_cosine(n_t, 1, 0.5)))
        profiles.append(('cos2', _make_cosine(n_t, 2, 0.3)))
        profiles.append(('cos_low', _make_cosine(n_t, 1, 1.0)))
        profiles.append(('poly', _make_poly(n_t, [1.0, 3.0, 1.0])))
        profiles.append(('poly2', _make_poly(n_t, [1.0, 0.0, 4.0])))
        profiles.append(('noise_s', _make_smooth_noise(n_t, rng, 0.15)))
        profiles.append(('noise_m', _make_smooth_noise(n_t, rng, 0.4)))
        profiles.append(('noise_h', _make_smooth_noise(n_t, rng, 0.8)))

        if parent is not None:
            profiles.append(('parent', _interpolate_seq(parent, n_t)))

        for name, a0 in profiles:
            sc = _score(a0)
            if np.isfinite(sc):
                candidates.append((sc, name, n_t, a0))

    candidates.sort(key=lambda x: x[0])
    print(f"Generated {len(candidates)} candidates. Best init: "
          f"score={candidates[0][0]:.8f} ({candidates[0][1]}, n={candidates[0][2]})")
    print(f"Top 5: " + ", ".join(
        [f"{c[1]}/n{c[2]}={c[0]:.6f}" for c in candidates[:5]]))

    # Select diverse set to optimize (max 5-6)
    selected = []
    length_count = {}
    for sc, name, n_t, a0 in candidates:
        if len(selected) >= 6:
            break
        lc = length_count.get(n_t, 0)
        if lc >= 2:
            continue
        selected.append((sc, name, n_t, a0))
        length_count[n_t] = lc + 1

    # Ensure parent is included if available
    if parent is not None:
        has_parent = any(x[1] in ('parent', 'parent_orig') for x in selected)
        if not has_parent:
            p_sc = _score(parent)
            selected.insert(0, (p_sc, 'parent_orig', parent.size, parent.copy()))

    print(f"Selected {len(selected)} candidates for optimization:")
    for sc, name, n_t, _ in selected:
        print(f"  {name} n={n_t} score={sc:.8f}")

    # Allocate time: give more to better candidates
    remaining = total_deadline - time.monotonic() - 250.0
    weights = []
    for i, (sc, name, n_t, a0) in enumerate(selected):
        if i == 0:
            w = 0.30
        elif i == 1:
            w = 0.22
        elif i == 2:
            w = 0.18
        elif i == 3:
            w = 0.14
        else:
            w = 0.09
        weights.append(w)
    total_w = sum(weights)
    time_budgets = [remaining * (w / total_w) for w in weights]

    # Optimize each selected candidate
    for i, (sc0, name, n_t, a0) in enumerate(selected):
        if time.monotonic() > total_deadline - 250:
            break
        if sc0 >= best_score + 0.3 and name != 'parent_orig':
            print(f"Skipping {name} (init score {sc0:.6f} too far from best {best_score:.6f})")
            continue

        tb = time_budgets[i]
        if tb < 15:
            continue

        print(f"\n=== Optimizing {name} (n={n_t}, init={sc0:.8f}, budget={tb:.0f}s) ===")

        a_work = a0.copy()

        # Phase A: Gradient polish (first 45% of budget)
        gp_budget = tb * 0.45
        gp_end = time.monotonic() + gp_budget
        a_gp, sc_gp = _gradient_polish(a_work, gp_end, lp_every=100, lp_nudge=True)
        if sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()
            print(f"  After GP: {sc_gp:.10f}")
        a_work = a_gp

        # Phase B: LP refinement (next 30%)
        lp_budget = tb * 0.30
        lp_end = time.monotonic() + lp_budget
        if n_t <= 1536 and time.monotonic() < lp_end:
            a_lp, sc_lp = _lp_refine(a_work, lp_end, max_iter=10000,
                                     inflate_start=1.003, inflate_min=1.0005)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  After LP: {sc_lp:.10f}")
            a_work = a_lp
        else:
            print(f"  Skipping LP (n={n_t} too large or no time)")

        # Phase C: Gradient + perturb restart (remaining)
        rest_end = time.monotonic() + tb * 0.25
        if time.monotonic() < rest_end:
            a_f, sc_f = _gradient_polish(a_work, rest_end, lp_every=200, lp_nudge=True)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  After final GP: {sc_f:.10f}")
            a_work = a_f

            # Perturbation restart if time remains
            if time.monotonic() < rest_end - 30:
                a_pert = _perturb_and_restart(a_work, rng, strength=0.01)
                pert_end = rest_end
                a_p, sc_p = _gradient_polish(a_pert, pert_end, lp_every=300, lp_nudge=False)
                if sc_p < best_score:
                    best_score = sc_p
                    best_seq = a_p.copy()
                    print(f"  After perturb restart: {sc_p:.10f}")

    # Final polish on the global best
    if best_seq is not None:
        final_budget = max(60.0, total_deadline - time.monotonic() - 40.0)
        if final_budget > 30:
            print(f"\n=== FINAL POLISH (budget={final_budget:.0f}s, best={best_score:.10f}) ===")

            # Gradient polish
            half_end = time.monotonic() + final_budget * 0.5
            a_f, sc_f = _gradient_polish(best_seq, half_end, lp_every=50, lp_nudge=True)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  GP: {sc_f:.10f}")

            # LP if n allows
            if best_seq.size <= 1536 and time.monotonic() < total_deadline - 40:
                lp_end = total_deadline - 40
                a_l, sc_l = _lp_refine(best_seq, lp_end, max_iter=5000,
                                       inflate_start=1.002, inflate_min=1.0003)
                if sc_l < best_score:
                    best_score = sc_l
                    best_seq = a_l.copy()
                    print(f"  LP: {sc_l:.10f}")

            # One more gradient pass
            if time.monotonic() < total_deadline - 40:
                a_g, sc_g = _gradient_polish(best_seq, total_deadline - 40,
                                             lp_every=100, lp_nudge=False)
                if sc_g < best_score:
                    best_score = sc_g
                    best_seq = a_g.copy()
                    print(f"  Final GP: {sc_g:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)
        print(f"Using fallback ramp: score={best_score:.8f}")

    # Final validation and evaluation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)
        s = float(final_seq.sum())
        final_seq = final_seq / s

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final_seq]
