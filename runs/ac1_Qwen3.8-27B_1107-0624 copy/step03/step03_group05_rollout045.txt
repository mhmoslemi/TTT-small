"""
Optimized step function coefficient minimizer using Nesterov-accelerated
smooth-max gradient, direct peak reduction, and fast active-set LP.
Focuses on aggressive peak targeting with adaptive step sizes.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return w / s * target_sum

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

def _grad_conv_max(a, weights):
    """Gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(weights, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h

def _nesterov_smoothmax(a, deadline, max_iter=50000, beta_init=0.02, lr_init=0.05):
    """Nesterov-accelerated gradient descent on smooth max of convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    beta = max(1e-10, beta_init * float(_fft_convolve(a).max()))
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step
        if it == 0:
            y = a
        else:
            momentum_cur = (1 + momentum) / 2  # Nesterov schedule
            y = a + momentum_cur * v
        
        # Compute gradient at y (clipped to non-negative)
        y_pos = np.maximum(y, 0)
        s_y = y_pos.sum()
        if s_y < 1e-10:
            y_pos = a
            s_y = 1.0
        y_norm = y_pos / s_y
        
        c = _fft_convolve(y_norm)
        w = _smooth_max_weights(c, beta)
        grad = _grad_conv_max(y_norm, w)
        
        # Add small regularization to keep away from boundary
        grad += 1e-8 * y_norm
        
        # Update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v, 1.0)
        a = a_new
        
        # Evaluate
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.97)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, beta_init * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
        
        # Anneal beta
        if it % 50 == 0:
            beta = max(1e-12, beta * 0.9)
    
    return best_a, best_score

    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - target) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target / n)
    return w * (target / s)


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_weighted_conv(a, weights):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(weights, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _nesterov_gradient(a, deadline, beta_init=0.03, lr_init=0.04, max_iter=80000):
    """Nesterov-accelerated smooth-max gradient with backtracking line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a, 1.0)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    beta = max(1e-10, beta_init * float(_fft_convolve(a).max()))
    lr = lr_init
    base_lr = lr
    momentum = 0.92
    v = np.zeros(n)
    y = a.copy()
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        if it > 0:
            y = a + momentum * v

        y_pos = np.maximum(y, 0.0)
        s_y = y_pos.sum()
        if s_y < 1e-10:
            y_pos = a
            s_y = 1.0
        y_norm = y_pos / s_y

        c = _fft_convolve(y_norm)
        w = _smooth_max_weights(c, beta)
        grad = _grad_weighted_conv(y_norm, w)
        grad += 1e-9 * y_norm

        # Backtracking line search
        a_try = _project_simplex(a - lr * grad, 1.0)
        score_try = float(2.0 * n * _fft_convolve(a_try).max())
        max_c = float(c.max())

        shrinks = 0
        while score_try > max_c * (1.0 - 1e-12 * lr) and shrinks < 15:
            lr = lr * 0.5
            a_try = _project_simplex(a - lr * grad, 1.0)
            score_try = float(2.0 * n * _fft_convolve(a_try).max())
            shrinks += 1

        # Accept if improvement
        if score_try < best_score + 1e-10:
            v = momentum * v - lr * grad
            a = a_try

        if score_try < best_score - 1e-12:
            best_score = score_try
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                lr = max(0.002, lr * 0.98)
            if no_imp > max(1500, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-10, beta_init * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 80 == 0:
            beta = max(1e-13, beta * 0.88)

    return best_a, best_score


def _peak_surgery(a, deadline, max_iter=2000, top_k=5, transfer_frac=0.05):
    """Directly reduce peaks by transferring mass from contributors to safe positions."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        peak_count = min(top_k, len(c))
        peak_indices = np.argsort(c)[::-1][:peak_count]
        peak_vals = c[peak_indices]
        current_max = float(c.max())

        # Compute contribution weight for each position
        contrib = np.zeros(n)
        for pk in peak_indices:
            weight = peak_vals[int(np.where(peak_indices == pk)[0][0])] if peak_indices.ndim > 0 else current_max
            for i in range(n):
                j = int(pk) - i
                if 0 <= j < n:
                    contrib[i] += weight * a[j]
                j2 = i - int(pk)
                if 0 <= j2 < n and j2 != j:
                    contrib[i] += 0.5 * weight * a[j2]

        # Weighted by peak value
        for idx, pk in enumerate(peak_indices):
            w = peak_vals[idx] / peak_vals.sum()
            for i in range(n):
                j = int(pk) - i
                if 0 <= j < n:
                    contrib[i] += w * a[j]

        # Find safe positions (low contribution)
        contrib_norm = contrib / (contrib.sum() + 1e-30)

        # Move mass from high-contrib positions to low-contrib positions
        n_high = max(1, n // 16)
        n_low = max(1, n // 16)
        high_idx = np.argsort(contrib)[-n_high:]
        low_idx = np.argsort(contrib)[:n_low]

        # Check that low positions don't overlap with peak contribution
        total_mass = transfer_frac
        for hi in high_idx:
            share = total_mass / len(high_idx)
            # Find best low position
            best_lo = -1
            best_gain = 0
            for lo in low_idx:
                if lo == hi:
                    continue
                # Gain: reduction in contrib at hi, cost at lo
                if contrib[hi] - contrib[lo] > best_gain:
                    best_gain = contrib[hi] - contrib[lo]
                    best_lo = lo
            if best_lo < 0:
                continue

            a_try = a.copy()
            delta = share * a[hi]
            if delta > a[hi]:
                delta = a[hi] * 0.9
            a_try[hi] -= delta
            a_try[best_lo] += delta
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = float(2.0 * n * _fft_convolve(a_try).max())

            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                no_imp = 0
            else:
                no_imp += 1

        if no_imp > 300:
            break

    return best_a, best_score


def _adam_logit_polish(a, deadline, lr=0.03, max_iter=50000):
    """Adam optimizer in logit parameterization for fine-tuning."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    beta = max(1e-10, 0.03 * float(_fft_convolve(a).max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weights(c, beta)
        grad_p = _grad_weighted_conv(a, w)
        dot = float(np.dot(a, grad_p))
        grad_z = a * (grad_p - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = np.log(np.maximum(best_a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v_arr = b2 * v_arr + (1 - b2) * grad_z ** 2
        mh = m / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))

        e = np.exp(z)
        a = e / e.sum()

        sc = float(2.0 * n * _fft_convolve(a).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-13, beta * 0.85)

        if no_imp > max(2000, 4 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.8 * np.log(np.maximum(best_a, 1e-12)) + 0.2 * rng.normal(size=n) * 0.3
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-10, 0.03 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            lr = 0.03
            no_imp = 0

    return best_a, best_score


def _fast_lp_refine(a, deadline, top_k_constraints=20, max_iter=5000, inflate=1.001, blend=0.25):
    """LP refinement using only top-k peak constraints for speed."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate_cur

        # Use only top-k constraints for speed
        k = min(top_k_constraints, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]

        # Build linear constraint: A @ g <= M for top constraints only
        # conv_k(g) = sum_i a[i]*g[k-i] <= M
        # This is linear in g
        rows = []
        for pk in top_idx:
            row = np.zeros(n)
            for i in range(n):
                j = int(pk) - i
                if 0 <= j < n:
                    row[i] += a[j]
            rows.append(row)
        A = np.array(rows)
        b = np.full(k, M)

        res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': 500, 'presolve': True})

        if not res.success or res.x is None:
            no_imp += 1
            if no_imp > 200:
                break
            inflate_cur = min(1.02, inflate_cur * 1.05)
            continue

        g = np.maximum(res.x, 0.0)
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 300:
                break
            continue
        g = g / sum_g

        a_new = (1 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = float(2.0 * n * _fft_convolve(a_new).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.95)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.01, inflate_cur * 0.97)

    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, c1=0.0, c2=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c1 * (t - 0.5) + c2 * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _full_optimize(a_init, deadline):
    """Run the full optimization pipeline on a single sequence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a, 1.0)

    best_a = a.copy()
    best_score = _score(a)
    print(f"      Init: {best_score:.10f}")

    # Phase A: Nesterov gradient (main)
    budget_a = (deadline - time.monotonic()) * 0.4
    if budget_a > 5:
        end_a = time.monotonic() + budget_a
        a_res, sc = _nesterov_gradient(a, end_a, beta_init=0.03, lr_init=0.04)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"      Nesterov: {sc:.10f}")

    # Phase B: Peak surgery
    budget_b = (deadline - time.monotonic()) * 0.15
    if budget_b > 3:
        end_b = time.monotonic() + budget_b
        a_res, sc = _peak_surgery(best_a, end_b, top_k=5, transfer_frac=0.04)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"      PeakSurgery: {sc:.10f}")

    # Phase C: Adam polish
    budget_c = (deadline - time.monotonic()) * 0.25
    if budget_c > 5:
        end_c = time.monotonic() + budget_c
        a_res, sc = _adam_logit_polish(best_a, end_c, lr=0.03)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"      Adam: {sc:.10f}")

    # Phase D: Fast LP
    budget_d = (deadline - time.monotonic()) * 0.2
    if budget_d > 5:
        end_d = time.monotonic() + budget_d
        a_res, sc = _fast_lp_refine(best_a, end_d, top_k_constraints=15,
                                    max_iter=2000, inflate=1.001, blend=0.2)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"      FastLP: {sc:.10f}")

    # Phase E: Final Nesterov
    if time.monotonic() < deadline - 10:
        a_res, sc = _nesterov_gradient(best_a, deadline, beta_init=0.01, lr_init=0.02,
                                       max_iter=20000)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        print(f"      Final Nesterov: {sc:.10f}")

    return best_a, best_score

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
