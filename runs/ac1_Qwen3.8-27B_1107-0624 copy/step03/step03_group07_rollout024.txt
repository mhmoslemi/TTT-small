"""
Multi-scale L-BFGS-B optimization with annealed soft-max for minimizing
2n * max(conv(a,a)) / sum(a)^2. Uses scipy.optimize.minimize with FFT-based
gradients, peak-equalization refinements, and simulated annealing restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def softmax_vec(z):
    """Compute softmax of z vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def soft_max_val(c, beta):
    """Compute soft-max value and weights."""
    m = c.max()
    if beta < 1e-14:
        return m, np.zeros_like(c)
    x = (c - m) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = ex.sum()
    val = m + beta * np.log(s)
    w = ex / s
    return val, w

def optimize_lbfgs(z0, beta, maxiter=500, gtol=1e-8):
    """Run L-BFGS-B on the soft-max objective."""
    result = minimize(
        fun=lambda z: obj_and_grad_neg_log_softmax(z, beta, len(z))[0],
        x0=z0,
        jac=lambda z: obj_and_grad_neg_log_softmax(z, beta, len(z))[1],
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 50}
    )
    return result

def anneal_lbfgs(z0, deadline, rng):
    """Anneal beta and run L-BFGS-B at each level."""
    z = z0.copy()
    n = z.size
    a = softmax_vec(z)
    c = full_convolve_fft(a)
    max_c = c.max()
    
    # Start with moderate beta
    beta = max(0.05 * max_c, 1e-6)
    best_z = z.copy()
    best_val = max_c
    
    while time.monotonic() < deadline:
        # Run L-BFGS-B at current beta
        result = optimize_lbfgs(z, beta, maxiter=200, gtol=1e-7)
        if result.success or result.fun < best_val:
            z = result.x
            a = softmax_vec(z)
            c = full_convolve_fft(a)
            max_c = c.max()
            if max_c < best_val:
                best_val = max_c
                best_z = z.copy()
        
        # Decrease beta
        beta *= 0.5
        if beta < 1e-10:
            break
    
    return best_z, best_val

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def make_smooth_profile(n, rng, method='random'):
    t = np.linspace(0.0, 1.0, n)
    if method == 'linear':
        eta = rng.uniform(0.5, 3.0)
        a = 1.0 + eta * (0.5 - t)
    elif method == 'cosine':
        k = rng.integers(1, 4)
        amp = rng.uniform(0.2, 0.6)
        a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    elif method == 'bump':
        width = rng.uniform(0.2, 0.6)
        offset = rng.uniform(0.2, 0.8)
        a = np.exp(-((t - offset) / width) ** 2)
    elif method == 'random_smooth':
        coeffs = rng.uniform(-0.5, 0.5, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def peak_equalize(a, n_peaks=5, n_iter=100, lr=0.01):
    """FNorm-style peak equalization."""
    a = a.copy()
    n = a.size
    for _ in range(n_iter):
        c = full_convolve_fft(a)
        max_c = c.max()
        # Find top peaks
        peak_idx = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        w[peak_idx] = 1.0 / n_peaks
        # Compute gradient
        ar = a[::-1]
        N = next_pow2(3 * n - 2)
        far = rfft(ar, N)
        fw = rfft(w, N)
        h = irfft(far * fw, N)[n-1:2*n-1]
        grad_a = 2.0 * h
        # Update in logit space
        z = np.log(a + 1e-30)
        dot_val = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot_val)
        z = z - lr * dz / (np.sqrt(np.mean(dz**2)) + 1e-10)
        a = softmax_vec(z)
    return a

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def softmax_vec(z):
    """Compute softmax of z vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


def soft_max_val(c, beta):
    """Compute soft-max value and weights on convolution values."""
    m = c.max()
    if beta < 1e-14:
        w = np.zeros_like(c)
        idx = np.argmax(c)
        w[idx] = 1.0
        return float(m), w

    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return float(m), w
    val = float(m + beta * np.log(s))
    w = ex / s
    return val, w


def compute_obj_grad(z, beta):

    n = len(z)

    c = full_convolve_fft(a)

    fw = rfft(w, N)

    h = conv_hw[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    dot_val = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot_val)
    return val, grad_z

    def __init__(self, beta):

    val = float(m + beta * np.log(s))
    w = ex / s
    return val, w


def compute_obj_grad(z, beta):
    """Compute objective value and gradient for soft_max(conv(softmax(z)), beta)."""
    n = len(z)
    a = softmax_vec(z)
    c = full_convolve_fft(a)
    val, w = soft_max_val(c, beta)
    ar = a[::-1]
    N = next_pow2(3 * n - 2)
    far = rfft(ar, N)
    fw = rfft(w, N)
    conv_hw = irfft(far * fw, N)
    h = conv_hw[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    dot_val = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot_val)
    return val, grad_z


class SoftMaxObjective:
    """Callable for L-BFGS-B."""
    def __init__(self, beta):
        self.beta = float(beta)

    def __call__(self, z):
        return compute_obj_grad(z, self.beta)


def run_lbfgs(z0, beta, maxiter=300, gtol=1e-8, maxls=50):
    """Run L-BFGS-B at given beta level."""
    obj = SoftMaxObjective(beta)
    try:
        result = minimize(
            fun=obj,
            x0=z0,
            jac=obj,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': gtol, 'maxls': maxls}
        )
        return result.x, float(result.fun)
    except Exception:
        return z0, np.inf


def annealed_lbfgs(z0, deadline, beta_start=None, beta_ratio=0.5, maxiter_per_stage=150):
    """Anneal beta and run L-BFGS-B at each level."""
    z = z0.copy()
    n = z.size
    a = softmax_vec(z)
    c = full_convolve_fft(a)
    max_c = float(c.max())

    if beta_start is None:
        beta = max(0.05 * max_c, 1e-6)
    else:
        beta = float(beta_start)

    best_z = z.copy()
    best_val = max_c
    cur_maxiter = maxiter_per_stage

    while time.monotonic() < deadline:
        z_new, fun_val = run_lbfgs(z, beta, maxiter=cur_maxiter, gtol=1e-7)
        a_test = softmax_vec(z_new)
        c_test = full_convolve_fft(a_test)
        max_test = float(c_test.max())
        if max_test < best_val:
            best_val = max_test
            best_z = z_new.copy()
        z = z_new
        beta *= beta_ratio
        if beta < 1e-10:
            break
        cur_maxiter = max(30, int(cur_maxiter * 0.85))

    return best_z, best_val


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_profile(n, rng, method):
    """Generate a starting profile."""
    t = np.linspace(0.0, 1.0, n)
    if method == 'linear':
        eta = rng.uniform(0.8, 2.5)
        a = 1.0 + eta * (0.5 - t)
    elif method == 'linear_asym':
        eta = rng.uniform(1.0, 4.0)
        a = 1.0 + eta * t
    elif method == 'cosine':
        k = int(rng.integers(1, 4))
        amp = rng.uniform(0.2, 0.5)
        a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    elif method == 'bump':
        width = rng.uniform(0.2, 0.5)
        offset = rng.uniform(0.2, 0.7)
        a = np.exp(-((t - offset) / width) ** 2)
    elif method == 'random_smooth':
        coeffs = rng.uniform(-0.4, 0.4, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2.0 * np.pi * i * t)
    elif method == 'ramp':
        p = rng.uniform(0.5, 3.0)
        a = (1.0 - t) ** p + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
