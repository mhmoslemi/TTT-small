Given a sequence a (normalized, sum=1), the score is 2n * max_k c[k] where c = conv(a,a).

If we change a[i] by delta (and then renormalize), the new convolution is:
c'[k] = c[k] + 2*delta*a[k-i] + delta^2 * 1_{k=2i}

(where a[k-i] = 0 if k-i is out of [0, n-1])

After renormalization (dividing by sum + delta), the score changes in a more complex way. But if delta is small, we can approximate.

For coordinate descent, a simpler approach:
1. Compute c = conv(a, a)
2. For each i, find the best delta_i (change to a[i]) that minimizes the score
3. Update a[i] and renormalize
4. Repeat

To find the best delta_i for a given i:
- The new convolution is c + 2*delta_i * (a shifted by i) + delta_i^2 * e_{2i}
- The new sum is S + delta_i (where S = sum(a) = 1)
- The new score is 2n * max_k (c[k] + 2*delta_i*a[k-i] + delta_i^2 * 1_{k=2i}) / (1 + delta_i)^2

This is a 1D optimization problem in delta_i. The maximum is a piecewise function (each c[k] + 2*delta_i*a[k-i] is linear in delta_i, and the +delta_i^2 term is at one point). The max of linear functions is convex, and dividing by (1+delta_i)^2 makes it more complex.

We can solve this by:
1. Evaluating the score at several candidate delta_i values (0, -a[i], and values that make the top few convolution entries equal)
2. Taking the best

Or more practically:
1. Compute the current c and identify the top few k values
2. For each candidate delta, compute the new max and score
3. Use ternary search or grid search over delta in [-a[i], some_positive_value]

For efficiency, we can limit the search to delta in [-a[i], a[i]] (i.e., we don't make any element more than double its current value in a single step).

```python
    """One full pass of coordinate descent."""
    for i in range(n):
        # Current value
        ai = a[i]
        # The shift vector: a[k-i] for k = 0..2n-2
        # c_new[k] = c[k] + 2*delta*a[k-i] + delta^2 * (1 if k==2i else 0)
        # Let's try a few candidate deltas
        
        best_delta = 0.0
        best_score = 2.0 * n * c.max()  # current score (sum=1)
        
        # Candidate deltas: 0, -ai/2, -ai, -ai/4, ai/4, ai/8
        candidates = [0.0, -0.5*ai, -0.25*ai, -0.1*ai, 0.1*ai, 0.25*ai]
        # Also try to equalize the top convolution entries
        if c.max() > 0:
            top_k = np.argmax(c)
            # c_new[top_k] = c[top_k] + 2*delta*a[top_k-i]
            # Set this to 0: delta = -c[top_k] / (2*a[top_k-i]) if a[top_k-i] > 0
            if 0 <= top_k - i < n and a[top_k - i] > 1e-15:
                d = -c[top_k] / (2.0 * a[top_k - i])
                candidates.append(max(-ai, min(ai, d)))
        
        for d in candidates:
            d = max(-ai, min(ai * 2, d))  # bound delta
            if d < -ai:
                d = -ai  # can't go below 0
            c_new = c + 2.0 * d * _shifted_a(a, i, n)
            if 2*i < 2*n - 1:
                c_new[2*i] += d * d
            c_new = np.maximum(c_new, 0)
            new_sum = 1.0 + d  # since sum(a) = 1
            if new_sum < 0.01:
                continue
            sc = 2.0 * n * c_new.max() / (new_sum * new_sum)
            if sc < best_score:
                best_score = sc
                best_delta = d
        
        if best_delta != 0:
            a[i] += best_delta
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = _fft_convolve(a)  # recompute (expensive but accurate)
    
    return a, c
```

```python
"""
Optimized step function coefficient search using LP-based block coordinate descent,
coordinate descent local search, smooth-max gradient polish, and adaptive restarts.
Improves upon previous LP approach with: (1) incremental coordinate descent,
(2) minmax LP as escape mechanism, (3) larger sequence lengths, (4) better line search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = conv(f, g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        l_lo = max(0, k - n + 1)
        l_hi = min(n - 1, k)
        for l in range(l_lo, l_hi + 1):
            rows.append(k)
            cols.append(l)
            vals.append(f[k - l])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_maxsum_lp(f, M, n):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _solve_minmax_lp(f, n):
    """Solve: min t s.t. conv(f,g)[k] <= t, sum(g)=1, g>=0. Returns (g, t)."""
    # Variables: g[0..n-1], t  => n+1 variables
    # Objective: min t => c = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: conv(f,g)[k] - t <= 0 for k=0..2n-2
    A_ineq = np.zeros((2 * n - 1, n + 1))
    Af = _build_conv_matrix(f, n)
    A_ineq[:, :n] = Af.toarray()
    A_ineq[:, -1] = -1.0
    b_ineq = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        g = _normalize(g)
        t = float(res.x[-1])
        return g, t
    return None, None


def _line_search_golden(a, g, n, max_iter=20):
    """Golden section search for best blend parameter t in [0,1]."""
    best_a = a.copy()
    best_sc = _score(a)

    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    d = hi - lo
    x1 = hi - phi * d
    x2 = lo + phi * d
    f1 = _score(_normalize(np.maximum((1 - x1) * a + x1 * g, 0)))
    f2 = _score(_normalize(np.maximum((1 - x2) * a + x2 * g, 0)))

    for _ in range(max_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            d = hi - lo
            x1 = hi - phi * d
            f1 = _score(_normalize(np.maximum((1 - x1) * a + x1 * g, 0)))
        else:
            lo = x1
            x1 = x2
            f1 = f2
            d = hi - lo
            x2 = lo + phi * d
            f2 = _score(_normalize(np.maximum((1 - x2) * a + x2 * g, 0)))

    # Check endpoints and midpoint
    for t in [lo, hi, (lo + hi) / 2, 0.0, 1.0]:
        a_cand = _normalize(np.maximum((1 - t) * a + t * g, 0))
        sc = _score(a_cand)
        if sc < best_sc:
            best_sc = sc
            best_a = a_cand
    return best_a, best_sc


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return _normalize(a)


def _make_symmetric_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - np.abs(2.0 * t - 1.0)) ** alpha + 1e-6
    return _normalize(a)


def _make_trapezoid(n, frac=0.6):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    a[t < 0.5 - frac / 2] = 0.0
    mask_rise = (t >= 0.5 - frac / 2) & (t < 0.5)
    mask_fall = (t >= 0.5) & (t <= 0.5 + frac / 2)
    mask_top = (t >= 0.5) & (t <= 0.5 + frac / 2)
    a[t < 0.5 - frac / 2] = 0.0
    a = np.clip((t - (0.5 - frac / 2)) / (frac / 2), 0, 1)
    a = np.minimum(a, np.clip(((0.5 + frac / 2) - t) / (frac / 2), 0, 1))
    a = np.maximum(a, 1e-8)
    return _normalize(a)


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _lp_refine(a0, deadline, max_iter=50000):
    """LP-based block coordinate descent with adaptive inflation."""
    a = _normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    inflate = 1.008
    no_imp = 0
    stuck_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        if M < 1e-15:
            break
        M_lp = M * inflate

        g = _solve_maxsum_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            stuck_count += 1
            if stuck_count % 20 == 0:
                inflate = min(inflate * 1.15, 0.05)
            if stuck_count > 200:
                # Try minmax LP as escape
                g_mm, t_mm = _solve_minmax_lp(a, n)
                if g_mm is not None:
                    a_new, sc_new = _line_search_golden(a, g_mm, n, max_iter=12)
                    if sc_new < best_score - 1e-12:
                        best_score = sc_new
                        best_a = a_new.copy()
                        a = a_new
                        no_imp = 0
                        stuck_count = 0
                        inflate = 1.008
                    else:
                        break
                else:
                    break
            continue
        else:
            stuck_count = 0

        g = _normalize(g)
        a_new, sc_new = _line_search_golden(a, g, n, max_iter=15)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            inflate = max(1.002, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(inflate * 1.1, 0.05)
            if no_imp > max(150, n):
                a = best_a.copy()
                inflate = 1.01
                no_imp = 0
                if inflate > 1.005:
                    break

    return best_a


def _coord_descent(a0, deadline, max_passes=500):
    """Coordinate descent local search with incremental convolution updates."""
    a = _normalize(a0.copy())
    n = len(a)
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    # Precompute shift vectors
    shifts = []
    for i in range(n):
        v = np.zeros(2 * n - 1)
        start = i
        length = min(n, 2 * n - 1 - i)
        if length > 0:
            v[start:start + length] = a[:length]
        shifts.append(v)

    no_imp_pass = 0

    for pass_idx in range(max_passes):
        if time.monotonic() > deadline:
            break

        pass_improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-15:
                continue

            shift_i = shifts[i]
            two_i = 2 * i

            # Try candidate deltas
            candidates = [0.0, -0.5 * ai, -0.25 * ai, -0.1 * ai,
                          0.1 * ai, 0.05 * ai, -0.75 * ai, 0.2 * ai]

            # Try to equalize peak
            k_peak = int(np.argmax(c))
            if shift_i[k_peak] > 1e-15:
                d_eq = -(c[k_peak] - c.mean()) / (2.0 * shift_i[k_peak])
                candidates.append(np.clip(d_eq, -ai, ai * 2))

            best_delta = 0.0
            best_local_sc = float(2.0 * n * c.max())  # current score (sum=1)

            for d in candidates:
                d = max(-ai, min(ai * 3.0, d))
                c_new = c + 2.0 * d * shift_i
                if two_i < 2 * n - 1:
                    c_new[two_i] += d * d
                c_new = np.maximum(c_new, 0.0)
                new_sum = 1.0 + d
                if new_sum < 0.01:
                    continue
                sc = 2.0 * n * c_new.max() / (new_sum * new_sum)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_delta = d

            if best_delta != 0.0:
                a[i] += best_delta
                a = np.maximum(a, 0.0)
                new_sum = a.sum()
                a = a / new_sum
                # Update c incrementally (approximate - will recompute after pass)
                # Actually just update c using the formula with the actual delta
                # But since we renormalized, c is now slightly off.
                # For correctness, we'll recompute c at end of pass.
                pass_improved = True

        # Recompute c after full pass
        c = _fft_convolve(a)
        current_score = float(2.0 * n * c.max())

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_imp_pass = 0
        else:
            no_imp_pass += 1
            if no_imp_pass > 30:
                a = best_a.copy()
                c = _fft_convolve(a)
                no_imp_pass = 0
                if pass_idx > 100:
                    break

        # Update shifts (since a changed)
        if pass_idx % 10 == 9:
            for i in range(n):
                v = np.zeros(2 * n - 1)
                start = i
                length = min(n, 2 * n - 1 - i)
                if length > 0:
                    v[start:start + length] = a[:length]
                shifts[i] = v

    return best_a


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Smooth-max gradient descent with momentum and adaptive step size."""
    a = _normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.03 * float(c.max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nested = a + momentum * v
        grad, maxc = _smooth_max_grad(a_nested, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue

        # Line search along gradient direction
        best_t = lr
        best_sc = _score(_project_simplex(a - lr * grad))
        for t in [lr * 0.5, lr * 0.25, lr * 2.0]:
            sc = _score(_project_simplex(a - t * grad))
            if sc < best_sc:
                best_sc = sc
                best_t = t

        v = momentum * v - best_t * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.93)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 200 == 199:
            beta = max(1e-10, beta * 0.7)

    return best_a


def _perturb_and_optimize(a0, n_perturb, deadline, rng):
    """Perturb a fraction of coordinates and re-optimize with LP."""
    a = a0.copy()
    n = len(a)
    indices = rng.choice(n, size=n_perturb, replace=False)
    a[indices] *= np.exp(rng.normal(0, 0.3, size=n_perturb))
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based block coordinate descent,
    coordinate descent, gradient polish, with adaptive restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initial candidates at multiple lengths
    lengths = [512, 768, 1024]
    candidates = []

    for n_t in lengths:
        if parent is not None:
            candidates.append(('parent_interp', n_t, _interpolate(parent, n_t)))
        candidates.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        candidates.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        candidates.append(('sym_ramp', n_t, _make_symmetric_ramp(n_t, 1.5)))
        candidates.append(('trapezoid', n_t, _make_trapezoid(n_t, 0.7)))

    # Score and sort candidates
    scored = []
    for name, n_t, a0 in candidates:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Top candidates: {[(f'{sc:.6f}', nm) for sc, nm, _, _ in scored[:5]]}")

    # Phase 1: LP refinement on top candidates
    for idx, (sc0, name, n_t, a0) in enumerate(scored):
        if time.monotonic() > deadline - 300:
            break
        if sc0 >= best_score + 0.2:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
            print(f"New best from init: {name} n={n_t} score={sc0:.10f}")

        # Allocate time
        remaining = deadline - time.monotonic() - 250
        if remaining < 20:
            break

        if sc0 < best_score + 0.05:
            lp_time = min(400.0, remaining * 0.4)
        elif idx < 3:
            lp_time = min(200.0, remaining * 0.25)
        else:
            lp_time = min(80.0, remaining * 0.1)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"Phase1 LP: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=50000)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> score={sc_lp:.10f}")
        else:
            print(f"  -> score={sc_lp:.8f}")

    # Phase 2: Coordinate descent on best
    if best_seq is not None and time.monotonic() < deadline - 150:
        cd_time = min(200.0, deadline - time.monotonic() - 120)
        if cd_time > 10:
            cd_end = time.monotonic() + cd_time
            print(f"Phase2 CoordDescent: budget={cd_time:.0f}s, n={len(best_seq)}")
            a_cd = _coord_descent(best_seq, cd_end, max_passes=2000)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  -> score={sc_cd:.10f}")

    # Phase 3: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 100:
        gp_time = min(150.0, deadline - time.monotonic() - 80)
        if gp_time > 10:
            gp_end = time.monotonic() + gp_time
            print(f"Phase3 GradPolish: budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=50000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> score={sc_gp:.10f}")

    # Phase 4: Another LP pass with tighter params
    if best_seq is not None and time.monotonic() < deadline - 80:
        lp_time = min(120.0, deadline - time.monotonic() - 60)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            print(f"Phase4 FinalLP: budget={lp_time:.0f}s")
            a_final = _lp_refine(best_seq, lp_end, max_iter=30000)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  -> score={sc_final:.10f}")

    # Phase 5: Final coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 30:
        cd_time = min(60.0, deadline - time.monotonic() - 20)
        if cd_time > 5:
            cd_end = time.monotonic() + cd_time
            print(f"Phase5 FinalCD: budget={cd_time:.0f}s")
            a_fc = _coord_descent(best_seq, cd_end, max_passes=1000)
            sc_fc = _score(a_fc)
            if np.isfinite(sc_fc) and sc_fc < best_score:
                best_score = sc_fc
                best_seq = a_fc.copy()
            print(f"  -> score={sc_fc:.10f}")

    # Phase 6: Try perturbation + quick re-optimization
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_pert = max(10, len(best_seq) // 20)
        a_pert = _perturb_and_optimize(best_seq, n_pert, deadline - 25, rng)
        a_pert = _coord_descent(a_pert, time.monotonic() + 10, max_passes=100)
        sc_pert = _score(a_pert)
        if np.isfinite(sc_pert) and sc_pert < best_score:
            best_score = sc_pert
            best_seq = a_pert.copy()
            print(f"Phase6 Perturb: score={sc_pert:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * s  # keep sum as is (>= 0.01)
    else:
        final_seq = _make_ramp(512, 1.0)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
```
