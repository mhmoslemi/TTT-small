"""
Hybrid optimization: Adam + L-BFGS + LP alternating for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses logit-space parameterization with smooth-max approximation.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        return mval
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        return mval
    return mval + beta * float(np.log(s))

def _smooth_max_grad(c, beta):
    """Returns (value, weights) where weights sum to 1."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _lp_solve(a, M):
    """Solve LP: maximize sum(b) subject to a*b <= M (element-wise), b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _lp_alternating(a, deadline, t=0.3, max_solves=30):
    """Alternating LP: solve for g given f, then for f' given g."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    
    f = a.copy()
    for solve_num in range(max_solves):
        if time.monotonic() > deadline:
            break
        
        # Solve LP for g given f
        c = _full_convolve(f)
        M = float(c.max())
        g = _lp_solve(f, M)
        if g is None:
            break
        s_g = float(g.sum())
        if s_g < 1e-10:
            break
        g_norm = g / s_g
        
        # Blend
        f_new = (1.0 - t) * f + t * g_norm
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        sc_new = _score_norm(f_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = f_new.copy()
        
        # Now solve LP for f' given g_norm
        c2 = _full_convolve(g_norm)
        M2 = float(c2.max())
        f_lp = _lp_solve(g_norm, M2)
        if f_lp is not None:
            s_f = float(f_lp.sum())
            if s_f > 1e-10:
                f_lp_norm = f_lp / s_f
                f_new2 = (1.0 - t) * f_new + t * f_lp_norm
                f_new2 = np.maximum(f_new2, 1e-15)
                f_new2 = f_new2 / f_new2.sum()
                sc_new2 = _score_norm(f_new2)
                if sc_new2 < current_score:
                    current_score = sc_new2
                    current_best = f_new2.copy()
                f = f_new2
            else:
                f = f_new
        else:
            f = f_new
        
        t = max(0.15, t * 0.92)
    
    return current_best, current_score

        # = sum_k w_k * (p_{k-j} + p_{k-j} reversed)

        n = len(p)

        h = np.fft.irfft(fw * fp_rev, N)

        # Actually, gradient of sum_k w_k * (p*p)_k w.r.t. p_j
        # (p*p)_k = sum_i p_i * p_{k-i}
        # d/dp_j (p*p)_k = p_{k-j} + p_{k-j} (if k-j is valid and j is valid)
        # = p_{k-j} (treating convolution)

        vp = g * 2.0  # factor of 2 because d(conv(p,p))/dp = 2*cross_corr

            v.fill(0.0)

def _lbfgs_refine(p0, deadline, beta=0.01, maxiter=200):
    """L-BFGS refinement using smooth-max approximation."""
    n = p0.size
    z0 = np.log(np.maximum(p0, 1e-15))
    z0 -= float(np.mean(z0))
    
    def objective(z):
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        return _smooth_max_val(c, beta)
    
    def grad(z):
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        _, w = _smooth_max_grad(c, beta)
        # grad of smooth_max w.r.t. p
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp_rev, N)
        g_p = h[n - 1:2 * n - 1] * 2.0
        # grad w.r.t. z
        dot = float(np.dot(p, g_p))
        gz = p * (g_p - dot)
        gz -= float(np.mean(gz))
        return gz
    
    start = time.monotonic()
    try:
        res = minimize(objective, z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': 500,
                                'gtol': 1e-8, 'iprint': -1})
        if res.success or res.fun < objective(z0):
            z_best = res.x
            p_best = _softmax_from_z(z_best)
            score = _score_norm(p_best)
            return p_best, score
    except Exception:
        pass
    
    return p0, _score_norm(p0)

def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_symmetric(n, nu):
    """Symmetric profile: a[i] = a[n-1-i] with linear decay from center."""
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 + nu * t**2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_fourier(n, freqs, amps):

    a = np.maximum(a, 1e-12)
    return a / a.sum()

"""
Hybrid optimization: Adam in logit space + L-BFGS refinement + LP alternating
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines smooth-max gradient-based optimization with structural LP improvements

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _smooth_max_grad(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _smooth_max_val(c, beta):
    val, _ = _smooth_max_grad(c, beta)
    return val


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_alternating(a, deadline, t_init=0.35, max_solves=40):

    current_best = a.copy()
    current_score = _score_norm(a)
    f = a.copy()

def _lbfgs_obj(z, p_cache, beta):

def _lbfgs_grad(z, p_cache, beta, n):

    gz = p * (g_p - dot)
    gz -= float(np.mean(gz))

            x1 = np.linspace(0, 1, target_n)
            p_int = np.interp(x1, x0, best_seq)
            p_int = np.maximum(p_int, 1e-12)
            p_int = p_int / p_int.sum()
            sc_int = _score_norm(p_int)
            print(f"  Scale to n={target_n}: {sc_int:.10f}")
            _try_update(p_int, sc_int, f'scale_n{target_n}')

            per = min(80.0, phase5_end - time.monotonic() - 30.0)
            if per > 20:
                p_e, sc_e, _ = _adam_logit(p_int, time.monotonic() + per, rng,
                                            lr=0.05, beta_frac=0.08, beta_decay=0.92,
                                            no_imp_limit=200)
                _try_update(p_e, sc_e, f'scale_adam_n{target_n}')

    # --- Phase 6: Multi-restart at best length ---
    phase6_end = deadline - 40.0
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(25.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5.0)
            if per < 12:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.04 + 0.04 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r, _ = _adam_logit(p_s, time.monotonic() + per, rng, lr=0.05,
                                        beta_frac=0.07, beta_decay=0.93,
                                        no_imp_limit=int(0.25 * n_best))
            _try_update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 7: Final polish ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            # Final L-BFGS with very low beta
            p_f, sc_f = _lbfgs_refine(best_seq, time.monotonic() + per * 0.6,
                                       beta=0.003, maxiter=int(per * 0.6 * 2))
            _try_update(p_f, sc_f, 'final_lbfgs')

        # Final LP if size allows
        if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < deadline - 5:
            per_lp = deadline - time.monotonic() - 5
            if per_lp > 15:
                p_final, sc_final = _lp_alternating(best_seq, time.monotonic() + per_lp,
                                                     t_init=0.2, max_solves=15)
                _try_update(p_final, sc_final, 'final_LP')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
