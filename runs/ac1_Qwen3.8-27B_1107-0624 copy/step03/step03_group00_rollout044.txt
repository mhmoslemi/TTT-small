"""
Multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses chirp-based initialization, cosine-annealed smooth-max in logit space,
interleaved LP refinement, and multi-scale escalation up to n=8192.
Key differences from previous: longer sequences, chirp/soft-binary inits,
slower annealing, more frequent LP insertion, peak-aware temperature.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if not np.isfinite(mval):
        return 0.0, np.zeros_like(c)
    if mval < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    if beta < 1e-10:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -30.0, 30.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    if n > 8192:
        return None
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=25):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            M *= 1.05
            continue
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.08, t * 0.9)
    return current_best, current_score

    and periodic LP refinement."""

    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1_, b2_, eps = 0.9, 0.999, 1e-8
    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta_init = max(1e-3, beta_init_frac * maxc)
    beta_final = max(1e-5, beta_final_frac * maxc)
    beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    total_steps = 0

    while time.monotonic() < deadline:

        g = _conv_cross_corr(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_.fill(0.0)
            v_.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        total_steps += 1
        m_ = b1_ * m_ + (1.0 - b1_) * gz
        v_ = b2_ * v_ + (1.0 - b2_) * (gz * gz)
        mh = m_ / (1.0 - b1_ ** t_adam)
        vh = v_ / (1.0 - b2_ ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_.fill(0.0)
            v_.fill(0.0)

        maxc = float(c.max())

                best_score = score
                best_p = p.copy()
                best_z = z.copy()

            else:

        total_elapsed = time.monotonic()
        remaining_frac = max(0.0, (deadline - time.monotonic()) / max(1e-6, deadline - (deadline - 0)))
        # Use step-based annealing instead

        progress = min(1.0, no_imp / float(no_imp_limit))
        # Better: use a fixed schedule based on total steps
        step_frac = min(1.0, total_steps / 5000.0)
        cosine_val = 0.5 * (1.0 + np.cos(np.pi * step_frac))
        beta = beta_final + (beta_init - beta_final) * cosine_val

        # Adaptively adjust beta based on peak gap

            gap = c[sidx[0]] - c[sidx[1]]

                beta = max(beta_final, beta * 0.5)

                    maxc = float(c.max())

        # Restart logic
        if no_imp > no_imp_limit:
            if best_score < 1.51:
                ns = 0.06
            elif best_score < 1.53:
                ns = 0.12
            else:
                ns = 0.25
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = beta_init
            m_.fill(0.0)
            v_.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            total_steps = 0

        # Learning rate decay
        if no_imp % 400 == 0 and no_imp > 0:
            lr = max(0.002, lr * 0.9)

    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_chirp(n, f0, f1, amp):
    """Chirp signal: linear frequency sweep, good autocorrelation properties."""
    t = np.arange(n)
    freq = f0 + (f1 - f0) * t / n
    phase = 2.0 * np.pi * np.cumsum(freq) / n
    a = 1.0 + amp * np.sin(phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_soft_binary(n, rng, temp=0.5):
    """Soft binary sequence: smoothed version of random ±1."""
    b = rng.choice([-1.0, 1.0], size=n)
    a = 1.0 + temp * b
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_smooth(n, rng, n_peaks=3):
    """Gaussian process-like smooth random sequence."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for _ in range(n_peaks):
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.2)
        amp = rng.uniform(0.3, 1.0) * (1 if rng.random() > 0.5 else -1)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_symmetric(n, half_profile):
    """Create a symmetric sequence from a half-profile."""
    a = np.zeros(n)
    a[:n // 2] = half_profile
    a[n // 2:] = half_profile[::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main optimization: multi-scale Adam+LP with diverse initialization."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]", flush=True)

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, deadline in {deadline - t_start:.0f}s", flush=True)

    # --- Phase 1: Generate diverse candidates ---
    phase1_end = t_start + 100.0
    candidates = []

    # Analytical profiles at multiple lengths
    for n in [512, 1024, 2048, 4096]:
        if time.monotonic() > phase1_end:
            break
        profs = [
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('lin3.0', _make_profile_linear(n, 3.0)),
            ('cubic1', _make_profile_cubic(n, 1.5, -1.0)),
            ('cubic2', _make_profile_cubic(n, 2.0, -1.5)),
            ('cubic3', _make_profile_cubic(n, 1.0, -0.5)),
            ('exp0.3', _make_profile_exponential(n, 0.3)),
            ('exp1.0', _make_profile_exponential(n, 1.0)),
            ('exp2.0', _make_profile_exponential(n, 2.0)),
            ('exp4.0', _make_profile_exponential(n, 4.0)),
            ('twob1', _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
            ('twob2', _make_profile_two_bump(n, 0.2, 0.8, 0.2)),
            ('cos1', _make_profile_cosine(n, 1, 0.7)),
            ('cos2', _make_profile_cosine(n, 2, 0.6)),
            ('cos3', _make_profile_cosine(n, 3, 0.5)),
            ('cos5', _make_profile_cosine(n, 5, 0.4)),
            ('chirp1', _make_chirp(n, 1, 5, 0.8)),
            ('chirp2', _make_chirp(n, 2, 8, 0.7)),
            ('chirp3', _make_chirp(n, 1, 3, 0.9)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    # Random smooth sequences (Gaussian-like)
    for n in [1024, 2048, 4096]:
        if time.monotonic() > phase1_end:
            break
        for i in range(4):
            p = _make_gaussian_smooth(n, rng, n_peaks=2 + i)
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'gauss{i}_n{n}'))

    # Soft binary
    for n in [1024, 2048]:
        for temp in [0.5, 1.0]:
            p = _make_soft_binary(n, rng, temp)
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'softbin_t{temp}_n{n}'))

    # Parent sequences at various lengths
    if parent is not None:
        for n in [512, 1024, 2048, 4096, 8192]:
            if time.monotonic() > phase1_end:
                break
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)

        if 0 < pn <= 8192:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f} n={candidates[0][2]}", flush=True)
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Adam optimization on top candidates ---
    phase2_end = t_start + 800.0
    polished = []
    top_k = 8
    count = 0

    for sc, p, n, label in candidates[:top_k]:
        if time.monotonic() > phase2_end or count >= top_k:

        per_end = min(phase2_end, time.monotonic() + time_per)
        if per_end - time.monotonic() < 10:

        # Vary parameters slightly for diversity
        lr = 0.04 + 0.01 * count
        b_init = 0.12 + 0.02 * (count % 3)
        lp_ev = 150 + 50 * count

            p, per_end, rng, lr=lr, beta_init_frac=b_init,
            beta_final_frac=0.003, no_imp_limit=300,
            lp_every=lp_ev, use_lp=(n <= 4096)
        )
        _update(p_b, sc_b, f'poly_{label}_n{n}')
        polished.append((sc_b, p_b, n, label))
        count += 1
        print(f"  Phase2 [{count}/{top_k}]: best={best_score:.10f} n={best_seq.size}", flush=True)

    # Also try the parent specifically at a longer length
    if parent is not None and time.monotonic() < phase2_end - 60:

        pi = _interp_profile(parent, n_try)

    # --- Phase 3: LP refinement on best ---
    phase3_end = t_start + 1200.0
