```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

The problem is: minimize max_k sum_i a[i]*a[k-i] subject to a ≥ 0, sum(a) = 1.

At the optimum, suppose the maximum is M and is achieved at lags k_1, k_2, ..., k_K. Then:

```python
"""
Optimizing non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses projected gradient with Nesterov momentum, logsumexp smooth max,
and multi-phase optimization with continuation and restarts.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def _score_norm(a):
    """Score assuming sum(a) is already 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _grad_logsumexp(a, beta):
    """
    Gradient of 2*n*logsumexp_beta(conv(a,a)) with respect to a.
    Returns grad_a of shape (n,).
    """
    n = len(a)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # h[j] = sum_m w[m+j]*a[m] for j=0..n-1
    # h[j] = conv(w, a_rev)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    w_conv = _fft_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * h

                         lr_init=0.01, momentum=0.9, n_max_iter=1000000):

    Nesterov accelerated projected gradient method with logsumexp smooth max.
    a is on the simplex (sum=1, non-negative).
    """
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)  # momentum velocity
    best_a = a.copy()
    best_score = _score_norm(a)
    
    c = _fft_convolve(a, a)
    max_c = float(c.max())
    beta = beta_start * max_c
    lr = lr_init
    
    for t in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Gradient at y
        grad = _grad_logsumexp(y, beta)
        
        # Step
        y_new = y - lr * grad
        
        # Project onto simplex
        y_proj = _proj_simplex(y_new)
        
        # Update velocity
        v = y_proj - a
        a = y_proj
        
        # Evaluate and update best
        if (t + 1) % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:

                best_a = a.copy()

            max_c = float(c.max())
            beta = max(beta_min * max_c, beta * 0.8)

    # Final evaluation
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score
```

"""
Minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences a.
Uses Adam optimization in softmax (log) space with logsumexp smooth max,
aggressive beta continuation, multi-restart, and final coordinate surgery.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _grad_z(a, beta, n):
    """
    Compute gradient of 2*n*logsumexp_beta(conv(a,a)) w.r.t. z (log-space).
    Returns grad_z of shape (n,).
    """
    c = _fft_convolve(a, a)
    cm = float(c.max())
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _fft_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _make_profile(n, type='tri', rng=None):
    if type == 'tri':
        t = np.linspace(0, 1, n)
        a = np.minimum(t, 1 - t)
        a = np.maximum(a, 1e-10)
    elif type == 'linear':
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        a = np.maximum(a, 1e-10)
    elif type == 'quad':
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (1 - (2*t - 1)**2)
        a = np.maximum(a, 1e-10)
    elif type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2*t - 1
        a = 1.0 - 0.3*u**3
        a = np.maximum(a, 1e-10)
    elif type == 'bump':
        t = np.linspace(0, 1, n)
        a = np.exp(-2*(2*t-1)**2)
        a = np.maximum(a, 1e-10)
    elif type == 'flat':
        a = np.ones(n)
    else:
        a = np.ones(n)
    return a / a.sum()

def _adam_optimize(a_init, deadline, rng, lr=0.1, beta_frac_start=0.1,
                   beta_frac_min=0.0001, n_max_iter=1000000,
                   beta_decay=0.9, beta_decay_period=300,

                   eval_period=100):

    Adam optimization in z-space with logsumexp smooth max.
    a_init should be on the simplex (sum=1, non-negative).
    Returns (best_a, best_score) where score = 2*n*max(conv(a,a)).
    """

def _coordinate_surgery(a, deadline, rng, n_rounds=50):
    """
    Try small coordinate changes to reduce the max convolution.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    c = _fft_convolve(best_a, best_a)
    max_c = float(c.max())
    max_idx = int(np.argmax(c))
    
    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(best_a, best_a)
        max_c = float(c.max())
        max_idx = int(np.argmax(c))
        
        # Find coordinates that contribute most to c[max_idx]
        # c[max_idx] = sum_i a[i]*a[max_idx-i]
        contributions = np.zeros(n)
        for j in range(n):
            k = max_idx - j
            if 0 <= k < n:
                contributions[j] = best_a[j] * best_a[k]
        
        # Try reducing top contributors and redistributing
        top_idx = np.argsort(contributions)[::-1][:10]
        
        for j in top_idx:
            k = max_idx - j
            if k < 0 or k >= n or k == j:
                continue
            
            # Try reducing a[j] slightly, adding to a random other coordinate
            delta = best_a[j] * 0.01 * rng.uniform(0.5, 2.0)
            candidates = [i for i in range(n) if i != j and i != k 
                         and abs(i - (max_idx - i)) != 0]
            if not candidates:
                continue
            # Pick a coordinate that doesn't contribute to c[max_idx]
            rand_idx = int(rng.integers(0, n))
            while rand_idx == j or rand_idx == k:
                rand_idx = int(rng.integers(0, n))
            
            a_try = best_a.copy()
            a_try[j] -= delta
            a_try[rand_idx] += delta
            if a_try[j] < 0:
                continue
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            
            sc = _score_norm(a_try)
            if sc < best_score - 1e-12:
                best_a = a_try.copy()
                best_score = sc
    
    return best_a, best_score
