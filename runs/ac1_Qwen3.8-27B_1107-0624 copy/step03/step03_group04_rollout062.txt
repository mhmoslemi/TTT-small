"""
Optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 using flat-half convolution
initialization combined with adaptive gradient descent, LP spreading, and coordinate
descent. The flat-half construction produces sequences where the first n lags of the
auto-convolution are equal, providing a much better starting point than Gaussian or
linear profiles.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _flat_half_init(n):
    """Construct sequence with (a*a)[k] = a[0]^2 for k = 0,...,n-1 (flat first half)."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        if k > 1:
            s = np.dot(a[1:k], a[k-1:0:-1])
        else:
            s = 0.0
        a[k] = (1.0 - s) / 2.0
        if a[k] < 1e-15:
            a[k] = 0.0
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def _flat_half_partial(n, flat_frac=0.7):
    """Construct sequence with flat first fraction of auto-convolution."""
    a = np.zeros(n)
    a[0] = 1.0
    n_flat = int(n * flat_frac)
    for k in range(1, n_flat):
        if k > 1:
            s = np.dot(a[1:k], a[k-1:0:-1])
        else:
            s = 0.0
        a[k] = (1.0 - s) / 2.0
        if a[k] < 1e-15:
            a[k] = 0.0
    # Fill remaining with geometric decay
    if n_flat < n and a[n_flat - 1] > 1e-15:
        ratio = 0.95
        for k in range(n_flat, n):
            a[k] = a[k-1] * ratio
    elif n_flat < n:
        a[n_flat:] = a[max(0, n_flat-1)] * 0.5
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _gradient_of_smooth_max(a, beta):
    """Compute smooth max value and gradient w.r.t. a using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return c_max, np.zeros(n)
    w = ex / s
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]
    grad = 2.0 * h
    val = c_max + beta * float(np.log(s))
    return val, grad, c_max


def _subgradient_of_max(a):
    """Compute subgradient of max(c) w.r.t. a (for the exact max, not smooth)."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    # All argmax positions
    threshold = c_max - 1e-10
    active = c >= threshold
    w = active.astype(np.float64)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]
    grad = 2.0 * h
    return c_max, grad

        # Update Nesterov

        z = z_new

        # Restart if stuck too long

            #     if k-i != i: c[k] = c_base[k] + 2*x*a[k-i]  [a[k-i] is current, unchanged]
            #     if k-i == i: c[k] = c_base[k] + x*x
            #   else: c[k] = c_base[k]

            # The objective is max_k c[k] (since score = 2n*max/s^2, and s = s_base + x)
            # Actually s also changes! s = s_base + x where s_base = sum(a) with a[i]=0.

            # So we minimize: 2n * max_k(c_base[k] + f_k(x)) / (s_base + x)^2

            # This is a 1D optimization in x >= 0. Let's solve it.
            s_base = float(a.sum())  # sum with a[i]=0

            # For efficiency, precompute the relevant coefficients
            ki = k - i  # this depends on k, so I need to loop over k
            # Let me just evaluate at a few candidate points and do a local search

            a[i] = ai  # restore for now, we'll set it again

            # Candidate values for x
            # The optimal x equalizes the active constraints
            # For simplicity, try: current ai, and a few perturbations
            candidates = [0.0, ai * 0.3, ai * 0.7, ai, ai * 1.3, ai * 1.7]
            # Also try the value that would equalize top peaks
            c_full = c_base.copy()
            for k in range(2*n-1):
                ki_val = k - i
                if 0 <= ki_val < n:
                    if ki_val != i:
                        c_full[k] += 2.0 * ai * a[ki_val]
                    else:
                        c_full[k] += ai * ai
            c_max_val = float(c_full.max())
            # Try to reduce the max by adjusting x
            # If increasing x reduces max (because s increases): try larger x
            # If decreasing x reduces max: try smaller x

            best_x = ai
            best_local_score = float(2.0 * n * c_max_val / (s_base + ai)**2)

            for x_cand in candidates:
                if x_cand < 0:
                    continue
                # Compute new max
                c_test = c_base.copy()
                for k in range(2*n-1):
                    ki_val = k - i
                    if 0 <= ki_val < n:
                        if ki_val != i:
                            c_test[k] += 2.0 * x_cand * a[ki_val]
                        else:
                            c_test[k] += x_cand * x_cand
                max_val = float(c_test.max())
                s_test = s_base + x_cand
                if s_test < 1e-10:
                    continue
                sc = 2.0 * n * max_val / (s_test * s_test)
                if sc < best_local_score:
                    best_local_score = sc
                    best_x = x_cand

            # Fine-tune around best_x with bisection
            lo = max(0.0, best_x * 0.5)
            hi = best_x * 1.5
            for _ in range(10):
                mid = (lo + hi) / 2
                c_test = c_base.copy()
                for k in range(2*n-1):
                    ki_val = k - i
                    if 0 <= ki_val < n:
                        if ki_val != i:
                            c_test[k] += 2.0 * mid * a[ki_val]
                        else:
                            c_test[k] += mid * mid
                max_val = float(c_test.max())
                s_test = s_base + mid
                sc = 2.0 * n * max_val / (s_test * s_test) if s_test > 1e-10 else np.inf

                # Check direction
                mid2 = (mid + hi) / 2
                c_test2 = c_base.copy()
                for k in range(2*n-1):
                    ki_val = k - i
                    if 0 <= ki_val < n:
                        if ki_val != i:
                            c_test2[k] += 2.0 * mid2 * a[ki_val]
                        else:
                            c_test2[k] += mid2 * mid2
                max_val2 = float(c_test2.max())
                s_test2 = s_base + mid2
                sc2 = 2.0 * n * max_val2 / (s_test2 * s_test2) if s_test2 > 1e-10 else np.inf

                if sc < sc2:
                    hi = mid2
                else:
                    lo = mid

            a[i] = best_x

        # End coordinate sweep
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
        else:
            # Check if we've converged
            if np.allclose(a, best_a, atol=1e-10):
                break

    return best_a


def _make_conv_matrix(a):
    n = len(a)
    # For LP: constraints are (a*b)[k] <= M for all k
    # (a*b)[k] = sum_j a[j]*b[k-j] = sum_j A[k,j] * b[j]
    # A[k,j] = a[j] if 0<=k-j<n else 0 ... wait
    # (a*b)[k] = sum_j a[j] * b[k-j] where 0<=j<n and 0<=k-j<n
    # So for fixed k, the valid j are: max(0, k-n+1) <= j <= min(k, n-1)
    # And A[k,j] = a[j] for valid j
    # Wait, that's not right either.
    # (a*b)[k] = sum_j a[j] * b[k-j]
    # For this to be valid: 0<=j<n and 0<=k-j<n => k-n+1<=j<=k and 0<=j<n
    # So j ranges from max(0, k-n+1) to min(k, n-1)
    # And the coefficient of b[k-j] is a[j]
    # Let m = k-j, so j = k-m, and m ranges from max(0, k-n+1) to min(k, n-1)
    # Wait, m = k-j, so as j goes from max(0,k-n+1) to min(k,n-1),
    # m goes from k-min(k,n-1) to k-max(0,k-n+1)
    # This is getting complicated. Let me just build the matrix directly.
    
    # Actually, for the LP we want: for each k (0 to 2n-2),
    # sum_j A[k,j] * b[j] <= M
    # where A[k,j] is the coefficient of b[j] in (a*b)[k]
    # (a*b)[k] = sum_j a[j] * b[k-j] (valid j)
    # The coefficient of b[m] (where m = k-j, so j = k-m) is a[k-m] (if 0<=k-m<n)
    # So A[k,m] = a[k-m] if 0 <= k-m < n, else 0
    # i.e., A[k,m] = a[k-m] if max(0, k-n+1) <= m <= min(k, n-1), else 0
    
    # This is a Toeplitz-like matrix (actually it's the "convolution matrix")
    # For efficiency with large n, we won't build the full matrix explicitly
    # but use the structure. However, for n up to ~2048, the matrix is (2n-1) x n
    # which is about 4M entries for n=2048. That's manageable.
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # a[j] contributes to (a*b)[j+m] for m=0,...,n-1 with coefficient a[j]*b[m]... 
        # Wait, I need to be more careful.
        # (a*b)[k] = sum_j a[j] * b[k-j]
        # Fix j: a[j] * b[k-j] contributes to (a*b)[k] for k = j, j+1, ..., j+n-1
        # (since k-j must be in [0, n-1])
        # So A[j+m, j] = a[j] for m = 0, ..., n-1
        A[j:j+n, j] = a[j]
    return A


def _lp_spread(a, M, deadline_lp=None):
    """LP: maximize sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(1, (deadline_lp - time.monotonic()) if deadline_lp else 300)})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    lp_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        lp_deadline = min(deadline, time.monotonic() + 60.0)
        g = _lp_spread(a, M_lp, deadline_lp=lp_deadline)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.08)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            lp_count += 1
            # Slowly tighten
            blend_cur = max(0.1, blend_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.97)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Generate flat-half initializations at multiple lengths
    2. Gradient optimization with adaptive beta and Nesterov momentum
    3. Coordinate descent refinement
    4. LP-based spreading
    5. Final gradient polish
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    # Focus on sizes that give good trade-off between quality and computation
    lengths = [512, 768, 1024]
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths.insert(0, pn)

    print(f"Starting optimization, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    print(f"Lengths to try: {lengths}", flush=True)

    # Phase 1: Generate initializations
    inits = []
    for n_t in lengths:
        # Flat-half (primary)
        a_fh = _flat_half_init(n_t)
        sc_fh = _score(a_fh)
        inits.append((sc_fh, f'flat_half_{n_t}', n_t, a_fh))

        # Partial flat-half
        a_pf = _flat_half_partial(n_t, 0.7)
        sc_pf = _score(a_pf)
        inits.append((sc_pf, f'partial_flat_{n_t}', n_t, a_pf))

        # Linear profile
        a_lin = _make_linear_profile(n_t, 1.5)
        sc_lin = _score(a_lin)
        inits.append((sc_lin, f'linear_{n_t}', n_t, a_lin))

        # Parent interpolation
        if parent is not None:
            a_par = _interpolate_profile(parent, n_t)
            sc_par = _score(a_par)
            inits.append((sc_par, f'parent_interp_{n_t}', n_t, a_par))

    inits.sort(key=lambda x: x[0])
    print("Top initializations:", flush=True)
    for sc, name, n_t, _ in inits[:6]:
        print(f"  {name}: {sc:.8f}", flush=True)

    # Phase 2: Optimize top candidates
    # Allocate time: top 3 get full treatment, rest get quick treatment
    top_k = min(4, len(inits))
    time_per_candidate = (deadline - time.monotonic() - 200) / top_k

    for idx in range(top_k):
        if time.monotonic() > deadline - 250:
            break

        sc0, name, n_t, a0 = inits[idx]
        remaining_for_this = min(time_per_candidate * (top_k - idx),
                                 deadline - time.monotonic() - 200 - idx * 50)
        if remaining_for_this < 60:
            break

        print(f"\n--- Optimizing {name} (start={sc0:.8f}, budget={remaining_for_this:.0f}s) ---", flush=True)

        # Step 2a: Gradient optimization
        grad_deadline = time.monotonic() + remaining_for_this * 0.55
        grad_deadline = min(grad_deadline, deadline - 150)
        a_grad, sc_grad = _optimize_gradient(a0, grad_deadline)
        print(f"  Gradient: {sc_grad:.10f}", flush=True)

        if sc_grad < best_score:
            best_score = sc_grad
            best_seq = a_grad.copy()

        # Step 2b: Coordinate descent
        cd_deadline = min(time.monotonic() + remaining_for_this * 0.15, deadline - 120)
        if cd_deadline - time.monotonic() > 30 and n_t <= 1024:
            a_cd = _coordinate_descent(a_grad, cd_deadline, max_sweeps=20)
            sc_cd = _score(a_cd)
            print(f"  CoordDesc: {sc_cd:.10f}", flush=True)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            a_grad = a_cd

        # Step 2c: LP refinement
        lp_deadline = min(time.monotonic() + remaining_for_this * 0.25, deadline - 80)
        if lp_deadline - time.monotonic() > 30 and n_t <= 1024:
            a_lp = _lp_refine(a_grad, lp_deadline, max_iter=2000, inflate=1.001, blend=0.25)
            sc_lp = _score(a_lp)
            print(f"  LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_grad = a_lp

    # Phase 3: Final polish on the best
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"\n--- Final polish (score={best_score:.10f}) ---", flush=True)
        polish_deadline = deadline - 20
        a_polish, sc_polish = _optimize_gradient(best_seq, polish_deadline, max_iter=100000)
        print(f"  Polish: {sc_polish:.10f}", flush=True)
        if sc_polish < best_score:
            best_score = sc_polish
            best_seq = a_polish.copy()

        # One more coordinate descent pass
        if time.monotonic() < deadline - 25 and best_seq.size <= 1024:
            cd_deadline2 = deadline - 20
            a_cd2 = _coordinate_descent(best_seq, cd_deadline2, max_sweeps=10)
            sc_cd2 = _score(a_cd2)
            print(f"  Final CD: {sc_cd2:.10f}", flush=True)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()

    # Final validation
    if best_seq is None:
        best_seq = _flat_half_init(512)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _flat_half_init(512)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
