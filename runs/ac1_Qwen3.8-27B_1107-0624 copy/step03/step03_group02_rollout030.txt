"""
Optimized step function search using L-BFGS with continued smooth max,
alternating LP block coordinate descent, and multi-start strategy.
Key improvements: L-BFGS quasi-Newton optimization, alternating LP (solve for both
sides), continued method with careful temperature scheduling, and structured
parametric initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _smooth_max_and_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * np.log(s / len(c))
    return val, w


def _grad_smooth_max_wrt_a(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _lbfgs_smooth_max(a0, deadline, beta_decay=0.85, n_levels=25, max_iter_per_level=200):
    """L-BFGS optimization with continued smooth max method."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    c0 = _fft_convolve(a)
    beta0 = 0.1 * float(c0.max())
    best_a = a.copy()
    best_score = float(2.0 * n * c0.max())

    for level in range(n_levels):
        if time.monotonic() > deadline:
            break
        beta = beta0 * (beta_decay ** level)
        if beta < 1e-10:
            beta = 1e-10

        def fun(z_v):
            a_v = _softmax(z_v)
            c_v = _fft_convolve(a_v)
            val, _ = _smooth_max_and_weights(c_v, beta)
            return 2.0 * n * val

        def grad(z_v):
            a_v = _softmax(z_v)
            c_v = _fft_convolve(a_v)
            _, w = _smooth_max_and_weights(c_v, beta)
            grad_a = _grad_smooth_max_wrt_a(a_v, w)
            dot = float(np.dot(a_v, grad_a))
            grad_z = a_v * (grad_a - dot)
            grad_z -= float(np.mean(grad_z))
            return 2.0 * n * grad_z

        try:
            res = minimize(fun, z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_level, 'gtol': 1e-6, 'maxls': 50})
            if res.success or res.fun < fun(z):
                z = res.x
        except Exception:
            pass

        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        a = _softmax(z)
        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _adam_smooth_max(a0, deadline, lr=0.03, beta_decay=0.82, max_iter=30000):
    """Adam optimizer in logit-space with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(0.08 * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_and_weights(c_curr, beta)[1]
        grad_a = _grad_smooth_max_wrt_a(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m, v_arr, adam_t = np.zeros(n), np.zeros(n), 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        z, a = z_new, a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_imp > max(5000, 8 * n):
            rng_r = np.random.default_rng(2000 + it)
            z = best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-10, 0.08 * float(_fft_convolve(a).max()))
            m, v_arr, adam_t = np.zeros(n), np.zeros(n), 0
            no_imp = 0

    return best_a, best_score


def _lp_solve(fixed, M, max_iter_lp=5000):
    """Solve LP: maximize sum(g) s.t. conv(fixed, g)[k] <= M, g >= 0."""
    n = len(fixed)
    c_obj = -np.ones(n)
    # Build constraint matrix A where A[k, j] = contribution of g[j] to conv(fixed, g)[k]
    # conv(fixed, g)[k] = sum_j fixed[j] * g[k-j]
    # For a given j, g[j] appears in conv[k] with coefficient fixed[k-j] (if valid)
    # So A[k, j] = fixed[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k in [j, j+n-1] with coefficient fixed[k-j]
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end + 1, j] = fixed[:n]
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'maxiter': max_iter_lp, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_alternating(a0, deadline, max_rounds=200, inflate=1.001, blend=0.4):
    """Alternating LP: fix one side, solve LP for the other, alternate."""
    a = np.asarray(a0, dtype=np.float64).copy()

    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    g = a.copy()  # Initialize g

    no_imp = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Solve LP for g given a
        c_a = _fft_convolve(a)
        M = float(c_a.max()) * inflate
        g_new = _lp_solve(a, M)
        if g_new is None:
            break
        sum_g = float(g_new.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 30:
                break
            continue

        g_new = g_new / sum_g
        # Update a
        a_new = (1.0 - blend) * a + blend * g_new
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Now solve LP for a given g_new (alternate direction)
        c_g = _fft_convolve(g_new)
        M2 = float(c_g.max()) * inflate
        a_from_lp = _lp_solve(g_new, M2)
        if a_from_lp is not None:
            sum_a2 = float(a_from_lp.sum())
            if sum_a2 > 1.0 + 1e-6:
                a_from_lp = a_from_lp / sum_a2
                a_mix = (1.0 - blend) * a_new + blend * a_from_lp
                a_mix = np.maximum(a_mix, 0.0)
                s2 = a_mix.sum()
                if s2 > 1e-10:
                    a_mix = a_mix / s2
                    sc2 = _score(a_mix)
                    if sc2 < best_score:
                        best_score = sc2
                        best_a = a_mix.copy()
                        a = a_mix
                        g = a_from_lp
                        continue

        a = a_new
        g = g_new

    return best_a

def _subgradient_multi(a0, deadline, max_iter=8000, top_k=3):
    """Multi-peak subgradient method."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy().astype(np.float64)
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    base_lr = 0.25 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]
        weights = np.array([1.0 - 0.2 * i for i in range(k)])
        weights = weights / weights.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            idx = int(idx)
            for j in range(n):
                # dc[k]/da[j] = 2*a[k-j] (with a[m]=0 for m outside [0,n-1])
                m = idx - j
                if 0 <= m < n:
                    sg[j] += weights[i] * 2.0 * a[m]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _make_profile(n, kind='linear', p1=1.5, p2=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'linear':
        a = 1.0 + p1 * u
    elif kind == 'cubic':
        a = 1.0 + p1 * u + p2 * u ** 3
    elif kind == 'cosine':
        a = 1.0 + p1 * np.cos(np.pi * t)
    elif kind == 'bell':
        a = np.exp(-0.5 * (u / p1) ** 2)
    elif kind == 'quartic':
        a = 1.0 + p1 * u ** 2 + p2 * u ** 4
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    Explores multiple initializations at n=512 and n=1024 with careful time allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Build initialization list
    n512 = 512
    n1024 = 1024

    inits = []
    labels = []

    if parent is not None:
        inits.append(_interpolate_profile(parent, n512))
        labels.append("parent@512")
        inits.append(_interpolate_profile(parent, n1024))
        labels.append("parent@1024")

    # Structured profiles
    inits.append(_make_profile(n512, 'linear', 1.8))
    labels.append("linear1.8@512")
    inits.append(_make_profile(n1024, 'linear', 1.8))
    labels.append("linear1.8@1024")
    inits.append(_make_profile(n512, 'cubic', 2.2, -1.8))
    labels.append("cubic@512")
    inits.append(_make_profile(n1024, 'cubic', 2.2, -1.8))
    labels.append("cubic@1024")
    inits.append(_make_profile(n512, 'quartic', 1.0, -0.5))
    labels.append("quartic@512")
    inits.append(_make_profile(n1024, 'cosine', 1.2))
    labels.append("cosine@1024")

    # Perturbed parent
    if parent is not None:
        pp = _interpolate_profile(parent, n512)
        pp = pp * (1.0 + 0.12 * rng.normal(size=n512))
        pp = np.maximum(pp, 0.0)
        inits.append(pp / pp.sum())
        labels.append("perturbed@512")

    # Random simplex
    for rn in [n512, n1024]:
        r = rng.dirichlet(np.ones(rn) * 2.0)
        inits.append(r)
        labels.append(f"random@{rn}")

    # Allocate time: give more time to parent-based inits
    n_total = len(inits)
    base_time = (deadline - t_start - 120.0) / n_total
    # First 2 (parent-based) get double time
    times = []
    for i in range(n_total):
        if i < 2 and parent is not None:
            times.append(base_time * 2.5)
        elif i < 4:
            times.append(base_time * 1.5)
        else:
            times.append(base_time)

    # Run pipeline on each initialization
    for i, (a0, label) in enumerate(zip(inits, labels)):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(times[i], remaining - 60)
        if alloc < 30:
            break
        d_end = time.monotonic() + alloc
        print(f"[{i+1}/{n_total}] {label}: allocating {alloc:.0f}s")
        try:
            res, sc = _full_pipeline(a0, d_end)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
                print(f"  -> NEW BEST: {sc:.10f} (n={res.size})")
            else:
                print(f"  -> {sc:.10f} (not better than {best_score:.10f})")
        except Exception as e:
            print(f"  -> Error: {e}")

    # Final consolidation: take best and do intensive refinement
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement (best={best_score:.10f}, n={best_seq.size}) ===")
        bn = best_seq.size

        # Try at both 512 and 1024 if different
        candidates_final = [best_seq.copy()]
        if bn == 512:
            candidates_final.append(_interpolate_profile(best_seq, 1024))
        elif bn == 1024:
            candidates_final.append(_interpolate_profile(best_seq, 512))

        for ci, cf in enumerate(candidates_final):
            if time.monotonic() > deadline - 15:
                break
            # Quick L-BFGS
            if time.monotonic() < deadline - 30:
                d_f = min(deadline - 25, time.monotonic() + 40)
                af, sf = _lbfgs_smooth_max(cf, d_f, beta_decay=0.9, n_levels=10, max_iter_per_level=80)
                if sf < best_score:
                    best_score = sf
                    best_seq = af.copy()
                    print(f"  Final L-BFGS [{ci}]: {sf:.10f}")

            # Quick LP
            if time.monotonic() < deadline - 10:
                d_lp = min(deadline - 5, time.monotonic() + 25)
                al = _lp_polish(best_seq, d_lp, max_iter=200, inflate=1.0003, blend=0.15)
                sl = _score(al)
                if sl < best_score:
                    best_score = sl
                    best_seq = al.copy()
                    print(f"  Final LP [{ci}]: {sl:.10f}")

            # Ultra-fine Adam
            if time.monotonic() < deadline - 5:
                af2, sf2 = _adam_smooth_max(best_seq, deadline - 3, lr=0.003, beta_decay=0.92, max_iter=3000)
                if sf2 < best_score:
                    best_score = sf2
                    best_seq = af2.copy()
                    print(f"  Final Adam [{ci}]: {sf2:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profile(512, 'linear', 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
