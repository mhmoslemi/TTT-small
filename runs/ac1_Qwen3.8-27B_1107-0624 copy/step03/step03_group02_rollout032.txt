import numpy as np
import time
from scipy.optimize import minimize_scalar

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    k = 1
    while k < 2*n-1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2*n-1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c

def _score_from_conv(a, c):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s*s)

def _score(a):
    c = _fft_convolve(a)
    return _score_from_conv(a, c)

import time
from scipy.optimize import minimize_scalar


def _fft_convolve(a):

def _score_from_c(a, c):

    return float(2.0 * n * c.max() / (s * s))

def _coord_descent(a, max_sweeps=300, time_limit=None, verbose=False):
    """Exact coordinate descent minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses incremental convolution updates for O(n) per coordinate evaluation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return a, np.inf
    c = _fft_convolve(a).copy()
    score = _score_from_c(a, c)
    best_a = a.copy()
    best_score = score

    for sweep in range(max_sweeps):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        improved = False
        max_change = 0.0

        for i in range(n):
            x_old = a[i]
            S0 = s - x_old
            if S0 < 1e-15:
                continue

            # D[k] = 2*a_zero[k-i] for valid k (where a_zero = a with a[i]=0)
            # Valid k: i <= k <= i+n-1 (so 0 <= k-i <= n-1), clipped to [0, 2n-2]
            k_start = max(0, i)
            k_end = min(2 * n - 1, i + n - 1)
            if k_start > k_end:
                continue

            D = np.zeros(2 * n - 1)
            ks = np.arange(k_start, k_end + 1)
            # vals = a_zero[k-i] = a[k-i] except where k-i == i (set to 0)
            vals = a[ks - i].copy()
            # Zero out position where k-i = i, i.e., k = 2*i
            pos_2i = 2 * i - k_start
            if 0 <= pos_2i < len(vals):
                vals[pos_2i] = 0.0
            D[ks] = 2.0 * vals

            e_idx = 2 * i
            has_e = (e_idx <= 2 * n - 2)
            x_old_sq = x_old * x_old

            # Objective: f(x) = 2n * max(c + (x-x_old)*D + (x^2-x_old^2)*E) / (S0+x)^2
            def _obj(x_val):
                c_new = c + (x_val - x_old) * D
                if has_e:
                    c_new[e_idx] = c_new[e_idx] + (x_val * x_val - x_old_sq)
                mx = float(c_new.max())
                s_new = S0 + x_val
                if s_new < 1e-15:
                    return 1e18
                return 2.0 * n * mx / (s_new * s_new)

            # Bounded minimization
            x_max = max(2.0 * x_old + 1e-8, min(S0, 10.0))
            if x_old > 0:

                s = S0 + x_new
                improved = True

            best_a = a.copy()

            print(f"        CD sweep {sweep}: score={cur_score:.10f}, max_change={max_change:.2e}")

        if not improved or max_change < 1e-13:
            break

    return best_a, best_score


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a)."""
    n = len(a)
    # c[k] = sum_j a[j]*a[k-j], dc/da[i] = sum_k w[k] * (a[k-i] + a[k-i] with proper bounds)
    # Using correlation: grad[i] = 2 * (w * a_reversed)[i] where a_reversed is padded
    N = 1
    while N < 3 * n:
        N <<= 1
    a_rev = np.zeros(N)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w_pad, N)
    corr = np.fft.irfft(fa * fw, N)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad


def _adam_logit(a, deadline, max_iter=30000, lr=0.04, beta_factor=0.04, verbose=False):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            continue

        t_count += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_count)
        vh = v / (1.0 - b2 ** t_count)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.92)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(5000 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v = np.zeros(n)
                t_count = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.75)

    return best_a, best_score


def _frank_wolfe(a, deadline, max_iter=5000, beta_init=0.03, verbose=False):
    """Frank-Wolfe (conditional gradients) on the simplex for smooth max."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    beta = max(beta_init * float(c.max()), 1e-10)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad = _grad_smooth_max(a, w)

        # FW step: move toward e_{argmin grad}
        v = np.zeros(n)
        v[int(np.argmin(grad))] = 1.0

        # Step size: gamma = 2/(it+2) (standard FW)
        gamma = 2.0 / (it + 2.0)
        a_new = (1.0 - gamma) * a + gamma * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        if it % 300 == 0:
            beta = max(1e-12, beta * 0.8)

    return best_a, best_score


def _interpolate_to_length(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Uniform
    profiles.append(np.ones(n) / n)

    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        p = 1.0 + eta * (t - 0.5)
        p = np.maximum(p, 1e-8)
        profiles.append(p / p.sum())

        p = np.maximum(p, 1e-8)
        profiles.append(p / p.sum())

    u = 2.0 * t - 1.0
    for a1, a3 in [(1.5, -1.0), (2.0, -2.0), (3.0, -3.0), (1.0, 0.5), (2.5, -0.5)]:
        p = 1.0 + a1 * u + a3 * u ** 3

        p = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
        profiles.append(p / p.sum())

    # Two-hump
    p = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
    profiles.append(p / p.sum())

    # W-shape
    p = 1.0 + 0.8 * np.cos(2 * np.pi * t)
    p = np.maximum(p, 1e-8)
    profiles.append(p / p.sum())

    return profiles


def _mcmc_perturb(a, rng, scale=0.1):
    """Random multiplicative perturbation."""
    n = len(a)
    perturbation = 1.0 + scale * rng.normal(size=n)
    a_new = a * perturbation
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining exact coordinate descent, Adam in logit space,
    Frank-Wolfe, and random restarts with diverse initializations at multiple lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # === Phase 1: Coordinate descent on parent at n=512 and n=1024 ===
    for target_n in [512, 1024]:
        if time.monotonic() > deadline - 120:
            break
        if parent is not None:
            a0 = _interpolate_to_length(parent, target_n)
        else:
            a0 = np.ones(target_n) / target_n

        cd_limit = min(deadline - 100, time.monotonic() + 45)
        a1, s1 = _coord_descent(a0, max_sweeps=200, time_limit=cd_limit, verbose=True)
        print(f"  CD n={target_n}: score={s1:.10f} (time={time.monotonic()-t_start:.1f}s)")
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()

    # === Phase 2: Adam refinement on best ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        a2, s2 = _adam_logit(best_seq, deadline - 60, max_iter=50000, lr=0.04, beta_factor=0.04)
        print(f"  Adam (best@{best_seq.size}): score={s2:.10f} (time={time.monotonic()-t_start:.1f}s)")
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()

    # === Phase 3: Coordinate descent after Adam ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        a3, s3 = _coord_descent(best_seq, max_sweeps=200, time_limit=deadline - 40)
        print(f"  CD-after-Adam: score={s3:.10f} (time={time.monotonic()-t_start:.1f}s)")
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()

    # === Phase 4: Diverse initializations + coordinate descent ===
    if time.monotonic() < deadline - 80:
        n_div = 512
        profiles = _make_profiles(n_div)

        # Also add perturbed parent
        if parent is not None:
            p512 = _interpolate_to_length(parent, n_div)
            for pscale in [0.1, 0.2, 0.3]:
                p_pert = _mcmc_perturb(p512, rng, scale=pscale)
                profiles.append(p_pert)

        # Interpolated parent variants
        if parent is not None:
            p400 = _interpolate_to_length(parent, 400)
            p512_back = _interpolate_to_length(p400, n_div)
            profiles.append(p512_back)

        per_budget = (deadline - 80 - time.monotonic()) / max(1, len(profiles))
        per_budget = min(per_budget, 30.0)

        for i, a0 in enumerate(profiles):
            if time.monotonic() > deadline - 80:
                break
            if per_budget < 8:
                break
            dl = min(deadline - 80, time.monotonic() + per_budget)
            a_d, s_d = _coord_descent(a0, max_sweeps=150, time_limit=dl)
            if s_d < best_score:
                best_score = s_d
                best_seq = a_d.copy()
                print(f"  Diverse-{i}: score={s_d:.10f} (NEW BEST)")

    # === Phase 5: Frank-Wolfe on best ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_fw, s_fw = _frank_wolfe(best_seq, deadline - 30, max_iter=3000, beta_init=0.02)
        print(f"  Frank-Wolfe: score={s_fw:.10f} (time={time.monotonic()-t_start:.1f}s)")
        if s_fw < best_score:
            best_score = s_fw
            best_seq = a_fw.copy()

    # === Phase 6: Random restarts from best + coordinate descent ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_restart = len(best_seq)
        n_restarts = min(10, int((deadline - 20 - time.monotonic()) / 15))
        n_restarts = max(n_restarts, 3)
        per_restart = (deadline - 20 - time.monotonic()) / max(1, n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            scale = 0.05 + 0.1 * (r % 3)
            a_pert = _mcmc_perturb(best_seq, rng, scale=scale)
            dl = min(deadline - 20, time.monotonic() + per_restart - 2)
            a_r, s_r = _coord_descent(a_pert, max_sweeps=100, time_limit=dl)
            if s_r < best_score:
                best_score = s_r
                best_seq = a_r.copy()
                print(f"  Restart-{r}: score={s_r:.10f} (NEW BEST)")

    # === Phase 7: Try n=2048 if time permits ===
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_2048 = _interpolate_to_length(best_seq, 2048)
        dl_2048 = min(deadline - 10, time.monotonic() + 40)
        a_2k, s_2k = _coord_descent(a_2048, max_sweeps=100, time_limit=dl_2048)
        print(f"  CD n=2048: score={s_2k:.10f} (time={time.monotonic()-t_start:.1f}s)")
        if s_2k < best_score:
            best_score = s_2k
            best_seq = a_2k.copy()

    # === Phase 8: Final Adam polish ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, s_fin = _adam_logit(best_seq, deadline - 3, max_iter=10000, lr=0.01, beta_factor=0.01)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
        # One more CD pass
        if time.monotonic() < deadline - 3:
            a_fin2, s_fin2 = _coord_descent(best_seq, max_sweeps=50, time_limit=deadline - 2)
            if s_fin2 < best_score:
                best_score = s_fin2
                best_seq = a_fin2.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        # Keep as-is (score is scale invariant)
        pass
    else:
        final_seq = np.ones(256) / 256.0

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
