Optimized hybrid search for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. Iterative LP with adaptive tight bounds (most powerful improvement)
2. Active-set LP refinement (constrain only top positions)

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M, top_k=None):
    """LP: maximize sum(g) s.t. conv(a,g) <= M (optionally only top_k positions), g >= 0."""
    n = len(a)
    c = _fft_convolve(a)
    if top_k is not None and top_k < 2 * n - 1:
        # Only constrain the top_k positions
        top_idx = np.argsort(c)[::-1][:top_k]
        A_full = _make_conv_matrix(a)
        A = A_full[top_idx, :]
        b = np.full(len(top_idx), M)
    else:
        A = _make_conv_matrix(a)
        b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _conv_grad_presmooth(a, w):
    """Gradient of smooth max of conv(a,a) w.r.t. a, given weights w on conv positions."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _logit_adam_optimize(a, deadline, max_iter=80000, lr=0.05):

    n = len(a)

    z -= float(np.mean(z))

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c_new = _fft_convolve(a)
        score = float(2.0 * n * c_new.max())

            best_score = score

            best_z = z_new.copy()

def _lp_refine_iterative(a, deadline, max_iter=50000):
    """
    Iterative LP with adaptive tight bounds.
    Key: tighten M aggressively, use full replacement when beneficial.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    best_a = a.copy()

    no_imp = 0
    eps = 0.001  # Initial inflation
    top_k = None  # Start with all constraints

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M_cur = float(c.max())
        M_lp = M_cur * (1.0 + eps)

        g = _lp_spread(a, M_lp, top_k=top_k)

        if g is None:
            # LP infeasible, increase eps
            eps = min(0.05, eps * 1.5)
            if eps > 0.04:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1e-10:
            eps = min(0.05, eps * 1.3)
            continue

        g = g / sum_g  # Normalize to same sum as a

        # Try full replacement first
        c_g = _fft_convolve(g)
        score_g = float(2.0 * n * c_g.max())

        if score_g < best_score - 1e-13:
            # Full replacement is better
            a = g.copy()
            best_score = score_g
            best_M = float(c_g.max())
            best_a = a.copy()
            no_imp = 0
            # Try to tighten eps further
            eps = max(0.0001, eps * 0.9)
            # Update top_k: use top positions
            if it > 50:
                top_k = min(2 * n - 1, max(20, int(0.3 * (2 * n - 1))))
        else:
            # Blend
            blend = 0.5 if no_imp < 100 else 0.3
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            c_new = _fft_convolve(a_new)
            score_new = float(2.0 * n * c_new.max())

            if score_new < best_score - 1e-13:
                best_score = score_new
                best_M = float(c_new.max())
                best_a = a_new.copy()
                a = a_new
                no_imp = 0
                eps = max(0.0001, eps * 0.95)
            else:
                a = a_new  # Still update to explore
                no_imp += 1
                if no_imp % 50 == 0:
                    eps = min(0.05, eps * 1.1)
                if no_imp > max(300, 3 * n):
                    a = best_a.copy()
                    eps = 0.001
                    top_k = None
                    no_imp = 0

    return best_a


def _active_set_lp(a, deadline, max_iter=20000):

    Active-set LP: only constrain positions near the max.
    This makes the LP less constrained and potentially finds better directions.
    """

    s = a.sum()
    if s < 1e-10:
        return a

    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    best_a = a.copy()

    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M_cur = float(c.max())

        # Identify active set: positions where c[k] > 0.95 * M_cur
        threshold = 0.95 * M_cur
        active_mask = c > threshold
        n_active = int(active_mask.sum())

        if n_active < 3:
            n_active = max(5, int(0.05 * (2 * n - 1)))
            top_idx = np.argsort(c)[::-1][:n_active]
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[top_idx] = True

        # LP with only active constraints
        M_lp = M_cur * 1.001
        n = len(a)
        A_full = _make_conv_matrix(a)
        A = A_full[active_mask, :]
        b = np.full(n_active, M_lp)
        c_obj = -np.ones(n)

        g = res.x

        else:
            blend = 0.3

            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:

            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                no_imp = 0
                threshold = 0.9 * float(_fft_convolve(a).max())

        a = a_new

def _gradient_polish(a, deadline, max_iter=50000):
    """Fine gradient polish with exact max subgradient."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.01
    no_imp = 0

    # Track top positions for subgradient averaging
    top_count = 3

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        c_max_cur = float(c.max())

        # Use average of top few subgradients
        top_idx = np.argsort(c)[::-1][:top_count]
        w = np.zeros_like(c)
        for idx in top_idx:
            w[idx] = 1.0
        w = w / w.sum()

        vp = _conv_grad_presmooth(p, w) + 1e-9 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.0005, lr * 0.9)
            if no_imp > max(5000, 8 * n):
                break

    return best_a


def _make_anti_symmetric(n, rng):
    """Create an anti-symmetric-like profile: high on left, low on right."""
    t = np.linspace(0.0, 1.0, n)
    # Rising ramp with some curvature
    a = t ** 1.5 + 0.1
    a = np.maximum(a, 1e-10)
    # Add small noise
    a = a * (1.0 + 0.05 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, rng):
    """Two separate peaks."""
    t = np.linspace(0.0, 1.0, n)
    center1 = 0.25 + 0.05 * rng.standard_normal()
    center2 = 0.75 + 0.05 * rng.standard_normal()
    width = 0.1
    a = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2) + 0.01
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_spectrum(n, rng):
    """Initialize from flat power spectrum."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_bell(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _multi_refine(a, deadline, n_attempts=6, rng=None):
    """Multi-start local refinement with perturbations."""
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_attempts):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.03 * (i + 1)
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.3:
            continue

        # Quick LP refinement
        sub_dl = min(deadline, time.monotonic() + 20.0)
        a_ref = _lp_refine_iterative(a_pert, sub_dl, max_iter=2000)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

    Phases:
    1. Diverse initialization (parent, anti-symmetric, bell, flat-spectrum, bimodal)
    2. Logit-Adam optimization with adaptive smooth/exact max
    3. Iterative LP with tight bounds (main workhorse)
    4. Active-set LP refinement
    5. Fine gradient polish with exact subgradient
    6. Multi-start local search
    7. Final LP pass
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    target_n = 768
    lengths = [target_n]
    if parent is not None and parent.size != target_n:
        lengths.insert(0, parent.size)

    # Phase 1: Generate diverse initializations
    print(f"Phase 1: Initialization, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))
        init_pool.append(('anti_sym', n_t, _make_anti_symmetric(n_t, rng)))
        init_pool.append(('bell', n_t, _make_bell(n_t)))
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum(n_t, rng)))
        init_pool.append(('linear', n_t, _make_linear(n_t, 1.5)))
        init_pool.append(('bimodal', n_t, _make_bimodal(n_t, rng)))
        # Additional flat-spectrum variants
        init_pool.append(('flat_spec2', n_t, _make_flat_spectrum(n_t, rng)))
        init_pool.append(('flat_spec3', n_t, _make_flat_spectrum(n_t, rng)))

    # Quick evaluation
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 6 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:6]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Optimize top candidates
    # Focus time on best 3 candidates
    top_inits = init_scores[:3]
    time_per_init = (deadline - t_start - 400) / max(1, len(top_inits))

    for idx, (sc0, name, n_t, a0) in enumerate(top_inits):
        if time.monotonic() > deadline - 350:
            break
        if sc0 >= best_score * 1.1 and name != 'parent':
            continue

        # Step 2a: Logit-Adam
        adam_dl = min(deadline - 300, time.monotonic() + time_per_init * 0.5)
        print(f"Phase 2a [{name}, n={n_t}]: Adam, start={sc0:.8f}, budget={adam_dl - time.monotonic():.0f}s", flush=True)
        a_opt = _logit_adam_optimize(a0, adam_dl, max_iter=60000, lr=0.04)
        sc_opt = _score(a_opt)
        print(f"  Adam result: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # Step 2b: Iterative LP (main workhorse)
        if time.monotonic() < deadline - 200:
            lp_dl = min(deadline - 150, time.monotonic() + time_per_init * 0.35)
            print(f"Phase 2b [{name}]: LP iterative, budget={lp_dl - time.monotonic():.0f}s", flush=True)
            a_lp = _lp_refine_iterative(a_opt, lp_dl, max_iter=30000)
            sc_lp = _score(a_lp)
            print(f"  LP result: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Step 2c: Active-set LP
        if time.monotonic() < deadline - 100:
            as_dl = min(deadline - 80, time.monotonic() + 60.0)
            print(f"Phase 2c [{name}]: Active-set LP, budget={as_dl - time.monotonic():.0f}s", flush=True)
            a_as = _active_set_lp(a_opt, as_dl, max_iter=10000)
            sc_as = _score(a_as)
            print(f"  Active-set result: {sc_as:.10f}", flush=True)
            if sc_as < best_score:
                best_score = sc_as
                best_seq = a_as.copy()
            a_opt = a_as

    # Phase 3: Fine gradient polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_dl = min(deadline - 40, time.monotonic() + 80.0)
        print(f"Phase 3: Gradient polish, budget={gp_dl - time.monotonic():.0f}s", flush=True)
        a_fine = _gradient_polish(best_seq, gp_dl, max_iter=30000)
        sc_fine = _score(a_fine)
        print(f"  Polish result: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Phase 4: Another LP pass on polished result
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp2_dl = min(deadline - 20, time.monotonic() + 60.0)
        print(f"Phase 4: LP pass 2, budget={lp2_dl - time.monotonic():.0f}s", flush=True)
        a_lp2 = _lp_refine_iterative(best_seq, lp2_dl, max_iter=10000)
        sc_lp2 = _score(a_lp2)
        print(f"  LP2 result: {sc_lp2:.10f}", flush=True)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()

    # Phase 5: Multi-start refinement
    if best_seq is not None and time.monotonic() < deadline - 15:
        ms_dl = min(deadline - 10, time.monotonic() + 40.0)
        print(f"Phase 5: Multi-start, budget={ms_dl - time.monotonic():.0f}s", flush=True)
        a_ms = _multi_refine(best_seq, ms_dl, n_attempts=4, rng=rng)
        sc_ms = _score(a_ms)
        print(f"  Multi-start result: {sc_ms:.10f}", flush=True)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()

    # Phase 6: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        fp_dl = deadline - 3
        if fp_dl - time.monotonic() > 5:
            a_fp = _gradient_polish(best_seq, fp_dl, max_iter=10000)
            sc_fp = _score(a_fp)
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
            print(f"  Final polish: {sc_fp:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_bell(768)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_bell(768)
        best_score = _score(final_seq)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
