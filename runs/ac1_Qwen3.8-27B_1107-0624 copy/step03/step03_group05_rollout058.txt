import numpy as np

from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _conv_matrix(a):
    """Build the convolution matrix: (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Equivalently: A[k, j] = conv(a,g)[k] contributes a[k-j]*g[j]
    # So A[k,j] = a[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For row k, A[k, j] = a[k-j] if 0 <= k-j < n
        # k-j in [0, n-1] means k in [j, j+n-1]
        A[j:j+n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, res
    return None, None


def _lp_spread_fast(a, M):
    """Faster LP using the structure of the convolution matrix."""
    n = len(a)
    # The constraint matrix is a convolution (Toeplitz-like) matrix.
    # For HiGHS, we can pass it as a sparse matrix for speed.
    from scipy.sparse import csr_matrix
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if a[k - j] > 0:
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_sp, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, res
    return None, None

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

                cols.append(j)
                vals.append(v)

    # Adaptive beta: start larger, decrease over time
    t_start = time.monotonic()

    for attempt in range(3):
        beta_cur = beta_init * (0.5 ** attempt)
        beta = beta_cur

        res = minimize(objective, best_z, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-12,
                                'gtol': 1e-10, 'maxfun': 50000, 'disp': False})

        val = res.fun
        a_final = np.exp(res.x - res.x.max())
        a_final = a_final / a_final.sum()
        sc = _score(a_final)

        if sc < best_val:
            best_val = sc
            best_z = res.x.copy()

        if time.monotonic() > deadline:
            break

    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out, _score(a_out)


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()
