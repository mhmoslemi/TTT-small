"""
Optimized step function sequence minimization using improved LP refinement,
variance-based gradient descent with multiplicative updates, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_bell(n, width=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2) + 1e-8
    return a / a.sum()


def _make_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 1e-8
    return a / a.sum()


def _make_triangle(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0) + 1e-8
    return a / a.sum()


def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    # Convolution matrix: A[i,j] = f[i-j] if valid, else 0
    # conv(f, g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    # For row k of A: A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] includes f[k-j]*g[j] where 0 <= k-j < n
        # So for fixed j, k ranges from j to j+n-1
        # A[k, j] = f[k-j] for k in [j, j+n-1]
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, n_grid=11):
    """Find best blending parameter t in [(1-t)*a + t*g]"""
    best_t = 0.0
    best_sc = _score(a)
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    ts = np.linspace(0.02, 1.0, n_grid)
    for t in ts:
        a_new = (1.0 - t) * a_norm + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:

        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    # Refine around best_t

        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        ts_fine = np.linspace(lo, hi, 9)
        for t in ts_fine:
            a_new = (1.0 - t) * a_norm + t * g_norm
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = _score(a_new)
            if sc < best_sc:

                best_t = t

def _variance_grad(a, beta=None):
    """Compute gradient of variance of convolution (flatness objective)."""
    n = len(a)
    c = _fft_convolve(a)
    mean_c = c.mean()
    
    if beta is None:
        beta = max(1e-8, 0.1 * (c.max() - mean_c))
    
    # Weight: emphasize entries above the mean
    x = (c - mean_c) / beta
    # Use softplus-like weighting for entries above mean
    w = np.maximum(x, 0.0) + 0.1  # baseline weight
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 50.0))  # clip to avoid overflow
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _gradient_descent(a0, deadline, max_iter=100000):

    Gradient descent with multiplicative updates to minimize the score.
    Uses smooth max of convolution as the objective.

    a = a / a.sum()

            a_new = np.maximum(a_new, 0)

        a = a_new

        score = float(2.0 * n * c.max())

            best_a = a.copy()
            no_imp = 0

                no_imp = 0

def _solve_lp_minmax(f, n):
    """
    Solve: min t s.t. conv(f, g)_k <= t, sum(g) = 1, g >= 0
    Variables: g_0,...,g_{n-1}, t (total n+1 variables)
    """
    # conv(f, g)_k <= t  =>  sum_j A[k,j] * g_j - t <= 0
    # A is the (2n-1) x n convolution matrix
    # 
    # In standard LP form (min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds):
    # x = [g_0, ..., g_{n-1}, t]
    # c = [0, ..., 0, 1]  (minimize t)
    # A_ub: for each k, [A[k,0], ..., A[k,n-1], -1] @ x <= 0
    #       i.e., sum_j A[k,j]*g_j - t <= 0
    # A_eq: [1, 1, ..., 1, 0] @ x = 1  (sum(g) = 1)
    # bounds: g_i >= 0, t >= 0 (or unbounded below? t should be >= 0 since conv >= 0 for g >= 0)
    
    A_conv = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_conv[j:j+n, j] = f
    
    # A_ub: (2n-1) x (n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1  # coefficient of t
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: 1 x (n+1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    bounds = [(0, None)] * n + [(0, None)]  # g_i >= 0, t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]  # return g and t*
    return None, None
