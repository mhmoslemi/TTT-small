"""
Optimized step function search using LSE-Adam with improved annealing,
Frank-Wolfe conditional gradient, and multi-length exploration.
Target: minimize 2n*max(conv(a,a))/sum(a)^2 below 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _lse_adam_step(a, z, m, v, t, T, lr):
    """One Adam step in logit space using LSE-smoothed max."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    # Gradient of LSE: 2 * conv(w, reverse(a)) centered at [n-1:, :2n-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    # Project gradient to tangent space of simplex
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, m, v, t, m_val

    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1.0 - b1) * gz
    v = b2 * v + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z -= float(np.mean(z))
    a_new = _softmax(z)

    return a_new, z, m, v, t, m_val


def _lse_optimize(a, deadline, max_iter, T_start, T_end, lr, patience):
    """LSE-Adam optimization with geometric annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr

    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a, z, m, v, t, m_val = _lse_adam_step(a, z, m, v, t, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

def _frank_wolfe(a, deadline, max_iter, ls_iters=30):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        
        # Find all near-max lags
        threshold = m_val * 0.999
        max_lags = np.where(c >= threshold)[0]
        
        # Subgradient: average over near-max lags
        # grad_i = sum over max_lags k of partial(conv_k / partial(a_i))
        # For simplicity, use the primary max
        k_star = int(np.argmax(c))
        
        # Subgradient of conv_k at a_i:
        # = a_{k-i} (if 0 <= k-i < n and 2i != k)
        # = 2*a_{k-i} (if 0 <= k-i < n and 2i != k) ... 
        # Actually let me just compute it directly
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = a[j]  # a_i * a_i, derivative is a_i
                else:
                    grad[i] = 2.0 * a[j]
        
        # Project to tangent space
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        
        # Linear minimization over simplex: argmin_j grad_tan_j
        i_min = int(np.argmin(grad_tan))
        
        # Line search along a -> e_{i_min}
        e = np.zeros(n)
        e[i_min] = 1.0
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = _score_norm((1 - c_pt) * a + c_pt * e)
        fd = _score_norm((1 - d_pt) * a + d_pt * e)
        for _ in range(min(ls_iters, 20)):
            if fc < fd:
                hi = d_pt
                d_pt, fd = c_pt, fc
                c_pt = hi - gr * (hi - lo)
                fc = _score_norm((1 - c_pt) * a + c_pt * e)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = _score_norm((1 - d_pt) * a + d_pt * e)
        
        t_opt = (lo + hi) / 2.0
        a_new = (1.0 - t_opt) * a + t_opt * e
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        score = _score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp > 300:
                break

    return best_a


def _lp_spread(a, M, top_k=100):
    """LP-based spreading: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_aa = _fft_convolve(a)
    
    # Use only near-active constraints
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    
    # Build constraint matrix
    m = len(active)
    A = np.zeros((m, n))
    for row, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[row, j] = a[j]
    b = np.full(m, M)
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.15):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    infl = inflate
    bl = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * infl
        g = _lp_spread(a, M)

        if g is None:
            infl = min(1.005, infl * 1.3)
            no_imp += 5
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 50 == 0:
                infl = min(1.005, infl * 1.1)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = _score_norm(np.maximum((1 - c_pt) * a + c_pt * g, 0))
        fd = _score_norm(np.maximum((1 - d_pt) * a + d_pt * g, 0))
        for _ in range(25):
            if fc < fd:
                hi = d_pt
                d_pt, fd = c_pt, fc
                c_pt = hi - gr * (hi - lo)
                fc = _score_norm(np.maximum((1 - c_pt) * a + c_pt * g, 0))
            else:
                lo = c_pt
                c_pt, fc = d_pt, fd
                d_pt = lo + gr * (hi - lo)
                fd = _score_norm(np.maximum((1 - d_pt) * a + d_pt * g, 0))
        t_opt = (lo + hi) / 2.0

        a_new = np.maximum((1.0 - t_opt) * a + t_opt * g, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score = _score_norm(a_new)
        if score < best_score - 1e-14:

            best_a = a_new.copy()

        else:
            no_imp += 1
            if no_imp % 50 == 0:
                bl = max(0.05, bl * 0.9)
            if no_imp > 200:
                a = best_a.copy()
                bl = blend
                infl = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_cos(n, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _adam_cosine_anneal_lr(step, total_steps, lr_max, lr_min):
    """Cosine annealing learning rate schedule."""
    if step >= total_steps:
        return lr_min
    return lr_min + 0.5 * (lr_max - lr_min) * (1 + np.cos(np.pi * step / total_steps))

def _subgrad_polish(a, deadline, max_iter=50000, lr=0.01, momentum=0.85):
    """Subgradient polish at the actual max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = _score_norm(a)

    v_mom = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = int(np.argmax(c))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = a[j]
                else:
                    grad[i] = 2.0 * a[j]

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v_mom = momentum * v_mom - base_lr * grad_tan
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        score = _score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                base_lr = max(0.002, base_lr * 0.95)
            if no_imp > 5000:
                a = best_a.copy()
                v_mom = np.zeros(n)
                base_lr = lr
                no_imp = 0

    return best_a

Wait, I need to reconsider. The subgradient computation in the Frank-Wolfe and subgradient polish is O(n) per step because I'm computing the gradient manually. For n=1024, that's fine. But let me make sure it's correct.

For `conv_k = sum_j a_j * a_{k-j}`, the derivative w.r.t. `a_i`:
- Term where `j = i`: `a_i * a_{k-i}`, derivative is `a_{k-i}` (treating `a_{k-i}` as constant)
- Term where `k-j = i`, i.e., `j = k-i`: `a_{k-i} * a_i`, derivative is `a_{k-i}`

So if `i ≠ k-i` (i.e., `i ≠ k/2`), the total derivative is `2 * a_{k-i}`.
If `i = k/2` (i.e., `k` is even and `i = k/2`), the two terms are the same, and the derivative is `a_{k-i} = a_i`.

Also need `0 <= k-i < n` for the term to exist.

Let me verify with a simple example. `a = (x, y, z)`, n=3.
`conv = (x^2, 2xy, y^2+2xz, 2yz, z^2)`

`∂conv_1/∂x = 2y`. Using formula: k=1, i=0: j=k-i=1, a_j=a_1=y. i≠j so derivative = 2*y. ✓
`∂conv_1/∂y = 2x`. k=1, i=1: j=k-i=0, a_j=x. i≠j so derivative = 2*x. ✓
`∂conv_1/∂z = 0`. k=1, i=2: j=k-i=-1, invalid. ✓

`∂conv_2/∂x = 2z`. k=2, i=0: j=2, a_j=z. i≠j so 2z. ✓
`∂conv_2/∂y = y`. k=2, i=1: j=1, a_j=y. i=j so y. ✓
`∂conv_2/∂z = 2x`. k=2, i=2: j=0, a_j=x. i≠j so 2x. ✓

Great, the formula is correct.

For the Frank-Wolfe and subgradient methods, the O(n) Python loop might be slow. Let me vectorize it.

- `2 * a[k-i]` if `0 <= k-i < n` and `i != k-i`
- `a[k-i]` if `0 <= k-i < n` and `i == k-i`
- `0` otherwise

"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LSE-Adam with cosine annealing, Frank-Wolfe conditional gradient,
LP-based spreading, and subgradient polish. Explores n=1024 and n=2048.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _conv_subgrad(a, k):
    """Compute subgradient of conv(a,a)_k with respect to a. Vectorized."""
    n = len(a)
    g = np.zeros(n)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g[mask] = 2.0 * a[j[mask]]
    if k % 2 == 0:
        i_mid = k // 2
        if 0 <= i_mid < n:
            g[i_mid] = a[i_mid]
    return g


def _lse_adam_step(a, z, m, v, t, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, m, v, t, m_val

    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1.0 - b1) * gz
    v = b2 * v + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z -= float(np.mean(z))
    a_new = _softmax(z)

    return a_new, z, m, v, t, m_val


def _lse_optimize(a, deadline, max_iter, T_start, T_end, lr_max, patience):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0

    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        progress = it / max(max_iter, 1)
        lr = 0.01 + (lr_max - 0.01) * 0.5 * (1 + np.cos(np.pi * progress))

        a, z, m, v, t, m_val = _lse_adam_step(a, z, m, v, t, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_imp += 10
            continue

        score = _score_norm(a)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > patience:
                rng_r = np.random.default_rng(9999 + it)
                a = best_a * (1.0 + 0.04 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_imp = 0

        T *= T_ratio
        T = max(T, T_end)

    return best_a


def _frank_wolfe(a, deadline, max_iter=2000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        k_star = int(np.argmax(c))

        grad = _conv_subgrad(a, k_star)
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        i_min = int(np.argmin(grad_tan))
        e = np.zeros(n)
        e[i_min] = 1.0

        # Line search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = _score_norm((1 - c_pt) * a + c_pt * e)
        fd = _score_norm((1 - d_pt) * a + d_pt * e)
        for _ in range(18):
            if fc < fd:
                hi = d_pt
                d_pt, fd = c_pt, fc
                c_pt = hi - gr * (hi - lo)
                fc = _score_norm((1 - c_pt) * a + c_pt * e)
            else:
                lo = c_pt
                c_pt, fc = d_pt, fd
                d_pt = lo + gr * (hi - lo)
                fd = _score_norm((1 - d_pt) * a + d_pt * e)

        t_opt = (lo + hi) / 2.0
        a_new = (1.0 - t_opt) * a + t_opt * e
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score = _score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp > 200:
                break

    return best_a


def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)

    m = len(active)
    A = np.zeros((m, n))
    for row, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[row, j] = a[j]
    b = np.full(m, M)

    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _subgrad_polish(a, deadline, max_iter=30000, lr=0.008, momentum=0.85):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    if best_seq is not None and time.monotonic() < deadline - 20:
        lp2_dl = time.monotonic() + min(40.0, deadline - time.monotonic() - 20)
        if lp2_dl > time.monotonic():
            a_final = _lp_refine(best_seq, lp2_dl, max_iter=800, inflate=1.0003)
            sc = _score_norm(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
            print(f"  Final LP: {sc:.10f}")

    # ===== Final validation =====
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
