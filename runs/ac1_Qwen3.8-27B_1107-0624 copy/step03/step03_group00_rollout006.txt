def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    """Score assuming sum(a) = 1: returns 2*n*max(conv)."""
    n = a.size
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Score with arbitrary scaling: 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    """Compute cross-correlation of w and p (reversed).
    Result[i] = sum_j w[j] * p[i + j - (n-1)]... 
    Actually, we want (w * p_rev)[n-1 + k] for k in range(2n-1)...
    Let me think about this more carefully.
    
    We want the gradient of smooth_max(conv(p,p)) with respect to p.
    d/dp[j] smooth_max = sum_k w[k] * d(conv(p,p)[k])/dp[j]
    d(conv(p,p)[k])/dp[j] = p[k-j] (for valid indices)
    So grad[j] = sum_k w[k] * p[k-j]
    
    This is the cross-correlation of w and p (with p reversed).
    Specifically, if we define q = p[::-1], then:
    grad[j] = sum_k w[k] * p[k-j] = (w * q)[n-1 + j - (n-1)] ... 
    Let me use FFT.
    
    Actually, let me just compute it directly with FFT:
    grad = ifft(fft(w) * conj(fft(p))) ... no that's circular correlation.
    
    For linear cross-correlation:
    We want grad[j] = sum_k w[k] * p[k-j] for k-j in [0, n-1]
    = sum_k w[k] * p[k-j]
    
    This is the cross-correlation (w ⊛ p)[j] where the correlation is defined as:
    (w ⊛ p)[j] = sum_k w[k] * p[k-j]
    
    In terms of convolution: (w * p_rev)[n-1+j] where p_rev = p[::-1]
    Because (w * p_rev)[m] = sum_k w[k] * p_rev[m-k] = sum_k w[k] * p[n-1-m+k]
    Setting m = n-1+j: sum_k w[k] * p[n-1-(n-1+j)+k] = sum_k w[k] * p[k-j]
    
    Yes! So grad = conv(w, p_rev)[n-1 : n-1+n]
    """
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _adam_symmetric(p0, deadline, rng, lr=0.05, beta_frac=0.12, beta_decay=0.93,
                    no_imp_limit=500, sym_period=100):
    """Adam optimizer in logit space with periodic symmetrization."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    # Initialize in logit space
    z = np.log(p0 + 1e-15)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    count = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        # Project gradient onto the simplex constraint (sum p = 1)
        # The gradient of the objective in logit space is: p * (g - dot(p, g))
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        
        # Periodic symmetrization
        if sym_period > 0 and count % sym_period == 0 and count > 0:
            p = 0.5 * (p + p[::-1])
            p = p / p.sum()
            z = np.log(p + 1e-15)
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        count += 1

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.003, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive temperature
        if count % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.15:
                    beta = max(1e-4, beta * 0.6)

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = min(0.2, 0.05 + 0.02 * (no_imp // no_imp_limit))
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    # Final symmetrization
    best_p = 0.5 * (best_p + best_p[::-1])
    best_p = best_p / best_p.sum()
    final_score = _score_norm(best_p)
    if final_score < best_score:
        best_score = final_score
    return best_p, best_score


def _lp_solve_symmetric(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k, b symmetric."""
    n = len(a)
    # For symmetry: b[j] = b[n-1-j]
    # Free variables: b[0], ..., b[(n-1)//2]
    half = (n + 1) // 2  # number of free variables
    
    # Build constraint matrix for the free variables
    # (a*b)[k] = sum_j a[j] * b[j] * ... wait, it's (a*b)[k] = sum_j a[j] * b[k-j]
    # But b is symmetric: b[k-j] = b[n-1-(k-j)] = b[n-1-k+j]
    # So (a*b)[k] = sum_j a[j] * b_sym[k-j] where b_sym is the full symmetric sequence
    
    # This is more complex. Let me just solve without the symmetry constraint
    # and symmetrize the result.
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        b = np.maximum(res.x, 0.0)
        # Symmetrize
        b = 0.5 * (b + b[::-1])
        return b
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        t0 = time.monotonic()
        b = _lp_solve_symmetric(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        # Symmetrize
        a_new = 0.5 * (a_new + a_new[::-1])
        a_new = a_new / a_new.sum()
        
        sc_new = _score_norm(a_new)
        if sc_new < current_score - 1e-15:
            current_score = sc_new
            current_best = a_new.copy()
        
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        n_solves += 1
        
        # Adaptive step size
        elapsed = time.monotonic() - t0
        if elapsed > 3:
            # LP is getting slow, reduce
            t = max(0.1, t * 0.8)
        if sc_new >= current_score:
            t = max(0.05, t * 0.8)
    
    return current_best, current_score


def _make_profile(n, kind='uniform', **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'triangular':
        a = np.minimum(t, 1 - t) * 2
    elif kind == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == 'exponential':
        lam = kwargs.get('lam', 1.0)
        a = np.exp(-lam * n * t)
    elif kind == 'two_bump':
        c1 = kwargs.get('c1', 0.3)
        c2 = kwargs.get('c2', 0.7)
        s = kwargs.get('s', 0.15)
        a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    elif kind == 'cosine':
        freq = kwargs.get('freq', 1)
        amp = kwargs.get('amp', 0.5)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    elif kind == 'linear':
        eta = kwargs.get('eta', 1.0)
        a = 1.0 + eta * (t - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    a = 0.5 * (a + a[::-1])  # symmetrize
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    p = 0.5 * (p + p[::-1])
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse initialization, Adam smooth-max with
    adaptive temperature and symmetrization, LP refinement, multi-scale
    escalation, and multi-restart with noise injection.
    Exploits symmetry of the optimal solution.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                # Symmetrize parent
                parent = 0.5 * (parent + parent[::-1])
                parent = parent / parent.sum()
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, start={t_start}")

    # === Phase 1: Initial candidates (0-100s) ===
    phase1_end = t_start + 100.0
    candidates = []

    sizes = [256, 512, 1024, 2048]
    profiles = [
        ('uniform', {}),
        ('triangular', {}),
        ('raised_cosine', {}),
        ('exponential', {'lam': 0.5}),
        ('exponential', {'lam': 1.0}),
        ('exponential', {'lam': 2.0}),
        ('two_bump', {'c1': 0.3, 'c2': 0.7, 's': 0.15}),
        ('two_bump', {'c1': 0.25, 'c2': 0.75, 's': 0.2}),
        ('cosine', {'freq': 1, 'amp': 0.5}),
        ('cosine', {'freq': 2, 'amp': 0.4}),
        ('linear', {'eta': 1.5}),
    ]

    for n in sizes:
        if time.monotonic() > phase1_end - 20:
            break
        for kind, kwargs_p in profiles:
            p = _make_profile(n, kind, **kwargs_p)
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'{kind}_n{n}'))

    # Parent interpolations
    if parent is not None:
        for n in [256, 512, 1024, 2048, 4096]:
            if time.monotonic() > phase1_end - 10:
                break
            if pn != n:
                pi = _interp_profile(parent, n)
                sc = _score_norm(pi)
                candidates.append((sc, pi, n, f'parent_n{n}'))
            elif 0 < pn <= 4096:
                sc = _score_norm(parent)
                candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f} ({candidates[0][3]})")
    _update(candidates[0][1], candidates[0][0], 'init')

    # === Phase 2: Adam polish top candidates (100s - 650s) ===
    phase2_end = t_start + 650.0
    polished = []
    count = 0
    max_polish = 8

    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end - 30 or count >= max_polish:
            break
        # Skip if already very similar to a polished one
        per_end = min(phase2_end - 10, time.monotonic() + 55.0)
        if per_end - time.monotonic() < 10:
            break
        
        # Use different hyperparameters for diversity
        lr_val = 0.05 + 0.02 * (count % 3)
        beta_frac_val = 0.10 + 0.03 * (count % 2)
        
        p_b, sc_b = _adam_symmetric(p, per_end, rng, lr=lr_val,
                                     beta_frac=beta_frac_val, beta_decay=0.93,
                                     no_imp_limit=400, sym_period=200)
        _update(p_b, sc_b, f'polish_{label}')
        polished.append((sc_b, p_b, n, label + '_pol'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP refinement (650s - 1000s) ===
    phase3_end = t_start + 1000.0
    if best_seq is not None and best_seq.size <= 2048:
        print(f"  LP refine n={best_seq.size}, score={best_score:.10f}")
        p_lp, sc_lp = _lp_refine(best_seq, min(phase3_end - 30, time.monotonic() + 200),
                                  t=0.4, max_solves=30)
        _update(p_lp, sc_lp, 'LP_refine')

        # Adam polish after LP
        remaining = phase3_end - time.monotonic() - 10
        if remaining > 20:
            p_ap, sc_ap = _adam_symmetric(p_lp, time.monotonic() + remaining, rng,
                                           lr=0.04, beta_frac=0.08, beta_decay=0.94,
                                           no_imp_limit=500, sym_period=100)
            _update(p_ap, sc_ap, 'post_LP')

    # === Phase 4: Multi-scale escalation (1000s - 1400s) ===
    phase4_end = t_start + 1400.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            remaining = phase4_end - time.monotonic() - 30
            if remaining > 20:
                per = min(120.0, remaining)
                p_e, sc_e = _adam_symmetric(p_int, time.monotonic() + per, rng,
                                             lr=0.05, beta_frac=0.08, beta_decay=0.92,
                                             no_imp_limit=400, sym_period=150)
                _update(p_e, sc_e, f'escalate_n{tn}')

                # Quick LP at new scale
                if time.monotonic() < phase4_end - 10 and p_e.size <= 2048:
                    p_lp2, sc_lp2 = _lp_refine(p_e, min(phase4_end - 5, time.monotonic() + 60),
                                                t=0.3, max_solves=8)
                    _update(p_lp2, sc_lp2, f'LP_n{p_e.size}')

    # === Phase 5: Multi-restart (1400s - 1750s) ===
    phase5_end = t_start + 1750.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break
            
            # Different noise strategies
            if i == 0:
                # Small noise
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + 0.08 * rng.normal(size=n_best)
            elif i == 1:
                # Medium noise
                z = np.log(np.maximum(best_seq, 1e-12))
                z -= z.mean()
                z = z + 0.15 * rng.normal(size=n_best)
            elif i == 2:
                # Local perturbation (modify subset)
                z = np.log(np
