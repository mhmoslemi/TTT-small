def solve_lp_direction(f, M, max_time=30):
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => c = -ones(n) for minimization
    c = -np.ones(n)
    
    # Constraints: for each k in 0..2n-2, sum_i f_i * g_{k-i} <= M
    # Build A_ub (spare)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, j] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # g >= 0 handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        s = g.sum()
        if s > 0:
            return g / s  # normalize to sum=1
    return None

"""
LP-based direction finding + multi-phase optimization for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Uses linear programming to find descent
directions, combined with subgradient polishing and L-BFGS fine-tuning.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _peak_and_lag(a):
    n = len(a)
    c = _full_convolve(a, a)
    return c, float(c.max()), int(np.argmax(c))


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _subgradient_direction(a, active_lags):
    """Compute subgradient of max conv at the given active lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx_start = k - j_end
            idx_end = k - j_start
            grad[j_start:j_end + 1] += 2.0 * a[idx_start:idx_end + 1][::-1]
    grad /= max(1, len(active_lags))
    return grad


def _subgradient_steps(a, num_steps, step_size, eps_frac=0.05):
    """Perform several subgradient steps with projection to simplex."""
    for _ in range(num_steps):
        c, c_max, _ = _peak_and_lag(a)
        if not np.isfinite(c_max) or c_max <= 0:
            break
        threshold = c_max * (1.0 - eps_frac)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        grad = _subgradient_direction(a, active)
        a_new = a - step_size * grad
        a_new = _project_simplex(a_new)
        a = a_new
    return a


def _solve_lp_direction(f, M, max_time=10.0):
    """
    Solve LP: max sum(g) s.t. conv(f, g)_k <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    c_obj = -np.ones(n)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[i])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    result = linprog(
        c_obj, A_ub=A_ub, b_ub=b_ub,
        bounds=[(0, None)] * n, method='highs',
        options={'time_limit': max_time, 'presolve': True}
    )

    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None


def _line_search_step(f, g, max_t=1.0):
    """Find best step size t for f_new = (1-t)*f + t*g."""
    best_t = 0.0
    best_peak = _score_norm_fn(f)
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_t = (1.0 - t) * f + t * g
        peak = _score_norm_fn(f_t)
        if peak < best_peak:
            best_peak = peak
            best_t = t
    if best_t > 0:
        return (1.0 - best_t) * f + best_t * g, best_peak
    return f, best_peak


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_start, beta_end, max_iter=300):
    """Run L-BFGS with beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    num_levels = 8
    betas = np.geomspace(beta_start * max_c0, beta_end * max_c0, num_levels)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        iters = max(20, int(max_iter * remaining / (beta_start * max_c0) * (beta / max_c0)))
        iters = min(iters, max_iter)

        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': iters, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 30, 'maxfun': iters * 2}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x
                a_new = _softmax_fn(z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _pickard_subgradient(a, deadline, lr_init, lr_decay):
    """
    Pickard-style subgradient method with averaging for diminishing step sizes.
    Minimizes the exact max of convolution.
    """
    n = len(a)
    a_curr = a.copy()
    a_avg = a.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    t = 0
    lr = lr_init

    while time.monotonic() < deadline:
        t += 1
        c, c_max, k_star = _peak_and_lag(a_curr)

        # Find active lags (within 2% of max)
        threshold = c_max * 0.98
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([k_star])

        grad = _subgradient_direction(a_curr, active)

        # Step
        a_curr = a_curr - lr * grad
        a_curr = _project_simplex(a_curr)

        # Running average
        a_avg = (t * a_avg + a_curr) / (t + 1)

        # Check improvement
        sc = _score_norm_fn(a_avg)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_avg.copy()

        # Periodic annealing
        if t % 100 == 0:
            lr = max(1e-8, lr * lr_decay)

    return best_a, best_score

        update_best(a_result, sc_result, "final_lbfgs")

        # Tiny subgradient steps
        if time.monotonic() < phase5_end - 3:
            a_work = best_seq.copy()
            a_work = _subgradient_steps(a_work, 100, 0.002, eps_frac=0.01)
            sc = _score_fn(a_work)
            update_best(a_work, sc, "final_subgrad")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 2.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(2048, 2.0)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
