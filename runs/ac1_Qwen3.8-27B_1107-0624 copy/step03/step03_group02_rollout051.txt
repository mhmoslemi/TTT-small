"""
Multi-scale optimization for minimum peak autocorrelation step functions.
Uses adaptive smooth-max Adam, top-k peak shaving, L-BFGS refinement,
and multi-resolution strategy exploring n=512, 1024, 2048.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution using soft weights."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Gradient of sum_k w[k]*conv[k] w.r.t. a
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h, c


def _adam_smooth_max(a0, deadline, max_iter=30000, lr=0.02, beta_init_frac=0.03,
                     beta_decay=0.97, restart_threshold=5000):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_init_frac * float(c.max()), 1e-12)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad_a, c_curr = _smooth_max_grad(a, beta)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp += 1
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.95)
            if no_imp > restart_threshold:
                rng_r = np.random.default_rng(7777 + it)
                z = best_z + 0.2 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-12, beta_init_frac * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new
        beta = max(1e-14, beta * beta_decay)

    return best_a, best_score


def _peak_shaving(a, deadline, max_iter=5000, top_k=5, strength=0.1):
    """Directly reduce elements contributing most to top peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a) * (a.sum() ** 2) / (a.sum() ** 2)
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())

        # Get top-k peak positions
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]

        # For each top peak, find which a[j] contributes most and reduce it
        delta = np.zeros(n)
        for ki in range(k):
            ki_idx = int(top_idx[ki])
            # conv[ki_idx] = sum_j a[j] * a[ki_idx - j]
            # Contribution of a[j] to this peak: a[ki_idx - j] (if valid)
            j_min = max(0, ki_idx - n + 1)
            j_max = min(ki_idx, n - 1)
            if j_min > j_max:
                continue
            contrib = np.zeros(n)
            valid_j = np.arange(j_min, j_max + 1)
            contrib[valid_j] = a[ki_idx - valid_j]
            # Reduce elements with highest contribution
            total_contrib = float(contrib.sum())
            if total_contrib < 1e-15:
                continue
            # Proportional reduction
            reduction = strength * contrib / max_val
            delta -= reduction

        # Apply update: reduce high contributors, redistribute
        a_new = a + delta * max_val
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend less
            a_new2 = (1.0 - 0.3) * a + 0.3 * (a + delta * max_val)
            a_new2 = np.maximum(a_new2, 0.0)
            s_new2 = float(a_new2.sum())
            if s_new2 > 1e-10:
                a_new2 = a_new2 / s_new2
                c_new2 = _fft_convolve(a_new2)
                score_new2 = float(2.0 * n * c_new2.max())
                if score_new2 < best_score - 1e-12:
                    best_score = score_new2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    a = a  # keep current

    return best_a, best_score


def _lbfgs_smooth(a0, deadline, beta=0.01):
    """L-BFGS refinement on smooth max objective in logit space."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()
    z0 = np.log(a)

    def objective(z):
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        e = np.exp(z)
        a_local = e / e.sum()
        c = _fft_convolve(a_local)
        mval = c.max()
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        return float(mval + beta * np.log(ex.sum()))

    def gradient(z):
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        e = np.exp(z)
        a_local = e / e.sum()
        c = _fft_convolve(a_local)
        mval = c.max()
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if s < 1e-30:
            return np.zeros(n)
        w = ex / s
        # Gradient of smooth max w.r.t. a
        N = _next_pow2(3 * n)
        w_pad = np.zeros(N)
        w_pad[:2 * n - 1] = w
        a_rev = a_local[::-1].copy()
        a_pad = np.zeros(N)
        a_pad[:n] = a_rev
        fw = np.fft.rfft(w_pad, N)
        fa = np.fft.rfft(a_pad, N)
        conv_result = np.fft.irfft(fw * fa, N)
        h = conv_result[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        # Chain rule: d/da -> d/dz (softmax parameterization)
        dot = float(np.dot(a_local, grad_a))
        grad_z = a_local * (grad_a - dot)
        grad_z -= grad_z.mean()
        return grad_z

    best_z = z0.copy()
    best_val = objective(z0)

    # Multiple L-BFGS restarts
    n_restarts = 3
    per_time = max(2.0, (deadline - time.monotonic()) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 2:
            break
        z_start = z0.copy() if r == 0 else best_z + 0.1 * np.random.default_rng(r * 31).normal(size=n)
        z_start = z_start - z_start.mean()
        d_end = min(deadline - 1.0, time.monotonic() + per_time)

        def cb(xk):
            return time.monotonic() > d_end

        try:
            res = minimize(objective, z_start, jac=gradient, method='L-BFGS-B',
                           options={'maxiter': 500, 'callback': cb, 'gtol': 1e-8})
            if res.fun < best_val:
                best_val = float(res.fun)
                best_z = res.x.copy()
        except Exception:
            pass

    e = np.exp(best_z - best_z.mean())
    best_a = e / e.sum()
    c = _fft_convolve(best_a)
    best_score = float(2.0 * n * c.max())
    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_asymmetric_bump(n, center_frac=0.3, width=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center_frac) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-14)
    return a / a.sum()


def _make_two_bump(n, c1=0.25, c2=0.75, width=0.15):

        0.7 * np.exp(-((t - c2) ** 2) / (2.0 * width ** 2))

                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception as ex:
        print(f"Parent load failed: {ex}")

    if parent is None:
        parent = _make_asymmetric_bump(256, 0.35, 0.25)
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent, using bump init: score={best_score:.10f}")

    # === Main optimization at n=1024 (where best was found) ===
    print(f"\n=== Main: n=1024 (budget: {deadline - time.monotonic():.0f}s) ===")
    # Allocate ~50% of time to this
    d_main = min(deadline - 400, time.monotonic() + 0.5 * (deadline - time.monotonic()))
    a1024, s1024 = _optimize_at_n(parent, 1024, d_main, "main1024")
    if s1024 < best_score:
        best_score = s1024
        best_seq = a1024.copy()

    # === Try n=2048 ===
    if time.monotonic() < deadline - 200:
        print(f"\n=== Explore: n=2048 ===")
        d_2048 = min(deadline - 200, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        a2048, s2048 = _optimize_at_n(best_seq, 2048, d_2048, "exp2048")
        if s2048 < best_score:
            best_score = s2048
            best_seq = a2048.copy()
            print(f"  New best at n=2048: {best_score:.10f}")

    # === Diverse restarts at n=1024 ===
    if time.monotonic() < deadline - 100:
        print(f"\n=== Diverse restarts @ n=1024 ===")
        n = 1024
        diverse_inits = []

        # Perturbed best
        pert = best_seq * (1.0 + 0.1 * rng.normal(size=best_seq.size))
        pert = np.maximum(pert, 0)
        diverse_inits.append(_interpolate_profile(pert, n))

        # Asymmetric bumps at various positions
        for cf in [0.2, 0.3, 0.4]:
            diverse_inits.append(_interpolate_profile(_make_asymmetric_bump(64, cf, 0.2), n))

        # Two-bump
        diverse_inits.append(_interpolate_profile(_make_two_bump(64, 0.25, 0.7, 0.12), n))

        # Random smooth profile
        x = np.linspace(0, 1, 32)
        y = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(0.5, 2.0) * x + rng.uniform(0, 2*np.pi))
        y = np.maximum(y, 0.1)
        diverse_inits.append(_interpolate_profile(y / y.sum(), n))

        remaining = deadline - 100 - time.monotonic()
        per_init = max(15.0, remaining / max(1, len(diverse_inits)))

        for i, a0 in enumerate(diverse_inits):
            if time.monotonic() > deadline - 100:
                break
            d_i = min(deadline - 100 + i, time.monotonic() + per_init)
            # Quick optimization: just Adam + peak shave
            if d_i - time.monotonic() > 5:
                a_q, s_q = _adam_smooth_max(a0, d_i, max_iter=5000, lr=0.015,
                                            beta_init_frac=0.04, beta_decay=0.97)
                if s_q < best_score:
                    best_score = s_q
                    best_seq = a_q.copy()
                    print(f"  Diverse-{i}: NEW BEST {s_q:.10f}")
                else:
                    print(f"  Diverse-{i}: {s_q:.10f}")

    # === Final polish on best ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish (best={best_score:.10f}, n={best_seq.size}) ===")
        n_best = best_seq.size

        # Fine Adam
        if time.monotonic() < deadline - 3:
            d_fin1 = deadline - 3
            a_f1, s_f1 = _adam_smooth_max(best_seq, d_fin1, max_iter=8000, lr=0.005,
                                          beta_init_frac=0.005, beta_decay=0.99)
            if s_f1 < best_score:
                best_score = s_f1
                best_seq = a_f1.copy()
            print(f"  Final Adam: {best_score:.10f}")

        # Final peak shave
        if time.monotonic() < deadline - 1:
            a_f2, s_f2 = _peak_shaving(best_seq, deadline - 1, max_iter=1000,
                                       top_k=2, strength=0.03)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final Shave: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_asymmetric_bump(512, 0.35, 0.25)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_asymmetric_bump(256, 0.35, 0.25)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
            best_score = min(best_score, sc_eval)
    except Exception as ex:
        print(f"Eval failed: {ex}")

    total_time = time.monotonic() - t_global_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={total_time:.1f}s")
    return [float(x) for x in final_seq]
