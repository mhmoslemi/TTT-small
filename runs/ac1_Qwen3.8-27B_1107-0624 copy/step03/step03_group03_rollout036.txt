So total: dc[k]/dp[m] = p[k-m] + p[k-m] = 2*p[k-m] (when k-m ≠ m)
Or: dc[k]/dp[m] = 2*p[m] (when k-m = m)

In both cases, dc[k]/dp[m] = 2*p[k-m] (since when k-m=m, p[k-m] = p[m]).

So the gradient of c[k] with respect to p is: (grad c[k])[m] = 2*p[k-m] (for valid m, i.e., 0 <= m <= n-1 and 0 <= k-m <= n-1).

This is just 2 * (reversed and shifted p)! Specifically, grad c[k] = 2 * reverse(p)[k-n+1 : k+1] (padded with zeros).

This is a very simple structure! The subgradient of max_k c[k] is 2 times the reversed p, shifted to align with an argmax k.

So the subgradient method is:
1. Find k* = argmax c[k]
2. grad = 2 * (reversed p, shifted by k* - (n-1)) [centered]
3. Update: p_new = p * exp(-eta * (grad - mean(grad))) [multiplicative weights, centered to preserve sum]

Wait, for multiplicative weights (mirror descent with KL divergence):
p_new[j] = p[j] * exp(-eta * grad[j]) / Z

where Z = sum_j p[j] * exp(-eta * grad[j]).

This automatically preserves the non-negativity and we normalize to sum 1.

The centering (subtracting mean) is needed because the constraint is sum(p) = 1, not sum(p) <= 1. In the mirror descent with the constraint sum(p) = 1, the update should be:
p_new[j] = p[j] * exp(-eta * grad[j]) / Z

where Z ensures sum(p_new) = 1. This is equivalent to:
p_new[j] = p[j] * exp(-eta * (grad[j] - mean_weighted(grad))) / Z'

The "mean_weighted" is with respect to p, not uniform. But since we normalize anyway, it doesn't matter.

  - t_{t+1} = (1 + sqrt(1 + 4*t_t^2)) / 2
  - y_{t+1} = x_{t+1} + (t_t - 1)/t_{t+1} * (x_{t+1} - x_t)

For our problem with multiplicative weights:

```python
def _nesterov_multiplicative(p, deadline, max_iter=100000, eta0=0.1, beta=0.9):
    n = len(p)
    p = p.copy()
    best_p = p.copy()
    best_score = _score_normalized(p)
    
    # Nesterov state
    y = p.copy()  # lookahead
    t_val = 1.0
    
    # Running average
    avg_p = p.copy()
    avg_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Evaluate subgradient at y
        c = _fft_convolve(y)
        M = c.max()
        score = 2.0 * n * M
        
        # Subgradient: 2 * reversed y, shifted to align with argmax
        k_star = int(np.argmax(c))
        # grad[j] = 2 * y[k_star - j] for valid indices
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * y[idx]
        # Center the gradient (for the sum constraint)
        grad = grad - grad.mean()
        
        # Step
        p_new = p * np.exp(-eta0 * grad)
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s < 1e-30:
            p_new = p
        else:
            p_new = p_new / s
        
        # Update Nesterov parameter
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y_new = p_new + ((t_val - 1.0) / t_new) * (p_new - p)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-30:
            y_new = p_new
        else:
            y_new = y_new / s_y
        
        p = p_new
        y = y_new
        t_val = t_new
        
        # Check improvement
        c_new = _fft_convolve(p)
        score_new = 2.0 * n * c_new.max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_p = p.copy()
        
        # Running average (last N iterates)
        # Simple: exponential moving average
        alpha = 2.0 / (it + 2.0)
        avg_p = alpha * p + (1.0 - alpha) * avg_p
    
    return best_p
```

```python
"""
Optimized step function sequence search using momentum-accelerated multiplicative weights
with weighted subgradients, full LP refinement with line search, and adaptive strategies.
Improves upon previous LSE-based approach by using exact subgradients with momentum,
full-constraint LP, and better convergence properties.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _linear_conv(a, b):
    na, nb = len(a), len(b)
    N = _next_pow2(na + nb - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:na + nb - 1]


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _weighted_subgradient(p, temperature=0.001):
    n = len(p)
    c = _fft_convolve(p)
    M = float(c.max())
    x = (c - M) / temperature
    x = np.clip(x, -60.0, 60.0)
    w = np.exp(x)
    sw = float(w.sum())
    if sw < 1e-30 or not np.isfinite(sw):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    w = w / sw
    rev_p = p[::-1]
    w_conv = _linear_conv(w, rev_p)
    grad = 2.0 * w_conv[n - 1:2 * n - 1]
    grad = grad - float(np.mean(grad))
    return grad, M


def _exact_subgradient(p):
    n = len(p)
    c = _fft_convolve(p)
    M = float(c.max())
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * p[k_star - j]
    grad = grad - float(np.mean(grad))
    return grad, M, k_star


def _make_conv_matrix(p):
    n = len(p)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = p[j]
    return A


def _make_conv_matrix_sparse(p):
    n = len(p)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = p[lo:hi + 1]
    return A


def _lp_solve(p, M):
    n = len(p)
    c_obj = -np.ones(n)
    A = _make_conv_matrix_sparse(p)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _line_search_1d(p, direction, max_evals=50):
    n = len(p)
    best_t = 0.0
    best_score = _score_normalized(p)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    def eval_at(t):
        q = p + t * direction
        q = np.maximum(q, 0.0)
        s = q.sum()
        if s < 1e-30:
            return np.inf
        q = q / s
        return _score_normalized(q)

    fc = eval_at(c_pt)
    fd = eval_at(d_pt)
    for _ in range(max_evals - 2):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_at(d_pt)
    best_t = (lo + hi) / 2.0
    q = p + best_t * direction
    q = np.maximum(q, 0.0)
    s = q.sum()
    if s < 1e-30:
        return p.copy(), 0.0
    q = q / s
    return q, best_t


def _momentum_optimize(p, deadline, max_iter=200000, eta_init=0.08, momentum_init=0.90,
                       temp_init=0.005, temp_min=1e-5):
    n = len(p)
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    v = np.zeros(n)
    best_p = p.copy()
    best_score = _score_normalized(p)
    no_imp = 0
    eta = eta_init
    momentum = momentum_init
    temp = temp_init
    base_eta = eta_init
    base_momentum = momentum_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, M = _weighted_subgradient(p, temp)

        if not np.all(np.isfinite(grad)):
            grad, M, _ = _exact_subgradient(p)

        v = momentum * v + grad
        v = v - float(np.mean(v))

        p_new = p * np.exp(-eta * v)
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-30:
            p_new = p.copy()
        else:
            p_new = p_new / s

        c_new = _fft_convolve(p_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_p = p_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                eta = max(0.005, eta * 0.95)
            if no_imp % 500 == 0:
                temp = max(temp_min, temp * 0.8)
            if no_imp > max(5000, 10 * n):
                p = best_p.copy()
                v = np.zeros(n)
                eta = base_eta
                momentum = base_momentum
                temp = temp_init
                no_imp = 0

        p = p_new
        temp = max(temp_min, temp * 0.9999)

    return best_p

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    n_target = 1024
    if parent is not None:
        n_target = parent.size
        if n_target != 1024:
            n_target = 1024

    print(f"\nTarget length: {n_target}")
    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.1f}s")

    if time.monotonic() > deadline - 100:
        if best_seq is None:
            best_seq = np.ones(n_target) / n_target
        return [float(x) for x in np.clip(best_seq, 0, 1000)]

    # Build initial candidates
    candidates = []
    if parent is not None:
        par_interp = _interpolate_profile(parent, n_target)
        candidates.append(par_interp)
        for trial in range(3):
            pert = par_interp * (1.0 + 0.1 * rng.normal(size=n_target))
            pert = np.maximum(pert, 0)
            if pert.sum() > 1e-10:
                candidates.append(pert / pert.sum())

    candidates.append(_make_raised_cosine(n_target))
    candidates.append(_make_gaussian(n_target, 0.35))
    candidates.append(_make_bimodal(n_target, 0.15))
    candidates.append(_make_skewed(n_target, 2.0))
    candidates.append(_make_skewed(n_target, 3.0))

    # Quick screening: score all candidates
    scored = []
    for ci, c in enumerate(candidates):
        sc = _score_normalized(c)
        scored.append((sc, c, ci))
        print(f"  Candidate {ci}: score={sc:.8f}")

    scored.sort(key=lambda x: x[0])

    # Phase 1: Optimize top candidates with momentum (short time each)
    print("\n=== Phase 1: Momentum optimization ===")
    phase1_deadline = t_start + min(600.0, deadline - t_start + 200.0)
    results = []

    for sc_init, cand, ci in scored[:4]:
        if time.monotonic() > phase1_deadline - 30:
            break
        sub_deadline = min(time.monotonic() + 120.0, phase1_deadline - 10.0)
        p_opt = _momentum_optimize(cand, sub_deadline, max_iter=50000,
                                   eta_init=0.06, momentum_init=0.90,
                                   temp_init=0.004, temp_min=1e-5)
        sc_opt = _score_normalized(p_opt)
        print(f"  Cand {ci}: {sc_init:.8f} -> {sc_opt:.10f}")
        results.append((sc_opt, p_opt))
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = p_opt.copy()

    results.sort(key=lambda x: x[0])

    # Phase 2: LP refinement on best from Phase 1
    print("\n=== Phase 2: LP refinement ===")
    phase2_deadline = t_start + min(1200.0, deadline - t_start + 400.0)

    if results:
        sc_best_p1, p_best_p1 = results[0]
        p_lp = _lp_refine_reduced(p_best_p1, phase2_deadline, max_iter=400,
                                  top_k=400, inflate_start=1.001)
        sc_lp = _score_normalized(p_lp)
        print(f"  LP reduced: {sc_lp:.10f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = p_lp.copy()

        # Full LP on the result
        if time.monotonic() < deadline - 200:
            phase2b_deadline = min(time.monotonic() + 200.0, deadline - 50.0)
            p_lp2 = _lp_refine(p_lp, phase2b_deadline, max_iter=200,
                               inflate_start=1.0005, inflate_rate=1.01)
            sc_lp2 = _score_normalized(p_lp2)
            print(f"  LP full: {sc_lp2:.10f}")
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = p_lp2.copy()

    # Phase 3: Try second-best candidate with full pipeline
    print("\n=== Phase 3: Second candidate pipeline ===")
    if len(results) > 1 and time.monotonic() < deadline - 300:
        sc_2, p_2 = results[1]
        sub_dl = min(time.monotonic() + 100.0, deadline - 200.0)
        p_2m = _momentum_optimize(p_2, sub_dl, max_iter=40000,
                                  eta_init=0.05, momentum_init=0.92,
                                  temp_init=0.003, temp_min=1e-5)
        sc_2m = _score_normalized(p_2m)
        print(f"  Cand 2 momentum: {sc_2:.8f} -> {sc_2m:.10f}")
        if time.monotonic() < deadline - 100:
            p_2l = _lp_refine_reduced(p_2m, min(time.monotonic() + 80.0, deadline - 30.0),
                                      max_iter=200, top_k=400, inflate_start=1.001)
            sc_2l = _score_normalized(p_2l)
            print(f"  Cand 2 LP: {sc_2l:.10f}")
            if sc_2l < best_score:
                best_score = sc_2l
                best_seq = p_2l.copy()

    # Phase 4: Final fine-tuning
    print("\n=== Phase 4: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        final_deadline = deadline
        p_final = _lp_refine(best_seq, final_deadline, max_iter=100,
                             inflate_start=1.0003, inflate_rate=1.005)
        sc_final = _score_normalized(p_final)
        print(f"  Final LP: {sc_final:.10f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = p_final.copy()

        # One more momentum pass with tiny steps
        if time.monotonic() < deadline - 10:
            p_fm = _momentum_optimize(best_seq, deadline, max_iter=20000,
                                      eta_init=0.01, momentum_init=0.97,
                                      temp_init=0.0005, temp_min=1e-7)
            sc_fm = _score_normalized(p_fm)
            print(f"  Final momentum: {sc_fm:.10f}")
            if sc_fm < best_score:
                best_score = sc_fm
                best_seq = p_fm.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = _score_normalized(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(n_target) / n_target

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
```
