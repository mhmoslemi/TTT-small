def _trust_region_lp_step(f, max_change=0.5):
    """
    Solve the linearized trust-region problem.
    Find delta that minimizes max_k (c_k + 2*(f*delta)_k)
    subject to -f_j*max_change <= delta_j <= f_j*max_change, sum(delta) = 0.
    """
    n = len(f)
    c = _fft_convolve(f)
    
    # Variables: [delta_0, ..., delta_{n-1}, M]
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: c_k + 2*(f*delta)_k - M <= 0
    # A_ub has shape (2n-1, n+1)
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        for d in range(n):
            j = k - d
            if 0 <= j < n:
                A_ub[k, d] = 2.0 * f[j]
        A_ub[k, n] = -1.0
        b_ub[k] = -c[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for j in range(n):
        bounds.append((-f[j]*max_change, f[j]*max_change))
    bounds.append((0, None))  # M >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[:n]
        return delta
    return None

Let me think about what "modify magnitude to be flatter" means. We want to reduce the dynamic range of |DFT(f)|. One way:

- Or: replace with |DFT(f)|^p for p < 1 (which compresses the dynamic range)
- Or: use a "water-filling" approach to equalize the magnitudes

**Spectral step:**
Given f (non-negative, sum=1), compute F = DFT(f). We want to make |F| flatter. 

The circular autocorrelation is r_circ[k] = (1/n) * IDFT(|F|^2)[k]. We want r_circ to be flat, which means |F|^2 should be flat, which means |F| should be constant.

But F[0] = sum(f) = 1, and for a non-negative f, F[0] is the largest component (by triangle inequality: |F[k]| <= sum|f[j]| = 1 = F[0]).

So we can't make |F| truly flat. The best we can do is make |F[k]| as close to 1/sqrt(n) as possible for k >= 1, while keeping F[0] = 1.

Actually, the constraint is that f >= 0. In the frequency domain, this means F = DFT(f) where f >= 0. The set of valid F is the image of the non-negative simplex under the DFT.

A "spectral shaping" step:
1. F = DFT(f)
2. target_mag = sqrt(mean(|F[1:]|^2)) (or some other target)
3. For each k >= 1: |F[k]|_new = (|F[k]| + target_mag) / 2 (soft clipping toward target)
4. F_new[0] = F[0] (keep DC)
5. F_new = |F_new| * exp(i * angle(F))
6. f_new = real(IDFT(F_new))
7. f_new = max(f_new, 0); f_new = f_new / sum(f_new)

This is a heuristic that might help flatten the spectrum.

"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. Trust-region linearized LP (main workhorse - new approach)
2. Spectral shaping (new approach - flatten DFT magnitude)
3. Smooth-max gradient with Adam (polish)
4. Multi-length, multi-init search
"""

import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    k = 1
    while k < 2*n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2*n-1]
    mn = c.min()
    if mn < 0:
        c -= mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s*s))

def _trust_region_lp(f, max_change=0.3):
    """Solve linearized trust-region LP."""
    n = len(f)
    c = _fft_convolve(f)
    
    # Build constraint matrix efficiently
    # A_ub[k, d] = 2*f[k-d] if 0 <= k-d < n, else 0
    # A_ub[k, n] = -1
    # b_ub[k] = -c[k]
    
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        d_start = max(0, k - (n-1))
        d_end = min(k, n-1)
        # d ranges from d_start to d_end
        # f[k-d] where k-d ranges from k-d_end to k-d_start
        idx = np.arange(d_start, d_end+1)
        vals = f[k - idx]
        A_ub[k, idx] = 2.0 * vals
        A_ub[k, n] = -1.0
    b_ub = -c.copy()
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lo = -f * max_change
    hi = f * max_change
    bounds = list(zip(lo, hi)) + [(0, None)]
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def _spectral_shape(f, alpha=0.5):
    """Spectral shaping: flatten the DFT magnitude."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Target: make mag[k] for k>=1 closer to their mean
    if len(mag) > 1:
        target = np.mean(mag[1:])
        # Soft clip: move toward target
        new_mag = mag.copy()
        new_mag[1:] = (1-alpha) * mag[1:] + alpha * target
        # Keep DC
        new_mag[0] = mag[0]
        
        F_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new = f_new / s
            return f_new
    return f.copy()

        x = (c - mval) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s <= 0 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        
        # Gradient in f-space: grad_f = 2 * (w ⋆ f)
        k = 1
        while k < 2*n - 1:
            k <<= 1
        Fw = np.fft.rfft(w, k)
        Ff = np.fft.rfft(f_cur, k)
        xcorr = np.fft.irfft(np.conj(Fw) * Ff, k)[:n]
        grad_f = 2.0 * xcorr
        
        # Project gradient to logit space
        # f = softmax(z) (approximately, since we're using exp(z)/sum(exp(z)))
        # df/dz_j = f_j * (delta_jk - f_k) (for the softmax parameterization)
        # Actually, if f_j = exp(z_j) / sum(exp(z)), then:
        # df_j/dz_k = f_j * (delta_jk - f_k)
        # So grad_z_k = sum_j grad_f_j * df_j/dz_k = sum_j grad_f_j * f_j * (delta_jk - f_k)
        # = f_k * (sum_j grad_f_j * f_j - sum_j grad_f_j * f_j * f_k)
        # = f_k * (sum_j grad_f_j * f_j) - f_k^2 * (sum_j grad_f_j / f_j * f_j)
        # Hmm, this is getting complicated. Let me use a simpler projection.
        # grad_z ≈ grad_f / f (elementwise), then subtract mean to preserve zero-sum
        grad_z = grad_f / f
        grad_z -= grad_z.mean()
        
        # Adam update
        t += 1
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * grad_z**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z -= z.mean()
        z = np.clip(z, -50, 50)
        
        f = np.exp(z)
        s = f.sum()
        if s < 1e-10 or not np.isfinite(s):
            z = z + 0.1 * np.random.default_rng(it).normal(size=n)
            z -= z.mean()
            f = np.exp(z)
            s = f.sum()
        f = f / s
        
        sc = _score(f)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.95)
        
        if it % 200 == 0:
            beta = max(1e-10, beta * 0.8)
        
        if no_imp > max(3000, 5*n):
            # Restart
            z = 0.8 * np.log(best_f) - 0.8*np.log(best_f).mean() + 0.3 * np.random.default_rng(hash(it)%2**32).normal(size=n)
            z -= z.mean()
            z = np.clip(z, -50, 50)

            m = np.zeros(n)
            v = np.zeros(n)

    return best_f

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    k = 1
    while k < 2 * n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())

def _build_lp_matrix(f):
    """Build the A_ub matrix for the trust-region LP efficiently."""
    n = len(f)
    A = np.zeros((2 * n - 1, n + 1))
    for d in range(n):
        start = d
        end = min(d + n, 2 * n - 1)
        length = end - start
        if length > 0:
            A[start:end, d] = 2.0 * f[:length]
    A[:, n] = -1.0
    return A


def _trust_region_lp(f, max_change=0.3):

    n = len(f)

    hi = f * max_change

    return None

def _spectral_shape(f, alpha=0.5):
    """Spectral shaping: flatten the DFT magnitude to reduce autocorrelation peaks."""
    n = len(f)
    if n < 4:
        return f.copy()
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)

    if len(mag) > 1:
        target = float(np.mean(mag[1:]))
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - alpha) * mag[1:] + alpha * target
        new_mag[0] = mag[0]
        F_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            return f_new / s
    return f.copy()

            best_sc = sc
            best_f = f_new.copy()
    return best_f


def _adam_polish(f, deadline, max_iter=20000, lr_init=0.05):
    """Adam optimization in logit space with smooth-max gradient."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 1e-12)
    f = f / f.sum()
    n = len(f)

    z = np.log(f)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    base_lr = lr
    c_init = _fft_convolve(f)
    beta = max(1e-8, 0.05 * float(c_init.max()))

    best_f = f.copy()
    best_score = _score(f)
    no_imp = 0
    rng = np.random.default_rng(42)

    k_fft = 1
    while k_fft < 2 * n - 1:
        k_fft <<= 1

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(f)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex <= 0 or not np.isfinite(s_ex):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        Fw = np.fft.rfft(w, k_fft)
        Ff = np.fft.rfft(f, k_fft)
        xcorr = np.fft.irfft(np.conj(Fw) * Ff, k_fft)[:n]
        grad_f = 2.0 * xcorr

        grad_z = grad_f / f
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_f)
            z -= float(np.mean(z))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * grad_z * grad_z
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z -= float(np.mean(z))
        z = np.clip(z, -50.0, 50.0)

        f = np.exp(z)
        s = f.sum()
        if s < 1e-10 or not np.isfinite(s):
            z = 0.8 * np.log(best_f) - 0.8 * float(np.mean(np.log(best_f)))
            z += 0.2 * rng.normal(size=n)
            z -= float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            f = np.exp(z)
            s = f.sum()
        f = f / s

        sc = _score(f)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.8)

        if no_imp > max(3000, 5 * n):
            z = 0.7 * np.log(best_f) - 0.7 * float(np.mean(np.log(best_f)))
            z += 0.4 * rng.normal(size=n)
            z -= float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            f = np.exp(z)
            f = f / f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = base_lr
            beta = max(1e-8, 0.05 * float(_fft_convolve(f).max()))
            no_imp = 0

    return best_f


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_mixture(n, centers, sigmas, weights):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, depth=1.0):
    """W-shaped profile: high at edges and center, low in between."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * (4.0 * t * t - 2.0 * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-10:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return p / s


def _gradient_polish_simple(f, deadline, max_iter=10000):
    """Simple gradient descent with momentum on the simplex."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_score = _score(f)
    momentum = 0.85
    v = np.zeros(n)
    lr = 0.02
    base_lr = lr
    c_init = _fft_convolve(f)
    beta = max(1e-8, 0.05 * float(c_init.max()))
    no_imp = 0

    k_fft = 1
    while k_fft < 2 * n - 1:
        k_fft <<= 1

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(f)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex <= 0 or not np.isfinite(s_ex):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        Fw = np.fft.rfft(w, k_fft)
        Ff = np.fft.rfft(f, k_fft)
        xcorr = np.fft.irfft(np.conj(Fw) * Ff, k_fft)[:n]
        grad = 2.0 * xcorr
        grad -= float(np.mean(grad))

        v = momentum * v - lr * grad
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f = f - v
            v = np.zeros(n)
            continue
        f_new = f_new / s
        f = f_new

        sc = _score(f)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                f = best_f.copy()
                v = np.zeros(n)
                lr = base_lr
                beta = max(1e-8, 0.05 * float(_fft_convolve(f).max()))
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_f


def _combined_optimize(f, deadline, max_lp_time=40, max_adam_time=20, max_grad_time=10):
    """Run combined LP + spectral + Adam + gradient optimization."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s

    best_f = f.copy()
    best_score = _score(f)
    n = len(f)

    # Phase A: Spectral shaping
    if time.monotonic() < deadline - max_lp_time - max_adam_time - max_grad_time - 5:
        f_sp = _spectral_shape_multi(f, [0.3, 0.5, 0.7], deadline)
        sc_sp = _score(f_sp)
        if sc_sp < best_score:
            best_score = sc_sp
            best_f = f_sp.copy()
        f = best_f.copy()

    # Phase B: LP refinement
    if time.monotonic() < deadline - max_adam_time - max_grad_time - 5:
        lp_end = min(deadline - max_adam_time - max_grad_time - 5,
                     time.monotonic() + max_lp_time)
        a_lp = _lp_refine(f, lp_end, max_iter=20000, max_change_init=0.3)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_f = a_lp.copy()
        f = best_f.copy()

    # Phase C: Adam polish
    if time.monotonic() < deadline - max_grad_time - 5:
        ad_end = min(deadline - max_grad_time - 5, time.monotonic() + max_adam_time)
        a_ad = _adam_polish(f, ad_end, max_iter=10000, lr_init=0.04)
        sc_ad = _score(a_ad)
        if sc_ad < best_score:
            best_score = sc_ad
            best_f = a_ad.copy()
        f = best_f.copy()

    # Phase D: Gradient polish
    if time.monotonic() < deadline - 3:
        gp_end = min(deadline - 3, time.monotonic() + max_grad_time)
        a_gp = _gradient_polish_simple(f, gp_end, max_iter=5000)
        sc_gp = _score(a_gp)
        if sc_gp < best_score:
            best_score = sc_gp
            best_f = a_gp.copy()

    # Phase E: Final LP
    if time.monotonic() < deadline - 2:
        lp2_end = min(deadline - 2, time.monotonic() + 15)
        a_lp2 = _lp_refine(best_f, lp2_end, max_iter=5000, max_change_init=0.2)
        sc_lp2 = _score(a_lp2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_f = a_lp2.copy()

    return best_f


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization using trust-region LP, spectral shaping,
    and Adam polish to minimize 2n*max(conv(a,a))/sum(a)^2.
    Novel contributions:
    - Trust-region LP: directly minimizes linearized peak via LP at each step
    - Spectral shaping: flattens DFT magnitude to reduce autocorrelation peaks
    - Combined pipeline with adaptive time allocation
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Lengths to try
    lengths = [768, 512, 1024, 640, 384]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    lengths = list(dict.fromkeys(lengths))

    print(f"Trying lengths: {lengths}")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: For each length, try several initializations
    time_per_length = max(30.0, (deadline - t_start - 250.0) / len(lengths))

    for n_t in lengths:
        if time.monotonic() > deadline - 200:
            break

        t_len_end = min(deadline - 150, time.monotonic() + time_per_length * 1.3)
        print(f"\n--- Length {n_t} ---")

        # Generate initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_u_profile(n_t, 2.0))
        inits.append(_make_u_profile(n_t, 4.0))
        inits.append(_make_u_profile(n_t, 6.0))
        inits.append(_make_u_profile(n_t, 10.0))
        inits.append(_make_linear_profile(n_t, 2.0))
        inits.append(_make_linear_profile(n_t, 3.0))
        inits.append(_make_gaussian_mixture(n_t, [0.1, 0.9], [0.15, 0.15], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_t, [0.05, 0.95], [0.1, 0.1], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_t, [0.2, 0.8], [0.2, 0.2], [1.0, 1.0]))
        inits.append(_make_w_profile(n_t, 1.0))
        inits.append(_make_w_profile(n_t, 2.0))
        # Random structured
        for _ in range(2):
            a_r = np.maximum(rng.normal(1.0, 0.7, n_t), 0.0)
            inits.append(a_r / a_r.sum())

        # Sort by initial score (try best first)
        scored_inits = []
        for a0 in inits:
            sc0 = _score(a0)
            scored_inits.append((sc0, a0))
        scored_inits.sort(key=lambda x: x[0])

        # Try top inits
        max_inits = min(len(scored_inits), max(3, int((t_len_end - time.monotonic()) / 35.0)))
        for sc0, a0 in scored_inits[:max_inits]:
            if time.monotonic() > t_len_end - 30:
                break

            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            print(f"  Init (n={n_t}): score={sc0:.8f}")

            # Run combined optimization
            t_opt_end = min(t_len_end - 10, time.monotonic() + 55.0)
            a_opt = _combined_optimize(a0, t_opt_end,
                                       max_lp_time=30, max_adam_time=12, max_grad_time=8)
            sc_opt = _score(a_opt)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  ** New best: {sc_opt:.10f} (n={n_t})")

    if best_seq is not None and time.monotonic() < deadline - 30:

                              max_iter=50000, max_change_init=0.25)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  2a LP ({t_lp:.0f}s): {sc:.10f}")

        # 2b: Spectral + LP
        if time.monotonic() < deadline - 20:
            f_sp = _spectral_shape_multi(best_seq, [0.2, 0.4, 0.6, 0.8])
            sc_sp = _score(f_sp)
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = f_sp.copy()
            t_lp2 = min(remaining * 0.15, 40.0)

                                   max_iter=20000, max_change_init=0.2)
                sc = _score(a_lp2)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_lp2.copy()
                print(f"  2b Spectral+LP: {sc:.10f}")

        # 2c: Adam

                a_lp3 = _lp_refine(best_seq, time.monotonic() + t_lp3,
                                   max_iter=30000, max_change_init=0.1)
                sc = _score(a_lp3)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_lp3.copy()
