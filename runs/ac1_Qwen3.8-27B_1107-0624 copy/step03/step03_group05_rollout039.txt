import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    if n <= 0:
        return 2
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_of_smooth_max(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _peak_weight(c, top_k=3, sharpness=10.0):
    """Create weights emphasizing top-k peaks with exponential falloff."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / max(mval * 0.01, 1e-15)
    ex = np.exp(-sharpness * x)
    s = float(ex.sum())
    if s < 1e-30:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_hump(n, sep=0.3, width=0.12):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5 + sep) / width) ** 2) + np.exp(-0.5 * ((t - 0.5 - sep) / width) ** 2)
    a += 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_beta_profile(n, alpha=0.7, beta_val=0.7):
    """Use beta distribution shape."""
    t = np.linspace(0.01, 0.99, n)
    a = t ** (alpha - 1) * (1 - t) ** (beta_val - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_minimize(a_fixed, deadline, max_iter=20000, inflate_start=1.003, blend_start=0.4):
    """LP-based refinement with adaptive parameters."""
    a = np.asarray(a_fixed, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflate = inflate_start
    blend = blend_start
    no_imp = 0
    stuck_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        if M < best_M * 1.00001:
            M = best_M
        M_lp = M * inflate

        g = _lp_spread(a, M_lp)
        if g is None:
            no_imp += 1
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 50 == 0:

                blend = max(0.1, blend * 0.95)
            if no_imp > 500:
                if stuck_count > 5:
                    break
                stuck_count += 1
                a = best_a.copy()
                no_imp = 0
            continue

        g = g / sum_g
        # Adaptive blend: use more of g when improvement is larger
        a_new = (1.0 - blend) * a + blend * g

        s_new = a_new.sum()

        score_new = float(2.0 * n * M_new)

            best_score = score_new
            best_M = M_new

            stuck_count = 0

                    break
                stuck_count += 1
                a = best_a.copy()
                no_imp = 0

def _gradient_polish(a_init, deadline, max_iter=30000, method="smooth_max"):
    """Gradient descent with smooth max or peak-aware weighting."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.04 * float(c.max()))
    lr = 0.035
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        if method == "peak_aware":
            c_curr = _fft_convolve(a + momentum * v)
            w = _peak_weight(c_curr, top_k=3, sharpness=15.0)
            w = 0.5 * w + 0.5 * _smooth_max_weight(c_curr, beta)
        else:
            c_curr = _fft_convolve(a + momentum * v)
            w = _smooth_max_weight(c_curr, beta)

        grad = _gradient_of_smooth_max(a + momentum * v, w)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.97)
            if no_imp > max(800, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.04 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 50 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a


def _adam_logit_polish(a_init, deadline, max_iter=25000, lr=0.06):

    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)

        w = 0.4 * w_peak + 0.6 * w_smooth

        grad_p = _gradient_of_smooth_max(p, w)
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))

        a = e / e.sum()

        score = float(2.0 * n * c.max())

            best_z = z.copy()
            no_imp = 0
        else:

            c = _fft_convolve(a)
            beta = max(1e-8, 0.04 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)

def _conv_equalize(a, deadline, max_iter=5000):
    """
    Novel: directly target convolution peaks by adjusting a.
    For each peak in conv(a,a), reduce contributing a values.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    eta = 0.02

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval = float(c.max())

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:5]

        # Compute gradient pointing toward reducing peaks
        w = np.zeros_like(c)
        for k in top_idx:
            w[k] = c[k] / mval
        w = w / w.sum()

        # Gradient of peak-weighted convolution
        grad = _gradient_of_smooth_max(a, w)

        # Update: reduce contribution to peaks
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
            eta = min(eta * 1.02, 0.1)
        else:
            no_imp += 1
            eta = max(0.003, eta * 0.97)
            if no_imp > 500:
                break

    return best_a


def _block_perturb(a, rng, block_frac=0.1):
    """Randomly perturb a block of the sequence."""
    n = len(a)
    a = a.copy()
    blen = max(2, int(n * block_frac))
    start = int(rng.integers(0, n - blen))
    a[start:start + blen] *= float(rng.uniform(0.5, 1.5))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s


def _multiscale_optimize(a_init, n_target, deadline):
    """
    Coarse-to-fine optimization: start small, gradually increase size.
    """
    sizes = []
    s = max(16, n_target // 16)
    while s < n_target:
        sizes.append(s)
        s = min(s * 2, n_target)
    sizes.append(n_target)

    a = a_init
    for n_s in sizes:
        a = _interpolate_profile(a, n_s)
        remaining = deadline - time.monotonic() - 60.0
        if remaining < 10:
            break
        budget = remaining * 0.4
        a = _lp_minimize(a, time.monotonic() + budget,
                        max_iter=3000, inflate_start=1.005, blend_start=0.3)
        a = _gradient_polish(a, time.monotonic() + budget * 0.5,
                           max_iter=5000, method="smooth_max")

    return a

            best_score = sc
            best_a = candidate.copy()
            return True
        return False

    rem = deadline - time.monotonic() - 30.0

    # Phase D: Adam in logit space

    if rem > 15:
        ad_time = min(rem * 0.18, 100.0)
        a_ad = _adam_logit_polish(best_a, time.monotonic() + ad_time,
                                 max_iter=15000, lr=0.05)

    # Phase E: Second LP pass (finer)

    if rem > 15:
        lp2_time = min(rem * 0.12, 80.0)
        a_lp2 = _lp_minimize(best_a, time.monotonic() + lp2_time,
                            max_iter=10000, inflate_start=1.001, blend_start=0.2)

    # Phase F: Final gradient polish
    rem = deadline - time.monotonic() - 10.0
    if rem > 10:
        gf_time = min(rem * 0.1, 50.0)
        a_gf = _gradient_polish(best_a, time.monotonic() + gf_time,
                              max_iter=10000, method="smooth_max")
        _try_update(a_gf)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LP spreading, peak-aware gradients,
    convolution equalization, multiscale refinement, and Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 896, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))
    lengths.sort()

    print(f"Target lengths: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Multiscale optimization from parent
    if parent is not None and time.monotonic() < deadline - 200:
        print("\n--- Phase 1: Multiscale from parent ---")
        ms_budget = min(deadline - time.monotonic() - 400, 250.0)
        a_ms = _multiscale_optimize(parent, parent.size, t_start + budget_s - 400)
        sc_ms = _score(a_ms)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()
            print(f"  Multiscale improved: {sc_ms:.10f}")

    # Phase 2: Try different lengths with full pipeline
    # Allocate time: more to lengths near the parent size
    n_base = parent.size if parent is not None else 768
    weights = []
    for ln in lengths:
        w = 1.0 / (1.0 + 0.1 * abs(ln - n_base))
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    # Reserve time for final polish
    reserved = 250.0
    available = max(100.0, deadline - time.monotonic() - reserved)

    for i, n_t in enumerate(lengths):
        if time.monotonic() > deadline - reserved:
            break

        phase_budget = available * weights[i]
        phase_budget = max(30.0, min(phase_budget, deadline - time.monotonic() - reserved))

        if phase_budget < 25:
            continue

        print(f"\n--- Phase 2: Length {n_t}, budget={phase_budget:.0f}s ---")

        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_linear_profile(n_t, 1.5))
        inits.append(_make_u_profile(n_t, 2.0))
        inits.append(_make_two_hump(n_t, 0.25, 0.15))
        inits.append(_make_beta_profile(n_t, 0.6, 0.6))

        # Pick best init
        best_init = inits[0]
        best_init_sc = np.inf
        for ini in inits:
            sc_i = _score(ini)
            if sc_i < best_init_sc:
                best_init_sc = sc_i
                best_init = ini

        a_res, sc_res = _full_pipeline(best_init, n_t, deadline - reserved)
        if sc_res < best_score:
            best_score = sc_res
            best_seq = a_res.copy()
            print(f"  ** New best at n={n_t}: {sc_res:.10f}")

    # Phase 3: Diverse restarts at best length
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        print(f"\n--- Phase 3: Diverse restarts at n={n_best} ---")

        restarts = []
        # Perturbed best
        for _ in range(4):
            a_p = best_seq * (1.0 + 0.1 * rng.normal(size=n_best))
            a_p = np.maximum(a_p, 0.0)
            a_p = a_p / a_p.sum()
            restarts.append(a_p)
        # Different profiles
        restarts.append(_make_u_profile(n_best, 3.0))
        restarts.append(_make_u_profile(n_best, 5.0))
        restarts.append(_make_two_hump(n_best, 0.35, 0.1))
        restarts.append(_make_beta_profile(n_best, 0.5, 0.5))
        # Random structured
        smooth_kernel = np.exp(-0.5 * (np.arange(30) - 15) ** 2 / 20.0)
        for _ in range(3):
            raw = np.maximum(rng.normal(size=n_best), 0)
            # Smooth it
            N_f = _next_pow2(2 * n_best - 1)
            fr = np.fft.rfft(raw, N_f)
            fk = np.fft.rfft(smooth_kernel, N_f)
            raw_s = np.fft.irfft(fr * fk, N_f)[:n_best]
            raw_s = np.maximum(raw_s, 0)
            if raw_s.sum() > 1e-10:
                restarts.append(raw_s / raw_s.sum())

        # Try each restart with reduced pipeline
        for a0 in restarts:
            if time.monotonic() > deadline - 60:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            sub_budget = min(deadline - time.monotonic() - 60, 40.0)
            if sub_budget < 15:
                break

            # Quick LP + gradient
            a_r = _lp_minimize(a0, time.monotonic() + sub_budget * 0.5,
                             max_iter=5000, inflate_start=1.002, blend_start=0.25)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart improved: {sc_r:.10f}")

            if time.monotonic() < deadline - 30:
                a_r2 = _gradient_polish(best_seq, time.monotonic() + sub_budget * 0.3,
                                       max_iter=5000, method="peak_aware")
                sc_r2 = _score(a_r2)
                if sc_r2 < best_score:
                    best_score = sc_r2
                    best_seq = a_r2.copy()

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n--- Phase 4: Final polish ---")
        final_budget = deadline - time.monotonic() - 15.0

        # LP with very tight inflation
        if final_budget > 30:
            t1 = min(final_budget * 0.35, 120.0)
            a_f = _lp_minimize(best_seq, time.monotonic() + t1,
                             max_iter=20000, inflate_start=1.0008, blend_start=0.2)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  Final LP: {sc_f:.10f}")

        # Peak-aware gradient
        if final_budget > 25 and time.monotonic() < deadline - 20:
            t2 = min(final_budget * 0.25, 80.0)
            a_g = _gradient_polish(best_seq, time.monotonic() + t2,
                                  max_iter=15000, method="peak_aware")
            sc_g = _score(a_g)
            if sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  Final PeakGrad: {sc_g:.10f}")

        # Conv equalize
        if final_budget > 20 and time.monotonic() < deadline - 15:
            t3 = min(final_budget * 0.15, 50.0)
            a_ce = _conv_equalize(best_seq, time.monotonic() + t3, max_iter=5000)
            sc_ce = _score(a_ce)
            if sc_ce < best_score:
                best_score = sc_ce
                best_seq = a_ce.copy()
            print(f"  Final ConvEq: {sc_ce:.10f}")

        # Adam polish
        if final_budget > 15 and time.monotonic() < deadline - 12:
            t4 = min(final_budget * 0.2, 60.0)
            a_ad = _adam_logit_polish(best_seq, time.monotonic() + t4,
                                     max_iter=10000, lr=0.03)
            sc_ad = _score(a_ad)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Final Adam: {sc_ad:.10f}")

        # One more LP with ultra-tight params
        if time.monotonic() < deadline - 10:
            t5 = min(final_budget * 0.1, 30.0)
            a_lp3 = _lp_minimize(best_seq, time.monotonic() + t5,
                                max_iter=5000, inflate_start=1.0005, blend_start=0.15)
            sc_lp3 = _score(a_lp3)
            if sc_lp3 < best_score:
                best_score = sc_lp3
                best_seq = a_lp3.copy()
            print(f"  Final LP2: {sc_lp3:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
