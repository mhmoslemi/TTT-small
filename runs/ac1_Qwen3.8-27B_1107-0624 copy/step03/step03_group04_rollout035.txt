Optimized step function coefficient search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses KKT-motivated pressure equalization, aggressive LP refinement, and gradient polish.
Key improvements: principled pressure equalization via KKT conditions, adaptive LP scheduling,
and focused time allocation on the best candidates.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _compute_pressure(a, c, top_idx):
    """Compute pressure vector: pressure[i] = sum_{k in top} c[k] * a[k-i] (valid idx)."""
    n = len(a)
    pressure = np.zeros(n)
    for k in top_idx:
        w = c[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_start = k - hi
        j_end = k - lo
        pressure[lo:hi + 1] += w * a[j_start:j_end + 1][::-1]
    return pressure

def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_step(a, beta, lr):
    """One gradient step using smooth max."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_val(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    vp = 2.0 * h
    dot = float(np.dot(a, vp))
    gz = a * (vp - dot)
    gz -= float(np.mean(gz))
    return c, w, gz

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = np.log(a).copy()

    z = best_z.copy()
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.93)

            if len(top5) > 1:

                if peak_gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif peak_gap > beta * 8.0:
                    beta = min(0.3 * c_max_new, beta * 1.5)

        # Restart if stuck
        if no_imp > max(4000, 6 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise_scale = 0.2
            z_restart = best_z + noise_scale * rng.standard_normal(n)
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _lp_refine_aggressive(a, deadline, max_iter=50000):
    """Aggressive LP refinement with adaptive inflation and blending."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    inflate = 1.0005
    blend = 0.4
    no_imp = 0
    consecutive_lp_fail = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_spread(a, M_lp)

        if g is None:
            consecutive_lp_fail += 1
            if consecutive_lp_fail > 5:
                break
            inflate = min(1.01, inflate * 1.2)
            continue

        consecutive_lp_fail = 0
        sum_g = float(g.sum())

        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            inflate = min(1.005, inflate * 1.1)
            blend = max(0.1, blend * 0.97)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Progressive: slightly increase blend for more aggressive moves
            blend = min(0.6, blend + 0.005)
            inflate = max(1.0003, inflate * 0.98)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend = max(0.15, blend * 0.95)
                inflate = min(1.003, inflate * 0.99)
            if no_imp > 200:
                a = best_a.copy()
                blend = 0.4
                inflate = 1.0005
                no_imp = 0

    return best_a


def _fine_grit_optimize(a, deadline, max_iter=20000):
    """Very fine gradient optimization for final polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.005 * c_max)
    lr = 0.005

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    z = np.log(a).copy()
    z -= np.mean(z)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-9 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.0005, lr * 0.95)
            if no_imp > 3000:
                break

        if it % 500 == 0:
            beta = max(1e-12, beta * 0.7)

    return best_a


def _zero_threshold(a, n_zeros=10):
    """Set the n_zeros smallest coordinates to exactly zero and renormalize."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    if n_zeros >= n:
        return a
    # Find coordinates to zero out
    order = np.argsort(a)
    zero_idx = order[:n_zeros]
    a[zero_idx] = 0.0
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def _animal_optimize(a, deadline):
    """
    Combined optimization: alternating pressure equalization, LP, and gradient steps.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    def try_update(candidate):
        nonlocal best_a, best_score
        s = _score(candidate)
        if s < best_score - 1e-14:
            best_score = s
            best_a = candidate.copy()
            return True
        return False

    # Round-based optimization
    round_size = max(20, (2 * n) // 10)
    it = 0
    total_iters = 30000

    while it < total_iters and time.monotonic() < deadline:
        phase = (it // round_size) % 3

        if phase == 0:
            # Pressure equalization
            sub_dl = min(deadline, time.monotonic() + 30.0)
            a_new = _kkt_pressure_optimize(a, sub_dl, n_top_frac=0.05, alpha_init=0.5, max_iter=200)
            if try_update(a_new):
                a = a_new
        elif phase == 1:
            # LP step
            c = _fft_convolve(a)
            M = float(c.max())
            M_lp = M * 1.001
            g = _lp_spread(a, M_lp)
            if g is not None:
                sum_g = float(g.sum())
                if sum_g > 1.0 + 1e-6:
                    g = g / sum_g
                    a_new = 0.6 * a + 0.4 * g
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        if try_update(a_new):
                            a = a_new
        else:
            # Gradient step
            sub_dl = min(deadline, time.monotonic() + 20.0)
            a_new = _gradient_optimize(a, sub_dl, max_iter=2000, lr=0.03)
            if try_update(a_new):
                a = a_new

        it += round_size

    return best_a


def _multi_length_search(parent, deadline, rng):
    """Try optimization at multiple lengths."""
    results = []

    # Try parent length and a couple others
    n_parent = parent.size
    lengths_to_try = [n_parent]
    for ln in [512, 1024, 1536]:
        if ln != n_parent and ln <= 2048:
            lengths_to_try.append(ln)

    # Sort by expected value (parent first)
    lengths_to_try.sort(key=lambda x: abs(x - n_parent))

    # Allocate time: more time to parent length
    time_per_length = max(30.0, (deadline - time.monotonic()) / len(lengths_to_try))

    for n_t in lengths_to_try:
        if time.monotonic() > deadline:
            break

        if n_t == n_parent:
            a0 = parent.copy()
        else:
            a0 = _interpolate_profile(parent, n_t)

        sub_deadline = min(deadline, time.monotonic() + time_per_length)
        print(f"  Optimizing n={n_t}, budget={sub_deadline - time.monotonic():.0f}s", flush=True)

        # Phase 1: Animal optimization (combined)
        a1 = _animal_optimize(a0, sub_deadline - 60.0 if sub_deadline - time.monotonic() > 120 else sub_deadline - 20.0)
        sc1 = _score(a1)
        print(f"    Animal: {sc1:.10f}", flush=True)

        # Phase 2: LP refinement
        if time.monotonic() < sub_deadline - 30:
            sub_dl2 = min(sub_deadline - 20, time.monotonic() + 60.0)
            a2 = _lp_refine_aggressive(a1, sub_dl2, max_iter=20000)
            sc2 = _score(a2)
            print(f"    LP: {sc2:.10f}", flush=True)
            if sc2 < sc1:
                a1 = a2
                sc1 = sc2

        # Phase 3: Fine polish
        if time.monotonic() < sub_deadline - 10:
            sub_dl3 = sub_deadline - 5
            a3 = _fine_grit_optimize(a1, sub_dl3, max_iter=10000)
            sc3 = _score(a3)
            print(f"    Fine: {sc3:.10f}", flush=True)
            if sc3 < sc1:
                a1 = a3
                sc1 = sc3

        results.append((sc1, n_t, a1))

    results.sort(key=lambda x: x[0])
    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Load parent sequence
    2. Apply KKT pressure equalization for quick improvement
    3. Combined animal optimization (pressure + LP + gradient)
    4. Aggressive LP refinement
    5. Fine gradient polish
    6. Try alternative lengths
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception as e:
        print(f"Parent load failed: {e}", flush=True)

    if parent is None:
        # Fallback: create a reasonable initial sequence
        n_fb = 512
        parent = np.ones(n_fb) / n_fb
        sc = _score(parent)
        best_score = sc
        best_seq = parent.copy()
        print(f"Fallback parent: score={sc:.10f}, n={parent.size}", flush=True)

    n = parent.size

    # Phase 1: Quick KKT pressure equalization
    print(f"\nPhase 1: KKT pressure equalization", flush=True)
    p1_deadline = min(deadline, t_start + 120.0)
    a_p1 = _kkt_pressure_optimize(parent, p1_deadline, n_top_frac=0.03, alpha_init=0.6, max_iter=5000)
    sc_p1 = _score(a_p1)
    print(f"  Result: {sc_p1:.10f} (improved by {best_score - sc_p1:.10f})", flush=True)
    if sc_p1 < best_score:
        best_score = sc_p1
        best_seq = a_p1.copy()

    # Phase 2: Animal optimization (combined pressure + LP + gradient)
    print(f"\nPhase 2: Animal optimization (deadline in {deadline - time.monotonic():.0f}s)", flush=True)
    p2_deadline = min(deadline - 120.0, t_start + 900.0)
    a_p2 = _animal_optimize(a_p1, p2_deadline)
    sc_p2 = _score(a_p2)
    print(f"  Result: {sc_p2:.10f}", flush=True)
    if sc_p2 < best_score:
        best_score = sc_p2
        best_seq = a_p2.copy()

    # Phase 3: Aggressive LP refinement
    print(f"\nPhase 3: LP refinement", flush=True)
    p3_deadline = min(deadline - 60.0, t_start + 1400.0)
    if p3_deadline - time.monotonic() > 30:
        a_p3 = _lp_refine_aggressive(best_seq, p3_deadline, max_iter=50000)
        sc_p3 = _score(a_p3)
        print(f"  Result: {sc_p3:.10f}", flush=True)
        if sc_p3 < best_score:
            best_score = sc_p3
            best_seq = a_p3.copy()

    # Phase 4: Fine gradient polish
    print(f"\nPhase 4: Fine gradient polish", flush=True)
    p4_deadline = min(deadline - 30.0, t_start + 1650.0)
    if p4_deadline - time.monotonic() > 20:
        a_p4 = _fine_grit_optimize(best_seq, p4_deadline, max_iter=30000)
        sc_p4 = _score(a_p4)
        print(f"  Result: {sc_p4:.10f}", flush=True)
        if sc_p4 < best_score:
            best_score = sc_p4
            best_seq = a_p4.copy()

    # Phase 5: Try zero-thresholding (set small coords to 0)
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 5: Zero-thresholding experiments", flush=True)
        for n_z in [5, 10, 20, 50]:
            if time.monotonic() > deadline - 20:
                break
            a_z = _zero_threshold(best_seq, n_z)
            # Quick gradient polish after zeroing
            sub_dl = min(deadline - 15, time.monotonic() + 30.0)
            a_zr = _gradient_optimize(a_z, sub_dl, max_iter=5000, lr=0.02)
            sc_z = _score(a_zr)
            print(f"  n_zeros={n_z}: {sc_z:.10f}", flush=True)
            if sc_z < best_score:
                best_score = sc_z
                best_seq = a_zr.copy()

    # Phase 6: Try alternative length
    if time.monotonic() < deadline - 60 and n != 1024:
        n_alt = 1024 if n == 768 else 768
        print(f"\nPhase 6: Try n={n_alt}", flush=True)
        a_alt = _interpolate_profile(best_seq, n_alt)
        sub_dl = deadline - 20
        a_alt_opt = _kkt_pressure_optimize(a_alt, min(sub_dl, time.monotonic() + 60.0),
                                            n_top_frac=0.03, alpha_init=0.5, max_iter=3000)
        sc_alt = _score(a_alt_opt)
        print(f"  n={n_alt} pressure: {sc_alt:.10f}", flush=True)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt_opt.copy()
            # Further optimize
            if time.monotonic() < deadline - 30:
                a_alt2 = _gradient_optimize(best_seq, deadline - 20, max_iter=10000, lr=0.03)
                sc_alt2 = _score(a_alt2)
                print(f"  n={n_alt} gradient: {sc_alt2:.10f}", flush=True)
                if sc_alt2 < best_score:
                    best_score = sc_alt2
                    best_seq = a_alt2.copy()

    # Phase 7: Final KKT + gradient cycle
    if time.monotonic() < deadline - 15:
        print(f"\nPhase 7: Final polish", flush=True)
        sub_dl = deadline - 10
        a_final = _kkt_pressure_optimize(best_seq, sub_dl, n_top_frac=0.02, alpha_init=0.3, max_iter=2000)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        if time.monotonic() < deadline - 5:
            a_final2 = _fine_grit_optimize(best_seq, deadline - 5, max_iter=5000)
            sc_final2 = _score(a_final2)
            if sc_final2 < best_score:
                best_score = sc_final2
                best_seq = a_final2.copy()

    # Final validation
    if best_seq is None:
        best_seq = parent.copy()
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(512) * sqrt(2 * 512) / 512  # fallback

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence score: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
