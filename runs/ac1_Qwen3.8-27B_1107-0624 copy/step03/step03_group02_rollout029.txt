def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    mval = float(c.max())

    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg

def _peeling_subgradient(a0, deadline, max_iter=20000, top_k=5, verbose=False):

def _lp_spread(a, M):
    n = len(a)
    if n > 2048:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_cosine_taper(n, beta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.5 * np.cos(np.pi * (2.0 * t - 1.0) * beta)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinc_like(n, bandwidth=1.0):
    t = np.linspace(0.0, 1.0, n)
    x = (t - 0.5) * bandwidth
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.sinc(x)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_shaped(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2

def _make_parabolic(n, alpha=1.0):

    a = np.maximum(a, 1e-12)

        per_candidate_time = alloc / 2.0
        for i in range(min(2, len(scored))):
            if time.monotonic() > d_n - 10:
                break
            sc_i, c_i = scored[i]
            d_ci = min(d_n - 10, time.monotonic() + per_candidate_time)
            label = f"n{n}-c{i}"
            
            a_best, s_best = _full_optimize(c_i, d_ci, label)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"  NEW BEST: {best_score:.10f} at n={n}")

    # Final polish on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish (best={best_score:.10f}, n={best_seq.size}) ===")
        
        # Try a different length if it might help
        best_n = best_seq.size
        if best_n <= 1024:
            # Try upsampling to 1536
            up = _interpolate_profile(best_seq, 1536)
            d_up = min(deadline - 10, time.monotonic() + 120)
            a_up, s_up = _full_optimize(up, d_up, "upsample")
            if s_up < best_score:
                best_score = s_up
                best_seq = a_up.copy()
                print(f"  Upsample improvement: {best_score:.10f}")
        
        # Final FISTA on the best
        if time.monotonic() < deadline - 5:
            a_f, s_f = _fista_smooth_max(best_seq, deadline - 5, 
                                         beta_init=max(1e-6, 0.005 * _fft_convolve(best_seq).max()),
                                         beta_decay=0.90, beta_min=1e-10)
            if s_f < best_score:
                best_score = s_f
                best_seq = a_f.copy()
            print(f"  Final: {best_score:.10f}")
        
        # Ultra-fine coordinate descent
        if time.monotonic() < deadline - 3:
            a_cd, s_cd = _coordinate_descent(best_seq, deadline - 3, max_iter=10000)
            if s_cd < best_score:
                best_score = s_cd
                best_seq = a_cd.copy()
            print(f"  After coord desc: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_cosine_taper(512, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_cosine_taper(256, 1.0)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
