"""
LP-guided optimization with subgradient mirror descent for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses LP-based direction finding, true subgradient
updates, and multi-scale search.
"""
import numpy as np
import time
import cvxpy as cp


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def cross_convolve_fft(f, g):
    """Full linear cross-convolution f*g using FFT."""
    n = len(f)
    m = len(g)
    N = next_pow2(n + m - 1)
    ff = np.fft.rfft(f, N)
    gf = np.fft.rfft(g, N)
    c = np.fft.irfft(ff * gf, N)[:n + m - 1]
    return c


def subgradient_of_max_conv(a, c, top_k=1):
    """
    Compute subgradient of max_k (a*a)[k] at the current point.
    Returns a vector of the same length as a.
    """
    n = len(a)
    # Find top-k maximizing indices
    if top_k >= len(c):
        top_k = len(c)
    peak_indices = np.argsort(c)[-top_k:]
    
    grad = np.zeros(n)
    for k in peak_indices:
        # d/d a_j [(a*a)[k]] = 2 * a[k-j] if 0 <= k-j < n, else 0
        # So grad[j] += 2 * a[k-j] for valid j
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # grad[j] = 2 * a[k-j] for j in [j_min, j_max]
            # k-j ranges from k-j_max to k-j_min
            grad[j_min:j_max+1] += 2.0 * a[k - j_max:k - j_min + 1]
    
    return grad / top_k


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_subgrad(a0, deadline, rng, lr=None, top_k=1, 
                           use_logit=False, logit_lr=0.05, restart_period=500):
    """
    Mirror descent / projected gradient with true subgradient.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    if lr is None:
        lr = 1.0 / np.sqrt(n)
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    
    # For logit mode
    if use_logit:
        z = np.log(a0)
        z = z - z.mean()
        best_z = z.copy()
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.9, 0.999, 1e-8
        current_lr = logit_lr
    
    while time.monotonic() < deadline:
        # Compute subgradient
        grad = subgradient_of_max_conv(a0, c, top_k)
        
        if use_logit:
            # Mirror descent in logit space
            dot = float(np.dot(a0, grad))
            dz = a0 * (grad - dot)
            t_adam += 1
            m = b1 * m + (1.0 - b1) * dz
            v = b2 * v + (1.0 - b2) * (dz * dz)
            mh = m / (1.0 - b1 ** t_adam)
            vh = v / (1.0 - b2 ** t_adam)
            z = z - current_lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            s = a0.sum()
            if s < 1e-15:
                z = best_z.copy()
                a0 = best_a.copy()
                continue
            a0 = a0 / s
        else:
            # Projected gradient
            a_new = a0 - lr * grad
            a0 = simplex_project(a_new)
        
        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a0.copy()
            if use_logit:
                best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if use_logit and no_improve > 200:
            current_lr = max(0.005, current_lr * 0.95)
        
        if not use_logit and no_improve > 200:
            lr = max(0.001, lr * 0.95)
        
        # Restart
        if no_improve > restart_period:
            if use_logit:
                z = best_z + 0.1 * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z = z - z.mean()
                a0 = np.exp(z - z.max())
                a0 = a0 / a0.sum()
                m.fill(0.0)
                v.fill(0.0)
                t_adam = 0
                current_lr = logit_lr
            else:
                a0 = best_a + 0.01 * rng.normal(size=n)
                a0 = np.maximum(a0, 0)
                a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            no_improve = 0
    
    return best_a, best_score


def lp_direction(f, M):
    """
    Solve the LP to find the best direction g.
    Maximize sum(g) subject to f*g <= M, g >= 0.
    Returns the normalized g (sum = 1) or None if LP fails.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Constraint: cross-convolution f*g <= M at all lags
    # (f*g)[k] = sum_i f_i * g_{k-i} for valid i
    # This is a linear constraint in g for fixed f
    
    # Build the constraint matrix (sparse)
    # Row k: sum_i f_i * g_{k-i} <= M
    # For each k, the non-zero entries are at positions j = k-i where i is valid
    
    rows = []
    cols = []
    vals = []
    rhs = []
    
    for k in range(2 * n - 1):
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i  # index into g
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
        rhs.append(M)
    
    # Build sparse matrix
    A = cp.sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: maximize sum(g) subject to A @ g <= M, g >= 0
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= np.array(rhs)]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None
    
    if g.value is None:
        return None
    
    g_val = np.asarray(g.value, dtype=np.float64).ravel()
    g_val = np.maximum(g_val, 0.0)
    s = float(g_val.sum())
    if s < 1e-10:
        return None
    
    return g_val / s

def inverse_f_direction(f):

def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def logit_adam_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.05,
                        beta_decay=0.93, beta_min=1e-10, restart_period=800):
    """
    Logit-space Adam with soft-max approximation of max.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Soft max weights
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s = float(np.sum(ex))
        if s < 1e-300:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s
        
        # Gradient via cross-correlation
        # h[j] = sum_k w[k] * a[k-j] for valid k-j
        # = cross-correlation of w and a
        n_w = len(w)
        N = next_pow2(n + n_w - 1)
        fa = np.fft.rfft(a0, N)
        fw = np.fft.rfft(w, N)
        # h[j] = sum_k w[k] * a[k-j]
        # This is the cross-correlation <w, shift_j(a)>
        # = sum_k w[k] * a[k-j]
        # Let's compute it as: reverse a, convolve with w
        # (w * a_rev)[j + n - 1] = sum_k w[k] * a_rev[j+n-1-k] = sum_k w[k] * a[n-1-(j+n-1-k)] = sum_k w[k] * a[k-j]
        a_rev = a0[::-1]
        conv = np.fft.irfft(fw * np.fft.rfft(a_rev, N), N)
        h = conv[n-1:2*n-1]
        
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)
        
        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s
        
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        if no_improve > 300:
            current_lr = max(0.003, current_lr * 0.97)
        
        if no_improve > restart_period:
            z = best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.06 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0
    
    return best_a, best_score
