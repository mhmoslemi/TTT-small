import time

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def subgradient_max_conv(c, a, top_k=1):

    else:

        w = np.zeros(len(c))
        w[idx] = 1.0 / top_k

    # Then a[k-j] = u[n-1-k+j]

def soft_max_value_and_weights(c, beta):

def fista_optimize(a0, deadline, rng, lr_init=0.01, max_iter=100000):

    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) for minimizing

    subject to sum(a) = 1, a >= 0.

    The proximal step handles the simplex constraint.

    best_score = score

    y = a0.copy()  # FISTA momentum variable

        # Compute gradient at y

        val, w = soft_max_value_and_weights(c_y, 0.01 * max_c * (0.95 ** (it // 100)))

        fw = np.fft.rfft(w, N)

        y = y / y.sum()
        t_fista = t_new

            y = best_a.copy()
            x = best_a.copy()
            break

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size

    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 0.0)
    s = result.sum()
    return result / s if s > 1e-10 else np.ones(n_new) / n_new


def make_tukey_window(n, alpha=0.5):
    t = np.linspace(0.0, 1.0, n)
    w = np.ones(n)
    m = np.minimum(2 * t / alpha, np.minimum(2.0 - 2 * t / alpha, 1.0))
    w[m < 1.0] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[m < 1.0] / alpha - 1.0)))
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def make_hamming_window(n):
    t = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * t / max(1, n - 1))
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def make_hann_window(n):
    t = np.arange(n)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * t / max(1, n - 1))
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def make_gaussian_window(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    w = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def make_raised_cosine(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    w = 0.5 + 0.5 * np.cos(np.pi * (2.0 * t - 1.0)) ** p
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def make_linear_ramp(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    w = 1.0 + eta * (1.0 - 2.0 * t)
    w = np.maximum(w, 1e-8)
    return w / w.sum()


def make_flat_taper(n, taper_frac=0.3):
    w = np.ones(n)
    ntaper = int(n * taper_frac / 2)
    if ntaper > 0:
        ramp = np.linspace(0.1, 1.0, ntaper)
        w[:ntaper] = ramp
        w[-ntaper:] = ramp[::-1]
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def soft_max_adam_optimize(a0, deadline, rng, lr=0.04, beta_init_frac=0.08,
                           beta_decay=0.95, beta_min=1e-10, top_k=1):
    """Logit-space Adam with adaptive soft-max."""
    n = a0.size
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())

    z = np.log(a0)
    z = z - z.mean()

    beta = beta_init_frac * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_score = 2.0 * n * max_c
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        mval, w = soft_max_grad(c, beta)

        if top_k > 1 and len(c) > top_k:
            idx = np.argpartition(c, -top_k)[-top_k:]
            w2 = np.zeros_like(c)
            w2[idx] = w[idx]
            s = w2.sum()
            if s > 0:
                w = w2 / s

        grad = gradient_from_weights(w, a0)
        dot = float(np.dot(a0, grad))
        dz = a0 * (grad - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps_adam)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            a0 = best_a.copy()
            z = np.log(a0)
            z = z - z.mean()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 500:
            current_lr = max(0.003, current_lr * 0.95)

        if no_improve > 3000:
            noise = 0.1 * rng.normal(size=n)
            z = z + noise
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            beta = max(1e-6, 0.06 * c.max())
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0

    return best_a, best_score


def polyak_polish(a0, deadline, rng, n_iters_max=500000):
    """
    Exact subgradient method with Polyak step size for final polish.
    Uses running average of convolution for stability.
    """
    n = a0.size
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c_avg = full_convolve_fft(a0).copy()
    best_a = a0.copy()
    best_max_c = float(c_avg.max())
    best_score = 2.0 * n * best_max_c

    lower_bound = 1.0 / (2.0 * n - 1.0)
    alpha_smooth = 0.05
    it = 0

    while time.monotonic() < deadline and it < n_iters_max:
        it += 1

        c_cur = full_convolve_fft(a0)
        c_avg = (1.0 - alpha_smooth) * c_avg + alpha_smooth * c_cur

        # Find the peak in the averaged convolution
        peak_idx = int(np.argmax(c_avg))
        w = np.zeros(len(c_avg))
        w[peak_idx] = 1.0

        grad = gradient_from_weights(w, a0)
        grad_tan = grad - float(np.dot(a0, grad))
        gnorm_sq = float(np.dot(grad_tan, grad_tan))

        if gnorm_sq < 1e-30:
            break

        current_max = float(c_avg.max())
        gap = current_max - lower_bound
        if gap < 1e-15:
            break

        eta = gap / gnorm_sq
        eta = min(eta, 1.0 / (np.sqrt(it + 1.0) + 1.0))
        eta = max(eta, 1e-12)

        a_new = a0 - eta * grad_tan
        a_new = project_simplex(a_new)

        c_new = full_convolve_fft(a_new)
        max_new = float(c_new.max())

        if max_new < best_max_c - 1e-15:
            best_max_c = max_new
            best_a = a_new.copy()
            best_score = 2.0 * n * best_max_c

        a0 = a_new

    return best_a, best_score


def exact_subgrad_adam(a0, deadline, rng, top_k=3, lr=0.02):
    """
    Subgradient method using exact top-k peak identification (no soft-max).
    Uses Adam for step size adaptation.
    """
    n = a0.size
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    b1, b2, eps_adam = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_score = 2.0 * n * full_convolve_fft(a0).max()
    no_improve = 0
    current_lr = lr

    # Running average for peak identification
    c_avg = full_convolve_fft(a0).copy()
    alpha = 0.03

    while time.monotonic() < deadline:
        c_cur = full_convolve_fft(a0)
        c_avg = (1.0 - alpha) * c_avg + alpha * c_cur

        # Top-k peaks from averaged convolution
        if top_k >= len(c_avg):
            idx = np.arange(len(c_avg))
        else:
            idx = np.argpartition(c_avg, -top_k)[-top_k:]
        w = np.zeros(len(c_avg))
        w[idx] = 1.0 / top_k

        grad = gradient_from_weights(w, a0)
        dot = float(np.dot(a0, grad))
        dz = a0 * (grad - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps_adam)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 200 == 0:
            current_lr = max(0.002, current_lr * 0.98)

        if no_improve > 5000:
            noise = 0.05 * rng.normal(size=n)
            z = np.log(np.maximum(best_a, 1e-15)) + noise
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0
            c_avg = full_convolve_fft(a0).copy()

    return best_a, best_score


def generate_diverse_profiles(n, rng):
    """Generate a diverse set of starting profiles."""
    profiles = []
    profiles.append(make_tukey_window(n, 0.3))
    profiles.append(make_tukey_window(n, 0.5))
    profiles.append(make_tukey_window(n, 0.7))
    profiles.append(make_hamming_window(n))
    profiles.append(make_hann_window(n))
    profiles.append(make_gaussian_window(n, 0.25))
    profiles.append(make_gaussian_window(n, 0.35))
    profiles.append(make_gaussian_window(n, 0.45))
    profiles.append(make_raised_cosine(n, 2.0))
    profiles.append(make_raised_cosine(n, 4.0))
    profiles.append(make_linear_ramp(n, 1.0))
    profiles.append(make_linear_ramp(n, 2.0))
    profiles.append(make_linear_ramp(n, 3.0))
    profiles.append(make_flat_taper(n, 0.2))
    profiles.append(make_flat_taper(n, 0.4))
    profiles.append(make_flat_taper(n, 0.6))
    # Random smooth profiles
    for _ in range(4):
        coeffs = rng.uniform(-0.3, 0.3, 5)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2.0 * np.pi * i * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining soft-max Adam, exact subgradient,
    Polyak polish, and diverse window-function initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to load previous best
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting optimization. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ========== Phase 1: Exploration at n=512 ==========
    n1 = 512
    phase1_end = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: n={n1} exploration ---")

    p1_base = interpolate_profile(init_seq, n1) if init_seq is not None else make_tukey_window(n1, 0.5)
    profiles_1 = generate_diverse_profiles(n1, rng)
    profiles_1.insert(0, p1_base)

    best_p1 = None
    best_s1 = np.inf

    # Score all profiles and pick the best few
    scored = []
    for p in profiles_1:
        sc = compute_score(p)
        scored.append((sc, p))
    scored.sort(key=lambda x: x[0])

    # Optimize the top 8 profiles
    for rank, (sc_init, p) in enumerate(scored[:8]):
        if time.monotonic() > phase1_end - 5:
            break
        per_profile_time = max(5.0, (phase1_end - time.monotonic()) / max(1, 8 - rank))
        dl = min(phase1_end, time.monotonic() + per_profile_time * 0.8)
        r, sc_r = soft_max_adam_optimize(p, dl, rng, lr=0.05,
                                          beta_init_frac=0.1, beta_decay=0.93)
        sc_f = compute_score(r)
        update(r, sc_f)
        if sc_f < best_s1:
            best_s1 = sc_f
            best_p1 = r.copy()
        print(f"  Profile rank {rank}: init={sc_init:.6f} -> {sc_f:.8f}")

    if best_p1 is not None:
        update(best_p1, best_s1)
    print(f"  Phase 1 best: {best_score:.10f}")

    # ========== Phase 2: Refinement at n=1024 ==========
    n2 = 1024
    phase2_end = min(deadline, t_start + 600.0)
    print(f"\n--- Phase 2: n={n2} refinement ---")

    p2_base = interpolate_profile(best_seq if best_seq is not None else p1_base, n2)
    sc2_init = compute_score(p2_base)
    update(p2_base, sc2_init)

    # Soft-max Adam at n=1024
    dl2a = min(phase2_end, time.monotonic() + 250.0)
    r2, sc2 = soft_max_adam_optimize(p2_base, dl2a, rng, lr=0.03,
                                       beta_init_frac=0.06, beta_decay=0.94)
    sc2_f = compute_score(r2)
    update(r2, sc2_f)
    print(f"  After soft-max Adam: {sc2_f:.10f}")

    # Exact subgradient Adam
    if time.monotonic() < phase2_end - 30:
        dl2b = min(phase2_end, time.monotonic() + 150.0)
        r2b, sc2b = exact_subgrad_adam(r2, dl2b, rng, top_k=3, lr=0.015)
        sc2b_f = compute_score(r2b)
        update(r2b, sc2b_f)
        print(f"  After exact subgrad Adam: {sc2b_f:.10f}")
        if sc2b_f < sc2_f:
            r2 = r2b

    # Polyak polish
    if time.monotonic() < phase2_end - 10:
        dl2c = min(phase2_end, time.monotonic() + 50.0)
        r2c, sc2c = polyak_polish(r2, dl2c, rng)
        sc2c_f = compute_score(r2c)
        update(r2c, sc2c_f)
        print(f"  After Polyak polish: {sc2c_f:.10f}")
        if sc2c_f < sc2_f:
            r2 = r2c

    # Try diverse starts at n=1024
    profiles_2 = generate_diverse_profiles(n2, rng)
    profiles_2.insert(0, p2_base)
    scored2 = []
    for p in profiles_2:
        sc = compute_score(p)
        scored2.append((sc, p))
    scored2.sort(key=lambda x: x[0])

    for rank, (sc_i, p) in enumerate(scored2[:4]):
        if time.monotonic() > phase2_end - 5:
            break
        if sc_i > best_score + 0.05:
            continue
        dl = min(phase2_end, time.monotonic() + 30.0)
        r, sc_r = soft_max_adam_optimize(p, dl, rng, lr=0.04,
                                          beta_init_frac=0.08, beta_decay=0.94)
        sc_f = compute_score(r)
        update(r, sc_f)
        if sc_f < best_score + 0.01:
            print(f"  Alt profile {rank}: {sc_f:.8f}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # ========== Phase 3: Large n = 2048 ==========
    n3 = 2048
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\n--- Phase 3: n={n3} large scale ---")

    p3_base = interpolate_profile(best_seq if best_seq is not None else r2, n3)
    sc3_init = compute_score(p3_base)
    update(p3_base, sc3_init)
    print(f"  Interpolated: {sc3_init:.10f}")

    # Long soft-max Adam
    dl3a = min(phase3_end, time.monotonic() + 400.0)
    r3, sc3 = soft_max_adam_optimize(p3_base, dl3a, rng, lr=0.02,
                                       beta_init_frac=0.04, beta_decay=0.95)
    sc3_f = compute_score(r3)
    update(r3, sc3_f)
    print(f"  After soft-max Adam: {sc3_f:.10f}")

    # Exact subgradient
    if time.monotonic() < phase3_end - 50:
        dl3b = min(phase3_end, time.monotonic() + 200.0)
        r3b, sc3b = exact_subgrad_adam(r3, dl3b, rng, top_k=5, lr=0.01)
        sc3b_f = compute_score(r3b)
        update(r3b, sc3b_f)
        print(f"  After exact subgrad: {sc3b_f:.10f}")
        if sc3b_f < sc3_f:
            r3 = r3b

    # Polyak polish at large scale
    if time.monotonic() < phase3_end - 20:
        dl3c = min(phase3_end, time.monotonic() + 80.0)
        r3c, sc3c = polyak_polish(r3, dl3c, rng)
        sc3c_f = compute_score(r3c)
        update(r3c, sc3c_f)
        print(f"  After Polyak: {sc3c_f:.10f}")
        if sc3c_f < sc3_f:
            r3 = r3c

    print(f"  Phase 3 best: {best_score:.10f}")

    # ========== Phase 4: Try n=4096 if time allows ==========
    n4 = 4096
    if time.monotonic() < phase3_end - 150 and best_score > 1.503:
        print(f"\n--- Phase 4: n={n4} ---")
        p4 = interpolate_profile(best_seq if best_seq is not None else r3, n4)
        sc4_i = compute_score(p4)
        update(p4, sc4_i)
        print(f"  Interpolated: {sc4_i:.10f}")

        dl4 = min(phase3_end, time.monotonic() + 120.0)
        r4, sc4 = soft_max_adam_optimize(p4, dl4, rng, lr=0.015,
                                           beta_init_frac=0.03, beta_decay=0.96)
        sc4_f = compute_score(r4)
        update(r4, sc4_f)
        print(f"  After Adam: {sc4_f:.10f}")

    # ========== Phase 5: Final polish at best size ==========
    print(f"\n--- Phase 5: Final polish ---")
    if best_seq is not None:
        np_best = best_seq.size
        # Multiple polish rounds
        for rnd, (method, lr_v) in enumerate([
            ("exact_k1", 0.01),
            ("exact_k3", 0.008),
            ("polyak", None),
            ("exact_k1", 0.005),
        ]):
            if time.monotonic() > deadline - 5:
                break
            dl = min(deadline, time.monotonic() + 80.0)
            if method == "polyak":
                r_p, sc_p = polyak_polish(best_seq, dl, rng)
            else:
                top_k = 1 if "k1" in method else 3
                p0 = best_seq + 0.0002 * rng.normal(size=np_best)
                p0 = np.maximum(p0, 1e-12)
                p0 = p0 / p0.sum()
                r_p, sc_p = exact_subgrad_adam(p0, dl, rng, top_k=top_k, lr=lr_v)
            sc_p_f = compute_score(r_p)
            update(r_p, sc_p_f)
            print(f"  Polish {rnd} ({method}): {sc_p_f:.10f}")

    # ========== Phase 6: Try nearby sizes ==========
    if best_seq is not None:
        n_curr = best_seq.size
        for n_try in [int(n_curr * 0.75), int(n_curr * 1.25)]:
            if n_try < 64 or time.monotonic() > deadline - 15:
                continue
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            update(p_try, sc_try)
            if sc_try < best_score + 0.005:
                dl = min(deadline, time.monotonic() + 30.0)
                r_t, _ = exact_subgrad_adam(p_try, dl, rng, top_k=3, lr=0.008)
                sc_t = compute_score(r_t)
                update(r_t, sc_t)
                print(f"  Size n={n_try}: {sc_t:.10f}")

    print(f"\n  Final best: n={best_seq.size if best_seq is not None else '?'}, score={best_score:.10f}")

    # ========== Validation and return ==========
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_tukey_window(1024, 0.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_tukey_window(512, 0.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
