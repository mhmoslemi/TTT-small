Now let me think about the coordinate descent more carefully. The key function is the line search for a single coordinate.

When we change a[i] from x_old to x, the convolution values change as:
- For k where 0 <= k-i < n: c_k changes by 2*(x-x_old)*a[k-i] (if k != 2i) or (x^2-x_old^2) (if k == 2i)
- For all other k: c_k unchanged

The new peak is max(unchanged_max, max of changed values).

To find the optimal x, we need to minimize:
f(x) = max(unchanged_max, max_{k in changed set} (base_k + slope_k * x + quad_k * x^2))

where base_k, slope_k, quad_k are computed from the current state.

Since f(x) is convex (max of affine and one quadratic, all with non-negative leading coefficients on x), we can find the minimum by:
1. Finding the subgradient at x=0: if it's >= 0, the minimum is at x=0
2. Otherwise, the minimum is at some x* > 0 where the subgradient contains 0

For step 2, since there's at most one quadratic term (at k=2i), and the rest are affine, the function f(x) is the max of:
- A constant (unchanged_max)
- n-1 affine functions (slope_k * x + base_k for k in changed set, k != 2i)
- One quadratic (x^2 + slope_2i * x + base_2i) [if 2i is in the changed set]

The minimum of the max of these occurs where either:
- The quadratic and one affine function intersect at the minimum, or
- The quadratic is above all others and we minimize the quadratic directly, or
- Two affine functions intersect at the minimum (and are above the quadratic)

This is a 1D problem that can be solved efficiently.

    x_old = a[i]

    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unchanged max
    unch_max = 0.0
    if k_lo > 0:
        unch_max = float(np.max(c[0:k_lo]))
    if k_hi < 2 * n - 2:
        v = float(np.max(c[k_hi+1:2*n-1]))
        unch_max = max(unch_max, v)

    # Quadratic part (k == 2i, if in range)

    # and slopes (coefficient of x)

    slopes = np.zeros(len(affected_ks))  # coefficient of x

            slopes[idx] = 2.0 * a[j]
            quads[idx] = 0.0

        vals = bases + slopes * x + quads * x * x
        peak = float(np.max(vals))
        if unch_max > peak:
            peak = unch_max
        return peak
    
    # Find minimum of f(x) for x >= 0
    # f is convex, so use golden section or subgradient

            best_x = x_mid

"""
Coordinate descent + LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Uses exact coordinate-wise line search exploiting the quadratic structure of
auto-convolution, combined with LP-based spreading and adaptive multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < -1e-10:
        c += -c.min()
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(n, a):
    """Build the matrix A for the LP: (a*g)_k = sum_j A[k,j] * g_j"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0"""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(n, a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new, dtype=np.float64) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new, dtype=np.float64) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _cd_line_search(c, a, i, n):
    """
    Exact line search for coordinate i of the auto-convolution peak.
    Returns (optimal_x, new_peak_value).
    """
    x_old = a[i]
    
    # Affected convolution indices: k where 0 <= k-i < n
    k_lo = i  # since i >= 0 and k-i >= 0 means k >= i; also k >= 0

    k_hi = min(2 * n - 2, k_hi)

    unch_max = 0.0
    if k_lo > 0:

    if k_hi < 2 * n - 2:
        v = float(np.max(c[k_hi + 1:]))
        if v > unch_max:

    affected = np.arange(k_lo, k_hi + 1)

    m = len(affected)
    base = np.empty(m)
    slope = np.empty(m)
    quad = np.zeros(m)

        j = j_vals[idx]
        if k == 2 * i:
            base[idx] = c[k] - x_old * x_old
            slope[idx] = 0.0
            quad[idx] = 1.0
        else:
            base[idx] = c[k] - 2.0 * x_old * a[j]
            slope[idx] = 2.0 * a[j]
            quad[idx] = 0.0
    
    def f(x):
        # f(x) = max(unch_max, base + slope*x + quad*x^2)
        vals = base + slope * x + quad * x * x
        p = float(np.max(vals))
        if unch_max > p:
            p = unch_max
        return p

    cands = [0.0, x_old]
    for mult in [0.05, 0.1, 0.25, 0.5, 0.75, 1.5, 2.0, 4.0]:

    best_f = f(x_old)
    for x in cands:
        val = f(x)
        if val < best_f:
            best_f = val
            best_x = x

    if best_x > 1e-15:

            best_f = fm
            best_x = x_mid
