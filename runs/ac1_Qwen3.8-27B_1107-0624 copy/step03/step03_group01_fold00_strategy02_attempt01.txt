    sum_a = np.sum(seq)
    # A simple normalization that keeps the objective finite and the vector valid
    if sum_a < 0.001 * n:
        # If sum is too small, we can ensemble flexibility but maintain the same
        norm_factor = 0.01 / sum_a if max(b_sequence) > 0 else 1.0
        seq_scaled = [x * norm_factor for x in seq]
        max_b_scaled = max(np.convolve(seq_scaled, seq_scaled))
        sum_a_scaled = np.sum(seq_scaled)
        # But careful: evaluator uses raw input sum, so we return the scaled version's objective
        if max_b_scaled == 0:
            return np.inf
        return 997.0 * n * max_b_scaled / (sum_a_scaled ** 2)
    else:
        return evaluate_sequence(seq)

def adaptive_optimization(initial_sequence, initial_beta, n, max_iterations):
    # Build a class of functions based on varying the shape in frequency domain
    # But note prevalence of increasing complexity: simpler is better for numerical stability
    def freq_convolve(x, n):
        a = np.exp(x) / (np.sum(np.exp(x)))  # Convert from log space to simplex in freq mode
        # But careful: constrained simplex is only valid if we maintain normalization
        time_limit_used = initial_beta * max_iterations / n
        # For simplicity, let the time limit be the product of beta and iterations
        time_budget = initial_beta * n * max_iterations
        if time_budget < 180*n:
            return np.inf
        return evaluate_sequence(a).item()

    def freq_objective(z):
        a = np.exp(z) / np.sum(np.exp(z))
        c = np.convolve(a, a)
        cm = max(c)
        s = np.sum(a)
        n_here = len(z)
        score_val = (2.0 * n_here * cm) / (s ** 2)
        
        # To protect against floating point monsters and avoid optimization pitfalls
        if ~(np.all(np.isfinite(a))) or s < 0.001 or cm > 1000000:
            return np.inf
        
        # Return the score and maintain the simplex representation
        return score_val

    def freq_constraint(z):
        if len(z) != n:
            return np.inf
        a = np.exp(z) / np.sum(np.exp(z))
        return np.sum(a) - 1.0  # Only valid if sum=1, which is common in stochastic optimization

    def optimize_from_z(z):
        # Optimization bounds and options with explicit simplex enforcement
        bounds_list = []
        for i in range(n):
            # Ensure extremely small positive values: this is for numerical consistency in the simplex
            if i < 10:  # Restrict early indices to avoid instability
                bounds_list.append([-10.0, 0.0])
            else:
                bounds_list.append([-12.0, 12.0])
                
        # Define a more adaptive preference for storing vectors in a simplex and not clamping them
        # Use L-BFGS-B with initial point z, bounds, and constraints
        res = minimize(
            freq_objective,
            z,
            method='L-BFGS-B',
            bounds=bounds_list,
            constraints={'type': 'ineq', 'fun': lambda z: 1.0 - freq_constraint(z)} if n > 7 else {},
            options={
                'maxiter': max_iterations,
                'disp': True,
                'ftol': 1e-4,
                'eps': 1e-3,
                'maxcor': 20,
                'maxls': 50
            }
        )
        if res.success and np.isfinite(res.x).all():
            new_vector = [max(x, 1e-12) for x in res.x]
            new_vector = new_vector / sum(new_vector)  # Enforce sum=1, explicit checks may be in evaluator anyway
            return new_vector
        else:
            return z  # No improvement, so return the same z

    # Begin the optimization loop with specified dimensions and initial simplex
    # Note: using a initial simplex of length n might be slow, so we use a two-stage optimization: coarse then fine
    # Returns a refined vector with approximately lower score
    refined_vectors = []
    for iter_attempt in range(max_iterations):
        if iter_attempt % 37 == 0:
            # Introduce a set of perturbations as a form of regional widening
            new_bounds = []
            for i, low in enumerate(bounds_list):
                new_low = low[0] - 1.0 if i % 2 == 0 else low[0]
                new_high = low[1] + 0.5 if i % 3 == 1 else low[1] + 0.8
                new_bounds.append([new_low, new_high])
        try:
            new_point = optimize_from_z(z)
            refined_vectors.append(new_point)
            if evaluate_sequence(new_point) < initial_beta + 1e-5:
                break
        except KeyboardInterrupt:
            print("Optimization interrupted manually. Returning best found.")
            break
    # End the outer optimizer loop and return the best found solution
    # But note: multiple runs might be needed, especially in higher dimensions
    # We try a fallback method if milestone dimensions are not met with improvement
    best_refined_vector = None
    for vec in refined_vectors:
        if best_refined_vector is None or evaluate_sequence(vec) < evaluate_sequence(best_refined_vector):
            best_refined_vector = vec
            
    if best_refined_vector is None:
        # Set a fallback optimization path if no improvement and no captured vector found
        z = [0.0] * n
        for i in range(n):
            if i % 5 == 0:
                continue
            if i % 100 == 0:
                z[i] = (i+1) * initial_beta * 0.1
            else:
                z[i] = initial_beta * (i % 7 + 1) / (n * 0.5)
                
        def score(z):
            if ~np.all(np.isfinite(z)) or any(x==0 for x in z):
                return np.inf
            res = minimize(lambda z: np.log(2.0 * n * freq_objective(z)),
                           z,
                           method='trust-constr',
                           bounds=[(-12.0, 12.0) for _ in range(n)],
                           options={'maxiter': 200})

        if res.success:
            best = minimize(freq_objective, res.x[:n], bounds=[(-12.0, 12.0),]*n, options={'maxiter': 5})
            return best.x.tolist()
        else:
            return refined_vectors[0][:n].tolist()

    return best_refined_vector.tolist()
