```python
"""
Hybrid L-BFGS + Exact Subgradient + Frank-Wolfe optimization for minimizing
2*n*max(conv(p,p)) over probability vectors p (non-negative, sum=1).
Uses continuation on smoothing parameter, exact subgradient polish,
and active-set Newton refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(p):
    """Compute 2*n*max(conv(p,p)) for a probability vector p."""
    n = len(p)
    s = float(p.sum())
    if s < 1e-15:
        return np.inf, 0
    p_norm = p / s
    c = _full_convolve(p_norm, p_norm)
    max_c = float(c.max())
    return 2.0 * n * max_c, max_c

def _score_raw(a):
    """Score for raw (unnormalized) sequence a, matching evaluate_sequence."""
    n = len(a)
    a = np.maximum(0, np.minimum(1000.0, a))
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_max_and_grad(p, n):
    """
    Compute max of conv(p,p) and its index, plus the gradient of that max component.
    Returns: (max_val, max_idx, grad) where grad = 2 * A_k * p for k = max_idx.
    """
    c = _full_convolve(p, p)
    k = int(np.argmax(c))
    max_val = float(c[k])
    # Gradient of Q_k(p) = sum_{i+j=k} p[i]*p[j] w.r.t. p
    # (Q_k * p)[m] = p[k-m] if 0 <= k-m <= n-1
    grad = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for m in range(lo, hi + 1):
        grad[m] = 2.0 * p[k - m]
    return max_val, k, grad

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = int(np.max(ind[cond]))
    theta = float(cssv[k - 1] / k)
    return np.maximum(v - theta, 0.0)
```

```python
    top2_diff = 0.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        max_val, k, c = _convolve_max_info(p)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()

        # Track stability of argmax
        if k == last_k:
            stable_count += 1
        else:
            stable_count = 0
        last_k = k
        last_max_val = max_val

        # Compute top-2 gap
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) >= 2:
            top2_diff = float(c_sorted[0] - c_sorted[1])

        # Gradient at argmax
        grad = _grad_at_k(p, k, n)

        # FW direction: vertex with minimum gradient
        j = int(np.argmin(grad))

        # Multi-candidate line search
        base_alpha = 2.0 / (it + 2.0)
        candidates = [base_alpha]
        if it > 0:
            candidates.extend([min(1.0, base_alpha * 1.5), max(0.01, base_alpha * 0.5)])
        candidates.extend([0.05, 0.02, 0.01])

        best_a = candidates[0]
        best_f = cur_score
        for a in candidates:
            p_try = (1.0 - a) * p + a * (1.0 / n * np.ones(n) if False else
                                         np.where(np.arange(n) == j, 1.0, 0.0))
            f_try, _, _ = _convolve_max_info(p_try)
            if f_try < best_f:
                best_f = f_try
                best_a = a

        # Also try "away-step" if we have a previous vertex
        p = (1.0 - best_a) * p + best_a * np.where(np.arange(n) == j, 1.0, 0.0)

        # Periodic: try moving toward uniform to escape degenerate solutions
        if it > 0 and it % 5000 == 0 and top2_diff < 1e-12:
            p = 0.95 * p + 0.05 * (1.0 / n)

    return best_p, best_score


def _run_nesterov_subgrad(p_init, deadline, base_lr=0.5, max_iter=200000):
    """Nesterov-accelerated subgradient on the simplex with exact max."""
    n = len(p_init)
    p = p_init.copy()
    y = p_init.copy()
    best_p = p.copy()
    best_score, _, _ = _convolve_max_info(p)
    best_score = 2.0 * n * best_score

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        max_val, k, _ = _convolve_max_info(y)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = y.copy()

        grad = _grad_at_k(y, k, n)

        # Adam-like normalization on the subgradient
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        lr = base_lr / (1.0 + 0.01 * t)
        z = y - lr * mh / (np.sqrt(vh) + eps)
        p_candidate = _project_simplex(z)

        # Nesterov interpolation
        theta = 2.0 / (np.sqrt(t) + 1.0)
        p = (1.0 - theta) * p + theta * p_candidate
        y = p

        # Periodic: nudge toward interior to avoid boundary issues
        if it > 0 and it % 10000 == 0:
            y = 0.99 * y + 0.01 * (1.0 / n)

    return best_p, best_score


def _run_active_newton(p_init, deadline, max_iter=5000):
    """
    When argmax is stable, minimize Q_k(p) on simplex via projected gradient.
    Uses the structure that grad = 2 * A_k * p.
    """
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_score, _, _ = _convolve_max_info(p)
    best_score = 2.0 * n * best_score

    prev_k = -1
    stable = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        max_val, k, c = _convolve_max_info(p)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()

        if k == prev_k:
            stable += 1
        else:
            stable = 0
        prev_k = k

        # If stable, do projected gradient on Q_k
        if stable >= 5:
            grad = _grad_at_k(p, k, n)
            # Backtracking line search
            alpha = 1.0 / (it + 10)
            for _ in range(20):
                p_try = _project_simplex(p - alpha * grad)
                f_try, _, _ = _convolve_max_info(p_try)
                if f_try <= max_val * (1.0 + 1e-10):
                    p = p_try
                    break
                alpha *= 0.5
        else:
            # Not stable, use FW step
            grad = _grad_at_k(p, k, n)
            j = int(np.argmin(grad))
            alpha = 2.0 / (it + 2.0)
            s = np.zeros(n)
            s[j] = 1.0
            p = (1.0 - alpha) * p + alpha * s

    return best_p, best_score


def _make_tapered(n, alpha=2.0):
    """Create a tapered (raised-cosine-like) probability vector."""
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** alpha
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_poly(n, coeffs):
    """Create a polynomial-shaped probability vector."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0
    for i, c in enumerate(coeffs):
        a = a + c * t ** i
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS continuation -> Frank-Wolfe ->
    Nesterov subgradient -> Active Newton polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print("  NEW BEST: {:.10f} (n={}) [{}]".format(best_score, len(p), label))

    # --- Get parent / initial point ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print("Parent: n={}, score={:.10f}".format(n, _score_prob(parent)))
    else:
        n = 2048
        parent = _make_tapered(n, 2.0)
        print("No parent, using tapered profile n={}".format(n))

    sc_parent = _score_prob(parent)
    update_best(parent, sc_parent, "parent")

    # --- Phase 0: Quick parametric scan (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = [(sc_parent, parent.copy(), "parent")]

    for shape_fn in [_make_tapered,
                     lambda nn: _make_poly(nn, [0.5, -0.2]),
                     lambda nn: _make_poly(nn, [1.0, -0.5, 0.1]),
                     lambda nn: _make_poly(nn, [0.3, 0.2, -0.1])]:
        try:
            p = shape_fn(n)
            sc = _score_prob(p)
            candidates.append((sc, p, "param"))
            update_best(p, sc, "param")
        except Exception:
            pass

    candidates.sort(key=lambda x: x[0])
    print("Phase 0: best={:.10f}".format(best_score))

    # --- Phase 1: L-BFGS continuation (up to 400s) ---
    phase1_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase1_end:
        p_curr = candidates[0][1].copy()
        sc_curr = candidates[0][0]
        per_time = min(350.0, phase1_end - time.monotonic() - 5)
        p_best, sc_best = _run_lbfgs_continuation(p_curr, time.monotonic() + per_time, rng,
                                                   max_wall_time=per_time)
        update_best(p_best, sc_best, "lbfgs_cont")
        print("Phase 1: best={:.10f}".format(best_score))

    # --- Phase 2: Frank-Wolfe with exact max (up to 800s total) ---
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        per_time = min(350.0, phase2_end - time.monotonic() - 5)
        p_best, sc_best = _run_fw(best_seq, time.monotonic() + per_time, max_iter=100000)
        update_best(p_best, sc_best, "fw")
        print("Phase 2: best={:.10f}".format(best_score))

    # --- Phase 3: Nesterov subgradient (up to 1200s total) ---
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_time = min(350.0, phase3_end - time.monotonic() - 5)
        p_best, sc_best = _run_nesterov_subgrad(best_seq, time.monotonic() + per_time,
                                                base_lr=0.3, max_iter=200000)
        update_best(p_best, sc_best, "nesterov")

        # Second pass with smaller lr
        if time.monotonic() < phase3_end - 30:
            per_time2 = min(150.0, phase3_end - time.monotonic() - 5)
            p_best2, sc_best2 = _run_nesterov_subgrad(best_seq, time.monotonic() + per_time2,
                                                      base_lr=0.1, max_iter=100000)
            update_best(p_best2, sc_best2, "nesterov_fine")
        print("Phase 3: best={:.10f}".format(best_score))

    # --- Phase 4: Active Newton polish (up to 1550s total) ---
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_time = min(250.0, phase4_end - time.monotonic() - 5)
        p_best, sc_best = _run_active_newton(best_seq, time.monotonic() + per_time, max_iter=10000)
        update_best(p_best, sc_best, "newton")
        print("Phase 4: best={:.10f}".format(best_score))

    # --- Phase 5: Final FW + subgradient micro-polish (remaining time) ---
    if time.monotonic() < deadline - 20 and best_seq is not None:
        Phase5_end = min(deadline, t_start + 1750.0)
        if time.monotonic() < Phase5_end - 10:
            per_time = min(100.0, Phase5_end - time.monotonic() - 5)
            p_best, sc_best = _run_fw(best_seq, time.monotonic() + per_time, max_iter=50000)
            update_best(p_best, sc_best, "fw_final")

        if time.monotonic() < deadline - 10:
            per_time = min(80.0, deadline - time.monotonic() - 5)
            p_best, sc_best = _run_nesterov_subgrad(best_seq, time.monotonic() + per_time,
                                                    base_lr=0.05, max_iter=100000)
            update_best(p_best, sc_best, "subgrad_final")

    # --- Try a different length (2048 -> try 1536 or 2560) ---
    if time.monotonic() < deadline - 100 and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr == 2048:
            for alt_n in [1536, 2560]:
                if time.monotonic() > deadline - 60:
                    break
                # Interpolate to new length
                x0 = np.linspace(0, 1, n_curr)
                x1 = np.linspace(0, 1, alt_n)
                p_interp = np.interp(x1, x0, best_seq)
                p_interp = np.maximum(p_interp, 0.0)
                s = p_interp.sum()
                if s > 0:
                    p_interp = p_interp / s
                    sc_interp = _score_prob(p_interp)
                    if sc_interp < best_score:
                        update_best(p_interp, sc_interp, "interp_n{}".format(alt_n))
                        # Quick FW polish
                        if time.monotonic() < deadline - 40:
                            p_b, sc_b = _run_fw(p_interp, time.monotonic() + 30, max_iter=20000)
                            update_best(p_b, sc_b, "fw_n{}".format(alt_n))

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_tapered(256, 2.0)
            s = final_seq.sum()
        # Keep normalized (score is scale-invariant)
        sc_final = _score_prob(final_seq)
        print("FINAL: score={:.10f}, n={}".format(sc_final, len(final_seq)))

        # Validate with evaluate_sequence if available
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print("Evaluator score: {:.10f}".format(sc_eval))
        except Exception as ex:
            print("evaluate_sequence check failed: {}".format(ex))

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_tapered(256, 2.0)
    print("FALLBACK: n=256, score={:.10f}".format(_score_prob(fallback)))
    return [float(x) for x in fallback]
```
