"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 with non-negative sequences.

Key techniques:
1. Spectral shaping (Gerchberg-Saxton style) for global exploration
2. Adaptive gradient descent with smooth max in logit space
3. LP-based spreading refinement
4. Peak pressure redistribution
5. Coarse-to-fine multi-length search
6. Diverse initialization strategies

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_val(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, float(c.max()), c


def _spectral_shape(a, n_iter=200, alpha=0.3, deadline=None):
    """
    Gerchberg-Saxton style spectral shaping to flatten autocorrelation.
    Modifies amplitude spectrum to reduce peaks, keeps phase, projects to non-negative.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    L = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute current spectrum
        A = np.fft.rfft(a, L)
        P = A.conjugate() * A  # Power spectrum (real)
        
        # Compute current autocorrelation
        c = _fft_convolve(a)
        c_mean = 1.0 / (2 * n - 1)
        
        # Find peak positions (above mean)
        excess = c - c_mean
        peak_mask = excess > 0
        if not np.any(peak_mask):
            break
        
        # Compute "damage" vector: how each frequency contributes to peaks
        # c[k] = (1/L) * sum_m P[m] * exp(2πimk/L)
        # dc[k]/dP[m] = (1/L) * exp(2πimk/L)
        # We want to reduce c[k] where k is a peak, so reduce P[m] where it contributes positively
        peak_indices = np.where(peak_mask)[0]
        # Compute contribution of each frequency to peak excess
        # contrib[m] = sum_{k in peaks} P[m] * cos(2πmk/L) * excess[k]
        # Simplified: use the top few peaks
        n_peaks = min(10, len(peak_indices))
        top_peaks = peak_indices[np.argsort(excess[peak_indices])[::-1][:n_peaks]]
        
        # For each frequency, compute its weighted contribution to peaks
        phases = np.zeros(L // 2 + 1)
        for k in top_peaks:
            weights = excess[k]
            m_vals = np.arange(L // 2 + 1)
            phases += weights * np.cos(2.0 * np.pi * m_vals * k / L)
        
        # Scale: reduce frequencies that contribute positively to peaks
        phases = np.maximum(phases, 0)
        phases = phases / (phases.max() + 1e-15)
        
        # New amplitude: reduce based on contribution
        amp = np.abs(A)
        new_amp = amp * np.exp(-alpha * phases)
        
        # Keep phase, use new amplitude
        phase_A = A / (np.abs(A) + 1e-30)
        A_new = new_amp * phase_A
        
        # Inverse DFT
        a_new = np.real(np.fft.irfft(A_new, L))[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            alpha = min(1.0, alpha * 1.02)
        else:
            no_imp += 1
            alpha = max(0.05, alpha * 0.95)
            if no_imp > 50:
                break
        
        a = a_new
    
    return best_a


def _spectral_shape_v2(a, n_iter=300, deadline=None):
    """
    Improved spectral shaping: iteratively remove peak contributions from spectrum.
    Uses a "notch filter" approach targeting the autocorrelation peaks.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    L = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, L)
        c = _fft_convolve(a)
        c_target = 1.0 / (2 * n - 1)
        
        # Compute error: how much each autocorrelation bin exceeds target
        error = np.maximum(c - c_target, 0)
        if error.max() < 1e-8:
            break
        
        # The error in frequency domain
        E = np.fft.rfft(error, L)
        
        # Modify spectrum: subtract error contribution (scaled)
        # Current power: P = |A|^2
        # We want to reduce P where it contributes to peaks
        # Simple approach: scale A by exp(-beta * Re(conj(A) * E) / |A|^2)
        amp = np.abs(A) + 1e-15
        # Projection of error onto current signal at each frequency
        proj = np.real(A.conjugate() * E) / amp
        proj = np.maximum(proj, 0)
        
        beta = 0.1 + 0.2 * (it / max(n_iter, 1))
        scale = np.exp(-beta * proj / (proj.max() + 1e-15))
        A_new = A * scale
        
        a_new = np.real(np.fft.irfft(A_new, L))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def _pressure_redirect(a, n_iter=500, deadline=None):
    """
    Peak pressure redistribution: reduce coordinates that contribute most to peaks.
    More aggressive than simple peak shaving.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    L = _next_pow2(2 * n - 1)
    
    best_a = a.copy()
    best_score = _score(a)
    eta = 0.1
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_mean = 1.0 / (2 * n - 1)
        
        # Compute pressure on each coordinate
        # p[i] = sum_k (c[k] - c_mean)_+ * a[k-i] / sum_j a[j]
        # Use FFT for efficiency
        excess = np.maximum(c - c_mean, 0)
        # p[i] = (excess * a)[i] in circular sense, but we need linear
        # p[i] = sum_k excess[k] * a[k-i] for valid k-i
        # This is a cross-correlation
        N = L
        fe = np.fft.rfft(excess, N)
        fa = np.fft.rfft(a, N)
        # cross-correlation: ifft(conj(Fa) * Fe)
        p = np.real(np.fft.irfft(fa.conjugate() * fe, N))[:n]
        
        p = np.maximum(p, 0)
        p_max = p.max()
        if p_max < 1e-15:
            break
        
        # Exponential tilting
        a_new = a * np.exp(-eta * p / p_max)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            eta = min(0.5, eta * 1.01)
        else:
            eta = max(0.02, eta * 0.97)
            a = best_a.copy()
            if eta < 0.03:
                break
        
        a = a_new
    
    return best_a

    adam_t = 0
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.03 * c_max)
    
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    base_lr = lr
    
    prev_z = z.copy()
    prev_grad = None

        if time.monotonic() > deadline:
            break

        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)

        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

            z = best_z.copy()
            m = np.zeros(n)

            continue
        
        # BB step size
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):

            if dot_ss > 1e-20:

        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        if not np.all(np.isfinite(z_new)):
            continue
        
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.9)
        
        # Adaptive beta

            top3 = np.sort(c)[::-1][:3]
            if len(top3) >= 2:

                    beta = max(1e-10, beta * 0.6)
        
        # Restart

            rng = np.random.default_rng(it * 31 + 7)
            z_restart = best_z + 0.2 * best_score * rng.standard_normal(n)
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))

        if n > 1024:
            # Use subset of constraints
            pass  # Still use all constraints for correctness
        
        g = _lp_spread(a, M_lp)
        
        if g is None:
            inflate = min(1.01, inflate * 1.1)
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(1.005, inflate * 1.05)
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate = max(1.0002, inflate * 0.98)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.95)
            if no_imp > max(100, n // 4):
                a = best_a.copy()
                blend = blend_init
                inflate = inflate_init
                no_imp = 0
        
        a = a_new
    
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_u_shape(n, depth=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_inv_u_shape(n, depth=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - depth * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cosine_window(n, power=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * (t - 0.5)) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_spectrum(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.abs(np.real(np.fft.ifft(X)))[:n]
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_smooth_random(n, rng, n_fourier=8):
    """Random smooth profile from few Fourier modes."""
    coeffs_real = rng.standard_normal(n_fourier)
    coeffs_imag = rng.standard_normal(n_fourier)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_fourier + 1):
        a += abs(coeffs_real[k-1]) * np.cos(2 * np.pi * k * t + coeffs_imag[k-1])
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _cascade_optimize(a0, deadline, n):
    """
    Run a cascade of optimizers on the given sequence.
    Returns the best sequence found.
    """
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best = a.copy()
    best_sc = _score(a)
    
    # Step 1: Spectral shaping (fast, global)
    t_sub = min(deadline, time.monotonic() + 60.0)
    a_spec = _spectral_shape(a, n_iter=200, alpha=0.2, deadline=t_sub)
    sc = _score(a_spec)
    if sc < best_sc:
        best_sc = sc
        best = a_spec.copy()
        a = a_spec
    
    # Step 2: Pressure redirect
    t_sub = min(deadline, time.monotonic() + 30.0)
    a_press = _pressure_redirect(a, n_iter=300, deadline=t_sub)
    sc = _score(a_press)
    if sc < best_sc:
        best_sc = sc
        best = a_press.copy()
        a = a_press
    
    # Step 3: Gradient optimization
    t_sub = min(deadline, time.monotonic() + 120.0)
    a_grad = _logit_adam(a, deadline=t_sub, max_iter=20000, lr=0.04)
    sc = _score(a_grad)
    if sc < best_sc:
        best_sc = sc
        best = a_grad.copy()
        a = a_grad
    
    # Step 4: LP refinement
    t_sub = min(deadline, time.monotonic() + 180.0)
    a_lp = _lp_refine(a, deadline=t_sub, max_iter=8000, inflate_init=1.0008, blend_init=0.15)
    sc = _score(a_lp)
    if sc < best_sc:
        best_sc = sc
        best = a_lp.copy()
        a = a_lp
    
    # Step 5: Final pressure + gradient
    t_sub = min(deadline, time.monotonic() + 30.0)
    a_p2 = _pressure_redirect(a, n_iter=200, deadline=t_sub)
    sc = _score(a_p2)
    if sc < best_sc:
        best_sc = sc
        best = a_p2.copy()
        a = a_p2
    
    t_sub = min(deadline, time.monotonic() + 60.0)
    a_g2 = _logit_adam(a, deadline=t_sub, max_iter=15000, lr=0.02)
    sc = _score(a_g2)
    if sc < best_sc:
        best_sc = sc
        best = a_g2.copy()
        a = a_g2
    
    return best


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-length optimization for minimizing autocorrelation peak.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_score = np.inf
    best_seq = None
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass
    
    # Sequence of lengths to try (coarse to fine)
    lengths = [256, 384, 512, 768]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)
    
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen and ln <= 2048:
            seen.add(ln)
            unique_lengths.append(ln)
    
    print(f"Trying lengths: {unique_lengths}", flush=True)
    
    # For each length, generate initializations and optimize
    prev_best = None  # Best from previous (shorter) length for interpolation
    
    for n_t in unique_lengths:
        if time.monotonic() > deadline - 60:
            break
        
        # Generate initializations for this length
        inits = []
        
        # From parent
        if parent is not None:
            inits.append(('parent', _interpolate_profile(parent, n_t)))
        
        # From previous shorter length result
        if prev_best is not None:
            inits.append(('interp_prev', _interpolate_profile(prev_best, n_t)))
        
        # U-shape
        for depth in [0.5, 1.0, 1.5, 2.0]:
            inits.append((f'u_{depth}', _make_u_shape(n_t, depth)))
        
        # Inverted U
        for depth in [0.3, 0.5, 0.7]:
            inits.append((f'invu_{depth}', _make_inv_u_shape(n_t, depth)))
        
        # Cosine window
        for pw in [0.5, 1.0, 2.0]:
            inits.append((f'cos_{pw}', _make_cosine_window(n_t, pw)))
        
        # Flat spectrum
        inits.append(('flat_spec', _make_flat_spectrum(n_t, rng)))
        
        # Smooth random
        for i in range(3):
            inits.append((f'rand_{i}', _make_smooth_random(n_t, rng, n_fourier=6)))
        
        # Evaluate all inits
        scored_inits = []
        for name, a0 in inits:
            a0 = np.maximum(a0, 0)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            sc = _score(a0)
            scored_inits.append((sc, name, a0))
        
        scored_inits.sort(key=lambda x: x[0])
        
        print(f"\n=== n={n_t} === Top inits:", flush=True)
        for sc, name, _ in scored_inits[:5]:
            print(f"  {name}: {sc:.6f}", flush=True)
        
        # Pick top candidates (those with score < best or close)
        # Also always include the best init
        top_candidates = []
        for sc, name, a0 in scored_inits:
            if sc < best_score * 1.1 or len(top_candidates) < 3:
                top_candidates.append((sc, name, a0))
            if len(top_candidates) >= 5:
                break
        
        # Optimize top candidates
        best_this_n = None
        best_this_n_sc = np.inf
        
        for sc0, name, a0 in top_candidates:
            if time.monotonic() > deadline - 30:
                break
            
            # Time budget per candidate
            time_per_candidate = min(150.0, max(30.0, (deadline - time.monotonic()) / max(1, len(top_candidates))))
            sub_deadline = time.monotonic() + time_per_candidate
            
            print(f"  Optimizing {name} (n={n_t}), init_score={sc0:.6f}, budget={time_per_candidate:.0f}s", flush=True)
            
            a_opt = _cascade_optimize(a0, sub_deadline, n_t)
            sc_opt = _score(a_opt)
            print(f"    Result: {sc_opt:.10f}", flush=True)
            
            if sc_opt < best_this_n_sc:
                best_this_n_sc = sc_opt
                best_this_n = a_opt.copy()
            
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"    *** New global best: {best_score:.10f} ***", flush=True)
        
        if best_this_n is not None:
            prev_best = best_this_n
    
    # Final polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal polish, remaining={deadline - time.monotonic():.0f}s", flush=True)
        
        # LP refinement on the best
        t_sub = min(deadline - 5, time.monotonic() + 60.0)
        a_final = _lp_refine(best_seq, deadline=t_sub, max_iter=5000, inflate_init=1.0005, blend_init=0.1)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  LP polish: {best_score:.10f}", flush=True)
        
        # Gradient polish
        t_sub = min(deadline - 3, time.monotonic() + 30.0)
        a_gp = _logit_adam(best_seq, deadline=t_sub, max_iter=10000, lr=0.01)
        sc_gp = _score(a_gp)
        if sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()
            print(f"  Gradient polish: {best_score:.10f}", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = _make_u_shape(512, 1.0)
        best_score = _score(best_seq)
    
    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_u_shape(512, 1.0)
    
    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
