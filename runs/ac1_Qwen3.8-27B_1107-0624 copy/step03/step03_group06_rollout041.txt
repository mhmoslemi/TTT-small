def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 200:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _solve_lp_full(f, M, n):
    """Solve the full LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _solve_lp_reduced(f, M, n, top_k=200):
    """Solve LP with only the top most-binding constraints."""
    # First compute f * f to identify which lags are close to max
    c_ff = _fft_convolve(f)
    # The constraint (f*g)[k] <= M: the most binding are where f has mass
    # Use the lags where f*f is largest as proxy for binding constraints
    top_idx = np.argsort(c_ff)[::-1][:top_k]
    top_idx = np.sort(top_idx)
    
    n_constr = len(top_idx)
    A = np.zeros((n_constr, n))
    for ci, k in enumerate(top_idx):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[ci, j] = f[idx]
    
    c_obj = -np.ones(n)
    b = np.full(n_constr, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_golden(a, g, max_iter=15):
    """Golden section line search for best blend parameter t."""
    best_sc = _score(a)
    best_t = 0.0
    
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - (hi - lo) / gr
    d_pt = lo + (hi - lo) / gr
    
    def _eval_t(t):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return _score(a_new)
    
    sc_c = _eval_t(c_pt)
    sc_d = _eval_t(d_pt)
    
    for _ in range(max_iter):
        if sc_c < sc_d:
            hi = d_pt
            d_pt = c_pt
            sc_d = sc_c
            c_pt = hi - (hi - lo) / gr
            sc_c = _eval_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            sc_c = sc_d
            d_pt = lo + (hi - lo) / gr
            sc_d = _eval_t(d_pt)
    
    best_t = (lo + hi) / 2.0
    best_sc = _eval_t(best_t)
    if best_sc >= _score(a):
        return 0.0, _score(a)
    return best_t, best_sc


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _peak_shaving_grad(a, top_k=5):
    """Gradient targeting only the top-k peaks of the convolution."""
    n = len(a)
    c = _fft_convolve(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Weighted gradient from top peaks
    peak_vals = c[top_idx]
    weights = np.exp(peak_vals / max(peak_vals.max(), 1e-10))
    weights = weights / weights.sum()
    
    N_pad = _next_pow2(2 * n - 1)
    grad = np.zeros(n)
    for ki, k in enumerate(top_idx):
        w_k = np.zeros(2 * n - 1)
        w_k[k] = weights[ki]
        fw = np.fft.rfft(w_k, N_pad)
        fa_rev = np.fft.rfft(a[::-1], N_pad)
        corr = np.fft.irfft(fw * fa_rev, N_pad)
        grad += 2.0 * corr[n - 1:2 * n - 1]
    
    return grad, float(c.max())

        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        if a_trial[i] < 0:
            a_trial[i] = 0
            a_trial[j] += delta  # redistribute
        
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        
        sc = _score(a_trial)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        
        # Occasionally try the reverse direction
        if it % 3 == 0:
            a_trial2 = a.copy()
            delta2 = step * a[j]
            if delta2 >= 1e-12:
                a_trial2[i] += delta2
                a_trial2[j] -= delta2
                a_trial2 = np.maximum(a_trial2, 0)
                s2 = a_trial2.sum()
                if s2 > 1e-10:
                    a_trial2 = a_trial2 / s2
                    sc2 = _score(a_trial2)
                    if sc2 < best_score - 1e-14:
                        best_score = sc2
                        best_a = a_trial2.copy()
                        a = a_trial2
        
        if it % 5000 == 4999:
            si += 1
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization -> LP refine -> gradient polish
    -> peak shaving -> coordinate polish. Tries multiple lengths and strategies.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Generate initializations at multiple lengths
    lengths = [512, 768, 1024, 1536]
    configs = []
    
    for n_t in lengths:
        # Parent interpolated
        if parent is not None:
            configs.append(('parent', n_t, _interpolate(parent, n_t)))
        
        # Diverse profiles
        configs.append(('uniform', n_t, _make_uniform(n_t)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('ramp_rev', n_t, _make_ramp_rev(n_t, 1.5)))
        configs.append(('edge_conc', n_t, _make_edge_conc(n_t, 1.5)))
        configs.append(('asym', n_t, _make_asym(n_t, 0.5)))
        configs.append(('asym2', n_t, _make_asym(n_t, 1.0)))
        configs.append(('two_bump', n_t, _make_two_bump(n_t, 0.35, 0.3, rng)))
        configs.append(('rand_exp', n_t, _make_random(n_t, rng, 'exponential')))
        configs.append(('rand_bimodal', n_t, _make_random(n_t, rng, 'bimodal')))
        configs.append(('rand_unif', n_t, _make_random(n_t, rng, 'uniform')))
        configs.append(('comb', n_t, _make_comb(n_t, period=4, depth=0.3, rng=rng)))
    
    # Score and sort initializations
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])
    
    print(f"Top 10 initializations:")
    for sc, name, n_t, _ in scored_configs[:10]:
        print(f"  {name} n={n_t}: score={sc:.8f}")
    
    # Phase 1: LP refinement on top initializations
    # Allocate time: top configs get more time
    n_configs_to_try = min(len(scored_configs), 12)
    time_per_config_base = (deadline - t_start - 200) / max(n_configs_to_try, 1)
    
    for idx in range(n_configs_to_try):
        if time.monotonic() > deadline - 180:
            break
        sc0, name, n_t, a0 = scored_configs[idx]
        
        # Skip if clearly not competitive
        if sc0 >= best_score + 0.3:
            continue
        
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
        
        # Time allocation: more for better configs
        if sc0 < best_score + 0.02:
            alloc = min(time_per_config_base * 1.5, deadline - time.monotonic() - 100)
        else:
            alloc = min(time_per_config_base * 0.7, deadline - time.monotonic() - 100)
        
        if alloc < 10:
            continue
        
        lp_end = time.monotonic() + alloc
        use_reduced = n_t > 1000
        print(f"LP: {name} n={n_t} init={sc0:.8f} alloc={alloc:.0f}s reduced={use_reduced}")
        
        a_lp = _lp_refine(a0, lp_end, max_iter=50000,
                          inflate_start=1.003, inflate_min=1.0002,
                          use_reduced=use_reduced, top_k=min(300, 2 * n_t))
        sc_lp = _score(a_lp)
        
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP score={sc_lp:.10f} (NEW BEST)")
        else:
            print(f"  -> LP score={sc_lp:.8f}")
    
    # Phase 2: Gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 120:
        gp_time = min(120.0, deadline - time.monotonic() - 80)
        if gp_time > 10:
            gp_end = time.monotonic() + gp_time
            print(f"\nGradient polish (smooth max): budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=50000, use_peak_shaving=False)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> score={sc_gp:.10f}")
    
    # Phase 3: Peak shaving gradient
    if best_seq is not None and time.monotonic() < deadline - 100:
        ps_time = min(80.0, deadline - time.monotonic() - 60)
        if ps_time > 10:
            ps_end = time.monotonic() + ps_time
            print(f"\nPeak shaving polish: budget={ps_time:.0f}s")
            a_ps = _gradient_polish(best_seq, ps_end, max_iter=30000, use_peak_shaving=True)
            sc_ps = _score(a_ps)
            if np.isfinite(sc_ps) and sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            print(f"  -> score={sc_ps:.10f}")
    
    # Phase 4: Another LP pass with tighter inflation
    if best_seq is not None and time.monotonic() < deadline - 80:
        lp_time = min(100.0, deadline - time.monotonic() - 50)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            n_t = len(best_seq)
            use_reduced = n_t > 1000
            print(f"\nFinal LP (tight): budget={lp_time:.0f}s")
            a_final = _lp_refine(best_seq, lp_end, max_iter=30000,
                                 inflate_start=1.001, inflate_min=1.0001,
                                 use_reduced=use_reduced, top_k=min(300, 2 * n_t))
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  -> score={sc_final:.10f}")
    
    # Phase 5: Coordinate polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        cp_time = min(60.0, deadline - time.monotonic() - 20)
        if cp_time > 10:
            cp_end = time.monotonic() + cp_time
            print(f"\nCoordinate polish: budget={cp_time:.0f}s")
            a_cp = _coordinate_polish(best_seq, cp_end, max_iter=200000)
            sc_cp = _score(a_cp)
            if np.isfinite(sc_cp) and sc_cp < best_score:
                best_score = sc_cp
                best_seq = a_cp.copy()
            print(f"  -> score={sc_cp:.10f}")
    
    # Phase 6: Final quick gradient
    if best_seq is not None and time.monotonic() < deadline - 5:
        gp_time = min(40.0, deadline - time.monotonic() - 2)
        if gp_time > 5:
            gp_end = time.monotonic() + gp_time
            a_g = _gradient_polish(best_seq, gp_end, max_iter=20000, use_peak_shaving=False)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  -> Final gradient score={sc_g:.10f}")
    
    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)
    
    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)
    
    sc_local = _score(final_seq)
    print(f"\nFINAL (local): score={sc_local:.10f}, n={final_seq.size}")
    
    # Verify with evaluate_sequence if available and sequence is small enough
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
    
    return [float(x) for x in final_seq]
