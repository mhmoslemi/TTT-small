from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def _smooth_max_logsumexp(c, beta):
    """Compute log-sum-exp of c/beta + beta*log(sum(exp((c-max)/beta)))"""
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0:
        return m, np.zeros_like(c)
    val = m + beta * np.log(s)
    w = ex / s
    return val, w

def _objective_and_grad(z, beta):
    """Compute smooth-max objective and gradient w.r.t. z (log-space)."""
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max_logsumexp(c, beta)
    
    # Gradient of smooth max w.r.t. a
    # d/d a_i [sum_k w_k * c_k] = 2 * sum_k w_k * a_{k-i} (with appropriate bounds)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]
    grad_a = 2.0 * h

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -80.0, 80.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        return m, np.zeros_like(c)
    val = m + beta * np.log(s)
    w = ex / s
    return val, w


def _obj_grad_smooth(z, beta):
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max_and_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    return 2.0 * n * val, grad_z


def _make_conv_matrix_rows(a, M):
    n = len(a)
    return np.full(2 * n - 1, M)


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n, power=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_decay(n, rate=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bilateral(n, decay=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-decay * np.abs(t - 0.5) * 2.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _lbfgs_smooth(a_init, deadline, betas=None, max_iter_per_beta=3000):
    """Multi-pass L-BFGS with decreasing beta (smooth max temperature)."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    if betas is None:
        c0 = _fft_convolve(a)
        cmax0 = float(c0.max())
        betas = [0.2 * cmax0, 0.1 * cmax0, 0.05 * cmax0, 0.02 * cmax0,
                 0.01 * cmax0, 0.005 * cmax0, 0.002 * cmax0, 0.001 * cmax0]

    best_a = a.copy()
    best_score = _score(a)
    z = np.log(a)
    z -= z.mean()

    for beta in betas:
        if time.monotonic() > deadline:
            break

        def func(zv):
            val, _ = _obj_grad_smooth(zv, beta)
            return val

        def jac(zv):
            _, g = _obj_grad_smooth(zv, beta)
            return g

        res = minimize(func, z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': max_iter_per_beta, 'ftol': 1e-12,
                                'gtol': 1e-10, 'maxls': 50})

        if res.x is not None and np.all(np.isfinite(res.x)):
            z = res.x
            a_new = _softmax(z)
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a


def _peak_shave_cd(a, deadline, max_sweeps=500):
    """Coordinate descent peak shaving: iteratively reduce elements contributing to peaks."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))
        med = float(np.median(c))
        excess = c_max - med

        if excess < 1e-10:
            break

        # Compute contribution of each element to the peak
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contrib = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = a[j]

        # Reduce high-contribution elements, boost low-contribution ones
        max_contrib = contrib.max()
        if max_contrib < 1e-15:
            break

        # Exponential reduction proportional to contribution
        reduction = np.exp(-0.3 * contrib / max_contrib)
        a_new = a * reduction
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break

        a = a_new

    return best_a


def _multi_peak_shave(a, deadline, n_peaks=3, max_sweeps=300):
    """Shave multiple peaks simultaneously."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        # Get top peaks above median
        med = float(np.median(c))
        above = np.where(c > med * 1.01)[0]
        if len(above) == 0:
            break

        top_k = min(n_peaks, len(above))
        peak_indices = np.argsort(c[above])[::-1][:top_k]
        peak_indices = above[peak_indices]

        # Weighted contribution
        peak_vals = c[peak_indices]
        weights = (peak_vals - med)
        weights = weights / weights.sum()

        contrib = np.zeros(n)
        for pk, w in zip(peak_indices, weights):
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            for i in range(lo, hi + 1):
                j = pk - i
                if 0 <= j < n:
                    contrib[i] += w * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        a_new = a * np.exp(-0.2 * contrib / max_c)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                break

        a = a_new

    return best_a


def _lp_spread_step(a, M, deadline):
    """Single LP step: find g maximizing sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(1, int(deadline - time.monotonic()))})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.2):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline - 5:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate
        g = _lp_spread_step(a, M, deadline)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.05, blend * 0.9)
            if no_imp > 200:
                a = best_a.copy()
                no_imp = 0
                inflate = min(1.01, inflate * 1.1)

        a = a_new

    return best_a


def _gradient_descent_fine(a, deadline, max_iter=20000, lr_init=0.02):
    """Fine gradient descent with very small beta for final polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    beta = 0.005 * float(c.max())
    lr = lr_init

    best_a = a.copy()
    best_score = _score(a)
    z = np.log(a)
    z -= z.mean()
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        if it % 500 == 0:
            beta = max(1e-6, beta * 0.9)

        val, grad_z = _obj_grad_smooth(z, beta)

        if not np.all(np.isfinite(grad_z)):
            break

        z = z - lr * grad_z
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()
        a_new = _softmax(z)

        sc = _score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)

        if no_imp > max(3000, 5 * n):
            break

        a = a_new

    return best_a


def _random_restart(a, rng, scale=0.1):
    """Perturb a sequence randomly."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    a_pert = np.maximum(a_pert, 1e-15)
    a_pert = a_pert / a_pert.sum()
    return a_pert


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diversified initialization (parent, parametric, flat-spectrum)
    2. L-BFGS smooth-max optimization (multiple beta values)
    3. Peak shaving coordinate descent
    4. LP refinement
    5. Fine gradient polish
    6. Multi-length search (768, 1024, 1536)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try (prioritize longer for better asymptotic score)
    lengths = [1024, 768, 1536]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)

    print(f"Lengths to try: {lengths}", flush=True)
    print(f"Total budget: {budget_s - 30:.0f}s", flush=True)

    # Phase 1: Quick initialization for each length
    print("Phase 1: Initialization", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append((f'parent_{n_t}', n_t, _interpolate_profile(parent, n_t)))

        init_pool.append((f'raised_cos_{n_t}', n_t, _make_raised_cosine(n_t, 2.0)))
        init_pool.append((f'raised_cos3_{n_t}', n_t, _make_raised_cosine(n_t, 3.0)))
        init_pool.append((f'linear_{n_t}', n_t, _make_linear_decay(n_t, 2.0)))
        init_pool.append((f'bilateral_{n_t}', n_t, _make_bilateral(n_t, 1.5)))
        init_pool.append((f'flat_spec_{n_t}', n_t, _make_flat_spectrum(n_t, rng)))

    # Score all inits quickly
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print("Top 8 initializations:", flush=True)
    for sc, name, n_t, _ in init_scores[:8]:
        print(f"  {name}: {sc:.8f}", flush=True)

    # Phase 2: L-BFGS optimization on top candidates
    # Budget allocation: give more time to longer sequences and better inits
    top_inits = init_scores[:min(6, len(init_scores))]

    # Allocate time based on remaining budget
    n_candidates = len(top_inits)
    time_per_candidate = (deadline - t_start - 300) / max(1, n_candidates)
    time_per_candidate = max(60.0, min(300.0, time_per_candidate))

    print(f"\nPhase 2: L-BFGS optimization ({n_candidates} candidates, ~{time_per_candidate:.0f}s each)", flush=True)

    for idx, (sc0, name, n_t, a0) in enumerate(top_inits):
        if time.monotonic() > deadline - 200:
            break

        # Skip if clearly worse and not parent
        if sc0 > best_score * 1.05 and 'parent' not in name:
            print(f"  Skipping {name} (n={n_t}): init score {sc0:.6f} too high", flush=True)
            continue

        opt_deadline = min(deadline - 200, time.monotonic() + time_per_candidate)
        remaining = opt_deadline - time.monotonic()
        if remaining < 10:
            continue

        print(f"  [{idx+1}/{n_candidates}] L-BFGS on {name} (n={n_t}), init={sc0:.8f}, budget={remaining:.0f}s", flush=True)

        # Multi-pass L-BFGS
        max_iter_per_beta = max(500, int(remaining / 8))
        a_opt = _lbfgs_smooth(a0, opt_deadline, max_iter_per_beta=max_iter_per_beta)
        sc_opt = _score(a_opt)
        print(f"    L-BFGS result: {sc_opt:.10f}", flush=True)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # Phase 3: Peak shaving
        if time.monotonic() < deadline - 150:
            ps_deadline = min(deadline - 150, time.monotonic() + 60.0)
            a_ps = _multi_peak_shave(a_opt, ps_deadline, n_peaks=3, max_sweeps=200)
            sc_ps = _score(a_ps)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
                print(f"    Peak shave: {sc_ps:.10f} (improved!)", flush=True)
            else:
                print(f"    Peak shave: {sc_ps:.10f}", flush=True)
            a_opt = a_ps if sc_ps < sc_opt else a_opt

        # Phase 4: LP refinement (if time allows)
        if time.monotonic() < deadline - 100 and n_t <= 1024:
            lp_deadline = min(deadline - 100, time.monotonic() + 60.0)
            a_lp = _lp_refine(a_opt, lp_deadline, max_iter=500, inflate=1.001, blend=0.15)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"    LP refine: {sc_lp:.10f} (improved!)", flush=True)
            else:
                print(f"    LP refine: {sc_lp:.10f}", flush=True)
            a_opt = a_lp if sc_lp < _score(a_opt) else a_opt

    # Phase 5: Fine polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 50:
        gp_deadline = min(deadline - 30, time.monotonic() + 120.0)
        print(f"\nPhase 5: Fine gradient polish, budget={gp_deadline - time.monotonic():.0f}s", flush=True)
        a_fine = _gradient_descent_fine(best_seq, gp_deadline, max_iter=30000, lr_init=0.01)
        sc_fine = _score(a_fine)
        print(f"  Fine polish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

        # Final peak shave
        if time.monotonic() < deadline - 20:
            ps2_deadline = min(deadline - 15, time.monotonic() + 30.0)
            a_ps2 = _multi_peak_shave(best_seq, ps2_deadline, n_peaks=5, max_sweeps=200)
            sc_ps2 = _score(a_ps2)
            if sc_ps2 < best_score:
                best_score = sc_ps2
                best_seq = a_ps2.copy()
                print(f"  Final peak shave: {sc_ps2:.10f} (improved!)", flush=True)

    # Phase 6: Try a few random restarts from the best if time allows
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\nPhase 6: Random restarts, budget={deadline - time.monotonic():.0f}s", flush=True)
        n_restarts = min(5, int((deadline - time.monotonic()) / 40))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            a_r = _random_restart(best_seq, rng, scale=0.05 * (r + 1))
            sc_r = _score(a_r)
            if sc_r < best_score * 1.02:
                sub_deadline = min(deadline - 10, time.monotonic() + 30.0)
                a_r = _lbfgs_smooth(a_r, sub_deadline, max_iter_per_beta=500)
                sc_r = _score(a_r)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = a_r.copy()
                    print(f"  Restart {r+1}: {sc_r:.10f} (improved!)", flush=True)
                else:
                    print(f"  Restart {r+1}: {sc_r:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_raised_cosine(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_raised_cosine(768, 2.0)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence score: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
