When `H(w)` is large (i.e., `w` is spread out), the smooth approximation is good and we can take larger steps.

This suggests an adaptive step size: `lr ∝ H(w)` or `lr ∝ β`.

Let me also think about the LP approach more carefully. The LP refinement step is:
1. Compute `M = max conv(a,a)`.
2. Solve `max sum(b) s.t. conv(a,b) <= M, b >= 0`.
3. Let `s = sum(b)`. If `s > 1`, then `b/s` is a feasible direction that reduces the max.
4. Update: `a <- (1-t)*a + t*(b/s)`.

The key question is: what's the optimal `t`? 

If we think of it as a line search: we want to minimize `max_k conv((1-t)*a + t*b/s, (1-t)*a + t*b/s)[k]` over `t ∈ [0,1]`.

By bilinearity of convolution, `conv((1-t)*a + t*b/s, (1-t)*a + t*b/s) = (1-t)²*conv(a,a) + t*(1-t)*(conv(a,b/s) + conv(b/s,a)) + t²*conv(b/s, b/s)`.

This is a quadratic in `t` for each `k`. We want to minimize the max over `k`. This is a 1D convex optimization problem (max of convex functions is convex). We can solve it with a bisection or golden section search.

This is a good improvement! Instead of a fixed `t=0.3`, do a proper line search.

"""
Optimized step function search combining:
1. FISTA with adaptive smooth-max in logit space
2. LP-based direction finding with 1D line search
3. Multi-length exploration (1024, 2048)
4. Adaptive restart with structured noise
5. Final micro-polish phase
Target: beat 1.506320, aim for ~1.5027 or better.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_grad(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_line_search(a, b_norm, M, max_iter=30):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*b_norm minimizing max conv."""
    n = len(a)
    # Binary search for optimal t
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_sc = _score_norm(a)
    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        a_mid = (1.0 - mid) * a + mid * b_norm
        a_mid = np.maximum(a_mid, 0.0)
        s = float(a_mid.sum())
        if s < 1e-15:
            hi = mid
            continue
        a_mid = a_mid / s
        sc = _score_norm(a_mid)
        if sc < best_sc:
            best_sc = sc
            best_t = mid
        # Check derivative sign by perturbing
        eps = 0.01
        if mid + eps > 1.0:
            hi = mid
            continue
        a_mid2 = (1.0 - (mid + eps)) * a + (mid + eps) * b_norm
        a_mid2 = np.maximum(a_mid2, 0.0)
        s2 = float(a_mid2.sum())
        if s2 < 1e-15:
            hi = mid
            continue
        a_mid2 = a_mid2 / s2
        sc2 = _score_norm(a_mid2)
        if sc2 > sc:
            hi = mid
        else:
            lo = mid
    return best_t, best_sc


def _fista_smoothmax(p0, deadline, rng, lr=0.05, beta_init_frac=0.12,
                     beta_decay=0.95, patience=250, noise_scales=None):
    """FISTA-style optimization in logit space with adaptive smooth-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    # FISTA state
    y = z.copy()
    t_fista = 1.0
    z_prev = z.copy()

    b1, b2, eps = 0.9, 0.999, 1e-9
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    if noise_scales is None:
        noise_scales = [0.1, 0.15, 0.2, 0.3]
    noise_idx = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_grad(c, beta)
        g = _conv_cross_corr(w, p)

        # Gradient in logit space: gz = p * (g - <p,g>)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            y = z.copy()
            z_prev = z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            t_fista = 1.0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        # Adam step on y (FISTA momentum is on z, but we use Adam on y)
        lr_eff = lr * (1.0 + 0.1 * np.cos(0.01 * t_adam))  # slight periodic modulation
        y = y - lr_eff * mh / (np.sqrt(vh) + eps)
        y = np.clip(y, -60.0, 60.0)

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        z = y + ((t_fista - 1.0) / t_new) * (y - z_prev)
        z_prev = z.copy()
        t_fista = t_new

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            y = z.copy()
            z_prev = z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            t_fista = 1.0
            continue

        # Adaptive beta
        if t_adam % 20 == 0:
            beta = max(1e-5, beta * beta_decay)
            # Check gap between top two
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.4)

        # Restart
        if no_imp > patience:
            ns = noise_scales[min(noise_idx, len(noise_scales) - 1)]
            noise_idx = min(noise_idx + 1, len(noise_scales) - 1)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            t_fista = 1.0
            y = z.copy()
            z_prev = z.copy()
            lr = base_lr
            no_imp = 0
            noise_idx = 0

    return best_p, best_score


def _lp_refine_v2(a, deadline, max_solves=40):
    """LP refinement with line search."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())

    for _ in range(max_solves):
        if time.monotonic() > deadline:
            break
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b

        # Line search for optimal step
        t_opt, sc_opt = _lp_line_search(a, b_norm, M, max_iter=25)

        if t_opt > 0.01 and sc_opt < current_score:
            current_score = sc_opt
            current_best = (1.0 - t_opt) * a + t_opt * b_norm
            current_best = np.maximum(current_best, 0.0)
            s = float(current_best.sum())
            if s > 1e-15:
                current_best = current_best / s
            a = current_best
            c = _full_convolve(a)
            M = float(c.max())
        else:
            # Try a small fixed step
            a_new = 0.7 * a + 0.3 * b_norm
            a_new = np.maximum(a_new, 0.0)
            s = float(a_new.sum())
            if s < 1e-15:
                break
            a_new = a_new / s
            sc_new = _score_norm(a_new)
            if sc_new < current_score:
                current_score = sc_new
                current_best = a_new.copy()
                a = a_new
                c = _full_convolve(a)
                M = float(c.max())
            else:
                break

    return current_best, current_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_w_jacobi(n, alpha, beta_p):
    """Weight function related to Jacobi polynomials - gives flat-ish convolution."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha * (1.0 - t) ** beta_p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _diversified_restart(p_best, rng, strength):
    """Create a diverse restart from the best solution."""
    n = len(p_best)
    # Mix of global noise and local perturbation
    z = np.log(np.maximum(p_best, 1e-14))
    z -= z.mean()
    # Global noise
    z += strength * rng.normal(size=n)
    # Local perturbations (blocks)
    n_blocks = rng.integers(3, 8)
    for _ in range(n_blocks):
        start = rng.integers(0, n - 10)
        length = rng.integers(5, n // 8)
        end = min(start + length, n)
        z[start:end] += strength * 0.5 * rng.normal(size=end - start)
    z = np.clip(z, -60.0, 60.0)
    z -= z.mean()
    return _softmax_from_z(z)

                                                   lr=0.03, beta_init_frac=0.08,
                                                   beta_decay=0.96, patience=150)
                _update(p_post, sc_post, 'post_LP_fista')

    # --- Phase 4: Multi-restart at best length ---
    phase4_end = t_start + 1550.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(25.0, (phase4_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            per = min(per_r, phase4_end - time.monotonic() - 5)
            if per < 15:
                break
            strength = 0.08 + 0.07 * i
            p_restart = _diversified_restart(best_seq, rng, strength)
            p_r, sc_r = _fista_smoothmax(p_restart, time.monotonic() + per, rng,
                                         lr=0.04, beta_init_frac=0.1,
                                         beta_decay=0.95, patience=int(0.25 * n_best))
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 5: Final LP + micro-polish ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        # Final LP
        if best_seq.size <= 3072:
            p_final_lp, sc_final_lp = _lp_refine_v2(best_seq, time.monotonic() + 80, max_solves=20)
            _update(p_final_lp, sc_final_lp, 'final_LP')

        # Micro-polish with very small steps
        if time.monotonic() < deadline - 10:
            per = deadline - time.monotonic() - 10
            if per > 10:
                p_mp, sc_mp = _fista_smoothmax(best_seq, time.monotonic() + per, rng,
                                               lr=0.01, beta_init_frac=0.04,
                                               beta_decay=0.97, patience=100,
                                               noise_scales=[0.03, 0.05, 0.08])
                _update(p_mp, sc_mp, 'micro_polish')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 2.0)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
