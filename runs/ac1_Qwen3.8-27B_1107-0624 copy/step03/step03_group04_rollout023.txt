"""
Optimized step function minimizer for 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. Multiplicative gradient descent (mirror descent on simplex) instead of logit-Adam
2. Combined smooth-max + L2 flattening objective for better landscape navigation
3. Aggressive multi-parameter LP refinement sweeps
4. Nesterov momentum for faster convergence
5. Coordinate descent polish for final refinement
6. Multi-length search focused on parent length (768) with alternatives
7. Adaptive smooth-max temperature with periodic restart

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_and_grad(a, beta, l2_weight=0.0):
    """
    Compute smooth objective and its gradient w.r.t. a.
    Objective: smooth_max(conv(a,a)) + l2_weight * ||conv(a,a)||^2 / (2*sum(a)^2)
    """
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())

    # Log-sum-exp for smooth max
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    S = float(ex.sum())
    w = ex / S
    smooth_val = c_max + beta * float(np.log(S))

    # Gradient of smooth max: weighted sum of individual gradients
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]
    grad = 2.0 * h

    # L2 penalty gradient: d/da [sum(c^2)] = 2 * sum_k c_k * dc_k/da
    # dc_k/da_m = 2*a_{k-m} (for valid k-m)
    # So d/da_m = 2 * sum_k c_k * 2*a_{k-m} = 4 * (a_rev * c)[m]
    if l2_weight > 0.0:
        N2 = _next_pow2(2 * n - 1)
        fc = np.fft.rfft(c, N2)
        fa2_rev = np.fft.rfft(a[::-1], N2)
        h2 = np.fft.irfft(fc * fa2_rev, N2)[n - 1:2 * n - 1]
        grad_l2 = 4.0 * l2_weight * h2
        grad = grad + grad_l2
        val = smooth_val + l2_weight * float(np.dot(c, c)) / (2.0 * float(a.sum()) ** 2)
    else:
        val = smooth_val

    # Project gradient to simplex tangent space
    a_sum = float(a.sum())
    grad_proj = grad - np.dot(a, grad) / (a_sum * a_sum) * a

    return val, grad_proj, c, c_max


def _multiplicative_ascent(a, deadline, max_iter=100000, beta_init=None,
                           l2_weight=0.0, lr=0.1, momentum=0.9,
                           beta_decay=0.999, restart_patience=3000):
    """
    Mirror descent (multiplicative updates) with Nesterov momentum.
    Optimizes: smooth_max(conv(a,a)) + l2_weight * ||conv(a,a)||^2
    on the simplex {a >= 0, sum(a) = 1}.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max0 = float(c.max())
    if beta_init is None:
        beta = 0.02 * c_max0
    else:
        beta = beta_init

    best_a = a.copy()
    best_score = float(2.0 * n * c_max0)

    momentum_vec = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient
        _, grad, c, c_max = _smooth_max_and_grad(a, beta, l2_weight)

        if not np.all(np.isfinite(grad)):
            break

        # Nesterov momentum
        momentum_vec = momentum * momentum_vec + (1.0 - momentum) * grad

        # Adaptive step size
        grad_norm = float(np.sqrt(np.dot(momentum_vec, momentum_vec)))
        if grad_norm < 1e-15:
            break
        eta = base_lr * (a.sum() / grad_norm) ** 0.5
        eta = min(eta, 2.0)

        # Multiplicative update
        log_update = -eta * momentum_vec
        log_update = np.clip(log_update, -10.0, 10.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        a_new = a_new / s

        # Evaluate
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        # Adaptive beta
        if it % 200 == 0:
            beta = max(1e-10, beta * beta_decay)

        # Slow down if stuck
        if no_imp > 0 and no_imp % 2000 == 0:
            base_lr = max(0.01, base_lr * 0.9)

        # Restart if very stuck
        if no_imp > restart_patience:
            # Perturb from best
            rng = np.random.default_rng(it * 31 + 7)
            noise = 0.1 * rng.standard_normal(n)
            a = best_a * (1.0 + noise)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            momentum_vec = np.zeros(n)
            base_lr = lr
            beta = 0.02 * float(_fft_convolve(a).max())
            no_imp = 0

    return best_a

    Aggressive multi-parameter LP refinement.
    Tries various inflation factors and blend parameters.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    inflations = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]
    blends = [0.1, 0.2, 0.3, 0.5, 0.7]

    no_imp = 0
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        found_improvement = False
        for infl in inflations:
            if time.monotonic() > deadline:
                break
            M_lp = M * (1.0 + infl)
            g = _lp_spread(a, M_lp)
            if g is None:

            sum_g = float(g.sum())
            if sum_g <= 1.0 + 1e-6:
                continue
            g = g / sum_g

            for blend in blends:
                a_new = (1.0 - blend) * a + blend * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-10:

                a_new = a_new / s_new

                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    found_improvement = True
                    break
            if found_improvement:
                break

        if found_improvement:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    lr = 0.05
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, grad, c, c_max_cur = _smooth_max_and_grad(a, beta, 0.0)

        if not np.all(np.isfinite(grad)):
            break

        # Simple gradient step in multiplicative space
        log_update = -lr * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            a = best_a if no_imp % 5000 == 0 else a

        if it % 1000 == 0:
            beta = max(1e-12, beta * 0.85)
            lr = max(0.005, lr * 0.97)

        if no_imp > max(5000, 5 * n):
            break

    return best_a


def _coordinate_descent(a, deadline, max_passes=50):
    """
    Coordinate descent: for each coordinate, try small perturbations.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-10:
                continue
            # Try reducing a[i] slightly and redistributing
            for frac in [0.95, 0.9, 0.8]:
                a_try = a.copy()
                delta = a_try[i] * (1.0 - frac)
                a_try[i] *= frac
                # Redistribute to neighbors
                a_try[0] += delta * 0.3
                a_try[-1] += delta * 0.3
                mid = n // 2
                a_try[mid] += delta * 0.4
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                sc = _score(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            # Restore if no improvement
            a[i] = orig

        if not improved:
            break

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_init_profiles(n, rng):
    """Generate diverse initialization profiles."""
    profiles = []

    # Uniform
    profiles.append(('uniform', np.ones(n) / n))

    # Linear
    t = np.linspace(0, 1, n)
    for eta in [1.0, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append((f'lin_{eta}', a / a.sum()))

    # U-shaped
    for width in [0.2, 0.4, 0.6]:
        a = np.exp(-((t - 0.5) / width) ** 2)
        a_u = 1.0 / (a + 0.1)
        profiles.append((f'u_{width}', a_u / a_u.sum()))

    # Cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    profiles.append(('cos', a / a.sum()))

    # Flat spectrum random
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.abs(np.real(np.fft.ifft(X)))[:n]
    a = np.maximum(a, 1e-10)
    profiles.append(('flat_spec', a / a.sum()))

    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

    Strategy:
    1. Start from parent (n=768) and diverse inits
    2. Mirror descent with smooth max + L2 penalty
    3. Aggressive multi-parameter LP refinement
    4. Fine gradient polish
    5. Try alternative lengths
    6. Final LP pass
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
    except Exception:
        pass

    # Primary length
    primary_n = parent.size if parent is not None else 768

    # Phase 1: Optimize parent with mirror descent
    if parent is not None:
        print(f"\nPhase 1: Mirror descent on parent (n={primary_n})", flush=True)
        md_deadline = min(deadline - 400, t_start + 300)
        a_md = _multiplicative_ascent(parent, md_deadline,
                                      max_iter=80000, beta_init=None,
                                      l2_weight=0.01, lr=0.15,
                                      momentum=0.85, beta_decay=0.998)
        sc_md = _score(a_md)
        print(f"  Mirror descent: {sc_md:.10f}", flush=True)
        if sc_md < best_score:
            best_score = sc_md
            best_seq = a_md.copy()

    # Phase 2: LP refinement (aggressive)
    if best_seq is not None:
        print(f"\nPhase 2: LP refinement (n={best_seq.size})", flush=True)
        lp_deadline = min(deadline - 300, t_start + 600)
        a_lp = _lp_refine_multi(best_seq, lp_deadline, max_rounds=300)
        sc_lp = _score(a_lp)
        print(f"  LP refine: {sc_lp:.10f}", flush=True)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

    # Phase 3: More mirror descent (with L2 penalty for flattening)
    if best_seq is not None:
        print(f"\nPhase 3: Mirror descent + L2 (n={best_seq.size})", flush=True)
        md2_deadline = min(deadline - 250, t_start + 800)
        a_md2 = _multiplicative_ascent(best_seq, md2_deadline,
                                       max_iter=60000, beta_init=0.005 * _fft_convolve(best_seq).max(),
                                       l2_weight=0.05, lr=0.08,
                                       momentum=0.8, beta_decay=0.999)
        sc_md2 = _score(a_md2)
        print(f"  MD+L2: {sc_md2:.10f}", flush=True)
        if sc_md2 < best_score:
            best_score = sc_md2
            best_seq = a_md2.copy()

    # Phase 4: LP refinement again
    if best_seq is not None:
        print(f"\nPhase 4: LP refinement round 2", flush=True)
        lp2_deadline = min(deadline - 200, t_start + 1000)
        a_lp2 = _lp_refine_multi(best_seq, lp2_deadline, max_rounds=200)
        sc_lp2 = _score(a_lp2)
        print(f"  LP refine 2: {sc_lp2:.10f}", flush=True)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()

    # Phase 5: Fine gradient polish
    if best_seq is not None:
        print(f"\nPhase 5: Fine gradient polish", flush=True)
        gp_deadline = min(deadline - 150, t_start + 1200)
        a_gp = _gradient_polish(best_seq, gp_deadline, max_iter=40000)
        sc_gp = _score(a_gp)
        print(f"  Gradient polish: {sc_gp:.10f}", flush=True)
        if sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()

    # Phase 6: Try alternative lengths
    if best_seq is not None:
        alt_lengths = [1024, 512]
        for alt_n in alt_lengths:
            if alt_n == best_seq.size:
                continue
            if time.monotonic() > deadline - 150:
                break
            print(f"\nPhase 6: Trying n={alt_n}", flush=True)
            a_alt = _interpolate_profile(best_seq, alt_n)
            sc_alt = _score(a_alt)
            print(f"  Interpolated: {sc_alt:.10f}", flush=True)

            if sc_alt < best_score * 1.01:  # Only optimize if competitive
                alt_deadline = min(deadline - 100, time.monotonic() + 120)
                a_alt_opt = _multiplicative_ascent(a_alt, alt_deadline,
                                                   max_iter=20000, lr=0.1, momentum=0.85)
                sc_alt_opt = _score(a_alt_opt)
                # Quick LP
                if time.monotonic() < deadline - 60:
                    a_alt_lp = _lp_refine_multi(a_alt_opt, min(deadline - 60, time.monotonic() + 40), max_rounds=50)
                    sc_alt_lp = _score(a_alt_lp)
                    if sc_alt_lp < sc_alt_opt:
                        sc_alt_opt = sc_alt_lp
                        a_alt_opt = a_alt_lp
                print(f"  Optimized n={alt_n}: {sc_alt_opt:.10f}", flush=True)
                if sc_alt_opt < best_score:
                    best_score = sc_alt_opt
                    best_seq = a_alt_opt.copy()

    # Phase 7: Final LP pass
    if best_seq is not None:
        print(f"\nPhase 7: Final LP pass", flush=True)
        final_lp_deadline = deadline - 10
        if final_lp_deadline - time.monotonic() > 20:
            a_final = _lp_refine_multi(best_seq, final_lp_deadline, max_rounds=100)
            sc_final = _score(a_final)
            print(f"  Final LP: {sc_final:.10f}", flush=True)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()

    # Phase 8: One more gradient polish if time
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\nPhase 8: Final gradient polish", flush=True)
        gp_final_deadline = deadline - 5
        a_fp = _gradient_polish(best_seq, gp_final_deadline, max_iter=20000,
                                beta=0.001 * _fft_convolve(best_seq).max())
        sc_fp = _score(a_fp)
        print(f"  Final polish: {sc_fp:.10f}", flush=True)
        if sc_fp < best_score:
            best_score = sc_fp
            best_seq = a_fp.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s <= 0.01:
        final_seq = np.ones(512) / 512

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence score: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
