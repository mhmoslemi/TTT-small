"""
Step function optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses min-peak LP with active-set reduction, coordinate descent polishing,
and multi-scale exploration with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming a is normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = float(a.sum())
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s


def _lp_min_peak(a, active_indices, top_k=None):
    """
    Solve the min-peak LP:
    minimize M
    subject to: (a*b)_k <= M for k in active_indices
                sum(b) = 1
                b >= 0
    
    Returns the optimal b (unit sum) and M.
    """
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality constraints: (a*b)_k - M <= 0 for each active k
    A_rows = []
    b_rows = []
    for k in active_indices:
        row = np.zeros(n + 1)
        row[-1] = -1.0  # -M
        # Coefficients for b_i: a_{k-i} if 0 <= k-i < n
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            row[i] = a[k - i]
        A_rows.append(row)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        b_opt = res.x[:n]
        M_opt = res.x[-1]
        # Ensure non-negative
        b_opt = np.maximum(b_opt, 0)
        s = b_opt.sum()
        if s > 1e-10:
            b_opt = b_opt / s
        return b_opt, float(M_opt)
    return None, np.inf


def _lp_min_peak_vectorized(a, active_indices):
    """Vectorized version of the min-peak LP for speed."""
    n = len(a)
    m = len(active_indices)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Build A_ub efficiently
    A_ub = np.zeros((m, n + 1))
    A_ub[:, -1] = -1.0
    
    for row, k in enumerate(active_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # row[i] = a[k-i] for i in [i_lo, i_hi]
        idx = np.arange(i_lo, i_hi + 1)
        A_ub[row, idx] = a[k - idx]
    
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        b_opt = np.maximum(res.x[:n], 0)
        M_opt = float(res.x[-1])
        s = b_opt.sum()
        if s > 1e-10:
            b_opt = b_opt / s
        else:
            b_opt = np.ones(n) / n
        return b_opt, M_opt
    return None, np.inf


def _lp_max_sum(a, M, active_indices):
    """
    Original LP: maximize sum(b) subject to (a*b)_k <= M, b >= 0.
    """
    n = len(a)
    c_obj = -np.ones(n)  # minimize -sum(b)
    
    A_rows = []
    for k in active_indices:
        row = np.zeros(n)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        row[idx] = a[k - idx]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        b_opt = np.maximum(res.x, 0)
        return b_opt
    return None


def _coordinate_descent(a, deadline, max_sweeps=50, n_inner=40):
    """
    Coordinate descent to minimize max(a*a)/(sum(a))^2.
    a is modified in place (and a copy is returned for the best found).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    
    rng = np.random.default_rng(12345)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Random order for this sweep
        order = rng.permutation(n)
        
        for idx_pos in range(n):
            i = order[idx_pos]
            
            # Compute rest_k: convolution with a_i = 0
            a_save = a[i]
            a[i] = 0.0
            rest = _fft_convolve(a)
            a[i] = a_save
            
            S_rest = float(a.sum())  # sum with a_i = 0
            
            # Define g(x) = max_k f_k(x) / (S_rest + x)^2
            # where f_k(x) = rest_k + delta_k(x)
            # delta_{2i}(x) = x^2
            # delta_k(x) = 2*x*a_{k-i} if 0<=k-i<n and k!=2i
            # delta_k(x) = 0 otherwise
            
            # Precompute the linear coefficients
            # f_k(x) = rest_k + c_k * x + d_k * x^2
            # d_k = 1 if k == 2i, else 0
            # c_k = 2*a_{k-i} if 0<=k-i<n and k!=2i, else (a_{k-i} if k==2i, else 0)
            # Wait, when k=2i: f_k(x) = rest_k + x^2 (the x^2 comes from a_i*a_i)
            # When k!=2i and 0<=k-i<n: f_k(x) = rest_k + 2*a_{k-i}*x
            # Otherwise: f_k(x) = rest_k
            
            # For the max computation, we need max over k of these.
            # Let's precompute:
            # linear_coeffs[k] = 2*a_{k-i} if 0<=k-i<n and k!=2i
            # quadratic: only at k=2i, coefficient is 1
            
            k_2i = 2 * i
            # Linear coefficients for each k
            # For k in [0, 2n-2]:
            #   if 0 <= k-i < n and k != 2i: coeff = 2 * a[k-i]
            #   else: coeff = 0
            linear = np.zeros(2 * n - 1)
            k_lo = max(0, i)  # k-i >= 0 => k >= i
            k_hi = min(2 * n - 2, i + n - 1)  # k-i <= n-1 => k <= i+n-1
            for k in range(k_lo, k_hi + 1):
                if k != 2 * i:
                    linear[k] = 2.0 * a[k - i]
            
            # Objective: g(x) = max_k [(rest_k + linear[k]*x) + (x^2 if k==2i)] / (S_rest + x)^2
            
            # Golden section search for x in [0, x_max]
            x_max = max(2.0 * a_save, 1.0)  # reasonable upper bound
            
            def _g(x):
                # Compute f_k(x) for all k
                f = rest + linear * x
                if 0 <= k_2i < 2 * n - 1:
                    f[k_2i] += x * x
                denom = (S_rest + x) ** 2
                if denom < 1e-30:
                    return np.inf
                return f.max() / denom
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = x_max
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = _g(c)
            fd = _g(d)
            
            for _ in range(n_inner):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _g(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _g(d)
            
            x_opt = (lo + hi) / 2.0
            a[i] = x_opt
            
            # Check if improved
            s = a.sum()
            if s < 1e-10:
                a[i] = a_save
                continue
            c_full = _fft_convolve(a)
            sc = 2.0 * n * c_full.max() / (s * s)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
    
    return best_a

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        b_opt = np.maximum(res.x[:n], 0)
        M_opt = float(res.x[-1])
        s = b_opt.sum()
        if s > 1e-10:
            b_opt = b_opt / s
        return b_opt, M_opt
    return None, np.inf


def _lp_max_sum(a, M, active_indices):
    """Maximize sum(b) s.t. (a*b)_k <= M for active k, b>=0."""
    n = len(a)
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        A_ub[row, idx] = a[k - idx]
    
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0)
    return None


def _get_active_indices(c, top_k=80, margin=0.0):
    """Get indices of convolution values near the max."""
    m = c.max()
    threshold = m * (1.0 - margin)
    # Get top_k indices
    idx = np.argsort(c)[::-1][:top_k]
    # Also include neighbors
    active = set()
    for i in idx:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < len(c):
                active.add(j)
    return sorted(active)


def _lp_optimize(a, deadline, max_iter=5000, top_k=80, inflate=1.001):
    """Iterative LP optimization using min-peak LP."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        active = _get_active_indices(c, top_k=top_k)
        
        # Min-peak LP
        b_opt, M_lp = _lp_min_peak(a, active)
        
        if b_opt is None:
            no_imp += 10
            if no_imp > 200:
                break
            continue
        
        # Try blending
        best_blend_score = best_score
        best_blend = None
        
        for t in [1.0, 0.7, 0.5, 0.3]:
            a_try = (1.0 - t) * a + t * b_opt
            a_try = np.maximum(a_try, 0)
            a_try = _normalize(a_try)
            s_try = _score(a_try)
            if s_try < best_blend_score:
                best_blend_score = s_try
                best_blend = a_try
        
        if best_blend is not None and best_blend_score < best_score - 1e-12:
            best_score = best_blend_score
            best_a = best_blend.copy()
            a = best_blend
            no_imp = 0
        else:
            # Try max-sum LP as alternative
            M_infl = M * inflate
            b_ms = _lp_max_sum(a, M_infl, active)
            if b_ms is not None and b_ms.sum() > 1.0 + 1e-6:
                b_ms = _normalize(b_ms)
                for t in [1.0, 0.5, 0.3]:
                    a_try = (1.0 - t) * a + t * b_ms
                    a_try = np.maximum(a_try, 0)
                    a_try = _normalize(a_try)
                    s_try = _score(a_try)
                    if s_try < best_score - 1e-12:
                        best_score = s_try
                        best_a = a_try.copy()
                        a = a_try
                        no_imp = 0
                        break
            
            no_imp += 1
            if no_imp > 300:
                # Try increasing top_k
                top_k = min(200, top_k * 2)
                no_imp = 0
                if top_k > 200:
                    break
    
    return best_a


def _coordinate_descent(a, deadline, max_sweeps=100):
    """Coordinate descent to minimize max(a*a)/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    
    rng = np.random.default_rng(777)
    no_imp_sweeps = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                return best_a
            
            a_save = a[i]
            a[i] = 0.0
            rest = _fft_convolve(a)
            a[i] = a_save
            
            S_rest = float(a.sum())
            
            k_2i = 2 * i
            linear = np.zeros(2 * n - 1)
            if k_2i < 2 * n - 1:
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k != 2 * i:
                        linear[k] = 2.0 * a[k - i]
            
            # Golden section search
            x_max = max(3.0 * a_save, 2.0 * S_rest / n)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = x_max
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            
            # Evaluate g(c) and g(d)
            def _eval_g(x):
                f = rest + linear * x
                if 0 <= k_2i < 2 * n - 1:
                    f[k_2i] += x * x
                denom = (S_rest + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return float(f.max()) / denom
            
            fc = _eval_g(c_pt)
            fd = _eval_g(d_pt)
            
            for _ in range(30):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = _eval_g(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = _eval_g(d_pt)
            
            x_opt = (lo + hi) / 2.0
            a[i] = x_opt
            
            # Quick score check (every few coordinates)
            if (i % 64 == 0) or (i == 0):
                s = a.sum()
                if s < 1e-10:
                    a[i] = a_save
                    continue
                c_full = _fft_convolve(a)
                sc = 2.0 * n * c_full.max() / (s * s)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_imp_sweeps = 0
        else:
            no_imp_sweeps += 1
            if no_imp_sweeps > 10:
                break
    
    # Final check
    s = a.sum()
    if s > 1e-10:
        sc = 2.0 * n * _fft_convolve(a).max() / (s * s)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    return _normalize(a)


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_flat(n):
    return np.ones(n) / n


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    return _normalize(p)
