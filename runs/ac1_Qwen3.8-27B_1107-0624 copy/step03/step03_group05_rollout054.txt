    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _alternating_lp(a_init, deadline, max_iter=3000):
    """Alternating LP: alternate between fixing probe and optimizing a."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    probe = a.copy()
    no_imp = 0
    blend = 0.6

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_new, t_val = _he_alp_solve(probe, n)
        if a_new is None:
            break

        # Evaluate actual self-convolution score
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        # Try blended version too
        a_blend = (1.0 - blend) * a + blend * a_new
        a_blend = np.maximum(a_blend, 0.0)
        sb = a_blend.sum()
        if sb > 1e-10:
            a_blend = a_blend / sb
            c_blend = _fft_convolve(a_blend)
            score_blend = float(2.0 * n * c_blend.max())
        else:
            score_blend = np.inf

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            probe = a.copy()
            no_imp = 0
            blend = min(1.0, blend * 1.05)
        elif score_blend < best_score - 1e-12:
            best_score = score_blend
            best_a = a_blend.copy()
            a = a_blend.copy()
            probe = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            # Update probe toward the LP solution for progress
            probe = 0.6 * probe + 0.4 * a_new
            probe = _project_simplex(probe)
            a = a_blend.copy() if sb > 1e-10 else a_new.copy()

            if no_imp % 100 == 0:
                blend = max(0.3, blend * 0.95)
            if no_imp > 300:
                a = best_a.copy()
                probe = best_a.copy()
                no_imp = 0
                blend = 0.6

    return best_a


def _peak_chasing(a_init, deadline, max_iter=80000, top_k=3):
    """Fast gradient method targeting top-k convolution peaks exactly."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.04
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        peak_idx = np.argsort(c)[::-1][:top_k]

        # Compute gradient of max peak
        grad = np.zeros(n)
        for m in peak_idx:
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += 2.0 * a[m - idx]

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.95)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _spread_lp_polish(a_init, deadline, max_iter=2000):
    """Use the max-spread LP to push mass away from peaks."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    blend = 0.4
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * 1.0005
        b = _hp_max_spread(a, M, n)
        if b is None:
            break
        sb = b.sum()
        if sb <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 200:
                break
            continue

        b = b / sb
        a_new = (1.0 - blend) * a + blend * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
            blend = min(0.8, blend * 1.02)
        else:
            no_imp += 1
            a = a_new.copy()
            if no_imp % 50 == 0:
                blend = max(0.15, blend * 0.93)
            if no_imp > 300:
                a = best_a.copy()
                blend = 0.4
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, center_dist=0.2, width=0.12):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - (0.5 - center_dist)) / width) ** 2) + \
        np.exp(-0.5 * ((t - (0.5 + center_dist)) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_stepped_profile(n, n_steps=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i in range(n_steps):
        mask = (t >= i / n_steps) & (t < (i + 1) / n_steps)
        a[mask] = 1.0 + 0.5 * np.abs(np.sin(i * np.pi / n_steps))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: Alternating LP + Peak-chasing gradient +
    Spread LP, with diverse initializations and multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [384, 512, 768, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))
    # Sort: parent length first, then by descending priority
    lengths.sort(key=lambda x: (0 if parent is not None and x == parent.size else 1, x))

    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, effective deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: For each length, try initializations with alternating LP
    for n_t in lengths:
        if time.monotonic() > deadline - 100:
            break

        remaining = deadline - time.monotonic() - 60.0
        if remaining < 30:
            break

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_u_profile(n_t, 2.0))
        inits.append(_make_u_profile(n_t, 4.0))
        inits.append(_make_u_profile(n_t, 1.0))
        inits.append(_make_linear_profile(n_t, 2.0))
        inits.append(_make_bimodal(n_t, 0.2, 0.12))
        inits.append(_make_cosine_profile(n_t, 1, 0.4))

        # Random smooth init
        a_rand = np.maximum(rng.normal(1.0, 0.3, n_t), 0.01)
        inits.append(a_rand / a_rand.sum())

        num_inits = len(inits)
        per_init_time = max(15.0, min(60.0, remaining / num_inits / 2.0))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 50:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Time allocation for this init
            t_alloc = min(per_init_time, deadline - time.monotonic() - 40.0)
            if t_alloc < 10:
                break

            # Step 1: Alternating LP (main workhorse for n <= 1024)
            if n_t <= 1024:
                t_lp = min(t_alloc * 0.5, deadline - time.monotonic() - 30.0)
                if t_lp > 5:
                    a_lp = _alternating_lp(a0, time.monotonic() + t_lp,
                                           max_iter=2000)
                    sc_lp = _score(a_lp)
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_seq = a_lp.copy()
                        print(f"  n={n_t}, init{idx}: LP score={sc_lp:.10f}")
                    a_current = a_lp
                else:
                    a_current = a0
            else:
                a_current = a0

            # Step 2: Spread LP polish
            t_sp = min(t_alloc * 0.2, deadline - time.monotonic() - 20.0)
            if t_sp > 5:
                a_sp = _spread_lp_polish(a_current, time.monotonic() + t_sp,
                                         max_iter=1000)
                sc_sp = _score(a_sp)
                if sc_sp < best_score:
                    best_score = sc_sp
                    best_seq = a_sp.copy()
                    print(f"  n={n_t}, init{idx}: SpreadLP score={sc_sp:.10f}")
                a_current = a_sp
            else:
                pass

            # Step 3: Peak-chasing gradient
            t_gc = min(t_alloc * 0.3, deadline - time.monotonic() - 15.0)
            if t_gc > 3:
                a_gc = _peak_chasing(a_current, time.monotonic() + t_gc,
                                     max_iter=50000, top_k=3)
                sc_gc = _score(a_gc)
                if sc_gc < best_score:
                    best_score = sc_gc
                    best_seq = a_gc.copy()
                    print(f"  n={n_t}, init{idx}: PeakChase score={sc_gc:.10f}")

    # Phase 2: Intensive refinement on the best found
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 25.0
        print(f"\nFinal refinement: n={n_best}, budget={final_budget:.0f}s")

        # 2a: Alternating LP
        t_a = min(final_budget * 0.3, 300.0)
        if t_a > 10 and n_best <= 1024:
            a_r = _alternating_lp(best_seq, time.monotonic() + t_a, max_iter=3000)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final ALP: {sc:.10f}")

        # 2b: Spread LP
        t_b = min(final_budget * 0.2, 150.0)
        if t_b > 5:
            a_r = _spread_lp_polish(best_seq, time.monotonic() + t_b, max_iter=2000)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final SpreadLP: {sc:.10f}")

        # 2c: Peak-chasing with varying top_k
        t_c = min(final_budget * 0.25, 200.0)
        if t_c > 5:
            for tk in [2, 3, 5]:
                if time.monotonic() > deadline - 20:
                    break
                t_sub = min(t_c / 3, deadline - time.monotonic() - 15.0)
                if t_sub < 3:
                    break
                a_r = _peak_chasing(best_seq, time.monotonic() + t_sub,
                                    max_iter=30000, top_k=tk)
                sc = _score(a_r)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_r.copy()
                print(f"  PeakChase k={tk}: {sc:.10f}")

        # 2d: One more LP round
        t_d = min(final_budget * 0.15, 100.0)
        if t_d > 5 and n_best <= 1024:
            a_r = _alternating_lp(best_seq, time.monotonic() + t_d, max_iter=1500)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final ALP2: {sc:.10f}")

        # 2e: Final peak-chase
        t_e = deadline - time.monotonic() - 15.0
        if t_e > 5:
            a_r = _peak_chasing(best_seq, time.monotonic() + t_e,
                                max_iter=20000, top_k=3)
            sc = _score(a_r)
            if sc < best_score:
                best_score = sc
                best_seq = a_r.copy()
            print(f"  Final PeakChase: {sc:.10f}")

    # Safety: ensure we have a valid sequence
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(768, 2.0)

    # Validate with authoritative function
    if final_seq.size <= 5000:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
