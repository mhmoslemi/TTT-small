"""
Optimized step function search using L-BFGS with LSE smoothing,
annealing, LP refinement, and diverse initialization.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lse_objective_and_grad(z, T, n):
    """Compute LSE-smoothed objective and gradient in logit space."""
    # Softmax to get a
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    a = e / e.sum()
    
    # Convolution
    c = _fft_convolve(a)
    
    # LSE
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    
    # LSE value (for objective)
    lse_val = m_val + T * np.log(s)
    obj = 2.0 * n * lse_val
    
    # Gradient of LSE w.r.t. a: 2 * cross_correlation(w, a)
    N = _next_pow2(2 * n - 1)
    N2 = _next_pow2(3 * n - 2)  # for conv of w (len 2n-1) and a (len n)
    # Actually, let's use the simpler formula:
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (k-i in [0,n-1])
    # This is the cross-correlation of w and a
    # = conv(w, a[::-1])[n-1 : 2n-1]
    Nw = _next_pow2((2*n-1) + n - 1)
    fw = np.fft.rfft(w, Nw)
    fa_rev = np.fft.rfft(a[::-1], Nw)
    h_full = np.fft.irfft(fw * fa_rev, Nw)
    grad_a = 2.0 * h_full[n-1:2*n-1]
    
    # Gradient w.r.t. z (softmax parameterization)
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z


def _lbfgs_lse(a0, deadline, T_start=0.1, T_end=1e-4, n_restarts=5):
    """Optimize using L-BFGS on LSE-smoothed objective in logit space."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    z = np.log(a0)
    z -= float(np.mean(z))
    
    best_a = a0.copy()
    best_score = _score_normalized(a0)
    
    T = T_start
    n_total_steps = 0
    
    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break
        
        # L-BFGS optimization at current T
        result = minimize(
            lambda z: _lse_objective_and_grad(z, T, n)[0],
            z,
            jac=lambda z: _lse_objective_and_grad(z, T, n)[1],
            method='L-BFGS-B',
            options={'maxiter': 500, 'gtol': 1e-6, 'maxls': 50}
        )
        
        n_total_steps += result.nit if hasattr(result, 'nit') else 0
        
        # Update best
        z_best = result.x
        z_shifted = z_best - np.max(z_best)
        a_new = np.exp(z_shifted) / np.exp(z_shifted).sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_best.copy()
        else:
            # Keep old z, perturb slightly
            z = z + 0.01 * np.random.default_rng(restart).normal(size=n)
        
        # Decrease T
        T *= 0.5
        T = max(T, T_end)
        
        print(f"    L-BFGS restart {restart}: T={T:.6f}, score={best_score:.10f}, func_calls={result.nfev}")
    
    return best_a, best_score


def _lp_refine(a, deadline, max_iter=2000, M_inflate=1.001):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    inflate = M_inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        # Build LP: maximize sum(b) s.t. (a*b)[k] <= M_lp, b >= 0
        # (a*b)[k] = sum_j a[j] * b[k-j] for valid j
        # Constraint matrix: row k, col j = a[j] if 0 <= k-j < n else 0
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j+n, j] = a[j]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            inflate = min(1.01, inflate * 1.3)
            no_imp += 5
            if no_imp > 200:
                break
            continue
        
        g = res.x
        sum_g = float(g.sum())
        
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.01, inflate * 1.1)
            if no_imp > 300:
                break
            continue
        
        g = g / sum_g
        
        # Line search for blend factor
        a_current = a.copy()
        best_t = 0.0
        best_local = _score_normalized(a_current)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1 - t) * a_current + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score_normalized(a_try)
            if sc < best_local:
                best_local = sc
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = _score_normalized(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(1.005, inflate * 0.9)
            if no_imp > max(200, n):
                a = best_a.copy()
                no_imp = 0
        
        a = a_new
    
    return best_a, best_score


def _make_raised_cosine(n):
    t = np.linspace(0, 2*np.pi, n, endpoint=False)
    a = 0.5 * (1 + np.cos(t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.25):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_super_gaussian(n, p=4):
    """Super-Gaussian: flat top with smooth edges."""
    t = np.linspace(-1, 1, n)
    a = np.exp(-np.abs(t) ** p)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_smooth_bump(n, width=0.6, order=3):
    """Compact bump with smooth edges using polynomial."""
    t = np.linspace(0, 1, n)
    x = (t - 0.5) / (0.5 * width)
    a = np.where(np.abs(x) < 1, (1 - x**2)**order, 0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with LSE-smoothed objective and geometric annealing in logit space,
combined with LP-based peak spreading and diverse smooth-profile initialization.
Exploits the fact that non-negative sequences can have self-convolution peaks
away from center, enabling sub-2 scores.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.ndimage import gaussian_filter1d


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lse_compute(z, T, n):
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    a = e / e.sum()
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    lse_val = m_val + T * np.log(s)
    obj = 2.0 * n * lse_val
    Nw = _next_pow2((2 * n - 1) + n)
    fw = np.fft.rfft(w, Nw)
    fa_rev = np.fft.rfft(a[::-1], Nw)
    h_full = np.fft.irfft(fw * fa_rev, Nw)
    grad_a = 2.0 * h_full[n - 1:2 * n - 1]
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    return obj, grad_z


def _obj_wrapper(z, T, n):
    val, _ = _lse_compute(z, T, n)
    return val


def _grad_wrapper(z, T, n):
    _, g = _lse_compute(z, T, n)
    return g

def _subgradient_polish(a, deadline, max_iter=30000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    momentum = 0.7
    lr = 0.02
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-10)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        Nw = _next_pow2((2 * n - 1) + n)
        fw = np.fft.rfft(w, Nw)
        fa_rev = np.fft.rfft(a[::-1], Nw)
        h_full = np.fft.irfft(fw * fa_rev, Nw)
        grad = 2.0 * h_full[n - 1:2 * n - 1]
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:

            no_imp = 0
            a = a_new

            no_imp += 1

            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.02
                no_imp = 0

    return best_a, best_score


def _make_raised_cosine(n):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                print(f"Parent: score={sc:.10f}, n={parent.size}")
                if sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
    except Exception:
        parent = None

    lengths = [1024]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    lengths.append(2048)
    lengths = sorted(set(lengths))
    print(f"Lengths to try: {lengths}")

    # Phase 1: L-BFGS + LSE annealing on diverse profiles
    print("\n=== Phase 1: L-BFGS + LSE annealing ===")
    best_per_length = {}

    for n_t in lengths:
        time_left = deadline - time.monotonic()
        if time_left < 200:
            break
        print(f"\n--- n={n_t}, time_left={time_left:.0f}s ---")

        candidates = []
        if parent is not None:
            candidates.append(_interpolate_profile(parent, n_t))
            pert = _interpolate_profile(parent, n_t)
            pert = np.maximum(pert * (1.0 + 0.08 * rng.normal(size=n_t)), 0)
            candidates.append(pert / pert.sum())
            pert2 = _interpolate_profile(parent, n_t)
            pert2 = np.maximum(pert2 * (1.0 + 0.15 * rng.normal(size=n_t)), 0)
            candidates.append(pert2 / pert2.sum())

        candidates.append(_make_raised_cosine(n_t))
        candidates.append(_make_gaussian(n_t, 0.3))
        candidates.append(_make_gaussian(n_t, 0.4))
        candidates.append(_make_super_gaussian(n_t, 4))
        candidates.append(_make_super_gaussian(n_t, 8))
        candidates.append(_make_smooth_bump(n_t, 0.8, 3))
        candidates.append(_make_smooth_bump(n_t, 1.0, 2))
        candidates.append(_make_two_peaks(n_t, 0.15))
        candidates.append(_random_smooth(n_t, rng, 5.0))
        candidates.append(_random_smooth(n_t, rng, 10.0))
        candidates.append(_random_smooth(n_t, rng, 2.0))

        best_here = None
        best_here_score = np.inf
        n_cands = len(candidates)
        time_per_candidate = (time_left - 150) / n_cands

        for ci, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 150:
                break
            sc0 = _score_normalized(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            alloc = min(time_per_candidate, deadline - time.monotonic() - 150)
            if alloc < 10:
                break
            lbfgs_end = time.monotonic() + alloc
            a_opt, sc_opt = _lbfgs_anneal(a0, lbfgs_end,
                                          T_start=0.1, T_end=0.001,
                                          n_restarts=4)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
            if sc_opt < best_here_score:
                best_here_score = sc_opt
                best_here = a_opt
            print(f"  cand {ci}: {sc0:.8f} -> {sc_opt:.10f} (best={best_score:.10f})")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best n={n_t}: {best_here_score:.10f}")

    # Phase 2: LP refinement on best per length
    print("\n=== Phase 2: LP refinement ===")
    for n_t in sorted(best_per_length.keys(), reverse=True):
        time_left = deadline - time.monotonic()
        if time_left < 120:
            break
        a_in = best_per_length[n_t]
        lp_end = min(time.monotonic() + min(300, time_left - 60), deadline - 60)
        a_lp, sc_lp = _lp_refine(a_in, lp_end, max_iter=1500, M_inflate_start=1.005)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP n={n_t}: {sc_lp:.10f} (global best={best_score:.10f})")

    # Phase 3: Subgradient polish
    if best_seq is not None:
        time_left = deadline - time.monotonic()
        if time_left > 60:
            print(f"\n=== Phase 3: Subgradient polish ===")
            sp_end = min(time.monotonic() + min(120, time_left - 30), deadline - 30)
            a_sp, sc_sp = _subgradient_polish(best_seq, sp_end, max_iter=50000)
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgradient: {sc_sp:.10f} (best={best_score:.10f})")

    # Phase 4: Final LP with tighter inflation
    if best_seq is not None:
        time_left = deadline - time.monotonic()
        if time_left > 30:
            print(f"\n=== Phase 4: Final tight LP ===")
            lp_end = min(time.monotonic() + min(80, time_left - 15), deadline - 15)
            a_fin, sc_fin = _lp_refine(best_seq, lp_end, max_iter=800, M_inflate_start=1.001)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
            print(f"  Final LP: {sc_fin:.10f} (best={best_score:.10f})")

    # Phase 5: Quick L-BFGS at very low T
    if best_seq is not None:
        time_left = deadline - time.monotonic()
        if time_left > 20:
            print(f"\n=== Phase 5: Low-T L-BFGS ===")
            lb_end = min(time.monotonic() + min(50, time_left - 10), deadline - 10)
            a_low, sc_low = _lbfgs_anneal(best_seq, lb_end,
                                          T_start=0.02, T_end=0.0005,
                                          n_restarts=3)
            if sc_low < best_score:
                best_score = sc_low
                best_seq = a_low.copy()
            print(f"  Low-T L-BFGS: {sc_low:.10f} (best={best_score:.10f})")

    # Final validation and return
    if best_seq is None:
        best_seq = _make_raised_cosine(1024)
        best_score = _score_normalized(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(1024) / 1024.0
    else:
        final_seq = final_seq / s

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
