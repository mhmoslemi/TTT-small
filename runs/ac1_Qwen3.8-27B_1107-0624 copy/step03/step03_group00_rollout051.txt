def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(p):

    s = float(p.sum())

def _subgradient_at_max(c, a):

    Returns the subgradient vector (length n) and the active set."""

    eps = 1e-10 * max(1.0, c.max())

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {a: a >= 0, sum(a) = target_sum}."""
    n = len(v)

        return np.zeros(n)
    chap = float(cssv[rho[-1]] - target_sum) / (rho[-1] + 1)
    return np.maximum(v - chap, 0.0)

def _make_profile(n, profile_type, **params):
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'uniform':
        a = np.ones(n)
    elif profile_type == 'linear':
        eta = params.get('eta', 1.5)
        a = 1.0 + eta * (t - 0.5)
    elif profile_type == 'quadratic':
        a = 1.0 + params.get('amp', 1.0) * (2*t - 1)**2
    elif profile_type == 'cubic':
        a = 1.0 + params.get('a1', 1.5) * (2*t-1) + params.get('a3', -1.0) * (2*t-1)**3
    elif profile_type == 'gaussian':
        c = params.get('center', 0.5)
        s = params.get('sigma', 0.3)
        a = np.exp(-0.5*((t-c)/s)**2)
    elif profile_type == 'two_gaussians':
        c1 = params.get('c1', 0.3)
        c2 = params.get('c2', 0.7)
        s = params.get('sigma', 0.15)
        a = np.exp(-0.5*((t-c1)/s)**2) + np.exp(-0.5*((t-c2)/s)**2)
    elif profile_type == 'cosine':
        freq = params.get('freq', 1)
        amp = params.get('amp', 0.8)
        a = 1.0 + amp * np.cos(2*np.pi*freq*t)
    elif profile_type == 'power':
        p = params.get('p', 2.0)
        a = t**p + (1-t)**p + 1e-6
    elif profile_type == 'bell':
        a = np.exp(-params.get('sharp', 5.0) * (t - 0.5)**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return np.maximum(p0, 1e-15) / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
