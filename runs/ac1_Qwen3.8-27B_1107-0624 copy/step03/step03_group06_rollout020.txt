"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov accelerated gradient (smooth max), LP refinement, and coordinate descent.
Key improvements: adaptive beta scheduling, backtracking line search, sparse LP,
coordinate-wise 1D optimization, and diverse initializations including cosine/gaussian windows.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_parts(a):
    """Returns (score, c, max_c, sum_a) for internal use."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf, None, 0.0, 0.0
    c = _fft_convolve(a)
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s)), c, mc, s


def _smooth_max_grad(a, beta):
    """Compute gradient of log-sum-exp smooth max of conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval, float(beta * (np.log(s)))


def _project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # First ensure non-negativity, then normalize
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return v / s


def _project_simplex_exact(v):
    """Exact projection onto simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_cosine(n):
    t = np.linspace(0.0, np.pi, n)
    a = np.cos(t / 2.0) ** 2 + 1e-8
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.4):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2) + 1e-8
    return a / a.sum()


def _make_triangular(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(0.0, 1.0 - 2.0 * np.abs(t - 0.5)) + 1e-8
    return a / a.sum()


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()


def _make_bell(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + ((t - 0.5) / width) ** 2) + 1e-8
    return a / a.sum()


def _interpolate_to(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _gradient_descent(a0, deadline, max_iter=50000, beta_start=None, lr_start=0.05):
    """Nesterov accelerated gradient descent with adaptive beta and backtracking."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    sc, c, mc, S = _score_parts(a)
    best_score = sc

    if beta_start is None:
        beta = max(1e-6, 0.1 * mc)
    else:
        beta = beta_start

    lr = lr_start
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        sl = a_look.sum()
        if sl < 1e-10:
            break
        a_look = a_look / sl

        grad, mval, logsum = _smooth_max_grad(a_look, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 5.0)
            v = np.zeros(n)
            continue

        # Gradient of the full objective: 2n * (grad*logsum - 2*mval) / S^3
        # Simplified: we minimize max(c)/S^2, gradient w.r.t. a is (grad - 2*max_c/s * a) * 2n / s^2
        # For the smooth version: (grad*beta - 2*mval/s*a) direction (dropping constants)
        S_cur = float(a.sum())
        grad_obj = grad * beta - 2.0 * mval / S_cur * a

        # Backtracking line search
        a_new = a - lr * grad_obj
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / sn

        sc_new = _score(a_new)

        # Accept if improved
        if sc_new < sc - 1e-15:
            # Update momentum
            v = momentum * v - lr * grad_obj
            sc = sc_new
            a = a_new
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
            # Adaptive lr
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            # Decrease beta slowly
            if it % 50 == 49:
                beta = max(1e-8, beta * 0.97)
        else:
            lr = lr * 0.7
            no_imp += 1
            if no_imp % 500 == 0:
                lr = base_lr
                v = np.zeros(n)
                beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            if no_imp > max(5000, 10 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
                no_imp = 0

    return best_a


def _build_lp_matrix(f, n):
    """Build sparse constraint matrix for LP."""
    # A has shape (2n-1, n), A[j:j+n, j] = f
    rows = []
    cols = []
    vals = []
    for j in range(n):
        r_start = j
        r_end = j + n
        rows.extend(range(r_start, r_end))
        cols.extend([j] * n)
        vals.extend(f)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_refine(a0, deadline, max_iter=20000, inflate_start=1.003, inflate_min=1.0005):
    """LP-based refinement with warm start and adaptive inflate."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    A = _build_lp_matrix(a, n)
    c_obj = -np.ones(n)

    no_imp = 0
    inflate = inflate_start
    prev_x = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        b_ub = np.full(2 * n - 1, M_lp)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})

        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())

        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > 200:
                # Reset
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate > 1.001:
                    break
            continue

        g = g / sum_g

        # Line search for blend parameter
        best_t = 0.0
        best_sc = best_score
        for t in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                break
            a_new = a_new / sn
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.1)
            if no_imp > max(100, n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate > 1.0015:
                    break

    return best_a


def _coord_descent_1d(a, i, c, S, n):
    """Find optimal value for coordinate i given current convolution c and sum S."""
    # c[k] changes as: c_new[k] = c[k] + 2*delta*a[k-i] for k in [i, i+n-1], k != 2i
    # c_new[2i] = c[2i] + 2*delta*a[i] + delta^2 (if 2i < 2n-1)
    # S_new = S + delta

    def obj(delta):
        a_new_i = a[i] + delta
        if a_new_i < 0:
            return np.inf
        S_new = S + delta
        if S_new < 1e-10:
            return np.inf
        # Compute new max correlation
        # Start with current c
        max_c = c.max()
        # Update affected positions
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        for k in range(k_lo, k_hi + 1):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    new_val = c[k] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_val = c[k] + 2.0 * delta * a[ki]
                if new_val > max_c:
                    max_c = new_val
        return 2.0 * n * max_c / (S_new * S_new)

    # Search over delta in [-a[i], upper_bound]
    lo = -a[i]
    # Upper bound: delta such that S_new is reasonable
    hi = max(10.0 * a[i] + S / n, 1.0)
    # Maybe try larger
    hi = max(hi, S * 2.0 / n)

    res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-10})
    if res.success:
        new_val = a[i] + res.x
        return max(0.0, new_val)
    return a[i]


def _coordinate_descent(a0, deadline, max_passes=500):
    """Cyclic coordinate descent with 1D optimization."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    rng = np.random.default_rng(12345)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Shuffle coordinates
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            new_val = _coord_descent_1d(a, i, c, S, n)
            if new_val != a[i]:
                # Update convolution incrementally
                delta = new_val - a[i]
                a[i] = new_val
                S += delta
                # Update c
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            c[k] += 2.0 * (a[i] - delta) * (a[i] - delta) - c[k] + 2.0 * delta * (a[i] - delta) + delta * delta
                        else:
                            c[k] += 2.0 * delta * a[ki]
                # Actually, let's just recompute c for correctness (expensive but safe)
                # For now, recompute every 10 coordinates
                if i % 10 == 0 or i == n - 1:
                    c = _fft_convolve(a)
                    S = float(a.sum())

        # Recompute at end of pass
        c = _fft_convolve(a)
        S = float(a.sum())
        if S < 1e-10:
            break
        a = a / S
        c = _fft_convolve(a)
        S = float(a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

    return best_a


def _fast_coord_descent(a0, deadline, max_passes=200):
    """Faster coordinate descent - only update top contributors to peaks."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    c = _fft_convolve(a)
    mc = float(c.max())
    best_score = float(2.0 * n * mc)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        S = float(a.sum())
        mc = float(c.max())
        sc = 2.0 * n * mc / (S * S)

        # Find peak positions
        peak_thresh = mc * 0.95
        peak_positions = np.where(c >= peak_thresh)[0]

        # Find which coordinates contribute most to peaks
        contrib = np.zeros(n)
        for k in peak_positions:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                ki = k - j
                if 0 <= ki < n:
                    contrib[j] += a[ki]

        # Update top contributors
        top_k = max(10, n // 10)
        top_indices = np.argsort(contrib)[::-1][:top_k]

        for i in top_indices:
            new_val = _coord_descent_1d(a, i, c, S, n)
            if new_val != a[i] and abs(new_val - a[i]) > 1e-12:
                a[i] = new_val

        # Renormalize
        S = float(a.sum())
        if S < 1e-10:
            break
        a = a / S

        # Check improvement
        c = _fft_convolve(a)
        mc = float(c.max())
        sc = 2.0 * n * mc / (S * S)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        elif p > 50:
            # No improvement in a while, try more aggressive
            a = best_a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: gradient -> LP -> fine gradient -> coordinate descent.
    Tries multiple initializations and sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations
    lengths = [512, 768, 1024]
    initializations = []

    for n_t in lengths:
        if parent is not None:
            initializations.append(('parent', n_t, _interpolate_to(parent, n_t)))
        initializations.append(('cosine', n_t, _make_cosine(n_t)))
        initializations.append(('gaussian', n_t, _make_gaussian(n_t, 0.35)))
        initializations.append(('gaussian_w', n_t, _make_gaussian(n_t, 0.45)))
        initializations.append(('bell', n_t, _make_bell(n_t, 0.3)))
        initializations.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        initializations.append(('tri', n_t, _make_triangular(n_t)))

    # Score and sort
    scored = []
    for name, n_t, a0 in initializations:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Top initializations:")
    for sc, name, n_t, _ in scored[:5]:
        print(f"  {name} n={n_t}: {sc:.8f}")

    # Phase 1: Quick gradient descent on top candidates
    phase1_deadline = t_start + 400.0
    candidates_after_grad = []

    for sc0, name, n_t, a0 in scored[:6]:
        if time.monotonic() > phase1_deadline:
            break
        if sc0 > best_score + 0.05:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        rem = phase1_deadline - time.monotonic()
        if rem < 20:
            break
        gd_deadline = time.monotonic() + min(rem * 0.8, 120.0)
        print(f"Grad: {name} n={n_t} init={sc0:.8f} budget={gd_deadline - time.monotonic():.0f}s")
        a_g = _gradient_descent(a0, gd_deadline, max_iter=30000)
        sc_g = _score(a_g)
        if np.isfinite(sc_g):
            candidates_after_grad.append((sc_g, name, n_t, a_g))
            if sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
                print(f"  -> {sc_g:.10f} (NEW BEST)")
            else:
                print(f"  -> {sc_g:.8f}")

    # Sort candidates by score
    if candidates_after_grad:
        candidates_after_grad.sort(key=lambda x: x[0])

    # Phase 2: LP refinement on best candidates
    phase2_deadline = t_start + 1000.0
    lp_candidates = candidates_after_grad[:3] if candidates_after_grad else \
        [(scored[0][0], scored[0][1], scored[0][2], scored[0][3])]

    for sc0, name, n_t, a0 in lp_candidates:
        if time.monotonic() > phase2_deadline:
            break
        rem = phase2_deadline - time.monotonic()
        if rem < 30:
            break
        lp_deadline = time.monotonic() + min(rem * 0.7, 250.0)
        print(f"LP: {name} n={n_t} init={sc0:.8f} budget={lp_deadline - time.monotonic():.0f}s")
        a_lp = _lp_refine(a0, lp_deadline, max_iter=10000, inflate_start=1.003)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp):
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  -> {sc_lp:.10f} (NEW BEST)")
            else:
                print(f"  -> {sc_lp:.8f}")

    # Phase 3: Fine gradient on the best
    if best_seq is not None:
        phase3_deadline = t_start + 1400.0
        if time.monotonic() < phase3_deadline:
            rem = phase3_deadline - time.monotonic()
            if rem > 30:
                gd_deadline = time.monotonic() + min(rem * 0.8, 200.0)
                mc = float(_fft_convolve(best_seq).max())
                print(f"Fine grad: budget={gd_deadline - time.monotonic():.0f}s")
                a_fg = _gradient_descent(best_seq, gd_deadline, max_iter=50000,
                                         beta_start=0.02 * mc, lr_start=0.02)
                sc_fg = _score(a_fg)
                if np.isfinite(sc_fg):
                    if sc_fg < best_score:
                        best_score = sc_fg
                        best_seq = a_fg.copy()
                        print(f"  -> {sc_fg:.10f} (NEW BEST)")
                    else:
                        print(f"  -> {sc_fg:.8f}")

    # Phase 4: Coordinate descent
    if best_seq is not None:
        phase4_deadline = t_start + budget_s - 40.0
        if time.monotonic() < phase4_deadline:
            rem = phase4_deadline - time.monotonic()
            if rem > 30:
                cd_deadline = time.monotonic() + min(rem * 0.85, 300.0)
                print(f"Coord descent: budget={cd_deadline - time.monotonic():.0f}s")
                a_cd = _fast_coord_descent(best_seq, cd_deadline, max_passes=200)
                sc_cd = _score(a_cd)
                if np.isfinite(sc_cd):
                    if sc_cd < best_score:
                        best_score = sc_cd
                        best_seq = a_cd.copy()
                        print(f"  -> {sc_cd:.10f} (NEW BEST)")
                    else:
                        print(f"  -> {sc_cd:.8f}")

    # Phase 5: Final LP + gradient squeeze
    if best_seq is not None and time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic()
        if rem > 20:
            # Quick LP
            lp_deadline = time.monotonic() + min(rem * 0.5, 80.0)
            a_sl = _lp_refine(best_seq, lp_deadline, max_iter=5000,
                             inflate_start=1.002, inflate_min=1.0003)
            sc_sl = _score(a_sl)
            if np.isfinite(sc_sl) and sc_sl < best_score:
                best_score = sc_sl
                best_seq = a_sl.copy()
                print(f"  Final LP: {sc_sl:.10f} (NEW BEST)")

            # Quick gradient
            if time.monotonic() < deadline - 5:
                gd_deadline = deadline - 2
                mc = float(_fft_convolve(best_seq).max())
                a_sg = _gradient_descent(best_seq, gd_deadline, max_iter=20000,
                                         beta_start=0.01 * mc, lr_start=0.01)
                sc_sg = _score(a_sg)
                if np.isfinite(sc_sg) and sc_sg < best_score:
                    best_score = sc_sg
                    best_seq = a_sg.copy()
                    print(f"  Final grad: {sc_sg:.10f} (NEW BEST)")

    # Fallback
    if best_seq is None:
        best_seq = _make_cosine(512)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_cosine(512)

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
