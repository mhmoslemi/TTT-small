"""
Optimized step function coefficient search using adaptive smooth-max Adam with
peak-directed subgradient refinement, diverse initialization strategies, and
annealing schedules. Targets score < 1.5027 by minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key improvements: power-law initializations, faster beta annealing, multi-peak
subgradient, and careful time management with multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.025, beta_factor=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    beta_decay = 0.75

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        gnorm = float(np.linalg.norm(grad_z))
        if not np.all(np.isfinite(grad_z)) or gnorm < 1e-15:
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.92)
            if no_imp > max(3000, 4 * n):
                rng_r = np.random.default_rng(5000 + it * 7)
                z = best_z + 0.4 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue
        z = z_new
        a = a_new
        if it % 100 == 0:
            beta = max(1e-12, beta * beta_decay)
            beta_decay = min(0.98, beta_decay * 1.005)
    return best_a, best_score


def _multi_peak_subgradient(a, deadline, max_iter=10000, top_k=5):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])
        weights = np.exp(-0.5 * np.array([(c_curr[top_indices[i]] - max_val) / (max_val * 0.1 + 1e-10) for i in range(k)]))
        weights = weights / weights.sum()
        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break
        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break
        a = a_new
    return best_a, best_score


def _lp_refine(a, deadline, max_iter=2000, inflate=1.0008, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    c_obj = -np.ones(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M_lp)
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.04)
            if no_imp > 100:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.005, inflate_cur * 0.96)
            if no_imp > max(150, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new
    return best_a


def _make_power_profile(n, p=1.0):
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.sin(0.5 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_skew_gaussian(n, mu=0.7, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - mu) ** 2) / (2.0 * sigma ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _adam_with_schedule(a, deadline, max_iter=30000):
    """Adam with aggressive annealing and periodic sharpening."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    beta = 0.06 * float(c.max())
    no_imp = 0
    lr_cur = 0.03
    base_lr = 0.03

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        gnorm = float(np.linalg.norm(grad_z))
        if not np.all(np.isfinite(grad_z)) or gnorm < 1e-15:
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr_cur = max(0.003, lr_cur * 0.94)
            if no_imp > max(2500, 3 * n):
                rng_r = np.random.default_rng(10000 + it * 13)
                z = best_z + 0.5 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, 0.06 * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue
        z = z_new
        a = a_new
        if it % 80 == 0:
            if it < max_iter * 0.5:
                beta = max(1e-12, beta * 0.7)
            else:
                beta = max(1e-12, beta * 0.88)
    return best_a, best_score


def _full_optimize(a0, deadline, label=""):
    """Full optimization pipeline on a0."""
    a0 = np.asarray(a0, dtype=np.float64).copy()
    n = a0.size
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-10:
        a0 = np.ones(n) / n
        s0 = 1.0
    a0 = a0 / s0
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f} (n={n})")

    best_a = a0.copy()
    best_sc = sc0
    elapsed = time.monotonic()

    # Phase A: Main Adam
    if time.monotonic() < deadline - 30:
        dA = min(deadline - 30, elapsed + 0.55 * (deadline - elapsed))
        a1, s1 = _adam_with_schedule(a0, dA, max_iter=40000)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam: {best_sc:.10f} (t={time.monotonic()-elapsed:.0f}s)")

    # Phase B: Multi-peak subgradient
    if time.monotonic() < deadline - 20:
        dB = min(deadline - 20, time.monotonic() + 0.2 * (deadline - time.monotonic()))
        a2, s2 = _multi_peak_subgradient(best_a, dB, max_iter=8000, top_k=5)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} MultiSG: {best_sc:.10f} (t={time.monotonic()-elapsed:.0f}s)")

    # Phase C: LP refine (only if n <= 512 for speed)
    if best_a.size <= 512 and time.monotonic() < deadline - 15:
        dC = min(deadline - 15, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        a3 = _lp_refine(best_a, dC, max_iter=1500, inflate=1.001, blend=0.25)
        s3 = _score(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} LP: {best_sc:.10f} (t={time.monotonic()-elapsed:.0f}s)")

    # Phase D: Fine Adam
    if time.monotonic() < deadline - 8:
        dD = deadline - 8
        a4, s4 = _adam_smooth_max(best_a, dD, max_iter=15000, lr=0.008, beta_factor=0.015)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} FineAdam: {best_sc:.10f} (t={time.monotonic()-elapsed:.0f}s)")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for minimum peak auto-convolution sequences.
    Combines diverse initializations (power-law, cosine, skew-Gaussian profiles),
    adaptive smooth-max Adam with aggressive annealing, multi-peak subgradient,
    and LP refinement. Explores n=512 and n=1024 with careful time management.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare initializations
    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
    else:
        p512 = _make_power_profile(512, 1.0)
        p1024 = _make_power_profile(1024, 1.0)

    # === Phase 1: Quick screening at n=512 (first 20% of time) ===
    phase1_deadline = t_start + 0.2 * (budget_s - 30)
    print(f"=== Phase 1: Screening (deadline at t={phase1_deadline - t_start:.0f}s) ===")

    candidates_512 = [
        ("parent512", p512),
        ("power1.0", _make_power_profile(512, 1.0)),
        ("power1.5", _make_power_profile(512, 1.5)),
        ("power0.7", _make_power_profile(512, 0.7)),
        ("power2.0", _make_power_profile(512, 2.0)),
        ("cosine", _make_cosine_ramp(512, 2.0)),
        ("skew_gauss", _make_skew_gaussian(512, 0.75, 0.35)),
        ("linear", _make_linear_profile(512, 3.0)),
    ]

    if parent is not None:
        p_pert = p512 * (1.0 + 0.2 * rng.normal(size=512))
        p_pert = np.maximum(p_pert, 0.0)
        candidates_512.append(("parent_pert", p_pert / p_pert.sum()))

    quick_results = []
    per_cand_time = max(8.0, (phase1_deadline - time.monotonic()) / len(candidates_512) - 2)

    for name, a0 in candidates_512:
        if time.monotonic() > phase1_deadline - 5:
            break
        d = min(phase1_deadline - 5, time.monotonic() + per_cand_time)
        a_b, s_b = _adam_smooth_max(a0, d, max_iter=3000, lr=0.03, beta_factor=0.05)
        quick_results.append((s_b, a_b, name))
        print(f"  Quick {name}: {s_b:.10f}")
        if s_b < best_score:
            best_score = s_b
            best_seq = a_b.copy()

    # Sort by score and keep top 3
    quick_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 best: {quick_results[0][2]} = {quick_results[0][0]:.10f}")

    # === Phase 2: Main optimization on top candidates (50% of time) ===
    phase2_deadline = t_start + 0.7 * (budget_s - 30)
    print(f"=== Phase 2: Main optimization (deadline at t={phase2_deadline - t_start:.0f}s) ===")

    top_candidates = []
    for s_c, a_c, name in quick_results[:3]:
        top_candidates.append(a_c)
    # Also include parent at 512 if it's good
    if parent is not None and p512 not in top_candidates:
        top_candidates.append(p512)

    # Deduplicate (by checking if they're very similar)
    unique_cands = [top_candidates[0]]
    for c in top_candidates[1:]:
        diff = np.linalg.norm(c - unique_cands[0])
        if diff > 0.1:
            unique_cands.append(c)

    per_cand_budget = max(30.0, (phase2_deadline - time.monotonic()) / max(1, len(unique_cands)) - 5)

    for idx, a0 in enumerate(unique_cands):
        if time.monotonic() > phase2_deadline - 10:
            break
        d = min(phase2_deadline - 10, time.monotonic() + per_cand_budget)
        a_b, s_b = _full_optimize(a0, d, label=f"P2-{idx}")
        if s_b < best_score:
            best_score = s_b
            best_seq = a_b.copy()

    # === Phase 3: Try n=1024 if time remains (20% of time) ===
    phase3_deadline = t_start + 0.9 * (budget_s - 30)
    if time.monotonic() < phase3_deadline - 60:
        print(f"=== Phase 3: n=1024 optimization ===")
        best_1024 = _interpolate_profile(best_seq, 1024) if best_seq is not None else p1024

        # Also try power-law at 1024
        p1024_pl = _make_power_profile(1024, 1.2)

        # Pick the better initialization for 1024
        sc_b1024 = _score(best_1024)
        sc_pl1024 = _score(p1024_pl)
        init_1024 = best_1024 if sc_b1024 < sc_pl1024 else p1024_pl
        print(f"  n=1024 init score: {min(sc_b1024, sc_pl1024):.10f}")

        d3 = min(phase3_deadline - 30, time.monotonic() + 0.2 * (budget_s - 30))
        a1024, s1024 = _full_optimize(init_1024, d3, label="P3-1024")
        if s1024 < best_score:
            best_score = s1024
            best_seq = a1024.copy()
            print(f"  n=1024 improved! score={s1024:.10f}")

    # === Phase 4: Final polish (last 10% of time) ===
    if best_seq is not None and time.monotonic() < deadline:
        print(f"=== Phase 4: Final polish (current best={best_score:.10f}) ===")
        n_best = best_seq.size

        # Fine-tune at current length
        d_fine = min(deadline - 10, time.monotonic() + 60)
        a_f1, s_f1 = _adam_smooth_max(best_seq, d_fine, max_iter=10000, lr=0.005, beta_factor=0.01)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Fine Adam: {best_score:.10f}")

        # Multi-peak subgradient
        if time.monotonic() < deadline - 10:
            d_f2 = min(deadline - 10, time.monotonic() + 30)
            a_f2, s_f2 = _multi_peak_subgradient(best_seq, d_f2, max_iter=5000, top_k=7)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final MultiSG: {best_score:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 5:
            a_f3, s_f3 = _adam_smooth_max(best_seq, deadline - 5, max_iter=5000, lr=0.002, beta_factor=0.005)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Ultra-fine Adam: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_power_profile(512, 1.0)
        best_score = _score(best_seq)

    # Final validation and cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        final_seq = final_seq / s_final
    else:
        final_seq = _make_power_profile(256, 1.0)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            print(f"WARN: evaluate_sequence returned {sc_eval}, falling back")
            final_seq = _make_power_profile(512, 1.0)
            final_seq = final_seq / final_seq.sum()
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            print(f"Fallback evaluate_sequence: {sc_eval:.10f}")
            best_score = sc_eval
    except Exception as ex:
        print(f"Evaluate error: {ex}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final_seq]
