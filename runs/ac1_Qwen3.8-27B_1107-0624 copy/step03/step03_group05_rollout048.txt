For n=768, we want `max(conv(p,p))` to be as small as possible, subject to `p >= 0`, `sum(p) = 1`.

The minimum possible value of `max(conv(p,p))` is at least `1/(2n-1)` (by the averaging argument), but this is achieved only if all convolution values are equal to `1/(2n-1)`. This would mean the autocorrelation is a constant (plus the DC component). In the frequency domain, this means the power spectrum is a single spike (at DC) - i.e., the sequence is constant. But a constant sequence has `conv(p,p)[n-1] = n*(1/n)^2 = 1/n`, not `1/(2n-1)`.

Wait, let me redo this. If p is constant: `p_i = 1/n` for all i.
`conv(p,p)[k] = sum_j (1/n) * (1/n)` (over valid j)
The number of valid j for lag k is `min(k+1, n, 2n-1-k)`.
So `conv(p,p)[k] = min(k+1, n, 2n-1-k) / n^2`.
The max is at k = n-1: `n/n^2 = 1/n`.
Score = `2n * 1/n = 2`.

The uniform gives `1/768 ≈ 0.001302`.
The lower bound is `1/(2*768-1) = 1/1535 ≈ 0.000651`.

So we need the max convolution to be between 0.000651 and 0.001302. The uniform achieves 0.001302 (score 2), and we need to get to 0.000978 (score 1.5027). That's a 25% reduction from uniform.

This suggests the optimal sequence is significantly non-uniform. It needs to have a structure that "spreads out" the autocorrelation more evenly.

- A peak at lag 0 (within-bump correlation)
- A peak at lag ~n/2 (cross-bump correlation)
- Smaller values elsewhere

By making the two bumps have different sizes or shapes, we can balance these peaks.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0)  # Remove numerical negatives
    return c

def _score(a):

def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Project to simplex {x : x >= 0, sum(x) = target_sum}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return w * (target_sum / s)

    # A[idx, j] = sum of a contributions
    # conv(a, g)[k] = sum_j a_j * g_{k-j}

    A = np.zeros((len(idx), n))

        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[i, j] = a[k - j]
    return A, idx

    a = np.asarray(self_a, dtype=np.float64)
    n = len(a)
    c = _fft_convolve(a)

        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        idx = np.arange(2 * n - 1)
    
    c_obj = -np.ones(n)
    b = np.full(A.shape[0], M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True})

        pass
    return None

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, maxc = _smooth_max_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta *= 2
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        # Project gradient onto simplex tangent: remove component in direction of (1,1,...,1)
        dot = float(np.mean(grad))
        grad = grad - dot
        
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                beta = max(1e-8, beta_init_scale * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
        
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)
    
    return best_a

def _diversify_and_eval(a_best, deadline, rng):
    """Generate diverse candidates and evaluate them."""
    n = len(a_best)
    results = []
    
    # Generate diverse inits
    inits = []
    
    # U-shaped
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        t = np.linspace(0, 1, n)
        p = 1.0 + alpha * (2*t - 1)**2
        p = np.maximum(p, 1e-12) / p.sum()
        inits.append(p)
    
    # Bimodal (Gaussian mixture)
    for c1, c2, s1, s2 in [(0.2, 0.8, 0.1, 0.1), (0.15, 0.85, 0.08, 0.08),
                             (0.1, 0.9, 0.05, 0.05), (0.3, 0.7, 0.15, 0.15),
                             (0.25, 0.75, 0.12, 0.12)]:
        t = np.linspace(0, 1, n)
        p = np.exp(-0.5*((t-c1)/s1)**2) + np.exp(-0.5*((t-c2)/s2)**2)
        p = np.maximum(p, 1e-12) / p.sum()
        inits.append(p)
    
    # Random with structure
    for _ in range(5):
        p = np.maximum(rng.normal(1.0, 0.7, n), 0.0)
        p = p / p.sum()
        inits.append(p)
    
    # Perturbed best
    for _ in range(3):
        p = a_best * (1.0 + 0.2 * rng.normal(size=n))
        p = np.maximum(p, 0.0)
        p = p / p.sum()
        inits.append(p)
    
    # Evaluate each
    for p in inits:
        if time.monotonic() > deadline:
            break
        sc = _score(p)
        results.append((sc, p.copy()))
    
    return results
