    return max(4, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n + 2)
    fw = np.fft.rfft(w, N)
    fq = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fq, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    p = e / s
    p = np.maximum(p, floor)
    p = p / p.sum()
    return p


def _lp_solve(a, M, max_iter=50000):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                  method='highs', options={'maxiter': max_iter})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_direction(a, M):
    b = _lp_solve(a, M)
    if b is None:
        return None
    s = float(b.sum())
    if s < 1e-12:
        return None
    return b / s


def _adam_optimize(p0, deadline, rng, lr=0.08, beta_init_frac=0.15,
                   beta_decay=0.97, no_imp_limit=500, lp_interval=100,
                   noise_floor=0.05):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-14)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    iter_count = 0
    base_lr = lr

    while time.monotonic() < deadline:
        iter_count += 1
        _, w = _smooth_max_weights(c, beta)
        g = _cross_corr(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        step = lr * mh / (np.sqrt(vh) + eps)
        z = z - step

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr * 0.5
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-15:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        # Adaptive beta: based on gap between top peaks
        if iter_count % 30 == 0 and c.size >= 2:
            sidx = np.argsort(c)[::-1]
            gap = c[sidx[0]] - c[sidx[1]]
            target_beta = max(1e-5, 0.3 * gap)
            if target_beta > beta:
                beta = min(beta * 1.1, target_beta)
            else:
                beta = max(1e-5, beta * beta_decay)

        # LP direction kick
        if lp_interval > 0 and iter_count % lp_interval == 0 and no_imp > 50:
            b_dir = _lp_direction(p, maxc)
            if b_dir is not None:
                mix = 0.3
                p_lp = (1.0 - mix) * p + mix * b_dir
                p_lp = np.maximum(p_lp, 1e-14)
                p_lp = p_lp / p_lp.sum()
                sc_lp = _score_norm(p_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_p = p_lp.copy()
                    z = np.log(np.maximum(p_lp, 1e-14))
                    z -= float(np.mean(z))
                    best_z = z.copy()
                    p = p_lp.copy()
                    c = _full_convolve(p)
                    maxc = float(c.max())
                    no_imp = 0

        # Restart logic
        limit = max(100, no_imp_limit)
        if no_imp > limit:
            # Structured restart: mix of Gaussian and structured noise
            ns = max(noise_floor, 0.05 + 0.05 * (no_imp - limit) / limit)
            noise_type = rng.integers(0, 3)
            if noise_type == 0:
                pert = ns * rng.normal(size=n)
            elif noise_type == 1:
                freq = rng.uniform(1, 5)
                amp = ns * 2
                x = np.arange(n)
                pert = amp * np.sin(2 * np.pi * freq * x / n)
            else:
                split = rng.integers(n // 3, 2 * n // 3)
                pert = ns * np.zeros(n)
                pert[:split] = -1.0
                pert[split:] = 1.0
            z = best_z + pert
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_refine(a, deadline, t_init=0.4, max_solves=25, lp_interval_check=5):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_direction(a, M)
        if b is None:
            break
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 1e-14)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score - 1e-15:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.15, t * 0.88)

    return current_best, current_score


def _interp_to(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        out = p.copy()
        out = np.maximum(out, 0.0)
        s = out.sum()
        if s > 0:
            out = out / s
        return out
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    out = np.interp(x1, x0, p)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 0:
        out = out / s
    return out


def _make_init_patterns(n):
    """Generate diverse initial patterns of length n."""
    patterns = []
    t = np.linspace(0.0, 1.0, n)

    # Exponential decays
    for lam in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-lam * t)
        a = np.maximum(a, 1e-14)
        patterns.append(('exp_%.1f' % lam, a / a.sum()))

    # Linear
    for eta in [1.0, 1.5, 2.0, 2.5]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-14)
        patterns.append(('lin_%.1f' % eta, a / a.sum()))

    # Two bump
    for c1, c2, s in [(0.25, 0.75, 0.12), (0.3, 0.7, 0.15), (0.2, 0.8, 0.2),
                      (0.35, 0.65, 0.1), (0.25, 0.75, 0.08)]:
        a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
        a = np.maximum(a, 1e-14)
        patterns.append(('bump_%.1f_%.1f_%.2f' % (c1, c2, s), a / a.sum()))

    # Cosine modulated
    for freq, amp in [(1, 0.5), (1, 0.8), (2, 0.4), (2, 0.6), (3, 0.3),
                      (1, 0.3), (4, 0.2)]:
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
        a = np.maximum(a, 1e-14)
        patterns.append(('cos_%d_%.1f' % (freq, amp), a / a.sum()))

    # Skewed (power law)
    for p in [0.5, 1.5, 2.0, 3.0]:
        a = t ** p
        a = np.maximum(a, 1e-14)
        patterns.append(('pow_%.1f' % p, a / a.sum()))

    # Sigmoid-like
    for k in [5, 10, 20]:
        a = 1.0 / (1.0 + np.exp(-k * (t - 0.5)))
        a = np.maximum(a, 1e-14)
        patterns.append(('sig_%d' % k, a / a.sum()))

    # Triangular / tent
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-14)
    patterns.append(('tent', a / a.sum()))

    # V-shape
    a = np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-14)
    patterns.append(('vshape', a / a.sum()))

    return patterns


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print("  BEST: %.10f n=%d [%s]" % (best_score, p.size, label))
            sys.stdout.flush()

    import sys

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print("Parent n=%d, budget=%ds" % (pn, int(budget_s)))

    # === Phase 1: Candidate generation (120s) ===
    phase1_end = t_start + 120.0
    candidates = []

    target_ns = [512, 1024, 2048]
    for n in target_ns:
        pats = _make_init_patterns(n)
        for label, p in pats:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))
        if time.monotonic() > phase1_end:
            break

    # Add parent at various resolutions
    if parent is not None:
        for n in [512, 1024, 2048]:
            if pn != n:
                pi = _interp_to(parent, n)
                sc = _score_norm(pi)
                candidates.append((sc, pi, n, 'parent_n%d' % n))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print("Phase 1: %d candidates, best=%.10f" % (len(candidates), candidates[0][0]))
    _update(candidates[0][1], candidates[0][0], 'init')

    # === Phase 2: Adam polish top candidates (600s) ===
    phase2_end = t_start + 600.0
    polished = []
    count = 0
    for i, (sc, p, n, label) in enumerate(candidates):
        if time.monotonic() > phase2_end or count >= 8:
            break
        per_end = min(phase2_end, time.monotonic() + 60.0)
        p_b, sc_b = _adam_optimize(p, per_end, rng, lr=0.08, beta_init_frac=0.15,
                                   beta_decay=0.97, no_imp_limit=400,
                                   lp_interval=150)
        _update(p_b, sc_b, 'polish_%s_n%d' % (label, n))
        polished.append((sc_b, p_b, n, label))
        count += 1
        if time.monotonic() > phase2_end:
            break

    polished.sort(key=lambda x: x[0])
    print("Phase 2 done: best=%.10f" % best_score)

    # === Phase 3: LP refinement (300s) ===
    phase3_end = t_start + 900.0
    if best_seq is not None and best_seq.size <= 2048:
        sc_lp = time.monotonic()
        p_lp, sc_lp = _lp_refine(best_seq, min(phase3_end, t_start + 1000.0),
                                  t_init=0.4, max_solves=20)
        _update(p_lp, sc_lp, 'LP_refine')
        print("  LP took %.1fs" % (time.monotonic() - sc_lp))

        # Adam after LP
        if time.monotonic() < phase3_end - 10:
            per = min(80.0, phase3_end - time.monotonic() - 5)
            if per > 15:
                p_ap, sc_ap = _adam_optimize(p_lp, time.monotonic() + per, rng,
                                             lr=0.04, beta_init_frac=0.08,
                                             beta_decay=0.98, no_imp_limit=300,
                                             lp_interval=200)
                _update(p_ap, sc_ap, 'post_LP')

    # === Phase 4: Multi-scale escalation (400s) ===
    phase4_end = t_start + 1300.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 30:
                break
            p_int = _interp_to(best_seq, tn)
            sc_int = _score_norm(p_int)
            print("  Escalate n=%d: %.10f" % (tn, sc_int))
            _update(p_int, sc_int, 'interp_n%d' % tn)

            per = min(120.0, phase4_end - time.monotonic() - 20)
            if per > 20:
                p_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                           lr=0.06, beta_init_frac=0.12,
                                           beta_decay=0.97, no_imp_limit=400,
                                           lp_interval=150)
                _update(p_e, sc_e, 'escalate_n%d' % tn)

            # LP at new scale
            if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase4_end - 10:
                p_lp2, sc_lp2 = _lp_refine(best_seq, min(phase4_end - 5, time.monotonic() + 80),
                                            t_init=0.35, max_solves=12)
                _update(p_lp2, sc_lp2, 'LP_n%d' % best_seq.size)

    # === Phase 5: Multi-restart with structured perturbations (300s) ===
    phase5_end = deadline - 30.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(25.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break
            z = np.log(np.maximum(best_seq, 1e-14))
            z -= z.mean()
            # Structured perturbation
            pert_type = i % 3
            if pert_type == 0:
                scale = 0.1 + 0.08 * i
                z_pert = z + scale * rng.normal(size=n_best)
            elif pert_type == 1:
                freq = rng.uniform(1, 4)
                amp = 0.15 + 0.1 * i
                x = np.arange(n_best)
                z_pert = z + amp * np.sin(2 * np.pi * freq * x / n_best + rng.uniform(0, 2 * np.pi))
            else:
                scale = 0.12 + 0.06 * i
                split = rng.integers(n_best // 4, 3 * n_best // 4)
                z_pert = z.copy()
                z_pert[:split] -= scale
                z_pert[split:] += scale
            z_pert = np.clip(z_pert, -60, 60)
            z_pert -= z_pert.mean()
            p_s = _softmax_from_z(z_pert)
            p_r, sc_r = _adam_optimize(p_s, time.monotonic() + per, rng, lr=0.06,
                                       beta_init_frac=0.1, beta_decay=0.96,
                                       no_imp_limit=int(0.4 * n_best),
                                       lp_interval=120)
            _update(p_r, sc_r, 'restart%d_n%d' % (i, n_best))

    # === Phase 6: Final polish (remaining time) ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            p_f, sc_f = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                       lr=0.03, beta_init_frac=0.05,
                                       beta_decay=0.98, no_imp_limit=150,
                                       lp_interval=200)
            _update(p_f, sc_f, 'final_polish')

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            t = np.linspace(0, 1, 1024)
            final = 1.0 + 1.5 * (t - 0.5)
            final = np.maximum(final, 1e-6)
            final = final / final.sum()

        exact_sc = _score_norm(final)
        print("FINAL: score=%.10f n=%d" % (exact_sc, final.size))

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print("EVAL: %.10f" % eval_sc)
                if eval_sc < best_score:
                    best_score = eval_sc
        except Exception as e:
            print("Eval error: %s" % str(e))

        return [float(x) for x in final]

    # Fallback
    t = np.linspace(0, 1, 1024)
    fallback = 1.0 + 1.5 * (t - 0.5)
    fallback = np.maximum(fallback, 1e-6)
    fallback = fallback / fallback.sum()
    print("FALLBACK n=1024")
    return [float(x) for x in fallback]
