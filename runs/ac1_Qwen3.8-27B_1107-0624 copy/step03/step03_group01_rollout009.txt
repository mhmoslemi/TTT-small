```python
import time
from scipy.optimize import linprog
import cvxpy as cp

def _solve_lp(f, M, n):
    # Solve: max sum(b) s.t. conv(f,b)_k <= M for all k, b >= 0
    # Variables: b_0, ..., b_{n-1}
    # conv(f,b)_k = sum_i f_i * b_{k-i} (where 0 <= i < n and 0 <= k-i < n)
    
    # cmap = M (upper bound for each convolution value)
    # A * b <= M * ones(2n-1)
    # where A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    
    # Build A matrix (sparse)
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # linprog: minimize -sum(b) s.t. A*b <= M, b >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
"""
Hybrid L-BFGS + LP active-set optimization for minimizing autocorrelation peak.
Uses continuation L-BFGS for coarse optimization, then LP-based best-response
iterations for fine tuning, with final L-BFGS polish.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score with sum(a) already = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_constraint_matrix(f, n):
    """Build sparse matrix A where A @ b = conv(f, b)"""
    data = np.tile(f, n)
    row_ind = np.concatenate([np.arange(j, j + n) for j in range(n)])
    col_ind = np.repeat(np.arange(n), n)
    A = csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))
    return A


def _lp_best_response(f, M, n):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0"""
    A = _build_constraint_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_continuation(a_init, deadline, max_wall_time=300.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.10, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = time.monotonic() + per_level
        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9,
                         'maxls': 30, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _lp_iterations(f_init, deadline, max_iters=500):
    """LP-based active set iterations"""
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_score = _score_norm_fn(f)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = float(_full_convolve(f, f).max())
        
        g = _lp_best_response(f, M, n)
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g < 1e-15:
            break
        g = g / sum_g  # normalize to sum=1
        
        # Line search for best t
        best_t = 0.0
        best_sc = best_score
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            candidate = (1 - t) * f + t * g
            sc = _score_norm_fn(candidate)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.001, 0.005, 0.02]:
                candidate = (1 - t) * f + t * g
                sc = _score_norm_fn(candidate)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        
        if best_t == 0.0:
            break  # No improvement possible
        
        f = (1 - best_t) * f + best_t * g
        if best_sc < best_score:
            best_score = best_sc
            best_f = f.copy()
        
        if it % 50 == 0:
            print(f"  LP iter {it}: score={best_score:.10f}, t={best_t:.4f}, M={M:.8f}")
    
    return best_f, best_score


def _adam_polish(a_init, deadline, lr=0.03, beta_frac=0.01, n_max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
```

"""
Hybrid L-BFGS + LP active-set optimization for minimizing the autocorrelation peak
of a non-negative step function. Uses continuation L-BFGS for coarse optimization,
then LP-based best-response iterations for fine tuning, with final Adam polish.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_convolve_matrix(f, n):
    """Build sparse matrix A (2n-1 x n) where A @ b = conv(f, b)"""
    data = np.tile(f, n)
    col_ind = np.repeat(np.arange(n), n)
    row_ind = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    A = csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))
    return A


def _lp_best_response(f, M, n):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0"""
    A = _build_convolve_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()

    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,

                         'maxls': 40, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
