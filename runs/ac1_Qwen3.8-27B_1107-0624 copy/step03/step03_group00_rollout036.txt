import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _conv_cross_corr(w, p):
    """Compute cross-correlation of w with reversed p, giving gradient direction."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

    peak_indices = np.argsort(conv_aa)[::-1][:max_constraints]

    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve_reduced(a, M, max_constraints)
        if b is None:

        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t_start) * a + t_start * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t_start = max(0.05, t_start * 0.85)
    
    return current_best, current_score

def _smooth_max_gradient(c, p, beta):
    """Compute gradient using smooth max approximation."""
    n = len(p)
    mval = float(c.max())
    if beta < 1e-12:
        # Hard max
        mask = c >= mval - 1e-10
        if mask.sum() == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = mask.astype(float) / mask.sum()
    else:
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
    
    g = _conv_cross_corr(w, p)
    return g

def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_exp(n, lam, side='left'):
    t = np.arange(n, dtype=float)
    if side == 'left':
        a = np.exp(-lam * t / n)
    else:
        a = np.exp(-lam * (n - 1 - t) / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profile_power(n, alpha):
    """p(i) proportional to (i+1)^alpha"""
    t = np.arange(1, n + 1, dtype=float)
    a = t ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

def _coord_descent_polish(p, deadline, max_rounds=50, K=3):
    """Polish using coordinate descent on top peaks."""
    n = len(p)
    p = p.copy()
    c = _full_convolve(p)
    best_score = float(2.0 * n * c.max())
    best_p = p.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1][:K]
        
        improved = False
        for k in sorted_idx:
            if time.monotonic() > deadline:
                break
            
            # For peak at position k, we want to reduce conv(k)
            # conv(k) = sum_j p(j) * p(k-j)
            # Gradient wrt p_i: 2*p(k-i) for valid i
            # So we should decrease p_i where p(k-i) is large, and vice versa
            
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            
            # The contribution to conv(k) from position i is p(i) * p(k-i)
            # To reduce conv(k), we can redistribute mass:
            # Move mass from high-contribution positions to low-contribution ones
            
            contrib = np.zeros(n)
            for j in range(j_start, j_end + 1):
                contrib[j] = p[j] * p[k - j]
            
            # Try to reduce the peak by moving mass from high-contrib to low-contrib
            # Simple approach: find the position with highest contribution and
            # move a small amount to a position with lowest contribution
            max_contrib_idx = j_start + int(np.argmax(contrib[j_start:j_end+1]))
            min_contrib_idx = j_start + int(np.argmin(contrib[j_start:j_end+1]))
            
            if max_contrib_idx == min_contrib_idx:
                continue
            
            # Try several small moves
            for frac in [0.01, 0.005, 0.001]:
                delta = p[max_contrib_idx] * frac
                p_new = p.copy()
                p_new[max_contrib_idx] -= delta
                p_new[min_contrib_idx] += delta
                if p_new[max_contrib_idx] < 0:
                    continue
                p_new = p_new / p_new.sum()
                c_new = _full_convolve(p_new)
                sc_new = float(2.0 * n * c_new.max())
                if sc_new < best_score:
                    p = p_new
                    c = c_new
                    best_score = sc_new
                    best_p = p.copy()
                    improved = True
                    break
        
        if not improved:
            # Try a different strategy: reduce all positions that contribute to the peak
            for k in sorted_idx[:1]:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                # Move mass from peak-contributing positions to positions far from the peak
                contrib = np.zeros(n)
                for j in range(j_start, j_end + 1):
                    contrib[j] = p[j] * p[k - j]
                
                # Find positions far from the peak (low contribution)
                top_contrib = np.argsort(contrib)[::-1][:5]
                bot_contrib = np.argsort(contrib)[:5]
                
                total_move = sum(p[i] for i in top_contrib) * 0.01
                per_move = total_move / 5
                p_new = p.copy()
                for i in range(5):
                    src = top_contrib[i]
                    dst = bot_contrib[i]
                    p_new[src] -= per_move * 0.5
                    p_new[dst] += per_move * 0.5
                p_new = np.maximum(p_new, 0)
                p_new = p_new / p_new.sum()
                c_new = _full_convolve(p_new)
                sc_new = float(2.0 * n * c_new.max())
                if sc_new < best_score:
                    p = p_new
                    c = c_new
                    best_score = sc_new
                    best_p = p.copy()
                    improved = True
            if not improved:
                break
    
    return best_p, best_score

If conv(k) = sum_j p(j) * p(k-j), then d/dp_i conv(k) = p(k-i) + p(k-i) = 2*p(k-i) (for the cross terms, when k-i ≠ i) or p(k-i) + p(k-i) = 2*p(i) (when k-i = i).

Actually, more carefully: conv(k) = sum_{j: 0<=j<n, 0<=k-j<n} p(j)*p(k-j)

d/dp_i conv(k) = p(k-i) [from the j=i term] + p(i) [from the k-j=i term, i.e., j=k-i]
If k-i ≠ i: d/dp_i conv(k) = 2*p(k-i)
If k-i = i (k=2i): d/dp_i conv(k) = 2*p(i)

The `_conv_cross_corr` function computes: for each i, sum over k of w[k] * p[k-i] (where k-i is valid). This is the cross-correlation.

```python
"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) hard-max and smooth-max Adam in log-space with corrected full gradient,
(2) fast LP refinement with reduced constraints, (3) multi-scale escalation,
(4) adaptive K peak focusing, (5) final coordinate descent polish.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _gradient_full(w, p):
    """Full gradient: sum_k w[k]*(p[k-i] + p[i]) = cross_corr(w,p) + p*sum(w)."""
    h = _conv_cross_corr(w, p)
    return h + p * float(w.sum())


def _lp_solve_reduced(a, M, max_constraints=200):
    n = len(a)
    conv_aa = _full_convolve(a)
    peak_indices = np.argsort(conv_aa)[::-1][:max_constraints]
    peak_indices = np.sort(peak_indices)
    n_c = len(peak_indices)
    k_to_row = {int(k): i for i, k in enumerate(peak_indices)}
    rows = []
    cols = []
    vals = []
    for k in peak_indices:
        j_start = max(0, int(k) - n + 1)
        j_end = min(n - 1, int(k))
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k_to_row[int(k)]] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[int(k) - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_c, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(n_c)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                      options={'maxiter': 10000})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_start=0.3, max_solves=30, max_constraints=150):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        mc = max_constraints if n <= 2048 else 100
        b = _lp_solve_reduced(a, M, mc)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t_start) * a + t_start * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t_start = max(0.05, t_start * 0.85)
    return current_best, current_score


def _adam_logit_hardmax(p0, deadline, rng, lr=0.05, K=5,
                        no_imp_limit=300, noise_scale=None, lr_decay=0.95):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    current_K = K
    while time.monotonic() < deadline:
        sorted_idx = np.argsort(c)[::-1][:current_K]
        w = np.zeros(2 * n - 1)
        for k in sorted_idx:
            w[int(k)] = 1.0 / current_K
        g = _gradient_full(w, p)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-30:
            g = g / g_norm
        else:
            break
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
                if t_adam % 300 == 0 and current_K < 15:
                    current_K = min(15, current_K + 1)
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.005, lr * lr_decay)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.08
                elif best_score < 1.53:
                    ns = 0.15
                else:
                    ns = 0.3
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            current_K = K
    return best_p, best_score


def _adam_logit_smooth(p0, deadline, rng, lr=0.05, beta_frac=0.08,

    b1, b2, eps = 0.9, 0.999, 1e-8

    v = np.zeros(n)

            w = np.zeros_like(c)
