def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _lp_spread(a, M):
    """Maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_active_set(a, active_lags):
    """
    Solve: min t s.t. q_k - 2*sum(d_i * a[k-i]) >= -t for k in active_lags
           sum(d_i) = 0, d_i >= -a_i
    Returns direction d and predicted t.
    """
    n = len(a)
    c_full = _fft_convolve(a)

    # Variables: d_0..d_{n-1}, t  (total n+1)
    # Minimize t: c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    rows = []
    rhs = []

    # Active lag constraints: q_k + 2*sum_i d_i*a[k-i] - t <= 0
    # i.e., sum_i (2*a[k-i]) * d_i - t <= -q_k
    for k in active_lags:
        row = np.zeros(n + 1)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                row[i] = 2.0 * a[j]
        row[-1] = -1.0
        rows.append(row)
        rhs.append(-float(c_full[k]))

    # Non-negativity: a_i + d_i >= 0 => -d_i <= a_i
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        rows.append(row)
        rhs.append(a[i] + 1e-15)

    # Upper bound: d_i <= 2*a_i + 1/n (prevent too-large positive steps)
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        rows.append(row)
        rhs.append(2.0 * a[i] + 1.0 / n)

    # Equality: sum(d_i) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    A_ub = np.array(rows)
    b_ub = np.array(rhs)

    bounds = [(None, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def _gradient_polish(a, deadline, max_iters=20000):
    """Fine gradient polish in logit space with decreasing step size."""
    a = np.asarray(a, dtype=np.float64).copy()

    n = len(a)

    no_imp = 0

    z -= np.mean(z)

    beta = 0.005  # smooth max temperature (small for sharp max)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Smooth max weights
        x = (c - c_max) / beta
        x = np.clip(x, -50, 50)
        w = np.exp(x)
        w = w / w.sum()

        # Gradient of smooth max w.r.t. a
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)

        h = h_full[n - 1:2 * n - 1]

        # Project gradient to simplex tangent (sum-preserving)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)

            best_score = sc2
            best_seq = a2.copy()
            a1 = a2

    # === Phase 3: Another round of active-set LP ===
    if time.monotonic() < deadline - 200:
        print(f"\n=== Phase 3: Active-set LP round 2 ===", flush=True)
        phase3_deadline = min(deadline - 200, time.monotonic() + budget_s * 0.2)
        a3 = _active_set_optimize(a1, phase3_deadline, max_iters=8000)
        sc3 = _score(a3)
        print(f"  Active-set LP r2: {sc3:.10f}", flush=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            a1 = a3

    # === Phase 4: Targeted peak shaving ===
    if time.monotonic() < deadline - 150:
        print(f"\n=== Phase 4: Peak shaving ===", flush=True)
        phase4_deadline = min(deadline - 150, time.monotonic() + 60)
        a4 = _targeted_shave(a1, phase4_deadline, n_peaks=10, max_iters=3000)
        sc4 = _score(a4)
        print(f"  Peak shave: {sc4:.10f}", flush=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            a1 = a4

    # === Phase 5: Gradient polish ===
    if time.monotonic() < deadline - 50:
        print(f"\n=== Phase 5: Gradient polish ===", flush=True)
        phase5_deadline = min(deadline - 50, time.monotonic() + 120)
        a5 = _gradient_polish(a1, phase5_deadline, max_iters=30000)
        sc5 = _score(a5)
        print(f"  Gradient polish: {sc5:.10f}", flush=True)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 6: Multi-length exploration ===", flush=True)
        lengths_to_try = []
        for ln in [n_parent + 128, n_parent + 256, n_parent - 128]:
            if ln >= 256 and ln not in lengths_to_try:
                lengths_to_try.append(ln)

        # Also try a fresh structured init at a different length
        for ln in [1024, 896]:
            if ln not in lengths_to_try and ln != n_parent:
                lengths_to_try.append(ln)

        for ln in lengths_to_try[:3]:
            if time.monotonic() > deadline - 30:
                break
            print(f"  Trying n={ln}...", flush=True)

            # Interpolate parent to this length
            a_interp = _interpolate_profile(best_seq, ln)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}", flush=True)

            # Also try structured init
            a_struct = _make_structured_init(ln, rng)
            sc_struct = _score(a_struct)
            print(f"    Structured: {sc_struct:.10f}", flush=True)

            # Pick better starting point
            if sc_interp < sc_struct:
                a_start = a_interp
            else:
                a_start = a_struct

            # Quick active-set optimization
            sub_deadline = min(deadline - 30, time.monotonic() + 60)
            a_opt = _active_set_optimize(a_start, sub_deadline, max_iters=2000)
            sc_opt = _score(a_opt)
            print(f"    Optimized: {sc_opt:.10f}", flush=True)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()

    # === Phase 7: Final fine polish on the best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 7: Final polish ===", flush=True)
        final_deadline = deadline - 5
        a_final = _gradient_polish(best_seq, final_deadline, max_iters=10000)
        sc_final = _score(a_final)
        print(f"  Final: {sc_final:.10f}", flush=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = parent.copy()

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_structured_init(512, rng)
        best_score = _score(final_seq)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in final_seq]
