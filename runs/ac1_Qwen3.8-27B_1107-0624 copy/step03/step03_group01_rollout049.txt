"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with adaptive logsumexp continuation, (2) subgradient peak shaving,
(3) multi-restart with diverse initializations, (4) population-based crossover.
Exploits convolution symmetry and uses FFT for efficiency.

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _autocorr(a):
    """Full autocorrelation: conv(a, a)"""
    return _full_convolve(a, a)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _autocorr(a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lbe_objective(z, beta, n):
    a = _softmax(z)
    c = _autocorr(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _lbe_gradient(z, beta, n):
    a = _softmax(z)
    c = _autocorr(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _lbfgs_phase(a_init, deadline, max_wall_time=60.0):
    """Run L-BFGS with continuation on logsumexp approximation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _autocorr(a_init)
    max_c0 = float(c0.max())
    
    # Adaptive beta schedule
    beta_starts = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]
    
    per_level = max_wall_time / len(betas)
    
    for beta in betas:
        level_deadline = time.monotonic() + per_level
        if time.monotonic() > deadline:
            break
        
        try:
            result = minimize(
                _lbe_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbe_gradient,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 20, 'maxfun': 1000}

        m = b1 * m + (1 - b1) * grad

        a = 1.0 + param * u**2
    elif profile_type == 2:  # Cubic

    a = np.maximum(a, 1e-12)

def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _crossover(a1, a2, rng):
    """Random crossover between two sequences of the same length."""
    n = len(a1)
    t = rng.random()
    a = t * a1 + (1 - t) * a2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def _peak_shave(a, rng, n_shaves=10, strength=0.05):
    """Targeted perturbation to reduce the current peak."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        c = _autocorr(a)
        k_star = int(np.argmax(c))
        # Perturb elements that contribute to peak k_star
        # Elements a_j where j and k_star-j are both in [0,n-1] contribute
        contributing = [j for j in range(n) if 0 <= k_star - j < n]
        if not contributing:
            break
        # Randomly select a subset to perturb
        n_perturb = max(1, len(contributing) // 4)
        selected = rng.choice(contributing, size=min(n_perturb, len(contributing)), replace=False)
        # Reduce these elements, increase others
        reduction = strength * a[selected]
        a[selected] -= reduction
        # Distribute to non-contributing elements
        others = [j for j in range(n) if j not in selected]
        if others:
            a[others] += strength * a[selected] / len(others)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
    return a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) L-BFGS with adaptive logsumexp continuation, (2) subgradient peak shaving,
(3) Adam polishing with temperature annealing, (4) multi-restart with diverse perturbations,
(5) population-based crossover. Uses FFT-based convolution for efficiency.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _autocorr(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbe_objective(z, beta, n):
    a = _softmax(z)
    c = _autocorr(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbe_gradient(z, beta, n):
    a = _softmax(z)
    c = _autocorr(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    max_c0 = float(c0.max())

    betas = [b * max_c0 for b in beta_fracs]

            break
        level_deadline = time.monotonic() + per_level

                options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 25, 'maxfun': 1500}
            )

def _subgradient_shave(a, deadline, n_steps=3000, lr_init=0.005, lr_end=0.00005):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    momentum = np.zeros(n)
    beta_m = 0.85

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        t_frac = step / max(1, n_steps)
        lr = lr_init * (lr_end / lr_init) ** t_frac

        c = _autocorr(a)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        a_rev = a[::-1]
        shift = n - 1 - k_star
        grad = np.zeros(n)
        if shift >= 0:
            grad[shift:] = 2.0 * a_rev[:n - shift]
        else:
            grad[:n + shift] = 2.0 * a_rev[-shift:]

        momentum = beta_m * momentum - lr * grad
        a = np.maximum(a + momentum, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a / s

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

    b1, b2, eps = 0.9, 0.999, 1e-8

    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

        if time.monotonic() > deadline:

        t += 1
        a = _softmax(z)
        c = _autocorr(a)
        cm = float(c.max())

        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

            beta = max(1e-12, beta * 0.82)
        if t % 5000 == 0 and best_score < 1.51:
            lr = max(0.003, lr * 0.93)

    return best_a, best_score


def _make_profile(n, profile_type=0, param=1.0):

    elif profile_type == 1:

    elif profile_type == 2:
        a = 1.0 + param * u ** 3
    elif profile_type == 3:
        a = 1.0 + param * np.cos(np.pi * u)
    elif profile_type == 4:
        a = 1.0 / (1.0 + np.exp(-param * u * 4.0))
    elif profile_type == 5:
        a = 1.0 + param * np.exp(-6.0 * u ** 2)
    elif profile_type == 6:
        a = 1.0 + param * np.sin(2.0 * np.pi * u)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _crossover(a1, a2, rng):
    n = len(a1)
    t = rng.random()
    a = t * a1 + (1.0 - t) * a2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _peak_shave(a, rng, n_shaves=8, strength=0.04):
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        c = _autocorr(a)
        k_star = int(np.argmax(c))
        contributing = np.array([j for j in range(n) if 0 <= k_star - j < n])
        if len(contributing) == 0:
            break
        n_perturb = max(1, len(contributing) // 5)
        selected = rng.choice(contributing, size=min(n_perturb, len(contributing)), replace=False)
        reduction = strength * a[selected]
        a[selected] -= reduction
        mask = np.ones(n, dtype=bool)
        mask[selected] = False
        others = np.where(mask)[0]
        if len(others) > 0:
            a[others] += strength * a[selected].sum() / len(others)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        a0 = parent.copy()
        sc_parent = _score_norm(a0)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = a0.copy()
        best_score = sc_parent
        best_n = n_parent
    else:
        n_parent = 2048
        a0 = _make_profile(n_parent, 0, 1.5)
        sc_parent = _score_norm(a0)
        print(f"No parent, init profile n={n_parent}, score={sc_parent:.10f}")
        best_seq = a0.copy()
        best_score = sc_parent
        best_n = n_parent

    # === Phase 1: Diverse initialization (60s) ===
    phase1_end = min(deadline, t_start + 60.0)
    for pt in range(7):
        if time.monotonic() > phase1_end:
            break
        a_test = _make_profile(n_parent, pt, 0.3 + 0.4 * pt)
        sc = _score_norm(a_test)
        if sc < best_score:
            best_seq = a_test.copy()
            best_score = sc
            best_n = n_parent
            print(f"  Phase1: new best {sc:.10f} (prof{pt})")

    # Try different lengths
    for target_n in [1024, 2048, 4096]:
        if time.monotonic() > phase1_end:
            break
        if target_n == best_n:
            continue
        a_interp = _interp_profile(best_seq, target_n)
        sc = _score_norm(a_interp)
        if sc < best_score:
            best_seq = a_interp.copy()
            best_score = sc
            best_n = target_n
            print(f"  Phase1: new best {sc:.10f} (n={target_n})")

    print(f"Phase 1 done: best={best_score:.10f} (n={best_n})")

    # === Phase 2: L-BFGS on best (400s) ===
    phase2_end = min(deadline, t_start + 460.0)
    if best_seq is not None:
        per_time = min(150.0, phase2_end - time.monotonic())
        dl = min(phase2_end, time.monotonic() + per_time)
        p_best, sc_best = _lbfgs_phase(best_seq, dl, max_wall_time=per_time)
        if sc_best < best_score:
            best_seq = p_best.copy()
            best_score = sc_best
            best_n = len(p_best)
            print(f"  Phase2: new best {sc_best:.10f} (lbfgs)")

    # L-BFGS on n=4096
    if time.monotonic() < phase2_end - 80:
        a_4096 = _interp_profile(best_seq, 4096)
        per_time = min(100.0, phase2_end - time.monotonic())
        dl = min(phase2_end, time.monotonic() + per_time)
        p_best, sc_best = _lbfgs_phase(a_4096, dl, max_wall_time=per_time)
        if sc_best < best_score:
            best_seq = p_best.copy()
            best_score = sc_best
            best_n = 4096
            print(f"  Phase2: new best {sc_best:.10f} (n=4096)")

    print(f"Phase 2 done: best={best_score:.10f} (n={best_n})")

    # === Phase 3: Multi-restart with peak shaving (400s) ===
    phase3_end = min(deadline, t_start + 860.0)
    if best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 10
        per_restart = (phase3_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break

            noise_scale = 0.03 + 0.02 * i
            a_pert = best_seq + noise_scale * rng.normal(size=n_best)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            a_shaved = _peak_shave(a_pert, rng, n_shaves=4, strength=0.02)

            per_dl = min(phase3_end - 5, time.monotonic() + per_restart * 0.5)
            p1, sc1 = _lbfgs_phase(a_shaved, per_dl, max_wall_time=per_restart * 0.5)
            if sc1 < best_score:
                best_seq = p1.copy()
                best_score = sc1
                best_n = n_best
                print(f"  Phase3: new best {sc1:.10f} (restart{i})")

            if time.monotonic() < phase3_end - 15:
                per_dl2 = min(phase3_end - 10, time.monotonic() + per_restart * 0.4)
                p2, sc2 = _adam_polish(p1, per_dl2, lr=0.03, beta_frac=0.005)
                if sc2 < best_score:
                    best_seq = p2.copy()
                    best_score = sc2
                    best_n = n_best
                    print(f"  Phase3: new best {sc2:.10f} (restart{i}_adam)")

    print(f"Phase 3 done: best={best_score:.10f} (n={best_n})")

    # === Phase 4: Population evolution (300s) ===
    phase4_end = min(deadline, t_start + 1160.0)
    if best_seq is not None:
        n_pop = len(best_seq)
        population = [best_seq.copy()]

        for i in range(3):
            if time.monotonic() > phase4_end:
                break
            a_new = _make_profile(n_pop, rng.integers(0, 7), 0.3 + 0.4 * rng.random())
            a_new = 0.75 * best_seq + 0.25 * a_new
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            population.append(a_new)

        n_gen = 5
        if n_gen > 0:
            per_gen = (phase4_end - time.monotonic()) / n_gen
        for gen in range(n_gen):
            if time.monotonic() > phase4_end - 40:
                break

            scores = [_score_norm(p) for p in population]
            idx_order = np.argsort(scores)
            top1 = population[idx_order[0]]
            top2 = population[idx_order[min(1, len(idx_order) - 1)]]

            child = _crossover(top1, top2, rng)
            child = _peak_shave(child, rng, n_shaves=3, strength=0.02)

            per_dl = min(phase4_end - 10, time.monotonic() + per_gen * 0.5)
            p_opt, sc_opt = _lbfgs_phase(child, per_dl, max_wall_time=per_gen * 0.5)
            if sc_opt < best_score:
                best_seq = p_opt.copy()
                best_score = sc_opt
                best_n = n_pop
                print(f"  Phase4: new best {sc_opt:.10f} (gen{gen})")

            population[idx_order[-1]] = p_opt

            if time.monotonic() < phase4_end - 30:
                per_dl2 = min(phase4_end - 20, time.monotonic() + per_gen * 0.4)
                p_adv, sc_adv = _adam_polish(p_opt, per_dl2, lr=0.025, beta_frac=0.004)
                if sc_adv < best_score:
                    best_seq = p_adv.copy()
                    best_score = sc_adv
                    best_n = n_pop
                    print(f"  Phase4: new best {sc_adv:.10f} (gen{gen}_adam)")

    print(f"Phase 4 done: best={best_score:.10f} (n={best_n})")

    # === Phase 5: Final intensive polish (remaining time) ===
    phase5_end = deadline
    if best_seq is not None:
        # Long Adam with very small beta
        if time.monotonic() < phase5_end - 80:
            per_dl = min(phase5_end - 40, time.monotonic() + 150.0)
            p_best, sc_best = _adam_polish(best_seq, per_dl, lr=0.015, beta_frac=0.002, n_max_iter=200000)
            if sc_best < best_score:
                best_seq = p_best.copy()
                best_score = sc_best
                print(f"  Phase5: new best {sc_best:.10f} (long_adam)")

        # L-BFGS fine
        if time.monotonic() < phase5_end - 30:
            per_dl = min(phase5_end - 10, time.monotonic() + 80.0)
            p_best, sc_best = _lbfgs_phase(best_seq, per_dl, max_wall_time=80.0)
            if sc_best < best_score:
                best_seq = p_best.copy()
                best_score = sc_best
                print(f"  Phase5: new best {sc_best:.10f} (fine_lbfgs)")

        # Micro restart
        if time.monotonic() < phase5_end - 20:
            a_micro = best_seq + 0.005 * rng.normal(size=len(best_seq))
            a_micro = np.maximum(a_micro, 0.0)
            a_micro = a_micro / a_micro.sum()
            per_dl = min(phase5_end - 5, time.monotonic() + 40.0)
            p_best, sc_best = _lbfgs_phase(a_micro, per_dl, max_wall_time=40.0)
            if sc_best < best_score:
                best_seq = p_best.copy()
                best_score = sc_best
                print(f"  Phase5: new best {sc_best:.10f} (micro)")

    print(f"Phase 5 done: best={best_score:.10f} (n={best_n})")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(256, 0, 1.5)

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_profile(256, 0, 1.5)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
