"""

Uses soft-max Adam with Nesterov momentum, subgradient peak-shaving, and LP spreading.
Primary target: n=2048 for more degrees of freedom to reduce peak/sum ratio.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _soft_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_soft_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgrad_at_lag(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _adam_nesterov(a, deadline, max_iter=30000, lr=0.025, beta_init=0.08,
                    beta_decay=0.75, beta_decay_interval=120,
                    nesterov=True, restart_interval=5000, verbose=False):
    """Adam with Nesterov momentum and adaptive soft-max beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    v_z = np.zeros(n)  # Nesterov velocity
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_init * float(c.max()), 1e-12)
    no_imp = 0
    lr_cur = lr
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _soft_max_weights(c_curr, beta)
        grad_a = _grad_soft_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            v_z = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        adam_step = mh / (np.sqrt(vh) + eps)

        if nesterov:
            v_z = 0.9 * v_z - lr_cur * adam_step
            z_new = z + v_z
        else:
            z_new = z - lr_cur * adam_step

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.95)
            if no_imp > restart_interval:
                rng_r = np.random.default_rng(777 + it)
                # Structured perturbation: mix of noise and sinusoidal
                pert = rng_r.normal(size=n) * 0.2
                # Add a few sinusoidal components
                for freq in [1, 2, 3, 5]:
                    pert += 0.1 * rng_r.normal() * np.cos(2 * np.pi * freq * np.arange(n) / n)
                z = best_z + pert
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-12, beta_init * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                v_z = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % beta_decay_interval == 0:
            beta = max(1e-14, beta * beta_decay)

        if verbose and it % 5000 == 0:
            print(f"      AdamN it={it}: best={best_score:.10f}, beta={beta:.2e}, no_imp={no_imp}")

    return best_a, best_score


def _peak_shave(a, deadline, max_iter=8000, lr_init=0.5, verbose=False):
    """Direct subgradient descent on the max lag with adaptive step size."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = lr_init / np.sqrt(n)
    momentum = np.zeros(n)
    mu = 0.7
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_idx = int(np.argmax(c_curr))
        sg = _subgrad_at_lag(a, max_idx)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        # Adaptive step size
        step = lr / np.sqrt(it + 1)
        momentum = mu * momentum + (1 - mu) * (sg / sg_norm)
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(lr * 0.9, 0.01 / np.sqrt(n))
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _lp_spread_topk(a, M, top_k=8):
    """LP that only constrains the top-k autocorrelation lags."""
    n = len(a)
    c = _fft_convolve(a)
    # Get top-k lag indices
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]

    # Build sparse constraint matrix
    rows = []
    for lag in top_idx:
        lag = int(lag)
        j1_start = max(0, lag - n + 1)
        j1_end = min(lag, n - 1)
        for j in range(j1_start, j1_end + 1):
            rows.append((lag, j, a[lag - j]))
        j2_start = max(lag, 0)
        j2_end = min(lag + n - 1, n - 1)
        for j in range(j2_start, j2_end + 1):
            rows.append((lag, j, a[j - lag]))

    if not rows:
        return np.ones(n) / n

    # Build dense matrix (small enough for top_k=8)
    A = np.zeros((k, n))
    for lag_i, lag in enumerate(top_idx):
        lag = int(lag)
        j1_start = max(0, lag - n + 1)
        j1_end = min(lag, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            A[lag_i, idx] += a[lag - idx]
        j2_start = max(lag, 0)
        j2_end = min(lag + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            A[lag_i, idx] += a[idx - lag]

    c_obj = -np.ones(n)
    b = np.full(k, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.002, blend=0.3,
               top_k=8, verbose=False):
    """LP-based refinement targeting top-k peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread_topk(a, M_lp, top_k)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.05)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(150, n // 4):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        if s > 1e-30:
            return p / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _optimize_at_length(a0, deadline, n, label=""):
    """Run full optimization pipeline at a given length."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  [{label}] n={n} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Coarse soft-max Adam
    if time.monotonic() < deadline - 80:
        d1 = min(deadline - 80, time.monotonic() + 150)
        a1, s1 = _adam_nesterov(a0, d1, max_iter=20000, lr=0.025,
                                beta_init=0.08, beta_decay=0.75,
                                beta_decay_interval=100)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  [{label}] n={n} Adam1: {best_sc:.10f}")

    # Phase 2: Peak shaving
    if time.monotonic() < deadline - 60:
        d2 = min(deadline - 60, time.monotonic() + 80)
        a2, s2 = _peak_shave(best_a, d2, max_iter=6000, lr_init=0.4)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  [{label}] n={n} Shave1: {best_sc:.10f}")

    # Phase 3: Fine soft-max Adam
    if time.monotonic() < deadline - 50:
        d3 = min(deadline - 50, time.monotonic() + 120)
        a3, s3 = _adam_nesterov(best_a, d3, max_iter=15000, lr=0.01,
                                beta_init=0.03, beta_decay=0.7,
                                beta_decay_interval=80)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  [{label}] n={n} Adam2: {best_sc:.10f}")

    # Phase 4: Peak shaving + LP (only for n <= 2048)
    if n <= 2048 and time.monotonic() < deadline - 40:
        d4 = min(deadline - 40, time.monotonic() + 80)
        a4 = _lp_refine(best_a, d4, max_iter=1500, inflate=1.001, blend=0.25, top_k=6)
        s4 = _score(a4)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  [{label}] n={n} LP: {best_sc:.10f}")

    # Phase 5: Ultra-fine Adam
    if time.monotonic() < deadline - 10:
        d5 = deadline - 10
        a5, s5 = _adam_nesterov(best_a, d5, max_iter=10000, lr=0.005,
                                beta_init=0.01, beta_decay=0.6,
                                beta_decay_interval=60)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  [{label}] n={n} Adam3: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimization: 512 -> 1024 -> 2048 with improved soft-max Adam,
    Nesterov momentum, peak shaving, and LP spreading.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_to_length(parent, 512)
        p1024 = _interpolate_to_length(parent, 1024)
    else:
        # Fallback initialization
        p512 = np.ones(512) / 512
        p1024 = np.ones(1024) / 1024

    # === Phase A: Optimize at n=1024 (quick, high priority) ===
    print("=== Phase A: n=1024 ===")
    dA = min(deadline - 600, t_start + 350)
    a1024, s1024 = _optimize_at_length(p1024, dA, 1024, "A")
    if s1024 < best_score:
        best_score = s1024
        best_seq = a1024.copy()
    print(f"  Phase A best: {best_score:.10f}")

    # === Phase B: Upsample to n=2048 and optimize ===
    if time.monotonic() < deadline - 400:
        print("=== Phase B: n=2048 ===")
        # Upsample the best 1024 result to 2048
        p2048_from_a = _interpolate_to_length(a1024, 2048)
        # Also try upscaling parent directly (might be slightly different)
        p2048_from_parent = _interpolate_to_length(parent, 2048) if parent is not None else p2048_from_a

        # Optimize the upscaled version
        dB = min(deadline - 300, time.monotonic() + 500)
        a2048, s2048 = _optimize_at_length(p2048_from_a, dB, 2048, "B")
        if s2048 < best_score:
            best_score = s2048
            best_seq = a2048.copy()
        print(f"  Phase B best: {best_score:.10f}")

    # === Phase C: Try a second 2048 init from parent ===
    if parent is not None and time.monotonic() < deadline - 250:
        print("=== Phase C: n=2048 (parent init) ===")
        # Add small perturbation
        p2048_pert = p2048_from_parent * (1.0 + 0.1 * rng.normal(size=2048))
        p2048_pert = np.maximum(p2048_pert, 0.0)
        p2048_pert = p2048_pert / p2048_pert.sum()

        dC = min(deadline - 200, time.monotonic() + 250)
        a2048b, s2048b = _optimize_at_length(p2048_pert, dC, 2048, "C")
        if s2048b < best_score:
            best_score = s2048b
            best_seq = a2048b.copy()
        print(f"  Phase C best: {best_score:.10f}")

    # === Phase D: Final intensive optimization on the best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"=== Phase D: Final optimization (best={best_score:.10f}, n={best_seq.size}) ===")
        n_best = best_seq.size

        # Fine Adam at current length
        dD1 = min(deadline - 15, time.monotonic() + 120)
        aD1, sD1 = _adam_nesterov(best_seq, dD1, max_iter=15000, lr=0.004,
                                  beta_init=0.005, beta_decay=0.5,
                                  beta_decay_interval=50)
        if sD1 < best_score:
            best_score = sD1
            best_seq = aD1.copy()
        print(f"  Phase D Adam: {best_score:.10f}")

        # Peak shave
        if time.monotonic() < deadline - 12:
            dD2 = min(deadline - 12, time.monotonic() + 50)
            aD2, sD2 = _peak_shave(best_seq, dD2, max_iter=4000, lr_init=0.2)
            if sD2 < best_score:
                best_score = sD2
                best_seq = aD2.copy()
            print(f"  Phase D Shave: {best_score:.10f}")

        # LP if n is manageable
        if best_seq.size <= 2048 and time.monotonic() < deadline - 10:
            dD3 = min(deadline - 10, time.monotonic() + 60)
            aD3 = _lp_refine(best_seq, dD3, max_iter=1000, inflate=1.0005,
                             blend=0.15, top_k=5)
            sD3 = _score(aD3)
            if sD3 < best_score:
                best_score = sD3
                best_seq = aD3.copy()
            print(f"  Phase D LP: {best_score:.10f}")

        # Final micro-Adam
        if time.monotonic() < deadline - 5:
            dD4 = deadline - 5
            aD4, sD4 = _adam_nesterov(best_seq, dD4, max_iter=8000, lr=0.002,
                                      beta_init=0.002, beta_decay=0.4,
                                      beta_decay_interval=40)
            if sD4 < best_score:
                best_score = sD4
                best_seq = aD4.copy()
            print(f"  Phase D Micro: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256

    # Validate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
