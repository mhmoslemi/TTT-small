"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with logsumexp smooth max, parametric initialization,
active-set equalization, Adam polishing, and multi-resolution upsampling.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smooth_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _smooth_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_run(z_init, beta, n, max_wall_time):
    """Run L-BFGS with given beta for a wall-time budget."""
    start = time.monotonic()
    try:
        result = minimize(

            options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12,
                     'maxls': 50, 'maxfun': 5000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return z_init


def _continuation_optimize(a_init, deadline, rng, base_betas=None):
    """Multi-temperature L-BFGS continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if base_betas is None:
        base_betas = [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001, 0.00005]

    n_levels = len(base_betas)

        beta = bf * max_c0

            best_a = a_new.copy()

        else:

def _adam_optimize(a_init, deadline, rng, lr=0.05, beta_frac=0.01, max_iter=50000):
    """Adam optimizer on the smooth max objective in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    beta = max(1e-10, beta_frac * float(c.max()))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        sc = _score_norm_fn(_softmax_fn(z))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _softmax_fn(z).copy()

        if t % 100 == 0:
            beta = max(1e-12, beta * 0.88)

        if t % 5000 == 0:
            lr = max(0.003, lr * 0.92)

    return best_a, best_score


def _parametric_gaussian(n, mu, sigma, shift):
    """Gaussian bump: a[j] = exp(-((j-nu)^2)/(2*sigma^2)) * (1+shift*(2j/(n-1)-1))"""
    j = np.arange(n, dtype=np.float64)
    u = 2.0 * j / (n - 1) - 1.0
    base = np.exp(-((j - mu) ** 2) / (2.0 * sigma * sigma))
    asym = 1.0 + shift * u
    a = base * np.maximum(asym, 1e-10)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_poly(n, p, mu, w):
    """Polynomial: a[j] = (1 + p*(2j/(n-1)-1) + w*(2j/(n-1)-1)^2) clipped"""
    j = np.arange(n, dtype=np.float64)
    u = 2.0 * j / (n - 1) - 1.0
    a = 1.0 + p * u + w * u * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _parametric_cosine(n, q, mu):
    """Raised cosine: a[j] = cos(pi*(j-mu)/(n))*q clipped"""
    j = np.arange(n, dtype=np.float64)
    x = np.pi * (j - mu) / n
    a = np.cos(x) ** q
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _active_set_equalize(a_init, deadline, rng, top_k=20):
    """Try to equalize the top-k peaks by solving a small optimization."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = 0.002
    for outer in range(200):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top-k active lags
        idx = np.argsort(c)[::-1][:top_k]
        active_vals = c[idx]
        mean_active = float(active_vals.mean())

        # Compute subgradient: average of gradients at active lags
        # dF/da[m] = 2n * sum_{k in active} (1/|active|) * 2*a[k-m]
        # = 4n * mean_k a[k-m]
        # Using FFT: this is a filtered version
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / top_k
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Projected gradient step on simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            lr = max(1e-6, lr * 0.7)

        if lr < 1e-8:
            break

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
