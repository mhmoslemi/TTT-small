import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _compute_subgrad(a, n_peaks=5):
    """Compute subgradient of max(conv(a,a)) using top peak positions."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    top_idx = np.argsort(c)[::-1][:n_peaks]
    peak_vals = c[top_idx]
    w = peak_vals / peak_vals.sum()

    grad = np.zeros(n)
    for k, wk in zip(top_idx, w):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j_lo = k - hi
            j_hi = k - lo
            seg = a[j_lo:j_hi + 1][::-1]
            grad[lo:hi + 1] += 2.0 * wk * seg

    return grad, c, c_max


def _logit_subgrad_line_search(a, deadline, max_iter=30000, eta_init=0.1, n_peaks=5):
    """Subgradient descent in log-space with backtracking line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a)
    c_max = float(c.max())

    no_imp = 0
    eta = eta_init
    eta_min = 1e-6
    momentum = np.zeros(n)
    mu = 0.9

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, c, c_max = _compute_subgrad(a, n_peaks)
        # Log-space gradient
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        # momentum
        momentum = mu * momentum + (1.0 - mu) * gz
        direction = -momentum

        # Backtracking line search
        found = False
        eta_try = eta
        cur_score = best_score
        for bs in range(20):
            z_new = z + eta_try * direction
            z_new = np.clip(z_new, -50.0, 50.0)
            z_new -= np.mean(z_new)
            e = np.exp(z_new)
            a_new = e / e.sum()
            sc = _score(a_new)
            if sc < cur_score - 1e-15:
                z = z_new
                a = a_new
                eta = min(eta * 1.1, 0.5)
                found = True
                break
            eta_try *= 0.5

        if not found:
            no_imp += 1
            eta = max(eta_min, eta * 0.7)
            if no_imp % 500 == 0:
                # Annealing: add noise
                rng = np.random.default_rng(it * 31 + 7)
                z += 0.1 * rng.standard_normal(n)
                z -= np.mean(z)
                momentum = np.zeros(n)
                eta = eta_init
            if no_imp > 5000:
                break
            continue

        # Update best
        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0

    return best_a, best_score

                blend_cur = blend_init

                no_imp = 0

def _spectral_shaping(a, deadline, max_iter=2000):
    """Reduce peak by shaping the power spectrum."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        peak_k = int(np.argmax(c))

        # Compute DFT of a
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # The peak at lag k is influenced by |A_m|^2 * cos(2*pi*m*k/N)
        # To reduce the peak, reduce |A_m| where cos(2*pi*m*k/N) is large and positive
        freqs = np.arange(N // 2 + 1)
        alignment = np.cos(2.0 * np.pi * freqs * peak_k / N)
        # Only reduce where alignment is positive (constructive interference at the peak)
        reduce_mask = alignment > 0.3
        power = mag ** 2
        # Reduce power at constructive frequencies
        power[reduce_mask] *= 0.95
        new_mag = np.sqrt(power)
        new_A = new_mag * np.exp(1j * phase)
        new_A_full = np.zeros(N, dtype=complex)
        new_A_full[:N // 2 + 1] = new_A
        for m in range(1, N // 2 + 1):
            new_A_full[N - m] = np.conj(new_A[m])
        a_new = np.real(np.fft.ifft(new_A_full))[:n]
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break

    return best_a, best_score

def _window_smooth(a, deadline, max_iter=5000):
    """Apply small window smoothing to reduce local variations that contribute to peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_score = _score(a)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_diverse_init(n, rng):
    """Create a diverse set of initializations."""
    inits = []

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    # Bell profiles
    for width in [0.2, 0.3, 0.4, 0.6]:
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.5) / width) ** 2)
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())

    # Edge-concentrated
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) * 4) ** 2) + 0.5 * np.exp(-(t / 0.1) ** 2) + 0.5 * np.exp(-((t - 1) / 0.1) ** 2)
    a = np.maximum(a, 1e-12)
    inits.append(a / a.sum())

    # Flat spectrum (random phases)
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.abs(np.real(np.fft.ifft(X)))[:n]
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())

    # cosine taper
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * t / 2) ** 2
    a = np.maximum(a, 1e-12)
    inits.append(a / a.sum())

    return inits

        if parent is not None:
            a0 = _interpolate_profile(parent, n_t)
            sc = _score(a0)
            candidates.append((sc, f'parent_n{n_t}', a0))

        # Diverse inits
        inits = _make_diverse_init(n_t, rng)
        for i, a0 in enumerate(inits):
            sc = _score(a0)
            candidates.append((sc, f'init{n_t}_{i}', a0))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 initializations:", flush=True)
    for sc, name, _ in candidates[:5]:
        print(f"    {name}: {sc:.8f}", flush=True)

    # Phase 2: Main optimization on top candidates
    # Allocate time proportionally
    remaining = deadline - time.monotonic()
    n_opt_candidates = min(4, len(candidates))

    for ci in range(n_opt_candidates):
        sc0, name, a0 = candidates[ci]
        n_t = len(a0)
        sub_deadline = min(deadline - 100, time.monotonic() + remaining * 0.45 / max(1, n_opt_candidates - ci))
        if time.monotonic() > deadline - 80:
            break
        if sc0 > best_score * 1.1 and ci > 0:
            continue

        print(f"Phase 2: Subgrad+LS on {name} (n={n_t}), start={sc0:.8f}, budget={sub_deadline - time.monotonic():.0f}s", flush=True)

        # Subgradient with line search
        a_opt, sc_opt = _logit_subgrad_line_search(a0, sub_deadline, max_iter=50000, eta_init=0.08, n_peaks=5)
        print(f"  Subgrad result: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # Spectral shaping
        if time.monotonic() < deadline - 60:
            ss_deadline = min(deadline - 50, time.monotonic() + 60.0)
            a_ss, sc_ss = _spectral_shaping(a_opt, ss_deadline, max_iter=2000)
            if sc_ss < best_score:
                best_score = sc_ss
                best_seq = a_ss.copy()
                a_opt = a_ss
            print(f"  Spectral: {sc_ss:.10f}", flush=True)

    # Phase 3: LP refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        lp_deadline = min(deadline - 40, time.monotonic() + 300.0)
        print(f"Phase 3: LP refine (n={n_best}), budget={lp_deadline - time.monotonic():.0f}s", flush=True)
        a_lp, sc_lp = _lp_refine(best_seq, lp_deadline, max_iter=15000, inflate_init=1.001, blend_init=0.15)
        print(f"  LP result: {sc_lp:.10f}", flush=True)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

    # Phase 4: Local search
    if best_seq is not None and time.monotonic() < deadline - 40:
        ps_deadline = min(deadline - 20, time.monotonic() + 200.0)
        print(f"Phase 4: Pairwise LS, budget={ps_deadline - time.monotonic():.0f}s", flush=True)
        a_ps, sc_ps = _pairwise_local_search(best_seq, ps_deadline, max_iter=80000)
        print(f"  Pairwise LS: {sc_ps:.10f}", flush=True)
        if sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()

    # Phase 5: Window smoothing
    if best_seq is not None and time.monotonic() < deadline - 20:
        ws_deadline = min(deadline - 10, time.monotonic() + 100.0)
        a_ws, sc_ws = _window_smooth(best_seq, ws_deadline, max_iter=5000)
        print(f"  Window smooth: {sc_ws:.10f}", flush=True)
        if sc_ws < best_score:
            best_score = sc_ws
            best_seq = a_ws.copy()

    # Phase 6: Another LP pass if time allows
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp2_deadline = deadline - 3
        if lp2_deadline - time.monotonic() > 15:
            a_lp2, sc_lp2 = _lp_refine(best_seq, lp2_deadline, max_iter=5000, inflate_init=1.0005, blend_init=0.1)
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()
            print(f"  Final LP: {sc_lp2:.10f}", flush=True)

    # Phase 7: Final subgrad polish
    if best_seq is not None and time.monotonic() < deadline - 3:
        gf_deadline = deadline - 2
        if gf_deadline - time.monotonic() > 10:
            a_gf, sc_gf = _logit_subgrad_line_search(best_seq, gf_deadline, max_iter=10000, eta_init=0.02, n_peaks=3)
            if sc_gf < best_score:
                best_score = sc_gf
                best_seq = a_gf.copy()
            print(f"  Final polish: {sc_gf:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(768) / 768

    # Validate
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
