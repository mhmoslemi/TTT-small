"""
Optimized step function search using smoothed gradient descent with FFT-based
gradients, LP polish (min-t formulation), parameterized initialization sweeps,
and multi-length search. Targets improvement over 1.506869 toward 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute gradient of smoothed max of conv(a,a) using FFT cross-correlation."""
    n = len(a)
    b = _fft_convolve(a)
    mval = float(b.max())
    x = (b - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w = ex / s

    # Cross-correlation: c[i] = sum_k w[k+i] * a[k] for i=0..n-1
    L = _next_pow2(2 * n + len(w) - 1)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(W * np.conj(A), L)
    grad = 2.0 * c[:n]
    return grad, mval


def _make_power(n, p=1.0, eps=0.01):
    t = np.linspace(0.0, 1.0, n)
    a = (t + eps) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * t)
    return a / a.sum()


def _make_poly(n, c1=1.0, c2=0.5, c3=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c1 * t + c2 * t**2 + c3 * t**3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, center=0.5, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _solve_lp_mint(a, n):
    """
    Solve: min t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None) on failure.
    """
    # Variables: [t, g_0, ..., g_{n-1}]
    # Objective: minimize t -> c = [1, 0, 0, ..., 0]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

    # Constraints: conv(a,g)_k - t <= 0 for k = 0, ..., 2n-2
    # Row k: [-1, C[k,0], C[k,1], ..., C[k,n-1]] where C[k,j] = a[j] if 0<=k-j<n
    n_constr = 2 * n - 1
    A_ub = np.zeros((n_constr, n + 1))
    A_ub[:, 0] = -1.0
    for j in range(n):
        A_ub[j:j + n, j + 1] = a

    b_ub = np.zeros(n_constr)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] + [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        t_opt = float(res.x[0])
        g = res.x[1:]
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g, t_opt
    return None, None

                if no_imp % 200 == 0:
                    lr = max(0.01, lr * 0.9)
                if no_imp > max(2000, 5 * n):
                    a = best_a.copy()
                    v = np.zeros(n)
                    beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
                    lr = base_lr
                    no_imp = 0

        # Anneal beta
        if it % 200 == 199:
            beta = max(1e-8, beta * 0.85)

        # Slowly increase momentum
        momentum = min(0.95, momentum + 0.0001)

    return best_a, best_score


def _parameter_sweep(n, deadline):
    """Sweep over parameterized families to find good initializations."""
    candidates = []

    # Power laws
    for p in [0.5, 0.75, 1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0]:
        for eps in [0.0, 0.01, 0.05, 0.1]:
            a = _make_power(n, p, eps)
            candidates.append(a)

    # Exponentials
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        candidates.append(_make_exp(n, alpha))

    # Polynomials
    for c1 in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        for c2 in [0.0, 0.5, 1.0, 2.0]:
            for c3 in [0.0, -0.5, 0.5, 1.0]:
                candidates.append(_make_poly(n, c1, c2, c3))

    # Off-center Gaussians (broader peaks)
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.2, 0.3, 0.4, 0.5]:
            candidates.append(_make_gaussian(n, center, width))

    # Evaluate and sort
    scored = []
    for a in candidates:
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((sc, a))

    scored.sort(key=lambda x: x[0])

    # Try top few with gradient descent
    best_a = None
    best_score = np.inf
    top_k = min(8, len(scored))

    for i in range(top_k):
        sc0, a0 = scored[i]
        if sc0 >= best_score:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        gd_time = min(120.0, remaining * 0.4)
        gd_end = time.monotonic() + gd_time
        a_gd, sc_gd = _gradient_descent(a0, gd_end, max_iter=50000)
        if sc_gd < best_score:
            best_score = sc_gd
            best_a = a_gd.copy()
            print(f"  Param sweep: init_score={sc0:.8f} -> gd_score={sc_gd:.10f}")

    return best_a, best_score


def _diversify_and_refine(a0, deadline, rng):
    """Add random perturbations and re-optimize to escape local optima."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    n_tries = 5
    for i in range(n_tries):
        if time.monotonic() > deadline - 60:
            break
        # Random perturbation
        noise_scale = 0.05 * (1.0 - i / n_tries)
        noise = rng.standard_normal(n) * noise_scale
        a_pert = a + noise
        a_pert = _project_simplex(a_pert)

        # Quick gradient descent
        gd_end = time.monotonic() + min(60.0, (deadline - time.monotonic()) / (n_tries - i))
        a_gd, sc_gd = _gradient_descent(a_pert, gd_end, max_iter=20000)
        if sc_gd < best_score - 1e-10:
            best_score = sc_gd
            best_a = a_gd.copy()
            a = a_gd.copy()
            print(f"  Diversify {i}: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: parameter sweep + smoothed gradient descent + LP polish
    + multi-length search + diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    lengths_to_try = [512, 768, 1024, 1536]

    # Phase 1: Parameter sweep on each length
    for n_t in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        print(f"\n=== Parameter sweep n={n_t} ===")
        a_sweep, sc_sweep = _parameter_sweep(n_t, deadline - 100)
        if a_sweep is not None and sc_sweep < best_score:
            best_score = sc_sweep
            best_seq = a_sweep.copy()
            print(f"  Sweep best: {sc_sweep:.10f}")

    # Phase 2: If parent available, interpolate and optimize
    if parent is not None:
        for n_t in lengths_to_try:
            if time.monotonic() > deadline - 200:
                break
            a0 = _interpolate(parent, n_t)
            sc0 = _score(a0)
            print(f"\n=== Parent interp n={n_t}: init={sc0:.8f} ===")

            # Gradient descent
            if time.monotonic() < deadline - 150:
                gd_end = time.monotonic() + min(180.0, deadline - time.monotonic() - 100)
                a_gd, sc_gd = _gradient_descent(a0, gd_end, max_iter=100000)
                if sc_gd < best_score:
                    best_score = sc_gd
                    best_seq = a_gd.copy()
                    print(f"  GD: {sc_gd:.10f}")

            # LP polish
            if time.monotonic() < deadline - 60:
                lp_end = time.monotonic() + min(120.0, deadline - time.monotonic() - 40)
                a_lp, sc_lp = _lp_refine_mint(a_gd, lp_end, max_iter=3000)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"  LP: {sc_lp:.10f}")

    # Phase 3: Diversification from best
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Diversification from best (score={best_score:.10f}) ===")
        a_div, sc_div = _diversify_and_refine(best_seq, deadline - 30, rng)
        if sc_div < best_score:
            best_score = sc_div
            best_seq = a_div.copy()

    # Phase 4: Final LP polish on the very best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        lp_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 10)
        print(f"\n=== Final LP polish (n={n_best}) ===")
        a_final, sc_final = _lp_refine_mint(best_seq, lp_end, max_iter=2000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  Final LP: {sc_final:.10f}")

    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        gd_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 2)
        a_gd2, sc_gd2 = _gradient_descent(best_seq, gd_end, max_iter=50000)
        if sc_gd2 < best_score:
            best_score = sc_gd2
            best_seq = a_gd2.copy()
            print(f"  Final GD: {sc_gd2:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_power(512, 1.5, 0.05)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_power(512, 1.5, 0.05)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
