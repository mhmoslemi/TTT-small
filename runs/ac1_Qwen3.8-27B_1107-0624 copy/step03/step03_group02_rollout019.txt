"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses Chebyshev polynomial parametrization, Lp-norm warm-up, adaptive smooth-max Adam,
and multi-length search with spectral gradient computation.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_conv_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a)."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _lp_norm_objective(c, p):
    """Compute sum(c^p) as a smooth proxy for max(c)."""
    return float(np.sum(c ** p))


def _lp_norm_weights(c, p):
    """Weights for Lp norm: w[k] = p * c[k]^(p-1) / sum(c^p)."""
    cp = c ** p
    s = float(cp.sum())
    if s < 1e-30:
        return np.ones_like(c) / len(c)
    return p * c ** (p - 1) / s


def _adam_logit(a, deadline, max_iter=30000, lr=0.03, beta_factor=0.04,
                use_lp_warmup=True, p_warm=4, warmup_frac=0.2, verbose=False):
    """Adam optimizer in logit-space with optional Lp warm-up and adaptive beta."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = beta_factor * float(c.max())
    beta_min = 1e-4 * float(c.max())
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    warmup_end = int(warmup_frac * max_iter)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)

        if use_lp_warmup and it < warmup_end:
            w = _lp_norm_weights(c_curr, p_warm)
        else:
            w = _smooth_max_weights(c_curr, beta)

        grad_a = _grad_conv_weighted(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.002, lr_cur * 0.9)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(7000 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = beta_factor * float(_fft_convolve(a).max())
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0

        z = z_new
        a = a_new

        # Adaptive beta: decrease when improving, keep stable otherwise
        if it > warmup_end:
            if no_imp == 0:
                beta = max(beta_min, beta * 0.85)
            elif it % 100 == 0:
                beta = max(beta_min, beta * 0.95)

        if verbose and it % 5000 == 0:
            print(f"        Adam it={it}: best={best_score:.10f}, beta={beta:.2e}, lr={lr_cur:.4f}")

    return best_a, best_score


def _chebyshev_profile(coeffs, n):
    """Evaluate Chebyshev polynomial profile: a[i] = exp(sum c_k T_k(2i/(n-1)-1)) / Z."""
    if n == 1:
        return np.array([1.0])
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    f = np.zeros(n)
    for k, ck in enumerate(coeffs):
        f += ck * np.cos(k * np.arccos(np.clip(x, -1, 1)))
    a = np.exp(np.clip(f, -30, 30))
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _fit_chebyshev(a, d):
    """Fit Chebyshev coefficients to log(a)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    log_a = np.log(a / a.sum())
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    # Build Chebyshev design matrix
    T = np.zeros((n, d + 1))
    for k in range(d + 1):
        T[:, k] = np.cos(k * np.arccos(np.clip(x, -1, 1)))
    # Least squares
    coeffs, _, _, _ = np.linalg.lstsq(T, log_a, rcond=None)
    return coeffs


def _optimcheb(deadline, n, d=6, seed=42, verbose=False):
    """Optimize Chebyshev coefficients using gradient descent."""
    rng = np.random.default_rng(seed)
    # Initialize with zero coefficients (uniform) plus small perturbation
    best_coeffs = np.zeros(d + 1)
    best_a = _chebyshev_profile(best_coeffs, n)
    best_score = _score(best_a)

    # Try a few initializations
    inits = [np.zeros(d + 1)]
    for _ in range(3):
        c_init = rng.uniform(-0.5, 0.5, size=d + 1)
        inits.append(c_init)

    for c0 in inits:
        coeffs = c0.copy()
        lr = 0.05
        for it in range(500):
            if time.monotonic() > deadline:
                break
            a = _chebyshev_profile(coeffs, n)
            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_coeffs = coeffs.copy()

            # Gradient via finite differences (low dim, cheap)
            grad = np.zeros(d + 1)
            h = 0.01
            for k in range(d + 1):
                c_plus = coeffs.copy()
                c_plus[k] += h
                a_plus = _chebyshev_profile(c_plus, n)
                c_plus_conv = _fft_convolve(a_plus)
                s_plus = float(2.0 * n * c_plus_conv.max())
                grad[k] = (s_plus - score) / h
            coeffs -= lr * grad
            lr *= 0.995

    if verbose:
        print(f"      Chebyshev opt: best={best_score:.10f}")
    return best_a, best_score


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        # Build LP: maximize sum(b) s.t. conv(a*b) <= M_lp, b >= 0
        # conv(a*b)[k] = sum_j a[j]*b[k-j] (with bounds)
        # This is a matrix-vector product: A @ b <= M_lp
        # A is (2n-1) x n, A[k,j] = a[j] if 0 <= k-j <= n-1 else 0
        # For large n, this is too big. Use a simpler heuristic instead.
        # Find positions where conv is at or near max, reduce a there.
        top_frac = 0.05
        n_top = max(1, int(top_frac * (2 * n - 1)))
        top_idx = np.argsort(c)[::-1][:n_top]
        # Reduce a at positions that contribute most to top peaks
        reduction = np.zeros(n)
        for k in top_idx:
            # dc[k]/da[j] = 2*a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                reduction[j_start:j_end + 1] += a[k - np.arange(j_start, j_end + 1)]
        # Apply reduction
        a_new = a - 0.1 * reduction / (np.max(reduction) + 1e-10)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        # Blend
        a_new = (1 - blend_cur) * a + blend_cur * a_new
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break
        a = a_new

    return best_a


def _subgradient_multi(a, deadline, max_iter=8000, top_k=5):
    """Multi-peak subgradient method."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    base_lr = 0.15 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        # Weighted subgradient
        weights = np.exp(-0.5 * np.arange(k))
        weights /= weights.sum()
        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            idx = int(idx)
            j_start = max(0, idx - n + 1)
            j_end = min(idx, n - 1)
            if j_start <= j_end:
                idx_arr = np.arange(j_start, j_end + 1)
                sg[idx_arr] += weights[i] * 2.0 * a[idx - idx_arr]
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break
        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profiles(n):
    """Generate diverse initial profiles of length n."""
    t = np.linspace(0.0, 1.0, n)
    profiles = []
    # Linear
    for eta in [0.8, 1.2, 1.8, 2.5]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cosine
    for alpha in [0.3, 0.6, 1.0, 1.5]:
        a = 1.0 + alpha * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cubic
    u = 2.0 * t - 1.0
    for a1, a3 in [(1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (0.5, 0.3)]:
        a = 1.0 + a1 * u + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Quintic
    for a1, a3, a5 in [(1.0, 0.0, -0.3), (0.5, 0.5, -0.5), (1.5, -0.3, 0.2)]:
        a = 1.0 + a1 * u + a3 * u ** 3 + a5 * u ** 5
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Asymmetric exponential
    for rate in [0.5, 1.0, 2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Two-bump
    for w in [0.2, 0.3]:
        a = np.exp(-((t - 0.3) ** 2) / (2 * w ** 2)) + 0.5 * np.exp(-((t - 0.7) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: Chebyshev param warm-start, Lp-norm + adaptive smooth-max Adam,
    multi-length search, and subgradient polish. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # === Phase 1: Chebyshev optimization at multiple lengths ===
    print("=== Phase 1: Chebyshev optimization ===")
    lengths = [512, 768, 1024, 1536]
    cheb_results = {}

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        d_cheb = min(8, max(4, n // 128))
        d1 = min(deadline - 300, time.monotonic() + 30)
        a_c, s_c = _optimcheb(d1, n, d=d_cheb, seed=seed, verbose=True)
        cheb_results[n] = a_c
        if s_c < best_score:
            best_score = s_c
            best_seq = a_c.copy()
        print(f"  Cheb n={n}: {s_c:.10f}")

    # === Phase 2: Full Adam optimization on best Chebyshev results ===
    print("=== Phase 2: Adam smooth-max optimization ===")

    # Determine which lengths to fully optimize
    opt_lengths = set()
    if parent is not None:
        pn = parent.size
        # Round to nearest in our list
        closest = min(lengths, key=lambda x: abs(x - pn))
        opt_lengths.add(closest)
        opt_lengths.add(min(closest + 256, 2048))

    # Add best Chebyshev length
    if cheb_results:
        best_cheb_n = min(cheb_results.keys(), key=lambda k: _score(cheb_results[k]))
        opt_lengths.add(best_cheb_n)

    # Always include 1024 and 512
    opt_lengths.add(1024)
    opt_lengths.add(512)

    # Sort by current best score at that length
    def _length_key(n):
        if parent is not None and n == min(lengths, key=lambda x: abs(x - parent.size)):
            return _score(_interpolate_profile(parent, n))
        if n in cheb_results:
            return _score(cheb_results[n])
        return 3.0  # default

    opt_lengths_sorted = sorted(opt_lengths, key=_length_key)

    for n in opt_lengths_sorted:
        if time.monotonic() > deadline - 100:
            break
        # Choose best init for this length
        if parent is not None:
            init = _interpolate_profile(parent, n)
        elif n in cheb_results:
            init = cheb_results[n].copy()
        else:
            init = _make_profiles(n)[4]  # cosine alpha=1.0

        # Try Chebyshev result too
        if n in cheb_results:
            sc_init = _score(init)
            sc_cheb = _score(cheb_results[n])
            if sc_cheb < sc_init:
                init = cheb_results[n].copy()

        sc_init = _score(init)
        print(f"  Adam n={n}, init_score={sc_init:.10f}")

        # Allocate time proportionally
        remaining = deadline - 100 - time.monotonic()
        n_left = len(opt_lengths_sorted) - opt_lengths_sorted.index(n)
        per_budget = max(60, remaining / max(1, n_left))

        d2 = min(deadline - 100, time.monotonic() + per_budget)
        a_opt, s_opt = _adam_logit(init, d2, max_iter=30000, lr=0.03,
                                   beta_factor=0.04, use_lp_warmup=True,
                                   p_warm=4, warmup_frac=0.15, verbose=False)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
        print(f"    Adam n={n}: {s_opt:.10f}")

        # Subgradient polish
        if time.monotonic() < deadline - 80:
            d2s = min(deadline - 80, time.monotonic() + 40)
            a_sg, s_sg = _subgradient_multi(a_opt, d2s, max_iter=5000, top_k=5)
            if s_sg < best_score:
                best_score = s_sg
                best_seq = a_sg.copy()
            print(f"    SubGrad n={n}: {s_sg:.10f}")

    # === Phase 3: Diverse initializations at promising lengths ===
    print("=== Phase 3: Diverse initializations ===")
    for n in [512, 1024]:
        if time.monotonic() > deadline - 80:
            break
        profiles = _make_profiles(n)
        # Score all profiles quickly
        scored = [(_score(p), i, p) for i, p in enumerate(profiles)]
        scored.sort(key=lambda x: x[0])

        # Take top 3 and optimize
        for rank in range(min(3, len(scored))):
            if time.monotonic() > deadline - 60:
                break
            _, idx, p0 = scored[rank]
            d3 = min(deadline - 60, time.monotonic() + 30)
            a_d, s_d = _adam_logit(p0, d3, max_iter=10000, lr=0.025,
                                   beta_factor=0.05, use_lp_warmup=True,
                                   p_warm=4, warmup_frac=0.2)
            if s_d < best_score:
                best_score = s_d
                best_seq = a_d.copy()
            print(f"  Diverse n={n} profile#{idx}: {s_d:.10f}")

    # === Phase 4: Final polish on best sequence ===
    print(f"=== Phase 4: Final polish (best={best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        best_n = best_seq.size

        # Very fine Adam
        d4a = min(deadline - 8, time.monotonic() + 80)
        a_f1, s_f1 = _adam_logit(best_seq, d4a, max_iter=20000, lr=0.008,
                                 beta_factor=0.015, use_lp_warmup=False, verbose=False)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Fine Adam: {s_f1:.10f}")

        # Subgradient
        if time.monotonic() < deadline - 6:
            d4b = min(deadline - 6, time.monotonic() + 40)
            a_f2, s_f2 = _subgradient_multi(best_seq, d4b, max_iter=5000, top_k=3)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final SubGrad: {s_f2:.10f}")

        # LP-style peak reduction
        if time.monotonic() < deadline - 4:
            d4c = min(deadline - 4, time.monotonic() + 30)
            a_f3 = _lp_refine(best_seq, d4c, max_iter=1500)
            s_f3 = _score(a_f3)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  LP Refine: {s_f3:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 2:
            a_f4, s_f4 = _adam_logit(best_seq, deadline - 2, max_iter=10000,
                                     lr=0.003, beta_factor=0.005,
                                     use_lp_warmup=False)
            if s_f4 < best_score:
                best_score = s_f4
                best_seq = a_f4.copy()
            print(f"  Ultra-fine Adam: {s_f4:.10f}")

    # === Phase 5: Try re-interpolating best to other lengths ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        best_n = best_seq.size
        for target_n in [512, 768, 1024, 1536]:
            if target_n == best_n or time.monotonic() > deadline - 3:
                continue
            interp = _interpolate_profile(best_seq, target_n)
            sc_i = _score(interp)
            if sc_i < best_score * 1.001:  # Close enough to try optimizing
                d5 = min(deadline - 3, time.monotonic() + 20)
                a_r, s_r = _adam_logit(interp, d5, max_iter=5000, lr=0.01,
                                       beta_factor=0.01, use_lp_warmup=False)
                if s_r < best_score:
                    best_score = s_r
                    best_seq = a_r.copy()
                print(f"  Re-interp to n={target_n}: {s_r:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        final_seq = final_seq / s_final
    else:
        final_seq = np.ones(256) / 256.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
        else:
            print(f"Evaluate_sequence returned inf, using fallback")
            final_seq = np.ones(512) / 512.0
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            print(f"Fallback score: {sc_eval:.10f}")
    except Exception as e:
        print(f"Evaluate error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
