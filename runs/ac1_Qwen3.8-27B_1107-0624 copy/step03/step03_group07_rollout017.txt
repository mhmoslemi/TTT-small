Uses scipy L-BFGS on logit-space with adaptive soft-max smoothing,
multi-scale beta annealing, multiple initializations, and coordinate descent polish.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_h(w, a):
    """
    Compute h[i] = sum_j a[j] * w[i+j] for i=0,...,n-1.
    This gives grad_a = 2*h for the soft-max objective.
    """
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = rfft(af, N)
    fw = rfft(w, N)
    conv = irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_value_weights(c, beta):
    """Compute soft-max value and softmax weights."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-300:
        return np.full_like(z, 1.0 / len(z))
    return a / s


def lbfgs_objective(z, n, beta):
    """Objective and gradient for L-BFGS in logit space."""
    a = softmax(z)
    c = full_convolve_fft(a)
    val, w = soft_max_value_weights(c, beta)
    h = compute_h(w, a)
    grad_a = 2.0 * h
    dot = float(np.dot(a, grad_a))
    grad_z = 2.0 * n * a * (grad_a - dot)
    return 2.0 * n * val, grad_z

def interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, a)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_random_smooth(n, rng, n_freqs=8):
    """Generate a random smooth non-negative sequence."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_freqs + 1):
        amp = rng.uniform(0, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_tapered(n, rng, taper_power=1.0):
    """Tapered (raised cosine or power-law) profile."""
    t = np.linspace(0.0, 1.0, n)
    # Raised cosine taper
    taper = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = taper ** taper_power + 0.1
    a = a + 0.05 * rng.normal(size=n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

"""
L-BFGS with adaptive soft-max beta annealing, subgradient refinement, and coordinate
descent polish for minimizing 2n*max(conv(a,a))/sum(a)^2. Uses scipy L-BFGS on
logit-space parameters with multi-scale optimization across sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_h(w, a):
    """
    Compute h[i] = sum_j a[j] * w[i+j] for i=0,...,n-1.
    grad_a = 2*h for the soft-max of convolution objective.
    """
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_value_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-300:
        return np.full_like(z, 1.0 / len(z))
    return a / s


def run_lbfgs(a0, deadline, beta_init=0.05, beta_decay=0.7,
              max_iter_per_phase=150, no_improve_limit=5):
    """
    Run L-BFGS with adaptive beta annealing on logit-space parameters.
    Returns (best_a, best_score) where score = 2n*max(conv(a,a)).
    """
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    z = np.log(a0)
    z = z - z.mean()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    beta = beta_init * max_c
    if beta < 1e-10:
        beta = 1e-6

    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    phase = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Define objective and gradient for this beta
        def obj_and_grad(z_val):
            a_val = softmax(z_val)
            c_val = full_convolve_fft(a_val)
            val, w_val = soft_max_value_weights(c_val, beta)
            h_val = compute_h(w_val, a_val)
            grad_a = 2.0 * h_val
            dot = float(np.dot(a_val, grad_a))
            grad_z = 2.0 * n * a_val * (grad_a - dot)
            return 2.0 * n * val, grad_z

        def fun_only(z_val):
            a_val = softmax(z_val)
            c_val = full_convolve_fft(a_val)
            val, _ = soft_max_value_weights(c_val, beta)
            return 2.0 * n * val

        def jac_only(z_val):
            a_val = softmax(z_val)
            c_val = full_convolve_fft(a_val)
            _, w_val = soft_max_value_weights(c_val, beta)
            h_val = compute_h(w_val, a_val)
            grad_a = 2.0 * h_val
            dot = float(np.dot(a_val, grad_a))
            return 2.0 * n * a_val * (grad_a - dot)

        # Run L-BFGS
        try:
            result = minimize(
                fun=fun_only,
                x0=z,
                jac=jac_only,
                method='L-BFGS-B',
                bounds=[(-40, 40)] * n,
                options={
                    'maxiter': max_iter_per_phase,
                    'maxfun': max_iter_per_phase * 2,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'iprint': 0
                }
            )
        except Exception:
            break

        z = result.x
        z = np.clip(z, -40, 40)
        z = z - z.mean()

        a = softmax(z)
        c = full_convolve_fft(a)
        max_c_new = float(c.max())
        score = float(2.0 * n * max_c_new)

        if np.isfinite(score) and score < best_score - 1e-15:
            best_score = score
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        phase += 1

        # Anneal beta
        beta *= beta_decay
        max_c = max(max_c_new, 1e-10)
        if beta < 1e-8 * max_c:
            beta = 1e-8 * max_c

        # If no improvement for several phases, try a small perturbation
        if no_improve >= no_improve_limit:
            z = best_z + 0.05 * np.random.randn(n)
            z = np.clip(z, -35, 35)
            z = z - z.mean()
            no_improve = 0

        # Check time
        if time.monotonic() > deadline - 5:
            break

    a_best = softmax(best_z)
    return a_best, best_score


def subgradient_optimize(a0, deadline, lr_init=0.008, beta_factor=0.02):
    """
    Soft-subgradient method with adaptive step size and restarts.
    """
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    z = np.log(a0)
    z = z - z.mean()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)

    lr = lr_init
    no_improve = 0
    beta = beta_factor * max_c

    while time.monotonic() < deadline:
        a = softmax(z)
        c = full_convolve_fft(a)
        max_c = float(c.max())

        # Compute soft-subgradient
        val, w = soft_max_value_weights(c, beta)
        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        # Adaptive step size (RMS-normalized)
        rms = np.sqrt(np.mean(dz * dz)) + 1e-15
        step = lr * dz / rms
        step = np.clip(step, -0.3, 0.3)

        z_new = z + step
        z_new = np.clip(z_new, -40, 40)
        z_new = z_new - z_new.mean()

        a_new = softmax(z_new)
        c_new = full_convolve_fft(a_new)
        score_new = float(2.0 * n * c_new.max())

        if np.isfinite(score_new) and score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        if no_improve % 100 == 0:
            lr *= 0.92
            beta = max(beta * 0.95, 1e-8 * max_c)

        if no_improve > 800:
            # Restart with perturbation
            z = best_z + 0.08 * np.random.randn(n)
            z = np.clip(z, -35, 35)
            z = z - z.mean()
            lr = lr_init
            no_improve = 0

    return best_a, best_score


def coordinate_descent_polish(a0, deadline):
    """
    Coordinate descent: try small perturbations of individual coordinates.
    Uses FFT for fast convolution evaluation.
    """
    n = a0.size

    best_score = float(2.0 * n * c.max())

    # Precompute FFT of a for fast updates (approximate)
    # For exact: just recompute convolution each time (fast enough for n<=2048)

    np.random.seed(0)

    while time.monotonic() < deadline - 2:
        improved = False

        # Try random subset of coordinates
        n_try = min(n, 200)
        idxs = np.random.choice(n, size=n_try, replace=False)

        for i in idxs:
            if time.monotonic() > deadline - 2:
                break
            base = a[i]
            for frac in [0.9, 1.1, 0.8, 1.2, 0.95, 1.05]:
                new_val = base * frac
                if new_val < 1e-12:
                    continue
                a_try = a.copy()
                a_try[i] = new_val
                a_try = a_try / a_try.sum()
                c_try = full_convolve_fft(a_try)
                s = float(2.0 * n * c_try.max())
                if s < best_score - 1e-14:
                    best_score = s
                    best_a = a_try.copy()
                    a = a_try.copy()
                    c = c_try
                    improved = True
                    break

        # Also try pairs of coordinates
        if time.monotonic() < deadline - 5:
            for _ in range(min(50, n // 4)):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                for fi in [0.9, 1.1]:
                    for fj in [0.9, 1.1]:
                        if time.monotonic() > deadline - 2:
                            break
                        a_try = a.copy()
                        a_try[i] = a[i] * fi
                        a_try[j] = a[j] * fj
                        a_try = np.maximum(a_try, 1e-12)
                        a_try = a_try / a_try.sum()
                        c_try = full_convolve_fft(a_try)
                        s = float(2.0 * n * c_try.max())
                        if s < best_score - 1e-14:
                            best_score = s
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break

        if not improved:
            # Try larger random perturbations
            n_pert = min(20, n // 10)
            any_improve = False
            for _ in range(n_pert):
                if time.monotonic() > deadline - 2:
                    break
                mask = np.random.choice(n, size=min(n, 20), replace=False)
                a_try = a.copy()
                for idx in mask:
                    a_try[idx] *= 1.0 + 0.1 * np.random.randn()

                    best_score = s
                    best_a = a_try.copy()
                    a = a_try.copy()
                    any_improve = True
                    break
            if not any_improve:
                break

    return best_a, best_score


def interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-10 else a.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, a)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_random_smooth(n, rng, n_freqs=10):
    """Generate a random smooth non-negative sequence."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_freqs + 1):
        amp = rng.uniform(0, 0.25) / max(1, k)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_tapered(n, rng, taper_power=1.0):
    """Tapered profile with noise."""
    t = np.linspace(0.0, 1.0, n)
    taper = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = taper ** taper_power + 0.15
    a = a + 0.03 * rng.normal(size=n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_bell(n, rng, width=0.4, center=0.5):
    """Gaussian bell profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2) + 0.1
    a = a + 0.02 * rng.normal(size=n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()
