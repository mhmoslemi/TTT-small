def _smooth_max_objective_and_grad(a_raw, beta):
    """Compute objective and gradient for L-BFGS-B.
    a_raw is the raw parameter vector (before projection)."""
    a = np.maximum(a_raw, 0.0)
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    c = _fft_convolve(a)
    m = float(c.max())
    x = (c - m) / beta
    ex = np.exp(np.clip(x, -50, 50))
    sm = float(ex.sum())
    
    if sm < 1e-300:
        sm_val = m
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        sm_val = m + beta * np.log(sm)
        w = ex / sm
    
    val = 2.0 * n * sm_val / (s * s)
    
    # Gradient of smooth_max(c) w.r.t. a
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad_c = 2.0 * corr[n - 1:2 * n - 1]
    
    # d(val)/da = 2*n/s^2 * grad_c - 2*val/s * ones
    grad = (2.0 * n / (s * s)) * grad_c - (2.0 * val / s)
    
    # Zero out gradient where a is 0 (for box constraint handling)
    grad[a < 1e-15] = 0.0
    
    return val, grad


def _lbfgs_optimize(a0, deadline, beta_start=0.1, beta_end=0.001, max_iter=200):
    """Run L-BFGS-B with decreasing beta."""
    a = np.asarray(a0, dtype=np.float64).copy()
    best_a = a.copy()
    best_score = _score(a)
    
    n = len(a)
    n_beta_steps = 8
    
    for bi in range(n_beta_steps):
        if time.monotonic() > deadline:
            break
        
        t_frac = bi / max(1, n_beta_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        
        def fun(x):
            v, g = _smooth_max_objective_and_grad(x, beta)
            return v, g
        
        x0 = a.copy()
        res = minimize(fun, x0, method='L-BFGS-B', 
                      jac=True, bounds=[(0, None)] * n,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        
        if res.success or res.fun < best_score:
            a_candidate = np.maximum(res.x, 0.0)
            s = a_candidate.sum()
            if s > 1e-10:
                sc = _score(a_candidate)
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
                    a = a_candidate.copy()
    
    return best_a, best_score


def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f, g) <= M, g >= 0."""
    # Build constraint matrix A where A[k, j] such that (A @ g)[k] = (conv(f, g))[k]
    # conv(f, g)[k] = sum_j f[j] * g[k-j] (with appropriate bounds)
    # For k = 0..2n-2:
    #   conv(f,g)[k] = sum_{j=max(0,k-n+1)}^{min(k, n-1)} f[j] * g[k-j]
    #                  = sum_{j} f[j] * g[k-j]
    # Let i = k-j, so j = k-i:
    #   = sum_i f[k-i] * g[i] for valid indices
    # So A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = f[k-j] if 0 <= k-j < n
        k_start = max(0, j)
        k_end = min(2 * n - 1, j + n - 1)
        # A[k, j] = f[k-j] for k in [j, j+n-1]
        A[j:j + n, j] = f
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.003, inflate_min=1.0002):
    """LP-based refinement with adaptive inflation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    consecutive_breaks = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.01, inflate * 1.4)
            if no_imp > 200:
                consecutive_breaks += 1
                if consecutive_breaks > 5:
                    break
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
            continue

        g = g / sum_g

        # Line search
        a_old = a.copy()
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a_old + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a = (1.0 - best_t) * a_old + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_score = best_sc
            best_a = a.copy()
            no_imp = 0
            consecutive_breaks = 0
            # Slowly decrease inflate
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.01, inflate * 1.2)
            if no_imp > max(100, n):
                consecutive_breaks += 1
                if consecutive_breaks > 8:
                    break
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0

    return best_a


def _make_tukey(n, r=0.5):
    """Tukey (tapered cos) window."""
    a = np.ones(n)
    m = int(r * n // 2)
    if m > 0:
        w = 0.5 * (1 + np.cos(np.pi * (np.arange(m) + 0.5) / m))
        a[:m] = w
        a[-m:] = w[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.3):
    t = (np.arange(n) - n / 2) / n
    a = np.exp(-t**2 / (2 * sigma**2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-10
    return a / a.sum()


def _make_bell(n):
    """Design a bell-shaped window optimized for flat spectrum."""
    t = (np.arange(n) - (n-1)/2) / ((n-1)/2)
    a = np.cos(np.pi * t / 2) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
