"""
Optimized step function coefficient search using fast coordinate descent (SOCA-style) with
incremental convolution updates, combined with Adam smooth-max, multi-peak subgradient,
and LP refinement. The key innovation is the O(n) incremental convolution update enabling
thousands of coordinate descent sweeps for significant local improvement.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = np.clip((c - mval) / beta, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_smooth_max(a0, deadline, max_iter=50000, lr=0.03, beta_factor=0.05):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.001, lr_cur * 0.93)
            if no_imp > max(4000, 6 * n):
                rng_r = np.random.default_rng(1000 + it)
                z = best_z + 0.3 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.78)

    return best_a, best_score


def _soca_fast(a0, deadline, max_sweeps=50000, verbose=False):
    """
    Fast SOCA-style coordinate descent with incremental convolution updates.
    For each coordinate, tries several perturbations and keeps the one that
    minimizes the max of the convolution. Uses O(n) incremental updates.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_max_c = float(c.max())
    best_a = a.copy()
    best_score = 2.0 * n * best_max_c

    no_improve_sweeps = 0
    sweep_count = 0

    # Adaptive trial multipliers
    rel_multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0]
    # Small step size for fine-tuning
    fine_step = 2.0 / n

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for i in range(n):
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small positive values
                candidates = [fine_step * 0.5, fine_step, fine_step * 2.0]
            else:
                # Generate candidates
                candidates = [old_val * f for f in rel_multipliers]
                # Add small absolute perturbations
                candidates.append(old_val + fine_step)
                candidates.append(old_val - fine_step)
                candidates.append(old_val + 0.5 * fine_step)
                candidates.append(old_val - 0.5 * fine_step)

            best_local_max = best_max_c
            best_new_val = old_val
            best_delta = 0.0

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            for new_val in candidates:
                if new_val < 0:
                    new_val = 0.0
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue

                # Incremental update: c[k] += 2 * delta * a[k-i] for k in [k_lo, k_hi]
                # (first-order approximation)
                idx = np.arange(k_lo, k_hi + 1)
                a_offset = a[idx - i]
                old_c_seg = c[idx].copy()
                c[idx] += 2.0 * delta * a_offset

                cur_max = float(c.max())
                if cur_max < best_local_max - 1e-14:
                    best_local_max = cur_max
                    best_new_val = new_val
                    best_delta = delta
                    # Keep the updated c for this best candidate
                    best_c_seg = c[idx].copy()
                else:
                    c[idx] = old_c_seg

            if best_delta != 0.0:
                # Apply the best change
                a[i] = best_new_val
                idx = np.arange(k_lo, k_hi + 1)
                a_offset = a[idx - i]
                # Note: a[i] has changed, so for the term where k-i == i (k=2i),
                # we should use the new value. But since delta is small, first-order is fine.
                c[idx] += 2.0 * best_delta * a_offset
                # Fix the k=2i term more precisely
                if 2 * i <= 2 * n - 2:
                    # The k=2i term should use new a[i] for both factors
                    # Our formula used old a[k-i] = old a[i] for the second factor
                    # The correction is delta * a_new[i] (the second factor uses new value)
                    c[2 * i] += best_delta * a[i]
                best_max_c = best_local_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max_c
                sweep_improved = True

        sweep_count += 1
        if verbose and sweep % 100 == 0:
            print(f"    SOCA sweep={sweep}: best_score={best_score:.10f}")

        if sweep_improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break

    return best_a, best_score


def _soca_multipeak(a0, deadline, top_k=3, max_sweeps=20000):
    """
    Coordinate descent that minimizes the sum of top-k convolution peaks
    instead of just the global max. This avoids the "peak shifting" problem.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())
    best_a = a.copy()
    best_max_c = float(c.max())

    no_improve = 0
    fine_step = 2.0 / n
    rel_multipliers = [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for i in range(n):
            old_val = a[i]
            if old_val < 1e-15:
                candidates = [fine_step, fine_step * 2.0]
            else:
                candidates = [old_val * f for f in rel_multipliers]
                candidates.append(old_val + fine_step)
                candidates.append(old_val - fine_step)

            # Current top-k sum
            sorted_c = np.sort(c)[::-1]
            current_topk_sum = float(sorted_c[:top_k].sum())
            best_local_topk = current_topk_sum
            best_new_val = old_val
            best_delta = 0.0

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            for new_val in candidates:
                if new_val < 0:
                    new_val = 0.0
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue

                idx = np.arange(k_lo, k_hi + 1)
                a_offset = a[idx - i]
                old_c_seg = c[idx].copy()
                c[idx] += 2.0 * delta * a_offset

                sorted_c_new = np.sort(c)[::-1]
                new_topk = float(sorted_c_new[:top_k].sum())

                if new_topk < best_local_topk - 1e-14:
                    best_local_topk = new_topk
                    best_new_val = new_val
                    best_delta = delta
                else:
                    c[idx] = old_c_seg

            if best_delta != 0.0:
                a[i] = best_new_val
                idx = np.arange(k_lo, k_hi + 1)
                a_offset = a[idx - i]
                c[idx] += 2.0 * best_delta * a_offset
                if 2 * i <= 2 * n - 2:
                    c[2 * i] += best_delta * a[i]

                cur_max = float(c.max())
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    best_max_c = cur_max
                sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_a, best_score


def _multi_subgradient(a0, deadline, max_iter=10000, top_k=5):
    """Multi-peak subgradient method."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Weight peaks by their relative magnitude
        weights = np.array([1.0 + 0.5 * i for i in range(k)])
        weights = weights / weights.sum()

        # Subgradient: for each peak at position p, the subgradient w.r.t. a[j]
        # is a[j-p] + a[p-j] (for valid indices)
        sg = np.zeros(n)
        for i, p in enumerate(top_indices):
            p = int(p)
            for j in range(n):
                if 0 <= p - j < n:
                    sg[j] += weights[i] * a[p - j]
                if 0 <= j - p < n:
                    sg[j] += weights[i] * a[j - p]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(float(it + 1))
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _lp_refine(a0, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Build LP: maximize sum(g) subject to (a*g)[k] <= M_lp for all k, g >= 0
        c_obj = -np.ones(n)
        # Constraint matrix: row k has a[j] at column j for valid (k-j)
        # a*g conv: (a*g)[k] = sum_j a[j] * g[k-j]
        # For each k in [0, 2n-2], the constraint is sum over valid j: a[j] * g[k-j] <= M_lp
        # This is complex; use a simplified version with only the binding constraints
        # Find top binding constraints
        conv_ag = _fft_convolve(a)  # proxy for where constraints bind
        # Actually, we need: for variable g, the constraint is a * g (conv) <= M_lp
        # Let's just use the full constraint matrix (2n-1 constraints, n variables)
        if n <= 512:
            A = np.zeros((2 * n - 1, n))
            for j in range(n):
                A[j:j + n, j] = a
            try:
                res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_lp),
                              bounds=[(0, None)] * n, method='highs')
                if res.success:
                    g = res.x
                    sum_g = float(g.sum())
                    if sum_g > 1.0 + 1e-8:
                        g = g / sum_g
                        a_new = (1.0 - blend_cur) * a + blend_cur * g
                        a_new = np.maximum(a_new, 0.0)
                        s_new = float(a_new.sum())
                        if s_new > 1e-10:
                            a_new = a_new / s_new
                            c_new = _fft_convolve(a_new)
                            score_new = float(2.0 * n * c_new.max())
                            if score_new < best_score - 1e-12:
                                best_score = score_new
                                best_a = a_new.copy()
                                no_imp = 0
                                a = a_new
                                continue
                    no_imp += 1
                else:
                    no_imp += 1
            except Exception:
                no_imp += 1
        else:
            break

        if no_imp % 30 == 0:
            inflate_cur = min(1.01, inflate_cur * 1.05)
            blend_cur = max(0.05, blend_cur * 0.9)
        if no_imp > 100:
            break

    return best_a


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    except Exception:
        pass

    if parent is None:
        parent = _make_linear_profile(512, 1.5)
        print(f"No parent available, using linear profile n=512")

    n_parent = parent.size

    # Also prepare a 512 version if parent is different length
    if n_parent != 512:
        parent_512 = _interpolate_profile(parent, 512)
    else:
        parent_512 = parent.copy()

    # === Phase 1: Quick Adam from parent (100s) ===
    print("=== Phase 1: Quick Adam ===")
    d1 = min(deadline - 1500, t_start + 100)
    a1, s1 = _adam_smooth_max(parent, d1, max_iter=8000, lr=0.03, beta_factor=0.05)
    if s1 < best_score:
        best_score = s1
        best_seq = a1.copy()
    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Fast SOCA coordinate descent (700s) ===
    print("=== Phase 2: Fast SOCA ===")
    d2 = min(deadline - 900, t_start + 800)
    a2, s2 = _soca_fast(best_seq, d2, max_sweeps=100000)
    if s2 < best_score:
        best_score = s2
        best_seq = a2.copy()
    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Multi-peak subgradient (200s) ===
    print("=== Phase 3: Multi-peak subgradient ===")
    d3 = min(deadline - 600, t_start + 1000)
    a3, s3 = _multi_subgradient(best_seq, d3, max_iter=5000, top_k=5)
    if s3 < best_score:
        best_score = s3
        best_seq = a3.copy()
    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Second SOCA pass (400s) ===
    print("=== Phase 4: Second SOCA ===")
    d4 = min(deadline - 300, t_start + 1400)
    a4, s4 = _soca_fast(best_seq, d4, max_sweeps=50000)
    if s4 < best_score:
        best_score = s4
        best_seq = a4.copy()
    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Multi-peak SOCA (200s) ===
    print("=== Phase 5: Multi-peak SOCA ===")
    d5 = min(deadline - 150, t_start + 1600)
    a5, s5 = _soca_multipeak(best_seq, d5, top_k=3, max_sweeps=20000)
    if s5 < best_score:
        best_score = s5
        best_seq = a5.copy()
    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Fine Adam (100s) ===
    print("=== Phase 6: Fine Adam ===")
    d6 = min(deadline - 50, t_start + 1700)
    a6, s6 = _adam_smooth_max(best_seq, d6, max_iter=20000, lr=0.005, beta_factor=0.01)
    if s6 < best_score:
        best_score = s6
        best_seq = a6.copy()
    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final SOCA polish (remaining time) ===
    print("=== Phase 7: Final SOCA polish ===")
    d7 = deadline - 10
    a7, s7 = _soca_fast(best_seq, d7, max_sweeps=50000)
    if s7 < best_score:
        best_score = s7
        best_seq = a7.copy()
    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Try 512 version ===
    if best_seq.size != 512:
        b512 = _interpolate_profile(best_seq, 512)
        if time.monotonic() < deadline - 30:
            print("=== Phase 8: 512 version ===")
            d8a = min(deadline - 20, time.monotonic() + 60)
            a8, s8 = _soca_fast(b512, d8a, max_sweeps=20000)
            if s8 < best_score:
                best_score = s8
                best_seq = a8.copy()
            print(f"  Phase 8 done: best={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)

    final_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s")

    return [float(x) for x in final_seq]
