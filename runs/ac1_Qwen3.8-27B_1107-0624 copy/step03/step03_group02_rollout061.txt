"""
Optimization of step function heights to minimize 2n*max(a*a)/(sum(a))^2.
Key insight: linear ramp [1,2,...,n] gives score ~4/3 (vs 2 for uniform),
providing a much better starting point. Combined with Adam optimization
in logit-space, LP refinement, and coordinate descent polish.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_optimize(a0, deadline, max_iter=50000, lr=0.03, beta_factor=0.05):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.001, lr_cur * 0.92)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(10000 + it)
                z = best_z + 0.5 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 100 == 0:
            beta = max(1e-12, beta * 0.75)

    return best_a, best_score


def _subgradient_opt(a0, deadline, max_iter=10000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    base_lr = 0.4 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_idx = int(np.argmax(c_curr))

        sg = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break
        a = a_new

    return best_a, best_score


def _multi_subgradient(a0, deadline, max_iter=8000, top_k=3):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    base_lr = 0.25 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        weights = np.array([1.0 - 0.2 * i for i in range(k)])
        weights = np.maximum(weights, 0.1)
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i in range(k):
            idx = int(top_indices[i])
            j1s = max(0, idx - n + 1)
            j1e = min(idx, n - 1)
            if j1s <= j1e:
                ii = np.arange(j1s, j1e + 1)
                sg[ii] += weights[i] * a[idx - ii]
            j2s = max(idx, 0)
            j2e = min(idx + n - 1, n - 1)
            if j2s <= j2e:
                ii = np.arange(j2s, j2e + 1)
                sg[ii] += weights[i] * a[ii - idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.08)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.9)
            if no_imp > 150:
                break

        a = a_new

    return best_a


def _coord_descent(a0, deadline, max_iter=5000):

    a = a / a.sum()
    best_a = a.copy()

    rng = np.random.default_rng(12345)

    for it in range(max_iter):

        i = int(rng.integers(0, n))

                a_trial = a.copy()
                a_trial[i] = max(0.0, a[i] + direction * delta_scale * a[i] + direction * delta_scale * 0.01)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = _score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break
        if it % 500 == 0:
            idx = int(np.argmax(_fft_convolve(a)))
            j1s = max(0, idx - n + 1)
            j1e = min(idx, n - 1)
            if j1s <= j1e:
                ii = rng.integers(j1s, j1e + 1)
                a_trial = a.copy()
                a_trial[ii] *= 0.5
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial = a_trial / s
                    sc = _score(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a


def _make_linear_ramp(n):

def _make_perturbed_ramp(n, rng, scale=0.1):

def _interpolate_to_length(a0, n_new):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n0 = a0.size
    if n0 == n_new:
        s = a0.sum()
        if s > 1e-30:
            return a0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a0)
    p = np.maximum(p, 1e-15)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    if sc_linear < best_score:
        best_score = sc_linear
        best_seq = linear_ramp.copy()

    sub_ramp = _make_sublinear_ramp(n, 0.5)
    sc_sub = _score(sub_ramp)
    print(f"Sublinear ramp init: score={sc_sub:.10f}")
    if sc_sub < best_score:
        best_score = sc_sub
        best_seq = sub_ramp.copy()

    pert_ramp = _make_perturbed_ramp(n, rng, 0.1)
    sc_pert = _score(pert_ramp)
    print(f"Perturbed ramp init: score={sc_pert:.10f}")
    if sc_pert < best_score:
        best_score = sc_pert
        best_seq = pert_ramp.copy()

    # If parent is good, interpolate to n
    if parent is not None:
        parent_interp = _interpolate_to_length(parent, n)
        sc_pi = _score(parent_interp)
        print(f"Parent interpolated: score={sc_pi:.10f}")
        if sc_pi < best_score:
            best_score = sc_pi
            best_seq = parent_interp.copy()

    # Phase 1: Adam optimization on best init (most time)
    print(f"\n=== Phase 1: Adam on best init (score={best_score:.10f}) ===")
    d1 = min(deadline - 300, t_start + 600)
    a1, s1 = _adam_optimize(best_seq, d1, max_iter=30000, lr=0.03, beta_factor=0.05)
    if s1 < best_score:
        best_score = s1
        best_seq = a1.copy()
    print(f"  Adam1: score={best_score:.10f}")

    # Phase 2: Multi-subgradient
    if time.monotonic() < deadline - 200:
        print("=== Phase 2: Multi-subgradient ===")
        d2 = min(deadline - 200, time.monotonic() + 150)
        a2, s2 = _multi_subgradient(best_seq, d2, max_iter=6000, top_k=3)
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()
        print(f"  MultiSG: score={best_score:.10f}")

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 150:
        print("=== Phase 3: LP refinement ===")
        d3 = min(deadline - 150, time.monotonic() + 150)
        a3 = _lp_refine(best_seq, d3, max_iter=2000, inflate=1.002, blend=0.3)
        s3 = _score(a3)
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()
        print(f"  LP: score={best_score:.10f}")

    # Phase 4: Fine Adam
    if time.monotonic() < deadline - 100:
        print("=== Phase 4: Fine Adam ===")
        d4 = min(deadline - 100, time.monotonic() + 200)
        a4, s4 = _adam_optimize(best_seq, d4, max_iter=20000, lr=0.01, beta_factor=0.02)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
        print(f"  Adam2: score={best_score:.10f}")

    # Phase 5: Try second-best initialization with Adam
    if time.monotonic() < deadline - 80:
        print("=== Phase 5: Second init exploration ===")
        # Try a different perturbation of the linear ramp
        alt_ramp = _make_perturbed_ramp(n, rng, 0.2)
        sc_alt = _score(alt_ramp)
        if sc_alt < best_score + 0.1:  # only if competitive
            d5 = min(deadline - 60, time.monotonic() + 80)
            a5, s5 = _adam_optimize(alt_ramp, d5, max_iter=10000, lr=0.02, beta_factor=0.04)
            if s5 < best_score:
                best_score = s5
                best_seq = a5.copy()
                print(f"  Alt ramp Adam: score={best_score:.10f}")

    # Phase 6: Subgradient polish
    if time.monotonic() < deadline - 40:
        print("=== Phase 6: Subgradient polish ===")
        d6 = min(deadline - 40, time.monotonic() + 60)
        a6, s6 = _subgradient_opt(best_seq, d6, max_iter=5000)
        if s6 < best_score:
            best_score = s6
            best_seq = a6.copy()
        print(f"  Subgrad: score={best_score:.10f}")

    # Phase 7: LP polish
    if time.monotonic() < deadline - 25:
        print("=== Phase 7: LP polish ===")
        d7 = min(deadline - 25, time.monotonic() + 40)
        a7 = _lp_refine(best_seq, d7, max_iter=1000, inflate=1.001, blend=0.2)
        s7 = _score(a7)
        if s7 < best_score:
            best_score = s7
            best_seq = a7.copy()
        print(f"  LP polish: score={best_score:.10f}")

    # Phase 8: Coordinate descent
    if time.monotonic() < deadline - 10:
        print("=== Phase 8: Coordinate descent ===")
        d8 = min(deadline - 10, time.monotonic() + 30)
        a8 = _coord_descent(best_seq, d8, max_iter=3000)
        s8 = _score(a8)
        if s8 < best_score:
            best_score = s8
            best_seq = a8.copy()
        print(f"  CoordDesc: score={best_score:.10f}")

    # Phase 9: Ultra-fine Adam on very short time
    if time.monotonic() < deadline - 3:
        d9 = deadline - 3
        a9, s9 = _adam_optimize(best_seq, d9, max_iter=5000, lr=0.005, beta_factor=0.01)
        if s9 < best_score:
            best_score = s9
            best_seq = a9.copy()
        print(f"  Ultra Adam: score={best_score:.10f}")

    # Final: also try n=2048 if time permits (quick check)
    if best_seq is not None:
        n2 = 2048
        best_2048 = _interpolate_to_length(best_seq, n2)
        sc_2048 = _score(best_2048)
        print(f"\n  n=2048 interpolation: score={sc_2048:.10f}")
        if sc_2048 < best_score and time.monotonic() < deadline - 5:
            a2k, s2k = _adam_optimize(best_2048, deadline - 3, max_iter=5000, lr=0.01, beta_factor=0.02)
            if s2k < best_score:
                best_score = s2k
                best_seq = a2k.copy()
                print(f"  n=2048 Adam: score={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_ramp(256)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_ramp(256)
        s = float(final_seq.sum())

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
