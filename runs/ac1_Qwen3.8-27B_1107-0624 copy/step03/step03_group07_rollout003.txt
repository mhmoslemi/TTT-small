import numpy as np
import time

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_gradient_weights(c, beta):
    """Compute soft-max weights for the convolution peaks."""
    mval = c.max()
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            w[c.argmax()] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    x = np.clip(x, -70.0, 0.0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[c.argmax()] = 1.0
        s = 1.0
    return ex / s

def compute_gradient(a, w):
    """Compute gradient of soft-max(conv(a,a)) w.r.t. a."""
    # grad_i = 2 * sum_k w_k * a_{k-i} for valid k
    # This is 2 * (w * a_reversed) circular conv, restricted to valid range
    n = len(a)
    # h[i] = sum_k w[k] * a[k-i] for k-i in [0, n-1]
    # = sum_j w[i+j] * a[j] for j in [0, n-1] and i+j in [0, 2n-2]
    # This is the cross-correlation of w and a
    # h = conv(w, a)[n-1 : 2n-1] (taking appropriate slice)
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # We need h[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i = 0, ..., n-1
    # This is the correlation: h = (a * w_reversed) [n-1 : 2n-1]
    # Or equivalently, h[i] = sum_j a[j] w[i+j]
    # Let's use: h = np.convolve(a, w)[n-1 : 2n-1]
    # Wait, np.convolve(a, w)[k] = sum_j a[j] w[k-j]
    # We want h[i] = sum_j a[j] w[i+j]
    # Let w_rev = w[::-1], then h[i] = sum_j a[j] w[i+j] = sum_j a[j] w_rev[len(w)-1-i-j]
    # = conv(a, w_rev)[len(w)-1-i]
    # This is getting complicated. Let me just use the same approach as the original.
    
    # Original approach: compute_h(w, a) where h[i] = sum_m a[m] * w[i+m]
    # Use FFT: h = irfft(fft(a[::-1]) * fft(w))[n-1 : 2n-1]
    af = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(af, N) * np.fft.rfft(w, N), N)
    return 2.0 * conv[n-1 : 2*n-1]

def proj_simplex(v, s=1.0):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for tau
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def projected_gradient_optimize(a0, deadline, rng, lr=0.01, beta_init=0.05,
                                  beta_decay=0.95, use_nesterov=True, 
                                  max_iters_per_restart=5000, restart_noise=0.1):
    """
    Projected gradient descent with Nesterov acceleration on the simplex.
    Uses soft-max approximation of max(conv(a,a)).
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    c = full_convolve_fft(a)
    max_c = c.max()
    best_a = a.copy()
    best_score = 2.0 * n * max_c
    
    beta = beta_init * max_c
    current_lr = lr
    no_improve = 0
    
    # Nesterov momentum
    a_prev = a.copy()
    t_nest = 1.0
    
    for it in range(max_iters_per_restart):
        if time.monotonic() > deadline:
            break
        
        # Compute weights and gradient
        w = compute_gradient_weights(c, beta)
        grad = compute_gradient(a, w)
        
        # Nesterov step
        if use_nesterov:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest**2)) / 2.0
            y = a + (t_nest - 1.0) / t_new * (a - a_prev)
            y = np.maximum(y, 0.0)  # Stay in non-negative orthant
            a_new = y - current_lr * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()  # Renormalize
            a_prev = a
            a = a_new
            t_nest = t_new
        else:
            a_new = a - current_lr * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            a = a_new
        
        # Evaluate
        c = full_convolve_fft(a)
        max_c = c.max()
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt beta
        if it % 50 == 49:
            beta = max(1e-10, beta * beta_decay)
        
        # Adapt learning rate
        if no_improve > 200:
            current_lr *= 0.98
        
        # Restart
        if no_improve > max_iters_per_restart:
            t_nest = 1.0
            a = best_a + restart_noise * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            a_prev = a.copy()
            c = full_convolve_fft(a)
            max_c = c.max()
            beta = beta_init * max_c
            current_lr = lr
            no_improve = 0
    
    return best_a, best_score


def mirror_descent_optimize(a0, deadline, rng, lr=0.01, beta_init=0.05,

    best_z = z.copy()

def deflation_refine(a0, deadline, rng, n_deflations=50):
    """
    Deflation: iteratively identify the peak and reduce it.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    c = full_convolve_fft(a)
    best_score = 2.0 * n * c.max()
    
    for d in range(n_deflations):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve_fft(a)
        max_c = c.max()
        peak_idx = c.argmax()
        
        # The peak at index peak_idx: c[peak_idx] = sum_{i+j=peak_idx} a_i * a_j
        # To reduce this, we need to reduce a_i for i that contribute to this peak
        # The contributing indices are i in [max(0, peak_idx-n+1), min(n-1, peak_idx)]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        
        # Reduce the contributing a_i values proportionally
        # a_i contributes a_i * a_{peak_idx - i} to c[peak_idx]
        # We want to reduce the total contribution
        # Simple approach: reduce all contributing a_i by a small factor
        factor = 0.02 + 0.03 * (d / n_deflations)  # Increasing deflation
        a[lo:hi+1] *= (1.0 - factor)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = full_convolve_fft(a)
        score = 2.0 * n * c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
