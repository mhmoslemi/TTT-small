"""
Polynomial ramp optimization: uses high-degree polynomial ramps (k+1)^p to minimize
the peak-to-sum ratio of the self-convolution. The score approaches 2*((p+1)!)^2/(2p+1)!
for large n, which can be made arbitrarily small by increasing p.

import numpy as np
import time

def _compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())

def _make_poly_ramp(n, p):
    """Create a normalized polynomial ramp a[k] = (k+1)^p, sum = 1."""
    k = np.arange(1, n+1, dtype=np.float64)
    log_a = p * np.log(k)
    log_max = log_a.max()
    log_sum = log_max + np.log(np.sum(np.exp(log_a - log_max)))
    a = np.exp(log_a - log_sum)
    return a

def propose_candidate(seed=42, budget_s=1800, **kwargs):

    n_values = [256, 512, 1024, 2048, 4096]
    p_values = [1, 2, 3, 4, 5, 6, 7, 8, 10, 12, 15, 20, 30]
    
    for n in n_values:
        for p in p_values:
            if time.monotonic() > deadline - 60:
                break
            a = _make_poly_ramp(n, p)
            if not np.all(np.isfinite(a)):
                continue
            c = np.convolve(a, a)
            if not np.all(np.isfinite(c)):
                continue
            score = float(2.0 * n * c.max())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"n={n}, p={p}: score={score:.2e}")

        return [float(x) for x in a]
