import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Score assuming sum(a) = 1"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_conv_matrix(a):
    """Create the constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_solve(a, M):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success and res.fun is not None:
        return -res.fun, res.x
    return 0.0, None

def _lp_bisection(a, M_current, max_iter=20, tol=1e-8):
    """Find minimum M* achievable, and corresponding g*.
    Returns (M_star, g_star, score_improvement)
    """
    n = len(a)
    M_low = 0.0
    M_high = M_current
    best_g = None
    best_M = M_current
    
    # First check: can we achieve M_current with sum >= 1?
    # (We can, since g = a/sum(a) is feasible)
    
    for it in range(max_iter):

        if M_mid <= 0:
            break
        val, g = _lp_solve(a, M_mid)
        if val >= 1.0 - 1e-9:
            # Feasible: M_mid works
            M_high = M_mid
            best_g = g
            best_M = M_mid
        else:
            M_low = M_mid
        if M_high - M_low < tol * M_current:

    # One final solve at M_high to get the best g
    val, g = _lp_solve(a, M_high)
    if g is not None and val > 0:

        best_M = M_high
    
    return best_M, best_g

def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

    h = h_full[n - 1:2 * n - 1]
    # Gradient of max(c) w.r.t. a

    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z -= float(np.mean(z))
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.05 * c_max)
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):

            break
        gz, c = _gradient_step(a, beta, base_lr)
        if not np.all(np.isfinite(gz)):
            break
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z_new)):
            continue
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()
        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:

            if no_imp % 2000 == 0:

        if it % 200 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.5:
                    beta = max(1e-10, beta * 0.7)

        if no_imp > max(5000, 8 * n):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _make_sine_profile(n):
    """Sinusoidal profile: a_i = sin^2(pi*(i+0.5)/n)"""
    t = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_parabolic_profile(n):
    """Parabolic profile"""
    t = np.linspace(0.0, 1.0, n)
    a = t * (1 - t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

"""
Optimized LP-bisection + line-search hybrid for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. LP with line search on blend parameter (optimal step in LP direction)
2. Adaptive inflation that starts tight and loosens when stuck
3. Bisection LP for exact minimum achievable peak (used periodically)
4. Multi-phase time allocation maximizing LP iterations
5. Gradient polish with adaptive smooth-max temperature
6. Peak shaving targeting top convolution peaks
7. Alternate-n exploration if time permits
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_solve(a, M, time_limit=5.0):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.fun is not None:
        return -res.fun, res.x
    return 0.0, None


def _smooth_max_val(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _logit_adam_optimize(a, deadline, max_iter=50000, lr=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z -= float(np.mean(z))
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.03 * c_max)
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max_val(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)

            best_z = z_new.copy()

        if it % 200 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]

                if peak_gap < beta * 0.5:
                    beta = max(1e-10, beta * 0.7)
                elif peak_gap > beta * 5.0:
                    beta = min(0.3 * c_max_new, beta * 1.3)

        if no_imp > max(6000, 10 * n):

            rng_re = np.random.default_rng(it * 7919 + 1)

            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            no_imp = 0
        z = z_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_sine_profile(n):
    t = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_parabolic_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = t * (1 - t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _lp_with_linesearch(a, M_lp, deadline):
    """Solve LP and do 1D line search on blend parameter."""
    val, g = _lp_solve(a, M_lp)
    if g is None or val < 1.0 + 1e-6:
        return a, 0.0
    g = g / g.sum()

    best_t = 0.3
    best_sc = _score(a)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        if time.monotonic() > deadline:
            break
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t

    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new, val

def _lp_bisection_step(a, deadline):

    best_g = None
    best_M_ach = M_current

    for bis in range(20):
        if time.monotonic() > deadline:
            break
        M_mid = (M_low + M_high) / 2.0
        if M_mid <= 1e-12:

        if val >= 1.0 - 1e-8:
            M_high = M_mid
            best_g = g
            best_M_ach = M_mid
        else:
            M_low = M_mid

    if best_g is None:
        return a

    sum_g = float(best_g.sum())

    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 1e-15)

def _peak_shave_fftopt(a, deadline, max_iter=5000, n_peaks=5, strength=0.04):
    """FFT-accelerated peak shaving."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(3 * n - 2)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        med = np.median(c)
        weights = np.maximum(c[top_idx] - med, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / weights.sum()

        # Compute pressure via FFT
        # pressure[i] = sum_k w[k] * a[k - i] for k in top_idx (valid range)
        # This is equivalent to: for each peak k, a shifted version of a weighted by w
        # Use: pressure = sum_k w_k * (a shifted by k)
        # In FFT: pressure = IFFT(FFT(w_vec) * FFT(a)) but need to be careful with indexing
        # Simpler: direct for small n_peaks
        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # a[lo:hi+1] corresponds to a[k-i] for i in [lo:hi+1]
                idx = np.arange(lo, hi + 1)
                j = k - idx
                pressure[lo:hi + 1] += w * a[j]

        pmax = pressure.max()
        if pmax < 1e-15:
            no_imp += 1
            if no_imp > 500:
                break
            continue

        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.12, str_cur * 1.008)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                str_cur = max(0.01, str_cur * 0.92)
            if no_imp > 800:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a
