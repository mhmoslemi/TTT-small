    alpha: weight for smooth max (1-alpha for L2)
    Returns (objective, gradient).
    """
    p = _softmax(z)
    n = len(p)
    c = _convolve(p)
    maxc = float(c.max())
    
    # Smooth max value and weights
    if beta < 1e-12:
        sm = maxc
        w = np.zeros_like(c)
        mask = c >= maxc - 1e-10
        cnt = mask.sum()
        if cnt == 0:
            w[np.argmax(c)] = 1.0
        else:
            w[mask] = 1.0 / cnt
    else:
        x = (c - maxc) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s = float(ex.sum())
        if s < 1e-300 or not np.isfinite(s):
            sm = maxc
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            sm = maxc + beta * float(np.log(s))
            w = ex / s
    
    # L2 norm squared of c
    l2 = float(np.dot(c, c))
    
    # Combined objective
    obj = alpha * sm + (1.0 - alpha) * l2
    
    # Gradient in p-space
    # d(sm)/dp_i = 2 * (w * p)_crosscorr[i]
    # d(l2)/dp_i = 4 * (c * p)_crosscorr[i]
    
    N = _npow2(3 * n - 2)  # Need this for cross-correlation of len(2n-1) and len(n)
    fw = np.fft.rfft(w, N)
    fc = np.fft.rfft(c, N)
    fp = np.fft.rfft(p, N)
    
    # Cross-correlation (w ⋆ p)[i] = IFt(FFT(w)*conj(FFT(p)))[i]
    corr_w = np.fft.irfft(fw * np.conj(fp), N)[:n]
    corr_c = np.fft.irfft(fc * np.conj(fp), N)[:n]
    
    grad_p = 2.0 * alpha * corr_w + 4.0 * (1.0 - alpha) * corr_c
    
    # Push through softmax: grad_z = p * (grad_p - <p, grad_p>)
    dot = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - dot)
    
    # Clean up
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    
    return obj, grad_z


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.arange(n, dtype=np.float64)
    t_norm = t / (n - 1) if n > 1 else np.array([0.0])

            a = np.maximum(a, 1e-12)

    # Cubic
    for c1 in [0.5, 1.0, 1.5]:

    # Spiky

    a[0] = 1.0
    a[n//3] = 0.5
    a[2*n//3] = 0.3
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    return profiles

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Adam with combined smooth-max + L2-flatness
objective in logit space, LP refinement, and frequency-domain perturbation for diversity.
Key innovation: combining peak reduction (smooth max) with convolution flatness (L2 norm)
to push the convolution closer to uniform, approaching the theoretical bound 2n/(2n-1)≈1.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _npow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _convolve(a):
    n = len(a)
    N = _npow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(p):
    n = len(p)
    c = _convolve(p)
    return 2.0 * n * float(c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - z.mean()
    z = np.clip(z, -50.0, 50.0)
    z = z - z.max()
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr(w, p):
    """Compute (w cross-correlate p)[i] for i=0..len(p)-1."""
    nw = len(w)
    np_ = len(p)
    N = _npow2(nw + np_ - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    result = np.fft.irfft(fw * np.conj(fp), N)
    return result[:np_]


def _gradient_combined(p, c, beta, alpha):
    """Compute gradient of alpha*smooth_max + (1-alpha)*||c||^2 in z-space."""
    n = len(p)
    _, w = _smooth_max_weights(c, beta)
    corr_w = _cross_corr(w, p)
    corr_c = _cross_corr(c, p)
    grad_p = 2.0 * alpha * corr_w + 4.0 * (1.0 - alpha) * corr_c
    dot = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - dot)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n)
    return grad_z


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                   alpha_init=0.3, alpha_end=0.9, no_imp_limit=400,
                   noise_scale=None, max_iters=None):
    """Adam optimization in logit space with combined smooth-max + L2 objective."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    it = 0

    p = p0.copy()
    c = _convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

        alpha = alpha_init + (alpha_end - alpha_init) * frac

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)

            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)

def _lbfgs_optimize(p0, deadline, beta, alpha):

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)

    def func_and_grad(z):
        nonlocal best_score, best_z, best_p
        p = _softmax(z)
        c = _convolve(p)
        maxc = float(c.max())
        sc = 2.0 * n * maxc

            best_score = sc
            best_z = z.copy()
            best_p = p.copy()

        corr_w = _cross_corr(w, p)

        grad_p = 2.0 * alpha * corr_w + 4.0 * (1.0 - alpha) * corr_c
        dot = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - dot)
        return alpha * (maxc + beta * float(np.log(np.exp((c - maxc) / beta).sum()))) + (1 - alpha) * float(np.dot(c, c)), grad_z

    from scipy.optimize import minimize
    res = minimize(func_and_grad, z0, method='L-BFGS',

        sc_res = _score(p_res)
        if sc_res < best_score:
            best_score = sc_res
            best_p = p_res
            best_z = res.x.copy()

    return best_p, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((a[k - idx]).tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    n = len(a)
    current_best = a.copy()
    current_score = _score(a)
    c = _convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _convolve(a)
        M = float(c.max())
        t = max(0.1, t * 0.9)
    return current_best, current_score


def _freq_perturb(p, rng, scale=0.3):
    """Perturb in frequency domain to escape local minima."""
    n = len(p)
    N = _npow2(2 * n - 1)
    fp = np.fft.rfft(p, N)
    magnitude = np.abs(fp)
    phase = np.angle(fp)
    # Add random phase perturbation (preserving DC)
    phase_noise = scale * rng.normal(size=len(fp))
    phase_noise[0] = 0  # Keep DC unchanged
    new_phase = phase + phase_noise
    new_fp = magnitude * np.exp(1j * new_phase)
    # Make it conjugate symmetric for real signal
    new_fp_full = np.fft.irfft(new_fp, N)
    p_new = np.maximum(new_fp_full[:n], 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy()
    return p_new / s

def _interp_profile(p0, n_new):

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Combines:
    1. Diverse non-symmetric profile initialization
    2. Adam optimization with combined smooth-max + L2-flatness objective
    3. LP-based refinement
    4. Frequency-domain perturbation for escaping local minima
    5. Multi-scale escalation
    6. Final polish with decreasing temperature
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # Try to access parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Generate and score diverse candidates ---
    phase1_end = t_start + 60.0
    candidates = []

    target_ns = [512, 1024, 2048]
    if pn > 0 and pn <= 4096:
        target_ns.append(pn)

    for n in target_ns:
        profs = _make_profiles(n)
        for i, p in enumerate(profs):
            sc = _score(p)
            candidates.append((sc, p, n, f'prof{i}'))

        if parent is not None and pn != n:
            pi = _interp_profile(parent, n)
            if pi is not None:
                sc = _score(pi)
                candidates.append((sc, pi, n, 'parent_interp'))

    if parent is not None and 0 < pn <= 4096:
        sc = _score(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Adam optimization on top candidates ---
    phase2_end = t_start + 700.0
    top_k = 15
    for idx in range(min(top_k, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        _, p, n, label = candidates[idx]
        per = min(40.0, phase2_end - time.monotonic() - 5.0)
        if per < 10:
            break
        p_b, sc_b = _adam_optimize(p, time.monotonic() + per, rng,
                                   lr=0.05, beta_frac=0.1, beta_decay=0.93,
                                   alpha_init=0.3, alpha_end=0.8,
                                   no_imp_limit=300)
        _update(p_b, sc_b, f'adam_{label}_n{n}')

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: L-BFGS on best few ---
    phase3_end = t_start + 900.0
    if best_seq is not None:
        # Run L-BFGS on current best
        per = min(60.0, phase3_end - time.monotonic() - 5.0)
        if per > 10:
            p_lb, sc_lb = _lbfgs_optimize(best_seq, time.monotonic() + per,
                                          beta=0.05, alpha=0.5)
            _update(p_lb, sc_lb, 'lbfgs')

        # LP refinement
        per = min(60.0, phase3_end - time.monotonic() - 5.0)
        if per > 10 and best_seq.size <= 2048:
            p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + per, t=0.3, max_solves=15)
            _update(p_lp, sc_lp, 'LP')

        # Adam after LP
        per = min(50.0, phase3_end - time.monotonic() - 5.0)
        if per > 10 and best_seq is not None:
            p_a, sc_a = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                       lr=0.03, beta_frac=0.08, beta_decay=0.94,
                                       alpha_init=0.5, alpha_end=0.9,
                                       no_imp_limit=200)
            _update(p_a, sc_a, 'post_LP_adam')

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Frequency perturbation + re-optimization ---
    phase4_end = t_start + 1300.0
    n_perturbs = 6
    if best_seq is not None:
        n_best = best_seq.size
        per_each = max(15.0, (phase4_end - time.monotonic()) / max(1, n_perturbs))
        for i in range(n_perturbs):
            if time.monotonic() > phase4_end - 10:
                break
            per = min(per_each, phase4_end - time.monotonic() - 5.0)
            if per < 10:
                break
            scale = 0.2 + 0.15 * i
            p_pert = _freq_perturb(best_seq, rng, scale=scale)
            sc_pert = _score(p_pert)
            if sc_pert < best_score:
                _update(p_pert, sc_pert, f'freq_pert{i}')
            # Optimize the perturbed version
            p_opt, sc_opt = _adam_optimize(p_pert, time.monotonic() + per, rng,
                                           lr=0.04, beta_frac=0.08, beta_decay=0.94,
                                           alpha_init=0.4, alpha_end=0.9,
                                           no_imp_limit=int(0.3 * n_best))
            _update(p_opt, sc_opt, f'freq_opt{i}')

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-scale escalation ---
    phase5_end = t_start + 1500.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_cur = best_seq.size
        if n_cur <= 1024:
            target_n = 2048
            p_int = _interp_profile(best_seq, target_n)
            if p_int is not None:
                sc_int = _score(p_int)
                _update(p_int, sc_int, f'escalate_n{target_n}')
                per = min(60.0, phase5_end - time.monotonic() - 10.0)
                if per > 15:
                    p_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                               lr=0.04, beta_frac=0.07, beta_decay=0.94,
                                               alpha_init=0.3, alpha_end=0.8,
                                               no_imp_limit=int(0.3 * target_n))
                    _update(p_e, sc_e, f'escalate_opt_n{target_n}')
                    # LP at new scale
                    if p_e.size <= 2048 and time.monotonic() < phase5_end - 20:
                        p_lp2, sc_lp2 = _lp_refine(p_e, min(phase5_end - 5, time.monotonic() + 60),
                                                    t=0.3, max_solves=10)
                        _update(p_lp2, sc_lp2, f'LP_n{p_e.size}')

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        per = deadline - time.monotonic() - 5
        if per > 10:
            # Very careful final Adam with low learning rate
            p_f, sc_f = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                       lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                       alpha_init=0.7, alpha_end=1.0,
                                       no_imp_limit=150)
            _update(p_f, sc_f, 'final_polish')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            t = np.linspace(0, 1, 512)
            final = (1 + 1.5 * t)
            final = final / final.sum()

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profiles(512)[0]
    print("FALLBACK n=512")
    return [float(x) for x in fb]
