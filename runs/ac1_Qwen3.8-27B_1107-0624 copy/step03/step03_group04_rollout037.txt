Let me also think about the LP step more carefully. The previous LP is:
`max sum(b_j) s.t. conv(a, b)_k <= M * inflate for all k, b_j >= 0`

For the "direct attack" method:
- Find the current peak at index k*
- The peak value is c[k*] = sum_i a[i]*a[k*-i]
- To reduce c[k*], we can reduce a[i] for i where a[k*-i] is large
- The "pressure" on a[i] from peak k* is a[k*-i]
- So we want to reduce a[i] proportionally to a[k*-i]
- But we also need to maintain sum(a) = 1, so we need to increase some other a[j]
- The best j to increase is one where it doesn't contribute much to any peak

This gives a specific direction:
- delta_a[i] = -alpha * a[k*-i] (for the "source" indices)
- delta_a[j] = +alpha * something (for the "sink" indices)

"""
Optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. Nesterov accelerated gradient with active-set smooth max
2. Fast LP-based spreading
3. Direct peak attack
4. Multi-length search
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_conv_matrix(a):
    n = len(a)
    # For the LP: conv(a, b)_k = sum_i a[i] * b[k-i]
    # This is a matrix-vector product: A[k, j] * b[j]
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = 0
        hi = min(n, 2 * n - 1 - j)
        A[j:hi, j] = a[:hi-j]
    return A

def _lp_spread(a, M, inflate=1.001):
    n = len(a)
    c_a = _fft_convolve(a)
    # We want: conv(a, b)_k <= (M*inflate - c_a[k]) / 2 for all k
    # (from the linearization conv(a+tb, a+tb) ≈ conv(a,a) + 2t*conv(a,b))
    rhs = (M * inflate - c_a) / 2.0
    rhs = np.maximum(rhs, 0)  # If c_a[k] > M*inflate, we can't increase b at all in that direction
    
    A = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    # Also add constraint sum(b) <= 1 (to normalize)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    res = linprog(c_obj, A_ub=A, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _nesterov_smooth_max(a, deadline, max_iter=50000, lr=0.1, m_top=10):
    """
    Nesterov accelerated gradient on the smooth max of the convolution.
    Uses a top-M active set for efficiency.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Work in log-space: a_i = exp(z_i) / sum(exp(z_j))
    z = np.log(a / a.mean())
    
    x = z.copy()  # Nesterov iterate
    z_prev = z.copy()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    c_max = float(c.max())
    beta = max(1e-6, 0.02 * c_max)
    
    t_curr = 1.0
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute a from x
        e = np.exp(np.clip(x, -60, 60))
        p = e / e.sum()
        
        # Compute convolution and gradient
        c = _fft_convolve(p)
        c_max_cur = float(c.max())
        score = float(2.0 * n * c_max_cur)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = p.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Smooth max with top-M weighting
        # Use a weighted sum of top M convolution values
        top_idx = np.argsort(c)[::-1][:min(m_top, len(c))]
        top_vals = c[top_idx]
        # Weights: softmax over top values with temperature beta
        x_w = (top_vals - top_vals[0]) / beta
        x_w = np.clip(x_w, -50, 50)
        w = np.exp(x_w)
        w = w / w.sum()
        
        # Full weight vector (sparse)
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w
        
        # Gradient of weighted sum w^T c with respect to a
        # d(c_k)/da_i = a_{k-i} (with bounds)
        # grad_i = sum_k w_full[k] * a_{k-i}
        # This is a convolution of w_full with a (reversed)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w_full, N)
        fa_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        grad_a = 2.0 * h_full[n-1:2*n-1]  # Factor of 2 because c_k = sum a_i a_{k-i}, d/da_i gives two terms
        
        # Actually, let me be more careful. c_k = sum_i a_i * a_{k-i}
        # d(c_k)/da_i = a_{k-i} + (if i != k-i) a_{i-k}
        # For the full convolution, d(c_k)/da_i = a_{k-i} + a_{i-k} (with the understanding that if k-i == i-k, i.e., 2i = k, then it's 2*a_{k/2})
        # Wait, no. c_k = sum_j a_j * a_{k-j}. d(c_k)/da_i: 
        #   Term where j = i: a_{k-i}
        #   Term where k-j = i, i.e., j = k-i: a_i (but this is the same term if j = i, i.e., 2i = k)
        # So d(c_k)/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} if 2i != k
        #                  = 2*a_{k-i} if 2i == k (since a_{k-i} = a_i, and both terms are a_i)
        # Wait, that's not right either. Let me think again.
        # c_k = sum_j a_j * a_{k-j}
        # d(c_k)/da_i: the terms involving a_i are:
        #   j = i: a_i * a_{k-i}, derivative w.r.t. a_i is a_{k-i}
        #   k-j = i, i.e., j = k-i: a_{k-i} * a_i, derivative w.r.t. a_i is a_{k-i}
        #   If i = k-i (i.e., 2i = k), these are the same term, so derivative is 2*a_{k-i} = 2*a_i
        #   If i != k-i, derivative is a_{k-i} + a_{k-i} = 2*a_{k-i}

        # where (w_full * a_reversed)_i = sum_k w_full[k] * a_{k-i}
        # This is the convolution of w_full with a, evaluated at position i... 
        # Actually it's a cross-correlation. Let me just use the FFT approach.
        
        # The gradient w.r.t. a is: 2 * (w_full cross-correlated with a)
        # cross_corr(i) = sum_k w_full[k] * a[k-i] (with bounds)
        # This is equivalent to: convolve(w_full, a_reversed)[n-1 + i]... hmm
        # Actually, let's think of it as: for each k, the contribution to grad from w_full[k] is:
        #   For each valid i (0 <= i < n, 0 <= k-i < n): grad[i] += 2 * w_full[k] * a[k-i]
        # This is: grad = 2 * sum_k w_full[k] * (a shifted right by k, truncated to length n)
        # Which is the same as: grad = 2 * convolve(w_full, a)[n-1 : 2n-1] ... no
        # Let me just verify with the FFT:
        # If we compute conv(w_full, a_rev) where a_rev = a[::-1], then:
        # conv(w_full, a_rev)[j] = sum_k w_full[k] * a_rev[j-k] = sum_k w_full[k] * a[n-1-(j-k)]
        # For j = n-1+i: sum_k w_full[k] * a[n-1-(n-1+i-k)] = sum_k w_full[k] * a[k-i]
        # Yes! So grad = 2 * conv(w_full, a_rev)[n-1 : 2n-1]
        # But we already computed this as 2 * h_full[n-1:2n-1] above. OK good.
        
        # Convert gradient from a-space to z-space
        # a_i = exp(z_i) / S where S = sum(exp(z_j))
        # da_i/dz_j = a_i * (delta_ij - a_j)
        # So grad_z_j = sum_i grad_a_i * a_i * (delta_ij - a_j) = a_j * (grad_a_j - sum_i grad_a_i * a_i)
        dot = float(np.dot(p, grad_a))
        grad_z = p * (grad_a - dot)
        grad_z -= np.mean(grad_z)  # Center for invariance
        
        if not np.all(np.isfinite(grad_z)):
            x = z_prev.copy()
            t_curr = 1.0
            continue
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0
        theta = (t_curr - 1.0) / t_new
        
        x_new = x - lr * grad_z
        z_new = x_new + theta * (x_new - x)
        
        # Clip and normalize
        z_new = np.clip(z_new, -60, 60)
        z_new -= np.mean(z_new)
        
        # Line search (backtracking)
        step = 1.0
        for _ in range(20):
            e_test = np.exp(np.clip(z_new * step + (1-step) * z_prev, -60, 60))
            a_test = e_test / e_test.sum()
            c_test = _fft_convolve(a_test)
            if float(c_test.max()) < c_max_cur + 1e-10:  # descent condition
                break
            step *= 0.5
        
        if step < 0.01:
            # No progress, try smaller lr
            lr = max(0.001, lr * 0.9)
            z_new = z_prev.copy()
        
        z_prev = z_new
        x = x + (z_new - z_prev)  # Update x for next Nesterov step
        # Actually, the standard Nesterov update is:
        # y_k = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
        # x_{k+1} = y_k - (lr/t_{k+1}) * grad(y_k)
        # Let me redo this properly.
        
        t_curr = t_new
        
        # Adaptive beta
        if it % 200 == 0 and it > 100:
            sorted_c = np.sort(c)[::-1]
            gap = sorted_c[0] - sorted_c[min(m_top, len(sorted_c))-1]
            if gap < beta * 0.3:
                beta = max(1e-10, beta * 0.6)
            elif gap > beta * 3.0:
                beta = min(0.5 * c_max_cur, beta * 1.3)
        
        # Restart if stuck
        if no_imp > max(3000, 5 * n):
            # Restart from best with noise
            rng = np.random.default_rng(it * 31 + 7)
            noise = 0.2 * best_score * rng.standard_normal(n)
            z_best = np.log(best_a / best_a.mean())
            z_new = z_best + noise
            z_new = np.clip(z_new, -60, 60)
            z_new -= np.mean(z_new)
            x = z_new.copy()
            z_prev = z_new.copy()
            t_curr = 1.0
            lr = 0.1
            beta = max(1e-6, 0.02 * float(_fft_convolve(best_a).max()))
            no_imp = 0
    
    return best_a

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2*n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:2*n-1]
    return np.maximum(c, 0)

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s*s)

def _grad_smooth_max(a, beta, top_m=20):
    """Return (value, gradient) of smooth max of conv(a,a)."""
    n = len(a)
    c = _fft_conv(a)
    # Top-m smooth max
    idx = np.argsort(c)[::-1][:min(top_m, len(c))]
    vals = c[idx]
    x = (vals - vals[0]) / beta
    x = np.clip(x, -50, 50)
    w = np.exp(x)
    w = w / w.sum()
    val = vals[0] + beta * np.log(float(w.sum() * np.exp(0)))  # = vals[0] + beta * log(sum(exp((vals-vals[0])/beta)))
    # Actually: val = max + beta * log(sum(exp((c - max)/beta)))
    # Let me compute it properly:
    val = vals[0] + beta * np.log(float(np.exp((vals - vals[0])/beta).sum()))
    
    # Gradient
    w_full = np.zeros(2*n-1)
    w_full[idx] = w
    N = _next_pow2(2*n-1)
    fw = np.fft.rfft(w_full, N)
    fa_r = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_r, N)
    grad = 2.0 * h[n-1:2*n-1]
    return val, grad, c

def _lp_spread_fast(a, M, inflate=1.001):
    """LP to find spreading direction."""
    n = len(a)
    c_a = _fft_conv(a)
    # conv(a,b)_k <= (M*inflate - c_a[k]) / 2
    rhs = np.maximum((M*inflate - c_a)/2, 1e-10)
    
    # Build constraint matrix
    # conv(a,b)_k = sum_j a[k-j] * b[j] for valid j
    # Matrix A: A[k,j] = a[k-j] if 0<=k-j<n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        length = min(n, 2*n-1-j)
        A[start:start+length, j] = a[:length]
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=rhs, bounds=[(0,None)]*n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None
