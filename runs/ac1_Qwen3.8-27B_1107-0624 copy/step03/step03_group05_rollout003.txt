        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, k=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(k * np.pi * (2.0 * t - 1.0))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_polynomial_profile(n, degree=3, coeffs=None):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if coeffs is None:
        coeffs = [1.0, 0.0, 0.3, 0.0, -0.1]
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * u ** i
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.0) / width) ** 2) + np.exp(-((t - 1.0) / width) ** 2)
    a += 0.3 * np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _smooth_max_objective_and_grad(z, beta):
    """Compute smooth max of a*a and its gradient w.r.t. z (log-param)."""
    n = len(z)
    a = _softmax(z)
    c = _fft_convolve(a)

    # Smooth max via log-sum-exp
    mval = float(c.max())
    x = (c - mval) * beta
    # Clip for numerical stability
    x = np.clip(x, -500.0, 500.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        obj = 2.0 * n * mval
        grad = np.zeros(n)
        return obj, grad
    w = ex / s

    # Gradient: d/dz_i [2n * smooth_max(c)]
    # d smooth_max(c) / da_i = sum_k w[k] * 2 * a[k-i]  (with a out of range = 0)
    # This is 2 * (w_reversed * a) truncated appropriately
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # w * a_reversed gives us what we need
    a_rev = np.zeros(N)
    a_rev[:n] = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    h = np.fft.irfft(fw * fa_rev, N)
    h_full = h[:2 * n - 1]

    # d smooth_max(c) / da_i = 2 * sum_k w[k] * a[k-i] for 0<=k-i<n
    # = 2 * (w * a_rev_flipped)[i] ... let me think more carefully
    # c[k] = sum_j a[j] * a[k-j], so dc[k]/da[i] = 2*a[k-i] (a out of range = 0)
    # d smoothing / da[i] = sum_k w[k] * 2 * a[k-i]
    # Let's compute this as: for each i, sum_k w[k] * a[k-i]
    # This is a cross-correlation: (w cross a)[i] = sum_k w[k] a[k-i]
    # Which equals (w * a_flipped)[n-1+i - (n-1)] ... 
    # Actually: cross-correlation r[i] = sum_k w[k] * a[k-i + n-1] ... 
    # Let me just use the FFT approach:
    # h[j] = sum_k w[k] * fa[k-j]... no.
    # FFT convolution: if I flip w, then conv(flip(w), a)[j] = sum_k w[n-1-k] * a[j-n+1+k]
    # That's getting complicated. Let me use a direct approach for correctness.

    # Actually the cleanest way:
    # grad_a[i] = 2 * sum_k w[k] * a[(k-i) mod handling]
    # where a[m] = 0 if m < 0 or m >= n
    # This is: grad_a = 2 * np.convolve(w, a[::-1], mode='same')... no.
    # 
    # Let me use: grad_a[i] = 2 * sum_{k} w[k] * a[k-i]
    #          = 2 * sum_{m} w[i+m] * a[m]   (substituting m = k-i)
    #          = 2 * (w * a)[i] in the "valid correlation" sense
    #
    # In numpy: np.correlate(w, a, mode='full') gives sum_m w[m] * a[m-k] ...
    # Actually np.correlate(v, w) computes sum_j v[j+k]*w[j]
    # 
    # Let me just do it via FFT properly:
    # grad_a[i] = 2 * sum_k w[k] * a[k-i]  (a[m]=0 outside [0,n-1])
    # = 2 * sum_{k=i}^{i+n-1} w[k] * a[k-i]   (only valid range)
    # = 2 * (w ⊛ a)[i]  where ⊛ is cross-correlation
    
    # Using FFT: cross-correlation of w and a = ifft(fft(w) * conj(fft(a))) 
    # but we need to be careful about the offset.
    # Actually: (w cross a)[i] = sum_k w[k+i] * a[k] = (w * flip(a))[n-1-i + offset]
    
    # Simplest: just use the already computed FFT
    # h[j] from irfft(fw * fa_rev) where a_rev = flip(a)
    # h[j] = sum_k w[k] * a_rev[j-k] = sum_k w[k] * a[n-1-(j-k)] = sum_k w[k] * a[n-1-j+k]
    # Let m = n-1-j+k, so k = m - n + 1 + j
    # h[j] = sum_m w[m-n+1+j] * a[m]
    # We want grad_a[i] = 2 * sum_k w[k] * a[k-i]
    # Let's set k-i = m, so k = m+i
    # grad_a[i] = 2 * sum_m w[m+i] * a[m]
    # From h[j]: h[j] = sum_m w[m + (j-n+1)] * a[m]
    # So if j-n+1 = i, i.e., j = i+n-1:
    # h[i+n-1] = sum_m w[m+i] * a[m] = grad_a[i]/2
    
    # So grad_a[i] = 2 * h[i+n-1] for i = 0,...,n-1
    # h has length N, and i+n-1 ranges from n-1 to 2n-2, which is within [0, N-1]
    
    grad_a = 2.0 * h[n - 1:2 * n - 1]

    # Convert to gradient w.r.t. z: gz = a * (grad_a - dot(a, grad_a))
    dot = float(np.dot(a, grad_a))
    gz = a * (grad_a - dot)
    gz -= gz.mean()  # softmax is invariant to constant shifts in z

    obj = 2.0 * n * (mval + np.log(s) / beta)
    return obj, gz


def _lbfgs_optimize(a_init, n, deadline, max_iter=5000):
    """Run L-BFGS optimization with adaptive temperature."""
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()

    z = np.log(a_init)
    z -= z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)
    best_z = z.copy()

    beta = 50.0
    no_imp_count = 0
    iteration = 0

    while time.monotonic() < deadline and iteration < max_iter:
        # Define objective for L-BFGS
        def obj_func(z_vec):
            obj, _ = _smooth_max_objective_and_grad(z_vec, beta)
            if not np.isfinite(obj):
                return 1e10
            return obj

        def grad_func(z_vec):
            _, grad = _smooth_max_objective_and_grad(z_vec, beta)
            if not np.all(np.isfinite(grad)):
                return np.zeros_like(z_vec)
            return grad

        # Run L-BFGS for a batch of iterations
        batch_iter = min(200, maxIter - iteration if 'maxIter' in dir() else max_iter - iteration)
        try:
            res = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': batch_iter, 'gtol': 1e-8, 'ftol': 1e-12, 'maxls': 20}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_new = res.x
                z_new = np.clip(z_new, -50.0, 50.0)
                z_new -= z_new.mean()
                a_new = _softmax(z_new)
                sc = _score(a_new)
                iteration += int(res.nit) if res.nit is not None else batch_iter

                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    best_z = z_new.copy()
                    no_imp_count = 0
                else:
                    no_imp_count += 1
                z = z_new
            else:
                no_imp_count += 1
                iteration += batch_iter
        except Exception:
            no_imp_count += 1
            iteration += batch_iter

        # Adaptive temperature schedule
        if no_imp_count % 5 == 0:
            beta *= 1.8
            if beta > 50000:
                # Reset and try different step
                z = best_z + 0.1 * np.random.default_rng(iteration).normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= z.mean()
                beta = 100.0
                no_imp_count = 0

        if no_imp_count > 30:
            # Significant stagnation: perturb and restart
            z = best_z + 0.3 * np.random.default_rng(iteration * 7).normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            beta = 50.0
            no_imp_count = 0
            iteration += 10

    return best_a, best_score


def _lbfgs_optimize_v2(a_init, deadline, beta_start=20.0, beta_max=20000.0):
    """Improved L-BFGS with smoother temperature ramp."""
    a_init = np.asarray(a_init, dtype=np.float64).ravel()
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()

    z = np.log(a_init)
    z -= z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)
    best_z = z.copy()

    beta = beta_start
    no_imp = 0
    total_iter = 0

    while time.monotonic() < deadline:
        def obj_func(zv):
            a_v = _softmax(zv)
            c = _fft_convolve(a_v)
            mval = float(c.max())
            x = (c - mval) * beta
            x = np.clip(x, -500, 500)
            ex = np.exp(x)
            s = float(ex.sum())
            if not np.isfinite(s) or s <= 0:
                return 1e10
            return 2.0 * n * (mval + np.log(s) / beta)

        def grad_func(zv):
            a_v = _softmax(zv)
            c = _fft_convolve(a_v)
            mval = float(c.max())
            x = (c - mval) * beta
            x = np.clip(x, -500, 500)
            ex = np.exp(x)
            s = float(ex.sum())
            if not np.isfinite(s) or s <= 0:
                return np.zeros(n)
            w = ex / s
            N = _next_pow2(2 * n - 1)
            fw = np.fft.rfft(w, N)
            a_rev = np.zeros(N)
            a_rev[:n] = a_v[::-1]
            fa_rev = np.fft.rfft(a_rev, N)
            h = np.fft.irfft(fw * fa_rev, N)
            grad_a = 2.0 * h[n - 1:2 * n - 1]
            dot = float(np.dot(a_v, grad_a))
            gz = a_v * (grad_a - dot)
            gz -= gz.mean()
            return gz

        try:
            res = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 300, 'gtol': 1e-7, 'ftol': 1e-14, 'maxls': 30}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = np.clip(res.x, -50.0, 50.0)
                z -= z.mean()
                a_new = _softmax(z)
                sc = _score(a_new)
                
                if np.isfinite(sc) and sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_new.copy()
                    best_z = z.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                
                total_iter += 1
            else:
                no_imp += 1
                total_iter += 1
        except Exception:
            no_imp += 1
            total_iter += 1

        # Temperature schedule
        if no_imp == 1:
            beta *= 1.5
        elif no_imp == 3:
            beta *= 2.0
        elif no_imp == 5:
            beta = min(beta_max, beta * 3.0)
        elif no_imp >= 8:
            # Restart with perturbation
            rng = np.random.default_rng(total_iter * 31 + 7)
            z = best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            beta = max(beta_start, beta * 0.5)
            no_imp = 0

        if beta > beta_max:
            beta = beta_start
            no_imp = 0

    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=50):
    """Coordinate descent with exact 1D optimization."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(42)
    order = rng.permutation(n)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = 0
        for i in order:
            if time.monotonic() > deadline:
                break
            # Try moving mass from i to a random j
            j = rng.integers(0, n)
            if i == j:
                continue
            best_frac = 0.0
            best_try_score = best_score
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                delta = frac * a[i]
                if delta >= a[i]:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                sc = _score(a_try)
                if sc < best_try_score - 1e-12:
                    best_try_score = sc
                    best_frac = frac
            if best_frac > 0:
                delta = best_frac * a[i]
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                improved += 1
                if best_try_score < best_score - 1e-12:
                    best_score = best_try_score
                    best_a = a.copy()
        if improved == 0:
            break

    return best_a, best_score


def _peak_redistribute(a, deadline, max_iter=200):
    """Target the peak of the convolution and redistribute mass from contributors."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        peak_idx = int(c.argmax())
        M = float(c.max())

        # Find which pairs (i, j) with i+j=peak_idx contribute most
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # We want to reduce this by moving mass away from high-contributing positions
        contribs = []
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(key=lambda x: -x[0])

        # Move mass from the top contributor's position
        if contribs:
            top_c, ti, tj = contribs[0]
            if ti != tj and a[ti] > 1e-6:
                # Find position to move to: one that increases the flat part
                # Move a small amount from ti to a random position
                target = np.random.default_rng(it).integers(0, n)
                if target != ti:
                    for frac in [0.01, 0.02, 0.05]:
                        delta = frac * a[ti]
                        a_try = a.copy()
                        a_try[ti] -= delta
                        a_try[target] += delta
                        sc = _score(a_try)
                        if sc < best_score - 1e-10:
                            a = a_try
                            best_score = sc
                            best_a = a.copy()
                            no_imp = 0
                            break
                    else:
                        no_imp += 1
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 100:
            break

    return best_a, best_score


def _block_swap(a, deadline, max_iter=3000):
    """Random block swaps to escape local minima."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(999)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        block_size = int(rng.integers(max(2, n // 32), max(3, n // 8)))
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        a_new[i:i + block_size], a_new[j:j + block_size] = \
            a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
        sc_new = _score(a_new)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 1500:
                break

    return best_a, best_score


def _generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for a given length."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = []

    if parent is not None:
        inits.append(_interpolate_profile(parent, n))

    # U-shaped profiles
    for alpha in [1.0, 2.0, 3.0, 4.0, 6.0]:
        inits.append(_make_u_profile(n, alpha))

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0]:
        inits.append(_make_linear_profile(n, eta))

    # Cosine profiles
    for k in [1, 2, 3]:
        inits.append(_make_cosine_profile(n, k))

    # Polynomial profiles
    inits.append(_make_polynomial_profile(n, [1.0, 0.0, 0.5, 0.0, -0.2]))
    inits.append(_make_polynomial_profile(n, [1.0, 0.0, 0.8, 0.0, -0.3]))
    inits.append(_make_polynomial_profile(n, [1.0, 0.2, 0.1, 0.0, 0.0]))

    # Bimodal
    for w in [0.1, 0.15, 0.25]:
        inits.append(_make_bimodal_profile(n, w))

    # Random structured
    for _ in range(3):
        a_rand = np.maximum(rng.normal(1.0, 0.4, n), 0.0)
        if a_rand.sum() > 1e-10:
            inits.append(a_rand / a_rand.sum())

    return inits

            # Budget per init
            per_init_budget = min(60.0, (deadline - time.monotonic() - 40) / max(1, len(inits) - idx))
            if per_init_budget < 10:
                per_init_budget = 10

            a_opt, sc_opt = _lbfgs_optimize_v2(a0, time.monotonic() + per_init_budget,
                                                beta_start=20.0, beta_max=20000.0)
            if sc_opt < best_at_res_score:
                best_at_res_score = sc_opt
                best_at_res = a_opt.copy()
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  Init {idx} (n={rn}): score={sc_opt:.10f} ** NEW BEST")
            else:
                print(f"  Init {idx} (n={rn}): score={sc_opt:.10f}")

        if best_at_res is not None:
            current_a = best_at_res
            print(f"  Best at n={rn}: {best_at_res_score:.10f}")

    # Phase 2: Intensive optimization at full resolution
    if time.monotonic() < deadline - 120:
        print(f"\nPhase 2: Intensive at n={target_n}")
        rem = deadline - time.monotonic() - 60

        # Generate fresh diverse inits at full resolution
        inits_p2 = _generate_initializations(target_n, parent, np.random.default_rng(seed + 100))
        if current_a is not None:
            inits_p2.insert(0, _interpolate_profile(current_a, target_n))

        # Also add perturbed best
        if best_seq is not None:
            for scale in [0.1, 0.2, 0.3]:
                pert = best_seq * (1.0 + scale * np.random.default_rng(seed + 200).normal(size=target_n))
                pert = np.maximum(pert, 0.0)
                pert = pert / pert.sum()
                inits_p2.append(pert)

        # Try each with appropriate budget
        num_p2 = len(inits_p2)
        per_p2_budget = max(15.0, (rem - 60) / max(1, num_p2))
        print(f"  {num_p2} inits, ~{per_p2_budget:.0f}s each")

        for idx, a0 in enumerate(inits_p2):
            if time.monotonic() > deadline - 60:
                break
            per_b = min(per_p2_budget, deadline - time.monotonic() - 50)
            if per_b < 10:
                break

            a_opt, sc_opt = _lbfgs_optimize_v2(a0, time.monotonic() + per_b,
                                                beta_start=30.0, beta_max=30000.0)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  P2 Init {idx}: {sc_opt:.10f} ** NEW BEST")
            else:
                print(f"  P2 Init {idx}: {sc_opt:.10f}")

    # Phase 3: Local refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\nPhase 3: Local refinement, score={best_score:.10f}")
        rem = deadline - time.monotonic() - 20

        # 3a: One more L-BFGS with very high beta (near-exact max)
        if rem > 30:
            b = min(rem * 0.4, 120.0)
            a_r, sc_r = _lbfgs_optimize_v2(best_seq, time.monotonic() + b,
                                            beta_start=500.0, beta_max=50000.0)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  3a HighBeta L-BFGS: {sc_r:.10f}")
            else:
                print(f"  3a HighBeta L-BFGS: {sc_r:.10f}")

        # 3b: Coordinate descent
        if time.monotonic() < deadline - 20:
            b = min(rem * 0.2, 60.0)
            a_cd, sc_cd = _coordinate_descent(best_seq, time.monotonic() + b, max_rounds=30)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"  3b CoordDescent: {sc_cd:.10f}")
            else:
                print(f"  3b CoordDescent: {sc_cd:.10f}")

        # 3c: Peak redistribution
        if time.monotonic() < deadline - 15:
            b = min(rem * 0.15, 40.0)
            a_pr, sc_pr = _peak_redistribute(best_seq, time.monotonic() + b, max_iter=300)
            if sc_pr < best_score:
                best_score = sc_pr
                best_seq = a_pr.copy()
                print(f"  3c PeakRedist: {sc_pr:.10f}")
            else:
                print(f"  3c PeakRedist: {sc_pr:.10f}")

        # 3d: Block swap
        if time.monotonic() < deadline - 10:
            b = min(rem * 0.15, 30.0)
            a_bs, sc_bs = _block_swap(best_seq, time.monotonic() + b, max_iter=2000)
            if sc_bs < best_score:
                best_score = sc_bs
                best_seq = a_bs.copy()
                print(f"  3d BlockSwap: {sc_bs:.10f}")
            else:
                print(f"  3d BlockSwap: {sc_bs:.10f}")

        # 3e: Final L-BFGS polish
        if time.monotonic() < deadline - 5:
            b = min(rem * 0.2, 40.0)
            if b > 5:
                a_final, sc_final = _lbfgs_optimize_v2(best_seq, time.monotonic() + b,
                                                        beta_start=100.0, beta_max=30000.0)
                if sc_final < best_score:
                    best_score = sc_final
                    best_seq = a_final.copy()
                print(f"  3e Final L-BFGS: {sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(768, 2.0)

    # Validate with evaluate_sequence if possible
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
