def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _smax_val(c, beta):
    m = float(c.max())
    cs = (c - m) / beta
    cs = np.clip(cs, -50, 50)
    return m + beta * float(np.log(np.sum(np.exp(cs))))


def _smax_weights(c, beta):
    cs = (c - c.max()) / beta
    cs = np.clip(cs, -50, 50)
    e = np.exp(cs)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_h(w, a, n):
    """g_j = sum_k w_k * a_{k-j} for j=0..n-1, via FFT."""
    a_rev = a[::-1]
    cw = _conv(a_rev, w)
    return cw[n-1:2*n-1]


def _score_norm(a):
    n = len(a)
    c = _conv(a, a)
    return 2.0 * n * float(c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a) / (s * s)


def _obj_fn(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    return 2.0 * n * _smax_val(c, beta)


def _grad_fn(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    w = _smax_weights(c, beta)
    W = float(np.dot(w, c))
    h = _grad_h(w, a, n)
    g = 2.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g


def _project_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rng_ = np.arange(1, n + 1)
    cond = u * rng_ > (css - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    tau = (css[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def _make_beta_shape(alpha, beta, n):
    """a_i ∝ (i+1)^alpha * (n-i)^beta, i=0..n-1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** alpha * (n - i) ** beta
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def _make_exp_shape(pa, pb, n):
    """a_i ∝ exp(pa * t + pb * t^2) where t=i/(n-1)."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.exp(pa * t + pb * t * t)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def _interp_to_n(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _lbfgs_continuation(a_init, deadline, max_wall):
    """Run L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    c0 = _conv(a_init, a_init)
    m0 = float(c0.max())
    
    # Beta schedule: start large, decrease
    beta_fracs = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    n_levels = len(beta_fracs)
    per_level = max_wall / n_levels

    for frac in beta_fracs:
        if time.monotonic() > deadline:
            break
        beta = max(1e-10, frac * m0)
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            res = minimize(
                _obj_fn, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_fn,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 1000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax(res.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass

    return best_a, best_sc


def _adam_descent(a_init, deadline, lr=0.02, beta_frac=0.01, max_iter=50000):
    """Adam descent in log-space with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _conv(a_init, a_init)
    beta = max(1e-10, beta_frac * float(c.max()))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax(z)
        c = _conv(a, a)
        w = _smax_weights(c, beta)
        W = float(np.dot(w, c))
        h = _grad_h(w, a, n)
        grad = 2.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            continue

        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        sc = _score_norm(_softmax(z))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = _softmax(z).copy()

        # Adaptively decrease beta
        if t % 100 == 0:
            beta = max(1e-12, beta * 0.9)
        if t % 2000 == 0 and best_sc < 1.51:
            lr = max(0.003, lr * 0.92)

    return best_a, best_sc


def _flatten_descent(a_init, deadline, lr=0.05, max_iter=20000):
    """Minimize variance of c (to flatten the convolution profile)."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _conv(a, a)
        mean_c = c.mean()
        # Gradient of sum((c - mean_c)^2) = 2 * sum((c_k - mean_c) * dc_k/da_i)
        # dc_k/da_i = a_{k-i} (if valid)
        # d(sum(c^2))/da_i = 2 * sum_k c_k * a_{k-i} = 2 * h_i (where h = conv with c)
        # d(mean_c)/da_i = 1/(2n-1) * sum_k a_{k-i} ... this is getting complicated
        # 
        # Simpler: minimize sum(c^2) - (sum(c))^2/(2n-1) = sum(c^2) - 1/(2n-1) (since sum(c)=(sum(a))^2=1)
        # So minimize sum(c^2).
        # d(sum(c^2))/da_i = 2 * sum_k c_k * dc_k/da_i = 2 * sum_k c_k * a_{k-i}
        # = 2 * h_i where h_i = sum_k c_k * a_{k-i}
        
        h_flat = _grad_h(c, a, n)  # h_i = sum_k c_k * a_{k-i}
        grad_a = 2.0 * h_flat  # gradient of sum(c^2) w.r.t. a
        
        # Convert to z-space gradient
        W_flat = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W_flat)
        
        if not np.all(np.isfinite(grad_z)):
            break
            
        z = z - lr * grad_z
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        sc = _score_norm(_softmax(z))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = _softmax(z).copy()
        
        # Decay learning rate
        if it % 1000 == 0:
            lr *= 0.95

    return best_a, best_sc
