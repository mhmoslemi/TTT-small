"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using LP-based first-order descent,
LSE-smoothed Adam, and subgradient methods with multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unit(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = x.size
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def _lp_descent_step(a, use_full=True, top_k=200):
    """
    LP-based first-order descent. Returns (direction, predicted_M).
    Solves: min M s.t. (a*a)_k + 2*(a*b)_k <= M, b >= -a, sum(b)=0.
    """
    n = len(a)
    c_aa = _fft_convolve(a)
    current_max = float(c_aa.max())
    
    # Build constraint matrix
    if use_full:
        k_values = np.arange(2 * n - 1)
    else:
        idx = np.argsort(c_aa)[::-1]
        k_values = idx[:top_k]
    
    m = len(k_values)
    # Variables: b_0,...,b_{n-1}, M (index n)
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    
    # Convolution constraints
    for row, k in enumerate(k_values):
        b_ub[row] = -c_aa[k]
        A_ub[row, n] = -1  # coefficient for M
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[row, lo:hi + 1] = a[lo:hi + 1]
    
    # Non-negativity of a+b: b_j >= -a_j => -b_j <= a_j
    for j in range(n):
        A_ub[m + j, j] = -1
        b_ub[m + j] = a[j]
    
    # Equality: sum(b) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: M >= 0
    bounds = [(None, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, current_max
    
    b_dir = res.x[:n]
    M_pred = res.x[n]
    
    return b_dir, M_pred


def _line_search_lp(a, b_dir, t_max):
    """Line search for t in [0, t_max] to minimize score(a + t*b_dir)."""
    if t_max <= 0:
        return 0.0
    
    # Try a few values
    candidates = [0.0, 0.1, 0.25, 0.5, 0.75, min(1.0, t_max)]
    # Also add t_max/2 and t_max
    candidates.append(t_max * 0.5)
    candidates.append(t_max)
    
    # Deduplicate and filter
    seen = set()
    t_candidates = []
    for t in candidates:
        t = min(t, t_max)
        t = max(0, t)
        key = round(t, 6)
        if key not in seen:
            seen.add(key)
            t_candidates.append(t)
    
    best_t = 0.0
    best_s = _score_unit(a)
    
    for t in t_candidates:
        a_new = a + t * b_dir
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        s = _score_unit(a_new)
        if s < best_s:
            best_s = s
            best_t = t
    
    return best_t

def _lse_gradient(a, T):
    """Compute LSE-smoothed gradient."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    
    # Gradient: 2 * cross-correlation of w with a
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, w


def _lse_adam(a, deadline, max_iter=50000, T_start=0.03, T_end=1e-5, lr=0.05):

    a = np.asarray(a, dtype=np.float64).copy()

    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    # Work in logit space for better conditioning
    z = np.log(a)
    z -= np.mean(z)
    
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    t_count = 0
    
    best_a = a.copy()
    best_score = _score_unit(a)
    T = T_start
    
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1)) if max_iter > 1 else 1.0
    b1, b2 = 0.9, 0.999
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, w = _lse_gradient(a, T)
        
        # Project gradient onto tangent space of simplex
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= np.mean(gz)
        
        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= np.mean(z)
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            t_count = 0
            continue
        
        t_count += 1
        state_m = b1 * state_m + (1 - b1) * gz
        state_v = b2 * state_v + (1 - b2) * (gz * gz)
        mh = state_m / (1 - b1 ** t_count)
        vh = state_v / (1 - b2 ** t_count)
        
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -30, 30)

        a_new = e / e.sum()
        
        sc = _score_unit(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 5 * n):
                # Restart from best

                a = np.maximum(a, 1e-15)

                z = np.log(a)
                z -= np.mean(z)

                state_v = np.zeros(n)
                t_count = 0

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using LP first-order descent,
LSE-smoothed Adam, L2-norm Adam, and subgradient methods with multi-length search.
Improved over previous state-of-art with better LP formulation and L2 proxy objective.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unit(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(x):
    n = x.size
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def _lp_descent_step(a, top_k=200):
    n = len(a)
    c_aa = _fft_convolve(a)
    current_max = float(c_aa.max())
    idx = np.argsort(c_aa)[::-1]
    k_values = idx[:top_k]
    m = len(k_values)
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    for row, k in enumerate(k_values):
        b_ub[row] = -c_aa[k]
        A_ub[row, n] = -1.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[row, lo:hi + 1] = a[lo:hi + 1]
    for j in range(n):
        A_ub[m + j, j] = -1.0
        b_ub[m + j] = a[j]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(None, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, current_max
    return res.x[:n], res.x[n]


def _line_search_t(a, b_dir, t_max):
    if t_max <= 1e-10:
        return 0.0
    best_t = 0.0
    best_s = _score_unit(a)
    for t in [0.1, 0.25, 0.5, 0.75, min(1.0, t_max * 0.8), t_max * 0.5, t_max]:

            continue
        a_new = a_new / s_new
        s = _score_unit(a_new)
        if s < best_s:
            best_s = s
            best_t = t
    return best_t

def _lse_gradient(a, T):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(np.concatenate([a[::-1], np.zeros(N - n)]), N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h
