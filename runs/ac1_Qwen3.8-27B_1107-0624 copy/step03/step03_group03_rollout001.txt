"""
Optimized step function search combining LSE gradient with BB step sizes,
full-matrix LP refinement, subgradient polishing, and diverse multi-start
initialization across multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score with sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_full_conv_matrix(n):
    """Create the (2n-1) x n convolution matrix for fixed a (sparsity pattern)."""
    # Row k has non-zeros in columns max(0,k-n+1) to min(n-1,k)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jStart = max(0, k - n + 1)
        jEnd = min(n, k + 1)
        for j in range(jStart, jEnd):
            rows.append(k)
            cols.append(j)
    return rows, cols


def _conv_matrix_as_dense(a):
    """Build dense conv matrix where row k has a[j] for valid j."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jStart = max(0, k - n + 1)
        jEnd = min(n, k + 1)
        A[k, jStart:jEnd] = a[jStart:jEnd]
    return A


def _lp_spread_full(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Uses full constraint set."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c_obj = -np.ones(n)
    A = _conv_matrix_as_dense(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True, "dual_ratio": 10.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_spread_reduced(a, M, top_k=80):
    """LP with reduced constraints for speed on larger n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    c_obj = -np.ones(n)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        jStart = max(0, k - n + 1)
        jEnd = min(n, k + 1)
        A[row, jStart:jEnd] = a[jStart:jEnd]
    try:

                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True})

    except Exception:
        pass

def _line_search(a, g, max_evals=25):
    """Golden section search for optimal blend t in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = _score_norm(np.maximum(a + x1 * (g - a), 0))
    f2 = _score_norm(np.maximum(a + x2 * (g - a), 0))
    for _ in range(max_evals - 2):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = _score_norm(np.maximum(a + x1 * (g - a), 0))
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = _score_norm(np.maximum(a + x2 * (g - a), 0))
    return (lo + hi) / 2.0


def _lse_adam_step(a, z, state, T, lr):
    """One Adam step in logit space using log-sum-exp smoothing."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a0, deadline, max_iter=200000, T_start=0.05, T_end=1e-6, lr=0.1):
    """LSE gradient optimization with cosine annealing and structured restarts."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    base_lr = lr
    rng = np.random.default_rng(12345)

    # Cosine annealing
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        progress = it / max(max_iter - 1, 1)
        T = T_end + 0.5 * (T_start - T_end) * (1.0 + np.cos(np.pi * progress))
        T = max(T, T_end)

        a, z, state, m_val = _lse_adam_step(a, z, state, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 8 * n):
                # Structured restart: add sinusoidal perturbation
                freq = rng.integers(1, max(2, n // 4))
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.05 + 0.03 * rng.random()
                t_idx = np.arange(n)
                pert = 1.0 + amp * np.sin(2 * np.pi * freq * t_idx / n + phase)
                a = best_a * pert
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
            elif no_imp % 2000 == 0:
                lr = max(0.02, lr * 0.92)

    return best_a


def _lp_refine(a0, deadline, max_iter=5000, use_full=False, inflate_start=1.001):
    """LP-based refinement with adaptive inflation and line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    inflate = inflate_start
    blend = 0.2

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        if use_full and n <= 1024:
            g = _lp_spread_full(a, M_lp)
        else:
            g = _lp_spread_reduced(a, M_lp, top_k=100)

        if g is None:
            inflate = min(1.02, inflate * 1.3)
            no_imp += 5
            if no_imp > 500:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(1.02, inflate * 1.1)
            if no_imp > 600:
                break
            continue

        g = g / sum_g

        # Line search
        t_opt = _line_search(a, g, max_evals=15)
        if t_opt < 0.01:
            t_opt = blend * 0.5

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend = max(0.03, blend * 0.9)
                inflate = max(1.0001, inflate * 0.95)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend = 0.2
                inflate = inflate_start
                no_imp = 0

        a = a_new

    return best_a


def _subgradient_polish(a0, deadline, max_iter=100000, lr=0.02):
    """Subgradient polish at the exact max with momentum."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        # Use top few peaks for smoother gradient
        threshold = m_val - 0.02 * m_val
        max_idx = np.where(c >= threshold)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 800 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(5000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_with_edges(n, edge_weight=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, center_sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5 + center_sep) / 0.12) ** 2) + \
        np.exp(-0.5 * ((t - 0.5 - center_sep) / 0.12) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_smooth(n, rng, order=3):
    """Generate a random smooth non-negative sequence via spline interpolation."""
    n_ctrl = max(4, n // order)
    ctrl = rng.random(n_ctrl)
    x_ctrl = np.linspace(0, 1, n_ctrl)
    x = np.linspace(0, 1, n)
    a = np.interp(x, x_ctrl, ctrl)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LSE gradient with cosine annealing and structured
    restarts, full-matrix LP refinement, and subgradient polishing.
    Explores lengths 512, 1024, 2048 with diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try based on available time
    lengths = [512, 1024]
    if budget_s > 1200:
        lengths.append(2048)

    print(f"\nTrying lengths: {lengths}, budget={budget_s-30:.0f}s")
    print(f"Initial best: {best_score:.10f}")

    # === Phase 1: LSE gradient optimization ===
    print("\n=== Phase 1: LSE Gradient Optimization ===")
    best_per_length = {}

    # Allocate time proportionally to lengths
    n_lengths = len(lengths)
    phase1_time = min(1000.0, deadline - time.monotonic() - 300.0)
    if phase1_time < 60:
        phase1_time = deadline - time.monotonic() - 30.0

    time_per_length = phase1_time / n_lengths

    for n_t in lengths:
        if time.monotonic() > deadline - 200:
            break

        t_phase_start = time.monotonic()
        t_phase_end = min(t_phase_start + time_per_length, deadline - 200)
        print(f"\n  --- Length n={n_t}, time budget={time_per_length:.0f}s ---")

        # Generate diverse candidates
        candidates = []
        if parent is not None:
            candidates.append(_interpolate_profile(parent, n_t))
            for sp in [0.05, 0.1, 0.15]:
                p_pert = _interpolate_profile(parent, n_t) * (1.0 + sp * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 1e-10)
                p_pert = p_pert / p_pert.sum()
                candidates.append(p_pert)
        candidates.append(_make_gaussian(n_t, 0.35))
        candidates.append(_make_raised_cosine(n_t))
        candidates.append(_make_flat_with_edges(n_t, 0.5))
        candidates.append(_make_bimodal(n_t, 0.25))
        candidates.append(_make_bimodal(n_t, 0.35))
        candidates.append(_make_random_smooth(n_t, rng, order=3))
        candidates.append(np.ones(n_t) / n_t)

        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(candidates):
            if time.monotonic() > t_phase_end:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
                print(f"    New best from init: {sc0:.10f}")

            # Run LSE optimization
            a_lse = _lse_optimize(a0, t_phase_end, max_iter=150000,
                                   T_start=0.05, T_end=1e-6, lr=0.1)
            sc_lse = _score(a_lse)

            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
                print(f"    New best from LSE: {sc_lse:.10f}")
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse

            elapsed = time.monotonic() - t_phase_start
            print(f"    cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f} ({elapsed:.0f}s)")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # === Phase 2: LP Refinement ===
    print("\n=== Phase 2: LP Refinement ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 150:
            break

        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"\n  LP on n={n_t}: start={sc_in:.10f}")

        # Allocate LP time
        lp_budget = min(300.0, deadline - time.monotonic() - 150.0)
        if lp_budget < 20:
            continue
        lp_end = time.monotonic() + lp_budget

        use_full = (n_t <= 1024)
        a_lp = _lp_refine(a_in, lp_end, max_iter=5000,
                          use_full=use_full, inflate_start=1.0005)
        sc_lp = _score(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  New best from LP: {sc_lp:.10f}")
        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: {sc_lp:.10f}")

    # === Phase 3: Subgradient Polish ===
    print("\n=== Phase 3: Subgradient Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sp_budget = min(150.0, deadline - time.monotonic() - 60.0)
        if sp_budget > 20:
            sp_end = time.monotonic() + sp_budget
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=100000, lr=0.015)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
                print(f"  New best from subgradient: {sc_sp:.10f}")
            print(f"  Subgradient: {sc_sp:.10f}")

    # === Phase 4: Final LP polish (fine) ===
    print("\n=== Phase 4: Final LP Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp_budget = min(100.0, deadline - time.monotonic() - 30.0)
        if lp_budget > 15:
            lp_end = time.monotonic() + lp_budget
            n_best = len(best_seq)
            use_full = (n_best <= 1024)
            a_final = _lp_refine(best_seq, lp_end, max_iter=3000,
                                 use_full=use_full, inflate_start=1.0002)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  New best from final LP: {sc_final:.10f}")
            print(f"  Final LP: {sc_final:.10f}")

    # === Phase 5: One more LSE pass on the very best ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        lse_budget = min(60.0, deadline - time.monotonic() - 15.0)
        if lse_budget > 10:
            lse_end = time.monotonic() + lse_budget
            a_last = _lse_optimize(best_seq, lse_end, max_iter=50000,
                                    T_start=0.02, T_end=1e-7, lr=0.05)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
                print(f"  New best from final LSE: {sc_last:.10f}")
            print(f"  Final LSE: {sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
