"""

import numpy as np
import time
from scipy.optimize import minimize

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1"""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def soft_max_value_and_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def gradient_descent_optimize(a0, deadline, rng, lr=0.03, beta_init=0.05,
                               beta_decay=0.94, beta_min=1e-10, 
                               restart_after=2000, use_adam=True):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    c = full_convolve_fft(a)
    max_c = float(c.max())
    
    z = np.log(a)
    z = z - z.mean()
    
    beta = beta_init * max_c
    best_a = a.copy()
    best_score = 2.0 * n * max_c
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() < deadline:
        _, w = soft_max_value_and_weights(c, beta)
        h = compute_h(w, a)
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        dz = a * (grad - dot)
        
        if use_adam:
            t += 1
            m_adam = b1 * m_adam + (1 - b1) * dz
            v_adam = b2 * v_adam + (1 - b2) * (dz * dz)
            mh = m_adam / (1 - b1**t)
            vh = v_adam / (1 - b2**t)
            step = cur_lr * mh / (np.sqrt(vh) + eps)
        else:
            rms = np.sqrt(np.mean(dz**2)) + 1e-10
            step = cur_lr * dz / rms
        
        z = z - step
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        a_new = np.exp(z - z.max())
        s = a_new.sum()
        if s < 1e-15:
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = full_convolve_fft(a)
            continue
        a_new = a_new / s
        
        c_new = full_convolve_fft(a_new)
        max_c_new = float(c_new.max())
        score = 2.0 * n * max_c_new
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            c = c_new
            max_c = max_c_new
        else:
            no_improve += 1
            # Revert if no improvement
            a = a_new  # Still move forward to explore
            c = c_new
            max_c = max_c_new
        
        # Adapt beta
        if t % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Adapt learning rate
        if no_improve > 200:
            cur_lr = max(0.003, cur_lr * 0.98)
        
        # Restart
        if no_improve > restart_after:
            noise = 0.2 * rng.normal(size=n)
            z_best = np.log(np.maximum(best_a, 1e-15))
            z = z_best + noise
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = full_convolve_fft(a)
            max_c = float(c.max())
            beta = max(beta_min, 0.06 * max_c)
            m_adam.fill(0)
            v_adam.fill(0)
            t = 0
            cur_lr = lr
            no_improve = 0
    
    return best_a, best_score

def peak_suppression(a0, deadline, rng, n_top=5, lr=0.02):
    """Directly target top peaks for suppression."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    z = np.log(a)
    z = z - z.mean()
    best_a = a.copy()
    best_score = 2.0 * n * full_convolve_fft(a).max()
    
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        # Get top peaks
        top_idx = np.argsort(c)[-n_top:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / n_top
        
        h = compute_h(w, a)
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        dz = a * (grad - dot)
        
        t += 1
        m_adam = b1 * m_adam + (1 - b1) * dz
        v_adam = b2 * v_adam + (1 - b2) * (dz * dz)
        mh = m_adam / (1 - b1**t)
        vh = v_adam / (1 - b2**t)
        step = lr * mh / (np.sqrt(vh) + eps)
        
        z = z - step
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        a_new = np.exp(z - z.max())
        s = a_new.sum()
        a_new = a_new / max(s, 1e-15)
        
        c_new = full_convolve_fft(a_new)
        score = 2.0 * n * c_new.max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def simulated_annealing_polish(a0, deadline, rng, n_trials=500):
    """Try random perturbations and keep improvements."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * full_convolve_fft(a).max()
    
    temperature = 0.01
    cooling = 0.995
    
    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        # Random perturbation: pick a random subset of indices and perturb them
        n_perturb = max(1, n // 10)
        idx = rng.integers(0, n, n_perturb)
        noise = rng.normal(0, 0.1, n_perturb)
        
        a_perturbed = a.copy()
        a_perturbed[idx] = a_perturbed[idx] + noise
        a_perturbed = np.maximum(a_perturbed, 1e-15)
        a_perturbed = a_perturbed / a_perturbed.sum()
        
        c = full_convolve_fft(a_perturbed)
        score = 2.0 * n * c.max()
        
        if score < best_score:
            best_score = score
            best_a = a_perturbed.copy()
            a = a_perturbed
        elif rng.random() < np.exp(-(score - best_score) / temperature):
            a = a_perturbed
        
        temperature *= cooling
    
    return best_a, best_score

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, p0)
    return result / max(result.sum(), 1e-15)

def compute_score(a):

    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def compute_h(w, a):
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval, w


def adaptive_beta(c, base_beta, min_beta=1e-12):
    """Adapt beta based on the gap between top peaks."""
    sorted_c = np.sort(c)[::-1]
    if len(sorted_c) < 2:
        return base_beta
    gap = sorted_c[0] - sorted_c[1]
    mval = sorted_c[0]
    if mval < 1e-15:
        return base_beta
    relative_gap = gap / mval
    # If gap is small, use larger beta (smoother); if large, use smaller beta
    scale = max(0.3, min(3.0, 0.5 / max(relative_gap, 1e-6)))
    return max(min_beta, base_beta * scale)

def simulated_annealing(a0, deadline, rng, n_trials=3000):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * full_convolve_fft(a).max()

    temperature = 0.005
    cooling = 0.997

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break

        n_perturb = max(2, n // 8)
        idx = rng.integers(0, n, n_perturb)
        noise = rng.normal(0, 0.05, n_perturb)

        a_new = a.copy()
        a_new[idx] += noise
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c = full_convolve_fft(a_new)
        score = 2.0 * n * c.max()

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        elif rng.random() < np.exp(-(score - best_score) / temperature):
            a = a_new

        temperature = max(1e-6, temperature * cooling)

    return best_a, best_score

    if beta < 1e-10:
        beta = 1e-6

        z_centered = z - z.mean()
        a_curr = np.exp(z_centered - z_centered.max())
        s = a_curr.sum()
        if s < 1e-15:
            return 1e10
        a_curr = a_curr / s
        c = full_convolve_fft(a_curr)
        mval = float(c.max())
        x = np.clip((c - mval) / beta, -80.0, 0.0)
        ex = np.exp(x)
        sm_val = mval + beta * np.log(ex.sum())
        return 2.0 * n * sm_val

    def gradient(z):
        z_centered = z - z.mean()
        a_curr = np.exp(z_centered - z_centered.max())
        s = a_curr.sum()
        if s < 1e-15:
            return np.zeros(n)
        a_curr = a_curr / s
        c = full_convolve_fft(a_curr)
        mval = float(c.max())
        x = np.clip((c - mval) / beta, -80.0, 0.0)
        ex = np.exp(x)
        w = ex / ex.sum()
        h = compute_h(w, a_curr)
        grad_a = 2.0 * h
        dot = float(np.dot(a_curr, grad_a))
        dz = a_curr * (grad_a - dot)
        return dz

    try:
        result = scipy_minimize(
            objective, z0, jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': 500,
                     'ftol': 1e-12, 'gtol': 1e-8}
        )
        z_opt = result.x - result.x.mean()
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        return a_opt
    except Exception:
        return a


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def diverse_starts(n, rng):
    starts = []
    t = np.linspace(0, 1, n)
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    for k in [1, 2, 3]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    for _ in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a += cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    print(f"=== Starting search. Budget={budget_s}s ===")

    # Load initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    if init_seq is not None:
        sc = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc:.10f}")
        update(init_seq, sc)
    else:
        print("No initial sequence found, starting fresh.")

    # Quick n-sweep to find promising sizes
    n_candidates = [256, 384, 512, 768, 1024, 1536, 2048]
    print("\n--- Quick n-sweep ---")
    if init_seq is not None:
        for nc in n_candidates:
            p = interpolate_profile(init_seq, nc)
            sc = compute_score(p)
            update(p, sc)
            print(f"  n={nc}: {sc:.10f}")

    # Determine focus n
    if best_seq is not None:
        n_focus = best_seq.size
    else:
        n_focus = 1024

    # Phase 1: Gradient optimization at focus n (long run)
    print(f"\n--- Phase 1: GD at n={n_focus} ---")
    phase1_deadline = min(deadline, t_start + 600.0)

    base = interpolate_profile(best_seq if best_seq is not None else np.ones(n_focus) / n_focus, n_focus)
    r1, s1 = gradient_optimize(base, phase1_deadline, rng,
                                lr=0.025, beta_factor=0.04,
                                beta_decay=0.95, restart_after=3000)
    s1f = compute_score(r1)
    update(r1, s1f)
    print(f"  GD result: {s1f:.10f}")

    # Phase 2: Peak suppression
    print(f"\n--- Phase 2: Peak suppression ---")
    phase2_deadline = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_deadline - 10:
        dl = min(phase2_deadline, time.monotonic() + 200.0)
        r2, s2 = peak_suppression_optimize(r1, dl, rng, n_top=3, lr=0.012)
        s2f = compute_score(r2)
        update(r2, s2f)
        print(f"  Peak suppression: {s2f:.10f}")
        if s2f < s1f:
            r1 = r2

    # Phase 3: Simulated annealing
    print(f"\n--- Phase 3: Simulated annealing ---")
    phase3_deadline = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_deadline - 10:
        dl = min(phase3_deadline, time.monotonic() + 100.0)
        r3, s3 = simulated_annealing(r1, dl, rng, n_trials=2000)
        s3f = compute_score(r3)
        update(r3, s3f)
        print(f"  SA result: {s3f:.10f}")
        if s3f < compute_score(r1):
            r1 = r3

    # Phase 4: Try other n values
    print(f"\n--- Phase 4: Multi-n ---")
    phase4_deadline = min(deadline, t_start + 1600.0)
    for n_try in [768, 1024, 1536, 2048]:
        if time.monotonic() > phase4_deadline - 20:
            break
        if n_try == n_focus:
            continue
        p = interpolate_profile(best_seq, n_try)
        sc_p = compute_score(p)
        if sc_p >= best_score + 0.02:
            continue
        print(f"  Trying n={n_try}: interpolated={sc_p:.10f}")
        dl = min(phase4_deadline, time.monotonic() + 80.0)
        r, sc_r = gradient_optimize(p, dl, rng, lr=0.02, beta_factor=0.03,
                                     beta_decay=0.95, restart_after=2000)
        sc_rf = compute_score(r)
        update(r, sc_rf)
        print(f"    Optimized: {sc_rf:.10f}")

    # Phase 5: Final scipy polish
    print(f"\n--- Phase 5: scipy polish ---")
    if best_seq is not None and time.monotonic() < deadline - 15:
        dl = min(deadline, time.monotonic() + 40.0)
        # Check if we can do scipy polish
        test_obj = objective_cache = None
        try:
            r_scipy = scipy_polish(best_seq, dl, maxiter=100)
            sc_scipy = compute_score(r_scipy)
            update(r_scipy, sc_scipy)
            print(f"  scipy polish: {sc_scipy:.10f}")
        except Exception as e:
            print(f"  scipy polish failed: {e}")

    # Final: ultra-fine gradient polish
    print(f"\n--- Final polish ---")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = best_seq.size
        p = best_seq + 0.0005 * rng.normal(size=n_fin)
        p = np.maximum(p, 1e-12)
        p = p / p.sum()
        dl = min(deadline, time.monotonic() + 30.0)
        r_fin, s_fin = gradient_optimize(p, dl, rng, lr=0.005,
                                          beta_factor=0.01, beta_decay=0.97)
        s_fin_f = compute_score(r_fin)
        update(r_fin, s_fin_f)
        print(f"  Ultra-fine: {s_fin_f:.10f}")

    # Return best
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        final = np.minimum(final, 1000.0)
        s = final.sum()
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(256) / 256.0
    else:
        final = np.ones(512) / 512.0

    final_score = compute_score(final)
    print(f"\n=== FINAL: n={final.size}, score={final_score:.10f} ===")

    try:
        ev = float(evaluate_sequence([float(x) for x in final]))
        print(f"evaluate_sequence confirms: {ev:.10f}")
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final]
