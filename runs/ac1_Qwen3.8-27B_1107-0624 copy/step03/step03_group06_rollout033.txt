def _subgradient_polish(a0, deadline, max_iter=50000):
    """Subgradient method for minimizing max(conv(a,a)) on the simplex."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    lr = 0.01
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        max_c = float(c.max())
        
        # Find active lags (within 1% of max)
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        
        # Subgradient: average of grad(c_k) over active lags
        # grad(c_k)_i = 2 * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k in active:
            # For each active lag k, the gradient is 2 * a reversed and shifted
            # (grad)_i = 2 * a[k-i] for 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # grad[lo:hi] += 2 * a[k - lo : k - hi + 1][::-1]
            # Actually: grad[i] += 2 * a[k-i] for i in [max(0,k-n+1), min(n-1,k)]
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            # For i in [i_start, i_end): grad[i] += 2 * a[k-i]
            indices = np.arange(i_start, i_end)
            grad[indices] += 2.0 * a[k - indices]
        
        grad /= len(active)
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a_new = a_new / s_new
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            # Slowly decrease lr
            if it % 100 == 99:
                lr *= 0.98
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                lr *= 0.95
            if no_imp > max(1000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.01
                no_imp = 0
        
        if lr < 0.0001:
            break
    
    return best_a

"""
Optimized step function coefficient search using binomial initialization,
symmetric LP refinement, and FFT-based subgradient polishing.
Key innovations: binomial coefficient init (near-flat autocorrelation),
adaptive LP inflation, active-set subgradient with FFT, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog
from math import comb


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _binomial_init(n):
    a = np.zeros(n, dtype=np.float64)
    for j in range(n):
        a[j] = comb(2 * j, j) / (4.0 ** j)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _symmetric_binomial_init(n):
    a = _binomial_init(n)
    return (a + a[::-1]) / 2.0


def _tilted_binomial_init(n, tilt):
    a = _binomial_init(n).copy()
    t = np.linspace(0.0, 1.0, n)
    a = a * (1.0 + tilt * t)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def _ramp_init(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _solve_lp_symmetric(f, M, n):
    if n % 2 == 0:
        n_half = n // 2
        f_half = f[:n_half]
        g_half = _solve_lp(f_half, M, n_half)
        if g_half is None:
            return None
        g = np.concatenate([g_half, g_half[::-1]])
    else:
        n_half = (n + 1) // 2
        f_half = f[:n_half]
        g_half = _solve_lp(f_half, M, n_half)
        if g_half is None:
            return None
        g = np.concatenate([g_half, g_half[-2::-1]])
    return g

    base_lr = lr
    momentum = 0.75

            v = np.zeros(n)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
