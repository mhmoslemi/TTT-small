"""
LP-driven optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. Tighter LP (minimal or no inflation) with direct updates
2. Adaptive perturbation using gradient of smooth max when stuck
3. Smarter blending: try direct LP solution first, blend only if needed
4. Precomputed convolution matrix structure for fast LP builds
5. Multi-phase: LP refinement -> gradient polish -> final LP
6. Periodic re-evaluation of strategy based on convergence rate

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Remove tiny negatives from FFT
    mx = c.max()
    mn = c.min()
    if mn < -1e-10 * max(abs(mx), 1.0):
        c = np.maximum(c, 0.0)
    else:
        c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(a):
    """Build dense convolution matrix A of shape (2n-1, n) where (A @ b)_k = (a*b)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _solve_lp_spread(a, M):
    """Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _smooth_max_weights(c, beta):
    """Compute softmax weights for smooth max approximation."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        return mval, np.zeros_like(c)
    w = ex / s
    return mval, w


def _grad_smooth_max(a, w):
    """
    Gradient of sum_k w_k * c_k where c = a*a (convolution).
    d(c_k)/d(a_i) = 2 * a_{k-i} (for valid indices).
    So grad_i = 2 * sum_k w_k * a_{k-i}
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        # a[k-i] for i = i_min, ..., i_max
        # k-i ranges from k-i_min down to k-i_max
        j_lo = k - i_max
        j_hi = k - i_min
        segment = a[j_lo:j_hi + 1][::-1]
        grad[i_min:i_max + 1] += 2.0 * wk * segment
    return grad


def _logit_gradient_step(a, beta, lr):
    """One gradient step in logit space to reduce smooth max of convolution."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    c = _fft_convolve(a)
    mval, w = _smooth_max_weights(c, beta)
    grad_c = _grad_smooth_max(a, w)
    # Project gradient: we want to minimize mval subject to sum(a) = 1, a >= 0
    # In logit space: z = log(a), update z -= lr * (grad_c - mean(a * grad_c)) * a
    p = a
    vp = grad_c
    dot = float(np.dot(p, vp))
    gz = p * (vp - dot)
    gz -= float(np.mean(gz))
    z = np.log(p)
    z -= np.mean(z)
    z = z - lr * gz
    z = np.clip(z, -50.0, 50.0)
    z -= np.mean(z)
    e = np.exp(z)
    a_new = e / e.sum()
    return a_new, mval


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

    last_print = 0

    # Track history for momentum
    recent_solutions = []

    while time.monotonic() < deadline:
        iter_count += 1
        c = _fft_convolve(a)
        M = float(c.max())

        # Solve LP
        b = _solve_lp_spread(a, M)

        if b is not None:
            sum_b = float(b.sum())
            if sum_b > 1e-10:
                b = b / sum_b
                c_b = _fft_convolve(b)
                M_b = float(c_b.max())

                improved = False
                if M_b < best_M * (1.0 - 1e-12):
                    # Direct update
                    a = b.copy()
                    best_M = M_b
                    best_score = float(2.0 * n * M_b)
                    best_a = a.copy()
                    no_imp = 0
                    improved = True
                elif M_b < M * (1.0 - 1e-10):
                    # Better than current but not better than best - try blend with best
                    for t in [0.3, 0.5, 0.7, 1.0]:
                        a_try = (1.0 - t) * a + t * b
                        a_try = np.maximum(a_try, 1e-15)
                        a_try = a_try / a_try.sum()
                        c_try = _fft_convolve(a_try)
                        M_try = float(c_try.max())
                        if M_try < best_M * (1.0 - 1e-12):
                            a = a_try
                            best_M = M_try
                            best_score = float(2.0 * n * M_try)
                            best_a = a.copy()
                            no_imp = 0
                            improved = True
                            break
                        elif M_try < M * (1.0 - 1e-10):
                            a = a_try
                            improved = True
                            break

                if not improved:
                    # LP didn't help directly - try small blends
                    for t in [0.1, 0.05, 0.02, 0.01]:
                        a_try = (1.0 - t) * a + t * b
                        a_try = np.maximum(a_try, 1e-15)
                        a_try = a_try / a_try.sum()
                        c_try = _fft_convolve(a_try)
                        M_try = float(c_try.max())
                        if M_try < M * (1.0 - 1e-10):
                            a = a_try
                            no_imp = 0
                            improved = True
                            break

                    if not improved:
                        no_imp += 1

        else:
            no_imp += 1

        # Perturbation if stuck
        if no_imp > 20:
            # Gradient-based perturbation
            beta = max(1e-8, 0.01 * best_M)
            c_cur = _fft_convolve(a)
            _, w = _smooth_max_weights(c_cur, beta)
            grad = _grad_smooth_max(a, w)

            # Apply perturbation in logit space
            lr_pert = 0.3 * (1.0 + min(no_imp, 100) * 0.1)
            z = np.log(np.maximum(a, 1e-15))
            z -= np.mean(z)
            p = a
            dot = float(np.dot(p, grad))
            gz = p * (grad - dot)
            gz -= np.mean(gz)
            z = z - lr_pert * gz
            z = np.clip(z, -50.0, 50.0)
            z -= np.mean(z)
            e = np.exp(z)
            a = e / e.sum()

            # Also add random noise for diversity
            noise_scale = 0.05 * min(no_imp / 20.0, 2.0)
            a = a * (1.0 + noise_scale * rng.standard_normal(n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

            # Check if perturbation helped
            c_new = _fft_convolve(a)
            M_new = float(c_new.max())
            if M_new < best_M:
                best_M = M_new
                best_score = float(2.0 * n * M_new)
                best_a = a.copy()
            no_imp = 0

        # ** Every 500 iterations, alternate with a gradient step
        elif iter_count % 500 == 0 and iter_count > 0:
            beta = max(1e-8, 0.05 * M)
            a_grad, mval = _logit_gradient_step(a, beta, lr=0.1)
            c_g = _fft_convolve(a_grad)
            M_g = float(c_g.max())
            if M_g < M * 0.999:
                a = a_grad
                if M_g < best_M:
                    best_M = M_g
                    best_score = float(2.0 * n * M_g)
                    best_a = a.copy()
                    no_imp = 0

        # Periodic reporting
        if verbose and iter_count - last_print > 2000:
            elapsed = time.monotonic() - (deadline - max_iter * 0.01)
            print(f"    LP iter {iter_count}: score={best_score:.10f}, M={best_M:.10f}, no_imp={no_imp}", flush=True)
            last_print = iter_count

        # Early termination if converged
        if best_M < 1e-15:
            break

    if verbose:
        print(f"  LP phase done: score={best_score:.10f} after {iter_count} iters", flush=True)

    return best_a


def _gradient_polish(a, deadline, max_iter=50000):
    """Fine gradient polish with decreasing beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    best_a = a.copy()

    beta = max(1e-8, 0.02 * best_M)
    lr = 0.05
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval, w = _smooth_max_weights(c, beta)
        grad = _grad_smooth_max(a, w)

        p = a
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= np.mean(gz)

        z = np.log(np.maximum(p, 1e-15))
        z -= np.mean(z)
        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        M = float(c.max())
        sc = float(2.0 * n * M)

        if sc < best_score - 1e-14:
            best_score = sc
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.95)
                beta = max(1e-12, beta * 0.8)

        if no_imp > max(5000, 10 * n):
            break

    return best_a


def _lp_final_pass(a, deadline):
    """Final LP pass with very tight constraints for last-mile improvement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    use_inflate = False
    inflate = 1.0 + 1e-6

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Try exact M first
        M_target = M
        b = _solve_lp_spread(a, M_target)

        if b is None:
            # Try with tiny inflation
            M_target = M * inflate
            b = _solve_lp_spread(a, M_target)

        if b is None:
            break

        sum_b = float(b.sum())
        if sum_b <= 1e-10:
            break

        b = b / sum_b
        c_b = _fft_convolve(b)
        M_b = float(c_b.max())

        if M_b < best_M * (1.0 - 1e-13):
            a = b.copy()
            best_M = M_b
            best_score = float(2.0 * n * M_b)
            best_a = a.copy()
            no_imp = 0
        else:
            # Try blend
            improved = False
            for t in [0.5, 0.3, 0.1]:
                a_try = (1.0 - t) * a + t * b
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                c_try = _fft_convolve(a_try)
                if float(c_try.max()) < best_M * (1.0 - 1e-13):
                    a = a_try
                    best_M = float(c_try.max())
                    best_score = float(2.0 * n * best_M)
                    best_a = a.copy()
                    no_imp = 0
                    improved = True
                    break
            if not improved:
                no_imp += 1
                if no_imp > 200:
                    break

    return best_a


def _multi_length_search(parent, deadline, best_score_current, rng):
    """Try optimizing at different lengths and interpolate back."""
    if parent is None:
        return None, np.inf

    best_a = None
    best_sc = best_score_current
    n_parent = parent.size

    # Try a few lengths
    for n_try in [n_parent]:
        if time.monotonic() > deadline:
            break
        a0 = parent.copy()
        a0 = a0 / a0.sum()
        sub_deadline = min(deadline, time.monotonic() + 120.0)
        a_opt = _lp_final_pass(a0, sub_deadline)
        sc = _score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: LP-driven with adaptive updates.
    
    Strategy:
    1. Start from parent sequence
    2. Phase 1: Aggressive LP refinement (majority of time)
    3. Phase 2: Gradient polish with decreasing smoothness
    4. Phase 3: Final tight LP pass
    5. Phase 4: Brief multi-perturbation restart
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.12f}, n={parent.size}", flush=True)
    except Exception as e:
        print(f"Could not load parent: {e}", flush=True)

    if parent is None:
        # Fallback: flat spectrum init
        n = 768
        parent = _make_flat_spectrum_init(n, rng)
        sc = _score(parent)
        best_score = sc
        best_seq = parent.copy()
        print(f"Generated init: score={sc:.12f}, n={n}", flush=True)

    a = parent.copy()

    # Phase 1: Main LP refinement (give ~70% of time)
    phase1_deadline = t_start + (budget_s - 15.0) * 0.65
    print(f"\nPhase 1: LP refinement, budget={phase1_deadline - time.monotonic():.0f}s", flush=True)
    a_lp = _lp_phase(a, phase1_deadline, max_iter=100000, verbose=True)
    sc_lp = _score(a_lp)
    print(f"  Phase 1 result: score={sc_lp:.12f}", flush=True)
    if sc_lp < best_score:
        best_score = sc_lp
        best_seq = a_lp.copy()
    a = a_lp

    # Phase 2: Gradient polish (~15% of time)
    if time.monotonic() < deadline:
        phase2_deadline = min(deadline - 60.0, time.monotonic() + (budget_s - 15.0) * 0.15)
        print(f"\nPhase 2: Gradient polish, budget={phase2_deadline - time.monotonic():.0f}s", flush=True)
        a_grad = _gradient_polish(a, phase2_deadline, max_iter=50000)
        sc_grad = _score(a_grad)
        print(f"  Phase 2 result: score={sc_grad:.12f}", flush=True)
        if sc_grad < best_score:
            best_score = sc_grad
            best_seq = a_grad.copy()
        a = a_grad

    # Phase 3: Final tight LP pass (~15% of time)
    if time.monotonic() < deadline - 20:
        phase3_deadline = min(deadline - 10.0, time.monotonic() + (budget_s - 15.0) * 0.15)
        print(f"\nPhase 3: Final LP pass, budget={phase3_deadline - time.monotonic():.0f}s", flush=True)
        a_final = _lp_final_pass(a, phase3_deadline)
        sc_final = _score(a_final)
        print(f"  Phase 3 result: score={sc_final:.12f}", flush=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        a = a_final

    # Phase 4: Perturbation restarts with local LP (~5% of time)
    if time.monotonic() < deadline - 10:
        phase4_deadline = deadline - 5.0
        remaining = phase4_deadline - time.monotonic()
        n_restarts = max(1, int(remaining / 30))
        print(f"\nPhase 4: {n_restarts} perturbation restarts, budget={remaining:.0f}s", flush=True)

        for r in range(n_restarts):
            if time.monotonic() > phase4_deadline:
                break
            noise_scale = 0.02 * (r + 1)
            a_pert = a * (1.0 + noise_scale * rng.standard_normal(len(a)))
            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / a_pert.sum()

            sc_pert = _score(a_pert)
            if sc_pert > best_score * 1.05:
                continue

            sub_dl = min(phase4_deadline, time.monotonic() + 25.0)
            a_opt = _lp_final_pass(a_pert, sub_dl)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"    Restart {r}: improved to {sc_opt:.12f}", flush=True)
            else:
                print(f"    Restart {r}: {sc_opt:.12f} (no improvement)", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = parent.copy()
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nAuthoritative eval: {sc_eval:.12f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.12f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
