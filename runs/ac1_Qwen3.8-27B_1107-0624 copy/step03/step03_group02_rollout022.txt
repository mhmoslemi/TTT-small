    b2 = 0.999

    m = np.zeros(n)

        z_new = np.clip(z_new, -50.0, 50.0)

        mag_new[0] = 1.0

        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s < 1e-10:
            a = best_a.copy()
            continue
        a_new = a_new / s
        a = a_new

        if it % 50 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a


def _lp_spread(a, M):
    """LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    """Iterative LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.08)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.92)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _subgradient_opt(a, deadline, max_iter=10000, top_k=3):
    """Multi-peak subgradient method."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.25 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        weights = np.array([1.0 + 0.5 * i for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            idx = int(idx)
            # Subgradient of c[idx] = sum_j a[j]*a[idx-j] w.r.t. a
            for j in range(n):
                j1 = idx - j
                if 0 <= j1 < 2 * n - 1:
                    pass  # already counted
            # Direct computation
            j1_start = max(0, idx - n + 1)
            j1_end = min(idx, n - 1)
            if j1_start <= j1_end:
                idx_arr = np.arange(j1_start, j1_end + 1)
                sg[idx_arr] += a[idx - idx_arr]
            j2_start = max(idx - (n - 1), 0)
            j2_end = min(idx, n - 1)
            if j2_start <= j2_end:
                idx_arr2 = np.arange(j2_start, j2_end + 1)
                sg[idx_arr2] += a[idx_arr2 - (idx - idx_arr2)]  # a[j2 - idx + j2]... 
                # Actually: d c[idx]/da[i] includes a[idx-i] and a[i-idx]
                # For the second term a[i-idx]: valid when 0 <= i-idx < n, i.e., idx <= i < idx+n
                # So for fixed i, the contribution is a[idx-i] (if valid) + a[i-idx] (if valid)
                # Let me redo: sg[i] += a[idx-i] if 0<=idx-i<n, plus a[i-idx] if 0<=i-idx<n
            # Simpler: sg[i] = a[idx-i] + a[i-idx] (both zero-padded)
            # Recompute properly:
            sg = np.zeros(n)
            for ii in range(n):
                val = 0.0
                j1 = idx - ii
                if 0 <= j1 < n:
                    val += a[j1]
                j2 = ii - idx
                if 0 <= j2 < n:
                    val += a[j2]
                sg[ii] = val
            # This is O(n) per peak, O(n*k) total - fine for n <= 2048
            break  # BUG: break out of the for i loop

        # Redo subgradient properly for all top peaks
        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            idx = int(idx)
            for ii in range(n):
                val = 0.0
                j1 = idx - ii
                if 0 <= j1 < n:
                    val += a[j1]
                j2 = ii - idx
                if 0 <= j2 < n:
                    val += a[j2]
                sg[ii] += weights[i] * val

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp_profile(n, direction=1):
    """Monotone ramp - key for asymmetric solutions."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, dtype=np.float64)
    return a / a.sum()


def _make_sigmoid_profile(n, sharpness=5.0):
    """Sigmoid-shaped profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + np.exp(-sharpness * (t - 0.5)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p=2.0):
    """Power-law profile."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 0.01
    return a / a.sum()


def _make_quadratic_profile(n):
    """Quadratic (parabolic) profile - asymmetric."""
    t = np.linspace(0.0, 1.0, n)
    a = (t ** 2) * 0.5 + t * 0.3 + 0.2
    return a / a.sum()


def _full_pipeline(a0, deadline, label=""):
    """Run full optimization pipeline."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: GS spectral shaping
    if time.monotonic() < deadline - 120:
        global deadline_global
        old_dl = deadline_global
        deadline_global = min(deadline - 120, time.monotonic() + 60)
        N = _next_pow2(2 * len(a0) - 1)
        a_gs = _gs_flat_spectrum(a0, N, n_iters=1000)
        deadline_global = old_dl
        s_gs = _score(a_gs)
        if s_gs < best_sc:
            best_a, best_sc = a_gs, s_gs
        print(f"  {label} GS: {best_sc:.10f}")

    # Phase 2: Adam smooth max
    if time.monotonic() < deadline - 80:
        d1 = min(deadline - 80, time.monotonic() + 200)
        a1, s1 = _adam_smooth_max(best_a, d1, max_iter=20000, lr=0.03, beta_factor=0.05)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 40:
        d2 = min(deadline - 40, time.monotonic() + 120)
        a2 = _lp_refine(best_a, d2, max_iter=1500, inflate=1.001, blend=0.3)
        s2 = _score(a2)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} LP: {best_sc:.10f}")

    # Phase 4: Fine Adam
    if time.monotonic() < deadline - 20:
        d3 = deadline - 20
        a3, s3 = _adam_smooth_max(best_a, d3, max_iter=15000, lr=0.01, beta_factor=0.02)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} Adam2: {best_sc:.10f}")

    # Phase 5: Subgradient polish
    if time.monotonic() < deadline - 5:
        d4 = deadline - 5
        a4, s4 = _subgradient_opt(best_a, d4, max_iter=5000, top_k=3)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} SubGrad: {best_sc:.10f}")

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: spectral shaping + Adam + LP spreading.
    Uses asymmetric initializations (ramps, sigmoids, power laws) to exploit
    the fact that asymmetric sequences can beat the symmetric bound of 2.
    Multi-scale: optimize at short lengths then interpolate up.
    """
    global deadline_global
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    deadline_global = deadline
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # === Phase A: Multi-scale optimization starting from short ===
    print("=== Phase A: Multi-scale from n=64 ===")
    if time.monotonic() < deadline - 400:
        # Start at n=64 with diverse asymmetric inits
        n_small = 64
        inits_small = [
            _make_ramp_profile(n_small, 1),
            _make_ramp_profile(n_small, -1),
            _make_sigmoid_profile(n_small, 5.0),
            _make_sigmoid_profile(n_small, 10.0),
            _make_power_profile(n_small, 3.0),
            _make_quadratic_profile(n_small),
        ]
        if parent is not None:
            inits_small.append(_interpolate_profile(parent, n_small))

        best_small = None
        for i, a0 in enumerate(inits_small):
            if time.monotonic() > deadline - 400:
                break
            per_time = 30.0
            d = min(deadline - 400, time.monotonic() + per_time)
            a_opt, s_opt = _adam_smooth_max(a0, d, max_iter=5000, lr=0.05, beta_factor=0.1)
            print(f"  A-{i}: n={n_small}, score={s_opt:.10f}")
            if best_small is None or s_opt < _score(best_small):
                best_small = a_opt

        # Interpolate to n=256 and refine
        if best_small is not None:
            a_256 = _interpolate_profile(best_small, 256)
            if time.monotonic() < deadline - 300:
                d = min(deadline - 300, time.monotonic() + 100)
                a_opt256, s_256 = _adam_smooth_max(a_256, d, max_iter=10000, lr=0.03, beta_factor=0.05)
                print(f"  A interp->256: score={s_256:.10f}")
                if s_256 < best_score:
                    best_score = s_256
                    best_seq = a_opt256.copy()

            # Interpolate to n=512
            a_512 = _interpolate_profile(a_256, 512)
            if time.monotonic() < deadline - 200:
                d = min(deadline - 200, time.monotonic() + 120)
                a_opt512, s_512 = _adam_smooth_max(a_512, d, max_iter=12000, lr=0.02, beta_factor=0.04)
                print(f"  A interp->512: score={s_512:.10f}")
                if s_512 < best_score:
                    best_score = s_512
                    best_seq = a_opt512.copy()

            # Interpolate to n=1024
            a_1024 = _interpolate_profile(a_512, 1024)
            if time.monotonic() < deadline - 100:
                d = min(deadline - 100, time.monotonic() + 150)
                a_opt1024, s_1024 = _adam_smooth_max(a_1024, d, max_iter=15000, lr=0.015, beta_factor=0.03)
                print(f"  A interp->1024: score={s_1024:.10f}")
                if s_1024 < best_score:
                    best_score = s_1024
                    best_seq = a_opt1024.copy()

    # === Phase B: Full pipeline on parent at n=1024 ===
    if parent is not None and time.monotonic() < deadline - 300:
        print("=== Phase B: Parent @1024 full pipeline ===")
        p1024 = _interpolate_profile(parent, 1024)
        d_b = min(deadline - 300, time.monotonic() + 350)
        a_b, s_b = _full_pipeline(p1024, d_b, "B")
        if s_b < best_score:
            best_score = s_b
            best_seq = a_b.copy()
        print(f"  Phase B best: {best_score:.10f}")

    # === Phase C: Diverse initializations at n=1024 ===
    if time.monotonic() < deadline - 150:
        print("=== Phase C: Diverse inits @1024 ===")
        n = 1024
        candidates = [
            _make_ramp_profile(n, 1),
            _make_sigmoid_profile(n, 3.0),
            _make_sigmoid_profile(n, 8.0),
            _make_power_profile(n, 2.0),
            _make_power_profile(n, 4.0),
            _make_quadratic_profile(n),
        ]
        # Perturbed parent
        if parent is not None:
            p1024 = _interpolate_profile(parent, n)
            pert = p1024 * (1.0 + 0.1 * rng.normal(size=n))
            pert = np.maximum(pert, 0.0)
            candidates.append(pert / pert.sum())
            # Also try flipped parent
            candidates.append(p1024[::-1] / p1024.sum())
            # Partial flip (first half flipped)
            half_flip = p1024.copy()
            half_flip[:n//2] = p1024[n//2-1::-1]
            half_flip = np.maximum(half_flip, 0)
            candidates.append(half_flip / half_flip.sum())

        # GS-initialized
        N_gs = _next_pow2(2 * n - 1)
        a_gs_init = _make_ramp_profile(n, 1)
        old_dl = deadline_global
        deadline_global = min(deadline - 150, time.monotonic() + 20)
        a_gs = _gs_flat_spectrum(a_gs_init, N_gs, n_iters=500)
        deadline_global = old_dl
        candidates.append(a_gs)

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 150:
                break
            per_time = min(25.0, (deadline - 150 - time.monotonic()) / max(1, len(candidates) - i))
            if per_time < 8:
                break
            d_c = time.monotonic() + per_time
            a_c, s_c = _adam_smooth_max(a0, d_c, max_iter=5000, lr=0.03, beta_factor=0.05)
            print(f"  C-{i}: score={s_c:.10f}")
            if s_c < best_score:
                best_score = s_c
                best_seq = a_c.copy()

    # === Phase D: Intensive final refinement ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"=== Phase D: Final refinement (best so far: {best_score:.10f}) ===")
        n_best = best_seq.size

        # Try to also work at n=2048 if time allows
        if time.monotonic() < deadline - 200 and n_best <= 1024:
            a_big = _interpolate_profile(best_seq, 2048)
            d_big = min(deadline - 200, time.monotonic() + 120)
            a_big_opt, s_big = _adam_smooth_max(a_big, d_big, max_iter=10000, lr=0.01, beta_factor=0.02)
            print(f"  D->2048: score={s_big:.10f}")
            if s_big < best_score:
                best_score = s_big
                best_seq = a_big_opt.copy()

        # Adam on current best
        if time.monotonic() < deadline - 60:
            d_d1 = min(deadline - 60, time.monotonic() + 100)
            a_d1, s_d1 = _adam_smooth_max(best_seq, d_d1, max_iter=10000, lr=0.008, beta_factor=0.01)
            if s_d1 < best_score:
                best_score = s_d1
                best_seq = a_d1.copy()
            print(f"  D Adam: {best_score:.10f}")

        # LP on current best
        if time.monotonic() < deadline - 30:
            d_d2 = min(deadline - 30, time.monotonic() + 80)
            a_d2 = _lp_refine(best_seq, d_d2, max_iter=1000, inflate=1.0005, blend=0.2)
            s_d2 = _score(a_d2)
            if s_d2 < best_score:
                best_score = s_d2
                best_seq = a_d2.copy()
            print(f"  D LP: {best_score:.10f}")

        # Final Adam fine-tune
        if time.monotonic() < deadline - 5:
            a_d3, s_d3 = _adam_smooth_max(best_seq, deadline - 5, max_iter=8000, lr=0.003, beta_factor=0.005)
            if s_d3 < best_score:
                best_score = s_d3
                best_seq = a_d3.copy()
            print(f"  D Final Adam: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp_profile(256, 1)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp_profile(256, 1)

    # Evaluate with authoritative function for real sequences
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
