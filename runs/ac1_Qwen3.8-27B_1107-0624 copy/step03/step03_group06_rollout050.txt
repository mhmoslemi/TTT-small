def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _solve_lp_max_sum(a, M, n):
    """Solve: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    a_rev = a[::-1]
    for j in range(n):
        A[j:j + n, j] = a_rev[j:]
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _solve_lp_min_max(a, n):
    """Solve: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0."""
    # variables: g_0,...,g_{n-1}, t
    # minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    # Inequality constraints: conv(a,g)_k - t <= 0 for each k
    A = np.zeros((2 * n - 1, n + 1))
    a_rev = a[::-1]
    for j in range(n):
        A[j:j + n, j] = a_rev[j:]
    A[:, -1] = -1.0  # -t

    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n]
    return None


def _line_search_golden(a, g, max_iter=30):
    """Golden section line search for best blend parameter t in [0,1]."""
    a = np.asarray(a, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()

    def obj(t):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return np.inf
        return _score(a_new / s)

    # First do a coarse grid to bracket the minimum
    n_grid = 11
    ts = np.linspace(0.01, 1.0, n_grid)
    vals = [obj(t) for t in ts]
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]

    # Refine with golden section around the best grid point
    if best_idx > 0 and best_idx < n_grid - 1:
        lo = ts[best_idx - 1]
        hi = ts[best_idx + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj(c)
        fd = obj(d)
        for _ in range(max_iter):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj(d)
            t_mid = (lo + hi) / 2.0
            fm = obj(t_mid)
            if fm < best_val:
                best_val = fm
                best_t = t_mid
                if abs(hi - lo) < 1e-6:
                    break

    return best_t, best_val


def _adam_state(n):
    return {
        'm': np.zeros(n),
        'v': np.zeros(n),
        't': 0
    }


def _adam_step(a, grad, state, lr):
    """One Adam step with simplex projection."""
    state['t'] += 1
    t = state['t']
    b1, b2 = 0.9, 0.999
    state['m'] = b1 * state['m'] + (1 - b1) * grad
    state['v'] = b2 * state['v'] + (1 - b2) * grad ** 2
    m_hat = state['m'] / (1 - b1 ** t)
    v_hat = state['v'] / (1 - b2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    return _project_simplex(a_new)


def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _soft_max_grad(a, beta):
    """Compute gradient of soft-max of conv(a,a) with temperature beta."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # Gradient: grad_i = sum_k w_k * a_{k-i}
    # This is a cross-correlation of w and a
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _lp_adam_refine(a0, deadline, max_iter=50000, use_min_max_lp=False):
    """Combined LP + Adam optimization loop."""
    a = _normalize(a0.copy())
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    # LP parameters
    inflate = 1.008
    inflate_min = 1.0008
    no_lp_imp = 0
    no_any_imp = 0

    # Adam parameters
    adam_state = _adam_state(n)
    beta = max(1e-5, 0.1 * float(_fft_convolve(a).max()))
    lr = 0.05
    base_lr = lr

    # Schedule: alternate between LP and Adam
    lp_every = 3  # do LP every 3 iterations
    adam_every = 1  # do Adam every iteration

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # --- LP Step ---
        if it % lp_every == 0 and no_lp_imp < 200:
            c = _fft_convolve(a)
            M_current = float(c.max())
            M_lp = M_current * inflate

            if use_min_max_lp:
                g = _solve_lp_min_max(a, n)
            else:
                g = _solve_lp_max_sum(a, M_lp, n)

            if g is not None:
                sum_g = float(g.sum())
                if sum_g > 1.0 + 1e-6:
                    g = g / sum_g
                    # Line search
                    t_best, sc_best = _line_search_golden(a, g, max_iter=15)
                    if t_best > 0.001 and sc_best < best_score - 1e-12:
                        a_new = (1.0 - t_best) * a + t_best * g
                        a_new = np.maximum(a_new, 0.0)
                        s_new = a_new.sum()
                        if s_new > 1e-10:
                            a_new = a_new / s_new
                            best_score = sc_best
                            best_a = a_new.copy()
                            a = a_new
                            no_lp_imp = 0
                            no_any_imp = 0
                            # Cool down
                            inflate = max(inflate_min, inflate * 0.97)
                            beta = max(1e-7, 0.1 * float(_fft_convolve(a).max()))
                            continue
                else:
                    no_lp_imp += 1

        if time.monotonic() > deadline:
            break

        # --- Adam Gradient Step ---
        if it % adam_every == 0:
            a_try = a.copy()
            grad, maxc = _soft_max_grad(a_try, beta)

            if not np.all(np.isfinite(grad)):
                beta = max(1e-5, beta * 3.0)
                continue

            a_new = _adam_step(a, grad, adam_state, lr)

            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                no_any_imp = 0
            else:
                no_any_imp += 1
                no_lp_imp += 1

            # Adapt learning rate
            if no_any_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)
            if no_any_imp > 2000:
                # Reset and try different direction
                a = best_a.copy()
                adam_state = _adam_state(n)
                lr = base_lr
                beta = max(1e-7, 0.1 * float(_fft_convolve(a).max()))
                no_any_imp = 0
                # Increase inflation to get different LP directions
                inflate = min(0.05, inflate * 1.5)

            # Slowly decrease beta (sharpen the max)
            if it % 50 == 49:
                beta = max(1e-8, beta * 0.85)

        # Inflate increase when stuck
        if no_lp_imp > 30:
            inflate = min(0.03, inflate * 1.2)
        if no_lp_imp > 500:
            # Major reset
            a = best_a.copy()
            adam_state = _adam_state(n)
            lr = base_lr
            beta = max(1e-7, 0.1 * float(_fft_convolve(a).max()))
            inflate = 1.008
            no_lp_imp = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-initialization LP + Adam optimizer for low-peak convolution sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations
    configs = []
    lengths = [256, 384, 512, 768, 1024]

    for n_t in lengths:
        if parent is not None:
            configs.append(('parent', n_t, _interpolate(parent, n_t)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 1.5)))
        configs.append(('ramp3', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('cos', n_t, _make_cosine(n_t)))
        configs.append(('exp_down', n_t, _make_exp(n_t, 2.0)))
        configs.append(('exp_up', n_t, _make_exp_up(n_t, 2.0)))
        configs.append(('bathtub', n_t, _make_bathtub(n_t, 0.3)))
        configs.append(('quad', n_t, _make_quadratic(n_t, 1.0, 1.0, 2.0)))

    # Score and sort
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        if np.isfinite(sc):
            scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Top 5 initializations:")
    for sc, name, n_t, _ in scored[:5]:
        print(f"  {name} n={n_t}: {sc:.10f}")

    # Phase 1: Optimize top initializations
    # Allocate time: give more time to better initializations
    n_configs = len(scored)
    time_per_config_base = (deadline - t_start) / max(n_configs, 1) * 0.7

    for idx, (sc0, name, n_t, a0) in enumerate(scored):
        if time.monotonic() > deadline - 200:
            break

        # Skip if initial score is much worse than current best
        if sc0 > best_score + 0.15 and idx > 3:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Time allocation: better configs get more time
        if sc0 < best_score + 0.03:
            alloc = min(300.0, time_per_config_base * 2.0)
        elif sc0 < best_score + 0.08:
            alloc = min(200.0, time_per_config_base * 1.5)
        else:
            alloc = min(100.0, time_per_config_base)

        remaining = deadline - time.monotonic() - 150
        alloc = min(alloc, max(10.0, remaining))

        if alloc < 15:
            break

        sub_deadline = time.monotonic() + alloc
        print(f"\nOptimizing {name} n={n_t} init={sc0:.8f} t={alloc:.0f}s")

        # Try both LP formulations
        use_mml = (idx % 2 == 0)  # alternate between formulations
        a_opt, sc_opt = _lp_adam_refine(a0, sub_deadline, max_iter=100000,
                                         use_min_max_lp=use_mml)
        sc_final = _score(a_opt)
        if np.isfinite(sc_final):
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_opt.copy()
                print(f"  -> NEW BEST: {sc_final:.10f} (LP {'minmax' if use_mml else 'maxsum'})")
            else:
                print(f"  -> score={sc_final:.10f} (best={best_score:.10f})")

    # Phase 2: Polish the best with a dedicated gradient pass
    if best_seq is not None and time.monotonic() < deadline - 30:
        polish_time = min(180.0, deadline - time.monotonic() - 20)
        if polish_time > 20:
            print(f"\nPolishing best (n={best_seq.size}): {polish_time:.0f}s")
            sub_dl = time.monotonic() + polish_time
            a_pol, sc_pol = _lp_adam_refine(best_seq, sub_dl, max_iter=100000,
                                             use_min_max_lp=False)
            sc_p = _score(a_pol)
            if np.isfinite(sc_p) and sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  -> Polished: {sc_p:.10f}")
            else:
                print(f"  -> Polished: {sc_p:.10f} (no improvement)")

    # Phase 3: Try a different length for the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        # Try a larger length
        n_try = min(n_best * 2, 2048)
        if n_try > n_best:
            a_interp = _interpolate(best_seq, n_try)
            sc_interp = _score(a_interp)
            print(f"\nTrying n={n_try} (interpolated from n={n_best}): {sc_interp:.10f}")
            remaining = deadline - time.monotonic() - 20
            if remaining > 30:
                sub_dl = time.monotonic() + min(120.0, remaining)
                a_new, sc_new = _lp_adam_refine(a_interp, sub_dl, max_iter=50000,
                                                 use_min_max_lp=True)
                sc_new = _score(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  -> NEW BEST at n={n_try}: {sc_new:.10f}")
                else:
                    print(f"  -> n={n_try}: {sc_new:.10f} (no improvement)")

    # Phase 4: Final quick polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            print(f"\nFinal polish: {remaining:.0f}s")
            a_f, sc_f = _lp_adam_refine(best_seq, time.monotonic() + remaining,
                                         max_iter=20000, use_min_max_lp=False)
            sc_f = _score(a_f)
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  -> Final: {sc_f:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(256, 1.0)
        best_score = _score(best_seq)

    # Final validation via evaluate_sequence
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(256, 1.0)

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
