"""
Optimized search for minimum-peak autocorrelation step functions.
Combines peak-cycling smooth-max Adam, aggressive LP spreading,
spectral initialization, and multi-length exploration to beat 1.5027.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_smooth_max(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a, computed via FFT."""
    n = len(a)
    a_rev = a[::-1].copy()
    N = _next_pow2(n + len(w))
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv_result = np.fft.irfft(fa * fw, N)
    g = 2.0 * conv_result[n - 1:2 * n - 1]
    return g


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.025, beta_factor=0.04,

    a = a / s

    m = np.zeros(n)

    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    cycle_idx = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_conv(a)
        peak_val = float(c_curr.max())

        # Peak cycling: use different beta ranges to target different peaks
       阶段 = it % (peak_cycle_k * 100)

def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        sg[idx] = 2.0 * a[k - idx]
    return sg


def _multi_subgradient(a, deadline, max_iter=12000, top_k=5, verbose=False):
    """Multi-peak subgradient with adaptive top_k."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    c = _fft_conv(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_conv(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        weights = np.exp(-0.5 * np.arange(k))
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

        if verbose and it % 5000 == 0:
            print(f"      MultiSG it={it}: best={best_score:.10f}")

    return best_a, best_score


def _lp_spread(a, M):
    """Solve LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: conv(a,b)[k] <= M for k=0,...,2n-2
    # conv(a,b)[k] = sum_j a[j]*b[k-j] (valid j, k-j)
    # For each k, the constraint is: sum_{j: 0<=k-j<n} a[j]*b[k-j] <= M
    # Let i = k-j, so j = k-i: sum_{i: 0<=k-i<n} a[k-i]*b[i] <= M
    # So coefficient of b[i] in constraint k is a[k-i] if 0<=k-i<n else 0
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # b[i] appears in constraint k with coefficient a[k-i] where 0<=k-i<n
        # k-i in [0, n-1] => k in [i, i+n-1]
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            A[k, i] = a[k - i]
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_conv(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_conv(a)
        M = float(c_curr.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.06)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_conv(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.92)
            if no_imp > max(150, n // 4):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exp_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * (t - 0.5))
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bell_profile(n, width=0.8):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_profile(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 1e-10
    return a / a.sum()


def _spectral_init(n, seed=0):
    """Initialize via spectral method: design a power spectrum, take IFFT, clip."""
    rng = np.random.default_rng(seed)
    # Target: roughly flat power spectrum with random phase
    N = n
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    # Roughly flat magnitude (with slight taper to ensure non-negativity after IFFT)
    freqs = np.fft.rfftfreq(N)
    # Taper: reduce high frequencies to help non-negativity
    taper = 1.0 / (1.0 + 10.0 * (freqs / 0.5) ** 4)
    mag = np.sqrt(taper)
    spec = mag * np.exp(1j * phase)
    # Ensure DC is real and positive
    spec[0] = abs(spec[0]) + 2.0
    # IFFT
    a = np.fft.irfft(spec, N)
    # Enforce non-negativity
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

                                        lr=0.005, beta_factor=0.008)
            if s_u < best_score:
                best_score = s_u
                best_seq = a_u.copy()
            print(f"  Ultra Adam: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    # Final validation and cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
