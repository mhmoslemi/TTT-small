import numpy as np
import time

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1."""
    n = len(a)
    ar = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    har = np.fft.rfft(ar, N)
    hw = np.fft.rfft(w, N)
    conv = np.fft.irfft(har * hw, N)
    return conv[n-1:2*n-1]

def softmax_weights(c, beta):
    """Compute softmax weights for the max approximation."""
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-10
        w = np.zeros_like(c)
        w[mask] = 1.0 / np.count_nonzero(mask)
        return w
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = ex.sum()
    return ex / s

def nesterov_optimize(a0, deadline, rng, lr=0.05, beta_init=0.05, 
                       beta_decay=0.95, momentum=0.9, anneal_start=0.3):
    """Nesterov-accelerated gradient descent with adaptive softmax and simulated annealing."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    z = np.log(a0)
    z = z - z.mean()
    
    c = full_convolve_fft(a0)
    max_c = c.max()
    beta = beta_init * max_c
    
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    
    # Nesterov momentum
    z_prev = z.copy()
    v = np.zeros(n)  # velocity
    mu = momentum  # momentum parameter
    
    # Annealing
    T = 1.0
    T_min = 0.01
    total_time = deadline - time.monotonic()
    
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Nesterov look-ahead
        z_new = z + mu * (z - z_prev)
        a_tmp = np.exp(z_new - z_new.max())
        a_tmp = a_tmp / a_tmp.sum()
        
        # Compute gradient at look-ahead point
        c_tmp = full_convolve_fft(a_tmp)
        w = softmax_weights(c_tmp, beta)
        h = compute_h(w, a_tmp)
        grad = 2.0 * h
        dot = float(np.dot(a_tmp, grad))
        dz = a_tmp * (grad - dot)
        
        # Update
        z = z - lr * dz
        z = np.clip(z, -40, 40)
        z = z - z.mean()
        
        # Simulated annealing kick
        elapsed_frac = (time.monotonic() - (deadline - total_time)) / total_time
        T = max(T_min, T * (1 - 0.001))
        if rng.random() < 0.05 * (1 + elapsed_frac):
            kick = rng.normal(size=n) * 0.1 * T * (1 + no_improve / 100)
            z = z + kick
            z = np.clip(z, -40, 40)
            z = z - z.mean()
        
        z_prev = z
        
        # Compute new a and score
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            continue
        a0 = a0 / s
        
        c = full_convolve_fft(a0)
        max_c = c.max()
        score = 2.0 * n * max_c
        
        if score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt beta
        if no_improve % 50 == 0:
            beta = max(1e-10, beta * beta_decay)
        
        # Restart if stuck
        if no_improve > 500:

            z = z - z.mean()
            z_prev = z.copy()
            no_improve = 0
            beta = max(1e-10, beta_init * 2.0 * c.max())
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines Nesterov-accelerated gradient descent in logit space with adaptive
softmax, simulated annealing kicks, multi-scale n search, and local refinement.
Key innovations: symmetrized gradient, entropy-regularized softmax ensemble,
and adaptive restart with neighborhood exploration.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_{m=0}^{n-1} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    """Compute softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def make_linear_profile(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (1.0 - 2.0 * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_dual_ramp(n, alpha=0.5):
    """Dual asymmetric ramp: high at one end, medium at other."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.sin(np.pi * t) + 0.3 * (1.0 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_polynomial_profile(n, deg=3):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, deg + 1):
        a = a + (0.5 / k) * (2.0 * t - 1.0) ** k
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_asymmetric(n, width=0.4, offset=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 0.3 + np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def nesterov_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.06,
                           beta_decay=0.94, momentum=0.9, b1=0.9, b2=0.999,
                           anneal_prob=0.03, restart_period=600):
    """
    Nesterov-accelerated Adam in logit space with adaptive softmax
    and simulated annealing kicks for exploration.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()
    z_prev = z.copy()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    beta = beta_factor * max_c
    if beta < 1e-10:
        beta = 1e-6

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    eps = 1e-8

    best_a = a0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_improve = 0
    current_lr = lr
    base_lr = lr
    total_time = deadline - time.monotonic()
    t_start_local = time.monotonic()

    while time.monotonic() < deadline:
        # Nesterov look-ahead
        z_look = z + momentum * (z - z_prev)
        a_look = np.exp(z_look - z_look.max())
        s_look = a_look.sum()
        if s_look < 1e-15:
            a_look = a0
        else:
            a_look = a_look / s_look

        # Compute gradient at look-ahead
        c_look = full_convolve_fft(a_look)
        w = soft_max_weights(c_look, beta)
        h = compute_h(w, a_look)
        grad_a = 2.0 * h
        dot = float(np.dot(a_look, grad_a))
        dz = a_look * (grad_a - dot)

        # Adam update
        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * dz
        v_adam = b2 * v_adam + (1.0 - b2) * (dz * dz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Annealing kick (probability increases when stuck)
        elapsed_frac = min(1.0, (time.monotonic() - t_start_local) / max(1e-6, total_time))
        kick_prob = anneal_prob * (1.0 + no_improve / 200.0)
        if rng.random() < min(kick_prob, 0.3):
            kick_size = 0.05 + 0.1 * no_improve / max(1, restart_period)
            z = z + kick_size * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

        z = np.clip(z, -45.0, 45.0)
        z = z - z.mean()
        z_prev = z + momentum * (z - z_prev)  # Update look-ahead base
        z_prev = z.copy()  # Reset for next step

        # Compute new a
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15 or not np.isfinite(s):
            z = best_z.copy()
            z_prev = z.copy()
            a0 = best_a.copy()
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive beta (sharpen)
        if t_adam % 30 == 0:
            beta = max(1e-10, beta * beta_decay)

        # Adaptive learning rate
        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.98)

        # Random restart
        if no_improve > restart_period:
            noise = 0.15 + 0.1 * rng.random()
            z = best_z + noise * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            z_prev = z.copy()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(1e-8, beta_factor * 1.5 * max_c)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def ensemble_softmax_optimize(a0, deadline, rng, lr=0.03, n_ensemble=3):

def local_search_refine(a0, budget_iters=5000, rng=None):
    """
    Coordinate-wise local search: try small perturbations of individual
    coordinates and accept if score improves.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_score = compute_score(a0)

    for _ in range(budget_iters):
        # Pick a random coordinate
        i = rng.integers(0, n)
        # Try a small multiplicative perturbation
        for factor in [0.95, 0.9, 1.05, 1.1]:
            a_test = a0.copy()
            a_test[i] *= factor
            if a_test[i] < 0:

            if s < 0.01:
                continue
            a_test = a_test / s
            sc = compute_score(a_test)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_test.copy()
                a0 = a_test.copy()
                break

    return best_a, best_score


def multiplicative_update_optimize(a0, deadline, rng, lr=0.05, beta_factor=0.05):
    """
    Multiplicative (Fisher) update method: a[i] *= exp(-eta * gradient[i]).
    This naturally preserves non-negativity and can be more stable.
    """

    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    beta = beta_factor * max_c

    best_a = a0.copy()
    best_score = 2.0 * n * max_c
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad = 2.0 * h

        # Multiplicative update in log-space
        log_a = np.log(a0)
        log_a = log_a - current_lr * (grad - np.mean(grad))
        log_a = np.clip(log_a, -30.0, 30.0)
        log_a = log_a - log_a.mean()

        a0 = np.exp(log_a - log_a.max())
        s = a0.sum()
        if s < 1e-15:
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 40 == 0:
            beta = max(1e-10, beta * 0.93)
            current_lr = max(0.005, current_lr * 0.98)

        if no_improve > 1000:
            a0 = best_a * (1.0 + 0.1 * rng.normal(size=n))
            a0 = np.maximum(a0, 1e-15)
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())

    return best_a, best_score
