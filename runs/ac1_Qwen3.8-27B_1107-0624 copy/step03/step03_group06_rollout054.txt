def _lp_refine(a0, deadline, max_iter=30000, inflate_start=1.003, inflate_min=1.0005):

    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    inflate = inflate_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        if g is None:

            if no_imp > 200:
                break
            continue

        best_t = 0.0
        best_sc = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

def _make_asym_init(n, alpha=2.0, beta=0.5):
    """Create asymmetric initialization."""
    t = np.linspace(0.0, 1.0, n)
    # Skewed distribution
    a = (1.0 - t) ** alpha * t ** beta + 1e-8
    return a / a.sum()

def _make_exp_init(n, rate=3.0):
    """Exponential decay initialization."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t) + 1e-8
    return a / a.sum()

def _make_beta_dist(n, alpha=2.0, beta=5.0):
    """Beta distribution initialization."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** (alpha - 1) * (1 - t) ** (beta - 1) + 1e-8
    return a / a.sum()

def _convolution_equalize(a0, deadline, max_iter=20000):
    """Iterative method to equalize the convolution."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    
    lr = 0.1
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mean_c = c.mean()
        # Deficit: how much each position is above mean
        excess = np.maximum(c - mean_c, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        
        # For each a[j], compute how much it "contributes" to the excess
        # a[j] contributes to c[k] for k in [j, j+n-1] with weight a[k-j]
        # Contribution to excess = sum_k excess[k] * a[k-j]
        # This is the cross-correlation of excess and a
        
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        fe = np.fft.rfft(excess, N)
        contrib = np.fft.irfft(np.conj(fa) * fe, N)[:n]
        
        # Update: reduce a[j] proportionally to its contribution
        # a[j] <- a[j] * (1 - lr * contrib[j] / (a[j] + eps))
        # But this might make some entries negative, so use exponential:
        # a[j] <- a[j] * exp(-lr * contrib[j])
        
        update = 1.0 - lr * contrib / (contrib.max() + 1e-15)
        update = np.clip(update, 0.5, 1.5)
        a_new = a * update
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > 2000:
                break
        
        if it % 200 == 199:
            lr = min(0.3, lr * 1.02)
    
    return best_a, best_score

def _coordinate_descent(a0, deadline, max_sweeps=1000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            # Current contribution of a[j] to c:
            # c[k] += a[j] * a[k-j] for k in [j, min(2n-2, j+n-1)]
            # If we change a[j] to a[j] + delta:
            # c[k] += delta * a[k-j] for k in [j, min(2n-2, j+n-1)]
            
            k_lo = j
            k_hi = min(2*n-2, j+n-1)
            
            # Current c values in this range
            c_range = c[k_lo:k_hi+1].copy()
            # a[k-j] for k in [k_lo, k_hi]: a[0], a[1], ..., a[min(n-1, k_hi-j)]
            delta_effects = a[:k_hi-k_lo+1]
            
            # New c_range = c_range + delta * delta_effects
            # We want to minimize max(c_range + delta * delta_effects) - (c_range + delta * delta_effects)[j related to sum]
            # Actually we want to minimize max(c) overall after the update
            # But the max might be outside this range, so:
            # global_max_new = max(max(c_range + delta * delta_effects), max(c outside range))
            # max outside doesn't change with delta
            
            outside_max = float(np.max(np.delete(c, range(k_lo, k_hi+1)))) if (2*n-1) > (k_hi-k_lo+1) else 0.0

            # subject to a[j] + delta >= 0, i.e., delta >= -a[j]

            # The optimum is where the largest few values are balanced
            
            # Use ternary search on delta in [-a[j], some_upper_bound]
            # Upper bound: we don't want to increase c too much
            # A reasonable upper bound: a[j] (doubling)
            
            lo = -a[j]
            hi = a[j] * 0.5  # don't increase by more than 50% in one step
            
            if hi <= lo:
                continue
            
            best_delta = 0.0
            best_val = float(np.max(c))
            
            for _ in range(30):  # 30 iterations of golden section
                mid1 = lo + 0.382 * (hi - lo)
                mid2 = lo + 0.618 * (hi - lo)
                
                val1 = c_range + mid1 * delta_effects
                v1 = max(float(np.max(val1)), outside_max)
                
                val2 = c_range + mid2 * delta_effects
                v2 = max(float(np.max(val2)), outside_max)

                    hi = mid2
                else:
                    lo = mid1
                
                # Track best

                vmid = max(float(np.max(val_mid)), outside_max)
                if vmid < best_val:
                    best_val = vmid

                a[j] = max(a[j], 0.0)

        # Renormalize

        if s > 0:

    return best_a, best_score

"""
Combined coordinate descent + LP + gradient optimization for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses asymmetric sequences and coordinate-wise
exact optimization to escape flat regions where gradient methods struggle.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _lp_solve(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            A[k, m] = f[k - m]
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr_full = np.fft.irfft(np.conj(fa) * fw, N)
    grad = 2.0 * corr_full[:n]
    return grad, mval


def _project_nonneg_sum1(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return v / s


def _coord_descent(a0, deadline, max_sweeps=5000):
    """Coordinate descent: for each j, find optimal a[j] minimizing max(c)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        for j in range(n):
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            length = k_hi - k_lo + 1
            c_range = c[k_lo:k_hi + 1]
            delta_eff = a[:length]

            # Find outside max (max of c outside [k_lo, k_hi])
            if k_lo > 0:
                outside = max(float(c[:k_lo].max()),
                              float(c[k_hi + 1:].max()) if k_hi + 1 < 2 * n - 1 else 0.0)
            else:
                outside = float(c[k_hi + 1:].max()) if k_hi + 1 < 2 * n - 1 else 0.0

            # Golden section search for optimal delta in [-a[j], a[j]]
            lo = -a[j] * 0.99
            hi = a[j] * 0.5

            if hi <= lo:
                continue

            inv_phi = 0.618
            d1 = lo + (1 - inv_phi) * (hi - lo)
            d2 = lo + inv_phi * (hi - lo)
            f1 = max(float(np.max(c_range + d1 * delta_eff)), outside)
            f2 = max(float(np.max(c_range + d2 * delta_eff)), outside)

            for _ in range(40):
                if f1 < f2:
                    hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = lo + (1 - inv_phi) * (hi - lo)
                    f1 = max(float(np.max(c_range + d1 * delta_eff)), outside)
                else:
                    lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = lo + inv_phi * (hi - lo)
                    f2 = max(float(np.max(c_range + d2 * delta_eff)), outside)

            best_delta = (lo + hi) / 2.0
            a[j] = max(a[j] + best_delta, 0.0)
            c[k_lo:k_hi + 1] += best_delta * delta_eff

        # Renormalize after sweep
        s = a.sum()
        if s > 1e-10:
            a = a / s
            c = _fft_convolve(a)

        score = float(2.0 * n * c.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def _lp_refine(a0, deadline, max_iter=30000, inflate_start=1.003, inflate_min=1.0005):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_solve(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 150 == 0:
                inflate = min(0.02, inflate * 1.2)
            if no_imp > 300:
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate_min >= 1.001:
                    break
                continue

        g = g / sum_g

        best_t = 0.0
        best_sc = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = _fft_convolve(a_new)
            sc = float(2.0 * n * c_new.max())
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 100 == 99:
                inflate = max(inflate_min, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp > 400:
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0

    return best_a, best_score


def _gradient_polish(a0, deadline, max_iter=50000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    beta = 0.02 * float(c.max())
    beta_min = 1e-8 * float(c.max())
    lr = 0.05
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    patience = 3000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_mom = a + momentum * v
        a_mom = np.maximum(a_mom, 0)
        if a_mom.sum() < 1e-10:
            a_mom = a

        grad, mval = _gradient_smooth_max(a_mom, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(beta_min, beta * 2)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_nonneg_sum1(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > patience:
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(beta_min, 0.02 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
                patience = max(500, patience // 2)

        if it % 200 == 199:
            beta = max(beta_min, beta * 0.85)

    return best_a, best_score


def _make_exp_init(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t) + 1e-10
    return a / a.sum()


def _make_beta_init(n, alpha, beta_p):
    t = np.linspace(0.0, 1.0, n)
    a = t ** (alpha - 0.5) * (1 - t) ** (beta_p - 0.5) + 1e-10
    return a / a.sum()


def _make_asym(n, asym=0.3):
    """Create sequence with controlled asymmetry."""
    t = np.linspace(0.0, 1.0, n)
    base = np.exp(-0.5 * ((t - 0.5) / 0.3) ** 2)
    skew = 1.0 + asym * (1.0 - 2.0 * t)
    a = base * skew
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to(parent, n_new):
    n0 = len(parent)
    if n0 == n_new:
        return _normalize(parent.copy())
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    a = np.interp(x1, x0, parent)
    a = np.maximum(a, 0.0)
    return _normalize(a)
