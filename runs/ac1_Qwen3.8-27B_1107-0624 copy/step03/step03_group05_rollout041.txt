Peak-equalizing optimization using QP subproblems with Hankel structure,
Nesterov-accelerated projected subgradient, simulated annealing, and Adam polish.
Key innovation: Convex QP subproblems exploiting Hankel matrix structure (W[i,j]=w[i+j])
solved via FFT-based matrix-vector products with simplex projection.

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _hankel_mv(w, b):
    """Compute H*b where H[i,j] = w[i+j] for 0<=i,j<n and w has length 2n-1."""
    n = len(b)
    # d[k] = w[2n-2-k]
    d = w[::-1]
    # (H*b)[i] = conv(d, b)[2n-2-i]
    # Full conv length: (2n-1) + n - 1 = 3n-2
    N = _next_pow2(3 * n - 2)
    fd = np.fft.rfft(d, N)
    fb = np.fft.rfft(b, N)
    conv_full = np.fft.irfft(fd * fb, N)[:3*n-2]
    # We need conv_full[n-1 : 2n-1] in reverse
    result = conv_full[n-1:2*n-1][::-1]
    # Clean up numerical noise
    mn = result.min()
    if mn < 0 and mn > -1e-10:
        result = result - mn
    return result

def _solve_qp_hankel(w, max_iter=200, lr=None):
    """
    Solve: min b^T H b s.t. sum(b)=1, b>=0
    where H[i,j] = w[i+j] (Hankel matrix).
    Uses projected gradient on the simplex.
    """
    n = (len(w) + 1) // 2  # n such that len(w) = 2n-1
    b = np.ones(n) / n
    
    # Estimate Lipschitz constant: L = 2 * lambda_max(H) <= 2 * max_row_sum
    # For Hankel: row i has sum = sum_{j=0}^{n-1} w[i+j]
    row_sums = np.array([w[i:i+n].sum() for i in range(n)])
    L = 2.0 * float(row_sums.max())
    if lr is None:
        lr = 1.0 / max(L, 1e-10)
    
    best_b = b.copy()
    best_val = float(b @ _hankel_mv(w, b))
    
    for it in range(max_iter):
        grad = 2.0 * _hankel_mv(w, b)
        b_new = _project_simplex(b - lr * grad)
        val = float(b_new @ _hankel_mv(w, b_new))
        if val < best_val:
            best_val = val
            best_b = b_new.copy()
        b = b_new
        # Decay step size slightly for convergence
        lr *= 0.999
    
    return best_b

"""
Convex optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key insight: The objective is CONVEX (max of convex quadratics on a convex set).
Uses FISTA with smooth-max approximation, Hankel-structure QP subproblems,
LP-based spreading, and multiple diverse initializations with adaptive beta.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad_fft(a, beta):
    """
    Compute gradient of smooth max of conv(a,a) using FFT.
    smooth_max = (1/beta) * log(sum exp(beta * c[k]))
    grad[i] = 2 * sum_k p[k] * a[k-i] where p is softmax(beta*c)
    Returns (gradient, max_c, c_array)
    """
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    p = ex / s

    # grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * conv(p, a_flipped)[n-1+i]
    a_flipped = a[::-1]
    N = _next_pow2(3 * n - 2)
    fp = np.fft.rfft(p, N)
    faf = np.fft.rfft(a_flipped, N)
    conv_full = np.fft.irfft(fp * faf, N)[:3 * n - 2]
    r = conv_full[n - 1:2 * n - 1]
    grad = 2.0 * r

    # Clean numerical noise
    mn = grad.min()
    if mn < 0 and mn > -1e-12:
        grad = grad - mn

    return grad, mval, c


def _fista_smooth_max(a_init, deadline, max_iter=50000, beta_init=None):
    """
    FISTA (accelerated proximal gradient) on the smooth max approximation.
    Minimizes smooth_max(conv(a,a), beta) over the simplex.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    c = _fft_convolve(a)
    max_c0 = float(c.max())
    if beta_init is None:
        beta = max(1e-6, 0.1 / max(max_c0, 1e-10))

    best_a = a.copy()
    best_score = float(2.0 * n * max_c0)

    x = a.copy()
    y = a.copy()
    t = 1.0
    lr = None

    # Estimate Lipschitz constant
    # L ≈ 2 * max_i sum_k p[k] * a[k-i] bound ~ 2 * sum_k p[k] * max(a) * n
    # Simpler: use backtracking
    if lr is None:
        lr = 1.0 / (4.0 * n * max_c0 + 1e-10)

    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, max_c, c_arr = _smooth_max_grad_fft(y, beta)

        if not np.all(np.isfinite(grad)):
            beta *= 2.0
            y = x.copy()
            t = 1.0
            continue

        # Backtracking line search for Lipschitz
        # For FISTA, we use a fixed step with periodic adjustment
        x_new = _project_simplex(y - lr * grad)

        # Evaluate
        c_new = _fft_convolve(x_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = x_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # FISTA update
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _project_simplex(y)
        x = x_new
        t = t_new

        # Adaptive step size
        if no_imp % 200 == 0:
            lr = max(1e-8, lr * 0.97)
        if no_imp > 3000:
            x = best_a.copy()
            y = best_a.copy()
            t = 1.0
            lr = base_lr
            beta = max(1e-6, 0.1 / max(float(_fft_convolve(x).max()), 1e-10))
            no_imp = 0

        # Periodically tighten beta
        if it % 500 == 0 and it > 0:
            beta = max(1e-8, beta * 0.85)

    return best_a, best_score


def _hankel_mv(w, b):
    """
    Compute H*b where H[i,j] = w[i+j] for 0<=i,j<n, w has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(b)
    d = w[::-1]  # d[k] = w[2n-2-k]
    # (H*b)[i] = conv(d, b)[2n-2-i] for i=0,...,n-1
    N = _next_pow2(3 * n - 2)
    fd = np.fft.rfft(d, N)
    fb = np.fft.rfft(b, N)
    conv_full = np.fft.irfft(fd * fb, N)[:3 * n - 2]
    result = conv_full[n - 1:2 * n - 1][::-1]
    mn = result.min()
    if mn < 0 and mn > -1e-10:
        result = result - mn
    return result


def _solve_qp_hankel(w, max_iter=200):
    """
    Solve: min b^T H b s.t. sum(b)=1, b>=0
    where H[i,j] = w[i+j] (Hankel matrix, PSD if w is PSD sequence).
    Uses projected gradient on simplex with adaptive step.
    """
    n = (len(w) + 1) // 2
    b = np.ones(n) / n

    # Estimate step size
    row_sums = np.array([w[i:i + n].sum() for i in range(n)])
    L = 2.0 * float(row_sums.max()) + 1e-12
    lr = 1.0 / L

    best_b = b.copy()
    h = _hankel_mv(w, b)
    best_val = float(b @ h)

    no_imp = 0
    for it in range(max_iter):
        grad = 2.0 * h
        b_new = _project_simplex(b - lr * grad)
        h_new = _hankel_mv(w, b_new)
        val = float(b_new @ h_new)

        if val < best_val - 1e-14:
            best_val = val
            best_b = b_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                lr *= 0.95
            if no_imp > 300:
                break

        b = b_new
        h = h_new

    return best_b


def _qp_refine(a_init, deadline, max_outer=30, qp_iter=150):
    """
    Iterative QP-based peak reduction.
    At each outer step: compute weights from current peaks, solve QP, blend.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        if M < 1e-15:
            break

        # Weight peaks: w[k] proportional to (c[k]/M)^2
        w = (c / M) ** 2
        w = w / w.sum() * (2 * n - 1)  # normalize so mean weight is 1

        # Solve QP
        b = _solve_qp_hankel(w, max_iter=qp_iter)

        # Try multiple blending factors
        improved = False
        for t_blend in [0.3, 0.2, 0.1, 0.5, 0.15]:
            a_new = (1.0 - t_blend) * a + t_blend * b
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = _score(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    from scipy.optimize import linprog
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def _lp_refine(a_init, deadline, max_iter=5000, inflate=1.002, blend_init=0.25):
    """LP-based refinement with adaptive inflation."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend = blend_init
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.03)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.93)
                inflate_cur = min(1.01, inflate_cur * 0.98)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend = blend_init
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _adam_logit_polish(a_init, deadline, max_iter=20000, lr=0.04):
    """Adam optimization in logit space for fine-tuning."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-6, 0.05 * float(c.max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, max_c, c_arr = _smooth_max_grad_fft(a, beta)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.97)

        if it % 200 == 0:
            beta = max(1e-8, beta * 0.8)

        if no_imp > max(2000, 3 * n):
            rng = np.random.default_rng(it * 7919 + n)
            z = 0.5 * best_z + 0.5 * rng.normal(size=n) * 0.3
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = 0.04
            no_imp = 0

    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, gamma=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(gamma * t)
    return a / a.sum()


def _make_sine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, beta=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + beta * t ** 2
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _subgradient_polish(a_init, deadline, max_iter=30000):
    """
    Projected subgradient with heavy averaging (Robbins-Monro).
    For the convex problem: min max_k a^T Q_k a s.t. a in simplex.
    Uses uniform averaging over top-k peaks for stability.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _score(a)

    # Heavy average iterate
    x_bar = a / 1.0
    T = 0

    lr_scale = 1.0 / (4.0 * n * float(_fft_convolve(a).max()) + 1e-10)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        lr = lr_scale / math.sqrt(float(T) + 1.0)

        c = _fft_convolve(a)
        # Use top-5 peaks for averaged subgradient
        top_k = min(5, 2 * n - 1)
        sorted_idx = np.argsort(c)[::-1][:top_k]
        peak_vals = c[sorted_idx]
        weights = peak_vals / peak_vals.sum()

        grad = np.zeros(n)
        for idx, w in zip(sorted_idx, weights):
            k = int(idx)
            valid_i = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
            grad[valid_i] += 2.0 * w * a[k - valid_i]

        a_new = _project_simplex(a - lr * grad)

        # Heavy average
        T += 1
        x_bar = ((T - 1) * x_bar + a_new) / T

        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

        # Also check the averaged iterate
        x_bar_proj = _project_simplex(x_bar)
        c_bar = _fft_convolve(x_bar_proj)
        sc_bar = float(2.0 * n * c_bar.max())

        if sc_bar < best_score - 1e-13:
            best_score = sc_bar
            best_a = x_bar_proj.copy()
            no_imp = 0
        elif sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        if no_imp > 5000:
            break

    return best_a, best_score


def _multi_phase_optimize(a_init, deadline):
    """Run multiple optimization phases on a single initialization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    phases = [
        ("FISTA", _fista_smooth_max),
        ("QP", _qp_refine),
        ("Subgrad", _subgradient_polish),
        ("LP", _lp_refine),
        ("Adam", _adam_logit_polish),
    ]

    # Allocate time: give more to FISTA and QP, less to others
    weights = [0.3, 0.2, 0.15, 0.15, 0.2]

    total_time = deadline - time.monotonic()
    for i, (name, func) in enumerate(phases):
        if time.monotonic() > deadline - 5:
            break

        phase_deadline = time.monotonic() + total_time * weights[i]
        phase_deadline = min(phase_deadline, deadline - 5)

        if phase_deadline - time.monotonic() < 5:
            continue

        print(f"      {name}: starting (budget={phase_deadline - time.monotonic():.0f}s)")

        try:
            result_a, result_score = func(best_a, phase_deadline)
            if result_score < best_score:
                best_score = result_score
                best_a = result_a.copy()
                print(f"      {name}: improved to {best_score:.10f}")
            else:
                print(f"      {name}: no improvement ({result_score:.10f})")
        except Exception as e:
            print(f"      {name}: error: {e}")

    # Second round if time remains
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"      Round 2: budget={remaining:.0f}s")
        for name, func in [("FISTA2", _fista_smooth_max), ("QP2", _qp_refine), ("Adam2", _adam_logit_polish)]:
            if time.monotonic() > deadline - 10:
                break
            phase_dl = min(time.monotonic() + remaining * 0.3, deadline - 5)
            if phase_dl - time.monotonic() < 5:
                continue
            try:
                result_a, result_score = func(best_a, phase_dl)
                if result_score < best_score:
                    best_score = result_score
                    best_a = result_a.copy()
                    print(f"      {name}: improved to {best_score:.10f}")
            except Exception:
                pass

    return best_a, best_score
