"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines: (1) adaptive smooth-max subgradient with Nesterov momentum,
(2) periodic LP-based spreading directions, (3) top-peak-focused gradient,

Key improvements over previous: faster inner loop via top-k peak tracking,

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _conv_gradient_smooth(a, w):
    """Gradient of sum_m w_m * (conv(a,a))_m w.r.t. a."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _top_k_gradient(a, c, top_k=8):
    """Gradient focusing on top-k peaks of the convolution."""
    n = len(a)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    peak_vals = c[sorted_idx]
    w = np.zeros_like(c)
    w[sorted_idx] = peak_vals ** 3
    s = w.sum()
    if s < 1e-30:
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    w = w / s
    return _conv_gradient_smooth(a, w)


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, gap=0.3, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - (0.5 - gap)) / width) ** 2) + \
        np.exp(-0.5 * ((t - (0.5 + gap)) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric_profile(n, skew=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + skew * (t - 0.5) ** 2 + 0.5 * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _greedy_spread(a, n_steps=50, lr=0.02):
    """Cheap greedy: move mass from high-gradient to low-gradient coordinates."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)
    rng = np.random.default_rng(42)
    
    best_a = a.copy()
    best_score = _score(a)
    
    for _ in range(n_steps):
        c = _fft_convolve(a)
        M = float(c.max())
        beta = max(1e-8, 0.02 * M)
        w = _smooth_max_weight(c, beta)
        grad = _conv_gradient_smooth(a, w)
        
        # Projected gradient step
        a_new = _project_simplex(a - lr * grad)
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try coordinate swap
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                break
            delta = 0.1 * a[i]
            if delta <= 0:
                break
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < best_score - 1e-12:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try
            else:
                break
    
    return best_a, best_score


def _nesterov_sentence(a, deadline, max_iter=50000):
    """Nesterov-accelerated projected subgradient with adaptive beta and top-k focus."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-8, 0.03 * float(c.max()))
    lr = 0.04
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        s_look = a_look.sum()
        if s_look < 1e-10:
            a_look = a
        else:
            a_look = a_look / s_look
        
        c_look = _fft_convolve(a_look)
        M = float(c_look.max())
        
        # Adaptive top-k
        top_k = min(n, max(3, int(5 * np.exp(-it / 2000.0))))
        grad = _top_k_gradient(a_look, c_look, top_k=top_k)
        
        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.003, lr * 0.97)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                no_imp = 0
        
        # Adaptive beta: decrease as we converge

        tgt_candidates = np.argsort(grad)[:max(5, n // 20)]
        
        i = int(src_candidates[rng.integers(0, len(src_candidates))])
        j = int(tgt_candidates[rng.integers(0, len(tgt_candidates))])
        
        if i == j:
            no_imp += 1
            if no_imp > 2000:
                break
            continue
        
        # Try multiple transfer fractions
        best_local = None
        best_local_score = best_score
        for frac in [0.05, 0.1, 0.2, 0.3]:
            delta = frac * a[i]
            if delta <= 0:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = _project_simplex(a_try)
            sc = _score(a_try)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = a_try
        
        if best_local is not None:
            a = best_local
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            if no_imp > 3000:
                break
    
    return best_a, best_score


def _block_rotate(a, deadline, max_iter=3000):
    """Block rotation/swap to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    n = len(a)
    rng = np.random.default_rng(777)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        block_size = int(rng.integers(max(2, n // 64), max(3, n // 16)))
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue
        
        a_new = a.copy()
        a_new[i:i + block_size], a_new[j:j + block_size] = \
            a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
        a_new = _project_simplex(a_new)
        
        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            # Occasionally accept with "simulated annealing"
            if rng.random() < 0.01 and sc < best_score * 1.01:
                a = a_new
            no_imp += 1
            if no_imp > 1500:
                break
    
    return best_a, best_score


def _optimize_at_length(a_init, n, deadline):
    """Full pipeline optimization at a fixed length."""
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    print(f"    Init: score={best_score:.10f}, n={n}")
    
    # Phase 1: Nesterov gradient (fast, many iterations)
    t1 = time.monotonic()
    budget1 = min(60.0, (deadline - t1) * 0.3)
    if budget1 > 5:
        a1, sc1 = _nesterov_sentence(a, t1 + budget1, max_iter=30000)
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
        print(f"    Nesterov: {sc1:.10f} (t={time.monotonic()-t1:.1f}s)")
    
    # Phase 2: LP polish
    t2 = time.monotonic()
    budget2 = min(80.0, (deadline - t2) * 0.3)
    if budget2 > 10:
        a2, sc2 = _lp_polish(best_a, t2 + budget2, max_iter=2000)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
        print(f"    LP: {sc2:.10f} (t={time.monotonic()-t2:.1f}s)")
    
    # Phase 3: Nesterov again with tighter params
    t3 = time.monotonic()
    budget3 = min(60.0, (deadline - t3) * 0.25)
    if budget3 > 5:
        a3, sc3 = _nesterov_sentence(best_a, t3 + budget3, max_iter=20000)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
        print(f"    Nesterov2: {sc3:.10f} (t={time.monotonic()-t3:.1f}s)")
    
    # Phase 4: Coordinate refinement
    t4 = time.monotonic()
    budget4 = min(40.0, (deadline - t4) * 0.2)
    if budget4 > 5:
        a4, sc4 = _coordinate_refine(best_a, t4 + budget4, max_iter=10000)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        print(f"    Coord: {sc4:.10f} (t={time.monotonic()-t4:.1f}s)")
    
    # Phase 5: Adam polish
    t5 = time.monotonic()
    budget5 = min(60.0, (deadline - t5) * 0.25)
    if budget5 > 5:
        a5, sc5 = _adam_logit_polish(best_a, t5 + budget5, max_iter=20000, lr=0.025)
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()
        print(f"    Adam: {sc5:.10f} (t={time.monotonic()-t5:.1f}s)")
    
    # Phase 6: Final LP + Nesterov
    t6 = time.monotonic()
    budget6 = max(5.0, deadline - t6 - 10)
    if budget6 > 10:
        a6, sc6 = _lp_polish(best_a, t6 + budget6 * 0.5, max_iter=1000)
        if sc6 < best_score:
            best_score = sc6
            best_a = a6.copy()
        t6b = time.monotonic()
        if t6b < deadline - 10:
            a6b, sc6b = _nesterov_sentence(best_a, deadline - 10, max_iter=10000)
            if sc6b < best_score:
                best_score = sc6b
                best_a = a6b.copy()
        print(f"    Final: {best_score:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: Nesterov gradient + LP + coordinate descent + Adam.
    Explores multiple lengths with diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    # Try a range of lengths around the parent
    for ln in [512, 640, 768, 896, 1024, 1152]:
        if ln not in lengths and ln <= 2048:
            lengths.append(ln)
    lengths = sorted(set(lengths))
    
    # If no parent, start with 768
    if parent is None:
        lengths = [640, 768, 896, 1024]
    
    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")
    
    # Phase 1: Optimize at each length with parent-based init
    n_lengths = len(lengths)
    # Allocate more time to the parent length
    parent_len = lengths[0] if parent is not None else 768
    total_budget = deadline - time.monotonic() - 100.0
    # Give 60% budget to best-length candidates, 40% to exploration
    parent_budget = total_budget * 0.5
    exp_budget = total_budget * 0.5
    per_exp = exp_budget / max(1, n_lengths - 1)
    
    # First: optimize at parent length (most promising)
    if parent is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Optimizing at parent length n={parent_len} (budget={parent_budget:.0f}s) ===")
        phase_deadline = time.monotonic() + parent_budget
        a_res, sc_res = _optimize_at_length(parent, parent_len, phase_deadline)
        if sc_res < best_score:
            best_score = sc_res
            best_seq = a_res.copy()
            print(f"  ** New best at n={parent_len}: {sc_res:.10f}")
    
    # Then: try other lengths
    for n_t in lengths:
        if n_t == parent_len:
            continue
        if time.monotonic() > deadline - 50:
            break
        
        rem = deadline - time.monotonic() - 50
        per_actual = min(per_exp, rem / max(1, len(lengths) - lengths.index(n_t) - 1)) if n_lengths > 1 else rem
        per_actual = max(30.0, per_actual)
        
        if per_actual < 20 or time.monotonic() > deadline - 40:
            continue
        
        print(f"\n=== Exploring n={n_t} (budget={per_actual:.0f}s) ===")
        phase_deadline = time.monotonic() + per_actual
        
        # Use parent-interpolated init
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_linear_profile(n_t, 1.5)
        
        a_res, sc_res = _optimize_at_length(a_init, n_t, phase_deadline)
        if sc_res < best_score:
            best_score = sc_res
            best_seq = a_res.copy()
            print(f"  ** New best at n={n_t}: {sc_res:.10f}")
    
    # Phase 2: Diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===")
        
        inits = []
        # Various structured profiles
        inits.append(_make_linear_profile(n_best, 1.0))
        inits.append(_make_linear_profile(n_best, 2.0))
        inits.append(_make_linear_profile(n_best, 0.5))
        inits.append(_make_u_profile(n_best, 1.5))
        inits.append(_make_u_profile(n_best, 3.0))
        inits.append(_make_bimodal_profile(n_best, 0.2, 0.12))
        inits.append(_make_bimodal_profile(n_best, 0.3, 0.15))
        inits.append(_make_asymmetric_profile(n_best, 0.5))
        inits.append(_make_asymmetric_profile(n_best, -0.3))
        
        # Perturbed best
        for scale in [0.1, 0.2, 0.3]:
            a_pert = best_seq * (1.0 + scale * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)
        
        # Interpolated parent (if different from best)
        if parent is not None and parent.size != n_best:
            inits.append(_interpolate_profile(parent, n_best))
        
        # Allocate time for diverse inits
        rem = deadline - time.monotonic() - 60
        n_inits = min(len(inits), max(2, int(rem / 45)))
        per_init = rem / max(1, n_inits)
        
        for i, a0 in enumerate(inits[:n_init]):
            if time.monotonic() > deadline - 40:
                break
            
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            sub_deadline = time.monotonic() + min(per_init, deadline - time.monotonic() - 30)
            if sub_deadline - time.monotonic() < 15:
                break
            
            a_res, sc_res = _optimize_at_length(a0, n_best, sub_deadline)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best (init {i}): {sc_res:.10f}")
    
    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        rem = deadline - time.monotonic() - 20
        print(f"\n=== Phase 3: Final polish, rem={rem:.0f}s ===")
        
        # 3a: Nesterov
        b1 = rem * 0.3
        if b1 > 10:
            a_r, sc_r = _nesterov_sentence(best_seq, time.monotonic() + b1, max_iter=20000)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            print(f"  Final Nesterov: {sc_r:.10f}")
        
        # 3b: LP
        b2 = rem * 0.25
        if b2 > 10 and time.monotonic() < deadline - 20:
            a_r, sc_r = _lp_polish(best_seq, time.monotonic() + b2, max_iter=3000)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            print(f"  Final LP: {sc_r:.10f}")
        
        # 3c: Adam
        b3 = rem * 0.2
        if b3 > 10 and time.monotonic() < deadline - 20:
            a_r, sc_r = _adam_logit_polish(best_seq, time.monotonic() + b3, max_iter=15000, lr=0.02)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            print(f"  Final Adam: {sc_r:.10f}")
        
        # 3d: Coordinate
        b4 = rem * 0.1
        if b4 > 5 and time.monotonic() < deadline - 15:
            a_r, sc_r = _coordinate_refine(best_seq, time.monotonic() + b4, max_iter=5000)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            print(f"  Final Coord: {sc_r:.10f}")
        
        # 3e: One last Nesterov
        if time.monotonic() < deadline - 15:
            a_r, sc_r = _nesterov_sentence(best_seq, deadline - 10, max_iter=10000)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            print(f"  Final Nesterov2: {sc_r:.10f}")
    
    # Safety: ensure valid output
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * 1.0  # normalize to sum=1 for stability
    else:
        final_seq = _make_linear_profile(768, 1.5)
    
    # Validate with evaluate_sequence
    if final_seq.size <= 5000:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  eval failed: {e}")
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
