    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_from_weights(a, w):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, max_iter=20000, inflate_init=1.001, blend_init=0.25):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend_init
    inflate_cur = inflate_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 1000:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.93)
                inflate_cur = min(1.008, inflate_cur * 1.01)
            if no_imp > max(500, 4 * n):
                a = best_a.copy()
                blend_cur = blend_init
                inflate_cur = inflate_init
                no_imp = 0

        a = a_new

    return best_a


def _exp_gradient(a, deadline, max_iter=30000, beta_init_frac=0.03):
    """Exponentiated gradient (multiplicative weights) with aggressive beta annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-10, beta_init_frac * float(c.max()))
    eta = 0.15
    no_imp = 0
    base_eta = eta

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weight(c, beta)
        grad = _gradient_from_weights(a, w)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 3.0)
            continue

        # Multiplicative update: a_new = a * exp(-eta * grad / (a + eps))
        # Normalized to maintain sum=1
        ratio = np.maximum(a, 1e-30)
        log_update = -eta * grad / ratio
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            beta = max(1e-10, beta_init_frac * float(c.max()))
            eta = base_eta
            no_imp = 0
            continue
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            eta = min(base_eta * 1.5, 0.5)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                eta = max(0.02, eta * 0.95)
            if no_imp > max(1000, 3 * n):
                a = best_a.copy()
                beta = max(1e-10, beta_init_frac * float(c.max()))
                eta = base_eta
                no_imp = 0

        # Aggressive beta annealing
        if it % 50 == 0:
            beta = max(1e-12, beta * 0.85)

        a = a_new

    return best_a


def _multi_scale_gradient(a, deadline, max_iter=20000):
    """Combine sharp and broad peak targeting in gradient."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta_sharp = max(1e-10, 0.01 * float(c.max()))
    beta_broad = max(1e-8, 0.1 * float(c.max()))
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w_sharp = _smooth_max_weight(c, beta_sharp)
        w_broad = _smooth_max_weight(c, beta_broad)
        w = 0.5 * w_sharp + 0.5 * w_broad
        w = w / w.sum()

        grad = _gradient_from_weights(a, w)

        if not np.all(np.isfinite(grad)):
            beta_sharp = max(1e-10, beta_sharp * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.96)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                c_best = _fft_convolve(a)
                beta_sharp = max(1e-10, 0.01 * float(c_best.max()))
                beta_broad = max(1e-8, 0.1 * float(c_best.max()))
                lr = base_lr
                no_imp = 0

        if it % 80 == 0:
            beta_sharp = max(1e-12, beta_sharp * 0.8)
            beta_broad = max(1e-10, beta_broad * 0.85)

    return best_a


def _directed_peak_reduction(a, deadline, max_iter=20000):
    """Directly target top peaks by moving mass from contributing positions."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(42)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        # Find top peaks
        top_k = min(5, len(c))
        peak_positions = np.argsort(c)[::-1][:top_k]

        for pk in peak_positions:
            # Position pk in conv: conv[pk] = sum_j a_j * a[pk-j]
            # Contributors: pairs (j, pk-j) for valid j
            # Find the largest contributor
            contrib = np.zeros(n)
            j_min = max(0, pk - n + 1)
            j_max = min(n - 1, pk)
            for j in range(j_min, j_max + 1):
                contrib[j] += a[pk - j]
            
            # Find position with highest contribution
            if contrib.max() < 1e-30:
                continue
            src = int(np.argmax(contrib))

            # Find target: position with low value that doesn't contribute to this peak
            # A position t contributes to peak pk if a[t]*a[pk-t] is large
            # So we want t where a[pk-t] is small (or t is far from contributing)
            target_scores = np.zeros(n)
            for t in range(n):
                if t == src:
                    continue
                # How much would adding mass at t increase peak pk?
                if 0 <= pk - t < n:
                    target_scores[t] = a[pk - t]
                else:
                    target_scores[t] = 0.0
            # We want to minimize target_scores (don't add to the peak)
            # Also prefer positions where a is already small
            target_scores += 0.1 * a
            dst = int(np.argmin(target_scores))

            # Transfer a small fraction
            frac = 0.02 + 0.03 * rng.random()
            delta = frac * a[src]
            if delta < 1e-15:
                continue

            a_try = a.copy()
            a_try[src] -= delta
            a_try[dst] += delta
            if a_try[src] < 0:
                continue
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            sc_try = _score(a_try)
            if sc_try < best_score - 1e-13:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                no_imp = 0
            else:
                no_imp += 1

        if no_imp > 3000:
            break

    return best_a


def _simulated_annealing_blocks(a, deadline, max_iter=10000):
    """Simulated annealing on block moves to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(999)

    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    
    T_init = 0.001
    T = T_init
    cooling = 0.9995
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Random block swap
        block_size = int(rng.integers(max(2, n // 64), max(3, n // 16)))
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        tmp = a_new[i:i + block_size].copy()
        a_new[i:i + block_size] = a_new[j:j + block_size]
        a_new[j:j + block_size] = tmp
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)

        # Accept or reject (simulated annealing)
        if sc_new < cur_score:
            accept = True
        else:
            delta = sc_new - cur_score
            if T > 1e-15 and rng.random() < np.exp(-delta / T):
                accept = True
            else:
                accept = False

        if accept:
            a = a_new
            cur_score = sc_new
            if sc_new < best_score - 1e-13:
                best_score = sc_new
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        T = max(1e-15, T * cooling)

        if no_imp > 5000:
            break

    return best_a


def _logit_adam_polish(a, deadline, max_iter=20000, lr=0.04):
    """Adam optimizer on logit parametrization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-10, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        grad_p = _gradient_from_weights(p, w)
        dot = float(np.dot(p, grad_p))
        gz = p * (grad_p - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.96)

        if it % 120 == 0:
            beta = max(1e-12, beta * 0.78)

        if no_imp > max(3000, 5 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-10, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _pipeline_length(a_init, n, deadline, budget_frac=1.0):
    """Run multi-method pipeline for a given initial sequence."""
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 1: LP refinement (most powerful for big moves)
    lp_time = min(remaining * 0.25, 120.0)
    if lp_time > 5:
        a_lp = _lp_refine(a, time.monotonic() + lp_time,
                         max_iter=15000, inflate_init=1.0008, blend_init=0.2)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"      LP: {sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 2: Exponentiated gradient
    eg_time = min(remaining * 0.2, 90.0)
    if eg_time > 5:
        a_eg = _exp_gradient(best_a, time.monotonic() + eg_time,
                            max_iter=20000, beta_init_frac=0.03)
        sc = _score(a_eg)
        if sc < best_score:
            best_score = sc
            best_a = a_eg.copy()
        print(f"      ExpGrad: {sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 3: Multi-scale gradient
    ms_time = min(remaining * 0.2, 80.0)
    if ms_time > 5:
        a_ms = _multi_scale_gradient(best_a, time.monotonic() + ms_time,
                                     max_iter=15000)
        sc = _score(a_ms)
        if sc < best_score:
            best_score = sc
            best_a = a_ms.copy()
        print(f"      MultiScale: {sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 4: Directed peak reduction
    dp_time = min(remaining * 0.15, 60.0)
    if dp_time > 5:
        a_dp = _directed_peak_reduction(best_a, time.monotonic() + dp_time,
                                        max_iter=15000)
        sc = _score(a_dp)
        if sc < best_score:
            best_score = sc
            best_a = a_dp.copy()
        print(f"      PeakReduce: {sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 5: Simulated annealing
    sa_time = min(remaining * 0.1, 40.0)
    if sa_time > 3:
        a_sa = _simulated_annealing_blocks(best_a, time.monotonic() + sa_time,
                                           max_iter=8000)
        sc = _score(a_sa)
        if sc < best_score:
            best_score = sc
            best_a = a_sa.copy()
        print(f"      SimAnneal: {sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 6: Adam polish
    ad_time = min(remaining * 0.15, 60.0)
    if ad_time > 5:
        a_ad = _logit_adam_polish(best_a, time.monotonic() + ad_time,
                                  max_iter=15000, lr=0.03)
        sc = _score(a_ad)
        if sc < best_score:
            best_score = sc
            best_a = a_ad.copy()
        print(f"      Adam: {sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 7: Final LP
    lp2_time = min(remaining * 0.15, 40.0)
    if lp2_time > 5:
        a_lp2 = _lp_refine(best_a, time.monotonic() + lp2_time,
                          max_iter=8000, inflate_init=1.0005, blend_init=0.15)
        sc = _score(a_lp2)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2.copy()
        print(f"      LP2: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining exponentiated gradient, directed peak
    reduction, adaptive LP, multi-scale smooth-max gradient, simulated annealing,
    and Adam polish. Explores multiple sequence lengths with focus on the best.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 768, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))

    print(f"Lengths to try: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Quick evaluation at each length to find the best
    print("\n=== Phase 1: Quick evaluation ===")
    length_scores = {}
    for n_t in lengths:
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_linear_profile(n_t, 1.5)
        sc = _score(a_init)
        length_scores[n_t] = sc
        print(f"  n={n_t}: init score={sc:.10f}")

    # Sort lengths by initial score (best first)
    lengths_sorted = sorted(lengths, key=lambda x: length_scores[x])
    print(f"  Order: {[(n, f'{length_scores[n]:.6f}') for n in lengths_sorted]}")

    # Phase 2: Full pipeline on best length
    print("\n=== Phase 2: Full pipeline on best length ===")
    n_best = lengths_sorted[0]
    if parent is not None:
        a_init_best = _interpolate_profile(parent, n_best)
    else:
        a_init_best = _make_linear_profile(n_best, 1.5)

    # Give most time to the best length
    main_time = min(800.0, (deadline - time.monotonic()) * 0.5)
    a_main, sc_main = _pipeline_length(a_init_best, n_best, time.monotonic() + main_time)
    if sc_main < best_score:
        best_score = sc_main
        best_seq = a_main.copy()
        print(f"  Main pipeline n={n_best}: {sc_main:.10f}")

    # Phase 3: Try second best length
    if len(lengths_sorted) > 1 and time.monotonic() < deadline - 200:
        print("\n=== Phase 3: Second length ===")
        n_second = lengths_sorted[1]
        if n_second != n_best:
            if parent is not None:
                a_init_2 = _interpolate_profile(parent, n_second)
            else:
                a_init_2 = _make_linear_profile(n_second, 1.5)
            t2 = min(300.0, (deadline - time.monotonic()) * 0.3)
            a_2, sc_2 = _pipeline_length(a_init_2, n_second, time.monotonic() + t2)
            if sc_2 < best_score:
                best_score = sc_2
                best_seq = a_2.copy()
                n_best = n_second
                print(f"  ** New best at n={n_second}: {sc_2:.10f}")

    # Phase 4: Diverse restarts on best length
    if time.monotonic() < deadline - 100:
        print(f"\n=== Phase 4: Diverse restarts at n={best_seq.size} ===")
        n_cur = best_seq.size
        inits = []

        # Perturbed best
        for scale in [0.05, 0.1, 0.2]:
            a_p = best_seq * (1.0 + scale * rng.normal(size=n_cur))
            a_p = np.maximum(a_p, 0.0)
            a_p = a_p / a_p.sum()
            inits.append(a_p)

        # U-shaped profiles
        for alpha in [1.0, 2.0, 4.0]:
            inits.append(_make_u_profile(n_cur, alpha))

        # Linear profiles
        for eta in [0.5, 1.5, 3.0]:
            inits.append(_make_linear_profile(n_cur, eta))

        # Random smooth
        for _ in range(2):
            freqs = rng.uniform(1, 5, 5)
            phases = rng.uniform(0, 2 * np.pi, 5)
            t = np.linspace(0, 1, n_cur)
            a_r = 1.0 + 0.3 * sum(np.sin(2 * np.pi * f * t + ph) for f, ph in zip(freqs, phases))
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / a_r.sum()
            inits.append(a_r)

        # Try each with quick pipeline
        remaining = deadline - time.monotonic() - 50
        per_init = min(60.0, remaining / max(1, len(inits)))

        for a0 in inits:
            if time.monotonic() > deadline - 40:
                break
            if per_init < 10:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            a_res, sc_res = _pipeline_length(a0, n_cur, time.monotonic() + per_init)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best: {sc_res:.10f}")

    # Phase 5: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Final polish ===")
        n_final = best_seq.size
        remaining = deadline - time.monotonic() - 15

        # 5a: LP
        t_lp = min(remaining * 0.3, 80.0)
        if t_lp > 10:
            a_f = _lp_refine(best_seq, time.monotonic() + t_lp,
                            max_iter=20000, inflate_init=1.0005, blend_init=0.15)
            sc = _score(a_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_f.copy()
            print(f"  Final LP: {sc:.10f}")

        # 5b: Exp gradient
        t_eg = min(remaining * 0.25, 60.0)
        if t_eg > 10:
            a_f = _exp_gradient(best_seq, time.monotonic() + t_eg,
                               max_iter=20000, beta_init_frac=0.02)
            sc = _score(a_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_f.copy()
            print(f"  Final ExpGrad: {sc:.10f}")

        # 5c: Directed peak reduction
        t_dp = min(remaining * 0.2, 50.0)
        if t_dp > 5:
            a_f = _directed_peak_reduction(best_seq, time.monotonic() + t_dp,
                                           max_iter=15000)
            sc = _score(a_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_f.copy()
            print(f"  Final PeakReduce: {sc:.10f}")

        # 5d: Adam
        t_ad = min(remaining * 0.15, 40.0)
        if t_ad > 5:
            a_f = _logit_adam_polish(best_seq, time.monotonic() + t_ad,
                                     max_iter=15000, lr=0.02)
            sc = _score(a_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_f.copy()
            print(f"  Final Adam: {sc:.10f}")

        # 5e: One more very tight LP
        t_lp2 = min(remaining * 0.1, 20.0)
        if t_lp2 > 5:
            a_f = _lp_refine(best_seq, time.monotonic() + t_lp2,
                            max_iter=5000, inflate_init=1.0003, blend_init=0.1)
            sc = _score(a_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_f.copy()
            print(f"  Final LP2: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(768, 1.5)

    # Authoritative evaluation
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
