"""
Optimized step function search using L-BFGS with logsumexp continuation.
Key improvements over previous:
- Better initial profiles (pure ramps instead of centered ramps)
- More aggressive beta continuation
- Multiple length attempts
- Coordinate descent polish phase
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        np.nan_to_num(grad, copy=False)
    return grad

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

def _coordinate_descent(a_init, deadline, n_steps=500, step_size=0.01):
    """Local coordinate descent to polish the solution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Try random coordinate
        i = rng.integers(0, n)
        
        # Try a few small perturbations
        for delta in [step_size, -step_size, step_size*0.5, -step_size*0.5]:
            a_new = a.copy()
            a_new[i] += delta * a.mean()
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = _score_norm_fn(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
    
    return best_a, best_score

def _make_ramp_profile(n, power=1.0):
    """Create a pure ramp: a[i] = (i+1)^power, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return a / a.sum()

def _make_log_profile(n):
    """Create a logarithmic profile."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.log(i + 1.0)
    return a / a.sum()

def _make_gaussian_mixture(n, rng):
    """Create a random sum of Gaussians."""
    a = np.zeros(n)
    for _ in range(3):
        center = rng.uniform(0.2, 0.8) * (n - 1)
        width = rng.uniform(0.05, 0.3) * n
        amplitude = rng.uniform(0.5, 2.0)
        t = np.arange(n)
        a += amplitude * np.exp(-(t - center) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

Optimized step function search using L-BFGS with logsumexp continuation.
Key improvements: pure ramp initializations (not centered), aggressive beta schedule,
multiple length attempts, and coordinate descent polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    exp_sum = np.sum(np.exp(c_shifted))
    if exp_sum < 1e-300 or not np.isfinite(exp_sum):
        return 2.0 * n * cm
    log_sum = float(np.log(exp_sum))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        np.nan_to_num(grad, copy=False, nan=0.0, posinf=0.0, neginf=0.0)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time=None, verbose=False):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_fracs = [0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-15) for b in beta_fracs]

    if max_wall_time is not None:
        per_level = max_wall_time / len(betas)
    else:
        per_level = 20.0

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        t_level = time.monotonic()
        remaining = deadline - t_level
        if remaining < 5:
            break

        max_iter = min(1200, int(remaining * 50))
        if bi < 3:

        elif bi < 7:

        try:

def _adam_smoothed(a_init, deadline, lr=0.05, beta_frac=0.01, max_iter=20000):
    """Adam optimizer on the logsumexp-smoothed objective in logit space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            beta = max(1e-12, beta * 0.7)
            if best_score < 1.4:
                lr = max(0.003, lr * 0.9)

    return best_a, best_score


def _coordinate_polish(a_init, deadline, step_frac=0.003, max_steps=10000):

    rng = np.random.default_rng(54321)

        if time.monotonic() > deadline:
            break

        i = int(rng.integers(0, n))
        improved = False
        for sign in [1, -1]:
            for frac in [step_frac, step_frac * 0.3]:
                delta = sign * frac * mean_val
                a_new = a.copy()
                a_new[i] = max(0.0, a_new[i] + delta)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new /= s
                sc = _score_norm_fn(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
            if improved:
                break

        if step % 1000 == 0:
            step_frac *= 0.7

    return best_a, best_score


def _make_ramp_profile(n, power=1.0):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return a / a.sum()


def _make_log_profile(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.log(i + 0.5)
    return a / a.sum()


def _make_power_ramp(n, p):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** p
    return a / a.sum()


def _make_quadratic_ramp(n):
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = t ** 2 + 0.01
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase L-BFGS + Adam + coordinate descent for minimizing
    2*n*max(conv(a,a))/sum(a)^2. Uses pure ramp initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Generate and evaluate initial profiles ---
    print("=== Generating initial profiles ===")
    profiles = []

    # Parent
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 10:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                profiles.append((sc, parent.copy(), "parent"))
                update_best(parent, sc, "parent")
    except Exception:
        pass

    # Pure ramps - these are key!
    ramp_configs = [
        (512, 1.0), (512, 1.5), (512, 2.0),
        (1024, 0.5), (1024, 1.0), (1024, 1.5), (1024, 2.0), (1024, 3.0),
        (2048, 0.5), (2048, 1.0), (2048, 1.5), (2048, 2.0), (2048, 3.0),
        (4096, 1.0), (4096, 1.5), (4096, 2.0),
    ]
    for n_try, power in ramp_configs:
        p = _make_ramp_profile(n_try, power)
        sc = _score_norm_fn(p)
        profiles.append((sc, p, f"ramp_p{power}_n{n_try}"))
        update_best(p, sc, f"ramp_p{power}_n{n_try}")

    # Log profiles
    for n_try in [1024, 2048]:
        p = _make_log_profile(n_try)
        sc = _score_norm_fn(p)
        profiles.append((sc, p, f"log_n{n_try}"))
        update_best(p, sc, f"log_n{n_try}")

    # Quadratic ramp
    for n_try in [1024, 2048]:
        p = _make_quadratic_ramp(n_try)
        sc = _score_norm_fn(p)
        profiles.append((sc, p, f"quad_n{n_try}"))
        update_best(p, sc, f"quad_n{n_try}")

    profiles.sort(key=lambda x: x[0])
    print(f"Initial best: {best_score:.10f}")
    print("Top 8 profiles:")
    for sc, p, label in profiles[:8]:
        print(f"  {sc:.10f} (n={len(p)}) [{label}]")

    # --- Phase 1: L-BFGS on diverse top profiles ---
    print(f"\n=== Phase 1: L-BFGS (time budget: {(deadline - t_start) / 60:.1f} min) ===")
    phase1_deadline = min(deadline, t_start + 450.0)

    # Select diverse profiles (different lengths)
    seen_n = set()
    selected = []
    for sc, p, label in profiles:
        n_p = len(p)
        if n_p not in seen_n:
            seen_n.add(n_p)
            selected.append((sc, p, label))
        if len(selected) >= 4:
            break

    # Always add the absolute best
    best_prof = profiles[0]
    if best_prof[2] not in [x[2] for x in selected]:
        selected.insert(0, best_prof)

    for idx, (sc, p, label) in enumerate(selected):
        if time.monotonic() > phase1_deadline - 30:
            break
        remaining = phase1_deadline - time.monotonic()
        alloc = min(remaining / max(1, len(selected) - idx), 120.0)
        if alloc < 20:
            break

        print(f"\n  [{label}] n={len(p)}, init={sc:.10f}, alloc={alloc:.0f}s")
        t0 = time.monotonic()
        a_best, sc_best = _run_lbfgs(p, t0 + alloc, max_wall_time=alloc, verbose=True)
        elapsed = time.monotonic() - t0
        print(f"    -> score={sc_best:.10f} ({elapsed:.1f}s)")
        update_best(a_best, sc_best, f"lbfgs_{label}")

    print(f"\nAfter Phase 1: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 2: Adam polish on best ---
    print(f"\n=== Phase 2: Adam optimization ===")
    phase2_deadline = min(deadline, t_start + 800.0)

    if time.monotonic() < phase2_deadline and best_seq is not None:
        alloc = min(phase2_deadline - time.monotonic() - 10, 250.0)
        if alloc > 30:
            t0 = time.monotonic()
            a_best, sc_best = _adam_smoothed(best_seq, t0 + alloc, lr=0.03, beta_frac=0.008, max_iter=50000)
            elapsed = time.monotonic() - t0
            print(f"  Adam: score={sc_best:.10f} ({elapsed:.1f}s)")
            update_best(a_best, sc_best, "adam")

    print(f"After Phase 2: best={best_score:.10f}")

    # --- Phase 3: Multi-restart L-BFGS ---
    print(f"\n=== Phase 3: Multi-restart ===")
    phase3_deadline = min(deadline, t_start + 1100.0)

    if time.monotonic() < phase3_deadline and best_seq is not None:
        n_restarts = 3
        per_restart = (phase3_deadline - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase3_deadline - 30:
                break
            noise = 0.03 + 0.04 * i
            z = np.log(np.maximum(best_seq, 1e-20))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            alloc = min(per_restart - 5, 80.0)
            if alloc < 15:
                break
            t0 = time.monotonic()
            a_best, sc_best = _run_lbfgs(p_start, t0 + alloc, max_wall_time=alloc, verbose=False)
            print(f"  Restart {i} (noise={noise:.3f}): score={sc_best:.10f} ({elapsed:.1f}s)" if False else f"  Restart {i} (noise={noise:.3f}): score={sc_best:.10f}")
            update_best(a_best, sc_best, f"restart{i}")

    print(f"After Phase 3: best={best_score:.10f}")

    # --- Phase 4: Coordinate descent polish ---
    print(f"\n=== Phase 4: Coordinate descent ===")
    phase4_deadline = min(deadline, t_start + 1400.0)

    if time.monotonic() < phase4_deadline and best_seq is not None:
        alloc = min(phase4_deadline - time.monotonic() - 10, 200.0)
        if alloc > 20:
            t0 = time.monotonic()
            a_best, sc_best = _coordinate_polish(best_seq, t0 + alloc, step_frac=0.002, max_steps=20000)
            elapsed = time.monotonic() - t0
            print(f"  Coord descent: score={sc_best:.10f} ({elapsed:.1f}s)")
            update_best(a_best, sc_best, "coord_polish")

    print(f"After Phase 4: best={best_score:.10f}")

    # --- Phase 5: Try longer sequence ---
    print(f"\n=== Phase 5: Longer sequence ===")
    phase5_deadline = min(deadline, t_start + 1650.0)

    if time.monotonic() < phase5_deadline and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr <= 2048:
            target_n = 4096
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            alloc = min(phase5_deadline - time.monotonic() - 20, 150.0)
            if alloc > 30:
                t0 = time.monotonic()
                a_best, sc_best = _run_lbfgs(p_interp, t0 + alloc, max_wall_time=alloc, verbose=True)
                print(f"  L-BFGS n={target_n}: score={sc_best:.10f} ({time.monotonic()-t0:.1f}s)")
                update_best(a_best, sc_best, f"lbfgs_n{target_n}")

    print(f"After Phase 5: best={best_score:.10f}")

    # --- Phase 6: Final combined polish ---
    print(f"\n=== Phase 6: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        # L-BFGS with very low beta
        alloc = min(deadline - time.monotonic() - 15, 100.0)
        if alloc > 20:
            t0 = time.monotonic()
            a_best, sc_best = _run_lbfgs(best_seq, t0 + alloc, max_wall_time=alloc, verbose=False)
            print(f"  Final L-BFGS: score={sc_best:.10f} ({time.monotonic()-t0:.1f}s)")
            update_best(a_best, sc_best, "final_lbfgs")

        # Quick Adam
        if time.monotonic() < deadline - 10:
            alloc2 = min(deadline - time.monotonic() - 5, 40.0)
            if alloc2 > 10:
                t0 = time.monotonic()
                a_best, sc_best = _adam_smoothed(best_seq, t0 + alloc2, lr=0.01, beta_frac=0.002, max_iter=10000)
                print(f"  Final Adam: score={sc_best:.10f} ({time.monotonic()-t0:.1f}s)")
                update_best(a_best, sc_best, "final_adam")

    # --- Final validation ---
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp_profile(1024, 1.0)

        sc_final = _score_fn(final_seq)
        print(f"Score (internal): {sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Score (evaluator): {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluator error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp_profile(1024, 1.0)
    print(f"FALLBACK: n=1024, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
