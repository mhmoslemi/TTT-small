"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses: adaptive-temperature smooth gradient descent in log-space, LP-based
spreading, 1D coordinate descent on hot coordinates, L-BFGS-B polish,
and multi-length search (768, 1024, 1536).

import numpy as np
import time

from scipy.special import softmax


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming a is normalized to sum=1."""
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_smooth_max(a, beta):
    """Compute gradient of smooth max of linear conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_and_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, c


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0

    a = np.maximum(a, 1e-10)
    s = a.sum()

def _adam_logspace(a, deadline, max_iter=80000, lr=0.05, beta_scale=0.05):

    Adam optimization in log-space with adaptive smooth-max temperature.
    Returns the best sequence found.
    """

    a = a / a.sum()
    n = len(a)

    z -= float(np.mean(z))

    b2 = 0.999

    t_adam = 0

    c_max = float(c.max())

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    base_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

        if time.monotonic() > deadline:
            break

        c = _fft_convolve(p)

            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:

                bb_cand = np.clip(bb_cand, 0.1, 5.0)
                bb_scale = 0.7 * bb_scale + 0.3 * bb_cand

            best_z = z_new.copy()

                gap = top3[0] - top3[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)

            prev_grad = None
            beta = max(1e-6, beta_scale * float(_fft_convolve(a).max()))

def _lp_refine(a, deadline, max_iter=5000, inflate=1.0005, blend=0.2):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.05)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _coordinate_descent_hot(a, deadline, n_hot=20, max_rounds=200):
    """
    1D coordinate descent on the hottest coordinates.
    For each hot coordinate, find the optimal value that minimizes max peak.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        score = float(2.0 * n * c_max)

        # Find hot coordinates: those that contribute most to the top peaks
        top_k = np.argsort(c)[::-1][:min(5, 2 * n - 1)]
        pressure = np.zeros(n)
        for k in top_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]

        hot_idx = np.argsort(pressure)[::-1][:n_hot]

        improved = False
        for i in hot_idx:
            if time.monotonic() > deadline:
                break

            # For coordinate i, find optimal a[i]
            # c[k] depends on a[i] for k in [i, i+n-1]
            # c[k] = 2*a[i]*a[k-i] + rest[k] (for k-i != i)
            # c[k] = a[i]^2 + rest[k] (for k == i, i.e., the self-term)

            ai_old = a[i]
            # Compute rest[k] = c[k] - contribution of a[i]
            # Contribution to c[k] from a[i]:
            #   if k >= i and k-i < n: add a[i]*a[k-i]
            #   if k >= i and i < n (always true): the term j=k-i gives a[k-i]*a[i]
            # So total contribution = a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
            # But when k-i == i (k=2i), both terms are a[i]*a[i], so contribution = a[i]^2

            # Affine part: for each k, c[k] = slope[k]*a[i] + intercept[k]
            slopes = np.zeros(2 * n - 1)
            intercepts = c.copy()
            for k in range(i, min(i + n, 2 * n - 1)):
                j = k - i
                if 0 <= j < n:
                    if j == i:  # k = 2i
                        slopes[k] = 0  # quadratic term
                        intercepts[k] = c[k] - ai_old * ai_old
                    else:
                        slopes[k] = 2.0 * a[j]
                        intercepts[k] = c[k] - slopes[k] * ai_old

            # Quadratic part at k=2i (if 2i < 2*n-1)
            quad_idx = 2 * i
            has_quad = quad_idx < 2 * n - 1

            # Find optimal a[i] >= 0 minimizing max over k of:
            #   slopes[k]*t + intercepts[k]  (linear in t)
            #   t^2 + intercepts[quad_idx]  (quadratic, if has_quad)
            #
            # Use bisection on the derivative / golden section

            def obj_1d(t):
                vals = slopes * t + intercepts
                if has_quad:
                    vals[quad_idx] = t * t + intercepts[quad_idx]
                return float(vals.max())

            # Golden section search on [0, 3*ai_old + 0.1/n]
            lo = 0.0
            hi = 3.0 * ai_old + 1.0 / n
            gr = (np.sqrt(5.0) + 1.0) / 2.0
            x1 = hi - (hi - lo) / gr
            x2 = lo + (hi - lo) / gr
            f1 = obj_1d(x1)
            f2 = obj_1d(x2)
            for _ in range(40):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - (hi - lo) / gr
                    f1 = obj_1d(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + (hi - lo) / gr
                    f2 = obj_1d(x2)

            t_opt = (lo + hi) / 2.0
            f_opt = obj_1d(t_opt)
            f_old = obj_1d(ai_old)

            if f_opt < f_old - 1e-15:
                a[i] = max(t_opt, 1e-15)
                # Renormalize slightly (keep sum close to 1)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                improved = True

        if improved:
            c = _fft_convolve(a)
            new_score = float(2.0 * n * c.max())
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_a = a.copy()
        else:
            if rd > 5:
                break

    return best_a


def _lbbfqs_polish(a, deadline, max_iter=5000):
    """
    L-BFGS-B polish using log-parametrization and sharp smooth max.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.01 * float(c.max()))

    def objective(z):
        e = np.exp(z - np.max(z))
        aa = e / e.sum()
        cc = _fft_convolve(aa)
        sm, _ = _smooth_max_and_weights(cc, beta)
        return float(2.0 * n * sm)

    def gradient(z):
        e = np.exp(z - np.max(z))
        aa = e / e.sum()
        cc = _fft_convolve(aa)
        _, w = _smooth_max_and_weights(cc, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        aa_rev = aa[::-1].copy()
        fa_rev = np.fft.rfft(aa_rev, N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * aa
        dot = float(np.dot(aa, vp))
        gz = aa * (vp - dot)
        gz -= float(np.mean(gz))
        return gz

    z0 = np.log(a)
    z0 -= np.mean(z0)

    try:
        res = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': 20000,
                                'ftol': 1e-15, 'gtol': 1e-10,
                                'iprint': 0})
        if res.success or res.fun < best_score:
            e = np.exp(res.x - np.max(res.x))
            aa = e / e.sum()
            sc = _score_normalized(aa)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
    except Exception:
        pass

    return best_a

"""
Multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses: multi-temperature smooth gradient descent in log-space, LP-based
spreading, 1D coordinate descent on hot coordinates, L-BFGS-B polish,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _grad_smooth(a, beta):

    _, w = _smooth_max_weights(c, beta)

    h_full = np.fft.irfft(fw * fa_rev, N)

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))

    return A


def _lp_spread(a, M):

    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)

                      bounds=[(0, None)] * n, method='highs')
        if res.success:

    except Exception:
        pass
    return None


def _interp_profile(p0, n_new):

    x0 = np.linspace(0.0, 1.0, n0)

    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:

                    x2 = lo_b + (hi_b - lo_b) / gr_phi
                    f2 = obj_fu(x2)

            t_opt = (lo_b + hi_b) / 2.0
            f_opt = obj_fu(t_opt)
            f_old = obj_fu(ai)

            if f_opt < f_old - 1e-14:
                a[i] = max(t_opt, 1e-15)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                improved = True

        if improved:
            sc = _score_norm(a)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
        elif rd > 3:
            break

    return best_a
