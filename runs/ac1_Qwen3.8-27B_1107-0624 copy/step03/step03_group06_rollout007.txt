"""
Multi-phase optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses fast symmetric LP refinement with adaptive inflation, FFT-based smooth-gradient
polishing, multi-scale initialization, and random perturbation restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _make_init(n, profile='edges'):
    t = np.linspace(0.0, 1.0, n)
    if profile == 'uniform':
        a = np.ones(n)
    elif profile == 'ramp_up':
        a = t + 1e-8
    elif profile == 'ramp_down':
        a = (1.0 - t) + 1e-8
    elif profile == 'edges':
        a = 1.0 - 0.4 * np.sin(np.pi * t) + 1e-8
    elif profile == 'center':
        a = 1.0 + 0.4 * np.cos(np.pi * t) + 1e-8
    elif profile == 'quad_edges':
        a = 1.0 + 1.5 * (2.0 * t - 1.0) ** 2 + 1e-8
    elif profile == 'exp_edges':
        a = np.exp(-1.5 * t) + np.exp(-1.5 * (1.0 - t)) + 1e-8
    elif profile == 'cosh':
        a = np.cosh(2.5 * (t - 0.5)) + 1e-8
    elif profile == 'sine_edges':
        a = np.cos(np.pi * t) ** 2 + 0.1 + 1e-8
    elif profile == 'mixed':
        a = 0.5 * (1.0 + np.cos(np.pi * t)) + 0.5 * (t ** 2 + (1 - t) ** 2) + 1e-8
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _build_A_matrix(n):
    """Build the (2n-1) x n constraint matrix for conv(f, g) <= M.
    A[k, j] = 1 if f[k-j] would appear in conv(f,g)[k], else 0.
    Actually A[k,j] stores the index into f: f[k-j].
    We return it as a sparse-like structure for fast matrix-vector multiply.
    """
    # For each row k, the non-zero column range is [max(0,k-n+1), min(k,n-1)]
    # and A[k,j] = f[k-j]
    # We'll build the full dense matrix (for n<=2048 it's at most ~4M entries)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = 1.0  # placeholder, actual values depend on f
    return A


def _build_A_for_f(f, n, n_cons=None):
    """Build the constraint matrix with actual f values."""
    if n_cons is None:
        n_cons = 2 * n - 1
    n_cons = min(n_cons, 2 * n - 1)
    A = np.zeros((n_cons, n), dtype=np.float64)
    for k in range(n_cons):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1]
    return A


def _solve_lp(f, M, n, n_cons=None, time_limit=5.0):
    """Solve: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    if n_cons is None:
        n_cons = 2 * n - 1
    n_cons = min(n_cons, 2 * n - 1)
    A = _build_A_for_f(f, n, n_cons)
    c_obj = -np.ones(n)
    b = np.full(n_cons, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _smooth_conv_grad(a, beta):
    """Compute smooth max gradient of conv(a,a) using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / max(beta, 1e-15)
    x = np.clip(x, -50.0, 50.0)
    w = np.exp(x)
    w_sum = float(w.sum())
    if not np.isfinite(w_sum) or w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum

    # grad[m] = 2 * sum_k w[k] * a[k-m] (valid range)
    # = 2 * convolve(w, a[::-1])[n-1 : 2n-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    corr = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    """Project vector onto the probability simplex (sum=1, non-negative)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _random_perturb(a, scale=0.01, seed=0):

            a_gp, sc_gp = _gradient_polish(best_seq, phase3_end, max_iter=100000)
            sc_gp = _score(a_gp)
            if sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> {sc_gp:.10f} (improved!)")
            else:
                print(f"  -> {sc_gp:.10f} (best={best_score:.10f})")

    # Phase 4: Subgradient refinement
    if best_seq is not None:
        phase4_time = min(100.0, deadline - time.monotonic() - 60)
        if phase4_time > 15:
            phase4_end = time.monotonic() + phase4_time
            print(f"Phase 4 subgradient: budget={phase4_time:.0f}s")
            a_sg, sc_sg = _subgradient_step(best_seq, phase4_end, max_iter=20000)
            sc_sg = _score(a_sg)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  -> {sc_sg:.10f} (improved!)")
            else:
                print(f"  -> {sc_sg:.10f}")

    # Phase 5: Random perturbations + quick re-optimization
    if best_seq is not None:
        phase5_time = min(200.0, deadline - time.monotonic() - 30)
        if phase5_time > 30:
            print(f"Phase 5 perturbations: budget={phase5_time:.0f}s")
            n_perturb = 5
            time_per = phase5_time / n_perturb - 10
            for i in range(n_perturb):
                if time.monotonic() > deadline - 30:
                    break
                pert_scale = 0.005 * (i + 1)
                a_pert = _random_perturb(best_seq, scale=pert_scale, seed=seed + 1000 + i)
                sc_pert = _score(a_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {i}: {sc_pert:.10f} (improved!)")
                    continue

                # Quick LP on perturbed
                pert_end = time.monotonic() + min(time_per, 30.0)
                a_pert_lp, sc_pert_lp = _lp_refine(a_pert, pert_end,
                                                   inflate_start=1.003, inflate_min=1.0002)
                sc_pert_lp = _score(a_pert_lp)
                if sc_pert_lp < best_score:
                    best_score = sc_pert_lp
                    best_seq = a_pert_lp.copy()
                    print(f"  Perturb {i} + LP: {sc_pert_lp:.10f} (improved!)")
                else:
                    print(f"  Perturb {i} + LP: {sc_pert_lp:.8f}")

    # Phase 6: Final LP squeeze
    if best_seq is not None:
        phase6_time = min(120.0, deadline - time.monotonic() - 10)
        if phase6_time > 10:
            phase6_end = time.monotonic() + phase6_time
            print(f"Phase 6 final squeeze: budget={phase6_time:.0f}s")
            a_final, sc_final = _lp_refine(best_seq, phase6_end,
                                           inflate_start=1.001, inflate_min=1.00005,
                                           max_iter=100000)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  -> {sc_final:.10f}")

    # Phase 7: Final gradient polish
    if best_seq is not None:
        phase7_time = min(60.0, deadline - time.monotonic() - 5)
        if phase7_time > 5:
            phase7_end = time.monotonic() + phase7_time
            a_g, sc_g = _gradient_polish(best_seq, phase7_end, max_iter=50000)
            sc_g = _score(a_g)
            if sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"Phase 7 final gradient: {sc_g:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_init(512, 'edges')
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_init(512, 'edges')

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")

    # Validate with evaluate_sequence if possible
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    return [float(x) for x in final_seq]
