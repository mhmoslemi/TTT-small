```python
"""
Search algorithm for minimizing the step-function convolution score.

Method:
- Work on the simplex using softmax coordinates y.
- Smooth the max-convolution term with a soft-max and anneal it while
  running L-BFGS-B in y-space.
- Start from several diverse non-palindromic initializations, including
  anti-palindromic sine patterns that break the all-ones symmetry.
- Polish with multiplicative / natural-gradient style updates, random
  block perturbations, and short additional L-BFGS stages.
"""

import numpy as np
import time
import scipy.optimize as so

_Y_LOW = -40.0
_Y_HIGH = 40.0


def softmax(y):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return y.copy()
    if not np.all(np.isfinite(y)):
        return np.ones_like(y, dtype=np.float64) / y.size
    m = float(np.max(y))
    if not np.isfinite(m):
        return np.ones_like(y, dtype=np.float64) / y.size
    e = np.exp(y - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(y, dtype=np.float64) / y.size
    return e / s


def fft_convolve(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.array([], dtype=np.float64)
    target = 2 * n - 1
    L = 1
    while L < target:
        L *= 2
    F = np.fft.rfft(f, n=L)
    out = np.fft.irfft(F * F, n=L)
    return out[:target]


def soft_max_and_p(b, tau):
    b = np.asarray(b, dtype=np.float64)
    tau = float(tau)
    if not np.isfinite(tau) or tau <= 0.0:
        tau = 1e-14
    m = float(np.max(b))
    if not np.isfinite(m):
        return 0.0, np.full_like(b, 1.0 / b.size, dtype=np.float64)
    shifted = (b - m) / tau
    e = np.exp(shifted)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(b, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
        return m, p
    soft = m + tau * float(np.log(s))
    p = e / s
    return soft, p


def xcorr(f, p):
    f = np.asarray(f, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.array([], dtype=np.float64)
    f_rev = f[::-1]
    target = n + p.size - 1
    L = 1
    while L < target:
        L *= 2
    F = np.fft.rfft(f_rev, n=L)
    P = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(F * P, n=L)
    return conv[n - 1:n - 1 + n]


def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.inf
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    b = fft_convolve(f)
    if b.size == 0:
        return np.inf
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * f.size * mb / (s * s))


def normalize_f(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(f, dtype=np.float64) / max(1, f.size)
    return f / s


def smooth_obj_y(y, tau, reg):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0 or not np.all(np.isfinite(y)):
        return 1e100
    f = softmax(y)
    b = fft_convolve(f)
    sm, _ = soft_max_and_p(b, tau)
    val = 2.0 * y.size * sm
    if reg > 0.0:
        val += 0.5 * reg * float(np.dot(y, y))
    if not np.isfinite(val):
        return 1e100
    return float(val)


def smooth_grad_y(y, tau, reg):
    y = np.asarray(y, dtype=np.float64)
    n = y.size
    if n == 0 or not np.all(np.isfinite(y)):
        return np.zeros_like(y, dtype=np.float64)
    f = softmax(y)
    b = fft_convolve(f)
    _, p = soft_max_and_p(b, tau)
    R = xcorr(f, p)
    avg_R = float(np.dot(f, R))
    grad = 4.0 * n * f * (R - avg_R)
    if reg > 0.0:
        grad = grad + reg * y
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(y, dtype=np.float64)
    return grad


def run_lbfgs_stage(start_y, tau, reg, bounds, maxiter):
    if start_y is None:
        return None
    start_y = np.clip(np.asarray(start_y, dtype=np.float64), _Y_LOW, _Y_HIGH)
    if not np.all(np.isfinite(start_y)):
        return None
    try:
        res = so.minimize(
            smooth_obj_y,
            start_y,
            method="L-BFGS-B",
            jac=smooth_grad_y,
            args=(tau, reg),
            bounds=bounds,
            options={
                "maxiter": int(max(1, maxiter)),
                "ftol": 1e-10,
                "gtol": 1e-8,
                "maxls": 10,
                "maxcor": 10,
            },
        )
    except Exception:
        return None
    if res is not None and hasattr(res, "x"):
        x = np.asarray(res.x, dtype=np.float64)
        if x.size == start_y.size and np.all(np.isfinite(x)):
            return np.clip(x, _Y_LOW, _Y_HIGH)
    return None


def f_to_y(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.maximum(f, 1e-12)
    y = np.log(f)
    y -= float(np.mean(y))
    return np.clip(y, _Y_LOW, _Y_HIGH)


def make_inits(n, rng, parent):
    inits = []
    idx = np.arange(n, dtype=np.float64)
    t = (idx + 0.5) / float(n)

    anti1 = np.sin(2.0 * np.pi * (idx + 0.5) / float(n))
    anti2 = np.sin(4.0 * np.pi * (idx + 0.5) / float(n))
    anti3 = np.sin(6.0 * np.pi * (idx + 0.5) / float(n))

    parent_arr = None
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and p.size == n:
            p = np.maximum(p, 0.0)
            if np.all(np.isfinite(p)) and float(np.sum(p)) > 0.0:
                p = p / float(np.sum(p))
                if float(np.std(p)) > 1e-8:
                    parent_arr = p
                else:
                    parent_arr = None

    if parent_arr is not None:
        y = f_to_y(parent_arr) + rng.normal(0.0, 0.03, size=n)
        y -= float(np.mean(y))
        inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Small anti-palindromic kick.
    y = 0.30 * anti1 + rng.normal(0.0, 0.02, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Strong anti-palindromic kick with a second harmonic.
    y = 1.0 * anti1 + 0.25 * anti2 + rng.normal(0.0, 0.05, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Very strong anti-palindromic kick.
    y = 2.0 * anti1 + 0.5 * anti3 + rng.normal(0.0, 0.10, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Generic Gaussian noise in y-space.
    for scale in (0.2, 0.8, 1.5):
        y = rng.normal(0.0, scale, size=n)
        y -= float(np.mean(y))
        inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Gaussian bell in f-space.
    h = np.exp(-0.5 * ((t - 0.5) / 0.25) ** 2)
    h = np.maximum(h, 1e-12)
    h = h / float(np.sum(h))
    y = f_to_y(h) + rng.normal(0.0, 0.05, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Sqrt-bell, vanishing at both ends.
    h = np.sqrt(np.maximum(t * (1.0 - t), 1e-10))
    h = h / float(np.sum(h))
    y = f_to_y(h) + rng.normal(0.0, 0.05, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # U-shaped cosine in f-space.
    h = 1.0 + 0.45 * np.cos(np.pi * t)
    h = np.maximum(h, 1e-8)
    h = h / float(np.sum(h))
    y = f_to_y(h) + rng.normal(0.0, 0.05, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    # Random positive weights.
    h = rng.random(n) + 0.05
    h = h / float(np.sum(h))
    y = f_to_y(h) + rng.normal(0.0, 0.05, size=n)
    y -= float(np.mean(y))
    inits.append(np.clip(y, _Y_LOW, _Y_HIGH))

    return inits[:10]


def weighted_mult_step(f, frac, eta):
    g = normalize_f(f)
    b = fft_convolve(g)
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return g
    scale = max(1.0, abs(m))
    if frac <= 0.0:
        mask = b >= (m - 1e-12 * scale)
    else:
        mask = b >= m * (1.0 - min(float(frac), 0.5))
    if not np.any(mask):
        mask = np.zeros_like(b, dtype=bool)
        mask[int(np.argmax(b))] = True
    p = np.zeros_like(b, dtype=np.float64)
    p[mask] = 1.0 / float(np.sum(mask))
    R = xcorr(g, p)
    cmax = float(np.max(R))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return g
    new = g * np.exp(-float(max(0.0, eta)) * R / cmax)
    new = np.maximum(new, 0.0)
    s = float(np.sum(new))
    if not np.isfinite(s) or s <= 0.0:
        return g
    return new / s


def soft_mult_step(f, rel, eta):
    g = normalize_f(f)
    b = fft_convolve(g)
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return g
    tau = max(m / max(float(rel), 1e-9), 1e-18)
    _, p = soft_max_and_p(b, tau)
    R = xcorr(g, p)
    cmax = float(np.max(R))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return g
    new = g * np.exp(-float(max(0.0, eta)) * R / cmax)
    new = np.maximum(new, 0.0)
    s = float(np.sum(new))
    if not np.isfinite(s) or s <= 0.0:
        return g
    return new / s


def global_noise_step(f, amp, rng):
    g = np.asarray(f, dtype=np.float64).copy()
    g = g * (1.0 + float(amp) * rng.normal(0.0, 1.0, size=g.size))
    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if not np.isfinite(s) or s <= 0.0:
        return normalize_f(f)
    return g / s


def random_block_step(f, rng):
    g = np.asarray(f, dtype=np.float64).copy()
    n = g.size
    low = max(1, n // 200)
    high = max(low + 1, n // 8)
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    g[start:stop] *= float(np.exp(rng.normal(0.0, 0.35, size=1)[0]))

    if rng.random() < 0.30:
        g = 0.5 * (g + g[::-1])

    if rng.random() < 0.25:
        g = g * np.exp(rng.normal(0.0, 0.04, size=n))

    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(g, dtype=np.float64) / max(1, n)
    return g / s


def random_y_step(f, rng):
    y = f_to_y(f)
    n = y.size
    idx = np.arange(n, dtype=np.float64)
    m = int(rng.integers(1, 4))
    amp = 0.4 + 1.6 * float(rng.random())
    pattern = np.sin(2.0 * np.pi * m * (idx + 0.5) / float(n))
    sign = 1.0 if rng.random() < 0.5 else -1.0
    y = y + sign * amp * pattern
    y = y + 0.25 * amp * rng.normal(0.0, 1.0, size=n)
    y -= float(np.mean(y))
    return softmax(np.clip(y, _Y_LOW, _Y_HIGH))


def refine_f(f, hard_deadline, rng):

    best_s = fast_score(best)
    if not np.isfinite(best_s):

    fracs = (0.0, 0.001, 0.005, 0.02, 0.08)

    cycles = 0
    no_improve = 0

    while cycles < 200000 and time.monotonic() < hard_deadline:
        method = int(rng.integers(0, 4))
        if method == 0:
            frac = float(fracs[int(rng.integers(0, len(fracs)))])
            eta = float(etas[int(rng.integers(0, len(etas)))])
            cand = weighted_mult_step(best, frac, eta)
        elif method == 1:
            rel = float(soft_rels[int(rng.integers(0, len(soft_rels)))])
            eta = float(etas[int(rng.integers(0, len(etas)))])
            cand = soft_mult_step(best, rel, eta)
        elif method == 2:
            if rng.random() < 0.55:
                cand = random_block_step(best, rng)
            else:
                cand = global_noise_step(best, 0.03 + 0.25 * float(rng.random()), rng)
        else:
            cand = random_y_step(best, rng)

        s = fast_score(cand)
        if np.isfinite(s) and s < best_s - 1e-13:
            best = cand
            best_s = s
            no_improve = 0
        else:
            no_improve += 1

        cycles += 1

        if no_improve == 100:
            cand = random_y_step(best, rng)
            s = fast_score(cand)
            if np.isfinite(s) and s < best_s - 1e-13:
                best = cand
                best_s = s
                no_improve = 0
            else:
                no_improve += 50

        if no_improve > 500:
            break

    return best, best_s


def optimize_y_candidate(y0, hard_deadline, rng, bounds, rels=None, jitter=0.15):
    if y0 is None:
        return None
    n = len(y0)
    if rels is None:
        rels = (50.0, 20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25)

    y = np.clip(np.asarray(y0, dtype=np.float64).copy(), _Y_LOW, _Y_HIGH)
    y -= float(np.mean(y))
    y = np.clip(y, _Y_LOW, _Y_HIGH)

    y_best = y.copy()
    f_best = softmax(y_best)
    s_best = fast_score(f_best)
    y_cur_s = s_best

    step_no = 0

    for rel in rels:
        if time.monotonic() > hard_deadline - 8.0:
            break

        f_cur = softmax(y)
        bmax = float(np.max(fft_convolve(f_cur)))
        tau = max(bmax / float(max(rel, 1e-9)), 1e-14)

        rem = hard_deadline - time.monotonic()
        if rem > 240.0:
            maxiter = 500
        elif rem > 120.0:
            maxiter = 250
        elif rem > 60.0:
            maxiter = 120
        else:
            maxiter = 50

        cand_y = run_lbfgs_stage(y, tau, reg, bounds, maxiter)
        if cand_y is not None:
            cand_y -= float(np.mean(cand_y))
            cand_y = np.clip(cand_y, _Y_LOW, _Y_HIGH)
            cand_f = softmax(cand_y)
            cand_s = fast_score(cand_f)
            if np.isfinite(cand_s):
                if cand_s < y_cur_s - 1e-13:
                    y = cand_y.copy()
                    y_cur_s = cand_s
                if cand_s < s_best - 1e-13:
                    y_best = cand_y.copy()
                    f_best = cand_f.copy()
                    s_best = cand_s

        if step_no % 2 == 0 and jitter > 0.0 and time.monotonic() < hard_deadline - 30.0:
            pert = y + rng.normal(0.0, float(jitter), size=n)
            pert -= float(np.mean(pert))
            pert = np.clip(pert, _Y_LOW, _Y_HIGH)
            cand_y2 = run_lbfgs_stage(pert, tau, reg, bounds, max(30, maxiter // 2))
            if cand_y2 is not None:
                cand_y2 -= float(np.mean(cand_y2))
                cand_y2 = np.clip(cand_y2, _Y_LOW, _Y_HIGH)
                cand_f2 = softmax(cand_y2)
                cand_s2 = fast_score(cand_f2)
                if np.isfinite(cand_s2):
                    if cand_s2 < y_cur_s - 1e-13:
                        y = cand_y2.copy()
                        y_cur_s = cand_s2
                    if cand_s2 < s_best - 1e-13:
                        y_best = cand_y2.copy()
                        f_best = cand_f2.copy()
                        s_best = cand_s2

        step_no += 1

    return y_best, f_best, s_best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    if budget_s is None:
        budget_s = 1800.0
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p_raw = height_sequence_1
        p_arr = np.asarray(p_raw, dtype=np.float64)
        if p_arr.ndim != 1:
            p_arr = p_arr.reshape(-1)
        if p_arr.size > 0 and np.all(np.isfinite(p_arr)):
            parent = p_arr
    except Exception:
        parent = None

    if parent is not None and 64 <= parent.size <= 8000:
        n = int(parent.size)
    elif parent is not None and parent.size > 8000:
        n = 4096
    else:
        n = 3809

    inits = make_inits(n, rng, parent)
    if not inits:
        inits = [np.zeros(n, dtype=np.float64)]
    inits = inits[:10]

    bounds = [(_Y_LOW, _Y_HIGH)] * n

    best_f = np.ones(n, dtype=np.float64) / float(n)
    best_s = fast_score(best_f)

    if parent is not None and parent.size == n:
        pf = normalize_f(parent)
        pf_s = fast_score(pf)
        if np.isfinite(pf_s) and pf_s < best_s:
            best_f = pf
            best_s = pf_s

    print(f"n={n} start_score={best_s:.10f} initial_candidates={len(inits)}", flush=True)

    reserve = 450.0

    for idx, y0 in enumerate(inits):
        if time.monotonic() > deadline - reserve:
            break

        now = time.monotonic()
        remaining = deadline - reserve - now
        left = len(inits) - idx
        if remaining <= 0.0:
            break

        slice_s = min(300.0, max(80.0, remaining / max(1, left)))
        hard = now + slice_s
        print(f"candidate {idx + 1}/{len(inits)} slice={slice_s:.1f}s", flush=True)

        res = optimize_y_candidate(y0, hard, rng, bounds, rels=None, jitter=0.15)
        if res is not None:
            _, f_c, s_c = res
            if f_c is not None and np.isfinite(s_c) and s_c < best_s:
                best_f = f_c
                best_s = s_c
                print(f"  new best={s_c:.10f}", flush=True)

        if best_s < 1.55:
            reserve = max(200.0, 0.5 * reserve)
            break
        if best_s < 1.65:
            reserve = max(250.0, 0.75 * reserve)

    if time.monotonic() < deadline - 200.0:
        y_best = f_to_y(best_f)
        extra = min(300.0, deadline - time.monotonic() - 200.0)
        hard = time.monotonic() + max(0.0, extra)
        print(f"polishing L-BFGS, time_left={deadline - time.monotonic():.1f}s", flush=True)
        res = optimize_y_candidate(
            y_best,
            hard,
            rng,
            bounds,
            rels=(2.0, 1.0, 0.6, 0.4, 0.25, 0.15),
            jitter=0.08,
        )
        if res is not None:
            _, f_c, s_c = res
            if f_c is not None and np.isfinite(s_c) and s_c < best_s:
                best_f = f_c
                best_s = s_c
                print(f"  polish new best={s_c:.10f}", flush=True)

    if time.monotonic() < deadline - 80.0:
        print(f"refining, time_left={deadline - time.monotonic():.1f}s", flush=True)
        best_f, best_s = refine_f(best_f, deadline - 80.0, rng)
        print(f"after refine score={best_s:.10f}", flush=True)

    if time.monotonic() < deadline - 40.0:
        y_best = f_to_y(best_f)
        extra = min(120.0, deadline - time.monotonic() - 40.0)
        hard = time.monotonic() + max(0.0, extra)
        print(f"final L-BFGS stage, time_left={deadline - time.monotonic():.1f}s", flush=True)
        res = optimize_y_candidate(
            y_best,
            hard,
            rng,
            bounds,
            rels=(1.0, 0.6, 0.4, 0.25),
            jitter=0.05,
        )
        if res is not None:
            _, f_c, s_c = res
            if f_c is not None and np.isfinite(s_c) and s_c < best_s:
                best_f = f_c
                best_s = s_c
                print(f"  final L-BFGS new best={s_c:.10f}", flush=True)

    if time.monotonic() < deadline - 5.0:
        best_f, best_s = refine_f(best_f, deadline - 5.0, rng)

    final_seq = [float(x) for x in best_f]

    try:
        eval_score = float(evaluate_sequence(final_seq))
    except Exception:
        eval_score = float(best_s)

    print(
        f"internal_best={best_s:.10f} evaluated={eval_score:.10f} "
        f"n={n} elapsed={time.monotonic() - t0:.1f}s",
        flush=True,
    )

    return final_seq
```
