```python
import time

def _compute_smoothed_objective(a, tau):
    """Compute smoothed log-sum-exp objective and gradient."""
    n = len(a)
    r = np.convolve(a, a, mode='full')
    r_max = r.max()
    # Smoothed objective
    r_shifted = tau * (r - r_max)
    p = np.exp(r_shifted)
    p_sum = p.sum()
    obj_smooth = (r_max + np.log(p_sum)) / tau
    # Gradient: g[i] = 2 * sum_m p[i+m] * a[m] / p_sum
    a_rev = a[::-1]
    corr = np.correlate(p, a_rev, mode='full')
    g = 2.0 * corr[n-1 : n-1+n] / p_sum
    return obj_smooth, g, r

def _project_simplex(v):
    """Project onto {x : x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s

def _fista_smoothed(a0, tau, lr, max_iter, deadline, rng):
    """FISTA with log-sum-exp smoothing."""
    n = len(a0)

    y = a0.copy()

        y = np.maximum(y, 0.0)
        y = y / y.sum()
        t = t_new
        a = a_new
```

$g[i] = 2 \sum_{m=0}^{n-1} p[i+m] a[m]$

This is the $i$-th element of the cross-correlation of $p$ and $a$. I can compute the entire cross-correlation using FFT in $O(n \log n)$:

```python
def _compute_gradient_fft(p, a):
    n = len(a)
    L = 1
    while L < 3*n:
        L *= 2
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    # Cross-correlation: sum_m p[m+k] a[m] for k = 0, 1, ...
    # This is IFFT(conj(FFT(a)) * FFT(p)) with appropriate indexing
    corr = np.fft.irfft(np.conj(A) * P, L)
    # corr[k] = sum_m a[m] p[m+k] (for the linear/correlation convention)
    # We want g[i] = 2 * corr[i] for i = 0, ..., n-1
    g = 2.0 * corr[:n]
    return g
```

Let me verify the convention. For DFT:
If $A[k] = \sum_m a[m] e^{-2\pi i km/L}$ and $P[k] = \sum_m p[m] e^{-2\pi i km/L}$, then:
$\text{IFFT}(\overline{A[k]} \cdot P[k]) = \sum_m a[m] \cdot p[m+k] \pmod{L}$

(This is the circular cross-correlation.) With sufficient zero-padding (L >= 3n-2), the circular and linear correlations agree for the range we care about.

So `corr[k] = sum_m a[m] p[m+k]` (linear, for valid indices).

And we want $g[i] = 2 \sum_m a[m] p[m+i] = 2 \cdot \text{corr}[i]$.

def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

def _smoothed_objective_and_grad(a, tau):
    """Compute smoothed log-sum-exp objective and its gradient."""
    n = len(a)
    r = np.convolve(a, a, mode='full')  # length 2n-1
    r_max = r.max()
    # For numerical stability, subtract r_max before exp
    exponents = tau * (r - r_max)
    # Clip to avoid overflow
    exponents = np.minimum(exponents, 500)
    weights = np.exp(exponents)
    w_sum = weights.sum()
    
    # Smoothed objective (this is an upper bound on max(r))
    obj = r_max + np.log(w_sum) / tau
    
    # Gradient
    p = weights / w_sum  # softmax weights
    L = 1
    while L < 3*n:
        L *= 2
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    corr = np.fft.irfft(np.conj(A) * P, L)
    g = 2.0 * corr[:n]
    
    return obj, g, r
