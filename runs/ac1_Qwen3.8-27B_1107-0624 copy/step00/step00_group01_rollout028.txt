    n = len(a)

    max_b = np.max(conv)
    sum_a = np.sum(a)

        return np.inf

    """Compute softmax weights over lags."""
    # c has length n (lags 0 to n-1)
    # For numerical stability, subtract max
    c_max = np.max(c)
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / np.sum(exp_c)

    if w_sum < 1e-15:
        return b
    return w / w_sum

def lp_refinement(best, n, S, verbose=False):
    """LP-based refinement: find g maximizing sum(g) subject to (best*g)_k <= M."""
    M = np.max(np.convolve(best, best))
    
    # Build constraint matrix (sparse)
    # Row k: sum_i best[i] * g[k-i] <= M for i in [max(0,k-n+1), min(n-1,k)]
    A = lil_matrix((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[k, k - i] = best[i]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    res = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 120})
    dt = time.monotonic() - t0
    
    if res.success and res.x is not None and res.x.sum() > 0:
        g = np.clip(res.x, 0, None)
        g = g * S / g.sum()
        if verbose:
            print(f"  LP refined in {dt:.1f}s, new score: {compute_score_fast(g):.6f}")
        return g
    return None

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2n*max(autocorr)/sum^2 for non-negative sequences using:
    1. Mirror descent (multiplicative weights) with log-sum-exp smoothing
    2. Periodic LP-based refinement
    3. Multiple restarts with perturbation when stuck
    
    The score is scale-invariant, so we fix sum(a)=sqrt(2n) and work in
    normalized space where sum(b)=1.
    """
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(parent)
    S = np.sqrt(2.0 * n)
    
    # --- Initialization: try several starts ---
    starts = []
    # 1. Given sequence
    starts.append(parent * S / np.sum(parent))
    # 2. Random
    starts.append(rng.random(n) * S / np.sum(rng.random(n)))
    # 3. Concentrated in middle (trapezoidal)
    trap = np.zeros(n)
    plateau = n // 3
    ramp = n // 6
    trap[n//2 - plateau//2 : n//2 + plateau//2 + 1] = 1.0
    for i in range(ramp):
        frac = (i + 1) / (ramp + 1)
        start_idx = n//2 - plateau//2 - 1 - i
        end_idx = n//2 + plateau//2 + i
        if 0 <= start_idx < n:
            trap[start_idx] = frac
        if 0 <= end_idx < n:
            trap[end_idx] = frac
    starts.append(trap * S / trap.sum())
    
    best = starts[0].copy()
    best_score = compute_score_fast(best)
    for s in starts[1:]:
        sc = compute_score_fast(s)
        if sc < best_score:
            best_score = sc
            best = s.copy()
    
    print(f"Initial: n={n}, best_score={best_score:.6f}")
    
    # --- Mirror descent ---
    b = best / S  # normalized, sum(b) = 1
    
    tau = 5.0
    tau_max = 80.0
    tau_rate = 1.003
    eta = float(n) * 0.5  # initial step size for mirror descent
    
    last_improve = time.monotonic()
    iter_count = 0
    
    while time.monotonic() < deadline:
        # Compute autocorrelation
        conv_b = np.convolve(b, b)
        c_b = conv_b[:n]
        max_c_b = np.max(conv_b)
        sum_b = b.sum()
        current_score = 2.0 * n * max_c_b / (sum_b ** 2)
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best = (b * S).copy()
            last_improve = time.monotonic()
        
        iter_count += 1
        if iter_count % 5000 == 0:
            print(f"Iter {iter_count}: score={current_score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.1f}, eta={eta:.1f}, gap={time.monotonic()-last_improve:.0f}s")
        
        # Stuck detection: restart with perturbation
        if time.monotonic() - last_improve > 120:
            noise = rng.normal(0, 0.005 * b.mean(), n)
            b = np.clip(b + noise, 0, None)
            b = b / b.sum()
            eta = float(n) * 0.5
            tau = 5.0
            last_improve = time.monotonic()
            print(f"  -> Perturbed & restarted. Score: {2*n*np.max(np.convolve(b,b))/b.sum()**2:.6f}")
            continue
        
        # Softmax weights over lags
        c_max = c_b.max()
        scaled = tau * (c_b - c_max)
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(n)
            p[int(np.argmax(c_b))] = 1.0
        else:
            p /= p_sum
        
        # Gradient: grad[j] = sum_k p[k] * (b[j+k]*[j+k<n] + b[j-k]*[j>=k])
        b_rev = b[::-1]
        forward = np.convolve(p, b_rev)[:n][::-1]
        backward = np.convolve(p, b)[:n]
        grad = forward + backward
        
        # Mirror descent step
        b = mirror_descent_step(b, grad, eta)
        
        # Adapt hyperparameters
        tau = min(tau * tau_rate, tau_max)
        eta *= 0.9995  # very slow decay
        
        # Periodic LP refinement (every ~60 seconds if time allows)
        if iter_count % 20000 == 0 and (deadline - time.monotonic()) > 180:
            print(f"  -> Attempting LP refinement...")
            lp_result = lp_refinement(b * S, n, S, verbose=True)
            if lp_result is not None:
                lp_score = compute_score_fast(lp_result)
                if lp_score < best_score:
                    best_score = lp_score
                    best = lp_result
                    b = lp_result / S
                    print(f"  -> LP improved! New best: {best_score:.6f}")
    
    # Final check
    final_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"Final: best_score={best_score:.6f}, verified={final_score:.6f}")
    return [float(x) for x in best]
