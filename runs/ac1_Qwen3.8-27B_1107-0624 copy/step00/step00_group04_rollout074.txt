```python
"""
Smooth-max mirror descent for nonnegative polynomial-square peak minimization.

The score is scale-invariant:
    score(a) = 2*n*max(conv(a,a)) / sum(a)^2.
We normalize sum(a)=1 and minimize max(conv(a,a)).  A smooth surrogate is
F_tau(a) = tau * log(sum_k exp(c_k/tau)), where c = conv(a,a).  Its gradient
has the form sum_k p_k * a shifted.  We do mirror-descent on the simplex,
annealing tau, with random restarts, jumps, and several parameterizations
(piecewise, sinusoidal, quadratic-phase, smoothed refs).  Best found is
returned; a length-4 construction [1/3,1/3,1/6,1/6] is guaranteed fallback.
"""

import numpy as np
import time


def next_pow2(m):
    m = int(max(1, m))
    return 1 << max(0, (m - 1).bit_length())


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    if floor > 0.0:
        a = np.maximum(a, float(floor))

    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        n = max(1, a.size)
        return np.ones(n, dtype=np.float64) / n

    a = a / s

    if floor > 0.0:
        a = np.maximum(a, float(floor))
        s = float(a.sum())
        if np.isfinite(s) and s > 0.0:
            a = a / s

    return a


def fft_square(a, L):
    n = int(a.size)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    try:
        cmax = float(c.max())
    except Exception:
        cmax = 0.0
    if not np.isfinite(cmax):
        cmax = 0.0
    return c, cmax


def smooth_value(c, tau):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    cmax = float(c.max())
    if not np.isfinite(cmax):
        cmax = 0.0
    x = (c - cmax) / tau
    x = np.clip(x, -700.0, 0.0)
    val = cmax + tau * float(np.log(float(np.exp(x).sum())))
    if not np.isfinite(val):
        val = cmax
    return val


def coeff_weights(c, tau):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(c.max())
    if not np.isfinite(cmax):
        cmax = 0.0

    x = (c - cmax) / tau
    x = np.clip(x, -700.0, 0.0)
    raw = np.exp(x)
    s = float(raw.sum())

    if not np.isfinite(s) or s <= 0.0:
        if c.size > 0:
            raw = np.zeros_like(c, dtype=np.float64)
            raw[int(np.argmax(c))] = 1.0
            s = 1.0
        else:
            raw = np.array([1.0], dtype=np.float64)
            s = 1.0

    p = raw / s
    F = cmax + tau * float(np.log(s))
    if not np.isfinite(F):
        F = cmax
    return p, F, cmax


def cross_corr_a_p(a, p, L):
    """
    Computes h_i = sum_k p_k a_{k-i}, with a_j=0 outside [0,n-1].
    This is the (unnormalized) gradient direction for F_tau.
    """
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    v = a[::-1]
    F1 = np.fft.rfft(v, L)
    F2 = np.fft.rfft(p, L)
    h = np.fft.irfft(F1 * F2, L)[n - 1: 2 * n - 1]
    np.maximum(h, 0.0, out=h)
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    return h


def tile_sequence(ref, n):
    ref = np.asarray(ref, dtype=np.float64).reshape(-1)
    n = int(max(1, n))

    if ref.size == 0:
        return normalize_positive(np.ones(n))

    if ref.size == n:
        return normalize_positive(ref)

    if ref.size > n:
        xs = np.linspace(0.0, 1.0, ref.size)
        ys = ref / max(1.0, float(ref.sum()))
        out = np.interp(np.linspace(0.0, 1.0, n), xs, ys,
                        left=float(ys[0]), right=float(ys[-1]))
        return normalize_positive(np.maximum(out, 0.0))

    reps = (n + ref.size - 1) // ref.size
    out = np.tile(ref, reps)[:n]
    return normalize_positive(out)


def make_initial(n, rng, mode, ref_a=None):
    n = int(max(1, n))
    if n < 4:
        return normalize_positive(np.ones(n))

    two_pi = 2.0 * np.pi
    i = np.arange(n, dtype=np.float64)

    if mode == 0:
        a = np.ones(n, dtype=np.float64)

    elif mode == 1:

        a = np.exp(rng.normal(0.0, sigma, n))

    elif mode == 2:

        k = int(rng.integers(2, max_seg + 1))
        cuts = rng.choice(np.arange(1, n), size=k - 1, replace=False)
        cuts = np.sort(cuts)
        vals = np.exp(rng.normal(0.0, 1.0, k))
        a = np.empty(n, dtype=np.float64)
        prev = 0
        for idx in range(k - 1):
            a[prev:cuts[idx]] = vals[idx]
            prev = cuts[idx]
        a[prev:] = vals[-1]

    elif mode == 3:
        a = np.ones(n, dtype=np.float64)
        n1 = max(2, n // 16)
        f1 = int(rng.integers(1, n1 + 1))
        ph1 = float(rng.uniform(0.0, two_pi))
        a += float(rng.uniform(0.3, 0.8)) * np.cos(two_pi * f1 * i / n + ph1)

        n2 = max(2, n // 4)
        f2 = int(rng.integers(1, n2 + 1))
        ph2 = float(rng.uniform(0.0, two_pi))
        a += float(rng.uniform(0.2, 0.7)) * np.cos(two_pi * f2 * i / n + ph2)

        ph3 = float(rng.uniform(0.0, two_pi))
        a += float(rng.uniform(0.2, 0.9)) * np.cos(np.pi * i / n + ph3)

    elif mode == 4:
        if ref_a is not None and np.size(ref_a) > 0:
            return tile_sequence(ref_a, n)
        a = np.exp(rng.normal(0.0, 1.0, n))

    elif mode == 5:
        if ref_a is not None and np.size(ref_a) > 0:
            base = tile_sequence(ref_a, n)
        else:
            base = np.exp(rng.normal(0.0, 1.0, n))
        sigma = float(rng.uniform(0.2, 0.9))
        a = base * np.exp(rng.normal(0.0, sigma, n))

    elif mode == 6:
        # Quadratic-phase positive part; useful chirp-like starts.
        amp = float(rng.uniform(0.5, 1.0))
        ph = float(rng.uniform(0.0, two_pi))
        a = 1.0 + amp * np.cos(two_pi * i * i / n + ph)
        ph2 = float(rng.uniform(0.0, two_pi))
        a += float(rng.uniform(0.1, 0.5)) * np.cos(np.pi * i / n + ph2)

    elif mode == 7:
        x = i / float(n)
        k = int(rng.integers(2, 5))
        centers = rng.uniform(0.0, 1.0, k)
        widths = rng.uniform(0.02, 0.2, k)
        weights = np.exp(rng.normal(0.0, 1.0, k))
        a = np.zeros(n, dtype=np.float64)
        for c0, w0, v0 in zip(centers, widths, weights):
            a += v0 * np.exp(-((x - c0) / (2.0 * w0)) ** 2)

    else:
        a = np.ones(n, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 1.0)
    a = np.maximum(a, 1e-3)
    return normalize_positive(a, floor=1e-12)


def safe_evaluate(vec):
    if "evaluate_sequence" not in globals():
        return float("inf")
    try:
        val = evaluate_sequence(vec)
        if val is None:
            return float("inf")
        val = float(val)
        if not np.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def run_optimization_budget(n, end_time, rng, best_a, best_score, best_C):
    n = int(n)
    if n < 4 or end_time <= time.monotonic():
        return best_a, best_score, best_C

    Lc = next_pow2(2 * n - 1)
    Lh = next_pow2(3 * n - 2)

    len_best_a = None
    len_best_c = float("inf")
    run_count = 0
    last_print_score = float(best_score)

    if n <= 32:
        base_run = 12.0
    elif n <= 128:
        base_run = 25.0
    elif n <= 512:
        base_run = 40.0
    elif n <= 2048:
        base_run = 55.0
    else:
        base_run = 75.0

    while time.monotonic() < end_time:
        remaining = end_time - time.monotonic()
        if remaining < 3.0:
            break

        run_time = min(base_run, remaining)
        ref = len_best_a if len_best_a is not None else best_a

        if run_count == 0:
            mode = 4 if ref is not None else 1
        else:
            r = float(rng.random())
            if r < 0.22 and ref is not None:
                mode = 4
            elif r < 0.40 and ref is not None:
                mode = 5
            elif r < 0.58:
                mode = 2
            elif r < 0.78:
                mode = 1
            elif r < 0.88:
                mode = 3
            elif r < 0.94:
                mode = 6
            else:
                mode = 7

        a = make_initial(n, rng, mode, ref)

        run_start = time.monotonic()
        run_deadline = min(end_time, run_start + run_time)
        eta = 1.0
        jumps = 0

        c, cmax = fft_square(a, Lc)
        if cmax < len_best_c:
            len_best_c = cmax
            len_best_a = a.copy()

        score = 2.0 * n * cmax
        if score < best_score - 1e-12:
            best_a = a.copy()
            best_score = float(score)
            best_C = float(n * cmax)
            if best_score < last_print_score - 1e-5:
                print(f"new best n={n} score={best_score:.6f} C={best_C:.6f}")
                last_print_score = best_score

        last_cmax_improve = time.monotonic()

        while time.monotonic() < run_deadline:
            now = time.monotonic()
            if now >= run_deadline:
                break

            progress = (now - run_start) / max(1e-9, run_deadline - run_start)
            progress = min(1.0, max(0.0, progress))

            # Anneal from a broad peak mixture to a sharp one.
            tau_factor = 0.8 * (0.03 / 0.8) ** progress
            tau = max(1e-15, tau_factor * max(float(cmax), 1e-15))

            p, F, cmax = coeff_weights(c, tau)
            if not np.isfinite(F):
                break

            h = cross_corr_a_p(a, p, Lh)
            hmax = float(h.max())
            if not np.isfinite(hmax) or hmax <= 1e-300:
                break

            d = h / hmax
            etas = [max(0.05, eta * 0.3), float(eta), min(10.0, eta * 3.0)]

            bestA = None
            bestC = None
            bestF = float(F)
            bestCM = float(cmax)
            bestIdx = -1

            for idx in range(3):
                et = float(etas[idx])
                an = a * np.exp(-et * d)
                an = normalize_positive(an, floor=1e-12)

                cn, cnmax = fft_square(an, Lc)
                fn = smooth_value(cn, tau)

                if fn < bestF - 1e-15:
                    bestF = fn
                    bestA = an
                    bestC = cn
                    bestCM = cnmax
                    bestIdx = idx
                elif abs(fn - bestF) <= 1e-15 and cnmax < bestCM * (1.0 - 1e-12):
                    bestF = fn
                    bestA = an
                    bestC = cn
                    bestCM = cnmax
                    bestIdx = idx

            if bestA is not None and (bestF < F - 1e-15 or bestCM < cmax * (1.0 - 1e-10)):
                if bestIdx == 0:
                    eta = max(0.05, eta * 0.8)
                elif bestIdx == 2:
                    eta = min(10.0, eta * 1.25)

                a = bestA
                c = bestC
                cmax = bestCM

                improved = False
                if cmax < len_best_c * (1.0 - 1e-12):
                    len_best_c = cmax
                    len_best_a = a.copy()
                    improved = True

                score = 2.0 * n * cmax
                if score < best_score - 1e-12:
                    best_a = a.copy()
                    best_score = float(score)
                    best_C = float(n * cmax)
                    improved = True
                    if best_score < last_print_score - 1e-5:
                        print(f"new best n={n} score={best_score:.6f} C={best_C:.6f}")
                        last_print_score = best_score

                if improved:
                    last_cmax_improve = time.monotonic()
            else:
                now = time.monotonic()
                if now - last_cmax_improve > 7.0:
                    mix = float(rng.uniform(0.05, 0.35))
                    sigma = float(rng.uniform(0.3, 1.0))
                    a = normalize_positive((1.0 - mix) * a + mix * (np.ones(n) / n),
                                           floor=1e-12)
                    a = normalize_positive(a * np.exp(rng.normal(0.0, sigma, n)),
                                           floor=1e-12)
                    jumps += 1

                    c, cmax = fft_square(a, Lc)
                    if cmax < len_best_c * (1.0 - 1e-12):
                        len_best_c = cmax
                        len_best_a = a.copy()
                        last_cmax_improve = now

                    if jumps >= 3 and now - last_cmax_improve > 12.0:
                        break
                else:
                    if eta > 0.06:
                        eta = max(0.05, eta * 0.6)

        run_count += 1

    return best_a, best_score, best_C


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1800.0

    rng = np.random.default_rng(seed)

    # Guaranteed sub-2 fallback.
    best_a = np.array([1.0 / 3.0, 1.0 / 3.0, 1.0 / 6.0, 1.0 / 6.0],
                      dtype=np.float64)
    best_score = 16.0 / 9.0
    best_C = 8.0 / 9.0

    # A few harmless deterministic sanity candidates.
    sanity_candidates = [
        [1.0 / 6.0, 1.0 / 6.0, 1.0 / 3.0, 1.0 / 3.0],
        [2.0, 2.0, 1.0, 1.0, 1.0, 1.0, 0.5, 0.5],
        [4.0, 3.0, 2.0, 2.0, 1.0, 1.0, 1.0, 1.0],
    ]
    for cand in sanity_candidates:
        try:
            arr = normalize_positive(np.asarray(cand, dtype=np.float64))
            if arr.size >= 1:
                c, cm = fft_square(arr, next_pow2(2 * arr.size - 1))
                s = 2.0 * arr.size * cm
                if s < best_score - 1e-12:
                    best_a = arr
                    best_score = float(s)
                    best_C = float(arr.size * cm)
        except Exception:
            pass

    # Use provided previous construction if it is valid.
    h = globals().get("height_sequence_1")
    if h is not None:
        try:
            arr = normalize_positive(np.asarray(h, dtype=np.float64))
            if arr.size >= 1:
                c, cm = fft_square(arr, next_pow2(2 * arr.size - 1))
                s = 2.0 * arr.size * cm
                if s < best_score - 1e-12:
                    best_a = arr
                    best_score = float(s)
                    best_C = float(arr.size * cm)
        except Exception:
            pass

    deadline = start_time + max(1.0, budget - 5.0)
    work = max(1.0, budget - 30.0)

    configs = [
        (16, 0.01),
        (32, 0.02),
        (64, 0.02),
        (128, 0.04),
        (256, 0.08),
        (293, 0.05),
        (512, 0.27),
        (1024, 0.36),
        (3809, 0.15),
    ]

    for n_alloc, frac in configs:
        sec = work * frac
        if sec < 3.0:
            continue
        end = min(deadline, time.monotonic() + sec)
        best_a, best_score, best_C = run_optimization_budget(
            int(n_alloc), end, rng, best_a, best_score, best_C
        )
        if time.monotonic() >= deadline:
            break

    # Use any remaining time to polish the best successful length.
    while time.monotonic() < deadline:
        n_pol = 1024
        if best_a is not None:
            sz = int(np.size(best_a))
            if 16 <= sz <= 4096:
                n_pol = sz

        end = min(deadline, time.monotonic() + 30.0)
        best_a, best_score, best_C = run_optimization_budget(
            n_pol, end, rng, best_a, best_score, best_C
        )
        if time.monotonic() + 5.0 >= deadline:
            break

    try:
        seq = [float(x) for x in np.asarray(best_a, dtype=np.float64).tolist()]
    except Exception:
        seq = [1.0 / 3.0, 1.0 / 3.0, 1.0 / 6.0, 1.0 / 6.0]

    val = safe_evaluate(seq)
    if not np.isfinite(val) or val > 2.0:
        seq = [1.0 / 3.0, 1.0 / 3.0, 1.0 / 6.0, 1.0 / 6.0]
        val = safe_evaluate(seq)

    try:
        print(f"final safe score={val:.6f} len={len(seq)}")
    except Exception:
        pass

    return seq
```
