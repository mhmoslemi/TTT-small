```python
import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def compute_convolution(f):
    """Compute the full convolution of f with itself."""
    return np.convolve(f, f)

def objective(f):
    """Compute 2*n*max(f*f)/(sum f)^2."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = max(conv)
    sum_a = sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a ** 2)

def lp_direction(f, M):
    """Find the LP direction to improve f."""
    n = len(f)
    # Build constraint matrix A where A[k, i] = f[k-i] if 0 <= k-i < n
    # Constraint: A @ g <= M * ones(2n-1)
    # Maximize: ones(n) @ g
    # Subject to: g >= 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def optimize_n(n, time_budget):
    """Optimize for a given n."""
    f = np.ones(n) / n
    best_f = f.copy()
    best_val = objective(f)
    
    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = max(conv)
        k_star = np.argmax(conv)
        
        # LP direction
        g = lp_direction(f, M)
        if g is None:
            break
        if sum(g) < 1e-10:
            break
            
        g_normalized = g / sum(g)
        
        # Line search for step size
        best_t = 0
        best_obj = objective(f)
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1-t)*f + t*g_normalized
            f_trial = np.clip(f_trial, 0, None)
            if sum(f_trial) < 0.01:
                continue
            obj = objective(f_trial)
            if obj < best_obj:
                best_obj = obj
                best_t = t
        
        if best_t == 0:
            # Try smaller steps or break
            break
        
        f = (1-best_t)*f + best_t*g_normalized
        f = np.clip(f, 0, None)
        if sum(f) > 0:
            f = f / sum(f)  # renormalize
        
        val = objective(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
    
    return best_f, best_val
```
