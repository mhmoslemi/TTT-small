    fa = np.fft.rfft(a, size)

    return np.fft.irfft(fa * fb, size)[:n]

def _compute_smooth_gradient(a, w, n):
    """Compute g[m] = 2 * sum_j a[j] * w[j+m] for m=0,...,n-1 using FFT."""
    # Pad to length >= 3n-2
    size = 1
    while size < 3*n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)
    # corr[m] = sum_j w[j+m] * a[j] for m=0,...,size-1
    # We want m=0,...,n-1
    return 2.0 * corr[:n]
