import numpy as np
import time

"""
Iterative spectral flattening algorithm to minimize 2n*max(conv(a,a))/sum(a)^2.

Key insight: The objective is scale-invariant, so we normalize sum(a)=1 and minimize
max(conv(a,a)). The minimum possible value is 2n/(2n-1) ≈ 1 for large n, achieved
when all convolution coefficients are equal. We use subgradient descent where the
gradient at each step is the correlation of the "excess" convolution (above mean)
with the sequence itself. Multiple random restarts and adaptive step sizes prevent
getting stuck in local minima.
"""


def fast_objective(a):
    """Fast internal objective computation (matches evaluate_sequence for valid inputs)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def run_flatten(a_init, max_iter, eta0, rng, deadline, verbose=False):
    """
    Iterative flattening: reduce peak of self-convolution via subgradient steps.
    
    At each step:
      1. Compute c = convolve(a, a)
      2. Compute excess e = max(0, c - mean(c))
      3. Compute gradient g = correlate(e, a) [the subgradient of max(c)]
      4. Update a = max(0, a - eta * g), renormalize
    """
    a = np.maximum(0.0, np.asarray(a_init, dtype=np.float64).copy())
    s = a.sum()
    if s < 1e-10:
        a = rng.random(len(a_init))
        s = a.sum()
    a /= s
    n = len(a)
    c_mean = 1.0 / (2.0 * n - 1.0)
    eta = eta0
    best_a = a.copy()
    best_obj = fast_objective(a)
    stagnation = 0
    iter_count = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        iter_count += 1

        c = np.convolve(a, a)
        e = np.maximum(0.0, c - c_mean)
        e_sum = e.sum()
        if e_sum < 1e-16:
            break

        # Subgradient: correlation of excess with sequence
        g = np.correlate(e, a, mode='valid')

        # Update
        a = a - eta * g
        a = np.maximum(0.0, a)
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
            s = a.sum()
        a /= s

        obj = fast_objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
            if verbose and it % 1000 == 0:
                print(f"  iter {it}: obj={obj:.8f}, eta={eta:.4f}")
        else:
            stagnation += 1
            if stagnation > 300:
                eta *= 0.6
                stagnation = 0
            elif stagnation > 800:
                # Escape: add perturbation
                noise_scale = 0.05 / n
                a = a + noise_scale * rng.standard_normal(n)
                a = np.maximum(0.0, a)
                a /= a.sum()
                stagnation = 0
                eta = eta0

    return best_a, best_obj


def run_flatten_multi_lag(a_init, max_iter, eta0, rng, deadline, top_k_frac=0.1):
    """
    Variant that uses top-k excess lags instead of all excess lags.
    This can be more aggressive and escape shallow local minima.
    """
    a = np.maximum(0.0, np.asarray(a_init, dtype=np.float64).copy())
    s = a.sum()
    if s < 1e-10:
        a = rng.random(len(a_init))
        s = a.sum()
    a /= s
    n = len(a)
    c_mean = 1.0 / (2.0 * n - 1.0)
    eta = eta0
    best_a = a.copy()
    best_obj = fast_objective(a)
    stagnation = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        c = np.convolve(a, a)
        # Use only the top-k% excess values for a more focused update
        e = c - c_mean
        e = np.maximum(0.0, e)
        if e.sum() < 1e-16:
            break

        # Threshold: keep only top fraction of excess values
        thresh = np.percentile(e, 100.0 * (1.0 - top_k_frac))
        e_thresh = np.where(e > thresh, e, 0.0)

        g = np.correlate(e_thresh, a, mode='valid')
        g = g / (g.max() + 1e-30) * e_thresh.max()  # Rescale

        a = a - eta * g
        a = np.maximum(0.0, a)
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
            s = a.sum()
        a /= s

        obj = fast_objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 200:
                eta *= 0.7
                stagnation = 0

    return best_a, best_obj
