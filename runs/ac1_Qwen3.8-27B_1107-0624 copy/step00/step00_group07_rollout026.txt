import time

def fast_eval(a):
    """Fast evaluation using FFT."""
    n = len(a)
    # Use FFT for convolution
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    R = np.fft.irfft(A * A, fft_size)[:size]
    max_r = np.max(R)
    sum_a = np.sum(a)
    return 2.0 * n * max_r / (sum_a ** 2)
