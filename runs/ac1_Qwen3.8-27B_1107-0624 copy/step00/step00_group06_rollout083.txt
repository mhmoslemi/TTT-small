def fft_gradient(a,w,L):
    n=len(a)
    fa=np.fft.rfft(a,n=L)
    fw=np.fft.rfft(w[::-1],n=L)
    conv=np.fft.irfft(fa*fw,n=L)
    h=conv[n-1:2*n-1][::-1]
    g=2.0*h
    if g.min() < 0:
       g=np.maximum(g,0.0)
    return g

"""
Search for a low-peak nonnegative coefficient sequence.

The algorithm uses a scale-normalized simplex representation and a smoothed
max-convolution objective.  Weighted quadratic gradients are computed with FFTs,
and multiplicative mirrored updates are applied with a small annealing/SA-style
acceptance rule.  Several heuristic seeds are run, and the best candidates are
finally verified using the provided evaluate_sequence function.
"""

import numpy as np
import time


def positive_normalize(x):
    """Convert x to a finite nonnegative float vector of unit sum."""
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1:
        x = x.reshape(-1)
    x = x.copy()
    x[~np.isfinite(x)] = 0.0
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-300:
        return None
    x /= s
    if x.size > 1:
        x = np.maximum(x, 1e-15)
        s2 = float(np.sum(x))
        if s2 > 1e-300:
            x /= s2
    return x


def add_if_valid(seeds, x):
    """Add a valid normalized seed to a list."""
    y = positive_normalize(x)
    if y is not None:
        seeds.append(y)


def fft_length_for(n):
    """Return a power-of-two FFT length large enough for useful products."""
    if n <= 0:
        return 1
    need = 3 * int(n) - 2
    L = 1
    while L < need:
        L <<= 1
    return L


def fft_square(a, L):
    """Full linear convolution a*a, clipped to nonnegative."""
    n = int(a.size)
    fa = np.fft.rfft(a, n=L)
    with np.errstate(invalid="ignore", over="ignore"):
        c = np.fft.irfft(fa * fa, n=L)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0)


def fft_gradient(a, w, L):
    """
    Gradient of sum_k w_k * (a*a)[k] with respect to a.
    Uses the identity grad_m = 2 * sum_j a_j w_{m+j}.
    """
    n = int(a.size)
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w[::-1], n=L)
    with np.errstate(invalid="ignore", over="ignore"):
        conv = np.fft.irfft(fa * fw, n=L)
    h = conv[n - 1:2 * n - 1][::-1]
    g = 2.0 * h
    return np.maximum(g, 0.0)


def make_weights(c, gamma):
    """Softmax weights over convolution coefficients, focused on large lags."""
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.full(c.shape, 1.0 / max(1, c.size), dtype=np.float64)

    tau = max(cmax / max(float(gamma), 1e-6), 1e-300)
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        w = np.exp((c - cmax) / tau)

    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(c.shape, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def multiplicative_update(a, grad, alpha):
    """One mirrored multiplicative descent step on the simplex."""
    gmax = float(np.max(grad))
    if not np.isfinite(gmax) or gmax <= 1e-300:
        return None

    eta = float(alpha) / gmax
    x = -eta * grad
    x = np.where(x < 0.0, x, 0.0)
    x = np.maximum(x, -40.0)

    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        m = np.exp(x)

    new_a = a * m
    s = float(np.sum(new_a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    new_a /= s
    return positive_normalize(new_a)


def random_perturbation(a, rng, frac, sigma):
    """Random multiplicative perturbation of a few coordinates."""
    n = int(a.size)
    mask = rng.random(n) < frac
    cnt = int(np.count_nonzero(mask))
    if cnt <= 0:
        return a.copy()
    noise = np.ones(n, dtype=np.float64)
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        noise[mask] = np.exp(rng.normal(0.0, sigma, cnt))
    return positive_normalize(a * noise)


def should_accept(trial_max, cur_max, temperature, rng):
    """Greedy plus mild simulated-annealing acceptance."""
    if not np.isfinite(trial_max) or not np.isfinite(cur_max):
        return False
    if trial_max < cur_max:
        return True
    if temperature <= 1e-300:
        return False
    ratio = (trial_max - cur_max) / temperature
    if ratio > 50.0:
        return False
    with np.errstate(over="ignore", under="ignore"):
        prob = np.exp(-ratio)
    return bool(rng.random() < prob)


def make_seeds(n, rng, parent=None):
    """Build a diverse list of normalized nonnegative seeds of length n."""
    seeds = []
    n = int(max(1, n))
    u = np.linspace(0.0, 1.0, n)

    add_if_valid(seeds, np.ones(n, dtype=np.float64))

    # Monotone / polynomial shapes.
    add_if_valid(seeds, np.sqrt(u) + 1e-3)
    add_if_valid(seeds, u + 1e-3)
    add_if_valid(seeds, u * u + 1e-3)
    add_if_valid(seeds, u ** 3 + 1e-3)

    # Exponential tilts.
    add_if_valid(seeds, np.exp(np.linspace(0.0, 1.0, n)))
    add_if_valid(seeds, np.exp(np.linspace(0.0, 2.0, n)))
    add_if_valid(seeds, np.exp(np.linspace(1.0, 0.0, n)))

    # Sinusoidal / cosine shapes.
    for amp in (0.3, 0.5, 0.7):
        add_if_valid(seeds, 1.0 + amp * np.sin(2.0 * np.pi * u))
        add_if_valid(seeds, 1.0 + amp * np.sin(4.0 * np.pi * u))
        add_if_valid(seeds, 1.0 + amp * np.cos(2.0 * np.pi * u))

    add_if_valid(seeds, 1.0 + 0.5 * np.sin(2.0 * np.pi * u) + 0.25 * np.sin(4.0 * np.pi * u))
    add_if_valid(seeds, 1.0 + 0.5 * np.sin(2.0 * np.pi * u) - 0.25 * np.sin(4.0 * np.pi * u))
    add_if_valid(seeds, 1.0 + 0.4 * np.cos(2.0 * np.pi * u) + 0.2 * np.cos(4.0 * np.pi * u))

    # Low-degree Legendre-like shapes on [0, 1].
    P1 = 2.0 * u - 1.0
    P2 = 6.0 * u * u - 6.0 * u + 1.0
    P3 = 20.0 * u ** 3 - 30.0 * u * u + 12.0 * u - 1.0

    coeffs = (
        (0.3, 0.0, 0.0),
        (0.4, 0.0, 0.0),
        (0.5, 0.0, 0.0),
        (0.6, 0.0, 0.0),
        (0.5, 0.1, 0.0),
        (0.5, 0.2, 0.0),
        (0.5, -0.1, 0.0),
        (0.5, 0.0, 0.1),
        (0.5, 0.0, -0.1),
        (0.4, 0.1, 0.05),
        (0.6, 0.1, -0.05),
    )
    for c1, c2, c3 in coeffs:
        add_if_valid(seeds, 1.0 + c1 * P1 + c2 * P2 + c3 * P3)

    # Random Dirichlet shapes.
    try:
        add_if_valid(seeds, rng.dirichlet(np.ones(n, dtype=np.float64)))
    except Exception:
        pass
    try:
        add_if_valid(seeds, rng.dirichlet(np.full(n, 0.5, dtype=np.float64)))
    except Exception:
        pass

    # Existing construction, if available.
    if parent is not None:
        add_if_valid(seeds, parent)

    if not seeds:
        add_if_valid(seeds, np.ones(n, dtype=np.float64))

    return seeds


def optimize_sequence(a, deadline, rng):
    """
    Run the mirrored smoothed-maxconvolution search.

    Returns (best_sequence, fast_score).  fast_score is computed with FFTs and
    is only used for local search; the safe exact score is verified later.
    """
    a = positive_normalize(a)
    if a is None:
        return None, np.inf

    n = int(a.size)
    L = fft_length_for(n)

    c = fft_square(a, L)
    cur_max = float(np.max(c))
    if not np.isfinite(cur_max) or cur_max <= 0.0:
        return a, float(2.0 * n * cur_max)

    best_a = a.copy()
    best_max = cur_max

    alpha_init = 0.35
    alpha_min = 0.05
    alpha_max = 1.6
    gamma_init = 3.0
    gamma_end = 12.0

    alpha = alpha_init
    gamma = gamma_init
    temperature = 0.05 * cur_max
    stale = 0

    while time.monotonic() < deadline:
        if not np.isfinite(cur_max) or cur_max <= 0.0:
            break

        w = make_weights(c, gamma)
        grad = fft_gradient(a, w, L)

        accepted = False
        for att in range(4):
            alpha_use = alpha * (0.7 ** att)
            if alpha_use < alpha_min:
                alpha_use = alpha_min

            trial = multiplicative_update(a, grad, alpha_use)
            if trial is None:
                continue

            c_trial = fft_square(trial, L)
            tmax = float(np.max(c_trial))
            if not np.isfinite(tmax):
                continue

            if should_accept(tmax, cur_max, temperature, rng):
                a = trial
                c = c_trial
                cur_max = tmax

                if tmax < best_max:
                    best_max = tmax
                    best_a = trial.copy()

                alpha = min(alpha_max, alpha * 1.015)
                gamma = min(gamma_end, gamma * 1.002)
                temperature = max(1e-12, temperature * 0.999)
                stale = 0
                accepted = True
                break

        if not accepted:
            stale += 1

            if stale > 80:
                a = best_a.copy()
                c = fft_square(a, L)
                cur_max = float(np.max(c))
                alpha = alpha_init
                gamma = gamma_init
                temperature = 0.05 * cur_max
                stale = 0
                continue

            if stale % 3 == 0:
                sigma = 0.10 + 0.04 * min(stale, 15)
                trial = random_perturbation(a, rng, 0.07, sigma)
                if trial is not None:
                    c_trial = fft_square(trial, L)
                    tmax = float(np.max(c_trial))
                    if np.isfinite(tmax) and should_accept(tmax, cur_max, temperature, rng):
                        a = trial
                        c = c_trial
                        cur_max = tmax
                        if tmax < best_max:
                            best_max = tmax
                            best_a = trial.copy()

            alpha = min(alpha_max, alpha * 1.2)
            temperature = max(0.05 * cur_max if cur_max > 0.0 else 1e-12, temperature * 1.05)

    best_a = positive_normalize(best_a)
    if best_a is None:
        return None, np.inf
    return best_a, float(2.0 * n * best_max)


def find_exact_best(candidates, final_deadline):
    """Evaluate candidate sequences with the authoritative evaluator."""
    best_seq = None
    best_score = np.inf
    scales = (1.0, 10.0, 0.1, 100.0)

    for seq in candidates:
        if seq is None:
            continue
        seq = positive_normalize(seq)
        if seq is None:
            continue

        for scale in scales:
            if time.monotonic() >= final_deadline:
                break

            cand = seq * scale
            cand = np.where(np.isfinite(cand), cand, 0.0)
            cand = np.clip(cand, 0.0, 1000.0)

            if float(np.sum(cand)) < 0.01:
                continue

            try:
                score = evaluate_sequence(cand.tolist())
            except Exception:
                score = np.inf

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()

        if time.monotonic() >= final_deadline:
            break

    if best_seq is None:
        for fallback_len in (1024, 256):
            if time.monotonic() >= final_deadline:
                break
            fallback = np.ones(fallback_len, dtype=np.float64) / float(fallback_len)
            try:
                score = evaluate_sequence(fallback.tolist())
            except Exception:
                score = np.inf
            if np.isfinite(score):
                best_seq = fallback.copy()
                best_score = score
                break

    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_score = np.inf

    return best_seq, best_score
