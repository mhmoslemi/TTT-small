```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(s, M):
    n = len(s)
    # Build constraint matrix A: A[k, j] = s[k-j] if 0 <= k-j <= n-1
    # A has shape (2n-1, n)
    # For row k, the non-zero entries are at columns j where 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
    # So row k has entries s[k], s[k-1], ..., s[max(0, k-n+1)] at columns min(k, n-1), min(k,n-1)-1, ..., max(0, k-n+1)
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(s[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(b) = [1,1,...,1]^T b
    # linprog minimizes, so minimize -sum(b)
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    else:
```
