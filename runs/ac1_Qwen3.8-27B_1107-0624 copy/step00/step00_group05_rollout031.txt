"""
LP-based iterative peak-shaving for minimum peak autocorrelation sequences.
Uses reduced linear programming to find descent directions, combined with
subgradient steps and simulated annealing perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_peak(a):
    """Compute the peak of the autocorrelation and the objective score."""
    n = len(a)
    b = np.convolve(a, a)
    peak = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return peak, np.inf
    score = 2.0 * n * peak / (s * s)
    return peak, score

def normalize_sequence(a, target=1.0):
    """Normalize a non-negative sequence to have the given sum."""
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a * (target / s)

def solve_direction_lp(a, M, K=30, top_lags=None):
    """
    Solve the reduced LP to find a direction g that maximizes sum(g)
    subject to (a*g)[k] <= M for the top K lags and g >= 0.
    Returns g or None if the LP fails.
    """
    n = len(a)
    
    if top_lags is None:
        b = np.convolve(a, a)
        top_lags = np.argsort(b)[-K:]
    
    c = -np.ones(n)  # Maximize sum(g) => minimize -sum(g)
    
    num_constraints = len(top_lags)
    # Build constraint matrix
    row_data = []
    col_data = []
    val_data = []
    
    for idx, k in enumerate(top_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_range = np.arange(j_start, j_end + 1)
        vals = a[k - j_range]
        row_data.extend(np.full(len(j_range), idx).tolist())
        col_data.extend(j_range.tolist())
        val_data.extend(vals.tolist())
    
    if not row_data:
        return None
    
    A = csr_matrix((val_data, (row_data, col_data)), shape=(num_constraints, n))
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
        if result.success:
            return result.x
    except Exception:
        pass
    
    return None

def line_search(a, g, n):
    """
    Find the best t in [0, 1] to mix a and g: new = (1-t)*a + t*g, normalize,
    and minimize the score.
    """
    best_score = np.inf
    best_t = 0.0
    best_candidate = None
    
    for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
        candidate = (1 - t) * a + t * g
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s  # Normalize
        b = np.convolve(candidate, candidate)
        score = 2.0 * n * np.max(b)  # Since sum = 1
        if score < best_score:
            best_score = score
            best_t = t
            best_candidate = candidate.copy()
    
    return best_candidate, best_score, best_t

def subgradient_step(a, num_lags=5, eta=0.1):
    """
    Perform a subgradient step to reduce the peak autocorrelation.
    Returns the updated sequence.
    """
    n = len(a)
    b = np.convolve(a, a)
    top_lags = np.argsort(b)[-num_lags:]
    
    grad = np.zeros(n)
    for k in top_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_range = np.arange(j_start, j_end + 1)
        grad[j_range] += a[k - j_range]
    
    candidate = a - eta * grad / n
    candidate = np.clip(candidate, 0.0, None)
    s = np.sum(candidate)
    if s < 1e-10:
        return a
    return candidate / s

def make_random_sequence(n, rng):
    """Generate a random non-negative sequence normalized to sum 1."""
    a = rng.random(n)
    return normalize_sequence(a, 1.0)

def make_binary_sequence(n, rng, p=0.5):
    """Generate a random binary sequence normalized to sum 1."""
    a = (rng.random(n) < p).astype(float)
    if a.sum() < 1:
        a[rng.integers(n)] = 1.0
    return normalize_sequence(a, 1.0)

def make_tapered_random(n, rng):
    """Generate a random sequence with cosine tapering at edges."""
    a = rng.random(n)
    # Apply a raised cosine window
    if n > 2:
        window = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
        a = a * window
        a = np.clip(a, 1e-6, None)  # Avoid exact zeros at edges
    return normalize_sequence(a, 1.0)

        print(f"  Start {i}: score = {score:.6f}")

    current_score = best_score
