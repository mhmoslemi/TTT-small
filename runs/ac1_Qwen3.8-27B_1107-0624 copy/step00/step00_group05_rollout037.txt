def fast_objective(a):
    """Compute 2*n*max(a*a)/(sum(a))^2 using FFT."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    # Full convolution via FFT
    size = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2*n-1]
    M = b.max()
    return 2.0 * n * M / (S * S)
```

This avoids the overhead of `evaluate_sequence` and is much faster. I'll use `evaluate_sequence` only for final verification.

- If it doesn't improve, decrease $\eta$ by a factor of 0.5
- Cap $\eta$ to ensure $a - \eta \cdot grad \geq 0$

import numpy as np
import time

def fast_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:out_len]

def fast_objective(a):
    """Compute 2*n*max(a*a)/(sum(a))^2."""
    S = a.sum()
    if S < 1e-10:
        return np.inf
    b = fast_conv(a)
    return 2.0 * len(a) * b.max() / (S * S)

def smooth_grad(a, beta):
    """
    Compute gradient of (2n/S^2) * logsumexp(beta * (a*a)).
    Returns (gradient_array, objective_value).
    """
    n = len(a)
    S = a.sum()
    b = fast_conv(a)
    
    # Log-sum-exp for numerical stability
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    log_sum = b_max + np.log(exp_b.sum())
    weights = exp_b / exp_b.sum()
    
    # Objective
    obj = 2.0 * n * log_sum / (S * S)
    
    # Gradient of log_sum_exp w.r.t. a:

def fast_cross(weights, a):
    """Compute cross-correlation: result[i] = sum_k weights[k] * a[k-i] for valid k-i."""
    n = len(a)
    # weights has length 2n-1
    # We need: for i=0 to n-1, sum_{j=0}^{n-1} weights[i+j] * a[j]
    # This is convolve(weights, a, mode='valid')
    # Using FFT:
    size = 1 << (2 * n - 2).bit_length()
    fw = np.fft.rfft(weights, size)
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fw * fa, size)
    return result[:n]  # 'valid' portion
```

- When the second array is shorter, 'valid' returns only those convolutions that do not rely on the zero-padded parts.
- For `np.convolve(v, a)` where $len(v) \geq len(a)$, the output has length $len(v) - len(a) + 1$.
