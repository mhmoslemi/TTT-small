Minimizes 2n*max(conv(s,s))/sum(s)^2 via FFT-accelerated gradient descent
with log-sum-exp smoothing, smooth restarts, and hybrid parametric search.

import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_convolve(s):
    n = len(s)
    size = 2 * n - 1
    nfft = _next_pow2(size)
    S = np.fft.rfft(s, nfft)
    return np.fft.irfft(S * S, nfft)[:size]


def _softmax_weights(c, temp):
    cmax = c.max()
    if not np.isfinite(cmax):
        return np.ones(len(c)) / len(c)
    logits = (c - cmax) / temp
    logits = np.clip(logits, -100.0, 100.0)
    w = np.exp(logits)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= s
    return w


def _grad_smooth_max(s, w, N):
    n = len(s)
    S = np.fft.rfft(s, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(W * np.conj(S), N)[:n]
    return 2.0 * corr


def _soft_max_val(c, w):
    return -np.log(w.sum())


def _project_simple(s, target_sum):
    s = np.clip(s, 0.0, 1000.0)
    ssum = s.sum()
    if ssum > 1e-15:
        s *= target_sum / ssum
    else:
        s = np.ones_like(s) / len(s)
    return s


def _project_exact(s, target_sum):
    s = np.clip(s, 0.0, 1000.0)
    n = s.size
    current_sum = s.sum()
    if np.abs(current_sum - target_sum) < 1e-12:
        return s
    if current_sum > target_sum:
        excess = current_sum - target_sum
        order = np.argsort(s)
        cum = 0.0
        result = s.copy()
        for idx in order:
            if cum >= excess:
                break
            can_take = s[idx]
            needed = min(can_take, excess - cum)
            result[idx] -= needed
            cum += needed
        return np.clip(result, 0.0, 1000.0)
    else:
        deficit = target_sum - current_sum
        s += deficit / n
        return np.clip(s, 0.0, 1000.0)

        if score < best_score:
            best_score = score
            best_s = s.copy()
    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Hybrid search: parametric init + gradient descent + block refinement."""
    global height_sequence_1
    global_start = time.monotonic()
    total_budget = budget_s - 15.0

    if seed % 7 == 0:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    elif seed % 3 == 0:
        _pf = 1.0 + 0.4 * np.cos(np.pi * np.arange(len(height_sequence_1)) / max(1, len(height_sequence_1) - 1))
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64) * _pf, 0.0, 1000.0)
    else:
        _rng_p = np.random.default_rng(seed + 777)
        _noise = np.exp(_rng_p.normal(0, 0.3, len(height_sequence_1)))
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64) * _noise, 0.0, 1000.0)

    if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all() or parent_arr.sum() < 0.01:
        parent_arr = np.ones(2048, dtype=np.float64)

    parent_n = len(parent_arr)

    lengths = sorted(set([
        parent_n,
        1024, 2048, 4096,
        max(512, parent_n // 2),
        min(8192, parent_n + parent_n // 3),
    ]))
    lengths = [l for l in lengths if 256 <= l <= 10000]

    n_descent = 5 if total_budget > 1200 else (4 if total_budget > 800 else 3)
    time_per_try = total_budget / (len(lengths) + n_descent)

    rng = np.random.default_rng(seed)
    global_best = None
    global_best_score = np.inf

    for n in lengths:
        remaining = total_budget - (time.monotonic() - global_start)
        if remaining < 10:
            break
        t_param = min(40, remaining * 0.2)
        s_p, score_p = _try_parametric(n)
        final_score_p = 2.0 * n * score_p
        if final_score_p < global_best_score:
            global_best_score = final_score_p
            global_best = s_p.copy()
        print(f"  n={n} parametric: score={final_score_p:.6f}", flush=True)

    for _ in range(n_descent):
        remaining = total_budget - (time.monotonic() - global_start)
        if remaining < 30:
            break
        n_desc = int(rng.integers(1024, 5000))
        mode = rng.integers(0, 5)
        if mode == 0:
            s_init = _try_parametric(n_desc)[0]
        elif mode == 1:
            x = np.linspace(-2.5, 2.5, n_desc)
            s_init = np.exp(-0.3 * x ** 2)
            s_init = np.clip(s_init, 0.0, None)
            s_init /= s_init.sum()
        elif mode == 2:
            x = np.linspace(0, 1, n_desc)
            s_init = (1 - x) ** 2 + 0.1 * np.ones(n_desc)
            s_init = np.clip(s_init, 0.0, None)
            s_init /= s_init.sum()
        elif mode == 3 and global_best is not None:
            s_init = np.interp(np.linspace(0, 1, n_desc), np.linspace(0, 1, len(global_best)), global_best)
            s_init = np.clip(s_init, 0.0, None)
            s_init /= max(s_init.sum(), 1e-15)
        else:
            s_init = np.ones(n_desc) / n_desc
            x = np.linspace(0, 1, n_desc)
            s_init += 0.2 * np.sin(np.pi * x)
            s_init = np.clip(s_init, 0.0, None)
            s_init /= s_init.sum()

        s_best, max_best = _gradient_descent(s_init, budget_s=remaining * 0.6, init_lr=0.04)
        final_score = 2.0 * n_desc * max_best
        if final_score < global_best_score:
            global_best_score = final_score
            global_best = s_best.copy()
        print(f"  n={n_desc} grad_desc: score={final_score:.6f}", flush=True)

    remaining = total_budget - (time.monotonic() - global_start)
    if global_best is not None and remaining > 120:
        n_ref = len(global_best)
        s_ref, max_ref = _block_refine(global_best, budget_s=remaining * 0.7, n_blocks=48)
        final_score = 2.0 * n_ref * max_ref
        if final_score < global_best_score:
            global_best_score = final_score
            global_best = s_ref.copy()
        print(f"  Refine n={n_ref}: score={final_score:.6f}", flush=True)

    if global_best is None:
        global_best = parent_arr.copy()
        global_best = np.clip(global_best, 0.0, 1000.0)
        if global_best.sum() < 0.01:
            global_best = np.ones(256)

    for _ in range(5):
        score = float(evaluate_sequence(global_best.tolist()))
        if np.isfinite(score):
            break
        global_best = np.clip(global_best * 0.9, 0.0, 1000.0)
        if global_best.sum() < 0.01:
            global_best = np.ones(256)

    print(f"  FINAL: n={len(global_best)}, score={global_best_score:.6f}, "
          f"time={time.monotonic()-global_start:.1f}s", flush=True)
    return [float(x) for x in global_best]
