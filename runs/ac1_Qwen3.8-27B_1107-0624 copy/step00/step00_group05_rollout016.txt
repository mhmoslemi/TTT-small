def gradient_step(a, lr, n):
    """One projected subgradient descent step."""
    conv = np.convolve(a, a)
    max_val = conv.max()
    # Active set: indices where conv is within tolerance of max
    active = np.where(conv >= max_val - 1e-8 * max_val)[0]
    
    # Subgradient: for each j, grad[j] = (1/|active|) * sum_{k in active} a[k-j] (if valid)
    grad = np.zeros(n)
    for k in active:
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        # a[k-j] for j in [j_min, j_max] is a[k-j_min : k-j_max+1] reversed... 
        # Actually, a[k-j] for j from j_min to j_max is a[k] down to a[k-j_max]
        # = a[k : k-j_max-1 : -1] = a[max(0,k-n+1) : k+1] in reverse
        # Let me just do it directly
        for j in range(j_min, j_max + 1):
            grad[j] += a[k - j]
    grad /= len(active)
    
    # Projected gradient step
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

        return result.x
    return None
