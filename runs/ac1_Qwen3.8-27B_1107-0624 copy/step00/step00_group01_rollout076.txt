    """Build matrix A where (A@g)[k] = (f*g)[k]."""

    return None

import numpy as np

def compute_score(a):
    """Fast internal score computation."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def build_conv_matrix(f, n):
    """Build matrix A where (A@g)[k] = (f*g)[k] for k=0..2n-2."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            A[k, m_start:m_end + 1] = f[k - np.arange(m_start, m_end + 1)]
    return A


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""

    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None

                candidates.append((s, n, p, a.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"  Evaluated {len(candidates)} parametric candidates in {time.monotonic()-t0:.1f}s")
    print("  Top 10:")
    for s, n, p, _ in candidates[:10]:
        print(f"    n={n}, p={p:.1f}, score={s:.6f}")

    for s, n, p, a in candidates[:8]:
        if s < best_score:
            best_score = s
            best_arr = a.copy()

    print(f"  Best parametric: score={best_score:.6f}")

    # Phase 2: LP optimization on top candidates
    print("\nPhase 2: LP optimization")
    t0 = time.monotonic()
    for s, n, p, a in candidates[:5]:
        if time.monotonic() > total_deadline - 120:
            break
        remaining = total_deadline - time.monotonic() - 120
        t_budget = min(180.0, remaining * 0.45)
        if t_budget < 20:
            break
        print(f"  LP: n={n}, p={p:.1f} (score={s:.4f}), budget={t_budget:.0f}s")
        f_opt, s_opt = lp_optimize(a, n, t_budget)
        if s_opt < best_score:
            best_score = s_opt
            best_arr = f_opt.copy()
            print(f"    LP improved to {s_opt:.6f}")
        else:
            print(f"    LP best: {s_opt:.6f} (no improvement)")

    print(f"  Phase 2 took {time.monotonic()-t0:.1f}s")

    # Phase 3: Gradient optimization on overall best
    if best_arr is not None and time.monotonic() < total_deadline - 60:
        print("\nPhase 3: Gradient optimization")
        t0 = time.monotonic()
        n = len(best_arr)
        t_budget = min(200.0, (total_deadline - time.monotonic() - 60) * 0.6)
        print(f"  Gradient: n={n}, budget={t_budget:.0f}s")
        f_opt, s_opt = gradient_optimize(best_arr, n, t_budget)
        if s_opt < best_score:
            best_score = s_opt
            best_arr = f_opt.copy()
            print(f"    Gradient improved to {s_opt:.6f}")
        else:
            print(f"    Gradient best: {s_opt:.6f}")
        print(f"  Phase 3 took {time.monotonic()-t0:.1f}s")

    # Phase 4: Final LP refinement
    if best_arr is not None and time.monotonic() < total_deadline - 30:
        print("\nPhase 4: Final LP refinement")
        t0 = time.monotonic()
        n = len(best_arr)
        t_budget = (total_deadline - time.monotonic() - 10) * 0.8
        if t_budget > 30:
            print(f"  Final LP: n={n}, budget={t_budget:.0f}s")
            f_opt, s_opt = lp_optimize(best_arr, n, t_budget)
            if s_opt < best_score:
                best_score = s_opt
                best_arr = f_opt.copy()
                print(f"    Final LP improved to {s_opt:.6f}")
            else:
                print(f"    Final LP best: {s_opt:.6f}")
        print(f"  Phase 4 took {time.monotonic()-t0:.1f}s")

    # Verify with actual evaluator
    if best_arr is not None:
        final_score = float(evaluate_sequence(best_arr.tolist()))
        print(f"\nFinal verified score: {final_score:.6f} (n={len(best_arr)}, internal={best_score:.6f})")
        if not np.isfinite(final_score):
            print("  WARNING: Evaluator rejected the sequence! Using fallback.")
            best_arr = np.arange(1, 1001, dtype=np.float64)
            final_score = float(evaluate_sequence(best_arr.tolist()))
            print(f"  Fallback score: {final_score:.6f}")

    if best_arr is None:
        best_arr = np.arange(1, 1001, dtype=np.float64)

    return [float(x) for x in best_arr]
