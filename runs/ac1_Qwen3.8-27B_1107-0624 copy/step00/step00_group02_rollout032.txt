```python
def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_conv_max(b):
    """Compute the max of conv(b,b) and the argmax."""
    conv = np.convolve(b, b)
    max_val = np.max(conv)
    max_idx = np.argmax(conv)
    return max_val, max_idx

def compute_gradient(b, k):
    """Compute gradient of (conv(b,b))_k with respect to b."""
    n = len(b)
    grad = np.zeros(n)
    # grad[i] = 2*b[k-i] if 0 <= k-i < n

        if 0 <= j < n:

    return grad

    """One step of projected gradient descent on max(conv(b,b))."""

    max_val = np.max(conv)

    for k in active:

    # Project onto simplex
```

```python
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
import sys

def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []  
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, 
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return np.zeros(n)

def conv_max(b):
    return float(np.max(np.convolve(b, b)))

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + budget_s - 10
    best_score = float('inf')
    best_seq = None
    
    n_list = [500, 1000, 2000, 3000]
    total_time = budget_s - 20
    time_per_n = total_time / (len(n_list) + 1)  # +1 for refinement
    
    for n in n_list:
        if time.monotonic() > deadline - 60:
            break
        
        phase_end = min(deadline - 30, time.monotonic() + time_per_n)
        
        f = np.ones(n) / n
        M = conv_max(f)
        print(f"n={n}, start: score={2*n*M:.6f}")
        
        t_mix = 0.3
        for it in range(500):
            if time.monotonic() > phase_end:
                break
            
            g = solve_lp(f, M)
            g_sum = float(np.sum(g))
            if g_sum < 1e-15:
                break
            g /= g_sum
            
            # Backtracking
            best_t = None
            for t in [t_mix, t_mix*0.5, t_mix*0.25, 0.05, 0.01]:
                f_new = (1-t)*f + t*g
                M_new = conv_max(f_new)
                if M_new < M:
                    best_t = t
                    f = f_new
                    M = M_new
                    break
            
            if best_t is None:
                t_mix *= 0.5
                if t_mix < 1e-4:
                    break
            
            if it % 50 == 0:
                print(f"  n={n}, it={it}, M={M:.8f}, score={2*n*M:.6f}")
        
        score = 2 * n * M
        print(f"n={n} done: it={it+1}, M={M:.8f}, score={score:.6f}")
        
        if score < best_score:
            best_score = score
            best_seq = f.copy()
    
    # Local refinement on best_seq
    if best_seq is not None and time.monotonic() < deadline - 30:
        rng = np.random.default_rng(seed)
        f = best_seq.copy()
        n = len(f)
        M = conv_max(f)
        score = 2*n*M
        
        for it in range(5000):
            if time.monotonic() > deadline:
                break
            # Random perturbation
            f_new = f.copy()
            width = max(1, n // 50)
            start = rng.integers(0, n - width)

    return [float(x) for x in best_seq]
```
