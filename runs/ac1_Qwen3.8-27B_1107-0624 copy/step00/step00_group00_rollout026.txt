def fast_full_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    # Use next power of 2 for FFT efficiency
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]
