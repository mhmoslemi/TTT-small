def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Uses a linear ramp as initial sequence and applies LP-based refinement
    to minimize the self-convolution peak. The LP at each step finds the
    optimal g (maximizing sum) subject to f*g <= M elementwise and g >= 0,
    then blends f towards g.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Choose sequence length
    n = 3000
    
    # Initialize with linear ramp
    f = np.arange(1, n+1, dtype=np.float64)
    f = f / f.max()  # normalize to [1/n, 1]
    
    # Evaluate initial
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial ramp: score = {best_score:.6f}, n = {n}")
    
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        t_start = time.monotonic()
        
        # Compute M = max of self-convolution
        conv = np.convolve(f, f)
        M = np.max(conv)
        
        # Set up LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        # Constraint matrix A: A[k, j] = f[k-j] if 0<=k-j<n, else 0
        # This is a "convolution matrix"
        
        # Build sparse constraint matrix
        # Row k has non-zeros at columns j where 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
        rows = []
        cols = []
        vals = []
        for k in range(2*n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # LP: max -c^T g where c = -1 (to maximize sum(g))
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
            if res.success:
                g = res.x
                # Normalize g to same scale as f
                if np.sum(g) > 0:
                    g = g / g.max() if g.max() > 0 else g
                    # Blend
                    t = 0.3  # blending parameter
                    f_new = (1 - t) * f + t * g
                    f_new = np.clip(f_new, 0.0, 1000.0)
