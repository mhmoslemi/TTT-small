```python
def active_set_step(a, i_star, step_size):
    n = len(a)
    # Compute pressure
    # g[j] = a[i_star - j] if 0 <= i_star - j < n, else 0
    # Plus the symmetric term
    g = np.zeros(n)
    # First term: a[i_star - j] valid when 0 <= i_star - j < n, i.e., i_star-n+1 <= j <= i_star
    lo1 = max(0, i_star - n + 1)
    hi1 = min(n - 1, i_star)
    if lo1 <= hi1:
        g[lo1:hi1+1] += a[i_star - hi1 : i_star - lo1 + 1][::-1]
    
    # Second term: a[j] valid when 0 <= i_star - j < n AND i_star - j in valid k range
    # i_star - j in [max(0, i_star-n+1), min(n-1, i_star)]
    # This means j in [i_star - min(n-1, i_star), i_star - max(0, i_star-n+1)]
    lo2 = max(0, i_star - min(n - 1, i_star))
    hi2 = min(n - 1, i_star - max(0, i_star - n + 1))
    if lo2 <= hi2:
        g[lo2:hi2+1] += a[lo2:hi2+1]
    
    # Now g[j] is the gradient of conv[i_star] w.r.t. a[j] (divided by... wait)
    # Actually d(conv[i])/d(a[j]) = a[i-j] * I(valid) + a[j] * I(valid)
    # So the "pressure" vector for reducing conv[i_star] is g as computed above
    
    # To reduce conv[i_star] while maintaining sum(a) = 1:
    # We want to transfer mass from high-g positions to low-g positions
    # Update: a_new = a - step * (g - mean(g))
    # This maintains the sum (since sum(g - mean(g)) = 0) and reduces conv[i_star]
    
    update = g - np.mean(g)
    a_new = a - step_size * update
    a_new = np.maximum(a_new, 0)
    a_new /= np.sum(a_new)
    return a_new
```

```python
def multi_constraint_step(a, top_k, step_size):
    n = len(a)
    conv = np.convolve(a, a)
    # Get top K lags by convolution value
    top_indices = np.argsort(conv)[-top_k:]
    top_indices = top_indices[::-1]  # descending order
    
    # Weighted gradient
    conv_vals = conv[top_indices]
    weights = np.exp(conv_vals / 0.1)  # softmax with T=0.1
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for w, i in zip(weights, top_indices):
        # Add w * gradient of conv[i]
        lo1 = max(0, i - n + 1)
        hi1 = min(n - 1, i)
        if lo1 <= hi1:
            g[lo1:hi1+1] += w * a[i - hi1 : i - lo1 + 1][::-1]
        lo2 = max(0, i - min(n - 1, i))
        hi2 = min(n - 1, i - max(0, i - n + 1))
        if lo2 <= hi2:
            g[lo2:hi2+1] += w * a[lo2:hi2+1]
    
    update = g - np.mean(g)
    a_new = a - step_size * update
    a_new = np.maximum(a_new, 0)
    a_new /= np.sum(a_new)
    return a_new
```

This means my simple formula `grad = 2 * conv_full[n-1:2*n-1][::-1]` is an APPROXIMATION that ignores edge effects. For large n, the edge effects are a small fraction (O(1) out of O(n) terms), so the approximation should be good.

Let me just use this approximation. The error is at most O(1)/n relative, which is negligible for n = 4806.

sum_i w[i] * a[i-j] (for valid i, j)

The condition max(0, i-n+1) <= j is always true when j >= 0 (since i-n+1 <= i, and for i >= n-1, i-n+1 >= 0, so max(0, i-n+1) = i-n+1 which is <= j when j >= i-n+1... hmm this is getting complicated).

For the "bulk" (i in [n-1, n-1], i.e., just the center), the valid range is all of [0, n-1], so both conditions are always satisfied. The approximation is exact at the center.

For edge convolutions (small i or large i), the valid range is smaller, and the approximation overestimates the gradient (it includes terms that shouldn't be there).

But since w[i] is concentrated around the argmax (due to the softmax), and the argmax is typically near the center (for reasonably uniform sequences), the approximation should be very good.

In the log-parameterization:

- a = softmax(θ) * n (or a = exp(θ) / sum(exp(θ)))

The gradient in θ-space is:
d(a[j])/d(θ[m]) = a[j] * (δ(j,m) - a[m]/sum(a)) = a[j] * (δ(j,m) - a[m]) (when sum(a) = 1)

```python
"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 via projected gradient descent on the simplex.
Uses smooth-max gradient with multiple initializations and periodic reheat.
"""
import numpy as np
import time


def normalize_to_sum(a, target_sum=1.0):
    s = np.sum(a)
    if s < 1e-12:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def compute_objective(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def compute_smooth_grad(a, T):
    """Compute approximate gradient of smoothed max of self-convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    w = np.exp((conv - max_c) / T)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # grad[j] = 2 * sum_k a[k] * w[k+j] (approx, ignoring O(1) edge effects)
    conv_full = np.convolve(a, w[::-1], mode='full')
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1].copy()
    return grad, conv, max_c


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s < 1e-30:
        return np.full_like(v, target_sum / len(v))
    return v * (target_sum / s)


def local_search(a0, deadline, rng, T_init=0.05, T_min=0.005, lr_init=0.01):
    """Projected gradient descent with adaptive temperature and step size."""
    a = a0.copy()
    n = len(a)
    target_sum = 1.0
    a = normalize_to_sum(a, target_sum)
    best_a = a.copy()
    best_obj = evaluate_sequence(a.tolist())
    if not np.isfinite(best_obj):
        best_obj = compute_objective(a)
    T = T_init
    lr = lr_init
    stall_count = 0
    it = 0

    while time.monotonic() < deadline:
        grad, conv, max_c = compute_smooth_grad(a, T)

        # Gradient in simplex coordinates: remove component along 1
        grad_centered = grad - np.mean(grad)

        # Line search (backtracking)
        improved = False
        for _ in range(8):
            a_new = a - lr * grad_centered
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, target_sum)
            obj_new = compute_objective(a_new)
            if np.isfinite(obj_new) and obj_new < best_obj * 1.0001:
                a = a_new
                improved = True
                break
            lr *= 0.5

        if improved:
            obj_now = compute_objective(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            # Anneal temperature
            T = max(T_min, T * 0.995)
            # Slowly increase lr as we get closer
            lr = min(lr_init, max(0.001, lr * 0.99))
        else:
            # Stuck: reheat
            stall_count += 1
            T = min(T_init, T * 2.0)
            lr = min(lr_init * 2, lr * 1.5)
            # Small random perturbation
            if stall_count > 20:
                noise = rng.normal(0, 0.01 * target_sum, size=n)
                a = np.maximum(a + noise, 0)
                a = normalize_to_sum(a, target_sum)
                stall_count = 0

        it += 1
        if it % 200 == 0:
            # Exact evaluation
            obj_exact = evaluate_sequence(a.tolist())
            if np.isfinite(obj_exact) and obj_exact < best_obj:
                best_obj = obj_exact
                best_a = a.copy()

    return best_a, best_obj


def make_initializations(n, rng, existing_seq=None):
    """Generate diverse initial sequences."""
    inits = []

    if existing_seq is not None:
        h = np.asarray(existing_seq, dtype=np.float64)
        h = np.clip(h, 0, 1000)
        if h.size == n and np.sum(h) > 0.01:
            inits.append(normalize_to_sum(h, 1.0))

    # Linearly increasing
    a = np.arange(1, n + 1, dtype=np.float64)
    inits.append(normalize_to_sum(a, 1.0))

    # Linearly decreasing
    a = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(normalize_to_sum(a, 1.0))

    # Two-block (first half high)
    a = np.ones(n)
    a[n // 2:] = 0.3
    inits.append(normalize_to_sum(a, 1.0))

    # Two-block (first quarter high)
    a = np.ones(n)
    a[n // 4:] = 0.2
    inits.append(normalize_to_sum(a, 1.0))

    # Exponential decay
    a = np.exp(-np.arange(n, dtype=np.float64) / (n * 0.3))
    inits.append(normalize_to_sum(a, 1.0))

    # Exponential growth
    a = np.exp(np.arange(n, dtype=np.float64) / (n * 0.3))
    inits.append(normalize_to_sum(a, 1.0))

    # Cosine modulation
    for phase in [0, np.pi / 2, np.pi / 3]:
        a = 1.0 + 0.7 * np.cos(2 * np.pi * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
        inits.append(normalize_to_sum(a, 1.0))

    # V-shape (non-symmetric: higher at start than end)
    a = 1.0 + 0.8 * np.abs(2 * np.arange(n) / n - 0.4)
    inits.append(normalize_to_sum(a, 1.0))

    # Random
    a = np.abs(rng.normal(1.0, 0.5, size=n))
    inits.append(normalize_to_sum(a, 1.0))

    # Random (another)
    a = np.abs(rng.normal(1.0, 1.0, size=n))
    inits.append(normalize_to_sum(a, 1.0))

    # Power law p=2
    a = (np.arange(1, n + 1, dtype=np.float64)) ** 2
    inits.append(normalize_to_sum(a, 1.0))

    # Power law p=0.5
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inits.append(normalize_to_sum(a, 1.0))

    return inits
```
