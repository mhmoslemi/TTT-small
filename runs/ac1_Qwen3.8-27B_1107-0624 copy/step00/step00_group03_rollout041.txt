with log-sum-exp smoothing of the max, combined with multi-length search
and perturbation polishing.

import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view


def _proj(v, target):
    """Euclidean projection onto {x>=0, sum(x)=target}."""
    lo, hi = np.min(v) - target, np.max(v) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.maximum(v - mid, 0.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - 0.5 * (lo + hi), 0.0)


def _conv(s):
    """Full linear convolution s*s via FFT."""
    n = len(s)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    S = np.fft.rfft(s, sz)
    return np.fft.irfft(S * S, sz)[:2 * n - 1]


def _grad(s, eps):
    """Return (c, obj, gradient) where c=conv(s,s), obj=logsumexp(c/eps)."""
    n = len(s)
    c = _conv(s)
    cm = c.max()
    e = np.exp((c - cm) / eps)

    p = e / Z
    g = (2.0 / eps) * (sliding_window_view(p, n) @ s)
    return c, cm / eps + np.log(Z), g
