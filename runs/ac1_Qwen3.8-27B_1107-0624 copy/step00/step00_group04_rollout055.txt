```python
"""
LP-based iterative refinement for minimizing 2*n*max(f*f)/(sum f)^2.
Uses linear programming to find the flattest cross-correlation direction,
then performs line search to update the sequence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix_row(f, k):
    """Build row k of the convolution matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    row = np.zeros(n)
    # (f*g)[k] = sum_i f[k-i] * g[i] for valid i
    # i ranges from max(0, k-n+1) to min(k, n-1)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    for i in range(i_start, i_end + 1):
        row[i] = f[k - i]
    return row


def solve_lp(f, active_indices, S):
    """
    Solve the LP: minimize M' subject to (f*g)[k] <= M' for k in active_indices,
    sum(g) = S, g >= 0, M' >= 0.
    Returns (g, M') or (None, None) if infeasible.
    """
    n = len(f)
    m = len(active_indices)
    
    if m == 0:
        return np.ones(n) * S / n, S  # degenerate case
    
    # Objective: minimize M' (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (f*g)[k] - M' <= 0 for k in active
    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A_ub[idx, i] = f[k - i]
        A_ub[idx, -1] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    g = np.clip(res.x[:n], 0, None)
    M_prime = res.x[-1]
    return g, M_prime


def compute_score(f):
    """Compute the objective value."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2 * n * np.max(conv) / s**2)

def local_search(f, time_budget, seed=42):

    Simple local search with various perturbation strategies.
    """

    while time.time() < deadline:
        # Choose perturbation type
        r = rng.random()
        
        if r < 0.3:
            # Block multiplicative perturbation
            candidate = best.copy()
            width = int(rng.integers(max(1, n // 50), max(2, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            end = min(n, start + width)
            candidate[start:end] *= np.exp(rng.normal(0, 0.15))
        
        elif r < 0.5:
            # Add random noise
            candidate = best + rng.normal(0, 0.05 * np.max(best), n)
        
        elif r < 0.7:
            # Reshape: multiply by a smooth random function
            candidate = best.copy()
            # Random smooth mask
            mask = np.ones(n)
            for _ in range(int(rng.integers(1, 5))):
                center = rng.uniform(0, n)
                width = rng.uniform(n/10, n/2)
                amplitude = np.exp(rng.normal(0, 0.3))
                mask *= 1 + (amplitude - 1) * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            candidate = best * mask
        
        elif r < 0.85:
            # Mirror/symmetrize variation
            candidate = best.copy()
            alpha = rng.uniform(0, 1)
            candidate = alpha * candidate + (1 - alpha) * candidate[::-1]
        
        else:
            # Pair swap: move mass from high to low positions
            candidate = best.copy()
            i = int(rng.integers(0, n - 1))
            j = (i + int(rng.integers(1, max(2, n // 4)))) % n
            transfer = rng.uniform(0, 0.1) * best[i]
            candidate[i] -= transfer
            candidate[j] += transfer
        
        candidate = np.clip(candidate, 0, 1000)
        score = compute_score(candidate)
        
        if score < best_score:
            best_score = score
            best = candidate
            iters += 1
            if iters % 100 == 0:
                print(f"    Local search iter {iters}: score={best_score:.6f}")
```

```python
def compute_score(f):
    """Compute the objective value 2*n*max(f*f)/(sum f)^2."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf

def solve_lp(f, active_indices, S):
    """
    Solve: minimize M' s.t. (f*g)[k] <= M' for k in active_indices,
           sum(g) = S, g >= 0, M' >= 0.
    Returns (g_array, M_prime) or (None, None).
    """
    n = len(f)
    m = len(active_indices)
    if m == 0:
        return np.full(n, S / n), S

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active_indices):

        i_end = min(k, n - 1)
```
