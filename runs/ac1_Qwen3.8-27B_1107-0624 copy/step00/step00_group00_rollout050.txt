import numpy as np
import time
from scipy.sparse import coo_matrix
import scipy.optimize as opt

def build_conv_matrix(a):
    """Build sparse matrix C where (C @ g)[k] = (a * g)[k] for full convolution."""
    n = len(a)
    i_vals = np.tile(np.arange(n), n)
    j_vals = np.repeat(np.arange(n), n)
    rows = i_vals + j_vals
    cols = j_vals
    vals = np.repeat(a, n)
    C = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C.tocsr()

def lp_step(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0. Return g or None."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = opt.linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
