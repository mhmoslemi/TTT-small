"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using LP-guided iterations,
projected gradient descent polishing, and multi-start with diverse
initializations. LP finds directions g maximizing sum(g) subject to
(a*g)_k <= M, g>=0; mixing a with g reduces the peak convolution.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    n_cons = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_cons):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _compute_subgrad(b, a, n):
    """Compute subgradient of max(b) w.r.t. a."""
    M = b.max()
    K = np.where(b >= M * (1 - 1e-9))[0]
    grad = np.zeros(n)
    for k in K:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= float(len(K))
    return grad, M
