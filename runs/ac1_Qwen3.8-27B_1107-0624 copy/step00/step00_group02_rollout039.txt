import numpy as np
import time
import math

try:
    from scipy.optimize import differential_evolution
    HAVE_DE = True
except Exception:
    HAVE_DE = False

TARGET = 1.5027
EARLY_RETURN = 1.4980


def score_p(p):
    """Exact objective for a nonnegative sequence, scale normalized to sum 1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf
    s = np.sum(p)
    if not math.isfinite(s) or s <= 0.0:
        return np.inf
    p = p / s
    c = np.convolve(p, p, mode="full")
    m = np.max(c)
    if not math.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def trim_sequence(p, tol=1e-12):
    """Remove exactly-zero (or numerically zero) endpoints and renormalize."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, None)
    n = p.size
    if n == 0:
        return p
    idx = np.where(p > tol)[0]
    if idx.size == 0:
        s = np.sum(p)
        if s > 0:
            return p / s
        return p
    p = p[idx[0]:idx[-1] + 1]
    s = np.sum(p)
    if s > 0:
        p = p / s
    return p


def finished_candidate_score(p):
    """Score after endpoint trimming, which is valid because zeros at the ends can be dropped."""
    q = trim_sequence(p)
    return score_p(q)


def p_from_y(y):
    """Convert simplex coordinates y (first weight is reference) to probabilities."""
    y = np.asarray(y, dtype=np.float64)
    m = y.size
    if m == 0:
        return np.array([1.0], dtype=np.float64)
    shift = np.max(y)
    e = np.exp(y - shift)
    denom = 1.0 + np.sum(e)
    if denom <= 0 or not math.isfinite(denom):
        return np.ones(m + 1, dtype=np.float64) / (m + 1)
    p0 = 1.0 / denom
    p = np.empty(m + 1, dtype=np.float64)
    p[0] = p0
    p[1:] = p0 * e
    s = np.sum(p)
    if s > 0:
        p /= s
    return p


def de_objective(y, n):
    """Objective for scipy differential_evolution.  n is passed via args."""
    p = p_from_y(np.asarray(y, dtype=np.float64))
    return score_p(p)


def softmax_weight(c, beta):
    """Stable softmax over convolution coefficients."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=np.float64)
    m = np.max(c)
    e = np.exp(beta * (c - m))
    z = np.sum(e)
    if z <= 0 or not math.isfinite(z):
        return np.full(c.shape, 1.0 / c.size, dtype=np.float64)
    return e / z


def hankel_direct(p, w):
    """
    Gradient of sum_k w_k * (p*p)_k with respect to p.
    This is 2 * T_w p where (T_w p)_i = sum_j p_j w_{i+j}.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    out = np.empty(n, dtype=np.float64)
    for i in range(n):
        out[i] = 2.0 * np.dot(p, w[i:i + n])
    return out


def improve_small(p, max_iter=2000, seed=0):
    """
    Small-scale local/simulated annealer for a fixed length sequence.
    Returns the best trimmed sequence and its score found during the run.
    """
    p = np.asarray(p, dtype=np.float64).copy()
    n = p.size
    if n <= 0 or max_iter <= 0:
        return p, score_p(p)

    s = np.sum(p)
    if not math.isfinite(s) or s <= 0:
        p = np.ones(n, dtype=np.float64) / n
    else:
        p = p / s

    p = np.maximum(p, 1e-300)
    p = p / np.sum(p)

    rng = np.random.default_rng(seed)
    curr = score_p(p)
    best = curr
    best_p = p.copy()

    beta_multipliers = np.array([2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0])
    step_factors = np.array([2.0, 1.0, 0.5, 0.25, 0.1, 0.05])

    phase1 = int(max(10, 0.6 * max_iter))
    phase2 = max(0, max_iter - phase1)
    temp = 0.10

    # Exploration phase: accept worse moves with metropolis probability.
    for it in range(phase1):
        if it % 10 == 0:
            temp *= 0.995
            if curr > best + 0.25 and it % 200 == 0:
                p = best_p.copy()
                curr = best

        r = rng.random()
        blob = p.size
        if r < 0.62 and blob > 1:
            c = np.convolve(p, p)
            w = softmax_weight(c, blob * float(rng.choice(beta_multipliers)))
            g = hankel_direct(p, w)
            max_g = np.max(np.abs(g))
            if (not math.isfinite(max_g)) or max_g < 1e-16:
                scale = float(rng.choice([0.2, 0.5, 1.0, 2.0]))
                lp = np.log(p + 1e-300) + rng.standard_normal(blob) * scale
                lp -= np.max(lp)
                p2 = np.exp(lp)
                p2 /= p2.sum()
            else:
                eta = float(rng.choice(step_factors)) / max_g
                lp = np.log(p + 1e-300) - eta * g
                lp -= np.max(lp)
                p2 = np.exp(lp)
                p2 /= p2.sum()
        else:
            scale = float(rng.choice([0.1, 0.2, 0.5, 1.0, 2.0]))
            lp = np.log(p + 1e-300) + rng.standard_normal(blob) * scale
            lp -= np.max(lp)
            p2 = np.exp(lp)
            p2 /= p2.sum()

        sc2 = score_p(p2)
        if math.isfinite(sc2):
            if sc2 < curr:
                p = p2
                curr = sc2
            else:
                if temp > 1e-12:
                    ratio = (curr - sc2) / temp
                    if ratio > -745.0:
                        if rng.random() < math.exp(ratio):
                            p = p2
                            curr = sc2
            if sc2 < best:
                best = sc2
                best_p = p2.copy()

    # Exploitation phase: only accept improvements, with small random escapes.
    if phase2 > 0:
        p = best_p.copy()
        curr = best
        stall = 0
        line_factors = np.array([1.0, 0.5, 0.25, 0.10, 0.05, 0.02])
        high_betas = np.array([20.0, 50.0, 100.0, 200.0, 500.0])

        for it in range(phase2):
            c = np.convolve(p, p)
            w = softmax_weight(c, p.size * float(rng.choice(high_betas)))
            g = hankel_direct(p, w)
            max_g = np.max(np.abs(g))

            accepted = False
            if math.isfinite(max_g) and max_g >= 1e-16:
                eta = 1.0 / max_g
                for f in line_factors:
                    lp = np.log(p + 1e-300) - eta * f * g
                    lp -= np.max(lp)
                    p2 = np.exp(lp)
                    s2 = p2.sum()
                    if s2 <= 0 or not math.isfinite(s2):
                        continue
                    p2 /= s2
                    sc2 = score_p(p2)
                    if math.isfinite(sc2) and sc2 < curr - 1e-13:
                        p = p2
                        curr = sc2
                        best = sc2
                        best_p = p2.copy()
                        accepted = True
                        break

            if not accepted:
                scale = float(rng.choice([0.3, 0.15, 0.07, 0.03]))
                lp = np.log(p + 1e-300) + rng.standard_normal(p.size) * scale
                lp -= np.max(lp)
                p2 = np.exp(lp)
                s2 = p2.sum()
                if s2 > 0 and math.isfinite(s2):
                    p2 /= s2
                    sc2 = score_p(p2)
                    if math.isfinite(sc2) and sc2 < curr:
                        p = p2
                        curr = sc2
                        best = sc2
                        best_p = p2.copy()
                        stall = 0
                        continue
                stall += 1
                if stall > 120 and it > 50:
                    break

    return best_p, best


def make_starts(n, seed, count=4):
    """Several cheap non-symmetric starts for a fixed small length."""
    rng = np.random.default_rng(seed)
    starts = []

    starts.append(rng.dirichlet(np.ones(n, dtype=np.float64) * 1.0))
    starts.append(rng.dirichlet(np.ones(n, dtype=np.float64) * 0.35))

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)

    p = np.exp(-2.0 * t) + 1e-6
    starts.append(p / p.sum())

    p = np.exp(-2.0 * (1.0 - t)) + 1e-6
    starts.append(p / p.sum())

    p = np.abs(rng.standard_normal(n)) + 0.05
    starts.append(p / p.sum())

    p = 1.0 + 0.45 * np.cos(2.0 * np.pi * (t + 0.17)) + 0.05
    p = np.maximum(p, 1e-4)
    starts.append(p / p.sum())

    p = rng.random(n)
    p = np.where(p > 0.2, p, 1e-5)
    starts.append(p / p.sum())

    if len(starts) > count:
        return starts[:count]
    return starts


def de_params(n):
    """DE hyperparameters by dimension."""
    if n <= 5:
        return 180, 90
    if n <= 8:
        return 140, 80
    if n <= 12:
        return 100, 70
    if n <= 16:
        return 70, 60
    if n <= 22:
        return 50, 50
    return 30, 40


def de_search(n, seed, maxiter, popsize):
    """Run differential evolution for the softmax-parameterized sequence of length n."""
    if not HAVE_DE:
        return None, np.inf
    m = n - 1
    if m <= 0:
        return np.array([1.0], dtype=np.float64), score_p(np.array([1.0], dtype=np.float64))

    bounds = [(-20.0, 20.0)] * m
    try:
        res = differential_evolution(
            de_objective,
            bounds,
            args=(n,),
            seed=seed,
            maxiter=maxiter,
            popsize=popsize,
            tol=1e-12,
            atol=1e-15,
            polish=False,
            disp=False
        )
    except Exception:
        return None, np.inf

    if res is None or not hasattr(res, "x"):
        return None, np.inf

    p = p_from_y(np.asarray(res.x, dtype=np.float64))
    sc = score_p(p)
    return p, sc


def fft_score(p, L, n):
    """FFT-based score for long sequences. Uses the true convolution DFT A^2."""
    A = np.fft.fft(p, L)
    c = np.fft.ifft(A * A)
    c = c[:2 * n - 1].real
    np.maximum(c, 0.0, out=c)
    m = c.max()
    if not math.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def softmax_large(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not math.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / z.size
    return e / s


def optimize_large(n, seed, time_budget):
    """
    Backup long-sequence gradient/annealer using FFT.
    This is less aggressive than the small-n global search, but may help if small n fails.
    """
    if n <= 0 or time_budget <= 5.0:
        return None, np.inf

    L = 1
    while L < 2 * n:
        L *= 2

    rng = np.random.default_rng(seed)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Non-symmetric positive start.
    base = np.exp(-2.0 * t) + 0.35 * np.exp(-1.0 * (1.0 - t)) + 0.25 * np.exp(-3.0 * (1.0 - t))
    p = base + 0.25 * np.abs(rng.standard_normal(n))
    p = np.maximum(p, 1e-8)
    p = p / p.sum()
    z = np.log(p)

    curr = fft_score(p, L, n)
    best = curr
    best_p = p.copy()

    beta_multipliers = np.array([5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0])
    step_factors = np.array([2.0, 1.0, 0.5, 0.25, 0.1, 0.05])

    start = time.monotonic()
    end = start + max(0.0, float(time_budget))
    it = 0

    while time.monotonic() < end:
        p = softmax_large(z)
        A = np.fft.fft(p, L)
        c = np.fft.ifft(A * A)
        c = c[:2 * n - 1].real
        np.maximum(c, 0.0, out=c)
        m = c.max()
        curr = float(2.0 * n * m)
        if math.isfinite(curr) and curr < best:
            best = curr
            best_p = p.copy()

        w = softmax_weight(c, n * float(rng.choice(beta_multipliers)))
        W = np.fft.fft(w, L)
        g = 2.0 * np.real(np.fft.ifft(np.conj(A) * W))[:n]
        max_g = np.max(np.abs(g))

        if (not math.isfinite(max_g)) or max_g < 1e-16:
            z = z + rng.standard_normal(n) * 0.02
            it += 1
            continue

        factor = float(rng.choice(step_factors))
        z2 = z - factor * (g / max_g)
        p2 = softmax_large(z2)
        sc2 = fft_score(p2, L, n)

        if math.isfinite(sc2) and sc2 < curr - 1e-12:
            z = z2
        elif rng.random() < 0.25:
            # Mild random kick to escape flat/oscillatory regions.
            z = z + rng.standard_normal(n) * 0.03

        it += 1
        if it % 500 == 0:
            print(f"large n={n} it={it} best={best:.10f}")

    return best_p, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    1) small fixed-length global DE plus annealing/local polishing,
    2) occasional trimming to shorter valid sequences,
    3) FFT-based large-length backup if small lengths did not beat the target.
    """
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s) - 30.0)
    deadline = t0 + budget

    # A valid conservative fallback.  score = 1.92.
    best_p = np.array([0.4, 0.0, 0.3, 0.3], dtype=np.float64)
    best_score = finished_candidate_score(best_p)
    print(f"initial best score={best_score:.10f}, len={best_p.size}")

    n_list = [4, 5, 6, 7, 8, 9, 10, 11, 12, 14, 16, 18, 20, 24, 28, 32, 40]

    def update_best(q):
        nonlocal best_p, best_score
        try:
            q = trim_sequence(q)
            sc = score_p(q)
            if math.isfinite(sc) and sc < best_score:
                best_p = q.copy()
                best_score = sc
        except Exception:
            pass

    # Initial cheap polishing of a few random starts at small lengths.
    for n in [4, 5, 6, 7, 8, 10]:
        if time.monotonic() > deadline - 60:
            break
        starts = make_starts(n, seed=100 + n, count=2)
        for si, st in enumerate(starts):
            if time.monotonic() > deadline - 60:
                break
            try:
                pp, ss_full = improve_small(st, max_iter=500, seed=seed + 1000 + n * 31 + si)
                q = trim_sequence(pp)
                sc = score_p(q)
                if math.isfinite(sc) and sc < best_score:
                    best_p = q.copy()
                    best_score = sc
                    print(f"early small n={n} score={best_score:.10f}")
            except Exception:
                pass

    if best_score < EARLY_RETURN:
        try:
            pp, ss = improve_small(best_p, max_iter=2000, seed=seed + 7777)
            q = trim_sequence(pp)
            sc = score_p(q)
            if math.isfinite(sc) and sc < best_score:
                best_p = q.copy()
                best_score = sc
        except Exception:
            pass
        return [float(x) for x in best_p]

    # Main small-length search.
    for n in n_list:
        now = time.monotonic()
        if now > deadline - 30:
            break
        if now > deadline - 120 and n > 10:
            # Keep enough time for the large backup and final return.
            break

        de_found = False
        if HAVE_DE:
            maxiter, popsize = de_params(n)
            # If time is tight, do not start heavier DE runs.
            if now > deadline - 90 and n > 8:
                continue
            try:
                p_de, sc_de = de_search(n, seed=10000 + n, maxiter=maxiter, popsize=popsize)
                if p_de is not None and math.isfinite(sc_de):
                    update_best(p_de)
                    de_found = True
                    try:
                        polish_iters = 2500 if n <= 10 else 1200 if n <= 20 else 600
                        pp, ss = improve_small(p_de, max_iter=polish_iters, seed=seed + 20000 + n * 131)
                        update_best(pp)
                    except Exception:
                        pass
            except Exception:
                pass

        # Random local/annealed starts.  More if DE is unavailable or did not help.
        start_count = 3
        if not HAVE_DE or not de_found:
            start_count = 6
        if n > 16 and HAVE_DE:
            start_count = min(start_count, 2)

        starts = make_starts(n, seed=30000 + n, count=start_count)
        for si, st in enumerate(starts):
            if time.monotonic() > deadline - 30:
                break
            try:
                polish_iters = 800 if n <= 12 else 450 if n <= 24 else 250
                pp, ss = improve_small(st, max_iter=polish_iters, seed=seed + 50000 + n * 97 + si)
                update_best(pp)
            except Exception:
                pass

        if best_score < EARLY_RETURN:
            try:
                pp, ss = improve_small(best_p, max_iter=2000, seed=seed + 99999 + n)
                update_best(pp)
            except Exception:
                pass
            return [float(x) for x in best_p]

        if best_score < TARGET and time.monotonic() > deadline - 60:
            return [float(x) for x in best_p]

        print(f"after n={n}: best={best_score:.10f}, len={best_p.size}, time={time.monotonic()-t0:.1f}s")

    # Large-length FFT backup.
    if best_score > TARGET or (time.monotonic() < deadline - 60):
        remaining = deadline - time.monotonic()
        if remaining > 90:
            try:
                p_big, sc_big = optimize_large(4096, seed=seed + 12345, time_budget=min(remaining - 40, 700))
                if p_big is not None:
                    update_best(p_big)
            except Exception:
                pass
            if best_score < EARLY_RETURN:
                return [float(x) for x in best_p]

        if time.monotonic() < deadline - 40:
            try:
                p_mid, sc_mid = optimize_large(2048, seed=seed + 67890, time_budget=min(deadline - time.monotonic() - 10, 300))
                if p_mid is not None:
                    update_best(p_mid)
            except Exception:
                pass

    # Final chance to polish the best known sequence if it is short.
    if best_p.size <= 80 and time.monotonic() < deadline - 5:
        try:
            pp, ss = improve_small(best_p, max_iter=3000, seed=seed + 88888)
            update_best(pp)
        except Exception:
            pass

    final_q = trim_sequence(best_p)
    try:
        sc_final = score_p(final_q)
        if math.isfinite(sc_final) and sc_final <= best_score:
            best_p = final_q
            best_score = sc_final
    except Exception:
        pass

    print(f"final best score={best_score:.10f}, len={best_p.size}")
    return [float(x) for x in best_p]
