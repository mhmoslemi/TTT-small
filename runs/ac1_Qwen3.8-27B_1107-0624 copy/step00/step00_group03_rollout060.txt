    sum_s = s.sum()

"""
Minimizes 2*n*max(conv(s,s)) / (sum(s))^2 for non-negative sequences of length n.
Uses gradient descent with analytical gradient, diverse initialization, and
iterative refinement. Exploits scale invariance by projecting gradient
orthogonal to the all-ones vector.
"""
import numpy as np
import time


def compute_score(s):
    n = len(s)
    c = np.convolve(s, s)
    max_c = c.max()
    sum_s = s.sum()
    if sum_s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_s ** 2)


def grad_descent(s, max_steps=5000, lr_init=0.1, lr_decay=0.998, time_limit=None):
    n = len(s)
    s = np.array(s, dtype=np.float64).copy()
    s = np.clip(s, 0.0, 1000.0)
    if s.max() > 0:
        s = s / s.max()
    
    best_s = s.copy()
    best_score = compute_score(s)
    lr = lr_init
    t_start = time.monotonic()
    
    for step in range(max_steps):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        
        c = np.convolve(s, s)
        k_star = int(c.argmax())
        max_c = c[k_star]
        sum_s = s.sum()
        if sum_s < 1e-10:
            break
        
        # Gradient of c[k_star] w.r.t. s[i]
        grad_c = np.zeros(n)
        i_low = max(0, k_star - n + 1)
        i_high = min(n - 1, k_star)
        if i_high >= i_low:
            grad_c[i_low:i_high+1] = 2.0 * s[k_star - i_high:k_star - i_low + 1][::-1]
        
        # Gradient of F = 2n * max_c / sum_s^2
        grad_F = (2.0 * n / sum_s**2) * (grad_c - max_c / sum_s)
        
        # Project to be orthogonal to all-ones (scale invariance)
        grad_F = grad_F - grad_F.mean()
        
        s_new = s - lr * grad_F
        s_new = np.clip(s_new, 0.0, 1000.0)
        
        if s_new.max() > 0:
            s_new = s_new / s_new.max()
        
        score = compute_score(s_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_s = s_new.copy()
            s = s_new
        else:
            lr *= lr_decay
        
        if lr < 1e-12:
            break
    
    return best_s, best_score

def compute_score_and_grad(s):
    """Compute score, argmax location, and gradient simultaneously."""
    n = len(s)
    c = np.convolve(s, s)
    k_star = int(c.argmax())
    max_c = c[k_star]
    sum_s = s.sum()
    if sum_s < 0.01:

    score = 2.0 * n * max_c / (sum_s ** 2)

    grad = (2.0 * n / sum_s**2) * (grad_c - max_c / sum_s)
    
    return score, k_star, grad
