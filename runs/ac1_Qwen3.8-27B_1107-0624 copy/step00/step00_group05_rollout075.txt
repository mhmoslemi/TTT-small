import numpy as np

def fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


def peak_gradient(a, k_star):
    """Gradient of conv[k_star] w.r.t. a. O(n)."""
    n = len(a)
    grad = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        j = np.arange(j_min, j_max + 1)
        grad[j] = 2.0 * a[k_star - j]
    return grad


def softmax_gradient(a, T):
    """
    Compute gradient of softmax(T * conv) w.r.t. a using FFT cross-correlation.
    O(n log n).
    """
    n = len(a)
    conv = fft_convolve(a)
    # Softmax weights
    conv_shifted = T * conv - T * conv.max()
    weights = np.exp(conv_shifted)
    weights /= weights.sum()
    # Cross-correlation: grad[j] = 2 * sum_i weights[j+i] * a[i]
    # = 2 * cross_correlation(weights, a)[j]

    fa = np.fft.rfft(a, N)

    # Build initializations
    inits = []

    # 1. Base (provided)
    s = np.sum(base)
    if s > 0.01:
        inits.append(base / s)

    # 2. Linear ramp (increasing)
    a = np.linspace(0.5, 1.5, n)
    inits.append(a / np.sum(a))

    # 3. Linear ramp (decreasing)
    a = np.linspace(1.5, 0.5, n)
    inits.append(a / np.sum(a))

    # 4. Cosine bump (high in middle)
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.4 * np.cos(np.pi * x)
    a = np.clip(a, 0.1, None)
    inits.append(a / np.sum(a))

    # 5. U-shape (high at ends)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * x)
    a = np.clip(a, 0.1, None)
    inits.append(a / np.sum(a))

    # 6. Exponential decay
    a = np.exp(-0.3 * x)
    inits.append(a / np.sum(a))

    # 7. Exponential rise
    a = np.exp(0.3 * x)
    inits.append(a / np.sum(a))

    # 8. Quadratic bowl
    a = 1.0 + 0.5 * (2.0 * x - 1.0) ** 2
    inits.append(a / np.sum(a))

    # 9. Inverse cosine (peaked at center)
    a = 1.0 + 0.3 * np.sin(np.pi * x)
    inits.append(a / np.sum(a))

    # 10. Two-hump
    a = 1.0 + 0.4 * np.cos(4.0 * np.pi * x)
    a = np.clip(a, 0.05, None)
    inits.append(a / np.sum(a))

    best_score = np.inf
    best_seq = inits[0].copy()

    # Phase 1: Try each initialization with gradient descent
    n_inits = len(inits)
    time_per_init = 1780.0 / n_inits - 200.0  # leave 200s for final refinement
    time_per_init = max(time_per_init, 30.0)

    for idx in range(n_inits):
        if time.monotonic() > deadline - 200:
            break
        init_seq = inits[idx].copy()
        phase_deadline = min(time.monotonic() + time_per_init, deadline - 200)

        # Run soft-max gradient descent
        seq, peak = run_gradient_descent(
            init_seq, phase_deadline,
            T_init=20.0, T_max=500.0, lr_init=0.1
        )

        # Quick score check (approximate: 2*n*peak for sum=1 sequence)
        approx_score = 2.0 * n * peak
        if approx_score < best_score:
            best_score = approx_score
            best_seq = seq.copy()
            print(f"Init {idx}: approx_score={approx_score:.6f}, peak={peak:.10f}")

    # Phase 2: Final refinement on the best sequence
    if time.monotonic() < deadline - 10:
        print(f"\nPhase 2: Refining best (approx={best_score:.6f})")

        a = best_seq.copy()
        s = np.sum(a)
        a = a / s

        # Hard peak refinement
        phase2_deadline = min(time.monotonic() + 300.0, deadline - 60)
        a, peak = run_hard_peak_refine(a, phase2_deadline)
        approx_score = 2.0 * n * peak
        if approx_score < best_score:
            best_score = approx_score
            best_seq = a.copy()
            print(f"After hard refine: approx={approx_score:.6f}")

        # Soft gradient again with higher T
        if time.monotonic() < deadline - 60:
            phase2b_deadline = min(time.monotonic() + 200.0, deadline - 60)
            a = best_seq.copy()
            a = a / np.sum(a)
            a, peak = run_gradient_descent(
                a, phase2b_deadline,
                T_init=200.0, T_max=2000.0, lr_init=0.05
            )
            approx_score = 2.0 * n * peak
            if approx_score < best_score:
                best_score = approx_score
                best_seq = a.copy()
                print(f"After soft refine 2: approx={approx_score:.6f}")

        # Phase 3: Block perturbation search
        if time.monotonic() < deadline - 10:
            print(f"\nPhase 3: Block perturbation search")
            a = best_seq.copy()
            a = a / np.sum(a)
            conv = fft_convolve(a)
            current_peak = float(np.max(conv))
            s = np.sum(a)
            best_block_score = 2.0 * n * current_peak

            it = 0
            no_improve = 0
            while time.monotonic() < deadline - 5:
                # Random block perturbation with varying width
                width = int(rng_global.integers(max(1, n // 500), max(2, n // 5)))
                start = int(rng_global.integers(0, max(1, n - width)))
                stop = min(n, start + width)

                a_new = a.copy()
                # Exponential perturbation
                scale = np.exp(rng_global.normal(0, 0.08, stop - start))
                a_new[start:stop] *= scale
                a_new = np.maximum(a_new, 0.0)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    it += 1
                    continue
                a_new = a_new / s_new

                conv_new = fft_convolve(a_new)
                peak_new = float(np.max(conv_new))

                if peak_new < current_peak:
                    current_peak = peak_new
                    a = a_new
                    conv = conv_new
                    no_improve = 0
                    block_score = 2.0 * n * current_peak
                    if block_score < best_block_score:
                        best_block_score = block_score
                        if block_score < best_score:
                            best_score = block_score
                            best_seq = a_new.copy()
                else:
                    no_improve += 1

                it += 1
                if it % 5000 == 0:
                    print(f"  Block iter {it}: peak={current_peak:.10f}, score~{2.0*n*current_peak:.6f}")
                if no_improve > 20000:
                    break

        # Final hard-peak polish
        if time.monotonic() < deadline - 3:
            a = best_seq.copy()
            a = a / np.sum(a)
            a, peak = run_hard_peak_refine(a, deadline - 2)
            approx_score = 2.0 * n * peak
            if approx_score < best_score:
                best_score = approx_score
                best_seq = a.copy()
                print(f"Final polish: approx={approx_score:.6f}")

    # Verify with actual evaluator
    final_list = [float(x) for x in best_seq]
    try:
        actual_score = float(evaluate_sequence(final_list))
        print(f"\nActual evaluated score: {actual_score:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
        actual_score = np.inf

    return final_list
