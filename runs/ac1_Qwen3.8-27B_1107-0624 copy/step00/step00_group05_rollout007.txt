"""
LP-based Frank-Wolfe (conditional gradient) method for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences a.

At each iteration, we solve the linear subproblem:
    max sum(g)  s.t.  (f*g)[k] <= M for all k,  g >= 0
where f is the current sequence and M = max(f*f).
We then take a line-search step towards the LP solution.
Multiple initializations (uniform, flat-spectrum, random) and
periodic restarts are used to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_A_csr(f):
    """Build sparse matrix A (shape (2n-1, n)) such that A @ g = convolve(f, g)."""
    n = len(f)
    n_rows = 2 * n - 1
    indptr = np.zeros(n_rows + 1, dtype=np.int32)
    indices = []
    data = []

    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        # A[k, j] = f[k - j] for j in [j_start, j_end]
        # values in order of j: f[k-j_end], f[k-j_end+1], ..., f[k-j_start]
        data.extend(f[k - j_end : k - j_start + 1])
        indices.extend(range(j_start, j_end + 1))
        indptr[k + 1] = indptr[k] + count

    indices = np.array(indices, dtype=np.int32)
    data = np.array(data, dtype=np.float64)
    return csr_matrix((data, indices, indptr), shape=(n_rows, n))


def solve_lp_direction(f, M, time_limit=25.0):
    """Solve LP: max sum(g) s.t. f*g <= M (element-wise), g >= 0. Returns g or None."""
    n = len(f)
    A = build_A_csr(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True,
                           'dual_feasibility_tolerance': 1e-7,
                           'primal_feasibility_tolerance': 1e-7})

    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def eval_fast(f):
    """Fast evaluation using direct numpy (avoids list conversion overhead)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

        A[n - k] = np.exp(-1j * phi)
    a = np.real(np.fft.ifft(A))
    # Shift to make non-negative
    a = a - a.min() + 0.01
    a = np.clip(a, 0.0, 1000.0)
    return a


def generate_random_init(n, rng):
    """Generate a random non-negative sequence."""
    a = rng.random(n)
    # Add some structure: mix with a smooth component
    smooth = np.ones(n)
    t = rng.uniform(0.3, 0.7)
    a = t * smooth + (1 - t) * a
    return np.clip(a, 0.0, 1000.0)


def lp_iteration(f, time_limit=25.0):
    """One Frank-Wolfe iteration: solve LP, take line-search step."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(conv.max())
    current_score = float(2.0 * n * M / (f.sum() ** 2))

    g = solve_lp_direction(f, M, time_limit=time_limit)
    if g is None:
        return f, current_score

    # Check if g gives improvement
    g_sum = g.sum()
    if g_sum <= f.sum() * (1 + 1e-10):
        return f, current_score

    # Normalize g to have same sum as f (optional, for step interpretation)
    # Actually, we search over all t in [0, 1] with the raw g
    # But to bound the step, let's normalize: g_norm = g * (sum_f / sum_g)
    # Then (1-t)f + t*g_norm has sum = sum_f (convex combo of two points with same sum)
    # This keeps the sum constant, isolating the shape change.
    g_norm = g * (f.sum() / g_sum)

    # Line search: try several step sizes
    best_t = 0.0
    best_score = current_score
    t = 1.0
    for _ in range(12):
        candidate = (1.0 - t) * f + t * g_norm
        candidate = np.clip(candidate, 0.0, 1000.0)
        if candidate.sum() < 0.01:
            t *= 0.5
            continue
        score = eval_fast(candidate)
        if score < best_score:
            best_score = score
            best_t = t
        t *= 0.5
        if t < 1e-6:
            break

    if best_t > 0:
        f_new = (1.0 - best_t) * f + best_t * g_norm
        f_new = np.clip(f_new, 0.0, 1000.0)
        return f_new, best_score
    return f, current_score


def restart_perturb(f, rng, strength=0.3):
    """Perturb the sequence to escape local minima."""
    n = len(f)
    f_new = f.copy()
    # Random block scaling
    n_blocks = rng.integers(1, 5)
    for _ in range(n_blocks):
        width = int(rng.integers(max(1, n // 50), max(2, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        f_new[start:start + width] *= np.exp(rng.normal(0, strength))
    f_new = np.clip(f_new, 0.0, 1000.0)
    return f_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: LP-based Frank-Wolfe with multiple initializations and restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    # --- Phase 1: Generate and evaluate initializations ---
    candidates = []

    # Given sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0.01:
            candidates.append(parent.copy())
    except Exception:
        pass

    # Uniform sequences of various lengths
    for n in [256, 512, 1024, 2048, 4096, 4498]:
        candidates.append(np.ones(n))

    # Flat-spectrum initializations
    for n in [512, 1024, 2048, 4096]:
        for trial in range(3):
            candidates.append(generate_flat_spectrum_init(n, rng))

    # Random initializations
    for n in [512, 1024, 2048, 4096]:
        for trial in range(2):
            candidates.append(generate_random_init(n, rng))

    # Evaluate all candidates
    best_score = np.inf
    best_f = None
    for c in candidates:
        if time.monotonic() > deadline:
            break
        s = eval_fast(c)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_f = c.copy()

    print(f"Phase 1 done. Best init: score={best_score:.6f}, n={len(best_f)}")

    # --- Phase 2: LP-based optimization ---
    f = best_f.copy()
    f_score = best_score
    no_improve_count = 0
    iteration = 0

    while time.monotonic() < deadline:
        iter_start = time.monotonic()
        time_left = deadline - iter_start
        if time_left < 30:
            break

        # Adaptive time limit for LP
        lp_time = min(25.0, time_left * 0.4)

        f, new_score = lp_iteration(f, time_limit=lp_time)
        iteration += 1

        if new_score < f_score - 1e-12:
            f_score = new_score
            no_improve_count = 0
            print(f"Iter {iteration}: score={f_score:.6f}, n={len(f)}, "
                  f"improved, t={iter_start - time.monotonic():.1f}s remain={time_left:.0f}s")
        else:
            no_improve_count += 1

        # Periodic restart if stuck
        if no_improve_count >= 20:
            print(f"Iter {iteration}: no improvement for {no_improve_count} iters, restarting")
            no_improve_count = 0
            # Try perturbation
            f_pert = restart_perturb(f, rng, strength=0.5)
            s_pert = eval_fast(f_pert)
            if s_pert < f_score:
                f = f_pert
                f_score = s_pert
            else:
                # Try a completely new initialization
                n_new = rng.choice([512, 1024, 2048, 4096, 4498])
                if rng.random() < 0.5:
                    f = generate_flat_spectrum_init(int(n_new), rng)
                else:
                    f = generate_random_init(int(n_new), rng)
                f_score = eval_fast(f)
                print(f"  New random init: score={f_score:.6f}, n={len(f)}")

        # Every 30 iterations, also try a fresh structured init
        if iteration % 30 == 0 and time.monotonic() < deadline:
            n_try = rng.choice([1024, 2048, 4096])
            f_try = generate_flat_spectrum_init(int(n_try), rng)
            s_try = eval_fast(f_try)
            if s_try < f_score * 0.95:  # significantly better
                print(f"  Found better init: {s_try:.6f} vs {f_score:.6f}")
                f = f_try
                f_score = s_try
                no_improve_count = 0

    # --- Phase 3: Final verification with exact evaluator ---
    final_score = float(evaluate_sequence([float(x) for x in f]))
    print(f"Final score (exact): {final_score:.6f}, n={len(f)}")

    return [float(x) for x in f]
