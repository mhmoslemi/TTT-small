```python
def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.fft(a, size)
    fb = np.fft.fft(b, size)
    result = np.fft.ifft(fa * fb).real
    return result[:n+m-1]
```

2. **Simplex projection** (proj onto {x : sum x = 1, x ≥ 0}):
```python
def project_simplex(v, s=1.0):
    n = len(v)
    if v.sum() == 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
