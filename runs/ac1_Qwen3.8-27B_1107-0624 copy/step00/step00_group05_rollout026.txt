    g = np.zeros(n)
    for m in active_positions:
        # First term: f[m-j] for j where 0 <= m-j < n
        # j ranges from max(0, m-n+1) to min(n-1, m)
        j_start = max(0, m - n + 1)

        j_start2 = max(0, m)

        if j_start2 < j_end2:
            idx2 = np.arange(j_start2, j_end2) - m

def _lp_step(f, max_b, M_limit=2.0):

    LP-based improvement: find g that maximizes sum(g)

    cols = []

                cols.append(j)

def compute_subgradient_fft(f, h, n):
    """
    Compute the subgradient g[j] = sum_m h[m] * (f[m-j]*I(valid) + f[j-m]*I(valid))
    using FFT.
    
    Parameters:
    f: array of length n (the sequence)
    h: array of length 2n-1 (weights at each convolution position)
    n: length of f
    """
    L = 2 * n - 1
    # Flip h
    h_flip = h[::-1]  # h_flip[k] = h[L-1-k]
    
    # FFT size
    N_fft = 1
    while N_fft < L + n - 1:  # need length for linear convolution
        N_fft *= 2
    
    F = np.fft.rfft(f, N_fft)
    H = np.fft.rfft(h, N_fft)
    HF = np.fft.rfft(h_flip, N_fft)
    
    # Second sum: (f * h)[j] for j = 0, ..., n-1
    conv_fh = np.fft.irfft(F * H, N_fft)[:n]
    
    # First sum: (f * h_flip)[L-1-j] for j = 0, ..., n-1
    # This is (f * h_flip)[m] where m = L-1-j, so m ranges from L-1 to L-1-(n-1) = n
    conv_fhf = np.fft.irfft(F * HF, N_fft)
    # We need conv_fhf[L-1-j] for j = 0, ..., n-1
    # That's conv_fhf[L-1], conv_fhf[L-2], ..., conv_fhf[L-1-(n-1)] = conv_fhf[n]
    # So it's conv_fhf[n:L][::-1] ... let me check indices
    # j=0: m = L-1 = 2n-2
    # j=n-1: m = L-1-(n-1) = n
    # So we need conv_fhf[n : 2n-1] reversed

    g = first_sum + conv_fh
    return g
