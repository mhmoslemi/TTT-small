def proj_simplex(v, target_sum=1.0):
    """Project v onto {x: x_i >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0)

def compute_smoothed_gradient(a, beta, N_fft=None):
    """Compute gradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    c = fftconvolve(a, a)
    # Softmax weights
    c_max = c.max()
    exp_vals = np.exp(beta * (c - c_max))
    w = exp_vals / exp_vals.sum()

    fa = np.fft.rfft(a, N_fft)
