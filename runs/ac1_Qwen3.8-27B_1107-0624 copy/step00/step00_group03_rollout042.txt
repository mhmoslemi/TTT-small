import numpy as np
from scipy.optimize import linprog
import time

def _solve_lp(f, M, n):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0."""
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    return res.x

def _solve_direction(f, n, mode='max_sum', eps=0.01):
    """Solve LP to find direction g.
    mode='max_sum': max sum(g) s.t. A@g <= 1, g>=0
    mode='min_peak': max sum(g) s.t. A@g <= 1, g*g[k] <= 1+eps for k near peak, g>=0
    """
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.ones(2 * n - 1)
    bounds = [(0, None)] * n

    if mode == 'min_peak':
        # Add extra constraints on g*self-convolution near the peak region
        # (f*g) is already bounded; add (g*g)[k] <= 1+eps for k in middle range
        mid_start = n // 2
        mid_end = 3 * n // 2
        extra_rows = mid_end - mid_start
        # Build sub-matrix for g*g constraints in the middle range
        # (g*g)[k] = sum_j g[j]*g[k-j] -- this is quadratic, can't use directly
        # Instead, use a linear approximation: for each k, sum_j g[j] * f[k-j] <= 1
        # which is already in A. So just use the standard LP.
        pass

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _evaluate(f):
    """Compute the score for sequence f (assumed non-negative, finite)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _local_search(f, n, deadline, rng, eval_budget_per_iter=6):
    """Iterative LP-guided local search."""
    best_f = f.copy()
    best_score = _evaluate(f)
    current_score = best_score
    M = np.max(np.convolve(f, f))
    sum_f = np.sum(f)

    # Normalize to avoid numerical issues
    scale = sum_f
    f_work = f / scale  # work with normalized sequence, sum=1
    M_norm = np.max(np.convolve(f_work, f_work))

    iteration = 0
    stagnant = 0
    t_base = 0.3

    while time.monotonic() < deadline:
        # Solve LP for direction
        g_star = _solve_direction(f_work, n)
        if g_star is None or np.sum(g_star) < 1e-10:
            break

        sum_g = np.sum(g_star)
        # Normalize g to have same sum as f_work (which is 1)
        g = g_star / sum_g * 1.0  # sum(g) = 1

        # Check if g is meaningfully different from f_work
        diff = np.max(np.abs(g - f_work))
        if diff < 1e-8:
            stagnant += 1
            if stagnant > 50:
                break
            # Perturb to escape
            f_work = f_work + 0.01 * rng.standard_normal(n)
            f_work = np.clip(f_work, 0, None)
            f_work /= np.sum(f_work)
            M_norm = np.max(np.convolve(f_work, f_work))
            stagnant -= 10
            iteration += 1
            continue

        # Line search: try several step sizes
        best_t = 0.0
        best_new_score = current_score

        step_sizes = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in step_sizes:
            f_new = (1 - t) * f_work + t * g
            f_new = np.clip(f_new, 0, None)
            s = np.sum(f_new)
            if s < 1e-10:
                continue
            f_new = f_new / s  # renormalize to sum=1

            # Quick score computation (no need to call evaluate_sequence for internal comparison)
            conv_new = np.convolve(f_new, f_new)
            max_b_new = np.max(conv_new)
            score_new = 2.0 * n * max_b_new  # since sum=1, score = 2n*max_b/1

            if score_new < best_new_score:
                best_new_score = score_new
                best_t = t

        if best_t > 0:
            f_work = (1 - best_t) * f_work + best_t * g
            f_work = np.clip(f_work, 0, None)
            s = np.sum(f_work)
            if s < 1e-10:
                break
            f_work = f_work / s
            current_score = best_new_score
            stagnant = 0

            if current_score < best_score / (scale ** 2):
                # Update best (actual score accounts for normalization)
                best_score = current_score
                best_f = f_work.copy()
                if iteration % 100 == 0:
                    elapsed = time.monotonic()
                    print(f"  Iter {iteration}: normalized score = {best_score:.6f}, t={best_t:.2f}")
        else:
            stagnant += 1
            if stagnant > 100:
                break
            # Adaptive: try smaller steps
            t_base = max(0.05, t_base * 0.8)

        iteration += 1
        if iteration % 5000 == 0:
            print(f"  Iter {iteration}: current_norm_score = {current_score:.6f}, "
                  f"best_norm_score = {best_score/(scale**2):.6f}")

    # Convert back to original scale for evaluation
    best_f_original = best_f * scale
    final_score = float(evaluate_sequence(best_f_original.tolist()))
    return best_f_original, final_score


def _random_start(n, rng, strategy='uniform'):
    """Generate a random initial sequence."""
    if strategy == 'uniform':
        return np.ones(n) / n
    elif strategy == 'random':
        f = rng.random(n)
        return f / np.sum(f)
    elif strategy == 'smooth':
        t = np.linspace(0, 1, n)
        f = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(0, 3) * t + rng.uniform(0, 6.28))
        f = np.clip(f, 0.01, None)
        return f / np.sum(f)
    elif strategy == 'bump':
        t = np.linspace(0, 1, n)
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        f = 1.0 + 0.5 * np.exp(-((t - center) / width) ** 2)
        return f / np.sum(f)
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    LP-guided local search to minimize the evaluation score.
    Uses multiple restarts with different initializations and strategy switching.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')

    # Strategy: try different n values and initializations
    n_candidates = [256, 128, 512]
    # Weight towards n=256 as it's a good balance
    n_weights = [0.5, 0.2, 0.3]

    strategies = ['uniform', 'random', 'smooth', 'bump']
    strategy_weights = [0.4, 0.2, 0.2, 0.2]

    n_tries = 0
    attempt_deadline = total_deadline

    # First, quick evaluation of uniform for various n to establish baseline
    for n in n_candidates:
        f_uniform = np.ones(n)
        score = float(evaluate_sequence(f_uniform.tolist()))
        print(f"Uniform n={n}: score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_sequence = f_uniform.copy()

    print(f"Initial best: {best_score:.6f}")

    # Main search loop with multiple restarts
    time_per_restart = max(30.0, (budget_s - 30.0) / 8.0)
    max_restarts = int((budget_s - 30.0) / time_per_restart)

    for restart in range(max_restarts):
        remaining = total_deadline - time.monotonic()
        if remaining < 10:
            break

        # Choose n and strategy
        n = rng.choice(n_candidates, p=n_weights)
        strategy = rng.choice(strategies, p=strategy_weights)

        # For later restarts, try to perturb the best found
        if best_sequence is not None and restart > 2 and rng.random() < 0.3:
            n = len(best_sequence)
            f0 = best_sequence / np.sum(best_sequence)
            # Add perturbation
            f0 = f0 + 0.05 * rng.standard_normal(n)
            f0 = np.clip(f0, 0, None)
            f0 = f0 / np.sum(f0)
            strategy = 'perturb_best'
        else:
            f0 = _random_start(n, rng, strategy)

        restart_deadline = min(total_deadline - 5.0, time.monotonic() + time_per_restart)

        print(f"Restart {restart+1}/{max_restarts}: n={n}, strategy={strategy}")

        f_best, score = _local_search(f0, n, restart_deadline, rng)

        if score < best_score:
            best_score = score
            best_sequence = f_best.copy()
            print(f"  NEW BEST: {best_score:.6f} (n={n})")
        else:
            print(f"  Score: {score:.6f} (best remains {best_score:.6f})")

        # Also try a quick random perturbation of the current best
        if best_sequence is not None and time.monotonic() < total_deadline - 20:
            n_best = len(best_sequence)
            for _ in range(20):
                if time.monotonic() >= total_deadline - 10:
                    break
                f_pert = best_sequence.copy()
                # Random block perturbation
                width = rng.integers(max(1, n_best // 50), max(2, n_best // 5))
                start = rng.integers(0, n_best - width + 1)
                f_pert[start:start + width] *= np.exp(rng.normal(0, 0.15))
                f_pert = np.clip(f_pert, 0, 1000)
                s = float(evaluate_sequence(f_pert.tolist()))
                if s < best_score and np.isfinite(s):
                    best_score = s
                    best_sequence = f_pert.copy()
                    print(f"  Perturbation improved: {best_score:.6f}")

    print(f"\nFinal best score: {best_score:.6f} (n={len(best_sequence) if best_sequence is not None else 0})")
    return [float(x) for x in best_sequence]
