```python
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2n*max(a*a)/(sum(a))^2 for non-negative sequences using
    subgradient descent on the auto-convolution peak, with FFT-accelerated
    convolution and periodic randomization to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    # Get parent sequence
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(parent)
    
    # Working normalization: sum = 1
    a = parent.copy()
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a_sum
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a))  # since sum = 1, obj = 2n * max(b)
    
    # FFT-based convolution for speed
    def fast_convolve(x, y):
        size = len(x) + len(y) - 1
        fsize = 1
        while fsize < size:
            fsize *= 2
        return np.real(np.fft.irfft(np.fft.rfft(x, fsize) * np.fft.rfft(y, fsize), fsize)[:size])
    
    K = 50  # number of active lags to use
    
    current_step = 0.01
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = fast_convolve(a, a)
        M = b.max()
        current_obj = 2.0 * n * M  # sum = 1
        
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top-K lags
        # Use partial sort for efficiency
        if len(b) > K:
            top_k_indices = np.argpartition(b, -K)[-K:]
        else:
            top_k_indices = np.arange(len(b))
        
        # Compute subgradient (approximate, using top-K lags)
        grad = np.zeros(n)
        for k in top_k_indices:
            if k < n:
                # dB_k/dA_j = a_{k-j} + a_j for j = 0..k
                # a_{k-j} for j=0..k is a[k::-1][:k+1]
                grad[:k+1] += a[k::-1]  # a[k], a[k-1], ..., a[0] → a[k-j] for j=0..k
                grad[:k+1] += a[:k+1]   # a[0], a[1], ..., a[k] → a[j] for j=0..k
            else:
                # k >= n: dB_k/dA_j = a_{k-j} + a_j for j = k-n+1..n-1
                j_start = k - n + 1
                j_end = n - 1
                # a_{k-j} for j = j_start..j_end: a[k-j_start] down to a[k-j_end]
                # k - j_start = k - (k-n+1) = n-1
                # k - j_end = k - (n-1) = k-n+1 = j_start
                # So a_{k-j} = a[n-1], a[n-2], ..., a[j_start]
                grad[j_start:] += a[n-1::-1][:n-j_start]  # a[n-1], a[n-2], ..., a[j_start]
                grad[j_start:] += a[j_start:]  # a[j_start], ..., a[n-1]
        
        # Normalize and take step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad /= grad_norm
            a = a - current_step * grad
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-10:
                a = a / a_sum
            else:
                a = best_a.copy()
        
        # Adaptive step size
        if no_improve > 100:
            current_step *= 0.95

            # Add random perturbation to escape
            a = a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            current_step = 0.01
        
        if no_improve > 2000:
            # Restart with different approach
            break
    
    # Convert back to original scale for evaluation
    # The objective is scale-invariant, so any scale works
    # But we need sum >= 0.01
    result = best_a * 1.0  # sum = 1, well above 0.01
    return [float(x) for x in result]
```

```python
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Subgradient-based minimization of 2n*max(conv(a,a))/(sum(a))^2 for
    non-negative sequences. Uses top-K lag approximation of the subgradient
    and adaptive step sizes with randomization to escape local optima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(parent)
    
    # Initialize: normalize parent to sum=1
    a = parent.copy()
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / a_sum
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a))
    
    print(f"Initial objective: {best_obj:.6f}, n={n}")
    
    K = 100  # number of top lags for subgradient approximation
    step = 0.05
    no_improve = 0
    iter_count = 0
    
    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    while time.monotonic() < deadline:
        # Compute convolution via FFT
        fa = np.fft.rfft(a, fft_size)
        b = np.real(np.fft.irfft(fa * fa, fft_size))[:2*n-1]
        
        M = b.max()
        current_obj = 2.0 * n * M  # sum(a) = 1
        
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        if iter_count % 1000 == 0:
            print(f"Iter {iter_count}, obj={best_obj:.6f}, step={step:.6f}")
        
        # Find top-K lags
        if 2*n-1 > K:
            top_k = np.argpartition(b, -K)[-K:]
        else:
            top_k = np.arange(2*n-1)
        
        # Compute approximate subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in top_k:
            if k < n:
                # grad[j] += 2*a[k-j] for j=0..k
                grad[:k+1] += 2.0 * a[:k+1][::-1]
            else:
                # grad[j] += 2*a[k-j] for j=k-n+1..n-1
                j_start = k - n + 1
                length = n - j_start
                grad[j_start:] += 2.0 * a[:length][::-1]
        
        # Gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad /= grad_norm
            a = a - step * grad
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-10:
                a = a / a_sum
            else:
                a = best_a.copy()
        
        # Adaptive step size
        if no_improve > 50:
            step *= 0.9
        elif no_improve > 200:
            step = max(0.001, step * 0.5)
        elif no_improve > 500:
            # Random kick to escape local minimum
            noise_scale = 0.05 * best_a.max()
            a = best_a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            step = 0.05
            no_improve = 0
        
        if no_improve > 5000:
            # Try a completely different approach: restart from random
            a = np.abs(rng.standard_normal(n)) + 0.1
            a = a / a.sum()
            step = 0.1
            no_improve = 0
    
    # Final: return best sequence (scale is irrelevant due to scale invariance)
    # Ensure sum >= 0.01
    result = best_a
    if result.sum() < 0.01:
        result = result * 1.0
    return [float(x) for x in result]
```

**Faster approach: Use the "excess" objective with smooth approximation**

Instead of directly minimizing $\max b_k$, minimize $f(a) = \sum_k \max(0, b_k - M)^2$ for a fixed threshold $M$. This is a smooth (well, piecewise smooth) function, and we can use gradient descent with a better convergence rate.

Actually, a simpler approach: use the "soft-min" of the objective. Define:
$f_\beta(a) = \frac{1}{\beta} \log \sum_k \exp(\beta b_k)$

This smooths the max and gives a well-defined gradient:
$\nabla_a f_\beta = \frac{1}{\sum_k \exp(\beta b_k)} \sum_k \exp(\beta b_k) \nabla_a b_k$

For large $\beta$, this approximates the subgradient of $\max b_k$. For small $\beta$, it's smoother and faster to optimize.

The gradient can be computed as:
$w_k = \exp(\beta b_k - \beta \max b)$ (normalized to avoid overflow)
$w = w / \sum w$
$\nabla_a f_\beta[j] = 2 \sum_k w_k a_{k-j} \mathbf{1}[j \in I_k]$

This is a weighted version of the subgradient, with weights $w_k$ that concentrate on the maximum lags for large $\beta$.

The computation is $O(K_{\text{eff}} \cdot n)$ where $K_{\text{eff}}$ is the number of lags with significant weight. For $\beta = 10$ and a "spread out" $b$, $K_{\text{eff}}$ might be 50-200.

```python
def compute_soft_gradient(a, b, beta):
    """Compute gradient of soft-max of b_k."""
    n = len(a)
    b_max = b.max()
    w = np.exp(beta * (b - b_max))
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w = w / w_sum
    
    grad = np.zeros(n)
    # For each k with significant weight:
    # grad[j] += 2 * w[k] * a[k-j] * 1[j in I_k]
    
    # For efficiency, only use lags with w[k] > threshold
    threshold = w.max() * 1e-4
    active = np.where(w > threshold)[0]
    
    for k in active:
        if k < n:
            grad[:k+1] += 2.0 * w[k] * a[:k+1][::-1]
        else:
            j_start = k - n + 1
            if j_start < n:
                length = n - j_start
                grad[j_start:] += 2.0 * w[k] * a[:length][::-1]
    
    return grad
```
