import math

"""
Algorithm: Multiplicative gradient descent (exponentiated gradient) to minimize
the peak autocorrelation of a non-negative sequence. The objective is
2*n*max(f*f)/(sum f)^2. We normalize sum(f)=1 and minimize max(f*f).
The gradient of sum_k (f*f)[k]^2 is computed and used in a multiplicative
update. Learning rate is adapted based on progress. Multiple initializations
and sequence lengths are tried.
"""

def compute_objective_normalized(a):
    """Compute 2*n*max(f*f) for a sequence normalized to sum=1."""
    n = len(a)
    R = np.convolve(a, a)
    return 2.0 * n * R.max()

def multiplicative_grad_descent(a, eta_init, max_iters, deadline, verbose=False):
    """Run multiplicative gradient descent to minimize max autocorrelation."""
    n = len(a)
    a = a.copy()
    # Normalize to sum = 1
    s = a.sum()
    if s < 1e-20:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = compute_objective_normalized(a)
    eta = eta_init
    patience = 0

def lp_direction(f, M):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid indices
    # This is a linear constraint in g.
    # A matrix: A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # So A is a (2n-1) x n matrix (Toeplitz-like)
    
    # For scipy linprog: minimize -sum(g) s.t. A @ g <= M, g >= 0
    # A[k, i] = f[k - i] if 0 <= k - i < n else 0
    
    # Build A efficiently
    # A is a lower-Hankel matrix (or "anti-Toeplitz")
    # A[k, i] = f[k-i] for k >= i (i.e., k-i >= 0 and k-i < n)
    #           = 0 otherwise
    
    # Actually A[k,i] = f[k-i] when 0 <= k-i <= n-1
    # For k from 0 to 2n-2, i from 0 to n-1:
    # k-i ranges from k-(n-1) to k
    # We need 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
    
    # This is a banded matrix. Let's build it.
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # A[k, i] = f[k-i] for k = i, i+1, ..., min(i+n-1, 2n-2)
        k_start = i
        k_end = min(i + n, 2*n - 1)  # exclusive
        A[k_start:k_end, i] = f[:k_end - k_start]
    
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return np.ones(n)  # fallback

def autocorr_max(a):
    """Compute max of full autocorrelation of a."""
    return np.convolve(a, a).max()


def objective_normalized(a):
    """Compute 2*n*max(f*f) for a sequence with sum=1."""
    n = len(a)
    return 2.0 * n * autocorr_max(a)


def l2_gradient(a):
    """
    Compute gradient of sum_k (f*f)[k]^2 with respect to f.
    grad[i] = 2 * sum_j a[j] * R[i+j] where R = f*f (full).
    """
    R = np.convolve(a, a)
    return 2.0 * np.correlate(R, a, mode='valid')


def soft_peak_gradient(a, tau):
    """
    Compute gradient of soft-max surrogate: log(sum exp(R[k]/tau)).
    """
    R = np.convolve(a, a)
    # Numerically stable soft-max
    R_shifted = R - R.max()
    w = np.exp(R_shifted / tau)
    w /= w.sum()
    grad = (2.0 / tau) * np.correlate(w, a, mode='valid')
    return grad


def run_mpgd(a, eta_init, max_iters, deadline, tau_mode='l2', tau_val=0.1, verbose=True):
    """
    Run multiplicative gradient descent.
    tau_mode: 'l2' for L2 surrogate, 'soft' for soft-max surrogate.
    """
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-20:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = objective_normalized(a)
    eta = eta_init
    patience = 0

            break
        
        if tau_mode == 'l2':
            grad = l2_gradient(a)
        else:
            grad = soft_peak_gradient(a, tau_val)

        max_R = R.max()
        scale = max_R if max_R > 1e-20 else 1.0

def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g vector.
    """
    n = len(f)
    # Build constraint matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = f[:k_end - k_start]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.status == 0:
        return res.x
    return np.ones(n) / n
