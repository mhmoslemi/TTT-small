import time

def propose_candidate(seed=42, budget_s=1800):

    Starts from multiple initializations (ramp, cosine, etc.) and refines
    with block-level mutations.
    """

        np.ones(n, dtype=np.float64),                       # constant
        np.exp(-0.5 * ((x - n/2) / (n/6))**2).copy(),      # Gaussian
        (x / n)**2 * (1 - x/n).copy(),                     # asymmetric bell
    ]
    
    best = None

        score = float(evaluate_sequence(s.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = s.copy()
    
    print(f"Best initial score: {best_score:.6f} (n={n})")
    
    # Simulated annealing
    T = 0.5
    T_min = 0.0001
    best_T = T
    improvements = 0
    total_evals = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        candidate = best.copy()
        
        # Choose mutation
        r = rng.random()
        if r < 0.35:
            # Block scaling
            width = int(rng.integers(max(2, n//200), max(3, n//5)))
            start = int(rng.integers(0, max(1, n - width)))
            scale = float(np.exp(rng.normal(0, 0.2 * T + 0.05)))
            candidate[start:start+width] *= scale
        elif r < 0.5:
            # Smooth replacement: replace block with a smooth curve
            width = int(rng.integers(max(2, n//50), max(3, n//8)))
            start = int(rng.integers(0, max(1, n - width)))
            t = np.linspace(0, 1, width)
            curve_type = rng.integers(0, 4)
            if curve_type == 0:
                new_block = t
            elif curve_type == 1:
                new_block = 1 - t
            elif curve_type == 2:
                new_block = np.sin(np.pi * t)
            else:
                new_block = t**2
            
            # Match the scale to the surrounding
            local_max = max(candidate[start:start+width].max(), 0.1) if width < len(candidate) else 1.0
            candidate[start:start+width] = new_block * local_max
        elif r < 0.65:
            # Block swap
            if n >= 4:
                w1 = int(rng.integers(2, max(3, n//20)))
                w2 = w1
                s1 = int(rng.integers(0, max(1, n - w1)))
                s2 = int(rng.integers(0, max(1, n - w2)))
                # Swapping might create discontinuities, so blend
                block1 = candidate[s1:s1+w1].copy()
                block2 = candidate[s2:s2+w2].copy()
                blend = rng.uniform(0.5, 1.0)
                candidate[s1:s1+w1] = blend * block2[:w1] + (1-blend) * block1
                candidate[s2:s2+w2] = blend * block1[:w2] + (1-blend) * block2
        elif r < 0.8:
            # Smooth perturbation (low-pass filtered noise)
            noise = rng.normal(0, 0.1 * (T + 0.1), size=n)
            # Simple smoothing: moving average
            k = max(1, n // 100)
            kernel = np.ones(k) / k
            smooth_noise = np.convolve(noise, kernel, mode='same')
            candidate += smooth_noise
        elif r < 0.9:

            half = n // 2
            factor = float(np.exp(rng.normal(0, 0.3 * T + 0.05)))
            if rng.random() < 0.5:
                candidate[:half] *= factor
            else:
                candidate[half:] *= factor
        else:
            # Random restart (rare)
            x = np.arange(n, dtype=np.float64)
            candidate = np.abs(x - rng.uniform(0, n) * (n/2)) / (n/2)
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        
        score = float(evaluate_sequence(candidate.tolist()))
        total_evals += 1
        
        if np.isfinite(score):
            if score < best_score:
                best = candidate
                best_score = score
                improvements += 1
                if time.monotonic() - last_print > 10 or best_score < 1.5027:
                    print(f"Score: {best_score:.6f} | evals: {total_evals} | T: {T:.4f}")
                    last_print = time.monotonic()
            else:
                # Accept with probability exp(-delta/T)
                delta = score - best_score
                if rng.random() < np.exp(-delta / max(T, T_min)):
                    best = candidate
        # else: invalid, skip
        
        # Cooling
        T = max(T_min, T * 0.9995)
        
        # Reheat if stuck
        if T <= T_min * 1.1:
            T = 0.3 * best_T
            # Try to diversify
            candidate = best * np.exp(rng.normal(0, 0.5, size=n))
            candidate = np.clip(candidate, 0, 1000)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
    
    print(f"Final: score={best_score:.6f}, evals={total_evals}")
    return [float(x) for x in best]

def _compute_score_grad(a):

            # Slightly increase lr on improvement
            lr = min(lr * 1.02, 5.0 / n)
        
        # Gradient step with centered gradient (preserve sum approximately)
        grad_centered = grad - np.mean(grad)
        candidate = a - lr * grad_centered
        candidate = np.clip(candidate, 0.0, 1000.0)
        
        # If candidate is all zeros after clipping, try without centering
        if np.sum(candidate) < 0.01:
            candidate = a - lr * grad
            candidate = np.clip(candidate, 0.0, 1000.0)
        
        c_score = float(evaluate_sequence(candidate.tolist()))
        
        if np.isfinite(c_score) and c_score < score:
            a = candidate
        else:
            patience += 1
        
        if patience >= max_patience:
            # Random restart
            r = rng.random()
            if r < 0.4:
                # Perturb current best
                a = best * np.exp(rng.normal(0, 0.5, size=n))
            elif r < 0.7:
                # New random shape
                p = rng.uniform(0.3, 3.0)
                a = (np.arange(n)**p) / (n**p)
                a = a + rng.normal(0, 0.05, size=n)
            else:
                # Take a step from best in random direction
                a = best.copy()
                width = int(rng.integers(n//10, n//2))
                start = int(rng.integers(0, n - width))
                a[start:start+width] *= rng.uniform(0.1, 3.0)
            a = np.clip(a, 0.0, 1000.0)
            lr = 0.5 / n
            patience = 0
            max_patience = min(5000, max_patience + 500)
        
        iters += 1
        if iters % 5000 == 0:
            print(f"  GD iter {iters}, score={best_score:.6f}, lr={lr:.6f}")
    
    return best, best_score
