import numpy as np
import time

def smooth_max_and_weights(b, tau):
    """Compute smooth max and softmax weights."""
    m = np.max(b)
    exp_b = np.exp((b - m) * tau)
    sum_exp = np.sum(exp_b)
    smooth_max = m + np.log(sum_exp) / tau
    weights = exp_b / sum_exp
    return smooth_max, weights

"""

    """Compute a*a (full linear convolution) using FFT."""
    n = len(a)

        L = 1

            L *= 2

def compute_grad_fft(a, p, L=None):
    """
    Compute grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]

    while time.monotonic() < deadline:
        iterations += 1
        
        # Pick a random block to modify
        block_size = max(2, n // 50)
        start = rng.integers(0, n - block_size)
        end = start + block_size
        
        # Try different perturbations
        for _ in range(5):
            candidate = a.copy()
            # Multiplicative perturbation
            perturbation = np.exp(rng.normal(0, 0.3, block_size))
            candidate[start:end] *= perturbation
            candidate = np.clip(candidate, 0, None)
            s = candidate.sum()
            if s < 1e-10:
                continue
            candidate = candidate / s
            
            b_cand = fft_self_convolve(candidate, L1)
            score_cand = 2.0 * n * np.max(b_cand)
            
            if score_cand < best_score:
                best_score = score_cand
                best_a = candidate.copy()
                a = candidate.copy()
        
        if verbose and iterations % 500 == 0:
            print(f"  Block search: iter={iterations}, best={best_score:.6f}")
    
    return best_a, best_score


def try_structured_initialization(n, rng):
    """Try various structured initializations and return the best one."""
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Triangle
    t = np.linspace(0, 1, 2 * n - 1)
    tri = np.maximum(0, 1 - np.abs(t - 0.5) * 2)
    tri = tri[:n]
    candidates.append(tri / tri.sum())
    
    # Sine-modulated
    x = np.arange(n)
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            s = 1.0 + amp * np.cos(2 * np.pi * freq * x / n)
            s = np.maximum(s, 0)
            if s.sum() > 0:
                candidates.append(s / s.sum())
    
    # Gaussian-like
    center = n / 2
    for width in [n/10, n/5, n/3]:
        g = np.exp(-0.5 * ((x - center) / width) ** 2)
        candidates.append(g / g.sum())
    
    # Random
    candidates.append(rng.random(n) / rng.random(n).sum())
    
    # Evaluate all candidates
    N_conv = 2 * n - 1
    L1 = 1
    while L1 < N_conv:
        L1 *= 2
    
    best = None
    best_sc = np.inf
    for c in candidates:
        b = fft_self_convolve(c, L1)
        sc = 2.0 * n * np.max(b)
        if sc < best_sc:
            best_sc = sc
            best = c
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses gradient-based optimization in log-space
    with FFT-accelerated computation, multi-phase annealing, and
    multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 15.0
    
    # Get starting sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size < 2:
            raise ValueError("too short")
    except (NameError, TypeError, ValueError):
        base = np.ones(5000)
    
    print(f"Starting with base sequence of length {base.size}")
    
    best_overall_a = None
    best_overall_score = np.inf
    
    # Define sequence lengths to try
    lengths_to_try = []
    if 2 < base.size < 15000:
        lengths_to_try.append(base.size)
    lengths_to_try.extend([2000, 5000, 10000, 20000])
    # Remove duplicates and sort
    lengths_to_try = sorted(set(lengths_to_try))
    
    # Allocate time: spend more time on promising lengths
    # Give the base length the most time
    n_lengths = len(lengths_to_try)
    # Weight: base length gets 50%, others split the rest
    if base.size in lengths_to_try:
        idx_base = lengths_to_try.index(base.size)
    else:
        idx_base = n_lengths // 2
    
    print(f"Trying lengths: {lengths_to_try}")
    
    for idx, n in enumerate(lengths_to_try):
        remaining = total_deadline - time.monotonic()
        if remaining < 30:
            break
        
        if n == base.size if base.size in lengths_to_try else (idx == idx_base):
            time_alloc = max(30, remaining * 0.4)
        else:
            time_alloc = max(20, remaining * 0.15)
        
        time_alloc = min(time_alloc, remaining - 10)
        sub_deadline = time.monotonic() + time_alloc
        
        print(f"\n=== Trying n={n}, time_budget={time_alloc:.0f}s ===")
        
        # Prepare initial sequence
        if base.size == n:
            a_init = np.clip(base, 0, None).copy()
        elif base.size > 1:
            old_n = base.size
            new_indices = np.linspace(0, old_n - 1, n)
            a_init = np.interp(new_indices, np.arange(old_n), np.clip(base.astype(np.float64), 0, None))
        else:
            a_init = np.ones(n)
        
        a_init = np.clip(a_init, 0, None)
        if a_init.sum() < 1e-10:
            a_init = np.ones(n) / n
        a_init = a_init / a_init.sum()
        
        # Also try structured initialization
        a_struct, sc_struct = try_structured_initialization(n, rng)
        print(f"  Structured init score: {sc_struct:.6f}")
        
        # Use the better initialization
        b_init = fft_self_convolve(a_init, None)
        sc_init = 2.0 * n * np.max(b_init)
        if sc_struct < sc_init:
            a_init = a_struct
            print(f"  Using structured init (score {sc_struct:.6f} < {sc_init:.6f})")
        
        # Phase 1: Gradient optimization
        phase1_deadline = time.monotonic() + time_alloc * 0.7
        a_opt, score_opt = run_gradient_optimization(a_init, phase1_deadline, lr=0.3)
        print(f"  Gradient opt done: score = {score_opt:.6f}")
        
        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(f"  *** New overall best: {best_overall_score:.6f} (n={n}) ***")
        
        # Phase 2: Random block search for additional improvement
        remaining = sub_deadline - time.monotonic()
        if remaining > 10 and score_opt < best_overall_score + 0.1:
            a_block, score_block = run_random_block_search(a_opt, sub_deadline)
            if score_block < best_overall_score:
                best_overall_score = score_block
                best_overall_a = a_block.copy()
                print(f"  *** New overall best after block search: {best_overall_score:.6f} ***")
    
    # Final verification
    if best_overall_a is not None:
        exact_score = evaluate_sequence(best_overall_a.tolist())
        print(f"\nFinal exact score: {exact_score}")
        print(f"Sequence length: {len(best_overall_a)}")
        return best_overall_a.tolist()
    
    # Fallback: uniform sequence
    fallback = np.ones(1000) / 1000
    print(f"Fallback score: {evaluate_sequence(fallback.tolist())}")
    return fallback.tolist()
