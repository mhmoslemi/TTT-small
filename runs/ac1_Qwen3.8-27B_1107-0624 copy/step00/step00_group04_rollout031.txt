def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    # Bisection on threshold
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    projected = np.maximum(v - mid, 0)
    projected /= projected.sum()  # Ensure exact sum = 1
    return projected

import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import linprog
import time

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    if v.sum() < 1e-10:
        return np.ones(n) / n
    # Efficient projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_max = np.max(rho[cond]) if np.any(cond) else 0
    theta = (cssv[rho_max - 1] - 1) / rho_max if rho_max > 0 else 0
    return np.maximum(v - theta, 0)

def compute_peak(f):
    """Compute the peak of the full auto-convolution."""
    return fftconvolve(f, f).max()

def subgradient_phase(f, n, budget_s, rng):
    """Run subgradient descent for the given budget."""
    deadline = time.monotonic() + budget_s
    best_f = f.copy()
    best_peak = compute_peak(best_f)
    lr = 2.0 / n
    iter_count = 0
    last_improve = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        conv = fftconvolve(f, f)
        k_star = conv.argmax()
        
        # Subgradient: 2 * f[k_star - j] for valid j
        lo_j = max(0, k_star - n + 1)
        hi_j = min(k_star, n - 1)
        f_new = f.copy()
        if lo_j <= hi_j:
            j_range = np.arange(lo_j, hi_j + 1)
            idx = k_star - j_range
            f_new[j_range] -= lr * 2.0 * f[idx]
        
        f = project_simplex(f_new)
        iter_count += 1
        
        # Evaluate every 50 iterations
        if iter_count % 50 == 0:
            peak = compute_peak(f)
            if peak < best_peak:
                best_f = f.copy()
                best_peak = peak
                last_improve = iter_count
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 200:
                    # Decrease learning rate and add perturbation
                    lr *= 0.9
                    if rng.random() < 0.05 and lr > 0.001/n:
                        f = best_f + rng.normal(0, 0.002, n)
                        f = project_simplex(f)
                    no_improve = 0
        
        # Adaptive learning rate
        if iter_count < 5000:
            lr = max(0.01/n, (2.0/n) * np.sqrt(5000.0 / iter_count))
        else:
            lr = max(0.001/n, (2.0/n) * (5000.0 / iter_count))
        
        # Print progress

            print(f"    Subgrad iter {iter_count}: best_peak={best_peak:.8f}, current_lr={lr:.6f}")
    
    return best_f, best_peak, iter_count

def lp_phase(f, n, budget_s, rng):

    deadline = time.monotonic() + budget_s
    best_f = f.copy()
    best_peak = compute_peak(best_f)
    stagnant = 0
    lp_iter = 0
    
    while time.monotonic() < deadline:
        conv = fftconvolve(f, f)
        M = conv.max()
        
        # Build constraint matrix
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            A[k, lo:hi+1] = f[hi:lo-1:-1]

        b_ub = np.full(2*n-1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 30})
        
        del A

        best_t = 0.0
