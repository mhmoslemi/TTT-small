from scipy import sparse
from scipy.optimize import linprog
import time

def build_convolve_matrix(n):
    """Precompute row/col indices for the convolution sparse matrix."""
    col_idx = np.repeat(np.arange(n), n)
    row_idx = col_idx + np.tile(np.arange(n), n)
    return row_idx, col_idx

def compute_objective(f, n):
    """Compute 2*n*max(conv) / sum^2 for normalized f (sum=1)."""
    conv = np.convolve(f, f)
    max_b = conv.max()
    return 2.0 * n * max_b

def lp_iteration(f, n, row_idx, col_idx):
    """One LP iteration. Returns new f or None."""
    conv = np.convolve(f, f)
    M = conv.max()
    if M < 1e-15:
        return None
    
    data = np.tile(f, n)
    A = sparse.csc_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return None
    
    g = np.clip(res.x, 0, None)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    
    g = g / g_sum
    return g
