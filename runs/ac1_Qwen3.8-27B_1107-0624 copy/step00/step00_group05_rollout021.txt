import numpy as np
import time

def smooth_max_grad(a, c):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    z = c * (b - max_b)
    exp_z = np.exp(z)
    w_sum = exp_z.sum()
    weights = exp_z / w_sum
    smooth_val = max_b + np.log(w_sum) / c
    
    # Gradient via cross-correlation
    cross_full = np.convolve(a[::-1], weights)
    cross = cross_full[n-1:2*n-1]
    
    # Diagonal correction
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2*n - 1)
    diag = np.zeros(n)
    diag[valid] = a[valid] * weights[diag_idx[valid]]
    
    grad = cross + diag
    return grad, smooth_val, b
