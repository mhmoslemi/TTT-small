This is a weighted sum of the shifted a's, where the weights are the softmax of the convolution values. This is a "soft" version of the subgradient.

The computation cost is O(n²) for the convolution + O(n²) for the gradient = O(n²) per iteration. With scipy's L-BFGS, we'd need maybe 100-1000 iterations, each taking 50ms, so 5-50 seconds total. That's feasible!

import numpy as np
import time

def _conv_max_and_peaks(a, peak_frac=0.005, max_peaks=10):
    """Compute max of conv(a,a) and return (max_val, peak_indices)."""
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    if max_b <= 0:
        return max_b, np.array([len(a) - 1])
    # Find peaks: top max_peaks values within peak_frac of max
    threshold = max_b * (1 - peak_frac)
    candidates = np.where(b >= threshold)[0]
    if len(candidates) > max_peaks:
        # Take the top max_peaks
        vals = b[candidates]
        top_idx = np.argsort(vals)[::-1][:max_peaks]
        candidates = candidates[top_idx]
    return max_b, candidates

def _subgradient_step(a, peak_indices, lr):
    """One subgradient step with multiplicative update."""
    n = len(a)
    s = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            s[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    s /= len(peak_indices)
    # Multiplicative update (normalize by max(s) for stability)
    max_s = np.max(s)
    if max_s > 0:
        a = a * np.exp(-lr * s / max_s)
    total = np.sum(a)
    if total > 1e-15:
        a = a / total
    else:
        a = np.ones(n) / n
    return a

    """Subgradient descent with multiplicative updates."""
    rng = np.random.default_rng(seed)
    n = len(a_init)

    while time.monotonic() < deadline:

        max_b, peaks = _conv_max_and_peaks(a)
        a = _subgradient_step(a, peaks, lr)
