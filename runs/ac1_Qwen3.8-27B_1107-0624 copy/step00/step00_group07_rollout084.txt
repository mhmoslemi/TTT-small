def normalize_nonneg(a, target=1.0):
    """Normalize a 1-D array to sum 1, keeping it non-negative."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    a = a / s
    return a


def compute_score(a):
    """Exact internal score, equal to evaluate_sequence for valid positive input."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def score_normalized(a):
    """Score for a normalized sum-1 nonnegative sequence."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m)


def downsample(a, N):
    """Downsample a sequence to length N by summing bins."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m <= 1:
        return np.full(N, 1.0 / max(1, N), dtype=np.float64)
    if m == N:
        return normalize_nonneg(a)
    out = np.zeros(N, dtype=np.float64)
    for i in range(N):
        lo = (i * m) // N
        hi = ((i + 1) * m) // N
        if hi > lo:
            out[i] = a[lo:hi].sum()
    return normalize_nonneg(out)


def upsample(a, N):
    """Upsample a sequence to length N by linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m <= 1:
        return np.full(N, 1.0 / max(1, N), dtype=np.float64)
    if m == N:
        return normalize_nonneg(a)
    x = np.linspace(0.0, float(m - 1), N)
    i0 = np.floor(x).astype(np.int64)
    i1 = np.minimum(i0 + 1, m - 1)
    frac = x - i0
    out = a[i0] * (1.0 - frac) + a[i1] * frac
    return normalize_nonneg(np.maximum(out, 0.0))

    if a.size == N:
        return normalize_nonneg(a)
    if a.size < N:

    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if max_rows >= m:
        return np.arange(m, dtype=np.int64)
    k = int(max(1, min(max_rows, m)))
    if k == m:
        return np.arange(m, dtype=np.int64)
    return np.argpartition(c, -k)[-k:].astype(np.int64)


def build_active_matrix(f, rows):
    """
    Build the dense active LP matrix A where row k has A[k,j] = f[k-j]
    whenever 0 <= k-j < len(f), else 0.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    rows = np.asarray(rows, dtype=np.int64)
    A = np.zeros((rows.size, n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            idx = jlo + np.arange(jhi - jlo, dtype=np.int64)
            A[r, jlo:jhi] = f[k - idx]
    return A


def solve_lp_direction(f, M, rows):
    """
    Solve the LP direction:
        maximize sum(g) subject to A g <= M, g >= 0.
    Returns g or None.
    """
    if linprog is None:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n > 4096:
        return None
    if M <= 1e-300:
        return None

    try:
        A = build_active_matrix(f, rows)
    except Exception:
        return None

    if A.size == 0 or not np.any(A):
        return None

    A = A / M
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.ones(A.shape[0], dtype=np.float64)

    try:
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=bvec,
            bounds=(0.0, None),
            method="highs",
        )
    except Exception:
        return None

    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        if float(g.sum()) > 1e-12:
            return g
    return None


def shift_correlation(p, f):
    """
    Return r[j] = sum_i f[i] * p[i+j] for j=0..n-1,
    where p has length 2n-1 and f has length n.
   Computed with FFT.
    """
    f = np.asarray(f, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = f.size
    m = p.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f

    pb = np.zeros(size, dtype=np.float64)
    pb[:m] = p[::-1]

    out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(pb, size), size)
    idx = (m - 1) - np.arange(n, dtype=np.int64)
    return out[idx]


def smooth_step(f, c, M, temp_frac, cur_score):
    """
    Smooth-max subgradient step for the normalized sequence f.
    Returns an improved sequence or None.
    """
    f = np.asarray(f, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = f.size
    if M <= 1e-300:
        return None

    temp = max(1e-15, float(temp_frac) * M)
    e = (c - M) / temp
    e = np.clip(e, -50.0, 0.0)
    p = np.exp(e)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / s

    r = 2.0 * shift_correlation(p, f)
    d = r - np.mean(r)
    scale = float(np.max(np.abs(d)))
    if not np.isfinite(scale) or scale <= 1e-300:
        return None

    if n > 3000:
        alphas = (0.05, 0.18, 0.45)
    else:
        alphas = (0.05, 0.15, 0.4, 0.8)

    for alpha in alphas:
        h = f - alpha * scale * d
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if sc < cur_score:
            return h
    return None


def random_perturb(f, rng, scale=0.1):
    """Random multiplicative block/noise perturbation for nonnegative sequences."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    h = np.maximum(f, 0.0).copy()

    if n > 1:
        h = h * np.exp(rng.normal(0.0, float(scale), size=n))

        w1 = max(1, n // 500)
        w2 = max(2, n // 8)
        width = int(rng.integers(w1, w2 + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        h[start:start + width] *= np.exp(float(sigma) * scale if False else 0.0)  # placeholder prevents linter? no
        h[start:start + width] *= np.exp(float(scale) * 1.5 * rng.normal(0.0, 1.0, size=width))

    return normalize_nonneg(h)


def update_global(state, seq, score):
    """Update the search state if score is finite and better."""
    if not np.isfinite(score):
        return False
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return False

    old = state["score"]
    if score < old - 1e-12:
        state["score"] = float(score)
        state["seq"] = normalize_nonneg(seq).copy()
        if not np.isfinite(old) or score < old - 1e-4:
            print(
                f"update_best score={state['score']:.12f} n={state['seq'].size}",
                flush=True,
            )
        return True
    return False


def optimize_trajectory(f, time_limit, rng, state):
    """
    Run a local search trajectory:
    - LP direction from the short-peak covering program,
    - smooth-max projected subgradient fallback,
    - random escape perturbations.
    """
    f = normalize_nonneg(f)
    n = f.size
    if n < 2:
        f = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
        n = f.size

    c = np.convolve(f, f)
    M = float(c.max())
    best_score = float(2.0 * n * M)
    best = f.copy()
    update_global(state, best, best_score)

    deadline = time.monotonic() + max(0.0, float(time_limit))
    it = 0
    no_improve = 0
    temp_frac = 0.22
    lp_every = 1 if n <= 512 else 2

    while time.monotonic() < deadline:
        it += 1

        c = np.convolve(f, f)
        M = float(c.max())
        cur = float(2.0 * n * M)

        if cur < best_score - 1e-12:
            best_score = cur
            best = f.copy()
            update_global(state, best, best_score)
            no_improve = 0
            temp_frac = max(0.03, temp_frac * 0.94)
        else:
            no_improve += 1

        if no_improve > 35:
            temp_frac = min(0.65, temp_frac * 1.25)
            f = random_perturb(f, rng, 0.16)
            continue

        moved = False

        if linprog is not None and n <= 4096 and (it % lp_every == 0 or no_improve > 4):
            if n <= 600:
                max_rows = 2 * n - 1
            elif n <= 2048:
                max_rows = min(2 * n - 1, 700)
            else:
                max_rows = min(2 * n - 1, 430)

            rows = choose_active_rows(c, max_rows)
            g = solve_lp_direction(f, M, rows)

            if g is not None:
                Sg = float(g.sum())
                if Sg > 1.0 + 1e-5:
                    g = g / Sg
                    best_h = f
                    best_cur = cur

                    if Sg > 10.0:
                        ts = (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5)
                    elif Sg > 2.0:
                        ts = (0.02, 0.05, 0.1, 0.2, 0.4, 0.7)
                    else:
                        ts = (0.05, 0.1, 0.2, 0.4, 0.7, 1.0)

                    for t in ts:
                        h = (1.0 - float(t)) * f + float(t) * g
                        h = normalize_nonneg(h)
                        sc = score_normalized(h)
                        if sc < best_cur:
                            best_cur = sc
                            best_h = h

                    if best_cur < cur - 1e-12:
                        f = best_h
                        moved = True

        if not moved and time.monotonic() < deadline:
            h = smooth_step(f, c, M, temp_frac, cur)
            if h is not None:
                f = h
                moved = True

        if not moved:
            f = random_perturb(f, rng, 0.05)

    c = np.convolve(f, f)
    final_sc = float(2.0 * f.size * float(c.max()))
    if final_sc < best_score:
        best_score = final_sc
        best = f.copy()
    update_global(state, best, best_score)
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for low-score step-function heights.

    Uses a portfolio of asymmetric/non-uniform starts, an LP direction from the
    covering LP on active convolution peaks, smooth max-subgradient fallback,
    and graduated length upsampling. Scale invariance is used by optimizing
    normalized sum-1 sequences.
    """
    rng = np.random.default_rng(seed)
    hard_deadline = time.monotonic() + max(0.0, float(budget_s) - 70.0)

    state = {"score": float("inf"), "seq": None}

    provided = load_provided_sequence()
    provided_score = float("inf")
    if provided is not None:
        provided_score = compute_score(provided)
        update_global(state, provided, provided_score)

    # Guaranteed valid fallbacks.
    update_global(state, np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64), 16.0 / 9.0)
    update_global(state, np.ones(256, dtype=np.float64), compute_score(np.ones(256)))

    tasks = []

    # If the provided construction is already good, spend real time on it.
    if (
        provided is not None
        and np.isfinite(provided_score)
        and provided_score < 1.85
        and 256 <= provided.size <= 8000
    ):
        tasks.append((provided.size, -2, 120.0))

    # Short trajectories to establish basins quickly.
    for N in (64, 128):
        for kind in (3, 0, 1):
            tasks.append((N, kind, 8.0))

    # Medium lengths.
    for kind in (3, 0, 1, 2, 5):
        tasks.append((256, kind, 14.0))

    for kind in (3, 0, 1, 2, 5, -1):
        tasks.append((512, kind, 32.0))

    # Long lengths, where the score can approach the continuous limit.
    for kind in (3, 0, 1, 5, -1, -2):
        tasks.append((1024, kind, 48.0))

    for kind in (3, 0, -1, -2):
        tasks.append((2048, kind, 64.0))

    for kind in (-1, -2):
        tasks.append((3072, kind, 72.0))

    for kind in (-1,):
        tasks.append((4096, kind, 78.0))

    # If the provided sequence is long but not obviously great, try a smaller slice.
    if provided is not None and 512 <= provided.size <= 4096:
        tasks.append((provided.size, -2, 30.0))

    for N, kind, base in tasks:
        now = time.monotonic()
        if now + 3.0 > hard_deadline:
            break
        limit = min(float(base), hard_deadline - now)
        if limit < 2.0:
            continue

        seq = make_initial(N, kind, rng, state["seq"], provided)
        seq_sc = compute_score(seq)

        # If we already have a strong result, skip obviously bad long random starts.
        if (
            np.isfinite(seq_sc)
            and state["score"] < 1.8
            and seq_sc > max(3.0, 2.0 * state["score"])
            and kind not in (-1, -2)
            and N > 512
        ):
            continue

        optimize_trajectory(seq, limit, rng, state)

    # Extra refinement: upscale the current best progressively and refine again.
    same_size_no_gain = 0
    while time.monotonic() + 25.0 < hard_deadline:
        if state["seq"] is None:
            break

        cur_n = state["seq"].size
        if cur_n < 512:
            next_n = 512
        elif cur_n < 1024:
            next_n = 1024
        elif cur_n < 2048:
            next_n = 2048
        elif cur_n < 4096:
            next_n = 4096
        else:
            next_n = cur_n

        seq = resize_to_length(state["seq"], next_n)
        limit = min(110.0, hard_deadline - time.monotonic() - 15.0)
        if limit < 18.0:
            break

        old_score = state["score"]
        optimize_trajectory(seq, limit, rng, state)

        if next_n == cur_n and state["score"] >= old_score - 1e-9:
            same_size_no_gain += 1
            if same_size_no_gain >= 2:
                break
        else:
            same_size_no_gain = 0

    if state["seq"] is None:
        return [float(1.0 / 3.0), float(2.0 / 3.0)]

    cand = [float(x) for x in np.asarray(state["seq"], dtype=np.float64)]

    try:
        official = evaluate_sequence(cand)
        if np.isfinite(official):
            print(f"official_best score={official:.12f} n={len(cand)}", flush=True)
    except Exception:
        pass

    return cand
