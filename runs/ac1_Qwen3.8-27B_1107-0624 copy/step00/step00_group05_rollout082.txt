def fast_score(a):
    a=np.asarray(a,float)
    s=a.sum()
    if s<0.01: return inf
    n=len(a)
    # full convolution via FFT
    m=1
    L=2*n-1
    # choose next power of 2 for speed
    while m < L: m <<= 1
    fa=np.fft.rfft(a, m)
    c=np.fft.irfft(fa*fa, m)[:L]
    return float(2*n*max(c)/s/s)
```

```python
"""
Search for non-negative sequences with small peak convolution.

The algorithm uses:
1. Binary anti-palindromic local search.  A sequence is constrained by
   a_i + a_{n-1-i} = 1, which kills the center convolution peak for binary
   data and leaves many degrees of freedom.  Moves flip one symmetric pair.
2. Continuous anti-palindromic polishing with a smooth log-sum-exp objective.
3. A short general continuous polish with softmax parameters.

The returned sequence is the best one found before the internal time budget.
"""

import numpy as np
import time


def next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def full_convolution(a):
    """Full real convolution using FFT.  Input is 1D."""
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    m = next_pow2(L)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, n=m)
    return c[:L]


def local_score(a):
    """Fast local equivalent of evaluate_sequence, without security checks."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = full_convolution(a)
    mx = float(np.max(c))
    if mx <= 0.0:
        return 0.0
    return float(2.0 * a.size * mx / (s * s))


def convolution_int32(a):
    """Exact integer convolution for binary 0/1 sequences, as int32."""
    af = a.astype(np.float64) if a.dtype != np.float64 else a
    cf = full_convolution(af)
    return np.clip(np.rint(cf), 0, None).astype(np.int32)


def generate_anti_pal_binary(n, rng, kind=0):
    """Generate a binary anti-palindromic sequence of even length n."""
    m = n // 2
    if kind == 0:
        bits = rng.integers(0, 2, m, dtype=np.uint8)
    elif kind == 1:
        bits = (np.arange(m, dtype=np.uint8) % 2).astype(np.uint8)
    elif kind == 2:
        bits = np.zeros(m, dtype=np.uint8)
        bits[:m // 2] = 1
    elif kind == 3:
        bits = (((np.arange(m) * 7) & 1).astype(np.uint8))
    elif kind == 4:
        bits = np.zeros(m, dtype=np.uint8)
        bits[m // 3:2 * m // 3] = 1
    elif kind == 5:
        bits = (((np.arange(m) * 11 + (np.arange(m) // 13) * 3) & 1).astype(np.uint8))
    else:
        bits = rng.integers(0, 2, m, dtype=np.uint8)

    a = np.empty(n, dtype=np.uint8)
    a[:m] = bits
    a[m:] = 1 - bits[::-1]
    return a


def project_anti_pal_binary(arr, n):
    """Project an array to a binary anti-palindromic sequence of length n."""
    a = np.asarray(arr, dtype=np.float64).reshape(-1)
    m = n // 2
    b = np.zeros(n, dtype=np.uint8)
    if a.size >= m:
        b[:m] = (a[:m] >= 0.5).astype(np.uint8)
    b[m:] = 1 - b[:m][::-1]
    return b


def choose_move_j(n, m, rng, top_idx, guided=0.70):
    """Choose a first-half index to flip in an anti-palindromic pair."""
    if top_idx is not None and len(top_idx) > 0 and rng.random() < guided:
        k = int(top_idx[int(rng.integers(len(top_idx)))])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            t = int(rng.integers(lo, hi + 1))
            j = t if t < m else n - 1 - t
            if 0 <= j < m:
                return j
    return int(rng.integers(m))


def flip_pair(c, a, two_a, n, j):
    """
    Flip the anti-palindromic pair {j, n-1-j} and update convolution in place.

    a and two_a must be kept consistent: two_a = 2*a.
    """
    p = j
    q = n - 1 - j

    if int(a[p]) == 0:
        # p: 0 -> 1, q: 1 -> 0
        c[p:p + n] += two_a
        c[2 * p] += 1
        a[p] = 1
        two_a[p] = 2

        c[q:q + n] -= two_a
        c[2 * q] += 1
        a[q] = 0
        two_a[q] = 0
    else:
        # p: 1 -> 0, q: 0 -> 1
        c[p:p + n] -= two_a
        c[2 * p] += 1
        a[p] = 0
        two_a[p] = 0

        c[q:q + n] += two_a
        c[2 * q] += 1
        a[q] = 1
        two_a[q] = 2


def perturb_anti_pal(base, rng, n, m, mode):
    """Return a perturbed copy of a binary anti-palindromic sequence."""
    a = base.copy()
    cnt = max(1, m // 10)

    if mode == 0:
        idx = rng.integers(0, m, cnt, dtype=np.uint8)
        a[idx] = 1 - a[idx]
    elif mode == 1:
        blen = int(rng.integers(16, max(17, min(m, 256))))
        if blen >= m:
            blen = max(1, m // 2)
        l = int(rng.integers(0, max(1, m - blen + 1)))
        a[l:l + blen] = a[l:l + blen][::-1]
    else:
        idx = rng.integers(0, m, cnt, dtype=np.uint8)
        a[idx] = rng.integers(0, 2, cnt, dtype=np.uint8)

    a[m:] = 1 - a[:m][::-1]
    return a


def search_anti_pal_binary(n, seed, time_limit, initial_a=None):
    """
    Randomly guided local search over binary anti-palindromic sequences.

    Returns (best sequence as uint8 array, local score), or (None, inf).
    """
    if time_limit <= 0 or n < 4 or (n % 2 != 0):
        return None, float("inf")

    rng = np.random.default_rng(seed)
    m = n // 2
    L = 2 * n - 1
    factor = 2.0 * n / (float(m) * float(m))

    inits = []
    if initial_a is not None:
        inits.append(project_anti_pal_binary(initial_a, n))

    for _ in range(8):
        inits.append(generate_anti_pal_binary(n, rng, 0))
    for kind in range(1, 6):
        inits.append(generate_anti_pal_binary(n, rng, kind))

    best_a = None
    best_score = float("inf")

    for a0 in inits:
        c0 = convolution_int32(a0)
        sc = factor * float(c0.max())
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()

    if best_a is None:
        return None, float("inf")

    a = best_a.copy()
    two_a = a.astype(np.int32) * 2
    c = convolution_int32(a)
    oldmax = int(c.max())
    cur = factor * float(oldmax)
    if cur < best_score:
        best_score = cur

    top_idx = np.arange(L, dtype=np.int64)
    batch = 6 if n <= 2048 else 4
    stuck_limit = 3500
    no_improve = 0
    outer = 0
    deadline = time.monotonic() + max(0.5, float(time_limit))

    while time.monotonic() < deadline:
        if outer % 2000 == 0:
            c = convolution_int32(a)
            oldmax = int(c.max())
            cur = factor * float(oldmax)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

            k = min(32, L)
            if k <= 1:
                top_idx = np.array([0], dtype=np.int64)
            else:
                top_idx = np.argpartition(c, -k)[-k:].astype(np.int64)

        best_j = -1
        best_nm = oldmax + 1

        for _ in range(batch):
            j = choose_move_j(n, m, rng, top_idx, 0.70)
            flip_pair(c, a, two_a, n, j)
            nm = int(c.max())
            if nm < best_nm:
                best_nm = nm
                best_j = j
            flip_pair(c, a, two_a, n, j)

        if best_j >= 0:
            flip_pair(c, a, two_a, n, best_j)
            oldmax = best_nm
            cur = factor * float(oldmax)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += batch
            if no_improve > stuck_limit:
                base = best_a if best_a is not None else a
                mode = int(rng.integers(3))
                a = perturb_anti_pal(base, rng, n, m, mode)
                two_a = a.astype(np.int32) * 2
                c = convolution_int32(a)
                oldmax = int(c.max())
                no_improve = 0

        outer += 1

    # Final exact recomputation to guard against any accumulated drift.
    try:
        cf = convolution_int32(best_a)
        final_sc = factor * float(cf.max())
        if final_sc < best_score:
            best_score = final_sc
    except Exception:
        pass

    return best_a, float(best_score)


def adam_update(mom, var, grad, lr, t, b1=0.9, b2=0.999, eps=1e-8):
    """Return updated Adam moments and the step vector."""
    mom = b1 * mom + (1.0 - b1) * grad
    var = b2 * var + (1.0 - b2) * (grad * grad)
    mhat = mom / (1.0 - (b1 ** t))
    vhat = var / (1.0 - (b2 ** t))
    step = lr * mhat / (np.sqrt(vhat) + eps)
    return mom, var, step


def search_anti_pal_continuous(n, init_a, time_limit, seed):
    """
    Continuous anti-palindromic polish with an LSE peak objective.

    The sequence is parameterized as:
        a = [x, 1 - reverse(x)]
    with x in (0, 1).
    """
    if time_limit <= 0 or n < 4 or (n % 2 != 0):
        return None, float("inf")

    rng = np.random.default_rng(seed)
    m = n // 2
    L = 2 * n - 1
    factor = 2.0 * n / (float(m) * float(m))

    init = np.asarray(init_a, dtype=np.float64).reshape(-1)
    if init.size != n:
        init = project_anti_pal_binary(init, n).astype(np.float64)

    best_a = init.copy()
    best_score = local_score(best_a)

    x = init[:m].copy()
    if np.std(x) < 0.05:
        x = np.clip(x + rng.normal(0.0, 0.12, m), 0.03, 0.97)
    else:
        x = np.clip(x, 0.02, 0.98)

    u = np.log(x / (1.0 - x))
    mom = np.zeros(m, dtype=np.float64)
    var = np.zeros(m, dtype=np.float64)
    t = 1
    lr = 0.05
    tau_factor = 0.25
    no_improve = 0
    m2 = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.5, float(time_limit))

    while time.monotonic() < deadline:
        u = np.clip(u, -40.0, 40.0)
        x = 1.0 / (1.0 + np.exp(-u))
        x = np.clip(x, 1e-6, 1.0 - 1e-6)

        a = np.empty(n, dtype=np.float64)
        a[:m] = x
        a[m:] = 1.0 - x[::-1]

        fa = np.fft.rfft(a, m2)
        c = np.fft.irfft(fa * fa, n=m2)[:L]
        maxc = float(np.max(c))
        score = factor * maxc

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            u = u + rng.normal(0.0, 0.3, m)
            no_improve = 0
            lr = max(0.01, lr * 0.7)
            tau_factor = min(0.6, max(0.05, tau_factor * 1.5))

        if maxc <= 0.0:
            break

        tau = max(tau_factor * maxc, 1e-12)
        z = c / tau
        z = z - float(np.max(z))
        w = np.exp(z)
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 0.0:
            break
        w = w / ws

        fw = np.fft.rfft(w[::-1], m2)
        y = np.fft.irfft(fa * fw, n=m2)
        grad_a = 2.0 * y[n - 1:2 * n - 1][::-1]
        grad_u = grad_a[:m] * (x * (1.0 - x))
        np.clip(grad_u, -100.0, 100.0, out=grad_u)

        mom, var, step = adam_update(mom, var, grad_u, lr, t)
        u = u - step
        u = np.clip(u, -40.0, 40.0)
        t += 1

        if t % 100 == 0:
            tau_factor = max(0.05, tau_factor * 0.98)
            lr = max(0.01, lr * 0.99)

    return best_a, float(best_score)


def search_general_continuous(a0, time_limit, seed):
    """
    Short general continuous polish using softmax/log-parameterization.
    The sequence is kept nonnegative and normalized to sum 1.
    """
    a = np.asarray(a0, dtype=np.float64).reshape(-1)
    if time_limit <= 0 or a.size < 8:
        return None, float("inf")

    rng = np.random.default_rng(seed)
    n = a.size

    a = np.maximum(a, 0.0)
    s0 = float(np.sum(a))
    if s0 < 0.01:
        a = np.ones(n, dtype=np.float64)
        s0 = float(n)

    # Start with a nonnegative normalized version, slightly smoothed.
    a = 0.05 + 0.95 * (a / s0)
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_score = local_score(a)

    loga = np.log(np.maximum(a, 1e-12))
    mom = np.zeros(n, dtype=np.float64)
    var = np.zeros(n, dtype=np.float64)
    t = 1
    lr = 0.03
    tau_factor = 0.25
    no_improve = 0
    L = 2 * n - 1
    m2 = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.5, float(time_limit))

    while time.monotonic() < deadline:
        fa = np.fft.rfft(a, m2)
        c = np.fft.irfft(fa * fa, n=m2)[:L]
        maxc = float(np.max(c))
        score = float(2.0 * n * maxc)  # sum(a) == 1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 800:
            noise = rng.normal(0.0, 0.10, n) * np.sqrt(a)
            loga = loga + noise
            no_improve = 0
            lr = max(0.005, lr * 0.7)
            tau_factor = min(0.5, max(0.05, tau_factor * 1.5))

        if maxc <= 0.0:
            break

        tau = max(tau_factor * maxc, 1e-12)
        z = c / tau
        z = z - float(np.max(z))
        w = np.exp(z)
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 0.0:
            break
        w = w / ws

        fw = np.fft.rfft(w[::-1], m2)
        y = np.fft.irfft(fa * fw, n=m2)
        grad_a = 2.0 * y[n - 1:2 * n - 1][::-1]

        # Remove the scale direction; normalization is handled by softmax.
        grad = grad_a - float(np.mean(grad_a))
        np.clip(grad, -100.0, 100.0, out=grad)

        mom, var, step = adam_update(mom, var, grad, lr, t)
        loga = loga - step
        loga = loga - float(np.mean(loga))
        loga = np.clip(loga, -30.0, 30.0)

        a = np.exp(loga)
        a = np.maximum(a, 1e-9)
        a = a / float(np.sum(a))
        t += 1

        if t % 200 == 0:
            tau_factor = max(0.05, tau_factor * 0.97)
            lr = max(0.005, lr * 0.995)

    return best_a, float(best_score)


def propose_candidate():
    """
    Main entry point.  Searches several nonnegative sequences and returns the
    best list found before the allotted time budget.
    """
    start = time.monotonic()
    hard_deadline = start + 1770.0
    candidates = []
    min_score = float("inf")

    # Seed with the provided construction, if it exists and is usable.
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        init = np.clip(init, 0.0, 1000.0)
        if init.size > 0 and np.all(np.isfinite(init)):
            sc = local_score(init)
            if np.isfinite(sc):
                candidates.append((init.astype(np.float64), float(sc)))
                min_score = float(sc)
                print(f"initial construction score={float(sc):.6f} len={init.size}")
    except Exception:
        pass

    # If nothing valid was available, fall back to a uniform sequence.
    if not candidates:
        fallback = np.ones(2048, dtype=np.float64)
        sc = local_score(fallback)
        candidates.append((fallback, float(sc)))
        min_score = float(sc)
        print(f"fallback uniform score={float(sc):.6f}")

    # Phase 1: smaller binary anti-palindromic search.
    n1 = 2048
    rem = hard_deadline - time.monotonic()
    if rem > 350:
        limit1 = min(600.0, rem - 300.0)
        if limit1 > 30.0:
            a1, sc1 = search_anti_pal_binary(n1, seed=101, time_limit=limit1)
            if a1 is not None and np.isfinite(sc1):
                arr1 = a1.astype(np.float64)
                candidates.append((arr1, float(sc1)))
                if sc1 < min_score:
                    min_score = float(sc1)
                print(f"binary anti-pal n={n1} local_score={float(sc1):.6f} elapsed={time.monotonic()-start:.1f}s")

    # Phase 2: larger binary anti-palindromic search.
    n2 = 4096
    rem = hard_deadline - time.monotonic()
    if rem > 300:
        limit2 = min(850.0, rem - 260.0)
        if limit2 > 30.0:
            a2, sc2 = search_anti_pal_binary(n2, seed=202, time_limit=limit2)
            if a2 is not None and np.isfinite(sc2):
                arr2 = a2.astype(np.float64)
                candidates.append((arr2, float(sc2)))
                if sc2 < min_score:
                    min_score = float(sc2)
                print(f"binary anti-pal n={n2} local_score={float(sc2):.6f} elapsed={time.monotonic()-start:.1f}s")

    # Phase 3: continuous anti-palindromic polish on the best even candidate.
    best_even_arr = None
    best_even_sc = float("inf")
    for arr, sc in candidates:
        if arr.size % 2 == 0 and arr.size >= 64:
            if sc < best_even_sc:
                best_even_sc = float(sc)
                best_even_arr = arr

    rem = hard_deadline - time.monotonic()
    if rem > 120 and best_even_arr is not None:
        limit3 = min(120.0, rem - 90.0)
        if limit3 > 10.0:
            a3, sc3 = search_anti_pal_continuous(
                n=best_even_arr.size,
                init_a=best_even_arr,
                time_limit=limit3,
                seed=303
            )
            if a3 is not None and np.isfinite(sc3):
                arr3 = np.clip(a3.astype(np.float64), 0.0, 1000.0)
                candidates.append((arr3, float(sc3)))
                if sc3 < min_score:
                    min_score = float(sc3)
                print(f"continuous anti-pal polish local_score={float(sc3):.6f} elapsed={time.monotonic()-start:.1f}s")

    # Phase 4: short general continuous polish on the current best candidate.
    best_arr = None
    best_sc = float("inf")
    for arr, sc in candidates:
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_arr = arr

    rem = hard_deadline - time.monotonic()
    if rem > 40 and best_arr is not None:
        limit4 = max(10.0, rem - 20.0)
        a4, sc4 = search_general_continuous(best_arr, time_limit=limit4, seed=404)
        if a4 is not None and np.isfinite(sc4):
            arr4 = np.clip(a4.astype(np.float64), 0.0, 1000.0)
            candidates.append((arr4, float(sc4)))
            if sc4 < min_score:
                min_score = float(sc4)
            print(f"general continuous polish local_score={float(sc4):.6f} elapsed={time.monotonic()-start:.1f}s")

    # Final authoritative evaluation of the few stored candidates.
    best_seq = np.ones(2048, dtype=np.float64)
    try:
        best_official = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_official = float("inf")

    for arr, sc in candidates:
        try:
            arr = np.asarray(arr, dtype=np.float64).reshape(-1)
            arr = np.clip(arr, 0.0, 1000.0)
            off = float(evaluate_sequence([float(x) for x in arr]))
            if np.isfinite(off) and off < best_official:
                best_official = off
                best_seq = arr
        except Exception:
            pass

    return [float(x) for x in best_seq]
