    # Compute full convolution to find active constraints
    c_full = np.convolve(a, a)

    threshold = 0.5 * M
    active = np.where(c_full > threshold)[0]
    
    if len(active) == 0:
        active = np.array([c_full.argmax()])

    # Build constraint matrix (sparse)
    # A[k, i] = a[k-i] if 0 <= k-i < n
