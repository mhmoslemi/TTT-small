"""
Minimizes the step-function evaluation
    score(a) = 2*n*max(conv(a,a)) / (sum(a))^2   (a_i >= 0)
by (1) a differentiable surrogate built from a sharp softmax-max of the
convolution peaks, optimized with L-BFGS-B using an O(n log n) FFT-based
analytic gradient, from many diverse restarts; and (2) a final exact
block-perturbation polish driven directly by evaluate_sequence.
FFT-based conv and gradient keep each iteration cheap so the 1800 s budget
can be spent on many restarts plus polishing.
"""

import time

from scipy.special import logsumexp
from scipy.optimize import minimize


def _next_pow2(x):
    """Smallest power of two >= x (x >= 1)."""
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def _conv_full_fft(a):
    """Full linear convolution a*a via FFT (real, non-negative length 2n-1)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[: 2 * n - 1]


def _smooth_objective_and_grad(a, sharpness):
    """Smooth surrogate value + analytic gradient.

    L(c) = (1/t) logsumexp(t*c) approximates max(c).
    f(a) = 2n L(c) / S^2.  Gradient computed by FFT cross-correlation.
    """
    n = a.size
    S = float(np.sum(a))
    if S < 1e-4:
        return 1e12, np.zeros(n)

    c = _conv_full_fft(a)
    t = float(sharpness)
    tc = t * c
    logmx = logsumexp(tc)
    L = logmx / t
    w = np.exp(tc - logmx)          # softmax weights, sum = 1
    score = 2.0 * n * L / (S * S)

    # dL/da_m = sum_k (a_{k-m} + a_{m-k}) w_k
    #   v2[m] = sum_j a_j w_{m-j} = (conv(a,w))[m]            = IFFT(A*W)[m]
    #   v1[m] = sum_j a_j w_{j+m} = cross-corr style          = IFFT(conj(A)*W)[m]
    N2 = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N2)
    W = np.fft.rfft(w, N2)
    v2 = np.fft.irfft(A * W, N2)[:n]
    v1 = np.fft.irfft(np.conj(A) * W, N2)[:n]
    dc_part = v1 + v2

    grad = (2.0 * n / (S * S)) * dc_part - (2.0 * n * L / (S * S * S))
    return score, grad


def _objective_value_only(a, sharpness):
    return _smooth_objective_and_grad(a, sharpness)[0]
