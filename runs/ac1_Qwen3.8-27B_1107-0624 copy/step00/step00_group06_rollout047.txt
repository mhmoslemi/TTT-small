from scipy.optimize import linprog
from scipy import sparse
import time

def compute_convolution(a):
    """Compute the full linear convolution of a with itself."""
    return np.convolve(a, a)

def lp_direction(f, M, top_k=300, rng=None):
    """
    Solve the LP to find a direction g that maximizes sum(g)
    subject to convolve(f, g) <= M and g >= 0.
    Uses only the top_k tightest constraints for speed.
    """
    n = len(f)
    c = np.convolve(f, f)
    
    # Select top_k constraints (where c is largest)
    k_indices = np.argsort(c)[::-1][:top_k]
    
    # Build the constraint matrix A for these constraints
    # Row i of A corresponds to constraint: sum_j A[i,j] * g[j] <= M
    # A[i,j] = f[j] if max(0, k_i - n + 1) <= j <= min(k_i, n-1)
    
    # For efficiency, build the sparse matrix
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    
    # Solve: max sum(g) s.t. A@g <= M (vector), g >= 0
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(len(k_indices), M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

    initial_points.append(two_bump)

        if best is not None:

LP-based iterative optimization for minimizing the peak autocorrelation ratio.
Uses restricted-constraint LP to find descent directions, with adaptive
step size and multiple restarts from different initial points.
"""

from scipy import sparse
import time

def _compute_score(a):
    """Fast score computation (numpy only, no list conversion)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_c / (S * S)

def _lp_direction(f, top_k=200):
    """Solve restricted LP for direction g."""
    n = len(f)
    c = np.convolve(f, f)
    M = c.max()
    
    k_idx = np.argsort(c)[::-1][:top_k].astype(np.int64)
    n_k = len(k_idx)
    
    j_start = np.maximum(0, k_idx - n + 1)
    j_end = np.minimum(k_idx, n - 1)
    lengths = j_end - j_start + 1
    
    indptr = np.zeros(n_k + 1, dtype=np.int64)
    np.cumsum(lengths, out=indptr[1:])
    total = int(indptr[-1])
    
    indices = np.empty(total, dtype=np.int64)

        e = int(indptr[i + 1])
        js = int(j_start[i])
        je = int(j_end[i])
        indices[s:e] = np.arange(js, je + 1)
        data[s:e] = f[js:je + 1]
