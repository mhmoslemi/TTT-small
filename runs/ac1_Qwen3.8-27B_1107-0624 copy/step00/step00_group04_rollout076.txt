```python
"""
Hybrid LP-driven search for low peak self-convolution ratios.

The objective is
    score = 2*n*max(f*f) / (sum(f))^2
where * is full convolution and f is nonnegative.

The search starts from several parametric and given initial sequences,
then repeatedly solves a small linear program that finds a direction g
with controlled cross-correlation against the current sequence at peak
lags:
    min t
    s.t. (f*g)[k] <= t  for selected peak lags k,
         sum(g) = sum(f),
         0 <= g <= 1000.
A line search over convex blends f_new = (1-alpha) f + alpha g is done
exactly using already computed convolutions. Random block perturbations
are used to escape stagnation and try different basins.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def clip_seq(x):
    """Clip a sequence to the valid range used by the evaluator."""
    x = np.asarray(x, dtype=np.float64).ravel()
    return np.clip(x, 0.0, 1000.0)


def normalize_seq(f):
    """
    Scale a nonnegative sequence to mean 1 whenever possible.
    Score is scale invariant, so this only improves numerical conditioning.
    """
    f = clip_seq(f)
    n = f.size
    if n == 0:
        return f
    S = float(np.sum(f))
    if S <= 0.0:
        return np.ones(n, dtype=np.float64)
    scale = float(n) / S
    mx = float(np.max(f))
    if mx * scale <= 1000.0:
        f = f * scale
    return clip_seq(f)


def score_array(f):
    """Fast local copy of the evaluation function for valid nonnegative f."""
    f = clip_seq(f)
    n = f.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(f)):
        return np.inf
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    C = np.convolve(f, f)
    M = float(np.max(C))
    if not np.isfinite(M) or M < 0.0:
        return np.inf
    return float((2.0 * n * M) / (S * S))


def blend_score(ff, fg, gg, alpha, S_f, S_g, n):
    """Exact score for (1-alpha)*f + alpha*g using precomputed convolutions."""
    S = (1.0 - alpha) * S_f + alpha * S_g
    if not np.isfinite(S) or S < 0.01:
        return np.inf
    a = 1.0 - alpha
    coef_ff = a * a
    coef_fg = 2.0 * alpha * a
    coef_gg = alpha * alpha
    M = float(np.max(coef_ff * ff + coef_fg * fg + coef_gg * gg))
    if not np.isfinite(M) or M < 0.0:
        return np.inf
    return float((2.0 * n * M) / (S * S))


def mix_two(f1, f2, score1, alphas):
    """Try convex blends between two sequences and return the best found."""
    f1 = normalize_seq(f1)
    f2 = normalize_seq(f2)
    n1 = f1.size
    n2 = f2.size
    if n1 != n2 or n1 == 0:
        return f1, score1 if np.isfinite(score1) else score_array(f1)

    cur = score1 if np.isfinite(score1) else score_array(f1)
    best_f = f1.copy()
    best_score = cur

    S1 = float(np.sum(f1))
    S2 = float(np.sum(f2))
    if S1 < 0.01 or S2 < 0.01:
        return best_f, best_score

    ff = np.convolve(f1, f1)
    fg = np.convolve(f1, f2)
    gg = np.convolve(f2, f2)

    for alpha in alphas:
        a = float(alpha)
        if a <= 0.0:
            continue
        est = blend_score(ff, fg, gg, a, S1, S2, n1)
        if est < best_score - 1e-12:
            f_new = (1.0 - a) * f1 + a * f2
            f_new = normalize_seq(f_new)
            new_score = score_array(f_new)
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
            else:
                best_score = min(best_score, cur)
    return best_f, float(best_score)


def build_A(f, peak_lags):
    """
    Build matrix A with rows representing (f*g)[k] = sum_j A[k,j] g[j].
    Only selected peak lags are included.
    """
    n = int(f.size)
    K = len(peak_lags)
    A = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(peak_lags):
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        lo = k - j_max
        hi = k - j_min
        A[idx, j_min:j_max + 1] = f[lo:hi + 1][::-1]
    return A


def lp_direction(f, peak_lags, S):
    """
    Solve:
        min t
        s.t. (f*g)[k] <= t for selected peak lags k,
             sum(g) = S,
             0 <= g <= 1000.
    Returns g or None.
    """
    if linprog is None:
        return None

    f = np.asarray(f, dtype=np.float64)
    n = int(f.size)
    if n == 0 or len(peak_lags) == 0 or S <= 0.0:
        return None

    A = build_A(f, peak_lags)
    row_mask = np.any(A, axis=1)
    if not np.any(row_mask):
        return None
    A = A[row_mask]
    K = A.shape[0]
    if K == 0:
        return None

    A_ub = np.empty((K, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([S], dtype=np.float64)

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    bounds = [(0.0, 1000.0)] * n + [(0.0, None)]

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'presolve': True, 'time_limit': 4.0}
        )
    except Exception:
        return None

    if res is None or not getattr(res, 'success', False):
        return None

    x = getattr(res, 'x', None)
    if x is None:
        return None

    g = np.asarray(x[:n], dtype=np.float64)
    if not np.all(np.isfinite(g)):
        return None
    g = np.clip(g, 0.0, 1000.0)

    # Mild anti-spike smoothing. This is heuristic and does not break the
    # score formula because the line search uses the actual resulting sum.
    avg = float(S) / float(n)
    cap = 50.0 * avg + 1.0
    if cap < 1000.0 and np.any(g > cap):
        g = np.minimum(g, cap)

    g_sum = float(np.sum(g))
    if not np.isfinite(g_sum) or g_sum <= 0.0:
        return None
    return g


def block_perturb(f, rng, sigma):
    """Multiplicative random perturbation on a contiguous block."""
    f = np.asarray(f, dtype=np.float64).copy()
    n = f.size
    if n == 0:
        return f
    width = int(max(1, n // 200))
    if width >= n:
        width = n
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    f[start:stop] *= np.exp(rng.normal(0.0, sigma, stop - start))
    return clip_seq(f)


def make_candidates(n, given):
    """Generate a diverse set of valid starting sequences."""
    cands = []
    n = int(n)
    if n <= 0:
        return cands

    if given is not None:
        try:
            f = np.asarray(given, dtype=np.float64).ravel()
            if f.size > 0:
                if f.size > n:
                    f = f[:n]
                elif f.size < n:
                    f = np.pad(f, (0, n - f.size), mode='constant')
                f = normalize_seq(f)
                if np.all(np.isfinite(f)) and float(np.sum(f)) > 0.0:
                    cands.append(f.copy())
        except Exception:
            pass

    cands.append(np.ones(n, dtype=np.float64))

    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    P1 = x
    P2 = x * x - 1.0 / 3.0
    P3 = 0.5 * (5.0 * x * x * x - 3.0 * x)

    if n <= 8000:
        vals = [-0.6, -0.3, 0.0, 0.3, 0.6]
        exp_vals = [-1.0, -0.5, 0.0, 0.5, 1.0]
        random_count = 4
    elif n <= 20000:
        vals = [-0.4, 0.0, 0.4]
        exp_vals = [-0.5, 0.0, 0.5]
        random_count = 2
    else:
        vals = [-0.3, 0.0, 0.3]
        exp_vals = []
        random_count = 1

    # Additive low-order polynomial shapes.
    for a1 in vals:
        for a2 in vals:
            for a3 in vals:
                f = 1.0 + a1 * P1 + a2 * P2 + a3 * P3
                f = normalize_seq(f)
                if np.all(np.isfinite(f)) and float(np.sum(f)) > 0.0:
                    cands.append(f)

    # Multiplicative exponential shapes, which stay positive before clipping.
    for a1 in exp_vals:
        for a2 in exp_vals:
            for a3 in exp_vals:
                z = a1 * P1 + a2 * P2 + a3 * P3
                f = np.exp(np.clip(z, -7.0, 7.0))
                f = normalize_seq(f)
                if np.all(np.isfinite(f)) and float(np.sum(f)) > 0.0:
                    cands.append(f)

    # Smooth random deterministically generated shapes.
    for r in range(random_count):
        rng = np.random.default_rng(12345 + r * 17)
        f = np.ones(n, dtype=np.float64)
        for _ in range(5):
            freq = int(rng.integers(1, 7))
            amp = float(rng.uniform(0.02, 0.25))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            f = f + amp * np.sin(2.0 * np.pi * freq * x + phase)
        f = normalize_seq(f)
        if np.all(np.isfinite(f)) and float(np.sum(f)) > 0.0:
            cands.append(f)

    # Exponential skewness in both directions.
    for b in [0.2, 0.4, 0.6, 0.8]:
        f = np.exp(b * x)
        f = normalize_seq(f)
        if np.all(np.isfinite(f)):
            cands.append(f)
        f = np.exp(-b * x)
        f = normalize_seq(f)
        if np.all(np.isfinite(f)):
            cands.append(f)

    return cands


def lp_refine(start_f, deadline, alphas, seed):
    """
    Run the main LP + line search + perturbation loop until deadline.
    Returns the best sequence and score found in this call.
    """
    rng = np.random.default_rng(seed)
    f = normalize_seq(start_f)
    n = f.size
    if n == 0:
        f = np.ones(1, dtype=np.float64)
        n = 1

    cur_score = score_array(f)
    if not np.isfinite(cur_score):
        f = np.ones(n, dtype=np.float64)
        cur_score = score_array(f)

    best_score = cur_score
    best_f = f.copy()
    S = float(np.sum(f))

    if n <= 8000:
        maxK = 80
        K_cycle = [10, 20, 50, 30, 80]
    elif n <= 20000:
        maxK = 40
        K_cycle = [5, 10, 20, 30, 40]
    else:
        maxK = 20
        K_cycle = [5, 10]

    K = min(20, maxK)
    cycle_i = 1
    stuck = 0
    it = 0

    while time.monotonic() < deadline - 3.0:
        if it > 0 and it % 25 == 0:
            K = K_cycle[cycle_i % len(K_cycle)]
            cycle_i += 1
        K = min(K, maxK)
        if K < 1:
            K = 1

        ff = np.convolve(f, f)
        M = float(np.max ff)
        if not np.isfinite(M):
            break

        topK = min(K, ff.size)
        if topK >= ff.size:
            peak_lags = np.arange(ff.size, dtype=np.int64)
        else:
            peak_lags = np.argpartition(ff, -topK)[-topK:].astype(np.int64)

        # Force inclusion of the exact global maximum lag.
        peak_lags = np.unique(np.concatenate((peak_lags, np.array([int(np.argmax(ff))], dtype=np.int64))))
        if peak_lags.size > topK + 1:
            order = np.argsort(ff[peak_lags])[::-1]
            peak_lags = peak_lags[order][:topK + 1]

        g = lp_direction(f, peak_lags.tolist(), S)
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            S_g = float(np.sum(g))
            if S_g > 0.0 and np.isfinite(S_g):
                fg = np.convolve(f, g)
                gg = np.convolve(g, g)

                best_alpha = None
                best_est = np.inf
                for alpha in alphas:
                    a = float(alpha)
                    if a <= 0.0:
                        continue
                    est = blend_score(ff, fg, gg, a, S, S_g, n)
                    if est < best_est:
                        best_est = est
                        best_alpha = a

                if best_alpha is not None and best_est < cur_score - 1e-12:
                    f_new = (1.0 - best_alpha) * f + best_alpha * g
                    f_new = normalize_seq(f_new)
                    new_score = score_array(f_new)
                    if new_score < cur_score - 1e-12:
                        f = f_new
                        S = float(np.sum(f))
                        cur_score = new_score
                        stuck = 0
                        if cur_score < best_score:
                            best_score = cur_score
                            best_f = f.copy()
                    else:
                        stuck += 1
                else:
                    stuck += 1
        else:
            stuck += 1

        if stuck >= 6:
            sigma = 0.05 + 0.01 * min(stuck, 20)
            f = block_perturb(best_f, rng, sigma)
            if rng.random() < 0.10:
                f = 0.5 * (f + f[::-1])
            f = normalize_seq(f)
            S = float(np.sum(f))
            cur_score = score_array(f)
            if not np.isfinite(cur_score):
                f = best_f.copy()
                S = float(np.sum(f))
                cur_score = best_score
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
            stuck = 0
            K = min(20, maxK)

        it += 1
        if it % 100 == 0:
            print("lp_refine iter", it, "current", float(cur_score), "best", float(best_score), "K", int(K))

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Searches for a nonnegative sequence minimizing
    evaluate_sequence. Always returns the best valid sequence found.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1800.0

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42

    deadline = time.monotonic() + budget - 5.0

    given = globals().get('height_sequence_1', None)
    n = 3809
    if given is not None:
        try:
            arr = np.asarray(given, dtype=np.float64).ravel()
            if arr.size > 0:
                n = int(arr.size)
        except Exception:
            n = 3809

    if not np.isfinite(n) or n <= 0:
        n = 3809

    # Safety cap for cases where the supplied sequence is extremely large.
    if n > 20000:
        n = 20000
    if n < 16:
        n = 3809

    cands = make_candidates(n, given)
    alphas = (0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0)

    scored = []
    max_score_candidates = 80 if n > 10000 else 250
    for i, f0 in enumerate(cands):
        if time.monotonic() > deadline - 60.0 and len(scored) > 5:
            break
        if len(scored) >= max_score_candidates:
            break
        f = normalize_seq(f0)
        if f.size != n:
            if f.size > n:
                f = f[:n]
            else:
                f = np.pad(f, (0, n - f.size), mode='constant')
        s = score_array(f)
        if np.isfinite(s):
            scored.append((float(s), i, f))

    if not scored:
        f = np.ones(n, dtype=np.float64)
        scored = [(float(score_array(f)), 0, f)]

    # Sorting tuple (score, index, array) avoids array comparisons.
    scored.sort()

    best_score = float(scored[0][0])
    best_f = scored[0][2].copy()

    print("initial best score", float(best_score), "length", int(best_f.size))

    # Cheap cross-basin mixing before expensive LP refinement.
    top_count = min(5, len(scored))
    for idx in range(1, top_count):
        mixed, mixed_score = mix_two(best_f, scored[idx][2], best_score, alphas)
        if np.isfinite(mixed_score) and mixed_score < best_score:
            best_score = float(mixed_score)
            best_f = mixed.copy()
    print("after mixing score", float(best_score))

    available = max(0.0, deadline - time.monotonic() - 5.0)
    if available > 30.0 and top_count > 0:
        if top_count == 1:
            shares = [0.9]
        else:
            shares = [0.45]
            rem = 0.45 / float(top_count - 1)
            for _ in range(1, top_count):
                shares.append(rem)

        for idx in range(top_count):
            if time.monotonic() >= deadline - 10.0:
                break
            start = scored[idx][2]
            slice_deadline = time.monotonic() + shares[idx] * available
            slice_deadline = min(slice_deadline, deadline - 5.0)
            if slice_deadline <= time.monotonic() + 5.0:
                continue

            print("lp phase", idx, "starting score", float(best_score), "time slice s", float(slice_deadline - time.monotonic()))
            f, sc = lp_refine(start, slice_deadline, alphas, seed=seed_int + 1000 + idx * 77)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_f = f.copy()
            print("lp phase", idx, "finished score", float(best_score))

        # Final polish from the global best.
        if time.monotonic() < deadline - 30.0:
            print("final polish starting score", float(best_score))
            f, sc = lp_refine(best_f, deadline, alphas, seed=seed_int + 99999)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_f = f.copy()
            print("final polish finished score", float(best_score))

    final_list = [float(x) for x in best_f]

    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = np.inf

    if not np.isfinite(final_score):
        # Conservative fallback. The score of ones is exactly 2.
        final_list = [1.0] * max(1, len(best_f))
        try:
            final_score = float(evaluate_sequence(final_list))
        except Exception:
            final_score = np.inf

    print("returned length", len(final_list), "evaluated score", float(final_score))
    return final_list
```
