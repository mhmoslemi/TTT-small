import numpy as np
from scipy.optimize import linprog
import time

def compute_convolution(s):
    return np.convolve(s, s)

def solve_lp(s, M, n):
    """
    Maximize sum(b) subject to:
    b >= 0
    (s*b)[k] <= M for all k = 0, ..., 2n-2
    """
    # Number of variables: n (for b_0, ..., b_{n-1})
    # Number of constraints: 2n-1 (for each k)
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (s*b)[k] <= M for each k
    # (s*b)[k] = sum_i s[i] * b[k-i]
    # This is a linear constraint in b.
    
    # Build constraint matrix A (size (2n-1) x n)
    # A[k, j] = s[j] if 0 <= k-j < n, else 0
    # Wait, (s*b)[k] = sum_i s[i] * b[k-i]
    # For a fixed k, the coefficient of b[j] is s[k-j] (if 0 <= k-j < n)
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = s[idx]
    
    b_vec = np.full(2*n-1, M)
    
    # Non-negativity: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

def compute_score(s):
    """Compute the score directly (same as evaluate_sequence but faster for iteration)."""
    s = np.clip(s, 0, 1000)
    n = len(s)
    total = s.sum()
    if total < 0.01:
        return np.inf
    c = np.convolve(s, s)
    return 2.0 * n * c.max() / (total**2)

def solve_lp_for_b(s, M, n):
    """
    Solve: maximize sum(b) subject to b >= 0, (s*b)[k] <= M for all k.
    """
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    # (s*b)[k] = sum_j s[k-j] * b[j] for valid k-j
    # A[k, j] = s[k-j] if 0 <= k-j < n
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For each j, find all k such that 0 <= k-j < n
        # k ranges from j to j+n-1

        k_end = min(2*n-1, j+n)
        if k_start < 2*n-1:
            A[k_start:k_end, j] = s[0:k_end-k_start]

    result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def solve_lp_for_b(s, M, n):
    c_obj = -np.ones(n)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(s[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_vec = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def compute_score_fast(s):
    """Fast score computation for internal use."""
    s = np.clip(s, 0.0, 1000.0)
    n = len(s)
    total = s.sum()
    if total < 0.01:
        return np.inf
    c = np.convolve(s, s)
    return float(2.0 * n * c.max() / (total ** 2))

def build_constraint_matrix(s, n):
    """Build sparse constraint matrix for (s*b)[k] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(s[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp_for_b(s, M, n):
    """Solve: max sum(b) s.t. b>=0, (s*b)[k] <= M for all k."""
    c_obj = -np.ones(n)
    A = build_constraint_matrix(s, n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def generate_initial_candidates(n):
    """Generate diverse initial candidate sequences."""
    candidates = []
    idx = np.arange(1, n + 1, dtype=np.float64)
    
    # Linear ramp
    candidates.append(idx / n)
    
    # Quadratic ramp
    candidates.append((idx / n) ** 2)
    
    # Cube root ramp (less aggressive)
    candidates.append((idx / n) ** 0.5)
    
    # Sine ramp
    candidates.append(np.sin(idx * np.pi / (2 * n)))
    
    # Exponential ramp (mild)
    candidates.append(np.exp(0.3 * idx / n) - 1)
    
    # Triangular (peak in middle)
    mid = n / 2.0
    candidates.append(np.abs(idx - mid) / mid)
    
    # Inverse linear (decreasing)
    candidates.append((n + 1 - idx) / n)
    
    # Combined: ramp with slight concavity
    candidates.append(idx / n * (1 - 0.2 * idx / n))
    
    # Power law p=0.7
    candidates.append((idx / n) ** 0.7)
    
    # Power law p=1.5
    candidates.append((idx / n) ** 1.5)
    
    return candidates

def lp_refine(s, n, deadline, best_score, best):
    """Apply LP-based refinement iterations."""
    s_work = s.copy()
    
    for iter_num in range(200):
        if time.monotonic() > deadline:
            break
        
        total = s_work.sum()
        if total < 0.01:
            break
        
        s_norm = s_work * np.sqrt(2 * n) / total
        c = np.convolve(s_norm, s_norm)
        M = float(c.max())
        
        b = solve_lp_for_b(s_norm, M, n)
        if b is None or b.sum() < 0.01:
            break
        
        b = b * np.sqrt(2 * n) / b.sum()
        
        # Try multiple mixing parameters
        for t in [0.2, 0.4, 0.6, 0.8]:
            if time.monotonic() > deadline:
                break
            s_try = (1 - t) * s_norm + t * b
            s_try = np.clip(s_try, 0.0, 1000.0)
            score = compute_score_fast(s_try)
            if score < best_score:
                best_score = score
                best = s_try.copy()
                s_work = s_try.copy()
                print(f"  LP iter {iter_num}, t={t}: score={best_score:.6f}")
    
    return best, best_score

def heuristic_refine(s, n, deadline, best_score, best):
    """Apply heuristic block perturbations."""
    rng = np.random.default_rng(123)
    s_work = s.copy()
    no_improve = 0
    
    for iter_num in range(5000):

        if no_improve > 500:
            break

        width = max(1, n // 50)

        score = compute_score_fast(candidate)

            best_score = score
            best = candidate.copy()
            s_work = candidate.copy()
            no_improve = 0
            if iter_num % 100 == 0:
                print(f"  Heur iter {iter_num}: score={best_score:.6f}")

            no_improve += 1
    
    return best, best_score
