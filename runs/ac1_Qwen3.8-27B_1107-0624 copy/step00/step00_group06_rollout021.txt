import time

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def fast_score(a):
    """Score assuming sum(a) = 1."""
    b = fft_convolve(a, a)
    return 2.0 * len(a) * b.max()

def soft_grad_descent(a, budget_s, T_start, T_end, lr_start, lr_end, seed=42):
    """Run soft gradient descent for the given budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    T = T_start
    lr = lr_start
    total_T = T_start - T_end
    v = np.zeros(n)
    momentum = 0.7
    step = 0
    best_a = a.copy()
    best_score = fast_score(a)
    
    while time.monotonic() < deadline:
        b = fft_convolve(a, a)
        b_shifted = b - b.max()
        s = np.exp(b_shifted / T)
        s_sum = s.sum()
        if s_sum > 1e-300:
            s /= s_sum
        else:
            s = np.zeros_like(b)
            s[b.argmax()] = 1.0
        
        # Gradient
        a_flipped = a[::-1]
        grad = 2.0 * fft_convolve(a_flipped, s)[n-1:2*n-1]
        
        # Update
        v = momentum * v - lr * grad
        a_new = a * np.exp(np.clip(v, -5, 5))
        a_sum = a_new.sum()
        if a_sum > 1e-300:
            a_new /= a_sum
        else:
            a_new = a.copy()
        
        score = fast_score(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
        
        a = a_new
        step += 1
        
        # Anneal
        frac = min(1.0, step / 5000)
        T = T_start + (T_end - T_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
