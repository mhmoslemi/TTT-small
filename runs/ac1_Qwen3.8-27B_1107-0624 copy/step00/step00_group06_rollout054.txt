def _softmax_weights(c, tau):
    """Compute softmax weights for smooth max approximation."""
    shifted = tau * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    return w

def _compute_grad(a, tau):
    """Compute gradient of smooth_max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    p = _softmax_weights(c, tau)
    # grad_j = 2 * sum_{i=0}^{n-1} a_i * p_{i+j}

    conv = np.convolve(a[::-1], p)

def _lp_iteration(f, M, solver='hihgs'):
    """One LP iteration: find g that maximizes sum(g) subject to (f*g)_k <= M, g >= 0."""
    n = len(f)
    # Build the constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a banded Toeplitz matrix
    
    # For efficiency, use scipy's linprog with sparse matrix
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(g) (to maximize sum(g))
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        g = res.x
        return g
    return None

    rng = np.random.default_rng(seed)
    candidates.append(rng.random(n))
    
    # Phase 1: Smooth gradient optimization on each candidate
    for idx, cand in enumerate(candidates):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        t_per = remaining / (len(candidates) - idx)
        
        cand = np.maximum(cand, 0)
        if cand.sum() < 1e-10:
            cand = np.ones(n)
        
        a_opt, score = _smooth_optimize(cand, max_time=min(t_per, 300))
        if score < best_score:
            best = a_opt
            best_score = score
            print(f"Candidate {idx}: score = {score:.6f}")
    
    print(f"After gradient optimization: {best_score:.6f}")
    
    # Phase 2: LP-based iterations on the best
    if time.monotonic() < deadline:
        a = best / best.sum()
        for i in range(20):
            if time.monotonic() > deadline:
                break
            c = np.convolve(a, a)
            M = c.max()
            g = _lp_iteration(a, M * 1.001)  # slight relaxation
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                # Blend
                t_blend = 0.3
                a_new = (1 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                
                score = float(evaluate_sequence(a_new.tolist()))
                if score < best_score:
                    best = a_new
                    best_score = score
                    print(f"LP iter {i}: score = {score:.6f}")
                a = a_new
    
    print(f"After LP: {best_score:.6f}")
    
    # Phase 3: Local search
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        a_opt, score = _local_search(best, max_time=remaining, seed=seed+1)
        if score < best_score:
            best = a_opt
            best_score = score
    
    print(f"Final score: {best_score:.6f}")
    return [float(x) for x in best]
