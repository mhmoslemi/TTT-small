def _fft_convolve_full(a, b):
    """Full linear convolution of two 1D arrays using FFT."""
    n = len(a)

    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def _self_conv(a):
    """Full self-convolution a * a using FFT."""
    return _fft_convolve_full(a, a)


def _conv_grad(a, w):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Returns grad[j] = 2 * sum_m a[m] * w[j+m] (valid indices only).
    Uses FFT cross-correlation: ifft(conj(FFT(a)) * FFT(w)).
    """
    n = len(a)
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * corr[:n]
