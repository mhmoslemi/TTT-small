def _next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b):

    size = _next_power_of_2(n)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

    """Compute (a ⋆ b)[i] = sum_j a[j] * b[j+i] for i = 0, ..., len(a)-1."""
    n = len(a)
    m = len(b)
    size = _next_power_of_2(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    corr = np.fft.irfft(np.conj(fa) * fb, size)

Key insight: For palindromic sequences, the center of the convolution is always ≥ S²/n (by Cauchy-Schwarz), giving a lower bound of 2. To beat 2.0, I need **non-palindromic** sequences where the convolution peak is off-center and smaller than S²/n.

1. **Soft-max subgradient descent** using FFT for fast gradient computation

import time


def _np2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

    """FFT-based full convolution."""
    n = len(a) + len(b) - 1
    sz = _np2(n)
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)[:n]


def _fft_xcorr(a, b):
    """Cross-correlation: result[i] = sum_j a[j]*b[j+i], i=0..len(a)-1."""
    n = len(a)
    sz = _np2(n + len(b) - 1)
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(np.conj(fa) * fb, sz)[:n]

    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    C = _fft_conv(a, a)
    return 2.0 * n * np.max(C) / (S * S)


def _soft_grad(a, T):
    """Gradient of soft-max approximation to max(conv(a,a))."""
    n = len(a)
    C = _fft_conv(a, a)
    Cmax = np.max(C)
    if Cmax < 1e-14:
        return np.zeros(n), C, Cmax
    # Softmax weights (shifted for stability)
    e = np.exp(T * (C - Cmax) / Cmax)
    p = e / e.sum()
    # g[i] = 2 * (a xcorr p)[i]
    g = 2.0 * _fft_xcorr(a, p)
    return g, C, Cmax


def _lp_direction(a, M, timeout=2.0, top_k=None):
    """LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for all k (or top_k constraints)."""
    n = len(a)
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    C = _fft_conv(a, a)
    if top_k is not None and top_k < 2 * n - 1:
        # Only use top constraints
        idx = np.argsort(C)[::-1][:top_k]
        idx = np.sort(idx)
    else:
        idx = np.arange(2 * n - 1)

    ncon = len(idx)
    rows = []
    cols = []
    vals = []
    b_vec = []
    for k in idx:
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jlo < jhi:
            js = np.arange(jlo, jhi)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[k - js].tolist())
            b_vec.append(M)
        else:
            rows.extend([k])
            cols.extend([0])
            vals.extend([0.0])
            b_vec.append(M)

    A = csr_matrix((vals, (rows, cols)), shape=(ncon, n))
    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=np.array(b_vec),
                    bounds=[(0, None)] * n, method='highs',
                    options={'time_limit': timeout})
        if r.status == 0 and r.x is not None:
            b = r.x
            sb = b.sum()
            if sb > 1e-10:
                return b * a.sum() / sb
    except Exception:
        pass
    return None


def _perturb(a, rng, strength=0.4):
    """Apply random block perturbation."""
    n = len(a)
    a = a.copy()
    nb = rng.integers(3, 12)
    bs = max(1, n // nb)
    for _ in range(nb):
        s0 = rng.integers(0, n - bs + 1)
        a[s0:s0 + bs] *= np.exp(rng.normal(0, strength, bs))
    a = np.clip(a, 0.001, 1000.0)
    S = a.sum()
    if S > 0.01:
        a = a * S / a.sum()
    return a


def _optimize_from(a, rng, deadline, best_sc, best_seq):
    """Run soft-max gradient descent with LP refinement from initial point a."""
    n = len(a)
    S = a.sum()
    T = 3.0
    step = 0.05 * S / n
    local_best = _obj(a)
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        # Gradient step
        g, C, Cmax = _soft_grad(a, T)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break

        a_try = a - step * g / gn
        a_try = np.clip(a_try, 0.0, 1000.0)
        st = a_try.sum()
        if st < 0.01:
            stagnant += 10
            step *= 0.5
            continue
        a_try = a_try * S / st
        sc = _obj(a_try)

        if sc < local_best:
            a = a_try
            local_best = sc
            stagnant = 0
            step = min(step * 1.03, 0.5 * S / n)
            if sc < best_sc[0]:
                best_sc[0] = sc
                best_seq[0] = a_try.copy()
        else:
            a = a_try
            stagnant += 1
            if stagnant > 40:
                step *= 0.7
                stagnant = 0

        # Increase temperature periodically
        if it % 120 == 119:
            T = min(T * 1.1, 80.0)

        # Perturbation to escape local minima
        if it % 350 == 349:
            a = _perturb(a, rng, strength=0.3)
            T = 3.0
            step = 0.05 * S / n
            stagnant = 0
            local_best = _obj(a)

        # LP refinement (reduced LP for speed)
        if it % 200 == 199 and time.monotonic() < deadline - 60:
            C = _fft_conv(a, a)
            M = np.max(C)
            # Use top constraints for speed
            b = _lp_direction(a, M, timeout=2.0, top_k=min(200, 2*n-1))
            if b is not None:
                for tm in [0.3, 0.5, 0.7, 0.9]:
                    am = (1 - tm) * a + tm * b
                    am = np.clip(am, 0.0, 1000.0)
                    sm = am.sum()
                    if sm > 0.01:
                        am = am * S / sm
                        scm = _obj(am)
                        if scm < best_sc[0]:
                            best_sc[0] = scm
                            best_seq[0] = am.copy()
                            a = am.copy()
                            local_best = scm
                            break

        it += 1

    return best_sc[0], best_seq[0]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2n*max(conv(a,a))/(sum(a))^2 using soft-max subgradient descent
    (FFT-based), periodic LP refinement, and random perturbations.
    Exploits non-palindromic structures to break below the palindromic bound of 2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0

    best_score = np.inf
    best_seq = None

    # Build diverse initializations
    inits = []

    # Use provided sequence as one option
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim == 1 and h.size > 50 and np.isfinite(h).all():
            inits.append(np.clip(h, 0.01, 1000.0))
    except Exception:
        pass

    # Non-symmetric initializations for various lengths
    for n in [1500, 2500, 3500, 5000]:
        # Random
        inits.append(np.clip(rng.random(n) + 0.05, 0.01, 1000))
        # Linear ramp (strongly non-palindromic)
        inits.append(np.clip(np.linspace(1.0, 0.02, n), 0.01, 1000))
        # Two-level step
        a = np.full(n, 0.4)
        a[:int(n * 0.6)] = 1.0
        inits.append(a)
        # Exponential decay
        inits.append(np.clip(np.exp(-4.0 * np.linspace(0, 1, n)) + 0.01, 0.01, 1000))
        # Asymmetric cosine
        x = np.linspace(0, np.pi, n)
        inits.append(np.clip(0.3 + 0.7 * np.sin(x) ** 2, 0.01, 1000))
        # Gaussian (asymmetric center)
        x = np.linspace(0, 1, n)
        inits.append(np.clip(np.exp(-20 * (x - 0.3) ** 2) + 0.02, 0.01, 1000))

    # Evaluate all initializations
    for a in inits:
        sc = _obj(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    print(f"Initial best: {best_score:.6f} (n={len(best_seq)})", flush=True)

    # Optimize from the best initialization
    a = best_seq.copy()
    sc_arr = [best_score]
    seq_arr = [best_seq]

    best_score, best_seq = _optimize_from(a, rng, deadline, sc_arr, seq_arr)

    print(f"After phase 1: {best_score:.6f}", flush=True)

    # Phase 2: Try reconstructing from scratch with different n values
    # and different perturbation strategies
    for n_try in [2000, 3000, 4500]:
        if time.monotonic() > deadline - 30:
            break
        for seed2 in range(3):
            if time.monotonic() > deadline - 15:
                break
            rng2 = np.random.default_rng(seed * 1000 + seed2)
            a2 = np.clip(rng2.random(n_try) * 0.5 + 0.05, 0.01, 1000)
            # Make it non-palindromic
            a2[:n_try // 3] *= 1.5
            a2 = np.clip(a2, 0.01, 1000)
            sc2 = _obj(a2)
            if sc2 < best_score + 0.5:  # Only pursue if competitive
                sc_arr2 = [best_score]
                seq_arr2 = [best_seq]
                bb, ss = _optimize_from(a2, rng2, deadline, sc_arr2, seq_arr2)
                if bb < best_score:
                    best_score = bb
                    best_seq = ss
                    print(f"Phase 2 n={n_try} seed={seed2}: {best_score:.6f}", flush=True)

    # Phase 3: Final polish with high temperature and small steps
    if time.monotonic() < deadline - 20:
        a = best_seq.copy()
        n = len(a)
        S = a.sum()
        T = 50.0  # High temperature (near-exact max)
        step = 0.01 * S / n
        local_best = best_score
        stagnant = 0

        for it in range(200000):
            if time.monotonic() > deadline:
                break
            g, C, Cmax = _soft_grad(a, T)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            a_try = a - step * g / gn
            a_try = np.clip(a_try, 0.0, 1000.0)
            st = a_try.sum()
            if st < 0.01:
                break
            a_try = a_try * S / st
            sc = _obj(a_try)
            if sc < local_best:
                a = a_try
                local_best = sc
                stagnant = 0
                if sc < best_score:
                    best_score = sc
                    best_seq = a_try.copy()
                    if best_score < 1.7:
                        print(f"  Polish it={it}: {best_score:.6f}", flush=True)
            else:
                stagnant += 1
                if stagnant > 50:
                    step *= 0.8
                    stagnant = 0
                    if step < 1e-8 * S / n:
                        break

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_sc = float(evaluate_sequence(result))
    if not np.isfinite(final_sc) or final_sc > best_score:
        # Fallback to constant sequence (score = 2.0)
        result = [1.0] * len(best_seq)
        final_sc = 2.0

    print(f"Final score: {final_sc:.6f}", flush=True)
    return result
