Minimize 2n*max(a*a)/(sum a)^2 using smoothed gradient descent (log-sum-exp),
multi-start with parametric families, and LP-based direction polish.
"""
import numpy as np

from scipy.signal import correlate

def _normalize(a):
    s = np.sum(a)
    if s < 1e-10:
        return a
    return a / s

def _objective_normalized(a):
    """Objective for sum(a)=1: just max(a*a) * 2n"""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)

def _smooth_grad_convex(a, p):
    """Compute gradient of log-sum-exp approximation.
    Returns 2 * correlate(p, a, 'valid')
    """
    return 2.0 * correlate(p, a, mode='valid')

def _gradient_descent(a, n, deadline, eps_start=0.5, eps_end=0.01, 
                      eta=0.5, max_iters=10000, rng=None, verbose=False):
    """Smoothed gradient descent with adaptive step size."""
    a = _normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_val = np.max(np.convolve(a, a))
    
    if rng is None:
        rng = np.random.default_rng()
    
    successful = 0
    failed = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal temperature
        frac = min(1.0, it / (max_iters * 0.8))
        eps = eps_start * (eps_end / eps_start) ** frac
        
        # Compute convolution
        conv = np.convolve(a, a)
        current_max = np.max(conv)
        
        # Softmax weights
        scaled = conv * eps
        scaled -= np.max(scaled)
        p = np.exp(scaled)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            break
        p /= p_sum
        
        # Gradient
        grad = _smooth_grad_convex(a, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        # Scale step so that the change in a is at most ~0.1 in L2
        step = min(eta, 0.1 / grad_norm)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 0.01:
            # Fallback: reduce step
            a_new = a - step * 0.1 * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new < 0.01:
                continue
        a_new = _normalize(a_new)
        
        # Evaluate
        conv_new = np.convolve(a_new, a_new)
        max_new = np.max(conv_new)
        
        if max_new < best_val:
            best_val = max_new
            best_a = a_new.copy()
            successful += 1
        else:
            failed += 1
            # Reduce step size
            eta = max(eta * 0.5, 0.001)
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            print(f"  iter {it}: max_conv={best_val:.6f} eta={eta:.4f} eps={eps:.4f}")
    
    return best_a, _objective_normalized(best_a)

def _lp_direction(a, n, deadline):
    """Find a good perturbation direction using LP."""
    a = _normalize(a.copy().astype(np.float64))
    conv = np.convolve(a, a)
    M = np.max(conv)
    
    # Active set
    threshold = M * 0.98
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        return a
    
    # Limit active set size for LP efficiency
    if len(active) > 200:
        # Keep the top 200
        idx = np.argsort(conv[active])[::-1][:200]
        active = active[idx]
    
    # Build LP constraints
    # Maximize sum(g) subject to:
    #   sum_i a[i]*g[k-i] <= 0 for k in active
    #   -a_j <= g_j <= a_j
    
    from scipy.optimize import linprog
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    # Build active constraint matrix
    A_rows = []
    for k in active:
        row = np.zeros(n)
        # (a*g)_k = sum_j a[k-j] * g[j] for 0<=k-j<n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)

            row[j] = a[k - j]
        A_rows.append(row)
    
    A_active = np.array(A_rows)

    A_bound = np.vstack([-np.eye(n), np.eye(n)])
    b_bound = np.concatenate([a, a])
    
    A_ub = np.vstack([A_active, A_bound])
    b_ub = np.concatenate([np.zeros(len(active)), b_bound])
