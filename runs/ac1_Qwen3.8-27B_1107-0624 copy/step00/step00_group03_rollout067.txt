"""
Algorithm: Soft-max gradient descent with multi-start U-shaped initializations.
We minimize the peak of the autocorrelation (convolution) of a non-negative sequence
by using a smooth approximation (soft-max) of the max, with adaptive T schedule.
U-shaped initializations (high at edges, low in center) exploit the insight that
the "deterministic" triangular shape of the convolution means edges need amplification.
A final local search with targeted perturbations refines the solution.
"""

import numpy as np
import time


def compute_score(a):
    """Fast computation of the evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * np.max(c) / (S * S))


def soft_max_conv(self_grad, a, T):
    """Compute the convolution, soft-max weights, and gradient in one pass."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    Tc = T * (c - np.max(c))
    p = np.exp(Tc)
    p /= np.sum(p)
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * p[j+l]
    # This equals 2 * correlate(p, a, mode='valid')[j]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return c, p, grad


def gd_optimize(a, num_steps, T_start, T_end, deadline):
    """Run gradient descent on the soft-max objective."""
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        S = 1.0
        a = a * (1.0 / S)
    best_a = a.copy()
    best_score = compute_score(a)
    patience = 0
    max_patience = max(30, num_steps // 5)

    for step in range(num_steps):
        if time.monotonic() > deadline:
            break

        frac = step / max(1, num_steps - 1)
        T = T_start * (T_end / T_start) ** frac

        c, p, grad = soft_max_conv(a, a, T)

        # Center gradient to preserve sum
        grad_c = grad - np.mean(grad)

        # Adaptive step size
        grad_std = np.std(grad_c)
        if grad_std < 1e-15:
            break
        step_size = 0.3 * S / (n * grad_std)

        # Apply step
        a_new = a - step_size * grad_c

        # Project onto non-negative
        a_new = np.maximum(a_new, 0.0)

        # Renormalize to preserve sum
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new * S / S_new

        score = float(2.0 * n * np.max(np.convolve(a_new, a_new, mode='full')) / (S_new * S))

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience >= max_patience:
            break

    return best_a, best_score

def make_u_shaped(n, curvature):
    """Create a U-shaped sequence: high at edges, low in center."""
    x = np.linspace(-1.0, 1.0, n)
    return 1.0 + curvature * x ** 2


def make_cosine_u(n):
    """Cosine window that is U-shaped (high at edges)."""
    x = np.linspace(0.0, np.pi, n)
    return 0.5 * (1.0 - np.cos(x))


def make_exp_u(n, power):
    """Exponential U-shape."""
    x = np.linspace(-1.0, 1.0, n)
    return np.exp(power * x ** 2)


def make_random_smooth(n, rng, n_freqs=5):
    """Random smooth sequence from sum of low-frequency cosines."""
    vals = np.ones(n)
    for _ in range(n_freqs):
        f = rng.uniform(0.05, 2.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        vals += amp * np.cos(2.0 * np.pi * f * np.arange(n) / n + ph)
    return np.maximum(vals, 0.05)
