def _compute_score(a):
    """Fast numpy score computation."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_c / (S * S)


def _lp_direction(f, top_k=200):
    """Solve restricted LP: max sum(g) s.t. A@g <= M, g >= 0.
    Uses only top_k tightest constraints."""
    n = len(f)

    k_idx = np.argsort(c)[::-1][:top_k].astype(np.int64)
    n_k = len(k_idx)

    j_start = np.maximum(0, k_idx - n + 1)
    j_end = np.minimum(k_idx, n - 1)
    lengths = j_end - j_start + 1

    indptr = np.zeros(n_k + 1, dtype=np.int64)
    np.cumsum(lengths, out=indptr[1:])
    total = int(indptr[-1])

    if total == 0:
        return None

    indices = np.empty(total, dtype=np.int64)
    data = np.empty(total, dtype=np.float64)

        data[s:e] = f[js:je + 1]

        A_ub=A,

    return None
