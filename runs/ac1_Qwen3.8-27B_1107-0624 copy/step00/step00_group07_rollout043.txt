def _fft_size(n):
    """Next power of 2 >= n."""
    s = 1
    while s < n:
        s *= 2
    return s

def _convolve_full(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = _fft_size(n)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]

def _smooth_max_and_grad(a, beta):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (L, grad_a) where L ≈ max(conv(a,a)) and grad_a = dL/da.
    """
    n = len(a)
    c = _convolve_full(a, a)  # length 2n-1
    
    # Softmax weights
    c_max = np.max(c)
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / np.sum(exp_bc)
    L = c_max + np.log(np.sum(exp_bc)) / beta

    h = _convolve_full(w, a[::-1])  # length (2n-1) + n - 1 = 3n-2
    grad_a = 2.0 * h[n-1 : 2*n-1]
    
    return L, grad_a, c

    deadline = time.monotonic() + max(0, float(budget_s)) - 5.0

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        parent = np.clip(parent, 0.0, 1000.0)
        n = len(parent)
    except Exception:
        parent = np.ones(4096, dtype=np.float64)
        n = len(parent)

    if n < 50 or not np.isfinite(parent).all() or np.sum(parent) < 0.01:
        parent = np.ones(4096, dtype=np.float64)
        n = len(parent)

    print(f"n={n}, starting search, budget={budget_s}s")
    sys.stdout.flush()

    inits = []
    p = parent / np.sum(parent) * n
    inits.append(("parent", p.copy()))

    for r in [0.99995, 0.9999, 0.9998, 0.9997, 0.9995, 0.999, 0.998, 0.997]:
        inits.append((f"geom_{r}", _make_geometric(n, r)))

    for cf, sf in [(0.5, 0.15), (0.5, 0.25), (0.5, 0.4), (0.4, 0.3), (0.6, 0.3)]:
        inits.append((f"gauss_{cf}_{sf}", _make_gaussian_bump(n, cf, sf)))

    lin = np.linspace(3.0, 0.3, n)
    inits.append(("linear", lin.copy()))

    rng0 = np.random.default_rng(seed)
    for i in range(3):
        a_r = np.exp(rng0.normal(0, 0.8, n))
        a_r = np.convolve(a_r, np.ones(30) / 30, mode='same')
        a_r = np.maximum(a_r, 1e-6)
        inits.append((f"rand{i}", a_r.copy()))

    best_score = _score(p)
    best_a = p.copy()
    print(f"Initial best: {best_score:.6f}")
    sys.stdout.flush()

    for idx, (name, init_a) in enumerate(inits):
        if time.monotonic() > deadline:
            break
        time_left = deadline - time.monotonic()
        if time_left < 20:
            break
        alloc = min(200, time_left * 0.7 / max(1, len(inits) - idx))
        t0 = time.monotonic()
        try:
            a_opt, sc = _optimize_logspace(init_a, alloc, seed + idx)
        except Exception as e:
            print(f"  {name}: error {e}")
            continue
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  {name}: {sc:.6f} (t={elapsed:.1f}s)")
        else:
            print(f"  {name}: {sc:.6f} (t={elapsed:.1f}s) [no improve]")
        sys.stdout.flush()

    # Local refinement
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        a_rf, sc = _local_refine(best_a, deadline - time.monotonic(), seed + 100)
        if sc < best_score:
            best_score = sc
            best_a = a_rf.copy()
        print(f"  refine: {sc:.6f} (t={time.monotonic()-t0:.1f}s)")
        sys.stdout.flush()

    # Second round log-space refinement on best
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        a_opt, sc = _optimize_logspace(best_a, deadline - time.monotonic(), seed + 200)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  round2: {sc:.6f} (t={time.monotonic()-t0:.1f}s)")
        sys.stdout.flush()

    print(f"FINAL: {best_score:.6f}")
    sys.stdout.flush()
    return [float(v) for v in best_a]
