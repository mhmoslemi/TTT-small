            idx[total]=d

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_CONV_PAT = {}
_BASIS_CACHE = {}


def _normalize(f):
    """Return a finite, non-negative, finite-sized sequence with sum 1."""
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.ones(1, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if s < 1e-14:
        f = np.ones(f.size, dtype=np.float64) / float(f.size)
    else:
        f = f / s
    return f


def _clip_nonneg(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    return np.clip(f, 0.0, 1000.0)


def _convolve_fast(a, b=None):
    """Fast linear convolution for real arrays."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)

    total = n + m - 1
    if max(n, m) <= 1024 and n * m <= 2000000:
        return np.convolve(a, b)

    L = 1
    while L < total:
        L <<= 1
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(A * B, L)
    c = c[:total]
    return c


def _score_fast(f):
    """Same objective as evaluate_sequence, but faster and assumes non-negative."""
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return float("inf")
    s = float(f.sum())
    if s < 1e-14:
        return float("inf")
    b = _convolve_fast(f, f)
    if b.size == 0:
        return float("inf")
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * f.size * mb / (s * s))


def _softmax_weights(b, tau_frac):
    """Normalize convolution peaks into weights for a smoothed max objective."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return b
    mb = float(np.max(b))
    if mb <= 0.0 or not np.isfinite(mb):
        return b.astype(np.float64)
    tau = max(1e-18, float(tau_frac) * mb)
    z = b / tau
    z = z - float(np.max(z))
    w = np.exp(z)
    sw = float(w.sum())
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= sw
    return w


def _grad_from_weights(w, f):
    """
    Gradient of sum_k w_k (f*f)_k with respect to f.
    If w has length 2n-1 and f has length n, returns length n.
    """
    f = np.asarray(f, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.zeros_like(f)
    if n <= 512:
        tmp = np.convolve(f, w[::-1])
        return 2.0 * tmp[n - 1:2 * n - 1][::-1]

    total = 3 * n - 2
    L = 1
    while L < total:
        L <<= 1
    F = np.fft.rfft(f, L)
    W = np.fft.rfft(w[::-1], L)
    tmp = np.fft.irfft(F * W, L)[:total]
    return 2.0 * tmp[n - 1:2 * n - 1][::-1]


def _hard_grad(f, k):
    """Exact subgradient for one active lag k."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    g = np.zeros(n, dtype=np.float64)
    k = int(k)
    j0 = max(0, k - n + 1)
    j1 = min(n - 1, k)
    if j0 <= j1:
        p0 = k - j0
        p1 = k - j1
        g[j0:j1 + 1] = 2.0 * f[p1:p0 + 1][::-1]
    return g


def _mult_update(f, dim, theta):
    """Multiplicative update f * exp(-theta * dim), renormalized."""
    f = np.asarray(f, dtype=np.float64)
    x = f * np.exp(-float(theta) * dim)
    x = np.maximum(x, 1e-14)
    s = float(x.sum())
    if s <= 1e-14:
        return None
    x /= s
    return x


def _add_update(f, dim, theta):
    """Additive/projected update with step relative to the mean element."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return None
    x = f - (float(theta) / float(n)) * dim
    x = np.maximum(x, 1e-14)
    s = float(x.sum())
    if s <= 1e-14:
        return None
    x /= s
    return x


def _random_mult_trial(f, sigma, rng):
    f = np.asarray(f, dtype=np.float64)
    x = f * np.exp(rng.normal(0.0, float(sigma), f.size))
    return _normalize(x)


def _smooth_antisym_trial(f, sigma, rng):
    """Add a smooth antisymmetric multiplicative perturbation."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    z = rng.normal(0.0, 1.0, n)
    z = 0.5 * (z - z[::-1])
    # very light smoothing by a few box averages
    for _ in range(3):
        z = 0.5 * (z + np.roll(z, 1))
    zm = float(np.max(np.abs(z)))
    if zm > 1e-14:
        z /= zm
    x = f * np.exp(float(sigma) * z)
    return _normalize(np.maximum(x, 0.0))


def _optimize_multiplicative(f0, budget, rng):
    """
    Annealed mirror-descent / subgradient search on the simplex.
    Uses smoothed max-convolution surrogates, hard peak subgradients, and escaping noise.
    """
    if budget <= 0:
        return _normalize(f0)
    f = _normalize(np.asarray(f0, dtype=np.float64).copy())
    n = f.size
    end = time.monotonic() + float(budget)

    best = f.copy()
    best_score = _score_fast(best)
    cur = f.copy()
    cur_score = best_score

    it = 0
    stall = 0
    last_print = time.monotonic()

    while time.monotonic() < end:
        it += 1
        now = time.monotonic()
        frac = max(0.0, min(1.0, (end - now) / max(1e-9, float(budget))))

        if stall > 70:
            cur = best.copy()
            cur_score = best_score
            stall = 0

        b = _convolve_fast(cur, cur)
        if b.size == 0 or not np.all(np.isfinite(b)):
            break
        mb = float(np.max(b))
        if mb <= 0.0:
            break
        cur_score = float(2.0 * n * mb)

        tau_frac = 0.02 + 0.48 * frac * frac
        if stall > 30:
            tau_frac = max(tau_frac, 0.12)

        best_trial = None
        best_trial_score = cur_score

        w = _softmax_weights(b, tau_frac)
        g = _grad_from_weights(w, cur)
        gc = g - float(np.mean(g))
        sc = float(np.max(np.abs(gc)))
        if sc > 1e-18:
            dim = gc / sc
            for theta in (0.8, 2.0, 5.0):
                if time.monotonic() >= end:
                    break
                trial = _mult_update(cur, dim, theta)
                if trial is None:
                    continue
                s = _score_fast(trial)
                if np.isfinite(s) and s < best_trial_score:
                    best_trial_score = s
                    best_trial = trial
            for a_theta in (0.7, 1.8, 3.5):
                if time.monotonic() >= end:
                    break
                trial = _add_update(cur, dim, a_theta)
                if trial is None:
                    continue
                s = _score_fast(trial)
                if np.isfinite(s) and s < best_trial_score:
                    best_trial_score = s
                    best_trial = trial

        if it % 4 == 0 or frac < 0.25 or stall > 15:
            k = int(np.argmax(b))
            h = _hard_grad(cur, k)
            hc = h - float(np.mean(h))
            hsc = float(np.max(np.abs(hc)))
            if hsc > 1e-18:
                hdim = hc / hsc
                for theta in (0.7, 2.0, 6.0):
                    trial = _mult_update(cur, hdim, theta)
                    if trial is None:
                        continue
                    s = _score_fast(trial)
                    if np.isfinite(s) and s < best_trial_score:
                        best_trial_score = s
                        best_trial = trial

        if it % 25 == 0:
            trial = _random_mult_trial(cur, 0.18 + 0.45 * frac, rng)
            s = _score_fast(trial)
            if np.isfinite(s) and s < best_trial_score:
                best_trial_score = s
                best_trial = trial

        if it % 120 == 0:
            trial = _smooth_antisym_trial(best, 0.35 + 0.65 * frac, rng)
            s = _score_fast(trial)
            if np.isfinite(s) and s < best_trial_score:
                best_trial_score = s
                best_trial = trial

        if best_trial is not None:
            if best_trial_score < best_score - 1e-14:
                best = best_trial
                best_score = best_trial_score

            accept = False
            if best_trial_score <= cur_score:
                accept = True
            elif best_trial_score <= cur_score * (1.0 + 0.0015):
                if rng.random() < 0.65:
                    accept = True
            elif rng.random() < 0.07:
                accept = True

            if accept:
                cur = best_trial.copy()
                cur_score = best_trial_score
                stall = 0
            else:
                stall += 1

            if stall > 90:
                cur = best.copy()
                cur_score = best_score
                stall = 0
        else:
            stall += 1

        if it % 2000 == 0 and time.monotonic() - last_print > 30:
            print("    multi it={:d} best={:.10f} cur={:.10f}".format(it, best_score, cur_score), flush=True)
            last_print = time.monotonic()

    if best_score > _score_fast(best):
        best_score = _score_fast(best)
    return best


def _make_basis(N, R):
    """Small antisymmetric-like trigonometric basis for compact shape search."""
    key = (int(N), int(R))
    if key in _BASIS_CACHE:
        return _BASIS_CACHE[key]

    js = np.arange(N, dtype=np.float64)
    c = (N - 1) / 2.0
    basis = []

    phi = js - c
    phi = 0.5 * (phi - phi[::-1])
    mx = float(np.max(np.abs(phi))) if phi.size else 0.0
    basis.append(phi / mx if mx > 0 else np.zeros(N, dtype=np.float64))

    while len(basis) < R:
        m = len(basis)
        phi = np.sin(2.0 * np.pi * float(m) * (js - c) / float(N))
        phi = 0.5 * (phi - phi[::-1])
        mx = float(np.max(np.abs(phi))) if phi.size else 0.0
        if mx > 0:
            phi /= mx
        else:
            phi = np.zeros(N, dtype=np.float64)
        basis.append(phi)

    basis = basis[:R]
    _BASIS_CACHE[key] = basis
    return basis


def _basis_score(x, basis, N):
    x = np.asarray(x, dtype=np.float64)
    f = np.ones(N, dtype=np.float64)
    for a, phi in zip(x, basis):
        f = f + float(a) * phi
    f = np.maximum(f, 1e-6)
    return _score_fast(f)


def _basis_spsa(N, R, init, budget, rng, basis=None):
    """SPSA-style search over a low-dimensional shape parameterization."""
    if basis is None:
        basis = _make_basis(N, R)
    x = np.asarray(init, dtype=np.float64).copy()
    if x.size != R:
        x = np.resize(x, R)
    x = np.clip(x, -1.5, 1.5)

    best_x = x.copy()
    best_score = _basis_score(x, basis, N)
    cur_score = best_score
    c = 0.30
    stall = 0
    end = time.monotonic() + max(0.0, float(budget))
    it = 0

    while time.monotonic() < end:
        it += 1
        d = np.where(rng.random(R) < 0.5, 1.0, -1.0)
        x1 = np.clip(x + c * d, -1.5, 1.5)
        x2 = np.clip(x - c * d, -1.5, 1.5)
        s1 = _basis_score(x1, basis, N)
        s2 = _basis_score(x2, basis, N)

        if s1 <= s2:
            x_new = x1
            s_new = s1
        else:
            x_new = x2
            s_new = s2

        if np.isfinite(s_new):
            if s_new < best_score - 1e-14:
                best_x = x_new.copy()
                best_score = s_new
            if s_new < cur_score - 1e-14:
                stall = 0
            else:
                stall += 1

            if s_new <= cur_score or rng.random() < 0.22:
                x = x_new.copy()
                cur_score = s_new
            elif stall > 120:
                x = best_x.copy()
                cur_score = best_score
                stall = 0

        c = max(0.02, c * 0.9992)

    return best_x, best_score


def _f_from_basis(n, coeffs, R):
    coeffs = np.asarray(coeffs, dtype=np.float64)
    basis = _make_basis(n, R)
    f = np.ones(n, dtype=np.float64)
    for a, phi in zip(coeffs, basis):
        f = f + float(a) * phi
    f = np.maximum(f, 1e-6)
    return _normalize(f)


def _basis_search_and_make(n_large, N_search, R, budget, rng):
    """Initialize low-dim shape search at a small size and sample it at n_large."""
    if budget <= 0:
        return _normalize(np.ones(n_large)), 0.0, None
    basis = _make_basis(N_search, R)
    starts = []

    z = np.zeros(R)
    starts.append(z)

    z2 = np.zeros(R)
    if R > 0:
        z2[0] = 0.8
    starts.append(z2)

    starts.append(0.45 * rng.normal(0.0, 1.0, R))

    if R >= 8:
        z3 = np.zeros(R)
        z3[1] = 0.6
        z3[4] = -0.4
        z3[7] = 0.25
        starts.append(z3)

    best_x = None
    best_score = float("inf")
    per = max(1.0, float(budget) / float(max(1, len(starts))))

    for st in starts:
        if time.monotonic() >= time.monotonic() + max(0.5, per) and best_x is not None:
            break
        x, sc = _basis_spsa(N_search, R, st, per, rng, basis)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_x = x

    if best_x is None:
        best_x = np.zeros(R)
        best_score = _basis_score(best_x, basis, N_search)

    f_large = _f_from_basis(n_large, best_x, R)
    return f_large, _score_fast(f_large), best_x


def _init_blocks(n, alpha, a):
    """Three-block antisymmetric-like start: low/high at edges, middle near 1."""
    n = int(n)
    L = max(1, int(float(alpha) * n))
    if 2 * L >= n:
        L = max(1, (n - 1) // 3)
    f = np.ones(n, dtype=np.float64)
    f[:L] = a
    f[-L:] = 2.0 - a
    f = np.maximum(f, 0.0)
    return _normalize(f)


def _init_sine(n, rng, amp=0.75):
    """Random smooth antisymmetric sine-mode start."""
    n = int(n)
    js = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    f = np.ones(n, dtype=np.float64)

    mode_pool = np.arange(1, 24)
    ms = rng.choice(mode_pool, size=min(5, mode_pool.size), replace=False)
    for m in ms:
        s = rng.choice([-1.0, 1.0])
        magnitude = rng.uniform(0.20, max(0.21, amp))
        phi = np.sin(2.0 * np.pi * float(m) * (js - c) / float(n))
        phi = 0.5 * (phi - phi[::-1])
        mx = float(np.max(np.abs(phi))) if phi.size else 0.0
        if mx > 1e-14:
            f += s * magnitude * phi / mx

    f = np.maximum(f, 1e-6)
    return _normalize(f)


def _init_random_block(n, K, rng):
    """Random positive block-valued start."""
    n = int(n)
    K = max(1, int(K))
    h = 1.0 + 0.45 * rng.normal(0.0, 1.0, K)
    h = np.maximum(h, 0.05)
    f = np.empty(n, dtype=np.float64)
    for i in range(K):
        a = i * n // K
        b = (i + 1) * n // K
        f[a:b] = h[i]
    return _normalize(f)


def _init_from_height(n, rng):
    """Use the provided previous construction if available."""
    hs = globals().get("height_sequence_1", None)
    if hs is None:
        return None
    try:
        arr = np.asarray(hs, dtype=np.float64).ravel()
        if arr.size == 0:
            return None
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        if arr.size == int(n):
            f = arr.copy()
        else:
            xold = np.linspace(0.0, 1.0, arr.size)
            xnew = np.linspace(0.0, 1.0, int(n))
            f = np.interp(xnew, xold, arr)
        f = np.maximum(f, 0.0)
        s = float(f.sum())
        if s < 1e-14:
            return None
        f = f * (1.0 + 0.02 * rng.normal(0.0, 1.0, int(n)))
        return _normalize(np.maximum(f, 0.0))
    except Exception:
        return None


def _downsample(f, n):
    f = np.asarray(f, dtype=np.float64)
    n = int(n)
    if f.size == n:
        return _normalize(f)
    if f.size == 1:
        return np.full(n, float(f[0]), dtype=np.float64)
    cuts = np.linspace(0, f.size, n + 1, dtype=np.int64)
    out = np.zeros(n, dtype=np.float64)
    for i in range(n):
        a = int(cuts[i])
        b = int(cuts[i + 1])
        if b > a:
            out[i] = float(np.mean(f[a:b]))
        else:
            out[i] = float(f[a])
    return _normalize(out)


def _upsample(f, n):
    f = np.asarray(f, dtype=np.float64)
    n = int(n)
    if f.size == n:
        return _normalize(f)
    if f.size == 1:
        return np.full(n, float(f[0]), dtype=np.float64)
    xold = np.linspace(0.0, 1.0, f.size)
    xnew = np.linspace(0.0, 1.0, n)
    y = np.interp(xnew, xold, f)
    return _normalize(np.maximum(y, 0.0))


def _get_conv_pattern(n):
    """
    Fixed sparsity pattern for linear convolution C g:
    (C g)[k] = sum_j g[j] f[k-j] for 0 <= k-j < n.
    """
    n = int(n)
    key = n
    if key in _CONV_PAT:
        return _CONV_PAT[key]

    rows = 2 * n - 1
    total = n * n
    indptr = np.zeros(rows + 1, dtype=np.int32)
    indices = np.empty(total, dtype=np.int32)
    idx = np.empty(total, dtype=np.int32)
    pos = 0

    for k in range(rows):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        L = j1 - j0 + 1
        indptr[k] = pos
        if L:
            indices[pos:pos + L] = np.arange(j0, j1 + 1, dtype=np.int32)
            idx[pos:pos + L] = np.arange(k - j0, k - j1 - 1, -1, dtype=np.int32)
            pos += L

    indptr[rows] = pos
    data = np.zeros(pos, dtype=np.float64)
    A = csr_matrix((data, indices, indptr), shape=(rows, n), dtype=np.float64)
    _CONV_PAT[key] = (A, idx, indptr)
    return (A, idx, indptr)


def _lp_update(f):
    """One LP direction step: maximize sum(g) subject to f*g <= 1 after scaling."""
    f = np.asarray(f, dtype=np.float64).copy()
    n = f.size
    s = float(f.sum())
    if s < 1e-14:
        return None
    f /= s

    b = _convolve_fast(f, f)
    if b.size == 0:
        return None
    M = float(np.max(b))
    if M <= 1e-18 or not np.isfinite(M):
        return None

    A, idx, indptr = _get_conv_pattern(n)
    A.data[:] = f[idx] / M

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(A.shape[0], dtype=np.float64)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs"
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = g[np.isfinite(g)]
    if g.size != n:
        return None
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1e-14:
        return None
    g /= gs
    return g


def _lp_optimize(f0, budget, rng):
    """LP-polish a smaller-length sequence, following the known SOTA idea."""
    if budget <= 0:
        f = _normalize(f0)
        return f, _score_fast(f)

    f = _normalize(np.asarray(f0, dtype=np.float64).copy())
    n = f.size
    end = time.monotonic() + float(budget)

    best = f.copy()
    best_score = _score_fast(best)
    cur = f.copy()
    cur_score = best_score
    stall = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < end:
        it += 1
        g = _lp_update(cur)
        if g is None:
            if stall < 5:
                stall += 1
                cur = _random_mult_trial(best, 0.35, rng)
                cur_score = _score_fast(cur)
                continue
            break

        best_trial = cur
        best_trial_score = cur_score

        for t in (0.05, 0.15, 0.35, 0.65, 1.0):
            trial = (1.0 - t) * cur + t * g
            trial = np.maximum(trial, 0.0)
            s = float(trial.sum())
            if s <= 1e-14:
                continue
            trial /= s
            sc = _score_fast(trial)
            if np.isfinite(sc) and sc < best_trial_score:
                best_trial_score = sc
                best_trial = trial

        if best_trial_score < best_score - 1e-14:
            best = best_trial.copy()
            best_score = best_trial_score

        accept = False
        if best_trial_score <= cur_score:
            accept = True
        elif best_trial_score <= cur_score * (1.0 + 0.001):
            if rng.random() < 0.5:
                accept = True
        elif rng.random() < 0.05:
            accept = True

        if accept:
            cur = best_trial.copy()
            cur_score = best_trial_score
            stall = 0
        else:
            stall += 1
            if stall > 12:
                cur = _random_mult_trial(best, 0.45, rng)
                cur_score = _score_fast(cur)
                stall = 0

        if it % 20 == 0 and time.monotonic() - last_print > 15:
            print("    LP it={:d} best={:.10f} cur={:.10f}".format(it, best_score, cur_score), flush=True)
            last_print = time.monotonic()

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-arm search for low max self-convolution step functions.

    Arms:
    1. Low-dimensional SPSA shape search on smooth antisymmetric bases.
    2. LP-based directional polish at n=1024, upsampled after polish.
    3. Several annealed mirror-descent/subgradient searches at n=4096.
    4. Optional final polish and optional longer-n refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = max(60.0, float(budget_s) - 10.0)

    n_large = 4096
    n_lp = 1024

    basis_budget = min(120.0, max(20.0, total * 0.07))
    lp_budget = min(240.0, max(0.0, total * 0.13))
    final_budget = min(320.0, max(60.0, total * 0.18))
    safety = 30.0

    best_f = None
    best_score = float("inf")
    exact_candidates = []

    def _register(f):
        nonlocal best_f, best_score
        if f is None:
            return
        f = np.asarray(f, dtype=np.float64)
        f = _normalize(f)
        sc = _score_fast(f)
        if np.isfinite(sc) and sc < best_score:
            best_f = f.copy()
            best_score = sc
        exact_candidates.append(f.copy())

    # Arm 1: compact low-dimensional shape search.
    if total > 120:
        print("basis search budget={:.1f}s".format(basis_budget), flush=True)
        f_basis, sc_basis, _ = _basis_search_and_make(n_large, 256, 16, basis_budget, rng)
        print("  basis score={:.10f}".format(sc_basis), flush=True)
        _register(f_basis)

    # Arm 2: LP polish at smaller n, then upsample.
    lp_best_large = None
    if lp_budget > 30 and total - (time.monotonic() - t_start) > 120:
        lp_inits = []
        if best_f is not None:
            lp_inits.append(_downsample(best_f, n_lp))
        lp_inits.append(_init_blocks(n_lp, 0.34, 0.58))
        lp_inits.append(_init_sine(n_lp, rng, 0.8))

        # Prefer the two best-looking LP starts by fast score.
        scored = []
        for init_f in lp_inits:
            init_f = _normalize(init_f)
            scored.append((_score_fast(init_f), init_f))
        vals = np.array([s for s, _ in scored], dtype=np.float64)
        order = np.argsort(vals)
        lp_starts = [scored[i][1] for i in order[:2]]

        per = lp_budget / float(max(1, len(lp_starts)))
        for init_f in lp_starts:
            if time.monotonic() - t_start + per > total - safety:
                per = max(0.0, total - safety - (time.monotonic() - t_start))
            if per < 5:
                break
            print("  LP novice polish on one start budget={:.1f}s".format(per), flush=True)
            f_lp, sc_lp = _lp_optimize(init_f, per, rng)
            print("    LP small-n score={:.10f}".format(sc_lp), flush=True)
            f_lp_large = _upsample(f_lp, n_large)
            sc_up = _score_fast(f_lp_large)
            print("    upsamped score={:.10f}".format(sc_up), flush=True)
            if sc_up < (float("inf") if lp_best_large is None else _score_fast(lp_best_large)):
                lp_best_large = f_lp_large.copy()
            _register(f_lp_large)

    # Arm 3: full-dimensional annealed mirror-descent/subgradient from diverse starts.
    inits = []
    if best_f is not None:
        inits.append(best_f.copy())
    if lp_best_large is not None:
        inits.append(lp_best_large.copy())
    inits.append(_init_blocks(n_large, 0.34, 0.58))
    inits.append(_init_sine(n_large, rng, 0.75))
    inits.append(_init_random_block(n_large, 96, rng))
    h = _init_from_height(n_large, rng)
    if h is not None:
        inits.append(h)

    valid_inits = []
    seen_scores = []
    for f in inits:
        if f is None:
            continue
        f = _normalize(np.asarray(f, dtype=np.float64).copy())
        sc = _score_fast(f)
        if np.isfinite(sc):
            valid_inits.append((sc, f))
            seen_scores.append(sc)

    # Run the four best-looking starts, but keep at least the compact/LP/blocks ones.
    if valid_inits:
        vals = np.array([s for s, _ in valid_inits], dtype=np.float64)
        order = np.argsort(vals)
        kept = [valid_inits[i][1] for i in order[:4]]

        elapsed = time.monotonic() - t_start
        cand_total = max(0.0, total - elapsed - final_budget - safety)
        per = cand_total / float(max(1, len(kept)))

        for j, init_f in enumerate(kept):
            if time.monotonic() - t_start + per > total - safety:
                per = max(0.0, total - safety - (time.monotonic() - t_start))
            if per < 8:
                break
            print("full mirror arm {}/{} budget={:.1f}s n={}".format(
                j + 1, len(kept), per, n_large), flush=True)
            f_res = _optimize_multiplicative(init_f, per, rng)
            sc = _score_fast(f_res)
            print("  full mirror result score={:.10f}".format(sc), flush=True)
            _register(f_res)

    # Arm 4: final polish of the best 4096-length object found so far.
    if best_f is not None:
        elapsed = time.monotonic() - t_start
        fb = min(final_budget, max(0.0, total - elapsed - 15.0))
        if fb > 20:
            print("final 4096 polish budget={:.1f}s".format(fb), flush=True)
            f_pol = _optimize_multiplicative(best_f, fb, rng)
            sc = _score_fast(f_pol)
            print("  final 4096 score={:.10f}".format(sc), flush=True)
            _register(f_pol)

    # Optional: try a longer length from the current best shape if time remains.
    elapsed = time.monotonic() - t_start
    if best_f is not None and elapsed < total - 150:
        n_extra = 8192
        f_extra = _upsample(best_f, n_extra)
        sc_extra = _score_fast(f_extra)
        print("extra upsample to n={} score={:.10f}".format(n_extra, sc_extra), flush=True)
        budge = min(180.0, max(0.0, total - elapsed - 45.0))
        if budge > 30:
            f_extra_pol = _optimize_multiplicative(f_extra, budge, rng)
            sc_extra_pol = _score_fast(f_extra_pol)
            print("  extra n={} polished score={:.10f}".format(n_extra, sc_extra_pol), flush=True)
            _register(f_extra_pol)

    # Choose the best by the actual evaluator when possible.
    final_best = None
    final_score = float("inf")
    # Prefer most recent / best fast-score candidates; evaluate a small set.
    eval_list = exact_candidates[-8:] if len(exact_candidates) > 8 else exact_candidates
    if best_f is not None and (len(eval_list) == 0 or not np.allclose(eval_list[-1], best_f)):
        eval_list = eval_list + [best_f]

    for f in eval_list:
        if f is None:
            continue
        seq = np.asarray(f, dtype=np.float64)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        seq = np.clip(seq, 0.0, 1000.0)
        seq_list = [float(x) for x in seq.tolist()]
        try:
            sc = float(evaluate_sequence(seq_list))
        except Exception:
            sc = _score_fast(seq)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_best = seq_list

    if final_best is None:
        if best_f is not None:
            arr = np.asarray(best_f, dtype=np.float64)
            arr = np.where(np.isfinite(arr), arr, 0.0)
            final_best = [float(x) for x in np.clip(arr, 0.0, 1000.0).tolist()]
        else:
            final_best = [1.0 / 16.0] * 16

    print("selected score={:.10f} len={}".format(final_score, len(final_best)), flush=True)
    return final_best
