def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    # Security Checks
    if not isinstance(sequence, list):
        return np.inf
    if not len(sequence):
        return np.inf

    try:
        for x in sequence:
            # Reject boolean types and non-integer/non-float values
            if isinstance(x, bool) or not isinstance(x, (int, float)) or not np.isfinite(x):
                return np.inf
            # Convert to float explicitly since Scipy might not handle integers for 64-bit sequences
            if isinstance(x, int):
                raise ValueError("Integer values rejected by evaluate_sequence")
    except (TypeError, ValueError):
        return np.inf

    # Convert to float and ensure nonnegative via clipping
    sequence_float = [float(x) for x in sequence]
    sequence_clipped = [max(0.0, x) for x in sequence_float]
    # Clip to [0, 1000] as per previous program's fallback
    sequence_clipped = [min(1000.0, x) for x in sequence_clipped]

    n = len(sequence_clipped)
    # Compute the convolution
    try:
        b_sequence = np.convolve(sequence_float, sequence_float, mode="full")
    except ValueError:
        raise np.inf

    max_b = np.max(b_sequence)
    sum_a = np.sum(sequence_float)

    # Reject Not-a-Number (NaN) or very small sums (less than 0.01)
    if np.isnan(sum_a) or sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a ** 2))
