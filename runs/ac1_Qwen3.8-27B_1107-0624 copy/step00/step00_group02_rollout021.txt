    n = len(a)

    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2*n-1]

    """Compute the objective: 2*n*max(autocorr) / (sum)^2"""
    c = _fft_autocorr(arr)
    max_c = c.max()
    s = arr.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using the standard algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - S)
    if not cond.any():
        theta = (v.sum() - S) / n
        return np.maximum(v - theta, 0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimize peak autocorrelation of non-negative sequences using
subgradient descent with adaptive step size, multi-peak gradients,
and random restarts.

import numpy as np
import time


def _fft_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2 * n - 1]


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - S))[0]
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _grad_topk(a, c, top_k=1):
    """Subgradient from top-K peaks."""
    n = len(a)
    k = min(top_k, len(c))
    part = np.argpartition(c, -k)[-k:]
    order = np.argsort(c[part])[::-1]
    top_idx = part[order]
    weights = c[top_idx].copy()
    wsum = weights.sum()
    if wsum <= 0:
        weights = np.ones(k) / k
    else:
        weights /= wsum
    g = np.zeros(n)
    for m, w in zip(top_idx, weights):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            g[lo:hi + 1] += w * 2.0 * a[m - np.arange(lo, hi + 1)]
    return g
