        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(x[0]) * float(x[0])], dtype=np.float64)
    if n <= 512:
        return np.convolve(x, x)
    L = 2 * n - 1
    F = np.fft.rfft(x, L)
    y = np.fft.irfft(F * F, L)
    y = np.asarray(y, dtype=np.float64)[:L]
    y = np.maximum(y, 0.0)
    if not np.all(np.isfinite(y)):
        return np.convolve(x, x)
    return y


def fast_score(seq):
    x = normalize_nonneg(seq)
    if x is None:
        return float("inf")
    c = fft_self_conv(x)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return float("inf")
    mc = float(np.max(c))
    if mc < 0.0:
        mc = 0.0
    return 2.0 * float(x.size) * mc


def lse_max(c, beta):
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    if beta <= 0.0:
        return m
    z = beta * (c - m)
    z = np.clip(z, -700.0, 700.0)
    return m + float(np.log(float(np.sum(np.exp(z)))) / beta)


def candidate_score(cand):
    x = normalize_nonneg(cand)
    if x is None:
        return None, float("inf")
    sc = fast_score(x)
    return x, sc


def add_candidate_bank(score, seq):
    global good_bank
    if not np.isfinite(score):
        return
    x = normalize_nonneg(seq)
    if x is None:
        return
    if score > best_score + 0.05:
        return
    for item in good_bank:
        if item[1].size == x.size:
            if np.max(np.abs(item[1] - x)) < 1e-8:
                return
    good_bank.append([float(score), x])
    if len(good_bank) > 35:
        idx = 0
        val = -1e300
        for i, item in enumerate(good_bank):
            if item[0] > val:
                val = item[0]
                idx = i
        good_bank.pop(idx)


def record_candidate(score, seq):
    global best_sequence, best_score
    x = normalize_nonneg(seq)
    if x is None or not np.isfinite(score):
        return
    if score < best_score - 1e-12:
        best_sequence = x.copy()
        best_score = float(score)
        print("new best %.10f len=%d" % (best_score, int(x.size)))
    add_candidate_bank(score, x)


def get_bases(n, q):
    key = (int(n), int(q))
    if key in TRIG_CACHE:
        return TRIG_CACHE[key]
    i = np.arange(int(n), dtype=np.float64) / float(int(n))
    arr = []
    for m in range(1, int(q) + 1):
        ang = 2.0 * math.pi * float(m) * i
        arr.append((np.cos(ang), np.sin(ang)))
    TRIG_CACHE[key] = arr
    return arr


def theta_to_p(theta, n):
    theta = np.asarray(theta, dtype=np.float64)
    n = int(n)
    if n < 2:
        return np.array([1.0], dtype=np.float64)
    if theta.size % 2 != 0:
        theta = theta[:int(theta.size // 2) * 2]
    q = theta.size // 2
    if q == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = theta[0::2]
    b = theta[1::2]
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b))):
        return np.ones(n, dtype=np.float64) / float(n)
    amp = float(np.sum(np.hypot(a, b)))
    if not np.isfinite(amp) or amp <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    if amp > 1.0:
        a = a / amp
        b = b / amp
    bases = get_bases(n, q)
    f = np.ones(n, dtype=np.float64)
    for m in range(q):
        cs, sn = bases[m]
        f = f + a[m] * cs + b[m] * sn
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return f / s


def theta_smooth_obj(theta, n, beta):
    p = theta_to_p(theta, int(n))
    c = fft_self_conv(p)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return float("inf")
    return 2.0 * float(p.size) * lse_max(c, float(beta))


def theta_actual_score(theta, n):
    return fast_score(theta_to_p(theta, int(n)))


def random_fourier_best(n, q, count, end_time, seed, initial_score, initial_theta):
    rng = np.random.default_rng(seed)
    best_t = np.array(initial_theta, dtype=np.float64).copy()
    best_s = float(initial_score)
    if best_t.size != 2 * int(q):
        best_t = np.zeros(2 * int(q), dtype=np.float64)
    for _ in range(int(count)):
        if time.monotonic() >= float(end_time):
            break
        t = rng.normal(0.0, 0.7, 2 * int(q))
        p = theta_to_p(t, n)
        sc = fast_score(p)
        if np.isfinite(sc):
            record_candidate(sc, p)
            add_candidate_bank(sc, p)
            if sc < best_s:
                best_s = float(sc)
                best_t = t.copy()
    return best_t, best_s


def optimize_fourier(n, qmax, end_time, seed=600, init_theta=None, do_de=True):
    n = int(n)
    if n < 2 or qmax < 1:
        return None
    if time.monotonic() >= float(end_time):
        return None
    qmax = int(qmax)
    best_theta = np.zeros(2 * qmax, dtype=np.float64)
    if init_theta is not None:
        t = np.asarray(init_theta, dtype=np.float64)
        take = min(int(t.size), 2 * qmax)
        if take > 0:
            best_theta[:take] = t[:take]
    p0 = theta_to_p(best_theta, n)
    best_sc = fast_score(p0)
    record_candidate(best_sc, p0)
    add_candidate_bank(best_sc, p0)

    qrand = min(4, qmax)
    if qrand >= 1:
        if init_theta is None:
            cnt = 1200 if n <= 600 else 400
            t, s = random_fourier_best(n, qrand, cnt, time.monotonic() + 12.0,
                                       int(seed), best_sc, best_theta[:2 * qrand])
            if s < best_sc:
                best_sc = s
                best_theta = np.zeros(2 * qmax, dtype=np.float64)
                best_theta[:2 * qrand] = t
        else:
            rng = np.random.default_rng(seed + 17)
            for _ in range(250):
                if time.monotonic() >= float(end_time) - 30.0:
                    break
                t = best_theta.copy()
                t += rng.normal(0.0, 0.12, t.size)
                p = theta_to_p(t, n)
                sc = fast_score(p)
                if np.isfinite(sc):
                    record_candidate(sc, p)
                    add_candidate_bank(sc, p)
                    if sc < best_sc:
                        best_sc = sc
                        best_theta = t.copy()

    if do_de and HAS_OPT:
        qde = min(4, qmax)
        if qde >= 1:
            bounds = [(-1.0, 1.0)] * (2 * qde)
            try:
                maxiter_de = 25 if n <= 1024 else 18
                res = differential_evolution(
                    theta_smooth_obj,
                    bounds,
                    args=(n, float(20 * n)),
                    seed=int(seed),
                    maxiter=maxiter_de,
                    popsize=10,
                    tol=1e-7,
                    polish=False
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    x = np.asarray(res.x, dtype=np.float64)
                    p = theta_to_p(x, n)
                    sc = fast_score(p)
                    if np.isfinite(sc):
                        record_candidate(sc, p)
                        add_candidate_bank(sc, p)
                        if sc < best_sc:
                            best_sc = sc
                            best_theta = np.zeros(2 * qmax, dtype=np.float64)
                            best_theta[:2 * qde] = x
            except Exception:
                pass

    if not HAS_OPT:
        return best_theta

    for qcur in range(2, qmax + 1):
        if time.monotonic() >= float(end_time):
            break
        bound = [(-1.0, 1.0)] * (2 * qcur)
        base = np.zeros(2 * qcur, dtype=np.float64)
        take = min(qcur, best_theta.size // 2)
        if take > 0:
            base[:2 * take] = best_theta[:2 * take]
        starts = [base.copy()]
        rng = np.random.default_rng(seed + 100 + qcur)
        starts_count = 4 if qcur <= 6 else 3
        for _ in range(starts_count):
            x = base.copy()
            x += rng.normal(0.0, 0.18, x.size)
            starts.append(x)

        maxiter_lb = 55 if n <= 1024 else 35
        for x0 in starts:
            if time.monotonic() >= float(end_time):
                break
            x = x0.copy()
            for beta_frac in (5, 20, 60):
                if time.monotonic() >= float(end_time):
                    break
                beta = float(beta_frac * n)
                try:
                    res = minimize(
                        theta_smooth_obj,
                        x,
                        args=(n, beta),
                        method="L-BFGS-B",
                        bounds=bound,
                        options={
                            "maxiter": maxiter_lb,
                            "ftol": 1e-13,
                            "gtol": 1e-8,
                            "maxls": 20
                        }
                    )
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        x = res.x
                except Exception:
                    pass
                p = theta_to_p(x, n)
                sc = fast_score(p)
                if np.isfinite(sc):
                    record_candidate(sc, p)
                    add_candidate_bank(sc, p)
                    if sc < best_sc:
                        best_sc = sc
                        best_theta = np.zeros(2 * qmax, dtype=np.float64)
                        best_theta[:2 * qcur] = x
    return best_theta


def softmax_z(z, n):
    n = int(n)
    zz = np.zeros(n, dtype=np.float64)
    if n > 1:
        zz[1:] = np.asarray(z, dtype=np.float64)[:n - 1]
    zz = zz - float(np.max(zz))
    p = np.exp(zz)
    s = float(p.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / s


def softmax_obj(z, n, beta):
    zz = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(zz)):
        return float("inf")
    p = softmax_z(zz, int(n))
    c = fft_self_conv(p)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return float("inf")
    return 2.0 * float(p.size) * lse_max(c, float(beta))

def active_gradient(p, active):
    n = int(p.size)
    grad = np.zeros(n, dtype=np.float64)
    for kk in active:
        k = int(kk)
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if hi < lo:
            continue
        if hi == lo:
            grad[lo] += 2.0 * p[lo]
        else:
            grad[lo:hi + 1] += 2.0 * p[hi:lo - 1:-1]
    return grad


def try_update_local(cand, best_local, best_local_sc, last_improve, step):
    x, sc = candidate_score(cand)
    if x is not None and sc < best_local_sc - 1e-15:
        record_candidate(sc, x)
        add_candidate_bank(sc, x)
        return x, x.copy(), sc, step
    return best_local, best_local, best_local_sc, last_improve


def refine_sequence(initial, end_time):
    x = normalize_nonneg(initial)
    if x is None:
        return
    end_time = float(end_time)
    cur = x.copy()
    best_local = x.copy()
    sc = fast_score(cur)
    best_local_sc = sc
    record_candidate(sc, cur)
    add_candidate_bank(sc, cur)
    last_improve = 0
    step = 0
    rng = np.random.default_rng(555)
    n = int(cur.size)

    while time.monotonic() < end_time:
        step += 1
        sigmas = (0.03, 0.07, 0.15, 0.35, 0.80)
        sigma = sigmas[step % len(sigmas)]

        if n <= 8192:
            noise = rng.normal(0.0, sigma, n)
            noise = np.clip(noise, -4.0, 4.0)
            cand = cur * np.exp(noise)
            cur, best_local, best_local_sc, last_improve = try_update_local(
                cand, cur, best_local, best_local_sc, last_improve, step)

        if step % 3 == 0 and n <= 8192:
            c = fft_self_conv(cur)
            if c.size and np.all(np.isfinite(c)):
                M = float(np.max(c))
                if np.isfinite(M):
                    tol = max(1e-12, 0.03 * M)
                    active = np.where(c >= M - tol)[0]
                    if active.size > 350:
                        vals = c[active]
                        order = np.argsort(vals)
                        active = active[order[-350:]]
                    g = active_gradient(cur, active)
                    gscale = float(np.mean(np.abs(g)))
                    if gscale > 1e-15:
                        gs = g / gscale
                        for eta in (0.05, 0.20, 0.80, 2.00):
                            cand = cur - eta * gs
                            cur, best_local, best_local_sc, last_improve = try_update_local(
                                cand, cur, best_local, best_local_sc, last_improve, step)
                        base = gs - float(np.mean(gs))
                        for eta in (0.20, 0.80, 2.50, 6.00):
                            cand = cur * np.exp(np.clip(-eta * base, -40.0, 40.0))
                            cur, best_local, best_local_sc, last_improve = try_update_local(
                                cand, cur, best_local, best_local_sc, last_improve, step)

        if step % 8 == 0 and len(good_bank) > 1:
            item = good_bank[int(rng.integers(len(good_bank)))]
            y = item[1]
            if y.size == n:
                for t in (0.10, 0.30):
                    cand = (1.0 - t) * cur + t * y
                    cur, best_local, best_local_sc, last_improve = try_update_local(
                        cand, cur, best_local, best_local_sc, last_improve, step)

        if step - last_improve > 80 and n <= 8192:
            cand = cur * np.exp(np.clip(rng.normal(0.0, 1.0, n), -4.0, 4.0))
            cur, best_local, best_local_sc, last_improve = try_update_local(
                cand, cur, best_local, best_local_sc, last_improve, step)

        if n > 8192 and (step - last_improve) > 60:
            cand = cur * np.exp(np.clip(rng.normal(0.0, 0.05, n), -2.0, 2.0))
            cur, best_local, best_local_sc, last_improve = try_update_local(
                cand, cur, best_local, best_local_sc, last_improve, step)

    record_candidate(best_local_sc, best_local)
    add_candidate_bank(best_local_sc, best_local)


def exact_refine(seq, end_time):
    global best_sequence, best_score
    x = normalize_nonneg(seq)
    if x is None:
        return x, float("inf")
    try:
        sc = evaluate_sequence([float(v) for v in x.tolist()])
    except Exception:
        sc = float("inf")
    best = x.copy()
    bestsc = float(sc) if np.isfinite(sc) else float("inf")
    record_candidate(bestsc, best)
    add_candidate_bank(bestsc, best)
    rng = np.random.default_rng(1234)
    n = int(best.size)
    count = 0
    while time.monotonic() < float(end_time) and count < 30:
        count += 1
        sigmas = (0.02, 0.05, 0.10, 0.20)
        sigma = sigmas[count % 4]
        cand = best * np.exp(np.clip(rng.normal(0.0, sigma, n), -4.0, 4.0))
        x2, sc_fast = candidate_score(cand)
        if x2 is not None and np.isfinite(sc_fast) and sc_fast < bestsc + 0.002:
            try:
                sc_ex = evaluate_sequence([float(v) for v in x2.tolist()])
            except Exception:
                sc_ex = float("inf")
            if np.isfinite(sc_ex) and sc_ex < bestsc:
                best = x2.copy()
                bestsc = float(sc_ex)
                record_candidate(bestsc, best)
                add_candidate_bank(bestsc, best)
    return best, bestsc


def finalize_candidates(deadline):
    fallback = np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)
    items = good_bank[:]
    items.append([float(best_score), best_sequence.copy()])
    items.sort(key=bank_score_key)

    best_seq = None
    best_exact = float("inf")
    seen = []
    evaluated = 0
    for item in items:
        if time.monotonic() > float(deadline) - 10.0:
            break
        if evaluated >= 14:
            break
        sc, arr = item
        if arr.size == 0 or not np.all(np.isfinite(arr)):
            continue
        dup = False
        for s in seen:
            if s.size == arr.size:
                if np.max(np.abs(s - arr)) < 1e-12:
                    dup = True
                    break
        if dup:
            continue
        seen.append(arr)
        try:
            ex = evaluate_sequence([float(v) for v in arr.tolist()])
        except Exception:
            ex = float("inf")
        evaluated += 1
        if np.isfinite(ex) and ex < best_exact:
            best_exact = float(ex)
            best_seq = arr.copy()
            print("exact best %.10f len=%d" % (best_exact, int(arr.size)))
        if time.monotonic() > float(deadline) - 30.0:
            break

    if best_seq is None:
        try:
            ex = evaluate_sequence(fallback.tolist())
        except Exception:
            ex = float("inf")
        best_seq = fallback.copy()
        best_exact = float(ex) if np.isfinite(ex) else float("inf")

    if np.isfinite(best_exact) and best_exact > TARGET and time.monotonic() < float(deadline) - 60.0:
        refined, refined_sc = exact_refine(best_seq, float(deadline) - 20.0)
        if refined is not None and np.isfinite(refined_sc) and refined_sc < best_exact:
            best_seq = refined
            best_exact = float(refined_sc)
            record_candidate(best_exact, best_seq)
            add_candidate_bank(best_exact, best_seq)
    elif best_seq is None:
        best_seq = fallback
    return best_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    global best_sequence, best_score, good_bank
    seed = int(seed)
    total = max(10.0, float(budget_s))
    start = time.monotonic()
    deadline = start + max(10.0, total - 90.0)

    best_sequence = np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)
    best_score = 16.0 / 9.0
    good_bank = [[best_score, best_sequence.copy()]]
    TRIG_CACHE.clear()

    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.size > 0:
            sc = fast_score(h)
            record_candidate(sc, h)
            add_candidate_bank(sc, h)
    except Exception:
        pass

    # Short exact-search phase for small lengths.
    small_end = min(deadline - 700.0, start + 45.0)
    for n in (6, 8, 10, 12, 16, 24):
        if time.monotonic() > small_end:
            break
        rem = small_end - time.monotonic()
        if rem > 2.0:
            optimize_small(n, time.monotonic() + min(8.0, rem), seed + 100 + n)

    theta = None
    if time.monotonic() < deadline - 500.0:
        theta = optimize_fourier(512, 8,
                                 min(deadline - 500.0, time.monotonic() + 280.0),
                                 seed + 300, None, True)

    if theta is not None and time.monotonic() < deadline - 420.0:
        theta2 = optimize_fourier(1024, 8,
                                  min(deadline - 420.0, time.monotonic() + 260.0),
                                  seed + 301, theta, False)
        if theta2 is not None:
            theta = theta2

    if theta is not None and time.monotonic() < deadline - 280.0:
        theta3 = optimize_fourier(2048, 6,
                                  min(deadline - 280.0, time.monotonic() + 160.0),
                                  seed + 302, theta, False)
        if theta3 is not None:
            theta = theta3

    if theta is not None:
        for n in (512, 1024, 2048, 4096, 6121):
            if time.monotonic() > deadline - 160.0:
                break
            p = theta_to_p(theta, n)
            sc = fast_score(p)
            record_candidate(sc, p)
            add_candidate_bank(sc, p)

    refine_end = deadline - 130.0
    if time.monotonic() < refine_end:
        cand_items = []
        if best_sequence.size >= 2:
            cand_items.append([float(best_score), best_sequence.copy()])
        for item in good_bank:
            if item[1].size >= 64 and item[0] <= best_score + 0.04:
                cand_items.append([float(item[0]), item[1].copy()])
        cand_items.sort(key=bank_score_key)
        used = set()
        tries = 0
        for item in cand_items:
            if time.monotonic() > refine_end - 25.0:
                break
            if tries >= 4:
                break
            key = (item[1].size, int(round(item[0], 8)))
            if key in used:
                continue
            used.add(key)
            tries += 1
            alloc = min(240.0, refine_end - time.monotonic())
            if alloc > 25.0:
                refine_sequence(item[1], time.monotonic() + alloc)

    final_seq = finalize_candidates(deadline)
    if final_seq is None or final_seq.size == 0:
        final_seq = np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)
    return [float(v) for v in np.asarray(final_seq, dtype=np.float64).tolist()]
