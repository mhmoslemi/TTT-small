import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_convolution(p):
    """Compute full convolution of p with itself using FFT."""
    n = len(p)
    size = 2 * n - 1
    # Use zero-padded FFT
    N = 1 << (size - 1).bit_length()  # next power of 2
    P = rfft(p, N)
    C = irfft(P * P, N)
    return C[:size]

def softmax(x, beta):
    """Numerically stable softmax."""
    x = beta * x
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()

def gradient_descent_step(p, beta, lr):
