def lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) subject to f*g <= M elementwise."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None

import time

def fast_score(a):
    """Fast score computation (assumes a is valid)."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def multiplicative_update(a, c, top_k, eta):
    """Apply multiplicative update based on top peaks."""
    n = len(a)
    # Get top k peak positions
    peak_positions = np.argsort(c)[-top_k:][::-1]  # sorted descending
    peak_values = c[peak_positions]
    
    # Compute pressure: for each i, sum over peaks of a[peak-i] (if valid)
    pressure = np.zeros(n)
    for k in peak_positions:
        weight = peak_values[list(peak_positions).index(k)] / peak_values[0]  # relative weight
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                pressure[i] += weight * a[j]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * pressure)
    a_new = np.clip(a_new, 0, 1000)
    return a_new

import numpy as np
import time

def fast_convolve(a, b):
    """FFT-based convolution for speed."""
    from scipy.signal import fftconvolve
    return fftconvolve(a, b, mode='full')

def fast_score(a):
    n = len(a)
    c = fast_convolve(a, a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def compute_pressure(a, peak_positions, peak_weights):
    n = len(a)
    pressure = np.zeros(n)
    for k, w in zip(peak_positions, peak_weights):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            pressure[idx] += w * a[k - idx]
    return pressure

def local_search(a, deadline, rng, top_k=10, eta=0.1):
    """Run local search from initial sequence a."""
    best = a.copy()
    best_score = fast_score(best)
    
    while time.monotonic() < deadline:
        c = fast_convolve(best, best)
        # Get top peaks
        peak_vals, peak_pos = np.sort(c)[::-1][:top_k], np.argsort(c)[::-1][:top_k]
        
        # Normalize weights (larger peaks get more weight)
        weights = peak_vals / peak_vals[0]
        
        # Compute pressure
        pressure = compute_pressure(best, peak_pos, weights)
        
        # Multiplicative update
        candidate = best * np.exp(-eta * pressure)
        candidate = np.clip(candidate, 0, 1000)
        
        score = fast_score(candidate)
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
        else:
            # Try smaller step
            candidate2 = best * np.exp(-eta * 0.1 * pressure)
            candidate2 = np.clip(candidate2, 0, 1000)
            score2 = fast_score(candidate2)
            if np.isfinite(score2) and score2 < best_score:
                best = candidate2
                best_score = score2
            else:
                eta = max(0.01, eta * 0.9)
