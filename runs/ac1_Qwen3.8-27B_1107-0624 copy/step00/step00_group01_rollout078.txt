def _compute_conv_and_grad(b, p, fft_size, n):

    B = np.fft.rfft(b, fft_size)

    tc = tau * conv
    tc_max = np.max(tc)

    start_wall = time.monotonic()

    iter_count = 0
    print_every = 3000

    while time.monotonic() < deadline:
        # Compute convolution and gradient
        conv, grad = _compute_conv_and_grad(b, None, fft_size, n)
        # We need p before computing grad, so recompute
        p = _softmax_from_conv(conv, tau)
        conv, grad = _compute_conv_and_grad(b, p, fft_size, n)

        max_c = np.max(conv)
        current_score = _score_from_max_c(max_c, n)

        # Track best
        if current_score < best_score:
            best_score = current_score
            best_b = b.copy()
            last_improve_wall = time.monotonic()

        iter_count += 1

        # Progress
        if iter_count % print_every == 0:
            elapsed = time.monotonic() - start_wall
            print(f"iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.1f}, eta={eta:.0f}, elapsed={elapsed:.0f}s")

        # Stuck: perturb and restart
        if time.monotonic() - last_improve_wall > stall_limit:
            noise_scale = 0.05 / np.sqrt(n)
            b = b + rng.normal(0.0, noise_scale, n)
            b = np.maximum(b, 0.0)
            b_sum = b.sum()
            if b_sum < 1e-15:
                b = np.ones(n) / n
            else:
                b = b / b_sum
            eta = float(n)
            tau = 5.0
            last_improve_wall = time.monotonic()
            conv_r, _ = _compute_conv_and_grad(b, None, fft_size, n)
            print(f"  RESTART (iter {iter_count}): score={_score_from_max_c(np.max(conv_r), n):.6f}")
            continue

        # Soft objective for step acceptance
        J_old = _soft_objective(conv, tau)

        # Mirror descent step in log domain
        log_b = np.log(np.maximum(b, 1e-30))
        log_w = log_b - eta * grad
        log_w -= np.max(log_w)
        w = np.exp(log_w)
        w_sum = w.sum()

        accepted = False
        if w_sum > 1e-15 and np.isfinite(w_sum):
            b_new = w / w_sum
            conv_new, _ = _compute_conv_and_grad(b_new, p, fft_size, n)
            J_new = _soft_objective(conv_new, tau)
            if J_new <= J_old + 1e-12:
                b = b_new
                eta = min(eta * eta_incr, eta_max)
                accepted = True
            else:
                eta = max(eta * eta_decr, eta_min)
        else:
            b = np.ones(n) / n
            eta = float(n)

        # Update temperature
        tau = min(tau * tau_rate, tau_max)

        # If we've been at max tau for a while and improving slowly,
        # try a small random kick to escape potential plateaus
        if tau >= tau_max and not accepted and iter_count % 10000 == 0:
            b = b + rng.normal(0.0, 0.001 / n, n)
            b = np.maximum(b, 0.0)
            b = b / b.sum()
            eta = float(n) * 0.5

    # Convert to output: scale to sum = sqrt(2n) for reasonable magnitudes
    S = np.sqrt(2.0 * n)
    result = best_b * S
    result = np.clip(result, 0.0, 1000.0)

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in result]))
    print(f"Final: best_score={best_score:.6f}, verified={final_score:.6f}")
    return [float(x) for x in result]
