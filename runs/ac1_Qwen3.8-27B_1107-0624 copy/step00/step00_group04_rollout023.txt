from scipy.optimize import linprog

def compute_conv_max(p):
    """Compute max of np.convolve(p, p)"""
    return np.max(np.convolve(p, p))

def solve_lp(p, M):
    """Solve the LP: max sum(q) s.t. conv(p,q) <= M elementwise, q >= 0"""
    n = len(p)
    # Variables: q[0], ..., q[n-1]
    # Objective: maximize sum(q) => minimize -sum(q)
    c = -np.ones(n)

            if 0 <= i < n:

    bounds = [(0, None)] * n

    n = len(p)

    cols = []

from scipy.optimize import linprog
import time


def build_conv_indices(n):
    """Build row/col index arrays for the convolution matrix."""
    n_entries = n * n  # Total non-zeros in the convolution matrix
    row_arr = np.empty(n_entries, dtype=np.int32)
    col_arr = np.empty(n_entries, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        count = i_end - i_start + 1
        row_arr[idx:idx + count] = k
        col_arr[idx:idx + count] = np.arange(i_start, i_end + 1, dtype=np.int32)

    print(f"FINAL RESULT: best_score={best_score:.6f}")
    return [float(x) for x in best_p]
