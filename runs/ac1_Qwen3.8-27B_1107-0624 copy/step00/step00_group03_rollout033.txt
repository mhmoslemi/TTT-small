def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find the threshold
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / (1 + np.arange(1, n+1)) * (cssv - 1) > 0)
    # Wait, the standard algorithm:
    # rho = max {j : u[j] - (sum_{i<=j} u[i] - 1) / (j+1) > 0}
    # theta = (sum_{i<=rho} u[i] - 1) / (rho + 1)
    # project[i] = max(0, v[i] - theta)
    
    nnz = np.nonzero(u * np.arange(1, n+1) - (cssv - 1) > 0)[0]
    if len(nnz) == 0:
        return np.ones(n) / n
