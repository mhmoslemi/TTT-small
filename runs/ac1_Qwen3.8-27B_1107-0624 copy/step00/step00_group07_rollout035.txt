    # Cosine taper (Hann window)
    cos_t = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    initializations.append(cos_t / cos_t.sum())

    # Asymmetric sequences (key: asymmetry reduces center convolution)
    ramp = i + 1.0
    initializations.append(ramp / ramp.sum())
    initializations.append(ramp[::-1] / ramp.sum())

    # Exponential decays at various rates
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0, 10.0, 20.0]:
        exp_s = np.exp(-alpha * i / n)
        initializations.append(exp_s / exp_s.sum())
        initializations.append(exp_s[::-1] / exp_s.sum())

    # Two-block structures (first half = 1, second half = ratio)
    half = n // 2
    for ratio in [0.1, 0.3, 0.5, 1.0, 2.0, 3.0, 5.0, 10.0]:
        tb = np.ones(n)
        tb[half:] = ratio
        initializations.append(tb / tb.sum())
        initializations.append(tb[::-1] / tb.sum())

    # Three-block: high-low-high
    third = n // 3
    for mid_ratio in [0.1, 0.3, 0.5]:
        tb3 = np.ones(n)
        tb3[third:2 * third] = mid_ratio
        initializations.append(tb3 / tb3.sum())

    # U-shaped (high at edges, low in middle)
    center = (n - 1) / 2.0
    for beta in [5.0, 10.0, 20.0, 50.0]:
        u = ((i - center) / (n * 0.4)) ** 2 * beta + 0.01
        initializations.append(u / u.sum())

    # Gaussian bumps
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        initializations.append(g / g.sum())

    # "Anti-gaussian" (inverse): high where gaussian is low
    for sigma_frac in [0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        inv_g = 1.0 / (g + 0.05)
        initializations.append(inv_g / inv_g.sum())

    # Frequency-based: 1 + a*cos(2*pi*f*i/n) for various f, a
    for f in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            seq = 1.0 + amp * np.cos(2 * np.pi * f * i / n)
            seq = np.clip(seq, 0.01, None)
            initializations.append(seq / seq.sum())

    # Random smooth sequences via random Fourier features
    for trial in range(50):
        n_freq = rng.integers(3, 15)
        freqs = rng.uniform(0.5, n // 3, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = rng.uniform(0.05, 1.0, size=n_freq)
        seq = np.ones(n) * 0.4
        for f, p, amp in zip(freqs, phases, amps):
            seq += amp * np.cos(2 * np.pi * f * i / n + p)
        seq = np.clip(seq, 0.01, None)
        initializations.append(seq / seq.sum())

    # Random asymmetric sequences
    for trial in range(30):
        # Random monotone-ish sequence
        increments = rng.exponential(1.0, size=n)
        seq = np.cumsum(increments)
        seq = seq - seq.min() + 0.01
        initializations.append(seq / seq.sum())

    # Evaluate all initializations
    init_scores = []
    for idx, init in enumerate(initializations):
        if time.monotonic() > deadline:
            break
        init = np.clip(init, 0.0, 1000.0)
        sc = float(evaluate_sequence(init.tolist()))
        if np.isfinite(sc):
            init_scores.append((sc, idx, init.copy()))
        if idx % 50 == 0:
            print(f"  Evaluated {idx}/{len(initializations)} initializations")

    init_scores.sort(key=lambda x: x[0])
    if init_scores:
        for sc_val, idx_val, init_val in init_scores[:5]:
            print(f"  Top init #{idx_val}: score={sc_val:.6f}")
        if init_scores[0][0] < best_score:
            best = init_scores[0][2].copy()
            best_score = init_scores[0][0]
            print(f"  Updated best from init: {best_score:.6f}")

    # ===== Phase 2: Local optimization from top starts =====
    num_starts = min(8, len(init_scores)) if init_scores else 1
    if not init_scores:
        init_scores = [(best_score, 0, best.copy())]

    remaining_time = deadline - time.monotonic()
    time_per_start = remaining_time / max(1, num_starts)

    for start_idx in range(num_starts):
        if time.monotonic() > deadline - 30:
            break

        sc_init, _, init_seq = init_scores[start_idx]
        start_deadline = time.monotonic() + time_per_start - 5.0

        print(f"  Optimizing start {start_idx} (init score={sc_init:.6f}, "
              f"time budget={time_per_start:.0f}s)")

        result, result_score = _local_optimize(
            init_seq, n, start_deadline,
            K=30, temperature=10.0, max_iters=50000,
            label=f"S{start_idx}"
        )

        if np.isfinite(result_score) and result_score < best_score:
            best = result.copy()
            best_score = result_score
            print(f"  New global best: {best_score:.6f}")

    # ===== Phase 3: Refinement with different hyperparameters =====
    if time.monotonic() < deadline - 60:
        # Try different K and temperature values
        for K_val, temp_val in [(20, 5.0), (50, 20.0), (10, 3.0), (100, 50.0)]:
            if time.monotonic() > deadline - 30:
                break
            # Start from current best
            a_start = _project_simplex(best.copy())
            sc = float(evaluate_sequence(a_start.tolist()))
            remaining = deadline - time.monotonic() - 10
            if remaining < 20:
                break
            result, result_score = _local_optimize(
                a_start, n, time.monotonic() + remaining * 0.4,
                K=K_val, temperature=temp_val, max_iters=30000,
                label=f"T(K={K_val},T={temp_val})"
            )
            if np.isfinite(result_score) and result_score < best_score:
                best = result.copy()
                best_score = result_score
                print(f"  Refinement K={K_val},temp={temp_val}: {best_score:.6f}")

    # ===== Phase 4: Final fine-tuning with very small steps =====
    if time.monotonic() < deadline - 10:
        a = _project_simplex(best.copy())
        velocity = np.zeros(n)
        momentum = 0.5
        lr = 0.005
        K = 50
        for it in range(20000):
            if time.monotonic() > deadline:
                break
            conv = _fast_convolve(a)
            grad = _compute_grad_topk(a, conv, K, temperature=20.0)
            velocity = momentum * velocity - lr * grad
            velocity = np.clip(velocity, -2.0, 2.0)
            a = a * np.exp(velocity)
            a = np.clip(a, 1e-15, None)
            a = a / a.sum()
            if it % 500 == 0:
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = a.copy()
                    print(f"  Fine-tune iter {it}: {best_score:.6f}")

    # Final evaluation
    final_score = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"Final score: {best_score:.6f}")
    return [float(x) for x in best]
