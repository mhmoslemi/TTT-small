"""
Minimize  obj = 2 * n * max_k [x^k] A(x)^2  /  (sum a)^2   for a >= 0.

Because the evaluator uses np.convolve(a, a) (a *linear convolution*, i.e. the
coefficients of A(x)^2 with A(x) = sum a_i x^i), the scale-invariant objective is
    obj = 2 * n * max(c) / S^2,   c = ifft(FFT(a)^2),  S = sum(a).
We fix S = 1 (scale invariance) and minimise  max(c), keeping a on the
non-negative simplex.  We use a smoothed-subgradient (top-L lags) projected
FISTA-style descent, each step is O(n) + O(n log n) (FFT for c), so tens of
thousands of steps fit easily inside the time budget.  Several structural
seeds (the provided sequence, its reverse, ramps, concave/convex bumps) are
run in parallel to avoid local minima, and we periodically confirm candidates
with the authoritative evaluate_sequence().
"""

import time

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) - standard Giftis-O'Donnell algorithm.
# ----------------------------------------------------------------------
def _proj_simplex(v, target=1.0):
    """Project vector v (any real) onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    thr = 1.0 / np.size(v)
    lam = 0.0
    if target > 0:
        sv = np.sort(v)[::-1]
        cssv = np.cumsum(sv)
        rho = np.nonzero(sv * np.arange(1, np.size(v) + 1) > (cssv - target))[0][-1]
        lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _to_sum1(a):
    s = a.sum()
    if s <= 1e-12:
        n = a.size
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def _conv_sq(a, m):
    """Linear convolution a*a (coefficients of A(x)^2) via FFT, length m>=2n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[: (2 * a.size - 1)]


def _smoothed_grad_topL(c, a, L, tau):
    """Return grad of J = sum_k w_k c_k, w_k = softmax(log(c_k)/tau) over the
    top-L lags (smoothed max).  grad_a[p] = 2 * sum_k w_k a[k-p] (valid idx)."""
    n = a.size
    L = min(L, n)
    order = np.argpartition(-c, L - 1)[:L]
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-12))
    w = w / w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(order, w):
        if wk <= 0.0:
            continue
        # add 2*wk*a[k-p] for p where 0 <= k-p < n  -> p in [max(0,k-n+1), min(k,n-1)]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        src = k - idx
        grad[idx] += 2.0 * wk * a[src]
    return grad


def _local_step(a, c, m, lr, L, tau, momentum, buf):
    """One projected smoothed-subgradient step with heavy-ball momentum."""
    g = _smoothed_grad_topL(c, a, L, tau)
    # normalise gradient scale to keep step stable
    gn = np.linalg.norm(g)
    if gn > 0:
        g = g / gn
    buf = momentum * buf - lr * g
    a_new = a + lr * buf
    a_new = _proj_simplex(a_new, 1.0)
    return a_new, buf

def _candidate_seeds(base):
    n = base.size
    ones = np.ones(n, dtype=np.float64)
    ramp_up = np.linspace(0.05, 1.0, n)

    concave = np.sin(np.linspace(0.0, math.pi, n)) ** 0.8

    bell = np.exp(-0.5 * ((np.arange(n) - 0.6 * n) / (0.25 * n)) ** 2)
    bell2 = np.exp(-0.5 * ((np.arange(n) - 0.35 * n) / (0.2 * n)) ** 2)
    corrupt = base.copy()
    return [base, base[::-1].copy(), ones, ramp_up, ramp_dn, concave, convex,
