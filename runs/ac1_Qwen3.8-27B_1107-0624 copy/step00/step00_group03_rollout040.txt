import numpy as np
import time
import cvxpy as cp

def solve_lp_for_g(f, M):
    """Solve the LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0"""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix
    # conv(f, g)[k] = sum_{i} f[i] * g[k-i] for valid i, k-i
    # For each k, the constraint is: sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * g[j] <= M
    
    objective = cp.Maximize(cp.sum(g))
    constraints = []
    
    for k in range(2*n - 1):
        # conv(f,g)[k] = sum_j f[k-j] * g[j] where 0<=j<n and 0<=k-j<n
        # = sum_j g[j] * f[k-j]
        # Valid j: max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            coeffs = f[k - np.arange(j_start, j_end + 1)]  # f[k-j] for j in range
            constraints.append(coeffs @ g[j_start:j_end+1] <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)
    
    if g.value is not None:
        return np.maximum(0, g.value)
    return None

    num_constraints = 2 * n - 1
    g = cp.Variable(n, nonneg=True)
    
    # Build sparse constraint matrix A where A @ g <= M * ones
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0

    cols = []
    vals = []

        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= np.ones(num_constraints) * M]
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)
    
    if g.value is not None:
        return np.maximum(0, g.value)
    return None

import time
from scipy import optimize
from scipy.signal import fftconvolve

def compute_score(a):
    """Compute the objective value for a non-negative sequence."""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = np.sum(a)
    return 2.0 * n * max_b / (sum_a ** 2)

def lp_direction(a, M):
    """Find the LP direction: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    # Use scipy linprog
    # min -sum(g) s.t. A @ g <= M*1, g >= 0
    
    # Build A: A[k, j] = a[k-j] if valid
    # This is a banded matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.ones(2*n-1) * M
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

        t0 = time.monotonic()

def solve_lp(a, M):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build sparse matrix A where (A @ g)[k] = conv(a, g)[k]
    # A[k, j] = a[k-j] if 0 <= k-j < n and 0 <= j < n
    # j range for row k: max(0, k-n+1) to min(k, n-1)
    
    rows = np.empty(n * n, dtype=np.int32)  # upper bound on nnz
    cols = np.empty(n * n, dtype=np.int32)

    nnz = 0

        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows[nnz] = k
            cols[nnz] = j
            vals[nnz] = a[k - j]
            nnz += 1
