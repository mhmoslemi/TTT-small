```python
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    # C is the matrix where (C b)[k] = sum_j f[k-j] * b[j]
    # C[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # This is a banded matrix of size (2n-1) x n
    
    # For efficiency, build the sparse matrix
    # Actually, for scipy linprog, we can use the standard form
    
    # Constraint: C @ b <= M (vector of length 2n-1)
    # Objective: minimize -sum(b)
    
    # Build C as a sparse matrix
    # C[k, j] = f[k-j] for 0 <= k-j <= n-1
    # For each column j, the non-zero entries are at rows j, j+1, ..., j+n-1
    # C[j, j] = f[0]
    # C[j+1, j] = f[1]
    # ...
    # C[j+n-1, j] = f[n-1]
    
    from scipy.sparse import diags, identity, csr_matrix, lil_matrix
    
    # Build the convolution matrix
    # Column j has f[0], f[1], ..., f[n-1] at positions j, j+1, ..., j+n-1
    row_indices = []
    col_indices = []
    values = []
    
    for j in range(n):
        for i in range(n):
            k = i + j  # row index
            if k < 2*n - 1:
                row_indices.append(k)
                col_indices.append(j)
                values.append(f[i])
    
    C = csr_matrix((values, (row_indices, col_indices)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    res = linprog(c, A_ub=C, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
def smooth_max(x, k=100, temperature=1.0):
    """Smooth approximation to max using log-sum-exp of top-k."""
    # Sort and take top-k
    idx = np.argsort(x)[-k:]
    top_k = x[idx]
    return np.log(np.sum(np.exp((top_k - top_k.max()) / temperature))) * temperature + top_k.max()
```
