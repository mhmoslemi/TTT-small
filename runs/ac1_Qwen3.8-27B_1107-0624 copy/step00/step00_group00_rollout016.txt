"""
Optimize non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based acceleration, multi-start initialization with good priors
(cosine-modulated sequences, Zadoff-Chu-inspired), and simulated annealing
with CUSUM-adapted step sizes and peak-driven mass transfer.
"""

import numpy as np
import time


def _fft_conv(a, size=None):
    """Full linear convolution via FFT."""
    n = a.size
    size = size if size is not None else 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _score_fast(a):
    """Fast objective: 2*n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return (2.0 * a.size * _fft_conv(a).max()) / (s * s)


def _anneal(start, deadline, rng, salt):
    """Simulated annealing with adaptive step on a single sequence."""
    a = start.astype(np.float64).copy()
    n = a.size
    if n < 2:
        return a, np.inf
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.ones(n)
    a = a / a.sum()

    size = 1 << (2 * n - 1).bit_length()
    S = 1.0  # sum is 1
    cur = 2.0 * n * _fft_conv(a, size).max()
    best, best_s = a.copy(), cur
    T0, Tend, cool = 0.4, 0.003, 0.9997
    T = T0
    base = 0.05 + 0.25 * (n / 4806.0)
    cusum = 0.0
    stall = 0
    w = np.exp(-np.linspace(0, 3.5, n))
    w /= w.sum()
    ns = 0
    last_out = time.monotonic()
    while time.monotonic() < deadline:
        A = np.fft.rfft(a, size)
        rem = np.maximum(np.fft.irfft(A * A, size)[:2 * n - 1], 0.0)
        for k in range(7):
            if time.monotonic() > deadline:
                return best, best_s
            pk = int(np.argmax(rem))
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            sg = np.zeros(n)
            if hi >= lo:
                sg[lo:hi + 1] = 2.0 * a[pk - np.arange(lo, hi + 1)]
            d = sg / sg.sum() if sg.sum() > 1e-15 else w
            r = rng.dirichlet(np.full(n, 12.0))
            move = np.sign(d - r * d.sum() / r.sum()) * base
            delta = np.where(np.maximum(a + move, 0) > a, move, 0)
            D = delta.sum()
            if D <= 1e-15:
                delta = np.where(np.maximum(a - move, 0) < a, -move, 0)
                D = -delta.sum()
            if D <= 1e-15:
                continue
            a = a + delta
            na = np.maximum(a, 0)
            if na.sum() < 1e-12:
                a = np.ones(n) / n
            else:
                a = na / na.sum()
            ns += 1
            if ns % 24 == 0:
                A = np.fft.rfft(a, size)
                rem = np.maximum(np.fft.irfft(A * A, size)[:2 * n - 1], 0.0)
            cur = 2.0 * n * rem.max()
            if cur < best_s - 1e-9:
                best, best_s = a.copy(), cur
                stall = 0
                cusum = max(cusum * 0.5, 0.0)
            else:
                acc = 1.0 if cur <= cur + T else np.exp(-(cur - cur) / max(T, 1e-9))
                if rng.random() < acc and cur <= cur:
                    pass
                stall += 1
                cusum += 1.0
            if cusum > 40:
                base = min(base * 1.3, 0.5)
                cusum = 0
            if stall > 600:
                a = np.maximum(a + rng.normal(0, 0.15, n), 0)
                a = a / a.sum()
                stall = 0
            T = max(Tend, T * cool)
        if time.monotonic() - last_out > 20:
            print(f"  [{salt}] n={n} t={time.monotonic()%100:.0f} best={best_s:.6f}")
            last_out = time.monotonic()
    return best, best_s
