    """Compute the objective value."""

def gradient_descent_step(a, alpha_max=1.0, max_backtracks=10):
    """One gradient descent step with backtracking line search."""
    n = len(a)
    C = np.convolve(a, a)
    k_star = int(np.argmax(C))
    s = np.sum(a)
    C_max = C[k_star]
    current_score = 2.0 * n * C_max / (s * s)
    
    # Gradient of C[k_star] w.r.t. a[j]
    grad_C = np.zeros(n)
    valid = (k_star - (np.arange(n) - 0)) # k_star - j for j = 0..n-1
    # k_star - j is in [0, n) when j is in (k_star-n, k_star]
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):
        grad_C[j] = 2.0 * a[k_star - j]

    # Backtracking line search
    alpha = alpha_max
    for _ in range(max_backtracks):
        a_new = np.clip(a - alpha * grad_f, 0.0, 1000.0)
        new_score = compute_score(a_new)
        if new_score < current_score:
            return a_new, new_score
        alpha *= 0.5
    
    # If no improvement found, return original
    return a.copy(), current_score

def lp_step(a, M=None, t=0.1):
    """One LP-based step."""
    n = len(a)
    if M is None:
        C = np.convolve(a, a)
        M = float(np.max(C))
    
    # Normalize
    target_sum = np.sqrt(2.0 * n)
    a_norm = a * (target_sum / np.sum(a))
    
    # LP: max sum(b) s.t. (a_norm * b)[k] <= M, b >= 0
    # (a_norm * b)[k] = sum_i a_norm[i] * b[k-i]
    
    # Build constraint matrix
    # For each k = 0..2n-2: sum_{i: valid} a_norm[i] * b[k-i] <= M
    # b[j] appears in constraint k with coefficient a_norm[k-j] (if 0 <= k-j < n)
    
    # This is a (2n-1) x n matrix A where A[k, j] = a_norm[k-j] if 0 <= k-j < n
    # A is actually a Toeplitz-like matrix (convolution matrix)
    
    # Using scipy's convolve_matrix or building it manually
    # For efficiency, I'll use the sparse matrix representation
    
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for j in range(n):
        # b[j] appears in C[k] for k = j, j+1, ..., j+n-1 (as a_norm[k-j])
        # Wait, C[k] = sum_i a[i]*b[k-i]. b[j] appears when k-i=j, i.e., i=k-j.
        # So b[j] appears in C[k] with coefficient a[k-j] for k = j, j+1, ..., j+n-1
        # (need 0 <= k-j < n, i.e., j <= k < j+n, and 0 <= k < 2n-1)
        for k in range(j, min(j + n, 2*n-1)):
            A[k, j] = a_norm[k - j]
    
    A = A.tocsr()
    
    # Objective: maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    # Constraints: A @ b <= M
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.status != 0:
        return a.copy()
    
    b = res.x
    b = np.clip(b, 0, 1000.0)
    
    # Rescale
    b_sum = np.sum(b)
    if b_sum < 1e-10:
        return a.copy()
    b = b * (target_sum / b_sum)
    
    # Blend
    a_new = (1 - t) * a_norm + t * b
    # Convert back to original scale
    a_new = a_new * (np.sum(a) / target_sum)
    
    return np.clip(a_new, 0, 1000.0)

def _fast_score(a):

def _neg_subgradient(a):
    """Compute negative subgradient of the objective at a.
    
    Returns (neg_grad, C, s) where neg_grad is the direction to step.
    """
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return -np.ones(n), np.convolve(a, a), s
    C = np.convolve(a, a)
    C_max = float(np.max(C))
    
    # Find active lags (within 0.1% of max for subgradient averaging)
    threshold = C_max * 0.999
    active = np.where(C >= threshold)[0]
    
    grad_f = np.zeros(n, dtype=np.float64)
    n_active = len(active)
    
    for k in active:
        k = int(k)
        # dC[k]/da[j] = 2*a[k-j] if 0 <= k-j < n else 0
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:

            grad_C = np.zeros(n, dtype=np.float64)
            grad_C[js] = 2.0 * a[k - js]
            # df/da[j] = 2n * (grad_C[j] * s - 2*C_max) / s^3
            grad_f += (2.0 * n / (s ** 3)) * (grad_C * s - 2.0 * C_max)
    
    grad_f /= n_active
    return -grad_f, C, s


def _backtrack_step(a, neg_grad, current_score, max_iter=20):
    """Take a descent step with backtracking line search."""
    n = len(a)
    # Initial step size based on gradient scale
    grad_norm = np.linalg.norm(neg_grad)
    if grad_norm < 1e-12:
        return a.copy(), current_score
    
    # Scale step: we want to move roughly O(1) in relative terms
    a_scale = np.linalg.norm(a)
    if a_scale < 1e-10:
        a_scale = 1.0
    step = 1.0 / (grad_norm / a_scale + 1e-10)
    step = min(step, 2.0 * a_scale / (grad_norm + 1e-10))
    
    best_a = a.copy()
    best_score = current_score
    
    for _ in range(max_iter):
        a_new = a + step * neg_grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = np.sum(a_new)

            step *= 0.5
            continue
        new_score = _fast_score(a_new)
        if new_score < best_score:
            best_a = a_new.copy()
            best_score = new_score
        step *= 0.6
    
    return best_a, best_score


def _lp_direction(a, M):
    """Solve LP to find a direction that maximizes sum while keeping conv <= M.
    
    Returns the LP solution b (or None if infeasible).
    """
    n = len(a)
    
    # Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # Row k: A[k, j] = a[k-j] for valid j
    for j in range(n):
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    # LP: max sum(b) s.t. A@b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    
    # Use highs solver for speed
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    
    if res.status != 0:
        return None
    return res.x

    t = 2.0 * np.arange(n) / (n - 1) - 1.0

    final_gt = float(evaluate_sequence(verified_best_seq.tolist()))
    print(f"Final verified score: {final_gt:.6f}")
    print(f"Best fast score: {global_best_score:.6f}")
    
    return [float(x) for x in verified_best_seq]
