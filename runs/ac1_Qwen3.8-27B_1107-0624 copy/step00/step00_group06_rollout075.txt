import numpy as np
import time
import math
from scipy.optimize import linprog

"""
Minimizes 2*n*max(f*f)/(sum(f))^2 using multiplicative gradient descent
(exponentiated gradient method) with L2 and soft-max surrogate gradients,
LP-based Frank-Wolfe refinement, and local perturbation search.
Explores multiple sequence lengths and structural initializations.
"""


def autocorr_full(a):
    """Full linear autocorrelation of array a."""
    return np.convolve(a, a)


def objective_normalized(a):
    """Compute 2*n*max(f*f) for a sequence normalized to sum=1."""
    n = len(a)
    return 2.0 * n * autocorr_full(a).max()


def l2_gradient(a):
    """Gradient of sum_k R[k]^2 wrt a, where R = a*a (full conv).
    Returns vector of length n: 2 * correlate(R, a, valid)
    """
    R = np.convolve(a, a)
    return 2.0 * np.correlate(R, a, mode='valid')


def soft_peak_gradient(a, tau):
    """Gradient of soft-max surrogate log(sum exp(R/tau))."""
    R = np.convolve(a, a)
    R_shifted = R - R.max()
    w = np.exp(R_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(R) / len(R)
    else:
        w /= w_sum
    return (2.0 / tau) * np.correlate(w, a, mode='valid')


def run_mpgd(a_init, eta_init, max_iters, deadline, tau_mode, tau_val, verbose, tag):
    """
    Multiplicative gradient descent to minimize max autocorrelation.
    tau_mode: 'l2' or 'soft'
    Returns (best_array_normalized, best_score).
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, None)
    s = a.sum()
    if s < 1e-20:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = objective_normalized(a)
    eta = eta_init
    patience = 0
    max_patience = 200

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute gradient
        R = np.convolve(a, a)
        max_R = R.max()
        scale = max_R if max_R > 1e-20 else 1.0

        if tau_mode == 'l2':
            grad = 2.0 * np.correlate(R, a, mode='valid')
        else:
            R_shifted = R - R.max()
            w = np.exp(R_shifted / tau_val)
            ws = w.sum()
            if ws < 1e-300:
                w = np.ones_like(R) / len(R)
            else:
                w /= ws
            grad = (2.0 / tau_val) * np.correlate(w, a, mode='valid')

        # Multiplicative update
        exponent = -eta * grad / scale
        # Clip to prevent underflow
        exponent = np.clip(exponent, -10.0, 10.0)
        a = a * np.exp(exponent)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
            break
        a = a / s

        score = objective_normalized(a)
        if score < best_score - 1e-13:
            best_a = a.copy()
            best_score = score
            patience = 0
            if verbose and it % 500 == 0:
                print(f"    {tag} mpGD iter {it}: score={score:.8f}")
        else:
            patience += 1

        # Learning rate decay
        if patience % 25 == 0 and patience > 0:
            eta *= 0.94
        if patience >= max_patience:
            # Try a restart with fresh noise
            rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = best_a * (1.0 + 0.05 * rng_local.normal(0, 1, n))
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 1e-20:
                a = best_a.copy()
            else:
                a = a / s
            eta = eta_init
            patience = 0
            if patience > max_patience:
                break

    return best_a, best_score


def solve_lp_dir(f, M):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = f[:k_end - k_start]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.status == 0:
        return res.x
    return None


def run_lp_refinement(a_init, max_iters, deadline, verbose, tag):
    """Frank-Wolfe with LP direction to minimize max autocorrelation."""
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, None)
    s = a.sum()
    if s < 1e-20:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = objective_normalized(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        a_work = a * target_sum
        R = np.convolve(a_work, a_work)
        M = R.max()

        try:
            g = solve_lp_dir(a_work, M)
        except Exception:
            break

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        gamma = 2.0 / (it + 2.0)
        a_new = (1.0 - gamma) * a + gamma * g
        a_new = np.clip(a_new, 0.0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score = objective_normalized(a_new)
        if score < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = score
            if verbose and it % 10 == 0:
                print(f"    {tag} LP iter {it}: score={score:.8f}")
        a = a_new

    return best_a, best_score


def local_perturb_search(a_init, deadline, seed, verbose, tag):
    """Local search with random multiplicative block perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, None)
    s = a.sum()
    if s < 1e-20:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = objective_normalized(a)
    sigma = 0.15
    no_improve = 0

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        width = max(1, n // (80 + int(rng.integers(0, 400))))
        start = int(rng.integers(0, max(1, n - width + 1)))
        noise = rng.normal(0.0, sigma, size=width)
        a_new = a.copy()
        a_new[start:start + width] *= np.exp(noise)
        a_new = np.clip(a_new, 0.0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        score = objective_normalized(a_new)
        if score < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = score
            a = a_new
            no_improve = 0
            if verbose and it % 3000 == 0:
                print(f"    {tag} perturb iter {it}: score={score:.8f}")
        else:
            no_improve += 1

        if no_improve % 300 == 0 and no_improve > 0:
            sigma *= 0.95
        if no_improve > 5000:
            break

    return best_a, best_score


def make_initializations(parent, rng):
    """Generate a list of (array, label) starting points."""
    inits = []

    if parent is not None:
        inits.append((parent.copy(), "parent"))
        pc = parent.size
        for n in [128, 256, 512, 1024, 2048, 4096]:
            if n != pc and n <= 8192:
                x_old = np.linspace(0.0, 1.0, pc)
                x_new = np.linspace(0.0, 1.0, n)
                resampled = np.interp(x_new, x_old, parent)
                inits.append((np.clip(resampled, 0.0, None), f"resamp_{n}"))

    for n in [64, 128, 256, 512, 1024, 2048, 4096]:
        inits.append((np.ones(n), f"uniform_{n}"))
        inits.append((rng.random(n) + 0.01, f"rand_{n}"))
        t = np.linspace(0.0, 1.0, n)
        inits.append((np.sin(np.pi * t), f"sin_{n}"))
        gauss = np.exp(-4.0 * (t - 0.5) ** 2)
        inits.append((gauss, f"gauss_{n}"))
        if n >= 128:
            u_shape = 1.0 + 3.0 * (2.0 * t - 1.0) ** 2
            inits.append((u_shape, f"ushape_{n}"))

    return inits
