"""
LP-guided local search for minimizing the autocorrelation peak ratio.
Uses iterative linear programming to find descent directions, combined
with line search and multiple restarts for robustness.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_score_fast(s):
    """Compute the evaluation score using FFT for speed."""
    n = len(s)
    size = 2 * n - 1
    f_s = np.fft.rfft(s, n=size)
    conv = np.fft.irfft(f_s * f_s, n=size)
    conv = np.maximum(conv, 0.0)
    max_b = np.max(conv)
    sum_a = np.sum(s)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_lp_structure(n):
    """Precompute the sparse structure of the LP constraint matrix."""
    row_list = []
    col_list = []
    s_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_list.append(k)
            col_list.append(j)
            s_list.append(k - j)
    return np.asarray(row_list, dtype=np.int32), np.asarray(col_list, dtype=np.int32), np.asarray(s_list, dtype=np.int32)

def run_lp_optimization(s, deadline, rng, verbose=True):

    n = len(s)
    row_idx, col_idx, s_idx = build_lp_structure(n)
    
    best_s = s.copy()
    best_score = compute_score_fast(s)
    
    step_sizes = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    iteration = 0
    no_improve = 0
    max_no_improve = 20
    
    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        
        # Compute autocorrelation
        size = 2 * n - 1
        f_s = np.fft.rfft(s, n=size)
        conv = np.fft.irfft(f_s * f_s, n=size)
        conv = np.maximum(conv, 0.0)
        M = float(np.max(conv))
        
        if M < 1e-15:
            break
        
        # Solve LP
        vals = s[s_idx]
        A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not result.success or result.x is None:
            if verbose:
                print(f"  LP failed at iter {iteration}, stopping.")
            break
        
        b = np.maximum(result.x, 0.0)
        sum_b = float(np.sum(b))
        
        if sum_b <= 0:
            break
        
        # Line search
        best_t = step_sizes[0]
        best_trial = np.inf
        for t in step_sizes:
            s_trial = (1.0 - t) * s + t * b
            s_trial = np.clip(s_trial, 0.0, 1000.0)

            sc = compute_score_fast(s_trial)
            if sc < best_trial:
                best_trial = sc
                best_t = t
        
        if not np.isfinite(best_trial):
            break
        
        s = (1.0 - best_t) * s + best_t * b
        s = np.clip(s, 0.0, 1000.0)

```python
"""
LP-guided local search minimizing the autocorrelation peak ratio of a step function.

Algorithm:
1. Start from an initial non-negative sequence.
2. Iteratively solve an LP to find a direction that maximizes the sum
   while keeping the cross-correlation bounded by the current peak.
3. Use line search to determine the step size.
4. Repeat until time budget is exhausted.
5. Use multiple initializations and noise injection to avoid local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score_fast(s):
    """Compute the evaluation score using FFT for speed."""
    n = len(s)
    size = 2 * n - 1
    f_s = np.fft.rfft(s, n=size)
    conv = np.fft.irfft(f_s * f_s, n=size)
    conv = np.maximum(conv, 0.0)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(s))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_lp_structure(n):
    """Precompute sparse structure for the LP constraint matrix."""
    row_list = []
    col_list = []
    s_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_list.append(k)
            col_list.append(j)
            s_list.append(k - j)
    return (np.asarray(row_list, dtype=np.int32),
            np.asarray(col_list, dtype=np.int32),
            np.asarray(s_list, dtype=np.int32))


def solve_direction(s, row_idx, col_idx, s_idx, time_limit=60):
    """Solve the LP to find the optimal descent direction."""
    n = len(s)
    
    # Compute current peak
    size = 2 * n - 1
    f_s = np.fft.rfft(s, n=size)
    conv = np.fft.irfft(f_s * f_s, n=size)
    conv = np.maximum(conv, 0.0)
    M = float(np.max(conv))
    
    if M < 1e-15:
        return None, M
    
    # Build constraint matrix
    vals = s[s_idx]
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # Solve: max sum(b) s.t. (s*b)[k] <= M, b >= 0
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None, M
    
    if result.x is not None and result.x.sum() > 0:
        return np.maximum(result.x, 0.0), M
    
    return None, M


def line_search(s, direction, step_candidates):
    """Find the best step size from candidates."""
    best_t = step_candidates[0]
    best_score = compute_score_fast(s)
    
    for t in step_candidates:
        s_trial = (1.0 - t) * s + t * direction
        s_trial = np.clip(s_trial, 0.0, 1000.0)
        if np.sum(s_trial) < 0.01:
            continue
        sc = compute_score_fast(s_trial)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    return best_t, best_score


def run_optimization(s_init, deadline, rng, label="", verbose=True):
    """Run LP-based local search from initial sequence."""
    n = len(s_init)
    s = s_init.copy()
    
    row_idx, col_idx, s_idx = build_lp_structure(n)
    
    best_s = s.copy()
    best_score = compute_score_fast(s)
    
    if verbose:
        print(f"  [{label}] n={n}, initial score={best_score:.6f}")
    
    normal_steps = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    fine_steps = [0.05, 0.1, 0.15, 0.2, 0.3]
    
    iteration = 0
    no_improve = 0
    max_no_improve = 15
    
    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        
        # Get direction from LP
        direction, M = solve_direction(s, row_idx, col_idx, s_idx, time_limit=60)
        
        if direction is None:
            # Fallback: use a heuristic direction
            # Push down the values at the peak lag
            size = 2 * n - 1
            f_s = np.fft.rfft(s, n=size)
            conv = np.fft.irfft(f_s * f_s, n=size)
            conv = np.maximum(conv, 0.0)
            k_star = int(np.argmax(conv))
            # Gradient of conv[k_star] w.r.t. s
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = s[j]
            # Move away from the gradient
            direction = s - 0.1 * grad
            direction = np.maximum(direction, 0.0)
        
        sum_dir = float(np.sum(direction))
        if sum_dir <= 0:
            break
        
        # Line search (use finer steps as we converge)
        if iteration < 20:
            steps = normal_steps
        else:
            steps = fine_steps
        
        best_t, trial_score = line_search(s, direction, steps)
        
        if trial_score >= compute_score_fast(s) + 1e-10:
            # No improvement, add noise
            no_improve += 1
            if no_improve >= max_no_improve:
                s = s * np.exp(rng.normal(0, 0.03, n))
                s = np.clip(s, 0.0, 1000.0)
                no_improve = 0
                if verbose:
                    print(f"  [{label}] Iter {iteration}: noise added")
            continue
        
        s = (1.0 - best_t) * s + best_t * direction
        s = np.clip(s, 0.0, 1000.0)
        
        score = trial_score
        elapsed = time.monotonic() - t0
        
        if score < best_score - 1e-9:
            best_score = score
            best_s = s.copy()
            no_improve = 0
            if verbose and (iteration <= 5 or iteration % 20 == 0 or best_score < 1.6):
                print(f"  [{label}] Iter {iteration}: score={best_score:.6f}, t={best_t:.2f}, {elapsed:.1f}s")
    
    if verbose:
        print(f"  [{label}] Done: {iteration} iters, best={best_score:.6f}")
    
    return best_s, best_score
