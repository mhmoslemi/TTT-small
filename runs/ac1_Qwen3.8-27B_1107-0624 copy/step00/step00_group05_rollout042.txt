Let me just use a simple iterative approach:

**Proximal method**:
1. Start with a0 = step function, normalized to sum = 1
2. At iteration t:
   a. Compute M_t = max(a_t * a_t)
   b. Identify active set: K_t = {k : (a_t*a_t)_k >= M_t - epsilon}
   c. Compute subgradient: g_t = (1/|K_t|) * sum_{k in K_t} grad_k where grad_k[j] = 2*a_{k-j} (if valid)
   d. Update: a_{t+1} = projection of (a_t - alpha_t * g_t) onto {a >= 0, sum(a) = 1}
   
The projection onto {a >= 0, sum(a) = 1} is the "water-filling" operation: subtract a constant from all elements, then clip at 0, and adjust the constant so the sum is 1.

This is simple and efficient! Each iteration is O(n) for the gradient computation (if we precompute the convolution) + O(n) for the projection.

Wait, the gradient computation: grad_k[j] = 2*a_{k-j} for j in [k-n+1, k] (clipped to [0, n-1]). For |K_t| active constraints, the total gradient computation is O(|K_t| * n). If |K_t| is small (like 10-100), this is fast.

But actually, I can compute the full gradient more efficiently. The gradient of F(a) = max_k (a*a)_k at a point where the max is achieved at k_1, ..., k_m (active set) is:
g = (1/m) * sum_{i=1}^m grad_{k_i}

where grad_{k_i}[j] = 2*a_{k_i - j} (if 0 <= k_i - j < n, else 0).

So g[j] = (1/m) * sum_{i: 0 <= k_i - j < n} 2*a_{k_i - j}

This can be computed as: for each active k_i, add 2*a reversed (shifted) to g. In total, O(m*n) operations.

For m = 10 and n = 4498, that's 45K operations per gradient step. Very fast.

The projection (water-filling) is O(n) with a sort or O(n log n).

So each iteration is O(m*n + n log n) ≈ O(n) for small m.

            best = a.copy()

        epsilon = 0.01 * M
        active = np.where(conv >= M - epsilon)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            # grad_k[j] = 2*a[k-j] if 0 <= k-j < n
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            g[j_min:j_max+1] += 2 * a[k - j_max:k - j_min + 1][::-1]

        if t > 1:
            beta = (t - 1) / (t + 2)
            y = a + beta * (a - a_prev)
        else:
            y = a
            beta = 0

        alpha_t = alpha / np.sqrt(t)
        a_new = y - alpha_t * g
        
        # Project onto {a >= 0, sum(a) = 1}
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        t += 1
    
    return best, best_M

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Water filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.clip(v - theta, 0, None)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using:
1. Optimal 2-level step initialization (objective = 27/16)
2. Subgradient descent with simplex projection
3. Window-level coordinate descent (O(n log n) per sweep)
4. LP-based global moves for escaping local minima
5. Adaptive local search refinement
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if v.sum() <= target_sum:
        # All non-negative, just scale up
        if v.sum() < 1e-15:
            return np.full(n, target_sum / n)
        return v * (target_sum / v.sum())
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_active_set(conv, M, frac=0.05, max_count=200):
    """Get indices where conv is near the max."""
    threshold = M * (1.0 - frac)
    active = np.where(conv >= threshold)[0]
    if len(active) > max_count:
        # Take the top max_count
        idx = np.argsort(conv)[::-1][:max_count]
        active = idx
    return active


def subgradient_descent(a, S_target, n, n_iter=5000, tol=1e-7):
    """Run subgradient descent to minimize max(a*a) given sum(a)=S_target."""
    conv = np.convolve(a, a)
    M_best = conv.max()
    a_best = a.copy()
    alpha = S_target * 0.1
    a_prev = a.copy()
    
    for t in range(n_iter):
        conv = np.convolve(a, a)
        M = conv.max()
        
        if M < M_best:
            M_best = M
            a_best = a.copy()
            if M_best < M_best_old * (1 - tol) if t > 0 else True:
                M_best_old = M_best
        
        # Active set
        eps = max(1e-6 * M, M_best * 1e-4)
        threshold = M - eps
        active = np.where(conv >= threshold)[0]
        if len(active) > 300:
            active = np.argsort(conv)[::-1][:300]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            idx = np.arange(j_min, j_max + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Nesterov momentum
        if t > 0:
            beta = (t - 1) / (t + 2)
            y = a + beta * (a - a_prev)
        else:
            y = a
        
        # Adaptive step size
        alpha_t = alpha / np.sqrt(t + 1)
        a_new = y - alpha_t * g
        a_new = project_simplex(a_new, S_target)
        
        a_prev = a
        a = a_new
    
    return a_best, M_best


def window_cd(a, S_target, n, n_sweeps=50, window_size=64):
    """Window-level coordinate descent."""
    best_conv = np.convolve(a, a)
    best_M = best_conv.max()
    rng = np.random.default_rng(12345)
    
    for sweep in range(n_sweeps):
        improved = False
        # Process windows in random order
        n_windows = n // window_size
        for _ in range(n_windows):
            w_start = int(rng.integers(0, n_windows))
            w_end = min(w_start + 1, n_windows)
            # Actually pick a random window position
            start = int(rng.integers(0, max(1, n - window_size)))
            end = min(start + window_size, n)
            
            # Current contribution from this window
            # Try scaling the window
            for scale in [0.85, 0.92, 1.08, 1.15]:
                a_test = a.copy()
                a_test[start:end] *= scale
                a_test = project_simplex(a_test, S_target)
                conv_test = np.convolve(a_test, a_test)
                M_test = conv_test.max()
                if M_test < best_M * (1 - 1e-8):
                    a = a_test
                    best_conv = conv_test
                    best_M = M_test
                    improved = True
                    break
        
        if not improved and sweep > 10:
            break
    
    return a, best_M


def try_lp_move(a, S_target, n, M_current, max_time=30.0):
    """Try an LP-based move. Returns improved sequence or None."""
    t0 = time.monotonic()
    
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    # Only include constraints where (a*a)_k is near max
    conv_aa = np.convolve(a, a)
    threshold = M_current * 0.9
    active_k = np.where(conv_aa >= threshold)[0]
    
    if len(active_k) == 0:
        return None
    
    # Limit to at most 5000 constraints for speed
    if len(active_k) > 5000:
        active_k = np.argsort(conv_aa)[::-1][:5000]
    
    n_constraints = len(active_k)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for ci, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(ci)
            cols.append(j)
            vals.append(float(a[k - j]))
    
    if not rows:
        return None
    
    A = np.array(rows, dtype=np.int32)
    B = np.array(cols, dtype=np.int32)
    C = np.array(vals, dtype=np.float64)
    
    import scipy.sparse as sp
    A_sparse = sp.coo_matrix((C, (A, B)), shape=(n_constraints, n)).tocsr()
    
    # LP: maximize sum(g) subject to A_sparse @ g <= M_current, 0 <= g <= S_target/n * 3
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0.0, S_target / n * 3.0)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=np.full(n_constraints, M_current),
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
    except Exception:
        return None
    
    if time.monotonic() - t0 > max_time:
        return None
    
    if res.status != 0:
        return None
    
    g = res.x
    if g is None or not np.all(np.isfinite(g)):
        return None
    
    sum_g = g.sum()
    if sum_g <= S_target * 1.01:
        return None
    
    # Normalize g to have target sum
    g = g * (S_target / sum_g)
    
    # Check if this direction actually helps
    # The LP ensures (a@g)_k <= M for active k
    # Let's try a blend
    for t_mix in [0.3, 0.5, 0.7]:
        a_try = (1 - t_mix) * a + t_mix * g
        a_try = project_simplex(a_try, S_target)
        conv_try = np.convolve(a_try, a_try)
        M_try = conv_try.max()
        if M_try < M_current * 0.999:
            return a_try
    
    return None


def local_search_perturb(a, S_target, n, rng, n_trials=200):
    """Adaptive local search with structured perturbations."""
    best = a.copy()
    conv_best = np.convolve(best, best)
    M_best = conv_best.max()
    
    for trial in range(n_trials):
        candidate = best.copy()
        op = rng.random()
        
        if op < 0.3:
            # Smooth sinusoidal perturbation
            freq = rng.uniform(0.5, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.1) * best.mean()
            x = np.arange(n)
            candidate += amp * np.sin(2 * np.pi * freq * x / n + phase)
        elif op < 0.5:
            # Block scaling
            width = int(rng.integers(max(10, n // 100), max(20, n // 8)))
            start = int(rng.integers(0, max(1, n - width)))
            scale = np.exp(rng.normal(0, 0.1))
            candidate[start:start + width] *= scale
        elif op < 0.7:
            # Gaussian noise
            sigma = rng.uniform(0.005, 0.05) * best.mean()
            candidate += rng.normal(0, sigma, n)
        elif op < 0.85:
            # Two-block swap (move mass)
            w1 = int(rng.integers(5, n // 20))
            w2 = w1
            s1 = int(rng.integers(0, max(1, n - w1)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            if s1 != s2:
                amnt = candidate[s1:s1 + w1].sum() * 0.1
                candidate[s1:s1 + w1] -= amnt / w1
                candidate[s2:s2 + w2] += amnt / w2
        else:
            # Taper ends (reduce boundary peaks)
            taper_len = int(rng.integers(n // 50, n // 10))
            decay = np.linspace(1.0, 0.5, taper_len)
            candidate[:taper_len] *= decay
            if rng.random() < 0.5:
                candidate[-taper_len:] *= decay[::-1]
        
        # Project
        candidate = np.maximum(candidate, 0)
        candidate = project_simplex(candidate, S_target)
        
        # Evaluate
        conv_c = np.convolve(candidate, candidate)
        M_c = conv_c.max()
        
        if M_c < M_best * (1 - 1e-9):
            best = candidate
            conv_best = conv_c
            M_best = M_c
    
    return best, M_best

        a_cd, M_cd = window_cd(best, S_target, n, n_sweeps=100)
        if M_cd < best_M:
            best = a_cd
            best_M = M_cd
            best_score = 2.0 * n * best_M / (S_target ** 2)
        elapsed = time.monotonic() - t_start
        if int(elapsed) % 30 < 2:
            print(f"[t={elapsed:.0f}s] CD: M={best_M:.6f}, score={best_score:.6f}")
        # Check deadline
        if time.monotonic() >= deadline - 600:
            break
    
    # ===== Phase 4: LP-based global moves =====
    t_start = time.monotonic()
    lp_moved = False
    while time.monotonic() < deadline - 300:
        # Try LP move
        a_lp = try_lp_move(best, S_target, n, best_M, max_time=20.0)
        if a_lp is not None:
            conv_lp = np.convolve(a_lp, a_lp)
            M_lp = conv_lp.max()
            if M_lp < best_M:
                best = a_lp
                best_M = M_lp
                best_score = 2.0 * n * best_M / (S_target ** 2)
                lp_moved = True
                print(f"[t={time.monotonic()-t_start:.0f}s] LP move: M={best_M:.6f}, score={best_score:.6f}")
        
        # Refine with local search after LP
        if time.monotonic() < deadline - 300:
            a_ls, M_ls = local_search_perturb(best, S_target, n, rng, n_trials=200)
            if M_ls < best_M:
                best = a_ls
                best_M = M_ls
                best_score = 2.0 * n * best_M / (S_target ** 2)
        
        # Additional subgradient refinement
        if time.monotonic() < deadline - 300:
            a_sg2, M_sg2 = subgradient_descent(best, S_target, n, n_iter=3000)
            if M_sg2 < best_M:
                best = a_sg2
                best_M = M_sg2
                best_score = 2.0 * n * best_M / (S_target ** 2)
    
    # ===== Phase 5: Final local search polish =====
    while time.monotonic() < deadline - 10:
        a_ls, M_ls = local_search_perturb(best, S_target, n, rng, n_trials=500)
        if M_ls < best_M:
            best = a_ls
            best_M = M_ls
            best_score = 2.0 * n * best_M / (S_target ** 2)
        
        # Quick subgradient
        if time.monotonic() < deadline - 10:
            a_sg3, M_sg3 = subgradient_descent(best, S_target, n, n_iter=2000)
            if M_sg3 < best_M:
                best = a_sg3
                best_M = M_sg3
                best_score = 2.0 * n * best_M / (S_target ** 2)
    
    # Final evaluation
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"[FINAL] Score={final_score:.6f}, M={best_M:.6f}")
    
    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    return result
