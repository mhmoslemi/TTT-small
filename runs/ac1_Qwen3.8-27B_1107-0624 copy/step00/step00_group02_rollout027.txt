def project_simplex(v, s=1.0):
    """Project v onto the simplex {x : sum(x) = s, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)

def _compute_conv(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    fa = fft(a, size)
    result = ifft(fa * fa).real[:2*n-1]
    return result
