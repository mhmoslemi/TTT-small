    delta = 1e-4  # Step size for finite differences
    
    for it in range(max_iterations):
        if it % 10 == 0:
            # Check convergence and fallback if stuck
            finite_differences = np.zeros(n)
            for i in range(n):
                a_plus = best_sequence.copy()
                a_plus[i] += delta
                a_plus /= np.sum(a_plus)
                obj_plus = evaluate_sequence(a_plus.tolist())
                
                a_minus = best_sequence.copy()
                a_minus[i] -= delta
                a_minus = np.maximum(a_minus, 0.0)
                a_minus /= np.sum(a_minus)
                obj_minus = evaluate_sequence(a_minus.tolist())
                
                # Compute finite-difference gradient at normalized level
                grad = (obj_plus - obj_minus) / (2 * delta)
                finite_differences[i] = grad
            
            # Avoid numerical explosion by gradient clipping
            norm = np.linalg.norm(finite_differences)
            if norm < 1e-7:
                # Convergence check: minimal gradient means stabilizing
                break
            finite_differences = finite_differences / norm
            
            # Optionally: consider flipping some velocity signs randomly to escape local minimum?
            # But we must ensure improvement, so we use gradient as primary guide.
            
            velocity[0] = 0.9 * velocity[0] + 0.2 * finite_differences
                
        # Update all elements simultaneously (stochastic gradient)
        candidate = best_sequence.copy()
        for i in range(n):
            candidate[i] += velocity[i]
            # Renormalized once to maintain unit-sum until next gradient computation
            if abs(candidate[i]) > 0.0001:
                candidate /= np.sum(candidate)
            else:
                # Avoid division by zero during updates
                candidate = candidate / np.sum(candidate) * 1.0001  # Adjust for epsilon increment
        
        # Renormalize after update
        candidate_sum = np.sum(candidate)
        candidate = candidate / candidate_sum
        
        # Clip values so that they are in [0,1000]
        clipped_candidate = np.clip(candidate, 0.0, 1000.0)
        clipped_candidate_sum = np.sum(clipped_candidate)
        # Scale back to avoid mass discrepancy
        clipped_candidate = clipped_candidate / clipped_candidate_sum
        
        # Safety: Ensure non-infinity unless sum is too small
        obj_val = evaluate_sequence(clipped_candidate.tolist())
        if np.isinf(obj_val):
            # Fallback: minimize by setting to uniform if not possible
            uniform = np.ones(n)
            uniform /= np.sum(uniform)
            clipped_uniform = np.clip(uniform, 0.0, 1000.0) / np.sum(clipped_uniform)
            obj_val_uniform = evaluate_sequence(clipped_uniform.tolist())
            if obj_val_uniform < np.isfinite(evaluate_sequence(best_sequence.tolist())):  # Check fallback improveness
                best_sequence = clipped_uniform.copy()
            break
        
        if np.isfinite(obj_val) and obj_val < evaluate_sequence(best_sequence.tolist()):
            best_sequence = clipped_candidate.copy()
    
    return best_sequence.tolist()
