Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 via gradient descent on the simplex.
Starts from a linear ramp (objective ~4/3 < target 1.5027) and refines using
FFT-based autocorrelation with top-peak gradient attack and adaptive learning rate.

import numpy as np
import time


def compute_gradient_multi_peak(p, q, top_m):
    """Compute combined subgradient from top-m peaks of autocorrelation q."""
    n = len(p)
    if top_m >= len(q):
        top_indices = np.argsort(q)[::-1]
    else:
        top_indices = np.argpartition(q, -top_m)[-top_m:]
        top_indices = top_indices[np.argsort(q[top_indices])[::-1]]
    g = np.zeros(n)
    for k in top_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        p_idx = k - np.arange(j_start, j_end + 1)
        g[j_start:j_end + 1] += 2.0 * p[p_idx]
    return g


def fast_autocorr(p):
    """Compute full linear autocorrelation using zero-padded FFT."""
    n = len(p)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(p, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]


def make_linear_ramp(n):
    """Create normalized linearly-ramping probability vector (ascending)."""
    p = np.arange(1, n + 1, dtype=np.float64)
    p /= p.sum()
    return p


def make_random_prob(rng, n, spread=1.0):
    """Create a random probability vector from log-normal distribution."""
    p = np.exp(rng.normal(0, spread, n))
    p = np.clip(p, 1e-15, None)
    p /= p.sum()
    return p
