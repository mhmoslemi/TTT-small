g[i] = a[k*-i] if 0 ≤ k*-i ≤ n-1, else 0

To minimize b[k*], we take a gradient descent step:
a ← a - η · g

Then project: a = max(0, a - λ · 1) where λ is chosen so that sum(a) = target_sum.

    a = a.copy()
    target_sum = a.sum()
    
    for _ in range(max_iter):
        b = np.convolve(a, a)
        k_star = np.argmax(b)
        
        # Gradient of b[k_star] w.r.t. a
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            g[i] = a[k_star - i]

import numpy as np

def project_onto_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    current_sum = v.sum()
    if current_sum <= target_sum:
        # Need to add mass
        # Add proportionally to existing values, or uniformly
        if current_sum > 0:
            v = v * target_sum / current_sum
        else:
            v = np.full_like(v, target_sum / len(v))
    else:
        # Need to remove mass - use water-filling
        # Sort in descending order
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, len(v) + 1) > (cssv - target_sum))[0]
        if len(rho) > 0:
            rho_max = rho[-1]
            lambda_ = (cssv[rho_max] - target_sum) / (rho_max + 1)
            v = np.maximum(v - lambda_, 0)
        else:
            v = np.maximum(v - (current_sum - target_sum) / len(v), 0)
    return v

def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u)
    # Find rho: largest rho such that u[rho-1] > (cssv[rho-1] - s) / rho
    rho_candidates = np.where(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        # All values are below the water level
        return np.full(n, s / n)
    rho = rho_candidates[-1] + 1  # 1-indexed
    lam = (cssv[rho-1] - s) / rho
    return np.maximum(v - lam, 0)
