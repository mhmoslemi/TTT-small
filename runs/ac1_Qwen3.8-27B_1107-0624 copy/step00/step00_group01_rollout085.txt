    for k in top_k:

       lo=max(0, k - n + 1)
       hi=min(n-1, k)
       vals = a[k-lo:k-hi+1][::-1]

```python
import numpy as np
import time

"""
Algorithm summary:

The score is
    2 * n * max(a * a) / (sum(a))**2
for a >= 0.  A useful large-n interpretation is that, after normalizing so that
sum(a) ~ n, the quantity
    n * max(a * a) / (sum(a))**2
converges to the maximum of the continuous self-convolution of a density f on
[0, 1].  The density
    f(x) = 1 - cos(pi x)
is nonnegative, has integral 1, and gives a continuous self-convolution maximum
about 0.665, hence asymptotic score about 1.331, which is already well below
the previous 2.0 bound.

This program:
 1. Uses the discrete sequence
        a_i = 1 - cos(pi * i/(n-1)),  i = 0,...,n-1
    with n = 4096.
 2. Explores a few nearby cosine-shape variants and keeps the best one by FFT
    convolution score (and by the official evaluator when promising).
 3. Runs a bounded local search using FFT convolution scores to try to improve
    the discrete sequence further.
 4. Returns the best sequence whose official score was verified if needed.
"""


def _pow2_ge(x):
    """Smallest power of two >= x."""
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _normalize(a, n):
    """Scale to sum n and keep values in the valid range."""
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return a
    a = a * (float(n) / s)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _candidate_cosine(n, exponent=1.0, amplitude=1.0, shift=0.0):
    """Generate a smooth nonnegative cosine-based candidate of length n."""
    n = int(max(2, n))
    m = n - 1
    t = np.linspace(0.0, 1.0, n)
    arg = np.pi * (t + float(shift))
    base = 1.0 - float(amplitude) * np.cos(arg)
    base = np.clip(base, 0.0, None)

    if abs(float(exponent) - 1.0) > 1e-12:
        base = np.power(base, float(exponent))

    if np.sum(base) <= 0.0:
        base = np.ones(n, dtype=np.float64)

    return base.astype(np.float64)


def _fft_score(a, size):
    """
    Evaluate the official expression using FFT convolution.

    Returns (score, convolution, peak).  The convolution is the full linear
    convolution truncated to the first 2*n-1 values.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(np.sum(a))
    if n == 0 or s <= 1e-8 or not np.isfinite(s):
        return np.inf, None, 0.0

    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    # Numerical noise can create tiny negative entries; the true convolution
    # is nonnegative.
    peak = float(np.max(conv))
    if peak < 0.0:
        peak = 0.0

    score = 2.0 * float(n) * peak / (s * s)
    if not np.isfinite(score):
        return np.inf, conv, peak
    return score, conv, peak


def _official_score(seq):
    """Call the provided official evaluator safely."""
    try:
        val = evaluate_sequence(seq)
        return float(val)
    except Exception:
        return np.inf


def _random_perturb(a, rng, sigma, n):
    """Small multiplicative block perturbation plus optional tiny lift."""
    n = int(a.size)
    cand = np.array(a, dtype=np.float64, copy=True)

    nblocks = 1
    if rng.random() < 0.35:
        nblocks = 2

    low_w = max(4, n // 1024)
    high_w = max(low_w + 2, n // 8)

    for _ in range(nblocks):
        w = int(rng.integers(low_w, high_w))
        start_hi = max(1, n - w + 1)
        start = int(rng.integers(0, start_hi))
        stop = min(n, start + w)
        factor = float(np.exp(rng.normal(0.0, float(sigma))))
        cand[start:stop] *= factor

    if rng.random() < 0.20:
        cand += 0.0015 * rng.random(n)

    np.clip(cand, 0.0, 1000.0, out=cand)
    return _normalize(cand, n)


def _peak_shave(a, conv, alpha):
    """
    Heuristic subgradient-like update: reduce components that participate in
    the largest convolution lags, while preserving the mean level.
    """
    n = int(a.size)
    if conv is None or conv.size == 0:
        return np.array(a, dtype=np.float64, copy=True)

    # A small number of the largest lags is enough and keeps the cost low.
    top = np.argsort(conv)[-16:]
    grad = np.zeros(n, dtype=np.float64)

    for k in top:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - lo:k - hi + 1][::-1]
            grad[lo:hi + 1] += seg

    gm = float(np.mean(grad))
    diff = grad - gm
    scale = float(np.max(np.abs(diff)))
    if scale <= 0.0:
        return np.array(a, dtype=np.float64, copy=True)

    diff *= 0.5 / scale
    np.clip(diff, -1.5, 1.5, out=diff)

    cand = a * np.exp(-float(alpha) * diff)
    return _normalize(cand, n)


def _initial_best(n):
    """
    Build and validate a small family of starting points.

    Returns:
      internal_seq   - best by FFT score
      internal_score - its FFT score
      internal_conv  - its convolution
      official_seq   - best by official evaluator
      official_score - its official score
      fft_size       - FFT size for this n
    """
    size = _pow2_ge(2 * n - 1)

    primary = _candidate_cosine(n, 1.0, 1.0, 0.0)
    primary = _normalize(primary, n)

    internal_seq = np.array(primary, dtype=np.float64, copy=True)
    internal_score, internal_conv, _ = _fft_score(internal_seq, size)

    official_seq = np.array(primary, dtype=np.float64, copy=True)
    official_score = _official_score(internal_seq.tolist())

    if not np.isfinite(internal_score):
        internal_score = np.inf

    # Search a small neighborhood of the analytic shape.  The FFT is fast;
    # the official evaluator is only called when a candidate is clearly
    # promising.
    amps = (1.0, 0.92, 0.85)
    shifts = (0.0, 0.08, -0.08)
    exponents = (1.0, 0.92, 1.08)

    for amp in amps:
        for sh in shifts:
            for ex in exponents:
                if amp == 1.0 and sh == 0.0 and ex == 1.0:
                    continue

                cand = _candidate_cosine(n, ex, amp, sh)
                cand = _normalize(cand, n)
                score, conv, peak = _fft_score(cand, size)

                if not np.isfinite(score):
                    continue

                # Update the internal best even if the official score is not
                # checked yet.
                if score < internal_score - 1e-8:
                    internal_seq = np.array(cand, dtype=np.float64, copy=True)
                    internal_score = float(score)
                    internal_conv = conv

                    # Check official evaluator for clearly improved points.
                    off = _official_score(cand.tolist())
                    if np.isfinite(off) and (
                        not np.isfinite(official_score) or off < official_score - 1e-10
                    ):
                        official_seq = np.array(cand, dtype=np.float64, copy=True)
                        official_score = float(off)

    return internal_seq, internal_score, internal_conv, official_seq, official_score, size


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """
    Generate the best sequence found within the time budget.

    The returned sequence is a list of non-negative floats.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0

    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1800.0

    start_time = time.monotonic()
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    # A moderate length gives a very good approximation of the continuous
    # optimum while keeping direct official evaluation fast.
    n = 4096

    internal_seq, internal_score, internal_conv, official_seq, official_score, size = _initial_best(n)

    print(
        "initial internal score: {:.9f}, official score: {:.9f}, n={}".format(
            internal_score, official_score, n
        ),
        flush=True,
    )

    # Spend the remaining time on local search, but leave time for final
    # official verification.
    available_local = max(0.0, budget - 35.0)
    if available_local > 15.0 and np.isfinite(internal_score):
        deadline = start_time + available_local
        working = np.array(internal_seq, dtype=np.float64, copy=True)
        working_score = float(internal_score)
        working_conv = internal_conv

        sigma = 0.045
        shave_alpha = 0.035
        no_improve = 0
        iteration = 0

        while time.monotonic() < deadline:
            # Occasionally do a peak-shaving step; most updates are random.
            if iteration % 25 == 0 and working_conv is not None:
                cand = _peak_shave(working, working_conv, shave_alpha)
            else:
                cand = _random_perturb(working, rng, sigma, n)

            score, conv, peak = _fft_score(cand, size)

            if np.isfinite(score) and score < working_score - 1e-8:
                working = cand
                working_score = float(score)
                working_conv = conv
                no_improve = 0

                # Keep a copy of the best internal point.
                if score < internal_score - 1e-10:
                    internal_seq = np.array(cand, dtype=np.float64, copy=True)
                    internal_score = float(score)
                    internal_conv = conv
            else:
                no_improve += 1

            if no_improve > 4000:
                sigma *= 0.94
                shave_alpha *= 0.96
                no_improve = 0
                if sigma < 0.003:
                    break

            iteration += 1

            if iteration % 20000 == 0:
                print(
                    "local iter {}: internal best {:.9f}, sigma {:.5f}, elapsed {:.1f}s".format(
                        iteration, internal_score, sigma, time.monotonic() - start_time
                    ),
                    flush=True,
                )

        print(
            "local search finished after {} iterations, internal best {:.9f}".format(
                iteration, internal_score
            ),
            flush=True,
        )

    # Final official check: if the local search produced an officially better
    # sequence, return it; otherwise return the safest officially verified
    # sequence.
    final_internal_official = _official_score(internal_seq.tolist())

    if np.isfinite(final_internal_official) and (
        not np.isfinite(official_score) or final_internal_official <= official_score + 1e-12
    ):
        return [float(x) for x in internal_seq]

    return [float(x) for x in official_seq]
```
