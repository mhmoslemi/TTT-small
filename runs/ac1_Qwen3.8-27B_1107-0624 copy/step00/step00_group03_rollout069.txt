"""
Search for sequences minimizing score = 2*n*max(conv(s,s))/(sum(s))^2.
Uses multi-start local search with parametric initializations (multi-block patterns,
smooth windows) and random perturbations. Targets score < 1.5027.
"""

import numpy as np
import time

def fast_score(s):
    """Compute the evaluation score without the security overhead."""
    n = len(s)
    if n == 0:
        return np.inf
    s_arr = np.array(s, dtype=np.float64)
    s_arr = np.clip(s_arr, 0.0, 1000.0)
    sum_a = np.sum(s_arr)
    if sum_a < 0.01:
        return np.inf
    c = np.convolve(s_arr, s_arr)
    max_b = np.max(c)
    return float(2.0 * n * max_b / (sum_a * sum_a))

def make_blocks(n, num_blocks, rng):
    """Create a sequence with alternating high/low blocks."""
    s = np.zeros(n)
    block_size = n // (2 * num_blocks - 1)
    remainder = n - block_size * (2 * num_blocks - 1)
    pos = 0
    for i in range(num_blocks):
        w = block_size + (1 if i < remainder else 0)
        s[pos:pos+w] = 1.0
        pos += w
        if i < num_blocks - 1:
            w2 = block_size + (1 if i < remainder else 0)
            pos += w2
    return s

def make_smooth(n, alpha, rng):
    """Create a smooth window sequence."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    s = np.exp(-alpha * (x - center)**2 / n)
    return np.maximum(s, 0.0)

def make_power_law(n, p, rng):
    """Power law tapering sequence."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = np.abs(x - center) / center
    s = np.power(1.0 - d * 0.5, p)
    return np.maximum(s, 0.0)

def perturb_sequence(s, rng, strength=0.1):
    """Random perturbation preserving non-negativity."""
    n = len(s)
    s_new = s.copy()
    # Block perturbation
    width = max(1, n // rng.integers(10, 100))
    start = rng.integers(0, max(1, n - width))
    scale = np.exp(rng.normal(0, strength))

    # Symmetrize sometimes
    if rng.random() < 0.3:
        s_new = 0.5 * (s_new + s_new[::-1])
    return np.clip(s_new, 0.0, 1000.0)

def perturb_blocks(s, n, num_blocks, rng, strength=0.05):
    """Perturb block boundaries and heights."""

    if s is not None and len(s) == n:
        # Blend with previous best
        base = 0.3 * base + 0.7 * s
    base += rng.normal(0, strength, size=n)
    return np.clip(base, 0.0, 1000.0)

def local_search(s, budget_s, rng, label=""):
    """Adaptive step-size local search."""
    n = len(s)
    best = s.copy()
    best_score = fast_score(best)
    step = 0.1
    no_improve = 0
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < budget_s:
        # Adaptive perturbation
        if no_improve > 50:
            step = max(0.01, step * 0.8)
            no_improve = 0
        elif no_improve > 10:
            step *= 1.1
        
        candidate = perturb_sequence(best, rng, strength=step)
        cand_score = fast_score(candidate)
        
        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
        
        # Occasionally try a bigger jump

            candidate = perturb_sequence(best, rng, strength=step * 5)
            cand_score = fast_score(candidate)
            if cand_score < best_score:
                best = candidate
                best_score = cand_score
                no_improve = 0
                step = 0.1
    
    return best, best_score

def block_param_search(n, num_blocks, budget_s, rng):
    """Search over block heights and widths."""
    best = None
    best_score = np.inf
    
    for trial in range(50):
        # Random block heights
        heights = np.ones(num_blocks)
        heights += rng.normal(0, 0.3, size=num_blocks)
        heights = np.maximum(heights, 0.1)
        
        # Random block widths (sum to n)
        raw = rng.dirichlet(np.ones(2*num_blocks - 1)) * n
        widths = np.round(raw).astype(int)
        # Adjust to sum to n
        diff = n - widths.sum()
        if diff > 0:
            widths[0] += diff
        elif diff < 0:
            widths[0] = max(0, widths[0] + diff)
        
        # Build sequence

        pos = 0

            best_score = sc
            best = s.copy()
            # Local refine
            best, best_score = local_search(best, min(2.0, budget_s/10), rng)
    
    return best, best_score
