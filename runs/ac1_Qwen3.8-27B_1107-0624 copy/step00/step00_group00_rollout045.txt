```python
from scipy.sparse import coo_matrix
from scipy.optimize import linprog

def solve_lp_for_g(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x
```
