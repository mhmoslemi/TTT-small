```python
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import numpy as np

def solve_lp(f, M, n):
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Size: (2n-1) x n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = lil_matrix((2*n-1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    
    A = A.tocsr()
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix for linear convolution with f."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()
```

```python
def solve_lp(A, M, n):
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
```
