    n = len(f)
    h = np.convolve(f, f, mode='full')
    M = h.max()

    top_k = np.argsort(h)[::-1][:K]

Multi-strategy optimizer for minimizing 2n*max(f*f)/(sum(f))^2.
Uses LP-based updates, smoothed gradient descent, and random restarts.

import numpy as np
import time
from scipy.optimize import linprog

def softmax_weights(h, tau):
    m = h.max()
    if tau < 1e-15:
        idx = np.argmax(h)
        s = np.zeros_like(h)
        s[idx] = 1.0
        return m, s
    exp_h = np.exp((h - m) / tau)
    s = exp_h / exp_h.sum()
    return m + tau * np.log(exp_h.sum()), s

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.full(n, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def grad_smoothed(f, tau):
    """Compute gradient of soft_max(f*f) w.r.t. f using FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)[:2*n-1]
    val, s = softmax_weights(h, tau)
    S = np.fft.rfft(s, size)
    corr = np.fft.irfft(S * np.conj(F), size)[:n]
    grad = 2.0 * corr
    return val, grad

def lp_update_step(f, K=300):
    """One LP-based update step. Returns updated f or None."""
    n = len(f)
    h = np.convolve(f, f, mode='full')
    M = h.max()
    
    # Select K most binding constraints
    if K >= 2*n - 1:
        top_k = np.arange(2*n - 1)
    else:
        top_k = np.argsort(h)[::-1][:K]
    
    # Build A (K x n), b (K,)
    A = np.zeros((len(top_k), n))
    for idx, k in enumerate(top_k):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        A[idx, i_start:i_end] = f[k - i_start : k - i_end : -1]
    
    c = -np.ones(n)
    b = np.full(len(top_k), M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 0:

    except Exception:
        pass
    return None

def run_lp_iterations(f, n_iters, deadline, t_base=0.1):
    """Run multiple LP-based update steps."""
    n = len(f)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

        g = lp_update_step(f, K)
        if g is None:

        # Normalize g to have same sum as f
        g = g / g.sum() * f.sum()

            best = f_new.copy()
            best_score = score
            f = f_new
        else:
            f = f_new  # still update to explore
    
    return best, best_score
