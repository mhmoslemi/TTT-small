    c = (n - 1) / 2.0
    sigma = frac * n
    return np.exp(-(x - c) ** 2 / (2.0 * sigma ** 2))


def generate_hann(n):
    """Hann (raised cosine) window."""
    x = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))


def generate_flat_taper(n, taper_frac, power):
    """Flat center with tapered edges."""
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    half_width = taper_frac * n / 2.0
    t = np.clip(np.abs(x - c) - (n / 2.0 - half_width), 0, half_width) / half_width
    t = np.clip(t, 0, 1)
    return 1.0 - t ** power


def generate_super_gaussian(n, frac, p):
    """Super-Gaussian: exp(-(x/sigma)^p)."""
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = frac * n
    return np.exp(-np.abs(x - c) / sigma) ** p


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best non-negative coefficient sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    Uses multiple analytical initializations and FFT-accelerated gradient descent
    with momentum, adaptive learning rate, and random restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # ---- Generate initial candidates ----
    candidates = []

    # Provided sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
        if parent.size > 0 and np.sum(parent) > 0.01:
            candidates.append(parent.copy())
            print(f"Loaded parent sequence, n={parent.size}")
    except Exception as e:
        print(f"Parent sequence unavailable: {e}")

    # Gaussian windows at various widths and lengths
    for n in [2048, 4096, 8192, 16384]:
        for frac in [0.20, 0.22, 0.25, 0.27, 0.30, 0.33, 0.36, 0.40, 0.45]:
            candidates.append(generate_gaussian(n, frac))

    # Hann windows
    for n in [2048, 4096, 8192, 16384]:
        candidates.append(generate_hann(n))

    # Flat-taper windows
    for n in [4096, 8192]:
        for tf in [0.3, 0.4, 0.5, 0.6]:
            for p in [2, 3, 4]:
                candidates.append(generate_flat_taper(n, tf, p))

    # Super-Gaussians
    for n in [4096, 8192]:
        for frac in [0.25, 0.3, 0.35]:
            for p in [2, 3, 4, 6]:
                candidates.append(generate_super_gaussian(n, frac, p))

    # Prime-length Gaussians (potential number-theoretic advantage)
    for n in [4093, 8191]:
        for frac in [0.22, 0.25, 0.28, 0.32]:
            candidates.append(generate_gaussian(n, frac))

    print(f"Generated {len(candidates)} initial candidates")

    # ---- Quick evaluation of all candidates ----
    scored = []
    for i, cand in enumerate(candidates):
        cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
        s = np.sum(cand)
        if s < 0.01:
            continue
        cand_n = cand / s
        score = get_objective(cand_n)
        scored.append((score, i, cand_n))

    scored.sort(key=lambda x: x[0])
    print(f"Top 10 initial scores:")
    for i, (sc, ci, _) in enumerate(scored[:10]):
        print(f"  {i+1}. n={len(scored[0][2])}, score={sc:.6f}")

    # ---- Run gradient descent on top candidates ----
    n_top = min(len(scored), 15)
    time_per = max(15.0, (budget_s - 30.0) / n_top)

    best_seq = None
    best_score_global = float(np.inf)

    for rank in range(n_top):
        if time.monotonic() > deadline:
            break
        init_score, ci, cand = scored[rank]
        n = len(cand)
        tl = min(time.monotonic() + time_per, deadline)
        print(f"  Optimizing candidate {rank} (n={n}, init={init_score:.6f}), "
              f"budget={time_per:.0f}s...", end=" ")

        res_a, res_score = run_gradient_descent(cand, tl, rng, verbose=False)
        improved = res_score < init_score
        print(f"final={res_score:.6f} {'(improved)' if improved else ''}")

        if res_score < best_score_global:
            best_score_global = res_score
            best_seq = res_a.copy()
            print(f"    ** NEW GLOBAL BEST: {best_score_global:.6f}")

    # ---- Additional refinement on the best found ----
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"  Final refinement, budget={remaining:.0f}s...")
            refined, refined_score = run_gradient_descent(
                best_seq, deadline, rng, verbose=True)
            if refined_score < best_score_global:
                best_score_global = refined_score
                best_seq = refined.copy()
                print(f"    ** REFINED BEST: {best_score_global:.6f}")

    # ---- Output ----
    if best_seq is None:
        # Fallback: uniform
        best_seq = np.ones(1024, dtype=np.float64)
        best_score_global = 2.0
        print("Using fallback uniform sequence")

    # Verify with actual evaluator
    final_check = evaluate_sequence(best_seq.tolist())
    print(f"\nBest internal score: {best_score_global:.6f}")
    print(f"Verified via evaluate_sequence: {final_check:.6f}")
    print(f"Total time used: {time.monotonic() - start_time:.1f}s")

    return [float(x) for x in best_seq]
