```
def _sequence_from_params_bases(n, params, bases):
    f = np.ones(n, dtype=np.float64)
    p = np.asarray(params, dtype=np.float64)
    maxh = min(len(bases), p.size//2)
    for idx in range(maxh):
        amp = float(p[2*idx])
        ph = float(p[2*idx+1])
        if amp == 0.0: continue
        s, c = bases[idx]
        f += amp * (s * np.cos(ph) + c * np.sin(ph))
    if not np.all(np.isfinite(f)): return None
    minf = float(np.min(f))
    if minf < 0.0:
        extra = f - 1.0
        min_extra = float(np.min(extra))
        if min_extra < -1.0:
            scale = -0.999999 / min_extra
            f = 1.0 + scale * extra
        else:
            f = np.maximum(f, 0.0)
    else:
        # no need but maybe f very small negative due round
        if minf < -1e-12: f=np.maximum(f,0.0)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 1e-300: return None
    return f / s
```

```python
"""
Fourier-shape local search for a nonnegative step function.

The objective is scale-invariant, so sequences are represented as probability
vectors of length n.  Strong candidates are generated from low-frequency
Fourier modulations of a flat density

    f(t) = 1 + sum_m a_m sin(2*pi*m*t + phi_m),

sampled at midpoints and normalized.  Candidate quality is scored quickly by
FFT convolution, then refined by a bounded coordinate/random-search local
optimizer.  The best internally-scored candidates are finally checked with the
exact evaluator.
"""

import numpy as np
import time


def _next_pow2_at_least(m):
    """Smallest power of two >= m."""
    L = 1
    while L < m:
        L <<= 1
    return L


def _score_array(p, L=None):
    """
    Fast score for a nonnegative sequence p:

        2 * len(p) * max(convolve(p, p)) / sum(p)^2

    Uses FFT for larger sequences.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf

    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf

    target_len = 2 * n - 1

    if n <= 256:
        c = np.convolve(p, p, mode="full")
        m = float(np.max(c))
        return float(2.0 * n * m / (s * s))

    if L is None:
        L = _next_pow2_at_least(target_len)

    F = np.fft.rfft(p, L)
    P = F.real * F.real + F.imag * F.imag
    c = np.fft.irfft(P, L)
    if c.size >= target_len:
        c = c[:target_len]
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))


def _make_bases(n, max_harm):
    """
    Precompute sine/cosine bases at midpoint locations:

        t_i = (i + 0.5) / n, i = 0,...,n-1.

    For each harmonic m, returns (sin(2*pi*m*t), cos(2*pi*m*t)).
    """
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    bases = []
    two_pi = 2.0 * np.pi
    for m in range(1, max_harm + 1):
        ang = two_pi * float(m) * t
        bases.append((np.sin(ang), np.cos(ang)))
    return bases


def _normalize_params(params, amp_bounds=None):
    """
    Normalize parameter vector [amp0, phase0, amp1, phase1, ...].
    Amplitudes are made nonnegative and clipped to modest bounds.
    Phases are wrapped to [-pi, pi].
    """
    p = np.asarray(params, dtype=np.float64).copy()
    two_pi = 2.0 * np.pi

    for i in range(0, p.size, 2):
        a = float(p[i])
        ph = float(p[i + 1])

        if not np.isfinite(a):
            a = 0.0
        if not np.isfinite(ph):
            ph = 0.0

        if a < 0.0:
            a = -a
            ph += np.pi

        h = i // 2
        if amp_bounds is not None and h < len(amp_bounds):
            bound = float(amp_bounds[h])
        else:
            bound = 3.0

        a = min(max(a, 0.0), bound)
        ph = (ph + np.pi) % two_pi - np.pi

        p[i] = a
        p[i + 1] = ph

    return p


def _sequence_from_params_bases(n, params, bases):
    """
    Build a normalized nonnegative sequence from Fourier parameters using
    precomputed bases.
    """
    p = np.asarray(params, dtype=np.float64)
    f = np.ones(n, dtype=np.float64)

    max_h = min(len(bases), p.size // 2)
    for idx in range(max_h):
        amp = float(p[2 * idx])
        ph = float(p[2 * idx + 1])
        if amp == 0.0:
            continue
        s, c = bases[idx]
        f += amp * (s * np.cos(ph) + c * np.sin(ph))

    if not np.all(np.isfinite(f)):
        return None

    # If the modulation creates negatives, rescale the zero-mean part around
    # the flat baseline to keep the sequence nonnegative.
    minf = float(np.min(f))
    if minf < 0.0:
        extra = f - 1.0
        min_extra = float(np.min(extra))
        if min_extra < -1.0:
            scale = -0.999999 / min_extra
            f = 1.0 + scale * extra

    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 1e-300:
        return None

    return f / s


def _param_score_bases(params, n, bases, L):
    """
    Internally fast score for a parameter vector.
    """
    seq = _sequence_from_params_bases(n, params, bases)
    if seq is None:
        return np.inf
    return _score_array(seq, L)


def _add_candidate(params, n, bases, L, candidates, amp_bounds=None):
    """
    Normalize, score, and append a candidate if finite.
    Returns (score, parameter tuple).
    """
    p = _normalize_params(params, amp_bounds)
    score = _param_score_bases(p, n, bases, L)
    tup = tuple(np.asarray(p, dtype=float).tolist())

    if np.isfinite(score):
        candidates.append((float(score), tup))

    return float(score), tup


def _local_search_params(start_params, n, bases, L, amp_bounds, deadline, rng,
                        min_score_stop=1.45):
    """
    Bounded coordinate + random-jump local search on Fourier parameters.
    """
    best = _normalize_params(start_params, amp_bounds)
    best_s = float(_param_score_bases(best, n, bases, L))
    D = best.size

    sigma = np.zeros(D, dtype=np.float64)
    for i in range(0, D, 2):
        h = i // 2
        if amp_bounds is not None and h < len(amp_bounds):
            bound = float(amp_bounds[h])
        else:
            bound = 1.0
        sigma[i] = max(1e-5, min(0.03, 0.05 * bound))
        sigma[i + 1] = 0.1

    step_multipliers = (1.0, 0.5, 2.0, 0.25, 4.0)
    stall = 0

    while time.monotonic() < deadline:
        if not np.isfinite(best_s) or best_s <= min_score_stop:
            break

        improved = False

        for i in range(D):
            base = float(sigma[i])
            if base <= 1e-7:
                if i % 2 == 0:
                    h = i // 2
                    if amp_bounds is not None and h < len(amp_bounds):
                        bound = float(amp_bounds[h])
                    else:
                        bound = 1.0
                    base = max(1e-5, 0.005 * bound)
                else:
                    base = 0.05

            accepted = False

            for mult in step_multipliers:
                delta = mult * base
                for sign in (1.0, -1.0):
                    cand = best.copy()
                    cand[i] += sign * delta

                    if i % 2 == 0:
                        if cand[i] < 0.0:
                            cand[i] = 0.0
                        h = i // 2
                        if amp_bounds is not None and h < len(amp_bounds):
                            bound = float(amp_bounds[h])
                        else:
                            bound = 1.0
                        if cand[i] > bound:
                            cand[i] = bound

                    cand = _normalize_params(cand, amp_bounds)
                    s = _param_score_bases(cand, n, bases, L)

                    if np.isfinite(s) and s < best_s:
                        best = cand
                        best_s = float(s)
                        improved = True
                        accepted = True
                        sigma[i] = min(
                            0.8 if i % 2 else 0.2,
                            sigma[i] * 1.05
                        )
                        break

                if accepted:
                    break

            if not accepted:
                sigma[i] *= 0.7

        if not improved:
            jump_improved = False
            for _ in range(8):
                cand = best.copy()
                for i in range(D):
                    scale = float(sigma[i]) * 2.0
                    cand[i] += float(rng.normal(0.0, scale))

                cand = _normalize_params(cand, amp_bounds)
                s = _param_score_bases(cand, n, bases, L)

                if np.isfinite(s) and s < best_s:
                    best = cand
                    best_s = float(s)
                    jump_improved = True
                    improved = True
                    break

            if not jump_improved:
                stall += 1

                # Occasionally enlarge steps to escape flat stalls.
                for i in range(D):
                    if i % 2 == 0:
                        h = i // 2
                        if amp_bounds is not None and h < len(amp_bounds):
                            bound = float(amp_bounds[h])
                        else:
                            bound = 1.0
                        sigma[i] = min(
                            0.1 * bound,
                            max(1e-5, sigma[i] * 1.6 + 0.002 * bound)
                        )
                    else:
                        sigma[i] = min(1.2, sigma[i] * 1.6 + 0.05)

                if stall >= 10:
                    break
        else:
            stall = 0

    return best, float(best_s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search and return a list of nonnegative heights minimizing the evaluator.
    """
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)
    rng = np.random.default_rng(seed)

    # Longer sequences reduce discretization error.  16384 is fast enough
    # for FFT scoring in the usual 1800-second budget.
    if budget_s >= 1500.0:
        n = 16384
    elif budget_s >= 700.0:
        n = 8192
    elif budget_s >= 250.0:
        n = 4096
    else:
        n = 2048

    MAX_HARM = 3
    bases = _make_bases(n, MAX_HARM)
    L = _next_pow2_at_least(2 * n - 1)

    # The first harmonic is the main shape; higher harmonics only polish it.
    amp_bounds = [1.10, 0.35, 0.20]
    D = 2 * MAX_HARM

    candidates = []
    best_score = np.inf
    best_tuple = None

    def _consider(s, tup):
        nonlocal best_score, best_tuple
        if np.isfinite(s) and s < best_score:
            best_score = float(s)
            best_tuple = tup

    # Uniform starting point / fallback.
    s, tup = _add_candidate(np.zeros(D), n, bases, L, candidates, amp_bounds)
    _consider(s, tup)

    # First harmonic grid.  This alone is already very strong: amplitudes
    # around 0.75-0.85 produce an approximately optimal sum-density shape.
    phases1 = np.linspace(-np.pi, np.pi, 16, endpoint=False)
    a1_values = [
        0.0, 0.30, 0.50, 0.60, 0.65, 0.70, 0.75,
        0.80, 0.85, 0.90, 1.00
    ]

    for a1 in a1_values:
        for ph in phases1:
            params = np.zeros(D)
            params[0] = a1
            params[1] = float(ph)
            s, tup = _add_candidate(params, n, bases, L, candidates, amp_bounds)
            _consider(s, tup)

    candidates.sort()

    # Second harmonic grid from the best first-harmonic seeds.
    top_first = candidates[:6]
    a2_values = [0.0, 0.05, 0.10, 0.15, 0.20]
    phases2 = np.linspace(-np.pi, np.pi, 8, endpoint=False)

    for score, tup in top_first:
        base = np.array(tup, dtype=np.float64)
        for a2 in a2_values:
            for ph in phases2:
                cand = base.copy()
                cand[2] = a2
                cand[3] = float(ph)
                s, nt = _add_candidate(cand, n, bases, L, candidates, amp_bounds)
                _consider(s, nt)

    candidates.sort()

    # Third harmonic grid from the best second-harmonic seeds.
    top_second = candidates[:8]
    a3_values = [0.0, 0.05, 0.10, 0.15]
    phases3 = np.linspace(-np.pi, np.pi, 8, endpoint=False)

    for score, tup in top_second:
        base = np.array(tup, dtype=np.float64)
        for a3 in a3_values:
            for ph in phases3:
                cand = base.copy()
                cand[4] = a3
                cand[5] = float(ph)
                s, nt = _add_candidate(cand, n, bases, L, candidates, amp_bounds)
                _consider(s, nt)

    # Random Fourier seeds for diversity.
    for _ in range(200):
        params = np.zeros(D)
        params[0] = float(rng.uniform(0.40, 1.00))
        params[1] = float(rng.uniform(-np.pi, np.pi))
        params[2] = float(rng.uniform(0.00, 0.25))
        params[3] = float(rng.uniform(-np.pi, np.pi))
        params[4] = float(rng.uniform(0.00, 0.15))
        params[5] = float(rng.uniform(-np.pi, np.pi))
        s, tup = _add_candidate(params, n, bases, L, candidates, amp_bounds)
        _consider(s, tup)

    candidates.sort()
    if candidates:
        best_score, best_tuple = candidates[0]

    print(f"[initial] n={n}, internal best={best_score:.9f}")

    # Local coordinate refinement.  The internal target here is deliberately
    # below the requested upper bound to leave a comfortable margin.
    if best_tuple is not None and time.monotonic() < deadline - 120.0:
        start = np.array(best_tuple, dtype=np.float64)

        # Cap local search time based on how close we already are.
        if best_score < 1.46:
            cap = 120.0
        elif best_score < 1.48:
            cap = 300.0
        elif best_score < 1.50:
            cap = 600.0
        else:
            cap = 1200.0

        local_deadline = min(deadline - 240.0, time.monotonic() + cap)
        if local_deadline > time.monotonic() + 20.0:
            refined, refined_score = _local_search_params(
                start, n, bases, L, amp_bounds, local_deadline, rng,
                min_score_stop=1.44
            )
            tup_ref = tuple(np.asarray(refined, dtype=float).tolist())
            if np.isfinite(refined_score):
                candidates.append((float(refined_score), tup_ref))
                if refined_score < best_score:
                    best_score = float(refined_score)
                    best_tuple = tup_ref
                print(f"[local1] internal best={best_score:.9f}")

            # If there is still time and score is not yet excellent, refine
            # another good seed.
            if (len(candidates) >= 2 and best_score > 1.47 and
                    time.monotonic() < deadline - 180.0):
                second = np.array(candidates[1][1], dtype=np.float64)
                second_deadline = min(deadline - 180.0, time.monotonic() + 180.0)
                if second_deadline > time.monotonic() + 20.0:
                    refined2, refined_score2 = _local_search_params(
                        second, n, bases, L, amp_bounds, second_deadline, rng,
                        min_score_stop=1.44
                    )
                    tup_ref2 = tuple(np.asarray(refined2, dtype=float).tolist())
                    if np.isfinite(refined_score2):
                        candidates.append((float(refined_score2), tup_ref2))
                        if refined_score2 < best_score:
                            best_score = float(refined_score2)
                            best_tuple = tup_ref2
                        print(f"[local2] internal best={best_score:.9f}")

    # Final exact verification on the best internal candidates.
    candidates.sort()
    final_candidates = candidates[:10]

    best_exact = np.inf
    best_seq = None

    eval_deadline = deadline - 2.0
    exact_count = 0

    for score, tup in final_candidates:
        if time.monotonic() > eval_deadline:
            break

        params = np.array(tup, dtype=np.float64)
        seq = _sequence_from_params_bases(n, params, bases)
        if seq is None:
            continue

        seq = np.asarray(seq, dtype=np.float64)
        # Ensure valid runtime type.
        seq = np.clip(seq, 0.0, 1000.0)

        exact_score = float(evaluate_sequence([float(x) for x in seq]))
        exact_count += 1

        if np.isfinite(exact_score) and exact_score < best_exact:
            best_exact = exact_score
            best_seq = seq
            print(f"[exact#{exact_count}] score={best_exact:.9f}")

        if exact_score < 1.45 and exact_count >= 2:
            break

    # Robust fallback if exact verification produced nothing usable.
    if best_seq is None or not np.isfinite(best_exact):
        params = np.zeros(D)
        params[0] = 0.80
        seq = _sequence_from_params_bases(n, params, bases)
        if seq is None:
            seq = np.full(n, 1.0 / float(n), dtype=np.float64)

        seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
        best_exact = float(evaluate_sequence([float(x) for x in seq]))
        best_seq = seq

    # Absolute safe final fallback.
    if best_seq is None or not np.isfinite(best_exact):
        best_seq = np.array([0.01], dtype=np.float64)
        best_exact = float(evaluate_sequence([float(x) for x in best_seq]))

    print(f"[final] internal best={best_score:.9f}, exact best={best_exact:.9f}")

    return [float(x) for x in best_seq]
```
