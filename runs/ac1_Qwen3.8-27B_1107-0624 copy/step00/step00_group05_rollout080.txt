    v = np.where(np.isfinite(v), v, 0.0)

```python
import numpy as np
import time
import math
from operator import itemgetter

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def next_power_of_two(x):
    """Smallest power of two >= x."""
    if x <= 1:
        return 1
    n = 1
    while n < x:
        n <<= 1
    return int(n)


def normalize_nonneg(v):
    """Clip to [0, inf) and normalize to unit sum."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        v = np.ones_like(v, dtype=np.float64)
        s = float(v.size)
    return v / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    target = 1.0
    u = np.sort(v)[::-1]
    n = int(u.size)
    cs = np.cumsum(u)
    cond = u * (np.arange(1, n + 1, dtype=np.float64)) > (cs - target)
    if not np.any(cond):
        return np.full(n, target / n, dtype=np.float64)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cs[rho] - target) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, target / n, dtype=np.float64)
    return out / s


def sequence_from_z(z):
    """Convert a log-density to a unit-sum nonnegative sequence."""
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -30.0, 30.0)
    a = np.exp(z)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a / s


def resample_sequence(a, target_len):
    """Linearly resample a unit-sum sequence to a new length."""
    a = np.asarray(a, dtype=np.float64)
    m = int(a.size)
    if m == 0 or target_len <= 0:
        return np.ones(int(target_len), dtype=np.float64) / max(1, int(target_len))
    xs = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    xt = (np.arange(int(target_len), dtype=np.float64) + 0.5) / float(target_len)
    out = np.interp(xt, xs, a)
    return normalize_nonneg(out)


def max_direct(a):
    """Exact max convolution for small sequences."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = normalize_nonneg(a)
    c = np.convolve(a, a)
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return cm


def max_full_fft(a, L):
    """FFT-based max convolution for long sequences. L must be large enough."""
    a = normalize_nonneg(a)
    n = int(a.size)
    if n <= 0:
        return np.inf
    F = np.fft.rfft(a, n=L)
    c = np.fft.irfft(F * F, n=L)[:2 * n - 1]
    cm = float(np.max(c))
    if not np.isfinite(cm):
        return np.inf
    return cm if cm > 0.0 else 0.0


def compute_c_fft(a, L):
    """Return (cmax, c, F) using FFT. L must be large enough."""
    a = normalize_nonneg(a)
    n = int(a.size)
    F = np.fft.rfft(a, n=L)
    c = np.fft.irfft(F * F, n=L)[:2 * n - 1]
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return cm, c, F


def make_weights(c, cmax, K_tau):
    """Softmax weights over convolution coefficients, blended with active peaks."""
    if len(c) == 0 or cmax <= 0.0 or not np.isfinite(cmax):
        return np.ones(len(c), dtype=np.float64) / max(1, len(c))
    tau = float(K_tau) / max(float(cmax), 1e-300)
    z = tau * (np.asarray(c, dtype=np.float64) - float(cmax))
    w = np.exp(z)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        return np.ones(len(c), dtype=np.float64) / max(1, len(c))
    w = w / sw
    active = c >= (float(cmax) * (1.0 - 1e-3))
    la = int(np.count_nonzero(active))
    if la > 0 and la < max(8, int(0.02 * len(c))):
        wa = np.zeros_like(w)
        wa[active] = 1.0 / float(la)
        w = 0.5 * w + 0.5 * wa
        sw = float(w.sum())
        if sw > 0.0 and np.isfinite(sw):
            w = w / sw
        else:
            w = np.ones(len(c), dtype=np.float64) / float(len(c))
    return w


def compute_gradient_fft(a, weights, L, idx):
    """Gradient of smooth max convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    weights = np.asarray(weights, dtype=np.float64)
    F = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(weights[::-1], n=L)
    conv = np.fft.irfft(F * Fw, n=L)
    grad = 2.0 * conv[idx]
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return grad


def add_to_top(top, score, seq, keep=8):
    """Keep a small list of best candidates."""
    if not np.isfinite(score):
        return
    seq = normalize_nonneg(seq)
    top.append((float(score), seq))
    if len(top) > keep:
        top.sort(key=itemgetter(0))
        del top[keep:]


def random_continuous_candidates(m, count, rng, keep=5, deadline=None):
    """Random log-polynomial/trigonometric shapes on a small grid."""
    top = []
    m = max(8, int(m))
    x = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    b1 = x
    b2 = x * x
    b3 = x * x * x
    b4 = b3 * x
    b5 = np.sin(2.0 * np.pi * x)
    b6 = np.cos(2.0 * np.pi * x)
    b7 = np.sin(4.0 * np.pi * x)
    b8 = np.cos(4.0 * np.pi * x)
    for _ in range(max(0, int(count))):
        if deadline is not None and time.monotonic() >= deadline:
            break
        p1 = rng.uniform(-3.0, 3.0)
        p2 = rng.uniform(-3.0, 3.0)
        p3 = rng.uniform(-2.5, 2.5)
        p4 = rng.uniform(-2.0, 2.0)
        p5 = rng.uniform(-2.0, 2.0)
        p6 = rng.uniform(-2.0, 2.0)
        p7 = rng.uniform(-1.5, 1.5)
        p8 = rng.uniform(-1.5, 1.5)
        z = (p1 * b1 + p2 * b2 + p3 * b3 + p4 * b4 +
             p5 * b5 + p6 * b6 + p7 * b7 + p8 * b8)
        a = sequence_from_z(z)
        cm = max_direct(a)
        sc = 2.0 * float(m) * cm
        add_to_top(top, sc, a, keep)
    return top


def continuous_pg(a0, deadline, rng, K_tau=60.0, lr0=0.30):
    """Continuous/projected-gradient optimization on a small sequence."""
    a = normalize_nonneg(a0)
    m = int(a.size)
    if m < 4:
        return a.copy(), max_direct(a)
    L = next_power_of_two(3 * m - 2)
    idx = 2 * m - 2 - np.arange(m, dtype=np.int64)
    cmax = max_direct(a)
    best = a.copy()
    bestc = cmax
    lr = float(lr0)
    stall = 0
    noacc = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        if a.size != m:
            a = normalize_nonneg(a)
        c = np.convolve(a, a)
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax < 0.0:
            cmax = 0.0

        if cmax < bestc - 1e-14:
            best = a.copy()
            bestc = cmax
            stall = 0
            noacc = 0
            lr = min(1.2, lr * 1.05)
        else:
            stall += 1

        if stall > 120:
            a = normalize_nonneg(best * np.exp(rng.normal(0.0, 0.08, m)))
            stall = 0
            noacc = 0
            continue

        w = make_weights(c, cmax, K_tau)
        grad = compute_gradient_fft(a, w, L, idx)
        gmax = float(np.max(np.abs(grad)))
        if gmax <= 1e-300:
            break

        gmean = float(np.dot(a, grad))
        rel = grad - gmean
        absrel = np.abs(rel)
        rscale = float(np.median(absrel))
        if rscale < 1e-300:
            rscale = float(np.mean(absrel))
        if rscale < 1e-300:
            break

        accepted = False

        # Multiplicative/log-style update.
        step = lr / max(rscale, 1e-300)
        for t in (1.0, 0.5, 0.25, 0.125):
            expo = np.clip(-t * step * rel, -50.0, 50.0)
            cand = normalize_nonneg(a * np.exp(expo) + 1e-300)
            cc = max_direct(cand)
            if cc < cmax - 1e-14:
                a = cand
                cmax = cc
                accepted = True
                noacc = 0
                break

        # Fallback: linear projected step.
        if not accepted:
            gscale = max(gmax, 1e-300)
            step2 = lr * (1.0 / float(m)) / gscale
            for t in (1.0, 0.5, 0.25, 0.125):
                cand = project_simplex(a - t * step2 * grad)
                cc = max_direct(cand)
                if cc < cmax - 1e-14:
                    a = cand
                    cmax = cc
                    accepted = True
                    noacc = 0
                    break

        if not accepted:
            noacc += 1
            if noacc >= 8:
                lr = max(0.03, lr * 0.72)

        if it % 500 == 0:
            print(f"  cont-pg iter={it} score={2.0 * m * cmax:.8f} best={2.0 * m * bestc:.8f}", flush=True)

    if max_direct(best) < bestc:
        bestc = max_direct(best)
    return best, bestc


def continuous_lp_improve(a0, deadline, rng):
    """LP direction from the stated strategy, used for small sequences."""
    if linprog is None or a0.size < 8:
        return normalize_nonneg(a0), max_direct(a0)
    a = normalize_nonneg(a0)
    best = a.copy()
    bestc = max_direct(best)
    if bestc <= 1e-14:
        return best, bestc

    while time.monotonic() < deadline:
        best = normalize_nonneg(best)
        M = bestc * (1.0 + 1e-8)
        m = int(a.size)
        A = np.zeros((2 * m - 1, m), dtype=np.float64)
        for p in range(m):
            A[p:p + m, p] = best

        try:
            res = linprog(
                -np.ones(m, dtype=np.float64),
                A_ub=A,
                b_ub=np.full(2 * m - 1, M, dtype=np.float64),
                bounds=[(0.0, None)] * m,
                method="highs",
                options={"presolve": True}
            )
        except Exception:
            break

        if not res.success:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.maximum(g, 0.0)
        G = float(np.sum(g))
        if not np.isfinite(G) or G <= 1.0 + 1e-6:
            break
        gs = normalize_nonneg(g)
        improved = False
        for t in (0.9, 0.6, 0.4, 0.2, 0.1):
            cand = normalize_nonneg(best * (1.0 - t) + gs * t)
            cc = max_direct(cand)
            if cc < bestc - 1e-14:
                best = cand
                bestc = cc
                improved = True
                break
        if not improved:
            break

    return best, bestc


def full_optimize(a0, L, idx, deadline, rng, K_tau=65.0, lr0=0.30):
    """Full FFT-based smooth-gradient refinement with line search."""
    a = normalize_nonneg(a0)
    n = int(a.size)
    if n < 4:
        return a.copy(), max_full_fft(a, L)

    cm, _, _ = compute_c_fft(a, L)
    best = a.copy()
    bestc = cm
    lr = float(lr0)
    stall = 0
    noacc = 0
    it = 0
    print("full-optimize start score=%.8f" % (2.0 * n * bestc), flush=True)

    while time.monotonic() < deadline:
        it += 1
        cmax, c, F = compute_c_fft(a, L)

        if cmax < bestc - 1e-14:
            best = a.copy()
            bestc = cmax
            stall = 0
            noacc = 0
            lr = min(1.5, lr * 1.04)
        else:
            stall += 1

        if cmax > bestc * 1.05:
            a = best.copy()
            stall = 0
            continue

        if stall > 250:
            if rng.random() < 0.35:
                a = resample_sequence(resample_sequence(best, 256), n)
            else:
                a = best.copy()
                wmin = max(2, n // 500)
                wmax = max(wmin + 1, min(n, n // 10))
                if wmax >= n:
                    wmax = max(wmin + 1, n // 2)
                w = int(rng.integers(wmin, wmax + 1))
                s = int(rng.integers(0, n - w + 1))
                a[s:s + w] *= np.exp(rng.normal(0.0, 0.25))
                a = normalize_nonneg(a)
            stall = 0
            noacc = 0
            continue

        w = make_weights(c, cmax, K_tau)
        grad = compute_gradient_fft(a, w, L, idx)
        gmax = float(np.max(np.abs(grad)))
        if gmax <= 1e-300:
            break

        gmean = float(np.dot(a, grad))
        rel = grad - gmean
        absrel = np.abs(rel)
        rscale = float(np.median(absrel))
        if rscale < 1e-300:
            rscale = float(np.mean(absrel))
        if rscale < 1e-300:
            continue

        step = lr / max(rscale, 1e-300)
        accepted = False

        for t in (1.0, 0.5, 0.25, 0.125):
            expo = np.clip(-t * step * rel, -50.0, 50.0)
            cand = normalize_nonneg(a * np.exp(expo) + 1e-300)
            Fc = np.fft.rfft(cand, n=L)
            cc_all = np.fft.irfft(Fc * Fc, n=L)
            cc = float(np.max(cc_all[:2 * n - 1]))
            if not np.isfinite(cc) or cc < 0.0:
                cc = 0.0
            if cc < cmax - 1e-14:
                a = cand
                cmax = cc
                accepted = True
                noacc = 0
                if cmax < bestc:
                    best = a.copy()
                    bestc = cmax
                break

        if not accepted:
            noacc += 1
            if noacc >= 8:
                lr = max(0.03, lr * 0.72)

        if it % 200 == 0:
            print(f"  full-opt iter={it} score={2.0 * n * cmax:.8f} best={2.0 * n * bestc:.8f}", flush=True)

    return best, bestc


def full_local_search(a0, L, deadline, rng, bestc=None):
    """Cheap block/local search on the full sequence."""
    a = normalize_nonneg(a0)
    n = int(a.size)
    if n < 8:
        return a.copy(), max_full_fft(a, L)
    cm = max_full_fft(a, L)
    if bestc is None or not np.isfinite(bestc):
        bestc = cm
    cur = a.copy()
    best = a.copy()
    sigma = 0.05
    stall = 0
    wmin = max(2, n // 500)
    wmax = max(wmin + 1, min(n, n // 10))
    if wmax >= n:
        wmax = max(wmin + 1, n // 2)

    while time.monotonic() < deadline:
        cand = cur.copy()
        if wmax <= 1:
            continue
        w = int(rng.integers(wmin, wmax + 1))
        s = int(rng.integers(0, n - w + 1))

        r = rng.random()
        if r < 0.70:
            cand[s:s + w] *= np.exp(rng.normal(0.0, sigma))
        elif r < 0.90:
            cand[s:s + w] = cand[s:s + w][::-1]
        else:
            w2 = int(rng.integers(wmin, wmax + 1))
            s2 = int(rng.integers(0, n - w2 + 1))
            scale = np.exp(rng.normal(0.0, sigma))
            cand[s:s + w] *= scale
            cand[s2:s2 + w2] /= max(scale, 1e-12)

        cand = normalize_nonneg(cand)
        cm = max_full_fft(cand, L)
        if np.isfinite(cm) and cm < bestc - 1e-14:
            cur = cand.copy()
            best = cand.copy()
            bestc = cm
            sigma = min(0.8, sigma * 1.08)
            stall = 0
        else:
            if np.isfinite(cm) and (cm - bestc) <= 1e-7 and rng.random() < 0.05:
                cur = cand.copy()
            else:
                cur = best.copy()
                stall += 1
                if stall > 120:
                    sigma = max(0.01, sigma * 0.9)
                    stall = 0

    return best, bestc

                break
            a_start = starts[idx_start][1]
            dl = min(pg_deadline, time.monotonic() + per)
            if dl <= time.monotonic() + 0.5:
                continue
            a_opt, _ = continuous_pg(a_start, dl, rng, K_tau=65.0, lr0=0.32)
            af = resample_sequence(a_opt, n)
            cm = max_full_fft(af, L)
            add_to_top(top, 2.0 * float(n) * cm, af, 12)

    # Small LP refinement
    if linprog is not None and time.monotonic() < cont_deadline:
        lp_start = None
        if starts:
            lp_start = resample_sequence(starts[0][1], 128)
        elif top:
            lp_start = resample_sequence(top[0][1], 128)
        if lp_start is not None:
            lp_deadline = min(cont_deadline, time.monotonic() + 30.0)
            if lp_deadline > time.monotonic() + 1.0:
                a_lp, _ = continuous_lp_improve(lp_start, lp_deadline, rng)
                af = resample_sequence(a_lp, n)
                cm = max_full_fft(af, L)
                add_to_top(top, 2.0 * float(n) * cm, af, 12)

    if top:
        best_full = top[0][1].copy()
        best_score = top[0][0]
    print(f"after continuous best full score={best_score:.8f}", flush=True)

    # ---- Full-resolution optimization ----
    rem = deadline - time.monotonic()
    local_reserve = min(300.0, max(2.0, rem * 0.25))
    full_deadline = deadline - local_reserve
    if full_deadline <= time.monotonic() + 1.0:
        full_deadline = max(time.monotonic() + 1.0, deadline - 0.5)

    if best_full is None:
        best_full = np.full(n, 1.0 / float(n), dtype=np.float64)

    if full_deadline > time.monotonic() + 2.0:
        best_full, best_cmax = full_optimize(best_full, L, idx, full_deadline, rng,
                                             K_tau=70.0, lr0=0.30)
        fs = 2.0 * float(n) * max_full_fft(best_full, L)
        if fs < best_score:
            best_score = fs

    if deadline > time.monotonic() + 1.0:
        best_full, best_cmax = full_local_search(best_full, L, deadline, rng)
        fs = 2.0 * float(n) * max_full_fft(best_full, L)
        if fs < best_score:
            best_score = fs

    # Final safety checks
    if best_full is None or best_full.size != n:
        best_full = np.full(n, 1.0 / float(n), dtype=np.float64)

    # Compare top full candidate one more time
    if top:
        cm = max_full_fft(top[0][1], L)
        sc = 2.0 * float(n) * cm
        if np.isfinite(sc) and sc < best_score:
            best_full = top[0][1].copy()
            best_score = sc

    cm = max_full_fft(best_full, L)
    if np.isfinite(cm):
        best_score = 2.0 * float(n) * cm
    else:
        best_full = np.full(n, 1.0 / float(n), dtype=np.float64)
        best_score = 2.0

    result = [float(v) for v in best_full]

    # Validate with official evaluator
    try:
        val = float(evaluate_sequence(result))
    except Exception:
        val = np.inf

    if not np.isfinite(val):
        # Try a scaled version
        s = float(np.sum(best_full))
        if s > 0:
            best_full = best_full * (10.0 / s)
        else:
            best_full = np.full(n, 10.0 / float(n), dtype=np.float64)
        result = [float(v) for v in best_full]
        try:
            val = float(evaluate_sequence(result))
        except Exception:
            val = np.inf

    if not np.isfinite(val):
        # Uniform fallback
        result = [1.0 / float(n)] * n
        try:
            val = float(evaluate_sequence(result))
        except Exception:
            val = np.inf

    print(f"finalInternalScore={best_score:.8f} finalEval={val if np.isfinite(val) else float('inf')}", flush=True)
    return result
```
