3. Move towards $g$

"""
Frank-Wolfe (conditional gradient) method for minimizing the peak of the
self-convolution of a non-negative sequence. At each step, solve an LP to
find the best direction that reduces the convolution peak while maintaining
non-negativity.
"""

import numpy as np
import time
import scipy.linalg
from scipy.optimize import linprog


def build_constraint_matrix(f):
    """Build the Toeplitz constraint matrix A where A[k,j] = f[k-j] if in range."""
    n = len(f)
    first_col = np.concatenate([f, np.zeros(n-1)])
    first_row = np.concatenate([[f[0]], np.zeros(n-1)])
    return scipy.linalg.toeplitz(first_col, first_row)


def solve_lp_direction(f, M, n):
    """Solve the LP to find the FW direction."""
    A = build_constraint_matrix(f)
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, method='highs',

$(a*a)_k \le t$ for all $k$

Each $(a*a)_k = \sum_i a_i a_{k-i}$ is a quadratic form in $a$. Specifically, it's $a^T Q_k a$ where $Q_k$ is a symmetric matrix with $(Q_k)_{ij} = 1/2$ if $i+j=k$ (and 1 if $i=j=k/2$ when $k$ is even).

"""
Frank-Wolfe (conditional gradient) method for minimizing the peak of the
self-convolution of a non-negative sequence. Uses LP-based direction finding.
"""

import numpy as np
import time
import scipy.linalg
from scipy.optimize import linprog


def build_toeplitz(f):
    """Build Toeplitz matrix A where A[k,j] = f[k-j] if 0<=k-j<len(f), else 0."""
    n = len(f)
    first_col = np.concatenate([f, np.zeros(n-1)])
    first_row = np.array([f[0]] + [0.0]*(n-1))
    return scipy.linalg.toeplitz(first_col, first_row)


def fw_step(f, M, n):
    """Solve the LP for one FW step. Returns direction g or None."""
    A = build_toeplitz(f)
    b = np.full(2*n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

"""
Frank-Wolfe (conditional gradient) method for minimizing the peak of the
self-convolution of a non-negative sequence. At each step, solve an LP to
find the direction that maximizes mass under the current peak constraint,
then move toward that direction. Multiple sequence lengths and initializations
are tried, followed by deep refinement of the best result.
"""

import numpy as np
import time
import scipy.linalg
from scipy.optimize import linprog


def build_toeplitz(f):
    """Build Toeplitz matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return scipy.linalg.toeplitz(first_col, first_row)


def fw_direction(f, M, n):
    """Solve the LP for one FW step. Returns direction g (sum-normalized) or None."""
    A = build_toeplitz(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def run_fw(f0, max_time, deadline, seed=0):
    """Run Frank-Wolfe iterations. Returns (best_f, best_score)."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = np.sum(f)
    if S < 1e-10:
        f = np.ones(n, dtype=np.float64)
        S = float(n)
    f = f / S  # normalize sum to 1

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    t = 0.6
    stall = 0
    it = 0
    start = time.monotonic()

    while True:
        if time.monotonic() > deadline or time.monotonic() - start > max_time:
            break
        it += 1

        conv = np.convolve(f, f)
        M = np.max(conv)

        g = fw_direction(f, M, n)
        if g is None:
            break

        # Convergence check
        if np.max(np.abs(g - f)) < 5e-6:
            if stall > 5:
                break
            stall += 1
            continue

        # Try multiple step sizes (larger first for faster progress)
        improved = False
        step_candidates = [t, t * 0.5, t * 0.25, 0.1, 0.05, 0.02]
        for trial_t in step_candidates:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.maximum(new_f, 0.0)
            if np.sum(new_f) < 1e-6:
                continue
            score = evaluate_sequence(new_f.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = new_f.copy()
                f = new_f.copy()
                improved = True
                break

        if improved:
            stall = 0
            t = min(1.0, t * 1.04)
        else:
            stall += 1
            t = max(0.02, t * 0.85)
            if stall > 60:
                # Random perturbation to escape local minimum
                noise_scale = 0.02 * max(1.0, np.max(best_f))
                f = best_f + rng.normal(0, noise_scale, n)
                f = np.maximum(f, 0.0)
                fs = np.sum(f)
                if fs > 1e-6:
                    f = f / fs
                stall = 0
                t = 0.4
