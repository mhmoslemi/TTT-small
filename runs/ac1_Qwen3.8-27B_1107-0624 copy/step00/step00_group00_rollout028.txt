from scipy.optimize import linprog
import time

def build_conv_matrix(f, n):
    """Build the matrix A such that A @ g = np.convolve(f, g)."""
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        A[k, js] = f[k - js]
    return A

def solve_lp_for_direction(f, n):

    A = build_conv_matrix(f, n)
    
    # LP: min t s.t. A @ g - t <= 0 (for each row), sum(g) = 1, g >= 0, t >= 0
    # Variables: g[0], ..., g[n-1], t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    A_ub = np.hstack([A, -np.ones((2*n - 1, 1))])
    b_ub = np.zeros(2*n - 1)

    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([1.0])

    if res.success:
        return res.x[:n]

import numpy as np
from scipy.optimize import linprog
import time

def _compute_obj(f, n):
    """Compute objective 2*n*max(conv(f,f))/sum(f)^2."""
    s = np.sum(f)
    if s < 1e-6:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * n * np.max(c) / (s * s)

def _build_A(f, n):
    """Build matrix A where A @ g = np.convolve(f, g)."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[k - hi:k - lo + 1][::-1]
    return A

def _solve_lp(f, n):
    """Solve min t s.t. A_f @ g <= t, sum(g)=1, g>=0."""
    A = _build_A(f, n)
    
    # Variables: g[0..n-1], t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)

                  bounds=bounds, method='highs', 
                  options={'presolve': True})
    
    if res.success:
