```python
"""
Power-law / polynomial-root initializations, followed by a multiplicative
peak-flattening local search in the non-negative simplex.

The objective is scale invariant.  We normalize every sequence to sum 1 and
minimize 2*n*max(full convolution).  The local search uses weights focused on
the largest convolution coefficients and applies an exponential tilt to the
logarithms of the sequence values, with exact-score line search and occasional
perturbations to escape local minima.
"""

import numpy as np
import time


def _as_normalized_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64)
    return a / s


def _downsample(a, m):
    a = _as_normalized_array(a)
    m = max(2, int(m))
    if a.size <= m:
        return a.copy()
    n = a.size
    out = np.empty(m, dtype=np.float64)
    for j in range(m):
        s = int((j * n) / m)
        e = int(((j + 1) * n) / m)
        if e <= s:
            e = s + 1
        out[j] = float(np.mean(a[s:e]))
    return _as_normalized_array(out)


def _convolution(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    l = a.size + b.size - 1
    if l <= 0:
        return np.array([0.0], dtype=np.float64)
    if l <= 1024:
        c = np.convolve(a, b)
    else:
        m = 1 << (l - 1).bit_length()
        fa = np.fft.rfft(a, m)
        fb = np.fft.rfft(b, m)
        c = np.fft.irfft(fa * fb, m)[:l]
    c = np.asarray(c, dtype=np.float64)
    if c.size and c.min() < 0.0:
        c = np.maximum(c, 0.0)
    return c


def _pressure(w, a):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    l = n + m - 1
    if l <= 1024:
        c = np.convolve(a, w[::-1])
    else:
        size = 1 << (l - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fw = np.fft.rfft(w[::-1], size)
        c = np.fft.irfft(fa * fw, size)[:l]
    c = np.asarray(c, dtype=np.float64)
    h = c[m - n:m][::-1]
    if h.size and h.min() < 0.0:
        h = np.maximum(h, 0.0)
    return h


def _relative_weights(c, gamma):
    c = np.asarray(c, dtype=np.float64)
    peak = float(np.max(c))
    if not np.isfinite(peak) or peak <= 0.0:
        return np.full(c.shape, 1.0 / max(1, c.size), dtype=np.float64)
    rel = c / peak
    w = np.power(rel, float(gamma))
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return w / s


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    a = a / s
    c = _convolution(a)
    peak = float(np.max(c))
    if not np.isfinite(peak):
        return float("inf")
    return float(2.0 * a.size * peak)


def _power_candidate(n, alpha, power, reverse=False):
    n = max(2, int(n))
    x = np.arange(n, dtype=np.float64) + float(alpha)
    a = np.power(x, -float(power))
    if reverse:
        a = a[::-1]
    return _as_normalized_array(a)


def _recurrence_candidate(n):
    # Coefficients of the finite truncation of sqrt(1 / (1 - z)).
    # This gives a natural heavy-tailed nonnegative sequence whose square has
    # a relatively flat early convolution.
    n = max(2, int(n))
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k > 1:
            known = float(np.dot(a[1:k], a[k - 1:0:-1]))
        else:
            known = 0.0
        val = 0.5 * (1.0 - known)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        a[k] = val
    return _as_normalized_array(a)


def _smooth_random_candidate(n, seed):
    n = max(32, int(n))
    rng = np.random.default_rng(int(seed))
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for f in (1, 2, 4, 8, 16, 32):
        freq = rng.uniform(0.5 * f, 1.5 * f) / (2.0 * np.pi * n)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.10, 0.40)
        a *= np.exp(amp * np.cos(2.0 * np.pi * freq * x + phase))
    a = np.maximum(a, 1e-8)
    return _as_normalized_array(a)


def _perturb(a, sigma, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    p = a * np.exp(rng.normal(0.0, float(sigma), a.size))
    return _as_normalized_array(p)


def _block_perturb(a, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    p = a.copy()
    width = int(round(n * rng.uniform(0.01, 0.12)))
    width = max(1, min(n, width))
    start = int(rng.integers(0, max(1, n - width + 1)))
    p[start:start + width] *= np.exp(rng.normal(0.0, 0.25, width))
    if rng.random() < 0.3:
        p = 0.5 * (p + p[::-1])
    return _as_normalized_array(p)


def _linear_candidate(a, c, peak, peak_fraction, factor):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    w = (c >= float(peak_fraction) * float(peak)).astype(np.float64)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    h = _pressure(w, a)
    hm = float(np.max(h))
    if not np.isfinite(hm) or hm <= 0.0:
        return a.copy()
    p = a - float(factor) * h
    return _as_normalized_array(p)


def _local_search(initial, limit_abs, tag, seed):
    rng = np.random.default_rng(int(seed))
    a = _as_normalized_array(initial)
    a = np.maximum(a, 1e-15)
    a /= float(np.sum(a))

    curr = _score(a)
    if not np.isfinite(curr):
        a = _power_candidate(512, 0.5, 0.5)
        a = np.maximum(a, 1e-15)
        a /= float(np.sum(a))
        curr = _score(a)

    best = a.copy()
    best_score = float(curr)

    tau = 0.8
    gamma = 8.0
    stall = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < limit_abs - 0.5:
        it += 1

        c = _convolution(a)
        peak = float(np.max(c))
        if not np.isfinite(peak) or peak <= 0.0:
            break

        w = _relative_weights(c, gamma)
        h = _pressure(w, a)
        hbar = float(np.dot(w, c))
        if not np.isfinite(hbar) or hbar <= 0.0:
            hbar = float(np.dot(a, h))
        if not np.isfinite(hbar) or hbar <= 0.0:
            break

        ratio = h / hbar
        delta = np.clip(ratio - 1.0, -4.0, 30.0)
        loga_base = np.log(np.maximum(a, 1e-300))

        tau_try = float(tau)
        accepted = False

        for _attempt in range(7):
            if tau_try <= 1e-5:
                break
            loga = loga_base - tau_try * delta
            loga -= float(np.max(loga))
            cand = np.exp(loga)
            s = float(np.sum(cand))
            if not np.isfinite(s) or s <= 0.0:
                break
            cand /= s
            sc = _score(cand)
            if np.isfinite(sc) and sc < curr - 1e-13:
                a = cand
                curr = float(sc)
                tau = min(float(tau_try) * 1.08, 6.0)
                accepted = True
                break
            tau_try *= 0.7

        if not accepted:
            stall += 1
            if tau > 1e-3:
                tau *= 0.7
            else:
                gamma = min(gamma * 1.4, 150.0)
                tau = 0.8

        if curr < best_score:
            best = a.copy()
            best_score = float(curr)

        if stall > 50:
            improved = False

            # Linear subgradient-style moves on the current hot peaks.
            for frac in (0.75, 0.60, 0.40):
                for fac in (1.0, 0.5, 0.2, 0.1):
                    p = _linear_candidate(a, c, peak, frac, fac)
                    sp = _score(p)
                    if np.isfinite(sp) and sp < curr - 1e-13:
                        a = p
                        curr = float(sp)
                        improved = True
                        break
                if improved:
                    break

            # Pure multiplicative perturbations.
            if not improved:
                for sig in (0.30, 0.15, 0.07):
                    p = _perturb(a, sig, rng)
                    sp = _score(p)
                    if np.isfinite(sp) and sp < curr - 1e-13:
                        a = p
                        curr = float(sp)
                        improved = True
                        break

            # Block perturbation.
            if not improved:
                p = _block_perturb(a, rng)
                sp = _score(p)
                if np.isfinite(sp) and sp < curr - 1e-13:
                    a = p
                    curr = float(sp)
                    improved = True

            # Very modest simulated-annealing move to escape flat basins.
            if not improved:
                p = _perturb(a, 0.25, rng)
                sp = _score(p)
                if (
                    np.isfinite(sp)
                    and sp < curr + 0.012
                    and (sp < best_score or rng.random() < 0.20)
                ):
                    a = p
                    curr = float(sp)
                    improved = True

            if improved:
                if curr < best_score:
                    best = a.copy()
                    best_score = float(curr)
                tau = 0.6
                gamma = min(max(gamma, 10.0), 80.0)
                stall = 0
            else:
                if gamma < 120.0:
                    gamma *= 1.5
                else:
                    gamma *= 0.6
                tau = 0.8
                stall = 0
                if gamma > 200.0 or gamma < 3.0:
                    break

        if it % 300 == 0 and (time.monotonic() - last_print) > 3.0:
            print(
                tag,
                "iter",
                it,
                "score",
                round(float(curr), 8),
                "best",
                round(float(best_score), 8),
                "gamma",
                round(float(gamma), 1),
                "tau",
                round(float(tau), 3),
                "n",
                int(a.size),
                flush=True
            )
            last_print = time.monotonic()

    if curr < best_score:
        best = a.copy()
        best_score = float(curr)

    return best, float(best_score)


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):

    for n in lengths:
        if n >= 2048:
            exponents = (0.45, 0.50, 0.55)
            alphas = (0.3, 1.0, 4.0)
        else:
            exponents = base_exponents
            alphas = base_alphas

        for alpha in alphas:
            for e in exponents:
                a = _power_candidate(n, alpha, e)
                sc = _score(a)
                if np.isfinite(sc):
                    candidates.append((sc, counter, a))
                    counter += 1

                if n <= 1024 and alpha == 0.3 and e == 0.5:
                    a = _power_candidate(n, alpha, e, reverse=True)
                    sc = _score(a)
                    if np.isfinite(sc):
                        candidates.append((sc, counter, a))
                        counter += 1

        if n <= 4096:
            a = _recurrence_candidate(n)
            sc = _score(a)
            if np.isfinite(sc):
                candidates.append((sc, counter, a))
                counter += 1

        if n <= 2048:
            a = _smooth_random_candidate(n, seed + 1000 + n)
            sc = _score(a)
            if np.isfinite(sc):
                candidates.append((sc, counter, a))
                counter += 1

            a = _smooth_random_candidate(n, seed + 2000 + n)
            sc = _score(a)
            if np.isfinite(sc):
                candidates.append((sc, counter, a))
                counter += 1

    # Use the provided global parent, if present, only as an extra downsampled
    # initial point.  This is optional and guarded.
    try:
        parent = height_sequence_1
        if parent is not None:
            pa = _as_normalized_array(parent)
            if pa.size > 0:
                for m in (1024, 2048):
                    a = _downsample(pa, m)
                    sc = _score(a)
                    if np.isfinite(sc):
                        candidates.append((sc, counter, a))
                        counter += 1
    except Exception:
        pass

    if not candidates:
        a = _power_candidate(512, 0.5, 0.5)
        return [float(x) for x in a]

    candidates.sort()

    chosen = []
    count_n = {}
    for sc, idx, a in candidates:
        n = int(a.size)
        if count_n.get(n, 0) >= 2:
            continue
        chosen.append((float(sc), a))
        count_n[n] = count_n.get(n, 0) + 1
        if len(chosen) == 6:
            break

    if not chosen:
        a = candidates[0][2]
        return [float(x) for x in _as_normalized_array(a)]

    global_best = chosen[0][1].copy()
    global_score = float(chosen[0][0])

    pool = []
    for sc, a in chosen:
        pool.append((float(sc), counter, a.copy()))
        counter += 1
    for sc, idx, a in candidates[:3]:
        pool.append((float(sc), counter, a.copy()))
        counter += 1

    for i, (sc, a) in enumerate(chosen):
        remaining = deadline - time.monotonic()
        if remaining < 15.0:
            break

        alloc = min(240.0, remaining / max(1, len(chosen) - i))
        if i == len(chosen) - 1:
            alloc = min(alloc, max(5.0, remaining - 5.0))
        alloc = max(5.0, alloc)

        limit = time.monotonic() + alloc
        tag = "run%d" % i
        b, bsc = _local_search(a, limit, tag, seed + 50 + i)

        pool.append((float(bsc), counter, b.copy()))
        counter += 1

        if bsc < global_score:
            global_best = b.copy()
            global_score = float(bsc)

        print("local done", tag, "score", round(float(bsc), 8),
              "global", round(float(global_score), 8), flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        b, bsc = _local_search(global_best, deadline - 3.0, "polish", seed + 900)
        pool.append((float(bsc), counter, b.copy()))
        counter += 1
        if bsc < global_score:
            global_best = b.copy()
            global_score = float(bsc)

    pool.sort()
    final_candidates = []
    for sc, idx, a in pool[:8]:
        final_candidates.append((float(sc), a))

    exact_best = np.asarray(global_best, dtype=np.float64).copy()
    exact_score = float(global_score)

    for sc, a in final_candidates:
        if time.monotonic() > deadline - 2.0:
            break
        seq = [float(x) for x in a]
        try:
            esc = evaluate_sequence(seq)
        except Exception:
            esc = sc
        if np.isfinite(esc) and float(esc) < exact_score:
            exact_best = np.asarray(a, dtype=np.float64).copy()
            exact_score = float(esc)

    out = _as_normalized_array(exact_best)
    return [float(x) for x in out]
```
